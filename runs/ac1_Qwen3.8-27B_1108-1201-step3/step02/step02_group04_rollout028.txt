"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase strategy:
1. Parametric shape sweep (asymmetric ramps, exponentials, power laws)
2. LP-guided alternating optimization with perturbation restarts
3. Log-sum-exp smoothed gradient descent polishing
4. Final combination and polish
Key insight: asymmetric sequences break the Cauchy-Schwarz barrier at score=2.
"""

import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def score_normalized(f):
    """Score for a normalized (sum=1) sequence."""
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()


def make_lp_structure(n):
    rows_list = []
    cols_list = []
    for i in range(n):
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def solve_lp(f, M, rows, cols, n, timeout=12):
    vals = np.tile(f, n)
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def lp_optimize(f0, time_limit, rows=None, cols=None, verbose=False):
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    s = f.sum()
    if s < 1e-12:
        f = np.ones(n) / n
    else:
        f /= s
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()
    if rows is None or cols is None:
        rows, cols = make_lp_structure(n)
    t0 = time.monotonic()
    t = 1.0
    patience = 0
    iter_count = 0
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        g = solve_lp(f, M, rows, cols, n)
        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g /= g_sum
        if np.max(np.abs(g - f)) < 1e-14:
            break
        improved = False
        for t_try in [t, 0.7 * t, 0.5 * t, 0.3 * t, 0.1 * t, 0.05 * t]:
            f_new = (1.0 - t_try) * f + t_try * g
            f_new = np.maximum(f_new, 0.0)
            ss = f_new.sum()
            if ss < 1e-12:
                continue
            f_new /= ss
            conv_new = fast_convolve(f_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj - 1e-13:
                f = f_new
                M = conv_new.max()
                best_obj = obj_new
                best_f = f.copy()
                patience = 0
                t = min(1.0, t * 1.05)
                improved = True
                break
        if not improved:
            patience += 1
            t *= 0.85
            if patience > 40:
                break
        if verbose and iter_count % 100 == 0:
            print(f"      LP iter {iter_count}, obj={best_obj:.8f}, t={t:.3f}, elapsed={time.monotonic()-t0:.1f}s")
    return best_f, best_obj


def compute_grad_smoothed(f, T):
    """Compute gradient of smoothed max convolution w.r.t. f (f normalized)."""
    n = len(f)
    conv = fast_convolve(f)
    conv_safe = np.maximum(conv, 1e-30)
    max_cv = conv.max()
    scaled = T * conv_safe / max_cv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-12:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= w_sum
    # grad[i] = sum_k w[k] * d conv[k]/d f[i]
    # d conv[k]/d f[i] = f[k-i] (if 0<=k-i<=n-1) + f[i-k] (if 0<=i-k<=n-1, i!=k-i)
    # Simplified: for each k, f[k-i] contributes to grad[i] for i in [max(0,k-n+1), min(n-1,k)]
    #            and f[i-k] contributes for i in [k, min(n-1, k+n-1)] (but k>=n makes this empty)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        # f[k-i] term: i from max(0,k-n+1) to min(n-1,k)
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        # f[k-i] for i in [i_lo, i_hi] => indices k-i_hi to k-i_lo
        grad[i_lo:i_hi + 1] += wk * f[k - i_hi:k - i_lo + 1][::-1]
        # f[i-k] term: i from max(0,k) to min(n-1, k+n-1), only if k < n
        if k < n:
            j_lo = k
            j_hi = min(n - 1, k + n - 1)
            # f[i-k] for i in [j_lo, j_hi] => f[0] to f[j_hi-k]
            if j_lo <= j_hi:
                grad[j_lo:j_hi + 1] += wk * f[0:j_hi - k + 1]
    return grad, conv


def gradient_polish(f0, time_limit, T=80.0, lr0=0.3, momentum=0.9):
    f = np.maximum(f0.astype(np.float64), 1e-10)
    n = len(f)
    f = f / f.sum()
    best_obj = score_normalized(f)
    best_f = f.copy()
    v = np.zeros(n)
    lr = lr0
    t0 = time.monotonic()
    patience = 0
    T_cur = T
    while time.monotonic() - t0 < time_limit:
        grad, conv = compute_grad_smoothed(f, T_cur)
        # Normalize: the "cost" we minimize is max conv, gradient points to increase it
        # We want to DECREASE it, so move opposite to weighted gradient
        # But we need to stay on the simplex (sum=1)
        grad_c = grad - grad.mean()
        v = momentum * v - lr * grad_c
        # Multiplicative update: f_new = f * exp(v), then normalize
        f_new = f * np.exp(np.clip(v, -2, 2))
        f_new = np.maximum(f_new, 1e-15)
        ss = f_new.sum()
        if ss < 1e-12:
            f_new = f.copy()
        else:
            f_new /= ss
        obj = score_normalized(f_new)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f_new.copy()
            f = f_new
            patience = 0
            lr = min(lr * 1.02, 1.0)
        else:
            patience += 1
            lr *= 0.7
            if patience % 50 == 0:
                T_cur = min(T_cur * 1.5, 500.0)
            if patience > 200:
                break
            if lr < 0.01:
                break
    return best_f, best_obj


def gen_parametric(n, rng):
    """Generate diverse parametric initializations exploiting asymmetry."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    xn = x / n
    inits['uni'] = np.ones(n) / n

    # Linear ramps (asymmetric)
    for c in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = 1.0 + c * xn
        a = np.maximum(a, 0.01)
        inits[f'lin_{c}'] = a / a.sum()

    # Power laws (growing)
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = (xn + 0.01) ** p
        inits[f'pow_{p}'] = a / a.sum()

    # Exponential
    for rate in [0.5, 1.0, 2.0, 3.0, 5.0]:
        a = np.exp(rate * xn)
        inits[f'exp_{rate}'] = a / a.sum()

    # Cosine-weighted (shifted)
    for shift in [0.3, 0.5, 0.7]:
        a = 0.5 * (1 + np.cos(2 * np.pi * (xn - shift) / (1.0)))
        a = np.maximum(a, 0.01)
        inits[f'cos_{shift}'] = a / a.sum()

    # Bimodal asymmetric
    w1 = max(1, n // 5)
    w2 = max(1, n // 8)
    a = np.zeros(n)
    a[:w1] = 1.0
    a[-w2:] = 1.0
    a = np.maximum(a, 0.01)
    inits['bimod'] = a / a.sum()

    # Gaussian (shifted right)
    for mu, sig in [(0.6, 0.15), (0.7, 0.2), (0.8, 0.1), (0.5, 0.3)]:
        a = np.exp(-0.5 * ((xn - mu) / sig) ** 2)
        a = np.maximum(a, 0.001)
        inits[f'gau_{mu}_{sig}'] = a / a.sum()

    # Random
    for seed in range(3):
        r = rng.dirichlet(np.ones(n) * 0.5)
        inits[f'rnd_{seed}'] = r

    return inits


def perturb(f, rng, strength=0.1):
    """Randomly perturb a sequence while keeping it non-negative."""
    f_new = f.copy()
    noise = rng normal(0, strength, len(f_new))
    f_new = f_new + noise
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-10:
        return f.copy()
    return f_new / s


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Try parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
    except Exception:
        pass

    results = []  # (score, n, f)

    def update_best(f, n):
        nonlocal best_seq, best_score
        sc = score_normalized(f)
        if sc < best_score:
            best_score = sc
            best_seq = f.copy()
        results.append((sc, n, f.copy()))

    # Phase 1: Parametric sweep
    print("Phase 1: Parametric sweep")
    t_phase1 = min(90.0, (deadline - time.monotonic()) * 0.08)
    t0 = time.monotonic()

    lengths = [256, 384, 512, 768, 1024]
    for n in lengths:
        if time.monotonic() - t0 > t_phase1:
            break
        inits = gen_parametric(n, rng)
        if parent is not None:
            if parent.size >= n:
                pp = np.maximum(parent[:n], 0.0)
                if pp.sum() > 1e-10:
                    inits['parent'] = pp / pp.sum()
            elif parent.size > 0:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
                pp = np.maximum(pp, 0.0)
                if pp.sum() > 1e-10:
                    inits['parent'] = pp / pp.sum()
        for name, a0 in inits.items():
            sc = score_normalized(a0)
            update_best(a0, n)
        # Report best for this n
        n_best = min([r for r in results if r[1] == n], key=lambda x: x[0])[0]
        print(f"  n={n}: best parametric score={n_best:.6f}, {len(inits)} shapes tried")

    # Phase 2: LP optimization on top candidates
    print(f"\nPhase 2: LP optimization")
    results.sort(key=lambda x: x[0])
    unique_ns = list(dict.fromkeys([r[1] for r in results[:30]]))
    # Pick top candidates per n
    candidates = []
    for n in unique_ns[:5]:
        n_results = [r for r in results if r[1] == n]
        n_results.sort(key=lambda x: x[0])
        candidates.extend(n_results[:3])
    candidates = candidates[:12]

    t0 = time.monotonic()
    time_per_lp = max(30.0, (deadline - time.monotonic() - 300) / max(1, len(candidates)))

    for idx, (sc0, n, f0) in enumerate(candidates):
        elapsed_total = time.monotonic() - t0
        if elapsed_total > (deadline - time.monotonic() - 250):
            break
        remaining = deadline - time.monotonic() - 200
        ta = min(time_per_lp, remaining)
        if ta < 15:
            break

        print(f"  LP [{idx}]: n={n}, init_score={sc0:.6f}, time={ta:.0f}s")
        f_opt, obj_opt = lp_optimize(f0, time_limit=ta * 0.8)
        sc_opt = score_normalized(f_opt)
        update_best(f_opt, n)
        print(f"    -> LP score: {sc_opt:.6f} (improved from {sc0:.6f})")

        # Gradient polish after LP
        if time.monotonic() < deadline - 150:
            tr = min(ta * 0.25, deadline - time.monotonic() - 150)
            if tr > 10:
                f_gp, obj_gp = gradient_polish(f_opt, time_limit=tr, T=60.0, lr0=0.3)
                sc_gp = score_normalized(f_gp)
                update_best(f_gp, n)
                print(f"    -> Grad polish: {sc_gp:.6f}")

    # Phase 3: Perturbation restarts from best
    print(f"\nPhase 3: Perturbation restarts. Current best: {best_score:.6f}")
    if best_seq is not None:
        t3_start = time.monotonic()
        n_best = len(best_seq)
        rows_best, cols_best = make_lp_structure(n_best)
        remaining = deadline - time.monotonic() - 100
        n_restarts = min(8, max(2, int(remaining / 60)))

        for r in range(n_restarts):
            if time.monotonic() > deadline - 80:
                break
            tr = min(60.0, (deadline - time.monotonic() - 50) / (n_restarts - r))
            if tr < 15:
                break
            strengths = [0.05, 0.1, 0.03, 0.15, 0.02, 0.2, 0.04, 0.25]
            f_pert = perturb(best_seq, rng, strength=strengths[r % len(strengths)])
            f_opt, obj_opt = lp_optimize(f_pert, time_limit=tr * 0.7, rows=rows_best, cols=cols_best)
            sc = score_normalized(f_opt)
            update_best(f_opt, n_best)
            if sc < best_score + 0.001:
                print(f"  Restart {r}: score={sc:.6f} (best={best_score:.6f})")

    # Phase 4: Final gradient polish with multiple temperatures
    print(f"\nPhase 4: Final polish. Current best: {best_score:.6f}")
    if best_seq is not None and time.monotonic() < deadline - 20:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic() - 15
        n_phases = 4
        t_each = remaining / n_phases
        for i, (T, lr) in enumerate([(40.0, 0.4), (100.0, 0.2), (200.0, 0.1), (400.0, 0.05)]):
            if time.monotonic() > deadline - 10:
                break
            tr = min(t_each, deadline - time.monotonic() - 10)
            if tr < 5:
                break
            f_gp, obj_gp = gradient_polish(best_seq, time_limit=tr, T=T, lr0=lr)
            sc = score_normalized(f_gp)
            update_best(f_gp, n_best)
            print(f"  Polish T={T:.0f}: score={sc:.6f}")

    # Phase 5: One more LP pass on the absolute best
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = len(best_seq)
        tr = deadline - time.monotonic() - 10
        if tr > 20:
            print(f"\nPhase 5: Final LP pass, {tr:.0f}s")
            rows_b, cols_b = make_lp_structure(n_best)
            f_final, obj_final = lp_optimize(best_seq, time_limit=tr, rows=rows_b, cols=cols_b)
            sc = score_normalized(f_final)
            update_best(f_final, n_best)
            print(f"  Final LP: score={sc:.6f}")

    # Return the best sequence
    if best_seq is None:
        best_seq = np.ones(512)
        best_score = score_normalized(best_seq)

    print(f"\nFINAL: n={len(best_seq)}, score={best_score:.8f}")
    return [float(x) for x in best_seq]
