def normalize(a, target):

    S = a.sum()
    if S < 1e-12:
        n = len(a)
        return np.full(n, target / n)
    return a * (target / S)


def softmax_gradient(a, tau):
    """Compute softmax-weighted gradient of max(conv(a,a))."""
    n = len(a)
    c = np.convolve(a, a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= w_sum
    # grad[i] = 2 * sum_j w[i+j] * a[j] via cross-correlation
    a_rev = a[::-1]
    conv_res = np.convolve(w, a_rev, mode='full')
    grad = 2.0 * conv_res[n - 1:n - 1 + n]
    return grad, w

def coordinate_descent_pass(a, max_rounds=5):
    """Full coordinate descent: iterate over all coordinates, update to best value."""
    n = len(a)
    for rnd in range(max_rounds):
        c = np.convolve(a, a)
        c_max = c.max()
        S = a.sum()
        any_change = False

def lp_refine(f, M, n_max=512):
    """
    LP: maximize sum(g) subject to convolve(f,g) <= M, g >= 0.
    Returns optimal g or None.
    """
    n = len(f)
    if n > n_max:
        return None

    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for m in range(j, j + n):
            A[m, j] = f[m - j]

    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def gradient_descent_phase(a, deadline, tau_schedule='linear'):
    """
    Softmax gradient descent with adaptive parameters.
    Returns best (a, score) found during this phase.
    """
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a.copy(), S_t)
    score = compute_obj(a)
    best = a.copy()
    best_score = score

    momentum = np.zeros(n)
    decay = 0.9
    it = 0
    no_improve = 0

    while time.monotonic() < deadline and it < 20000:
        it += 1

        # Adaptive tau: start moderate, increase over time
        if tau_schedule == 'linear':
            tau = 3.0 + 47.0 * min(1.0, it / 5000.0)
        elif tau_schedule == 'exponential':
            tau = 3.0 * (1.01 ** min(it, 2000))
        else:
            tau = 10.0

        lr = 0.15 * (0.999 ** it)
        lr = max(lr, 0.002)

        grad, _ = softmax_gradient(a, tau)
        # Project out mean to maintain sum
        grad -= grad.mean()

        momentum = decay * momentum - lr * grad
        a = a + momentum
        a = np.maximum(a, 0.0)
        a = normalize(a, S_t)

        if it % 100 == 0:
            s = compute_obj(a)
            if s < best_score - 1e-9:
                best_score = s
                best = a.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 15:
                    # Reset momentum, try different tau
                    momentum[:] = 0
                    no_improve = 0
                    if tau_schedule == 'linear':
                        tau_schedule = 'exponential'
                    tau = tau * 2
                    tau = min(tau, 100.0)

    return best, best_score


def peak_equalization(a, max_iters=100):
    """
    Targeted peak reduction: identify peak lag, reduce contributing coordinates.
    """
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a.copy(), S_t)
    score = compute_obj(a)
    best = a.copy()
    best_score = score

    for it in range(max_iters):
        c = np.convolve(a, a)
        peak_idx = int(np.argmax(c))
        M = c[peak_idx]
        S = a.sum()

        # Find coordinates contributing to peak
        # c[peak_idx] = sum_i a[i]*a[peak_idx-i] for valid i
        contrib = {}
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n and i != j:
                contrib[i] = contrib.get(i, 0) + a[j]
                contrib[j] = contrib.get(j, 0) + a[i]
            elif i == j:
                contrib[i] = contrib.get(i, 0) + a[i]

        # Reduce the coordinate with largest contribution/weight ratio
        # Objective sensitivity: reducing a[p] by d changes M by ~contrib[p]*d
        # and S by d. Score change ~ 2n*(dM*S^2 - M*2S*dS)/S^4 = 2n/S^3*(dM*S - 2*M*dS)
        # = 2n/S^3 * (-contrib[p]*d*S + 2*M*d) = 2n*d/S^3 * (2*M/S - contrib[p])
        # Negative when contrib[p] > 2*M/S, meaning reducing helps.

        threshold = 2.0 * M / S
        reducible = [(p, v) for p, v in contrib.items() if v > threshold and a[p] > 1e-6]

        if not reducible:
            break

        # Pick the one with highest contrib ratio
        reducible.sort(key=lambda x: -x[1])
        p_best = reducible[0][0]

        # Try reducing a[p_best] by a small amount
        reduction = a[p_best] * 0.1
        a[p_best] -= reduction
        a = np.maximum(a, 0.0)
        a = normalize(a, S_t)

        s = compute_obj(a)
        if s < best_score - 1e-10:
            best_score = s
            best = a.copy()
        else:
            # Restore
            a[p_best] += reduction
            a = normalize(a, S_t)
            break

    return best, best_score


def generate_initializations(rng, n, known_seq=None):
    """Generate diverse initial sequences of length n."""
    inits = []

    # Known best (truncated or padded)
    if known_seq is not None and len(known_seq) > 0:
        ks = np.asarray(known_seq, dtype=np.float64)
        ks = np.maximum(ks, 0)
        if len(ks) >= n:
            inits.append(ks[:n].copy())
        else:
            # Pad with interpolated tail
            pad = n - len(ks)
            tail = np.interp(np.linspace(len(ks)-1, n-1, pad),
                             np.linspace(0, len(ks)-1, len(ks)),
                             ks[-min(10, len(ks)):])
            inits.append(np.concatenate([ks, tail * 0.5]).copy())

    # Uniform
    inits.append(np.full(n, 1.0))

    # Random
    for _ in range(3):
        inits.append(rng.random(n) * 0.5 + 0.25)

    # Cosine modulation (various frequencies)
    i = np.arange(n, dtype=np.float64)
    for freq in [1, 2, 3, 5, 7, 13]:
        inits.append(1.0 + 0.5 * np.cos(2 * np.pi * freq * i / n))
        inits.append(1.0 + 0.3 * np.cos(2 * np.pi * freq * i / n + 0.5))

    # Quadratic phase (Zadoff-Chu inspired)
    for alpha in [1, 3, 5, 7]:
        phase = alpha * i * i / n
        inits.append(1.0 + 0.4 * np.cos(2 * np.pi * phase))
        inits.append(0.5 + 0.5 * np.abs(np.exp(1j * 2 * np.pi * phase)))

    # Gaussian bumps
    for center_frac in [0.3, 0.5, 0.7]:
        center = n * center_frac
        width = n * 0.15
        inits.append(np.exp(-0.5 * ((i - center) / width) ** 2) + 0.1)

    # Two-level: high/low pattern
    pattern = np.zeros(n)
    pattern[::2] = 1.0
    inits.append(pattern * 0.5 + 0.5)

    # Sparse-ish
    a = np.zeros(n)
    a[::3] = rng.random(n // 3 + 1) * 0.8 + 0.5
    inits.append(a)

    # Periodic blip train
    period = max(4, n // 32)
    a = np.zeros(n)
    a[::period] = 1.0
    a = a + 0.1  # baseline
    inits.append(a)

    return inits


def optimize_single(a0, deadline, verbose=True):
    """
    Full optimization pipeline for a single starting point.
    Returns (best_seq, best_score).
    """
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n) * 0.5
    S_t = np.sqrt(2.0 * n)
    a = normalize(a, S_t)
    score = compute_obj(a)
    best = a.copy()
    best_score = score

    if verbose:
        print(f"      n={n}, initial_score={score:.6f}")

    # Phase 1: Gradient descent (softmax)
    p1_deadline = time.monotonic() + max(5.0, (deadline - time.monotonic()) * 0.30)
    a1, s1 = gradient_descent_phase(a, p1_deadline, tau_schedule='linear')
    if s1 < best_score:
        best_score = s1
        best = a1
    if verbose:
        print(f"      After grad descent: {best_score:.6f}")

    # Phase 2: Peak equalization
    if time.monotonic() < deadline - 10:
        p2_deadline = time.monotonic() + max(3.0, (deadline - time.monotonic()) * 0.15)
        a2, s2 = peak_equalization(best, max_iters=200)
        if s2 < best_score:
            best_score = s2
            best = a2
        if verbose:
            print(f"      After peak eq: {best_score:.6f}")

    # Phase 3: Coordinate descent
    if time.monotonic() < deadline - 15 and n <= 512:
        a3 = best.copy()
        a3 = coordinate_descent_pass(a3, max_rounds=3)
        s3 = compute_obj(a3)
        if s3 < best_score:
            best_score = s3
            best = a3
        if verbose:
            print(f"      After coord desc: {best_score:.6f}")

    # Phase 4: LP refinement
    if n <= 384 and time.monotonic() < deadline - 10:
        c = np.convolve(best, best)
        M = float(c.max())
        for _ in range(10):
            if time.monotonic() > deadline - 10:
                break
            g = lp_refine(best, M)
            if g is None or g.sum() < 1e-6:
                break
            g = normalize(g, S_t)
            improved = False
            for t_blend in [0.5, 0.3, 0.2, 0.1, 0.05]:
                a_new = (1.0 - t_blend) * best + t_blend * g
                a_new = np.maximum(a_new, 0.0)
                a_new = normalize(a_new, S_t)
                s_new = compute_obj(a_new)
                if s_new < best_score - 1e-10:
                    best = a_new
                    best_score = s_new
                    improved = True
                    break
            if not improved:
                break
            c = np.convolve(best, best)
            M = float(c.max())
        if verbose:
            print(f"      After LP: {best_score:.6f}")

    # Phase 5: Final gradient polish
    if time.monotonic() < deadline - 5:
        p5_deadline = time.monotonic() + max(2.0, (deadline - time.monotonic()) * 0.5)
        a5, s5 = gradient_descent_phase(best, p5_deadline, tau_schedule='exponential')
        if s5 < best_score:
            best_score = s5
            best = a5
        if verbose:
            print(f"      After final grad: {best_score:.6f}")

    return best, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Multi-phase optimization with diverse starts.
    Returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Get known best sequence
    known = None
    try:
        known = np.asarray(height_sequence_1, dtype=np.float64)
        if known.size >= 8 and np.all(np.isfinite(known)):
            known = np.maximum(known, 0)
            print(f"Known best sequence: length={len(known)}, score={compute_obj(known):.6f}")
    except (NameError, Exception):
        pass

    # Lengths to try
    lengths = [384, 512, 256]

    print(f"Starting optimization, budget={budget_s}s")
    print(f"Deadline in {deadline - t_start:.1f}s")

    # === Phase A: Quick exploration ===
    # Try each length with the known best and a couple of random starts
    time_per_group = (deadline - t_start) * 0.40 / len(lengths)

    for n in lengths:
        if time.monotonic() > deadline - 120:
            break
        inits = generate_initializations(rng, n, known)
        # Limit initializations for time management
        inits = inits[:min(6, len(inits))]

        for idx, a0 in enumerate(inits):
            if time.monotonic() > deadline - 120:
                break
            single_deadline = min(max(t_start + 30, time.monotonic() + time_per_group / len(inits)),
                                  deadline - 100)
            seq, score = optimize_single(a0, single_deadline, verbose=True)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"    [n={n}, init {idx}] NEW BEST: {best_score:.6f}")

    print(f"\nAfter Phase A: best_score={best_score:.6f}, n={len(best_seq) if best_seq is not None else 0}")

    # === Phase B: Focused optimization on best length ===
    if best_seq is not None and time.monotonic() < deadline - 60:
        best_n = len(best_seq)
        time_remaining = deadline - time.monotonic() - 30
        n_tries = min(5, int(time_remaining / 60) + 1)

        # Generate diverse starts at the best length
        focused_inits = generate_initializations(rng, best_n, best_seq)
        # Add perturbed versions of the best
        for _ in range(3):
            noise = rng.normal(0, 0.05, best_n)
            perturbed = np.maximum(best_seq + noise, 0)
            focused_inits.append(perturbed)

        focused_inits = focused_inits[:n_tries]
        time_per_try = time_remaining / max(1, n_tries)

        for idx, a0 in enumerate(focused_inits):
            if time.monotonic() > deadline - 30:
                break
            single_deadline = min(time.monotonic() + time_per_try, deadline - 20)
            seq, score = optimize_single(a0, single_deadline, verbose=True)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"    [Phase B, try {idx}] NEW BEST: {best_score:.6f}")

    print(f"\nAfter Phase B: best_score={best_score:.6f}")

    # === Phase C: Deep refinement on the champion ===
    if best_seq is not None and time.monotonic() < deadline - 30:
        n = len(best_seq)
        print(f"\nPhase C: Deep refinement on best (n={n}, score={best_score:.6f})")

        # Extended coordinate descent
        time_for_cd = min(120.0, (deadline - time.monotonic() - 20) * 0.5)
        cd_deadline = time.monotonic() + time_for_cd

        a_cd = best_seq.copy()
        S_t = np.sqrt(2.0 * n)
        c = np.convolve(a_cd, a_cd)
        c_max = c.max()
        S = a_cd.sum()
        improved = True
        while improved and time.monotonic() < cd_deadline:
            improved = False
            for p in range(n):
                if time.monotonic() > cd_deadline:
                    break
                best_v, new_score = coord_desc_1d(a_cd, p, c, c_max, S)
                if abs(best_v - a_cd[p]) > 1e-8:
                    a_cd[p] = best_v
                    c = np.convolve(a_cd, a_cd)
                    c_max = c.max()
                    S = a_cd.sum()
                    s = compute_obj(a_cd)
                    if s < best_score - 1e-10:
                        best_score = s
                        best_seq = a_cd.copy()
                        improved = True

        print(f"  After deep CD: {best_score:.6f}")

        # Final gradient polish
        if time.monotonic() < deadline - 10:
            a_fin, s_fin = gradient_descent_phase(best_seq, deadline - 5, tau_schedule='exponential')
            if s_fin < best_score:
                best_score = s_fin
                best_seq = a_fin
            print(f"  After final polish: {best_score:.6f}")

    # === Final validation ===
    if best_seq is not None:
        final_score = compute_obj(best_seq)
        print(f"\n{'='*50}")
        print(f"FINAL: n={len(best_seq)}, score={final_score:.6f}")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence confirms: {eval_score:.6f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in best_seq]

    # Fallback
    print("FALLBACK: returning uniform sequence")
    return [1.0] * 256
