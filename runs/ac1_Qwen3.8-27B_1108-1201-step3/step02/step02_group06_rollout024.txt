```python
"""
Multi-phase optimization for minimizing normalized peak autocorrelation.
Uses p-norm gradient descent in log-space with Adam, peak-targeted refinement,
and simulated annealing. Exploits sequence symmetry for efficiency.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def pnorm_gradient(a, p, eps=1e-10):
    """Compute gradient of (1/p)*sum(c^p) with respect to a, where c=conv(a,a)."""
    c = fft_convolve(a, a)
    w = (c + eps) ** (p - 1)
    grad = 2.0 * np.correlate(w, a, mode='valid')
    return grad, c


def logspace_pnorm_optimize(a_init, time_limit, p_start=4, p_end=100, 
                             lr=0.5, verbose=False):

    n = len(a_init)
    a = np.clip(a_init, 1e-10, None).astype(np.float64)
    # Normalize to sum = 1
    a = a / np.sum(a)
    # Symmetrize
    a = (a + a[::-1]) / 2.0
    a = a / np.sum(a)
    
    theta = np.log(a)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps_adam = 0.85, 0.999, 1e-8
    step = 0
    
    best_a = a.copy()

        grad_a, c = pnorm_gradient(a, p)
        
        # Convert to theta-space gradient
        dot_ag = np.dot(a, grad_a)
        grad_theta = a * (grad_a - dot_ag)
        
        # Adam update
        step += 1
        m = beta1 * m + (1 - beta1) * grad_theta
        v = beta2 * v + (1 - beta2) * (grad_theta ** 2)
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)

        lr_t = lr / (1 + 0.001 * step) ** 0.5
        
        theta = theta - lr_t * m_hat / (np.sqrt(v_hat) + eps_adam)
        
        # Convert back to a-space
        theta_shifted = theta - np.max(theta)
        a_new = np.exp(theta_shifted)
        a_new = a_new / np.sum(a_new)
        
        # Symmetrize
        a_new = (a_new + a_new[::-1]) / 2.0
        a_new = a_new / np.sum(a_new)
        
        a = a_new
        score = local_score(a)
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
            stagnant = 0
        else:

            # Perturb to escape

def peak_reduction(a_init, time_limit, verbose=False):

    a = np.clip(a_init, 1e-10, None).astype(np.float64).copy()

        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:

        # Decrease top contributors, increase others
        top_k = max(3, n // 16)
        top_idx = np.argsort(contribs)[-top_k:]

        a_new = a.copy()

        a_new[top_idx] *= np.exp(rng.normal(-0.02, 0.01, size=top_k))
        # Increase some others
        if len(other_idx) > 0:
            n_inc = min(4, len(other_idx))
            inc_idx = rng.choice(other_idx, size=n_inc, replace=False)
            a_new[inc_idx] *= np.exp(rng.normal(0.02, 0.01, size=n_inc))
        
        a_new = np.clip(a_new, 1e-10, None)
        a_new = a_new / np.sum(a_new)

        if score < best_score:

def make_symmetric_init(provided_seq, n, rng):
    """Create a symmetric initialization from a provided sequence."""
    if len(provided_seq) == n:
        a = provided_seq.copy()
    else:
        x_old = np.linspace(0, 1, len(provided_seq))
        x_new = np.linspace(0, 1, n)
        a = np.interp(x_new, x_old, provided_seq)
    a = np.clip(a, 0.01, None)
    a = (a + a[::-1]) / 2.0
    a = a / np.sum(a)
    return a
```
