import numpy as np
from scipy.sparse import lil_matrix
from scipy.optimize import linprog
import time

def compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (a.sum()**2))

def normalize(a, target):
    s = a.sum()
    if s < 1e-15:
        return a.copy()
    return a * (target / s)

def build_constraint_matrix(f):
    """Build the sparse constraint matrix for the LP."""
    n = len(f)
    A = lil_matrix((2*n-1, n))
    for k in range(2*n-1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n-1, k)
        for i in range(i_lo, i_hi+1):
            A[k, i] = f[k - i]
    return A.tocsr()

def solve_lp(f, M, A=None):
    """Solve the LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    if A is None:
        A = build_constraint_matrix(f)
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.status == 0:
        return res.x
    return None

def lp_iteration(f, n, M=None, A=None):
    """One LP iteration. Returns (new_f, improved)."""
    S_t = np.sqrt(2.0 * n)
    if M is None:
        c = np.convolve(f, f)
        M = float(c.max())
    
    g = solve_lp(f, M, A)
    if g is None:
        return f, False
    
    g_sum = g.sum()
    if g_sum < 1e-10:
        return f, False
    g = g * (S_t / g_sum)
    
    f_old_score = compute_obj(f)
    
    best_f = f.copy()
    best_score = f_old_score
    
    for t in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05, 0.02]:
        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        f_new = normalize(f_new, S_t)
        s = compute_obj(f_new)
        if s < best_score - 1e-12:
            best_score = s
            best_f = f_new
    
    improved = best_score < f_old_score - 1e-10
    return best_f, improved
