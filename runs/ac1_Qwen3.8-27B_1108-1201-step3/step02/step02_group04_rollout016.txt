Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using improved LP-alternation,
subgradient polishing, simulated annealing restarts, and diverse initializations.
"""

import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time


def fast_auto_conv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def make_conv_matrix_indices(n):
    """Precompute sparse matrix indices for (f*g) convolution constraints."""
    rows = []
    cols = []
    for i in range(n):
        for j in range(n):
            rows.append(i + j)
            cols.append(i)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp_direction(f, M, n, rows, cols, timeout=10):
    """Solve max sum(g) s.t. g>=0, (f*g)[k]<=M for all k."""
    vals = np.tile(f, n)
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def compute_score_normalized(f):
    """Compute 2*n*max(conv) for normalized f (sum=1)."""
    conv = fast_auto_conv(f)
    return 2.0 * len(f) * conv.max()
