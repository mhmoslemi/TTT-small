        momentum = (momentum + 1.0) / beta
        y = a + momentum * (a - a_prev)
        y = np.maximum(y, 0.0)

        # Compute subgradient at y
        grad, mx = compute_subgradient(y, rel_tol=1e-5)

        # Backtracking line search
        a_prev = a.copy()
        improved = False
        fortrial_alpha in [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.1, alpha * 0.05]:
            a_new = a - trial_alpha * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = np.sum(a_new)
            if s_new < 0.01:
                continue
            a_new = a_new * (s0 / s_new)
            new_score = compute_score(a_new)
            if new_score < current_score:
                a = a_new
                current_score = new_score
                alpha = min(alpha * 1.1, alpha_max)
                stall_count = 0
                improved = True
                break

        if not improved:
            stall_count += 1
            alpha *= 0.7
            if alpha < alpha_min:
                alpha = alpha_min

        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()

        # Periodic restart to escape local minima
        if stall_count > 200:
            # Perturb and restart
            noise_scale = 0.02 * np.std(a)
            a = a + rng.normal(0, noise_scale, n)
            a = np.maximum(a, 0.0)
            s_a = np.sum(a)
            if s_a > 0.01:
                a = a * (s0 / s_a)
            current_score = compute_score(a)
            a_prev = a.copy()
            stall_count = 0
            alpha = 0.5 / (n ** 0.5)
            momentum = 0.0

        # Reduce relative tolerance as we converge
        if iteration % 500 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"    iter={iteration}, score={current_score:.6f}, best={best_score:.6f}, "
                  f"alpha={alpha:.4f}, stall={stall_count}, elapsed={elapsed:.1f}s")

    return best_a, best_score


def peak_shaving(a, rng, n_steps=100):
    """
    Targeted peak reduction: identify top contributors to peaks and reduce them,
    while increasing elements that don't contribute to peaks.
    """
    a = a.copy()
    n = len(a)
    s0 = np.sum(a)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > 60:
            break

        conv = fft_convolve(a, a)
        mx = np.max(conv)
        # Find top peak positions
        peak_idx = np.argsort(conv)[-3:]
        # For each peak, find top contributors
        reduce_mask = np.zeros(n)
        for k in peak_idx:
            for i in range(max(0, k - n + 1), min(k + 1, n)):
                j = k - i
                if 0 <= j < n:
                    reduce_mask[i] += a[i] * a[j]
                    reduce_mask[j] += a[i] * a[j]

        # Normalize reduce_mask
        rm_sum = reduce_mask.sum()
        if rm_sum > 0:
            reduce_mask /= rm_sum

        # Elements to increase: those with low reduce_mask
        increase_mask = 1.0 - reduce_mask
        im_sum = increase_mask.sum()
        if im_sum > 0:
            increase_mask /= im_sum

        # Apply perturbation
        scale = 0.03
        a_new = a * (1.0 - scale * reduce_mask)
        a_new = a_new + scale * s0 / n * increase_mask * 0.1
        a_new = np.maximum(a_new, 0.0)
        s_new = np.sum(a_new)
        if s_new > 0.01:
            a_new = a_new * (s0 / s_new)
            new_score = compute_score(a_new)
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                a = a_new

    return best_a, best_score


def local_search_polish(a_init, time_limit, rng):
    """
    Simulated annealing local search with diverse move types.
    """
    current = a_init.astype(np.float64).copy()
    n = len(current)
    s0 = np.sum(current)
    current_score = compute_score(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()

    temperature = 0.005
    cooling = 0.9995

    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()

        if r < 0.25:
            # Targeted peak reduction
            conv = fft_convolve(current, current)
            peak_idx = int(np.argmax(conv))
            contribs = np.zeros(n)
            for i in range(max(0, peak_idx - n + 1), min(peak_idx + 1, n)):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
                    contribs[j] = current[i] * current[j]
            # Reduce top contributors
            top_k = max(3, n // 15)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.03, 0.015, size=len(top_idx)))
            # Increase non-peak contributors
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 5:
                n_inc = min(10, len(other) // 3)
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.03, 0.015, size=n_inc))

        elif r < 0.40:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.025, size=width))

        elif r < 0.55:
            # Multi-block small perturbations
            for _ in range(4):
                width = max(1, int(rng.integers(1, max(2, n // 20))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.02, size=width))

        elif r < 0.70:
            # Pair redistribution
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                delta = rng.uniform(-0.12, 0.12) * current[i]
                candidate[i] = max(0.0, current[i] - delta)
                candidate[j] = max(0.0, current[j] + delta)

        elif r < 0.80:
            # Trilinear: move mass from one peak contributor to two others
            conv = fft_convolve(current, current)
            peak_idx = int(np.argmax(conv))
            i = int(rng.integers(max(0, peak_idx - n + 1), min(peak_idx + 1, n)))
            j = int(rng.integers(0, n))
            k = int(rng.integers(0, n))
            if len(set([i, j, k])) == 3:
                delta = rng.uniform(0.02, 0.08) * current[i]
                candidate[i] = max(0.0, current[i] - delta)
                candidate[j] = max(0.0, candidate[j] + delta * 0.5)
                candidate[k] = max(0.0, candidate[k] + delta * 0.5)

        elif r < 0.90:
            # Frequency-domain perturbation
            fa = np.fft.rfft(current)
            mag = np.abs(fa)
            phase = np.angle(fa)
            mag_noise = 1.0 + rng.normal(0, 0.02, len(mag))
            mag = mag * np.maximum(mag_noise, 0.1)
            candidate = np.fft.irfft(mag * np.exp(1j * phase), n=n)
            candidate = np.maximum(candidate, 0.0)

        else:
            # Small global perturbation
            candidate = current * np.exp(rng.normal(0.0, 0.008, size=n))

        candidate = np.clip(candidate, 0.0, 1000.0)
        s_cand = np.sum(candidate)
        if s_cand < 0.01:
            continue
        candidate = candidate * (s0 / s_cand)
        new_score = compute_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            current = candidate
            current_score = new_score

        temperature *= cooling
        if temperature < 1e-8:
            temperature = 1e-8

    return best, best_score


def coordinate_descent(a_init, time_limit, rng):
    """
    Coordinate-wise optimization: for each coordinate, find the optimal value
    by 1D search, cycling through all coordinates.
    """
    a = a_init.astype(np.float64).copy()
    n = len(a)
    s0 = np.sum(a)
    current_score = compute_score(a)
    best_a = a.copy()
    best_score = current_score
    t0 = time.monotonic()

    coord_order = np.arange(n)
    alpha_cd = 0.1

    while time.monotonic() - t0 < time_limit:
        rng.shuffle(coord_order)
        for p in coord_order:
            if time.monotonic() - t0 > time_limit:
                break
            # Try a few values for coordinate p
            base = a[p]
            for factor in [0.7, 0.85, 1.0, 1.15, 1.3]:
                a[p] = base * factor
                a[p] = max(0.0, a[p])
                s_a = np.sum(a)
                if s_a < 0.01:
                    a[p] = base
                    continue
                # Renormalize
                a_temp = a * (s0 / s_a)
                new_score = compute_score(a_temp)
                if new_score < current_score:
                    a = a_temp
                    current_score = new_score
                    if current_score < best_score:
                        best_score = current_score
                        best_a = a.copy()
                    break
            a[p] = base  # Restore if not improved in this inner loop

    return best_a, best_score


def make_alternating_init(n, ratio, rng):
    """Create alternating high-low initialization."""
    a = np.ones(n)
    a[::2] = 1.0
    a[1::2] = ratio
    a += rng.normal(0, 0.02, n)
    return np.maximum(a, 0.01)


def make_cosine_weighted_init(n, freq, rng):
    """Create cosine-windowed sinusoidal initialization."""
    t = np.arange(n, dtype=np.float64)
    window = 0.5 + 0.5 * np.cos(np.pi * (t / (n - 1) - 0.5))
    oscillation = 1.0 + 0.4 * np.cos(2 * np.pi * freq * t / n)
    a = window * oscillation
    a += rng.normal(0, 0.03, n)
    return np.maximum(a, 0.01)


def resample_sequence(seq, new_n):
    """Resample a sequence to a new length."""
    x_old = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, seq)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: Fast subgradient descent (60% time)
    Phase 2: Peak shaving + alternating subgradient (20% time)
    Phase 3: Local search polish (15% time)
    Phase 4: Final verification (5% time)
    """
    t_global = time.monotonic()
    deadline = t_global + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Get initial sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if len(start_seq) > 0:
            s = compute_score(start_seq)
            print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
            if s < best_score:
                best_score = s
                best_seq = start_seq.copy()

    n = len(best_seq) if best_seq is not None else 512

    # === Phase 1: Subgradient descent from provided sequence ===
    print("\n=== Phase 1: Subgradient descent (primary) ===")
    if best_seq is not None:
        t1_max = 0.55 * (deadline - t_global)
        t1_max = min(t1_max, 900.0)
        f_best, s_best = subgradient_descent(best_seq, t1_max, rng, verbose=True)
        print(f"  Phase 1 result: score={s_best:.6f}")
        if s_best < best_score:
            best_score = s_best
            best_seq = f_best.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # === Phase 2: Peak shaving and targeted optimization ===
    print(f"\n=== Phase 2: Peak shaving ===")
    time_left = deadline - time.monotonic()
    if time_left > 100 and best_seq is not None:
        t2_max = min(0.25 * (budget_s - 20), time_left * 0.5)
        # Multiple rounds of peak shaving with different parameters
        for round_id in range(3):
            if time.monotonic() > deadline - 200:
                break
            shaved, shave_score = peak_shaving(best_seq, rng, n_steps=200)
            if shave_score < best_score:
                best_score = shave_score
                best_seq = shaved.copy()
                print(f"  Round {round_id}: peak shaving improved to {best_score:.6f}")
            else:
                print(f"  Round {round_id}: no improvement ({shave_score:.6f} vs {best_score:.6f})")
            # Follow up with short subgradient
            if time.monotonic() < deadline - 200:
                f_best, s_best = subgradient_descent(best_seq, 30.0, rng, verbose=False)
                if s_best < best_score:
                    best_score = s_best
                    best_seq = f_best.copy()
                    print(f"    After subgrad: {best_score:.6f}")

    # === Phase 3: Local search polish ===
    print(f"\n=== Phase 3: Local search polish ===")
    if best_seq is not None and time.monotonic() < deadline - 50:
        polish_time = min(250.0, max(30.0, deadline - time.monotonic() - 30))
        print(f"  Starting polish, score={best_score:.6f}, time={polish_time:.1f}s")
        polished, polish_score = local_search_polish(best_seq, polish_time, rng)
        if polish_score < best_score:
            best_score = polish_score
            best_seq = polished.copy()
            print(f"  After polish: {best_score:.6f}")
        else:
            print(f"  Polish didn't improve (best={best_score:.6f})")

    # === Phase 3b: Try alternative initializations ===
    if time.monotonic() < deadline - 100:
        print(f"\n=== Phase 3b: Alternative starts ===")
        inits = [
            make_alternating_init(n, 0.5, rng),
            make_alternating_init(n, 0.3, rng),
            make_alternating_init(n, 0.7, rng),
            make_cosine_weighted_init(n, 1.0, rng),
            make_cosine_weighted_init(n, 2.0, rng),
            make_cosine_weighted_init(n, 3.7, rng),
        ]
        # Also resample best to check
        if best_seq is not None:
            inits.append(best_seq + 0.05 * rng.standard_normal(n))
            inits = [np.maximum(x, 0.01) for x in inits]

        time_per_init = max(15.0, (deadline - time.monotonic() - 80) / len(inits))
        for ii, a0 in enumerate(inits):
            if time.monotonic() > deadline - 80:
                break
            s0 = compute_score(a0)
            print(f"  Init {ii}: initial score={s0:.6f}")
            if s0 < best_score:
                best_score = s0
                best_seq = a0.copy()
            # Short subgradient from this init
            t_avail = min(time_per_init, deadline - time.monotonic() - 60)
            if t_avail > 10:
                f_best, s_best = subgradient_descent(a0, t_avail, rng, verbose=False)
                if s_best < best_score:
                    best_score = s_best
                    best_seq = f_best.copy()
                    print(f"    *** New best from init {ii}: {best_score:.6f} ***")

    # === Phase 4: Final refinement ===
    print(f"\n=== Phase 4: Final refinement ===")
    if best_seq is not None and time.monotonic() < deadline - 15:
        # Short subgradient
        final_sg_time = min(60.0, max(5.0, deadline - time.monotonic() - 10))
        if final_sg_time > 5:
            f_best, s_best = subgradient_descent(best_seq, final_sg_time, rng, verbose=False)
            if s_best < best_score:
                best_score = s_best
                best_seq = f_best.copy()
                print(f"  After final subgrad: {best_score:.6f}")

        # Final local search if time permits
        if time.monotonic() < deadline - 10:
            remaining = deadline - time.monotonic() - 5
            if remaining > 15:
                polished, polish_score = local_search_polish(best_seq, remaining, rng)
                if polish_score < best_score:
                    best_score = polish_score
                    best_seq = polished.copy()
                    print(f"  After final polish: {best_score:.6f}")

    # === Final verification ===
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 512
