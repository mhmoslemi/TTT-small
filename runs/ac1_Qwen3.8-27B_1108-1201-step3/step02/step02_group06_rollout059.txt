    n = len(f)

    # For fixed j, A[:,j] is f shifted by j positions (with zeros)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A

                     method='highs', options={'time_limit': time_limit})
    if result.success:
        return result.x
    return None

Since L >= 3n-2 and m < n, j < n, we have m+j-L < n + n - (3n-2) = 2-2n < 0, so (m+j-L) mod L = m+j-L+L = m+j.

So `R[L-j] = sum_m a[m] * w2[m+j]` (since m+j < 2n <= L for our case).

So `grad[j] = 2 * R[L-j]` where `R = ifft(fft(a, L) * conj(fft(w2, L)))`.

Or equivalently, `grad = 2 * R_reversed` where `R_reversed[j] = R[L-j]`.

def compute_gradient(a, w, tau):
    """Compute gradient of soft-max objective.
    a: length n
    w: soft-max weights, length 2n-1
    Returns: gradient of length n
    """
    n = len(a)
    L = 1
    while L < 3*n - 1:
        L <<= 1
    A = np.fft.rfft(a, L)
    W = np.fft.rfft(w, L)
    R = np.fft.irfft(A * np.conj(W), L)
    # grad[j] = 2 * R[L-j] for j=1,...,L-1, and grad[0] = 2*R[0]
    grad = np.zeros(n)
    for j in range(n):
        idx = (L - j) % L
        grad[j] = 2.0 * R[idx]
    return grad

Actually, the gradient in theta space is:

`= sum_j a[i]*(delta_ij - a[j]/S) * grad_a[j]`  (since softmax(theta)[j] = a[j]/S)

`= a[i] * (grad_a[i] - sum(a*grad_a)/S)`

So: `grad_theta = a * (grad_a - mean_weighted)` where `mean_weighted = sum(a*grad_a)/S`.

"""
Multi-phase optimization for minimizing normalized peak autocorrelation of
non-negative step functions. Combines LP-based direction finding, soft-max
gradient descent with exact FFT-based gradients, top-k peak weighting, and
coordinate descent polishing. Explores multiple sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def build_toeplitz(f):
    """Build constraint matrix for LP: A[k,j] = f[k-j] if 0<=k-j<n."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_direction(f, M, time_limit=20):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    A = build_toeplitz(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                     method='highs', options={'time_limit': time_limit, 'presolve': True})
    if result.success:
        return result.x
    return None


def compute_grad_a(a, w):
    """Compute gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a.
    Uses: grad[j] = 2 * sum_m a[m] * w[j+m]
    Computed via FFT cross-correlation.
    """
    n = len(a)
    L = 1
    while L < 3 * n - 1:
        L <<= 1
    A = np.fft.rfft(a, L)
    W = np.fft.rfft(w, L)
    R = np.fft.irfft(A * np.conj(W), L)
    grad = np.zeros(n)
    for j in range(n):
        idx = (L - j) % L
        grad[j] = 2.0 * R[idx]
    return grad


def gradient_ascent_soft(a_init, time_limit, tau_start=None, tau_end=None,
                         base_lr=0.5, top_k_frac=0.3, verbose=False):
    """Gradient descent on soft-max parameterization with exact gradients.
    Uses top-k peak weighting combined with soft-max for better focus.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a_safe = np.clip(a_init, 1e-10, None)
    theta = np.log(a_safe / np.sum(a_safe))

    if tau_start is None:
        tau_start = 5.0 / n
    if tau_end is None:
        tau_end = 0.1 / n

    momentum1 = np.zeros(n)
    momentum2 = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step_count = 0
    best_theta = theta.copy()
    best_score = local_score(a_safe)
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** elapsed_frac

        # Compute a from theta
        t_shifted = theta - np.max(theta)
        e = np.exp(t_shifted)
        a = e / np.sum(e) * S

        # Compute convolution and weights
        conv = fft_convolve(a, a)
        conv_shifted = conv - np.max(conv)
        exp_conv = np.exp(conv_shifted / tau * tau)  # = exp(conv - max)
        w = exp_conv / np.sum(exp_conv)

        # Top-k reweighting: boost the top peaks
        k_top = max(1, int(top_k_frac * len(conv)))
        top_indices = np.argsort(conv)[-k_top:]
        w_boosted = w.copy()
        top_weight = np.sum(w[top_indices])
        if top_weight > 1e-15:
            w_boosted[top_indices] *= 3.0 / (1.0 + 2.0 * top_weight)
            w_boosted /= np.sum(w_boosted)

        # Compute exact gradient
        grad_a = compute_grad_a(a, w_boosted)

        # Convert to theta-space gradient
        inner = np.dot(a, grad_a) / S
        grad_theta = a * (grad_a - inner)

        # Adam update
        step_count += 1
        momentum1 = beta1 * momentum1 + (1.0 - beta1) * grad_theta
        momentum2 = beta2 * momentum2 + (1.0 - beta2) * (grad_theta ** 2)
        m1_hat = momentum1 / (1.0 - beta1 ** step_count)
        m2_hat = momentum2 / (1.0 - beta2 ** step_count)
        lr = base_lr / (1.0 + 0.05 * np.sqrt(step_count))
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)

        # Evaluate
        t_shifted = theta - np.max(theta)
        a_new = np.exp(t_shifted) / np.sum(np.exp(t_shifted)) * S
        score = local_score(a_new)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_theta = theta.copy()

        if verbose and step_count % 500 == 0:
            print(f"    Grad step={step_count}, score={score:.6f}, "
                  f"best={best_score:.6f}, tau={tau:.6f}")

    t_shifted = best_theta - np.max(best_theta)
    a_best = np.exp(t_shifted) / np.sum(np.exp(t_shifted)) * S
    return a_best, local_score(a_best)

def lp_optimize(f_init, time_limit, t_init=0.25, verbose=False):
    """LP-based iterative optimization with adaptive step size and backtracking."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    sum_f = np.sum(f)
    best_f = f.copy()
    best_score = local_score(f)
    t0 = time.monotonic()
    t = t_init
    stagnant = 0
    rng_local = np.random.default_rng(777)
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(f, f)
        M = np.max(conv)

        g = lp_direction(f, M, time_limit=min(15, max(3, (time_limit - (time.monotonic()-t0)) * 0.3)))
        if g is None:
            g = lp_direction(f, M * 1.005, time_limit=min(15, max(3, (time_limit - (time.monotonic()-t0)) * 0.3)))
            if g is None:
                break

        sum_g = np.sum(g)
        if sum_g < 1e-10:
            break

        g_rescaled = g * (sum_f / sum_g)

        # Try multiple step sizes (backtracking)
        best_trial_score = np.inf
        best_trial_f = None
        for trial_t in [t, t * 0.7, t * 0.5, t * 0.3, t * 0.1]:
            f_trial = (1.0 - trial_t) * f + trial_t * g_rescaled
            f_trial = np.clip(f_trial, 0.0, None)
            s = np.sum(f_trial)
            if s < 1e-10:
                continue
            f_trial = f_trial * (sum_f / s)
            score_trial = local_score(f_trial)
            if score_trial < best_trial_score:
                best_trial_score = score_trial
                best_trial_f = f_trial

        if best_trial_f is None:
            t *= 0.5
            continue

        f = best_trial_f
        current_score = best_trial_score
        iteration += 1

        if current_score < best_score - 1e-13:
            best_score = current_score
            best_f = f.copy()
            stagnant = 0
            if t < 0.6:
                t = min(0.6, t * 1.02)
        else:
            stagnant += 1
            if stagnant % 15 == 0:
                t = max(0.03, t * 0.75)
            if stagnant > 40:
                # Add perturbation
                scale = 0.005 + 0.005 * rng_local.random()
                f = f * np.exp(rng_local.normal(0, scale, size=n))
                f = np.clip(f, 0.0, None)
                f = f * (sum_f / np.sum(f))
                t = 0.15
                stagnant = 0

        if verbose and iteration % 30 == 0:
            print(f"    LP iter={iteration}, t={t:.3f}, score={current_score:.6f}, "
                  f"best={best_score:.6f}, elapsed={time.monotonic()-t0:.1f}s")

    return best_f, best_score


def coordinate_descent_polish(f_init, time_limit, rng, verbose=False):
    """Coordinate descent / random coordinate polishing."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    sum_f = np.sum(f)
    best_f = f.copy()
    best_score = local_score(f)
    t0 = time.monotonic()
    alpha = 0.02

    while time.monotonic() - t0 < time_limit:
        # Perturb a random subset of coordinates
        n_perturb = max(1, int(n * 0.05))
        idx = rng.choice(n, size=n_perturb, replace=False)
        f_trial = f.copy()
        f_trial[idx] *= np.exp(rng.normal(0, alpha, size=n_perturb))
        f_trial = np.clip(f_trial, 0.0, None)
        s = np.sum(f_trial)
        if s < 1e-10:
            continue
        f_trial = f_trial * (sum_f / s)

        score = local_score(f_trial)
        # Simple annealing acceptance
        if score < best_score:
            f = f_trial
            best_score = score
            best_f = f_trial.copy()
            alpha = min(0.05, alpha * 1.01)
        elif score < local_score(f):
            f = f_trial
        else:
            # Occasionally accept bad moves to escape
            if rng.random() < 0.05 and score < best_score + 0.01:
                f = f_trial
            alpha = max(0.001, alpha * 0.999)

        if verbose and (time.monotonic() - t0) % 10 < 0.01:
            print(f"    CoordDesc: score={score:.6f}, best={best_score:.6f}, "
                  f"alpha={alpha:.5f}, elapsed={time.monotonic()-t0:.1f}s")

    return best_f, best_score


def make_chirp_init(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_init(n, rng, spread=0.3):
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multitone_init(n, rng, n_tones=8):
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for _ in range(n_tones):
        k = int(rng.integers(1, max(2, n // 2)))
        amp = rng.uniform(0.05, 0.25)
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_interpolated_init(provided_seq, n, rng, noise=0.0):
    """Interpolate provided sequence to new length."""
    if len(provided_seq) == n:
        a = provided_seq.copy()
    else:
        x_old = np.linspace(0, 1, len(provided_seq))
        x_new = np.linspace(0, 1, n)
        a = np.interp(x_new, x_old, provided_seq)
    if noise > 0:
        a += noise * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: LP-based optimization on multiple initializations/lengths.
    Phase 2: Soft-max gradient descent with exact FFT gradients.
    Phase 3: Coordinate descent / SA polishing.
    Phase 4: Final LP refinement.
    """
    t_global = time.monotonic()
    deadline = t_global + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check provided sequence
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s_prov = local_score(provided_seq)
        print(f"Provided sequence: n={len(provided_seq)}, score={s_prov:.6f}")
        if s_prov < best_score:
            best_score = s_prov
            best_seq = provided_seq.copy()

    # === Phase 1: LP-based optimization ===
    print("\n" + "=" * 60)
    print("Phase 1: LP-based iterative optimization")
    print("=" * 60)

    n_prov = len(provided_seq) if provided_seq is not None else 256

    # Define search: (length, init_type, time_fraction)
    search_configs = [
        (n_prov, 'provided', 0.25),
        (n_prov, 'chirp', 0.12),
        (512, 'chirp', 0.12),
        (512, 'multitone', 0.10),
        (768, 'random', 0.10),
        (n_prov, 'multitone', 0.10),
        (384, 'chirp', 0.08),
        (1024, 'random', 0.08),
        (n_prov, 'random', 0.05),
    ]

    total_frac = sum(c[2] for c in search_configs)
    available_time = max(60.0, deadline - time.monotonic())
    lp_total_time = available_time * 0.40

    for n, init_name, frac in search_configs:
        elapsed = time.monotonic() - t_global
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        time_alloc = min(frac / total_frac * lp_total_time, remaining - 15)
        if time_alloc < 10:
            break

        # Generate initialization
        if init_name == 'provided' and provided_seq is not None:
            a0 = make_interpolated_init(provided_seq, n, rng)
        elif init_name == 'chirp':
            alpha = 0.2 + 0.5 * rng.random()
            a0 = make_chirp_init(n, alpha, rng)
        elif init_name == 'multitone':
            a0 = make_multitone_init(n, rng, n_tones=max(4, n // 32))
        else:
            a0 = make_random_init(n, rng, spread=0.15 + 0.2 * rng.random())

        print(f"\n  n={n}, init='{init_name}', alloc={time_alloc:.0f}s, "
              f"init_score={local_score(a0):.6f}")
        t_start = time.monotonic()

        f_best, score = lp_optimize(a0, time_alloc, t_init=0.25, verbose=False)
        elapsed_phase = time.monotonic() - t_start
        print(f"  LP done: score={score:.6f}, time={elapsed_phase:.1f}s")

        if score < best_score:
            best_score = score
            best_seq = f_best.copy()
            print(f"  *** New best: {best_score:.6f} (n={n}, init={init_name}) ***")

    # === Phase 2: Gradient descent polishing ===
    print("\n" + "=" * 60)
    print("Phase 2: Soft-max gradient descent (exact FFT gradients)")
    print("=" * 60)

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        grad_time = min(remaining * 0.35, 400.0)
        if grad_time > 30:
            n_best = len(best_seq)
            print(f"  Polishing n={n_best}, time={grad_time:.0f}s")

            # Try both top-k boosted and plain gradient
            half_time = grad_time * 0.55
            a_grad, score_grad = gradient_ascent_soft(
                best_seq, half_time,
                tau_start=5.0 / n_best, tau_end=0.05 / n_best,
                base_lr=0.4, top_k_frac=0.2, verbose=False)
            print(f"  Topk-Grad result: {score_grad:.6f}")
            if score_grad < best_score:
                best_score = score_grad
                best_seq = a_grad.copy()
                print(f"  *** New best: {best_score:.6f} ***")

            remaining2 = deadline - time.monotonic()
            if remaining2 > 30:
                a_grad2, score_grad2 = gradient_ascent_plain(
                    best_seq, min(remaining2 - 10, grad_time * 0.4),
                    tau_start=15.0 / n_best, tau_end=0.2 / n_best,
                    base_lr=0.2, verbose=False)
                print(f"  PlainGrad result: {score_grad2:.6f}")
                if score_grad2 < best_score:
                    best_score = score_grad2
                    best_seq = a_grad2.copy()
                    print(f"  *** New best: {best_score:.6f} ***")

    # === Phase 3: Coordinate descent / SA polishing ===
    print("\n" + "=" * 60)
    print("Phase 3: Coordinate descent polishing")
    print("=" * 60)

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        cd_time = min(remaining * 0.3, 180.0)
        if cd_time > 10:
            print(f"  CoordDesc, n={len(best_seq)}, time={cd_time:.0f}s")
            a_cd, score_cd = coordinate_descent_polish(best_seq, cd_time, rng, verbose=False)
            print(f"  CoordDesc result: {score_cd:.6f}")
            if score_cd < best_score:
                best_score = score_cd
                best_seq = a_cd.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # === Phase 4: Final LP refinement ===
    print("\n" + "=" * 60)
    print("Phase 4: Final LP refinement")
    print("=" * 60)

    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        final_lp_time = min(remaining - 10, 80.0)
        if final_lp_time > 10:
            print(f"  Final LP, n={len(best_seq)}, time={final_lp_time:.0f}s")
            a_final, score_final = lp_optimize(best_seq, final_lp_time, t_init=0.15)
            print(f"  Final LP result: {score_final:.6f}")
            if score_final < best_score:
                best_score = score_final
                best_seq = a_final.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # === Phase 5: One more gradient pass if time ===
    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        if remaining > 10:
            n_best = len(best_seq)
            a_final2, score_final2 = gradient_ascent_soft(
                best_seq, remaining - 5,
                tau_start=3.0 / n_best, tau_end=0.02 / n_best,
                base_lr=0.2, top_k_frac=0.15, verbose=False)
            if score_final2 < best_score:
                best_score = score_final2
                best_seq = a_final2.copy()
                print(f"  Final gradient: {best_score:.6f} *** New best ***")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'=' * 60}")
        print(f"FINAL RESULT: score={final_score:.6f}, n={len(final_list)}")
        print(f"Local score check: {local_score(best_seq):.6f}")
        print(f"{'=' * 60}")
        return final_list

    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]
    return [1.0] * 256
