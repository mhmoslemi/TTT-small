```python
import time

def fast_convolve(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    # Euclidean projection onto {x >= 0, sum(x) = 1}
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_smooth_grad(a, conv, temperature):
    """Compute gradient of the log-sum-exp approximation of max(conv)."""
    n = len(a)
    # Softmax weights
    scaled = conv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    # Gradient: sum_k w[k] * grad_conv[k]
    # grad_conv[k][j] = 2*a[k-j] for valid j

        # Valid j range for lag k

        j_end = min(n - 1, k)
        if j_start > j_end:
            continue
        idx = np.arange(j_start, j_end + 1)
        grad[idx] += 2.0 * w[k] * a[k - idx]
    
    return grad
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth-max gradient descent
with log-sum-exp approximation, mirror descent, spectrum flattening,
and multi-phase search with diverse initializations.
"""

import numpy as np
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_gradient(a, conv, temperature):
    n = len(a)
    scaled = conv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        # Fallback: uniform weights
        w = np.ones(len(conv)) / len(conv)
    else:
        w /= w_sum
    
    grad = np.zeros(n)
    # Only process lags with significant weight
    thresh = w.max() * 1e-6
    active = np.where(w > thresh)[0]
    for k in active:
        ki = int(k)
        j_start = max(0, ki - n + 1)
        j_end = min(n - 1, ki)
        idx = np.arange(j_start, j_end + 1)
        grad[idx] += 2.0 * w[ki] * a[ki - idx]
    return grad


def compute_obj(n, conv):
    return 2.0 * n * conv.max()


def optimize_gradient(a0, time_limit, lr=0.05, momentum=0.9, temp_init=0.001,
                      temp_growth=1.001, batch=20, perturb_every=500,
                      perturb_scale=0.01):
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = compute_obj(n, conv)
    best_a = a.copy()
    
    t0 = time.monotonic()
    temp = temp_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit:
                break
            grad = smooth_gradient(a, conv, temp)
            v = momentum * v - lr * grad
            a = project_simplex(a + v)
            conv = fast_convolve(a)
            cur_obj = compute_obj(n, conv)
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        if no_improve >= perturb_every:
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
            noise = rng.standard_normal(n) * perturb_scale
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            v *= 0.3
            no_improve = 0
            lr = max(0.001, lr * 0.7)
        
        lr *= 0.9999
        temp *= temp_growth
    
    return best_a, best_obj


def optimize_mirror(a0, time_limit, eta=0.5, t_param=20.0):

def spectrum_flatten(a, n_iter=10, target_ratio=0.5):
    """Flatten the magnitude spectrum to reduce auto-convolution peak."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    a_cur = a.copy()
    for _ in range(n_iter):
        A = np.fft.rfft(a_cur, fft_size)
        mag = np.abs(A)
        # Target: geometric mean of magnitudes (preserves total energy)
        log_mag = np.log(mag + 1e-30)
        target_log_mag = log_mag * target_ratio + np.log(mag + 1e-30).mean() * (1 - target_ratio)
        target_mag = np.exp(target_log_mag)
        # Preserve phase
        phase = A / (mag + 1e-30)
        A_new = target_mag * phase
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a_cur = a_new / s
        else:
            break
    
    return a_cur
```

One important detail: the gradient `smooth_grad` should return the gradient of `max_k g(k)` (approximated by log-sum-exp), NOT divided by `(sum a)^2`. Since I'm working with `sum(a) = 1` (normalized), the objective is just `2*n*max_k g(k)`, and the gradient is `2*n * grad(max_k g(k))`. But since `2*n` is a constant factor, I can ignore it and just minimize `max_k g(k)` (or its smooth approximation).

So the gradient I compute is for `log(sum exp(g(k)/temp))` which approximates `max g(k)`. The gradient is:
`grad = sum_k (exp(g(k)/temp) / sum_j exp(g(j)/temp)) * grad(g(k))`

And `grad(g(k))[j] = 2*a[k-j]` for valid j.

This is the gradient of the smooth approximation, and I use it as the search direction (negate it for descent).

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth log-sum-exp gradient descent,
mirror descent, spectrum flattening, and multi-phase search with diverse initializations.
Key innovations: relative-temperature smooth-max gradient, adaptive backtracking,
spectrum compression preprocessing, and multi-method ensemble polishing.
"""

import numpy as np
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_grad(a, conv, frac):
    """Gradient of log-sum-exp approximation of max(conv), with temp = frac * max(conv)."""
    n = len(a)
    max_conv = conv.max()
    if max_conv < 1e-30:
        return np.zeros(n)
    temp = frac * max_conv
    scaled = conv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(conv)) / len(conv)
    else:
        w /= w_sum
    grad = np.zeros(n)
    thresh = w.max() * 1e-7
    active = np.where(w > thresh)[0]
    for k in active:
        ki = int(k)
        j_start = max(0, ki - n + 1)
        j_end = min(n - 1, ki)
        if j_start > j_end:
            continue
        idx = np.arange(j_start, j_end + 1)
        grad[idx] += 2.0 * w[ki] * a[ki - idx]
    return grad


def compute_obj(n, conv):
    return 2.0 * n * conv.max()


def opt_grad(a0, time_limit, lr=0.04, momentum=0.88, frac_init=0.5,
             frac_end=0.01, perturb_every=400, perturb_scale=0.008):
    """Projected gradient descent with smooth-max and momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = compute_obj(n, conv)
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    total_expected = max(1, int(time_limit / max(0.001, 2e-5)))
    for step in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        prog = min(1.0, step / max(1, perturb_every * 3))
        frac = frac_init * (frac_end / frac_init) ** prog
        grad = smooth_grad(a, conv, frac)
        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)
        conv_new = fast_convolve(a_new)
        obj_new = compute_obj(n, conv_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        a = a_new
        conv = conv_new
        if no_improve >= perturb_every:
            rng_local = np.random.default_rng(step * 7919 + n)
            noise = rng_local.standard_normal(n) * perturb_scale
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            v *= 0.2
            no_improve = 0
            lr = max(0.001, lr * 0.6)
        lr *= 0.99995
    return best_a, best_obj


def opt_mirror(a0, time_limit, eta=0.4, t_param=25.0):
    """Multiplicative mirror descent."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    conv = fast_convolve(a)
    best_obj = compute_obj(n, conv)
    best_a = a.copy()
    t0 = time.monotonic()
    t_cur = t_param
    while time.monotonic() - t0 < time_limit:
        max_conv = conv.max()
        if max_conv < 1e-15:
            break
        scaled = t_cur * conv / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(conv))
            peak_idx = int(np.argmax(conv))
            w[peak_idx] = 1.0
        else:
            w /= w_sum
        sg = np.zeros(n)
        thresh = w.max() * 0.005
        active = np.where(w > thresh)[0]
        for k in active:
            ki = int(k)
            j_start = max(0, ki - n + 1)
            j_end = min(n - 1, ki)
            if j_start > j_end:
                continue
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += 2.0 * w[ki] * a[ki - idx]
        sg = sg - sg.mean()
        log_mult = -eta * sg
        log_mult -= log_mult.max()
        a = a * np.exp(log_mult)
        a = np.maximum(a, 0.0)
        s = a.sum()
        a = a / s if s > 1e-15 else np.ones(n) / n
        conv = fast_convolve(a)
        cur_obj = compute_obj(n, conv)
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
    return best_a, best_obj


def spectrum_flatten(a, n_iter=8, alpha=0.4):
    """Compress dynamic range of |DFT(a)|^2 to flatten auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    a_cur = a.copy()
    for _ in range(n_iter):
        A = np.fft.rfft(a_cur, fft_size)
        mag = np.abs(A)
        log_mag = np.log(mag + 1e-30)
        mean_log = log_mag.mean()
        target_log = alpha * log_mag + (1.0 - alpha) * mean_log
        target_mag = np.exp(target_log)
        phase = A / (mag + 1e-30)
        A_new = target_mag * phase
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a_cur = a_new / s
        else:
            break
    return a_cur


def make_inits(n, parent, rng):
    inits = {}
    x = np.arange(n, dtype=np.float64)
    inits['uniform'] = np.ones(n) / n
    # Anti-palindromic perturbation
    f = np.sin(2 * np.pi * x * 3.0 / n)
    f = f - f.mean()
    a = (1.0 / n) * (1.0 + 0.4 * f / (np.abs(f).max() + 1e-10))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        inits['anti_pal'] = a / s
    # Cosine windows
    for nm, w in [('cos_w', 1.0), ('cos_m', 0.6), ('cos_n', 0.35)]:
        c = 0.5 + 0.5 * np.cos(2 * np.pi * x / (w * n))
        c = np.maximum(c, 0.005)
        c /= c.sum()
        inits[nm] = c
    # Ramp
    r = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = r / r.sum()
    # Bimodal
    bi = np.exp(-x / (0.12 * n)) + np.exp(-(n - 1 - x) / (0.12 * n))
    bi /= bi.sum()
    inits['bimodal'] = bi
    # Random
    inits['rand'] = rng.dirichlet(np.ones(n) * 0.4)
    # Log-uniform random
    lu = 10.0 ** rng.uniform(-2.5, 0.0, size=n)
    inits['log_rand'] = lu / lu.sum()
    # Two-scale structure
    period = max(3, n // 8)
    two = np.ones(n)
    for i in range(n):
        two[i] = 1.0 + 0.3 * np.sin(2 * np.pi * i / period)
    two = np.maximum(two, 0.1)
    two /= two.sum()
    inits['two_scale'] = two
    # Parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0.0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            elif len(p) > n:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0.0)
                ps2 = pp.sum()
                if ps2 > 1e-10:
                    inits['parent_ds'] = pp / ps2
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0.0)
                ps2 = pp.sum()
                if ps2 > 1e-10:
                    inits['parent_up'] = pp / ps2
    return inits
```
