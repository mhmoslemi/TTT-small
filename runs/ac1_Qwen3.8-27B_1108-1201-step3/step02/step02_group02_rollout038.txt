"""
Hybrid optimization for minimizing peak autocorrelation of non-negative sequences.
Uses mirror descent, subgradient methods, and simulated annealing.
"""

import numpy as np
import time

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def get_score(a):
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)

def get_conv_and_peaks(a):
    c = fft_convolve(a, a)
    max_c = np.max(c)
    peak_mask = c >= max_c * (1 - 1e-7)
    peaks = np.where(peak_mask)[0]
    return c, max_c, peaks

def compute_subgradient(a, peaks):
    """g[l] = (1/|S|) * sum_{k in S} 2*a[k-l] for valid k-l"""
    n = len(a)
    g = np.zeros(n)
    for k in peaks:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
    g /= len(peaks)
    return g

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def mirror_descent_step(a, g, lr):
    """Exponentiated gradient step."""
    log_a = np.log(a + 1e-30) - lr * g
    log_a -= np.max(log_a)
    a_new = np.exp(log_a)
    s = np.sum(a_new)
    if s < 1e-30:
        return a.copy()
    return a_new / s

def gradient_step_projected(a, g, lr):
    """Gradient step with projection onto simplex."""
    a_new = a - lr * g
    return simplex_project(a_new)

def make_asymmetric_init(n, rng):
    """Initialize with asymmetric pattern to break palindrome symmetry."""
    t = np.arange(n)
    a = 1.0 + 0.3 * np.sin(2 * np.pi * t / n) + 0.2 * np.cos(4 * np.pi * t / n)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

def make_ramp_init(n, rng):
    """Ramp initialization."""
    t = np.arange(n) / n
    a = 1.0 + 0.5 * t + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

def make_bimodal_init(n, rng):
    """Bimodal initialization."""
    t = np.arange(n)
    a = 1.0 + 0.4 * np.cos(2 * np.pi * t / n)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

def make_random_init(n, rng):
    a = 1.0 + 0.3 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)

def optimize_mirror(a, time_limit, rng, verbose=False):
    """Mirror descent optimization."""
    n = len(a)
    a = a.copy()
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 0.01:
        a = np.ones(n) / n
    else:
        a = a / s
    
    current_score = 2 * n * np.max(fft_convolve(a, a))
    best_score = current_score
    best_a = a.copy()
    lr = 0.1
    t0 = time.monotonic()
    iter_count = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        c, max_c, peaks = get_conv_and_peaks(a)
        g = compute_subgradient(a, peaks)
        
        # Try mirror descent step
        a_new = mirror_descent_step(a, g, lr)
        new_score = 2 * n * np.max(fft_convolve(a_new, a_new))
        
        if new_score < current_score:
            a = a_new
            current_score = new_score
            lr = min(lr * 1.02, 1.0)
            stall = 0
            iter_count += 1
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
                if verbose and iter_count % 1000 == 0:
                    print(f"    iter={iter_count}, score={best_score:.6f}, lr={lr:.4f}")
        else:

            lr *= 0.9

                    a = np.clip(a, 0.001, None)
                    a /= np.sum(a)
                    current_score = 2 * n * np.max(fft_convolve(a, a))
                    lr = 0.1
                    stall = 0
    
    return best_a, best_score

def optimize_subgrad(a, time_limit, rng, verbose=False):

    n = len(a)
    a = a.copy()
    a = np.clip(a, 0, None)
    s = np.sum(a)

    else:

    stall = 0

        g = compute_subgradient(a, peaks)

def polish_anneal(a, time_limit, rng):

    n = len(a)
    current = a.copy()
    current = np.clip(current, 0, None)
    s = np.sum(current)
    if s < 0.01:
        current = np.ones(n) / n
    else:
        current = current / s

    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()
    temp = 0.002
    cooling = 0.9998
    
    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()

            # Targeted: reduce contribution to peak
            c = fft_convolve(current, current)
            peak_idx = int(np.argmax(c))
            # Find indices that contribute most to this peak
            lo = max(0, peak_idx - n + 1)
            hi = min(peak_idx, n - 1)
            contribs = np.zeros(hi - lo + 1)
            for i, l in enumerate(range(lo, hi + 1)):
                contribs[i] = current[l] * current[peak_idx - l]
            # Reduce top contributors, increase others
            top_k = max(2, len(contribs) // 5)
            top_local = np.argsort(contribs)[-top_k:]
            top_global = np.array(range(lo, hi+1))[top_local]
            candidate[top_global] *= np.exp(rng.normal(-0.02, 0.01, len(top_global)))
            # Increase some random others
            others = np.setdiff1d(np.arange(n), top_global)
            if len(others) > 3:
                inc_idx = rng.choice(others, min(5, len(others)), replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.02, 0.01, len(inc_idx)))
        elif r < 0.6:
            # Pair swap
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                delta = rng.uniform(-0.1, 0.1) * current[i]
                candidate[i] = max(0, current[i] - delta)
                candidate[j] = max(0, current[j] + delta)
        elif r < 0.8:
            # Block perturbation
            width = int(rng.integers(2, max(3, n // 8)))
            start = int(rng.integers(0, n - width))

        else:

            current_score = new_score

"""
Hybrid optimization for minimizing peak autocorrelation of non-negative sequences.
Combines mirror descent (exponentiated gradient), projected subgradient descent,
adaptive coordinate moves, and simulated annealing polish. Uses FFT-accelerated
convolution for fast evaluation. Multiple diverse restarts to escape local minima.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def get_score(a):
    """Compute the evaluation score (scale-invariant version)."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)


def get_conv_and_peaks(a, tol=1e-7):
    """Get convolution, max value, and peak indices."""
    c = fft_convolve(a, a)
    max_c = np.max(c)
    peak_mask = c >= max_c * (1.0 - tol)
    peaks = np.where(peak_mask)[0]
    return c, max_c, peaks


def compute_subgradient(a, peaks):
    """Compute subgradient of max_k (a*a)[k] w.r.t. a.
    g[l] = (1/|S|) * sum_{k in S} 2*a[k-l] for valid indices."""
    n = len(a)
    g = np.zeros(n)
    for k in peaks:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            # g[lo:hi+1] += 2 * a[k-hi : k-lo+1] reversed
            segment = a[k - hi:k - lo + 1][::-1]
            g[lo:hi + 1] += 2.0 * segment
    g /= len(peaks)
    return g


def simplex_project(v):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    s = np.sum(v)
    if s <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    rho_mask = u * idx > (cssv - 1.0)
    if not np.any(rho_mask):
        return np.ones(n) / n
    rho = np.nonzero(rho_mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def normalize(a):
    """Normalize a to sum=1, ensuring non-negativity."""
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def mirror_step(a, g, lr):
    """Exponentiated gradient (mirror descent) step."""
    log_a = np.log(a + 1e-30) - lr * g
    log_a -= np.max(log_a)
    a_new = np.exp(log_a)
    s = np.sum(a_new)
    if s < 1e-30:
        return a.copy()
    return a_new / s


def gradient_project_step(a, g, lr):
    """Gradient step with projection onto simplex."""
    a_new = a - lr * g
    return simplex_project(a_new)


def make_asymmetric_init(n, rng):
    """Asymmetric sinusoidal initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + 0.35 * np.sin(2.0 * np.pi * t / n) + 0.15 * np.cos(4.0 * np.pi * t / n)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_ramp_init(n, rng):
    """Ramp initialization for anti-palindromic structure."""
    t = np.arange(n) / n
    a = 1.0 + 0.4 * t + 0.08 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_bimodal_init(n, rng):

    a += 0.04 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multi_sine_init(n, rng):

    a = np.ones(n)

        phase = rng.uniform(0, 2 * np.pi)

    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

def coordinate_descent_pass(a, n_peaks_check=5):
    """One pass of coordinate descent trying to reduce peak."""
    n = len(a)
    a = a.copy()
    c, max_c, peaks = get_conv_and_peaks(a)

    for _ in range(n):
        j = _  # Visit coordinate j
        # Try small perturbations of a[j]
        best_val = a[j]
        best_local_score = 2.0 * n * max_c  # current score (normalized)

        for delta_frac in [0.7, 0.85, 1.15, 1.3, 0.5, 1.5]:
            candidate = a.copy()
            candidate[j] = a[j] * delta_frac
            if candidate[j] < 0:
                continue
            cs = np.sum(candidate)
            if cs < 0.01:
                continue
            conv = fft_convolve(candidate, candidate)
            ms = np.max(conv)
            local_score = 2.0 * n * ms / (cs * cs)
            if local_score < best_local_score:
                best_local_score = local_score
                best_val = candidate[j]

        a[j] = best_val

    a = normalize(a)
    return a

    cooling = 0.99975
    it = 0

    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()

        if r < 0.35:
            # Targeted peak reduction
            c = fft_convolve(current, current)
            max_c = np.max(c)
            peak_mask = c >= max_c * (0.999)
            peak_indices = np.where(peak_mask)[0]
            # Pick a random peak
            pk = peak_indices[rng.integers(0, len(peak_indices))]
            lo = max(0, pk - n + 1)
            hi = min(pk, n - 1)
            # Contribution of each index to this peak
            contribs = np.zeros(hi - lo + 1)
            for i, l in enumerate(range(lo, hi + 1)):
                contribs[i] = current[l] * current[pk - l]
            top_k = max(3, (hi - lo + 1) // 4)
            top_local = np.argsort(contribs)[-top_k:]
            top_global = np.array(range(lo, hi + 1))[top_local]
            candidate[top_global] *= np.exp(rng.normal(-0.015, 0.008, len(top_global)))
            others = np.setdiff1d(np.arange(n), top_global)
            if len(others) > 5:
                n_inc = min(8, len(others) // 3)
                inc_idx = rng.choice(others, n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.015, 0.008, n_inc))
        elif r < 0.55:
            # Pair redistribution
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                delta = rng.uniform(-0.08, 0.08) * current[i]
                candidate[i] = max(0, current[i] - delta)
                candidate[j] = max(0, current[j] + delta)
        elif r < 0.75:
            # Block perturbation
            width = int(rng.integers(2, max(3, n // 6)))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.01, width))
        elif r < 0.9:
            # Multi-point adjustment guided by subgradient
            c, max_c, peaks = get_conv_and_peaks(current)
            g = compute_subgradient(current, peaks)
            g_norm = np.linalg.norm(g)
            if g_norm > 0:
                g = g / g_norm
                n_adj = int(rng.integers(3, min(15, n // 4)))
                adj_idx = rng.integers(0, n, n_adj)
                for idx in adj_idx:
                    candidate[idx] = max(0, current[idx] - 0.01 * g[idx])
        else:
            # Global small perturbation
            candidate *= np.exp(rng.normal(0, 0.004, n))

        candidate = np.clip(candidate, 0, None)
        cs = np.sum(candidate)
        if cs < 0.01:
            continue
        candidate = candidate / cs

        new_score = 2.0 * n * np.max(fft_convolve(candidate, candidate))

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        else:
            accept_prob = np.exp(-(new_score - current_score) / max(temp, 1e-15))
            if rng.random() < accept_prob:
                current = candidate
                current_score = new_score

        temp *= cooling
        it += 1

    return best, best_score
