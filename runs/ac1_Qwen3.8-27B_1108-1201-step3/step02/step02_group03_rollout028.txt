"""
Optimizes non-negative sequences to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Strategy: Multi-start search with L-BFGS on soft-max smoothed objective,
adaptive temperature annealing, LP-based refinement, and subgradient polishing.
Explores multiple sequence lengths and diverse initializations.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def normalize_sum(a, target=1.0):
    """Normalize a to have the given sum."""
    S = a.sum()
    if S < 1e-15:
        return np.full_like(a, target / len(a))
    return a * (target / S)

def softmax_value_and_grad(a, tau):
    """
    Compute soft-max of convolve(a,a) and its gradient w.r.t. a.
    soft_max = (1/tau) * log(sum exp(tau * c_k))
    grad = 2 * correlate(weights, a, 'valid') where weights = exp(tau*(c-cmax))/sum
    """
    n = len(a)
    c = np.convolve(a, a)
    c_max = c.max()
    # For numerical stability
    shifted = tau * (c - c_max)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -700, 700)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= w_sum
    soft_max = (c_max + np.log(w_sum)) / tau
    # Gradient: grad[p] = 2 * sum_j w[p+j] * a[j]
    grad = 2.0 * np.correlate(w, a, mode='valid')
    return soft_max, grad, c_max

def objective_and_grad(a, tau):
    """Objective (scaled) and gradient for L-BFGS.
    We minimize 2n * soft_max / S^2 subject to a >= 0.
    For efficiency, we fix sum(a) = 1 and minimize 2n * soft_max.
    """
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return 2.0 * n, np.zeros_like(a)
    soft_max, grad_c, _ = softmax_value_and_grad(a, tau)
    # f = 2n * soft_max / S^2
    # df/da = (2n/S^2) * grad_c - (4n * soft_max / S^3) * ones
    obj = 2.0 * n * soft_max / (S * S)
    grad = (2.0 * n / (S * S)) * grad_c - (4.0 * n * soft_max / (S * S * S)) * 1.0
    return obj, grad

def gradient_step_hard(a, lr, momentum, decay=0.9):
    """One step of gradient descent on the hard-max objective."""
    n = len(a)
    S = a.sum()
    c = np.convolve(a, a)
    k_star = int(np.argmax(c))
    
    # Gradient of max(c) w.r.t. a:
    # grad[p] = 2*a[k_star - p] if 0 <= k_star - p < n
    grad = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    for p in range(lo, hi + 1):
        grad[p] = 2.0 * a[k_star - p]
    
    # Full gradient of objective 2n*max/S^2

    a_new = a + momentum
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize_sum(a_new, S)
    
    return a_new, momentum

def lp_refine(f, M, time_limit=None):
    """
    LP: max sum(g) s.t. conv(f,g) <= M, g >= 0.
    """
    n = len(f)
    if n > 1024:
        return None
    
    c_obj = -np.ones(n)
    # Build constraint matrix sparsely
    # A[m, j] = f[m-j] if 0 <= m-j < n
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = f
    
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                     options={'time_limit': time_limit or 30, 'presolve': True})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None
