Hybrid optimizer: subgradient + LP + coordinate descent for minimizing
peak convolution of non-negative sequences.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]

def score_fast(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))

def conv_and_grad(a):
    """Return (convolution, subgradient of max conv)."""
    n = len(a)
    c = fft_convolve(a, a)
    max_c = np.max(c)
    # Positions within 1% of max
    thresh = max_c * 0.95
    peak_pos = np.where(c >= thresh)[0]
    g = np.zeros(n)
    for k in peak_pos:
        k = int(k)
        # dC[k]/da[j] = 2*a[k-j] if 0 <= k-j < n
        lo = max(0, k - (n-1))
        hi = min(n-1, k)
        g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
    g /= len(peak_pos)
    return c, g, max_c

def project_simplex(a, target_sum=1.0):
    """Project a onto {x >= 0, sum(x) = target_sum}."""
    n = len(a)
    # Find lambda such that sum(max(a - lambda, 0)) = target_sum
    a_sorted = np.sort(a)[::-1]
    cumulative = np.cumsum(a_sorted)
    rho = np.where(a_sorted > (cumulative - target_sum) / np.arange(1, n+1))[0]
    if len(rho) == 0:
        rho = 0
    else:
        rho = rho[-1]
    lambda_val = (cumulative[rho] - target_sum) / (rho + 1)
    return np.clip(a - lambda_val, 0.0, None)

def subgradient_step(a, lr=0.01):
    """One subgradient step to reduce max conv, maintaining sum=1, a>=0."""
    c, g, max_c = conv_and_grad(a)
    a_new = a - lr * g
    a_new = project_simplex(a_new, target_sum=1.0)
    return a_new

def coordinate_descent(a, n_iterations=100):
    """Coordinate descent: optimize pairs of coordinates to reduce peak."""
    n = len(a)
    a = a.copy()
    S = np.sum(a)
    for it in range(n_iterations):
        c = fft_convolve(a, a)
        k_peak = int(np.argmax(c))
        # Find the main contributors to C[k_peak]
        # C[k_peak] = sum_j a[j]*a[k_peak-j]
        contribs = np.zeros(n)
        for j in range(n):

            if 0 <= idx < n:

        # Reduce the largest contributors
        top_k = min(10, n//4)
        top_indices = np.argsort(contribs)[-top_k:]

                else:

        # Re-normalize
