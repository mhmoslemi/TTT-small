"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase strategy:
1. Fast subgradient descent from multiple initializations and lengths
2. LP-guided direction finding with line search
3. Coordinate descent and pairwise local search for polishing
4. Simulated annealing with structured moves for escaping local minima
Combines analytical subgradient computation with periodic LP refinement.
"""

import numpy as np
import scipy.optimize
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)[:size]
    result[result < 0] = 0.0  numerical noise fix
    return result


def obj_norm(a):
    """Objective: 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def compute_subgradient(a):
    """Compute subgradient of max_k conv(a,a)[k] at point a."""
    n = len(a)
    conv = fast_convolve(a)
    k_star = int(np.argmax(conv))
    rev_a = a[::-1]
    shift = n - 1 - k_star
    grad = np.zeros(n)
    i_lo = max(0, -shift)
    i_hi = min(n, n - shift)
    if i_lo < i_hi:
        grad[i_lo:i_hi] = 2.0 * rev_a[shift + i_lo:shift + i_hi]
    return grad, k_star, conv.max()


def subgradient_step(a, lr):
    """One subgradient step with projection onto non-negative simplex."""
    grad, _, _ = compute_subgradient(a)
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-12:
        return a.copy()
    return a_new / s


def subgradient_optimize(a0, time_limit, lr_init=0.08, lr_min=0.002, decay=0.9995):
    """Subgradient descent with adaptive step size."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s

    best_a = a.copy()
    best_obj = obj_norm(a)
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    max_no_improve = 500

    while time.monotonic() - t0 < time_limit:
        a = subgradient_step(a, lr)
        obj = obj_norm(a)
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve % 100 == 0:
            lr *= decay
            if lr < lr_min:
                break
        if no_improve >= max_no_improve:
            break

    return best_a, best_obj


def build_lp_matrix(n):
    """Build the constraint matrix for LP: rows are conv indices, cols are sequence indices."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(2 * n - 1, j + n)
        A[j:end, j] = 1.0  # placeholder, will be filled per f
    return A


def solve_lp_direction(f, M):
    """Find g >= 0 maximizing sum(g) subject to conv(f,g) <= M for all k."""
    n = len(f)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(2 * n - 1, j + n)
        A_ub[j:end, j] = f[:end - j]

    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return f.copy()


def line_search(f, g, max_iter=40):
    """Golden section search for best t in [0,1] for (1-t)*f + t*g."""
    def f_of_t(t):
        a = (1.0 - t) * f + t * g
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            return np.inf
        return obj_norm(a / s)

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f_of_t(c)
    fd = f_of_t(d)
    for _ in range(max_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f_of_t(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f_of_t(d)
    t = (lo + hi) / 2.0
    return t, f_of_t(t)


def lp_refine(a0, time_limit, max_no_improve=80):
    """LP-based iterative refinement."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s

    best_a = a.copy()
    best_obj = obj_norm(a)
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        if M < 1e-14:
            break

        g = solve_lp_direction(a, M)

        if np.allclose(g, a, atol=1e-10):
            no_improve += 5
        else:
            t, obj_t = line_search(a, g)
            if obj_t < best_obj - 1e-10:
                a_new = (1.0 - t) * a + t * g
                a_new = np.maximum(a_new, 0.0)
                ns = a_new.sum()
                if ns < 1e-12:
                    break
                a = a_new / ns
                best_obj = obj_t
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

        if no_improve >= max_no_improve:
            break

    return best_a, best_obj

def pairwise_local_search(a0, time_limit):
    """Targeted pairwise local search: move mass from peak contributors to safe positions."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        return a.copy(), np.inf
    a = a / s
    best_obj = obj_norm(a)
    t0 = time.monotonic()
    stuck = 0
    max_stuck = 200

    while time.monotonic() - t0 < time_limit and stuck < max_stuck:
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))

        # Compute contributions to the peak
        contrib = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        for i in range(lo, hi + 1):
            contrib[i] = a[i] * a[k_star - i]

        max_c = contrib.max()
        if max_c < 1e-15:
            break

        # Also consider top few peaks
        top_peaks = np.argsort(conv)[-5:][::-1]

        danger_threshold = 0.5 * max_c
        safe_threshold = 0.1 * max_c
        danger = np.where(contrib > danger_threshold)[0]
        safe = np.where(contrib < safe_threshold)[0]

        if len(danger) == 0 or len(safe) == 0:
            stuck += 1
            continue

        found = False
        # Try multiple fractions and target peaks
        for kp in top_peaks[:3]:
            if found:
                break
            c_kp = np.zeros(n)
            lo2 = max(0, kp - n + 1)
            hi2 = min(n - 1, kp)
            for i in range(lo2, hi2 + 1):
                c_kp[i] = a[i] * a[kp - i]
            max_c2 = c_kp.max()
            if max_c2 < 1e-15:
                continue
            danger2 = np.where(c_kp > 0.5 * max_c2)[0]
            safe2 = np.where(c_kp < 0.1 * max_c2)[0]
            if len(danger2) == 0 or len(safe2) == 0:
                continue

            for di in danger2[:20]:
                if found:
                    break
                for si in safe2[:20]:
                    if found or di == si:
                        continue
                    for frac in [0.05, 0.1, 0.2, 0.35, 0.5]:
                        amount = a[di] * frac
                        if amount < 1e-14:
                            continue
                        a_trial = a.copy()
                        a_trial[di] -= amount
                        a_trial[si] += amount
                        a_trial = np.maximum(a_trial, 0.0)
                        ts = a_trial.sum()
                        if ts < 1e-12:
                            continue
                        a_trial /= ts
                        o = obj_norm(a_trial)
                        if o < best_obj - 1e-10:
                            best_obj = o
                            a = a_trial.copy()
                            found = True
                            break

        if not found:
            stuck += 1
        else:
            stuck = 0

    return a, best_obj


def simulated_annealing(a0, time_limit, T_init=0.1, T_final=0.001):
    """Simulated annealing with structured moves."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_obj = obj_norm(a)
    current_obj = best_obj
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    total_time = max(1.0, time_limit)
    n_iter = max(100, int(total_time * 50))  # estimate iterations
    for it in range(n_iter):
        if time.monotonic() - t0 >= time_limit:
            break
        # Annealing schedule
        frac = it / n_iter
        T = T_init * (T_final / T_init) ** frac

        # Generate a move
        move_type = rng.integers(0, 4)
        a_trial = a.copy()

        if move_type == 0:
            # Single coordinate perturbation
            i = rng.integers(0, n)
            delta = rng.normal(0, 0.05 / n)
            a_trial[i] = max(0.0, a[i] + delta)
        elif move_type == 1:
            # Pair exchange
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                amount = rng.uniform(0, 0.1) * min(a[i], a[j])
                a_trial[i] = max(0.0, a[i] - amount)
                a_trial[j] = a[j] + amount
        elif move_type == 2:
            # Block perturbation
            w = max(1, n // 16)
            i = rng.integers(0, n - w)
            scale = rng.uniform(0.5, 1.5)
            a_trial[i:i+w] *= scale
        else:
            # Random redistribution in a window
            w = max(2, n // 8)
            i = rng.integers(0, n - w)
            window = a[i:i+w].copy()
            window = window + rng.normal(0, 0.02/n, w)
            window = np.maximum(window, 0)
            sw = window.sum()
            if sw > 0:
                window = window * (a[i:i+w].sum() / sw)
                a_trial[i:i+w] = window

        a_trial = np.maximum(a_trial, 0.0)
        st = a_trial.sum()
        if st < 1e-12:
            continue
        a_trial = a_trial / st
        trial_obj = obj_norm(a_trial)

        # Metropolis acceptance
        delta_obj = trial_obj - current_obj
        if delta_obj < 0 or (T > 1e-6 and rng.random() < np.exp(-delta_obj / T)):
            a = a_trial
            current_obj = trial_obj
            if trial_obj < best_obj - 1e-10:
                best_obj = trial_obj
                best_a = a_trial.copy()

    return best_a, best_obj


def get_parent_sequence(n):
    """Get parent sequence clipped/resized to length n, normalized."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim != 1 or p.size == 0 or not np.isfinite(p).all():
            return None
        if p.size >= n:
            pc = p[:n].copy()
        else:
            pc = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, p.size), p)
        pc = np.maximum(pc, 0.0)
        s = pc.sum()
        if s < 1e-10:
            return None
        return pc / s
    except Exception:
        return None


def make_initializations(n, rng):
    """Generate diverse initializations."""
    x = np.arange(n, dtype=np.float64)
    inits = {}

    inits['uniform'] = np.ones(n) / n

    for sf in [0.2, 0.3, 0.5, 0.7]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * 0.5 * n)) ** 2)
        g = np.maximum(g, 0)
        gs = g.sum()
        if gs > 0:
            inits['gauss_{}'.format(sf)] = g / gs

    if n > 1:
        hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1)))
        hann = np.maximum(hann, 0)
        hs = hann.sum()
        if hs > 0:
            inits['hann'] = hann / hs

    # Asymmetric distributions
    for skew in [0.3, 0.5, 0.7]:
        g = np.exp(-0.5 * ((x - skew * n) / (0.3 * n)) ** 2)
        g = np.maximum(g, 0)
        gs = g.sum()
        if gs > 0:
            inits['skew_{}'.format(skew)] = g / gs

    # Edge concentration
    m = max(1, n // 5)
    edges = np.zeros(n)
    edges[:m] = 1.0
    edges[-m:] = 1.0
    inits['edges'] = edges / edges.sum()

    # Random
    r = rng.dirichlet(np.ones(n) * 0.2)
    inits['random'] = r

    # Parent
    parent = get_parent_sequence(n)
    if parent is not None:
        inits['parent'] = parent

    # Chamfering: high in middle, low at edges (opposite of hann)
    chamf = x.astype(np.float64)
    chamf = chamf / chamf.sum() if chamf.sum() > 0 else np.ones(n)/n
    inits['ramp'] = chamf

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase optimization combining subgradient descent,
    LP refinement, coordinate descent, pairwise search, and simulated annealing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Track all candidates for later phases
    candidates = []  # list of (score, n, name, sequence)

    def update_best(seq, label=""):
        nonlocal best_sequence, best_score
        if seq is None:
            return
        score = float(evaluate_sequence([float(x) for x in seq]))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = seq.copy() if isinstance(seq, np.ndarray) else np.array(seq)
            print("  NEW BEST [{}]: score={:.6f}, n={}".format(label, score, len(best_sequence)))

    # ==================== PHASE 1: Quick multi-start scan ====================
    print("Phase 1: Multi-start subgradient optimization...")
    phase1_end = time.monotonic() + (float(budget_s) - 30.0) * 0.30

    lengths = [256, 384, 512, 768]

    # Build list of (n, name, init) to try
    all_configs = []
    for n in lengths:
        inits = make_initializations(n, rng)
        for name, a0 in inits.items():
            all_configs.append((n, name, a0))

    # Prioritize parent and shorter sequences
    def priority_key(item):
        n, name, _ = item
        if name == 'parent':
            return (0, n)
        if n <= 512:
            return (1, n)
        return (2, n)

    all_configs.sort(key=priority_key)

    phase1_time = phase1_end - time.monotonic()
    n_configs = len(all_configs)
    time_per_config = max(5.0, phase1_time / max(1, n_configs))

    for i, (n, name, a0) in enumerate(all_configs):
        remaining = deadline - time.monotonic() - 30.0
        if remaining < 30:
            break
        t_alloc = min(time_per_config, remaining - 20.0)
        if t_alloc < 3.0:
            break

        a_opt, obj_opt = subgradient_optimize(a0, time_limit=t_alloc)
        candidates.append((obj_opt, n, name, a_opt))

        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print("  P1: n={}, {}, score={:.6f}".format(n, name, score))

    print("Phase 1 complete. Best: {:.6f}".format(best_score))

    # ==================== PHASE 2: LP + subgradient on top candidates ====================
    print("Phase 2: LP refinement on top candidates...")
    phase2_end = time.monotonic() + (float(budget_s) - 30.0) * 0.35

    # Sort candidates by objective and pick top ones
    candidates.sort(key=lambda x: x[0])
    top_candidates = []
    seen = set()
    for score, n, name, seq in candidates:
        key = (n, name)
        if key not in seen and len(top_candidates) < 6:
            top_candidates.append((n, name, seq))
            seen.add(key)

    # Also add the current best
    if best_sequence is not None:
        top_candidates.append((len(best_sequence), 'best', best_sequence.copy()))

    remaining = phase2_end - time.monotonic()
    if remaining < 10:
        remaining = max(10.0, (deadline - time.monotonic()) * 0.3)
    time_per_candidate = max(20.0, remaining / max(1, len(top_candidates)))

    for i, (n, name, a0) in enumerate(top_candidates):
        remaining_dl = deadline - time.monotonic() - 60.0
        if remaining_dl < 30:
            break
        t_alloc = min(time_per_candidate, remaining_dl - 30.0)
        if t_alloc < 10.0:
            break

        # LP refinement
        a_lp, obj_lp = lp_refine(a0, time_limit=t_alloc * 0.5)
        score_lp = float(evaluate_sequence(a_lp.tolist()))
        if np.isfinite(score_lp) and score_lp < best_score:
            best_score = score_lp
            best_sequence = a_lp.copy()
            print("  P2 LP: n={}, {}, score={:.6f}".format(n, name, score_lp))

        # Subgradient on LP result
        remaining_sg = deadline - time.monotonic() - 60.0
        if remaining_sg > 10:
            a_sg, obj_sg = subgradient_optimize(a_lp, time_limit=min(remaining_sg * 0.3, 60.0))
            score_sg = float(evaluate_sequence(a_sg.tolist()))
            if np.isfinite(score_sg) and score_sg < best_score:
                best_score = score_sg
                best_sequence = a_sg.copy()
                print("  P2 SG: n={}, {}, score={:.6f}".format(n, name, score_sg))

        # Coordinate descent
        remaining_cd = deadline - time.monotonic() - 60.0
        if remaining_cd > 15:
            a_cd, obj_cd = coordinate_descent(a_lp, time_limit=min(remaining_cd * 0.3, 90.0))
            score_cd = float(evaluate_sequence(a_cd.tolist()))
            if np.isfinite(score_cd) and score_cd < best_score:
                best_score = score_cd
                best_sequence = a_cd.copy()
                print("  P2 CD: n={}, {}, score={:.6f}".format(n, name, score_cd))

    print("Phase 2 complete. Best: {:.6f}".format(best_score))

    # ==================== PHASE 3: Intensive polishing ====================
    print("Phase 3: Intensive polishing on best candidate...")
    if best_sequence is not None:
        polish_deadline = deadline - 15.0
        current = best_sequence.copy()
        current_score = best_score
        iterations = 0

        while time.monotonic() < polish_deadline:
            iterations += 1
            remaining = polish_deadline - time.monotonic()
            if remaining < 10:
                break

            # Alternate between different methods
            method = iterations % 5

            if method == 0:
                # LP step
                t_alloc = min(remaining * 0.2, 120.0)
                a_opt, obj_opt = lp_refine(current, time_limit=t_alloc)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < current_score - 1e-10:
                    current = a_opt.copy()
                    current_score = score
                    if score < best_score:
                        best_score = score
                        best_sequence = current.copy()
                        print("  P3 LP iter {}: score={:.6f}".format(iterations, score))

            elif method == 1:
                # Coordinate descent
                t_alloc = min(remaining * 0.2, 120.0)
                a_opt, obj_opt = coordinate_descent(current, time_limit=t_alloc)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < current_score - 1e-10:
                    current = a_opt.copy()
                    current_score = score
                    if score < best_score:
                        best_score = score
                        best_sequence = current.copy()
                        print("  P3 CD iter {}: score={:.6f}".format(iterations, score))

            elif method == 2:
                # Pairwise local search
                t_alloc = min(remaining * 0.15, 90.0)
                a_opt, obj_opt = pairwise_local_search(current, time_limit=t_alloc)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < current_score - 1e-10:
                    current = a_opt.copy()
                    current_score = score
                    if score < best_score:
                        best_score = score
                        best_sequence = current.copy()
                        print("  P3 PS iter {}: score={:.6f}".format(iterations, score))

            elif method == 3:
                # Simulated annealing (to escape local minima)
                t_alloc = min(remaining * 0.15, 90.0)
                a_opt, obj_opt = simulated_annealing(current, time_limit=t_alloc)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    current = a_opt.copy()
                    current_score = score
                    print("  P3 SA iter {}: score={:.6f}".format(iterations, score))

            else:
                # Subgradient with small step (fine tuning)
                t_alloc = min(remaining * 0.1, 60.0)
                a_opt, obj_opt = subgradient_optimize(current, time_limit=t_alloc,
                                                      lr_init=0.03, lr_min=0.001, decay=0.999)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < current_score - 1e-10:
                    current = a_opt.copy()
                    current_score = score
                    if score < best_score:
                        best_score = score
                        best_sequence = current.copy()
                        print("  P3 SG iter {}: score={:.6f}".format(iterations, score))

    # ==================== PHASE 4: Final verification ====================
    print("Phase 4: Final check...")
    if best_sequence is not None:
        # One more coordinate descent pass
        remaining = deadline - time.monotonic() - 5.0
        if remaining > 5:
            a_final, _ = coordinate_descent(best_sequence, time_limit=min(remaining, 60.0))
            score_final = float(evaluate_sequence(a_final.tolist()))
            if np.isfinite(score_final) and score_final < best_score:
                best_score = score_final
                best_sequence = a_final.copy()
                print("  Final CD: score={:.6f}".format(score_final))

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(512)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    final_score = float(evaluate_sequence([float(x) for x in best_sequence]))
    print("FINAL SCORE: {:.6f}".format(final_score))
    print("Sequence length: {}".format(len(best_sequence)))

    return [float(x) for x in best_sequence]
