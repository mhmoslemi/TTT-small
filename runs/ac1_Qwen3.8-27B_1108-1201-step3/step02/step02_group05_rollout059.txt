"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using soft-max gradient descent
with multiplicative (mirror) updates, followed by LP-based refinement
and local search. Uses analytic subgradient of the auto-convolution peak.
"""

import numpy as np
import scipy.optimize
import time


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_norm(a):
    """Objective: 2n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def compute_soft_max_grad(a, beta):
    """Compute gradient of soft_max(conv(a,a), beta) w.r.t. a.
    grad[j] = 2 * sum_k w[k] * a[k-j] where w[k] = softmax(beta*conv)[k]
    """
    n = len(a)
    conv = fast_convolve(a)
    max_c = conv.max()
    exp_c = np.exp(beta * (conv - max_c))
    w = exp_c / exp_c.sum()
    
    grad = np.zeros(n)
    # grad[j] = 2 * sum_{k=j}^{j+n-1} w[k] * a[k-j] (with bounds on k)
    for j in range(n):
        k_min = j
        k_max = min(j + n - 1, 2 * n - 2)
        if k_min <= k_max:
            grad[j] = 2.0 * np.dot(w[k_min:k_max+1], a[k_min-j:k_max+1-j])
    return grad


def soft_max_value(a, beta):
    """Compute soft_max(conv(a,a), beta)."""
    conv = fast_convolve(a)
    max_c = conv.max()
    return max_c + np.log(np.exp(beta * (conv - max_c)).sum()) / beta


def mirror_descent(a0, beta, lr_init, time_limit, max_iter=10000):

    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()

    lr = lr_init
    t0 = time.monotonic()

        if time.monotonic() - t0 > time_limit:
            break
        
        grad = compute_soft_max_grad(a, beta)

        a_new = np.maximum(a_new, 0.0)
        ns = a_new.sum()

def solve_lp(f, M):
    """Find g >= 0 maximizing sum(g) subject to conv(f,g)[k] <= M for all k."""
    n = len(f)
    # Build constraint matrix
    # conv(f, g)[k] = sum_j f[k-j] * g[j] (with appropriate bounds)
    # For k = 0, ..., 2n-2:
    #   conv[k] = sum_{j=max(0,k-n+1)}^{min(k, n-1)} f[k-j] * g[j]
    
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # g[j] appears in conv[k] for k = j, j+1, ..., j+n-1
        # with coefficient f[k-j]
        for k in range(j, min(j + n, 2 * n - 1)):
            A[k, j] = f[k - j]
    
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return f.copy()


def lp_optimize(f0, time_limit):
    """Iterative LP-based optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0).copy()
    s = f.sum()
    if s < 1e-12:
        f = np.ones_like(f) / len(f)
    else:
        f = f / s

    best_f = f.copy()
    best_obj = obj_norm(f)
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()
        if M < 1e-15:
            break

        g = solve_lp(f, M)

        if np.allclose(g, f, atol=1e-8):
            no_improve += 5
        else:
            # Line search
            best_t = 0.0
            best_obj_t = best_obj
            for t in [0.1, 0.25, 0.5, 0.75, 0.9]:
                a_new = (1.0 - t) * f + t * g
                a_new = np.maximum(a_new, 0.0)
                ns = a_new.sum()
                if ns < 1e-12:
                    continue
                a_new = a_new / ns
                obj_t = obj_norm(a_new)
                if obj_t < best_obj_t:
                    best_obj_t = obj_t
                    best_t = t
            
            if best_obj_t < best_obj - 1e-10:
                a_new = (1.0 - best_t) * f + best_t * g
                a_new = np.maximum(a_new, 0.0)
                ns = a_new.sum()
                a_new = a_new / ns
                f = a_new
                best_obj = best_obj_t
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1

        if no_improve >= 30:
            break

    return best_f, best_obj


def local_search(a, time_limit):

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-stage approach:
1. Soft-max gradient descent with multiplicative (mirror) updates on the simplex
2. LP-based refinement for precise local optimization

import numpy as np
import scipy.optimize
import time


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_norm(a):
    """Objective: 2n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def soft_max_value(a, beta):
    """Compute soft_max(conv(a,a), beta) for smooth approximation of max."""
    conv = fast_convolve(a)
    max_c = conv.max()
    return max_c + np.log(np.exp(beta * (conv - max_c)).sum()) / beta


def compute_soft_max_grad(a, beta):
    """FFT-based gradient of soft_max(conv(a,a), beta) w.r.t. a.
    grad[j] = 2 * sum_i w[i+j] * a[i] where w = softmax(beta * conv)
    """
    n = len(a)
    conv = fast_convolve(a)
    max_c = conv.max()
    exp_c = np.exp(beta * (conv - max_c))
    w = exp_c / exp_c.sum()

    # Cross-correlation via FFT: corr[j] = sum_i w[i+j] * a[i]
    L = 1 << (3 * n).bit_length()
    W = np.fft.rfft(w, L)
    A = np.fft.rfft(a, L)
    corr = np.fft.irfft(W * np.conj(A), L)
    grad = 2.0 * corr[:n]
    return grad


def mirror_descent(a0, beta, lr_init, time_limit, max_iter=20000):

def solve_lp(f, M):
    """Find g >= 0 maximizing sum(g) subject to conv(f,g)[k] <= M for all k."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_end = min(j + n, 2 * n - 1)
        A[j:k_end, j] = f[:k_end - j]

    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return None

                obj_t = obj_norm(a_new)

                best_f = f.copy()

def make_initializations(n, rng):
    """Generate diverse initial sequences of length n."""
    x = np.arange(n, dtype=np.float64)
    inits = []

    # Uniform
    inits.append(('uniform', np.ones(n) / n))

    # Gaussian
    for sf in [0.2, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        inits.append((f'gauss_{sf}', g / g.sum()))

    # Hann window
    if n > 1:
        hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1)))
    else:
        hann = np.ones(n)
    hann = np.maximum(hann, 0.0)
    inits.append(('hann', hann / hann.sum()))

    # Two bumps
    tb = np.zeros(n)
    m = max(1, n // 4)
    tb[:m] = 1.0
    tb[-m:] = 1.0
    inits.append(('two_bump', tb / tb.sum()))

    # Random
    r = rng.dirichlet(np.ones(n) * 0.3)
    inits.append(('random', r))

    # Parent
    parent = get_parent_sequence(n)
    if parent is not None:
        inits.append(('parent', parent))

    return inits

        nonlocal best_sequence, best_score
        a = np.maximum(a.astype(np.float64), 0.0)
        sm = a.sum()
        if sm < 0.01:
            return
        a = a / sm
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  {label}: score={score:.6f}, n={len(a)}")

    # Try multiple lengths
    lengths_to_try = [256, 512, 1024]

    # Phase 1: Gradient descent scan (45% of budget)
    phase1_frac = 0.45
    phase1_end = time.monotonic() + (total_budget - 30.0) * phase1_frac
    print(f"Phase 1: Soft-max gradient descent (budget: {phase1_end - time.monotonic():.0f}s)")

    all_results = []

    for n in lengths_to_try:
        inits = make_initializations(n, rng)

        # Prioritize parent initialization
        init_order = sorted(inits, key=lambda x: 0 if x[0] == 'parent' else 1)

        for name, a0 in init_order:
            for beta in [30, 80, 200]:
                for lr_init in [0.02, 0.1]:
                    remaining = phase1_end - time.monotonic()
                    if remaining < 15:
                        break

                    # Allocate time: more for parent, less for others
                    if name == 'parent':
                        t_alloc = min(remaining * 0.2, 80.0)
                    else:
                        t_alloc = min(remaining * 0.08, 30.0)

                    if t_alloc < 3:
                        break

                    a_opt, obj_opt = mirror_descent(a0, beta, lr_init, t_alloc, max_iter=10000)
                    all_results.append((obj_opt, a_opt, n, name, beta))
                    update_best(a_opt, f"MD n={n}, {name}, beta={beta}")

            # Check time
            if phase1_end - time.monotonic() < 15:
                break

    # Sort by objective
    all_results.sort(key=lambda x: x[0])
    print(f"  Phase 1 done. Best: {best_score:.6f}")

    # Phase 2: LP refinement (30% of budget)
    phase2_frac = 0.30
    phase2_end = time.monotonic() + (total_budget - 30.0) * phase2_frac
    print(f"Phase 2: LP refinement (budget: {phase2_end - time.monotonic():.0f}s)")

    # Take top unique candidates
    top_candidates = []
    seen_keys = set()
    for obj_opt, a_opt, n, name, beta in all_results:
        if n > 600 and len(top_candidates) >= 4:
            continue  # Limit LP to smaller sequences for speed
        key = (n, tuple(np.round(a_opt[:10], 4)))
        if key not in seen_keys:
            top_candidates.append(a_opt)
            seen_keys.add(key)
        if len(top_candidates) >= 8:
            break

    if best_sequence is not None:
        top_candidates.append(best_sequence.copy())

    for i, a0 in enumerate(top_candidates):
        remaining = phase2_end - time.monotonic()
        if remaining < 20:
            break
        n = len(a0)
        t_alloc = min(remaining * 0.25, 60.0)
        if t_alloc < 5:
            break

        if n <= 512:
            a_opt, obj_opt = lp_optimize(a0, t_alloc)
            update_best(a_opt, f"LP n={n}, cand={i}")

        # Local search
        remaining = phase2_end - time.monotonic()
        if remaining > 15:
            ls_time = min(remaining * 0.2, 25.0)
            a_ls, obj_ls = local_search(a_opt, ls_time)
            update_best(a_ls, f"LS n={n}, cand={i}")

    print(f"  Phase 2 done. Best: {best_score:.6f}")

    # Phase 3: Intensive polish (remaining time)
    print("Phase 3: Intensive polish...")
    if best_sequence is not None:
        polish_deadline = deadline - 15.0
        polish_iter = 0
        while time.monotonic() < polish_deadline:
            remaining = polish_deadline - time.monotonic()
            if remaining < 25:
                break
            n = len(best_sequence)

            # Cycle through methods
            method = polish_iter % 4

            if method == 0:
                # Gradient descent with high beta
                beta = min(200 + polish_iter * 30, 800)
                t_alloc = min(remaining * 0.25, 50.0)
                a_opt, obj_opt = mirror_descent(best_sequence, beta, 0.01, t_alloc, max_iter=5000)
                update_best(a_opt, f"Polish MD it={polish_iter} b={beta}")
            elif method == 1:
                # LP (only for n <= 512)
                if n <= 512:
                    t_alloc = min(remaining * 0.25, 50.0)
                    a_opt, obj_opt = lp_optimize(best_sequence, t_alloc)
                    update_best(a_opt, f"Polish LP it={polish_iter}")
                else:
                    # For large n, use gradient with moderate beta
                    t_alloc = min(remaining * 0.25, 50.0)
                    a_opt, obj_opt = mirror_descent(best_sequence, 100, 0.02, t_alloc, max_iter=5000)
                    update_best(a_opt, f"Polish MD-lg it={polish_iter}")
            elif method == 2:
                # Local search
                t_alloc = min(remaining * 0.2, 30.0)
                a_ls, obj_ls = local_search(best_sequence, t_alloc)
                update_best(a_ls, f"Polish LS it={polish_iter}")
            else:
                # Group search
                t_alloc = min(remaining * 0.2, 30.0)
                a_gs, obj_gs = group_search(best_sequence, t_alloc)
                update_best(a_gs, f"Polish GS it={polish_iter}")

            polish_iter += 1

    # Final evaluation
    if best_sequence is not None:
        final_score = float(evaluate_sequence(best_sequence.tolist()))
        if np.isfinite(final_score):
            print(f"FINAL SCORE: {final_score:.6f}")
            print(f"Sequence length: {len(best_sequence)}")
            return [float(x) for x in best_sequence]

    # Fallback
    fallback = np.ones(128)
    fb_score = float(evaluate_sequence(fallback.tolist()))
    print(f"FALLBACK SCORE: {fb_score:.6f}")
    return [float(x) for x in fallback]
