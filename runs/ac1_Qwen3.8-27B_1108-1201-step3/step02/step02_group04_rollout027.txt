"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using log-sum-exp smoothed projected gradient
with tau annealing, multi-start initialization, and LP-based polishing.
"""

import numpy as np
import time
import scipy.optimize as optimize
import scipy.sparse as sparse


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def lse_objective_and_grad(a, tau):
    """Compute log-sum-exp smoothed max of auto-convolution and gradient.
    Returns (objective_value, gradient)."""
    n = len(a)
    conv = fast_convolve(a)
    m = conv.max()
    if m < 1e-30:
        return 0.0, np.zeros(n)
    
    exp_terms = np.exp((conv - m) / tau)
    log_sum = m + tau * np.log(exp_terms.sum())
    obj = log_sum  # This is the smoothed max
    
    w = exp_terms / exp_terms.sum()
    
    # Gradient: grad[i] = 2 * sum_k w[k] * a[k-i]
    # Cross-correlation of w (length 2n-1) and a (length n)

    A = np.fft.rfft(a, size)
    corr = np.fft.irfft(W * np.conj(A), size)
    grad = 2.0 * corr[:n]
    
    return obj, grad

def lse_optimize(a0, time_limit, tau_start=None, tau_end=None, max_iters=10000):

def max_subgradient_refine(a0, time_limit, max_iters=5000):
    """Refine using the true max subgradient with projected gradient."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n

        a /= s

    best_a = a.copy()
    
    eta = 0.1
    t0 = time.monotonic()
    patience = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit - 2:
            break

        max_val = conv.max()
        
        # Get all positions where conv is (close to) max
        threshold = max_val * (1 - 1e-10)
        active = np.where(conv >= threshold)[0]
        
        # Subgradient: grad[i] = sum_{k in active} 2 * a[k-i] (normalized by |active|)
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            grad[j_start:j_end + 1] += 2.0 * a[k - j_end:k - j_start + 1][::-1]
        grad /= len(active)
        
        # Project onto tangent space
        grad_tan = grad - grad.mean()
        
        # Backtracking
        obj = max_val
        alpha = eta
        improved = False
        for back in range(15):
            a_new = a - alpha * grad_tan
            a_new = project_simplex(a_new)
            conv_new = fast_convolve(a_new)
            if conv_new.max() < obj - 1e-14:
                a = a_new
                improved = True
                break
            alpha *= 0.5

            obj_new = fast_convolve(a).max()
            if 2.0 * n * obj_new < best_obj - 1e-12:
                best_obj = 2.0 * n * obj_new
                best_a = a.copy()
            patience = 0
            eta = min(eta * 1.02, 1.0)
        else:
            eta *= 0.5
            patience += 1
            if patience > 100:

def make_lp_structure(n):
    """Precompute for LP."""
    rows_list = []
    cols_list = []
    for i in range(n):
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def solve_lp(f, M, rows, cols, n, timeout=10):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)[k]<=M."""
    vals = np.tile(f, n)
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                               options={'disp': False, 'time_limit': timeout})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def gen_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uni'] = np.ones(n) / n
    
    for sf in [0.1, 0.2, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5*n) / (sf*n))**2)
        g /= g.sum()
        inits[f'g{int(sf*100)}'] = g
    
    # Bimodal
    bi = np.zeros(n)
    w = max(1, n // 3)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    inits['bi'] = bi / bi.sum()
    
    # Cosine
    cb = 0.5 * (1 + np.cos(2*np.pi*x/n))
    inits['cos'] = cb / cb.sum()
    
    # Random
    rr = rng.dirichlet(np.ones(n) * 0.5)
    inits['rnd'] = rr
    
    # Triangle
    tri = np.maximum(1.0 - np.abs(x - 0.5*n)/(0.5*n), 0.05)
    inits['tri'] = tri / tri.sum()
    
    # Parent
    if parent is not None:
        p = np.maximum(parent[:n], 0.0) if parent.size >= n else \
            np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
        p = np.maximum(p, 0.0)
        if p.sum() > 1e-10:
            inits['par'] = p / p.sum()
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using log-sum-exp smoothed projected gradient
with geometric tau annealing, FFT-based gradients, multi-start initialization,
and LP-based polishing. Key innovation: smooth surrogate enables fast FFT-based
gradient computation, allowing many more iterations than LP-based methods.
"""

import numpy as np
import time
import scipy.optimize as optimize
import scipy.sparse as sparse


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def lse_obj_grad(a, tau):
    """Log-sum-exp smoothed max of auto-convolution and its gradient.
    f_tau(a) = tau * log(sum_k exp(conv_k / tau))
    grad[i] = 2 * sum_k w[k] * a[k-i] where w is the softmax weights.
    """
    n = len(a)
    conv = fast_convolve(a)
    m = conv.max()
    if m < 1e-30:
        return 0.0, np.zeros(n)
    
    scaled = (conv - m) / tau
    max_scaled = scaled.max()
    exp_scaled = np.exp(scaled - max_scaled)
    log_sum = m + tau * (max_scaled + np.log(exp_scaled.sum()))
    obj = log_sum
    
    w = exp_scaled / exp_scaled.sum()
    
    # Cross-correlation: grad[i] = 2 * sum_k w[k] * a[k-i]
    corr_size = 1 << (len(w) + n - 1).bit_length()
    W = np.fft.rfft(w, corr_size)
    A = np.fft.rfft(a, corr_size)
    corr = np.fft.irfft(W * np.conj(A), corr_size)
    grad = 2.0 * corr[:n]
    
    return obj, grad


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum(x) = 1}. O(n log n)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    conv0 = fast_convolve(a)
    m0 = conv0.max()
    best_obj = 2.0 * n * m0
    best_a = a.copy()
    
    if tau_fractions is None:
        tau_fractions = [0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]

        tau = frac * m0
        if tau < 1e-12:
            tau = 1e-12

            # Tangent projection (remove mean for simplex constraint)
            grad_tan = grad - grad.mean()
            gnorm = np.linalg.norm(grad_tan)
            if gnorm < 1e-15:
                break
            
            # Backtracking line search
            alpha = eta
            improved = False
            for _ in range(25):
                y = a - alpha * grad_tan
                a_new = project_simplex(y)
                conv_new = fast_convolve(a_new)
                obj_new, _ = lse_obj_grad(a_new, tau)
                
                # Armijo: f(a_new) <= f(a) - c * alpha * grad^T(a - a_new)
                descent = -alpha * np.dot(grad_tan, (a_new - a))
                if obj_new <= obj - 1e-4 * descent + 1e-14:
                    a = a_new
                    improved = True
                    break
                alpha *= 0.5
            
            if improved:
                patience = 0
                eta = min(eta * 1.03, 5.0)
                
                # Track true objective
                conv_now = fast_convolve(a)
                true_obj = 2.0 * n * conv_now.max()
                if true_obj < best_obj - 1e-10:
                    best_obj = true_obj
                    best_a = a.copy()
            else:
                eta *= 0.5
                patience += 1
                if patience > max_stall:
                    break
            
            # Early stop if objective barely changing
            if obj_prev is not None and abs(obj - obj_prev) < 1e-14 * max(1.0, abs(obj)):
                patience += 5
            obj_prev = obj
    
    return best_a, best_obj


def subgradient_refine(a0, time_limit):
    """Refine using true max subgradient with projected gradient."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    eta = 0.05
    t0 = time.monotonic()
    patience = 0
    
    for it in range(20000):
        if time.monotonic() - t0 > time_limit - 1:
            break
        
        conv = fast_convolve(a)
        max_val = conv.max()
        
        # Active set: positions where conv is within 1e-10 of max
        threshold = max_val - 1e-12
        active = np.where(conv >= threshold)[0]
        
        # Subgradient
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            grad[j_start:j_end + 1] += 2.0 * a[k - j_end:k - j_start + 1][::-1]
        grad /= len(active)
        
        grad_tan = grad - grad.mean()
        
        # Backtracking
        alpha = eta
        improved = False
        for _ in range(15):
            a_new = project_simplex(a - alpha * grad_tan)
            conv_new = fast_convolve(a_new)
            if conv_new.max() < max_val - 1e-14:
                a = a_new
                improved = True
                break
            alpha *= 0.5
        
        if improved:
            obj_new = 2.0 * n * fast_convolve(a).max()
            if obj_new < best_obj - 1e-10:
                best_obj = obj_new
                best_a = a.copy()
            patience = 0
            eta = min(eta * 1.01, 0.5)
        else:
            eta *= 0.5
            patience += 1
            if patience > 60:
                break
    
    return best_a, best_obj


def make_lp_structure(n):
    """Precompute sparse structure for cross-convolution constraint matrix."""
    rows_list = []
    cols_list = []
    for i in range(n):
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def lp_polish(f0, time_limit):
    """LP-based alternating optimization for final polishing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f_sum = f.sum()
    if f_sum < 1e-12:
        f = np.ones(n) / n
    else:
        f /= f_sum
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()
    
    rows, cols = make_lp_structure(n)
    
    t0 = time.monotonic()
    t = 1.0
    patience = 0
    
    while time.monotonic() - t0 < time_limit - 2:
        vals = np.tile(f, n)
        A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        c = -np.ones(n)
        b = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        try:
            res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                                   options={'disp': False, 'time_limit': 8})
            if not res.success:
                break
            g = np.maximum(res.x, 0.0)
        except Exception:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g /= g_sum
        
        if np.max(np.abs(g - f)) < 1e-13:
            break
        
        improved = False
        for t_try in [t, 0.7 * t, 0.5 * t, 0.3 * t, 0.1 * t]:
            f_new = (1.0 - t_try) * f + t_try * g
            f_new = np.maximum(f_new, 0.0)
            ss = f_new.sum()
            if ss < 1e-12:
                continue
            f_new /= ss
            conv_new = fast_convolve(f_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj - 1e-12:
                f = f_new
                M = conv_new.max()
                best_obj = obj_new
                best_f = f.copy()
                patience = 0
                t = min(1.0, t * 1.05)
                improved = True
                break
        
        if not improved:
            patience += 1
            t *= 0.8
            if patience > 30:
                break
    
    return best_f, best_obj


def gen_inits(n, rng, parent=None):
    """Generate diverse initializations on the simplex."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    mid = 0.5 * (n - 1)
    
    inits['uni'] = np.ones(n) / n
    
    for sf in [0.1, 0.2, 0.3, 0.5, 0.7]:
        g = np.exp(-0.5 * ((x - mid) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'g{int(sf*100)}'] = g
    
    # Bimodal (two bumps)
    for frac in [0.2, 0.3, 0.4]:
        bi = np.zeros(n)
        w = max(2, int(n * frac))
        bi[:w] = 1.0
        bi[-w:] = 1.0
        inits[f'bi{int(frac*100)}'] = bi / bi.sum()
    
    # Cosine window
    cb = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cb[cb < 0.01] = 0.01
    inits['cos'] = cb / cb.sum()
    
    # Triangle
    tri = np.maximum(1.0 - np.abs(x - mid) / (0.5 * n), 0.02)
    inits['tri'] = tri / tri.sum()
    
    # Random (concentrated)
    rr = rng.dirichlet(np.ones(n) * 0.3)
    inits['rnd'] = rr
    
    # Random (spread)
    rr2 = rng.dirichlet(np.ones(n) * 1.0)
    inits['rnd2'] = rr2
    
    # Flat with small random perturbation
    pert = np.ones(n) + 0.1 * rng.standard_normal(n)
    pert = np.maximum(pert, 0)
    inits['uni_pert'] = pert / pert.sum()
    
    # Parent
    if parent is not None:
        if parent.size >= n:
            p = np.maximum(parent[:n], 0.0)
        else:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
        p = np.maximum(p, 0.0)
        if p.sum() > 1e-10:
            inits['par'] = p / p.sum()
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main entry: multi-phase optimization combining LSE smoothed gradient,
    subgradient refinement, and LP polishing across multiple lengths and inits."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.5, float(budget_s) - 20.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: n={p.size}, sum={p.sum():.2f}, max={p.max():.4f}")
    except Exception:
        pass
    
    # Try multiple lengths
    lengths = [256, 512, 1024]
    if parent is not None:
        # Put parent's length first
        if parent.size in lengths:
            lengths.remove(parent.size)
        lengths.insert(0, min(max(parent.size, 128), 2048))
    
    all_candidates = []
    
    for n in lengths:
        if time.monotonic() > deadline - 30:
            break
        
        print(f"\n{'='*50}")
        print(f"Optimizing n={n}")
        print(f"{'='*50}")
        
        inits = gen_inits(n, rng, parent)
        
        # Score initializations
        scored = []
        for name, a0 in inits.items():
            a0n = a0 / a0.sum()
            conv = fast_convolve(a0n)
            sc = 2.0 * n * conv.max()
            scored.append((sc, name, a0n))
        scored.sort(key=lambda x: x[0])
        
        # LSE optimize top initializations
        top_k = min(5, len(scored))
        remaining = deadline - time.monotonic() - 60
        time_per = max(5.0, remaining / (top_k + 1))
        
        for idx in range(top_k):
            if time.monotonic() > deadline - 30:
                break
            sc_init, name, a0 = scored[idx]
            t_alloc = min(time_per, deadline - time.monotonic() - 30)
            if t_alloc < 5:
                break
            
            print(f"  [{idx+1}/{top_k}] init='{name}' obj={sc_init:.6f} budget={t_alloc:.0f}s")
            
            a_opt, obj_opt = lse_optimize(a0, time_limit=t_alloc * 0.7)
            
            # Subgradient refinement
            if time.monotonic() < deadline - 20:
                t_sub = min(15.0, deadline - time.monotonic() - 20)
                a_ref, obj_ref = subgradient_refine(a_opt, time_limit=t_sub)
                if obj_ref < obj_opt:
                    a_opt = a_ref
                    obj_opt = obj_ref
            
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc):
                all_candidates.append((sc, n, name, a_opt))
                if sc < best_score:
                    best_score = sc
                    best_seq = a_opt.copy()
                    print(f"      -> score={sc:.6f} (NEW BEST)")
                else:
                    print(f"      -> score={sc:.6f}")
        
        # LP polish on best for this length
        local_best = None
        for sc, nn, nm, aa in all_candidates:
            if nn == n:
                if local_best is None or sc < local_best[0]:
                    local_best = (sc, aa)
        
        if local_best is not None and time.monotonic() < deadline - 15:
            t_lp = min(90.0, (deadline - time.monotonic() - 15) / 2.0)
            if t_lp > 10:
                print(f"  LP polish n={n} (best={local_best[0]:.6f}) budget={t_lp:.0f}s")
                a_lp, _ = lp_polish(local_best[1], time_limit=t_lp)
                sc_lp = float(evaluate_sequence(a_lp.tolist()))
                if np.isfinite(sc_lp):
                    all_candidates.append((sc_lp, n, 'lp', a_lp))
                    if sc_lp < best_score:
                        best_score = sc_lp
                        best_seq = a_lp.copy()
                        print(f"    -> LP score={sc_lp:.6f} (NEW BEST)")
    
    # Final phase: LP polish on overall best
    if best_seq is not None and time.monotonic() < deadline - 10:
        n_best = len(best_seq)
        t_lp = min(120.0, deadline - time.monotonic() - 10)
        if t_lp > 15:
            print(f"\nFinal LP polish n={n_best} budget={t_lp:.0f}s")
            a_final, _ = lp_polish(best_seq, time_limit=t_lp * 0.6)
            sc_f = float(evaluate_sequence(a_final.tolist()))
            if np.isfinite(sc_f) and sc_f < best_score:
                best_score = sc_f
                best_seq = a_final.copy()
                print(f"  -> score={sc_f:.6f} (NEW BEST)")
            
            # Subgradient polish
            if time.monotonic() < deadline - 5:
                t_sub = min(20.0, deadline - time.monotonic() - 5)
                a_sf, _ = subgradient_refine(best_seq, time_limit=t_sub)
                sc_sf = float(evaluate_sequence(a_sf.tolist()))
                if np.isfinite(sc_sf) and sc_sf < best_score:
                    best_score = sc_sf
                    best_seq = a_sf.copy()
                    print(f"  -> subgrad score={sc_sf:.6f} (NEW BEST)")
    
    if best_seq is None:
        best_seq = np.ones(512)
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Ensure valid output
    best_seq = np.maximum(best_seq, 0.0)
    final_score = float(evaluate_sequence([float(x) for x in best_seq]))
    if not np.isfinite(final_score):
        best_seq = np.ones(512)
        final_score = float(evaluate_sequence(best_seq.tolist()))
    
    print(f"\n{'='*50}")
    print(f"FINAL: n={len(best_seq)}, score={final_score:.6f}")
    print(f"{'='*50}")
    return [float(x) for x in best_seq]
