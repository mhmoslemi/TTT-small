import time

from scipy.fft import fft, ifft

def compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def normalize(a, target=None):
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-15:
        return a.copy()
    return a * (target / S)

def soft_max_gradient(a, tau):
    """Compute gradient of soft-max autocorrelation using FFT."""
    n = len(a)
    c = np.convolve(a, a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    # grad[i] = 2 * sum_j w[i+j] * a[j]
    # Use FFT: convolve(w, a_reversed)
    a_rev = a[::-1]

    # result = np.convolve(w, a_rev, mode='full')
    # grad = 2 * result[n-1 : n-1+n]
    # Use FFT for speed
    size = 2 * n
    W = fft(w, size=size)
    A_rev = fft(a_rev, size=size)
    result = ifft(W * A_rev).real
    grad = 2.0 * result[n-1 : n-1+n]
    return grad

Tertiary: coordinate descent for final polish.

import time
from scipy.optimize import linprog

    max_c = float(c.max())

    if target is None:

    b = np.full(2 * n - 1, M)

    except Exception:

def gradient_step(a, lr, momentum, decay, tau):
    """One step of gradient descent with momentum."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a, S_t)
    if tau is None:
        g = hard_grad(a)
    else:
        g = soft_max_grad(a, tau)
    g -= g.mean()  # Project onto sum constraint
    momentum[:] = decay * momentum - lr * g
    a_new = a + momentum
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize(a_new, S_t)
    return a_new

def coordinate_descent(a, deadline, max_rounds=5):
    """Coordinate descent polish."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a.copy(), S_t)
    best_score = compute_obj(a)
    patience = 0
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline - 5:
            break
        improved = False
        c = np.convolve(a, a)
        k_star = int(np.argmax(c))
        
        for i in range(n):
            if time.monotonic() > deadline - 5:
                break
            orig = a[i]
            if orig < 1e-10:
                continue
            j = k_star - i
            # Try several values
            if 0 <= j < n:
                # The gradient direction suggests: if a[j] is large, reduce a[i]
                candidates = [orig * 0.5, orig * 0.7, orig * 1.3, orig * 0.2, 0.0]
            else:
                candidates = [orig * 0.5, orig * 1.5, orig * 0.8]
            
            best_val = orig
            best_s = best_score
            for val in candidates:
                a[i] = val
                a_norm = normalize(a, S_t)
                s = compute_obj(a_norm)
                if s < best_s - 1e-10:
                    best_s = s
                    best_val = val
                a = a_norm  # keep normalized version for next trial
            a[i] = best_val
            a = normalize(a, S_t)
            if best_s < best_score:
                best_score = best_s
                improved = True
            # Update peak for next iteration
            if improved:
                c = np.convolve(a, a)
                k_star = int(np.argmax(c))
        
        if not improved:
            patience += 1
            if patience >= 2:
                break
        else:
            patience = 0
    
    return a, best_score

    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target=None):
    """Normalize sequence to given sum (default sqrt(2n))."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-15:
        return a.copy()
    return a * (target / S)


def soft_max_grad(a, tau):
    """Gradient of soft-max of autocorrelation values."""
    n = len(a)
    c = np.convolve(a, a)
    c_max = c.max()
    w = np.exp(tau * (c - c_max))
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= w_sum
    a_rev = a[::-1]
    conv = np.convolve(w, a_rev, mode='full')
    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad


def hard_grad(a):
    """Gradient of hard-max autocorrelation."""
    n = len(a)
    c = np.convolve(a, a)
    k_star = int(np.argmax(c))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad


def build_lp_matrix(f):
    """Build the (2n-1) x n constraint matrix for the LP."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_refine_standard(f, M, time_limit=30):
    """Standard LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    A = build_lp_matrix(f)
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_refine_tight(f, delta_frac=0.01, time_limit=30):
    """LP with tighter constraint at the peak lag."""
    n = len(f)
    c = np.convolve(f, f)
    k_star = int(np.argmax(c))
    A = build_lp_matrix(f)
    c_obj = -np.ones(n)
    b = c.copy()
    b[k_star] = c[k_star] * (1.0 - delta_frac)
    b = np.maximum(b, 0.0)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_refine_multi_peak(f, n_peaks=3, delta_frac=0.01, time_limit=30):
    """LP with tighter constraints at the top-n peak lags."""
    n = len(f)
    c = np.convolve(f, f)
    A = build_lp_matrix(f)
    c_obj = -np.ones(n)
    b = c.copy()
    top_indices = np.argsort(c)[-n_peaks:]
    for k in top_indices:
        b[k] = c[k] * (1.0 - delta_frac)
    b = np.maximum(b, 0.0)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def gradient_step(a, lr, momentum, decay, tau):
    """One step of gradient descent with momentum (sum-preserving)."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a, S_t)
    if tau is None:
        g = hard_grad(a)
    else:
        g = soft_max_grad(a, tau)
    g -= g.mean()
    momentum[:] = decay * momentum - lr * g
    a_new = a + momentum
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize(a_new, S_t)
    return a_new


def gradient_escape(a, n_steps=100, tau=20.0):
    """Execute several gradient steps to escape a local minimum."""
    n = len(a)
    momentum = np.zeros(n)
    for i in range(n_steps):
        lr = max(0.05 * (0.99 ** i), 0.005)
        a = gradient_step(a, lr, momentum, 0.9, tau)
    return a


def coordinate_descent(a, deadline, max_rounds=5):
    """Coordinate descent: try perturbing each coordinate."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a.copy(), S_t)
    best_score = compute_obj(a)
    patience = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline - 5:
            break
        improved = False
        c = np.convolve(a, a)
        k_star = int(np.argmax(c))

        for i in range(n):
            if time.monotonic() > deadline - 5:
                break
            orig = a[i]
            if orig < 1e-12:
                continue

            j = k_star - i
            if 0 <= j < n and a[j] > 0:
                candidates = [orig * 0.5, orig * 0.7, orig * 0.3, orig * 0.1, 0.0,
                              orig * 1.2, orig * 1.5]
            else:
                candidates = [orig * 0.5, orig * 1.5, orig * 0.8, orig * 1.2]

            best_val = orig
            best_s = best_score
            for val in candidates:
                a[i] = val
                a_norm = normalize(a, S_t)
                s = compute_obj(a_norm)
                if s < best_s - 1e-10:
                    best_s = s
                    best_val = val
                a = a_norm

            a[i] = best_val
            a = normalize(a, S_t)
            if best_s < best_score - 1e-10:
                best_score = best_s
                improved = True
                c = np.convolve(a, a)
                k_star = int(np.argmax(c))

        if not improved:
            patience += 1
            if patience >= 2:
                break
        else:
            patience = 0

    return a, best_score


def optimize_one(a0, deadline, verbose=False):
    """Full optimization pipeline from a single starting point."""
    n = len(a0)
    S_t = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n) * 0.5
    a = normalize(a, S_t)
    score = compute_obj(a)
    best = a.copy()
    best_score = score

    if verbose:
        print(f"      n={n}, initial={score:.6f}")

    # Phase 1: Quick gradient descent
    momentum = np.zeros(n)
    phase1_time = min(30.0, (deadline - time.monotonic()) * 0.08)
    t_phase1_end = time.monotonic() + phase1_time
    it = 0
    while time.monotonic() < t_phase1_end and it < 2000:
        it += 1
        tau = 3.0 + 20.0 * (it / 2000.0)
        lr = max(0.15 * (0.999 ** it), 0.01)
        a = gradient_step(a, lr, momentum, 0.9, tau)
        if it % 200 == 0:
            s = compute_obj(a)
            if s < best_score:
                best_score = s
                best = a.copy()

    a = best.copy()
    if verbose:
        print(f"      After grad: {best_score:.6f}")

    # Phase 2: LP refinement loop (main engine)
    if time.monotonic() < deadline - 20:
        M = float(np.convolve(a, a).max())
        no_improve = 0
        lp_iter = 0
        use_tight = True

        while time.monotonic() < deadline - 15 and lp_iter < 2000:
            lp_iter += 1

            # Choose LP variant
            if use_tight and lp_iter % 3 != 0:
                delta = 0.005 + 0.01 * (lp_iter % 10) / 10.0
                g = lp_refine_tight(a, delta_frac=delta, time_limit=20)
            elif lp_iter % 10 == 0:
                g = lp_refine_multi_peak(a, n_peaks=3, delta_frac=0.01, time_limit=20)
            else:
                g = lp_refine_standard(a, M, time_limit=20)

            if g is None:
                if lp_iter > 5:
                    break
                g = lp_refine_standard(a, M, time_limit=20)
                if g is None:
                    break

            g_sum = g.sum()
            if g_sum < 1e-6:
                break

            g_norm = normalize(g, S_t)
            a_norm = normalize(a, S_t)

            # Check for fixed point
            if np.linalg.norm(g_norm - a_norm) < 1e-4:
                no_improve += 1
                if no_improve > 4:
                    a = gradient_escape(a, n_steps=80, tau=15.0)
                    a = np.maximum(a, 0.0)
                    a = normalize(a, S_t)
                    s = compute_obj(a)
                    if s < best_score:
                        best = a.copy()
                        best_score = s
                    else:
                        a = best.copy()
                    M = float(np.convolve(a, a).max())
                    no_improve = 0
                continue

            # Line search
            improved = False
            for alpha in [0.5, 0.3, 0.7, 0.2, 0.1, 0.8, 0.05]:
                a_new = (1.0 - alpha) * a_norm + alpha * g_norm
                a_new = np.maximum(a_new, 0.0)
                a_new = normalize(a_new, S_t)
                s_new = compute_obj(a_new)
                if s_new < best_score - 1e-11:
                    a = a_new
                    best = a.copy()
                    best_score = s_new
                    improved = True
                    break

            if not improved:
                no_improve += 1
                if no_improve > 6:
                    a = a + 0.02 * np.random.randn(n)
                    a = np.maximum(a, 0.0)
                    a = normalize(a, S_t)
                    no_improve = 0
            else:
                no_improve = 0

            M = float(np.convolve(a, a).max())

            if verbose and lp_iter % 100 == 0:
                print(f"      LP iter {lp_iter}: {best_score:.6f}, M={M:.4f}")

        if verbose:
            print(f"      After LP: {best_score:.6f}, iters={lp_iter}")

    # Phase 3: Coordinate descent polish
    if n <= 512 and time.monotonic() < deadline - 10:
        a_polished, s_polished = coordinate_descent(best, deadline - 5, max_rounds=3)
        if s_polished < best_score:
            best = a_polished
            best_score = s_polished
        if verbose:
            print(f"      After coord: {best_score:.6f}")

    return best, best_score


def generate_starts(rng, n):
    """Generate diverse starting sequences of length n."""
    starts = []
    i = np.arange(n, dtype=np.float64)

    # Random
    starts.append(rng.random(n) * 0.5 + 0.25)
    starts.append(rng.random(n) * 0.8 + 0.1)
    starts.append(rng.random(n) * 0.3 + 0.35)

    # Cosine bases
    for freq in [1, 2, 3, 5, 7, 11, 13]:
        starts.append(0.5 + 0.5 * np.cos(2 * np.pi * i * freq / n))

    # Sine
    for freq in [1, 3, 5]:
        starts.append(np.abs(np.sin(2 * np.pi * i * freq / n)) + 0.05)

    # Tapers and bumps
    starts.append(np.linspace(1.0, 0.1, n))
    starts.append(np.linspace(0.1, 1.0, n))
    starts.append(np.ones(n) * 0.5)
    starts.append(np.exp(-0.5 * ((i - n / 2) / (n * 0.3)) ** 2))
    starts.append(np.exp(-0.5 * ((i - n * 0.25) / (n * 0.1)) ** 2) +
                  np.exp(-0.5 * ((i - n * 0.75) / (n * 0.1)) ** 2))

    # Sparse
    a = np.zeros(n)
    a[::2] = rng.random(n // 2 + 1) * 0.5 + 0.1
    starts.append(a)
    a = np.zeros(n)
    a[::3] = rng.random(n // 3 + 1) * 0.6 + 0.1
    starts.append(a[:n])

    # Block
    a = np.zeros(n)
    a[:n // 3] = 0.8
    a[2 * n // 3:] = 0.3
    starts.append(a + rng.random(n) * 0.1)

    # Triangular
    starts.append(1.0 - 2.0 * np.abs(i / n - 0.5))

    # From height_sequence_1 if available
    try:
        hs = np.asarray(height_sequence_1, dtype=np.float64)
        if hs.size >= n and np.all(np.isfinite(hs)):
            starts.append(hs[:n].copy())
            # Slight perturbation
            starts.append(np.maximum(hs[:n] + 0.02 * rng.normal(size=n), 0))
    except NameError:
        pass

    return starts


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Uses LP-based refinement (with tighter-at-peak variant)
    as primary engine, gradient descent for exploration, coordinate descent for polish.
    Multi-start with diverse initializations across multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    print(f"Budget: {budget_s}s, effective deadline: {deadline - t_start:.0f}s")
    print(f"Start time: {time.monotonic():.2f}")

    # Try lengths sorted by proximity to known-good 384
    lengths = [384, 448, 256, 512, 640]

    # Phase 1: Quick exploration across lengths
    print("\n=== Phase 1: Quick exploration ===")
    for n in lengths:
        if time.monotonic() > deadline - 150:
            break

        starts = generate_starts(rng, n)
        rng.shuffle(starts)

        # Try up to 2 quick starts per length
        n_tries = min(2, len(starts))
        time_per = min(45.0, (deadline - time.monotonic() - 100) / max(1, len(lengths) - lengths.index(n)))
        time_per = max(15.0, time_per)

        for idx in range(n_tries):
            if time.monotonic() > deadline - 100:
                break
            t_remain = deadline - time.monotonic() - 50
            if t_remain < 15:
                break
            sub_deadline = time.monotonic() + min(time_per, t_remain)

            a0 = starts[idx]
            seq, score = optimize_one(a0, sub_deadline, verbose=False)

            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  n={n}, start {idx}: *** NEW BEST {best_score:.6f} ***")
            else:
                print(f"  n={n}, start {idx}: {score:.6f}")

    # Phase 2: Focused optimization on best length
    print("\n=== Phase 2: Focused optimization ===")
    if best_seq is not None:
        best_n = len(best_seq)
        print(f"Focused on n={best_n}, current best={best_score:.6f}")

        # Generate more diverse starts at this length
        extra_starts = generate_starts(rng, best_n)
        rng.shuffle(extra_starts)

        # Add perturbed versions of current best
        for _ in range(3):
            extra_starts.append(np.maximum(best_seq + 0.05 * rng.normal(size=best_n), 0))
        for _ in range(2):
            extra_starts.append(np.maximum(best_seq * (0.8 + 0.4 * rng.random()), 0))

        n_extra = min(4, len(extra_starts))
        for idx in range(n_extra):
            if time.monotonic() > deadline - 60:
                break
            t_remain = deadline - time.monotonic() - 40
            t_alloc = min(200.0, t_remain / (n_extra - idx))
            if t_alloc < 20:
                break
            sub_deadline = time.monotonic() + t_alloc - 5

            a0 = np.maximum(np.asarray(extra_starts[idx], dtype=np.float64), 0)
            if a0.sum() < 1e-6:
                a0 = np.ones(best_n) * 0.5

            seq, score = optimize_one(a0, sub_deadline, verbose=True)

            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  [focused {idx}] *** NEW BEST: {best_score:.6f} ***")

    # Phase 3: Long intensive LP on the best
    print("\n=== Phase 3: Intensive LP refinement ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        n = len(best_seq)
        a = best_seq.copy()
        S_t = np.sqrt(2.0 * n)
        a = normalize(a, S_t)
        M = float(np.convolve(a, a).max())
        no_improve = 0
        total_iter = 0
        momentum = np.zeros(n)

        print(f"  Starting intensive LP: n={n}, M={M:.4f}, score={best_score:.6f}")

        while time.monotonic() < deadline - 10 and total_iter < 5000:
            total_iter += 1

            # Cycle through LP variants
            variant = total_iter % 4
            if variant == 0:
                g = lp_refine_tight(a, delta_frac=0.003, time_limit=15)
            elif variant == 1:
                g = lp_refine_tight(a, delta_frac=0.01, time_limit=15)
            elif variant == 2:
                g = lp_refine_multi_peak(a, n_peaks=2, delta_frac=0.005, time_limit=15)
            else:
                g = lp_refine_standard(a, M, time_limit=15)

            if g is None:
                g = lp_refine_standard(a, M, time_limit=15)
                if g is None:
                    if total_iter > 10:
                        break
                    continue

            g_sum = g.sum()
            if g_sum < 1e-6:
                break

            g_norm = normalize(g, S_t)
            a_norm = normalize(a, S_t)

            if np.linalg.norm(g_norm - a_norm) < 1e-4:
                no_improve += 1
                if no_improve > 5:
                    # Escape using gradient with random perturbation
                    a = a + 0.03 * np.random.randn(n)
                    a = np.maximum(a, 0.0)
                    a = gradient_escape(a, n_steps=50, tau=10.0)
                    a = np.maximum(a, 0.0)
                    a = normalize(a, S_t)
                    s = compute_obj(a)
                    if s < best_score:
                        best_score = s
                        best_seq = a.copy()
                    no_improve = 0
                    M = float(np.convolve(a, a).max())
                elif no_improve == 4:
                    a = gradient_escape(a, n_steps=30, tau=20.0)
                    a = np.maximum(a, 0.0)
                    a = normalize(a, S_t)
                    M = float(np.convolve(a, a).max())
                continue

            # Line search with more options
            improved = False
            alphas = [0.3, 0.5, 0.2, 0.7, 0.1, 0.4, 0.15, 0.8]
            for alpha in alphas:
                a_new = (1.0 - alpha) * a_norm + alpha * g_norm
                a_new = np.maximum(a_new, 0.0)
                a_new = normalize(a_new, S_t)
                s_new = compute_obj(a_new)
                if s_new < best_score - 1e-12:
                    a = a_new
                    best_score = s_new
                    best_seq = a.copy()
                    improved = True
                    break

            if not improved:
                no_improve += 1
                if no_improve > 10:
                    a = a + 0.01 * np.random.randn(n)
                    a = np.maximum(a, 0.0)
                    a = normalize(a, S_t)
                    no_improve = 0
            else:
                no_improve = 0

            M = float(np.convolve(a, a).max())

            if total_iter % 200 == 0:
                print(f"    Intensive iter {total_iter}: score={best_score:.6f}, M={M:.4f}")

        print(f"  Intensive LP done: {best_score:.6f}, iters={total_iter}")

    # Phase 4: Final coordinate descent polish
    if best_seq is not None and time.monotonic() < deadline - 10:
        print(f"\n=== Phase 4: Final coordinate polish ===")
        n = len(best_seq)
        if n <= 512:
            a_polished, s_polished = coordinate_descent(best_seq, deadline - 5, max_rounds=4)
            if s_polished < best_score:
                best_score = s_polished
                best_seq = a_polished
                print(f"  After polish: {best_score:.6f}")

    # Final validation
    if best_seq is not None:
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = np.clip(best_seq, 0.0, 1000.0)
        final_score = compute_obj(best_seq)
        print(f"\n=== FINAL RESULT ===")
        print(f"  n={len(best_seq)}, computed_score={final_score:.6f}")

        result = [float(x) for x in best_seq]
        try:
            eval_score = float(evaluate_sequence(result))
            print(f"  evaluate_sequence: {eval_score:.6f}")
            if np.isfinite(eval_score):
                return result
        except Exception as e:
            print(f"  evaluate_sequence error: {e}")

        if np.isfinite(final_score):
            return result

    # Fallback
    print("Fallback: using height_sequence_1 or uniform")
    try:
        hs = np.asarray(height_sequence_1, dtype=np.float64)
        if hs.size > 0 and np.all(np.isfinite(hs)):
            return [float(max(0.0, x)) for x in hs]
    except NameError:
        pass
    return [1.0] * 384
