"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based iterative refinement
combined with soft-max smoothed L-BFGS polishing and multi-restart exploration.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def soft_max_objective_and_grad(x, t):
    """
    Compute soft-max smoothed objective and gradient.
    x: parameters on the log-simplex (unconstrained)
    t: temperature parameter
    Returns (objective_value, gradient)
    """
    n = len(x)
    # Convert to simplex
    a = np.exp(x - x.max())
    a /= a.sum()
    
    # Compute convolution
    conv = fast_convolve(a)
    
    # Soft max
    scaled = t * conv
    scaled -= scaled.max()
    exp_vals = np.exp(scaled)
    sum_exp = exp_vals.sum()
    
    # Objective: 2*n*soft_max(conv) (since sum(a)=1)
    obj = 2.0 * n * (1.0 / t) * np.log(sum_exp)
    
    # Gradient w.r.t. a
    p = exp_vals / sum_exp  # normalized weights
    grad_a = np.zeros(n)
    for j in range(n):
        k_start = max(0, j)
        k_end = min(2*n-2, j+n-1)
        if k_start <= k_end:
            grad_a[j] = np.dot(p[k_start:k_end+1], a[k_start-j:k_end-j+1])
    grad_a *= 2.0 * n / t
    
    # Gradient w.r.t. x (chain rule through softmax parameterization)
    # a = softmax(x), so da_i/dx_j = a_i * (delta_ij - a_j)
    # grad_x_j = sum_i grad_a_i * da_i/dx_j = sum_i grad_a_i * a_i * (delta_ij - a_j)
    # = a_j * (grad_a_j - sum_i grad_a_i * a_i)
    ga_squared = np.dot(grad_a, a)
    grad_x = a * (grad_a - ga_squared)
    
    return obj, grad_x

def lp_refinement_step(a, M, max_iter=50):
    """
    Solve the LP: max sum(b) s.t. b >= 0, conv(a,b)[k] <= M for all k.
    Returns the optimal b.
    """
    n = len(a)
    # Build constraint matrix A_con
    # For each k in 0..2n-2: sum_j a[j]*b[k-j] <= M (where 0<=k-j<n)
    num_constraints = 2 * n - 1
    A_con = np.zeros((num_constraints, n))
    for k in range(num_constraints):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_con[k, j] = a[k - j]
    
    b_const = np.full(num_constraints, M)
    
    # Objective: maximize sum(b) = minimize -sum(b)
    c = -np.ones(n)
    
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_con, b_ub=b_const, bounds=bounds, method='highs')
    
    if res.status == 0:
        return np.maximum(res.x, 0)
    return np.ones(n) / n

def optimize_lp(a0, time_limit, tol=1e-6, max_iters=200):
    """
    LP-based iterative refinement.
    a0: initial sequence (will be normalized to sum=1)
    Returns (best_a, best_obj)
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    step = 0.3
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        M = conv.max()
        b = lp_refinement_step(a, M)
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b / b_sum  # normalize to sum=1
        
        # Check if b is different from a
        if np.linalg.norm(b - a) < 1e-10:
            break
        
        # Update
        a_new = (1.0 - step) * a + step * b
        a_new = np.maximum(a_new, 0.0)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:
            break
        a_new = a_new / a_new_sum
        
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            # Try to increase step
            step = min(0.8, step * 1.1)
        else:
            no_improve += 1
            step = max(0.05, step * 0.8)
        
        a = a_new
        conv = conv_new
        
        if no_improve > 50:
            break
    
    return best_a, best_obj

def optimize_softmax_lbfgs(a0, time_limit, t_init=50.0, t_min=500.0):
    """
    Optimize using soft-max smoothed objective with L-BFGS-B.
    Uses log-simplex parameterization.
    Returns (best_a, best_obj)
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    # Initialize in log-space
    x = np.log(np.maximum(a, 1e-15))
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    t_cur = t_init
    
    while time.monotonic() - t0 < time_limit and t_cur <= t_min:
        def obj_fun(x):
            val, _ = soft_max_objective_and_grad(x, t_cur)
            return val
        
        def grad_fun(x):
            _, g = soft_max_objective_and_grad(x, t_cur)
            return g
        
        res = minimize(obj_fun, x, jac=grad_fun, method='L-BFGS-B',
                       options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-8})
        
        # Convert back
        a_opt = np.exp(res.x - res.x.max())
        a_opt /= a_opt.sum()
        a_opt = np.maximum(a_opt, 0)
        
        conv_opt = fast_convolve(a_opt)
        obj_opt = 2.0 * n * conv_opt.max()
        
        if obj_opt < best_obj - 1e-12:
            best_obj = obj_opt
            best_a = a_opt.copy()
            x = res.x
            t_cur *= 2.0  # increase temperature (sharper approximation)
        else:
            t_cur *= 1.5  # more aggressive increase to break through
        
        if time.monotonic() - t0 > time_limit - 10:
            break
    
    # Final: use the actual max (not soft-max) for the best found
    conv_final = fast_convolve(best_a)
    best_obj = 2.0 * n * conv_final.max()

def lp_step(a):

    Solve: min M' s.t. sum(b)=1, b>=0, conv(a,b)[k] <= M' for all k.
    Returns (optimal_b, optimal_M)
    """

    # Objective: minimize M' => c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Equality constraint: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [1.0]
    
    # Inequality constraints: conv(a,b)[k] - M' <= 0 for each k
    num_cons = 2 * n - 1
    A_ub = np.zeros((num_cons, n + 1))
    for k in range(num_cons):

        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):

        A_ub[k, -1] = -1.0  # coefficient of M'

    if res.status == 0:
        return res.x[:n], res.x[-1]
    return None, None

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based iterative refinement
with soft-max L-BFGS polishing and multi-restart exploration.
Uses the identity that the problem is scale-invariant to normalize sum(a)=1,
then iteratively solves LPs to find directions that reduce the peak auto-correlation.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build the matrix A where (A @ b)[k] = conv(a, b)[k]."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def lp_maximize_sum(a, M):
    """Solve: max sum(b) s.t. b>=0, conv(a,b)[k] <= M for all k."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x, 0)
    return None


def lp_minimize_peak(a):
    """Solve: min M' s.t. sum(b)=1, b>=0, conv(a,b)[k] <= M' for all k."""
    n = len(a)
    A = build_conv_matrix(a)
    num_rows = 2 * n - 1
    
    # Variables: b_0..b_{n-1}, M'
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0  # M' doesn't appear in sum constraint
    b_eq = [1.0]
    
    A_ub = np.zeros((num_rows, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0  # -M'
    b_ub = np.zeros(num_rows)
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x[:n], 0), res.x[n]
    return None, None


def optimize_lp_iterative(a0, time_limit, method='max_sum', max_iters=500,
                          step_init=0.3, step_min=0.01):
    """
    LP-based iterative refinement.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    step = step_init
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        M = conv.max()
        
        if method == 'max_sum':
            b = lp_maximize_sum(a, M)
            if b is None:
                break
            b_sum = b.sum()
            if b_sum < 1e-15:
                break
            b = b / b_sum
        else:  # 'min_peak'
            b, M_achieved = lp_minimize_peak(a)
            if b is None:
                break
            b_sum = b.sum()
            if b_sum < 1e-15:
                break
            b = b / b_sum
        
        # Skip if b is essentially the same as a
        diff = np.linalg.norm(b - a)
        if diff < 1e-10:
            return best_a, best_obj
        
        # Update with adaptive step
        a_new = (1.0 - step) * a + step * b
        a_new = np.maximum(a_new, 0.0)
        a_sum = a_new.sum()
        if a_sum < 1e-15:
            break
        a_new = a_new / a_sum
        
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            step = min(0.7, step * 1.05)
        else:
            no_improve += 1
            step = max(step_min, step * 0.9)
        
        a = a_new
        conv = conv_new
        
        if no_improve > 100:
            # Try a larger step or different method
            step = step_init
            no_improve = 0
            if method == 'max_sum':
                method = 'min_peak'
            else:
                method = 'max_sum'
        
        if no_improve > 300:
            break
    
    return best_a, best_obj


def softmax_obj_grad(a, t):
    """Compute soft-max smoothed 2n*max(conv(a,a)) and gradient w.r.t. a."""
    n = len(a)
    conv = fast_convolve(a)
    
    scaled = t * conv
    scaled -= scaled.max()
    exp_vals = np.exp(scaled)
    sum_exp = exp_vals.sum()
    
    obj = 2.0 * n * (1.0 / t) * np.log(sum_exp)
    
    p = exp_vals / sum_exp
    
    # Gradient: grad_a[j] = (2n/t) * sum_k p[k] * a[k-j] (valid k range)
    grad_a = np.zeros(n)
    for j in range(n):
        k_lo = j
        k_hi = min(2 * n - 2, j + n - 1)
        if k_lo <= k_hi:
            grad_a[j] = 2.0 * n / t * np.dot(p[k_lo:k_hi + 1], a[k_lo - j:k_hi - j + 1])
    
    return obj, grad_a


def optimize_projected_grad(a0, time_limit, t_init=20.0, lr_init=0.01,
                            momentum=0.9, max_iters=2000):
    """
    Projected gradient (on simplex) with momentum for the soft-max smoothed objective.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    v = np.zeros(n)
    t0 = time.monotonic()
    lr = lr_init
    t_cur = t_init
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        _, grad = softmax_obj_grad(a, t_cur)
        
        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)
        a = a_new
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive learning rate
        if no_improve > 200:
            lr *= 0.5
            no_improve = 0
        if no_improve > 1000:
            break
        
        # Slowly increase temperature (sharper approximation)
        if it % 500 == 0 and t_cur < 500:
            t_cur *= 2.0
    
    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    
    inits['uniform'] = np.ones(n) / n
    
    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps
    
    # Cosine-modulated
    x = np.arange(n, dtype=np.float64)
    for freq in [1, 2, 3]:
        a = 1.0 + 0.9 * np.cos(2 * np.pi * freq * x / n)
        a = np.maximum(a, 0)
        a /= a.sum()
        inits[f'cos_{freq}'] = a
    
    # Gaussian centered
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.25 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Sparse peaks
    for num_peaks in [2, 4, 8]:
        peaks = rng.choice(n, size=num_peaks, replace=False)
        a = np.zeros(n)
        for p in peaks:
            a[p] = 1.0
        # Smooth slightly
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
        a /= a.sum()
        inits[f'peaks_{num_peaks}'] = a
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based iterative refinement
combined with projected subgradient polishing and multi-restart exploration.
The LP approach solves: max sum(b) s.t. conv(a,b) <= M, b >= 0, providing
principled search directions that reduce peak auto-correlation iteratively.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = conv(a, b)[k] for k=0..2n-2."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def lp_maximize_sum(a, M):
    """Solve: max sum(b) s.t. b >= 0, conv(a,b)[k] <= M for all k."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x, 0)
    return None


def lp_minimize_peak(a):
    """Solve: min M' s.t. sum(b)=1, b>=0, conv(a,b)[k] <= M' for all k."""
    n = len(a)
    A = build_conv_matrix(a)
    num_rows = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    A_ub = np.zeros((num_rows, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(num_rows)
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x[:n], 0), res.x[n]
    return None, None

    else:

def softmax_obj_grad(a, t):
    """Compute soft-max smoothed objective and gradient."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = t * conv
    scaled -= scaled.max()
    exp_vals = np.exp(scaled)
    sum_exp = exp_vals.sum()
    obj = 2.0 * n * (1.0 / t) * np.log(sum_exp)
    p = exp_vals / sum_exp
    grad_a = np.zeros(n)
    for j in range(n):
        k_lo = j
        k_hi = min(2 * n - 2, j + n - 1)
        if k_lo <= k_hi:
            grad_a[j] = (2.0 * n / t) * np.dot(p[k_lo:k_hi + 1], a[k_lo - j:k_hi - j + 1])
    return obj, grad_a

def subgradient_multi_peak(a, conv, top_k, temperature):
    """Weighted subgradient over top-k convolution peaks."""
    n = len(a)
    if top_k >= len(conv):

        w_sum = w.sum()

            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum
    else:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0

        k = int(peaks[i])

        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)

    return sg
