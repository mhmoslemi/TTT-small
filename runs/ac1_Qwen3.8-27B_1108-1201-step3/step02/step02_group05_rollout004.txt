Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid of subgradient projection,
LP-based direction finding, and aggressive multi-type local search.
Explores sequence lengths up to 2048 for better "attack surface."
"""

import numpy as np
import scipy.optimize
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_norm(a):
    """Objective assuming sum(a)=1: 2n*max(conv(a,a))."""
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    return 2.0 * n * fast_convolve(a_norm).max()


def extend_sequence(a, new_len):
    """Extend or compress sequence to new_len via linear interpolation."""
    n = len(a)
    if new_len == n:
        return a.copy()
    if n == 1:
        return np.full(new_len, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_len)
    a_ext = np.interp(x_new, x_old, a)
    return np.maximum(a_ext, 0.0)


def subgrad_step(a, lr):
    """One subgradient step on max(conv(a,a)) with projection onto non-negative simplex."""
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    tol = 1e-10 * max(1.0, M)
    peak_mask = conv >= (M - tol)
    num_peaks = peak_mask.sum()
    if num_peaks == 0:
        return a.copy()
    
    # Compute subgradient: grad[j] = (2/|S|) * sum_{k in S} a[k-j] (valid k-j)
    grad = np.zeros(n)
    peak_indices = np.where(peak_mask)[0]
    for k in peak_indices:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        for j in range(lo, hi + 1):
            grad[j] += 2.0 * a[k - j]
    grad /= num_peaks
    
    # Gradient descent step
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-12:
        return a.copy()
    a_new /= s
    return a_new


def subgrad_optimize(a0, time_limit, lr_init=0.05, lr_decay=0.995, min_lr=1e-6):
    """Iterative subgradient optimization with adaptive learning rate."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones_like(a) / len(a)
    else:
        a /= s
    
    best_a = a.copy()
    best_obj = obj_norm(a)
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    max_no_improve = 200
    
    while time.monotonic() - t0 < time_limit:
        a_new = subgrad_step(a, lr)
        obj_new = obj_norm(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        elif obj_new < obj_norm(a) + 1e-10:
            # Accept if not much worse (exploration)
            a = a_new
            no_improve += 1
        else:
            no_improve += 1
            # Reduce learning rate
            lr = max(min_lr, lr * lr_decay)
        
        if no_improve >= max_no_improve:
            break
    
    return best_a, best_obj

def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_norm(a):
    """Objective assuming sum(a)=1: 2n*max(conv(a,a))."""
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    return 2.0 * n * fast_convolve(a_norm).max()


def extend_sequence(a, new_len):
    """Extend or compress sequence to new_len via linear interpolation."""
    n = len(a)
    if new_len == n:
        return a.copy()
    if n == 1:
        return np.full(new_len, a[0])
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_len)
    a_ext = np.interp(x_new, x_old, a)
    return np.maximum(a_ext, 0.0)

def make_inits(n, rng, parent):
    """Generate diverse initializations for length n."""
    x = np.arange(n, dtype=np.float64)
    inits = {}

    inits['uniform'] = np.ones(n) / n

    # Gaussian with different widths
    for sf in [0.15, 0.25, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - 0.4 * n) / (sf * n)) ** 2)
        g = np.maximum(g, 0)
        gs = g.sum()
        if gs > 1e-12:
            inits['gauss_%s' % str(sf)] = g / gs

    # Asymmetric ramp
    ramp = x + 1.0
    inits['ramp'] = ramp / ramp.sum()

    # Inverse ramp
    inv_ramp = (n - x) + 1.0
    inits['inv_ramp'] = inv_ramp / inv_ramp.sum()

    # Two-bump asymmetric
    bump1 = np.exp(-0.5 * ((x - 0.2 * n) / (0.1 * n)) ** 2)
    bump2 = np.exp(-0.5 * ((x - 0.7 * n) / (0.15 * n)) ** 2)

    es = edges.sum()
    if es > 1e-10:
