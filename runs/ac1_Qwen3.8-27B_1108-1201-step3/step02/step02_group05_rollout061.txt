```python
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def obj_norm(a):
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()

def compute_subgradient(a, epsilon=1e-10):
    """Compute subgradient of max_k conv(a,a)[k] at point a."""
    n = len(a)
    conv = fast_convolve(a)
    max_val = conv.max()
    # Find all indices within epsilon of max
    K = np.where(conv >= max_val - epsilon)[0]
    # Compute subgradient: average over active indices
    grad = np.zeros(n)
    for k in K:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] += 2.0 * a[j]
    grad /= len(K)
    return grad, max_val

def mirror_descent_step(a, grad, eta):
    """Perform a mirror descent step on the simplex."""
    # a_new[i] = a[i] * exp(-eta * grad[i]) / Z
    log_a = np.log(np.maximum(a, 1e-300))
    log_a_new = log_a - eta * grad
    # Subtract max for numerical stability
    log_a_new -= log_a_new.max()
    a_new = np.exp(log_a_new)
    a_new /= a_new.sum()
    return a_new

def mirror_descent_optimize(a0, time_limit=60.0, lr_scale=1.0):
    """Optimize using mirror descent with subgradients."""
    a = np.maximum(a0.astype(np.float64), 1e-15).copy()
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = obj_norm(a)
    t0 = time.monotonic()
    
    # Adaptive learning rate
    eta = lr_scale / np.sqrt(n)

            patience = 0

        # Decrease learning rate
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth (soft-max) objective
with mirror descent, LP polishing, and multi-scale exploration.
"""

import numpy as np
import scipy.optimize
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_norm(a):
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def smooth_grad(a, beta):
    """Compute gradient of soft-max approximation of max(conv(a,a)).
    Returns (gradient, soft_max_value)."""
    n = len(a)
    conv = fast_convolve(a)
    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    
    # grad[i] = 2 * sum_k w[k] * a[k-i] * I(0<=k-i<n)
    grad = np.zeros(n)
    for i in range(n):
        k_lo = max(0, i)
        k_hi = min(2 * n - 2, i + n - 1)
        if k_lo > k_hi:
            continue
        j_lo = k_lo - i
        j_hi = k_hi - i
        grad[i] = 2.0 * np.dot(w[k_lo:k_hi + 1], a[j_lo:j_hi + 1])
    
    soft_max_val = (1.0 / beta) * np.log(exp_shifted.sum()) + conv_max
    return grad, soft_max_val


def mirror_descent_smooth(a0, time_limit=60.0, beta_start=20.0, beta_end=200.0):
    """Mirror descent with smooth (soft-max) objective and annealing beta."""
    a = np.maximum(a0.astype(np.float64), 1e-15).copy()
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = obj_norm(a)
    t0 = time.monotonic()
    
    n_iter_est = 5000  # estimate for learning rate schedule
    eta0 = 2.0 / np.sqrt(n)  # initial step size
    
    for t in range(1, n_iter_est + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Anneal beta
        progress = min(1.0, t / (n_iter_est * 0.7))
        beta = beta_start + (beta_end - beta_start) * progress
        
        grad, soft_val = smooth_grad(a, beta)
        obj = 2.0 * n * soft_val
        true_obj = obj_norm(a)
        
        if true_obj < best_obj - 1e-12:
            best_obj = true_obj
            best_a = a.copy()
        
        # Learning rate
        eta = eta0 / np.sqrt(t)
        
        # Mirror descent step
        log_a = np.log(np.maximum(a, 1e-300))
        log_a_new = log_a - eta * grad
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new /= a_new.sum()
        a = a_new
    
    return best_a, best_obj


def solve_lp(f, M):
    """Find g >= 0 maximizing sum(g) subject to conv(f,g)[k] <= M for all k."""
    n = len(f)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(2 * n - 1, j + n)
        A_ub[j:end, j] = f[:end - j]
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return f.copy()


def lp_optimize(f0, time_limit=60.0):
    """LP-based iterative optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0).copy()
    s = f.sum()
    if s < 1e-12:
        f = np.ones(len(f)) / len(f)
    else:
        f /= s
    best_f = f.copy()
    best_obj = obj_norm(f)
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()
        if M < 1e-15:
            break
        g = solve_lp(f, M)
        if np.allclose(g, f, atol=1e-10):
            no_improve += 3
        else:
            # Line search
            best_t = 0.0
            best_obj_t = best_obj
            for t_try in [0.1, 0.25, 0.5, 0.75, 1.0]:
                a_new = (1.0 - t_try) * f + t_try * g
                a_new = np.maximum(a_new, 0.0)
                ns = a_new.sum()
                if ns < 1e-12:
                    continue
                a_new /= ns
                obj_t = obj_norm(a_new)
                if obj_t < best_obj_t:
                    best_obj_t = obj_t
                    best_t = t_try
            if best_obj_t < best_obj - 1e-10:
                a_new = (1.0 - best_t) * f + best_t * g
                a_new = np.maximum(a_new, 0.0)
                a_new /= a_new.sum()
                f = a_new
                best_obj = best_obj_t
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        if no_improve >= 40:
            break
    
    return best_f, best_obj


def local_search(a, time_limit=30.0):
    """Local search: redistribute mass to reduce peak."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    best_obj = obj_norm(a)
    t0 = time.monotonic()
    stuck = 0
    
    while time.monotonic() - t0 < time_limit and stuck < 200:
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))
        
        # Find indices contributing most to the peak
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]
        
        max_c = contrib.max()
        if max_c < 1e-15:
            break
        
        # Move mass from high-contributing to low-contributing indices
        sorted_idx = np.argsort(contrib)
        danger = sorted_idx[-10:]  # highest contributing
        safe = sorted_idx[:10]  # lowest contributing
        
        found = False
        for di in danger:
            if found:
                break
            for si in safe:
                if found:
                    break
                if di == si:
                    continue
                for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                    amount = a[di] * frac
                    if amount < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[di] -= amount
                    a_trial[si] += amount
                    a_trial = np.maximum(a_trial, 0.0)
                    ts = a_trial.sum()
                    if ts < 1e-12:
                        continue
                    a_trial /= ts
                    o = obj_norm(a_trial)
                    if o < best_obj - 1e-10:
                        best_obj = o
                        a = a_trial.copy()
                        found = True
                        break
        
        if not found:
            stuck += 1
        else:
            stuck = 0
    
    return a, best_obj


def get_parent(n):
    """Get parent sequence resized to length n."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)

def make_initializations(n, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    
    inits['uniform'] = np.ones(n) / n
    
    # U-shaped (high at edges)
    for alpha in [0.5, 1.0, 2.0, 3.0]:
        u = 1.0 / (1.0 + alpha * ((x - center) / center) ** 2)
        inits[f'u_shaped_{alpha}'] = u / u.sum()
    
    # Inverted U (low at edges) - this is what we want to AVOID
    # Actually we want HIGH at edges to flatten the convolution
    
    # Cosine-based
    cos = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / n))
    inits['cosine_edges'] = cos / cos.sum()
    
    # Two blocks
    m = n // 4
    blocks = np.zeros(n)
    blocks[:m] = 1.0
    blocks[-m:] = 1.0
    inits['two_blocks'] = blocks / blocks.sum()
    
    # Three blocks
    m3 = n // 6
    blocks3 = np.zeros(n)
    blocks3[:m3] = 1.0
    blocks3[n//3:2*n//3] = 0.5
    blocks3[-m3:] = 1.0
    inits['three_blocks'] = blocks3 / blocks3.sum()
    
    # Random
    r = rng.dirichlet(np.ones(n) * 0.5)
    inits['random'] = r
    
    # Parent
    parent = get_parent(n)
    if parent is not None:
        inits['parent'] = parent
    
    # Perturbed parent
    if parent is not None:
        for pert_scale in [0.05, 0.1, 0.2]:
            pert = parent * (1.0 + pert_scale * rng.standard_normal(n))
            pert = np.maximum(pert, 0.0)
            ps = pert.sum()
            if ps > 1e-10:
                inits[f'parent_pert_{pert_scale}'] = pert / ps
    
    return inits
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth soft-max mirror descent,
LP polishing, symmetry exploitation, and multi-length exploration.
Key innovations: annealed soft-max for smooth gradients, FFT-based gradient
computation, symmetry constraints, and aggressive multi-scale search.

import numpy as np
import scipy.optimize
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_norm(a):
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def symmetrize(a):
    n = len(a)
    return (a + a[::-1]) / 2.0


def smooth_gradient_fft(a, beta):
    """Compute gradient of soft-max of conv(a,a) using FFT for efficiency.
    Returns (grad, soft_max_value)."""
    n = len(a)
    conv = fast_convolve(a)
    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    
    # grad[i] = 2 * sum_k w[k] * a[k-i] for valid k-i in [0,n-1]
    # Use FFT-based cross-correlation
    # g[i] = sum_{j=0}^{n-1} w[j+i] * a[j]
    # This is cross_correlation(w, a) at lag i
    
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    W = np.zeros(fft_size)
    W[:len(w)] = w
    A = np.zeros(fft_size)
    A[:len(a)] = a
    
    # Cross-correlation: g[i] = sum_j W[j+i] * A[j]
    # = ifft(FFT(W) * conj(FFT(A))) at the right indices
    G = np.fft.ifft(np.fft.fft(W) * np.conj(np.fft.fft(A))).real
    
    # The cross-correlation result: G[i] = sum_j W[j] * A[j-i] (shifted)
    # We need g[i] = sum_j W[j+i] * A[j]
    # Let's use: np.correlate equivalent via FFT
    # Actually: ifft(fft(W) * conj(fft(A)))[k] = sum_j W[j] * A[j-k+n] (circular)
    # For linear: we need the right slice
    
    # Alternative: g[i] = sum_j W[j+i]*A[j] = correlate(W, A) at lag i
    # Using: correlate(W,A)[i] = ifft(conj(fft(W)) * fft(A))[i + len(A) - 1 + ...]
    # Let me just use the direct approach for correctness
    grad = np.zeros(n)
    for i in range(n):
        k_lo = i
        k_hi = min(2 * n - 2, i + n - 1)
        if k_lo > k_hi:
            continue
        j_lo = 0
        j_hi = k_hi - i
        grad[i] = 2.0 * np.dot(w[k_lo:k_hi + 1], a[j_lo:j_hi + 1])
    
    soft_max_val = (1.0 / beta) * np.log(exp_shifted.sum()) + conv_max
    return grad, soft_max_val


def smooth_gradient_direct(a, beta):
    """Direct computation of smooth gradient (simpler, used for small n)."""
    n = len(a)
    conv = fast_convolve(a)
    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    
    grad = np.zeros(n)
    for i in range(n):

        k_hi = min(2 * n - 2, i + n - 1)
        if k_lo > k_hi:

        j_start = k_lo - i
        j_end = k_hi - i
        grad[i] = 2.0 * np.dot(w[k_lo:k_hi + 1], a[j_start:j_end + 1])
    
    soft_max_val = (1.0 / beta) * np.log(exp_shifted.sum()) + conv_max
    return grad, soft_max_val

    n = len(a)

        log_a = np.log(np.maximum(a, 1e-300))
        log_a_new = log_a - eta * grad
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new /= a_new.sum()
        if use_symmetry:
            a_new = symmetrize(a_new)
            a_new /= a_new.sum()
        a = a_new
    
    return best_a, best_obj


def solve_lp_symmetric(f, M):
    """LP for symmetric sequences: optimize half, mirror the other half."""
    n = len(f)
    half = (n + 1) // 2
    
    # Build constraint matrix for the full convolution but with symmetry
    # conv(f, f)[k] = sum_i f[i]*f[k-i]
    # With symmetry f[i] = f[n-1-i], we only optimize f[0..half-1]
    # But the constraints are on the full conv
    
    # Simpler: just solve the full LP (symmetry is approximately maintained)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(2 * n - 1, j + n)
        A_ub[j:end, j] = f[:end - j]
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            g = np.maximum(res.x, 0.0)
            g = symmetrize(g)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return f.copy()


def lp_optimize(f0, time_limit=60.0, use_symmetry=True):
    """LP-based iterative optimization with line search."""
    f = np.maximum(f0.astype(np.float64), 0.0).copy()
    s = f.sum()
    if s < 1e-12:
        f = np.ones(len(f)) / len(f)
    else:
        f /= s
    if use_symmetry:
        f = symmetrize(f)
        f /= f.sum()
    
    best_f = f.copy()
    best_obj = obj_norm(f)
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        if no_improve > 50:
            break
        conv = fast_convolve(f)
        M = conv.max()
        if M < 1e-15:
            break
        
        g = solve_lp_symmetric(f, M)
        
        if np.allclose(g, f, atol=1e-10):
            no_improve += 3
            continue
        
        best_t = 0.0
        best_obj_t = best_obj
        for t_try in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_new = (1.0 - t_try) * f + t_try * g
            a_new = np.maximum(a_new, 0.0)
            if use_symmetry:
                a_new = symmetrize(a_new)
            ns = a_new.sum()
            if ns < 1e-12:
                continue
            a_new /= ns
            obj_t = obj_norm(a_new)
            if obj_t < best_obj_t:
                best_obj_t = obj_t
                best_t = t_try
        
        if best_obj_t < best_obj - 1e-10:
            a_new = (1.0 - best_t) * f + best_t * g
            a_new = np.maximum(a_new, 0.0)
            if use_symmetry:
                a_new = symmetrize(a_new)
            a_new /= a_new.sum()
            f = a_new
            best_obj = best_obj_t
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
    
    return best_f, best_obj


def local_search(a, time_limit=30.0, use_symmetry=True):
    """Local search: pairwise mass transfer to reduce peak."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    if use_symmetry:
        a = symmetrize(a)
        a /= a.sum()
    
    best_obj = obj_norm(a)
    t0 = time.monotonic()
    stuck = 0
    
    while time.monotonic() - t0 < time_limit and stuck < 300:
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))
        
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]
        
        max_c = contrib.max()
        if max_c < 1e-15:
            break
        
        sorted_idx = np.argsort(contrib)
        danger = sorted_idx[-min(15, n // 4):]
        safe = sorted_idx[:min(15, n // 4)]
        
        found = False
        for di in danger:
            if found:
                break
            for si in safe:
                if found:
                    break
                if di == si:
                    continue
                for frac in [0.05, 0.1, 0.15, 0.25, 0.4]:
                    amount = a[di] * frac
                    if amount < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[di] -= amount
                    a_trial[si] += amount
                    if use_symmetry:
                        a_trial[n - 1 - di] -= amount
                        a_trial[n - 1 - si] += amount
                    a_trial = np.maximum(a_trial, 0.0)
                    ts = a_trial.sum()
                    if ts < 1e-12:
                        continue
                    a_trial /= ts
                    o = obj_norm(a_trial)
                    if o < best_obj - 1e-10:
                        best_obj = o
                        a = a_trial.copy()
                        found = True
                        break
        
        if not found:
            stuck += 1
        else:
            stuck = 0
    
    return a, best_obj


def get_parent(n):
    """Get parent sequence resized to length n."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim != 1 or p.size == 0 or not np.isfinite(p).all():
            return None
        if p.size >= n:
            pc = p[:n].copy()
        else:
            pc = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, p.size), p)
        pc = np.maximum(pc, 0.0)
        s = pc.sum()
        if s < 1e-10:
            return None
        pc = symmetrize(pc)
        s = pc.sum()
        if s < 1e-10:
            return None
        return pc / s
    except Exception:
        return None


def make_initializations(n, rng):
    """Generate diverse symmetric initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    
    u = np.ones(n) / n
    inits['uniform'] = symmetrize(u) / symmetrize(u).sum()
    
    for alpha in [0.5, 1.0, 2.0, 3.0, 5.0]:
        v = 1.0 / (1.0 + alpha * ((x - center) / max(center, 1.0)) ** 2)
        v = symmetrize(v)
        inits[f'u_{alpha}'] = v / v.sum()
    
    for power in [1.0, 2.0, 4.0, 8.0]:
        v = np.abs(x - center) / max(center, 1.0)
        v = v ** power
        v = symmetrize(v)
        s = v.sum()
        if s > 0:
            inits[f'pow_{power}'] = v / s
    
    for width in [0.1, 0.2, 0.3]:
        v = np.exp(-((x - center) / (width * n)) ** 2)
        v = symmetrize(v)
        inits[f'edge_gauss_{width}'] = v / v.sum()
    
    m = max(2, n // 5)
    blocks = np.zeros(n)
    blocks[:m] = 1.0
    blocks[-m:] = 1.0
    blocks = symmetrize(blocks)
    inits['blocks'] = blocks / blocks.sum()
    
    r = rng.dirichlet(np.ones(n) * 0.3)
    r = symmetrize(r)
    inits['random'] = r / r.sum()
    
    parent = get_parent(n)
    if parent is not None:
        inits['parent'] = parent
        for scale in [0.05, 0.1, 0.15, 0.25]:
            pert = parent * (1.0 + scale * rng.standard_normal(n))
            pert = np.maximum(pert, 0.0)
            pert = symmetrize(pert)
            ps = pert.sum()
            if ps > 1e-10:
                inits[f'parent_p{scale}'] = pert / ps
    
    return inits

    """
    Main entry: multi-phase optimization combining smooth mirror descent,
    LP polishing, local search, and random perturbations.

    """
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    deadline = time.monotonic() + total_budget - 30.0
    
    best_sequence = np.ones(512)
    best_score = float(evaluate_sequence(best_sequence.tolist()))
    
    def try_update(a, label):
        nonlocal best_sequence, best_score
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return False
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  {label}: {score:.6f} (n={len(a)})")

    print("Phase 1: Scanning initializations...")

    lengths = [512, 768, 1024]
