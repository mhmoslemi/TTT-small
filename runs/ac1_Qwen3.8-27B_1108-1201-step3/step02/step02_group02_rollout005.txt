    # Equivalently: A[k, i] = f[k-i] if max(0, k-n+1) <= i <= min(k, n-1)
    # 
    # Actually, for the convolution: b[k] = sum_i f[i] * g[k-i]
    # The constraint is b[k] <= M for all k
    # So A[k, i] = f[k-i] if 0 <= k-i < n, else 0
    
    # Efficient construction using the fact that each row is a shifted version of f
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        # A[k, i] = f[k-i] where 0 <= k-i < n => max(0, k-n+1) <= i <= min(k, n-1)
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        if i_start <= i_end:
            f_idx = k - np.arange(i_start, i_end + 1)
            A[k, i_start:i_end + 1] = f[f_idx]
    return A


def solve_lp(f, M, previous_solution=None):
    """Solve LP: max sum(b) s.t. b>=0, A@b <= M."""
    n = len(f)
    A = build_A(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    if previous_solution is not None:
        return previous_solution.copy()
    return np.ones(n) / n


def subgradient_estimate(f):
    """Compute subgradient of max(conv(f,f)) with respect to f."""
    n = len(f)
    c = fft_convolve(f)
    max_c = np.max(c)
    # Find all positions within tolerance of max
    tol = max_c * 1e-8
    peak_mask = c >= max_c - tol
    peak_indices = np.where(peak_mask)[0]
    
    # For each peak position p, the subgradient at index i is 2*f[p-i] if 0<=p-i<n
    # Average over all peak positions
    sg = np.zeros(n)
    for p in peak_indices:
        # sg[i] += 2*f[p-i] if 0<=p-i<n
        i_min = max(0, p - n + 1)
        i_max = min(p, n - 1)
        if i_min <= i_max:
            f_idx = p - np.arange(i_min, i_max + 1)
            sg[i_min:i_max + 1] += 2.0 * f[f_idx]
    sg /= len(peak_indices)
    return sg, max_c


def coordinate_descent(f, max_rounds=100, time_limit=60.0, rng=None):
    """
    Coordinate descent: for each element, try to find the best value.
    We use the fact that the subgradient tells us the direction.
    """
    n = len(f)
    if rng is None:
        rng = np.random.default_rng(42)
    
    f = f.astype(np.float64).copy()
    S = np.sum(f)
    current_score = fast_score(f)
    best_f = f.copy()
    best_score = current_score
    t0 = time.monotonic()
    
    for round_i in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Get subgradient
        sg, max_c = subgradient_estimate(f)
        
        # For each coordinate, try to move in the direction that reduces the peak
        # The idea: if sg[i] is small, increasing a[i] won't increase the peak much
        # If sg[i] is large, decreasing a[i] will reduce the peak
        
        # Process coordinates in order of increasing sg (most "safe" to increase first)
        order = np.argsort(sg)
        improved = False
        
        for i in order:
            # Try increasing a[i]
            for delta_frac in [0.05, 0.02, 0.01]:
                f_new = f.copy()
                f_new[i] *= (1.0 + delta_frac)
                # Renormalize (except we might want to fix the sum)
                f_new = normalize_to_target(f_new, S)
                new_score = fast_score(f_new)
                if new_score < current_score:
                    f = f_new
                    current_score = new_score
                    improved = True
                    if current_score < best_score:
                        best_score = current_score
                        best_f = f.copy()
                    break
            else:
                # Try decreasing
                for delta_frac in [0.05, 0.02, 0.01]:
                    f_new = f.copy()
                    f_new[i] *= (1.0 - delta_frac)
                    if f_new[i] < 0:
                        continue
                    f_new = normalize_to_target(f_new, S)
                    new_score = fast_score(f_new)
                    if new_score < current_score:
                        f = f_new
                        current_score = new_score
                        improved = True
                        if current_score < best_score:
                            best_score = current_score
                            best_f = f.copy()
                        break
        
        if not improved:
            break
    
    return best_f, best_score


def simulated_annealing(f, time_limit=120.0, rng=None):
    """Simulated annealing with diverse moves."""
    if rng is None:
        rng = np.random.default_rng(123)
    
    n = len(f)
    f = f.astype(np.float64).copy()
    S = np.sum(f)
    current_score = fast_score(f)
    best_f = f.copy()
    best_score = current_score
    t0 = time.monotonic()
    
    temperature = 0.002
    cooling = 0.9998
    min_temp = 1e-7
    
    while time.monotonic() - t0 < time_limit and temperature > min_temp:
        f_new = f.copy()
        r = rng.random()
        
        if r < 0.25:
            # Peak-targeted: reduce elements that contribute to the peak
            c = fft_convolve(f)
            peak_idx = int(np.argmax(c))
            # Find top contributors to the peak
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = f[i] * f[j]
            top_k = max(3, n // 15)
            top_idx = np.argsort(contribs)[-top_k:]
            # Reduce top contributors
            f_new[top_idx] *= np.exp(rng.normal(-0.03, 0.015, size=len(top_idx)))
            # Increase some others
            others = np.setdiff1d(np.arange(n), top_idx)
            if len(others) > 5:
                n_inc = min(8, len(others) // 3)
                inc_idx = rng.choice(others, size=n_inc, replace=False)
                f_new[inc_idx] *= np.exp(rng.normal(0.03, 0.015, size=n_inc))
        elif r < 0.45:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            f_new[start:start + width] *= np.exp(rng.normal(0, 0.025, size=width))
        elif r < 0.60:
            # Multi-block
            for _ in range(int(rng.integers(3, 8))):
                width = max(1, int(rng.integers(1, max(2, n // 12))))
                start = int(rng.integers(0, n - width))
                f_new[start:start + width] *= np.exp(rng.normal(0, 0.02, size=width))
        elif r < 0.75:
            # Pair swap (transfer mass)
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                delta = rng.uniform(0.02, 0.1) * min(f[i], f[j])
                f_new[i] = max(0, f[i] + delta)
                f_new[j] = max(0, f[j] - delta)
        elif r < 0.85:
            # Gaussian perturbation (small)
            mask = rng.random(n) < 0.3
            f_new[mask] *= np.exp(rng.normal(0, 0.02, size=np.sum(mask)))
        elif r < 0.95:
            # Mirror perturbation (perturb both i and n-1-i)
            i = int(rng.integers(0, n // 2))
            j = n - 1 - i
            f_new[i] *= np.exp(rng.normal(0, 0.03))
            f_new[j] *= np.exp(rng.normal(0, 0.03))
        else:
            # Large random restart of a portion
            frac = rng.uniform(0.1, 0.3)
            n_mod = max(1, int(n * frac))
            start = int(rng.integers(0, n - n_mod))
            f_new[start:start + n_mod] *= np.exp(rng.normal(0, 0.05, size=n_mod))
        
        f_new = np.clip(f_new, 0.0, 1000.0)
        # Renormalize to maintain sum
        S_new = np.sum(f_new)
        if S_new < 0.01:
            continue
        f_new = f_new * (S / S_new)
        
        new_score = fast_score(f_new)
        
        if new_score < current_score:
            f = f_new
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best_f = f_new.copy()
        else:
            delta_e = new_score - current_score
            if rng.random() < np.exp(-delta_e / max(temperature, 1e-15)):
                f = f_new
                current_score = new_score
        
        temperature *= cooling
    
    return best_f, best_score


def lp_iteration(f0, time_limit=300.0, rng=None, verbose=False):
    """LP-based iterative optimization."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    n = len(f0)
    f = np.clip(f0.astype(np.float64), 0, None)
    S = np.sum(f)
    if S < 0.01:
        f = np.ones(n)
    else:
        f = f * (np.sqrt(2.0 * n) / S)
    
    S_target = np.sqrt(2.0 * n)
    current_score = fast_score(f)
    best_f = f.copy()
    best_score = current_score
    t0 = time.monotonic()
    t_mix = 0.1
    stall = 0
    prev_b = None
    
    while time.monotonic() - t0 < time_limit:
        M = np.max(fft_convolve(f))
        
        # Solve LP with warm start
        if prev_b is None:
            prev_b = f.copy()
        b_sol = solve_lp(f, M, prev_b)
        prev_b = b_sol.copy()
        
        sum_b = np.sum(b_sol)
        if sum_b < 1e-10:
            stall += 1
            if stall > 50:
                break
            continue
        
        g = (S_target / sum_b) * b_sol
        
        # Line search
        improved = False
        for trial_t in [t_mix, t_mix * 0.7, t_mix * 0.5, t_mix * 0.3, t_mix * 0.1]:
            f_new = (1.0 - trial_t) * f + trial_t * g
            f_new = np.clip(f_new, 0, None)
            f_new = normalize_to_target(f_new, S_target)
            new_score = fast_score(f_new)
            if new_score < current_score:
                f = f_new
                current_score = new_score
                t_mix = min(0.2, trial_t * 1.1)
                improved = True
                stall = 0
                if current_score < best_score:
                    best_score = current_score
                    best_f = f.copy()
                break
        
        if not improved:
            stall += 1
            t_mix *= 0.8
            if stall > 40:
                # Perturb
                f = f * np.exp(rng.normal(0, 0.02, n))
                f = np.clip(f, 0, None)
                f = normalize_to_target(f, S_target)
                current_score = fast_score(f)
                stall = 0
                t_mix = 0.08
        
        if verbose and (int((time.monotonic() - t0) * 2) % 100 == 0):
            print(f"    LP: score={current_score:.6f}, best={best_score:.6f}, t={t_mix:.4f}, elapsed={time.monotonic()-t0:.1f}s")
    
    # Final coordinate descent
    cd_f, cd_score = coordinate_descent(best_f, max_rounds=50, time_limit=min(30.0, max(5.0, (time_limit - (time.monotonic() - t0)))), rng=rng)
    if cd_score < best_score:
        best_score = cd_score
        best_f = cd_f
    
    return best_f, best_score


def make_chirp(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * (t + 1) / n)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multi_tone(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in range(1, 5):
        amp = 0.3 / k
        freq = rng.uniform(0.5, 3.0) * k
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * freq * t / n + phase)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def make_random(n, rng, spread=0.3):
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def make_tapered(n, rng, power=1.5):
    """Tapered sequence (higher in the middle)."""
    t = np.linspace(-1, 1, n)
    a = 1.0 - power * t**2
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def resample_to_length(seq, new_n):
    """Resample sequence to new length."""
    x_old = np.linspace(0, 1, len(seq), endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, seq)

"""
Hybrid optimization for minimizing peak autocorrelation of non-negative step functions.
Combines LP-based direction finding, subgradient descent, coordinate descent,
and simulated annealing with multi-length exploration.
Key insight: the peak of conv(a,a) for non-negative a is typically near the center,
and the subgradient at position i is proportional to a[p-i] where p is the peak position.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fft_convolve(a, b=None):
    """Compute linear convolution of two 1-D arrays using FFT."""
    if b is None:
        b = a
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def fast_score(a):
    """Compute the evaluation score (fast, no security checks)."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a)
    return float(2.0 * n * np.max(c) / (S * S))


def normalize_to_target(a, target_sum=None):
    """Normalize sequence to target sum."""
    if target_sum is None:
        target_sum = np.sqrt(2.0 * len(a))
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(len(a)) * target_sum / len(a)
    return a * (target_sum / S)


def build_constraint_matrix(f):
    """Build the (2n-1) x n constraint matrix A where A[k,i] = f[k-i] if valid."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        if i_start <= i_end:
            f_idx = k - np.arange(i_start, i_end + 1)
            A[k, i_start:i_end + 1] = f[f_idx]
    return A


def solve_lp_direction(f, M, prev_sol=None):
    """Solve LP: max sum(b) s.t. b>=0, A@b <= M. Returns b."""
    n = len(f)
    A = build_constraint_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    if prev_sol is not None:
        return prev_sol.copy()
    return np.ones(n) / n


def compute_subgradient(f):
    """Compute subgradient of max(conv(f,f)) with respect to f.
    Returns (subgradient_vector, max_value, peak_position)."""
    n = len(f)
    c = fft_convolve(f)
    max_c = np.max(c)
    peak_pos = int(np.argmax(c))
    # Find all near-peak positions
    tol = max_c * 1e-6
    peak_mask = c >= max_c - tol
    peak_indices = np.where(peak_mask)[0]
    
    # sg[i] = sum over peaks p of 2*f[p-i] (if 0<=p-i<n), averaged
    sg = np.zeros(n)
    for p in peak_indices:
        i_min = max(0, p - n + 1)
        i_max = min(p, n - 1)
        if i_min <= i_max:
            f_idx = p - np.arange(i_min, i_max + 1)
            sg[i_min:i_max + 1] += 2.0 * f[f_idx]
    sg /= len(peak_indices)
    return sg, max_c, peak_pos


def subgradient_descent_step(f, lr, peak_positions=None):
    """One subgradient descent step, followed by projection and normalization."""
    n = len(f)
    S_target = np.sum(f)
    sg, max_c, peak_pos = compute_subgradient(f)
    
    f_new = f - lr * sg
    f_new = np.clip(f_new, 0.0, None)
    s = np.sum(f_new)
    if s < 0.01:
        return f.copy()
    f_new = f_new * (S_target / s)
    return f_new

    if rng is None:

    best_score = current_score
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:

                break
            continue

def coordinate_descent(f, time_limit=60.0, rng=None):
    """
    Coordinate descent: cycle through elements, adjust each to improve score.
    Uses subgradient to prioritize which elements to adjust.
    """
    n = len(f)
    if rng is None:
        rng = np.random.default_rng(42)
    
    f = f.astype(np.float64).copy()
    S_target = np.sum(f)
    current_score = fast_score(f)
    best_f = f.copy()
    best_score = current_score
    t0 = time.monotonic()
    round_num = 0
    
    while time.monotonic() - t0 < time_limit:
        round_num += 1
        # Get subgradient to prioritize elements
        sg, max_c, peak_pos = compute_subgradient(f)
        
        # Process elements in order of increasing subgradient (safest to increase)
        order = np.argsort(sg)
        improved_this_round = False
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Try a few adjustments
            best_delta = 0.0
            best_local_score = current_score
            
            for scale in [0.08, 0.04, 0.02, 0.01]:
                # Try increase
                f_test = f.copy()
                f_test[i] *= (1.0 + scale)
                f_test = normalize_to_target(f_test, S_target)
                s_test = fast_score(f_test)
                if s_test < best_local_score:
                    best_local_score = s_test
                    best_delta = scale
                
                # Try decrease
                f_test = f.copy()
                f_test[i] *= max(0.0, 1.0 - scale)
                f_test = normalize_to_target(f_test, S_target)
                s_test = fast_score(f_test)
                if s_test < best_local_score:
                    best_local_score = s_test
                    best_delta = -scale
            
            if best_delta != 0.0:
                f[i] *= (1.0 + best_delta)
                f = np.clip(f, 0, None)
                f = normalize_to_target(f, S_target)
                current_score = fast_score(f)
                improved_this_round = True
                if current_score < best_score:
                    best_score = current_score
                    best_f = f.copy()
        
        if not improved_this_round:
            # Try random restart of a small portion
            n_mod = max(3, n // 20)
            start = int(rng.integers(0, n - n_mod))
            f_test = f.copy()
            f_test[start:start + n_mod] *= np.exp(rng.normal(0, 0.05, n_mod))
            f_test = np.clip(f_test, 0, None)
            f_test = normalize_to_target(f_test, S_target)
            s_test = fast_score(f_test)
            if s_test < current_score:
                f = f_test
                current_score = s_test
                if current_score < best_score:
                    best_score = current_score
                    best_f = f.copy()
            else:
                break
    
    return best_f, best_score


def simulated_annealing(f, time_limit=120.0, rng=None):
    """Simulated annealing with diverse moves for fine optimization."""
    if rng is None:
        rng = np.random.default_rng(123)
    
    n = len(f)
    f = f.astype(np.float64).copy()
    S_target = np.sum(f)
    current_score = fast_score(f)
    best_f = f.copy()
    best_score = current_score
    t0 = time.monotonic()
    
    temperature = 0.0015
    cooling = 0.99985
    min_temp = 5e-7
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit and temperature > min_temp:
        f_new = f.copy()
        r = rng.random()
        iter_count += 1
        
        if r < 0.20:
            # Peak-targeted move: reduce top contributors to the peak
            c = fft_convolve(f)
            peak_idx = int(np.argmax(c))
            contribs = np.zeros(n)
            i_min = max(0, peak_idx - n + 1)
            i_max = min(peak_idx, n - 1)
            for i in range(i_min, i_max + 1):
                contribs[i] = f[i] * f[peak_idx - i]
            top_k = max(4, n // 12)
            top_idx = np.argsort(contribs)[-top_k:]
            f_new[top_idx] *= np.exp(rng.normal(-0.035, 0.015, size=len(top_idx)))
            others = np.setdiff1d(np.arange(n), top_idx)
            if len(others) > 5:
                n_inc = min(10, len(others) // 3)
                inc_idx = rng.choice(others, size=n_inc, replace=False)
                f_new[inc_idx] *= np.exp(rng.normal(0.035, 0.015, size=n_inc))
        elif r < 0.35:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 6))))
            start = int(rng.integers(0, n - width))
            f_new[start:start + width] *= np.exp(rng.normal(0, 0.025, size=width))
        elif r < 0.50:
            # Multi-block small
            for _ in range(int(rng.integers(3, 8))):
                width = max(1, int(rng.integers(1, max(2, n // 12))))
                start = int(rng.integers(0, n - width))
                f_new[start:start + width] *= np.exp(rng.normal(0, 0.018, size=width))
        elif r < 0.65:
            # Pair mass transfer
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                delta = rng.uniform(0.01, 0.08) * min(f[i], f[j])
                direction = rng.choice([1, -1])
                f_new[i] = max(0, f[i] + direction * delta)
                f_new[j] = max(0, f[j] - direction * delta)
        elif r < 0.75:
            # Subgradient-guided move
            sg, _, _ = compute_subgradient(f)
            step = rng.uniform(0.01, 0.04)
            f_new = f - step * sg * rng.normal(1, 0.3, n)
            f_new = np.clip(f_new, 0, None)
        elif r < 0.85:
            # Symmetric perturbation (affect i and n-1-i together)
            n_pairs = int(rng.integers(2, max(3, n // 10)))
            for _ in range(n_pairs):
                i = int(rng.integers(0, n // 2))
                j = n - 1 - i
                f_new[i] *= np.exp(rng.normal(0, 0.025))
                f_new[j] *= np.exp(rng.normal(0, 0.025))
        elif r < 0.95:
            # Sparse Gaussian
            mask = rng.random(n) < 0.2
            n_mask = np.sum(mask)
            if n_mask > 0:
                f_new[mask] *= np.exp(rng.normal(0, 0.03, size=n_mask))
        else:
            # Larger perturbation (escape)
            frac = rng.uniform(0.05, 0.15)
            n_mod = max(3, int(n * frac))
            start = int(rng.integers(0, n - n_mod))
            f_new[start:start + n_mod] *= np.exp(rng.normal(0, 0.04, size=n_mod))
        
        f_new = np.clip(f_new, 0.0, 1000.0)
        S_new = np.sum(f_new)
        if S_new < 0.01:
            temperature *= cooling
            continue
        f_new = f_new * (S_target / S_new)
        
        new_score = fast_score(f_new)
        
        if new_score < current_score:
            f = f_new
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best_f = f_new.copy()
                temperature = max(0.0015, temperature * 1.02)  # warm up slightly
        else:
            delta_e = new_score - current_score
            if delta_e < 0.05:  # Only accept if not too bad
                if rng.random() < np.exp(-delta_e / max(temperature, 1e-15)):
                    f = f_new
                    current_score = new_score
        
        temperature *= cooling
    
    return best_f, best_score


def make_chirp(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * (t + 1) / n)
    a += 0.04 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multi_tone(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in range(1, 6):
        amp = 0.25 / k
        freq = rng.uniform(0.4, 2.5) * k
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    a += 0.04 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def make_random(n, rng, spread=0.3):
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def make_tapered(n, rng, power=1.0):
    t = np.linspace(-1, 1, n)
    a = 1.0 - power * t**2
    a += 0.04 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def make_log_taper(n, rng):
    """Logarithmic taper."""
    t = np.linspace(0.1, 1.0, n)
    a = np.log(t + 0.1) + 0.5
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def resample_to_length(seq, new_n):
    """Resample sequence to new length via linear interpolation."""
    x_old = np.linspace(0, 1, len(seq), endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, seq)

        print(f"\n=== Phase 3: Coordinate descent ===")
        cd_time = min(150.0, max(30.0, deadline - time.monotonic() - 250))
        cd_f, cd_score = coordinate_descent(best_seq, cd_time, rng)
        if cd_score < best_score:
            best_score = cd_score
            best_seq = cd_f.copy()
            print(f"  After CD: {best_score:.6f}")
        else:
            print(f"  CD no improvement (best={best_score:.6f})")
    
    # Phase 4: Simulated annealing polish
    if best_seq is not None and time.monotonic() < deadline - 150:
        print(f"\n=== Phase 4: Simulated annealing polish ===")
        sa_time = min(200.0, max(30.0, deadline - time.monotonic() - 120))
        print(f"  Starting SA, score={best_score:.6f}, time={sa_time:.0f}s")
        sa_f, sa_score = simulated_annealing(best_seq, sa_time, rng)
        if sa_score < best_score:
            best_score = sa_score
            best_seq = sa_f.copy()
            print(f"  After SA: {best_score:.6f}")
        else:
            print(f"  SA no improvement (best={best_score:.6f})")
    
    # Phase 5: Final LP refinement
    if best_seq is not None and time.monotonic() < deadline - 40:
        print(f"\n=== Phase 5: Final LP refinement ===")
        final_time = max(10.0, deadline - time.monotonic() - 30)
        f_best, s_best = lp_iteration(best_seq, final_time, rng, verbose=False)
        if s_best < best_score:
            best_score = s_best
            best_seq = f_best.copy()
            print(f"  After final LP: {best_score:.6f}")
    
    # Phase 6: Final SA
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic() - 5
        if remaining > 15:
            print(f"\n=== Phase 6: Final SA ({remaining:.0f}s) ===")
            sa_f, sa_score = simulated_annealing(best_seq, remaining, rng)
            if sa_score < best_score:
                best_score = sa_score
                best_seq = sa_f.copy()
                print(f"  Final SA: {best_score:.6f}")
    
    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list
    
    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
