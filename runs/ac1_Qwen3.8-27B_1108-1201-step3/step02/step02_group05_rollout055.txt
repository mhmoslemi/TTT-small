3. Multiple starting points and sequence lengths

import numpy as np
import scipy.optimize
import time

def fast_autoconvolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_obj_grad(a):
    """Compute objective and (sub)gradient of 2n*max(conv)/(sum)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf, np.zeros(n)
    conv = fast_autoconvolve(a)
    max_val = conv.max()
    obj = 2.0 * n * max_val / (S * S)
    
    # Find peak lags (within tolerance)
    tol = max(1e-12, 1e-9 * max_val)
    peak_lags = np.where(conv >= max_val - tol)[0]
    
    # Compute subgradient
    # df/da_p = (4n/(m*S^2)) * sum_l a_{k_l - p} - 4nM/S^3
    grad = np.zeros(n)
    for k in peak_lags:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] += a[j]
    grad = grad / len(peak_lags)
    grad = (4.0 * n / (S * S)) * grad - (4.0 * n * max_val) / (S * S * S)
    
    return obj, grad

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS-B with analytic
subgradients, coordinate descent polishing, and diverse initializations.
"""

import numpy as np
import scipy.optimize
import time


def fast_autoconvolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_obj_grad(a):
    """
    Compute objective value and subgradient of f(a) = 2n*max_k(a*a)_k / (sum a)^2.
    Returns (objective, gradient).
    """
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf, np.zeros(n)
    
    conv = fast_autoconvolve(a)
    max_val = conv.max()
    obj = 2.0 * n * max_val / (S * S)
    
    # Find all near-peak lags
    tol = max(1e-10, 1e-8 * max_val)
    peak_mask = conv >= max_val - tol
    peak_lags = np.where(peak_mask)[0]
    m = len(peak_lags)
    
    # Subgradient: (4n/(m*S^2)) * sum_l (shifted a at lag k_l) - 4nM/S^3
    grad = np.zeros(n)
    for k in peak_lags:
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            grad[i] += a[k - i]
    grad /= m
    grad = (4.0 * n / (S * S)) * grad - (4.0 * n * max_val) / (S * S * S)
    
    return obj, grad


def lbfgs_optimize(a0, time_limit, ftol=1e-12, gtol=1e-10):
    """Optimize using L-BFGS-B."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        a = np.ones(len(a))
        S = len(a)
    a /= S  # Normalize to sum=1 (objective is scale-invariant)
    
    t0 = time.monotonic()
    
    def func_and_grad(x):
        x = np.maximum(x, 0.0)  # Ensure non-negativity (L-BFGS-B should handle bounds, but just in case)
        obj, grad = compute_obj_grad(x)
        if np.isinf(obj):
            return 1e10, np.zeros(len(x))
        return obj, grad
    
    bounds = [(0.0, None)] * len(a)
    
    # Run L-BFGS-B with time-based stopping
    n_iter = 0
    result = None
    
    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        max_iter = max(50, int(remaining * 50))  # Estimate iterations
        try:
            result = scipy.optimize.minimize(
                func_and_grad, a, method='L-BFGS-B',
                jac=True, bounds=bounds,
                options={'maxiter': max_iter, 'ftol': ftol, 'gtol': gtol, 'maxls': 50}
            )
            a = np.maximum(result.x, 0.0)
        except Exception as e:
            print(f"  L-BFGS error: {e}")
            break
        
        if result.success and result.status in [1, 2, 4]:  # Converged or max iter
            n_iter += 1
            if result.success and result.status != 4:
                break
            # If max iterations, restart from current point with fresh BFGS memory
            if time.monotonic() - t0 >= time_limit:
                break
        else:
            break
    
    S = a.sum()
    if S > 1e-12:
        a /= S
    else:
        a = a0.copy()
    
    obj = compute_obj_grad(a)[0]
    return a, obj


def coordinate_polish(a, time_limit):
    """
    Coordinate descent polishing: try to reduce the peak by moving mass
    from high-contributing positions to low-contributing ones.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a, np.inf
    a /= S
    
    best_obj = compute_obj_grad(a)[0]
    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 200
    
    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        conv = fast_autoconvolve(a)
        k_star = int(np.argmax(conv))
        
        # Compute contribution of each position to the peak
        contrib = np.zeros(n)
        i_min = max(0, k_star - n + 1)
        i_max = min(n - 1, k_star)
        for i in range(i_min, i_max + 1):
            contrib[i] = a[i] * a[k_star - i]
        
        # Find positions with high and low contribution
        max_c = contrib.max()
        if max_c < 1e-15:
            break
        
        # Sort by contribution
        sorted_idx = np.argsort(contrib)
        
        found = False
        # Try moving mass from high-contrib to low-contrib positions
        for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
            if found:
                break
            for hi_pos in sorted_idx[-20:][::-1]:  # High contribution positions
                if found:
                    break
                for lo_pos in sorted_idx[:20]:  # Low contribution positions
                    if hi_pos == lo_pos:
                        continue
                    amount = a[hi_pos] * frac
                    if amount < 1e-12:
                        continue
                    a_trial = a.copy()
                    a_trial[hi_pos] -= amount
                    a_trial[lo_pos] += amount
                    a_trial = np.maximum(a_trial, 0.0)
                    ts = a_trial.sum()
                    if ts < 1e-12:
                        continue
                    a_trial /= ts
                    o = compute_obj_grad(a_trial)[0]
                    if o < best_obj - 1e-12:
                        best_obj = o
                        a = a_trial.copy()
                        found = True
                        break
        
        if not found:
            # Try small random perturbations
            improved = False
            for _ in range(50):
                mask = np.random.random(n) < 0.05
                perturbation = np.random.normal(0, 0.01, n) * mask
                a_trial = a + perturbation
                a_trial = np.maximum(a_trial, 0.0)
                ts = a_trial.sum()
                if ts < 1e-12:
                    continue
                a_trial /= ts
                o = compute_obj_grad(a_trial)[0]
                if o < best_obj - 1e-12:
                    best_obj = o
                    a = a_trial.copy()
                    improved = True
                    break
            if not improved:
                no_improve += 1
        else:
            no_improve = 0
    
    return a, best_obj


def make_start(n, type_name, rng, parent=None):
    """Create a starting sequence of length n."""
    if type_name == 'uniform':
        a = np.ones(n)
    elif type_name == 'gauss':
        x = np.arange(n)
        a = np.exp(-0.5 * ((x - n/2) / (n * 0.3)) ** 2)
    elif type_name == 'edges':
        a = np.zeros(n)
        m = max(1, n // 3)
        a[:m] = 1.0
        a[-m:] = 1.0
    elif type_name == 'random':
        a = rng.random(n) + 0.1
    elif type_name == 'hann':
        x = np.arange(n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1))) if n > 1 else np.ones(n)
    elif type_name == 'triangle':
        x = np.arange(n)
        a = np.minimum(x, n - 1 - x).astype(float) + 1.0
    elif type_name == 'parent':
        if parent is not None and len(parent) > 0:
            p = np.clip(np.array(parent, dtype=np.float64), 0, 1000)
            if len(p) >= n:
                a = p[:n].copy()
            else:
                a = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
        else:
            a = np.ones(n)
    elif type_name == 'sparse':
        a = np.zeros(n)
        k = max(2, n // 10)
        idx = rng.choice(n, size=k, replace=False)
        a[idx] = 1.0
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n)
        S = n
    return a / S

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS-B with analytic
subgradients, coordinate descent polishing with peak-targeting heuristics,
and diverse initializations across multiple sequence lengths.
"""

import numpy as np
import scipy.optimize
import time


def fast_autoconvolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_obj_grad(a):
    """
    Compute objective value and subgradient of f(a) = 2n*max_k(a*a)_k / (sum a)^2.
    
    The subgradient at a point where the max is achieved at lags {k_1,...,k_m} is:
    grad = (4n/(m*S^2)) * sum_l shifted_a(k_l) - 4n*M/S^3
    """
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf, np.zeros(n)
    
    conv = fast_autoconvolve(a)
    M = conv.max()
    obj = 2.0 * n * M / (S * S)
    
    # Find near-peak lags
    tol = max(1e-12, 1e-8 * M)
    peak_lags = np.where(conv >= M - tol)[0]
    m = len(peak_lags)
    
    # Compute sum of shifted vectors
    shifted_sum = np.zeros(n)
    for k in peak_lags:
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min <= i_max:
            src_end = k - i_min + 1
            src_start = k - i_max
            shifted_sum[i_min:i_max+1] += a[src_start:src_end][::-1]
    shifted_sum /= m
    
    grad = (4.0 * n / (S * S)) * shifted_sum - (4.0 * n * M) / (S * S * S)
    
    return obj, grad


def lbfgs_run(a0, time_limit):
    """Run L-BFGS-B optimization with time-based stopping."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        a = np.ones(len(a)) / len(a)
    else:
        a /= S
    
    n = len(a)
    bounds = [(0.0, None)] * n
    t0 = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj_grad(a)[0]
    
    def fun(x):
        x = np.maximum(x, 0.0)
        if x.sum() < 1e-12:
            return 1e15
        obj, grad = compute_obj_grad(x)
        if np.isinf(obj) or not np.isfinite(obj):
            return 1e15
        if not np.all(np.isfinite(grad)):
            grad = np.zeros(n)
        return obj, grad
    
    # Run multiple L-BFGS calls (restarts) until time runs out
    stall_count = 0
    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        max_iter = max(100, int(remaining * 100))
        try:
            res = scipy.optimize.minimize(
                fun, a, method='L-BFGS-B', jac=True,
                bounds=bounds,
                options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-12, 'maxls': 100}
            )
        except Exception:
            break
        
        new_a = np.maximum(res.x, 0.0)
        new_S = new_a.sum()
        if new_S < 1e-12:
            break
        new_a /= new_S
        
        new_obj = compute_obj_grad(new_a)[0]
        
        if new_obj < best_obj - 1e-14:
            best_obj = new_obj
            best_a = new_a.copy()
            a = new_a.copy()
            stall_count = 0
        else:
            stall_count += 1
            # Try a small perturbation to escape
            if stall_count >= 3:
                break
        
        if res.success and res.status in [1, 2]:
            # Converged, try perturbation
            pert = np.random.default_rng().normal(0, 0.01, n)
            a_pert = a + pert
            a_pert = np.maximum(a_pert, 0.0)
            pS = a_pert.sum()
            if pS > 1e-12:
                a = a_pert / pS
            stall_count = 0
    
    return best_a, best_obj


def peak_reduction_step(a, n_trials=200):
    """
    Single peak reduction step: find the peak lag, identify contributing
    positions, and try to move mass from high-contrib to low-contrib positions.
    Returns (new_a, improved).
    """
    n = len(a)
    conv = fast_autoconvolve(a)
    k_star = int(np.argmax(conv))
    
    # Contribution of each position to the peak
    contrib = np.zeros(n)
    i_min = max(0, k_star - n + 1)
    i_max = min(n - 1, k_star)
    for i in range(i_min, i_max + 1):
        contrib[i] = a[i] * a[k_star - i]
    
    # Sort positions by contribution
    order = np.argsort(contrib)
    
    best_obj = compute_obj_grad(a)[0]
    best_a = a.copy()
    
    rng = np.random.default_rng()
    
    for _ in range(n_trials):
        # Pick a high-contrib position (from top 30%)
        hi_frac = rng.random()
        hi_idx_pos = int(hi_frac * len(order[order > np.percentile(contrib, 50)]))
        if hi_idx_pos >= len(order[order > np.percentile(contrib, 50)]):
            hi_pos = order[-1]
        else:
            hi_pos = order[order > np.percentile(contrib, 50)][hi_idx_pos]
        
        # Pick a low-contrib position (from bottom 30%)
        low_mask = order < len(order) * 0.3
        if not np.any(low_mask):
            continue
        lo_pos = order[rng.choice(np.where(low_mask)[0])]
        
        if hi_pos == lo_pos:
            continue
        
        # Try moving a fraction of mass
        frac = rng.choice([0.05, 0.1, 0.15, 0.2, 0.3, 0.5])
        amount = a[hi_pos] * frac
        if amount < 1e-12:
            continue
        
        a_trial = a.copy()
        a_trial[hi_pos] -= amount
        a_trial[lo_pos] += amount
        a_trial = np.maximum(a_trial, 0.0)
        ts = a_trial.sum()
        if ts < 1e-12:
            continue
        a_trial /= ts
        
        o = compute_obj_grad(a_trial)[0]
        if o < best_obj - 1e-12:
            best_obj = o
            best_a = a_trial.copy()
    
    improved = best_obj < compute_obj_grad(a)[0] - 1e-12
    return best_a, improved


def coordinate_polish(a, time_limit):
    """Iterative coordinate descent polish using peak reduction."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return a, np.inf
    a /= S
    
    best_obj = compute_obj_grad(a)[0]
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and no_improve < 300:
        a_new, improved = peak_reduction_step(a, n_trials=100)
        if improved:
            a = a_new
            obj = compute_obj_grad(a)[0]
            if obj < best_obj:
                best_obj = obj
            no_improve = 0
        else:
            no_improve += 1
    
    return a, best_obj
