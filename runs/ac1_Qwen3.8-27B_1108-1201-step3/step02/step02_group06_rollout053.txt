            break
            
        sum_g = np.sum(g)
        if sum_g < 1e-10:
            break
        
        # Rescale g to have same sum as f
        g_rescaled = g * (sum_f / sum_g)
        
        # Backtracking line search
        best_t = 0.0
        best_local_score = compute_score(f)
        for t in [0.3, 0.5, 0.7, 0.2, 0.4, 0.6, 0.8, 0.1]:
            f_trial = (1.0 - t) * f + t * g_rescaled
            f_trial = np.clip(f_trial, 0.0, None)
            s_trial = np.sum(f_trial)
            if s_trial < 1e-10:
                continue
            f_trial = f_trial * (sum_f / s_trial)
            score_trial = compute_score(f_trial)
            if score_trial < best_local_score:
                best_local_score = score_trial
                best_t = t
        
        if best_t == 0.0:
            # Try very small steps
            for t in [0.05, 0.1, 0.15]:
                f_trial = (1.0 - t) * f + t * g_rescaled
                f_trial = np.clip(f_trial, 0.0, None)
                s_trial = np.sum(f_trial)
                if s_trial < 1e-10:
                    continue
                f_trial = f_trial * (sum_f / s_trial)
                score_trial = compute_score(f_trial)
                if score_trial < best_local_score:
                    best_local_score = score_trial
                    best_t = t
        
        if best_t == 0.0:
            stagnant += 1
            if stagnant > 30:
                # Perturb to escape
                f = f + 0.03 * rng.standard_normal(n)
                f = np.clip(f, 0.01, None)
                f = f * (sum_f / np.sum(f))
                stagnant = 0
            continue
        
        f = (1.0 - best_t) * f + best_t * g_rescaled
        f = np.clip(f, 0.0, None)
        s = np.sum(f)
        if s < 1e-10:
            break
        f = f * (sum_f / s)
        
        iter_count += 1
        current_score = compute_score(f)
        
        if current_score < best_score:
            best_score = current_score
            best_f = f.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        if verbose and iter_count % 20 == 0:
            print(f"    LP-active iter={iter_count}, score={current_score:.6f}, "
                  f"best={best_score:.6f}, t={best_t:.2f}, elapsed={time.monotonic()-t0:.1f}s")
    
    return best_f, best_score

def coordinate_descent(f_init, time_limit, verbose=False):
    """Coordinate descent: optimize one coordinate at a time."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()
    sweep_count = 0
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            a_i = f[i]
            S_prime = np.sum(f) - a_i  # sum without element i
            
            # Compute c_k for this coordinate
            # For lag k: c_k = a[i+k] (if valid) + a[i-k] (if valid)
            # For k=0: special case (quadratic)
            
            # We'll do a 1D search for the optimal value of f[i]
            # Score(t) = 2n * max_k b_k(t) / (S' + t)^2
            
            # Precompute current conv
            # Actually, let's just evaluate at a few candidate t values
            
            # Candidate t values: current, 0, a_i * [0.5, 0.7, 1.3, 1.5, 2.0]
            candidates = [0.0, a_i * 0.3, a_i * 0.7, a_i, a_i * 1.3, a_i * 1.7, a_i * 2.5]
            # Add some around the current value
            delta = a_i * 0.1
            candidates.extend([a_i + delta, a_i - delta, a_i + 2*delta, max(0, a_i - 2*delta)])
            
            best_t = a_i
            best_sc = compute_score(f)
            
            for t in candidates:
                if t < 0:
                    continue
                f[i] = t
                sc = compute_score(f)
                f[i] = a_i
                if sc < best_sc:
                    best_sc = sc
                    best_t = t
            
            if best_t != a_i:
                f[i] = best_t
                improved = True
                if best_sc < best_score:
                    best_score = best_sc
                    best_f = f.copy()
        
        sweep_count += 1
        if not improved and sweep_count > 3:
            break
        if verbose and sweep_count % 5 == 0:
            print(f"    CD sweep={sweep_count}, score={compute_score(f):.6f}, "
                  f"best={best_score:.6f}, elapsed={time.monotonic()-t0:.1f}s")
    
    return best_f, best_score

def sa_optimize(f_init, time_limit, rng, verbose=False):
    """Simulated annealing with diverse moves."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    sum_f = np.sum(f)
    best_f = f.copy()
    best_score = compute_score(f)
    current_score = best_score
    t0 = time.monotonic()
    temperature = 0.002
    cooling = 0.9999
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit and temperature > 1e-6:
        candidate = f.copy()
        r = rng.random()
        
        if r < 0.3:
            # Peak-targeted move
            conv = fft_convolve(f, f)
            peak_idx = int(np.argmax(conv))
            # Reduce contributors to the peak
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = f[i] * f[j]
            # Find top contributors
            top_k = max(5, n // 15)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.03, 0.01, size=top_k))
            # Boost some others
            other_idx = np.setdiff1d(np.arange(n), top_idx)
            if len(other_idx) > 0:
                n_inc = min(8, len(other_idx))
                inc_idx = rng.choice(other_idx, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.03, 0.01, size=n_inc))
        elif r < 0.5:
            # Segment perturbation
            width = max(1, int(rng.integers(2, max(3, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.03, size=width))
        elif r < 0.7:
            # Pair swap
            i, j = rng.choice(n, size=2, replace=False)
            scale = np.exp(rng.normal(0, 0.04))
            candidate[i] *= scale
            candidate[j] /= scale
        elif r < 0.85:
            # Global scale perturbation
            candidate *= np.exp(rng.normal(0.0, 0.005, size=n))
        else:
            # Random two-point change
            i, j = rng.choice(n, size=2, replace=False)
            di = rng.normal(0, 0.05) * f[i]
            dj = rng.normal(0, 0.05) * f[j]
            candidate[i] = max(0, f[i] + di)
            candidate[j] = max(0, f[j] + dj)
        
        candidate = np.clip(candidate, 0.0, None)
        cand_sum = np.sum(candidate)
        if cand_sum < 1e-10:
            continue
        candidate = candidate * (sum_f / cand_sum)
        
        new_score = compute_score(candidate)
        iter_count += 1
        
        if new_score < current_score:
            f = candidate
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = candidate.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            f = candidate
            current_score = new_score
        
        temperature *= cooling
    
    return best_f, best_score

def make_chirp_init(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

def make_random_init(n, rng, spread=0.3):
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

def make_multitone_init(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for _ in range(8):
        k = int(rng.integers(1, max(2, n // 4)))
        amp = rng.uniform(0.04, 0.15)
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

```python
"""
Fast active-set LP + coordinate descent + SA for minimizing
normalized peak autocorrelation of non-negative step functions.
"""

import numpy as np
import time
from scipy.optimize import linprog

# Helper functions (all top-level, no closures, no lambdas)

def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def build_active_matrix(f, top_k):
    """Build constraint matrix A for active set LP."""
    n = len(f)
    K = len(top_k)
    A = np.zeros((K, n))
    for idx, k in enumerate(top_k):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            A[idx, js:je+1] = f[k - je : k - js + 1][::-1]
    return A

def solve_active_lp(f, M, K=40):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for top K lags, g >= 0."""
    n = len(f)
    conv = fft_convolve(f, f)
    top_k = np.argsort(conv)[-K:]
    A = build_active_matrix(f, top_k)
    c_obj = -np.ones(n)
    b_ub = np.full(K, M)
    bounds = [(0.0, None)] * n
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x
    return None

def lp_optimize_fast(f_init, time_limit, K=40, verbose=False):
    """Fast LP optimization with analytic line search."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    S = np.sum(f)
    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()
    iter_count = 0
    stagnant = 0
    rng = np.random.default_rng(98765)
    t_grid = np.linspace(0.01, 0.95, 100)
    
    while time.monotonic() - t0 < time_limit:
        conv_f = fft_convolve(f, f)
        M = np.max(conv_f)
        
        g = solve_active_lp(f, M, K)
        if g is None:
            if stagnant > 100:
                break
            stagnant += 1
            continue
        
        sum_g = np.sum(g)
        if sum_g < 1e-10:
            stagnant += 1
            if stagnant > 100:
                break
            continue
        
        g_rescaled = g * (S / sum_g)
        
        # Analytic line search
        conv_fg = fft_convolve(f, g_rescaled)
        conv_g = fft_convolve(g_rescaled, g_rescaled)
        
        # Precompute coefficients for each lag
        # conv_new[k] = (1-t)^2 * conv_f[k] + 2t(1-t) * conv_fg[k] + t^2 * conv_g[k]
        # = conv_f[k] + t*(2*conv_fg[k] - 2*conv_f[k]) + t^2*(conv_f[k] - 2*conv_fg[k] + conv_g[k])
        A_k = conv_f  # constant term
        B_k = 2.0 * (conv_fg - conv_f)  # linear coefficient
        C_k = conv_f - 2.0 * conv_fg + conv_g  # quadratic coefficient
        
        # For each t, find max over k of A_k + B_k*t + C_k*t^2

    best_f = f.copy()

    t0 = time.monotonic()

        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:

            a_i = f[i]
            candidates = [0.0, a_i * 0.5, a_i * 0.7, a_i * 1.3, a_i * 1.5, a_i * 2.0,
                          max(0.0, a_i * 0.8), max(0.0, a_i * 1.1)]

                if best_sc < best_score:

            conv = fft_convolve(f, f)

                j = peak_idx - i
                if 0 <= j < n:

        elif r < 0.7:

            candidate[i] *= s

        elif r < 0.85:

            candidate *= np.exp(rng.normal(0.0, 0
