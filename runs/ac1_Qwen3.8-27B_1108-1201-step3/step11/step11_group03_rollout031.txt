            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                ss = b_val.sum()
                if ss > 1e-15:
                    b_val = b_val / ss
                    return b_val, float(max(0, M_var.value))
        except Exception:
            continue
    return None, None


def nesterov_lp_refine(a0, time_limit, solver_name, n_max_iters=500000):
    """Nesterov-accelerated LP alternating with adaptive restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(123)
    use_highs = True

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        if use_highs:
            b_new, M_b = lp_best_response_highs(y)
        else:
            b_new, M_b = lp_best_response_cvxpy(y, solver_name)

        if b_new is None:
            use_highs = False
            b_new, M_b = lp_best_response_cvxpy(y, solver_name)
            if b_new is None:
                continue

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 15:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 8)
                noise = 0.02 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                sy2 = y.sum()
                if sy2 > 1e-15:
                    y = y / sy2
                a = y.copy()
                stall_count = 0

            if (it + 1) % 2000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def build_q_sparse(k, n):
    """Build sparse Q matrix for conv(a,a)[k] = a^T Q a."""
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)


def solve_qp_on_active_set(a_init, active_set, solver_name):
    """Solve min t s.t. a^T Q_k a <= t for k in active_set, sum(a)=1, a>=0."""
    n = len(a_init)
    a_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [cp.sum(a_var) == 1]

    for k in active_set:
        Q = build_q_sparse(int(k), n)
        constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)

    prob = cp.Problem(cp.Minimize(t_var), constraints)

    solvers_to_try = [cp.MOSEK, cp.GUROBI, solver_name, cp.CLARABEL, cp.ECOS, cp.SCS, cp.MOSEK]
    seen = set()
    for s in solvers_to_try:
        if s is None or s in seen:
            continue
        seen.add(s)
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.asarray(a_var.value, dtype=np.float64)
                a_val = np.maximum(a_val, 0)
                ss = a_val.sum()
                if ss < 1e-15:
                    return None, None
                a_val = a_val / ss
                return a_val, float(max(0, t_var.value))
        except Exception:
            continue
    return None, None


def active_set_qcqp(a_init, solver_name, time_limit, max_active_set_size=200, converge_tol=1e-10):
    """Iterative active-set method for the convex QCQP.
    Repeatedly identifies violated constraints and solves the reduced QP.
    """
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    t0 = time.monotonic()
    active = set()
    best_a = a.copy()
    best_val = fast_convolve(a).max()

    # Start with the argmax position
    b = fast_convolve(a)
    k_max = int(np.argmax(b))
    active.add(k_max)

    # Add some nearby high positions
    threshold = b.max() * 0.99
    high_pos = np.where(b >= threshold)[0]
    for k in high_pos[:10]:
        active.add(int(k))

    for iteration in range(50):
        if time.monotonic() - t0 > time_limit:
            break
        if len(active) > max_active_set_size:
            # Keep only the most violated
            b_cur = fast_convolve(a)
            active_list = sorted(active, key=lambda k: -b_cur[k])
            active = set(active_list[:max_active_set_size])

        sol_a, sol_t = solve_qp_on_active_set(a, sorted(active), solver_name)
        if sol_a is None:
            break

        a = sol_a
        b = fast_convolve(a)
        max_b = b.max()

        if max_b < best_val:
            best_val = max_b
            best_a = a.copy()

        # Check for violated constraints outside active set
        violation = 0.0
        worst_k = -1
        for k in range(len(b)):
            if k not in active:
                if b[k] > sol_t + converge_tol and b[k] > violation:
                    violation = b[k]
                    worst_k = k

        if worst_k >= 0 and violation > sol_t + converge_tol:
            active.add(worst_k)
            print(f"    Active-set QCQP iter {iteration}: |A|={len(active)}, max_b={max_b:.10f}, violated at k={worst_k} (val={violation:.10f} > t={sol_t:.10f})")
        else:
            print(f"    Active-set QCQP converged at iter {iteration}: |A|={len(active)}, max_b={max_b:.10f}")
            break

    return best_a, best_val


def fenchel_dual_refine(a_init, time_limit, n_steps=20000):
    """Fenchel dual mirror descent: max_p min_a a^T M(p) a over simplex p.
    Inner QP solved via projected gradient (warm-started).
    """
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    # Dual variable: uniform distribution over convolution positions
    n_pos = 2 * n - 1
    p = np.ones(n_pos) / n_pos

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 0.1  # Step size for mirror descent
    inner_iters = 50  # Projected gradient iterations for inner QP

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute M(p) @ a = sum_k p_k * Q_k @ a
        # Q_k @ a = element of conv(a, a) ... no, that's a^T Q_k a
        # Actually Q_k @ a gives the k-th "row" of the gradient
        # (Q_k @ a)_i = sum_j (Q_k)_{i,j} a_j = contribution to (a*a)_k from position i
        # Specifically: (Q_k @ a)_i = (1/2)*a_{k-i} if k-i != i, and a_i if k-i == i
        # = a_{k-i}/2 * (1 + delta(k-2i))... let me just compute it directly

        # Gradient of a^T M(p) a w.r.t. a = 2 * M(p) @ a
        # M(p) @ a = sum_k p_k * Q_k @ a
        # (Q_k @ a)_i = sum_j (Q_k)_{i,j} a_j
        # (Q_k)_{i,j} = 1/2 if i+j=k, i!=j; 1 if i+j=k, i=j
        # So (Q_k @ a)_i = a_{k-i}/2 if k-i != i and 0 <= k-i < n
        #                   + a_i/2 if k-i == i (since we already counted i!=j case)
        # Actually: (Q_k @ a)_i = sum_{j: i+j=k} (Q_k)_{i,j} a_j
        # If k-i is in [0, n-1]:
        #   (Q_k @ a)_i = (Q_k)_{i, k-i} * a_{k-i}
        # If k-i == i (i.e., 2i=k): (Q_k)_{i,i} = 1, so (Q_k @ a)_i = a_i
        # Else: (Q_k)_{i, k-i} = 1/2, so (Q_k @ a)_i = a_{k-i}/2

        # M(p) @ a = sum_k p_k * (Q_k @ a)
        # Let's compute this efficiently:
        # (sum_k p_k * Q_k @ a)_i = sum_k p_k * a_{k-i}/2 * [0<=k-i<n] + p_k * a_i * [k=2i] * [0<=k-i<n]
        # = (1/2) * sum_{j:0<=j<n} p_{i+j} * a_j + p_{2i} * a_i * (1/2)
        # Wait, let me redo:
        # (sum_k p_k Q_k)_{i,j} = p_{i+j} * (Q_{i+j})_{i,j}
        # (Q_{i+j})_{i,j} = 1/2 if i!=j, 1 if i==j
        # So (sum_k p_k Q_k)_{i,j} = p_{i+j} * 1/2 if i!=j and i+j <= 2n-2
        #                           = p_{2i} * 1 if i==j
        # = p_{i+j} * (1/2 + 1/2*delta_{i,j})

        # M(p) @ a has components:
        # (M a)_i = sum_j p_{i+j} * (1/2 + 1/2*delta_{i,j}) * a_j
        # = (1/2) * sum_j p_{i+j} * a_j + (1/2) * p_{2i} * a_i

        # Compute sum_j p_{i+j} * a_j for all i:
        # This is like a cross-correlation of p and a!
        # Let c_i = sum_j p_{i+j} * a_j
        # If we define p' as p extended with zeros, this is a cross-correlation.

        # Actually, let's compute it via FFT:
        # c_i = sum_j p_{i+j} a_j for i = 0, ..., n-1
        # This is the "anti-convolution" or cross-correlation.
        # c = correlate(p, a) but only the first n elements.
        # correlate(p, a)[i] = sum_j p[j] a[j+i] ... hmm, let me be careful.
        # sum_j p_{i+j} a_j: let m = i+j, then j = m-i, sum_m p_m a_{m-i} for m from i to i+n-1
        # This is (a flipped) convolved with p, taking specific elements.
        # Actually it's: cross_correlation(a, p) where (a ⋆ p)[i] = sum_j a_j p_{i+j}

        # Let's just compute it directly for now (n=1200, it's O(n^2) = 1.4M ops, fast enough)
        # Actually for the gradient we need 2*M(p)@a, and we can compute M(p)@a as:
        ma = np.zeros(n)
        for i in range(n):
            # (Ma)_i = (1/2)*sum_j p[i+j]*a_j + (1/2)*p[2i]*a_i
            js = range(n)
            # p[i+j] is valid when i+j < n_pos
            valid_max = min(n, n_pos - i)
            ma[i] = 0.5 * np.dot(a[:valid_max], p[i:i + valid_max])
            if 2 * i < n_pos:
                ma[i] += 0.5 * p[2 * i] * a[i]

        grad = 2.0 * ma  # gradient of a^T M a w.r.t. a

        # Inner: minimize a^T M a s.t. a in simplex via projected gradient
        inner_step = 1.0 / (2.0 * np.linalg.norm(grad, np.inf) + 1e-15)
        for _ in range(inner_iters):
            a_new = a - inner_step * grad
            # Project onto simplex
            a_new = _project_simplex(a_new)
            # Update gradient for new a
            ma2 = np.zeros(n)
            for i2 in range(n):
                valid_max2 = min(n, n_pos - i2)
                ma2[i2] = 0.5 * np.dot(a_new[:valid_max2], p[i2:i2 + valid_max2])
                if 2 * i2 < n_pos:
                    ma2[i2] += 0.5 * p[2 * i2] * a_new[i2]
            grad2 = 2.0 * ma2
            a = a_new
            grad = grad2

        # Update best
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Mirror descent update on p
        # Subgradient of g(p) = min_a a^T M(p) a is s_k = a^T Q_k a for all k
        # s_k = (a*a)_k = b_k
        b_val = fast_convolve(a)
        # s = b_val (the subgradient vector)
        # Mirror descent: p <- p * exp(-eta * s) / Z  (since we maximize, use -eta*g for ascent... wait)
        # We want to MAXIMIZE g(p). Gradient ascent: p <- p * exp(eta * s)
        # But s = b_val has large values, so we need to normalize
        s_shifted = b_val - b_val.max()
        log_p = np.log(p) + eta * s_shifted
        log_p -= log_p.max()
        p_new = np.exp(log_p)
        p_new = p_new / p_new.sum()
        p = p_new

        if (step + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FenchelDual iter={step+1} ({elapsed:.0f}s): score={best_score:.10f}")

        # Adaptive step size
        if step % 2000 == 1999 and step > 0:
            eta *= 0.7

    return best_a, best_score


def _project_simplex(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def lse_lbgfs_refine(a0, time_limit, T_stages=None):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    if T_stages is None:
        T_stages = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]

    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()
    deadline = t0 + time_limit
    n_stages = len(T_stages)
    iters_per_stage = max(30, 3000 // n_stages)

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def lse_obj(z_vec):
            a_vec = _softmax(z_vec)
            b_vec = fast_convolve(a_vec)
            max_b = b_vec.max()
            exp_terms = np.exp((b_vec - max_b) / stage_T)
            lse_val = max_b + stage_T * np.log(exp_terms.sum())
            return 2.0 * n * lse_val

        def lse_grad(z_vec):
            a_vec = _softmax(z_vec)
            b_vec = fast_convolve(a_vec)
            max_b = b_vec.max()
            exp_terms = np.exp((b_vec - max_b) / stage_T)
            w = exp_terms / exp_terms.sum()

            grad_a = np.zeros(n)
            for k in range(len(b_vec)):
                if w[k] < 1e-18:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_vec[k - idx]

            S_val = np.dot(grad_a, a_vec)
            grad_z = a_vec * (grad_a - S_val)
            return 2.0 * n * grad_z

        try:
            res = minimize(
                fun=lse_obj, x0=z, jac=lse_grad,
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-25,
                         'gtol': 1e-18, 'maxfun': iters_per_stage * 500, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return _softmax(z)


def _softmax(z_vec):
    """Stable softmax."""
    z_shifted = z_vec - z_vec.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z_vec)) / len(z_vec)
    return exp_z / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def multiplicative_subgradient(a0, time_limit, n_steps=100000):
    """Mirror descent (multiplicative weights) targeting active peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta_base = 0.05
    no_improve = 0
    scale = 1.0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        threshold = max_b * (1.0 - 2e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([0])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        eta = eta_base * scale
        log_scale = -eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 50:
            scale *= 0.8
            no_improve = 0
            if scale < 0.003:
                break

    return best_a, best_score


def find_best_solver():
    """Find the best available CVXPY solver for QP problems."""
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum_squares(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Phase 1: Nesterov LP alternating (main workhorse, ~55% time)
    Phase 2: Iterative active-set QCQP (exact convex solve, ~20% time)
    Phase 3: Fenchel dual mirror descent (alternative refinement, ~10% time)
    Phase 4: Low-temp L-BFGS (final precision, ~8% time)
    Phase 5: Multi-n exploration (n=1199, 1201, ~7% time)
    Phase 6: Final QCQP polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')
    solver = find_best_solver()
    print(f"Using CVXPY QP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        x = np.linspace(0, 1, n_init)
        best_seq = np.cos(np.pi * x / 2) ** 2
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")

    def update_best(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # === Phase 1: Nesterov LP Alternating (main workhorse) ===
    n_best = len(best_seq)
    print(f"\n=== Phase 1: Nesterov LP n={n_best} (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.50, 900.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver, n_max_iters=1000000)
        update_best(a_lp, "lp_phase1")

    # === Phase 2: Active-set QCQP (iterative constraint addition) ===
    print(f"\n=== Phase 2: Active-set QCQP (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_qcqp = min(rem * 0.20, 300.0)
    if t_qcqp > 20:
        print(f"  Budget: {t_qcqp:.0f}s")
        a_qp, val_qp = active_set_qcqp(best_seq, solver, t_qcqp,
                                        max_active_set_size=150, converge_tol=1e-12)
        if a_qp is not None:
            update_best(a_qp, "active_qcqp")

    # === Phase 3: Nesterov LP round 2 (from QCQP result) ===
    print(f"\n=== Phase 3: LP Round 2 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, solver, n_max_iters=500000)
        update_best(a_lp2, "lp_phase3")

    # === Phase 4: Fenchel dual refinement ===
    print(f"\n=== Phase 4: Fenchel Dual (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_fd = min(rem * 0.10, 150.0)
    if t_fd > 20:
        print(f"  Budget: {t_fd:.0f}s")
        a_fd, _ = fenchel_dual_refine(best_seq, t_fd, n_steps=50000)
        update_best(a_fd, "fenchel_dual")

    # === Phase 5: Low-temperature L-BFGS ===
    print(f"\n=== Phase 5: L-BFGS Low-Temp (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_lb = min(rem * 0.08, 120.0)
    if t_lb > 15:
        T_stages = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
        a_lb = lse_lbgfs_refine(best_seq, t_lb, T_stages)
        update_best(a_lb, "lbgfs_lowtemp")

    # === Phase 6: Multi-n exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration (score={best_score:.10f}) ===")
    for n_try in [1199, 1201]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_n = min(rem * 0.15, 150.0)
        print(f"  Trying n={n_try}, budget={t_n:.0f}s")
        init_n = interpolate_sequence(best_seq, n_try)
        a_n, _ = nesterov_lp_refine(init_n, t_n * 0.6, solver, n_max_iters=200000)
        update_best(a_n, f"lp_n{n_try}")
        rem = deadline - time.monotonic()
        if best_seq is not None and len(best_seq) == n_try and rem > 15:
            a_q, _ = active_set_qcqp(best_seq, solver, min(rem * 0.3, 60.0),
                                       max_active_set_size=100, converge_tol=1e-11)
            if a_q is not None:
                update_best(a_q, f"qcqp_n{n_try}")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        a_ms, _ = multiplicative_subgradient(best_seq, min(rem * 0.3, 30.0), n_steps=100000)
        update_best(a_ms, "final_mult")
        rem = deadline - time.monotonic()
        if rem > 10:
            a_qf, _ = active_set_qcqp(best_seq, solver, min(rem * 0.5, 60.0),
                                       max_active_set_size=100, converge_tol=1e-13)
            if a_qf is not None:
                update_best(a_qf, "final_qcqp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
