"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using palindromic symmetry exploitation.
Key insight: The optimal solution is palindromic (a_j = a_{n-1-j}), reducing
the problem to n/2 free variables. This enables:
1. Fast LP alternating (half the variables/constraints)
2. Direct QCQP solve with palindromic constraint
3. Active-set refinement for exact optimality
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.sparse import lil_matrix, csc_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def pal_expand(h):
    """Expand half-palindromic vector to full palindromic vector.
    For even n=2m: h has length m, full = [h, h_reversed]
    For odd n=2m+1: h has length m+1, full = [h, h_reversed[:-1]]
    """
    if len(h) > 0 and (2 * len(h) - 1) % 2 == 1:
        # Could be odd: 2m+1 where m+1 = len(h), so 2*(len(h))-1 = 2*len(h)-1
        # Full length = 2*len(h) - 1
        return np.concatenate([h, h[-2::-1]])
    else:
        # Even: 2m where m = len(h)
        # Full length = 2*len(h)
        return np.concatenate([h, h[::-1]])


def pal_half(full):
    """Extract the first half of a palindromic vector."""
    n = len(full)
    if n % 2 == 0:
        return full[:n // 2].copy()
    else:
        return full[:(n + 1) // 2].copy()


def eval_score(a):
    """Compute score = 2*n*max_b/(sum_a)^2 locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_lp_matrix_pal(a, n):
    """Build matrix M (n x m) where M @ g = (a * b)[:n] when b is palindromic with half g.
    For even n=2m: g has length m, b_j = g[min(j, n-1-j)]
    For odd n=2m+1: g has length m+1, b_j = g[min(j, n-1-j)]
    """
    n = len(a)
    m = (n + 1) // 2  # number of free variables in palindromic b
    M = np.zeros((n, m))
    for k in range(n):
        for i in range(max(0, k - n + 1), min(k, n - 1) + 1):
            j = k - i  # index into b
            if 0 <= j < n:
                # b_j = g[min(j, n-1-j)]
                g_idx = min(j, n - 1 - j)
                M[k, g_idx] += a[i]
    return M


def solve_lp_best_response_pal(a, solver):
    """Given palindromic a (full), find palindromic b minimizing max (a*b)[k]."""
    n = len(a)
    m = (n + 1) // 2
    M = build_lp_matrix_pal(a, n)
    
    g_var = cp.Variable(m, nonneg=True)
    t_var = cp.Variable()
    constraints = [M @ g_var <= t_var, cp.sum(g_var * (2 * np.ones(m) - np.arange(m) - np.arange(m)[::-1])) == 1]
    
    # Wait, the sum constraint is tricky. sum(b) = sum(g_j * multiplicity)
    # For even n=2m: sum(b) = 2*sum(g)
    # For odd n=2m+1: sum(b) = 2*sum(g) - g_m (center is counted once)
    
    if n % 2 == 0:
        # 2m: sum(b) = 2 * sum(g)
        sum_constraint = 2 * cp.sum(g_var) == 1
    else:
        # 2m+1: sum(b) = 2*sum(g) - g[m-1] (center)
        # Actually: sum(b) = sum_{j=0}^{2m} b_j = 2*sum_{j=0}^{m-1} g_j + g_m
        # Wait: b = [g_0, g_1, ..., g_m, g_{m-1}, ..., g_0]
        # sum(b) = 2*(g_0 + ... + g_{m-1}) + g_m
        sum_constraint = 2 * cp.sum(g_var[:-1]) + g_var[-1] == 1
    
    constraints = [M @ g_var <= t_var, sum_constraint]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    solvers_to_try = [solver, cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP, cp.MOSEK, cp.GUROBI]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.asarray(g_var.value, dtype=np.float64)
                g_val = np.maximum(g_val, 0)
                b_full = pal_expand(g_val)
                # Normalize to sum=1
                s_sum = b_full.sum()
                if s_sum < 1e-15:
                    return None
                b_full = b_full / s_sum
                return b_full, float(max(0, t_var.value))
        except Exception:
            continue
    return None, None


def build_q_sparse_k(k, n):
    """Build sparse matrix Q_k (n x n) such that a^T Q_k a = (a*a)[k]."""
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)


def solve_qcqp_active_pal(a, solver, rel_tol=1e-5, max_active=20):
    """Solve active-set QCQP with palindromic constraint.
    Minimize t s.t. (a*a)[k] <= t for active k, sum(a)=1, a>=0, a palindromic.
    """
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    
    # By palindromic symmetry, only need k = 0 to n-1
    b_half = b[:n]
    threshold = max_b * (1.0 - rel_tol)
    active = np.where(b_half >= threshold)[0]
    
    if len(active) > max_active:
        active = active[np.argsort(b_half[active])[-max_active:]]
    if len(active) == 0:
        active = np.array([np.argmax(b_half)])
    
    a_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [cp.sum(a_var) == 1]
    
    # Palindromic constraint
    constraints.append(a_var == a_var[::-1])
    
    for k in active:
        Q = build_q_sparse_k(int(k), n)
        constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    solvers_to_try = [cp.MOSEK, cp.GUROBI, solver, cp.CLARABEL, cp.ECOS, cp.SCS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.asarray(a_var.value, dtype=np.float64)
                a_val = np.maximum(a_val, 0)
                s_sum = a_val.sum()
                if s_sum < 1e-15:
                    return None, None
                a_val = a_val / s_sum
                # Enforce palindromicity (average with reverse)
                a_val = 0.5 * (a_val + a_val[::-1])
                a_val = a_val / a_val.sum()
                return a_val, float(max(0, t_var.value))
        except Exception as e:
            print(f"    QCQP solver {s} failed: {e}")
            continue
    return None, None

def nesterov_lp_pal(a0, time_limit, solver, n_max_iters):
    """Nesterov-accelerated LP alternating with palindromic constraint."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    # Enforce palindromicity
    a = 0.5 * (a + a[::-1])
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    y = a.copy()
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(42)
    lp_times = []
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response_pal(y, solver)
        lp_times.append(time.monotonic() - lp_t0)
        
        if b_new is None:
            continue
        
        # Enforce palindromicity on b_new
        b_new = 0.5 * (b_new + b_new[::-1])
        b_new = b_new / b_new.sum()
        
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        y = 0.5 * (y + y[::-1])
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy
        
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        a = 0.5 * (a + a[::-1])
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        score_a = eval_score(a)
        score_y = eval_score(y)
        
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-9:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            
            if stall_count >= 10:
                # Perturb
                m = (n + 1) // 2
                h = pal_half(y)
                h = h + 0.02 * rng.standard_normal(m)
                h = np.maximum(h, 0)
                y_new = pal_expand(h)
                sy2 = y_new.sum()
                if sy2 > 1e-15:
                    y_new = y_new / sy2
                    y = y_new
                stall_count = 0
            
            elapsed = time.monotonic() - t0
            avg_lp = np.mean(lp_times[-50:]) * 1000 if len(lp_times) >= 50 else 0
            print(f"    NesterovLP-pal iter={it+1} ({elapsed:.0f}s): score={best_score:.10f} avg_lp={avg_lp:.1f}ms")
        
        if len(lp_times) > 100 and np.mean(lp_times[-100:]) > 3.0:
            print(f"    LP too slow, stopping at iter {it+1}")
            break
    
    return best_a, best_score


def active_set_qcqp_iterative_pal(a, solver, max_rounds=20):
    """Iteratively add violated constraints to the QCQP active set."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b[:n].max()
    
    threshold = max_b * (1.0 - 1e-4)
    active = set(np.where(b[:n] >= threshold)[0].tolist())
    
    for round_idx in range(max_rounds):
        a_new = solve_qcqp_active_pal(a, solver, rel_tol=1e-3, max_active=30)
        if a_new is None:
            break
        
        b_new = fast_convolve(a_new)
        max_b_new = b_new[:n].max()
        
        # Check for violations
        violated = np.where(b_new[:n] > max_b_new * (1 - 1e-6))[0]
        new_active = active.union(violated.tolist())
        
        if new_active == active and len(new_active) < 50:
            # Converged
            break
        
        active = new_active
        a = a_new
    
    return a


def lbfgs_refine_pal(a0, time_limit, T_stages):
    """L-BFGS polish with palindromic constraint using log-sum-exp smoothing."""
    n = len(a0)
    m = (n + 1) // 2
    h = pal_half(a0)
    z = np.log(np.maximum(h, 1e-300))
    t0 = time.monotonic()
    deadline = t0 + time_limit
    n_stages = len(T_stages)
    iters_per_stage = max(20, 5000 // n_stages)
    
    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        def lse_obj(z_in):
            if time.monotonic() > deadline:
                return 1e15
            z_in = z_in - z_in.max()
            exp_z = np.exp(z_in)
            h_in = exp_z / exp_z.sum()
            a_in = pal_expand(h_in)
            b_in = fast_convolve(a_in)
            max_b = b_in[:n].max()
            lse = max_b + stage_T * np.log(np.exp((b_in[:n] - max_b) / stage_T).sum())
            return 2.0 * n * lse
        
        def lse_grad(z_in):
            if time.monotonic() > deadline:
                return np.zeros_like(z_in)
            z_in = z_in - z_in.max()
            exp_z = np.exp(z_in)
            h_in = exp_z / exp_z.sum()
            a_in = pal_expand(h_in)
            b_in = fast_convolve(a_in)
            max_b = b_in[:n].max()
            w = np.exp((b_in[:n] - max_b) / stage_T)
            w /= w.sum()
            
            # Gradient of lse wrt h_in
            # lse = max_b + T * log(sum exp((b_k - max_b)/T)) for k=0..n-1
            # db_k/dh_j is complex due to palindromic expansion
            # Let's use finite differences for the gradient
            eps = 1e-7
            grad_h = np.zeros(m)
            for j in range(m):
                h_plus = h_in.copy()
                h_plus[j] += eps
                a_plus = pal_expand(h_plus)
                a_plus = a_plus / a_plus.sum()
                b_plus = fast_convolve(a_plus)
                max_plus = b_plus[:n].max()
                lse_plus = max_plus + stage_T * np.log(np.exp((b_plus[:n] - max_plus) / stage_T).sum())
                
                h_minus = h_in.copy()
                h_minus[j] = max(h_in[j] - eps, 1e-15)
                a_minus = pal_expand(h_minus)
                a_minus = a_minus / a_minus.sum()
                b_minus = fast_convolve(a_minus)
                max_minus = b_minus[:n].max()
                lse_minus = max_minus + stage_T * np.log(np.exp((b_minus[:n] - max_minus) / stage_T).sum())
                
                grad_h[j] = 2.0 * n * (lse_plus - lse_minus) / (2 * eps)
            
            # Convert to z-domain: grad_z = h_in * (grad_h - sum(grad_h * h_in))
            grad_z = h_in * (grad_h - np.dot(grad_h, h_in))
            return grad_z
        
        try:
            res = minimize(
                fun=lse_obj, x0=z, jac=lse_grad,
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-12, 'maxfun': iters_per_stage * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    h_final = exp_z / exp_z.sum()
    a_final = pal_expand(h_final)
    a_final = np.maximum(a_final, 0)
    a_final = a_final / a_final.sum()
    return a_final


def interpolate_pal(f, new_n):
    """Interpolate a palindromic sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    # Enforce palindromicity
    f_interp = 0.5 * (f_interp + f_interp[::-1])
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
