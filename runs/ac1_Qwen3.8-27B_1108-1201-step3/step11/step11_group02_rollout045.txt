    """FISTA on LogSumExp smoothed objective."""
    n = len(a0)
    x = np.maximum(a0, 0)
    s = x.sum()
    x = x / s
    y = x.copy()
    t_k = 1.0
    t0 = time.monotonic()
    best_x = x.copy()
    best_score = score(x)
    
    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute gradient at y
        grad = lse_gradient(y, T)
        
        # Gradient step
        z = y - (1.0/L) * grad
        
        # Project onto simplex {a >= 0, sum(a) = 1}
        x_new = project_simplex(z)
        
        # FISTA update
        t_new = (1 + np.sqrt(1 + 4*t_k*t_k)) / 2
        y = x_new + ((t_k - 1) / t_new) * (x_new - x)
        y = np.maximum(y, 0)
        y = y / y.sum()
        
        x = x_new
        t_k = t_new
        
        sc = score(x)
        if sc < best_score:
            best_score = sc
            best_x = x.copy()
    
    return best_x, best_score

1. LP-based alternating with Nesterov acceleration
2. LogSumExp-smoothed FISTA with FFT-based gradients
3. Multi-resolution (n) exploration

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence (scale-invariant, assumes non-negative)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w


def build_conv_matrix_values(n):
    """Return arrays to build the sparse conv matrix T (T @ b = convolve(ones/n, b)).
    Actually, T depends on the fixed sequence a. Let me just return the indices."""
    size = 2 * n - 1
    row_ind = []
    col_ind = []
    for i in range(n):
        row_ind.extend(range(i, i + n))
        col_ind.extend([i] * n)
    return np.array(row_ind), np.array(col_ind)


def build_sparse_conv_matrix(a):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    row_ind = []
    col_ind = []
    data = []
    for i in range(n):
        row_ind.extend(range(i, i + n))
        col_ind.extend([i] * n)
        data.extend(a)
    return csr_matrix((data, (np.array(row_ind, dtype=np.int32), np.array(col_ind, dtype=np.int32))),
                       shape=(size, n))


_lp_solver = None

    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:

def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max conv(a,b) subject to b>=0, sum(b)=1."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = build_sparse_conv_matrix(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [T @ b_var <= t_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver)
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            for alt in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
                if alt == solver:
                    continue
                try:
                    prob.solve(solver=alt)
                    if prob.status in ('optimal', 'optimal_inaccurate'):
                        break
                except Exception:
                    continue
    except Exception:
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    b_val = b_val / s
    return b_val


def nesterov_lp_refine(a0, time_limit, solver, max_iters=100000):
    """Nesterov LP alternating method."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score
    rng = np.random.default_rng(42)
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        
        # Nesterov step
        alpha = 2.0 / (np.sqrt(it + 1) + 1.0)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy
        
        # Weighted average
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        score_a = evaluate_local(a)
        score_y = evaluate_local(y)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        
        if (it + 1) % 50 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            
            if stall >= 10:
                # Perturb
                b_cur = fast_convolve(y)
                peak_idx = np.argmax(b_cur)
                js = max(0, peak_idx - n + 1)
                je = min(n - 1, peak_idx)
                idx = np.arange(js, je + 1)
                contribs = y[idx] * y[peak_idx - idx]
                n_reduce = max(3, len(idx) // 8)
                top_pos = idx[np.argsort(contribs)[::-1][:n_reduce]]
                for pos in top_pos:
                    if y[pos] > 1e-10:
                        y[pos] *= 0.95
                noise = rng.standard_normal(n) * 0.02
                y = np.maximum(y + noise, 0)
                y = y / y.sum()
                a = y.copy()
                stall = 0
        
        if (it + 1) % 1000 == 0:
            print(f"    NLP iter={it+1}: score={best_score:.8f}")
    
    return best_a, best_score


def lse_gradient_fft(a, T):
    """Compute LSE_T(conv(a,a)) and its gradient w.r.t. a using FFT.
    Returns (lse_value, gradient).
    """
    n = len(a)
    b = fast_convolve(a)  # length 2n-1
    max_b = b.max()
    shifted = (b - max_b) / T
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    
    lse_val = max_b + T * np.log(exp_shifted.sum())
    
    # Gradient: grad[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    # This is conv(w, flip(a))[n-1 : 2n-2]
    # Use FFT
    fft_size = 1 << (3 * n - 2).bit_length()  # Need at least (2n-1) + n - 1 = 3n-2
    W = rfft(w, fft_size)
    A_flip = rfft(np.flip(a), fft_size)
    conv_result = irfft(W * A_flip, fft_size)
    corr = conv_result[n-1 : 2*n-2]  # length n-1? No, should be length n
    
    # Wait, conv(w, flip_a) has length (2n-1) + n - 1 = 3n-2
    # corr[i] = conv[i + n - 1] for i = 0, ..., n-1
    # So we need indices n-1 to n-1+n-1 = 2n-2
    # That's n values. ✓
    
    grad = 2.0 * corr
    
    return lse_val, grad


def lse_fista(a0, T_stages, time_limit):
    """FISTA on LogSumExp smoothed objective with decreasing temperature."""
    n = len(a0)
    x = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = x.sum()
    if s < 1e-15:
        x = np.ones(n) / n
    x = x / s
    t0 = time.monotonic()
    best_x = x.copy()
    best_score = evaluate_local(x)
    
    for T in T_stages:
        if time.monotonic() - t0 > time_limit - 2:
            break
        
        x = project_simplex(x)
        y = x.copy()
        t_k = 1.0
        # Initial step size estimate
        L = 4.0 * n / T  # Conservative estimate
        iters_in_stage = 0
        
        while time.monotonic() - t0 < time_limit - 2:
            # Compute gradient at y
            lse_val, grad = lse_gradient_fft(y, T)
            
            # Backtracking line search for L
            # For now, use fixed L (conservative)
            step = 1.0 / L
            
            # Gradient step
            z = y - step * grad
            
            # Project onto simplex
            x_new = project_simplex(z)
            
            # FISTA momentum update
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            y = x_new + ((t_k - 1.0) / t_new) * (x_new - x)
            y = project_simplex(y)
            
            x = x_new
            t_k = t_new
            iters_in_stage += 1
            
            # Check score periodically
            if iters_in_stage % 50 == 0:
                sc = evaluate_local(x)
                if sc < best_score:
                    best_score = sc
                    best_x = x.copy()
            
            if iters_in_stage > 5000:
                break
        
        # Check score at end of stage
        sc = evaluate_local(x)
        if sc < best_score:
            best_score = sc
            best_x = x.copy()
        
        print(f"    LSE-FISTA T={T:.4f}: score={best_score:.8f}")
    
    return best_x, best_score


def coordinate_descent(a0, time_limit, max_steps=200000):
    """Coordinate descent focusing on peak-contributing elements."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall = 0
    rng = np.random.default_rng(99)
    
    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_m = min(6, len(idx))
        top_positions = idx[np.argsort(contribs)[::-1][:top_m]]
        
        improved = False
        for pos in top_positions:
            if a[pos] < 1e-12:
                continue
            for factor in [0.999, 0.997, 0.995, 0.99, 0.985, 0.98]:
                a_trial = a.copy()
                a_trial[pos] *= factor
                diff = a[pos] - a_trial[pos]
                if diff < 1e-15:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    break
                a_trial[mask] = others + diff * others / others_sum
                a_trial = a_trial / a_trial.sum()
                score = evaluate_local(a_trial)
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall > 200:
                # Random perturbation
                pos = rng.integers(0, n)
                a_trial = a.copy()
                a_trial[pos] *= 0.95
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = evaluate_local(a_trial)
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                stall = 0
            elif stall > 5000:
                break
        else:
            stall = 0
        
        if (step + 1) % 10000 == 0:
            print(f"    CoordDesc step={step+1}: score={best_score:.8f}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create structured initial sequence."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = np.exp(-3 * x * (1 - x))
    elif choice == 3:
        # Gaussian
        a = np.exp(-8 * (x - 0.5) ** 2)
    elif choice == 4:
        # Bimodal
        a = np.exp(-10 * (x - 0.3) ** 2) + np.exp(-10 * (x - 0.7) ** 2)
    elif choice == 5:
        # Sine modulated
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 6:
        # Power law
        a = x ** 0.5 * (1 - x) ** 0.5
    else:
        # Triangle
        a = 1 - np.abs(2 * x - 1)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
