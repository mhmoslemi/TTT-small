            # Adam step
            lr = lr0 / (1.0 + 0.1 * step / steps_per)
            m = b1 * m + (1 - b1) * gz
            v = b2 * v + (1 - b2) * gz ** 2
            mh = m / (1 - b1 ** (step + 1))
            vh = v / (1 - b2 ** (step + 1))
            loga -= lr * mh / (np.sqrt(vh) + eps)
            loga -= loga.mean()

    return best_a, best_s


def build_conv_T(a):
    """Build matrix T (shape 2n-1 x n) such that T @ b = convolve(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a_fixed, timeout_s=10.0):
    """Solve LP: min M s.t. conv(a,b)_k <= M for all k, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    T = build_conv_T(a_fixed)
    ones = np.ones(2 * n - 1)
    cons = [T @ b <= M * ones, cp.sum(b) == 1]
    prob = cp.Problem(cp.Minimize(M), cons)
    for solver in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate') and b.value is not None:
                bv = np.maximum(np.asarray(b.value, dtype=np.float64), 0)
                s = bv.sum()
                if s > 1e-15:
                    bv /= s
                    return bv, max(0.0, float(M.value))
        except Exception:
            continue
    return None, None


def lp_alternate(a0, time_limit, max_iter=5000):
    """Nesterov-accelerated LP alternating method."""
    n = len(a0)
    a = np.maximum(a0, 1e-15).copy()
    a /= a.sum()
    y = a.copy()
    best_a = a.copy()
    best_s = eval_score(a)
    t0 = time.monotonic()
    lp_times = []

    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        t_lp = time.monotonic()
        b_new, M_val = lp_best_response(y)
        lp_times.append(time.monotonic() - t_lp)
        if b_new is None:
            break

        # If LP is consistently too slow, stop
        if len(lp_times) > 30:
            avg = np.mean(lp_times[-30:])
            if avg > 8.0:
                print(f"    LP avg {avg:.1f}s/iter, stopping at iter {it+1}")
                break

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 1e-15:
            y /= sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / ((it + 1) + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa > 1e-15:
            a /= sa

        sc_a = eval_score(a)
        sc_y = eval_score(y)
        if sc_a < best_s:
            best_s = sc_a
            best_a = a.copy()
        if sc_y < best_s:
            best_s = sc_y
            best_a = y.copy()

        if (it + 1) % 50 == 0:
            print(f"    LP-alt iter {it+1} ({time.monotonic()-t0:.0f}s): best={best_s:.10f}")

    return best_a, best_s


def interpolate_seq(a, new_n):
    """Interpolate sequence to new length, preserving sum=1."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a_norm = a / a.sum()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a_norm)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase search minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Returns the best non-negative sequence found.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, budget_s - 15.0)
    rng = np.random.default_rng(seed)

    best_a = None
    best_score = float('inf')

    # --- Load parent sequence ---
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all() and parent.sum() > 0.01:
            p_norm = parent / parent.sum()
            ps = eval_score(p_norm)
            print(f"Parent: n={len(parent)}, score={ps:.10f}")
            best_a = p_norm
            best_score = ps
    except Exception as e:
        print(f"No parent loaded: {e}")

    # --- Try uniform sequences at various lengths (optimal by Cauchy-Schwarz) ---
    for n_u in [1000, 1200, 1400, 1500, 2000, 100]:
        u = np.ones(n_u) / n_u
        us = eval_score(u)
        print(f"Uniform n={n_u}: score={us:.10f}")
        if us < best_score:
            best_score = us
            best_a = u.copy()

    if best_a is None:
        best_a = np.ones(1200) / 1200.0
        best_score = eval_score(best_a)

    n_best = len(best_a)
    print(f"\nInitial best: n={n_best}, score={best_score:.10f}")

    def try_update(a_cand, label):
        nonlocal best_a, best_score
        if a_cand is None:
            return
        a_cand = np.maximum(np.asarray(a_cand, dtype=np.float64), 0.0)
        s = a_cand.sum()
        if s < 0.01:
            return
        a_cand = a_cand / s
        sc = eval_score(a_cand)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_cand.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.10f} (n={len(a_cand)}) ***")

    # --- Phase 1: Adam warm-up ---
    rem = deadline - time.monotonic()
    if rem > 60:
        t1 = min(rem * 0.25, 300.0)
        print(f"\n=== Phase 1: Adam warm-up ({t1:.0f}s), n={n_best} ===")
        T_warm = [20, 10, 5, 3, 2, 1, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01]
        a1, s1 = adam_smooth(best_a, t1, T_warm, lr0=0.01, topk_frac=0.2)
        try_update(a1, "adam_warm")

    # --- Phase 2: LP Alternating ---
    rem = deadline - time.monotonic()
    if rem > 90:
        t2 = min(rem * 0.30, 400.0)
        n_lp = len(best_a)
        print(f"\n=== Phase 2: LP Alternating ({t2:.0f}s), n={n_lp} ===")
        a2, s2 = lp_alternate(best_a, t2, max_iter=5000)
        try_update(a2, "lp_alt_1")

    # --- Phase 3: Fine Adam ---
    rem = deadline - time.monotonic()
    if rem > 40:
        t3 = min(rem * 0.25, 300.0)
        print(f"\n=== Phase 3: Fine Adam ({t3:.0f}s) ===")
        T_fine = [1, 0.5, 0.3, 0.2, 0.1, 0.07, 0.05, 0.04, 0.03, 0.02, 0.015, 0.01, 0.007, 0.005, 0.003, 0.002]
        a3, s3 = adam_smooth(best_a, t3, T_fine, lr0=0.005, topk_frac=0.08)
        try_update(a3, "adam_fine")

    # --- Phase 4: LP Round 2 ---
    rem = deadline - time.monotonic()
    if rem > 60:
        t4 = min(rem * 0.25, 250.0)
        print(f"\n=== Phase 4: LP Round 2 ({t4:.0f}s) ===")
        a4, s4 = lp_alternate(best_a, t4, max_iter=3000)
        try_update(a4, "lp_alt_2")

    # --- Phase 5: Try larger n ---
    rem = deadline - time.monotonic()
    for n_try in [1400, 1600, 2000, 2500]:
        if rem < 90:
            break
        print(f"\n=== Phase 5: Interpolate to n={n_try} ===")
        a_interp = interpolate_seq(best_a, n_try)
        t5a = min(rem * 0.35, 200.0)
        T_list5 = [5, 3, 2, 1, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005]
        a5, s5 = adam_smooth(a_interp, t5a, T_list5, lr0=0.01, topk_frac=0.15)
        try_update(a5, f"adam_n{n_try}")
        rem = deadline - time.monotonic()
        if rem > 60:
            t5b = min(rem * 0.3, 120.0)
            a5b, s5b = lp_alternate(a_interp, t5b, max_iter=2000)
            try_update(a5b, f"lp_n{n_try}")
        rem = deadline - time.monotonic()

    # --- Phase 6: Multi-restart perturbation ---
    rem = deadline - time.monotonic()
    if rem > 60:
        print(f"\n=== Phase 6: Perturbation restarts ===")
        n_cur = len(best_a)
        for trial in range(range(3) if rem > 30 else range(2)):
            if time.monotonic() >= deadline:
                break
            # Structured perturbation
            x = np.linspace(0, 2 * np.pi, n_cur)
            freq = rng.integers(2, 8)
            phase = rng.uniform(0, 2 * np.pi)
            amp = 0.02 + 0.01 * trial
            pert = amp * np.cos(freq * x + phase)
            a_pert = best_a + pert
            a_pert = np.maximum(a_pert, 0)
            sp = a_pert.sum()
            if sp > 0.01:
                a_pert /= sp
            t6 = min((deadline - time.monotonic()) * 0.25, 80.0)
            if t6 < 10:
                break
            T6 = [2, 1, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.01, 0.005]
            a6, s6 = adam_smooth(a_pert, t6, T6, lr0=0.008, topk_frac=0.1)
            try_update(a6, f"pert_{trial}")

    # --- Phase 7: Final polish ---
    rem = deadline - time.monotonic()
    if rem > 20:
        t7 = min(rem * 0.5, 60.0)
        print(f"\n=== Phase 7: Final polish ({t7:.0f}s) ===")
        T7 = [0.5, 0.3, 0.2, 0.1, 0.07, 0.05, 0.03, 0.02, 0.015, 0.01, 0.007, 0.005, 0.003, 0.002, 0.001]
        a7, s7 = adam_smooth(best_a, t7, T7, lr0=0.003, topk_frac=0.05)
        try_update(a7, "final_polish")

    # --- Final verification ---
    final_local = eval_score(best_a)
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(best_a)}, local score={final_local:.10f}")
    print(f"Elapsed: {time.monotonic() - t_start:.1f}s / {budget_s:.0f}s")
    print(f"{'='*60}")

    final_list = [float(max(0.0, min(1000.0, x))) for x in best_a]
    try:
        fs = evaluate_sequence(final_list)
        print(f"evaluate_sequence score: {fs:.10f}")
    except Exception as e:
        print(f"evaluate_sequence error: {e}")

    return final_list
