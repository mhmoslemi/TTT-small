def fast_conv(a, b=None):
    """Full linear convolution a*b using FFT. If b is None, computes a*a."""
    if b is None:

    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)

def conv_matrix_rows(a_fixed, k_list):

def subgradient_mirror_descent(a0, time_limit, eta_init=0.1, eta_decay=0.999):
    """
    Subgradient + mirror descent (multiplicative weights) for the nonconvex min-max problem.
    Each iteration is O(n log n) via FFT.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    eta = eta_init
    
    # Running average with exponential weighting
    ema_a = a.copy()
    ema_alpha = 0.05
    ema_best_score = best_score
    ema_best_a = a.copy()
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute convolution and find argmax
        conv = fast_autocorr(a)
        k_star = int(np.argmax(conv))
        
        # Subgradient: grad_i = 2 * a[k_star - i] (if valid)
        grad = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        for j in range(js, je + 1):
            grad[j] = 2.0 * a[k_star - j]
        
        # Mirror descent update (multiplicative weights)
        # a_i <- a_i * exp(-eta * grad_i), then normalize
        update = np.exp(-eta * grad)
        a_new = a * update
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        a = a_new
        eta *= eta_decay
        
        # Check score
        if it % 50 == 1 or time.monotonic() - t0 > time_limit - 1:
            score = compute_score(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
        
        # EMA
        ema_a = (1 - ema_alpha) * ema_a + ema_alpha * a
        if it % 200 == 0:
            ema_score = compute_score(ema_a)
            if ema_score < ema_best_score:
                ema_best_score = ema_score
                ema_best_a = ema_a.copy()
            if ema_score < best_score:
                best_score = ema_score
                best_a = ema_a.copy()
        
        if it % 10000 == 0:
            elapsed = time.monotonic() - t0
            sc = compute_score(a)
            print(f"      SGD it={it} ({elapsed:.0f}s): score={sc:.6f}, best={best_score:.6f}, eta={eta:.6f}")
    
    return best_a, best_score


def lse_gradient_optimize(a0, time_limit, T_start=50.0, T_end=0.5, n_stages=15):
    """LSE-smoothed optimization with L-BFGS and temperature annealing."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = compute_score(a_init)
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        iters_per_stage = max(20, min(100, int(30 + 30 * (si / n_stages))))
        
        def func(z_in):
            a = softmax_fn(z_in)
            b = fast_autocorr(a)
            Tb = stage_T * b
            max_Tb = Tb.max()
            return (max_Tb + np.log(np.sum(np.exp(Tb - max_Tb)))) / stage_T
        
        def jac(z_in):
            a = softmax_fn(z_in)
            b = fast_autocorr(a)
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            
            # grad_a[i] = 2 * sum_k w[k] * a[k-i]
            conv_len = n + (2 * n - 1) - 1
            fft_size = 1 << (conv_len - 1).bit_length()
            A_rev = rfft(a[::-1].copy(), fft_size)
            W = rfft(w, fft_size)
            conv = irfft(A_rev * W, fft_size)
            grad_a = 2.0 * conv[n - 1:n - 1 + n]
            
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)
            return grad_z
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': iters_per_stage * 300,
                         'disp': False}
            )
            z = res.x
            a = softmax_fn(z)
            score = compute_score(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                print(f"      LSE stage {si+1}/{n_stages} T={stage_T:.2f}: new best={score:.6f}")
        except Exception as e:
            print(f"      LSE stage {si+1} error: {e}")
    
    return best_a, best_score

                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation to escape local minimum
            for attempt in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    frac = rng.uniform(0.003, 0.02)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        a_trial = a_trial / a_trial.sum()
                        score = compute_score(a_trial)
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
                            break
            if stall > 300:
                break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = int(rng.integers(0, 6))
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 1.5
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.6 + 0.3 * np.cos(3 * x) + 0.1 * np.cos(7 * x)
    else:
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: multi-phase optimization with LP Nesterov, subgradient MD,
    LSE refinement, and peak shaving at multiple lengths.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)
    
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = compute_score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = compute_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    if parent is None:
        n_base = 1200
        best_seq = structured_init(n_base, rng)
        best_score = compute_score(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)
    
    remaining = time_budget - (time.monotonic() - t_global)
    
    # === Phase 1: LP Nesterov at base length ===
    print(f"\n=== Phase 1: LP Nesterov at n={n_base} ===")
    t1 = min(remaining * 0.25, 400.0)
    if t1 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t1, solver, max_active=250)
        try_update(a_out, f"lp_n{n_base}")
    
    # === Phase 2: Subgradient mirror descent (fast iterations) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Subgradient MD at n={n_base} ===")
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 10:
        a_out, _ = subgradient_mirror_descent(best_seq, t2, eta_init=0.05, eta_decay=0.9995)
        try_update(a_out, f"sgd_n{n_base}")
    
    # === Phase 3: LSE refinement at base length ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LSE at n={n_base} ===")
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 10:
        a_out, _ = lse_gradient_optimize(best_seq, t3, T_start=30.0, T_end=0.5, n_stages=12)
        try_update(a_out, f"lse_n{n_base}")
    
    # === Phase 4: Try longer lengths ===
    for length_mult, phase_name in [(1.1, 4), (1.25, 5), (1.5, 6)]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        n_long = int(round(n_base * length_mult))
        n_long = max(n_long, n_base + 50)
        
        print(f"\n=== Phase {phase_name}: LP at n={n_long} ===")
        t_ph = min(remaining * 0.15, 200.0)
        if t_ph > 20:
            init_long = interpolate_sequence(best_seq, n_long)
            a_out, _ = nesterov_lp_optimize(init_long, t_ph, solver, max_active=250)
            try_update(a_out, f"lp_n{n_long}")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 30:
            break
        print(f"\n=== Phase {phase_name}b: LSE at n={n_long} ===")
        t_ph2 = min(remaining * 0.10, 100.0)
        if t_ph2 > 10:
            init_long2 = interpolate_sequence(best_seq, n_long)
            a_out, _ = lse_gradient_optimize(init_long2, t_ph2, T_start=100.0, T_end=0.5, n_stages=10)
            try_update(a_out, f"lse_n{n_long}")
    
    # === Phase 5: Back to best length, more LP polishing ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_best = len(best_seq)
        print(f"\n=== Phase 7: LP polish at n={n_best} ===")
        t7 = min(remaining * 0.20, 200.0)
        if t7 > 20:
            a_out, _ = nesterov_lp_optimize(best_seq, t7, solver, max_active=300)
            try_update(a_out, f"lp_polish_n{n_best}")
    
    # === Phase 6: Subgradient MD for further refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Phase 8: Subgradient MD polish ===")
        t8 = min(remaining * 0.15, 100.0)
        if t8 > 10:
            a_out, _ = subgradient_mirror_descent(best_seq, t8, eta_init=0.01, eta_decay=0.9999)
            try_update(a_out, f"sgd_polish")
    
    # === Phase 7: Peak shaving (final polish) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        print(f"\n=== Phase 9: Peak shaving ===")
        t9 = min(remaining * 0.30, 120.0)
        if t9 > 5:
            a_out, _ = peak_shaving_cd(best_seq, t9, seed=456)
            try_update(a_out, "peak_shave_1")
    
    # === Phase 8: One more round of LSE + peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        print(f"\n=== Phase 10: Final LSE polish ===")
        t10 = min(remaining * 0.25, 80.0)
        if t10 > 8:
            a_out, _ = lse_gradient_optimize(best_seq, t10, T_start=5.0, T_end=0.2, n_stages=8)
            try_update(a_out, "final_lse")
    
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        print(f"\n=== Phase 11: Final peak shaving ===")
        t11 = min(remaining * 0.40, 60.0)
        if t11 > 5:
            a_out, _ = peak_shaving_cd(best_seq, t11, seed=789)
            try_update(a_out, "peak_shave_final")
    
    # === Phase 9: Try nearby lengths for the best solution ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30 and best_seq is not None:
        n_cur = len(best_seq)
        for delta in [-4, -2, 2, 4, 8]:
            if remaining < 15:
                break
            n_try = n_cur + delta
            if n_try < 100:
                continue
            print(f"\n=== Try n={n_try} ===")
            init_try = interpolate_sequence(best_seq, n_try)
            t_try = min(remaining * 0.15, 60.0)
            a_out, _ = nesterov_lp_optimize(init_try, t_try, solver, max_active=200)
            try_update(a_out, f"nearby_n{n_try}")
            remaining = time_budget - (time.monotonic() - t_global)
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            if abs(fs - best_score) > 0.01:
                print(f"  Warning: eval mismatch: computed={best_score:.6f}, actual={fs:.6f}")
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
    
    return [float(x) for x in best_seq]
