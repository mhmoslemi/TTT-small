import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix_sparse(a):
    """Build sparse matrix T (2n-1 x n) such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    total = n * n
    rows = np.empty(total, dtype=np.int32)
    cols = np.empty(total, dtype=np.int32)
    vals = np.empty(total, dtype=np.float64)
    idx = 0
    for k in range(size):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        c = je - js + 1
        rows[idx:idx + c] = k
        cols[idx:idx + c] = np.arange(js, je + 1)
        vals[idx:idx + c] = a[k - np.arange(js, je + 1)]
        idx += c
    return csr_matrix((vals, (rows, cols)), shape=(size, n))


def solve_lp_best_response_hg(a):
    """Solve LP: min M s.t. T@b <= M, sum(b)=1, b>=0 using HiGHS."""
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix_sparse(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    # A_ub: [T | -I_slice] such that T@b - M <= 0
    # We need to add M column: A_ub = [T, -ones(size,1)]
    from scipy.sparse import eye as speye, hstack as sphstack, vstack as spvstack
    M_col = -csr_matrix(np.ones((size, 1)))
    A_ub = sphstack([T, M_col], format='csr')
    b_ub = np.zeros(size)

    # Equality: sum(b) = 1
    A_eq = np.hstack([np.ones((1, n)), np.zeros((1, 1))])
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True})
        if result.success:
            b_val = np.maximum(result.x[:n], 0)
            s = b_val.sum()
            if s > 1e-10:
                b_val = b_val / s
            return b_val, float(result.x[-1])

def solve_lp_best_response_cvx(a, solver_name):

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):

def logsumexp_obj_grad(z, n, T_temp):
    """Compute LSE objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T_temp
    scaled_shifted = scaled - scaled.max()
    exp_terms = np.exp(scaled_shifted)
    lse = max_b + T_temp * (scaled.max() + np.log(exp_terms.sum() + 1e-300))
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z

            continue
        else:
            lp_fail_count = 0

        n_iter += 1
        alpha = 2.0 / (it + 2)

        y_old = y.copy()
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = y_old
        else:
            y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 50 == 0:
            if best_score < prev_best - 1e-12:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 10:
                # Perturb to escape local optimum
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 6)
                phase = rng.uniform(0, 2 * np.pi)
                noise = 0.01 * np.cos(freq * x + phase)
                y = np.maximum(y + noise, 0)
                sy = y.sum()
                if sy > 1e-10:
                    y = y / sy
                stall_count = 0

            if (it + 1) % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def multiplicative_weights_refine(a0, time_limit, n_steps=500000, eta0=0.03):
    """Fast multiplicative weights with backtracking."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * (1.0 - 5e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        w_active = b[active] / b[active].sum()

        g = np.zeros(n)
        for w, k in zip(w_active, active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += w * a[k - idx]

        log_mult = -eta * g
        log_mult -= log_mult.max()
        a_new = a * np.exp(log_mult)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.02, 0.15)
        else:
            eta *= 0.9
            if eta < 1e-9:
                break

    return best_a, best_score


def lbgfs_polish(a0, time_limit, T_stages=None):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    if T_stages is None:
        T_stages = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]

    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def obj_func(z_val):
            if time.monotonic() > deadline:
                return 1e15
            o, _ = logsumexp_obj_grad(z_val, n, stage_T)
            return o

        def obj_grad(z_val):
            if time.monotonic() > deadline:
                return np.zeros(len(z_val))
            _, g = logsumexp_obj_grad(z_val, n, stage_T)
            return g

        maxiter = max(50, int(remaining * 30 / max(1, len(T_stages.index(stage_T)) + 1)))
        maxiter = min(maxiter, 1000)
        try:
            res = minimize(fun=obj_func, x0=z, jac=obj_grad, method='L-BFGS-B',
                           options={'maxiter': maxiter, 'ftol': 1e-20,
                                    'gtol': 1e-15, 'disp': False})
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def structured_init(n, rng, style=None):
    """Create diverse structured initial sequences."""
    if style is None:
        style = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if style == 0:
        a = 0.5 * (1 + np.cos(2 * np.pi * x))  # Hann window
    elif style == 1:
        a = np.cos(np.pi * x - np.pi / 2) ** 2
    elif style == 2:
        a = np.exp(-4 * x * (1 - x))
    elif style == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif style == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif style == 5:
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    elif style == 6:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 5 * x + 0.5)
        a = np.maximum(a, 0)
    elif style == 7:
        a = 1.0 / (1.0 + 5.0 * np.abs(x - 0.5))
    elif style == 8:
        a = np.exp(-2 * (x - 0.2) ** 2) + 0.3 * np.exp(-2 * (x - 0.8) ** 2) + 0.1
    else:
        a = np.ones(n) + 0.2 * rng.standard_normal(n)
        a = np.maximum(a, 0)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def make_palindromic(a):
    """Make a sequence palindromic."""
    n = len(a)
    a_pal = (a + a[::-1]) / 2.0
    a_pal = np.maximum(a_pal, 0)
    s = a_pal.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a_pal / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses LP alternating (HiGHS), MW, LSE L-BFGS, multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng, style=0)  # Hann
        best_score = evaluate_local(best_seq)
        print(f"Init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 0: Quick MW to warm up ===
    print(f"\n=== Phase 0: MW warmup n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t0_mw = min(rem * 0.03, 40.0)
    if t0_mw > 5:
        a_mw, _ = multiplicative_weights_refine(best_seq, t0_mw, n_steps=200000, eta0=0.03)
        try_update(a_mw, "mw_warmup")

    # === Phase 1: Main LP alternating ===
    print(f"\n=== Phase 1: LP alternating n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.55, 800.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        a_lp, sc_lp = nesterov_lp_refine(best_seq, t1)
        try_update(a_lp, "lp_main")

    # === Phase 2: Palindromic LP ===
    print(f"\n=== Phase 2: Palindromic LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t2 = min(rem * 0.15, 200.0)
        a_pal = make_palindromic(best_seq)
        a_pal_lp, _ = nesterov_lp_refine(a_pal, t2)
        try_update(a_pal_lp, "pal_lp")

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    lengths_to_try = [1000, 1500]
    for ln in lengths_to_try:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        if len(best_seq) == ln:
            continue
        t_ln = min(rem * 0.2, 180.0)
        if t_ln < 25:
            continue
        print(f"  Trying n={ln}, budget={t_ln:.0f}s")
        init_ln = interpolate_sequence(best_seq, ln)
        a_ln_mw, _ = multiplicative_weights_refine(init_ln, t_ln * 0.2, n_steps=100000, eta0=0.03)
        try_update(a_ln_mw, f"mw_n{ln}")
        rem = deadline - time.monotonic()
        t_lp = min(rem * 0.5, t_ln * 0.7)
        if t_lp > 20:
            a_ln_lp, _ = nesterov_lp_refine(a_ln_mw if a_ln_mw is not None else init_ln, t_lp)
            try_update(a_ln_lp, f"lp_n{ln}")

    # === Phase 4: LSE L-BFGS polishing ===
    print(f"\n=== Phase 4: LSE L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t4 = min(rem * 0.25, 350.0)
        print(f"  Budget: {t4:.0f}s")
        T_stages = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
        a_lse = lbgfs_polish(best_seq, t4, T_stages)
        try_update(a_lse, "lse_fine")

    # === Phase 5: Perturbation escape + re-optimize ===
    print(f"\n=== Phase 5: Perturbation (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        for trial in range(3):
            rem = deadline - time.monotonic()
            if rem < 25:
                break
            n = len(best_seq)
            x = np.linspace(0, 2 * np.pi, n)
            freq = rng.integers(1, 8)
            phase = rng.uniform(0, 2 * np.pi)
            strength = 0.01 + 0.01 * trial
            pert = strength * np.cos(freq * x + phase)
            a_pert = np.maximum(best_seq + pert, 0)
            sp = a_pert.sum()
            if sp > 1e-10:
                a_pert = a_pert / sp
            t_sub = min(rem * 0.25, 50.0)
            a_opt, _ = nesterov_lp_refine(a_pert, t_sub)
            try_update(a_opt, f"pert{trial}")

    # === Phase 6: Final LP polish ===
    print(f"\n=== Phase 6: Final LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t6 = min(rem * 0.4, 100.0)
        a_fin, _ = nesterov_lp_refine(best_seq, t6)
        try_update(a_fin, "final_lp")

    # === Phase 7: Final MW polish ===
    print(f"\n=== Phase 7: Final MW (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 10:
        t7 = min(rem * 0.5, 30.0)
        a_fin_mw, _ = multiplicative_weights_refine(best_seq, t7, n_steps=200000, eta0=0.01)
        try_update(a_fin_mw, "final_mw")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} n={len(best_seq)}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
