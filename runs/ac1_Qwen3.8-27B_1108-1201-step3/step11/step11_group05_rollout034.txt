from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v, exclude=None):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(1 + u * np.arange(1, n + 1) - cssv > 0)
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def subgradient_at(a, conv_result, frac=0.01):
    """Compute subgradient of max(conv) at point a, using active set."""
    n = len(a)
    max_b = conv_result.max()
    threshold = max_b * (1.0 - 1e-8)
    active = np.where(conv_result >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(conv_result)])

    # = (1/|active|) * correlate(w, a)[:n] where w = indicator of active
    w = np.zeros(2 * n - 1)
    w[active] = 1.0
    g = np.correlate(w, a, 'full')[:n]
    g /= len(active)
    return g

        a = a / s
    
    y = a.copy()

    t0 = time.monotonic()
    t = 0

    z = np.zeros(n)  # Accumulated subgradients
    eta = 1.0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        # Nesterov interpolation for y

            alpha = 2.0 / (t + 1.0)
            y = (1.0 - alpha) * y + alpha * a

            y = a.copy()

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated mirror descent (multiplicative weights) with FFT
2. LP best-response via cvxpy PDLP (periodic, for precision)
3. Coordinate descent polish (exact 1D optimization per coordinate)
4. Multi-length exploration (n=1200 primary, n=1500/2000 secondary)
5. Adaptive time management

import numpy as np
import time
from scipy.fft import rfft, irfft
import math

# Will use cvxpy for LP
import cvxpy as cp

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Project onto probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.max(np.where(1 + u * np.arange(1, n+1) - cssv > 0)[0])
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def compute_subgradient(a, conv_result, active_frac=0.005):
    """Compute subgradient of max(conv) using active set."""
    n = len(a)
    max_b = conv_result.max()
    if max_b <= 0:
        return np.zeros(n)
    # Use top fraction of peaks
    k = max(1, int(active_frac * (2*n-1)))
    idx = np.argpartition(conv_result, -k)[-k:]
    w = np.zeros(2*n-1)
    w[idx] = 1.0
    g = np.correlate(w, a, 'full')[:n]
    g /= k
    return g

def mirror_descent_step(a, y, g, eta):
    """Mirror descent (multiplicative weights) step."""
    log_a = np.log(np.maximum(y, 1e-300)) - eta * g
    log_a -= log_a.max()
    a_new = np.exp(log_a)
    s = a_new.sum()
    if s < 1e-300:
        return y.copy()
    return a_new / s

def nesterov_mirror_descent(a0, time_limit, eta_0=1.0, decay_rate=0.0, verbose=True):
    """Nesterov-accelerated mirror descent for max-conv minimization."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(n)/n
    a_prev = a.copy()
    best_a = a.copy()
    best_f = fast_convolve(a).max()
    t0 = time.monotonic()
    t = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        # Nesterov extrapolation
        if t <= 2:
            y = a.copy()
        else:
            beta = (t-1) / (t+1)
            y = a + beta * (a - a_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y = y / sy
        
        # Subgradient at y
        conv_y = fast_convolve(y)
        g = compute_subgradient(y, conv_y)
        
        # Step size
        eta = eta_0 / (1.0 + 0.1 * t * decay_rate)
        
        # Mirror descent step
        a_new = mirror_descent_step(a, y, g, eta)
        a_prev = a
        a = a_new
        
        # Track best
        f_a = fast_convolve(a).max()
        if f_a < best_f:
            best_f = f_a
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive restart
        if stall > 10000:
            # Restart with smaller eta
            eta_0 *= 0.5
            a_prev = a.copy()
            stall = 0
        
        if verbose and t % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  NMD t={t} ({elapsed:.0f}s): best={best_f:.8f} eta_0={eta_0:.4f}")
    
    return best_a, best_f

def solve_lp_best_response(a_fixed, M, solver=None):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a_fixed)
    size = 2 * n - 1
    # Build constraint matrix A (size x n) where A[k,j] = a[k-j]
    # This is a Toeplitz-like matrix
    A = np.zeros((size, n))
    for j in range(n):
        A[j:j+n, j] = a_fixed
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [A @ b_var <= M * np.ones(size)]
    objective = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(objective, constraints)
    
    solvers = [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS, cp.GLOP, cp.CBC]
    if solver:
        solvers = [solver] + [s for s in solvers if s != solver]
    
    for s in solvers:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                b_val[np.isnan(b_val)] = 0
                sv = b_val.sum()
                if sv > 1e-15:
                    return b_val / sv
                return None
        except Exception:
            continue
    return None

def lp_refine(a0, time_limit, n_max_iters=100000, verbose=True):

    n = len(a0)

        it += 1
        M = fast_convolve(y).max()
        if M < best_f:

            best_a = y.copy()
        
        lp_t0 = time.monotonic()
        b = solve_lp_best_response(y, M)
        lp_times.append(time.monotonic() - lp_t0)
        
        if b is None:
            continue
        
        # Nesterov update
        alpha = 2.0 / (it + 2)
        y_old = y.copy()
        y = (1 - alpha) * y + alpha * b
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        
        # Also update a with stronger weight on b
        a = (it * a + alpha * (it+2) * b) / (it + alpha * (it+2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:

        if f_a < best_f:
            best_f = f_a
            best_a = a.copy()
            stall = 0
        elif f_y < best_f:
            best_f = f_y
            best_a = y.copy()
            stall = 0
        else:

            y = y + 0.01 * rng.standard_normal(n)
            y = np.maximum(y, 0)
            y = y / y.sum()
            stall = 0

1. Recompute conv after each coordinate update (expensive: O(n log n) per coordinate, O(n² log n) per sweep)

This is O(n) per update. So one sweep is O(n²) total, which for n=1200 is about 1.4M operations per sweep. In Python, this might take ~0.1-0.5s per sweep.
