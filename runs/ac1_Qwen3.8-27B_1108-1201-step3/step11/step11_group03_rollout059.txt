    n = len(a)
    size = 2 * n - 1

    return irfft(A * A, fft_size)[:size]

    """Build sparse Q matrix for conv(a,a)[k] = a^T Q a."""
    from scipy.sparse import lil_matrix, csc_matrix
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)


def solve_qcqp(active_set, n, solver):

    Solve: min t s.t. a^T Q_k a <= t for k in active_set,
           sum(a) = 1, a >= 0

    constraints = [cp.sum(a_var) == 1]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using cutting-plane QCQP.

The problem is convex: minimize max of (2n-1) PSD quadratic forms over the
simplex {a >= 0, sum(a) = 1}. The cutting-plane method iteratively solves
reduced QCQPs with the active set of constraints, adding the most violated
constraint at each step. This converges to the global optimum in a small
number of iterations (typically 3-15).

Key difference from previous approach: instead of relying on LP alternating
(slow convergence), we directly solve the convex QCQP to optimality using
a cutting-plane scheme. This is exact for convex problems and much faster
when the active set is small (which it is for this problem).
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.sparse import lil_matrix, csc_matrix
import itertools


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_q_sparse(k, n):
    """Build sparse Q matrix such that a^T Q a = conv(a,a)[k]."""
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)


def solve_qcqp_active(active_set, n, solver_hint=None):
    """Solve min t s.t. a^T Q_k a <= t for k in active_set, sum(a)=1, a>=0."""
    a_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [cp.sum(a_var) == 1]

    for k in sorted(active_set):
        Q = build_q_sparse(int(k), n)
        constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)

    prob = cp.Problem(cp.Minimize(t_var), constraints)

    solvers_to_try = []
    if solver_hint:
        solvers_to_try.append(solver_hint)
    for s in [cp.MOSEK, cp.GUROBI, cp.SCIP, cp.CLARABEL, cp.XPRESS, cp.ECOS, cp.SCS, cp.PDLP, cp.CBC, cp.CVXOPT]:
        if s not in solvers_to_try and s is not None:
            solvers_to_try.append(s)

    best_result = None
    for s in solvers_to_try:
        try:
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.asarray(a_var.value, dtype=np.float64)
                a_val = np.maximum(a_val, 0)
                s_sum = a_val.sum()
                if s_sum < 1e-15:
                    continue
                a_val = a_val / s_sum
                t_val = float(t_var.value)
                if best_result is None or t_val < best_result[1]:
                    best_result = (a_val, t_val)
        except Exception:
            continue

    return best_result


def cutting_plane_optimize(a_init, n, time_limit, solver_hint=None, verbose=True):
    """
    Cutting-plane method for the convex QCQP:
    min max_k (a*a)[k] s.t. a >= 0, sum(a) = 1.
    
    Returns (optimal_a, optimal_max_b).
    """
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    b = fast_convolve(a)
    max_b = float(b.max())

    # Initial active set: top constraints
    threshold = max_b * (1.0 - 2e-3)
    active = sorted(int(k) for k in np.where(b >= threshold)[0])
    if not active:
        active = [int(np.argmax(b))]

    best_a = a.copy()
    best_max_b = max_b
    t0 = time.monotonic()

    for iteration in range(200):
        if time.monotonic() - t0 > time_limit:
            break

        iter_t0 = time.monotonic()
        result = solve_qcqp_active(active, n, solver_hint)
        iter_time = time.monotonic() - iter_t0

        if result is None:
            if verbose:
                print(f"    QCQP failed at iter {iteration+1} (|A|={len(active)})")
            break

        a_sol, t_sol = result
        b_sol = fast_convolve(a_sol)
        max_violation = float(b_sol.max())

        if max_violation < best_max_b:
            best_max_b = max_violation
            best_a = a_sol.copy()

        # Convergence: no constraint is violated
        if max_violation <= t_sol + 1e-10:
            if verbose:
                print(f"    Converged in {iteration+1} iters: max_b={best_max_b:.12f}")
            break

        # Add most violated constraint
        k_violated = int(np.argmax(b_sol))
        if k_violated not in active:
            active.append(k_violated)
        else:
            # Already in set; try second most violated
            b_sorted = np.argsort(b_sol)[::-1]
            added = False
            for k in b_sorted:
                k = int(k)
                if k not in active and b_sol[k] > t_sol + 1e-12:
                    active.append(k)
                    added = True
                    break
            if not added:
                if verbose:
                    print(f"    No new constraint to add at iter {iteration+1}, stopping")
                break

        # Limit active set size (keep most violated)
        if len(active) > 20:
            active = sorted(active, key=lambda k: -b_sol[k])[:15]

        if verbose and (iteration + 1) % 3 == 0:
            print(f"    Iter {iteration+1}: max_b={max_violation:.10f}, t_qp={t_sol:.10f}, "
                  f"|A|={len(active)}, time={iter_time:.1f}s, total={time.monotonic()-t0:.1f}s")

    return best_a, best_max_b

                init2 = a_opt.copy()
                t2 = min(rem2 * 0.15, 60.0)
                a_opt2, max_b2 = cutting_plane_optimize(init2, n_try, t2, verbose=False)
                if a_opt2 is not None:
                    sc2 = local_score(a_opt2)
                    if sc2 < best_score:
                        update_best(a_opt2, f"cp2_n{n_try}")
                        results[n_try] = (a_opt2, sc2)

    # Phase 3: Try a few random perturbations of the best solution
    print(f"\n=== Phase 3: Perturbation Search ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        rng = np.random.default_rng(seed)
        n_best = len(best_seq)
        t_per_perturb = min(rem * 0.2, 40.0)

        for pert in range(5):
            if time.monotonic() > deadline - 10:
                break
            if t_per_perturb < 10:
                break

            noise = 0.05 * rng.standard_normal(n_best)
            a_pert = best_seq + noise
            a_pert = np.maximum(a_pert, 0)
            sp = a_pert.sum()
            if sp < 1e-15:
                continue
            a_pert = a_pert / sp

            a_opt, _ = cutting_plane_optimize(a_pert, n_best, t_per_pert, verbose=False)
            if a_opt is not None:
                update_best(a_opt, f"perturb{pert}_n{n_best}")

    # Phase 4: Fine n-search around the best
    print(f"\n=== Phase 4: Fine n-Search ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_best = len(best_seq)
        for n_fine in range(max(100, n_best - 5), min(2000, n_best + 6)):
            if time.monotonic() > deadline - 20:
                break
            if n_fine in results:
                continue
            rem2 = deadline - time.monotonic()
            if rem2 < 20:
                break
            init = interpolate_sequence(best_seq, n_fine)
            t_fine = min(rem2 * 0.3, 45.0)
            a_opt, _ = cutting_plane_optimize(init, n_fine, t_fine, verbose=False)
            if a_opt is not None:
                sc = local_score(a_opt)
                if sc < best_score:
                    update_best(a_opt, f"fine_n{n_fine}")
                results[n_fine] = (a_opt, sc)

    # Final
    final_score = local_score(best_seq)
    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {final_score:.10f} (n={len(best_seq)})")
    print(f"{'='*60}")

    # Print all results
    for n_key in sorted(results.keys()):
        a_k, sc_k = results[n_key]
        marker = " <== BEST" if sc_k <= best_score + 1e-12 else ""
        print(f"  n={n_key}: score={sc_k:.10f}{marker}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
