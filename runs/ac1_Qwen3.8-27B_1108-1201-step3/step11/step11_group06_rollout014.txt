import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix_sparse(a):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    rows = np.zeros(n * n, dtype=np.int32)
    cols = np.zeros(n * n, dtype=np.int32)
    vals = np.zeros(n * n, dtype=np.float64)
    count = 0
    for j in range(n):
        k_start = j
        k_end = min(j + n, size)
        for k in range(k_start, k_end):
            rows[count] = k
            cols[count] = j
            vals[count] = a[k - j]
            count += 1
    return csr_matrix((vals[:count], (rows[:count], cols[:count])), shape=(size, n))


def build_conv_matrix_sparse_fast(a):
    """Build sparse convolution matrix using vectorized approach."""
    n = len(a)
    size = 2 * n - 1
    # For each column j, nonzeros are at rows j, j+1, ..., j+n-1 (clipped to [0, size-1])
    rows_list = []
    cols_list = []
    vals_list = []
    for j in range(n):
        r = np.arange(j, min(j + n, size))
        c = np.full(len(r), j, dtype=np.int32)
        v = a[r - j]
        rows_list.append(r)
        cols_list.append(c)
        vals_list.append(v)
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    vals = np.concatenate(vals_list)
    return csr_matrix((vals, (rows.astype(np.int32), cols)), shape=(size, n))


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x[0] <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def solve_lp_best_response(a_fixed, solver, T_mat):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = [solver, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
    seen = set()
    for s in solvers_to_try:
        if s is None or s in seen:
            continue
        seen.add(s)
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_val = b_val.sum()
                if s_val < 1e-15:
                    return None, None
                b_val = b_val / s_val
                return b_val, float(max(0, M_var.value))
        except Exception:
            continue
    return None, None


def evaluate_local(a):

    if s < 0.01:

def multiplicative_weights_adaptive(a0, time_limit, n_steps, eta0=0.05):
    """Fast multiplicative weights with adaptive backtracking step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * (1.0 - 5e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        w_active = b[active] / b[active].sum()
        g = np.zeros(n)
        for w, k in zip(w_active, active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += w * a[k - idx]
        log_mult = -eta * g
        log_mult -= log_mult.max()
        a_new = a * np.exp(log_mult)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.02, 0.3)
        else:
            eta *= 0.85
            if eta < 1e-9:
                break
        if step % 50000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    MW {elapsed:.1f}s: score={best_score:.8f} eta={eta:.6f}")
    return best_a, best_score

                break

def _lse_objective_and_grad(z, n, T_temp, deadline):
    """Compute LSE objective and gradient w.r.t. z."""
    if time.monotonic() > deadline:
        return 1e15, np.zeros(len(z))
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        a = np.ones(n) / n
    else:
        a = exp_z / s
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T_temp
    scaled_shifted = scaled - scaled.max()
    exp_terms = np.exp(scaled_shifted)
    lse = max_b + T_temp * (scaled.max() + np.log(exp_terms.sum() + 1e-300))
    obj = 2.0 * n * lse
    w = exp_terms / (exp_terms.sum() + 1e-300)
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def lse_lbfgs_refine(a0, time_limit, T_stages, maxiter_per_stage=500):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()
    deadline = t0 + time_limit

    class LSEFunc:
        def __init__(self, n, T, deadline):
            self.n = n
            self.T = T
            self.deadline = deadline

        def __call__(self, z):
            if time.monotonic() > self.deadline:
                return 1e15
            obj, _ = _lse_objective_and_grad(z, self.n, self.T, self.deadline)
            return obj

    class LSEJac:
        def __init__(self, n, T, deadline):
            self.n = n
            self.T = T
            self.deadline = deadline

        def __call__(self, z):
            if time.monotonic() > self.deadline:
                return np.zeros(len(z))
            _, g = _lse_objective_and_grad(z, self.n, self.T, self.deadline)
            return g

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        func = LSEFunc(n, stage_T, deadline)
        jac = LSEJac(n, stage_T, deadline)
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-28,
                         'gtol': 1e-20, 'maxfun': maxiter_per_stage * 200,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(n) / n
    return exp_z / s


def subgradient_adagrad(a0, time_limit, n_steps=500000):
    """Subgradient descent with AdaGrad on the smoothed max."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    G = np.zeros(n)  # AdaGrad accumulator
    eta0 = 0.1
    eps = 1e-8

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        # Soft active set
        active_thresh = max_b * (1.0 - 1e-3)
        active = np.where(b >= active_thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        w_active = b[active] / b[active].sum()
        g = np.zeros(n)
        for w, k in zip(w_active, active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += w * a[k - idx]
        G += g * g
        a_new = a - eta0 * g / np.sqrt(G + eps)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
        else:
            eta0 *= 0.999
            if eta0 < 1e-10:
                break
        if step % 50000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    SG {elapsed:.1f}s: score={best_score:.8f} eta={eta0:.8f}")
    return best_a, best_score


def structured_init(n, rng):

        return np.ones(n) / n
    return a / s_val


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def perturb_sequence(a, rng, strength=0.03):
    """Create a perturbed version with structured frequency components."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq1 = rng.integers(1, 8)
    freq2 = rng.integers(2, 12)
    p1 = rng.uniform(0, 2 * np.pi)
    p2 = rng.uniform(0, 2 * np.pi)
    pert = strength * (np.cos(freq1 * x + p1) + 0.5 * np.sin(freq2 * x + p2))
    a_pert = np.maximum(a + pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Key strategies:
    - LP alternating with CLARABEL and Nesterov acceleration (main workhorse)
    - Multi-length exploration (n=1000, 1200, 2000)
    - LSE L-BFGS with ultra-fine temperature cooling for final polish
    - Subgradient descent with AdaGrad as diversity escape
    - Structured diverse restarts
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}, budget: {budget_s:.0f}s")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s_val = a.sum()
        if s_val < 0.01:
            return
        a = a / s_val
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Quick MW warmup ===
    print(f"\n=== Phase 1: MW n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.04, 60.0)
    if t1 > 10:
        a_mw, _ = multiplicative_weights_adaptive(best_seq, t1, n_steps=300000, eta0=0.04)
        try_update(a_mw, "mw_1")

    # === Phase 2: LP alternating on parent length (main workhorse) ===
    print(f"\n=== Phase 2: LP n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.35, 550.0)
    if t2 > 60:
        print(f"  Budget: {t2:.0f}s")
        a_lp, _ = lp_alternating_nesterov(best_seq, t2, solver, n_max_iters=1000000)
        try_update(a_lp, "lp_1")

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length (score={best_score:.8f}) ===")
    lengths_to_try = [1000, 2000]
    for ln in lengths_to_try:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        if len(best_seq) == ln:
            continue
        t_ln = min(rem * 0.18, 250.0)
        if t_ln < 40:
            continue
        print(f"  Trying n={ln}, budget={t_ln:.0f}s")
        init_ln = interpolate_sequence(best_seq, ln)
        # Quick MW warmup on new length
        a_ln_mw, _ = multiplicative_weights_adaptive(init_ln, min(t_ln * 0.15, 30.0), n_steps=100000, eta0=0.04)
        try_update(a_ln_mw, f"mw_n{ln}")
        # LP on new length
        rem = deadline - time.monotonic()
        t_lp = min(rem * 0.4, t_ln * 0.75)
        if t_lp > 30:
            a_ln_lp, _ = lp_alternating_nesterov(
                a_ln_mw if a_ln_mw is not None else init_ln,
                t_lp, solver, n_max_iters=500000
            )
            try_update(a_ln_lp, f"lp_n{ln}")

    # === Phase 4: Subgradient + AdaGrad on best ===
    print(f"\n=== Phase 4: Subgradient n={len(best_seq)} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t4 = min(rem * 0.12, 150.0)
        a_sg, _ = subgradient_adagrad(best_seq, t4, n_steps=300000)
        try_update(a_sg, "sg")

    # === Phase 5: LSE L-BFGS fine polishing ===
    print(f"\n=== Phase 5: LSE L-BFGS n={len(best_seq)} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t5 = min(rem * 0.25, 400.0)
        T_stages = [20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
        print(f"  {len(T_stages)} stages, budget={t5:.0f}s")
        a_lse = lse_lbfgs_refine(best_seq, t5, T_stages, maxiter_per_stage=600)
        try_update(a_lse, "lse_fine")

    # === Phase 6: Perturbation escape + LP re-optimization ===
    print(f"\n=== Phase 6: Perturbation escape (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        for trial in range(5):
            rem = deadline - time.monotonic()
            if rem < 25:
                break
            strengths = [0.01, 0.02, 0.03, 0.015, 0.025]
            a_pert = perturb_sequence(best_seq, rng, strength=strengths[trial])
            t_sub = min(rem * 0.2, 50.0)
            a_opt, _ = lp_alternating_nesterov(a_pert, t_sub, solver, n_max_iters=200000)
            try_update(a_opt, f"pert{trial}")

    # === Phase 7: Final LP polish on best ===
    print(f"\n=== Phase 7: Final LP n={len(best_seq)} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t7 = min(rem * 0.35, 180.0)
        a_fin, _ = lp_alternating_nesterov(best_seq, t7, solver, n_max_iters=500000)
        try_update(a_fin, "final_lp")

    # === Phase 8: Very fine LSE on best ===
    print(f"\n=== Phase 8: Ultra-fine LSE (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t8 = min(rem * 0.5, 100.0)
        T_stages_fine = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
        a_ultra = lse_lbfgs_refine(best_seq, t8, T_stages_fine, maxiter_per_stage=1000)
        try_update(a_ultra, "ultra_lse")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} n={len(best_seq)}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
