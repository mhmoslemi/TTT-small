Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multiplicative Weights (MW) minimax algorithm with FFT-accelerated QP subproblems
2. Palindromic symmetry exploitation (halves effective dimension)

4. LP-based alternating polish for final refinement
5. Multi-length exploration with adaptive time allocation
Key insight: The problem is a minimax game min_a max_k (a*a)[k]. We use MW on the
dual variable (distribution over k), solving the QP subproblem via projected gradient
with FFT-based matrix-vector products for the Hankel matrix Q_p.

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def hankel_matvec_fft(p, a, fft_size, q_fft):
    """Compute Q_p @ a where Q_p[i,j] = p[i+j]. Uses precomputed FFT of reversed p."""
    n = len(a)
    conv_full = irfft(np.asarray(a, dtype=np.float64).astype(np.float64) * 0 + rfft(np.asarray(a, dtype=np.float64), fft_size) * q_fft, fft_size)
    conv = conv_full[:3 * n - 1]
    return conv[n - 1:2 * n - 1][::-1]


def hankel_matvec(p, a):
    """Compute Q_p @ a where Q_p[i,j] = p[i+j]. p has length 2n-1, a has length n."""
    n = len(a)
    q = p[::-1]
    size = 3 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    Q = rfft(np.asarray(q, dtype=np.float64), fft_size)
    conv_full = irfft(A * Q, fft_size)
    conv = conv_full[:size]
    return conv[n - 1:2 * n - 1][::-1]


def hankel_matvec_precomputed(a, q_fft, fft_size):
    """Compute Q_p @ a using precomputed FFT of reversed p."""
    n = len(a)
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    conv_full = irfft(A * q_fft, fft_size)
    size = 3 * n - 1
    conv = conv_full[:size]
    return conv[n - 1:2 * n - 1][::-1]


def project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target} using the efficient algorithm."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    if target <= 0:
        return np.zeros(n)
    # Scale to standard simplex projection
    u = v / max(target, 1e-300)
    mu = np.sort(u)[::-1]
    css = np.cumsum(mu)
    rho = np.nonzero(u * np.arange(1, n + 1) > (css - 1))[0][-1]
    theta = (css[rho] - 1) / (rho + 1.0)
    result = np.maximum(u - theta, 0) * target
    return result

    """Convert half-sequence (length n/2) to full palindromic sequence (length n). n must be even."""
    m = n // 2
    a = np.empty(n)
    a[:m] = h
    a[m:] = h[::-1]
    return a

    m = n // 2

def evaluate_local(a):
    """Fast local evaluation of 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def qpg_solve_palindromic(p, h_init, n, n_iters=50, initial_step=0.1):
    """
    Solve min a(h)^T Q_p a(h) s.t. a palindromic, sum(a)=1, a>=0.
    Uses projected gradient in half-sequence space.
    Returns optimal half-sequence h.
    """
    m = n // 2
    h = np.maximum(np.asarray(h_init, dtype=np.float64), 0).copy()
    s = h.sum()
    if s < 1e-15:
        h = np.full(m, 0.5 / m)
    else:
        h = h / s * 0.5

    # Precompute FFT of reversed p
    q = p[::-1]
    size = 3 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    q_fft = rfft(q, fft_size)

    a = half_to_full(h, n)
    obj_current = float(np.dot(a, hankel_matvec_precomputed(a, q_fft, fft_size)))

    step = initial_step
    for it in range(n_iters):
        # Compute gradient in h-space
        a = half_to_full(h, n)
        grad_a = hankel_matvec_precomputed(a, q_fft, fft_size)
        # grad_h[j] = 2*(grad_a[j] + grad_a[n-1-j])
        grad_h = 2.0 * (grad_a[:m] + grad_a[:m][::-1])

        # Backtracking line search
        for ls in range(10):
            h_new = project_simplex(h - step * grad_h, target=0.5)
            a_new = half_to_full(h_new, n)
            obj_new = float(np.dot(a_new, hankel_matvec_precomputed(a_new, q_fft, fft_size)))
            if obj_new <= obj_current + 1e-15:
                break
            step *= 0.5
        else:
            break

        if obj_new < obj_current:
            h = h_new
            obj_current = obj_new
            step = min(step * 1.1, 0.5)
        else:
            step *= 0.7

    return h, obj_current


def minimax_mw_palindromic(h0, n, time_budget, eta=0.003, n_pg_iters=40,
                           check_interval=200):
    """
    Minimax MW algorithm restricted to palindromic sequences.
    p is the dual variable (distribution over convolution indices 0..n-1 by symmetry).
    """
    m = n // 2
    deadline = time.monotonic() + time_budget

    # Initialize p uniformly over k=0..n-1 (by symmetry, we only need half)
    # Actually, use full p of length 2n-1, but only the first n matter for palindromic a
    p = np.ones(2 * n - 1) / (2 * n - 1)

    h = np.maximum(np.asarray(h0, dtype=np.float64), 0).copy()
    s = h.sum()
    if s < 1e-15:
        h = np.full(m, 0.5 / m)
    else:
        h = h / s * 0.5

    # Initialize a and compute initial convolution
    a = half_to_full(h, n)
    conv_a = fast_convolve(a)
    max_conv = conv_a.max()
    best_score = 2.0 * n * max_conv  # since sum(a)=1
    best_h = h.copy()

    # Running average
    h_avg = h.copy()
    count = 0

    # Concentrate p initially on the peaks
    peak_threshold = max_conv * 0.95
    p_init = np.where(np.convolve(a, a) >= peak_threshold, 5.0, 1.0)
    p_init = p_init / p_init.sum()
    p = p_init.copy()

    t0 = time.monotonic()
    total_iters = 0

    while time.monotonic() < deadline:
        # Solve QP subproblem
        h_new, _ = qpg_solve_palindromic(p, h, n, n_iters=n_pg_iters)
        h = h_new
        a = half_to_full(h, n)

        # Compute convolution
        conv_a = fast_convolve(a)
        max_conv = conv_a.max()
        score = 2.0 * n * max_conv

        # Update running average
        count += 1
        h_avg = (h_avg * (count - 1) + h) / count
        a_avg = half_to_full(h_avg, n)
        score_avg = 2.0 * n * fast_convolve(a_avg).max()

        # Track best
        if score < best_score:
            best_score = score
            best_h = h.copy()
        if score_avg < best_score:
            best_score = score_avg
            best_h = h_avg.copy()

        # Multiplicative weights update on p
        # Use full convolution values for the MW update
        log_mult = eta * conv_a
        log_mult -= log_mult.max()
        exp_mult = np.exp(log_mult)
        p = p * exp_mult
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            p = p / p_sum

        total_iters += 1

        if total_iters % check_interval == 0:
            elapsed = time.monotonic() - t0
            print(f"    MW iter={total_iters} ({elapsed:.0f}s): "
                  f"score={best_score:.8f}, current={score:.8f}, avg={score_avg:.8f}")

    return best_h, best_score


def build_palindromic_conv_matrix(a_full):
    """
    Build matrix T (n x m) such that T @ h gives (a * b_pal)[0:n] where b_pal = half_to_full(h).
    This is for the LP best response.
    """
    n = len(a_full)
    m = n // 2
    # (a * b_pal)[k] = sum_j a_full[j] * b_pal[k-j]
    # b_pal[i] = h[i] for i < m, h[n-1-i] for i >= m
    # So (a * b_pal)[k] = sum_{i=0}^{m-1} h[i] * (a_full[k-i] + a_full[k-(n-1-i)])
    #                   = sum_{i=0}^{m-1} h[i] * (a_full[k-i] + a_full[k-n+1+i])
    # (with out-of-range terms being 0)

    T = np.zeros((n, m))
    for i in range(m):
        for k in range(n):
            val = 0.0
            idx1 = k - i
            if 0 <= idx1 < n:
                val += a_full[idx1]
            idx2 = k - (n - 1 - i)
            if 0 <= idx2 < n:
                val += a_full[idx2]
            T[k, i] = val
    return T


def lp_best_response_palindromic(a_full, T_mat, solver=None):
    """Solve LP: min M s.t. T@h <= M, sum(h)=1/2, h>=0. Returns h (half of b)."""
    m = T_mat.shape[1]
    n = T_mat.shape[0]

    h_var = cp.Variable(m, nonneg=True)
    M_var = cp.Variable()
    constraints = [T_mat @ h_var <= M_var, cp.sum(h_var) == 0.5]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = []
    if solver is not None:
        solvers_to_try.append(solver)
    solvers_to_try.extend([cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS])

    solved = False
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solved = True
                break
        except Exception:
            continue

    if not solved:
        return None, None

    h_val = np.asarray(h_var.value, dtype=np.float64)
    h_val = np.maximum(h_val, 0)
    s_val = h_val.sum()
    if s_val < 1e-15:
        return None, None
    h_val = h_val / s_val * 0.5
    return h_val, float(max(0.0, M_var.value))


def lp_alternating_palindromic(h0, n, time_budget, solver=None, n_max_iters=50000):
    """LP alternating method restricted to palindromic sequences with Nesterov acceleration."""
    m = n // 2
    deadline = time.monotonic() + time_budget

    h = np.maximum(np.asarray(h0, dtype=np.float64), 0).copy()
    s = h.sum()
    if s < 1e-15:
        h = np.full(m, 0.5 / m)
    else:
        h = h / s * 0.5

    y = h.copy()  # Nesterov momentum variable
    h_avg = h.copy()
    count = 0

    a = half_to_full(h, n)
    conv_a = fast_convolve(a)
    best_score = 2.0 * n * conv_a.max()
    best_h = h.copy()

    t0 = time.monotonic()
    T_mat = build_palindromic_conv_matrix(a)
    prev_best = best_score
    stall_count = 0
    lp_times = []

    for it in range(n_max_iters):
        if time.monotonic() > deadline:
            break

        # Solve LP best response to y
        lp_start = time.monotonic()
        h_br, M_br = lp_best_response_palindromic(half_to_full(y, n), T_mat, solver)
        lp_times.append(time.monotonic() - lp_start)

        if h_br is None:
            continue

        # Check LP speed
        if len(lp_times) > 20:
            avg_lp = np.mean(lp_times[-20:])
            if avg_lp > 5.0:
                print(f"    LP too slow ({avg_lp:.2f}s/iter), stopping at iter {it+1}")
                break

        # Nesterov update
        alpha = 2.0 / (it + 2)
        y_old = y.copy()
        y = (1.0 - alpha) * y + alpha * h_br
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = y_old
            sy = y_old.sum()
        y = y / sy * 0.5

        h = ((count + 1) * h + alpha * (count + 2) * h_br) / ((count + 2))
        h = np.maximum(h, 0)
        sh = h.sum()
        if sh < 1e-15:
            h = y.copy()
        else:
            h = h / sh * 0.5

        count += 1
        h_avg = (h_avg * (count - 1) + h) / count

        # Rebuild T matrix
        y_full = half_to_full(y, n)
        T_mat = build_palindromic_conv_matrix(y_full)

        # Evaluate
        a = half_to_full(h, n)
        conv_a = fast_convolve(a)
        score_h = 2.0 * n * conv_a.max()

        a_avg = half_to_full(h_avg, n)
        score_avg = 2.0 * n * fast_convolve(a_avg).max()

        if score_h < best_score:
            best_score = score_h
            best_h = h.copy()
        if score_avg < best_score:
            best_score = score_avg
            best_h = h_avg.copy()

        if (it + 1) % 50 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 6:
                # Add small perturbation
                rng = np.random.default_rng(it)
                pert = 0.01 * rng.standard_normal(m)
                y = np.maximum(y + pert, 0)
                y = y / y.sum() * 0.5
                T_mat = build_palindromic_conv_matrix(half_to_full(y, n))
                stall_count = 0

            if (it + 1) % 200 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_h, best_score


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving palindromic structure."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    m_old = old_n // 2
    m_new = new_n // 2
    h_old = f[:m_old]
    s = h_old.sum()
    if s < 1e-15:
        h_new = np.full(m_new, 0.5 / m_new)
    else:
        old_x = np.linspace(0, 1, m_old)
        new_x = np.linspace(0, 1, m_new)
        h_new = np.interp(new_x, old_x, h_old)
        h_new = np.maximum(h_new, 0)
        sh = h_new.sum()
        if sh > 1e-15:
            h_new = h_new / sh * 0.5
    return half_to_full(h_new, new_n)


def structured_init_pal(n, rng):
    """Create diverse structured palindromic initial sequences."""
    m = n // 2
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, m)
    if choice == 0:
        h = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        h = np.sin(np.pi * x) ** 2
    elif choice == 2:
        h = np.exp(-3 * x * (1 - x))
    elif choice == 3:
        h = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        h = np.maximum(h, 0)
    elif choice == 4:
        h = (1 - x) ** 1.5 + 0.05
    else:
        h = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    h = np.maximum(h, 0)
    s = h.sum()
    if s < 1e-15:
        h = np.full(m, 0.5 / m)
    else:
        h = h / s * 0.5
    return half_to_full(h, n)


def lse_refine(a0, time_budget):
    """LogSumExp L-BFGS refinement on the full sequence."""
    from scipy.optimize import minimize as sp_minimize

    n = len(a0)
    deadline = time.monotonic() + time_budget

    # Work in log space for non-negativity
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-10)
    s = a.sum()
    a = a / s
    z = np.log(a)

    T_stages = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]

    best_a = a.copy()
    best_score = evaluate_local(a)

    for T in T_stages:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break

        def lse_obj(z_val):
            a_val = np.exp(z_val - z_val.max())
            a_val = a_val / a_val.sum()
            b = fast_convolve(a_val)
            max_b = b.max()
            scaled = (b - max_b) / T
            lse_val = max_b + T * np.log(np.exp(scaled - scaled.max()).sum())
            return 2.0 * n * lse_val

        def lse_grad(z_val):
            a_val = np.exp(z_val - z_val.max())
            a_val = a_val / a_val.sum()
            b = fast_convolve(a_val)
            max_b = b.max()
            scaled = (b - max_b) / T
            shifted = scaled - scaled.max()
            exp_terms = np.exp(shifted)
            w = exp_terms / exp_terms.sum()
            # Gradient of max_b + T*log(sum(exp((b-max_b)/T))) w.r.t. a
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-18:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += w[k] * a_val[k - idx]
            S = np.dot(grad_a, a_val)
            # d/dz = a * (grad_a - S) ... but we need the chain rule for log-sum-exp
            # Actually for the LSE objective: dLSE/da = (1/T) * (b - b_exp_secure)
            # where b_exp_secure[k] = sum_j a[j]*a[k-j] weighted...
            # Simpler: use the formula dLSE/da[j] = (1/T) * sum_k w[k] * (db[k]/da[j])
            # db[k]/da[j] = 2*a[k-j] (with range restriction)
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-18:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += (2.0 / T) * w[k] * a_val[k - idx]
            # Project gradient to log-space: dz -> da -> dz
            S2 = np.dot(grad_a, a_val)
            grad_z = a_val * (grad_a - S2)
            return 2.0 * n * grad_z

        try:
            res = sp_minimize(
                fun=lse_obj, x0=z, jac=lse_grad, method='L-BFGS-B',
                options={'maxiter': 500, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxfun': 2000, 'disp': False}
            )
            z = res.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            score = evaluate_local(a_new)
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
        except Exception:
            pass

    return best_a


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Minimax MW (main workhorse, palindromic, FFT-accelerated)
    2. LP alternating polish (palindromic)
    3. LSE L-BFGS refinement
    4. Multi-length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}, budget: {budget_s:.0f}s")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init_pal(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST: {sc:.8f} (n={len(a)})")

    # === Phase 1: Minimax MW on parent (main workhorse) ===
    print(f"\n=== Phase 1: Minimax MW n={n_best} (score={best_score:.8f}) ===")
    if n_best % 2 != 0:
        # Make even
        n_best = n_best + 1
        best_seq = interpolate_sequence(best_seq, n_best)
        best_score = evaluate_local(best_seq)

    m = n_best // 2
    h_parent = full_to_half(best_seq, n_best)

    rem = deadline - time.monotonic()
    t1 = min(rem * 0.45, 700.0)
    if t1 > 60:
        print(f"  Budget: {t1:.0f}s")
        # Run minimax MW
        h_mw, score_mw = minimax_mw_palindromic(
            h_parent, n_best, t1, eta=0.003, n_pg_iters=40, check_interval=300
        )
        a_mw = half_to_full(h_mw, n_best)
        try_update(a_mw)

        # Also try with different eta
        rem = deadline - time.monotonic()
        if rem > 60:
            t1b = min(rem * 0.5, 300.0)
            print(f"  MW phase 1b (eta=0.01): budget={t1b:.0f}s")
            h_mw2, score_mw2 = minimax_mw_palindromic(
                h_mw, n_best, t1b, eta=0.01, n_pg_iters=60, check_interval=200
            )
            a_mw2 = half_to_full(h_mw2, n_best)
            try_update(a_mw2)

    # === Phase 2: LP Alternating Polish ===
    print(f"\n=== Phase 2: LP Alternating (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 120:
        t2 = min(rem * 0.30, 400.0)
        print(f"  Budget: {t2:.0f}s")
        h_cur = full_to_half(best_seq, n_best)
        h_lp, score_lp = lp_alternating_palindromic(
            h_cur, n_best, t2, solver=solver, n_max_iters=50000
        )
        a_lp = half_to_full(h_lp, n_best)
        try_update(a_lp)

    # === Phase 3: LSE L-BFGS Refinement ===
    print(f"\n=== Phase 3: LSE L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        t3 = min(rem * 0.20, 250.0)
        print(f"  Budget: {t3:.0f}s")
        a_lse = lse_refine(best_seq, t3)
        # Make palindromic
        a_lse_pal = (a_lse + a_lse[::-1]) / 2.0
        a_lse_pal = np.maximum(a_lse_pal, 0)
        try_update(a_lse_pal)
        try_update(a_lse)

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (score={best_score:.8f}) ===")
    for ln in [1500, 1000]:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        if len(best_seq) == ln:
            continue
        t_ln = min(rem * 0.20, 200.0)
        if t_ln < 40:
            continue
        print(f"  Trying n={ln}, budget={t_ln:.0f}s")
        init_ln = interpolate_sequence(best_seq, ln)
        h_ln = full_to_half(init_ln, ln)

        # Quick MW
        t_mw = t_ln * 0.5
        h_ln_mw, _ = minimax_mw_palindromic(h_ln, ln, t_mw, eta=0.005, n_pg_iters=30)
        a_ln_mw = half_to_full(h_ln_mw, ln)
        try_update(a_ln_mw)

        # LP polish
        rem = deadline - time.monotonic()
        if rem > 30:
            t_lp = min(rem * 0.4, t_ln * 0.4)
            if t_lp > 20:
                h_ln_lp, _ = lp_alternating_palindromic(h_ln_mw, ln, t_lp, solver=solver)
                a_ln_lp = half_to_full(h_ln_lp, ln)
                try_update(a_ln_lp)

    # === Phase 5: Final LP polish on best ===
    print(f"\n=== Phase 5: Final LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        n_fin = len(best_seq)
        if n_fin % 2 != 0:
            n_fin += 1
            best_seq = interpolate_sequence(best_seq, n_fin)
            best_score = evaluate_local(best_seq)
        t5 = min(rem * 0.4, 120.0)
        h_fin = full_to_half(best_seq, n_fin)
        h_fin_lp, _ = lp_alternating_palindromic(h_fin, n_fin, t5, solver=solver)
        a_fin = half_to_full(h_fin_lp, n_fin)
        try_update(a_fin)

    # === Phase 6: Final MW polish ===
    print(f"\n=== Phase 6: Final MW (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        n_fin = len(best_seq)
        if n_fin % 2 != 0:
            n_fin += 1
            best_seq = interpolate_sequence(best_seq, n_fin)
            best_score = evaluate_local(best_seq)
        t6 = min(rem * 0.5, 60.0)
        h_fin = full_to_half(best_seq, n_fin)
        h_fin_mw, _ = minimax_mw_palindromic(h_fin, n_fin, t6, eta=0.001, n_pg_iters=50)
        a_fin = half_to_full
