import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):

def hankel_mv(p, v):
    """Multiply Hankel matrix H (H[i,j] = p[i+j]) by vector v, using FFT."""
    n = len(v)
    # H @ v has components: (Hv)_i = sum_j p[i+j] v[j]
    # This is a cross-correlation: (p * reverse(v)) shifted
    # Let's use: (Hv)_i = sum_{j=0}^{n-1} p[i+j] v[j]
    # = correlation of p (length 2n-1) with v (length n), at shift i
    # Use FFT: pad v to length 2n-1, flip, correlate
    
    # Actually: sum_j p[i+j] v[j] for i=0,...,n-1
    # Let u = reverse(v), so u[n-1-j] = v[j]
    # sum_j p[i+j] v[j] = sum_j p[i+j] u[n-1-j]
    # = sum_k p[i+1+k] u[n-1-k] (substituting k = n-1-j, j = n-1-k)
    #   wait, let me just do it directly with the definition
    
    # (Hv)_i = sum_{j=0}^{n-1} p[i+j] * v[j]
    # This is the cross-correlation of p and v, evaluated at lags i to i+n-1
    # More specifically, it's (p correlated with v)[i] where correlation is:
    #   (p ⋆ v)[i] = sum_j p[i+j] v[j]
    
    # Using FFT: if we define the standard convolution (p * u)[k] = sum_j p[j] u[k-j]
    # then (p ⋆ v)[i] = sum_j p[i+j] v[j] = sum_j p[i+j] v[i+j - i]
    # Let m = i+j: = sum_m p[m] v[m-i] = (p * reverse(v))[n-1-i + ...]
    
    # Simpler: just use the direct formula with FFT
    # Define w = v reversed: w[j] = v[n-1-j]
    # (Hv)_i = sum_j p[i+j] v[j] = sum_j p[i+j] w[n-1-j]
    # Let k = n-1-j: j = n-1-k
    # = sum_k p[i+n-1-k] w[k]
    # = (p * w)[i+n-1] where (p*w)[m] = sum_k p[m-k] w[k]
    
    w = v[::-1]  # w[j] = v[n-1-j]
    # (Hv)_i = (p * w)[i + n - 1] where convolution is standard
    # p has length 2n-1, w has length n
    # p * w has length (2n-1) + n - 1 = 3n - 2
    # We need indices n-1, n, ..., 2n-2 (which are 0 to n-1 in 0-indexed from n-1)
    
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    P = rfft(p, fft_size)
    W = rfft(w, fft_size)
    conv = irfft(P * W, fft_size)
    
    # (Hv)_i = conv[i + n - 1] for i = 0, ..., n-1
    return conv[n-1:2*n-1]


def hankel_grad(p, f):
    """Gradient of f^T H_p f / 2 w.r.t. f, where H_p is Hankel with entries p[i+j].
    Returns H_p @ f."""
    return hankel_mv(p, f)


def evaluate_local(a):

    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage, deadline=None):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    if deadline is None:
        deadline = t0 + time_limit
    n_stages = len(T_stages)

    for stage_i, stage_T in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 3:
            break

        stages_left = n_stages - stage_i
        avail_time = remaining
        iters = max(20, min(maxiter_per_stage, int(avail_time * 30 / stages_left)))

        def func(z_in):
            if time.monotonic() > deadline:
                return 1e15
            obj, _ = logsumexp_obj_grad(z_in, n, stage_T)
            return obj

        def jac(z_in):
            if time.monotonic() > deadline:
                return np.zeros_like(z_in)
            _, g = logsumexp_obj_grad(z_in, n, stage_T)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-25,
                         'gtol': 1e-20, 'maxfun': iters * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def spectral_flatten(a, n_iters=100, blend=0.3, deadline=None):
    """Spectral flattening (Gerchberg-Saxton style)."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)

        mags = np.abs(F)
        non_dc = mags[1:]
        if non_dc.min() < 1e-15:
            target = np.mean(non_dc)
        else:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        phases = np.angle(F)
        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend) * mags[k] + blend * target
            F[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F[L - k] = new_mag * np.exp(1j * phases[L - k])

        a_new = np.fft.ifft(F).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new

    return a

        dual_vals.append(M_b)  # LP value is dual (lower bound)

        # Stall detection and restart
        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-12:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 6:
                # Perturb y to escape
                x = np.linspace(0, 2 * np.pi, n)
                for trial in range(3):
                    freq = rng.integers(1, 10)
                    phase = rng.uniform(0, 2 * np.pi)
                    strength = 0.02 + 0.01 * trial
                    noise = strength * np.cos(freq * x + phase)
                    y_new = y + noise
                    y_new = np.maximum(y_new, 0)
                    sy_new = y_new.sum()
                    if sy_new > 1e-15:
                        y_new = y_new / sy_new
                        sc = 2.0 * n * fast_convolve(y_new).max()
                        if sc < best_score:
                            y = y_new
                            a = y_new.copy()
                            break
                        elif sc < 2.0 * n * fast_convolve(y).max() * 1.01:
                            y = y_new
                            break
                stall_count = 0

            if (it + 1) % 1000 == 0 and verbose:
                elapsed = time.monotonic() - t0
                gap = abs(primal_vals[-1] - dual_vals[-1]) if len(primal_vals) > 0 else 0
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): "
                      f"primal={best_score/(2*n):.10f} dual={dual_vals[-1]:.10f} "
                      f"gap={gap:.2e} avg_lp={np.mean(lp_times[-50:])*1000:.0f}ms")

    return best_a, best_score


def multiplicative_weights_refine(a0, time_limit, n_steps=500000, verbose=True):
    """
    Multiplicative weights (Hedge) for the game:
    P: distribution over lags, F: point in simplex.
    F's best response: few steps of projected gradient on f^T Q_p f.
    P's update: exponential weights based on losses.
    """
    n = len(a0)
    n_lags = 2 * n - 1
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    # P's strategy: uniform over lags initially
    log_q = np.zeros(n_lags)  # log q
    q = np.ones(n_lags) / n_lags

    # F's current point
    f = a.copy()
    
    # Loss history for P
    # loss_k = (f*f)[k] (the loss P pays for each lag k)

    eta_p = 1.0 / np.sqrt(np.log(n_lags))  # P's learning rate
    eta_f = 0.1  # F's gradient step for inner QP
    n_f_steps = 20  # Number of gradient steps for F's best response

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        # F's best response to current q: minimize f^T Q_q f
        # Using projected gradient: f <- proj_simplex(f - eta * 2 * Q_q @ f)
        f_new = f.copy()
        for _ in range(n_f_steps):
            grad = 2.0 * hankel_mv(q, f_new)  # gradient of f^T Q_q f
            # Projected gradient step onto simplex
            f_new = f_new - eta_f * grad
            # Project onto simplex
            f_new = project_simplex(f_new)
            eta_f *= 0.95  # decay
            
        f = f_new
        f = np.maximum(f, 0)
        sf = f.sum()
        if sf < 1e-15:
            f = np.ones(n) / n
        else:
            f = f / sf

        # Compute losses for P: loss_k = (f*f)[k]
        conv_f = fast_convolve(f)
        losses = conv_f  # length 2n-1

        # Update P's strategy: q_k proportional to exp(-eta * cumulative_loss_k)
        # Use online update: q <- q * exp(-eta * loss) / sum(...)
        log_q -= eta_p * losses
        log_q -= log_q.max()  # numerical stability
        q = np.exp(log_q)
        q = q / q.sum()

        # Evaluate
        score = 2.0 * n * conv_f.max()
        if score < best_score:
            best_score = score
            best_a = f.copy()

        # Also evaluate the "average" strategy
        # (The Nesterov-style average would be better, but for simplicity)
        if step % 1000 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"    MW step={step} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def project_simplex(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cv - 1))[0][-1]
    theta = (cv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def perturbed_sequence(a, rng, strength=0.03):
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 10)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s
