    a = np.asarray(a, dtype=np.float64)

    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

    if s < 0.01:

    return float(2.0 * n * b.max() / (s * s))

        if time.monotonic() - t0 > time_limit - 5:
            break
        
        stage_time_start = time.monotonic()
        
        def func(z):
            a_trial = np.maximum(z, 0)
            st = a_trial.sum()
            if st < 1e-15:
                return 1e15
            return power_mean_obj_and_grad(a_trial, p)[0]
        
        def jac(z):
            a_trial = np.maximum(z, 0)
            st = a_trial.sum()
            if st < 1e-15:
                return np.zeros_like(z)
            return power_mean_obj_and_grad(a_trial, p)[1]
        
        try:
            res = minimize(
                fun=func, x0=a, jac=jac,
                method='L-BFGS-B',
                bounds=[(0, None)] * n,
                options={
                    'maxiter': iters_per_stage,
                    'ftol': 1e-18,
                    'gtol': 1e-10,
                    'maxfun': iters_per_stage * 50,
                    'disp': False,
                    'iprint': -1
                }
            )
            a = np.maximum(res.x, 0)
            sa = a.sum()
            if sa > 1e-15:
                a = a / sa
        except Exception as e:
            print(f"    L-BFGS stage p={p} failed: {e}")
        
        score = evaluate_local(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        elapsed = time.monotonic() - t0
        stage_elapsed = time.monotonic() - stage_time_start
        print(f"    PM-LB p={p:.0f} ({stage_elapsed:.0f}s, total {elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    # Use linear interpolation
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def peak_redistribution(a, time_limit, n_iters=5000):
    """
    Targeted optimization: reduce the dominant peaks by adjusting
    the coordinates that contribute most to them.
    Uses a multiplicative (mirror descent) update.
    """
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    
    # Adam-like state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    eta = 0.1
    no_improve = 0
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Find active peaks (top few)
        threshold = max_b * 0.99
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        # Compute gradient from active peaks only
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= max(1, len(active))
        
        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-15:
            g = g / g_norm
        
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Multiplicative (mirror) update
        log_a = np.log(np.maximum(a, 1e-300))
        log_a = log_a - eta * m_hat
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = a_new / a_new.sum()
        
        score = evaluate_local(a_new)
        if score < best_score - 1e-12:
            a = a_new
            best_score = score
            best_a = a.copy()
            no_improve = 0
            # Decay eta slowly
            eta = max(0.01, eta * 0.9999)
        else:
            no_improve += 1
            if no_improve > 200:
                # Reduce learning rate
                eta *= 0.9
                if eta < 0.001:
                    # Add perturbation
                    noise = rng.standard_normal(n) * 0.005
                    a = a + noise
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    m = np.zeros(n)
                    v = np.zeros(n)
                    t = 0
                    eta = 0.05
                no_improve = 0
        
        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakRedist iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score


def coordinate_descent_refine(a0, time_limit, n_steps=100000):
    """Final coordinate descent polish."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall = 0
    rng = np.random.default_rng(777)
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        # Focus on top peaks
        peaks = np.argsort(b)[::-1][:3]
        
        improved = False
        for peak_k in peaks:
            if improved:
                break
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            # Try reducing top contributors
            top_m = min(5, len(idx))
            top_positions = idx[np.argsort(contribs)[::-1][:top_m]]
            
            for pos in top_positions:
                if a[pos] < 1e-12:
                    continue
                for factor in [0.999, 0.998, 0.995, 0.99, 0.98]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] - a_trial[pos]
                    if diff < 1e-16:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    # Redistribute to positions NOT contributing to this peak
                    a_trial[mask] = others + diff * others / others_sum
                    a_trial = a_trial / a_trial.sum()
                    score = evaluate_local(a_trial)
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            stall += 1
            if stall % 500 == 0:
                # Try a random perturbation
                n_perturb = max(1, n // 100)
                positions = rng.choice(n, size=n_perturb, replace=False)
                a_trial = a.copy()
                for pos in positions:
                    a_trial[pos] *= (1.0 + rng.uniform(-0.02, 0.02))
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = evaluate_local(a_trial)
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
            if stall > 5000:
                break
        else:
            stall = 0
        
        if (step + 1) % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc {elapsed:.0f}s: score={best_score:.8f}")
    
    return best_a, best_score
