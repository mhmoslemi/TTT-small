    s = b_val.sum()

def solve_peak_shave(a, active_indices, solver=None):

def interpolated_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def fourier_init(n, rng):
    """Initialize based on flat-spectrum principle in frequency domain."""
    # We want |F(a)| to be approximately constant
    # Start with a base shape and add controlled harmonic content
    x = np.arange(n) / n
    a = np.ones(n) / n  # uniform base
    
    # Add anti-correlation harmonics to flatten the convolution
    n_harmonics = min(10, n // 4)
    for h in range(1, n_harmonics + 1):
        freq = h * rng.uniform(0.8, 1.2)
        phase = rng.uniform(0, 2 * np.pi)
        amplitude = rng.uniform(0.02, 0.08) / h
        a += amplitude * np.cos(2 * np.pi * freq * x + phase)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def cosine_taper_init(n):

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:

    return a / s

def bell_init(n):

        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            stall_count = 0
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        scores_window.append(best_score)
        if len(scores_window) > 30:
            scores_window = scores_window[-30:]

        if stall_count > 100:
            y = a.copy()
            momentum = base_momentum * 0.3
            stall_count = 0
        elif stall_count > 50:
            momentum *= 0.95

        if (it + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Nesterov iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} M_lp={M_lp:.6f} mom={momentum:.3f}")

    return best_a, best_score


def peak_shave_polish(a0, time_limit, solver=None, n_rounds=200):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    no_improve = 0

    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 40:
            break

        b = fast_convolve(a)
        M = b.max()
        threshold = M * 0.995
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        if len(active) > 20:
            top_idx = np.argsort(b[active])[::-1][:20]
            active = active[top_idx]

        delta = solve_peak_shave(a, active, solver)
        if delta is None:
            no_improve += 1
            continue

        best_alpha = 0.0
        best_obj = score_normalized(a)
        for alpha_try in [0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5]:
            a_try = a + alpha_try * delta
            a_try = np.maximum(a_try, 0)
            st = a_try.sum()
            if st < 1e-15:
                continue
            a_try = a_try / st
            obj = score_normalized(a_try)
            if obj < best_obj:
                best_obj = obj
                best_alpha = alpha_try

        if best_alpha > 0:
            a = a + best_alpha * delta
            a = np.maximum(a, 0)
            st = a.sum()
            if st < 1e-15:
                break
            a = a / st
            if best_obj < best_score:
                best_score = best_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if (rnd + 1) % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave round={rnd+1} ({elapsed:.0f}s): score={best_score:.8f} no_imp={no_improve}")

    return best_a, best_score


def alternating_best_response(a0, time_limit, solver=None, max_iters=200000):
    n = len(a0)
    a_left = np.asarray(a0[:n // 2], dtype=np.float64).copy()
    a_right = np.asarray(a0[n // 2:], dtype=np.float64).copy()
    s = a_left.sum() + a_right.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        return a, score_normalized(a)
    a_left = a_left / s
    a_right = a_right / s

    best_a = np.concatenate([a_left, a_right])
    best_score = score_normalized(best_a)
    t0 = time.monotonic()
    lr = 0.3

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_old_left = a_left.copy()
        b_old_right = a_right.copy()

        # Optimize left given right: min M s.t. (a_right * b)_k <= M, sum(b)=1, b>=0
        b_new_l, _ = solve_best_response(a_right, solver)
        if b_new_l is not None:
            a_left = (1 - lr) * b_old_left + lr * b_new_l
            a_left = np.maximum(a_left, 0)
            sl = a_left.sum()
            if sl > 1e-15:
                a_left = a_left / sl

        # Optimize right given left
        b_new_r, _ = solve_best_response(a_left, solver)
        if b_new_r is not None:
            a_right = (1 - lr) * b_old_right + lr * b_new_r
            a_right = np.maximum(a_right, 0)
            sr = a_right.sum()
            if sr > 1e-15:
                a_right = a_right / sr

        a_cur = np.concatenate([a_left, a_right])
        sc = score_normalized(a_cur)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()

        if (it + 1) % 500 == 0:
            lr *= 0.9
            elapsed = time.monotonic() - t0
            print(f"    AltBR iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def cosine_taper_init(n):
    x = np.arange(n)
    center = (n - 1) / 2.0
    a = np.cos(np.pi * (x - center) / n) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def harmonic_init(n, rng):
    x = np.arange(n) / n
    a = np.ones(n) / n
    n_harmonics = min(8, n // 5)
    for h in range(1, n_harmonics + 1):
        freq = h * rng.uniform(0.9, 1.1)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.03, 0.07) / h
        a += amp * np.cos(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def propose_candidate(seed=42, budget_s=1800.0):
    global _global_deadline
    rng = np.random.default_rng(seed)
    _global_deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        pass

    n_base = len(best_seq) if best_seq is not None else 1200

    if best_seq is None:
        best_seq = cosine_taper_init(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            remaining = _global_deadline - time.monotonic()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)}, remaining={remaining:.0f}s)")

    total_budget = budget_s - 30

    # Phase 1: Primary Nesterov LP (40%)
    print(f"\n=== Phase 1: Nesterov Adaptive (score={best_score:.8f}) ===")
    t1 = min(0.40 * total_budget, _global_deadline - time.monotonic())
    if t1 > 10:
        a_out, _ = nesterov_lp_adaptive(best_seq, t1, solver=solver)
        try_update(a_out, "nesterov_p1")

    # Phase 2: Peak Shaving (15%)
    print(f"\n=== Phase 2: Peak Shaving (score={best_score:.8f}) ===")
    t2 = min(0.15 * total_budget, _global_deadline - time.monotonic())
    if t2 > 10:
        a_ps, _ = peak_shave_polish(best_seq, t2, solver=solver, n_rounds=300)
        try_update(a_ps, "peak_shave_1")

    # Phase 3: Second Nesterov from improved point (20%)
    print(f"\n=== Phase 3: Nesterov Round 2 (score={best_score:.8f}) ===")
    t3 = min(0.20 * total_budget, _global_deadline - time.monotonic())
    if t3 > 10:
        a_out2, _ = nesterov_lp_adaptive(best_seq, t3, solver=solver)
        try_update(a_out2, "nesterov_p3")

    # Phase 4: Multi-length exploration (15%)
    print(f"\n=== Phase 4: Length Exploration (score={best_score:.8f}) ===")
    t4_total = min(0.15 * total_budget, _global_deadline - time.monotonic())
    if t4_total > 30:
        lengths_to_try = []
        for dn in [-100, 100, -200, 200, 300, -300]:
            nl = n_base + dn
            if 600 <= nl <= 2500:
                lengths_to_try.append(nl)

        if lengths_to_try:
            per_len = t4_total / len(lengths_to_try)
            for nl in lengths_to_try[:3]:
                remaining = _global_deadline - time.monotonic()
                if remaining < 15:
                    break
                t_len = min(per_len, remaining)
                if t_len < 10:
                    break
                print(f"  Trying n={nl} (budget={t_len:.0f}s)")
                init_try = interpolate_sequence(best_seq, nl)
                a_try, _ = nesterov_lp_adaptive(init_try, t_len, solver=solver, max_iters=100000)
                try_update(a_try, f"len_{nl}")
                remaining = _global_deadline - time.monotonic()
                if remaining < 10:
                    break
                t_ps = min(8.0, remaining)
                if t_ps > 5:
                    a_ps2, _ = peak_shave_polish(best_seq, t_ps, solver=solver, n_rounds=50)
                    try_update(a_ps2, f"ps_len_{nl}")

    # Phase 5: Final alternating + peak shave (10%)
    print(f"\n=== Phase 5: Final Polish (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    if remaining > 20:
        t_alt = min(0.5 * remaining, 60.0)
        if t_alt > 10:
            a_alt, _ = alternating_best_response(best_seq, t_alt, solver=solver)
            try_update(a_alt, "alt_br")
        remaining = _global_deadline - time.monotonic()
        if remaining > 10:
            t_fin = min(remaining * 0.7, 40.0)
            if t_fin > 5:
                a_fin, _ = peak_shave_polish(best_seq, t_fin, solver=solver, n_rounds=100)
                try_update(a_fin, "final_polish")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs) and fs < best_score:
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
