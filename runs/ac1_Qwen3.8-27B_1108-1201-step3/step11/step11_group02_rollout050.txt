def fosil_lp_refine(a0, time_limit, solver, n_max_iters=50000):
    """
    FOSIL-accelerated LP alternating minimization.
    FOSIL uses a heavy-ball scheme with adaptive momentum.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()  # Extrapolated point
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # FOSIL parameters
    mu = 5.0  # heavy ball parameter
    beta = 0.9  # momentum decay
    step_seq = 1.0 / (np.sqrt(n) + 1.0)
    
    prev_a = a.copy()
    prev_b = None
    stall = 0
    stall_threshold = 15
    rng = np.random.default_rng(9999)
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Determine extrapolation step
        if it == 0:
            z = a.copy()
        else:
            # FOSIL: z = y + beta_i * (y - prev_a) where beta_i is adaptive
            step_ratio = min(1.0, it / max(1, mu))
            beta_i = step_seq * step_ratio
            z = y + beta_i * (y - prev_a)
        
        # Clip and normalize z
        z = np.maximum(z, 0)
        sz = z.sum()
        if sz < 1e-15:
            z = a.copy()
            sz = 1.0
        z = z / sz
        
        # Solve LP: find best response b to z
        b_new, _ = solve_lp_best_response(z, solver, warm_start=prev_b)
        if b_new is None:
            continue
        
        # Update y (extrapolated point)
        y = b_new.copy()
        prev_a = a.copy()
        
        # Weighted average update for a (FOSIL uses step sequence)
        # Use a time-averaged iterate for better convergence guarantees
        a_new = (it * a + (1.0 + beta) * b_new) / (it + 1 + beta)
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
            sa = 1.0
        a = a_new / sa
        prev_b = b_new
        
        # Evaluate
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        
        # Stall detection and restart
        if (it + 1) % 5 == 0:
            if best_score > 2.0 * n * fast_convolve(best_a).max() - 1e-12:
                stall += 1
            if stall >= stall_threshold and it > 100:
                # Perturb toward second-best peak
                b_cur = fast_convolve(y)
                sorted_idx = np.argsort(b_cur)[::-1]
                peak1 = sorted_idx[0]
                peak2 = sorted_idx[1] if len(sorted_idx) > 1 else peak1
                
                # Reduce contribution to peak1, increase toward peak2
                js1 = max(0, peak1 - n + 1)
                je1 = min(n - 1, peak1)
                idx1 = np.arange(js1, je1 + 1)
                contribs1 = y[idx1] * y[peak1 - idx1]
                
                n_reduce = max(2, len(idx1) // 15)
                top_pos = idx1[np.argsort(contribs1)[::-1][:n_reduce]]

                        y[pos] *= 0.93

def logsumexp_smoothed_grad(a, T_temp):
    """
    Compute LogSumExp smoothed objective gradient.
    Objective: (2n/T) * LogSumExp(conv(a,a)/T) with sum(a)=1 normalization.
    Returns gradient w.r.t. a (unconstrained, then we project).
    """
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp((b - max_b) / T_temp)
    w = exp_b / exp_b.sum()

    # = (1/T) * sum_k w[k] * dc[k]/da[i]
    # dc[k]/da[i] = a[k-i] + a[i] (when both valid, with factor 2 if k=2i)
    # More precisely: dc[k]/da[i] = a[k-i] (if 0<=k-i<n) + a[i] (if 0<=i<n and i!=k-i... )
    # Actually: c[k] = sum_j a[j]*a[k-j], so dc[k]/da[i] = a[k-i] (from j=i) + a[i] (from k-j=i, i.e., j=k-i)
    # If k-i == i (i.e., k=2i), then dc[k]/da[i] = 2*a[i]
    
    # Efficient computation using cross-correlation:
    # grad[i] = sum_k w[k] * (a[k-i] + a[i]) (with boundary handling)
    # = sum_k w[k]*a[k-i] + sum_k w[k]*a[i]
    # = cross_correl(w, a)[i] + a[i] * sum(w)
    # = cross_correl(w, a)[i] + a[i]
    # (since sum(w) = 1)
    
    # cross_correl(w, a)[i] = sum_k w[k] * a[k-i] = sum_j w[i+j] * a[j]
    # This is the convolution of w (reversed) with a, or equivalently:
    # = (w_rev * a)[i + n - 1] where * is linear convolution
    
    # Use FFT for the cross-correlation
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # w[k] * a[k-i] summed over k = (w * flip(a))[i + n - 1]... 
    # Actually: sum_k w[k]*a[k-i] = (w correlations with a)
    # Let's just compute it directly using the convolution structure
    
    # sum_k w[k] * a[k-i]: for each i, sum over k where 0 <= k-i < n
    # = sum_{j=0}^{n-1} w[i+j] * a[j] (where j = k-i, so k = i+j)
    # This is a matrix-vector product where M[i,j] = w[i+j]
    
    # Use FFT: this is the cross-correlation of w and a
    # corr[i] = sum_j w[i+j] * a[j]
    # = (w_pad * flip_a_pad)[offset]
    
    # Simpler: use the fact that this is a Toeplitz matrix times vector
    # For efficiency with FFT:
    # corr = ifft(fft(w_pad) * conj(fft(a_pad)))... not quite right for linear case
    
    # Just do it with the convolution:
    # corr[i] = sum_{j=0}^{min(n-1, 2n-2-i)} w[i+j] * a[j]
    # This is the same as the (i)-th element of the "anti-convolution"
    
    # Actually the simplest FFT approach:
    # Define w_ext of length 2n-1 (already is)
    # corr[i] = sum_j a[j] * w[i+j] = (anti-convolution)
    # = convolve(flip(a), w)[i + n - 1]... let me just verify with small example
    
    # For now, use a direct but vectorized approach:
    # corr[i] = dot(w[i:i+n], a[:n]) but w has length 2n-1, so w[i:i+n] has length min(n, 2n-1-i)
    
    # Use FFT-based approach:
    # corr = real(ifft(fft(w, N) * conj(fft(a_ext, N))))[n-1:n-1+n]
    # where a_ext = [0, 0, ..., 0, a[0], ..., a[n-1], 0, ...] (shifted by n-1)
    
    a_shifted = np.zeros(fft_size)
    a_shifted[:n] = a
    A_shifted = rfft(a_shifted, fft_size)
    
    # w of length 2n-1, pad to fft_size
    w_padded = np.zeros(fft_size)
    w_padded[:2*n-1] = w
    W_padded = rfft(w_padded, fft_size)
    
    # cross-correlation: sum_k w[k]*a[k-i]
    # = real(ifft(W * conj(A_shifted))) shifted appropriately
    corr_full = irfft(W_padded * np.conj(A_shifted), fft_size)
    corr = corr_full[n-1:n-1+n]
    
    grad = (corr + a) / T_temp * (2.0 * n)
    return grad


def logsumexp_lbgfs(a0, time_limit, T_stages=None, maxiter_per_stage=3000):
    """L-BFGS on LogSumExp smoothed objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [10.0, 6.0, 4.0, 2.5, 1.5, 1.0, 0.6, 0.35, 0.2, 0.1]
    
    n_stages = len(T_stages)
    iters_per_stage = max(10, maxiter_per_stage // n_stages)
    best_a = a.copy()
    best_score = evaluate_local(a)
    
    for T in T_stages:
        if time.monotonic() - t0 > time_limit - 3:
            break
        
        def func(z_inner):
            a_trial = np.maximum(z_inner, 0)
            st = a_trial.sum()
            if st < 1e-15:
                return 1e15
            a_trial = a_trial / st
            b = fast_convolve(a_trial)
            max_b = b.max()
            exp_b = np.exp((b - max_b) / T)
            lse = max_b / T + np.log(exp_b.sum())
            return (2.0 * n / T) * lse
        
        def jac(z_inner):
            a_trial = np.maximum(z_inner, 0)
            st = a_trial.sum()
            if st < 1e-15:
                return np.zeros_like(z_inner)
            a_trial = a_trial / st
            grad = logsumexp_smoothed_grad(a_trial, T)
            return grad
        
        try:
            res = minimize(fun=func, x0=a, jac=jac, method='L-BFGS-B',
                           options={'maxiter': iters_per_stage, 'ftol': 1e-14,
                                    'gtol': 1e-10, 'maxfun': iters_per_stage * 100,
                                    'disp': False})
            a = np.maximum(res.x, 0)
            sa = a.sum()
            if sa > 1e-15:
                a = a / sa
        except Exception:
            pass
        
        sc = evaluate_local(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        elapsed = time.monotonic() - t0
        print(f"    LSE-LB T={T:.2f} ({elapsed:.0f}s): score={sc:.8f}")
    
    return best_a, best_score


def coordinate_descent_refine(a0, time_limit, n_steps=200000):
    """Targeted coordinate descent focusing on peak-contributing elements."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.9999
        peaks = np.where(b >= threshold)[0]
        
        improved = False
        for peak_k in peaks[:2]:
            if improved:
                break
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_m = min(6, len(idx))
            top_positions = idx[np.argsort(contribs)[::-1][:top_m]]
            
            for pos in top_positions:
                if a[pos] < 1e-12:
                    continue
                for factor in [0.999, 0.995, 0.99, 0.98]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] - a_trial[pos]
                    if diff < 1e-16:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others_sum
                    a_trial = a_trial / a_trial.sum()
                    score = evaluate_local(a_trial)
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            stall += 1
            if stall > 300:
                peak_k = int(np.argmax(b))
                js = max(0, peak_k - n + 1)
                je = min(n - 1, peak_k)
                idx = np.arange(js, je + 1)
                contribs = a[idx] * a[peak_k - idx]
                top_positions = idx[np.argsort(contribs)[::-1][:3]]
                a_trial = a.copy()
                for pos in top_positions:
                    if a_trial[pos] > 1e-12:
                        a_trial[pos] *= 0.998
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = evaluate_local(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
                elif stall > 1500:
                    break
        else:
            stall = 0
        
        if (step + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc {elapsed:.0f}s: score={best_score:.8f}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create structured initial sequence."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = np.exp(-3 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = np.exp(-2.0 * (x - 0.5) ** 2 / 0.2)
    else:
        a = np.exp(-8 * (x - 0.3) ** 2) + np.exp(-8 * (x - 0.7) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses FOSIL-LP, multi-n exploration, LogSumExp, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    
    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}")
    
    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
    
    n_parent = len(best_seq)
    
    # === Phase 1: FOSIL-LP at parent's n (main workhorse) ===
    print(f"\n=== Phase 1: FOSIL-LP (score={best_score:.8f}, n={n_parent}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = fosil_lp_refine(best_seq, t_alloc, solver)
        try_update(a_lp, "FOSIL_p1")
    
    # === Phase 2: Multi-n Exploration ===
    print(f"\n=== Phase 2: Multi-n Exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        n_best = len(best_seq)
        # Try nearby n values and some larger ones
        candidate_ns = sorted(set([
            max(400, n_best - 200),
            max(600, n_best + 100),
            max(800, n_best + 300),
            min(3000, n_best * 2),
        ]))
        # Filter out duplicates and the current best
        candidate_ns = [x for x in candidate_ns if x != n_best]
        per_n = t_alloc / max(1, len(candidate_ns))
        print(f"  Trying n values: {candidate_ns}, {per_n:.0f}s each")
        for n_try in candidate_ns:
            if time.monotonic() > deadline - 10:
                break
            a_interp = interpolate_sequence(best_seq, n_try)
            t_sub = min(per_n, deadline - time.monotonic() - 5)
            if t_sub < 10:
                continue
            print(f"  n={n_try}, budget={t_sub:.0f}s")
            a_sub, _ = fosil_lp_refine(a_interp, t_sub, solver, n_max_iters=30000)
            try_update(a_sub, f"multi_n_{n_try}")
    
    # === Phase 3: Deep FOSIL-LP Refinement at best n ===
    print(f"\n=== Phase 3: Deep FOSIL-LP (score={best_score:.8f}, n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_deep, _ = fosil_lp_refine(best_seq, t_alloc, solver, n_max_iters=50000)
        try_update(a_deep, "deep_fosil")
    
    # === Phase 4: LogSumExp L-BFGS ===
    print(f"\n=== Phase 4: LogSumExp L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 15:
        T_stages = [8.0, 5.0, 3.0, 2.0, 1.2, 0.7, 0.4, 0.2, 0.1]
        print(f"  Budget: {t_alloc:.0f}s")
        a_lb, _ = logsumexp_lbgfs(best_seq, t_alloc, T_stages=T_stages,
                                   maxiter_per_stage=5000)
        try_update(a_lb, "lbgfs")
    
    # === Phase 5: Coordinate Descent Polish ===
    print(f"\n=== Phase 5: Coord Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 10:
        a_cd, _ = coordinate_descent_refine(best_seq, t_alloc, n_steps=300000)
        try_update(a_cd, "coord_desc")
    
    # === Phase 6: Final short FOSIL-LP ===
    print(f"\n=== Phase 6: Final FOSIL-LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.05, 60.0)
    if t_alloc > 5:
        a_fin, _ = fosil_lp_refine(best_seq, t_alloc, solver, n_max_iters=20000)
        try_update(a_fin, "final")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
