with mirror descent, adaptive annealing, and multi-phase refinement.

Key innovations:

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence using local computation (scale-invariant)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def compute_gradient(b, w, fft_size):
    """
    Compute gradient of smoothed objective.
    grad_m = 2 * sum_j w[m+j] * b[j] for m = 0..n-1
    
    Uses FFT: grad = 2 * (b_rev * w)[n-1 : 2n-1]
    where b_rev = b[::-1]

    n = len(b)
    b_rev = b[::-1]

    B_rev = rfft(b_rev, fft_size)
    W = rfft(w, fft_size)
    conv = irfft(B_rev * W, fft_size)
    # Extract the relevant part
    grad = conv[n-1 : 2*n-1] * 2.0
    return grad

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated BBS (Braun-Blahut-Strybulevich)
with mirror descent in log-domain, adaptive annealing, multi-restart strategy,
and multi-length exploration.

Key innovations over previous approach:
- O(n log n) per iteration via FFT-based gradient (cross-correlation trick)
- Mirror descent in log-domain (no projection, automatically stays in simplex)
- Multiple restarts with controlled perturbation to escape local minima
- Adaptive temperature with restart on stall
- Fast coordinate descent polish using FFT evaluation
- Multi-n exploration with interpolation

import numpy as np
import time
from scipy.fft import rfft, irfft


def next_pow2(m):
    """Next power of 2 >= m."""
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = next_pow2(size)
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence (scale-invariant)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def softmax_stable(x, T):
    """Numerically stable softmax(x/T)."""
    x_shifted = x / T - (x / T).max()
    exp_x = np.exp(x_shifted)
    s = exp_x.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return exp_x / s


def bbs_mirror_descent(b0, time_limit, T_init=None, eta_base=1.0,
                       n_restarts_max=5, verbose=True):
    """
    BBS algorithm with mirror descent in log-domain.
    
    Minimizes max_k conv_k(b) subject to sum(b)=1, b>=0.
    Uses LogSumExp smoothing with annealing temperature T.
    
    Returns: (best_b, best_score)
    """
    n = len(b0)
    N = next_pow2(3 * n - 2)  # FFT size for gradient computation
    b = np.maximum(np.asarray(b0, dtype=np.float64), 0.0)
    s = b.sum()
    if s < 1e-15:
        b = np.ones(n) / n
        s = 1.0
    b = b / s
    
    conv = fast_convolve(b)
    max_conv = conv.max()
    
    if T_init is None:
        T_init = max_conv
    
    best_b = b.copy()
    best_score = 2.0 * n * max_conv
    t0 = time.monotonic()
    
    # Precompute b_rev FFT won't work since b changes each iteration
    # But we can optimize the inner loop
    
    for restart in range(n_restarts_max):
        if time.monotonic() - t0 > time_limit:
            break
        
        if restart > 0:
            # Perturb current best
            rng = np.random.default_rng(1000 + restart * 7919)
            noise_scale = 0.05 * np.sqrt(restart)
            b = best_b + rng.standard_normal(n) * noise_scale
            b = np.maximum(b, 0.0)
            s = b.sum()
            if s < 1e-15:
                b = best_b.copy()
            else:
                b = b / s
            T = T_init * (1.0 + 0.5 * restart)
        else:
            T = T_init
        
        # Annealing parameters
        T_min = T_init * 0.01
        T_decay = 0.9995  # Slight per-iteration decay
        T_decay_batch = 0.9  # Every 500 iters, bigger drop
        batch_count = 0
        
        stall = 0
        stall_threshold = 3000
        no_improve = 0
        best_local = best_score
        
        for it in range(10000000):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Compute auto-convolution
            B = rfft(b, N)
            conv = irfft(B * B, N)[:2 * n - 1]
            max_conv = conv.max()
            score = 2.0 * n * max_conv
            
            if score < best_score:
                best_score = score
                best_b = b.copy()
                no_improve = 0
            elif score < best_local:
                best_local = score
                no_improve = 0
            else:
                no_improve += 1
            
            if no_improve >= stall_threshold:
                break
            
            # Compute weights w = softmax(conv / T)
            w = softmax_stable(conv, T)
            
            # Compute gradient: grad_m = 2 * sum_j w[m+j] * b[j]
            # = 2 * (b_rev * w)[n-1 : 2n-1]
            b_rev = b[::-1]
            B_rev = rfft(b_rev, N)
            W = rfft(w, N)
            grad_conv = irfft(B_rev * W, N)
            grad = grad_conv[n - 1:2 * n - 1] * 2.0
            
            # Project gradient to maintain sum constraint
            grad_mean = grad.mean()
            grad_proj = grad - grad_mean
            
            # Adaptive step size
            grad_norm = np.sqrt(np.mean(grad_proj ** 2))
            if grad_norm < 1e-15:
                break
            eta = eta_base * T / (grad_norm * np.sqrt(T))
            # Clamp eta for stability
            eta = np.clip(eta, 0.01, 10.0)
            
            # Mirror descent update in log-domain
            logits = np.log(b + 1e-300)
            logits -= eta * grad_proj
            # Normalize logits for numerical stability
            logits -= logits.max()
            b_new = np.exp(logits)
            s_new = b_new.sum()
            if s_new < 1e-300:
                break
            b = b_new / s_new
            
            # Anneal temperature
            T = max(T * T_decay, T_min)
            batch_count += 1
            if batch_count % 500 == 0:
                T = max(T * T_decay_batch, T_min)
            
            # Restart on long stall within this restart
            if it > 0 and it % 5000 == 0:
                cur_score = 2.0 * n * fast_convolve(b).max()
                if cur_score >= best_local - 1e-10:
                    stall += 1
                    if stall >= 3:
                        break
                else:
                    stall = 0
                    best_local = cur_score
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    BBS restart {restart}: score={best_score:.8f} ({elapsed:.0f}s elapsed)")
    
    return best_b, best_score


def bbs_with_restarts(b0, time_limit, n_restarts=8, verbose=True):
    """
    Run BBS multiple times with different initializations and temperatures.
    Returns the best (b, score) found.
    """
    n = len(b0)
    b = np.maximum(np.asarray(b0, dtype=np.float64), 0.0)
    s = b.sum()
    if s < 1e-15:
        b = np.ones(n) / n
    else:
        b = b / s
    
    conv = fast_convolve(b)
    max_conv = conv.max()
    best_b = b.copy()
    best_score = 2.0 * n * max_conv
    t0 = time.monotonic()
    
    rng = np.random.default_rng(42)
    
    for r in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break
        
        remaining = time_limit - (time.monotonic() - t0)
        per_restart = remaining / max(1, n_restarts - r)
        # Give more time to early restarts
        per_restart = min(per_restart, remaining * 0.4)
        
        if r == 0:
            start_b = b
            T_init = max_conv * 0.5
        else:
            # Perturb from best
            noise_scale = 0.03 * (1 + 0.3 * r)
            start_b = best_b + rng.standard_normal(n) * noise_scale
            start_b = np.maximum(start_b, 0.0)
            ss = start_b.sum()
            if ss < 1e-15:
                start_b = best_b.copy()
            else:
                start_b = start_b / ss
            T_init = max_conv * (0.3 + 0.2 * r)
        
        b_out, score_out = bbs_mirror_descent(
            start_b, per_restart, T_init=T_init,
            eta_base=1.0, n_restarts_max=1, verbose=False
        )
        
        if score_out < best_score:
            best_score = score_out
            best_b = b_out.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    BBS-R restart {r}: NEW BEST {best_score:.8f} ({elapsed:.0f}s)")
        elif verbose:
            elapsed = time.monotonic() - t0
            print(f"    BBS-R restart {r}: score={score_out:.8f} (best={best_score:.8f}, {elapsed:.0f}s)")
    
    return best_b, best_score


def coordinate_descent_fast(b0, time_limit, n_iter_max=50000):
    """
    Fast coordinate descent: modify one coordinate at a time,
    using FFT to evaluate the score.
    Focuses on peak-contributing coordinates.
    """
    n = len(b0)
    N = next_pow2(2 * n - 1)
    b = np.maximum(np.asarray(b0, dtype=np.float64), 0.0)
    s = b.sum()
    if s < 1e-15:
        b = np.ones(n) / n
    else:
        b = b / s
    
    best_b = b.copy()
    best_score = 2.0 * n * fast_convolve(b).max()
    t0 = time.monotonic()
    stall = 0
    
    for it in range(n_iter_max):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(b)
        max_conv = conv.max()
        peak_idx = int(np.argmax(conv))
        
        # Find top contributing pairs to the peak
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        if je < js:
            break
        idx = np.arange(js, je + 1)
        contribs = b[idx] * b[peak_idx - idx]
        
        # Try reducing top contributors
        top_k = min(20, len(idx))
        top_positions = idx[np.argsort(contribs)[::-1][:top_k]]
        
        improved = False
        for pos in top_positions:
            if b[pos] < 1e-12:
                continue
            # Try several reduction factors
            for factor in [0.999, 0.995, 0.99, 0.98, 0.97]:
                a_trial = b.copy()
                a_trial[pos] *= factor
                diff = b[pos] - a_trial[pos]
                if diff < 1e-15:
                    continue
                # Redistribute to other positions proportionally
                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    break
                a_trial[mask] += diff * a_trial[mask] / others_sum
                a_trial = a_trial / a_trial.sum()
                
                # Quick score evaluation
                B_t = rfft(a_trial, N)
                conv_t = irfft(B_t * B_t, N)[:2 * n - 1]
                score_t = 2.0 * n * conv_t.max()
                
                if score_t < best_score - 1e-14:
                    b = a_trial
                    best_score = score_t
                    best_b = b.copy()
                    stall = 0
                    improved = True
                    break
            
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall > 2000:
                break
            # Try increasing a non-peak coordinate
            if stall % 50 == 0:
                # Find position far from peak that has low value
                dist_from_peak = np.abs(np.arange(n) - peak_idx)
                low_pos = np.argsort(b * (1 + dist_from_peak / n))[:5]
                for pos in low_pos:
                    a_trial = b.copy()
                    a_trial[pos] *= 1.5
                    a_trial = a_trial / a_trial.sum()
                    B_t = rfft(a_trial, N)
                    conv_t = irfft(B_t * B_t, N)[:2 * n - 1]
                    score_t = 2.0 * n * conv_t.max()
                    if score_t < best_score - 1e-14:
                        b = a_trial
                        best_score = score_t
                        best_b = b.copy()
                        stall = 0
                        break
        
        if (it + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc {elapsed:.0f}s: score={best_score:.8f}")
    
    return best_b, best_score


def interpolate_sequence(b0, new_n):
    """Interpolate sequence to new length, preserving sum=1."""
    old_n = len(b0)
    if old_n == new_n:
        return b0.copy()
    s = b0.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    b_norm = b0 / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    b_interp = np.interp(new_x, old_x, b_norm)
    b_interp = np.maximum(b_interp, 0.0)
    s2 = b_interp.sum()
    if s2 < 1e-15:
        return np.ones(new_n) / new_n
    return b_interp / s2


def structured_init(n, rng):
    """Create a structured initial sequence."""
    choice = rng.integers(0, 5)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.exp(-2.0 * (x - 0.5) ** 2 / 0.15)
    elif choice == 2:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 3:
        a = np.exp(-5 * (x - 0.3) ** 2) + 0.7 * np.exp(-5 * (x - 0.7) ** 2)
    else:
        # Tukey window (smooth uniform)
        r = 0.3
        a = np.ones(n)
        m = int(r * n)
        for i in range(m):
            t = 0.5 + 0.5 * np.cos(np.pi * i / m)
            a[i] = t
            a[n - 1 - i] = t
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

    # === Phase 1: BBS with multiple restarts (main workhorse) ===

    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s, n={n_curr}")
        a_bbs, sc_bbs = bbs_with_restarts(best_seq, t_alloc, n_restarts=10, verbose=True)
        try_update(a_bbs, "BBS_p1")
    
    # === Phase 2: Multi-n exploration ===
    print(f"\n=== Phase 2: Multi-n Exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = rem * 0.25
    if t_alloc > 30:
        n_best = len(best_seq)
        # Try different lengths
        candidate_ns = []
        for delta in [-400, -200, 200, 400, 600]:
            n_try = n_best + delta
            if 400 <= n_try <= 4000 and n_try != n_best:
                candidate_ns.append(n_try)
        # Also try halving and doubling
        n_half = max(400, n_best // 2)
        n_double = min(4000, n_best * 2)
        if n_half != n_best:
            candidate_ns.append(n_half)
        if n_double != n_best:
            candidate_ns.append(n_double)
        
        candidate_ns = sorted(set(candidate_ns))
        per_n = t_alloc / max(1, len(candidate_ns))
        print(f"  Trying n values: {candidate_ns}, ~{per_n:.0f}s each")
        
        for n_try in candidate_ns:
            if time.monotonic() > deadline - 15:
                break
            t_sub = min(per_n, deadline - time.monotonic() - 5)
            if t_sub < 10:
                continue
            print(f"  n={n_try}, budget={t_sub:.0f}s")
            a_interp = interpolate_sequence(best_seq, n_try)
            a_sub, sc_sub = bbs_with_restarts(a_interp, t_sub, n_restarts=3, verbose=False)
            try_update(a_sub, f"multi_n_{n_try}")
    
    # === Phase 3: Coordinate descent polish ===
    print(f"\n=== Phase 3: Coordinate Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = rem * 0.12
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        a_cd, sc_cd = coordinate_descent_fast(best_seq, t_alloc)
        try_update(a_cd, "coord_desc")
    
    # === Phase 4: Final BBS refinement ===
    print(f"\n=== Phase 4: Final BBS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = rem * 0.10
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        a_fin, sc_fin = bbs_with_restarts(best_seq, t_alloc, n_restarts=4, verbose=True)
        try_update(a_fin, "final_bbs")
    
    # === Phase 5: One more coordinate descent on final ===
    rem = deadline - time.monotonic()
    if rem > 10:
        t_alloc = rem * 0.6
        print(f"\n=== Phase 5: Final Coord Desc (score={best_score:.8f}) ===")
        a_cd2, sc_cd2 = coordinate_descent_fast(best_seq, t_alloc)
        try_update(a_cd2, "final_cd")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
