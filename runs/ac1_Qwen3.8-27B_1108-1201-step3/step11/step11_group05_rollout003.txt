            prob.solve(solver=solver_name)

    compute Anderson acceleration coefficients.
    Returns: x_new = sum c_i * x_i
    """
    k = len(iterates)
    if k <= 1:
        return iterates[-1].copy()

    m = min(m, k)
    X = np.array(iterates[-m:])  # shape (m, n)
    R = np.array(residuals[-m:])  # shape (m, n)

    # Using the standard formulation:
    # Let B = R @ R.T (m x m), d = R[0] @ R (m-vector)

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def solve_lp_response(a_fixed, solver):
    """Solve min_b M s.t. conv(a,b)_k <= M for all k, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = [solver, cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]
    solved = False
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
                solved = True
                break
        except Exception:
            continue
    if not solved or b_var.value is None:
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    b_val[np.isnan(b_val)] = 0
    s_val = b_val.sum()
    if s_val < 1e-15:
        return None, None
    b_val = b_val / s_val
    return b_val, float(max(0.0, M_var.value))


def evaluate_local(a):
    """Fast local evaluation of the objective."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to probability vector."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def anderson_mix(iterates, residuals, m=6):
    """Anderson acceleration: compute convex combination of recent iterates."""
    k = len(iterates)
    if k <= 1:
        return iterates[-1].copy()
    m = min(m, k)
    X = np.array(iterates[-m:])
    R = np.array(residuals[-m:])
    # min ||R @ c||^2 s.t. sum(c) = 1
    # KKT system: [2 R^T R, 1; 1^T, 0] [c; lam] = [0; 1]
    G = R.T @ R
    A = np.zeros((m + 1, m + 1))
    A[:m, :m] = 2.0 * G
    A[:m, m] = 1.0
    A[m, :m] = 1.0
    rhs = np.zeros(m + 1)
    rhs[m] = 1.0
    try:
        sol = np.linalg.solve(A + 1e-10 * np.eye(m + 1), rhs)
        c = sol[:m]
        c = np.maximum(c, 0)
        cs = c.sum()
        if cs < 1e-10:
            c = np.ones(m) / m
        else:
            c /= cs
        return c @ X
    except Exception:
        return X[-1].copy()


def lp_alternating_anderson(a0, time_limit, solver, m_anderson=6):
    """LP alternating with Anderson acceleration."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()

    iterates = [a.copy()]
    residuals = []
    prev_b = None

    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, m_val = solve_lp_response(a, solver)
        if b_new is None:
            continue

        # Anderson acceleration
        if len(iterates) >= 3:
            residual = a - b_new
            residuals.append(residual)
            if len(residuals) > m_anderson:
                residuals.pop(0)
            a_new = anderson_mix(iterates, residuals, m_anderson)
            a_new = normalize(a_new)
        else:
            a_new = b_new.copy()

        # Check improvement
        score = evaluate_local(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        # Update state
        iterates.append(a_new.copy())
        if len(iterates) > m_anderson + 2:
            iterates.pop(0)

        a = a_new

        if (it + 1) % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP+AA iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} lp_val={m_val:.8f}")

    return best_a, best_score


def adam_lse(a0, time_limit, T_schedule, lr=0.003, top_k_frac=0.1):
    """Adam optimization with log-sum-exp smoothing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    z -= z.mean()
    t0 = time.monotonic()
    deadline = t0 + time_limit

    best_a = normalize(a0.copy())
    best_score = evaluate_local(best_a)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step_count = 0

    total_steps = max(10000, int(time_limit * 30000))
    steps_per_stage = total_steps // len(T_schedule)

    for stage_idx, T in enumerate(T_schedule):
        if time.monotonic() > deadline:
            break
        top_k = max(5, int(top_k_frac * (2 * n - 1)))

        for step in range(steps_per_stage):
            if time.monotonic() > deadline:
                break

            a = np.exp(z - z.max())
            s = a.sum()
            if s < 1e-300:
                a = np.ones(n) / n
            else:
                a /= s

            b = fast_convolve(a)

            if T < 0.01:
                max_b = b.max()
                threshold = max_b * (1.0 - 1e-5)
                active = np.where(b >= threshold)[0]
                if len(active) == 0:
                    active = np.array([np.argmax(b)])
                w = np.zeros(len(b))
                w[active] = 1.0 / len(active)
            else:
                if top_k < len(b):
                    idx = np.argpartition(b, -top_k)[-top_k:]
                    b_top = b[idx]
                    max_top = b_top.max()
                    exp_top = np.exp((b_top - max_top) / T)
                    w_top = exp_top / exp_top.sum()
                    w = np.zeros(len(b))
                    w[idx] = w_top
                else:
                    max_b = b.max()
                    exp_all = np.exp((b - max_b) / T)
                    w = exp_all / exp_all.sum()

            # Gradient: d/dz of 2n * lse(b) where b = conv(a,a)
            # dLSE/da = (2/T) * correlate(w, a) [appropriately indexed]
            g_a = np.correlate(w, a, mode='full')
            g_a = g_a[n - 1:2 * n - 1] * 2.0

            # Chain rule to z-space
            S = np.dot(g_a, a)
            grad_z = a * (g_a - S)

            step_count += 1
            lr_eff = lr / (1.0 + 0.05 * step / steps_per_stage)
            m_buf = beta1 * m_buf + (1 - beta1) * grad_z
            v_buf = beta2 * v_buf + (1 - beta2) * grad_z ** 2
            m_hat = m_buf / (1 - beta1 ** step_count)
            v_hat = v_buf / (1 - beta2 ** step_count)
            z = z - lr_eff * m_hat / (np.sqrt(v_hat) + eps)
            z -= z.mean()

            if (step + 1) % 5000 == 0:
                cur_a = np.exp(z - z.max())
                cs = cur_a.sum()
                if cs > 1e-15:
                    cur_a /= cs
                    sc = evaluate_local(cur_a)
                    if sc < best_score:
                        best_score = sc
                        best_a = cur_a.copy()

    # Final evaluation
    a_final = np.exp(z - z.max())
    sf = a_final.sum()
    if sf > 1e-15:
        a_final /= sf
        sc = evaluate_local(a_final)
        if sc < best_score:
            best_score = sc
            best_a = a_final.copy()

    return best_a, best_score


def lbfgs_refine(a0, time_limit, T=0.1):
    """L-BFGS-B refinement on log-sum-exp objective."""
    n = len(a0)
    z0 = np.log(np.maximum(a0, 1e-300))
    z0 -= z0.mean()
    t0 = time.monotonic()

    best_a = normalize(a0.copy())
    best_score = evaluate_local(best_a)

    top_k = max(5, int(0.05 * (2 * n - 1)))

    def objective(z):
        a = np.exp(z - z.max())
        s = a.sum()
        if s < 1e-300:
            return 1e10
        a = a / s
        b = fast_convolve(a)
        if top_k < len(b):
            idx = np.argpartition(b, -top_k)[-top_k:]
            b_top = b[idx]
            max_top = b_top.max()
            exp_top = np.exp((b_top - max_top) / T)
            lse = max_top + T * np.log(exp_top.sum())
        else:
            max_b = b.max()
            exp_all = np.exp((b - max_b) / T)
            lse = max_b + T * np.log(exp_all.sum())
        return 2.0 * n * lse

    def gradient(z):
        a = np.exp(z - z.max())
        s = a.sum()
        if s < 1e-300:
            return np.zeros(n)
        a = a / s
        b = fast_convolve(a)
        if top_k < len(b):
            idx = np.argpartition(b, -top_k)[-top_k:]
            b_top = b[idx]
            max_top = b_top.max()
            exp_top = np.exp((b_top - max_top) / T)
            w_top = exp_top / exp_top.sum()
            w = np.zeros(len(b))
            w[idx] = w_top
        else:
            max_b = b.max()
            exp_all = np.exp((b - max_b) / T)
            w = exp_all / exp_all.sum()
        g_a = np.correlate(w, a, mode='full')
        g_a = g_a[n - 1:2 * n - 1] * 2.0
        S = np.dot(g_a, a)
        return a * (g_a - S)

    def time_callback():
        return time.monotonic() - t0 > time_limit

    # Use scipy L-BFGS-B with a stopping criterion based on time
    best_z = z0.copy()
    best_obj = objective(z0)

    options = {
        'maxiter': 500,
        'maxfun': 2000,
        'ftol': 1e-15,
        'gtol': 1e-12,
        'maxls': 20,
    }

    try:
        res = minimize(objective, z0, jac=gradient, method='L-BFGS-B',
                       options=options)
        if time.monotonic() - t0 < time_limit:
            z_new = res.x - res.x.mean()
            obj_new = objective(z_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_z = z_new
    except Exception as e:
        print(f"    L-BFGS-B error: {e}")

    a_final = np.exp(best_z - best_z.max())
    sf = a_final.sum()
    if sf > 1e-15:
        a_final /= sf
        sc = evaluate_local(a_final)
        if sc < best_score:
            best_score = sc
            best_a = a_final.copy()

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def perturb_and_optimize(a, rng, time_limit, solver, strength):

    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = lp_alternating_anderson(best_seq, t_alloc, solver, m_anderson=6)
        local_update(a_lp, "aa_lp_1")

    # === Phase 2: Adam log-sum-exp (20% time) ===
    print(f"\n=== Phase 2: Adam LSE (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        T_schedule = [10.0, 5.0, 3.0, 2.0, 1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.01]
        a_adam, _ = adam_lse(best_seq, t_alloc, T_schedule, lr=0.004, top_k_frac=0.15)
        local_update(a_adam, "adam_1")

    # === Phase 3: Second LP round with smaller Anderson window (15% time) ===
    print(f"\n=== Phase 3: AA-LP Round 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = lp_alternating_anderson(best_seq, t_alloc, solver, m_anderson=4)
        local_update(a_lp2, "aa_lp_2")

    # === Phase 4: L-BFGS-B fine-tuning (10% time) ===
    print(f"\n=== Phase 4: L-BFGS-B (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lbfgs, _ = lbfgs_refine(best_seq, t_alloc, T=0.1)
        local_update(a_lbfgs, "lbfgs_1")

    # === Phase 5: Fine Adam with very low temperature (10% time) ===
    print(f"\n=== Phase 5: Fine Adam (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        T_fine = [0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005]
        a_adam2, _ = adam_lse(best_seq, t_alloc, T_fine, lr=0.002, top_k_frac=0.05)
        local_update(a_adam2, "adam_fine")

    # === Phase 6: Try n=1500 (10% time) ===
    print(f"\n=== Phase 6: n=1500 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        t_n1500 = min(rem * 0.10, 150.0)
        print(f"  Budget: {t_n1500:.0f}s")
        init_1500 = interpolate_sequence(best_seq, 1500)
        a_1500_lp, _ = lp_alternating_anderson(init_1500, t_n1500 * 0.6, solver, m_anderson=5)
        local_update(a_1500_lp, "lp_1500")
        rem = deadline - time.monotonic()
        if rem > 30:
            a_1500_adam, _ = adam_lse(a_1500_lp, min(rem * 0.4, 60),
                                       [2.0, 1.0, 0.5, 0.2, 0.1, 0.05],
                                       lr=0.003, top_k_frac=0.1)
            local_update(a_1500_adam, "adam_1500")

    # === Phase 7: Perturbation escapes (remaining time) ===
    print(f"\n=== Phase 7: Perturbations (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        n_trials = min(5, int(rem // 30))
        for trial in range(n_trials):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 20:
                break
            strength = 0.005 + 0.005 * trial
            t_sub = min(rem_sub * 0.3, 50.0)
            print(f"  Perturbation trial {trial} (strength={strength:.4f}, budget={t_sub:.0f}s)")
            a_pert = perturb_and_optimize(best_seq, rng, t_sub, solver, strength)
            local_update(a_pert, f"pert_{trial}")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.5, 40.0)
        a_fin, _ = adam_lse(best_seq, t_fin,
                            [0.3, 0.1, 0.05, 0.02, 0.01, 0.005],
                            lr=0.001, top_k_frac=0.05)
        local_update(a_fin, "final_adam")

    rem = deadline - time.monotonic()
    if rem > 15:
        a_fin_lp, _ = lp_alternating_anderson(best_seq, min(rem * 0.6, 20.0), solver, m_anderson=3)
        local_update(a_fin_lp, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = min(best_score, fs)
        final_eval = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
        print(f"\nFINAL SCORE (local): {best_score:.8f} (n={len(best_seq)})")
        print(f"FINAL SCORE (evaluate_sequence): {final_eval:.8f}")
    else:
        best_seq = np.ones(1200) / 1200.0

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
