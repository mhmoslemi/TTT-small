def emd_optimize(a0, time_limit, eta_scale=0.1):
    """Exact Mirror Descent: use exact argmax subgradient."""
    n = len(a0)
    a = np.maximum(a0, 0).copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    
    avg_a = np.zeros(n)
    n_avg = 0
    
    t0 = time.monotonic()
    eta0 = eta_scale * n
    
    for t in range(1, 10_000_000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(a)
        k_star = np.argmax(b)
        
        # Gradient: g[i] = 2 * a[k_star - i] (with a_x = 0 for x out of [0,n-1])
        # This is just a shifted/reversed version of a
        g = np.zeros(n)
        # g[i] = 2 * a[k_star - i] if 0 <= k_star - i < n
        # i.e., k_star - i in [0, n-1], so i in [k_star - n + 1, k_star]
        # and i in [0, n-1]
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        for i in range(i_start, i_end + 1):
            g[i] = 2.0 * a[k_star - i]
        
        # Actually, for efficiency, let's just index directly:
        # g[i] = 2 * a[k_star - i] for valid i
        # This is 2 * a[k_star - i_start : k_star - i_end + 1][::-1]... 
        # Let me just do it vectorized:
        idx = np.arange(i_start, i_end + 1)
        g[idx] = 2.0 * a[k_star - idx]
        
        # Step size
        eta = eta0 / (1.0 + t / 1000.0)
        
        # Multiplicative update
        # a_new[i] = a[i] * exp(-eta * g[i])
        # For numerical stability:
        log_update = np.log(np.maximum(a, 1e-300)) - eta * g
        log_update -= log_update.max()
        a_new = np.exp(log_update)
        a_new = a_new / a_new.sum()
        
        a = a_new
        avg_a += a
        n_avg += 1
        
        if n_avg % 1000 == 0:
            avg_check = avg_a / n_avg
            score = 2.0 * n * fast_autocorr(avg_check).max()
            if score < best_score:
                best_score = score
                best_a = avg_check.copy()
            
            # Also check current a
            score_a = 2.0 * n * fast_autocorr(a).max()
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()
    
    return best_a, best_score
