- Find the set of active constraints: `A = {k : (a*a)[k] = f(a)}` (or near-active: `(a*a)[k] >= f(a) - ε`)
- Compute weights: `w_k = (a*a)[k] / sum_{k' in A} (a*a)[k']` (or uniform on active)

APSG with Nesterov acceleration:

    beta_{t-1} = (t-1) / (t+1 + 2/sqrt(alpha))  # or 2/(t+2) for simple Nesterov

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_gradient(w, a):
    """
    Compute grad[i] = sum_k w[k] * a[k-i] for i=0..n-1.
    Uses FFT: grad = conv(reversed(w), a)[0:n][::-1]
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    w_rev = np.asarray(w, dtype=np.float64)[::-1]
    W = rfft(w_rev, fft_size)
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    C = irfft(W * A, fft_size)[:n]
    return C[::-1]


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def spectral_init(n):
    """
    Compute the 'ideal' (possibly negative) sequence whose square has
    nearly flat convolution, then project to non-negative simplex.
    """
    M = 2 * n - 1
    # Evaluate sqrt of Dirichlet kernel on frequency grid
    N_fft = 1 << (4 * M).bit_length()
    omega = 2 * np.pi * np.arange(N_fft) / N_fft
    # Dirichlet kernel: sin(M*omega/2) / sin(omega/2)
    # Avoid singularity at omega=0
    D = np.ones(N_fft)
    mask = np.abs(omega) > 1e-12
    D[mask] = np.sin(M * omega[mask] / 2.0) / np.sin(omega[mask] / 2.0)
    D = np.maximum(D, 0)  # Should be non-negative for odd M
    sqrt_D = np.sqrt(D)
    # Inverse FFT to get time-domain
    a_full = np.real(ifft(sqrt_D))
    # The peak should be at index 0 (DC) and symmetric
    # We need to center it: the convolution of our sequence has length 2n-1
    # The ideal a has its "center" at the middle of the convolution
    # a_full is periodic with period N_fft, centered at 0
    # We want a sequence of length n
    # The convolution a*a should have length 2n-1, centered at n-1
    # So a should be centered at position 0 (in the DFT sense)
    # Extract n coefficients centered at 0
    start = (N_fft - n) // 2
    a = a_full[start:start + n]
    # Make non-negative and normalize
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def multiplicative_weights(a0, time_limit, max_steps=1000000, eta0=0.05):
    """Fast multiplicative weights with adaptive backtracking step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        # Active set: coefficients within 0.5% of max
        threshold = max_b * (1.0 - 1e-3)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        # Weighted subgradient
        w_active = b[active] / b[active].sum()
        g = np.zeros(n)
        for w, k in zip(w_active, active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += w * a[k - idx]
        # Multiplicative update
        log_mult = -eta * g
        log_mult -= log_mult.max()
        a_new = a * np.exp(log_mult)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.02, 0.5)
        else:
            eta *= 0.85
            if eta < 1e-10:
                # Reset with perturbation
                eta = eta0 * 0.5
                a = a + 0.01 * np.random.default_rng(step).random(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
    return best_a, best_score


def apsg_refine(a0, time_limit, max_iters=500000):
    """Accelerated projected subgradient with Nesterov momentum and FFT gradients."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        # Nesterov momentum
        beta = 2.0 / (it + 2.0)
        y = (1.0 - beta) * y + beta * a
        y = project_simplex(y)

        # Compute subgradient using FFT
        b = fast_convolve(y)
        max_b = b.max()
        # Use exponential weights on near-active constraints
        scaled = (b - max_b) / (max_b * 0.01 + 1e-30)
        w = np.exp(np.minimum(scaled, 0))
        sw = w.sum()
        if sw < 1e-30:
            w = np.zeros(len(b))
            w[np.argmax(b)] = 1.0
        else:
            w /= sw

        # Use FFT-based gradient (full weight vector for speed)
        g = fast_gradient(w, y)

        # Step size
        eta = 1.0 / np.sqrt(it + 1.0)
        a_new = y - eta * g
        a_new = project_simplex(a_new)

        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = a_new  # Still update for momentum
    return best_a, best_score


def build_conv_matrix_sparse(fixed_seq):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(fixed_seq, b)."""
    n = len(fixed_seq)
    size = 2 * n - 1
    rows = np.zeros(n * n, dtype=np.int32)
    cols = np.zeros(n * n, dtype=np.int32)
    vals = np.zeros(n * n, dtype=np.float64)
    count = 0
    for j in range(n):
        for k in range(j, min(j + n, size)):
            rows[count] = k
            cols[count] = j
            vals[count] = fixed_seq[k - j]
            count += 1
    return csr_matrix((vals[:count], (rows[:count], cols[:count])), shape=(size, n))


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS, cp.MOSEK]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver, T_mat, n_constraints=None):
    """
    Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k].
    Optionally use only top n_constraints binding constraints.
    """
    n = len(a_fixed)
    size = 2 * n - 1

    if n_constraints is not None and n_constraints < size:
        # Use only top constraints
        b_uniform = np.ones(n) / n
        conv_u = T_mat @ b_uniform
        top_idx = np.argsort(conv_u)[::-1][:n_constraints]
        T_sub = T_mat[top_idx, :]
        n_c = len(top_idx)
    else:
        T_sub = T_mat
        n_c = size

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T_sub @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s, max_time=30)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                break
        except Exception:
            continue
    else:
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None, None
    b_val = np.maximum(b_val, 0)
    ss = b_val.sum()
    if ss < 1e-15:
        return None, None
    b_val = b_val / ss
    return b_val, float(max(0, M_var.value if M_var.value is not None else 0))

                break

        alpha = 2.0 / (it + 2.0)

def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T_temp
    scaled_shifted = scaled - scaled.max()
    exp_terms = np.exp(scaled_shifted)
    lse = max_b + T_temp * (scaled.max() + np.log(exp_terms.sum() + 1e-300))
    obj = 2.0 * n * lse
    w = exp_terms / (exp_terms.sum() + 1e-300)
    # Gradient via FFT
    grad_a = fast_gradient(w, a)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


class LSEFunc:
    def __init__(self, n, T, deadline):
        self.n = n
        self.T = T
        self.deadline = deadline

    def __call__(self, z):
        if time.monotonic() > self.deadline:
            return 1e15
        obj, _ = logsumexp_obj_grad(z, self.n, self.T)
        return obj


class LSEJac:
    def __init__(self, n, T, deadline):
        self.n = n
        self.T = T
        self.deadline = deadline

    def __call__(self, z):
        if time.monotonic() > self.deadline:
            return np.zeros(len(z))
        _, g = logsumexp_obj_grad(z, self.n, self.T)
        return g


def lse_lbfgs_refine(a0, time_limit, T_stages=None, maxiter_per_stage=600):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    if T_stages is None:
        T_stages = [20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        func = LSEFunc(n, stage_T, deadline)
        jac = LSEJac(n, stage_T, deadline)
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-25,
                         'gtol': 1e-18, 'maxfun': maxiter_per_stage * 200,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass
    return softmax_from_z(z)


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Fast local evaluation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def perturb_sequence(a, rng, strength=0.02):
    """Create a perturbed version with structured frequency components."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq1 = rng.integers(1, 10)
    freq2 = rng.integers(2, 15)
    p1 = rng.uniform(0, 2 * np.pi)
    p2 = rng.uniform(0, 2 * np.pi)
    pert = strength * (np.cos(freq1 * x + p1) + 0.5 * np.sin(freq2 * x + p2))
    a_pert = np.maximum(a + pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses spectral init, adaptive MW, APSG, LP alternating, and LSE-LBFGS.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}, budget: {budget_s:.0f}s")

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = spectral_init(n_init)
        best_score = evaluate_local(best_seq)
        print(f"Spectral Init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 0: Spectral init + quick optimize (may improve over parent) ===
    t_spectral = min(deadline - time.monotonic(), 30.0)
    if t_spectral > 5:
        print(f"\n=== Phase 0: Spectral init n={n_best} ===")
        a_spec = spectral_init(n_best)
        sc_spec = evaluate_local(a_spec)
        print(f"  Spectral raw score: {sc_spec:.8f}")
        if sc_spec < best_score:
            a_mw_spec, _ = multiplicative_weights(a_spec, min(t_spectral * 0.5, 15.0), max_steps=50000)
            try_update(a_mw_spec, "spectral_mw")
        else:
            a_mw_p, _ = multiplicative_weights(best_seq, min(t_spectral * 0.7, 20.0), max_steps=50000)
            try_update(a_mw_p, "parent_mw0")

    # === Phase 1: MW optimization ===
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.10, 120.0)
    print(f"\n=== Phase 1: MW n={len(best_seq)} score={best_score:.8f} budget={t1:.0f}s ===")
    if t1 > 10:
        a_mw, _ = multiplicative_weights(best_seq, t1, max_steps=1000000, eta0=0.08)
        try_update(a_mw, "mw_main")

    # === Phase 2: APSG ===
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.15, 200.0)
    print(f"\n=== Phase 2: APSG score={best_score:.8f} budget={t2:.0f}s ===")
    if t2 > 15:
        a_apsg, _ = apsg_refine(best_seq, t2, max_iters=500000)
        try_update(a_apsg, "apsg")

    # === Phase 3: LP alternating (main workhorse) ===
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.30, 450.0)
    print(f"\n=== Phase 3: LP alternating score={best_score:.8f} budget={t3:.0f}s ===")
    if t3 > 30:
        # Use constraint generation: start with fewer constraints
        n_cons = min(40, 2 * len(best_seq) - 1)
        a_lp, _ = lp_alternating(best_seq, t3, solver, n_max_iters=500000, n_cons=n_cons)
        try_update(a_lp, "lp_main")

    # === Phase 4: Second LP round with more constraints ===
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.15, 200.0)
    print(f"\n=== Phase 4: LP full constraints score={best_score:.8f} budget={t4:.0f}s ===")
    if t4 > 30:
        a_lp2, _ = lp_alternating(best_seq, t4, solver, n_max_iters=2
