```python
def solve_lp_fixed_a(a, n):
    """Given fixed a (length n, sum=1), find b minimizing max(conv(a,b))."""
    # Build constraint matrix
    # Variables: [b_0, ..., b_{n-1}, M]
    # Minimize M: c = [0,...,0,1]
    
    # Inequality constraints: A_ub @ x <= b_ub
    # Part 1: conv(a,b)[k] - M <= 0 for k=0,...,2n-2
    # Part 2: -b_i <= 0 for i=0,...,n-1
    
    # Build sparse A_ub
    rows = []
    cols = []
    vals = []
    
    # Part 1: convolution constraints
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            # b[j] appears with coefficient a[k-j]
            rows.append(k)
            cols.append(j)
            vals.append(a[k-j])
        # M appears with coefficient -1
        rows.append(k)
        cols.append(n)  # index of M
        vals.append(-1.0)
    
    # Part 2: -b_i <= 0
    for i in range(n):
        rows.append(2*n-1+i)
        cols.append(i)
        vals.append(-1.0)
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(3*n-1, n+1))
    b_ub = np.zeros(3*n-1)
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    c = np.zeros(n+1)
    c[n] = 1.0  # minimize M
    
    bounds = [(0, None)] * n + [(None, None)]  # b >= 0, M free
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs', options={'presolve': True})
    
    if res.success:
        b = np.maximum(res.x[:n], 0)
        s = b.sum()
        if s > 1e-15:
            b = b / s
        M = res.x[n]
        return b, M
    return None, None
```

"""
Multi-phase optimization of 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast LP alternating (scipy HiGHS), Adam logsumexp in log-space,
projected subgradient, multiplicative weights, and multi-length exploration.
Key improvements: scipy-based LP (faster than cvxpy), active-set reduction,
direct subgradient polish, U-shaped/bimodal initializations, n up to 2000.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
from scipy.sparse import coo_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_score_scaled(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def eval_score_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    """Project vector onto probability simplex {x>=0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_constraint(a, n):
    """Build sparse (2n-1 + n) x (n+1) constraint matrix for LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(float(a[k - j]))
        rows.append(k)
        cols.append(n)
        vals.append(-1.0)
    for i in range(n):
        rows.append(2 * n - 1 + i)
        cols.append(i)
        vals.append(-1.0)
    A = coo_matrix((vals, (rows, cols)), shape=(3 * n - 1, n + 1))
    return A.tocsr()


def solve_lp_response(a_fixed):
    """Given fixed a (length n, sum=1), find b minimizing max(conv(a,b))."""
    n = len(a_fixed)
    A_ub = build_conv_constraint(a_fixed, n)
    b_ub = np.zeros(3 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    c = np.zeros(n + 1)
    c[n] = 1.0
    bounds = [(0, None)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success and res.x is not None:
            b = np.maximum(res.x[:n], 0.0)
            s = b.sum()
            if s > 1e-15:
                b = b / s
            return b
    except Exception:
        pass
    return None


def lp_alternate_nesterov(a0, time_limit, stall_threshold=30):
    """Nesterov dual averaging LP alternating."""
    n = len(a0)
    a = normalize(a0.copy())
    y = a.copy()
    best_a = a.copy()
    best_score = eval_score_normalized(a)
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(12345)
    n_iters = 0

    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        b_new = solve_lp_response(y)
        if b_new is None:
            continue
        n_iters += 1

        alpha = 2.0 / (it + 2.0)
        y = (1.0 - alpha) * y + alpha * b_new
        y = normalize(y)

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / ((it + 1) + alpha * (it + 2))
        a = normalize(a)

        score_a = eval_score_normalized(a)
        score_y = eval_score_normalized(y)

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 20 == 0:
            if best_score < prev_best - 1e-12:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= stall_threshold:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 6)
                noise = 0.02 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = normalize(y)
                a = a + 0.5 * noise
                a = normalize(a)
                stall_count = 0

            if (it + 1) % 200 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP-Alt iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def adam_logsumexp(a0, time_limit, T_stages, lr_init=0.004, lr_decay=0.85, top_k_frac=0.15):
    """Adam optimization in log-space with temperature annealing."""
    n = len(a0)
    a_init = normalize(a0)
    z = np.log(np.maximum(a_init, 1e-300))
    z -= z.mean()
    t0 = time.monotonic()

    best_a = a_init.copy()
    best_score = eval_score_normalized(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step_count = 0

    for stage_idx, T_stage in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        steps_this_stage = max(100, int(remaining * 60000 / max(1, len(T_stages) - stage_idx)))

        top_k = max(5, int(top_k_frac * (2 * n - 1)))

        for step in range(steps_this_stage):
            elapsed = time.monotonic() - t0
            if elapsed > time_limit:
                break

            a = np.exp(z - z.max())
            a = a / a.sum()
            b = fast_convolve(a)
            max_b = b.max()
            score = 2.0 * n * max_b
            step_count += 1

            if score < best_score:
                best_score = score
                best_a = a.copy()

            if T_stage < 0.01:
                threshold = max_b * (1.0 - 5e-5)
                active = np.where(b >= threshold)[0]
                if len(active) == 0:
                    active = np.array([np.argmax(b)])
                w = np.zeros(len(b))
                w[active] = 1.0 / len(active)
            else:
                if top_k < len(b):
                    idx = np.argpartition(b, -top_k)[-top_k:]
                    b_top = b[idx]
                    max_top = b_top.max()
                    exp_top = np.exp((b_top - max_top) / T_stage)
                    w_top = exp_top / exp_top.sum()
                    w = np.zeros(len(b))
                    w[idx] = w_top
                else:
                    exp_all = np.exp((b - max_b) / T_stage)
                    w = exp_all / exp_all.sum()

            g_a_raw = np.correlate(w, a, mode='full')
            g_a = g_a_raw[n - 1:2 * n - 1] * 2.0

            S = np.dot(g_a, a)
            grad_z = a * (g_a - S)

            lr = lr_init * (lr_decay ** stage_idx)
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * grad_z ** 2
            m_hat = m / (1.0 - beta1 ** (step + 1))
            v_hat = v / (1.0 - beta2 ** (step + 1))
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            z -= z.mean()

            if (step + 1) % 10000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    Adam s{stage_idx}(T={T_stage:.3f}) st{step+1} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def init_u_shaped(n, rng):
    """U-shaped initialization: high at edges, low in middle."""
    x = np.linspace(0, 1, n)

def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a_norm = a / a.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0)
    return normalize(a_new)
