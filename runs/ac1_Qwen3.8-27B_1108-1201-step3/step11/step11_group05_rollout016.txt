from scipy.fft import rfft, irfft

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS

            if s > 1e-15:
                b_val /= s
                return b_val
    except Exception:
        pass
    return None

def nesterov_lp(a0, time_limit, solver):

    s = a.sum()
    a = a / s

        y /= y.sum()
        
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))

            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
    
    return best_a, best_score

def lp_best_response(a_fixed, solver):
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            b_val[np.isnan(b_val)] = 0
            s = b_val.sum()
            if s > 1e-15:
                b_val /= s
                return b_val
    except Exception:
        pass
    return None

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for it in range(10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new = lp_best_response(y, solver)
        if b_new is None:
            if verbose and it < 10:
                print("  LP failed, skipping")
            continue
        
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = a.sum()
        y = y / sy
        
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
            sa = y.sum()
        a = a / sa
        
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if verbose and (it + 1) % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  NesterovLP iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} a_score={score_a:.8f} y_score={score_y:.8f}")
    
    return best_a, best_score

def adam_refine(a0, time_limit, T_stages=None, lr_init=0.003, lr_decay=0.85, top_k_frac=0.1, verbose=True):
    n = len(a0)
    if T_stages is None:
        T_stages = [5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005]
    
    z = np.log(np.maximum(a0, 1e-300))
    z -= z.mean()
    
    t0 = time.monotonic()
    deadline = t0 + time_limit
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    step_count = 0
    total_stages = len(T_stages)
    
    for stage_idx, T_stage in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 1:
            break
        
        top_k = max(5, int(top_k_frac * (2 * n - 1)))
        steps_per_stage = max(1000, int(remaining * 10000 / (total_stages - stage_idx)))
        
        for step in range(steps_per_stage):
            if time.monotonic() > deadline:
                break
            
            # Compute a from z
            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)
            s_exp = exp_z.sum()
            if s_exp < 1e-300:
                a = np.ones(n) / n
            else:
                a = exp_z / s_exp
            
            b = fast_convolve(a)
            max_b = b.max()
            score = 2.0 * n * max_b
            step_count += 1
            
            if score < best_score:
                best_score = score
                best_a = a.copy()
            
            # Compute gradient using weighted peaks
            if T_stage < 0.01:
                threshold = max_b * (1.0 - 1e-4)
                active = np.where(b >= threshold)[0]
                if len(active) == 0:
                    active = np.array([np.argmax(b)])
                w = np.zeros(len(b))
                w[active] = 1.0 / len(active)
            else:
                if top_k < len(b):
                    idx = np.argpartition(b, -top_k)[-top_k:]
                    b_top = b[idx]
                    max_top = b_top.max()
                    exp_top = np.exp((b_top - max_top) / T_stage)
                    w_top = exp_top / exp_top.sum()
                    w = np.zeros(len(b))
                    w[idx] = w_top
                else:
                    exp_all = np.exp((b - max_b) / T_stage)
                    w = exp_all / exp_all.sum()
            
            # Gradient: g_a[i] = 2 * sum_k w[k] * a[k-i] (correlation)
            g_a_raw = np.correlate(w, a, mode='full')
            g_a = g_a_raw[n - 1:2 * n - 1] * 2.0
            
            # z-gradient
            S = np.dot(g_a, a)
            grad_z = a * (g_a - S)
            
            # Adam update
            progress = step / max(1, steps_per_stage)
            lr = lr_init * (lr_decay ** stage_idx) / (1.0 + 0.5 * progress)
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** (step + 1))
            v_hat = v / (1 - beta2 ** (step + 1))
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            z -= z.mean()
            
            if verbose and (step + 1) % 5000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    Adam stage {stage_idx} (T={T_stage:.3f}) step {step+1} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))
