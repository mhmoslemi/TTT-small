def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_fn(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s


def weighted_convolve(a, peaks, weights):
    """
    Compute smoothed subgradient: sum_k weights[k] * 2*a[k-j] for valid k-j.
    Returns vector g where g[j] = sum_k weights[k] * 2*a[k-j]*1_{0<=k-j<n}
    """
    n = len(a)
    g = np.zeros(n)
    for k, w in zip(peaks, weights):
        k = int(k)
        if w < 1e-15:
            continue
        # g[j] += w * 2 * a[k-j] for j where 0 <= k-j < n
        # i.e., j ranges from max(0, k-(n-1)) to min(n-1, k)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            idx = np.arange(j_lo, j_hi + 1)
            g[j_lo:j_hi + 1] += 2.0 * w * a[k - idx]
    return g

def lp_best_response(a_fixed, time_budget=5.0):
    """Solve LP: min_b max_k conv(a_fixed, b) s.t. sum(b)=1, b>=0."""
    try:
        import cvxpy as cp
        n = len(a_fixed)
        size = 2 * n - 1
        
        # Build convolution matrix: T @ b = conv(a_fixed, b)
        # T[k, j] = a_fixed[k-j] if 0 <= k-j < n, else 0
        b_var = cp.Variable(n, nonneg=True)
        t_var = cp.Variable()
        
        # Build constraints using the structure
        # conv(a,b)[k] = sum_j a[k-j]*b[j] for valid k-j
        T = np.zeros((size, n))
        for j in range(n):
            k_lo = j
            k_hi = j + n - 1
            T[k_lo:k_hi+1, j] = a_fixed[n-1::-1][:n]  # a_fixed reversed
            # Actually: T[k,j] = a_fixed[k-j] if 0<=k-j<n
            # For fixed j: k-j ranges from 0 to n-1 as k goes from j to j+n-1
            # T[k,j] = a_fixed[k-j]
        
        # Rebuild properly
        T = np.zeros((size, n))
        for j in range(n):
            for k in range(j, j + n):
                if k < size:
                    T[k, j] = a_fixed[k - j]
        
        T_sparse = cp.csr_matrix(T)
        constraints = [T_sparse @ b_var <= t_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        
        for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver_name)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    b_val = np.asarray(b_var.value, dtype=np.float64)
                    b_val = np.maximum(b_val, 0)
                    s = b_val.sum()
                    if s > 1e-15:
                        b_val = b_val / s
                        return b_val
            except Exception:
                continue
    except Exception:
        pass
    return None


def logsumexp_obj_grad(a, T_param):
    """
    Log-sum-exp smoothed objective and gradient.
    f_smooth(a) = (2n/T) * logsumexp(conv(a,a)/T)
    """
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp(b / T_param - max_b / T_param)
    w = exp_b / exp_b.sum()
    
    # Gradient: df/da[j] = (2n/T) * sum_k w[k] * 2*a[k-j] * 1_{valid}
    g = weighted_convolve(a, np.arange(len(b)), w)
    g *= (2.0 * n) / T_param
    
    obj = (2.0 * n / T_param) * (max_b / T_param + np.log(exp_b.sum()))
    return obj, g

        eta *= lr_decay
        
        # Adaptive: increase beta (sharper peak selection) over time
        beta = min(beta * 1.0001, 80.0)
        
        # Perturbation if stuck
        if no_improve > 5000:
            a = a + rng.standard_normal(n) * 0.003
            a = normalize(a)
            no_improve = 0
            eta = 0.05
        
        if (t + 1) % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  MW t={t+1} ({elapsed:.0f}s): best={best_score:.8f}, eta={eta:.5f}")
    
    return best_a, best_score


def accelerated_subgradient(a0, time_limit, rng=None):
    """
    Nesterov-accelerated subgradient method with extrapolation.
    Uses the smooth structure of the problem for faster convergence.
    """
    if rng is None:
        rng = np.random.default_rng()
    
    n = len(a0)
    a = normalize(a0)
    y = a.copy()  # Nesterov's extrapolated point
    best_a = a.copy()
    best_score = score_fn(a)
    t0 = time.monotonic()
    
    t_start = 0
    for t in range(200000):
        if time.monotonic() - t0 > time_limit:
            break
        t_start = t
        
        # Compute subgradient at y
        b = fast_convolve(y)
        max_b = b.max()
        k_star = int(np.argmax(b))
        
        # Single peak subgradient
        g = np.zeros(n)
        j_lo = max(0, k_star - n + 1)
        j_hi = min(n - 1, k_star)
        idx = np.arange(j_lo, j_hi + 1)
        g[j_lo:j_hi+1] = 2.0 * y[k_star - idx]
        
        # Also use smoothed version (top 3 peaks)
        top3 = np.argsort(b)[::-1][:3]
        w3 = np.exp(5.0 * (b[top3] - b[top3].max()))
        w3 = w3 / w3.sum()
        g_smooth = np.zeros(n)
        for k, w in zip(top3, w3):
            k = int(k)
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            idx = np.arange(j_lo, j_hi + 1)
            g_smooth[j_lo:j_hi+1] += w * 2.0 * y[k - idx]
        
        g = 0.5 * g + 0.5 * g_smooth
        g = g - g.mean()
        
        # Nesterov step
        alpha = 2.0 / (np.sqrt(t + 4) + 1.0)
        eta = 1.0 / (np.sqrt(t + 1) + 1.0) * 2.0
        
        # Projected step
        a_new = a - eta * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        # Nesterov extrapolation
        y_new = a_new + (alpha - 1.0) * (a_new - a)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy > 1e-15:
            y_new = y_new / sy
        else:
            y_new = a_new.copy()
        
        a = a_new
        y = y_new
        
        sc = score_fn(a)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
        
        sc_y = score_fn(y)
        if sc_y < best_score - 1e-12:
            best_score = sc_y
            best_a = y.copy()
        
        if (t + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  AccSG t={t+1} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def interpolate_to_n(a, new_n):
    """Interpolate sequence to new length, maintaining structure."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a = normalize(a)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a)
    a_interp = np.maximum(a_interp, 0)
    s = a_interp.sum()
    if s > 1e-15:
        a_interp = a_interp / s
    return a_interp


def asymmetric_init(n, rng):
    """Create asymmetric initial sequence that breaks symmetry."""
    x = np.linspace(0, 1, n)
    choice = rng.integers(0, 8)
    if choice == 0:
        # Skewed cosine
        a = np.cos(np.pi * x * rng.uniform(0.3, 0.7)) ** 2
    elif choice == 1:
        # Exponential decay
        rate = rng.uniform(1.0, 4.0)
        a = np.exp(-rate * x)
    elif choice == 2:
        # Inverse exponential
        rate = rng.uniform(1.0, 4.0)
        a = np.exp(-rate * (1 - x))
    elif choice == 3:
        # Skewed Gaussian
        center = rng.uniform(0.2, 0.8)
        width = rng.uniform(0.05, 0.3)
        a = np.exp(-((x - center) ** 2) / (2 * width ** 2))
    elif choice == 4:
        # Power law
        p = rng.uniform(0.5, 3.0)
        a = x ** p
    elif choice == 5:
        # Power law inverted
        p = rng.uniform(0.5, 3.0)
        a = (1 - x) ** p
    elif choice == 6:
        # Bimodal asymmetric
        c1, c2 = sorted(rng.uniform(0, 1, 2))
        a = np.exp(-20*(x-c1)**2) + 0.5*np.exp(-20*(x-c2)**2)
    else:
        # Linear + noise
        a = 0.5 + 0.5 * x + rng.standard_normal(n) * 0.05
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def blended_optimization(a0, time_limit, solver=None, rng=None):
    """
    Combined optimization: FTRL + periodic LP blending + MW perturbation.
    """
    if rng is None:
        rng = np.random.default_rng()
    
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = score_fn(a)
    t0 = time.monotonic()
    
    # State
    Phi = np.zeros(n)
    eta = 0.3
    beta = 3.0
    m = 12
    lp_interval = 500  # Every 500 iterations, try LP
    no_improve = 0
    prev_score = best_score
    stall_counter = 0
    
    for t in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute convolution
        b = fast_convolve(a)
        max_b = b.max()
        
        # Top-m peaks
        if m >= len(b):
            peaks = np.arange(len(b))
            top_vals = b.copy()
        else:
            top_idx = np.argsort(b)[::-1][:m]
            peaks = top_idx
            top_vals = b[top_idx]
        
        # Softmax weights
        shifted = top_vals - top_vals.max()
        exp_w = np.exp(beta * shifted)
        weights = exp_w / exp_w.sum()
        
        # Smoothed subgradient
        g = weighted_convolve(a, peaks, weights)
        g = g - g.mean()
        
        # FTRL update
        eta_t = 0.3 / (1.0 + 0.05 * np.sqrt(t + 1))
        Phi += eta_t * g
        phi_shifted = -Phi - (-Phi).max()
        a_ftrl = np.exp(phi_shifted)
        sa = a_ftrl.sum()
        if sa < 1e-15:
            a_ftrl = a.copy()
        else:
            a_ftrl = a_ftrl / sa
        
        # Periodic LP blending
        if (t + 1) % lp_interval == 0 and solver is not None:
            lp_time = min(3.0, time_limit - (time.monotonic() - t0))
            if lp_time > 1.0:
                b_lp = lp_best_response(a, time_budget=lp_time)
                if b_lp is not None and len(b_lp) == n:
                    # Blend: move toward LP solution
                    blend = 0.3
                    a_trial = (1 - blend) * a_ftrl + blend * b_lp
                    a_trial = normalize(a_trial)
                    sc_trial = score_fn(a_trial)
                    if sc_trial < best_score:
                        a = a_trial
                        best_score = sc_trial
                        best_a = a.copy()
                        no_improve = 0
                    else:
                        a = a_ftrl
                else:
                    a = a_ftrl
            else:
                a = a_ftrl
        else:
            a = a_ftrl
        
        # Evaluate
        sc = score_fn(a)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
            stall_counter = 0
        else:
            no_improve += 1
        
        # Adaptive parameters
        beta = min(beta * 1.0002, 60.0)
        
        # Restart if stuck
        if no_improve > 2000:
            # Try a random perturbation
            if no_improve % 1000 == 0:
                a = a + rng.standard_normal(n) * (0.002 + 0.001 * no_improve / 2000)
                a = normalize(a)
                Phi = Phi * 0.3
                no_improve = 0
                stall_counter += 1
                eta = 0.3
        
        if (t + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  Blend t={t+1} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def coordinate_descent_refine(a0, time_limit, max_steps=500000):
    """
    Coordinate descent: reduce the largest contributors to the peak.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = score_fn(a)
    t0 = time.monotonic()
    stall = 0
    
    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        k_star = int(np.argmax(b))
        
        # Find contributors to the peak
        j_lo = max(0, k_star - n + 1)
        j_hi = min(n - 1, k_star)
        idx = np.arange(j_lo, j_hi + 1)
        contribs = a[idx] * a[k_star - idx]
        
        # Try reducing top contributors
        n_top = min(10, len(idx))
        top_positions = idx[np.argsort(contribs)[::-1][:n_top]]
        
        improved = False
        for pos in top_positions:
            if a[pos] < 1e-10:
                continue
            for factor in [0.9999, 0.9998, 0.9995, 0.999, 0.998, 0.995, 0.99]:
                a_trial = a.copy()
                a_trial[pos] *= factor
                diff = a[pos] - a_trial[pos]
                if diff < 1e-15:
                    break
                # Redistribute to others proportionally
                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    break
                a_trial[mask] = others + diff * others / others_sum
                a_trial = normalize(a_trial)
                
                sc = score_fn(a_trial)
                if sc < best_score - 1e-13:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
                elif sc > best_score * 1.001:
                    break  # This direction is bad, try next
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall > 10000:
                # Multi-coordinate reduction
                k_star = int(np.argmax(fast_convolve(a)))
                j_lo = max(0, k_star - n + 1)
                j_hi = min(n - 1, k_star)
                idx = np.arange(j_lo, j_hi + 1)
                contribs = a[idx] * a[k_star - idx]
                top_positions = idx[np.argsort(contribs)[::-1][:5]]
                a_trial = a.copy()
                for pos in top_positions:
                    if a_trial[pos] > 1e-10:
                        a_trial[pos] *= 0.999
                a_trial = normalize(a_trial)
                sc = score_fn(a_trial)
                if sc < best_score - 1e-13:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    stall = 0
                elif stall > 50000:
                    break
        else:
            stall = 0
        
        if (step + 1) % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  CoordDesc {elapsed:.0f}s: score={best_score:.8f}")
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining FTRL, multiplicative weights,
    LP blending, log-sum-exp L-BFGS, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    # === Load parent or create initial ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = score_fn(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        n_init = 1200
        best_seq = asymmetric_init(n_init, rng)
        best_score = score_fn(best_seq)
        print(f"Generated init: n={len(best_seq)}, score={best_score:.8f}")
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = score_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.10f} (n={len(a)}) ***")
    
    # === Check LP solver availability ===
    solver = None
    try:
        import cvxpy as cp
        for s_name in [cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                x = cp.Variable(5, nonneg=True)
                prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
                prob.solve(solver=s_name)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solver = s_name
                    break
            except Exception:
                continue
        print(f"LP solver: {solver}")
    except Exception:
        print("No LP solver available")
    
    # === Phase 1: FTRL Smoothed Mirror Descent (main workhorse) ===
    print(f"\n=== Phase 1: FTRL Mirror Descent (score={best_score:.8f}, n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a1, _ = ftrl_smoothed_mirror_descent(best_seq, t_alloc, solver=solver, rng=rng)
        try_update(a1, "FTRL_p1")
    
    # === Phase 2: Multiplicative Weights (different dynamics) ===
    print(f"\n=== Phase 2: Multiplicative Weights (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a2, _ = multiplicative_weights_o(best_seq, t_alloc, rng=rng)
        try_update(a2, "MW_p2")
    
    # === Phase 3: Blended FTRL + LP ===
    print(f"\n=== Phase 3: Blended FTRL+LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a3, _ = blended_optimization(best_seq, t_alloc, solver=solver, rng=rng)
        try_update(a3, "Blend_p3")
    
    # === Phase 4: Multi-n exploration ===
    print(f"\n=== Phase 4: Multi-n Exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 30:
        n_best = len(best_seq)
        # Try nearby n values
        candidate_ns = sorted(set([
            n_best - 100, n_best + 100, n_best + 200,
            max(600, n_best - 200), n_best + 300
        ]))
        per_n = t_alloc / max(1, len(candidate_ns))
        print(f"  Trying n values: {candidate_ns}, ~{per_n:.0f}s each")
        for n_try in candidate_ns:
            if time.monotonic() > deadline - 20:
                break
            if n_try == n_best or n_try < 200:
                continue
            a_interp = interpolate_to_n(best_seq, n_try)
            t_sub = min(per_n, deadline - time.monotonic() - 10)
            if t_sub < 15:
                continue
            print(f"  n={n_try}, budget={t_sub:.0f}s")
            a_sub, _ = ftrl_smoothed_mirror_descent(a_interp, t_sub, rng=rng)
            try_update(a_sub, f"multi_n_{n_try}")
    
    # === Phase 5: Log-sum-exp L-BFGS ===
    print(f"\n=== Phase 5: LogSumExp L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 15:
        T_stages = [25.0, 15.0, 10.0, 6.0, 4.0, 2.5, 1.5, 1.0, 0.6, 0.3, 0.15, 0.08]
        print(f"  Budget: {t_alloc:.0f}s")
        a5, _ = logsumexp_lbgfs(best_seq, t_alloc, T_stages=T_stages)
        try_update(a5, "LSE_LBFGS")
    
    # === Phase 6: Accelerated subgradient ===
    print(f"\n=== Phase 6: Accelerated Subgradient (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 120.0)
    if t_alloc > 10:

    # === Phase 7: Coordinate Descent Polish ===
    print(f"\n=== Phase 7: Coordinate Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.07, 100.0)
    if t_alloc > 10:
        a7, _ = coordinate_descent_refine(best_seq, t_alloc)
        try_update(a7, "CoordDesc")
    
    # === Final verification ===
    if best_seq is not None:
        fs = score_fn(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    
    # Ensure output is valid
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return result
