import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS, cp.MOSEK, cp.GUROBI]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

        target = (1 - 0.5 * t) * geo_mean + 0.5 * t * ideal_target
        
        # Actually, let's use a simpler approach:
        # Target = geometric mean of current non-DC magnitudes
        # This naturally converges to a stationary point
        target = geo_mean

        # Apply target to spectrum
        F_new = F.copy()
        blend = 0.5  # Fixed blend for stability
        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend) * mags[k] + blend * target
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])

        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        score = evaluate_local(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a

def peak_subgradient_nesterov(a0, time_limit, n_steps=500000):
    """
    Peak-focused multiplicative subgradient with Nesterov momentum.
    Directly targets the top peaks of the autocorrelation.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    # Nesterov momentum
    y = a.copy()  # momentum variable
    v = np.zeros(n)  # velocity
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 0.1
    gamma = 0.9  # momentum coefficient
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(y)
        max_b = b.max()

        # Find top K peaks
        K = min(10, n)
        top_k = np.argsort(b)[::-1][:K]
        
        # Weight peaks by their excess over the median
        median_b = np.median(b)
        weights = np.maximum(b[top_k] - median_b, 0)
        w_sum = weights.sum()
        if w_sum < 1e-30:
            weights = np.ones(K)
            w_sum = K
        weights = weights / w_sum

        # Compute subgradient
        g = np.zeros(n)
        for i, k in enumerate(top_k):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += weights[i] * y[k - idx]

        # Nesterov update
        alpha = 2.0 / (step + 2)
        y = (1.0 - alpha) * y + alpha * a  # This is for the dual... 
        # Actually let's use a simpler momentum scheme:
        
        # Multiplicative update on y
        log_scale = -eta * g
        log_scale -= log_scale.max()
        y_new = y * np.exp(log_scale)
        y_new = np.maximum(y_new, 0)
        s_new = y_new.sum()
        if s_new < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_new

        # Momentum: blend previous y with new y
        y = gamma * y + (1 - gamma) * y_new

        # Update a as average of all y's seen (dual averaging)
        a = ((step + 1) * a + y_new) / (step + 2)
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y_new.copy()
        else:
            a = a / sa

        # Track best
        for candidate in [a, y, y_new]:
            sc = 2.0 * n * fast_convolve(candidate).max()
            if sc < best_score:
                best_score = sc
                best_a = candidate.copy()
                no_improve = 0
            else:
                no_improve += 1

        # Adaptive step size
        if no_improve >= 20:
            eta *= 0.8
            no_improve = 0
            if eta < 0.001:
                break

        if step % 50000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakSG iter={step+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score

def geometric_mean_lp_alternating(a0, time_limit, solver, n_max_iters=500000):
    """
    LP alternating with geometric mean update.
    This is the multiplicative weights version of LP alternating.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)

        a = np.ones(n) / n

    b = a.copy()  # The other variable

    t0 = time.monotonic()

    stall = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        deadline = time.monotonic() + min(60, time_limit - (time.monotonic() - t0))

        lp_times.append(time.monotonic() - lp_t0)

        if b_new is None:
            # Fallback: use subgradient step
            conv_b = fast_convolve(a)
            k_max = np.argmax(conv_b)
            js = max(0, k_max - n + 1)
            je = min(n - 1, k_max)
            idx = np.arange(js, je + 1)
            g = np.zeros(n)
            g[idx] = a[k_max - idx]
            log_s = -0.05 * g
            log_s -= log_s.max()
            b_new = b * np.exp(log_s)
            b_new = np.maximum(b_new, 0)
            sb = b_new.sum()
            if sb > 1e-15:
                b_new = b_new / sb
            else:
                continue

        # Geometric mean update
        geo_mean = np.sqrt(a * b_new)
        sg = geo_mean.sum()
        if sg < 1e-15:
            geo_mean = a.copy()
        else:
            geo_mean = geo_mean / sg

        # Also try simple average
        avg = 0.5 * (a + b_new)
        sa = avg.sum()
        if sa > 1e-15:
            avg = avg / sa

            b = b_new
        else:
            a = avg
            b = b_new

            sc = 2.0 * n * fast_convolve(candidate).max()

                a = np.maximum(a, 0)
                sa = a.sum()
                if sa > 1e-15:
                    a = a / sa
                stall = 0

            if (it + 1) % 500 == 0:
                elapsed = time.monotonic() - t0

def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=500000):
    """Nesterov dual averaging LP alternating."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        deadline_sub = time.monotonic() + min(60, time_limit - (time.monotonic() - t0))
        b_new, M_b, _ = solve_lp_best_response(y, solver, deadline_sub)

        if b_new is None:
            continue

        # Nesterov dual averaging
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 10:
                rng = np.random.default_rng(42 + it)
                x = np.linspace(0, 2 * np.pi, n)
                y = y + 0.02 * np.cos(2 * x + rng.uniform(0, 2 * np.pi))
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                stall_count = 0

            if (it + 1) % 1000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score

def logsumexp_lbgfs(a0, time_limit, deadline=None):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    from scipy.optimize import minimize
    n = len(a0)
    
    def softmax_from_z(z):
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        s = exp_z.sum()
        if s < 1e-300:
            return np.ones(len(z)) / len(z)
        return exp_z / s

    def z_from_a(a):
        return np.log(np.maximum(a, 1e-300))

    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    if deadline is None:
        deadline = t0 + time_limit

    T_stages = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
    n_stages = len(T_stages)

    for stage_i, T in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break

        stages_left = n_stages - stage_i
        avail_time = remaining
        maxiter = max(10, min(200, int(avail_time * 30 / stages_left)))

        def func(z_in):
            a_s = softmax_from_z(z_in)
            b = fast_convolve(a_s)
            max_b = b.max()
            lse = max_b + T * np.log(np.sum(np.exp((b - max_b) / T)))
            return 2.0 * n * lse

        def jac(z_in):
            a_s = softmax_from_z(z_in)
            b = fast_convolve(a_s)
            max_b = b.max()
            exp_terms = np.exp((b - max_b) / T)
            w = exp_terms / exp_terms.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_s[k - idx]
            S = np.dot(grad_a, a_s)
            return 2.0 * n * a_s * (grad_a - S)

        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                          options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-16,
                                   'maxfun': maxiter * 100, 'disp': False})
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def structured_init(n, rng):
    """Create a structured initial sequence."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    else:
        a = 1.0 - np.abs(2 * x - 1)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

Wait, I'm using a closure (`local_update`) which violates the rules. Let me restructure.

Also, let me reconsider the algorithm. The key issue is that I need to make all helper functions top-level with no closures.

Let me also think about what would actually improve the score the most:

1. The LP approach is the most precise - it finds the exact best response at each iteration. The Nesterov acceleration helps convergence.

2. The spectral approach is fast but not as precise.

3. The subgradient approach is in between.

I think the most impactful improvement would be:
- Better LP solving (faster, more reliable)
- More LP iterations (by making each one faster)
- Better initial point (from spectral + subgradient pre-processing)

For the LP, the bottleneck is building the sparse matrix and solving. For n=1200, the matrix is (2399) x 1200. This is manageable.

Let me also think about whether we can exploit the symmetry of the problem. The autocorrelation is symmetric: `b[k] = b[2n-2-k]`. So we only need constraints for k = 0, ..., n-1 (half the constraints). This could speed up the LP.

Actually, for the LP `minimize M subject to T @ b <= M, sum(b) = 1, b >= 0`, the constraint matrix T has (2n-1) rows and n columns. Due to symmetry, rows k and 2n-2-k are "mirrored" but not identical (they correspond to different lags). So we can't simply halve the constraints.

However, we CAN note that `T[k]` and `T[2n-2-k]` have a specific relationship. Let's see: `T[k, j] = a[k-j]` if `0 <= k-j < n`, else 0. And `T[2n-2-k, j] = a[2n-2-k-j]` if `0 <= 2n-2-k-j < n`.

If we set `j' = n-1-j`, then `2n-2-k-j = 2n-2-k-(n-1-j') = n-1-k+j'`. So `T[2n-2-k, j] = a[n-1-k+j']` where `j' = n-1-j`. This means `T[2n-2-k, j] = T[k, n-1-j]` (checking: `T[k, n-1-j] = a[k-(n-1-j)] = a[k-n+1+j]`... hmm that's not quite right).

import numpy as np
import cvxpy as cp
import time
from scipy.optimize import minimize as scipy_minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix_sparse(a):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    import scipy.sparse as sp
    n = len(a)
    size = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(size):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])
    T = sp.csr_matrix((vals, (rows, cols)), shape=(size, n))
    return T


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS, cp.MOSEK, cp.GUROBI]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver, deadline=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix_sparse(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.MOSEK]
    seen = set()
    for s in solvers_to_try:
        if s is None or s in seen:
            continue
        seen.add(s)
        try:
            if deadline is not None and time.monotonic() > deadline:
                return None, None
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_sum = b_val.sum()
                if s_sum < 1e-15:
                    continue
                b_val = b_val / s_sum
                return b_val, float(max(0, M_var.value))
        except Exception:
            continue
    return None, None


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def spectral_flatten_adaptive(a, n_iters=200, deadline=None):
    """
    Adaptive spectral flattening with self-correcting target.
    The target magnitude adapts based on the achieved flatness after projection.
    """
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    best_a = a.copy()
    best_score = evaluate_local(a)

    blend = 0.4
    prev_neg_fraction = 0.0

    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)
        mags = np.abs(F)
        phases = np.angle(F)

        # Geometric mean of non-DC magnitudes as target
        non_dc = mags[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        # Adaptive blend: if many negatives last time, reduce aggression
        if prev_neg_fraction > 0.3:
            blend_eff = blend * 0.5
        elif prev_neg_fraction < 0.1:
            blend_eff = blend * 1.3
        else:
            blend_eff = blend

        # Apply target
        F_new = F.copy()
        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend_eff) * mags[k] + blend_eff * target
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])

        a_new_raw = np.fft.ifft(F_new).real[:n]
        neg_fraction = np.mean(a_new_raw < 0)
        prev_neg_fraction = neg_fraction

        a_new = np.maximum(a_new_raw, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        score = evaluate_local(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def peak_subgradient_exp(a0, time_limit, n_steps=300000):
    """
    Peak-focused multiplicative subgradient with exponential peak weighting.
    Uses Nesterov momentum and adaptive step size.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 0.08
    beta = 50.0  # Exponential weighting sharpness
    K = 5  # Number of top peaks to consider
    no_improve = 0
    scale = 1.0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(y)
        max_b = b.max()
        median_b = np.median(b)

        # Top K peaks
        if K < len(b):
            top_k = np.argpartition(b, -K)[-K:]
        else:
            top_k = np.arange(len(b))

        # Exponential weighting: heavier weight on higher peaks
        excess = b[top_k] - median_b
        excess = np.maximum(excess, 1e-30)
        w = np.exp(beta * (np.log(excess) - np.log(excess.max())))
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(len(top_k)) / len(top_k)
        else:
            w = w / w_sum

        # Compute weighted subgradient
        g = np.zeros(n)
        for i, k in enumerate(top_k):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += w[i] * y[k - idx]

        # Multiplicative update
        eta_eff = eta * scale
        log_scale = -eta_eff * g
        log_scale -= log_scale.max()
        y_new = y * np.exp(log_scale)
        y_new = np.maximum(y_new, 0)
        s_new = y_new.sum()
        if s_new < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_new

        # Nesterov-like momentum
        alpha = 2.0 / (step + 2)
        y = (1.0 - alpha) * y + alpha * y_new

        # Dual averaging for a
        a = ((step + 1) * a + y_new) / (step + 2)
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y_new.copy()
        else:
            a = a / sa

        # Track best
        improved = False
        for candidate in [a, y, y_new]:
            sc = 2.0 * n * fast_convolve(candidate).max()
            if sc < best_score:
                best_score = sc
                best_a = candidate.copy()
                improved = True
        if improved:
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive step size
        if no_improve >= 30:
            scale *= 0.75
            no_improve = 0
            if scale < 0.005:
                break

        if step % 50000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakSG_exp iter={step+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score

    rng = np.random.default_rng(777)
    lp_times = []

    for it in range(n_max_iters):

            break

        if b_new is None:

                print(f"    LP too slow ({avg_lp:.1f}s/iter), stopping at {it+1}")
                break

        # Nesterov dual averaging
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))

        score_a = 2.0 * n * fast_convolve(a).max()

def logsumexp_lbgfs(a0, time_limit, deadline=None):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)

    def softmax_from_z(z_in):
        z_shifted = z_in - z_in.max()
        exp_z = np.exp(z_shifted)
        s = exp_z.sum()
        if s < 1e-300:
            return np.ones(len(z_in)) / len(z_in)
        return exp_z / s

    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    if deadline is None:
        deadline = t0 + time_limit

    T_stages = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    n_stages = len(T_stages)

    for stage_i, T in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        stages_left = n_stages - stage_i
        maxiter = max(10, min(300, int(remaining * 40 / stages_left)))

        def func(z_in):
            a_s = softmax_from_z(z_in)
            b = fast_convolve(a_s)
            max_b = b.max()
            lse = max_b + T * np.log(np.sum(np.exp((b - max_b) / T)))
            return 2.0 * n * lse

        def jac(z_in):
            a_s = softmax_from_z(z_in)
            b = fast_convolve(a_s)
            max_b = b.max()
            exp_terms = np.exp((b - max_b) / T)
            w = exp_terms / exp_terms.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_s[k - idx]
            S = np.dot(grad_a, a_s)
            return 2.0 * n * a_s * (grad_a - S)

        try:
            res = scipy_minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                                 options={'maxiter': maxiter, 'ftol': 1e-20,
                                          'gtol': 1e-16, 'maxfun': maxiter * 200,
                                          'disp': False})
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create a structured initial sequence."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    else:
        a = 1.0 - np.abs(2 * x - 1)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def try_local_update(a, current_best, current_score):
    """Try to update the best sequence. Returns (new_seq, new_score, improved)."""
    if a is None:
        return current_best, current_score, False
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return current_best, current_score, False
    a = a / s
    sc = evaluate_local(a)
    if np.isfinite(sc) and sc < current_score:
        return a.copy(), sc, True
    return current_best, current_score, False


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.

    Strategy:
    1. Load parent or create structured init
    2. Quick spectral flattening for global improvement
    3. Peak subgradient with exponential weighting for fast local improvement
    4. Geometric-mean LP alternating (multiplicative weights) for precise optimization
    5. Nesterov LP alternating for further refinement
    6. L-BFGS LogSumExp for smooth local optimization
    7. Multi-n exploration
    8. Final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)
    rem = deadline - time.monotonic()
    print(f"Budget: {rem:.0f}s, starting score: {best_score:.8f}")

    # === Phase 1: Spectral Flattening (~100s) ===
    print(f"\n--- Phase 1: Spectral Flattening ---")
    t_spec = min(rem * 0.06, 100.0)
    if t_spec > 10:
        a_spec = spectral_flatten_adaptive(best_seq, n_iters=500,
                                           deadline=time.monotonic() + t_spec)
        best_seq, best_score, improved = try_local_update(a_spec, best_seq, best_score)
        if improved:
            print(f"  Spectral improved: {best_score:.8f}")

    rem = deadline - time.monotonic()

    # === Phase 2: Peak Subgradient (~80s) ===
    print(f"--- Phase 2: Peak Subgradient (score={best_score:.8f}) ---")
    t_sg = min(rem * 0.05, 80.0)
    if t_sg > 10:
        a_sg, _ = peak_subgradient_exp(best_seq, t_sg, n_steps=200000)
        best_seq, best_score, improved = try_local_update(a_sg, best_seq, best_score)
        if improved:
            print(f"  PeakSG improved: {best_score:.8f}")

    rem = deadline - time.monotonic()

    # === Phase 3: Geometric Mean LP Alternating (~450s) ===
    print(f"--- Phase 3: GeoLP (score={best_score:.8f}) ---")
    t_geolp = min(rem * 0.25, 450.0)
    if t_geolp > 30:
        print(f"  Budget: {t_geolp:.0f}s")
        a_geo, _ = geometric_mean_lp(best_seq, t_geolp, solver, n_max_iters=500000)
        best_seq, best_score, improved = try_local_update(a_geo, best_seq, best_score)
        if improved:
            print(f"  GeoLP improved: {best_score:.8f}")

    rem = deadline - time.monotonic()

    # === Phase 4: Nesterov LP (~400s) ===
    print(f"--- Phase 4: Nesterov LP (score={best_score:.8f}) ---")
    t_nlp = min(rem * 0.24, 400.0)
    if t_nlp > 30:
        print(f"  Budget: {t_nlp:.0f}s")
        a_nlp, _ = nesterov_lp_refine(best_seq, t_nlp, solver, n_max_iters=500000)
        best_seq, best_score, improved = try_local_update(a_nlp, best_seq, best_score)
        if improved:
            print(f"  NesterovLP improved: {best_score:.8f}")

    rem = deadline - time.monotonic()

    # === Phase 5: L-BFGS LogSumExp (~250s) ===
    print(f"--- Phase 5: L-BFGS (score={best_score:.8f}) ---")
    t_lbfgs = min(rem * 0.18, 250.0)
    if t_lbfgs > 30:
        a_lb = logsumexp_lbgfs(best_seq, t_lbfgs,
                               deadline=time.monotonic() + t_lbfgs)
        best_seq, best_score, improved = try_local_update(a_lb, best_seq, best_score)
        if improved:
            print(f"  L-BFGS improved: {best_score:.8f}")

    rem = deadline - time.monotonic()

    # === Phase 6: Second GeoLP round on updated best (~200s) ===
    print(f"--- Phase 6: Second GeoLP (score={best_score:.8f}) ---")
    t_geolp2 = min(rem * 0.15, 200.0)
    if t_geolp2 > 30:
        a_geo2, _ = geometric_mean_lp(best_seq, t_geolp2, solver, n_max_iters=300000)
        best_seq, best_score, improved = try_local_update(a_geo2, best_seq, best_score)
        if improved:
            print(f"  GeoLP2 improved: {best_score:.8f}")

    rem = deadline - time.monotonic()

    # === Phase 7: Perturbation escapes ===
    print(f"--- Phase 7: Perturbation Escapes (score={best_score:.8f}) ---")
    if rem > 150:
        for trial in range(3):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 50:
                break
            n_cur = len(best_seq)
            x = np.linspace(0, 2 * np.pi, n_cur)
            strength = 0.015 + 0.01 * trial
            a_pert = best_seq + strength * np.cos((trial + 2) * x + rng.uniform(0, 2 * np.pi))
            a_pert = np.maximum(a_pert, 0)
            s_p = a_pert.sum()
            if s_p < 1e-15:
                continue
            a_pert = a_pert / s_p

            t_sub = min(rem_sub * 0.25, 60.0)
            if t_sub > 15:
                a_opt, _ = nesterov_lp_refine(a_pert, t_sub, solver, n_max_iters=100000)
                best_seq, best_score, improved = try_local_update(a_opt, best_seq, best_score)
                if improved:
                    print(f"  Pert trial {trial} improved: {best_score:.8f}")

    rem = deadline - time.monotonic()

    # === Phase 8: Multi-n exploration ===
    print(f"--- Phase 8: Multi-n (score={best_score:.8f}) ---")
    if rem > 120:
        for n_try in [1000, 1500, 2000]:
            rem_sub = deadline - time.monotonic()
            if rem_sub < 50:
                break
            t_sub = min(rem_sub * 0.2, 70.0)
            if t_sub < 20:
                continue
            print(f"  Trying n={n_try}")
            init_try = interpolate_sequence(best_seq, n_try)
            # Quick spectral prep
            a_prep = spectral_flatten_adaptive(init_try, n_iters=30,
                                               deadline=time.monotonic() + 5)
            a_opt, _ = nesterov_lp_refine(a_prep, t_sub, solver, n_max_iters=50000)
            best_seq, best_score, improved = try_local_update(a_opt, best_seq, best_score)
            if improved:
                print(f"  n={n_try} improved: {best_score:.8f}")
                n_best = n_try

    rem = deadline - time.monotonic()

    # === Phase 9: Final polish ===
    print(f"--- Phase 9: Final Polish (score={best_score:.8f}) ---")
    if rem > 40:
        t_fin = min(rem * 0.4, 80.0)
        a_fin, _ = nesterov_lp_refine(best_seq, t_fin, solver, n_max_iters=50000)
        best_seq, best_score, improved = try_local_update(a_fin, best_seq, best_score)
        if improved:
            print(f"  Final LP improved: {best_score:.8f}")

    rem = deadline - time.monotonic()
    if rem > 20:
        a_spec_fin = spectral_flatten_adaptive(best_seq, n_iters=100,
                                               deadline=time.monotonic() + min(rem, 25))
        best_seq, best_score, improved = try_local_update(a_spec_fin, best_seq, best_score)
        if improved:
            print(f"  Final spectral improved: {best_score:.8f}")

    rem = deadline - time.monotonic()
    if rem > 15:
        a_sg_fin, _ = peak_subgradient_exp(best_seq, min(rem, 20), n_steps=50000)
        best_seq, best_score, improved = try_local_update(a_sg_fin, best_seq, best_score)
        if improved:
            print(f"  Final SG improved: {best_score:.8f}")

    # Final output
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_start
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}) at {elapsed:.1f}s")

        test_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        try:
            eval_score = evaluate_sequence(test_seq)
            print(f"Verified with evaluate_sequence: {eval_score:.8f}")
        except Exception as e:
            print(f"Verification error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
