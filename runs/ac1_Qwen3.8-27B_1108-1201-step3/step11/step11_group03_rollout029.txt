**Simpler approach: Frank-Wolfe (Conditional Gradient) method**

The problem is: `minimize f(a) = max_k (a*a)[k]` subject to `a in S` where `S = {a : sum(a) = 1, a >= 0}` (the simplex).

Frank-Wolfe iterates:
1. Compute the linear approximation of `f` at the current point `a_t`: `f(a) ≈ f(a_t) + <∇f(a_t), a - a_t>`.
2. Find `s_t = argmin_{s in S} <∇f(a_t), s>`.
3. Update: `a_{t+1} = (1 - gamma_t) * a_t + gamma_t * s_t`.

Step 2 is easy: `s_t = e_i` where `i = argmin_j ∇f(a_t)_j` (a vertex of the simplex).

Step 1: The subgradient of `f(a) = max_k (a*a)[k]` at `a` is `conv{∇(a*a)[k] : k in argmax(a*a)}`. We can use the average of the subgradients at the active positions.

def multiplicative_weights_refine(a0, time_limit, n_steps=20000):
    """Mirror descent (multiplicative weights) for final polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = 0.05
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        log_scale = -eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 50:
            eta *= 0.85
            no_improve = 0
            if eta < 0.001:
                break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    
    Strategy:
    1. Load parent sequence as initialization
    2. For each target n (1200, 1199, 1201):
       a. Try full QCQP (all constraints) with MOSEK
       b. Fall back to constraint-generation QCQP
       c. Polish with Nesterov LP alternating
    3. Final multiplicative weights polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')
    solver = get_solver()
    print(f"Using primary solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        x = np.linspace(0, 1, n_init)
        best_seq = np.cos(np.pi * x / 2) ** 2
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")

    def update_best(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # === Phase 1: Constraint-generation QCQP for n=1200 ===
    n_target = 1200
    print(f"\n=== Phase 1: CG-QCQP n={n_target} (current best={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 500.0)
    if rem > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_init = interpolate_sequence(best_seq, n_target) if len(best_seq) != n_target else best_seq.copy()

        # Try full QCQP first (all 2n-1 constraints)
        print("  Attempting full QCQP...")
        a_full, M_full = full_qcqp(n_target, solver, time_limit=min(t_alloc * 0.4, 90))
        update_best(a_full, "full_qcqp")

        # Constraint generation (from current best)
        rem2 = deadline - time.monotonic()
        t_cg = min(rem2 * 0.5, t_alloc * 0.6)
        if t_cg > 20:
            a_cg, _ = constraint_generation(a_init, n_target, solver, t_cg)
            update_best(a_cg, "cg_n1200")

    # === Phase 2: Nesterov LP polish for n=1200 ===
    print(f"\n=== Phase 2: Nesterov LP polish n={n_target} (best={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 350.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver, n_max_iters=100000)
        update_best(a_lp, "nesterov_lp")

    # === Phase 3: CG-QCQP again (from improved point) ===
    print(f"\n=== Phase 3: CG-QCQP round 2 (best={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 250.0)
    if t_alloc > 20:
        a_cg2, _ = constraint_generation(best_seq, len(best_seq), solver, t_alloc)
        update_best(a_cg2, "cg_round2")

    # === Phase 4: Try n=1199 ===
    print(f"\n=== Phase 4: CG-QCQP n=1199 (best={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_1199 = min(rem * 0.15, 200.0)
        print(f"  Budget: {t_1199:.0f}s")
        init_1199 = interpolate_sequence(best_seq, 1199)
        a_1199, _ = constraint_generation(init_1199, 1199, solver, t_1199)
        update_best(a_1199, "cg_n1199")

    # === Phase 5: Try n=1201 ===
    print(f"\n=== Phase 5: CG-QCQP n=1201 (best={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_1201 = min(rem * 0.15, 200.0)
        print(f"  Budget: {t_1201:.0f}s")
        init_1201 = interpolate_sequence(best_seq, 1201)
        a_1201, _ = constraint_generation(init_1201, 1201, solver, t_1201)
        update_best(a_1201, "cg_n1201")

    # === Phase 6: Final multiplicative weights polish ===
    print(f"\n=== Phase 6: Multiplicative weights polish (best={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 10:
        t_ms = min(rem * 0.3, 100.0)
        a_ms, _ = multiplicative_weights_refine(best_seq, t_ms, n_steps=50000)
        update_best(a_ms, "mult_weights")

    # === Phase 7: Final CG-QCQP tight ===
    rem = deadline - time.monotonic()
    if rem > 15:
        print(f"\n=== Phase 7: Final CG polish (best={best_score:.10f}) ===")
        a_final, _ = constraint_generation(best_seq, len(best_seq), solver, min(rem * 0.8, 60))
        update_best(a_final, "final_cg")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")

    # Ensure valid output
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    # Verify non-empty
    if not result:
        result = [1.0]
    return result
