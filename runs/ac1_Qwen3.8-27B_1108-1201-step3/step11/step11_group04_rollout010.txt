def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():

    for solver_name in [cp.GUROBI, cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:

def solve_lp_best_response(a_fixed, solver, get_dual=False):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    ineq_constraint = T_mat @ b_var <= M_var
    eq_constraint = cp.sum(b_var) == 1
    constraints = [ineq_constraint, eq_constraint]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.CLARABEL, cp.ECOS, cp.SCS]
    solved = False
    for s in solvers_to_try:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solved = True
                break
        except Exception:
            continue

    if not solved:
        return None, None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, None

def fw_step(a, step_size=None, top_k=5):

    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    
    # Find top_k peaks
    if top_k > 1:
        # Use softmax-weighted subgradient
        threshold = max_b * (1.0 - 0.01)
        active = np.where(b >= threshold)[0]
        if len(active) > top_k:
            # Keep only top_k
            idx = np.argsort(b[active])[-top_k:]
            active = active[idx]
        # Weight by how close to max
        weights = b[active] - b[active].min() + 1e-10
        weights = weights / weights.sum()
    else:
        active = np.array([np.argmax(b)])
        weights = np.array([1.0])
    
    # Compute subgradient: g[j] = 2 * sum_k w_k * a[k-j] (for valid k-j)
    g = np.zeros(n)
    for i, k in enumerate(active):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += 2.0 * weights[i] * a[k - idx]
    
    # FW step: move toward argmin of g
    j_min = np.argmin(g)
    
    if step_size is None:
        step_size = 2.0 / (len(fw_step.iters) + 2)  # placeholder
    
    a_new = (1.0 - step_size) * a + step_size * np.zeros(n)
    a_new[j_min] = step_size
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a, max_b
    a_new = a_new / s
    return a_new, max_b

    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)

    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    a_prev = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for it in range(1, n_max_iters + 1):
        if time.monotonic() - t0 > time_limit:

            b_new = a.copy()
            continue
        
        # Nesterov update

        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy
        
        a_new = ((it) * a + (it + 2) * alpha * b_new) / (it + (it + 2) * alpha)
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = y.copy()
            sa = 1.0

            best_a = a_new.copy()

def lbgfs_refine(a0, time_limit, T_stages, deadline=None):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    # Work in log-space
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    s = a.sum()
    a = a / s
    z = np.log(a)
    
    t0 = time.monotonic()
    if deadline is None:
        deadline = t0 + time_limit
    
    for stage_i, stage_T in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        
        stages_left = len(T_stages) - stage_i
        iters = max(50, int(min(remaining * 30 / stages_left, 10000)))
        
        def func(z_in):
            if time.monotonic() > deadline:
                return 1e15
            a_in = _softmax(z_in)
            b_in = fast_convolve(a_in)
            max_b = b_in.max()
            lse = max_b + stage_T * np.log(np.exp((b_in - max_b) / stage_T).sum())
            return 2.0 * n * lse
        
        def jac(z_in):
            if time.monotonic() > deadline:
                return np.zeros_like(z_in)
            a_in = _softmax(z_in)
            b_in = fast_convolve(a_in)
            max_b = b_in.max()
            exp_terms = np.exp((b_in - max_b) / stage_T)
            w = exp_terms / exp_terms.sum()
            
            grad_a = np.zeros(n)
            for k in range(len(b_in)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_in[k - idx]
            
            S = np.dot(grad_a, a_in)
            grad_z = a_in * (grad_a - S)
            return 2.0 * n * grad_z
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-30,
                         'gtol': 1e-20, 'maxfun': iters * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    a_final = _softmax(z)
    return a_final


def _softmax(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast mirror descent (multiplicative weights) with FFT-based subgradients
2. Extended LP alternating with Nesterov acceleration
3. LogSumExp L-BFGS fine polish (multi-temperature)
4. Direct subgradient polishing with line search

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.signal import correlate


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def find_best_solver():
    for name in [cp.GUROBI, cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            v = cp.Variable(3, nonneg=True)
            p = cp.Problem(cp.Minimize(cp.sum(v)), [cp.sum(v) == 1])
            p.solve(solver=name)
            if p.status in ('optimal', 'optimal_inaccurate'):
                return name
        except Exception:
            pass
    return cp.ECOS


def solve_lp_response(a_fixed, solver):
    n = len(a_fixed)
    bv = cp.Variable(n, nonneg=True)
    mv = cp.Variable()
    T = build_conv_matrix(a_fixed)
    con = [T @ bv <= mv, cp.sum(bv) == 1]
    prob = cp.Problem(cp.Minimize(mv), con)
    for s in [solver, cp.CLARABEL, cp.ECOS, cp.SCS, cp.GUROBI, cp.MOSEK]:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b = np.asarray(bv.value, dtype=np.float64)
                b = np.maximum(b, 0)
                ss = b.sum()
                if ss < 1e-15:
                    return None
                return b / ss
        except Exception:
            continue
    return None
