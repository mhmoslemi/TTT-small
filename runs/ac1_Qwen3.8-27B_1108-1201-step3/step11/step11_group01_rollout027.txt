"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated LP best-response with adaptive restarts (main workhorse)
2. Natural gradient (softmax) descent with Nesterov acceleration for polishing
3. Multi-length exploration with interpolation
4. Symmetrization and structured perturbations to escape local minima

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def build_T_dense(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp(a_fixed, solver_name=None):
    n = len(a_fixed)
    g_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_T_dense(a_fixed)
    constraints = [T @ g_var <= M_var, cp.sum(g_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers = [solver_name, 'PDLP', 'ECOS', 'SCS'] if solver_name else ['PDLP', 'ECOS', 'SCS']
    seen = set()
    for sn in solvers:
        if sn is None or sn in seen:
            continue
        seen.add(sn)
        try:
            prob.solve(solver=sn)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                break
        except Exception:
            continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    g_val = np.asarray(g_var.value, dtype=np.float64)
    g_val = np.maximum(g_val, 0)
    s = g_val.sum()
    if s < 1e-15:
        return None, None
    g_val /= s
    return g_val, float(M_var.value)

def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, 1, n)

        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-3 * x ** 2)
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif choice == 4:
        a = np.ones(n) * (1.0 + 0.15 * np.cos(2 * np.pi * 3 * np.arange(n) / n))

        x = np.linspace(0, 2 * np.pi, n)

        a[::2] *= 1.1

    else:

    a = np.maximum(a, 0.01)
    s = a.sum()
    if s < 1e-15:

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def symmetrize(a):
    a_sym = (a + a[::-1]) / 2.0
    a_sym = np.maximum(a_sym, 0)
    s = a_sym.sum()
    if s < 1e-15:
        return a.copy()
    return a_sym / s

            a_ng2, _ = natural_gradient(best_seq, t6, eta0=0.1, verbose=False)
            try_update(a_ng2, "ng_final")

    # Final symmetrization attempt
    a_sym_f = symmetrize(best_seq)
    try_update(a_sym_f, "sym_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")

    return [float(x) for x in best_seq]
