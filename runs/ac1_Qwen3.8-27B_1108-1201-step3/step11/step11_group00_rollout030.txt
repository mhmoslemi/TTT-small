Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based Nesterov accelerated game iteration (main engine) with running average
2. FFT-based LogSumExp smoothed gradient descent with multi-stage L-BFGS annealing
3. Subgradient-guided peak shaving with multiple active peaks
4. Multi-length exploration with interpolation transfer
5. Perturbation-based restarts to escape local minima

import numpy as np

import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_autocorr(a):
    """Full linear convolution a*a using FFT. O(n log n)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence using the provided evaluate_sequence."""
    try:
        a_arr = np.asarray(a, dtype=np.float64)
        a_arr = np.clip(a_arr, 0.0, 1000.0)
        a_list = [float(x) for x in a_arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def build_conv_matrix(a_fixed):
    """Build the (2n-1) x n convolution matrix T where (T@b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        row_start = j
        row_end = min(j + n, 2 * n - 1)
        T[row_start:row_end, j] = a_fixed[:row_end - j]
    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. T(a_fixed)@b <= M, sum(b)=1, b>=0. Returns (b, M)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        prob = None
    if prob is None or prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob2 = cp.Problem(cp.Minimize(M_var), constraints)
                prob2.solve(solver=alt)
                if prob2.status in ('optimal', 'optimal_inaccurate'):
                    prob = prob2
                    break
            except Exception:
                continue
    if prob is None or prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, max_iters=500000):
    """
    Nesterov-accelerated LP best-response iteration for the min-max game.
    Tracks running average which often outperforms any single iterate.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0
    last_progress = 0
    score_unchanged = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            score_unchanged += 1
            if score_unchanged > 1000:
                break
            continue

        score_unchanged = 0
        alpha_t = 2.0 / (t + 2.0)

        y_new = (1.0 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y

        if t <= 2:
            a_new = y_new.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_a

        y = y_new
        a = a_new

        # Running average (weight recent iterates more via exponential window)
        avg_a += a
        n_avg += 1

        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score - 1e-15:
                best_score = score_avg
                best_a = avg_check.copy()
                last_progress = t

        if n_avg > 100 and n_avg % 100 == 0:
            # Also check a windowed average (last 200 iterates)
            if n_avg >= 200:
                # We only track cumulative average, so just check current a and y
                score_a = 2.0 * n * fast_autocorr(a).max()
                if score_a < best_score - 1e-15:
                    best_score = score_a
                    best_a = a.copy()
                    last_progress = t
                score_y = 2.0 * n * fast_autocorr(y).max()
                if score_y < best_score - 1e-15:
                    best_score = score_y
                    best_a = y.copy()
                    last_progress = t

        if t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.7f}, M={M_val:.8f}")

        # Early stop if no progress for a long time
        if t - last_progress > 5000 and t > 1000:
            break

    return best_a, best_score


def logsumexp_optimize(a0, time_limit, T_start=100.0, T_end=0.1, n_stages=15):
    """
    LogSumExp-smoothed optimization with L-BFGS and multi-stage annealing.
    Uses FFT-based gradients for O(n log n) per function evaluation.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining_local = time_limit - (time.monotonic() - t0)
        if remaining_local < 2:
            break

        iters_per_stage = max(10, min(60, int(30 + 30 * (si / n_stages))))

        def _func(z_in):
            val, _ = lse_obj_grad_z(z_in, stage_T)
            return val

        def _jac(z_in):
            _, g = lse_obj_grad_z(z_in, stage_T)
            return g

        try:
            res = minimize(
                fun=_func, x0=z, jac=_jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': iters_per_stage * 500,
                         'disp': False}
            )
            z = res.x
            a_new = softmax_fn(z)
            score = 2.0 * n * fast_autocorr(a_new).max()
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
        except Exception:
            pass

    return best_a, best_score


def subgradient_peak_shave(a0, time_limit, seed=123):
    """
    Subgradient-guided peak shaving: identify the active max peaks and
    move mass from high-contributing positions to reduce them.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_val = b.max()
        # Find all peaks within 1% of max
        threshold = max_val * 0.99
        active_peaks = np.where(b >= threshold)[0]

        improved = False
        for k in active_peaks[:10]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            total_contrib = contrib.sum()
            if total_contrib < 1e-15:
                continue

            # Subgradient direction: reduce the most contributing pair
            src_local = np.argmax(contrib)
            src = idx[src_local]
            partner = k - src

            # Find good destination: low value, not contributing to any active peak
            in_peaks = np.array([False] * n)
            for k2 in active_peaks[:5]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                in_peaks[js2:je2 + 1] = True

            candidates = np.where(~in_peaks)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]
            # Prefer low-value positions
            cand_vals = a[candidates]
            # Pick from lowest 30%
            n_low = max(1, len(candidates) // 3)
            low_idx = np.argsort(cand_vals)[:n_low]
            dst = candidates[low_idx[rng.integers(n_low)]]

            # Try multiple step sizes
            for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                new_b = fast_autocorr(a_trial)
                new_max = new_b.max()
                new_score = 2.0 * n * new_max
                if new_score < best_score - 1e-14:
                    a = a_trial
                    best_score = new_score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

        if not improved:
            # Random exploration
            stall += 1
            for _ in range(5):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-12:
                    frac = rng.uniform(0.003, 0.02)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_autocorr(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
                            improved = True
                            break

            if stall > 100:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length using cubic-like smoothing."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    # Slight smoothing
    kernel = np.array([0.25, 0.5, 0.25])
    f_interp = np.convolve(f_interp, kernel, mode='same')
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def perturbation_restart(a0, rng, n_perturb=10):
    """Try small random perturbations and return the best."""
    n = len(a0)

    for _ in range(n_perturb):

        a_pert = a_pert / s
        score = 2.0 * n * fast_autocorr(a_pert).max()
        if score < best_score:
            best_score = score
            best = a_pert.copy()

    if t1 > 30:
        try:
            a_out, _ = nesterov_lp_optimize(best_seq, t1, solver)
            try_update(a_out, f"lp_n{n_parent}")
        except Exception as e:
            print(f"  Phase 1 error: {e}")

    # === Phase 2: LogSumExp refinement at parent length ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LSE L-BFGS n={n_parent} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 180.0)
    if t2 > 10:
        try:
            a_out, _ = logsumexp_optimize(best_seq, t2, T_start=50.0, T_end=0.1, n_stages=12)
            try_update(a_out, f"lse_n{n_parent}")
        except Exception as e:
            print(f"  Phase 2 error: {e}")

    # === Phase 3: Peak shaving at parent length ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Peak shave n={n_parent} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.10, 100.0)
    if t3 > 5:
        try:
            a_out, _ = subgradient_peak_shave(best_seq, t3, seed=777)
            try_update(a_out, f"shave_n{n_parent}")
        except Exception as e:
            print(f"  Phase 3 error: {e}")

    # === Phase 4: Try longer length ===
    n_long = n_parent + 400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LP Nesterov n={n_long} (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 30:
        init_long = interpolate_sequence(best_seq, n_long)
        try:
            a_out, _ = nesterov_lp_optimize(init_long, t4, solver)
            try_update(a_out, f"lp_n{n_long}")
        except Exception as e:
            print(f"  Phase 4 error: {e}")

    # === Phase 5: LSE at longer length ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: LSE n={n_long} (remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 150.0)
    if t5 > 10:
        try:
            a_out, _ = logsumexp_optimize(best_seq, t5, T_start=100.0, T_end=0.05, n_stages=15)
            if len(a_out) == n_long:
                pass
            try_update(a_out, f"lse_n{n_long}")
        except Exception as e:
            print(f"  Phase 5 error: {e}")

    # === Phase 6: Perturbation + quick refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Perturbation restarts (remaining={remaining:.0f}s) ===")
    if remaining > 20:
        t6 = min(remaining * 0.15, 120.0)
        perturb_t = min(20.0, t6 * 0.3)
        retry_t = t6 - perturb_t
        a_pert, _ = perturbation_restart(best_seq, rng, n_perturb=20)
        try_update(a_pert, "perturbed")
        if retry_t > 10:
            try:
                a_out, _ = subgradient_peak_shave(a_pert, retry_t, seed=999)
                try_update(a_out, "pert_shave")
            except Exception:
                pass

    # === Phase 7: Final LP Nesterov refinement on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final LP refinement n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t7 = min(remaining * 0.25, 200.0)
    if t7 > 20:
        try:
            a_out, _ = nesterov_lp_optimize(best_seq, t7, solver)
            try_update(a_out, "final_lp")
        except Exception as e:
            print(f"  Phase 7 error: {e}")

    # === Phase 8: Final peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Final peak shave (remaining={remaining:.0f}s) ===")
    t8 = min(remaining * 0.40, 100.0)
    if t8 > 5:
        try:
            a_out, _ = subgradient_peak_shave(best_seq, t8, seed=1234)
            try_update(a_out, "final_shave")
        except Exception:
            pass

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
