import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_autocorr(a):
    """Compute auto-convolution of a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def lse_objective_and_gradient(a, T):
    """Compute LSE-smoothed objective and gradient for a (assumed sum=1, a>=0)."""
    n = len(a)
    c = fast_autocorr(a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    lse = c_max + T * np.log(exp_c.sum())
    w = exp_c / exp_c.sum()
    
    # Gradient: g[m] = 2/T * sum_k w[k]*a[k-m] - a[m]*w[2m]/T
    # Cross-correlation of w and a
    N = 1 << (len(w) + n - 1).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    R = irfft(W * np.conj(A), N)
    cross = R[:n]  # cross[m] = sum_k w[k]*a[k-m]
    
    g = 2.0 * cross
    idx = 2 * np.arange(n)
    valid = idx < (2 * n - 1)
    g[valid] -= a[valid] * w[idx[valid]]
    g /= T
    
    obj = 2.0 * n * lse
    return obj, g

def softmax(z):
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()

    """Gerchberg-Saxton style iterative projection.
    Alternates between:
    1. Making the power spectrum flat (in frequency domain)
    2. Making the time-domain signal non-negative
    """

    # Initialize with random phases

    phases = np.exp(1j * rng.uniform(0, 2 * np.pi, fft_size // 2 + 1))
    A *= phases
    
    best_a = None
    best_score = float('inf')
    
    for it in range(n_iters):
        a = irfft(A, fft_size)[:n]
        # Project to non-negative

        if s < 1e-15:

        # Go to frequency domain
        A_new = rfft(a, fft_size)
        # Project to flat spectrum (keep phase, set magnitude to constant)
        P = np.abs(A_new)
        P_mean = P.mean()
        A = A_new / (P + 1e-300) * P_mean
        A[0] = P_mean  # DC component
    
    if best_a is None:
        best_a = a
        best_score = evaluate_local(a)
    
    return best_a, best_score

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated LogSumExp gradient with aggressive temperature annealing
2. Gerchberg-Saxton spectral projection for diverse initialization
3. L-BFGS optimization in log-space with multi-stage annealing
4. Parameterized shape families (sigmoid, polynomial, kernel-based)
5. Adaptive restart strategy to escape local minima
6. Multi-length exploration

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_autocorr(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(a)) / len(a)
    return a / s


def evaluate_local(a):
    """Evaluate score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    c = fast_autocorr(a)
    return float(2.0 * n * c.max() / (s * s))


def lse_obj_grad_a(a, T):
    """Compute LSE-smoothed objective and gradient w.r.t. a (a assumed sum=1, a>=0).
    Returns (obj, grad) where obj = 2n * LSE(c, T), grad = d(obj)/da.
    """
    n = len(a)
    c = fast_autocorr(a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    lse = c_max + T * np.log(exp_c.sum() + 1e-300)
    w = exp_c / (exp_c.sum() + 1e-300)
    obj = 2.0 * n * lse

    # Gradient: g[m] = (2/T) * sum_k w[k]*a[k-m] - (1/T)*a[m]*w[2m]
    N = 1 << (len(w) + n - 1).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    R = irfft(W * np.conj(A), N)
    cross = R[:n]

    g = 2.0 * cross
    idx = 2 * np.arange(n)
    valid = idx < (2 * n - 1)
    g[valid] -= a[valid] * w[idx[valid]]
    g /= T

    return obj, g


def softmax_fn(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def lse_obj_grad_z(z, T):
    """LSE objective and gradient in log-parameterization."""
    a = softmax_fn(z)
    obj, g_a = lse_obj_grad_a(a, T)
    S = np.dot(g_a, a)
    g_z = a * (g_a - S)
    return obj, g_z

    Alternates between non-negativity (time) and flat spectrum (frequency).
    Returns (best_a, best_score).
    """

    fft_size = 1 << (2 * n - 1).bit_length()

    # Initialize with random phases
    A = np.ones(fft_size // 2 + 1, dtype=np.complex128)
    phases = np.exp(1j * rng.uniform(0, 2 * np.pi, fft_size // 2 + 1))
    A *= phases

    best_a = None
    best_score = float('inf')
    current_a = np.ones(n) / n

    for it in range(n_iters):
        a = irfft(A, fft_size)[:n].copy()
        # Project to non-negative
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            # Reset
            a = np.ones(n) / n
            s = 1.0
        a = a / s
        current_a = a

        # Periodic evaluation
        if it % 200 == 0 or it == n_iters - 1:
            sc = evaluate_local(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        # Go to frequency domain
        A_new = rfft(a, fft_size)
        P = np.abs(A_new)
        P_mean = P.mean()
        # Blend between current magnitude and flat (annealing)
        alpha = flatten_strength * min(1.0, it / 500.0)
        P_target = np.abs((1 - alpha) * A_new + alpha * (A_new / (P + 1e-300) * P_mean))
        A = A_new / (P + 1e-300) * P_target
        # Ensure DC is correct
        A[0] = A_new[0]

    if best_a is None:
        best_a = current_a
        best_score = evaluate_local(current_a)

    return best_a, best_score


def optimize_lbgfs(a0, time_limit, T_stages=None):
    """L-BFGS optimization with LSE smoothing and temperature annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [10, 5, 3, 2, 1.5, 1, 0.7, 0.5, 0.35, 0.25, 0.18,
                    0.13, 0.1, 0.07, 0.05, 0.035, 0.025, 0.018, 0.013,
                    0.01, 0.007, 0.005, 0.003, 0.002, 0.001]

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func(z_in):
            obj, _ = lse_obj_grad_z(z_in, T)
            return obj

        def jac(z_in):
            _, g = lse_obj_grad_z(z_in, T)
            return g

        iters = max(10, int(remaining * 15))
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-22, 'gtol': 1e-20,
                         'maxcor': 50, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_fn(z)


def generate_shape_initializations(n, rng):
    """Generate diverse shape-based initializations."""
    x = np.linspace(0, 1, n)
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Sigmoid family (asymmetric)
    for beta in [3, 5, 8, 12]:
        for gamma in [0.3, 0.35, 0.4, 0.45, 0.55, 0.6, 0.7]:
            a = 1.0 / (1.0 + np.exp(-beta * (x - gamma)))
            inits.append(normalize(a))

    # Polynomial tilt
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = x ** p + 0.01
        inits.append(normalize(a))
        b = (1 - x) ** p + 0.01
        inits.append(normalize(b))

    # Gaussian (off-center)
    for mu in [0.3, 0.4, 0.6, 0.7]:
        for sigma in [0.15, 0.2, 0.3]:
            a = np.exp(-((x - mu) / sigma) ** 2)
            inits.append(normalize(a))

    # Cosine tilt
    for freq in [1, 2]:
        for phase in [0, np.pi / 4, np.pi / 2, -np.pi / 4]:
            a = 0.5 + 0.4 * np.cos(2 * np.pi * freq * x + phase)
            a = np.maximum(a, 0.01)
            inits.append(normalize(a))

    # Power-law with bump
    for alpha in [0.5, 1, 2]:
        a = x ** alpha * (1 - x) ** 0.5 + 0.01
        inits.append(normalize(a))
        a = (1 - x) ** alpha * x ** 0.5 + 0.01
        inits.append(normalize(a))

    # Random smooth (Fourier basis)
    for _ in range(5):
        coeffs = rng.normal(0, 1, 6)
        phases = rng.uniform(0, 2 * np.pi, 6)
        a = np.ones(n) * 0.5
        for i in range(6):
            a += abs(coeffs[i]) * 0.3 * np.cos(2 * np.pi * (i + 1) * x + phases[i])
        a = np.maximum(a, 0.01)
        inits.append(normalize(a))

    # Edge-weighted
    for w in [0.5, 1.0, 2.0]:
        a = np.exp(-w * x) + 0.01
        inits.append(normalize(a))
        a = np.exp(-w * (1 - x)) + 0.01
        inits.append(normalize(a))

    return inits


def perturbed_init(a0, rng, strength=0.05):
    """Create a perturbed version of a0."""
    n = len(a0)
    method = rng.integers(0, 4)
    if method == 0:
        noise = rng.normal(0, strength, n)
        a = a0 + noise
    elif method == 1:
        # Low-frequency perturbation
        x = np.linspace(0, 1, n)
        freq = rng.integers(1, 6)
        phase = rng.uniform(0, 2 * np.pi)
        noise = strength * np.cos(2 * np.pi * freq * x + phase)
        a = a0 + noise
    elif method == 2:
        # Block swap
        a = a0.copy()
        bs = rng.integers(max(5, n // 20), max(10, n // 5))
        i = rng.integers(0, n - bs)
        j = rng.integers(0, n - bs)
        if abs(i - j) > bs:
            a[i:i + bs], a[j:j + bs] = a[j:j + bs].copy(), a[i:i + bs].copy()
    else:
        # Multiplicative perturbation
        a = a0 * np.exp(rng.normal(0, strength * 0.5, n))
    return normalize(a)
