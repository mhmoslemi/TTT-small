def fast_convolve(a):

    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_of(a):
    """Compute 2*n*max(conv(a,a)) / (sum(a))^2"""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max() / (s * s)

def lse_polish(a0, time_limit, T_start=50.0, T_end=0.1, n_stages=10):
    """LogSumExp smoothing + L-BFGS polishing."""
    n = len(a0)
    a0 = normalize_to_simplex(a0)
    # Parameterize as a = softmax(z)
    z = np.log(np.maximum(a0, 1e-30))
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        
        def lse_objective(z_val):
            a_val = softmax(z_val)
            b = fast_convolve(a_val)
            Tb = T * b
            max_Tb = np.max(Tb)
            if max_Tb > 700:
                max_Tb = 700
            exp_Tb = np.exp(Tb - max_Tb)
            lse = max_Tb + np.log(np.sum(exp_Tb))
            return 2.0 * n / T * lse
        
        def lse_gradient(z_val):
            a_val = softmax(z_val)
            b = fast_convolve(a_val)
            Tb = T * b
            max_Tb = np.max(Tb)
            if max_Tb > 700:
                max_Tb = 700
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / np.sum(exp_Tb)
            
            # Gradient wrt a: (2n/T) * sum_k w[k] * a[k-j] for a[j]
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-20:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += w[k] * a_val[k - idx]
            grad_a *= 2.0 * n / T
            
            # Chain rule: grad_z = a * (grad_a - sum(grad_a * a))
            S = np.dot(grad_a, a_val)
            return a_val * (grad_a - S)
        
        try:
            res = minimize(
                fun=lse_objective, x0=z, jac=lse_gradient,
                method='L-BFGS-B',
                options={'maxiter': 50, 'ftol': 1e-15, 'gtol': 1e-12, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    return softmax(z)

def softmax(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def fourier_init(n, rng):
    """Initialize with a Fourier-based pattern."""
    x = np.arange(n)
    # Try different harmonic structures
    choice = rng.integers(0, 4)
    if choice == 0:

def subgradient_refine(a0, time_limit, n_steps=100000):
    """Subgradient descent with momentum for final refinement."""
    n = len(a0)
    a = normalize_to_simplex(a0)
    z = np.log(np.maximum(a, 1e-30))
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    momentum = np.zeros(n)
    beta = 0.9
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Find active constraints (top few peaks)
        threshold = max_b * 0.98
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        # Compute gradient
        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * a[k - idx]
        grad /= len(active)
        
        # Normalize gradient
        S = np.dot(grad, a)
        grad_z = a * (grad - S)
        
        # Momentum update
        eta = 0.1 / (1 + 0.01 * step)
        momentum = beta * momentum + (1 - beta) * grad_z
        z_new = z - eta * momentum
        a_new = softmax(z_new)
        
        sc = score_of(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            z = z_new
            a = a_new
    
    return best_a

def interpolate_to_length(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a_norm = a / a.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0)
    s = a_interp.sum()
    if s > 1e-15:
        a_interp /= s
    else:
        a_interp = np.ones(new_n) / new_n
    return a_interp

            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * a[k - idx]
        grad /= len(active)

        S = np.dot(grad, a)
        grad_z = a * (grad - S)

        eta = 0.3 / (1.0 + 0.005 * step)
        mom = beta * mom + (1.0 - beta) * grad_z
        z = z - eta * mom
        a_new = softmax_func(z)

        sc = score_of(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new

    return best_a, best_score


def interpolate_len(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a_norm = a / a.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0)
    s = a_interp.sum()
    if s > 1e-15:
        a_interp /= s
    else:
        a_interp = np.ones(new_n) / new_n
    return a_interp


def fourier_init(n, rng):
    x = np.arange(n)
    choice = rng.integers(0, 5)
    if choice == 0:
        a = 0.5 + 0.25 * np.cos(2 * np.pi * x / n)
    elif choice == 1:
        a = 0.5 + 0.15 * np.cos(4 * np.pi * x / n) + 0.1 * np.cos(2 * np.pi * x / n)
    elif choice == 2:
        a = 0.5 + 0.2 * np.cos(6 * np.pi * x / n)
    elif choice == 3:
        a = np.ones(n) + 0.05 * np.cos(2 * np.pi * 5 * x / n)
    else:
        a = 0.5 + 0.2 * np.cos(2 * np.pi * x / n) + 0.1 * np.cos(2 * np.pi * 7 * x / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()
