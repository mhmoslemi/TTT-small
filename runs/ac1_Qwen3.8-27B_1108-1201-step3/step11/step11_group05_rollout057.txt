def fast_conv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    """Compute the objective value."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    n = len(a)
    return 2.0 * n * fast_conv(a).max() / (s * s)

def grad_max_conv(a, w):
    """
    Compute gradient of sum_k w[k] * conv(a,a)[k] w.r.t. a.
    Result: 2 * correlate(w, a)[:n]
    """
    n = len(a)
    # correlate(w, a, 'full')[i] = sum_j w[j] * a[j-i] ... 
    # Actually numpy: np.correlate(v, w, 'full')[k] = sum_n v[n+k] * conj(w[n])
    # We want: g[i] = 2 * sum_j a[j] * w[i+j]

    # = sum_n a[n] * w[n+i]  -- YES this is what we want!
    c = np.correlate(w, a, mode='full')
    return 2.0 * c[:n]

def adam_logspace(a0, time_limit, T_list, lr0=0.005, top_k_frac=0.2):
    """Adam optimization in log-space with temperature annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    z -= z.max()  # center for numerical stability
    
    best_a = a0.copy()
    best_score = compute_score(a0)
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t0 = time.monotonic()
    step = 0
    
    for stage, T in enumerate(T_list):
        if time.monotonic() - t0 > time_limit:
            break
        
        top_k = max(10, int(top_k_frac * (2 * n - 1)))
        stage_steps = max(1000, int(time_limit * 10000 / len(T_list)))
        
        for s in range(stage_steps):
            if time.monotonic() - t0 > time_limit:
                break
            step += 1
            
            a = _softmax(z)
            b = fast_conv(a)
            cur_score = 2.0 * n * b.max()
            
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
            
            # Compute weights for log-sum-exp
            if T < 0.01:
                # Subgradient on active set
                max_b = b.max()
                threshold = max_b * (1.0 - 1e-5)
                active = np.where(b >= threshold)[0]
                w = np.zeros(len(b))
                w[active] = 1.0 / len(active)
            else:
                if top_k < len(b):
                    idx = np.argpartition(b, -top_k)[-top_k:]
                    b_top = b[idx]
                    max_top = b_top.max()
                    exp_top = np.exp((b_top - max_top) / T)
                    w_top = exp_top / exp_top.sum()
                    w = np.zeros(len(b))
                    w[idx] = w_top
                else:
                    max_b = b.max()
                    exp_all = np.exp((b - max_b) / T)
                    w = exp_all / exp_all.sum()
            
            # Gradient in a-space
            g_a = grad_max_conv(a, w)
            
            # Gradient in z-space
            S = np.dot(g_a, a)
            g_z = a * (g_a - S)
            
            # Adam update
            lr = lr0 * (0.85 ** stage) / (1.0 + 0.05 * s / stage_steps)
            m = beta1 * m + (1 - beta1) * g_z
            v = beta2 * v + (1 - beta2) * g_z ** 2
            m_hat = m / (1 - beta1 ** (step + 1))
            v_hat = v / (1 - beta2 ** (step + 1))
            z -= lr * m_hat / (np.sqrt(v_hat) + eps)
            z -= z.mean()  # recenter
            
    return best_a, best_score

def _softmax(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

**Alternative: Direct optimization with proximal gradient**

The problem is: min f(a) = 2n * max_k conv(a,a)[k] / (sum(a))^2

This is not convex in a (due to the ratio), but if we fix sum(a) = 1, it becomes:
min g(a) = 2n * max_k conv(a,a)[k] subject to a >= 0, sum(a) = 1

Which IS convex! (max of linear functions is convex, and the constraint set is convex.)

So we can use projected subgradient methods or accelerated projected subgradient (FISTA).

The subgradient of g(a) at a is: 2n * (1/|A|) * sum_{k in A} ∂_a conv(a,a)[k] where A is the active set (argmax).

The projection onto {a >= 0, sum(a) = 1} is: project onto the simplex.

FISTA algorithm:
- y_k = x_k + (t_{k-1}-1)/(t_k+1) * (x_k - x_{k-1})

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based fast auto-convolution and gradient computation
2. Adam optimization in log-space with adaptive top-k peak focusing
3. LP best-response with Nesterov acceleration (cutting-plane style)
4. Multi-length exploration with interpolation
5. Perturbation-based escape with simulated annealing acceptance
6. FISTA (proximal gradient with momentum) for the convex subproblem

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp
import math


def fast_conv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the objective value for a non-negative sequence."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    n = len(a)
    return 2.0 * n * fast_conv(a).max() / (s * s)


def grad_conv_weighted(a, w):
    """
    Gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a.
    Returns 2 * correlate(w, a)[:n].
    """
    n = len(a)
    c = np.correlate(w, a, mode='full')
    return 2.0 * c[:n]


def softmax_vec(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def adam_logspace(a0, time_limit, T_list, lr0=0.004, top_k_start=0.3, top_k_end=0.05):
    """
    Adam optimization in log-space with temperature annealing and adaptive top-k.
    """
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    z -= z.max()

    best_a = a0.copy()
    best_score = compute_score(a0)
    n_conv = 2 * n - 1

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t0 = time.monotonic()
    step = 0
    total_stages = len(T_list)

    for stage_idx, T in enumerate(T_list):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break

        # Adaptive top-k: larger early, smaller late
        frac = top_k_start + (top_k_end - top_k_start) * (stage_idx / max(1, total_stages - 1))
        top_k = max(10, int(frac * n_conv))

        stage_time_frac = remaining / max(1, total_stages - stage_idx)
        stage_steps = max(500, int(stage_time_frac * 8000))

        for s in range(stage_steps):
            if time.monotonic() - t0 > time_limit:
                break
            step += 1

            a = softmax_vec(z)
            b = fast_conv(a)
            cur_score = 2.0 * n * b.max()

            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

            # Compute peak weights
            if T < 0.005:
                max_b = b.max()
                threshold = max_b - max_b * 1e-6
                active = np.where(b >= threshold)[0]
                w = np.zeros(n_conv)
                w[active] = 1.0 / len(active)
            elif top_k < n_conv:
                idx = np.argpartition(b, -top_k)[-top_k:]
                b_top = b[idx]
                max_top = b_top.max()
                exp_top = np.exp((b_top - max_top) / T)
                w_top = exp_top / exp_top.sum()
                w = np.zeros(n_conv)
                w[idx] = w_top
            else:
                max_b = b.max()
                exp_all = np.exp((b - max_b) / T)
                w = exp_all / exp_all.sum()

            # Gradient in a-space
            g_a = grad_conv_weighted(a, w)

            # Gradient in z-space (Jacobian of softmax)
            S = np.dot(g_a, a)
            g_z = a * (g_a - S)

            # Adaptive learning rate
            progress = stage_idx / max(1, total_stages)
            decay = 1.0 / (1.0 + 0.3 * s / max(1, stage_steps))
            lr = lr0 * (0.82 ** stage_idx) * decay

            # Gradient clipping
            g_norm = np.linalg.norm(g_z)
            if g_norm > 10.0:
                g_z = g_z * 10.0 / g_norm

            # Adam update
            m = beta1 * m + (1 - beta1) * g_z
            v = beta2 * v + (1 - beta2) * g_z ** 2
            m_hat = m / (1 - beta1 ** (step + 1))
            v_hat = v / (1 - beta2 ** (step + 1))
            z -= lr * m_hat / (np.sqrt(v_hat) + eps)
            z -= z.mean()

            if (step + 1) % 50000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    Adam stage {stage_idx} T={T:.3f} step={step} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def build_T_matrix(a_fixed):
    """Build convolution matrix T such that T @ b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    size = 2 * n - 1

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    T = build_T_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = []
    if solver is not None:
        solvers_to_try.append(solver)
    solvers_to_try.extend([cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS])

    result_status = None
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
                result_status = prob.status
                break
        except Exception:
            continue

    if b_var.value is None:
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64).flatten()
    b_val = np.maximum(b_val, 0)
    b_val[np.isnan(b_val)] = 0
    s_val = b_val.sum()
    if s_val < 1e-15:
        return None, None
    b_val /= s_val
    M_val = float(M_var.value) if M_var.value is not None else fast_conv(a_fixed * 0 + b_val).max()
    return b_val, max(0.0, M_val)


def find_lp_solver():
    """Find the best available LP solver."""
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS, cp.GLPK, cp.CVXOPT]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def nesterov_lp_alternating(a0, time_limit, solver, max_iters=100000):
    """
    Nesterov-accelerated LP alternating method.
    Alternates between best-response LP solutions for a and b.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = compute_score(a0)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    stall_count = 0
    prev_best = best_score
    lp_times = []

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver)
        lp_time = time.monotonic() - lp_t0
        lp_times.append(lp_time)

        if b_new is None:
            continue

        # Nesterov dual averaging
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = best_a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = compute_score(a)
        score_y = compute_score(y)

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 30 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 15:
                # Perturb to escape
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(2, 15)
                noise = 0.02 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = np.maximum(y + noise, 0)
                y = y / y.sum()
                stall_count = 0

            if (it + 1) % 200 == 0:
                elapsed = time.monotonic() - t0
                avg_lp = np.mean(lp_times[-20:]) if len(lp_times) >= 20 else lp_time
                print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} lp={avg_lp*1000:.0f}ms")

        # Stop if LP is too slow
        if len(lp_times) > 30:
            avg_lp = np.mean(lp_times[-20:])
            if avg_lp > 3.0:
                print(f"    LP too slow ({avg_lp:.2f}s), stopping at iter {it+1}")
                break

    return best_a, best_score


def fista_optimize(a0, time_limit, n_steps=50000):
    """
    FISTA (Fast Iterative Shrinkage-Thresholding Algorithm) for:
    min 2n * max_k conv(a,a)[k] subject to a >= 0, sum(a) = 1.
    Uses log-sum-exp smoothing for near-differentiability.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = compute_score(a0)
    t0 = time.monotonic()

    x = a.copy()  # current iterate
    y = a.copy()  # lookahead
    t_k = 1.0
    T = 0.5  # temperature for log-sum-exp
    base_lr = 0.01
    n_conv = 2 * n - 1
    top_k = max(20, n_conv // 10)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute smoothed objective gradient
        b = fast_conv(y)
        if top_k < n_conv:
            idx = np.argpartition(b, -top_k)[-top_k:]
            b_top = b[idx]
            max_top = b_top.max()
            exp_top = np.exp((b_top - max_top) / T)
            w_top = exp_top / exp_top.sum()
            w = np.zeros(n_conv)
            w[idx] = w_top
        else:
            max_b = b.max()
            exp_all = np.exp((b - max_b) / T)
            w = exp_all / exp_all.sum()

        g = grad_conv_weighted(y, w)

        # Step size (adaptive)
        lr = base_lr / (1.0 + step * 1e-5)

        # FISTA update
        t_new = (1.0 + math.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        momentum = (t_k - 1.0) / t_new

        z = y - lr * g
        # Project onto simplex {a >= 0, sum(a) = 1}
        x_new = _project_simplex(z)
        y = x_new + momentum * (x_new - x)
        x = x_new
        t_k = t_new

        score = compute_score(x)
        if score < best_score:
            best_score = score
            best_a = x.copy()

        # Decrease temperature
        if step % 5000 == 0 and step > 0:
            T *= 0.7
        if step % 20000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA step={step} ({elapsed:.0f}s): score={best_score:.8f} T={T:.4f}")

    return best_a, best_score


def _project_simplex(v):
    """Project vector onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    csv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (csv - 1))[0][-1]
    theta = (csv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def multiplicative_weights(a0, time_limit, n_steps=100000):
    """
    Multiplicative weights (mirror descent with entropic regularization).
    Targets the active peaks of the auto-convolution.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = compute_score(a0)
    t0 = time.monotonic()
    eta = 0.05
    n_conv = 2 * n - 1
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_conv(a)
        max_b = b.max()

        # Active set: top 5% of peaks
        top_k = max(5, n_conv // 20)
        idx = np.argpartition(b, -top_k)[-top_k:]
        b_top = b[idx]
        max_top = b_top.max()
        # Soft weights within active set
        exp_w = np.exp((b_top - max_top) * 2.0)
        w_top = exp_w / exp_w.sum()
        w = np.zeros(n_conv)
        w[idx] = w_top

        # Gradient
        g = grad_conv_weighted(a, w)
        g = np.maximum(g, 0)

        # Multiplicative update
        a_new = a * np.exp(-eta * g)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 30:
            eta *= 0.7
            no_improve = 0
            if eta < 0.001:
                break

        if step % 30000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    MW step={step} ({elapsed:.0f}s): score={best_score:.8f} eta={eta:.5f}")

    return best_a, best_score

def interpolate_sequence(a, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0)
    ss = a_interp.sum()
    if ss > 1e-15:
        a_interp = a_interp / ss
    return a_interp


def perturb_sequence(a, rng, strength=0.03):
    """Create a perturbed version of the sequence."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 20)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    a_pert = np.maximum(a + perturbation, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s

        print(f"  Time: {t1:.0f}s")
        a_out, _ = adam_logspace(best_seq, t1, [10.0, 5.0, 3.0, 2.0, 1.0, 0.5], lr0=0.005)
        try_update(a_out, "adam_warm")

    # === Phase 2: FISTA ===
    print(f"\n=== Phase 2: FISTA ({best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.10, 150.0)
    if t2 > 15:
        print(f"  Time: {t2:.0f}s")
        a_out, _ = fista_optimize(best_seq, t2)
        try_update(a_out, "fista")

    # === Phase 3: Main LP Alternating ===
    print(f"\n=== Phase 3: Nesterov LP n={len(best_seq)} ({best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.25, 400.0)
    if t3 > 60:
        print(f"  Time: {t3:.0f}s")
        a_out, _ = nesterov_lp_alternating(best_seq, t3, solver)
        try_update(a_out, "nesterov_lp_1")

    # === Phase 4: Fine Adam ===
    print(f"\n=== Phase 4: Fine Adam ({best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.15, 250.0)
    if t4 > 20:
        print(f"  Time: {t4:.0f}s")
        a_out, _ = adam_logspace(best_seq, t4,
                                  [3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02],
                                  lr0=0.003, top_k_start=0.2, top_k_end=0.03)
        try_update(a_out, "adam_fine")

    # === Phase 5: Second LP ===
    print(f"\n=== Phase 5: LP Round 2 ({best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.15, 250.0)
    if t5 > 40:
        print(f"  Time: {t5:.0f}s")
        a_out, _ = nesterov_lp_alternating(best_seq, t5, solver)
        try_update(a_out, "nesterov_lp_2")

    # === Phase 6: Multiplicative Weights ===
    print(f"\n=== Phase 6: Multiplicative Weights ({best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.08, 100.0)
    if t6 > 15:
        print(f"  Time: {t6:.0f}s")
        a_out, _ = multiplicative_weights(best_seq, t6)
        try_update(a_out, "mult_weights")

    # === Phase 7: Multi-length exploration ===
    print(f"\n=== Phase 7: Multi-length ({best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        n_cur = len(best_seq)
        target_ns = [n for n in [1000, 1100, 1400, 1600] if n != n_cur and rem > 80]
        for n_try in target_ns[:2]:
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t_try = min(rem * 0.3, 120.0)
            print(f"  Trying n={n_try}, time={t_try:.0f}s")
            a_interp = interpolate_sequence(best_seq, n_try)
            # Quick Adam
            t_ad = t_try * 0.5
            a_out, _ = adam_logspace(a_interp, t_ad, [3.0, 1.5, 0.5, 0.2, 0.05], lr0=0.003)
            try_update(a_out, f"adam_n{n_try}")
            # Quick LP if time remains
            rem = deadline - time.monotonic()
            if rem > 30:
                t_lp = min(rem * 0.4, 50.0)
                a_out2, _ = nesterov_lp_alternating(a_out if a_out is not None else a_interp, t_lp, solver)
                try_update(a_out2, f"lp_n{n_try}")

    # === Phase 8: Perturbation escape ===
    print(f"\n=== Phase 8: Perturbation Escape ({best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        for trial in range(3):
            rem = deadline - time.monotonic()
            if rem < 25:
                break
            strength = 0.01 + 0.015 * trial
            a_pert = perturb_sequence(best_seq, rng, strength)
            t_sub = min(rem * 0.3, 40.0)
            a_out, _ = adam_logspace(a_pert, t_sub, [2.0, 1.0, 0.5, 0.2, 0.05, 0.01], lr0=0.003)
            try_update(a_out, f"pert_{trial}")

    # === Phase 9: Final combined polish ===
    print(f"\n=== Phase 9: Final Polish ({best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t_fin = min(rem * 0.3, 30.0)
        a_out, _ = adam_logspace(best_seq, t_fin, [0.5, 0.2, 0.1, 0.05, 0.01], lr0=0.001,
                                  top_k_start=0.1, top_k_end=0.02)
        try_update(a_out, "final_adam")

    rem = deadline - time.monotonic()
    if rem > 10:
        a_out, _ = fista_optimize(best_seq, min(rem * 0.4, 20.0))
        try_update(a_out, "final_fista")

    rem = deadline - time.monotonic()
    if rem > 8:
        a_out, _ = nesterov_lp_alternating(best_seq, min(rem * 0.5, 15.0), solver)
        try_update(a_out, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = compute_score(best_seq)
        best_score = min(best_score, fs)
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_eval = evaluate_sequence(final_list)
        print(f"\nFINAL: local={best_score:.8f}, evaluate_sequence={final_eval:.8f}, n={len(best_seq)}")
    else:
        best_seq = np.ones(1200) / 1200.0

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
