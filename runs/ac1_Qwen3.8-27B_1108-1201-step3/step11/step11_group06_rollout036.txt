import numpy as np

from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_cross(a, b):
    """Linear cross-convolution via FFT."""
    n = max(len(a), len(b))
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def evaluate_internal(a):
    """Fast local evaluation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix_sparse(a):
    """Build sparse matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    # T is size x n, T[k,j] = a[k-j] if 0 <= k-j < n, else 0
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(j, j + n):
            if k < size:
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(size, n))


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver, T_mat):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = [solver, cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP, cp.MOSEK]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                sv = b_val.sum()
                if sv < 1e-15:
                    return None, None
                return b_val / sv, max(0.0, float(M_var.value))
        except Exception:
            continue
    return None, None

def lp_alternating(a0, time_limit, solver, verbose=True):
    """LP alternating with Nesterov acceleration and adaptive step size."""

    t0 = time.monotonic()
    
    T_mat = build_conv_matrix_sparse(y)
    acc = a.copy()  # Accumulator for Nesterov

        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_b = solve_lp_best_response(y, solver, T_mat)
        if b_new is None:
            break
        
        # Nesterov step
        alpha = 2.0 / (it + 2)
        beta = (np.sqrt(1 + 4 * ((it) / (it + 2))**2) - 1) / 2  # momentum
        
        y_old = y.copy()
        # Update y with momentum
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = y_old
        y = y / sy
        
        # Update a (the "gradient" step averaged)
        acc = (1 - alpha) * acc + alpha * b_new
        acc = np.maximum(acc, 0)
        sa = acc.sum()
        if sa > 1e-15:
            a = acc / sa
        
        # Update constraint matrix
        T_mat = build_conv_matrix_sparse(y)
        
        # Evaluate
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if verbose and (it + 1) % 200 == 0:
            print(f"    LP iter={it+1} ({time.monotonic()-t0:.0f}s): score={best_score:.8f}")

def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def make_palindromic(a):
    """Make a sequence palindromic."""
    n = len(a)
    a_pal = (a + a[::-1]) / 2.0
    a_pal = np.maximum(a_pal, 0)
    s = a_pal.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a_pal / s


def perturb(a, rng, strength=0.02):
    """Create a perturbed version."""
    n = len(a)

    phase = rng.uniform(0, 2 * np.pi)
    freq2 = rng.integers(2, 15)
    phase2 = rng.uniform(0, 2 * np.pi)
    pert = strength * (np.cos(freq * x + phase) + 0.5 * np.sin(freq2 * x + phase2))
    a_pert = np.maximum(a + pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_internal(a):
    """Fast local evaluation (assumes a >= 0)."""

        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix_sparse(a):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for j in range(n):
        k_start = j
        k_end = min(j + n, size)
        for k in range(k_start, k_end):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(size, n))

def project_simplex(v):
    """Project vector onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def make_palindromic(a):
    """Make a sequence palindromic (a[i] = a[n-1-i])."""
    n = len(a)
    a_pal = (a + a[::-1]) / 2.0
    a_pal = np.maximum(a_pal, 0)
    s = a_pal.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a_pal / s

        print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses LP alternating, proximal gradient, MW, multi-length, and perturbation.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}, budget: {budget_s:.0f}s")
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_internal(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_internal(best_seq)
        print(f"Init: n={len(best_seq)}, score={best_score:.8f}")
    
    n_best = len(best_seq)
    
    # Phase 1: Quick MW warmup
    print(f"\n=== Phase 1: MW warmup n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.04, 60.0)
    if t1 > 10:
        a_mw, _ = multiplicative_weights(best_seq, t1, eta0=0.05)
        best_seq, best_score = try_update_state(best_seq, best_score, a_mw, "mw_1")
    
    # Phase 2: Main LP alternating (primary workhorse)
    print(f"\n=== Phase 2: LP main n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.38, 580.0)
    if t2 > 60:
        print(f"  Budget: {t2:.0f}s")
        a_lp, _ = lp_alternating(best_seq, t2, solver)
        best_seq, best_score = try_update_state(best_seq, best_score, a_lp, "lp_main")
    
    # Phase 3: Proximal gradient refinement
    print(f"\n=== Phase 3: Proximal Gradient (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t3 = min(rem * 0.12, 180.0)
        a_pg, _ = proximal_gradient(best_seq, t3, verbose=True)
        best_seq, best_score = try_update_state(best_seq, best_score, a_pg, "pg_1")
    
    # Phase 4: Palindromic LP
    print(f"\n=== Phase 4: Palindromic LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t4 = min(rem * 0.12, 180.0)
        a_pal = make_palindromic(best_seq)
        a_pal_lp, _ = lp_alternating(a_pal, t4, solver, verbose=False)
        best_seq, best_score = try_update_state(best_seq, best_score, a_pal_lp, "pal_lp")
    
    # Phase 5: Multi-length exploration
    print(f"\n=== Phase 5: Multi-length (score={best_score:.8f}) ===")
    lengths_to_try = [1000, 1500, 1800]
    for ln in lengths_to_try:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        if len(best_seq) == ln:
            continue
        t_ln = min(rem * 0.14, 220.0)
        if t_ln < 30:
            continue
        print(f"  Trying n={ln}, budget={t_ln:.0f}s")
        init_ln = interpolate_to_length(best_seq, ln)
        # Quick MW + PG
        a_ln, _ = multiplicative_weights(init_ln, min(t_ln * 0.12, 25), eta0=0.05, verbose=False)
        best_seq, best_score = try_update_state(best_seq, best_score, a_ln, f"mw_n{ln}")
        a_ln_pg, _ = proximal_gradient(a_ln, min(t_ln * 0.12, 25), verbose=False)
        best_seq, best_score = try_update_state(best_seq, best_score, a_ln_pg, f"pg_n{ln}")
        # LP on this length
        rem = deadline - time.monotonic()
        t_lp = min(rem * 0.35, t_ln * 0.6)
        if t_lp > 30:
            a_ln_lp, _ = lp_alternating(a_ln_pg, t_lp, solver, verbose=False)
            best_seq, best_score = try_update_state(best_seq, best_score, a_ln_lp, f"lp_n{ln}")
    
    # Phase 6: Perturbation escapes
    print(f"\n=== Phase 6: Perturbation (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        for trial in range(5):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            a_pert = perturb_sequence(best_seq, rng, strength=0.008 + 0.004 * trial)
            # Quick PG then LP
            t_pg = min(rem * 0.1, 30.0)
            a_pg2, _ = proximal_gradient(a_pert, t_pg, verbose=False)
            best_seq, best_score = try_update_state(best_seq, best_score, a_pg2, f"pg_pert{trial}")
            rem = deadline - time.monotonic()
            t_sub = min(rem * 0.15, 60.0)
            if t_sub > 15:
                a_opt, _ = lp_alternating(a_pg2, t_sub, solver, verbose=False)
                best_seq, best_score = try_update_state(best_seq, best_score, a_opt, f"lp_pert{trial}")
    
    # Phase 7: Final LP polish on global best
    print(f"\n=== Phase 7: Final LP polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t7 = min(rem * 0.3, 100.0)
        a_fin, _ = lp_alternating(best_seq, t7, solver, verbose=False)
        best_seq, best_score = try_update_state(best_seq, best_score, a_fin, "final_lp")
    
    # Phase 8: Final PG polish
    print(f"\n=== Phase 8: Final PG polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t8 = min(rem * 0.4, 60.0)
        a_fin_pg, _ = proximal_gradient(best_seq, t8, verbose=False)
        best_seq, best_score = try_update_state(best_seq, best_score, a_fin_pg, "final_pg")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_internal(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} n={len(best_seq)}")
    
    if best_seq is None:
        best_seq = np.ones(1200) / 1200
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
