"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Peak-aware multiplicative weights with backtracking
2. LP game iteration with Nesterov acceleration (efficient sparse matrix updates)
3. Multi-stage LogSumExp L-BFGS with rapid cooling schedule
4. Palindromic constraint exploitation for speedup
5. Adaptive multi-length exploration
6. Frequency-domain guided perturbation for escape from local minima
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def precompute_conv_structure(n):
    """Precompute (rows, cols) structure of the convolution matrix T where T@b = conv(a_ref, b).
    The values depend on a_ref, but the structure (row, col) pairs are fixed."""
    size = 2 * n - 1
    rows_list = []
    cols_list = []
    for j in range(n):
        for k in range(j, min(j + n, size)):
            rows_list.append(k)
            cols_list.append(j)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def make_conv_matrix(rows, cols, a_ref):
    """Build sparse convolution matrix using precomputed structure."""
    n = len(a_ref)
    size = 2 * n - 1
    offset = cols  # k - j for each entry
    vals = a_ref[offset.astype(np.int64)]
    return csr_matrix((vals, (rows, cols)), shape=(size, n))


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, T_mat, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    for s in [solver, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        if s is None:
            continue
        try:
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                sv = b_val.sum()
                if sv < 1e-15:
                    return None, None
                b_val = b_val / sv
                return b_val, float(max(0, M_var.value))
        except Exception:
            continue
    return None, None


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_grad(z, n, T):
    """Compute LogSumExp objective and gradient w.r.t. z for softmax parameterization."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T
    scaled_shifted = scaled - scaled.max()
    exp_terms = np.exp(scaled_shifted)
    lse = max_b + T * (scaled.max() + np.log(exp_terms.sum() + 1e-300))
    obj = 2.0 * n * lse
    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def multiplicative_weights_peaked(a0, time_limit, n_steps, eta0=0.04, n_peaks=5):
    """Peak-aware multiplicative weights targeting top-k peaks with backtracking."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        # Target top n_peaks positions
        top_k = min(n_peaks, len(b))
        peak_positions = np.argsort(b)[-top_k:]
        # Weight proportional to how close to max
        peak_vals = b[peak_positions]
        threshold = max_b * (1.0 - 1e-3)
        active = peak_positions[peak_vals >= threshold]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        w_active = b[active] / b[active].sum()
        g = np.zeros(n)
        for w, k in zip(w_active, active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += w * a[k - idx]
        log_mult = -eta * g
        log_mult -= log_mult.max()
        a_new = a * np.exp(log_mult)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.02, 0.25)
        else:
            eta *= 0.75
            if eta < 1e-10:
                break
        if step % 50000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    MW {elapsed:.0f}s: score={best_score:.8f} eta={eta:.6f}")
    return best_a, best_score


class LSEFunc:
    def __init__(self, n, T, deadline):
        self.n = n
        self.T = T
        self.deadline = deadline

    def __call__(self, z):
        if time.monotonic() > self.deadline:
            return 1e15
        obj, _ = logsumexp_obj_grad(z, self.n, self.T)
        return obj


class LSEJac:
    def __init__(self, n, T, deadline):
        self.n = n
        self.T = T
        self.deadline = deadline

    def __call__(self, z):
        if time.monotonic() > self.deadline:
            return np.zeros(len(z))
        _, g = logsumexp_obj_grad(z, self.n, self.T)
        return g


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage=1000):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        func = LSEFunc(n, stage_T, deadline)
        jac = LSEJac(n, stage_T, deadline)
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-28,
                         'gtol': 1e-20, 'maxfun': maxiter_per_stage * 300,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass
    return softmax_from_z(z)

def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 5 * x + 0.5)
        a = np.maximum(a, 0)
    elif choice == 7:
        a = 1.0 / (1.0 + 3.0 * np.abs(x - 0.5))
    elif choice == 8:
        # Noise-like (good for flat autocorrelation)
        a = np.abs(rng.standard_normal(n)) + 0.1
    else:
        # Wavelet-like
        a = np.abs(np.cos(2 * np.pi * x) * np.exp(-2 * (x - 0.5) ** 2)) + 0.05 * np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_palindromic(a):
    """Make a sequence palindromic."""
    n = len(a)
    a_pal = (a + a[::-1]) / 2.0
    a_pal = np.maximum(a_pal, 0)
    s = a_pal.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a_pal / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Fast local evaluation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def perturb_with_spectrum(a, rng, strength=0.02):
    """Perturb using spectral components that target peak positions."""

    phase = rng.uniform(0, 2 * np.pi)

def coordinate_descent_peak_shave(a, time_limit, max_rounds=500):
    """Directly reduce the peak by adjusting elements that contribute to it."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = 0.01

    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))
        # Elements contributing to peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        # Reduce contributing elements, increase non-contributing
        contrib = a[js:je+1].copy()
        total_contrib = contrib.sum()
        if total_contrib < 1e-15:
            break
        # Reduce by a small fraction
        frac = eta
        a[js:je+1] = contrib * (1 - frac)
        # Distribute to non-contributing positions
        non_contrib_mask = np.ones(n, dtype=bool)
        non_contrib_mask[js:je+1] = False
        n_non = non_contrib_mask.sum()
        if n_non > 0:
            added = total_contrib * frac / n_non
            a[non_contrib_mask] += added
        a = np.maximum(a, 0)
        s_new = a.sum()
        if s_new < 1e-15:
            a = best_a.copy()
            break
        a = a / s_new
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
            # Slowly decrease eta
            eta *= 0.99
        else:
            a = best_a.copy()
            eta *= 0.7
            if eta < 1e-8:
                break
    return best_a

        a_ln_mw, _ = multiplicative_weights_peaked(init_ln, t_ln * 0.2, n_steps=50000, eta0=0.03, n_peaks=3)
        try_update(a_ln_mw, f"mw_n{ln}")
        rem = deadline - time.monotonic()
        t_lp = min(rem * 0.4, t_ln * 0.6)
        if t_lp > 30:
            a_ln_lp, _ = nesterov_lp_refine(a_ln_mw if a_ln_mw is not None else init_ln, t_lp, solver, n_max_iters=500000)
            try_update(a_ln_lp, f"lp_n{ln}")

    # === Phase 6: Coordinate descent peak shaving ===
    print(f"\n=== Phase 6: Peak shaving (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t6 = min(rem * 0.08, 60.0)
        a_cs = coordinate_descent_peak_shave(best_seq, t6, max_rounds=2000)
        try_update(a_cs, "peak_shave")

    # === Phase 7: LP + LSE combination on best ===
    print(f"\n=== Phase 7: LP+LSE combo (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        # Short LP
        t7a = min(rem * 0.3, 120.0)
        if t7a > 20:
            a_lp2, _ = nesterov_lp_refine(best_seq, t7a, solver, n_max_iters=300000)
            try_update(a_lp2, "lp_2")
        # Then LSE
        rem = deadline - time.monotonic()
        if rem > 30:
            t7b = min(rem * 0.5, 180.0)
            T_stages2 = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
            a_lse2 = lbgfs_refine(best_seq, t7b, T_stages2, maxiter_per_stage=800)
            try_update(a_lse2, "lse_2")

    # === Phase 8: Perturbation escapes ===
    print(f"\n=== Phase 8: Perturbation (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        for trial in range(3):
            rem = deadline - time.monotonic()
            if rem < 25:
                break
            a_pert = perturb_with_spectrum(best_seq, rng, strength=0.01 + 0.005 * trial)
            t_sub = min(rem * 0.3, 50.0)
            a_opt, _ = nesterov_lp_refine(a_pert, t_sub, solver, n_max_iters=200000)
            try_update(a_opt, f"pert{trial}")

    # === Phase 9: Final polish ===
    print(f"\n=== Phase 9: Final polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t9 = min(rem * 0.5, 60.0)
        T_final = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
        a_fin = lbgfs_refine(best_seq, t9, T_final, maxiter_per_stage=1000)
        try_update(a_fin, "final_lse")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} n={len(best_seq)}")
    else:
        best_seq = np.ones(1200) / 1200
        best_score = 2.0
        print(f"\nFALLBACK: score={best_score:.8f} n=1200")

    # Ensure output is valid
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    return [float(x) for x in best_seq]
