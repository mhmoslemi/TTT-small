def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def lse_and_weights(conv_vals, T):
    """Compute log-sum-exp and softmax weights."""
    max_val = conv_vals.max()
    exp_vals = np.exp((conv_vals - max_val) / T)
    sum_exp = exp_vals.sum()
    lse = max_val + T * np.log(sum_exp)
    w = exp_vals / sum_exp
    return lse, w

def compute_grad_fft(w, a):

Let me also think about a completely different approach: **direct QP formulation**.

The problem: min M s.t. a^T Q_k a <= M for all k, sum(a) = 1, a >= 0.

This is still non-convex, but we could use a second-order method (Newton) on the KKT conditions.

mu is the vector of multipliers for a >= 0 (mu_j >= 0, mu_j * a_j = 0).

This is a system of nonlinear equations that could be solved with a Newton method, but it's complex and may not be more efficient than the LP alternating.

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def lse_and_weights(conv_vals, T):
    max_val = conv_vals.max()
    shifted = (conv_vals - max_val) / T
    # Clip to prevent overflow
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    sum_exp = exp_vals.sum()
    if sum_exp < 1e-300:
        w = np.ones(len(conv_vals)) / len(conv_vals)
        lse = max_val
    else:
        w = exp_vals / sum_exp
        lse = max_val + T * np.log(sum_exp)
    return lse, w

def compute_grad_fft(w, a):
    n = len(a)
    fft_n = 1 << (3 * n - 2).bit_length()
    A = rfft(a, fft_n)
    W = rfft(w, fft_n)
    c = irfft(np.conj(A) * W, fft_n)[:n]
    d = irfft(A * W, fft_n)[:n]
    return c + d

    s = a.sum()

        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

    a = np.exp(z_shifted)
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return a / s

def _z_from_a(a):

def _obj_fn(z):

        idx = np.argpartition(conv, -k)[-k:]
        conv_sub = conv[idx]

        w = np.zeros(len(conv))
        w[idx] = w_sub
    else:
        lse, w = lse_and_weights(conv, _g_temperature)
    return 2.0 * _g_n * lse
