    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_with(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_grad(z, n, T):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T)
    lse = max_b + T * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_refine(a0, time_limit, T_stages, deadline):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    n_stages = len(T_stages)

    for stage_i, stage_T in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 2:
            break

        def func(z_in):
            if time.monotonic() > deadline:
                return 1e15
            obj, _ = logsumexp_obj_grad(z_in, n, stage_T)
            return obj

        def jac(z_in):
            if time.monotonic() > deadline:
                return np.zeros_like(z_in)
            _, g = logsumexp_obj_grad(z_in, n, stage_T)
            return g

        iters = max(50, int(remaining * 30 / max(1, n_stages - stage_i)))
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-30,
                         'gtol': 1e-20, 'maxfun': iters * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
        if time.monotonic() > deadline:
            break

    return softmax_from_z(z)


def peak_shave(a, time_limit, n_steps=200000, top_k=5):
    """
    Peak shaving with focus on top-k peaks and adaptive weighting.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return None, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 0.03
    scale = 1.0
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        if max_b < 1e-15:
            break

        # Find top peaks
        sorted_idx = np.argsort(b)[::-1]
        active = sorted_idx[:top_k]
        # Include all peaks within 0.5% of max
        threshold = max_b * (1.0 - 5e-4)
        more_active = np.where(b >= threshold)[0]
        active = np.unique(np.concatenate([active, more_active]))
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # Weight peaks by their excess (recency bias)
        excess = b[active] - b.min()
        if excess.max() < 1e-30:
            weights = np.ones(len(active))
        else:
            weights = (excess / excess.max()) ** 2
            weights = weights / weights.sum()

        g = np.zeros(n)
        for i, k in enumerate(active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += weights[i] * a[k - idx]

        eta_eff = eta * scale
        log_scale_vec = -eta_eff * g
        log_scale_vec -= log_scale_vec.max()
        a_new = a * np.exp(log_scale_vec)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 20:
            scale *= 0.65
            no_improve = 0
            if scale < 0.003:
                break

    return best_a, best_score


def nesterov_alternating(a0, time_limit, n_max_iters=5000000):
    """
    Fast alternating subgradient with Nesterov acceleration.
    Alternates between optimizing a and b (both on simplex).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    b = a.copy()
    best_a = a.copy()
    b_conv = fast_convolve(a)
    best_score = 2.0 * n * b_conv.max()
    t0 = time.monotonic()

    # Nesterov momentum variables
    y_a = a.copy()
    y_b = b.copy()
    v_a = np.zeros(n)
    v_b = np.zeros(n)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        alpha = 2.0 / (it + 2)

        # --- Update b given y_a ---
        conv_ab = fast_convolve_with(y_a, b)
        max_ab = conv_ab.max()
        # Subgradient w.r.t. b: average of rows of T(y_a) at active peaks
        threshold = max_ab * (1.0 - 1e-3)
        active = np.where(conv_ab >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(conv_ab)])

        g_b = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g_b[idx] += y_a[k - idx]
        g_b /= len(active)

        # Multiplicative (mirror descent) step for b
        lr = 0.05 * alpha * (it + 2) / max(1, it * 0.01 + 1)
        lr = min(lr, 0.1)
        log_update = -lr * g_b
        log_update -= log_update.max()
        b_new = b * np.exp(log_update)
        b_new = np.maximum(b_new, 0)
        sb = b_new.sum()
        if sb < 1e-15:
            b_new = b.copy()
        else:
            b_new = b_new / sb

        # Nesterov extrapolation
        y_b = (1.0 - alpha) * y_b + alpha * b_new
        y_b = np.maximum(y_b, 0)
        sy = y_b.sum()
        if sy > 1e-15:
            y_b = y_b / sy
        b = y_b.copy()

        # --- Update a given y_b ---
        conv_ba = fast_convolve_with(y_b, a)
        max_ba = conv_ba.max()
        threshold2 = max_ba * (1.0 - 1e-3)
        active2 = np.where(conv_ba >= threshold2)[0]
        if len(active2) == 0:
            active2 = np.array([np.argmax(conv_ba)])

        g_a = np.zeros(n)
        for k in active2:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g_a[idx] += y_b[k - idx]
        g_a /= len(active2)

        log_update_a = -lr * g_a
        log_update_a -= log_update_a.max()
        a_new = a * np.exp(log_update_a)
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / sa

        y_a = (1.0 - alpha) * y_a + alpha * a_new
        y_a = np.maximum(y_a, 0)
        sya = y_a.sum()
        if sya > 1e-15:
            y_a = y_a / sya
        a = y_a.copy()

        # Check score
        if (it + 1) % 10 == 0:
            b_conv = fast_convolve(a)
            score = 2.0 * n * b_conv.max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

            b_conv_y = fast_convolve(y_a)
            score_y = 2.0 * n * b_conv_y.max()
            if score_y < best_score:
                best_score = score_y
                best_a = y_a.copy()

            if (it + 1) % 10000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NesterovAlt iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    # Final check
    b_conv = fast_convolve(a)
    score = 2.0 * n * b_conv.max()
    if score < best_score:
        best_score = score
        best_a = a.copy()
    b_conv = fast_convolve(y_a)
    score = 2.0 * n * b_conv.max()
    if score < best_score:
        best_score = score
        best_a = y_a.copy()

    return best_a, best_score


def spectral_flatten(a, n_iters=100, blend=0.3, deadline=None):
    """Gerchberg-Saxton style spectral flattening."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)

        mags = np.abs(F)
        non_dc = mags[1:]
        if non_dc.min() < 1e-15:
            target = np.mean(non_dc)
        else:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        phases = np.angle(F)
        F_new = F.copy()
        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend) * mags[k] + blend * target
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])

        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new

    return a


def coordinate_refine(a, time_limit, n_sweeps=100):
    """
    Coordinate descent exploiting the fact that fixing all but one coordinate
    gives a 1D convex problem (max of convex quadratics).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            # Parametrize: a[i] = x, others scaled to maintain sum = 1
            # a[j] = (1-x)/(n-1) for j != i
            # Find optimal x by ternary search on [0, 1]
            x_lo = 0.0
            x_hi = 1.0

            # Quick check: evaluate at current value and a few test points
            x_current = a[i]

            def obj_at_x(x):
                a_trial = np.full(n, (1.0 - x) / (n - 1))
                a_trial[i] = x
                b = fast_convolve(a_trial)
                return b.max()

            # Golden section search (10 iterations ≈ good precision)
            gr = (np.sqrt(5) - 1) / 2
            c = x_hi - gr * (x_hi - x_lo)
            d = x_lo + gr * (x_hi - x_lo)
            fc = obj_at_x(c)
            fd = obj_at_x(d)

            for _ in range(15):
                if fc < fd:
                    x_hi = d
                    d = c
                    fd = fc
                    c = x_hi - gr * (x_hi - x_lo)
                    fc = obj_at_x(c)
                else:
                    x_lo = c
                    c = d
                    fc = fd
                    d = x_lo + gr * (x_hi - x_lo)
                    fd = obj_at_x(d)

            x_new = (c + d) / 2
            a[i] = x_new
            # Renormalize (already normalized by construction)
            s_check = a.sum()
            if abs(s_check - 1.0) > 1e-10:
                a = a / s_check

        # Check improvement
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_a


def perturbed_sequence(a, rng, strength=0.03):
    """Create a perturbed version of sequence a."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 12)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    # Add a second frequency component
    freq2 = rng.integers(1, 20)
    phase2 = rng.uniform(0, 2 * np.pi)
    perturbation += strength * 0.5 * np.cos(freq2 * x + phase2)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

            a_sp_try = spectral_flatten(init_try, n_iters=20, blend=0.3,
                                        deadline=time.monotonic() + 15)
            a_opt_try, _ = nesterov_alternating(a_sp_try, t_sub * 0.6, n_max_iters=1000000)
            local_update(a_opt_try, f"nesterov_n{n_try}")

            rem_sub2 = deadline - time.monotonic()
            if rem_sub2 > 20 and best_seq is not None and len(best_seq) == n_try:
                T_try = [3.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
                a_lb_try = lbgfs_refine(best_seq, min(rem_sub2, 40.0), T_try,
                                        deadline=time.monotonic() + min(rem_sub2, 40.0))
                local_update(a_lb_try, f"lbgfs_n{n_try}")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        # Ultra-fine L-BFGS
        t_fin = min(rem * 0.4, 60.0)
        if t_fin > 10:
            T_fin = [1.0, 0.5, 0.3, 0.2, 0.1, 0.06, 0.04, 0.025,
                     0.015, 0.01, 0.007, 0.005, 0.003, 0.002]
            a_fin = lbgfs_refine(best_seq, t_fin, T_fin,
                                 deadline=time.monotonic() + t_fin)
            local_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 15:
        # Final peak shave
        t_ps = min(rem * 0.4, 30.0)
        if t_ps > 5:
            a_ps, _ = peak_shave(best_seq, t_ps, n_steps=100000, top_k=3)
            local_update(a_ps, "final_peak")

    rem = deadline - time.monotonic()
    if rem > 10:
        # Final coordinate refine
        t_cr = min(rem * 0.5, 20.0)
        if t_cr > 5:
            a_cr = coordinate_refine(best_seq, t_cr, n_sweeps=20)
            local_update(a_cr, "final_coord")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_start
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}) at {elapsed:.1f}s")
        test_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        try:
            eval_score = evaluate_sequence(test_seq)
            print(f"Verified with evaluate_sequence: {eval_score:.8f}")
            if eval_score < best_score:
                best_score = eval_score
        except Exception as e:
            print(f"Verification error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
