    return irfft(A * A, fft_size)[:size]

    """Safely evaluate a sequence using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a_fixed):
    """Build the (2n-1) x n convolution matrix T where (T@b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0. Returns (b, M)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """
    Nesterov-accelerated LP best-response with EMA tracking and adaptive schedule.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    # Exponential moving average (faster-adapting than simple running average)
    ema_a = a.copy()
    ema_alpha = 0.05  # Initial EMA rate, will adapt

    # Also track a longer-window average
    avg_a = np.zeros(n)
    n_avg = 0

    iter_count = 0
    stall_count = 0
    stall_threshold = 0.99999

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        # Solve LP best response
        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue
        iter_count += 1

        # Adaptive Nesterov schedule: start with larger steps, decay
        if t < 100:
            alpha_t = 2.0 / (t + 1.0)
        else:
            alpha_t = 2.0 / (t + 2.0)

        # Nesterov update
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_new_s = y_new.sum()
        if y_new_s < 1e-15:
            y_new = y.copy()
            y_new_s = 1.0
        y_new = y_new / y_new_s

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            a_s = a_new.sum()
            if a_s < 1e-15:
                a_new = y_new.copy()
                a_s = 1.0
            a_new = a_new / a_s

        y = y_new
        a = a_new

        # Score a
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        else:
            if score_a > best_score * stall_threshold:
                stall_count += 1

        # Score y
        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        # EMA update
        ema_a = (1 - ema_alpha) * ema_a + ema_alpha * a_new
        if t > 50 and t % 5 == 0:
            score_ema = 2.0 * n * fast_autocorr(ema_a).max()
            if score_ema < best_score:
                best_score = score_ema
                best_a = ema_a.copy()
                stall_count = 0

        # Running average
        avg_a += a_new
        n_avg += 1
        if n_avg >= 20 and n_avg % 25 == 0:
            avg_check = avg_a / n_avg
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
                stall_count = 0

        # Adapt EMA rate: increase if stalling (want faster adaptation)
        if stall_count > 200:
            ema_alpha = min(0.1, ema_alpha * 1.05)

        if t % 500 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}, ema_alpha={ema_alpha:.4f}")

        # Early stopping if severely stalled
        if stall_count > 3000 and t > 500:
            if verbose:
                print(f"      Early stop at t={t}, stall={stall_count}")
            break

    return best_a, best_score, iter_count


def qp_active_set_refinement(a0, time_limit, n_active=30):
    """
    QP refinement: identify active constraints (peaks near max) and solve QP.
    min M s.t. (a*a)_k <= M for k in active set, sum(a)=1, a>=0
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    b = fast_autocorr(a)
    max_b = b.max()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * max_b

    for round_num in range(20):
        if time.monotonic() - t0 > time_limit:
            break

        # Identify active lags
        threshold = max_b * (1 - 0.01)
        active_lags = np.where(b >= threshold)[0]
        if len(active_lags) > n_active:
            active_lags = np.argsort(b[active_lags])[::-1][:n_active] + np.where(b >= threshold)[0][:n_active]
            active_lags = np.unique(active_lags[:n_active])

        # Build QP: min M s.t. (a*a)_k <= M for k in active_lags, sum(a)=1, a>=0
        a_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()

        constraints = [cp.sum(a_var) == 1, M_var >= 0]
        for k in active_lags:
            # (a*a)_k = sum_j a_j * a_{k-j} (with valid indices)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            other_idx = k - idx
            q_expr = cp.sum(a_var[idx] * a_var[other_idx])
            constraints.append(q_expr <= M_var)

        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iter=10000)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, max_iter=10000)
            except Exception:
                break

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            try:
                prob.solve(solver=cp.CLARABEL)
            except Exception:
                break
            if prob.status not in ('optimal', 'optimal_inaccurate'):
                break

        a_new = np.asarray(a_var.value, dtype=np.float64)
        a_new = np.maximum(a_new, 0)
        a_new_s = a_new.sum()
        if a_new_s < 1e-15:
            break
        a_new = a_new / a_new_s

        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            b = fast_autocorr(a)
            max_b = b.max()
            print(f"      QP round {round_num}: improved to {best_score:.6f}")
        else:
            # Try mixing
            for frac in [0.3, 0.5, 0.7]:
                a_mix = (1 - frac) * a + frac * a_new
                a_mix = np.maximum(a_mix, 0)
                a_mix_s = a_mix.sum()
                if a_mix_s < 1e-15:
                    continue
                a_mix = a_mix / a_mix_s
                score_mix = 2.0 * n * fast_autocorr(a_mix).max()
                if score_mix < best_score:
                    best_score = score_mix
                    best_a = a_mix.copy()
                    a = a_mix
                    b = fast_autocorr(a)
                    max_b = b.max()
                    print(f"      QP mix {frac}: improved to {best_score:.6f}")
                    break
            else:
                # No improvement, try with more active constraints
                if round_num >= 3:
                    break
                # Widen active set
                n_active = min(n_active * 2, 80)

    return best_a, best_score


def mirror_descent_refine(a0, time_limit, lr_start=0.1, lr_end=0.001, n_iters=None):
    """
    Mirror descent (exponential gradient) on the simplex.
    minimizes max_k (a*a)_k using the entropy mirror map.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    if n_iters is None:
        n_iters = 5000

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Learn rate: decay over time
        progress = it / max(1, n_iters)
        lr = lr_start * (lr_end / lr_start) ** progress

        # Compute subgradient of max_k (a*a)_k
        b = fast_autocorr(a)
        max_b = b.max()
        # Subgradient: 2 * shift_k(a) where k is argmax
        # For average subgradient: use top few peaks
        top_k = np.argsort(b)[::-1][:5]
        # Weight by how close to max
        weights = np.exp((b[top_k] - max_b) * 10)
        weights = weights / weights.sum()

        # Compute gradient as weighted average of 2*shift_k(a)
        grad = np.zeros(n)
        for w, k in zip(weights, top_k):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            other = k - idx
            # grad contribution from lag k: 2*w*a[other] added to positions idx
            # and 2*w*a[idx] added to positions other
            grad[idx] += 2 * w * a[other]
            grad[other] += 2 * w * a[idx]

        # Mirror descent step: a_new(i) = a(i) * exp(-lr * grad(i)) / Z
        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a - lr * grad
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new_s = a_new.sum()
        if a_new_s < 1e-30:
            break
        a_new = a_new / a_new_s

        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        # Use a_new as next iterate regardless (for the algorithm to work)
        a = a_new

        if it % 500 == 0:
            print(f"      MD it={it}: best={best_score:.6f}, cur={2.0*n*fast_autocorr(a).max():.6f}")

    return best_a, best_score


def peak_shaving_v2(a0, time_limit, seed=123):
    """
    Improved peak shaving with:
    - Top peak identification with contribution analysis
    - Multi-fraction line search
    - Random restarts from perturbed states
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    max_stall = 500

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        # Top peaks
        n_peaks = min(10, 2 * n - 1)
        top_peaks = np.argsort(b)[::-1][:n_peaks]

        improved = False
        for k in top_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            total_contrib = contrib.sum()
            if total_contrib < 1e-15:
                continue

            # Find top contributors
            contrib_sorted = np.argsort(contrib)[::-1][:5]
            for ci in contrib_sorted:
                src = idx[ci]
                src_other = k - src

                # Find best target: position that contributes least to top peaks
                peak_positions = set()
                for k2 in top_peaks[:5]:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    peak_positions.update(range(js2, je2 + 1))

                # Score each candidate position by how much it would help
                candidates = []
                for j in range(n):
                    if j == src and j == src_other:
                        continue
                    # Estimate reduction: how much does moving mass to j help?
                    # It helps if j is far from peak positions
                    min_dist_to_peak = min(abs(j - ps) for ps in peak_positions) if peak_positions else 0
                    score_j = min_dist_to_peak - a[j] * n * 0.1
                    candidates.append((score_j, j))
                candidates.sort(reverse=True)

                # Try top candidates
                for _, dst in candidates[:10]:
                    if dst == src and dst == src_other:
                        continue
                    for frac in [0.002, 0.005, 0.01, 0.02, 0.04, 0.06]:
                        amt = min(a[src], a[src_other] if src != src_other else a[src]) * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        if src != src_other:
                            a_trial[src_other] -= amt * 0.5
                        a_trial[dst] += amt
                        if (a_trial < -1e-12).any():
                            continue
                        a_trial = np.maximum(a_trial, 0)
                        a_trial_s = a_trial.sum()
                        if a_trial_s < 1e-15:
                            continue
                        a_trial = a_trial / a_trial_s
                        score = 2.0 * n * fast_autocorr(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation
            if stall % 10 == 0:
                n_perturb = rng.integers(1, 5)
                sources = rng.choice(n, size=n_perturb, replace=False)
                targets = rng.choice(n, size=n_perturb, replace=False)
                a_trial = a.copy()
                for src, dst in zip(sources, targets):
                    frac = rng.uniform(0.003, 0.02)
                    amt = a_trial[src] * frac
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                if (a_trial < -1e-12).any():
                    continue
                a_trial = np.maximum(a_trial, 0)
                a_trial_s = a_trial.sum()
                if a_trial_s < 1e-15:
                    continue
                a_trial = a_trial / a_trial_s
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0

            if stall > max_stall:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: multi-phase optimization at multiple lengths with
    LP Nesterov, QP refinement, mirror descent, and peak shaving.
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    # Get parent sequence
    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Determine starting length
    n_base = len(best_seq) if best_seq is not None else 1600

    if parent is None:
        n_base = 1600
        x = np.linspace(0, np.pi, n_base)
        best_seq = np.cos(x / 2) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # Phase 1: Extended LP Nesterov at base length (most time)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: LP Nesterov at n={n_base} (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.35, 550.0)
    if t1 > 60:
        a_out, sc_out, iters = nesterov_lp_optimize(best_seq, t1, solver)
        print(f"  LP Nesterov done: {iters} iters, score={sc_out:.6f}")
        try_update(a_out, f"lp_n{n_base}")

    # Phase 2: QP Active Set Refinement
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: QP Refinement at n={n_base} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.12, 150.0)
    if t2 > 20:
        a_out, sc_out = qp_active_set_refinement(best_seq, t2, n_active=25)
        print(f"  QP done: score={sc_out:.6f}")
        try_update(a_out, f"qp_n{n_base}")

    # Phase 3: Mirror Descent
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: Mirror Descent at n={n_base} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.10, 120.0)
    if t3 > 15:
        a_out, sc_out = mirror_descent_refine(best_seq, t3, lr_start=0.05, lr_end=0.0005)
        print(f"  MD done: score={sc_out:.6f}")
        try_update(a_out, f"md_n{n_base}")

    # Phase 4: Try nearby lengths
    for n_try in [n_base - 100, n_base + 100, n_base + 200]:
        if n_try < 200:
            continue
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining < 120:
            break
        print(f"\n=== Phase 4: LP Nesterov at n={n_try} (remaining={remaining:.0f}s) ===")
        t4 = min(remaining * 0.20, 200.0)
        if t4 < 40:
            continue
        init_try = interpolate_sequence(best_seq, n_try)
        a_out, sc_out, iters = nesterov_lp_optimize(init_try, t4, solver, verbose=False)
        print(f"  LP at n={n_try}: {iters} iters, score={sc_out:.6f}")
        try_update(a_out, f"lp_n{n_try}")

        # Quick QP on the best at this length
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining > 60:
            t_qp = min(remaining * 0.08, 60.0)
            if t_qp > 10:
                a_qp, sc_qp = qp_active_set_refinement(a_out, t_qp, n_active=20)
                try_update(a_qp, f"qp_n{n_try}")

    # Phase 5: Peak shaving on global best
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: Peak Shaving at n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.25, 200.0)
    if t5 > 10:
        a_out, sc_out = peak_shaving_v2(best_seq, t5, seed=456)
        print(f"  Peak shave 1: score={sc_out:.6f}")
        try_update(a_out, "peak_shave_1")

    # Phase 6: Another LP Nesterov round (shorter) to exploit improvements
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: LP Nesterov round 2 at n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t6 = min(remaining * 0.25, 200.0)
    if t6 > 30:
        a_out, sc_out, iters = nesterov_lp_optimize(best_seq, t6, solver, verbose=False)
        print(f"  LP round 2: {iters} iters, score={sc_out:.6f}")
        try_update(a_out, "lp_round2")

    # Phase 7: Final QP + Peak shave
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 7: Final QP + Peak Shave (remaining={remaining:.0f}s) ===")
    if remaining > 30:
        t7_qp = min(remaining * 0.3, 60.0)
        if t7_qp > 10:
            a_out, sc_out = qp_active_set_refinement(best_seq, t7_qp, n_active=15)
            try_update(a_out, "final_qp")

    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 15:
        t7_ps = min(remaining * 0.5, 60.0)
        if t7_ps > 5:
            a_out, sc_out = peak_shaving_v2(best_seq, t7_ps, seed=789)
            print(f"  Final peak shave: score={sc_out:.6f}")
            try_update(a_out, "peak_shave_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
