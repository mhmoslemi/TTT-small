    v = np.exp(conv / t_weights)

            break

        s = len(support)

        for ii, i in enumerate(support):
            for jj, j in enumerate(support):

        b_aug = np.zeros(s+1)
        b_aug[s] = 1.0
        
        try:

        except np.linalg.LinAlgError:
            sol = np.linalg.lstsq(A_aug, b_aug, rcond=None)[0]
        
        a_s = sol[:s]
        c_val = sol[s]
        
        # Check for negative values
        if np.all(a_s >= -1e-12):

            a = a / a.sum()

        else:

Optimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Active-set pressure equalization (linear system solve)
2. LogSumExp-smoothed L-BFGS refinement
3. Bidirectional coordinate descent
4. Multi-n exploration with interpolation
5. Diversified multi-start

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.sparse import csr_matrix, lil_matrix

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_of(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

    n = len(a)

    exp_c = np.exp((conv - max_c) / T)
    w = exp_c / exp_c.sum()

    support = np.where(a > 1e-10)[0]

    for outer_it in range(n_outer):
        if time.time() - t0 > time_limit - 2:
            break

        conv = fast_convolve(a)
        max_conv = conv.max()

        # Find active peaks (within 1% of max)
        threshold = max_conv * 0.99
        active_peaks = np.where(conv >= threshold)[0]
        if len(active_peaks) == 0:
            active_peaks = np.array([np.argmax(conv)])

        # Build weight vector (uniform over active peaks)
        v = np.zeros(2 * n - 1)
        v[active_peaks] = 1.0 / len(active_peaks)

        # Build sparse M: M[i,j] = v[i+j] for valid indices
        # M[i,j] = sum_{m in active} (1/|A|) * [i+j = m]
        M = lil_matrix((n, n))
        for m in active_peaks:
            i_start = max(0, m - n + 1)
            i_end = min(n - 1, m)
            for i in range(i_start, i_end + 1):
                j = m - i
                if 0 <= j < n:
                    M[i, j] += 1.0 / len(active_peaks)
        M = M.tocsc()

        # Solve: M @ a_s = c * ones_s, sum(a_s) = 1
        # Augmented system: [M_S, -1; 1', 0] @ [a_S; c] = [0; 1]
        s = len(support)
        if s == 0:
            a = np.ones(n) / n
            support = np.arange(n)
            continue

        M_S = M[np.ix_(support, support)].toarray()

        A_aug = np.zeros((s + 1, s + 1))
        A_aug[:s, :s] = M_S
        A_aug[:s, s] = -1.0
        A_aug[s, :s] = 1.0
        A_aug[s, s] = 0.0
        b_aug = np.zeros(s + 1)
        b_aug[s] = 1.0

        try:
            # Add small regularization for stability
            A_aug[:s, :s] += 1e-8 * np.eye(s)
            sol = np.linalg.solve(A_aug, b_aug)
        except np.linalg.LinAlgError:
            try:
                sol = np.linalg.lstsq(A_aug, b_aug, rcond=None)[0]
            except Exception:
                continue

        a_S = sol[:s]
        c_val = sol[s]

        # Check for negative values - reduce support if needed
        if np.min(a_S) < -1e-6:
            # Remove most negative from support
            min_local = np.argmin(a_S)
            removed = support[min_local]
            support = np.delete(support, min_local)
            a[removed] = 0.0
            a = normalize(a)
            continue

        # Damped update
        a_new = np.zeros(n)
        a_new[support] = np.maximum(a_S, 0)
        a_new = normalize(a_new)

        # Also try a damped version
        a_damped = (1.0 - damp) * a + damp * a_new
        a_damped = normalize(a_damped)

        sc_new = score_of(a_new)
        sc_damped = score_of(a_damped)

        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
        elif sc_damped < best_score:
            best_score = sc_damped
            best_a = a_damped.copy()
            a = a_damped
        else:
            # Try smaller damping
            for d in [0.3, 0.1, 0.05]:
                a_try = (1.0 - d) * a + d * a_new
                a_try = normalize(a_try)
                sc_try = score_of(a_try)
                if sc_try < best_score:
                    best_score = sc_try
                    best_a = a_try.copy()
                    a = a_try
                    break
            else:
                # No improvement - try to expand support
                if len(support) < n:
                    # Add back the smallest removed element
                    zeros = np.where(a < 1e-10)[0]
                    if len(zeros) > 0:
                        # Add the one with highest "pressure"
                        a_trial = a.copy()
                        for z_idx in zeros[:5]:
                            a_trial[z_idx] = 1e-6
                        a_trial = normalize(a_trial)
                        conv_t = fast_convolve(a_trial)
                        pressures = np.zeros(n)
                        for m in active_peaks:
                            i_start = max(0, m - n + 1)
                            i_end = min(n - 1, m)
                            for i in range(i_start, i_end + 1):
                                j = m - i
                                if 0 <= j < n:
                                    pressures[i] += a_trial[j]
                        best_zero = zeros[np.argmax(pressures[zeros])]
                        a[best_zero] = 1e-4
                        a = normalize(a)
                        support = np.where(a > 1e-10)[0]

        if (outer_it + 1) % 30 == 0:
            print(f"    ActiveSet iter={outer_it+1} ({time.time()-t0:.0f}s): "
                  f"best={best_score:.9f}, |A|={len(active_peaks)}, |S|={len(support)}")

    return best_a, best_score


def bidirectional_coord_descent(a0, time_limit, n_steps=50000):
    """
    Coordinate descent that tries both increasing and decreasing values.
    Focuses on coordinates that contribute to the active peaks.
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.time()
    stall = 0
    rng = np.random.default_rng(777)

    for step in range(n_steps):
        if time.time() - t0 > time_limit - 2:
            break

        conv = fast_convolve(a)
        max_conv = conv.max()
        threshold = max_conv * 0.9995
        peaks = np.where(conv >= threshold)[0]

        improved = False
        # Try top 5 peaks
        for peak_k in peaks[:5]:
            if improved:
                break
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            # Top contributors
            top_m = min(10, len(idx))
            top_positions = idx[np.argsort(contribs)[::-1][:top_m]]

            for pos in top_positions:
                if a[pos] < 1e-14:
                    continue
                # Try both decreasing AND increasing
                factors = [0.999, 0.997, 0.993, 0.985, 0.97, 0.95,
                           1.01, 1.02, 1.04, 1.07, 1.12]
                for factor in factors:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    if a_trial[pos] < 0:
                        a_trial[pos] = 0
                    diff = a[pos] - a_trial[pos]
                    if abs(diff) < 1e-16:
                        continue
                    # Redistribute the difference to other coordinates proportionally
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others_sum = a_trial[mask].sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] += (-diff) * a_trial[mask] / others_sum
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = normalize(a_trial)
                    sc = score_of(a_trial)
                    if sc < best_score - 1e-12:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            if stall > 100:
                # Perturb: move mass from a peak contributor to a random non-contributor
                peak_k = int(np.argmax(conv))
                js = max(0, peak_k - n + 1)
                je = min(n - 1, peak_k)
                idx = np.arange(js, je + 1)
                contribs = a[idx] * a[peak_k - idx]
                top_pos = idx[np.argmax(contribs)]
                # Find a position NOT in the peak
                all_pos = np.arange(n)
                non_peak = np.setdiff1d(all_pos, idx)
                if len(non_peak) > 0:
                    target = non_peak[rng.integers(len(non_peak))]
                    amount = a[top_pos] * 0.01
                    a[top_pos] -= amount
                    a[target] += amount
                    a = normalize(a)
                stall = 0
        else:
            stall = 0

        if (step + 1) % 10000 == 0:
            print(f"    CoordDesc {time.time()-t0:.0f}s: score={best_score:.9f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def make_asymmetric_init(n, rng, kind=None):
    """Create asymmetric initial sequences."""
    if kind is None:
        kind = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if kind == 0:
        a = np.exp(-2.0 * x)
    elif kind == 1:
        a = np.exp(-3.0 * (1 - x))
    elif kind == 2:
        a = 0.3 + 0.7 * x
    elif kind == 3:
        a = np.exp(-4.0 * (x - 0.3) ** 2 / 0.05)
    elif kind == 4:
        a = np.exp(-1.5 * x) * (1 + 0.5 * np.sin(4 * np.pi * x))
        a = np.maximum(a, 0)
    elif kind == 5:
        # Bimodal asymmetric
        a = 0.6 * np.exp(-10 * (x - 0.25) ** 2) + 0.4 * np.exp(-10 * (x - 0.7) ** 2)
    elif kind == 6:
        # Power law
        a = (1 - x) ** 2 * (1 + 0.3 * x)
    else:
        # Cosine-based asymmetric
        a = np.cos(np.pi * x * 0.7) ** 2
        a = np.maximum(a, 0)
    a = np.maximum(a, 1e-10)
    s = a.sum()
    return a / s


def proposed_peaks_shift_init(n, rng):
    """
    Initialize by placing mass at positions that would create
    a shifted peak in the convolution.
    """
    # If we want the peak at position k in the convolution,
    # we need a[i] * a[k-i] to be large for many i
    # This is maximized when a is concentrated around k/2 and k-k/2 = k/2
    # So for a shifted peak, put mass at positions that are NOT symmetric about (n-1)/2
    
    center = (n - 1) / 2.0
    offset = n * (0.1 + 0.1 * rng.random())
    
    # Put mass off-center
    mu = center + offset
    sigma = n * 0.15
    x = np.arange(n)
    a = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    # Add a secondary bump
    mu2 = center - offset * 0.5
    a2 = 0.4 * np.exp(-0.5 * ((x - mu2) / (sigma * 0.7)) ** 2)
    a = a + a2
    a = np.maximum(a, 0)
    return normalize(a)


def multi_start_lse(a0, time_limit, n_starts=4):
    """Run LSE optimization from multiple perturbed starting points."""
    n = len(a0)
    rng = np.random.default_rng(999)
    best_a = a0.copy()
    best_score = score_of(a0)
    t0 = time.time()
    per_start = time_limit / (n_starts + 1)

    for s in range(n_starts):
        if time.time() - t0 > time_limit - 10:
            break
        # Create perturbation
        a_pert = a0.copy()
        # Shift mass
        shift_amt = 0.05 + 0.05 * s
        src = int(n * (0.3 + 0.1 * rng.random()))
        dst = int(n * (0.6 + 0.1 * rng.random()))
        if src != dst:
            a_pert[src] -= shift_amt * a_pert[src]
            a_pert[dst] += shift_amt * a_pert[src] / max(a_pert[dst], 1e-10) * 0.1
        a_pert = normalize(a_pert)

        remaining = time_limit - (time.time() - t0) - 5
        if remaining < 10:
            break
        t_sub = min(per_start, remaining)

        a_sub, sc_sub = lse_lbgfs_optimize(a_pert, t_sub,
                                           T_stages=[20.0, 10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.15])
        if sc_sub < best_score:
            best_score = sc_sub
            best_a = a_sub.copy()
            print(f"    Multi-start {s}: NEW BEST {sc_sub:.9f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = score_of(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.9f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = make_asymmetric_init(n_init, rng, kind=0)
        best_score = score_of(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.9f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = score_of(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.9f} (n={len(a)})")

    # === Phase 1: LSE L-BFGS (main workhorse) ===
    print(f"\n=== Phase 1: LSE L-BFGS (score={best_score:.9f}) ===")
    rem = deadline - time.time()
    t_alloc = min(rem * 0.35, 600.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        T_stages = [40.0, 25.0, 15.0, 10.0, 7.0, 5.0, 3.5, 2.5, 1.8, 1.2,
                    0.8, 0.5, 0.35, 0.25, 0.15, 0.1, 0.06, 0.04]
        a_lse, _ = lse_lbgfs_optimize(best_seq, t_alloc, T_stages=T_stages)
        try_update(a_lse, "LSE_p1")

    # === Phase 2: Active-set equalization ===
    print(f"\n=== Phase 2: Active-Set Equalization (score={best_score:.9f}) ===")
    rem = deadline - time.time()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_as, _ = active_set_equalize(best_seq, t_alloc, n_outer=500, damp=0.5)
        try_update(a_as, "ActiveSet")

    # === Phase 3: LSE refinement after active set ===
    print(f"\n=== Phase 3: LSE Refinement (score={best_score:.9f}) ===")
    rem = deadline - time.time()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        T_stages = [15.0, 8.0, 5.0, 3.0, 2.0, 1.2, 0.7, 0.4, 0.25, 0.15, 0.1]
        a_lse2, _ = lse_lbgfs_optimize(best_seq, t_alloc, T_stages=T_stages)
        try_update(a_lse2, "LSE_p3")

    # === Phase 4: Multi-start exploration ===
    print(f"\n=== Phase 4: Multi-Start (score={best_score:.9f}) ===")
    rem = deadline - time.time()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_ms, _ = multi_start_lse(best_seq, t_alloc, n_starts=4)
        try_update(a_ms, "MultiStart")

    # === Phase 5: Multi-n exploration ===
    print(f"\n=== Phase 5: Multi-n (score={best_score:.9f}) ===")
    rem = deadline - time.time()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 30:
        n_best = len(best_seq)
        candidate_ns = [max(400, n_best - 200), n_best + 200, max(300, n_best // 2)]
        candidate_ns = sorted(set([x for x in candidate_ns if x != n_best and x >= 200]))
        if candidate_ns:
            per_n = t_alloc / len(candidate_ns)
            print(f"  Trying n values: {candidate_ns}")
            for n_try in candidate_ns:
                if time.time() > deadline - 10:
                    break
                a_interp = interpolate_sequence(best_seq, n_try)
                t_sub = min(per_n, deadline - time.time() - 5)
                if t_sub < 15:
                    continue
                print(f"  n={n_try}, budget={t_sub:.0f}s")
                a_sub, _ = lse_lbgfs_optimize(a_interp, t_sub,
                                              T_stages=[30.0, 15.0, 8.0, 4.0, 2.0, 1.0, 0.5])
                try_update(a_sub, f"multi_n_{n_try}")

    # === Phase 6: Coordinate descent polish ===
    print(f"\n=== Phase 6: Coord Descent (score={best_score:.9f}) ===")
    rem = deadline - time.time()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 10:
        a_cd, _ = bidirectional_coord_descent(best_seq, t_alloc, n_steps=100000)
        try_update(a_cd, "CoordDesc")

    # === Phase 7: Final LSE polish ===
    print(f"\n=== Phase 7: Final Polish (score={best_score:.9f}) ===")
    rem = deadline - time.time()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 10:
        T_stages = [5.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.06, 0.04]
        a_fin, _ = lse_lbgfs_optimize(best_seq, t_alloc, T_stages=T_stages)
        try_update(a_fin, "FinalLSE")

    # Final verification
    if best_seq is not None:
        fs = score_of(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.9f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
