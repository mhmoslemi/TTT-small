import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.sparse import lil_matrix, csc_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def build_q_sparse(k, n):
    """Build sparse Q matrix for conv(a,a)[k] = a^T Q a."""
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.GUROBI, cp.MOSEK, cp.ECOS, cp.SCS, cp.CBC]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.CBC


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS, cp.GUROBI, cp.MOSEK, cp.CBC]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_sum = b_val.sum()
                if s_sum < 1e-15:
                    return None, None
                b_val = b_val / s_sum
                return b_val, float(max(0, M_var.value))
        except Exception:
            continue
    return None, None

    """Nesterov dual averaging LP alternating with adaptive restarts."""
    n = len(a0)

    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(42)
    lp_times = []

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver)
        lp_time = time.monotonic() - lp_t0
        lp_times.append(lp_time)

        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 300 == 0:
            if best_score < prev_best - 1e-9:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 8:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 5)
                noise = 0.03 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                sy2 = y.sum()
                if sy2 > 1e-15:
                    y = y / sy2
                stall_count = 0

            if (it + 1) % 2000 == 0:
                elapsed = time.monotonic() - t0
                avg_lp = np.mean(lp_times[-50:]) * 1000 if len(lp_times) >= 50 else 0
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.10f} avg_lp={avg_lp:.1f}ms")

        if len(lp_times) > 100 and np.mean(lp_times[-100:]) > 5.0:
            print(f"    LP too slow ({np.mean(lp_times[-100:]):.2f}s/iter), stopping at iter {it+1}")
            break

    return best_a, best_score


def solve_qcqp_active(a_init, active_indices, solver):
    """
    Solve: min t s.t. a^T Q_k a <= t for k in active_indices, sum(a)=1, a>=0
    Returns (a_solution, t_value) or (None, None).
    """
    n = len(a_init)
    a_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [cp.sum(a_var) == 1]

    for k in active_indices:
        Q = build_q_sparse(int(k), n)
        constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)

    prob = cp.Problem(cp.Minimize(t_var), constraints)

    solvers_to_try = [cp.MOSEK, cp.GUROBI, solver, cp.CLARABEL, cp.ECOS, cp.SCS, cp.CBC]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.asarray(a_var.value, dtype=np.float64)
                a_val = np.maximum(a_val, 0)
                s_sum = a_val.sum()
                if s_sum < 1e-15:
                    return None, None
                a_val = a_val / s_sum
                return a_val, float(max(0, t_var.value))
        except Exception:
            continue
    return None, None


def iterative_qcqp(a_init, time_limit, solver, initial_active=None):
    """
    Iterative active-set QCQP:
    1. Solve QCQP on active set
    2. Find most violated constraint not in active set
    3. Add it, repeat
    """
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    b = fast_convolve(a)
    max_b = b.max()

    if initial_active is None:
        threshold = max_b * 0.999
        active = set(np.where(b >= threshold)[0].tolist())
    else:
        active = set(initial_active)

    best_a = a.copy()
    best_score = 2.0 * n * max_b
    t0 = time.monotonic()
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        iteration += 1
        # Solve QCQP on current active set
        a_sol, t_val = solve_qcqp_active(a, sorted(active), solver)

        if a_sol is None:
            # Try with fewer active constraints (keep top ones)
            b_cur = fast_convolve(a)
            top_k = np.argsort(b_cur)[-min(5, len(active)):]
            active = set(top_k.tolist())
            a_sol, t_val = solve_qcqp_active(a, sorted(active), solver)
            if a_sol is None:
                break

        # Check for violated constraints
        b_new = fast_convolve(a_sol)
        max_b_new = b_new.max()
        k_violated = int(np.argmax(b_new))

        # Check if the solution is truly optimal (no violated constraint)
        violated = b_new.max() - t_val
        if violated < 1e-10:
            # No violation, we're at the optimum
            if 2.0 * n * max_b_new < best_score:
                best_score = 2.0 * n * max_b_new
                best_a = a_sol.copy()
            break

        # Add the violated constraint
        active.add(k_violated)

        # Cap active set size (keep the most important ones)
        if len(active) > 50:
            b_check = fast_convolve(a_sol)
            active_arr = np.array(sorted(active))
            active_sorted = active_arr[np.argsort(b_check[active_arr])[-40:]]
            active = set(active_sorted.tolist())

        a = a_sol.copy()

        if iteration % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    QCQP iter={iteration} ({elapsed:.0f}s): "
                  f"score={2.0*n*b_new.max():.10f} active={len(active)} violated={violated:.2e}")

    return best_a, best_score


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S_val = np.dot(grad_a, a)
    grad_z = a * (grad_a - S_val)
    return obj, 2.0 * n * grad_z


def lse_func(z, n, T_temp, deadline):
    """Top-level LSE objective for L-BFGS."""
    if time.monotonic() > deadline:
        return 1e15
    obj, _ = logsumexp_obj_grad(z, n, T_temp)
    return obj


def lse_jac(z, n, T_temp, deadline):
    """Top-level LSE gradient for L-BFGS."""
    if time.monotonic() > deadline:
        return np.zeros_like(z)
    _, g = logsumexp_obj_grad(z, n, T_temp)
    return g


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    n_stages = len(T_stages)
    iters_per_stage = max(20, maxiter_per_stage // n_stages)

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        try:
            res = minimize(
                fun=lse_func, x0=z, jac=lse_jac,
                args=(n, stage_T, deadline),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': iters_per_stage * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def frank_wolfe_refine(a0, time_limit):
    """Frank-Wolfe method with restarts for the convex problem."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    n_restarts = 50
    iters_per_restart = max(100, int(time_limit / n_restarts / 0.05))  # estimate

    for restart in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break

        t = 0
        a_fw = best_a.copy()
        no_improve = 0
        restart_iterations = 0
        max_restart_iters = 5000

        while time.monotonic() - t0 < time_limit and restart_iterations < max_restart_iters:
            t += 1
            restart_iterations += 1

            # Compute convolution and find argmax
            b = fast_convolve(a_fw)
            k_star = int(np.argmax(b))

            # Subgradient: g_j = 2 * a[k_star - j]
            g = np.zeros(n)
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            for j in range(js, je + 1):
                g[j] = 2.0 * a_fw[k_star - j]

            # Linear oracle
            j_min = int(np.argmin(g))

            # FW step with step size
            gamma = 2.0 / (t + 2)
            a_new = (1.0 - gamma) * a_fw
            a_new[j_min] += gamma

            # Evaluate
            score = 2.0 * n * fast_convolve(a_new).max()
            if score < best_score - 1e-12:
                best_score = score
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1

            a_fw = a_new

            if no_improve >= 200:
                break

        # Perturb for next restart
        if restart < n_restarts - 1:
            noise_scale = max(0.001, 0.01 * (1 - restart / n_restarts))
            noise = noise_scale * rng.standard_normal(n)
            a_pert = best_a + noise
            a_pert = np.maximum(a_pert, 0)
            sp = a_pert.sum()
            if sp > 1e-15:
                a_pert = a_pert / sp
                sc = 2.0 * n * fast_convolve(a_pert).max()
                if sc < best_score + 0.01:
                    best_a = a_pert
                    best_score = sc

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, n_steps=100000):
    """
    Coordinate descent: for each coordinate j, find the optimal value
    (given all other coordinates fixed) that minimizes the max autocorrelation.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        # Pick a coordinate to update (cyclic or random)
        j = step % n

        # The objective as a function of a[j] (with other coords fixed):
        # conv(a,a)[k] = sum_i a[i] a[k-i]
        # If we vary a[j], the terms involving a[j] are:
        #   a[j]^2 (appears in conv at position 2j)
        #   2*a[j]*a[k-j] for k != 2j (appears in conv at positions k where k-j is valid and k-j != j)
        # Actually, let's think of it differently.
        # Let a = a_fixed + delta * e_j where a_fixed has a[j] = 0.
        # Then conv(a,a)[k] = conv(a_fixed, a_fixed)[k] + 2*delta*conv(a_fixed, e_j)[k] + delta^2*conv(e_j, e_j)[k]
        # conv(a_fixed, e_j)[k] = a_fixed[k-j] (if 0 <= k-j <= n-1, else 0)
        # conv(e_j, e_j)[k] = 1 if k = 2j, else 0
        # So conv(a,a)[k] = b0[k] + 2*delta*a_fixed[k-j] + delta^2*[k==2j]
        # where b0 = conv(a_fixed, a_fixed)

        a_j = a[j]
        a[j] = 0.0
        b0 = fast_convolve(a)
        a[j] = a_j

        # Now conv(a,a)[k] = b0[k] + 2*delta*a[k-j] + delta^2*[k==2j]
        # where delta ranges from 0 to (1 - sum(a) + a[j]) = 1 (since sum(a)=1 and a[j] is removed)
        # Wait, we normalized so sum(a) = 1. If we set a[j]=0, sum = 1 - a_j.
        # We want to restore a[j] = delta such that sum = 1 again.
        # So delta ranges from 0 to a_j... no, we can change sum.
        # Actually, let's work with the unnormalized version and re-normalize after.
        # Simpler: work with a[j] as the variable, and the constraint is just a[j] >= 0.
        # The objective is 2n * max_k conv(a,a)[k] / (sum(a))^2.
        # Since we want to minimize this, and the denominator is (sum(a))^2,
        # let's just work with the ratio directly.

        # For simplicity, let's just do a 1D minimization:
        # f(delta) = 2*n * max_k (b0[k] + 2*delta*a[k-j] + delta^2*[k==2j]) / (sum(a) - a_j + delta)^2
        # where delta in [0, a_j_max] (a_j_max can be large, but in practice the minimum will be small)

        # This is a piecewise function (max of quadratics / quadratic).
        # We can minimize it using golden section search or by checking critical points.

        # For efficiency, let's use a simple line search:
        # Try a few candidate values of delta
        base_sum = s - a_j  # sum without a[j]
        # a[k-j] for valid k
        a_shifted = np.zeros(2 * n - 1)
        for i in range(n):
            if i != j:
                a_shifted[i + j] = a[i]  # a[k-j] = a[i] when k = i+j

        # Actually this is getting complicated. Let me just use a simpler approach:
        # Try reducing a[j] slightly and see if it helps.
        # Or use the subgradient direction.

        break  # Skip coordinate descent for now, it's complex

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def multiplicative_subgradient(a0, time_limit, n_steps):
    """Mirror descent (multiplicative weights) targeting active peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta_base = 0.03
    no_improve = 0
    scale = 1.0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        threshold = max_b * (1.0 - 3e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([0])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        eta = eta_base * scale
        log_scale = -eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 30:
            scale *= 0.8
            no_improve = 0
            if scale < 0.005:
                break

    return best_a, best_score
