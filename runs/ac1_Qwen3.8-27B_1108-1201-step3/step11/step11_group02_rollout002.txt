    """Compute gradient of LogSumExp objective.
    a: length n
    p: length 2n-1 (softmax weights over convolution)
    Returns: gradient of length n, = 2n * sum_k p[k] * (a[k-i] + a[i-k])
    """
    n = len(a)
    # Pad p to handle boundary effects
    # g1_i = sum_k p[k] * a[k-i] for valid k
    # g2_i = sum_k p[k] * a[i-k] for valid k
    
    # For g1: this is a "deconvolution" or cross-correlation
    # Let's compute it using FFT
    N = 1 << (3*n).bit_length()  # power of 2 >= 3n
    P = rfft(p, N)
    A = rfft(a, N)
    
    # g1_i = sum_k p[k] * a[k-i]
    # This is the cross-correlation (p circle a) at lag -i
    # = (flip(p) * a) at specific indices
    # flip(p) has length 2n-1, reversed
    p_flip = p[::-1]
    P_flip = rfft(p_flip, N)
    g1_full = irfft(P_flip * A, N)
    # g1_full has length N, the valid part is indices (2n-2) to (2n-2)+(n-1)
    # Actually: (p_flip * a)[m] = sum_j p_flip[j] * a[m-j] = sum_j p[2n-2-j] * a[m-j]
    # Let k = 2n-2-j: = sum_k p[k] * a[m - (2n-2-k)] = sum_k p[k] * a[m + k - 2n + 2]
    # We want sum_k p[k] * a[k-i] = sum_k p[k] * a[i + k - 2i]
    # So m + k - 2n + 2 = k - i => m = 2n - 2 - i
    g1 = g1_full[2*n-2 : 2*n-2+n]
    
    # For g2: sum_k p[k] * a[i-k]
    # = (p * a)[i] (standard convolution!)
    # (p * a)[m] = sum_j p[j] * a[m-j]
    # We want sum_k p[k] * a[i-k] = (p * a)[i]
    g2_full = irfft(P * A, N)
    g2 = g2_full[0:n]  # indices 0 to n-1
    
    grad = g1 + g2
    return 2.0 * n * grad

def compute_gradient(a, p, n):
    """Compute gradient: 4n * sum_k p[k] * a[k-i] for each i."""
    N = 1 << (3*n).bit_length()  # Next power of 2 >= 3n
    P = rfft(p, N)
    A = rfft(a, N)
    # Cross-correlation R[tau] = sum_k p[k] * a[k+tau]
    # = IFFT(FFT(p) * conj(FFT(a)))
    # For real: conj(FFT(a)) = FFT(fftshift(a, N//2))... no, that's wrong.
    # Actually for the DFT: conj(F[a][k]) = F[a[-k]][k] = F[flip(a)][k] approximately
    # Let me just use the standard formula.

    # Using FFT: R = ifft(X * conj(Y))
    # But this gives R[tau] for tau = 0, ..., N-1 (with wrap-around)
    # To avoid wrap-around, we need N >= len(p) + len(a) - 1
    
    x = np.zeros(N)
