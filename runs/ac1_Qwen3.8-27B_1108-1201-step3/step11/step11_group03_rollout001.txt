```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using active-set QCQP.
The problem is convex (min of max of PSD quadratics over simplex),
so the active-set method converges to the global optimum.
Tries n=1199,1200,1201 and picks the best.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.sparse import lil_matrix, csc_matrix

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_q_sparse(k, n):
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)

def get_solver():
    for s in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except:
            continue
    return cp.ECOS

def score_of(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * len(a) * b.max() / (s * s))

def solve_active_qcqp(a0, solver, max_iters=100, tol=1e-10):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    
    b = fast_convolve(a)
    t = b.max()
    
    # Initial active set: top constraints
    threshold = t * 0.99
    active = set(np.where(b >= threshold)[0].tolist())
    if not active:
        active = {int(np.argmax(b))}
    
    for it in range(max_iters):
        # Build problem
        a_var = cp.Variable(n, nonneg=True)
        t_var = cp.Variable()
        constraints = [cp.sum(a_var) == 1]
        
        for k in sorted(active):
            Q = build_q_sparse(k, n)
            constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        
        try:
            prob.solve(solver=solver)
        except Exception as e:
            try:
                prob.solve(solver=cp.ECOS)
            except:
                break
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using active-set QCQP with MOSEK.
The problem is convex (max of PSD quadratics over simplex), so active-set
converges to the global optimum. Tries multiple n values (1198-1202) and
uses LP alternating as warm start when QCQP is slow.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.sparse import lil_matrix, csc_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_q_sparse(k, n):
    """Build sparse Q matrix such that a^T Q a = conv(a,a)[k]."""
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)


def get_solver():
    """Find the best available QCQP solver."""
    for s in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(10, nonneg=True)
            t = cp.Variable()
            Q = cp.psd_wrap(np.eye(10) * 0.01 + np.eye(10) * 0.99)
            prob = cp.Problem(cp.Minimize(t), [cp.sum(x) == 1,
                                               cp.quad_form(x, Q, assume_PSD=True) <= t])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS


def score_of(a):
    """Compute the objective score."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * len(a) * b.max() / (s * s))


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build T such that T @ b = convolve(a_fixed, b)
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    T = csc_matrix(T)
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    for s in [solver, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0)
                b_val = b_val / b_val.sum()
                return b_val
        except Exception:
            continue
    return None
```
