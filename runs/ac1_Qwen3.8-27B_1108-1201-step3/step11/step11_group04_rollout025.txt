def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def compute_subgradient_from_dual(a, b, dual_p, n_peaks=50):
    """Compute subgradient using LP dual weights: g[j] = sum_k p[k] * b[k-j] (valid indices)."""
    n = len(a)
    if dual_p is None:
        # Fallback: use current b as the "gradient direction"
        g = np.zeros(n)
        max_b = fast_convolve(b).max()
        active = np.where(fast_convolve(b) >= max_b * 0.999)[0]
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += b[k - idx]
        g /= max(1, len(active))
        return g
    
    # Use dual weights
    n_conv = 2 * n - 1
    g = np.zeros(n)
    # Only use top-weighted dual entries for speed
    top_k = min(n_peaks, n_conv)
    top_idx = np.argsort(dual_p)[-top_k:]
    for k in top_idx:
        w = dual_p[k]
        if w < 1e-12:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += w * b[k - idx]
    return g


def spectral_flatten(a, n_iters=100, blend=0.4, deadline=None):
    """Iterative spectral flattening (Gerchberg-Saxton style)."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)
        mags = np.abs(F)
        non_dc = mags[1:]
        if non_dc.min() < 1e-15:
            target = np.mean(non_dc)
        else:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        phases = np.angle(F)
        F_new = F.copy()
        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend) * mags[k] + blend * target
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])
        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
    return best_a


def lbgfs_refine(a0, time_limit, T_stages, deadline=None):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    if deadline is None:
        deadline = t0 + time_limit

    n_stages = len(T_stages)
    for stage_i, stage_T in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 3:
            break
        stages_left = n_stages - stage_i
        iters = max(20, min(100, int(remaining * 30 / stages_left)))

        def func(z_in):
            if time.monotonic() > deadline:
                return 1e15
            a = _softmax_from_z(z_in)
            b = fast_convolve(a)
            max_b = b.max()
            exp_terms = np.exp((b - max_b) / stage_T)
            lse = max_b + stage_T * np.log(exp_terms.sum())
            return 2.0 * n * lse

        def jac(z_in):
            if time.monotonic() > deadline:
                return np.zeros_like(z_in)
            a = _softmax_from_z(z_in)
            b = fast_convolve(a)
            max_b = b.max()
            exp_terms = np.exp((b - max_b) / stage_T)
            w = exp_terms / exp_terms.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)
            return 2.0 * n * grad_z

        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': iters, 'ftol': 1e-25,
                                    'gtol': 1e-18, 'maxfun': iters * 200, 'disp': False})
            z = res.x
        except Exception:
            pass

    return _softmax_from_z(z)


def _softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = 0.03
    scale = 1.0
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * (1.0 - 5e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        eta_eff = eta * scale
        log_scale = -eta_eff * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= 40:
            scale *= 0.7
            no_improve = 0
            if scale < 0.003:
                break
    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, solver, n_max_iters=1000000):
    """Nesterov dual averaging LP alternating with adaptive strategies."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(777)

    # Adaptive alpha: start aggressive, decay
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b, dual_p = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        # Nesterov dual averaging
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Stall detection
        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-12:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 6:
                # Perturb in frequency domain
                a_padded = np.zeros(1 << (2 * n - 1).bit_length())
                a_padded[:n] = a
                F = np.fft.fft(a_padded)
                F_mag = np.abs(F)
                F_phase = np.angle(F)
                noise = rng.normal(0, 0.02, F_mag.shape)
                F_mag = np.maximum(F_mag + noise, 0)
                F_noisy = F_mag * np.exp(1j * F_phase)
                a_new = np.fft.ifft(F_noisy).real[:n]
                a_new = np.maximum(a_new, 0)
                sn = a_new.sum()
                if sn > 1e-15:
                    a = a_new / sn
                    y = a.copy()
                stall_count = 0

            if (it + 1) % 2000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def dual_guided_update(a0, time_limit, solver, n_steps=50000):
    """Mirror descent using LP dual variables for gradient computation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = 0.05
    scale = 1.0
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, M_b, dual_p = solve_lp_best_response(a, solver)
        if b_new is None:
            continue
        g = compute_subgradient_from_dual(a, b_new, dual_p)
        log_update = -eta * scale * g
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            continue
        a_new = a_new / sn
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= 30:
            scale *= 0.7
            no_improve = 0
            if scale < 0.003:
                break
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate sequence score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def freq_perturb(a, rng, strength=0.03):
    """Frequency-domain perturbation that preserves non-negativity better."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a_padded = np.zeros(L)
    a_padded[:n] = a
    F = np.fft.fft(a_padded)
    F_mag = np.abs(F)
    F_phase = np.angle(F)
    # Perturb magnitudes
    noise = rng.normal(0, strength, F_mag.shape)
    F_mag_new = np.maximum(F_mag + noise, 0)
    F_new = F_mag_new * np.exp(1j * F_phase)
    a_new = np.fft.ifft(F_new).real[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Primary engine: LP alternating with Nesterov acceleration.
    Secondary: dual-guided mirror descent, spectral flattening, L-BFGS.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')
    solver = get_best_solver()
    print(f"Using solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        x = np.linspace(0, 1, n_init)
        best_seq = np.cos(np.pi * x / 2) ** 2
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  ** NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    print(f"Total budget: {(deadline - time.monotonic()):.0f}s, starting score: {best_score:.8f}")

    # === Phase 0: Quick spectral flatten ===
    rem = deadline - time.monotonic()
    if rem > 30:
        print("\n=== Phase 0: Spectral Flatten ===")
        t0 = time.monotonic()
        a_sf = spectral_flatten(best_seq, n_iters=50, blend=0.4, deadline=time.monotonic() + 15)
        local_update(a_sf, "spec_flatten")
        print(f"  Done in {time.monotonic()-t0:.1f}s, score={best_score:.8f}")

    # === Phase 1: Main LP Nesterov (largest budget) ===
    rem = deadline - time.monotonic()
    if rem > 60:
        print(f"\n=== Phase 1: Nesterov LP (score={best_score:.8f}) ===")
        t_lp = min(rem * 0.45, 500.0)
        print(f"  Budget: {t_lp:.0f}s")
        a_lp, _ = nesterov_lp_optimize(best_seq, t_lp, solver, n_max_iters=1000000)
        local_update(a_lp, "nesterov_lp_main")

    # === Phase 2: Dual-guided mirror descent ===
    rem = deadline - time.monotonic()
    if rem > 60:
        print(f"\n=== Phase 2: Dual-Guided Mirror Descent (score={best_score:.8f}) ===")
        t_dgm = min(rem * 0.20, 200.0)
        print(f"  Budget: {t_dgm:.0f}s")
        a_dg, _ = dual_guided_update(best_seq, t_dgm, solver, n_steps=50000)
        local_update(a_dg, "dual_mirror")

    # === Phase 3: L-BFGS LogSumExp fine annealing ===
    rem = deadline - time.monotonic()
    if rem > 60:
        print(f"\n=== Phase 3: L-BFGS LogSumExp (score={best_score:.8f}) ===")
        t_lb = min(rem * 0.25, 300.0)
        T_stages = [5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05,
                    0.03, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001]
        print(f"  T_stages: {len(T_stages)}, budget={t_lb:.0f}s")
        a_lb = lbgfs_refine(best_seq, t_lb, T_stages, deadline=time.monotonic() + t_lb)
        local_update(a_lb, "lbgfs_fine")

    # === Phase 4: Peak shaving ===
    rem = deadline - time.monotonic()
    if rem > 30:
        print(f"\n=== Phase 4: Peak Shave (score={best_score:.8f}) ===")
        t_pk = min(rem * 0.08, 60.0)
        a_pk, _ = peak_shave(best_seq, t_pk, n_steps=100000)
        local_update(a_pk, "peak_shave")

    # === Phase 5: Second Nesterov LP round (perhaps with perturbation) ===
    rem = deadline - time.monotonic()
    if rem > 60:
        print(f"\n=== Phase 5: Nesterov LP Round 2 (score={best_score:.8f}) ===")
        t_lp2 = min(rem * 0.20, 150.0)
        # Slight perturbation to escape potential local minimum
        a_pert = freq_perturb(best_seq, rng, strength=0.01)
        a_lp2, _ = nesterov_lp_optimize(a_pert, t_lp2, solver, n_max_iters=500000)
        local_update(a_lp2, "nesterov_lp_2")

    # === Phase 6: Try different n values ===
    rem = deadline - time.monotonic()
    if rem > 80:
        print(f"\n=== Phase 6: Multi-n exploration (score={best_score:.8f}) ===")
        for n_try in [1500, 1800]:
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            t_n = min(rem * 0.25, 100.0)
            if t_n < 30:
                break
            print(f"  Trying n={n_try}, budget={t_n:.0f}s")
            init_n = interpolate_sequence(best_seq, n_try)
            a_sf_n = spectral_flatten(init_n, n_iters=20, blend=0.3)
            a_n, _ = nesterov_lp_optimize(a_sf_n, t_n * 0.7, solver, n_max_iters=200000)
            local_update(a_n, f"lp_n{n_try}")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.4, 60.0)
        T_fin = [2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001]
        a_fin = lbgfs_refine(best_seq, t_fin, T_fin, deadline=time.monotonic() + t_fin)
        local_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_ps = min(rem * 0.4, 30.0)
        a_ps, _ = peak_shave(best_seq, t_ps, n_steps=30000)
        local_update(a_ps, "final_peak")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_flp = min(rem * 0.5, 25.0)
        a_flp, _ = nesterov_lp_optimize(best_seq, t_flp, solver, n_max_iters=30000)
        local_update(a_flp, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_start
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}) at {elapsed:.1f}s")
        test_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        try:
            eval_score = evaluate_sequence(test_seq)
            print(f"Verified with evaluate_sequence: {eval_score:.8f}")
        except Exception as e:
            print(f"Verification error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
