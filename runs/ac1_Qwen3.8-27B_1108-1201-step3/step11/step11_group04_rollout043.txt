import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), n=fft_size)

def evaluate_local(a):

def softmax_from_z(z):

    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):

def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver, warm_start=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    solved = False
    b_val = None
    M_val = None

    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solved = True
                break
        except Exception:
            continue

    if not solved:
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s_sum = b_val.sum()
    if s_sum < 1e-15:
        return None, None
    b_val = b_val / s_sum
    M_val = float(max(0, M_var.value))
    return b_val, M_val


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=1000000):
    """Nesterov dual averaging LP alternating with adaptive restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)

        if b_new is None:

        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

                y = y + noise

                sy2 = y.sum()
                if sy2 > 1e-15:

                stall_count = 0

            if (it + 1) % 2000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

def logsumexp_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage, deadline=None):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    if deadline is None:
        deadline = t0 + time_limit
    n_stages = len(T_stages)

    for stage_i, stage_T in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 3:
            break

        stages_left = n_stages - stage_i
        iters = max(20, min(max(50, maxiter_per_stage // n_stages) * 3, int(remaining * 80 / stages_left)))

        def func(z_in):
            if time.monotonic() > deadline:
                return 1e15
            obj, _ = logsumexp_obj_grad(z_in, n, stage_T)
            return obj

        def jac(z_in):
            if time.monotonic() > deadline:
                return np.zeros_like(z_in)
            _, g = logsumexp_obj_grad(z_in, n, stage_T)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-25,
                         'gtol': 1e-18, 'maxfun': iters * 300, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def peak_shave(a, time_limit, n_steps=200000, eta_base=0.03):
    """Directly target the maximum of the autocorrelation using multiplicative updates."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return None, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    scale = 1.0
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        threshold = max_b * (1.0 - 2e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        eta = eta_base * scale
        log_scale = -eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 40:
            scale *= 0.75
            no_improve = 0
            if scale < 0.003:
                break

    return best_a, best_score


def spectral_flatten(a, n_iters=50, blend=0.4, deadline=None):
    """Iterative spectral flattening (Gerchberg-Saxton style)."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)

        mags = np.abs(F)
        non_dc = mags[1:L // 2 + 1]
        if non_dc.min() < 1e-15:
            target = np.mean(non_dc)
        else:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        phases = np.angle(F)
        F_new = F.copy()
        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend) * mags[k] + blend * target
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = (1 - blend) * mags[L - k] + blend * target
                F_new[L - k] *= np.exp(1j * phases[L - k])

        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new

    return a


def fourier_optimize(n, time_limit, n_modes=15, n_restarts=5, deadline=None):
    """
    Optimize using Fourier parameterization:
    a_i = exp(c0 + sum_k (ck*cos + dk*sin)) / sum(...)
    This reduces dimensionality and allows fast L-BFGS convergence.
    """
    t0 = time.monotonic()
    if deadline is None:
        deadline = t0 + time_limit
    best_a = None
    best_score = float('inf')
    rng = np.random.default_rng(12345)

    x = np.linspace(0, 2 * np.pi, n)
    n_params = 1 + 2 * n_modes

    for restart in range(n_restarts):
        if time.monotonic() > deadline:
            break

        # Random initialization in Fourier space
        if restart == 0:
            params = np.zeros(n_params)
        elif restart == 1:
            # Cosine-like init
            params[1] = 0.5  # first cosine mode
        elif restart == 2:
            # Slight parabola
            params[1] = -0.3
        else:
            params = rng.normal(0, 0.3, n_params)

        T_temp = 1.0

        def fourier_a(params_vec):
            c0 = params_vec[0]
            cos_coeff = params_vec[1:n_modes + 1]
            sin_coeff = params_vec[n_modes + 1:]
            f = np.full(n, c0)
            for k in range(1, n_modes + 1):
                f += cos_coeff[k - 1] * np.cos(k * x)
                f += sin_coeff[k - 1] * np.sin(k * x)
            f = f - f.max()
            exp_f = np.exp(f)
            s = exp_f.sum()
            if s < 1e-300:
                return np.ones(n) / n
            return exp_f / s

        def fourier_obj(params_vec):
            a = fourier_a(params_vec)
            b = fast_convolve(a)
            max_b = b.max()
            exp_terms = np.exp((b - max_b) / T_temp)
            lse = max_b + T_temp * np.log(exp_terms.sum())
            return 2.0 * n * lse

        def fourier_grad(params_vec):
            a = fourier_a(params_vec)
            b = fast_convolve(a)
            max_b = b.max()
            exp_terms = np.exp((b - max_b) / T_temp)
            w = exp_terms / exp_terms.sum()

            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]

            # grad_a is the gradient of the w.r.t. a (before normalization)
            # Chain rule: a = exp(f) / sum(exp(f))
            # da_i/df_j = a_i * (delta_ij - a_j)
            # So grad_f = a * (grad_a / a - sum(grad_a * a) / sum(a))
            # Since sum(a) = 1: grad_f = grad_a - a * sum(grad_a * a)
            # But we need gradient w.r.t. params, not f.
            # f = c0 + sum_k (ck*cos_k + dk*sin_k)
            # So df/dparams = [1, cos_1, ..., cos_m, sin_1, ..., sin_m]

            S = np.dot(grad_a, a)
            grad_f = grad_a - a * S

            grad_params = np.zeros(n_params)
            grad_params[0] = np.dot(grad_f, a)  # Wait, this isn't right
            # Actually: obj depends on a, a depends on f, f depends on params
            # d_obj/d_params = sum_i (d_obj/da_i) * (da_i/df_i) * (df_i/d_params)
            # d_obj/da_i is complicated (involves the max), but we can use the chain through f
            # Simpler: use the identity that for softmax, d_obj/d_params = a^T (J^-1)^T grad_f where J is Jacobian
            # Actually the cleanest way:
            # grad_params[0] = sum_i grad_f[i] * (df_i/dc0) = sum_i grad_f[i] * 1 = sum(grad_f)
            # grad_params[1+k] = sum_i grad_f[i] * cos((k+1)*x_i)
            # grad_params[1+m+k] = sum_i grad_f[i] * sin((k+1)*x_i)

            grad_params[0] = np.sum(grad_f)
            for k in range(1, n_modes + 1):
                grad_params[k] = np.dot(grad_f, np.cos(k * x))
                grad_params[n_modes + k] = np.dot(grad_f, np.sin(k * x))

            # Scale by 2n (from the objective)
            return 2.0 * n * grad_params

        # Anneal temperature
        T_schedule = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02]
        time_per_T = max(2.0, (time_limit / max(1, n_restarts)) / len(T_schedule))

        for T in T_schedule:
            if time.monotonic() > deadline:
                break
            T_temp = T

            def func_pv(pv):
                if time.monotonic() > deadline:
                    return 1e15
                return fourier_obj(pv)

            def jac_pv(pv):
                if time.monotonic() > deadline:
                    return np.zeros_like(pv)
                return fourier_grad(pv)

            try:
                res = minimize(func_pv, params, jac=jac_pv, method='L-BFGS-B',
                               options={'maxiter': 200, 'ftol': 1e-20, 'gtol': 1e-14, 'maxfun': 10000})
                params = res.x
            except Exception:
                pass

            a_candidate = fourier_a(params)
            sc = evaluate_local(a_candidate)
            if sc < best_score:
                best_score = sc
                best_a = a_candidate.copy()
                print(f"    FourierOpt n={n} restart={restart} T={T}: score={sc:.8f}")

    return best_a, best_score


def perturbed_sequence(a, rng, strength=0.03):
    """Create a perturbed version of sequence a."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 10)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def structured_init(n, rng, style=None):
    """Create a structured initial sequence."""
    x = np.linspace(0, 1, n)
    if style is None:
        style = rng.integers(0, 8)

    if style == 0:
        a = np.sin(np.pi * x) ** 2
    elif style == 1:
        a = np.exp(-4 * x * (1 - x))
    elif style == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif style == 3:
        # Raised cosine
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif style == 4:
        # Parabolic (inverted - more mass at edges)
        a = x * (1 - x) * 4 + 0.1
    elif style == 5:
        # Triangular
        a = 1.0 - np.abs(2 * x - 1)
    elif style == 6:
        # Sinc-like
        center = n // 2
        t = (np.arange(n) - center) / (n / 4.0)
        a = np.sinc(t / np.pi)
        a = np.maximum(a, 0.01)
    else:
        # Gaussian
        a = np.exp(-8 * (x - 0.5) ** 2)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    Combines Fourier-parameterized optimization, Nesterov LP alternating,
    ultra-fine L-BFGS, and adaptive peak shaving across multiple n values.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    rem = deadline - time.monotonic()
    print(f"Total budget: {rem:.0f}s, starting score: {best_score:.8f}")

    # === Phase 1: Multi-n initialization with diverse methods (250s) ===
    print(f"\n=== Phase 1: Multi-n Initialization ===")
    n_candidates = [1000, 1200, 1400, 1600]
    if parent is not None:
        n_candidates.insert(0, len(parent))
    # Remove duplicates
    n_candidates = list(dict.fromkeys(n_candidates))

    t_per_n = max(20.0, min(60.0, rem * 0.25 / len(n_candidates)))

    for n_cand in n_candidates:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        print(f"  Trying n={n_cand}, budget={t_per_n:.0f}s")

        # If we have a parent of this length, use it
        if parent is not None and len(parent) == n_cand:
            init_a = parent.copy()
        else:
            # Try multiple structured inits
            best_init = None
            best_init_sc = float('inf')
            for style in range(8):
                init_try = structured_init(n_cand, rng, style=style)
                sc_tmp = evaluate_local(init_try)
                if sc_tmp < best_init_sc:
                    best_init_sc = sc_tmp
                    best_init = init_try
            init_a = best_init

        # Quick peak shaving
        t_quick = min(t_per_n * 0.4, 20.0)
        a_ps, _ = peak_shave(init_a, t_quick, n_steps=50000)
        local_update(a_ps, f"init_ps_n{n_cand}")

        # Fourier optimization
        t_fourier = min(t_per_n * 0.5, 30.0)
        if t_fourier > 5 and rem - (deadline - time.monotonic()) < t_per_n - 5:
            a_four, _ = fourier_optimize(n_cand, t_fourier, n_modes=12, n_restarts=3,
                                         deadline=time.monotonic() + t_fourier)
            local_update(a_four, f"fourier_n{n_cand}")

        # Spectral flatten
        rem2 = deadline - time.monotonic()
        if rem2 > 10 and best_seq is not None and len(best_seq) == n_cand:
            a_spec = spectral_flatten(best_seq, n_iters=30, blend=0.3,
                                       deadline=time.monotonic() + min(10, rem2))
            local_update(a_spec, f"spec_n{n_cand}")

    # === Phase 2: Nesterov LP on best sequence (700s) ===
    print(f"\n=== Phase 2: Nesterov LP (score={best_score:.8f}, n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    t_lp = min(rem * 0.40, 700.0)
    if t_lp > 60:
        print(f"  Budget: {t_lp:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_lp, solver, n_max_iters=2000000)
        local_update(a_lp, "nesterov_lp_1")

    # === Phase 3: L-BFGS LogSumExp fine (400s) ===
    print(f"\n=== Phase 3: L-BFGS LogSumExp (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_lbfgs = min(rem * 0.25, 400.0)
    if t_lbfgs > 30:
        T_stages = [5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1,
                    0.05, 0.03, 0.02, 0.015, 0.01, 0.007, 0.005,
                    0.003, 0.002, 0.001]
        print(f"  {len(T_stages)} stages, budget={t_lbfgs:.0f}s")
        a_lb = lbgfs_refine(best_seq, t_lbfgs, T_stages, maxiter_per_stage=30000,
                            deadline=time.monotonic() + t_lbfgs)
        local_update(a_lb, "lbgfs_fine")

    # === Phase 4: Peak shaving (150s) ===
    print(f"\n=== Phase 4: Peak Shaving (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_peak = min(rem * 0.10, 150.0)
    if t_peak > 15:
        a_pk, _ = peak_shave(best_seq, t_peak, n_steps=500000, eta_base=0.02)
        local_update(a_pk, "peak_shave")

    # === Phase 5: Second LP round + perturbation escape (200s) ===
    print(f"\n=== Phase 5: LP Round 2 + Escape (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        # Perturbation escape
        for trial in range(2):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 40:
                break
            a_pert = perturbed_sequence(best_seq, rng, strength=0.01 + 0.01 * trial)
            a_sp = spectral_flatten(a_pert, n_iters=15, blend=0.3)
            t_sub = min(rem_sub * 0.3, 50.0)
            if t_sub > 10:
                a_opt, _ = nesterov_lp_refine(a_sp, t_sub, solver, n_max_iters=100000)
                local_update(a_opt, f"escape_{trial}")

    # === Phase 6: Try nearby n values (100s) ===
    print(f"\n=== Phase 6: Try nearby n (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 50 and best_seq is not None:
        n_best = len(best_seq)
        for n_try in [n_best - 200, n_best + 200]:
            rem = deadline - time.monotonic()
            if rem < 30 or n_try < 200:
                continue
            t_try = min(rem * 0.3, 40.0)
            if t_try < 10:
                continue
            print(f"  Trying n={n_try}")
            init_try = interpolate_sequence(best_seq, n_try)
            a_sp = spectral_flatten(init_try, n_iters=20, blend=0.3)
            a_opt, _ = nesterov_lp_refine(a_sp, t_try, solver, n_max_iters=50000)
            local_update(a_opt, f"n{n_try}")

    # === Phase 7: Final polish (remaining time) ===
    print(f"\n=== Phase 7: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        # L-BFGS final with lowest temps
        t_fin = min(rem * 0.35, 60.0)
        if t_fin > 10:
            T_fin = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.003, 0.002]
            a_fin = lbgfs_refine(best_seq, t_fin, T_fin, maxiter_per_stage=10000,
                                  deadline=time.monotonic() + t_fin)
            local_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 15:
        # Peak shave final
        t_ps = min(rem * 0.35, 30.0)
        if t_ps > 5:
            a_ps, _ = peak_shave(best_seq, t_ps, n_steps=100000, eta_base=0.01)
            local_update(a_ps, "final_peak")

    rem = deadline - time.monotonic()
    if rem > 10:
        # Final quick LP
        t_flp = min(rem * 0.5, 25.0)
        if t_flp > 5:
            a_flp, _ = nesterov_lp_refine(best_seq, t_flp, solver, n_max_iters=20000)
            local_update(a_flp, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_start
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}) at {elapsed:.1f}s")
        test_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        try:
            eval_score = evaluate_sequence(test_seq)
            print(f"Verified with evaluate_sequence: {eval_score:.8f}")
        except Exception as e:
            print(f"Verification error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
