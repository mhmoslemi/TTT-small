    A = rfft(a, fft_size)

    # grad[i] = 2 * sum_k w[k] * a[k-i] (with valid indices)
    # This is a cross-correlation: (w * a_flip)[n-1+i]
    conv_len = n + (2 * n - 1) - 1  # = 3n-2
    fft_size = 1 << (conv_len - 1).bit_length()
    if conv_len > fft_size:
        fft_size *= 2

    A_flip = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_flip * W, fft_size)

    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad, w


def mirror_descent_update(a, grad, eta):
    """Mirror descent step with negative entropy regularizer on the simplex."""
    log_a = np.log(np.maximum(a, 1e-300))
    unnormalized = log_a - eta * grad
    unnormalized -= unnormalized.max()
    exp_vals = np.exp(unnormalized)
    s = exp_vals.sum()
    if s < 1e-300:
        return np.ones(len(a)) / len(a)
    return exp_vals / s


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    sorted_v = np.sort(v)[::-1]
    cssv = np.cumsum(sorted_v)
    rho = np.nonzero(sorted_v * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 1.5
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(3 * x) + 0.15 * np.cos(7 * x) + 0.05 * np.cos(13 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + 10.0 * x * (1.0 - x))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def mirror_descent_optimize(a0, time_limit, T_start=50.0, T_end=0.5, n_stages=15,
                            C_base=1.5, seed=42):
    """
    Mirror descent with soft-max gradient and temperature annealing.
    Uses running average for the final output.
    Returns (best_a, best_score).
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    # Running average
    avg_a = np.zeros(n)
    n_avg = 0

    temps = np.geomspace(T_start, T_end, n_stages)

    for stage, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        # Scale C inversely with T for stability
        C = C_base / max(1.0, T / 10.0)

        # Run mirror descent for this temperature
        max_iters = max(50, min(20000, int(remaining * 50)))
        stag_start = time.monotonic()

        for t in range(1, max_iters + 1):
            if time.monotonic() - stag_start > remaining:
                break

            grad, w = compute_soft_max_grad(a, T)
            eta = C / np.sqrt(t + 1.0)
            a = mirror_descent_update(a, grad, eta)

            # Track running average
            avg_a += a
            n_avg += 1

            # Check score periodically
            if t % 50 == 0 or t == 1:
                score_a = 2.0 * n * fast_autocorr(a).max()
                if score_a < best_score:
                    best_score = score_a
                    best_a = a.copy()

                if n_avg >= 20 and n_avg % 100 == 0:
                    avg_check = avg_a / n_avg
                    score_avg = 2.0 * n * fast_autocorr(avg_check).max()
                    if score_avg < best_score:
                        best_score = score_avg
                        best_a = avg_check.copy()

        if stage % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MD stage {stage}/{n_stages} (T={T:.2f}, {elapsed:.0f}s): best={best_score:.6f}")

    # Final check on running average
    if n_avg > 0:
        avg_check = avg_a / n_avg
        score_avg = 2.0 * n * fast_autocorr(avg_check).max()
        if score_avg < best_score:
            best_score = score_avg
            best_a = avg_check.copy()

    return best_a, best_score


def get_lp_solver():
    """Find a working LP solver."""
    try:
        import cvxpy as cp
        for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            try:
                x = cp.Variable(5, nonneg=True)
                prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
                prob.solve(solver=solver_name)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    return solver_name
            except Exception:
                continue
        return cp.ECOS
    except Exception:
        return None


def build_conv_matrix(a_fixed):
    """Build the (2n-1) x n convolution matrix T where (T@b)[k] = sum_j a_fixed[k-j]*b[j]."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = j
        hi = min(2 * n - 1, j + n)
        T[lo:hi, j] = a_fixed[:hi - lo]
    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. (a_fixed * b)[k] <= M for all k, sum(b)=1, b>=0."""
    try:
        import cvxpy as cp
    except Exception:
        return None, None

    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, max_iters=50000):
    """
    Nesterov-accelerated LP best-response iteration for the problem:
      min_a max_k (a*a)_k  s.t. sum(a)=1, a>=0
    Uses running average to track best solution.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    y = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / s_y

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_a

        y = y_new
        a = a_new

        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        avg_a += a
        n_avg += 1
        if n_avg >= 20 and n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()

        if t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}, LP_M={M_val:.8f}")

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=123):
    """
    Coordinate-descent peak shaving: reduce top autocorrelation peaks
    by moving mass from high-contributing positions to low-contributing ones.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:min(12, 2 * n - 1)]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]

            src_local = int(np.argmax(contrib))
            src = int(idx[src_local])

            # Find target positions not in top peak areas
            peak_areas = set()
            for k2 in top_k[:6]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_areas.update(range(js2, je2 + 1))

            candidates = [j for j in range(n) if j not in peak_areas]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            if not candidates:
                continue
            candidates.sort(key=lambda j: a[j])
            dst_pool = candidates[:max(1, len(candidates) // 4)]
            dst = int(dst_pool[rng.integers(len(dst_pool))])

            for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > 300:
                break

    return best_a, best_score


def subgradient_polish(a0, time_limit, n_peaks=5, seed=99):
    """
    Fast subgradient polishing: use top-k peaks to form a subgradient,
    then take mirror descent steps with decaying step size.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a, 2.0 * n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    # Use soft-max with very low temperature (almost hard max)
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        # Top-k peaks with exponential weighting
        sorted_idx = np.argsort(b)[::-1]
        top = sorted_idx[:n_peaks]
        weights = np.zeros(len(b))
        vals = b[top] - b[top[0]]
        exp_vals = np.exp(vals / 5.0)  # very sharp
        weights[top] = exp_vals / exp_vals.sum()

        # Compute gradient: grad[i] = 2 * sum_k weights[k] * a[k-i]
        conv_len = n + len(b) - 1
        fft_size = 1 << (conv_len - 1).bit_length()
        A_flip = rfft(a[::-1].copy(), fft_size)
        W = rfft(weights, fft_size)
        conv = irfft(A_flip * W, fft_size)
        grad = 2.0 * conv[n - 1:n - 1 + n]

        eta = 0.3 / np.sqrt(t + 1.0)
        a = mirror_descent_update(a, grad, eta)

        if t % 100 == 0:
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

    score_final = 2.0 * n * fast_autocorr(a).max()
    if score_final < best_score:
        best_score = score_final
        best_a = a.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization:
    1. Mirror descent (fast) at parent length
    2. LP Nesterov (accurate) at parent length
    3. Multi-length exploration (longer sequences)
    4. Subgradient polish and peak shaving
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    # Get parent sequence
    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    n_base = len(best_seq) if best_seq is not None else 1200

    if parent is None:
        n_base = 1200
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        n_a = len(a)
        sc = 2.0 * n_a * fast_autocorr(a).max()
        if sc < best_score:
            # Verify with actual evaluator
            real_sc = evaluate_safe(a)
            if np.isfinite(real_sc) and real_sc < best_score:
                best_score = real_sc
                best_seq = a.copy()
                elapsed = time.monotonic() - t_global_start
                print(f"  *** NEW BEST [{label}]: {real_sc:.6f} (n={n_a}, t={elapsed:.0f}s)")

    # === Phase 1: Mirror descent at base length (fast) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: Mirror descent at n={n_base} (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.20, 300.0)
    if t1 > 30:
        a_out, _ = mirror_descent_optimize(best_seq, t1, T_start=50.0, T_end=0.5,
                                           n_stages=12, C_base=1.5, seed=42)
        try_update(a_out, f"md_n{n_base}")

    # === Phase 2: Subgradient polish at base length ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: Subgradient polish at n={n_base} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.10, 120.0)
    if t2 > 10:
        a_out, _ = subgradient_polish(best_seq, t2, n_peaks=5, seed=99)
        try_update(a_out, f"sgrad_n{n_base}")

    # === Phase 3: LP Nesterov at base length (accurate) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: LP Nesterov at n={n_base} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.25, 400.0)
    if t3 > 30 and solver is not None:
        a_out, _ = nesterov_lp_optimize(best_seq, t3, solver)
        try_update(a_out, f"lp_n{n_base}")

    # === Phase 4: Multi-length exploration ===
    # Try longer lengths for better theoretical bound
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: Multi-length exploration (remaining={remaining:.0f}s) ===")

    lengths_to_try = []
    base = n_base
    lengths_to_try.append(int(base * 1.25))
    lengths_to_try.append(int(base * 1.5))
    lengths_to_try.append(int(base * 1.75))
    # Also try a shorter length (sometimes helps)
    lengths_to_try.append(int(base * 0.75))
    lengths_to_try = sorted(set(lengths_to_try), key=lambda x: -abs(x - int(base * 1.5)))

    for li, n_try in enumerate(lengths_to_try[:3]):
        if n_try <= 50 or n_try > 5000:
            continue
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining < 60:
            break

        print(f"\n  --- Trying n={n_try} (remaining={remaining:.0f}s) ---")
        t_alloc = min(remaining * 0.25, 200.0)
        if t_alloc < 30:
            continue

        # Interpolate from best solution
        init_try = interpolate_sequence(best_seq, n_try)

        # Mirror descent at new length
        a_out, _ = mirror_descent_optimize(init_try, t_alloc * 0.6,
                                           T_start=30.0, T_end=0.5,
                                           n_stages=8, C_base=1.5, seed=100 + li)
        try_update(a_out, f"md_n{n_try}")

        # If time remains, subgradient polish
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining > 20:
            t_sp = min(remaining * 0.1, 50.0)
            if t_sp > 10:
                a_out2, _ = subgradient_polish(a_out, t_sp, n_peaks=5, seed=200 + li)
                try_update(a_out2, f"sgrad_n{n_try}")

    # === Phase 5: LP Nesterov at the best length found ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: LP Nesterov refinement (remaining={remaining:.0f}s) ===")
    n_best = len(best_seq)
    t5 = min(remaining * 0.30, 300.0)
    if t5 > 30 and solver is not None:
        a_out, _ = nesterov_lp_optimize(best_seq, t5, solver)
        try_update(a_out, f"lp_refine_n{n_best}")

    # === Phase 6: Final mirror descent polish ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: Final MD polish (remaining={remaining:.0f}s) ===")
    t6 = min(remaining * 0.20, 120.0)
    if t6 > 10:
        a_out, _ = mirror_descent_optimize(best_seq, t6, T_start=10.0, T_end=0.1,
                                           n_stages=8, C_base=0.8, seed=500)
        try_update(a_out, "final_md")

    # === Phase 7: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 7: Peak shaving (remaining={remaining:.0f}s) ===")
    t7 = min(remaining * 0.25, 100.0)
    if t7 > 5:
        a_out, _ = peak_shaving(best_seq, t7, seed=789)
        try_update(a_out, "peak_shave_1")

    # === Phase 8: Multi-start at best length ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 8: Multi-start (remaining={remaining:.0f}s) ===")
    n_final = len(best_seq)
    if remaining > 60:
        for ms in range(3):
            remaining = time_budget - (time.monotonic() - t_global_start)
            if remaining < 20:
                break
            t_ms = min(remaining * 0.25, 60.0)
            if t_ms < 10:
                break
            init_ms = structured_init(n_final, rng)
            # Blend with current best
            init_ms = 0.5 * init_ms + 0.5 * best_seq
            init_ms = init_ms / init_ms.sum()
            a_out, _ = mirror_descent_optimize(init_ms, t_ms, T_start=20.0, T_end=0.5,
                                               n_stages=6, C_base=1.0, seed=600 + ms)
            try_update(a_out, f"multi_start_{ms}")

    # === Phase 9: Final subgradient polish ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 9: Final sgrad polish (remaining={remaining:.0f}s) ===")
    t9 = min(remaining * 0.4, 60.0)
    if t9 > 5:
        a_out, _ = subgradient_polish(best_seq, t9, n_peaks=3, seed=999)
        try_update(a_out, "final_sgrad")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
