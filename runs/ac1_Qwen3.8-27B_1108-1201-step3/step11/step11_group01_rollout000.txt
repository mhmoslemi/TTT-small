import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):

def evaluate_safe(a):

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:

    _lp_solver = cp.ECOS
    return _lp_solver


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp(a0, time_limit, solver=None, max_iters=500000, restart_every=5000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0
    it_since_best = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            it_since_best = 0
        else:
            it_since_best += 1
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            it_since_best = 0

        # Random restart to escape local minima
        if (it + 1) % restart_every == 0 and it_since_best > restart_every // 2:
            rng_restart = np.random.default_rng(it)
            noise = rng_restart.normal(0, 0.01, n)
            a_perturbed = a + noise
            a_perturbed = np.maximum(a_perturbed, 0)
            sp = a_perturbed.sum()
            if sp > 1e-15:
                a_perturbed = a_perturbed / sp
            sc_pert = score_normalized(a_perturbed)
            if sc_pert < best_score * 1.02:  # accept if not too bad
                a = a_perturbed
                y = a_perturbed.copy()
                it_since_best = 0

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 5:
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} M_lp={M_lp:.6f}")
                last_print = elapsed

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n * T_val) * log_sum_exp

    # Gradient with respect to a
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * T_val * w[k] * a[k - idx]

    # Gradient with respect to z = log(a), using softmax parameterization
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_logsumexp(a0, time_limit, T_start=100.0, T_end=0.05, n_stages=20):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = 40

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, stage_T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, stage_T, n)
            return grad

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-25,
                         'gtol': 1e-18, 'maxfun': iters_per_stage * 500,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(res.x)
            sc = score_normalized(a_check)
            if (si + 1) % 5 == 0:
                print(f"    LSE stage={si+1}/{n_stages} T={stage_T:.2f}: score={sc:.8f}")
        except Exception:
            pass

    return softmax_from_z(z)


def projected_grad_momentum(a0, time_limit, n_steps=200000, eta_start=0.3, eta_end=0.005):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()
    v = np.zeros(n)
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    beta = 0.9
    gamma = 0.95

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        active = np.where(b >= max_b * 0.90)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        S = np.dot(g, a)
        grad_z = a * (g - S)

        eta = eta_start * (eta_end / eta_start) ** (step / max(1, n_steps))
        v = beta * v - eta * grad_z
        z_new = z + v
        a_new = softmax_from_z(z_new)

        sc_new = score_normalized(a_new)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            z = z_new
            a = a_new
        else:
            v = gamma * v
            # Don't update z, keep current
            pass

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng, variant=0):

        x = np.linspace(0, 1, n)

    elif variant == 5:
        x = np.linspace(0, 2 * np.pi, n)

    elif variant == 6:
        # Random smooth
        freqs = rng.integers(1, 10, 5)
        phases = rng.uniform(0, 2 * np.pi, 5)
        amps = rng.uniform(0.1, 0.3, 5)
        x = np.arange(n)
        a = np.ones(n)
        for f, p, amp in zip(freqs, phases, amps):
            a += amp * np.cos(2 * np.pi * f * x / n + p)
    elif variant == 7:
        # Bump function
        x = np.linspace(0, 1, n)
        center = 0.5
        width = 0.2 + 0.1 * rng.random()
        a = np.exp(-((x - center) / width) ** 2)
    else:
        a = np.ones(n) * (1.0 + 0.15 * np.cos(2 * np.pi * 5 * np.arange(n) / n))

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    def remaining_time():
        return _global_deadline - time.monotonic()

    # === Phase 1: Multi-start Nesterov LP (30% of budget) ===
    print(f"\n=== Phase 1: Multi-start Nesterov LP at n={n_base} ===")
    t1 = 0.30 * (budget_s - 30)
    if remaining_time() > t1 + 60:
        a_ms, _ = multi_start_nesterov(n_base, solver, t1, rng, n_starts=4)
        try_update(a_ms, "multi_start")

    # Also run from parent if different
    if remaining_time() > 120:
        t1b = min(0.10 * (budget_s - 30), remaining_time() * 0.6)
        if t1b > 30:
            print(f"\n=== Phase 1b: Nesterov LP from parent (budget={t1b:.0f}s) ===")
            a_out, _ = nesterov_lp(best_seq, t1b, solver=solver)
            try_update(a_out, "nesterov_parent")

    print(f"  After Phase 1: best_score={best_score:.8f}")

    # === Phase 2: Long Nesterov LP refinement (25% of budget) ===
    print(f"\n=== Phase 2: Long Nesterov LP refinement ===")
    remaining = remaining_time()
    t2 = min(0.25 * (budget_s - 30), remaining * 0.5)
    if t2 > 30:
        a_out2, _ = nesterov_lp(best_seq, t2, solver=solver, max_iters=500000)
        try_update(a_out2, "nesterov_long")

    print(f"  After Phase 2: best_score={best_score:.8f}")

    # === Phase 3: LogSumExp L-BFGS polishing (20% of budget) ===
    print(f"\n=== Phase 3: LSE L-BFGS polishing ===")
    remaining = remaining_time()
    t3 = min(0.20 * (budget_s - 30), remaining * 0.5)
    if t3 > 20:
        # Multiple LSE passes with different temperature schedules
        for pass_i in range(3):
            remaining = remaining_time()
            if remaining < 15:
                break
            t3_sub = min(t3 / 3.0, remaining * 0.6)
            if t3_sub < 10:
                break
            T_start = 50.0 * (pass_i + 1)
            T_end = 0.02
            a_lse = lbgfs_logsumexp(best_seq, t3_sub, T_start=T_start, T_end=T_end, n_stages=20)
            try_update(a_lse, f"lse_pass{pass_i+1}")

    print(f"  After Phase 3: best_score={best_score:.8f}")

    # === Phase 4: Projected gradient with momentum (10% of budget) ===
    print(f"\n=== Phase 4: Projected gradient momentum ===")
    remaining = remaining_time()
    t4 = min(0.10 * (budget_s - 30), remaining * 0.5)
    if t4 > 10:
        a_pg, _ = projected_grad_momentum(best_seq, t4, n_steps=300000, eta_start=0.2, eta_end=0.003)
        try_update(a_pg, "proj_grad")

    print(f"  After Phase 4: best_score={best_score:.8f}")

    # === Phase 5: Length exploration (10% of budget) ===
    print(f"\n=== Phase 5: Length exploration ===")
    remaining = remaining_time()
    t5_total = min(0.10 * (budget_s - 30), remaining * 0.5)
    if t5_total > 30:
        n_candidates = [n_base - 200, n_base - 100, n_base + 100, n_base + 200, n_base + 400, n_base - 400]
        n_candidates = [n for n in n_candidates if 500 <= n <= 3000]
        n_unique = list(dict.fromkeys(n_candidates))[:4]
        per_len = t5_total / max(1, len(n_unique))

        for n_try in n_unique:
            remaining = remaining_time()
            if remaining < 15:
                break
            t5 = min(per_len, remaining * 0.4)
            if t5 < 10:
                break
            print(f"  Trying n={n_try}, budget={t5:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            a_try, _ = nesterov_lp(init_try, t5, solver=solver, max_iters=100000)
            try_update(a_try, f"len_{n_try}")

    print(f"  After Phase 5: best_score={best_score:.8f}")

    # === Phase 6: Final LSE fine-tune ===
    print(f"\n=== Phase 6: Final LSE fine-tune ===")
    remaining = remaining_time()
    if remaining > 15:
        t6 = min(remaining * 0.5, 60.0)
        if t6 > 10:
            a_final = lbgfs_logsumexp(best_seq, t6, T_start=20.0, T_end=0.01, n_stages=15)
            try_update(a_final, "final_lse")

    # === Phase 7: Random perturbation + quick polish ===
    remaining = remaining_time()
    if remaining > 20:
        print(f"\n=== Phase 7: Random perturbation polish ===")
        t7 = min(remaining * 0.4, 40.0)
        if t7 > 10:
            for perturb_i in range(3):
                remaining = remaining_time()
                if remaining < 10:
                    break
                noise_scale = 0.005 * (perturb_i + 1)
                a_perturbed = best_seq + rng.normal(0, noise_scale, len(best_seq))
                a_perturbed = np.maximum(a_perturbed, 0)
                sp = a_perturbed.sum()
                if sp > 1e-15:
                    a_perturbed = a_perturbed / sp
                t7_sub = min(t7 / 3.0, remaining * 0.5)
                if t7_sub < 5:
                    break
                a_pol, _ = nesterov_lp(a_perturbed, t7_sub, solver=solver, max_iters=50000)
                try_update(a_pol, f"perturb_{perturb_i+1}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
