import numpy as np
import time
import math
from scipy import sparse
from scipy.fft import rfft, irfft
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_sparse_conv_mat(a):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    n_rows = 2 * n - 1
    # Precompute row/col indices
    total_nnz = n * n
    rows = np.empty(total_nnz, dtype=np.int32)
    cols = np.empty(total_nnz, dtype=np.int32)
    vals = np.empty(total_nnz, dtype=np.float64)
    idx = 0
    for k in range(n_rows):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        cnt = je - js + 1
        rows[idx:idx+cnt] = k
        cols[idx:idx+cnt] = np.arange(js, je + 1)
        vals[idx:idx+cnt] = a[k - np.arange(je, js - 1, -1)]
        idx += cnt
    return sparse.csr_matrix((vals, (rows, cols)), shape=(n_rows, n))

Now, for the spectral initialization:

The idea is to use the DFT to create a sequence with a flat-ish power spectrum. For a sequence a of length n:
- DFT: A[k] = sum_j a[j] * exp(-2*pi*i*j*k/n)
- Power spectrum: |A[k]|^2
- Circular autocorrelation: inverse DFT of |A[k]|^2

For linear autocorrelation, it's more complex, but a good approximation is to make |A[k]|^2 as flat as possible (subject to a >= 0 and sum(a) = 1).

We can use an iterative algorithm:
1. Start with random non-negative a
2. Compute DFT A
3. Set |A[k]| = target (e.g., constant or some smooth function)
4. Keep the phase of A
5. Take IDFT to get new a
6. Project onto non-negative (clip)
7. Repeat

This is essentially the Gerchberg-Saxton algorithm adapted for our problem.

Let me also think about what "target spectrum" to use. If we want the linear autocorrelation to be flat, the circular autocorrelation should be approximately flat (for long sequences, the linear and circular autocorrelations are similar away from the edges).

The flat circular autocorrelation corresponds to |A[k]|^2 = constant = (sum a)^2 / n = 1/n (when sum(a) = 1). So |A[k]| = 1/sqrt(n) for all k.

This is the "flat spectrum" target. Let me implement this:

```python
def spectral_init(n, rng, iterations=1000):

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    n_rows = 2 * n - 1
    total_nnz = n * n
    rows = np.empty(total_nnz, dtype=np.int64)
    cols = np.empty(total_nnz, dtype=np.int64)
    vals = np.empty(total_nnz, dtype=np.float64)
    idx = 0
    for k in range(n_rows):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        cnt = je - js + 1
        rows[idx:idx + cnt] = k
        cols[idx:idx + cnt] = np.arange(js, je + 1)
        vals[idx:idx + cnt] = a[k - je:k - js + 1][::-1]
        idx += cnt
    return sparse.csr_matrix((vals, (rows, cols)), shape=(n_rows, n))


def get_lp_solver():
    """Find a working LP solver."""
    candidates = [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]
    for s in candidates:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.SCS


def solve_best_response(a, solver):
    """Given fixed a (sum=1, a>=0), find b minimizing max_k conv(a,b)[k]."""
    n = len(a)
    T = build_sparse_conv_mat(a)
    b = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [T @ b <= M, cp.sum(b) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate') and b.value is not None:
            b_val = np.maximum(np.asarray(b.value, dtype=np.float64), 0)
            s = b_val.sum()
            if s > 1e-15:
                b_val /= s
                return b_val
    except Exception:
        pass
    return None


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def spectral_init(n, rng, n_iters=200):
    """Initialize using DFT spectral method (target flat spectrum)."""
    a = rng.random(n) + 0.1
    a = a / a.sum()
    target_mag = 1.0 / math.sqrt(n)
    for it in range(n_iters):
        A = np.fft.fft(a)
        mag = np.abs(A)
        phase = np.where(mag > 1e-300, A / (mag + 1e-300), 1.0 + 0j)
        A_new = target_mag * phase
        a = np.fft.ifft(A_new).real
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            a = rng.random(n) + 0.1
            a = a / a.sum()
        else:
            a = a / s
    return a


def score_fn(a):
    """Compute the objective value."""

def fista_lp_alternate(a0, time_limit, solver):
    """FISTA-accelerated LP alternating."""
    n = len(a0)

    y = a.copy()

    t_mom = 1.0
    
    best_a = a.copy()
    best_score = score_fn(a)
    t0 = time.monotonic()
    it = 0
    stall = 0
    rng = np.random.default_rng(123)
    
    while time.monotonic() - t0 < time_limit:

        if b_best is None:

            if it % 50 == 0:
                break
            continue

        momentum = (t_mom - 1.0) / t_new
        
        y_raw = b_best + momentum * (b_best - a_prev)
        y = project_simplex(y_raw)
        
        a_prev = a.copy()
        a = b_best.copy()
        t_mom = t_new
        it += 1
        
        # Evaluate
        sc_a = score_fn(a)
        sc_y = score_fn(y)
        
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            stall = 0
        elif sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        
        # Restart if stalled
        if stall == 100:
            noise = 0.05 * rng.standard_normal(n)
            y = a + noise
            y = project_simplex(y)
            a_prev = a.copy()
            t_mom = 1.0
            stall = 0
            print(f"    FISTA-LP restart at it={it}")
        
        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA-LP it={it} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

Let me also consider: instead of the LP alternating, what if I use a **direct coordinate descent** approach?

At each step, pick a coordinate i, and find the optimal value of a[i] (given all other a[j] fixed) subject to the constraint that max_k (a*a)[k] doesn't increase.

The optimal a[i] is the one that minimizes the maximum of the "affected" convolution entries. Only entries (a*a)[k] where k-i or k+i is in [0,n-1] are affected, which is all k in [i, i+n-1]... actually all k from 0 to 2n-2 are potentially affected.

This is a 1D optimization problem: minimize max_{k in K} (c_k + 2*a[i]*d_k + a[i]^2 * e_k) where c_k, d_k, e_k depend on the other coordinates. For e_k = 0 (which is the case since a[i]^2 only appears in (a*a)[2i]), it's a linear optimization: minimize max_k (c_k + 2*a[i]*d_k) subject to a[i] >= 0 and sum constraint.

This is solvable in O(|K|) time per coordinate update. But doing a full sweep over n coordinates is O(n * n) = O(n^2) per sweep, which for n=1200 is 1.44M operations per sweep. That's fast!

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA-accelerated LP alternating with sparse matrices (main workhorse)
2. Spectral (DFT) initialization for diverse starting points
3. Momentum subgradient method for fast local optimization
4. Multi-length exploration (n = 1000 to 2000)
5. Adaptive restart to escape local minima
"""

import numpy as np
import time
import math
from scipy import sparse
from scipy.fft import rfft, irfft
import cvxpy as cp


def fast_convolve(a):
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_sparse_conv_mat(a):
    n = len(a)
    n_rows = 2 * n - 1
    total_nnz = n * n
    rows = np.empty(total_nnz, dtype=np.int64)
    cols = np.empty(total_nnz, dtype=np.int64)
    vals = np.empty(total_nnz, dtype=np.float64)
    idx = 0
    for k in range(n_rows):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        cnt = je - js + 1
        rows[idx:idx + cnt] = k
        cols[idx:idx + cnt] = np.arange(js, je + 1)
        vals[idx:idx + cnt] = a[k - je:k - js + 1][::-1]
        idx += cnt
    return sparse.csr_matrix((vals, (rows, cols)), shape=(n_rows, n))


def get_lp_solver():
    candidates = [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]
    for s in candidates:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.SCS


def solve_best_response(a, solver):
    n = len(a)
    T = build_sparse_conv_mat(a)
    b = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    constraints = [T @ b <= M, cp.sum(b) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate') and b.value is not None:
            b_val = np.maximum(np.asarray(b.value, dtype=np.float64), 0)
            s = b_val.sum()
            if s > 1e-15:
                b_val /= s
                return b_val
    except Exception:
        pass
    return None


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def spectral_init(n, rng, n_iters=300):
    a = rng.random(n) + 0.1
    a = a / a.sum()
    target_mag = 1.0 / math.sqrt(n)
    for it in range(n_iters):
        A = np.fft.fft(a)
        mag = np.abs(A)
        phase = np.where(mag > 1e-300, A / (mag + 1e-300), 1.0 + 0j)
        A_new = target_mag * phase
        a_new = np.fft.ifft(A_new).real
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a = a_new / s
    return a


def score_fn(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * max(b.max(), 0.0) / (s * s))
