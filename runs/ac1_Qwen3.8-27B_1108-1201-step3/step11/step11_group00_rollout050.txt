1. LogSumExp-smoothed objective with FFT-based O(n log n) gradients

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_autocorr(a):
    """Full autocorrelation a*a using FFT, O(n log n)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_obj_grad(a, T):
    """
    Compute LogSumExp-smoothed objective and gradient.
    Objective: 2*n/T/s^2 * logsumexp(T * conv(a,a))
    This approximates 2*n*max(conv(a,a))/s^2 for large T.
    """
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return 0.0, np.zeros_like(a)
    
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    sum_exp = exp_Tb.sum()
    if sum_exp < 1e-300:
        return 0.0, np.zeros_like(a)
    
    w = exp_Tb / sum_exp
    logsum = max_Tb + np.log(sum_exp)
    L = logsum / T  # LSE-smoothed max
    
    obj = 2.0 * n * L / (s * s)
    
    # Gradient computation
    # dL/da[i] = sum_k w[k] * db[k]/da[i]
    # db[k]/da[i] = 2*a[k-i] if 0 <= k-i < n (with edge effects handled by FFT)
    # This is 2 * cross_correlation(w, a)
    
    conv_len = len(w) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W_fft = rfft(w, fft_size)
    A_rev_fft = rfft(a[::-1].copy(), fft_size)
    conv_result = irfft(W_fft * A_rev_fft, fft_size)[:conv_len]
    grad_L = 2.0 * conv_result[n - 1:n - 1 + n]
    
    # df/da = 2n * grad_L / s^2 - 4n * L / s^3
    grad = 2.0 * n * grad_L / (s * s) - 4.0 * n * L / (s * s * s)
    
    return obj, grad

def interpolate_sequence(f, new_n):
    """Interpolate a normalized nonnegative sequence to new length using PCHIP-like method."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp

def direct_gradient_descent(a0, time_limit, lr=0.01, T=50.0):
    """
    Direct projected gradient descent on the LSE-smoothed objective.
    More memory-efficient than L-BFGS for large n.
    """
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    v = np.zeros(n)  # momentum
    beta = 0.9
    
    step_count = 0
    while time.monotonic() - t0 < time_limit:
        obj_val, grad = lse_obj_grad(a, T)
        
        v = beta * v + (1 - beta) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a = np.ones(n) / n
            continue
        a_new = a_new / s
        
        # Adaptive learning rate
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        elif score < obj_val + lr * 0.01:
            a = a_new
        else:
            lr *= 0.9
        lr = min(lr, 0.1)
        
        step_count += 1
        if step_count % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      GD step {step_count} (t={elapsed:.0f}s): best={best_score:.8f}, lr={lr:.6f}")
        
        # Reduce temperature over time
        T = max(1.0, T * 0.999)
    
    return best_a, best_score

        t6 = min(remaining_time() * 0.25, 200.0)
        a_out, _ = peak_shaving(best_seq, t6, seed=123)
        try_update(a_out, "peak_shave_1")
    
    # Phase 7: Final LSE polish with lower temperature
    if remaining_time() > 15:
        print(f"\n=== Phase 7: Final LSE polish ===")
        t7 = min(remaining_time() * 0.30, 150.0)
        a_out, _ = lse_optimize(best_seq, t7, T_start=50.0, T_end=0.1, n_stages=15)
        try_update(a_out, "final_lse")
    
    # Phase 8: One more peak shave
    if remaining_time() > 10:
        t8 = min(remaining_time() * 0.40, 80.0)
        print(f"\n=== Phase 8: Final peak shaving ===")
        a_out, _ = peak_shaving(best_seq, t8, seed=789)
        try_update(a_out, "peak_shave_2")
    
    # Phase 9: Try a few random inits at optimal-ish lengths
    if remaining_time() > 30:
        print(f"\n=== Phase 9: Random init exploration ===")
        for ri in range(3):
            if remaining_time() < 20:
                break
            n_rand = n_base + rng.integers(-200, 400)
            n_rand = max(100, n_rand)
            init_r = structured_init(n_rand, rng)
            t_r = min(remaining_time() * 0.15, 100.0)
            a_out, _ = lse_optimize(init_r, t_r, T_start=100.0, T_end=1.0, n_stages=10)
            try_update(a_out, f"rand_init_{ri}_n{n_rand}")
    
    # Final verification
    if best_seq is not None:
        try:
            fs = evaluate_sequence([float(x) for x in best_seq])
            if np.isfinite(fs):
                best_score = fs
        except Exception:
            pass
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")
    else:
        print("\nFALLBACK: returning uniform sequence")
        best_seq = np.ones(1000)
        best_score = 2.0
    
    return [float(x) for x in best_seq]
