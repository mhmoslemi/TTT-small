def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Fast local evaluation (assumes a >= 0, sum > 0)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix_sparse(a):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    # Each column j has nonzeros in rows j..j+n-1 (clipped to [0, size-1])
    total = n * min(n, size)
    rows = np.empty(total, dtype=np.int32)
    cols = np.empty(total, dtype=np.int32)
    vals = np.empty(total, dtype=np.float64)
    count = 0
    for j in range(n):
        r_lo = j
        r_hi = min(j + n, size)
        length = r_hi - r_lo
        rows[count:count+length] = np.arange(r_lo, r_hi)
        cols[count:count+length] = j
        vals[count:count+length] = a[:length]
        count += length
    return csr_matrix((vals[:count], (rows[:count], cols[:count])), shape=(size, n))


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver, T_mat):
    """Given fixed a (nonneg, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)

    M_var = cp.Variable()
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = [solver, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    for s in solvers_to_try:
        if s is None:

        except Exception:

    return None, None


def multiplicative_weights_topk(a0, time_limit, n_steps, eta0=0.05, k_top=5):
    """MW with top-k exponential weighting of active peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()

    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0
    consecutive_fail = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        # Top-k peaks with exponential weighting
        top_k = min(k_top, len(b))
        idx_sorted = np.argpartition(b, -top_k)[-top_k:]
        b_top = b[idx_sorted]
        weights = b_top / b_top.sum()
        g = np.zeros(n)
        for w, k in zip(weights, idx_sorted):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            g[idx] += w * a[k - idx]
        log_mult = -eta * g
        log_mult -= log_mult.max()
        a_new = a * np.exp(log_mult)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        score_new = 2.0 * n * fast_convolve(a_new).max()

            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            consecutive_fail = 0
            eta = min(eta * 1.005, 0.4)
        else:

            eta *= 0.7
            if consecutive_fail > 200:
                break
        if step % 20000 == 0 and step > 0:

            print(f"    MW {elapsed:.0f}s: score={best_score:.8f} eta={eta:.6f} k={k_top}")

def dual_hedge_refine(a0, time_limit, eta0=0.01, n_steps=500000):
    """Solve the dual game using multiplicative weights (Hedge).
    Dual: max min_k (T_a^T y)[k] s.t. sum(y)=1, y>=0
    This gives a y that "weights" the lags. Then b* = argmin max is derived.
    We alternate: use y to form a surrogate gradient for updating a.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    size = 2 * n - 1
    y = np.ones(size) / size
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        # Find the worst lag for current a
        k_star = int(np.argmax(b))
        # Update y: increase weight on the worst lag
        y[k_star] *= (1.0 + eta)
        y = y / y.sum()
        # Compute surrogate gradient: g[j] = sum_k y[k] * a[|k-j|] (convolution with y)
        # g = convolve(a, y) (partial, taking the relevant part)
        # Actually g[j] = sum_k y[k] * (a[k-j] + a[j-k]) / 2 ... 
        # Simpler: g = the derivative of sum_k y[k]*b[k] w.r.t. a
        # sum_k y[k]*b[k] = sum_k y[k] * sum_j a[j]*a[k-j]
        # d/da[m] = sum_k y[k]*(a[k-m] + a[m-k]) (with bounds)
        # = conv(a, y)[m] + conv(a_rev, y)[m] ... 
        # Actually: = sum_k y[k] * (a[k-m]_valid + a[m-k]_valid)
        # Let h = convolve(y, a) (length 2n-1+size-1, but we only need the middle)
        # Simpler to just compute directly:
        g = np.zeros(n)
        # For each k with y[k] > threshold:
        active = np.where(y > 1e-10)[0]
        for k in active:
            wk = y[k]
            # Contribution: g[j] += wk * (a[k-j] + a[j-k]) for valid indices
            # a[k-j] valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
            # a[j-k] valid when 0 <= j-k < n, i.e., k <= j <= k+n-1
            js1 = max(0, k - n + 1)
            je1 = min(n - 1, k)
            if js1 <= je1:
                idx1 = np.arange(js1, je1 + 1)
                g[idx1] += wk * a[k - idx1]
            js2 = max(0, k)
            je2 = min(n - 1, k + n - 1)
            if js2 <= je2 and k != 0:
                idx2 = np.arange(js2, je2 + 1)
                # a[j-k] for j in [k, k+n-1]
                g[idx2] += wk * a[idx2 - k]
        # Update a multiplicatively
        log_mult = -eta * g
        log_mult -= log_mult.max()
        a_new = a * np.exp(log_mult)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.01, 0.5)
        else:
            eta *= 0.9
            if eta < 1e-10:
                break
        if step % 50000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    Hedge {elapsed:.0f}s: score={best_score:.8f} eta={eta:.6f}")
    return best_a, best_score

                stall_count += 1
            prev_best = best_score
            if stall_count >= 8:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 12)
                phase = rng.uniform(0, 2 * np.pi)
                noise = 0.03 * np.cos(freq * x + phase)
                y = np.maximum(y + noise, 0)
                sy2 = y.sum()
                if sy2 > 1e-15:
                    y = y / sy2
                T_mat = build_conv_matrix_sparse(y)
                stall_count = 0
            if (it + 1) % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP iter={it+1} ({elapsed:.0f}s, {lp_time:.3f}s/iter): score={best_score:.8f}")
    return best_a, best_score


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    sv = exp_z.sum()
    if sv < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / sv


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def lse_obj_grad_z(z, n, T_temp):
    """Compute LSE objective and gradient w.r.t. z for softmax parameterization."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T_temp
    scaled_shifted = scaled - scaled.max()
    exp_terms = np.exp(scaled_shifted)
    lse = max_b + T_temp * (scaled.max() + np.log(exp_terms.sum() + 1e-300))
    obj = 2.0 * n * lse
    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    active = np.where(w > 1e-16)[0]
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage=500):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func(x):
            if time.monotonic() > deadline:
                return 1e15
            o, _ = lse_obj_grad_z(x, n, stage_T)
            return o

        def jac(x):
            if time.monotonic() > deadline:
                return np.zeros(len(x))
            _, g = lse_obj_grad_z(x, n, stage_T)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-30,
                         'gtol': 1e-20, 'maxfun': maxiter_per_stage * 200,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass
    return softmax_from_z(z)


def coordinate_descent_polish(a0, time_limit, n_iterations=100000):
    """Coordinate descent: for each coordinate, find the optimal value given others fixed."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    for it in range(n_iterations):
        if time.monotonic() - t0 > time_limit:
            break
        # Pick a random coordinate
        j = rng.integers(0, n)
        # Current b (precompute for efficiency every few steps)
        if it % 10 == 0:
            b = fast_convolve(a)
        # Try a small range of values for a[j]
        # The contribution of a[j] to b[k] is: a[j]*a[k-j] + a[k-j]*a[j] (symmetric)
        # Simplify: just try perturbing a[j]
        current_val = a[j]
        # Try: current_val * (1 + delta) for delta in a grid
        best_delta = 0
        best_local_score = best_score
        # Quick evaluation: only update affected b values
        for delta in [-0.1, -0.05, -0.02, 0.0, 0.02, 0.05, 0.1, 0.2]:
            a_trial_val = current_val * (1.0 + delta)
            if a_trial_val < 0:
                a_trial_val = 0.0
            a[j] = a_trial_val
            s_new = a.sum()
            if s_new < 1e-15:
                a[j] = current_val
                continue
            # Quick score (full recompute for correctness)
            b_new = fast_convolve(a)
            sc = 2.0 * n * b_new.max() / (s_new * s_new)
            if sc < best_local_score:
                best_local_score = sc
                best_delta = delta
        a[j] = current_val * (1.0 + best_delta)
        if a[j] < 0:
            a[j] = 0
        a = a / a.sum()
        if best_local_score < best_score - 1e-12:
            best_score = best_local_score
            best_a = a.copy()
    return best_a, best_score


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 2 + 0.05
    elif choice == 5:
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.7 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * 7 * x + 0.3)
        a = np.maximum(a, 0)
    elif choice == 7:
        # Power law (one-sided)
        a = (x + 0.01) ** 3
    elif choice == 8:
        # W-shaped
        a = 0.3 + 0.5 * np.cos(2 * np.pi * 2 * x) ** 2
    else:
        # Random smooth
        a = np.ones(n) + 0.3 * np.sin(2 * np.pi * rng.uniform(1, 5) * x + rng.uniform(0, 6))
        a = np.maximum(a, 0)
    a = np.maximum(a, 0)
    sv = a.sum()
    if sv < 1e-15:
        return np.ones(n) / n
    return a / sv


def make_palindromic(a):
    """Make a sequence palindromic."""
    n = len(a)
    a_pal = (a + a[::-1]) / 2.0
    a_pal = np.maximum(a_pal, 0)
    sv = a_pal.sum()
    if sv < 1e-15:
        return np.ones(n) / n
    return a_pal / sv


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length using cubic spline."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    sv = f.sum()
    if sv < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / sv
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def spectral_init(n, rng):
    """Initialize via spectral method: try to make power spectrum flat."""
    # Start with a random phase spectrum
    fft_size = 1 << (2 * n - 1).bit_length()
    # Random amplitude (will be adjusted)
    A_magnitude = np.ones(fft_size // 2 + 1)
    A_phase = rng.uniform(0, 2 * np.pi, size=A_magnitude.shape)
    A_phase[0] = 0
    if fft_size % 2 == 0 and fft_size // 2 < len(A_phase):
        A_phase[fft_size // 2] = 0
    A = A_magnitude * np.exp(1j * A_phase)
    # Iteratively: compute time domain, enforce non-negativity, go back
    for _ in range(50):
        a_full = irfft(A, fft_size)
        a = a_full[:n]
        a = np.maximum(a, 0)
        sv = a.sum()
        if sv < 1e-15:
            break
        a = a / sv
        # Recompute spectrum
        A_new = rfft(a, fft_size)
        # Keep the phase from A_new but try to flatten magnitude
        A_magnitude_new = np.abs(A_new)
        # Blend towards flat
        A_magnitude = 0.7 * A_magnitude_new + 0.3 * A_magnitude_new.mean()
        A = A_magnitude * np.exp(1j * np.angle(A_new))
    a = np.maximum(a, 0)
    sv = a.sum()
    if sv < 1e-15:
        return np.ones(n) / n
    return a / sv


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses top-k MW, LP alternating, dual Hedge, multi-length, LSE L-BFGS, coord descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}, budget: {budget_s:.0f}s")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Initialized: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        sv = a.sum()
        if sv < 0.01:
            return
        a = a / sv
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    n_best = len(best_seq)

    # === Phase 1: Fast MW with top-k ===
    print(f"\n=== Phase 1: MW top-k n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.08, 100.0)
    if t1 > 15:
        a_mw, _ = multiplicative_weights_topk(best_seq, t1, n_steps=1000000, eta0=0.04, k_top=7)
        try_update(a_mw, "mw_topk")

    # === Phase 2: LP alternating (main workhorse) ===
    print(f"\n=== Phase 2: LP n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.38, 650.0)
    if t2 > 60:
        print(f"  Budget: {t2:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t2, solver, n_max_iters=2000000)
        try_update(a_lp, "lp_main")

    # === Phase 3: Dual Hedge ===
    print(f"\n=== Phase 3: Dual Hedge (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t3 = min(rem * 0.1, 150.0)
        a_hedge, _ = dual_hedge_refine(best_seq, t3, eta0=0.005, n_steps=200000)
        try_update(a_hedge, "hedge")

    # === Phase 4: Palindromic variant ===
    print(f"\n=== Phase 4: Palindromic LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t4 = min(rem * 0.1, 150.0)
        a_pal = make_palindromic(best_seq)
        a_pal_lp, _ = nesterov_lp_refine(a_pal, t4, solver, n_max_iters=300000)
        try_update(a_pal_lp, "pal_lp")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (score={best_score:.8f}) ===")
    lengths_to_try = [1500, 2000, 1000]
    for ln in lengths_to_try:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        if abs(len(best_seq) - ln) < 50:
            continue
        t_ln = min(rem * 0.12, 200.0)
        if t_ln < 30:
            continue
        print(f"  Trying n={ln}, budget={t_ln:.0f}s")
        # Two inits: interpolate + spectral
        init_ln = interpolate_sequence(best_seq, ln)
        init_ln_spec = spectral_init(ln, rng)
        # Pick the better init
        sc1 = evaluate_local(init_ln)
        sc2 = evaluate_local(init_ln_spec)
        if sc2 < sc1:
            init_ln = init_ln_spec
            print(f"    Spectral init better: {sc2:.8f} vs {sc1:.8f}")
        # Quick MW
        t_mw = t_ln * 0.25
        a_ln_mw, _ = multiplicative_weights_topk(init_ln, t_mw, n_steps=200000, eta0=0.04, k_top=5)
        try_update(a_ln_mw, f"mw_n{ln}")
        # LP
        rem = deadline - time.monotonic()
        t_lp = min(rem * 0.4, t_ln * 0.6)
        if t_lp > 30:
            a_ln_lp, _ = nesterov_lp_refine(a_ln_mw, t_lp, solver, n_max_iters=300000)
            try_update(a_ln_lp, f"lp_n{ln}")

    # === Phase 6: LSE L-BFGS fine-tune ===
    print(f"\n=== Phase 6: LSE L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 50:
        t6 = min(rem * 0.18, 300.0)
        T_stages = [20.0, 10.0, 5.0, 2.5, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
        print(f"  {len(T_stages)} stages, budget={t6:.0f}s")
        a_lse = lbgfs_refine(best_seq, t6, T_stages, maxiter_per_stage=600)
        try_update(a_lse, "lse")

    # === Phase 7: Coordinate descent polish ===
    print(f"\n=== Phase 7: Coord Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t7 = min(rem * 0.12, 120.0)
        a_cd, _ = coordinate_descent_polish(best_seq, t7, n_iterations=50000)
        try_update(a_cd, "coord")

    # === Phase 8: Final LP polish ===
    print(f"\n=== Phase 8: Final LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 25:
        t8 = min(rem * 0.3, 100.0)
        a_fin, _ = nesterov_lp_refine(best_seq, t8, solver, n_max_iters=200000)
        try_update(a_fin, "final_lp")

    # === Phase 9: Final MW polish ===
    print(f"\n=== Phase 9: Final MW (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t9 = min(rem * 0.4, 40.0)
        a_fin_mw, _ = multiplicative_weights_topk(best_seq, t9, n_steps=300000, eta0=0.01, k_top=3)
        try_update(a_fin_mw, "final_mw")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} n={len(best_seq)}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
