"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated auto-convolution
2. LogSumExp-smoothed objective with multi-stage temperature annealing
3. L-BFGS-B optimization in log-parameterization
4. Diverse bump-based and random initializations
5. Discrete coordinate descent refinement
6. Multi-length exploration with interpolation
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate the objective locally (scale-invariant)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def lse_obj_grad_z(z, n, T):
    """
    Compute LSE-smoothed objective and gradient in log-parameterization.
    z: log-space parameters, a = softmax(z) (normalized).
    Returns (obj, grad_z).
    """
    # Compute a from z (stable softmax)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    
    # Compute convolution
    c = fast_convolve(a)  # length 2n-1
    c_max = c.max()
    
    # LSE approximation of max(c)
    exp_c = np.exp((c - c_max) / T)
    lse_val = c_max + T * np.log(exp_c.sum())
    obj = 2.0 * n * lse_val
    
    # Softmax weights over c
    w = exp_c / exp_c.sum()  # length 2n-1

    # But since c[k] = sum_i a[i]*a[k-i], d(c[k])/da[m] = a[k-m] (if valid) + a[m-k] (if valid)
    # Wait, more carefully: c[k] = sum_{i} a[i] * a[k-i]
    # d(c[k])/da[m] = a[k-m] (from the a[i] term, when i=m) + a[m-k] (from the a[k-i] term, when k-i=m, i.e., i=k-m)

    # The a[m] appears in: (i=m) term: a[m]*a[k-m], and (k-i=m) term: a[k-m]*a[m].
