import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def solve_lp(f, M, active_set=None):
    """Solve the LP: max sum(g) s.t. (f*g)[k] <= M for k in active_set, g >= 0"""
    n = len(f)
    if active_set is None:
        active_set = range(2*n - 1)
    
    # A_eq constraint: sum(g) is maximized, so we minimize -sum(g)
    c = -np.ones(n)
    
    # Inequality constraints: (f*g)[k] <= M for k in active_set
    # A_ub @ g <= b_ub
    A_ub = np.zeros((len(active_set), n))
    b_ub = M * np.ones(len(active_set))
    
    for idx, k in enumerate(active_set):
        # (f*g)[k] = sum_j f[k-j] * g[j] for 0 <= k-j <= n-1
        for j in range(max(0, k-n+1), min(n, k+1)):
            A_ub[idx, j] = f[k-j]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs',
                  bounds=[(0, None)] * n)
    
    if res.success:
        return res.x
    return None

def lp_iteration(f, t_min=0.01, t_max=1.0, n_backtrack=5):
    """One LP-based iteration: solve LP, get direction, update f."""
    n = len(f)
    b = fast_convolve(f)
    M = b.max()
    
    # Active set: positions where b[k] >= 0.9 * M
    threshold = 0.9 * M
    active = np.where(b >= threshold)[0]
    
    # Solve LP
    g = solve_lp(f, M, active)
    if g is None:
        return f, M
    
    # Rescale
    g_sum = g.sum()
    if g_sum < 1e-15:
        return f, M
    g = g / g_sum
    
    # Line search: try different t values
    best_f = f.copy()
    best_obj = 2.0 * n * M  # current objective (normalized)
    
    for t in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05, t_min]:

        f_new = normalize(f_new)
        b_new = fast_convolve(f_new)
        M_new = b_new.max()
        obj_new = 2.0 * n * M_new
        if obj_new < best_obj:
            best_obj = obj_new
            best_f = f_new.copy()
    
    return best_f, best_obj / (2.0 * n)

import numpy as np

from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate score locally (normalized, sum=1)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def cosine_taper(n):

    x = np.linspace(0, np.pi, n)

def solve_lp_direction(f, M, active_set):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in active_set, g >= 0.
    Returns g (unnormalized) or None if infeasible."""
    n = len(f)
    m = len(active_set)
    if m == 0:
        return None

    c = -np.ones(n)
    A_ub = np.zeros((m, n))
    b_ub = M * np.ones(m)

    for idx, k in enumerate(active_set):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        if j_end > j_start:
            A_ub[idx, j_start:j_end] = f[k - j_start:k + 1 - j_start + 1][:j_end - j_start]
            # More carefully: for j in [j_start, j_end), A_ub[idx, j] = f[k-j]
            for j in range(j_start, j_end):
                A_ub[idx, j] = f[k - j]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs',
                      bounds=[(0, None)] * n, options={'time_limit': 10})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_step(f, active_size=15, n_line_search=6):
    """One LP-based iteration. Returns (new_f, new_M, improved)."""
    n = len(f)
    b = fast_convolve(f)
    M = b.max()

    # Active set: top positions where b[k] is close to max
    sorted_idx = np.argsort(b)[::-1]
    threshold = M * 0.85
    active = sorted_idx[np.where(b[sorted_idx] >= threshold)[0]][:min(active_size, 2 * n - 1)]
    if len(active) < 3:
        active = sorted_idx[:active_size]

    g = solve_lp_direction(f, M, active)
    if g is None or g.sum() < 1e-15:
        return f, M, False

    g = normalize(g)

    # Line search
    best_f = f.copy()
    best_obj = 2.0 * n * M

    for t in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]:
        f_new = normalize((1.0 - t) * f + t * g)
        b_new = fast_convolve(f_new)
        M_new = b_new.max()
        obj_new = 2.0 * n * M_new
        if obj_new < best_obj:
            best_obj = obj_new
            best_f = f_new

    improved = best_obj < 2.0 * n * M - 1e-15
    return best_f, best_obj / (2.0 * n), improved


def lse_obj_grad(a, T):
    """LSE-smoothed objective and gradient. a is normalized (sum=1)."""
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    lse = c_max + T * np.log(exp_c.sum())
    obj = 2.0 * n * lse

    w = exp_c / exp_c.sum()

    # Gradient computation via FFT
    fft_size = 1 << (len(w) + n - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    R = irfft(W * np.conj(A), fft_size)
    b_grad = R[:n]

    g = 2.0 * b_grad
    idx = 2 * np.arange(n)
    valid = idx < (2 * n - 1)
    g[valid] -= a[valid] * w[idx[valid]]

    return obj, g


def lse_lbgfs(a0, time_limit, T_stages):
    """L-BFGS with LSE smoothing and temperature annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func(z_in):
            z_s = z_in - z_in.max()
            exp_z = np.exp(z_s)
            a = exp_z / exp_z.sum()
            obj, _ = lse_obj_grad(a, stage_T)
            return obj

        def jac(z_in):
            z_s = z_in - z_in.max()
            exp_z = np.exp(z_s)
            a = exp_z / exp_z.sum()
            obj, grad_a = lse_obj_grad(a, stage_T)
            S = np.dot(grad_a, a)
            return a * (grad_a - S)

        iters = max(10, int(remaining * 20))
        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': iters, 'ftol': 1e-18,
                                    'gtol': 1e-14, 'maxfun': iters * 100, 'disp': False})
            z = res.x
        except Exception:
            pass

    z_s = z - z.max()
    exp_z = np.exp(z_s)
    return exp_z / exp_z.sum()


def spectral_refine(a0, time_limit, n_iters=2000):
    """Flatten power spectrum to reduce autocorrelation peaks."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    t0 = time.monotonic()
    lr = 0.003

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        A = rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        r = (P - P_mean) / (P_mean + 1e-15)

        H = np.zeros(fft_size // 2 + 1, dtype=np.complex128)
        mask = P > 1e-10
        H[mask] = -r[mask] * P_mean * A[mask] / (P[mask] + 1e-10)
        h = irfft(H * 2, fft_size)[:n]
        h -= h.mean()
        h_norm = np.linalg.norm(h)
        if h_norm < 1e-15:
            break
        h = h / h_norm

        for scale in [lr, lr * 0.3, lr * 0.1]:
            a_new = np.maximum(a + scale * h, 0)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new = a_new / s
            sc = evaluate_local(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
            break

        if it % 300 == 0:
            lr *= 0.8

    return best_a, best_score


def subgradient_polish(a0, time_limit, n_steps=50000):
    """Subgradient method with exact max evaluation."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    eta = 0.02

        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()

        # Active set: top positions
        top_k = min(5, 2 * n - 1)
        active = np.where(b >= M * 0.95)[0][:top_k]
        if len(active) == 0:
            active = np.array([n - 1])

        # Subgradient
        g = np.zeros(n)
        for k in active:
            j_start = max(0, k - n + 1)
            j_end = min(n, k + 1)
            for j in range(j_start, j_end):

        log_alpha -= log_alpha.max()
        a_new = a * np.exp(log_alpha)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s
        sc = evaluate_local(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()

        if step % 500 == 0:
            eta *= 0.95

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    s = f_interp.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Phase 1: LP-based coarse optimization with adaptive active set
    Phase 2: LSE-smoothed L-BFGS fine-tuning
    Phase 3: Spectral shaping
    Phase 4: Multi-start exploration
    Phase 5: Final polish (subgradient + very fine LSE)
    """
    total_time = max(10.0, float(budget_s))
    deadline = time.monotonic() + total_time - 15.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    def update_best(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # === Initialization ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            sc = evaluate_local(parent)
            print(f"Loaded parent: n={len(parent)}, score={sc:.10f}")
            best_seq = parent.copy()
            best_score = sc
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        candidates = diverse_init(n_init, rng)
        scores = [evaluate_local(c) for c in candidates]
        best_idx = np.argmin(scores)
        parent = candidates[best_idx].copy()
        best_seq = parent.copy()
        best_score = scores[best_idx]
        print(f"Default init: n={len(parent)}, score={best_score:.10f}")

    n = len(best_seq)
    print(f"Starting: n={n}, score={best_score:.10f}")

    # === Phase 1: LP-based optimization (50% of time) ===
    print(f"\n=== Phase 1: LP Optimization ===")
    t_lp_end = time.monotonic() + (deadline - time.monotonic()) * 0.45
    a = best_seq.copy()
    lp_iter = 0
    active_size = 10
    no_improve_count = 0

    while time.monotonic() < t_lp_end:
        if time.monotonic() > deadline:
            break

        # Adaptive active set: grow as we converge
        if lp_iter % 50 == 0:
            active_size = min(50, 10 + lp_iter // 50)

        a_new, M_new, improved = lp_step(a, active_size=active_size)

        if improved:
            a = a_new
            new_score = 2.0 * n * M_new
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                no_improve_count = 0
                if lp_iter % 50 == 0:
                    print(f"  LP iter {lp_iter}: score={best_score:.10f}")
            else:
                no_improve_count += 1
        else:
            no_improve_count += 1

        lp_iter += 1

        # Periodic LSE refinement every 100 LP iterations
        if lp_iter % 100 == 0 and time.monotonic() < t_lp_end - 30:
            T_quick = [1.0, 0.3, 0.1]
            a_lse = lse_lbgfs(a, 20.0, T_quick)
            sc_lse = evaluate_local(a_lse)
            if sc_lse < best_score:
                best_score = sc_lse
                best_seq = a_lse.copy()
                a = a_lse.copy()
                print(f"  LSE refinement at LP iter {lp_iter}: score={best_score:.10f}")

        # If stuck for too long, try a different approach
        if no_improve_count > 200:
            print(f"  LP stuck at iter {lp_iter}, trying perturbation...")
            noise = rng.normal(0, 0.01, n)
            a_pert = normalize(a + noise)
            a_new, _, imp = lp_step(a_pert, active_size=active_size)
            if imp and evaluate_local(a_new) < best_score:
                a = a_new
                best_score = evaluate_local(a)
                best_seq = a.copy()
            no_improve_count = 0
            active_size = max(10, active_size - 5)

    update_best(a, "lp_phase")
    print(f"  Phase 1 done: {lp_iter} LP iterations, best={best_score:.10f}")

    # === Phase 2: LSE fine-tuning (25% of time) ===
    print(f"\n=== Phase 2: LSE Fine-Tuning ===")
    t_lse_end = time.monotonic() + (deadline - time.monotonic()) * 0.30
    T_stages = [5.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001]

    # Split time among stages, giving more time to lower T
    n_stages = len(T_stages)
    weights = np.ones(n_stages)
    weights[:3] = 0.5  # Less time for high T
    weights[-5:] = 2.0  # More time for low T
    weights /= weights.sum()
    total_lse_time = min(t_lse_end - time.monotonic(), total_time * 0.25)

    for i, T in enumerate(T_stages):
        t_alloc = total_lse_time * weights[i]
        if t_alloc < 2:
            continue
        a_lse = lse_lbgfs(best_seq, t_alloc, [T])
        update_best(a_lse, f"lse_T{T}")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # === Phase 3: Spectral shaping (10% of time) ===
    print(f"\n=== Phase 3: Spectral Shaping ===")
    t_spec = min((deadline - time.monotonic()) * 0.10, 150.0)
    if t_spec > 10:
        a_spec, _ = spectral_refine(best_seq, t_spec, n_iters=5000)
        update_best(a_spec, "spectral")
    print(f"  Phase 3 done: best={best_score:.10f}")

    # === Phase 4: Multi-start exploration (10% of time) ===
    print(f"\n=== Phase 4: Multi-Start Exploration ===")
    t_explore = min((deadline - time.monotonic()) * 0.15, 250.0)
    if t_explore > 30:
        n_starts = 3
        t_per_start = t_explore / n_starts
        n_current = len(best_seq)
        n_tries = [n_current, n_current + 200, n_current - 200]

        for i in range(min(n_starts, len(n_tries))):
            n_try = n_tries[i]
            if n_try < 100 or n_try > 5000:
                continue
            rem = deadline - time.monotonic()
            if rem < 20:
                break

            init = interpolate_sequence(best_seq, n_try)
            # Quick LP warm-up
            a_try = init.copy()
            n_lp_iter = min(100, int(t_per_start * 3))
            for _ in range(n_lp_iter):
                if time.monotonic() > deadline - 20:
                    break
                a_try, _, _ = lp_step(a_try, active_size=10)
            # Quick LSE
            T_short = [1.0, 0.3, 0.1, 0.05]
            a_try = lse_lbgfs(a_try, t_per_start * 0.4, T_short)
            update_best(a_try, f"explore_n{n_try}")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # === Phase 5: Final polish (remaining time) ===
    print(f"\n=== Phase 5: Final Polish ===")
    rem = deadline - time.monotonic()
    if rem > 10:
        # Subgradient polish
        t_sub = min(rem * 0.3, 60.0)
        if t_sub > 5:
            a_sub, _ = subgradient_polish(best_seq, t_sub, n_steps=30000)
            update_best(a_sub, "subgradient")

        # Very fine LSE
        rem = deadline - time.monotonic()
        if rem > 10:
            T_final = [0.01, 0.005, 0.002, 0.001]
            a_fin = lse_lbgfs(best_seq, min(rem * 0.5, 60.0), T_final)
            update_best(a_fin, "final_lse")

        # One more LP pass
        rem = deadline - time.monotonic()
        if rem > 10:
            a = best_seq.copy()
            for _ in range(min(100, int(rem * 5))):
                if time.monotonic() > deadline - 5:
                    break
                a_new, _, imp = lp_step(a, active_size=30)
                if imp:
                    a = a_new
                    sc = evaluate_local(a)
                    if sc < best_score:
                        best_score = sc
                        best_seq = a.copy()
            update_best(a, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        # Also verify with the actual evaluate_sequence
        try:
            test_result = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
            if np.isfinite(test_result) and test_result < best_score:
                best_score = test_result
        except Exception:
            pass
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
