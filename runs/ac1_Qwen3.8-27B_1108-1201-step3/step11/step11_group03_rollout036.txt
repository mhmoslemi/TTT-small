import time
from scipy.fft import rfft, irfft
import cvxpy as cp

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def local_score(a):
    n = len(a)
    b = fast_convolve(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return float(2 * n * b.max() / (s * s))

def get_solver():
    for s in [cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status == 'optimal':
                return s
        except:
            continue
    return cp.ECOS

def solve_lp(a_fixed, solver):
    n = len(a_fixed)
    b = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    # Build constraint matrix T where T @ b = conv(a_fixed, b)
    # T[k, j] = a_fixed[k-j] if 0 <= k-j < n, else 0
    T = np.zeros((2*n-1, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed
    cons = [T @ b <= M, cp.sum(b) == 1]
    prob = cp.Problem(cp.Minimize(M), cons)
    prob.solve(solver=solver)
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(b.value, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s

    a = np.maximum(a0, 0)
    a = a / a.sum()
    y = a.copy()
    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()

        if time.monotonic() - t0 > time_limit:
            break
        b_new = solve_lp(y, solver)
        if b_new is None:
            continue
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        y = y / y.sum()
        a = ((it+1) * a + alpha * (it+2) * b_new) / (it+1 + alpha*(it+2))
        a = np.maximum(a, 0)
        a = a / a.sum()
        
        sa = local_score(a)
        sy = local_score(y)
        if sa < best_score:
            best_score = sa
            best_a = a.copy()
        if sy < best_score:
            best_score = sy
            best_a = y.copy()
    
    return best_a, best_score

import numpy as np

from scipy.fft import rfft, irfft
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def local_score(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (sum=1, nonneg), find b minimizing max conv(a,b)."""
    n = len(a_fixed)

    M_var = cp.Variable()

    # This is T @ b where T[k,j] = a_fixed[k-j] if 0<=k-j<n
    # T is (2n-1) x n, each column j is a_fixed shifted
    T_data = []
    for j in range(n):
        T_data.append(a_fixed)
    # T[k,j] = a_fixed[k-j] for 0<=k-j<n

    T_cols = []

    for j in range(n):

        if time.monotonic() - t0 > time_limit:
            break
        
        b_new = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        
        alpha = 2.0 / (it + 2)

        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        score_a = local_score(a)
        score_y = local_score(y)

            best_a = a.copy()

            prev_best = best_score

                noise = 0.01 * rng.standard_normal(n)

                sy2 = y.sum()
                if sy2 > 1e-15:

def build_q_sparse(k, n):
    """Build sparse Q matrix for conv(a,a)[k] = a^T Q a."""
    from scipy.sparse import lil_matrix, csc_matrix
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)


def solve_active_qcqp(a_init, active_set, solver, verbose=False):

    t_var = cp.Variable()

        Q = build_q_sparse(int(k), n)
        constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    solvers_to_try = [cp.MOSEK, solver, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.asarray(a_var.value, dtype=np.float64)
                a_val = np.maximum(a_val, 0)

                    continue
                a_val = a_val / s_sum
                t_val = float(max(0, t_var.value))
                if verbose:
                    c = fast_convolve(a_val)
                    print(f"      QCQP: t={t_val:.15e}, actual_max={c.max():.15e}")
                return a_val, t_val
        except Exception as e:
            if verbose:
                print(f"      Solver {s} failed: {e}")
            continue
    return None, None


def newton_kkt_solve(a_init, active_set, max_iters=50, verbose=False):
    """
    Direct Newton method on the KKT system for the active-set problem.
    
    Minimize t s.t. a^T Q_k a = t for k in active, sum(a)=1, a>=0
    KKT: 2*sum(lambda_j * Q_kj * a) = lambda_0 * 1 on support
         a^T Q_kj a = t for all j
         sum(a) = 1
         sum(lambda) = 1
    """
    n = len(a_init)
    m = len(active_set)
    
    if m == 0:
        return None, None
    
    # Start from a_init, normalize
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    # Initial guesses
    c = fast_convolve(a)
    t = float(np.max(c[active_set]))
    lam = np.ones(m) / m
    lam0 = 2.0 * t / (1.0 / n) * n  # rough initial guess
    
    for it in range(max_iters):
        # Compute convolution
        c = fast_convolve(a)
        
        # Compute residuals
        # F_j = c[k_j] - t for j = 0,...,m-1
        # F_m = sum(a) - 1
        # F_{m+1} = sum(lam) - 1
        # F_{m+2+i} = 2*sum_j lam_j * a[k_j - i] - lam0 for i=0,...,n-1
        
        F = np.zeros(n + m + 2)
        F[0:m] = c[active_set] - t
        F[m] = a.sum() - 1.0
        F[m + 1] = lam.sum() - 1.0
        
        # Gradient part: 2*sum_j lam_j * (Q_kj a)[i] = 2*sum_j lam_j * a[k_j - i]
        grad = np.zeros(n)
        for j in range(m):
            kj = active_set[j]
            js = max(0, kj - n + 1)
            je = min(n - 1, kj)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * lam[j] * a[kj - idx]
        F[m + 2:] = grad - lam0
        
        # Check convergence
        res_norm = np.max(np.abs(F))
        if res_norm < 1e-14:
            if verbose:
                print(f"      Newton converged at iter {it}, residual={res_norm:.2e}")
            break
        
        # Build Jacobian
        # J is (n+m+2) x (n+m+2)
        # Variables: [a(0..n-1), t, lam(0..m-1), lam0]
        J = np.zeros((n + m + 2, n + m + 2))
        
        # Block [0:m, 0:n] = dc[k_j]/da[l] = 2*a[k_j - l] (with bounds)
        for j in range(m):
            kj = active_set[j]
            # For column l: a[kj - l] if 0 <= kj-l < n, else 0
            # This is a shifted version of a
            row = np.zeros(n)
            js = max(0, kj - n + 1)
            je = min(n - 1, kj)
            idx = np.arange(js, je + 1)
            row[idx] = 2.0 * a[kj - idx]
            J[j, 0:n] = row
        
        # Block [0:m, n] = -1
        J[0:m, n] = -1.0
        
        # Block [m, 0:n] = 1
        J[m, 0:n] = 1.0
        
        # Block [m+1, n+1:n+1+m] = 1
        J[m + 1, n + 1:n + 1 + m] = 1.0
        
        # Block [m+2:m+2+n, 0:n] = d/dl [2*sum_j lam_j * a[k_j - i]]
        # = 2*sum_j lam_j * Q_kj[i,l] where Q_kj[i,l]=1 if i+l=kj
        # So J[m+2+i, l] = 2*sum_j lam_j * (1 if 0<=kj-j... 
        # Actually: d/dl of a[kj - i] w.r.t. a[l] is delta(kj - i, l)
        # So J[m+2+i, l] = 2 * sum_j lam_j * delta(l, kj - i)
        # = 2 * sum_j lam_j * (1 if l = kj - i and valid)
        for j in range(m):
            kj = active_set[j]
            for i in range(n):
                l = kj - i
                if 0 <= l < n:
                    J[m + 2 + i, l] += 2.0 * lam[j]
        
        # Block [m+2:m+2+n, n+1:n+1+m] = 2*a[k_j - i] for each j
        for j in range(m):
            kj = active_set[j]
            col = np.zeros(n)
            js = max(0, kj - n + 1)
            je = min(n - 1, kj)
            idx = np.arange(js, je + 1)
            col[idx] = 2.0 * a[kj - idx]
            J[m + 2:m + 2 + n, n + 1 + j] = col
        
        # Block [m+2:m+2+n, n+1+m] = -1
        J[m + 2:m + 2 + n, n + 1 + m] = -1.0
        
        # Solve J * dx = -F
        try:
            dx = np.linalg.solve(J, -F)
        except np.linalg.LinAlgError:
            # Fallback: use lstsq
            dx, _, _, _ = np.linalg.lstsq(J, -F, rcond=None)
        
        # Line search for non-negativity preservation
        a_step = dx[0:n]
        # Find max alpha that keeps a + alpha*a_step >= 0
        neg_mask = a_step < -1e-16
        if np.any(neg_mask):
            max_alpha = np.min(-a[neg_mask] / a_step[neg_mask]) * 0.99
            max_alpha = max(max_alpha, 1e-6)
        else:
            max_alpha = 1.0
        
        alpha = min(1.0, max_alpha)
        
        # Update
        a_new = a + alpha * a_step
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
            s_new = 1.0
        a = a_new / s_new
        
        t = t + alpha * dx[n]
        t = max(t, 0)
        lam = lam + alpha * dx[n + 1:n + 1 + m]
        lam = np.maximum(lam, 0)
        lam_sum = lam.sum()
        if lam_sum > 1e-15:
            lam = lam / lam_sum
        else:
            lam = np.ones(m) / m
        lam0 = lam0 + alpha * dx[n + 1 + m]
        
        if verbose and it % 5 == 0:
            print(f"      Newton iter {it}: residual={res_norm:.2e}, alpha={alpha:.3f}")
    
    c_final = fast_convolve(a)
    t_final = float(np.max(c_final[active_set]))
    return a, t_final


def refine_active_set(a, solver, max_rounds=20, verbose=False):
    """
    Iteratively refine the active set:
    1. Find active set from current solution
    2. Solve QCQP/Newton with that active set
    3. Check if any inactive constraint is violated
    4. Update active set and repeat
    """
    n = len(a)
    c = fast_convolve(a)
    max_c = c.max()
    
    for round_num in range(max_rounds):
        # Identify active set: all k where c[k] is within 1e-8 of max
        threshold = max_c - max(1e-12, 1e-8 * max_c)
        active = np.where(c >= threshold)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        
        # Cap active set size for numerical stability
        if len(active) > 50:
            # Keep the top 50
            order = np.argsort(c[active])[::-1][:50]
            active = active[order]
        
        if verbose:
            print(f"    Active set round {round_num}: {len(active)} constraints, max_c={max_c:.15e}")
        
        # Try Newton first (faster, higher precision for small active sets)
        a_sol, t_sol = newton_kkt_solve(a, active, max_iters=50, verbose=verbose)
        
        if a_sol is None or not np.all(np.isfinite(a_sol)):
            # Fallback to QCQP solver
            a_sol, t_sol = solve_active_qcqp(a, active, solver, verbose=verbose)
        
        if a_sol is None:
            break
        
        # Check for violated constraints
        c_new = fast_convolve(a_sol)
        max_new = c_new.max()
        new_max_idx = np.argmax(c_new)
        
        # Check if any currently inactive constraint is violated
        inactive = np.setdiff1d(np.arange(2 * n - 1), active)
        if len(inactive) > 0:
            viol = c_new[inactive] > max_new + 1e-12
            if np.any(viol):
                # Add violated constraints
                viol_idx = inactive[viol]
                active = np.unique(np.concatenate([active, viol_idx]))
                a = a_sol
                c = c_new
                max_c = max_new
                if verbose:
                    print(f"    Added {len(viol_idx)} violated constraints")
                continue
        
        # Check if solution improved
        score_new = local_score(a_sol)
        score_old = local_score(a)
        if score_new < score_old - 1e-15:
            a = a_sol
            c = c_new
            max_c = max_new
        else:
            # No improvement, check stability
            if np.max(np.abs(c_new[active] - max_new)) < 1e-12:
                if verbose:
                    print(f"    Active set stable at round {round_num}")
                break
