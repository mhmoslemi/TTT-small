import numpy as np

from scipy.fft import rfft, irfft, fft, ifft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_corr_fft(p, a):
    """
    Compute G[i] = sum_k p[k] * a[k-i] (with a[j]=0 for j not in [0,n-1])
    for i = 0, ..., n-1.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    # p has length 2n-1
    L = 1 << ((len(p) + n - 1) - 1).bit_length()
    P = fft(p, L)
    A = fft(a, L)
    C = ifft(P * np.conj(A))
    return np.real(C[:n])


def lse_objective_and_grad(z, n, T, deadline):
    """
    Compute LSE-smoothed objective and gradient w.r.t. z (log-domain).
    a = softmax(z), objective = 2*n*T*log(sum(exp(conv(a,a)/T)))
    """
    if time.monotonic() > deadline:
        return 1e15, np.zeros(n)
    
    # Softmax
    z_shift = z - z.max()
    exp_z = np.exp(z_shift)
    a = exp_z / exp_z.sum()
    
    # Convolution
    b = fast_convolve(a)
    b_max = b.max()
    
    # LSE objective
    scaled = (b - b_max) / T
    exp_scaled = np.exp(scaled)
    log_sum = b_max + T * np.log(exp_scaled.sum() + 1e-300)
    obj = 2.0 * n * log_sum
    
    # Weights
    p = exp_scaled / exp_scaled.sum()
    
    # Gradient of LSE w.r.t. a: 2 * cross_corr(p, a)
    G = cross_corr_fft(p, a)
    grad_a = 2.0 * G
    
    # Convert to gradient w.r.t. z (Jacobian of softmax)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return obj, grad_z


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def multiplicative_weights_fast(a0, time_limit, n_steps=500000, T=1.0):
    """
    Fast multiplicative weights / mirror descent with LSE smoothing.
    Uses FFT-based gradient computation.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    a = a / a.sum()
    z = z_from_a(a)
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    eta = 0.05
    momentum = np.zeros(n)
    mu = 0.9
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        obj, grad_z = lse_objective_and_grad(z, n, T, t0 + time_limit)
        
        # Nesterov momentum
        momentum = mu * momentum + (1 - mu) * grad_z
        z_new = z - eta * momentum
        
        a_new = softmax_from_z(z_new)
        score = 2.0 * n * fast_convolve(a_new).max()
        
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            z = z_new
            eta = min(eta * 1.02, 0.5)
        else:
            eta *= 0.85
            if eta < 1e-6:
                break
        
        if step % 5000 == 0 and step > 0:
            print(f"      MW: {time.monotonic()-t0:.0f}s, score={best_score:.8f}, eta={eta:.6f}")
    
    return best_a, best_score

    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    if T_stages is None:

def build_conv_matrix_sparse(a):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    rows = np.zeros(n * n, dtype=np.int32)
    cols = np.zeros(n * n, dtype=np.int32)
    vals = np.zeros(n * n, dtype=np.float64)
    count = 0
    for j in range(n):
        for k in range(j, min(j + n, size)):
            rows[count] = k
            cols[count] = j
            vals[count] = a[k - j]
            count += 1
    return csr_matrix((vals[:count], (rows[:count], cols[:count])), shape=(size, n))


def get_lp_solver():
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver, T_mat=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    size = 2 * n - 1
    if T_mat is None:
        T_mat = build_conv_matrix_sparse(a_fixed)
    
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solvers_to_try = [solver, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    for s in solvers_to_try:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_sum = b_val.sum()
                if s_sum < 1e-15:
                    return None
                return b_val / s_sum
        except Exception:
            continue
    return None


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=200000):
    """Nesterov LP alternating with sparse matrices."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(12345)
    T_mat = build_conv_matrix_sparse(y)
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new = solve_lp_best_response(y, solver, T_mat)
        if b_new is None:
            continue
        
        alpha = 2.0 / (it + 2)
        y_old = y.copy()
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = y_old
        else:
            y = y / sy
        
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        T_mat = build_conv_matrix_sparse(y)
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if (it + 1) % 50 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            if stall_count >= 8:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 15)
                phase = rng.uniform(0, 2 * np.pi)
                noise = 0.03 * np.cos(freq * x + phase)
                y = np.maximum(y + noise, 0)
                y = y / y.sum()
                T_mat = build_conv_matrix_sparse(y)
                stall_count = 0
            if (it + 1) % 200 == 0:
                print(f"      LP iter={it+1} ({time.monotonic()-t0:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score


def spectral_init(n, rng, n_iters=30):
    """Flat-spectrum initialization via iterative projection."""
    L = 2 * n - 1
    # Start with random non-negative
    a = np.maximum(rng.standard_normal(n), 0) + 0.1
    a /= a.sum()
    
    for it in range(n_iters):
        # Compute DFT of zero-padded a
        A = fft(a, L)
        mag = np.abs(A)
        target_mag = np.mean(mag)
        # Flatten magnitude
        A_flat = A * (target_mag / np.maximum(mag, 1e-10))
        # Inverse DFT
        a_new = np.real(ifft(A_flat, L))[:n]
        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        # Check convergence
        if np.max(np.abs(a_new - a)) < 1e-10:
            a = a_new
            break
        a = a_new
    return a


def structured_init(n, rng):

    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x + 0.5)
        a = np.maximum(a, 0) + 0.01
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.05
    else:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2) + 0.05
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Fast local evaluation (assumes a >= 0, sum > 0)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def perturb_sequence(a, rng, strength=0.02):
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq1 = rng.integers(1, 12)
    freq2 = rng.integers(2, 20)
    p1 = rng.uniform(0, 2 * np.pi)
    p2 = rng.uniform(0, 2 * np.pi)
    pert = strength * (np.cos(freq1 * x + p1) + 0.5 * np.sin(freq2 * x + p2))
    a_pert = np.maximum(a + pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses FFT-based LSE L-BFGS, fast MW, LP alternating, spectral init,
    and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}, budget: {budget_s:.0f}s")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = spectral_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Init (spectral): n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: FFT-based LSE L-BFGS on parent ===
    print(f"\n=== Phase 1: LSE L-BFGS n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.20, 350.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        T_stages_1 = [30.0, 15.0, 10.0, 7.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
        a_lse = lbgfs_lse_refine(best_seq, t1, T_stages=T_stages_1, maxiter_per_stage=200)
        try_update(a_lse, "lse_1")

    # === Phase 2: Fast MW ===
    print(f"\n=== Phase 2: MW n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.08, 100.0)
    if t2 > 15:
        a_mw, _ = multiplicative_weights_fast(best_seq, t2, n_steps=200000, T=0.5)
        try_update(a_mw, "mw_1")

    # === Phase 3: LP Alternating ===
    print(f"\n=== Phase 3: LP n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.25, 500.0)
    if t3 > 60:
        print(f"  Budget: {t3:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t3, solver, n_max_iters=500000)
        try_update(a_lp, "lp_1")

    # === Phase 4: Multi-length with spectral init ===
    print(f"\n=== Phase 4: Multi-length (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    lengths_to_try = [1000, 1500, 2000]
    for ln in lengths_to_try:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        if len(best_seq) == ln:
            continue
        t_ln = min(rem * 0.15, 200.0)
        if t_ln < 40:
            continue
        print(f"  Trying n={ln}, budget={t_ln:.0f}s")
        # Spectral init
        a_spec = spectral_init(ln, rng, n_iters=50)
        sc_spec = evaluate_local(a_spec)
        print(f"    Spectral init score: {sc_spec:.8f}")
        # Also try interpolated parent
        a_interp = interpolate_sequence(best_seq, ln)
        sc_interp = evaluate_local(a_interp)
        print(f"    Interp score: {sc_interp:.8f}")
        # Use better one
        a_start = a_spec if sc_spec < sc_interp else a_interp
        # LSE L-BFGS
        T_stages_4 = [10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.1, 0.05, 0.01]
        a_opt = lbgfs_lse_refine(a_start, t_ln * 0.6, T_stages=T_stages_4, maxiter_per_stage=150)
        try_update(a_opt, f"lse_n{ln}")
        # LP refinement
        rem = deadline - time.monotonic()
        t_lp = min(rem * 0.3, t_ln * 0.35)
        if t_lp > 30:
            a_lp2, _ = nesterov_lp_refine(a_opt, t_lp, solver, n_max_iters=100000)
            try_update(a_lp2, f"lp_n{ln}")

    # === Phase 5: Perturbation + re-optimize ===
    print(f"\n=== Phase 5: Perturbation (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        for trial in range(5):
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            a_pert = perturb_sequence(best_seq, rng, strength=0.01 + 0.01 * trial)
            t_sub = min(rem * 0.2, 80.0)
            if t_sub < 20:
                continue
            a_opt, _ = multiplicative_weights_fast(a_pert, t_sub, n_steps=100000, T=0.3)
            try_update(a_opt, f"pert_mw_{trial}")
            # Quick LSE polish
            rem = deadline - time.monotonic()
            if rem > 30:
                a_pol = lbgfs_lse_refine(a_opt, min(rem * 0.3, 50.0),
                                         T_stages=[1.0, 0.5, 0.2, 0.1, 0.05, 0.01],
                                         maxiter_per_stage=100)
                try_update(a_pol, f"pert_lse_{trial}")

    # === Phase 6: Final LP polish ===
    print(f"\n=== Phase 6: Final LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t6 = min(rem * 0.25, 200.0)
        a_fin_lp, _ = nesterov_lp_refine(best_seq, t6, solver, n_max_iters=300000)
        try_update(a_fin_lp, "final_lp")

    # === Phase 7: Final ultra-fine LSE ===
    print(f"\n=== Phase 7: Final LSE ultra-fine (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t7 = min(rem * 0.3, 150.0)
        T_fine = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
        a_fin = lbgfs_lse_refine(best_seq, t7, T_stages=T_fine, maxiter_per_stage=300)
        try_update(a_fin, "final_lse")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} n={len(best_seq)}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
