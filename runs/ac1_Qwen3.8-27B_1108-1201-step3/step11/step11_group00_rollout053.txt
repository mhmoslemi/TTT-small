def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_mw_gradient(a, w):
    """
    Compute gradient g[i] = 2 * sum_j w[i+j] * a[j] using FFT.
    w has length 2n-1, a has length n, result has length n.

    g = 2.0 * conv[2*n-2 : 2*n-2-n : -1]
