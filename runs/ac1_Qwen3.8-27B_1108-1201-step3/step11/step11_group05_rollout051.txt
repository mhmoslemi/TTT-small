def mirror_descent_step(a, sg, eta):
    """Apply mirror descent (multiplicative weights) step.
    a_new = a * exp(-eta * sg), normalized."""
    # Scale sg to be in a reasonable range
    sg_max = np.abs(sg).max()
    if sg_max < 1e-30:
        return a.copy()
    sg_scaled = sg / sg_max
    log_ratio = -eta * sg_scaled
    log_ratio -= log_ratio.max()  # numerical stability
    a_new = a * np.exp(log_ratio)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    return a_new / s


def adam_step(param, grad, state, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer step. Returns updated param and state."""
    m, v, t = state
    t = t + 1
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad ** 2
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    param_new = param - lr * m_hat / (np.sqrt(v_hat) + eps)
    new_state = (m, v, t)
    return param_new, new_state

            w = exp_c / exp_c.sum()
        
        # Gradient of LogSumExp w.r.t. a:
        # d/d a[j] of sum_k w[k]*c[k] where c[k] = sum_i a[i]*a[k-i]
        # = sum_k w[k] * (2 * a[k-j]) approximately (for the "cross" terms)
        # More precisely: = sum_k w[k] * (a[k-j] + a[j-k]) for valid indices
        
        # FFT-based computation:
        # The gradient is the "filtered" version of a
        fft_size = 1 << (2 * n - 1).bit_length()
        W = rfft(w[:n], fft_size)  # weight for first n lags... 
        
        # Actually, let's compute the gradient directly for efficiency
        # grad_a[j] = sum_k w[k] * (a[k-j] + a[j-k])
        # = (w * a)_correlation at lag j (from both sides)
        # This is 2 * correlate(w, a)[j] approximately
        
        # Use FFT: correlate(w_full, a) where w_full is the full (2n-1) weight
        W_full = rfft(w, fft_size)
        A_full = rfft(np.concatenate([a, np.zeros(n - 1)]), fft_size)
        corr = irfft(W_full * A_full, fft_size)
        # corr[i] = sum_k w[k] * a_ext[k-i] where a_ext is zero-padded
        # We want grad[j] = sum_k w[k] * a[k-j] (term 1) + sum_k w[k] * a[j-k] (term 2)
        # Term 1: (w * a) at lag j = corruncated correlation
        # This is getting complex, let me just use the direct method for top peaks
        
        # Direct method with weighted peaks:
        grad_a = np.zeros(n)
        top_k = min(10, n)
        idx_top = np.argpartition(c, -top_k)[-top_k:]
        w_top = c[idx_top]
        w_top = w_top / w_top.sum()
        
        for k, wi in zip(idx_top, w_top):
            k = int(k)
            lo1 = max(0, k - n + 1)
            hi1 = min(n - 1, k)
            if lo1 <= hi1:
                grad_a[lo1:hi1 + 1] += 2.0 * wi * a[k - hi1:k - lo1 + 1][::-1]
            if k < n:
                grad_a[k:n] += 2.0 * wi * a[0:n - k]
        
        # Convert to z-space gradient
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * grad_z ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr_eff = lr / (1 + 0.01 * t / 1000)
        z = z - lr_eff * m_hat / (np.sqrt(v_hat) + eps)
        z -= z.mean()
        
        step += 1
        
        # Track best
        a_cur = np.exp(z - z.max())
        a_cur = a_cur / a_cur.sum()
        sc = 2.0 * n * fast_convolve(a_cur).max()
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
        
        if verbose and time.monotnostic() - last_print > 10 if hasattr(time, 'monotnsic') else time.monotonic() - last_print > 10:
            last_print = time.monotonic()
            elapsed = time.monotonic() - t0
            print(f"    Adam {elapsed:.0f}s: best={best_score:.8f} T={T:.4f} step={step}")
    
    return best_a, best_score


def make_initialization(n, init_type, rng):
    """Create diverse initializations."""
    if init_type == 0:
        # Parent-based or uniform
        a = np.ones(n) / n
    elif init_type == 1:
        # Linear ramp (increasing)
        a = np.linspace(0.1, 1.0, n)
    elif init_type == 2:
        # Exponential (increasing)
        a = np.exp(np.linspace(0, 2, n))
    elif init_type == 3:
        # Power law
        p = rng.uniform(0.5, 2.0)
        a = (np.arange(1, n + 1) ** p)
    elif init_type == 4:
        # Asymmetric cosine
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x * rng.uniform(0.3, 0.7)) ** 2
        a = np.maximum(a, 0.01)
    elif init_type == 5:
        # Two bumps (asymmetric)
        center1 = rng.uniform(0.2, 0.4)
        center2 = rng.uniform(0.6, 0.8)
        width1 = rng.uniform(0.05, 0.15)
        width2 = rng.uniform(0.05, 0.15)
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - center1) / width1) ** 2) + 0.5 * np.exp(-((x - center2) / width2) ** 2)
    elif init_type == 6:
        # Oscillatory (asymmetric envelope)
        x = np.linspace(0, 1, n)
        freq = rng.uniform(3, 15)
        envelope = x ** rng.uniform(0.5, 2.0)
        a = envelope * (0.5 + 0.5 * np.cos(2 * np.pi * freq * x))
        a = np.maximum(a, 0.01)
    elif init_type == 7:
        # Random with smooth structure
        a = np.ones(n)
        for _ in range(rng.integers(2, 6)):
            center = rng.uniform(0, 1)
            width = rng.uniform(0.05, 0.3)
            amp = rng.uniform(-0.5, 0.5)
            x = np.linspace(0, 1, n)
            a += amp * np.exp(-((x - center) / width) ** 2)
        a = np.maximum(a, 0.01)
    elif init_type == 8:
        # Sawtooth (strongly asymmetric)
        a = np.linspace(1.0, 0.01, n)
    elif init_type == 9:
        # Optimal control: quadratic
        x = np.linspace(0, 1, n)
        a = 1.0 - x ** rng.uniform(1.0, 3.0)
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n) / n
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def quick_optimize(a0, time_budget, n_peaks=5, eta=0.1):
    """Quick mirror descent optimization."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    
    eta_cur = eta
    no_improve = 0
    
    while time.monotonic() - t0 < time_budget:
        sg, max_c, _ = subgradient_top_peaks(a, n_peaks)
        eta_eff = np.clip(eta_cur * max_c / (best_score / (2 * n) + 1e-15), 1e-4, 3.0)
        a_new = mirror_descent_step(a, sg, eta_eff)
        score_new = 2.0 * n * fast_convolve(a_new).max()
        
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 50:
                eta_cur *= 0.7
                no_improve = 0
                if eta_cur < 1e-5:
                    # Add perturbation
                    a = a + np.random.randn(n) * 0.005
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    eta_cur = eta
    
    return best_a, best_score


def lp_polish(a0, time_budget):
    """LP-based polishing using alternating minimization."""
    try:
        import cvxpy as cp
    except ImportError:
        return a0, score_of(a0)
    
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, score_of(a0)
    a = a / s
    
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    
    # Find a working solver
    solver = None
    for sol in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=sol)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solver = sol
                break
        except Exception:
            continue
    
    if solver is None:
        return best_a, best_score
    
    y = a.copy()
    for it in range(100000):
        if time.monotonic() - t0 > time_budget:
            break
        
        # Solve LP: minimize max_k conv(y, b)[k] subject to b>=0, sum(b)=1
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        
        # Build constraint: conv(y, b) <= M
        # conv(y,b)[k] = sum_i y[i]*b[k-i]
        # We need this <= M for all k
        # This is a set of linear inequalities
        
        # For efficiency, only enforce on a subset of k values (the ones near current peaks)
        c_ab = fast_convolve2(y, a)
        peak_threshold = c_ab.max() * 0.9
        active_k = np.where(c_ab >= peak_threshold)[0]
        # Always include center
        active_k = np.unique(np.concatenate([active_k, [n - 1]]))
        
        constraints = [cp.sum(b_var) == 1]
        for k in active_k[:50]:  # Limit to 50 constraints for speed
            k = int(k)
            # conv(y, b)[k] = sum_i y[i] * b[k-i]
            # Valid i: 0 <= i < n and 0 <= k-i < n
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                row = np.zeros(n)
                row[lo:hi + 1] = y[hi:lo - 1:-1] if lo < hi else y[k:k+1]
                # Actually: sum_i y[i]*b[k-i], for i in [lo, hi], k-i is in [k-hi, k-lo]
                # We want: sum_{i=lo}^{hi} y[i] * b[k-i]
                # Let j = k-i, so i = k-j, j from k-hi to k-lo
                # sum_{j=k-hi}^{k-lo} y[k-j] * b[j]
                # So the coefficient of b[j] is y[k-j] for j in [k-hi, k-lo]
                # j ranges from k-hi to k-lo, i.e., from max(0,k-hi) to min(n-1,k-lo)
                j_lo = max(0, k - hi)
                j_hi = min(n - 1, k - lo)
                if j_lo <= j_hi:
                    coeff = y[k - j_hi:k - j_lo + 1][::-1]
                    constraints.append(row @ b_var <= M_var)
                    # Wait, I need to construct the row properly
                    # row[j] = y[k-j] for j in [j_lo, j_hi]
                    row2 = np.zeros(n)
                    for jj in range(j_lo, j_hi + 1):
                        ii = k - jj
                        if 0 <= ii < n:
                            row2[jj] = y[ii]
                    constraints.append(row2 @ b_var <= M_var)
        
        # Remove duplicate constraints (we may have added empty ones)
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                sb = b_val.sum()
                if sb > 1e-15:
                    b_val = b_val / sb
                    # Nesterov update
                    alpha = 2.0 / (it + 2)
                    y = (1 - alpha) * y + alpha * b_val
                    y = np.maximum(y, 0)
                    sy = y.sum()
                    if sy > 1e-15:
                        y = y / sy
                    a = ((it + 1) * a + alpha * (it + 2) * b_val) / (it + 1 + alpha * (it + 2))
                    a = np.maximum(a, 0)
                    sa = a.sum()
                    if sa > 1e-15:
                        a = a / sa
                    
                    sa_score = score_of(a)
                    sy_score = score_of(y)
                    if sa_score < best_score:
                        best_score = sa_score
                        best_a = a.copy()
                    if sy_score < best_score:
                        best_score = sy_score
                        best_a = y.copy()
        except Exception:
            continue
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):

    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses fast mirror descent, Adam in log-space, multi-n exploration,
    and diverse asymmetric initializations.
    """

    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    print(f"Budget: {budget_s:.0f}s")
    
    # Try to load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = score_of(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        n_init = 1200
        best_seq = make_initialization(n_init, 3, rng)
        best_score = score_of(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")
    
    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = score_of(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}) ***")
    
    # === Phase 1: Quick scan over n values and initializations ===
    print(f"\n=== Phase 1: Quick Scan (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    
    n_values = [1200, 1500, 2000, 2500, 3000]
    if parent is not None:
        n_values.insert(0, len(parent))
    n_values = sorted(set(n_values))
    
    init_types = [1, 2, 3, 4, 5, 6, 7, 8, 9, 0]

    quick_budget_per = min(15.0, rem * 0.3 / (len(n_values) * 5))

        if parent is not None:

        # Various structured inits
        for it in [1, 3, 5, 6]:

            rem = deadline - time.monotonic()
            if rem < 20:
                break
            t_sub = min(quick_budget_per, rem * 0.4)
            if t_sub < 5:
                continue
            
            a_opt, sc = quick_optimize(a_init, t_sub, n_peaks=5, eta=0.1)
            scan_results.append((sc, n_val, a_opt, ii))
    
    # Sort by score
    scan_results.sort(key=lambda x: x[0])
    print(f"  Scan complete. Top results:")
    for sc, nv, _, ii in scan_results[:5]:
        print(f"    n={nv}, init={ii}: score={sc:.8f}")
    
    # Update best from scan
    for sc, nv, a_opt, ii in scan_results:
        if sc < best_score:

            best_seq = a_opt.copy()
            print(f"  *** Best from scan: {sc:.8f} (n={nv}) ***")

    # === Phase 2: Deep optimization of top candidates ===
    print(f"\n=== Phase 2: Deep Optimization (best={best_score:.8f}) ===")

        local_update(a_lp, "lp_polish")
    
    # === Phase 4: Perturbation and re-optimization ===
    print(f"\n=== Phase 4: Perturbation Escape (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    
    if rem > 80:
        n_best = len(best_seq)
        for trial in range(5):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            
            # Different perturbation strategies
            if trial == 0:

                a_pert = best_seq + np.random.randn(n_best) * 0.005
            elif trial == 1:

            elif trial == 3:

                third = n_best // 3
                a_pert = best_seq.copy()
                a_pert[:third] *= 1.1
                a_pert[2*third:] *= 0.9
            else:
                # Random mask perturbation
                mask = np.random.binomial(1, 0.1, n_best)
                a_pert = best_seq * (1 + 0.05 * mask)
