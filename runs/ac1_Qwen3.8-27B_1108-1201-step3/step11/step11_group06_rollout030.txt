```python
def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix_sparse(a):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(j, min(j + n, size)):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(size, n))


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver, T_mat):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = [solver, cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_val = b_val.sum()
                if s_val < 1e-15:
                    return None, None
                b_val = b_val / s_val
                return b_val, float(max(0, M_var.value))
        except Exception:
            continue
    return None, None


def evaluate_local(a):
    """Fast local evaluation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T_temp
    exp_terms = np.exp(scaled - scaled.max())
    lse = max_b + T_temp * (scaled.max() + np.log(exp_terms.sum() + 1e-300))
    obj = 2.0 * n * lse
    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z
```

```python
def coordinate_descent_step(a, conv, i):
    """
    Solve the 1D subproblem for coordinate i.
    Returns the optimal value of a[i].
    """
    n = len(a)
    S = a.sum() - a[i]  # sum of all other coordinates
    
    # Determine affected k range: k where 0 <= k-i < n
    k_min = max(0, i)
    k_max = min(2*n-2, i + n - 1)
    
    # For each affected k, determine base[k] and coef[k]
    # g_k(x) = base[k] + coef[k]*x + quad[k]*x^2
    affected_ks = np.arange(k_min, k_max + 1)
    rel_idx = affected_ks - i  # k - i, ranges from max(0,-i) to min(n-1, 2n-2-i)
    
    # coef[k] = 2*a[k-i] if k != 2i, else 0 (quadratic handles it)
    # quad[k] = 1 if k == 2i, else 0
    # base[k] = conv[k] - coef[k]*a[i] - quad[k]*a[i]^2
    
    a_ikh = a[rel_idx]  # a[k-i] for affected k
    is_quad = (affected_ks == 2 * i)
    
    coef = np.where(is_quad, 0.0, 2.0 * a_ikh)
    quad = np.where(is_quad, 1.0, 0.0)
    base = conv[affected_ks] - coef * a[i] - quad * a[i]**2
    
    # Unaffected k have constant values
    all_ks = np.arange(2*n - 1)
    mask_affected = (all_ks >= k_min) & (all_ks <= k_max)
    base_all = np.zeros(2*n - 1)
    coef_all = np.zeros(2*n - 1)
    quad_all = np.zeros(2*n - 1)
    base_all[mask_affected] = base
    coef_all[mask_affected] = coef
    quad_all[mask_affected] = quad
    base_all[~mask_affected] = conv[~mask_affected]
    
    # Now minimize f(x) = 2n * max_k (base[k] + coef[k]*x + quad[k]*x^2) / (S+x)^2
    # for x >= 0
    
    def f_1d(x):
        vals = base_all + coef_all * x + quad_all * x**2
        return 2.0 * n * vals.max() / (S + x)**2
    
    # Find a reasonable upper bound
    x_current = a[i]
    x_max = max(1.0, 20.0 * S / n)
    
    # Use bounded minimization
    try:
        res = minimize_scalar(f_1d, bounds=(0, x_max), method='bounded', 
                             options={'xatol': 1e-12})
        x_opt = res.x
    except Exception:
        x_opt = x_current
    
    return max(0.0, x_opt)
```

```python
def solve_1d_subproblem(base, coef, quad, C_un, S, n, x_init):
    """Solve min_{x>=0} 2n * max(C_un, max_k (base[k]+coef[k]*x+quad[k]*x^2)) / (S+x)^2"""
    
    def f_and_deriv(x):
        vals = base + coef * x + quad * x**2
        aff_max = vals.max()
        k_star = vals.argmax()
        M = max(aff_max, C_un)
        if M == C_un and C_un > aff_max:
            # The max is achieved by the constant, derivative is 0 from the affected part
            # But the subgradient includes 0 from the constant
            M_prime = 0.0  # not quite right, but for the constant part, derivative is 0
        else:
            M_prime = coef[k_star] + 2 * quad[k_star] * x
            if C_un > aff_max:
                # Both are active? This is a tie case
                # Subgradient is convex hull
                # For simplicity, use the affected part's derivative
                pass
        f = 2.0 * n * M / (S + x)**2
        f_prime = 2.0 * n * (M_prime * (S + x) - 2 * M) / (S + x)**3
        return f, f_prime
    
    # Bisection on f'(x) = 0
    lo = 0.0
    hi = max(1.0, 50.0 * S / n)
    
    # Check if f'(0) >= 0 (minimum is at x=0)
    _, fp_lo = f_and_deriv(lo)
    if fp_lo >= 0:
        return 0.0
    
    # Check if f'(hi) <= 0 (minimum is at hi or beyond)
    _, fp_hi = f_and_deriv(hi)
    if fp_hi <= 0:
        return hi
```

```python
    """Compute conv_w[j] = sum_k w[k]*a[k-j] for j=0..n-1."""
    conv_w = np.zeros(n)
    # Only iterate over k where w[k] is significant
    significant = w > 1e-15
```
