```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP alternating (zero-sum game)
with Nesterov dual averaging, multiple sequence lengths, and gradient polishing.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def solve_lp_best_response(a_fixed, n, solver, T_cached=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    if T_cached is None:
        T_mat = build_conv_matrix(a_fixed)
    else:
        T_mat = T_cached
    
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
        if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            b_val[np.isnan(b_val)] = 0
            s = b_val.sum()
            if s < 1e-15:
                return None, None
            b_val = b_val / s
            return b_val, float(max(0.0, M_var.value))
    except Exception:
        pass
    
    # Fallback to SCS
    try:
        prob.solve(solver=cp.SCS, verbose=False)
        if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s > 1e-15:
                b_val = b_val / s
                return b_val, float(max(0.0, M_var.value))
    except Exception:
        pass
    
    return None, None


def lp_alternating(a0, time_limit, solver, verbose=True):
    """Nesterov dual averaging LP alternating method."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    y = a.copy()  # Dual/mirror variable
    best_a = a.copy()
    best_score = float(2.0 * n * fast_convolve(a).max())
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    
    for it in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP for best response to y
        T_mat = build_conv_matrix(y)
        b_new, M_b = solve_lp_best_response(y, n, solver, T_mat)
        
        if b_new is None:
            time.sleep(0.1)
            continue
        
        # Nesterov dual averaging
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        
        # Primal average
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        # Evaluate
        score_a = 2.0 * n * fast_convolve(a).max()
        score_b = 2.0 * n * fast_convolve(b_new).max()
        
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_b < best_score:
            best_score = score_b
            best_a = b_new.copy()
        # Also check y
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if (it + 1) % 100 == 0:
            if verbose:
                if best_score < prev_best - 1e-12:
                    stall_count = 0
                else:
                    stall_count += 1
                prev_best = best_score
                
                if stall_count >= 30 and it > 500:
                    # Stuck, perturb
                    rng = np.random.default_rng(it)
                    noise = 0.01 * rng.standard_normal(n)
                    y = y + noise
                    y = np.maximum(y, 0)
                    sy = y.sum()
                    if sy > 1e-15:
                        y = y / sy
                    stall_count = 0
                
                if (it + 1) % 1000 == 0:
                    elapsed = time.monotonic() - t0
                    print(f"      LP iter={it+1} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

    a = a / s
    
    best_a = a.copy()
    best_score = float(2.0 * n * fast_convolve(a).max())

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
```

```python
def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

def fast_cross_corr(w, a):
    """Compute cross-correlation: result[i] = sum_k w[k] * a[k-i] for i=0..n-1."""
    n = len(a)

    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)

def build_conv_matrix(a):

        if elapsed > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, n, solver)

        if b_new is None:
            time.sleep(0.05)
            continue

        alpha = 2.0 / (it + 2)

        # Dual update
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        # Primal (Nesterov) average
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        # Evaluate all candidates
        score_a = 2.0 * n * fast_convolve(a).max()
        score_b = 2.0 * n * fast_convolve(b_new).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_b < best_score:
            best_score = score_b
            best_a = b_new.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Convergence / stall detection
        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 30 and it > 2000:
                noise = 0.02 * rng.standard_normal(n)
                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                stall_count = 0

            if verbose and (it + 1) % 2000 == 0:
                print(f"      LP iter={it+1} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def gradient_polish(a0, time_limit):
    """Adam optimization in log-space with LogSumExp smoothing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = float(2.0 * n * fast_convolve(a).max())
    t0 = time.monotonic()

    T_stages = [10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
    lr_base = 0.005

    z = np.log(a + 1e-300)
    m = np.zeros(n)
    v = np.zeros(n)
    step_count = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break

        steps = 3000
        for step in range(steps):
            if time.monotonic() - t0 > time_limit:
                break

            a = np.exp(z - z.max())
            a = a / a.sum()

            b = fast_convolve(a)
            max_b = b.max()
            score = 2.0 * n * max_b

            if score < best_score:
                best_score = score
                best_a = a.copy()

            # LogSumExp weights
            exp_b = np.exp((b - max_b) / T)
            w = exp_b / exp_b.sum()

            # Gradient: g_a[i] = (2/T) * sum_k w[k] * a[k-i]
            g_a = (2.0 / T) * fast_cross_corr(w, a)

            # Convert to z-space
            S = np.dot(g_a, a)
            grad_z = a * (g_a - S)

            # Adam update
            lr = lr_base / (1.0 + step / steps)
            step_count += 1
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * grad_z ** 2
            m_hat = m / (1.0 - beta1 ** step_count)
            v_hat = v / (1.0 - beta2 ** step_count)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            z -= z.mean()

    return best_a, best_score


def structured_init(n, seed=0):
    """Create diverse structured initial sequences."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    choice = rng.integers(0, 8)

    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.2 * np.cos(2 * np.pi * 3 * x) + 0.1 * np.cos(2 * np.pi * 7 * x + 0.3)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = np.ones(n) / n
    elif choice == 5:
        a = 0.6 + 0.2 * np.cos(2 * np.pi * 5 * x + 1.0) + 0.1 * np.cos(2 * np.pi * 11 * x)
        a = np.maximum(a, 0)
    elif choice == 6:
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.7 * np.exp(-3 * (x - 0.7) ** 2)
    else:
        a = 1.0 / (1.0 + 0.5 * np.cos(2 * np.pi * 2 * x))

    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate score locally (fast, no security checks)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def perturb_sequence(a, rng, strength):
    """Create a perturbed version."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 12)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Primary: LP alternating (zero-sum game). Secondary: multi-n + gradient polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Solver: {solver}, Budget: {budget_s:.0f}s")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={parent_score:.8f}")
            best_seq = parent.copy()
            best_score = parent_score
    except Exception:
        pass

    if parent is None:
        best_seq = structured_init(1200, seed)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    def update_best(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}) ***")

    # === Phase 1: LP alternating on parent length (primary) ===
    n_parent = len(best_seq)
    print(f"\n=== Phase 1: LP alternating n={n_parent} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 500.0)
    if t_alloc > 60:
        print(f"  Time: {t_alloc:.0f}s")
        a1, _ = lp_alternating(best_seq, t_alloc, solver)
        update_best(a1, f"lp1_n{n_parent}")

    # === Phase 2: Try n=1400 ===
    print(f"\n=== Phase 2: n=1400 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 200:
        n2 = 1400
        t_alloc = min(rem * 0.20, 300.0)
        init_2 = interpolate_sequence(best_seq, n2)
        init_score = evaluate_local(init_2)
        print(f"  Time: {t_alloc:.0f}s, init score: {init_score:.8f}")
        a2, _ = lp_alternating(init_2, t_alloc, solver)
        update_best(a2, f"lp_n{n2}")

    # === Phase 3: Second LP round on current best ===
    print(f"\n=== Phase 3: LP round 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 150:
        t_alloc = min(rem * 0.25, 350.0)
        print(f"  Time: {t_alloc:.0f}s, n={len(best_seq)}")
        a3, _ = lp_alternating(best_seq, t_alloc, solver)
        update_best(a3, "lp_round2")

    # === Phase 4: Try n=1600 ===
    print(f"\n=== Phase 4: n=1600 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 150:
        n4 = 1600
        t_alloc = min(rem * 0.20, 250.0)
        init_4 = interpolate_sequence(best_seq, n4)
        print(f"  Time: {t_alloc:.0f}s, init score: {evaluate_local(init_4):.8f}")
        a4, _ = lp_alternating(init_4, t_alloc, solver)
        update_best(a4, f"lp_n{n4}")

    # === Phase 5: Perturbation escape + quick LP ===
    print(f"\n=== Phase 5: Perturbation escape (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        n_cur = len(best_seq)
        for trial in range(3):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 30:
                break
            strength = 0.01 + 0.015 * trial
            a_pert = perturb_sequence(best_seq, rng, strength)
            t_sub = min(rem_sub * 0.3, 60.0)
            a_p, _ = lp_alternating(a_pert, t_sub, solver, verbose=False)
            update_best(a_p, f"pert_{trial}")

    # === Phase 6: Gradient polish ===
    print(f"\n=== Phase 6: Gradient polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_alloc = min(rem * 0.30, 200.0)
        print(f"  Time: {t_alloc:.0f}s")
        a6, _ = gradient_polish(best_seq, t_alloc)
        update_best(a6, "grad_polish")

    # === Phase 7: Final LP round ===
    print(f"\n=== Phase 7: Final LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_alloc = min(rem * 0.60, 250.0)
        print(f"  Time: {t_alloc:.0f}s")
        a7, _ = lp_alternating(best_seq, t_alloc, solver)
        update_best(a7, "final_lp")

    # === Phase 8: Last gradient polish ===
    print(f"\n=== Phase 8: Final gradient polish ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_alloc = min(rem * 0.5, 60.0)
        a8, _ = gradient_polish(best_seq, t_alloc)
        update_best(a8, "final_grad")

    # Final
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        best_score = min(best_score, fs)
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")
        # Verify with evaluate_sequence
        test_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        ev = evaluate_sequence(test_seq)
        print(f"evaluate_sequence: {ev:.8f}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200.0

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
```
