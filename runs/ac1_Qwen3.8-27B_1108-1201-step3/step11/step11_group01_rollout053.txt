For the peak shaving, here's the key idea:
- We have a probability vector a (sum = 1)
- b = conv(a, a), let k* be argmax(b)
- We want to find a direction d such that moving a in direction d reduces b[k*]
- The derivative of b[k*] w.r.t. a is: ∂b[k*]/∂a[i] = a[k*-i] (for valid i, i.e., 0 ≤ k*-i ≤ n-1 and 0 ≤ i ≤ n-1)
- So the gradient is a "shifted" version of a
- To reduce b[k*] while keeping sum(a) constant, we project the negative gradient onto the tangent space {d : sum(d) = 0}
- projected_grad[i] = grad[i] - mean(grad)
- Step: a_new = a - eta * projected_grad, then clip to non-negative, then normalize

This is a valid step that should reduce the peak at k* (for small enough eta).

To make it more effective, we can:
- Use multiple peaks (all k where b[k] > 0.9*max(b))
- Use an adaptive eta (line search or backtracking)
- Combine with the LP approach

def peak_shaving_step(a, eta=0.1, threshold=0.95):
    """
    Direct peak-shaving: compute conv(a,a), find peaks,
    take a gradient step in log-domain to reduce peaks.
    Returns new a (normalized to sum=1).
    """
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    
    # Find active peaks
    active = np.where(b >= threshold * max_b)[0]
    
    # Compute subgradient of max(conv(a,a)) w.r.t. a
    # For each active peak at position k, gradient w.r.t. a[i] is a[k-i]
    # Average over active peaks
    grad = np.zeros(n)
    for k in active:
        # a[k-i] is valid when 0 <= k-i <= n-1, i.e., k-n+1 <= i <= k
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        idx = np.arange(i_lo, i_hi + 1)
        grad[idx] += a[k - idx]
    grad /= len(active)
    
    # In log-domain, the gradient of f w.r.t. z = log(a) is:
    # df/dz[i] = a[i] * (df/da[i] - sum_j a[j] * df/da[j])
    # (this is the gradient on the simplex in log-coordinates)
    S = np.dot(grad, a)
    grad_z = a * (grad - S)
    
    # Step in log-domain
    z = z_from_a(a)
    z_new = z - eta * grad_z
    a_new = softmax_from_z(z_new)
    
    return a_new

def nesterov_lp_with_shaving(a0, time_limit, solver=None, max_iters=1000000, shave_every=50, shave_eta=0.3):
    """
    Nesterov-accelerated LP with periodic peak-shaving steps.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0

        if time.monotonic() - t0 > time_limit:
            break

        # Every shave_every iterations, do a peak-shaving step
        if it > 0 and it % shave_every == 0:

            sc_shaved = score_normalized(a_shaved)

            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a_shaved.copy()
            else:
                y = y / sy
            a = a_shaved
            continue

        if b_new is None:
            continue

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0

        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

def lse_objective_grad(z_in, T_val, n):
    """LogSumExp approximation of the objective, with gradient."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / (T_val * len(b))) * log_sum_exp  # normalized by len(b) for stability
    
    # Gradient w.r.t. a
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += T_val * w[k] * a[k - idx]
    grad_a *= 2.0 * n / (T_val * len(b))
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)

        return f.copy()

    if s < 1e-15:

    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def peak_shaving_step(a, eta=0.1, threshold=0.95):
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    active = np.where(b >= threshold * max_b)[0]
    if len(active) == 0:
        active = np.array([np.argmax(b)])
    
    grad = np.zeros(n)
    for k in active:
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        idx = np.arange(i_lo, i_hi + 1)
        grad[idx] += a[k - idx]
    grad /= len(active)
    
    S = np.dot(grad, a)
    grad_z = a * (grad - S)
    
    z = z_from_a(a)
    z_new = z - eta * grad_z
    a_new = softmax_from_z(z_new)
    return a_new


def nesterov_lp_with_shaving(a0, time_limit, solver=None, max_iters=1000000, shave_every=50, shave_eta=0.3):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        if it > 0 and it % shave_every == 0:
            a_shaved = peak_shaving_step(a, eta=shave_eta)
            sc_shaved = score_normalized(a_shaved)
            if sc_shaved < best_score:
                best_score = sc_shaved
                best_a = a_shaved.copy()
            y = 0.5 * y + 0.5 * a_shaved
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a_shaved.copy()
            else:
                y = y / sy
            a = a_shaved
            continue

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 10:
                print(f"    NLP it={it+1} ({elapsed:.0f}s): score={best_score:.8f} M={M_lp:.6f}")
                last_print = elapsed

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += T_val * w[k] * a[k - idx]
    grad_a *= 2.0 * n

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_logsumexp(a0, time_limit, T_start=50.0, T_end=0.02, n_stages=20):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        max_iter = max(10, min(50, int(remaining * 10)))

        # Use a wrapper that captures stage_T and n
        # Since we can't use closures/lambda, we'll inline the logic
        try:
            result = _lbfgs_lse_stage(z, stage_T, n, max_iter, t0, time_limit)
            if result is not None:
                z = result
                a_check = softmax_from_z(z)
                sc = score_normalized(a_check)
                if sc < best_score:
                    best_score = sc
                    best_a = a_check.copy()
                if (si + 1) % 5 == 0:
                    print(f"    LSE stage={si+1}/{n_stages} T={stage_T:.3f}: score={sc:.8f}")
        except Exception:
            pass

    return best_a, best_score


def _lbfgs_lse_stage(z0, stage_T, n, max_iter, t0, time_limit):
    """Single stage of L-BFGS for LSE objective."""
    # We need func and jac that don't use closures
    # Use a class-free approach with a helper that takes all args
    class _FuncWrapper:
        def __init__(self, T, n):
            self.T = T
            self.n = n
        def __call__(self, z_val):
            obj, _ = lse_objective_grad(z_val, self.T, self.n)
            return obj
    class _JacWrapper:
        def __init__(self, T, n):
            self.T = T
            self.n = n
        def __call__(self, z_val):
            _, grad = lse_objective_grad(z_val, self.T, self.n)
            return grad

    func_w = _FuncWrapper(stage_T, n)
    jac_w = _JacWrapper(stage_T, n)

    res = minimize(
        fun=func_w, x0=z0, jac=jac_w, method='L-BFGS-B',
        options={'maxiter': max_iter, 'ftol': 1e-18,
                 'gtol': 1e-14, 'maxfun': max_iter * 100,
                 'disp': False}
    )
    if time.monotonic() - t0 > time_limit:
        return z0
    return res.x


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def try_update_impl(best_seq, best_score, a, label):
    """Try to update best sequence. Returns (new_seq, new_score, improved)."""
    if a is None:
        return best_seq, best_score, False
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score, False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
        return a.copy(), sc, True
    return best_seq, best_score, False
