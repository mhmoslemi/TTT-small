def lse_objective_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, stage_T, n)
            return grad

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-16, 'maxfun': iters_per_stage * 500,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)
            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()
            if verbose and (si + 1) % 3 == 0:
                print(f"    LSE stage={si+1}/{n_stages} T={stage_T:.3f}: score={best_score:.8f}")
        except Exception:
            pass

    return best_a, best_score


def subgradient_polish(a0, time_limit, n_steps=200000, eta_start=0.5, eta_end=0.01, verbose=True):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        active = np.where(b >= max_b * 0.95)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        S = np.dot(g, a)
        grad_z = a * (g - S)

        eta = eta_start * (eta_end / eta_start) ** (step / max(1, n_steps))
        z = z - eta * grad_z
        a = softmax_from_z(z)

        sc = score_normalized(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose and (step + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 10:
                print(f"    SubGrad step={step+1} ({elapsed:.0f}s): score={best_score:.8f}")
                last_print = elapsed

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-2 * x ** 2)
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif choice == 4:
        a = np.ones(n) * (1.0 + 0.2 * np.cos(2 * np.pi * 3 * np.arange(n) / n))
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(3 * x) + 0.15 * np.cos(7 * x)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * (1 + 0.3 * np.cos(4 * np.pi * x))
    else:
        x = np.linspace(0, 1, n)
        a = 0.4 + 0.3 * np.sin(2 * np.pi * 2 * x) + 0.2 * np.cos(2 * np.pi * 5 * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def perturb_sequence(a, rng, strength=0.01):
    n = len(a)
    noise = rng.normal(0, strength, size=n)
    a_perturbed = a + noise
    a_perturbed = np.maximum(a_perturbed, 0)
    s = a_perturbed.sum()
    if s < 1e-15:
        return a.copy()
    return a_perturbed / s


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    time_start = time.monotonic()

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}, budget={budget_s:.0f}s")

    # Get parent or default initialization
    best_seq = None
    best_score = float('inf')
    n_base = 1200

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - time_start
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}) at {elapsed:.0f}s ***")

    # === Phase 1: Long Nesterov LP (40% of budget) ===
    print(f"\n=== Phase 1: Nesterov LP main (score={best_score:.8f}) ===")
    t1 = 0.40 * (budget_s - 30)
    remaining = deadline - time.monotonic()
    t1 = min(t1, remaining * 0.5)
    a_out, sc = nesterov_lp(best_seq, t1, solver=solver, max_iters=2000000, verbose=True)
    try_update(a_out, "nlp_main")
    print(f"  Phase 1 done: best={best_score:.8f}")

    # === Phase 2: LSE L-BFGS polish (10% of budget) ===
    print(f"\n=== Phase 2: LSE L-BFGS polish ===")
    remaining = deadline - time.monotonic()
    t2 = min(0.10 * (budget_s - 30), remaining * 0.3)
    if t2 > 10:
        a_lse, sc_lse = lse_lbgfs(best_seq, t2, T_start=100.0, T_end=0.02, n_stages=20, iters_per_stage=40, verbose=True)
        try_update(a_lse, "lse_polish1")
    print(f"  Phase 2 done: best={best_score:.8f}")

    # === Phase 3: Continue Nesterov LP with tighter tolerance (20% of budget) ===
    print(f"\n=== Phase 3: Nesterov LP refined ===")
    remaining = deadline - time.monotonic()
    t3 = min(0.20 * (budget_s - 30), remaining * 0.4)
    if t3 > 10:
        a_out2, sc2 = nesterov_lp(best_seq, t3, solver=solver, max_iters=2000000, adaptive_tol=False, verbose=True)
        try_update(a_out2, "nlp_refined")
    print(f"  Phase 3 done: best={best_score:.8f}")

    # === Phase 4: LSE L-BFGS with lower temperature (10% of budget) ===
    print(f"\n=== Phase 4: LSE L-BFGS fine ===")
    remaining = deadline - time.monotonic()
    t4 = min(0.10 * (budget_s - 30), remaining * 0.3)
    if t4 > 10:
        a_lse2, sc_lse2 = lse_lbgfs(best_seq, t4, T_start=20.0, T_end=0.01, n_stages=15, iters_per_stage=60, verbose=True)
        try_update(a_lse2, "lse_polish2")
    print(f"  Phase 4 done: best={best_score:.8f}")

    # === Phase 5: Perturbation restarts (15% of budget) ===
    print(f"\n=== Phase 5: Perturbation restarts ===")
    remaining = deadline - time.monotonic()
    t5_total = min(0.15 * (budget_s - 30), remaining * 0.5)
    if t5_total > 30:
        n_restarts = max(3, int(t5_total / 30))
        per_restart = t5_total / n_restarts
        strengths = [0.02, 0.01, 0.005, 0.05, 0.001, 0.03, 0.015, 0.008]
        for ri in range(n_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            per_budget = min(per_restart, remaining * 0.6)
            if per_budget < 5:
                break
            strength = strengths[ri % len(strengths)]
            a_pert = perturb_sequence(best_seq, rng, strength=strength)
            # Short Nesterov LP
            t_lp = per_budget * 0.6
            a_r, sc_r = nesterov_lp(a_pert, t_lp, solver=solver, max_iters=50000, verbose=False)
            try_update(a_r, f"restart_{ri}_s{strength}")
            # Short LSE
            remaining = deadline - time.monotonic()
            if remaining > 5 and a_r is not None:
                t_lse = min(per_budget * 0.3, remaining * 0.4)
                if t_lse > 3:
                    a_r2, _ = lse_lbgfs(a_r, t_lse, T_start=50.0, T_end=0.05, n_stages=8, iters_per_stage=20, verbose=False)
                    try_update(a_r2, f"restart_{ri}_lse")
    print(f"  Phase 5 done: best={best_score:.8f}")

    # === Phase 6: Length exploration (5% of budget) ===
    print(f"\n=== Phase 6: Length exploration ===")
    remaining = deadline - time.monotonic()
    t6_total = min(0.05 * (budget_s - 30), remaining * 0.5)
    if t6_total > 20:
        n_tries = [n_base + 100, n_base - 100, n_base + 200, n_base + 50, n_base - 50]
        per_try = t6_total / len(n_tries)
        for n_try in n_tries:
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            t_try = min(per_try, remaining * 0.5)
            if t_try < 5:
                break
            print(f"  Trying n={n_try}, budget={t_try:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            a_try, _ = nesterov_lp(init_try, t_try * 0.7, solver=solver, max_iters=100000, verbose=False)
            try_update(a_try, f"len_{n_try}")
            if a_try is not None and t_try > 10:
                a_try2, _ = lse_lbgfs(a_try, t_try * 0.25, T_start=30.0, T_end=0.1, n_stages=6, iters_per_stage=20, verbose=False)
                try_update(a_try2, f"len_{n_try}_lse")
    print(f"  Phase 6 done: best={best_score:.8f}")

    # === Phase 7: Final subgradient + LSE polish (5% of budget) ===
    print(f"\n=== Phase 7: Final polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 10:
        t_sg = min(remaining * 0.3, 60.0)
        if t_sg > 5:
            a_sg, _ = subgradient_polish(best_seq, t_sg, n_steps=100000, eta_start=0.3, eta_end=0.005, verbose=False)
            try_update(a_sg, "final_subgrad")

        remaining = deadline - time.monotonic()
        t_final = min(remaining * 0.6, 60.0)
        if t_final > 5:
            a_fin, _ = lse_lbgfs(best_seq, t_final, T_start=10.0, T_end=0.01, n_stages=12, iters_per_stage=40, verbose=True)
            try_update(a_fin, "final_lse")
    print(f"  Phase 7 done: best={best_score:.8f}")

    # === Final verification ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - time_start
        print(f"\nFINAL: score={best_score:.8f} (n={len(best_seq)}), time={elapsed:.0f}s")

    return [float(x) for x in best_seq]
