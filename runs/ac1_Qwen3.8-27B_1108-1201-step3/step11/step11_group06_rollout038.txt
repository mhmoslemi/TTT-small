    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A*A, fft_size)[:size]

def build_conv_matrix_sparse(a):
    n = len(a)
    size = 2*n - 1
    entries = []
    for j in range(n):
        for k in range(max(0, j-n+1), min(j+n, size)):
            entries.append((k, j, a[k-j]))
    rows = [e[0] for e in entries]
    cols = [e[1] for e in entries]
    vals = [e[2] for e in entries]
    return csr_matrix((vals, (rows, cols)), shape=(size, n))

def solve_lp_best_response(a_fixed, solver, T_mat):

    prob = cp.Problem(cp.Minimize(M_var), constraints)

            if prob.status in ('optimal', 'optimal_inaccurate'):

                b_val = np.maximum(b_val, 0)

    for j in range(n):
        for k in range(max(0, j - n + 1), min(j + n, size)):
            rows[count] = k
            cols[count] = j
            vals[count] = a[k - j]
            count += 1
    return csr_matrix((vals[:count], (rows[:count], cols[:count])), shape=(size, n))


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver, T_mat):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = [solver, cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                ssum = b_val.sum()
                if ssum < 1e-15:
                    return None, None
                b_val /= ssum
                return b_val, float(max(0, M_var.value))
        except Exception:
            continue
    return None, None


def evaluate_local(a):
    """Fast local evaluation (assumes a >= 0, sum > 0)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def multiplicative_weights(a0, time_limit, n_steps=500000, eta0=0.05):
    """Fast multiplicative weights with adaptive backtracking step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(n) / n
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * (1.0 - 2e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        w_active = b[active] / b[active].sum()
        g = np.zeros(n)
        for w, k in zip(w_active, active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += w * a[k - idx]
        log_mult = -eta * g
        log_mult -= log_mult.max()
        a_new = a * np.exp(log_mult)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        a_new = a_new / s_new if s_new > 1e-15 else a.copy()
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.008, 0.4)
        else:
            eta *= 0.85
            if eta < 1e-10:
                break
    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=500000):
    """Nesterov dual averaging LP alternating with sparse matrices."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(n) / n
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(777)
    lp_times = []
    T_mat = build_conv_matrix_sparse(y)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver, T_mat)
        lp_elapsed = time.monotonic() - lp_t0
        lp_times.append(lp_elapsed)
        if b_new is None:
            continue
        if len(lp_times) > 20:
            avg_lp = np.mean(lp_times[-20:])
            if avg_lp > 10.0:
                break
        alpha = 2.0 / (it + 2)
        y_old = y.copy()
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = y_old
            sy = y_old.sum()
        y = y / sy
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        T_mat = build_conv_matrix_sparse(y)
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        if (it + 1) % 50 == 0:
            if best_score < prev_best - 1e-12:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            if stall_count >= 8:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 10)
                phase = rng.uniform(0, 2 * np.pi)
                noise = 0.015 * np.cos(freq * x + phase)
                y = np.maximum(y + noise, 0)
                y = y / y.sum()
                T_mat = build_conv_matrix_sparse(y)
                stall_count = 0
            if (it + 1) % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.10f}")
    return best_a, best_score


def coordinate_descent(a0, time_limit, verbose=True):
    """
    Coordinate descent to minimize max_k conv(a,a)[k] with a >= 0.
    Uses in-place O(n) convolution updates per coordinate change.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    b = fast_convolve(a)
    best_score = 2.0 * n * b.max() / (a.sum() ** 2)
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    n_total = 2 * n - 1

    for it in range(20000000):
        if time.monotonic() - t0 > time_limit:
            break

        i = rng.integers(0, n)
        old_ai = a[i]
        if old_ai < 1e-18:
            # Try setting to a small positive value
            candidates = np.array([1e-6 / n, 2e-6 / n, 5e-6 / n, 1e-5 / n])
            a[i] = candidates[0]
            # Quick eval
            b = fast_convolve(a)
            sc = 2.0 * n * b.max()
            a[i] = 0.0
            if sc < best_score * a.sum() ** 2:
                best_score = sc
                best_a = a.copy()
            continue

        k_start = i
        k_end = min(n_total - 1, i + n - 1)
        span = k_end - k_start + 1

        # Compute R_linear[k] = b[k] - 2*old_ai*a[k-i] for k in [k_start, k_end], k != 2i
        # And R_quad = b[2i] - old_ai^2
        R_lin = b[k_start:k_end + 1].copy()
        j_off = np.arange(k_start, k_end + 1) - i
        # j_off is in [0, n-1] for all valid k
        mask_quad = (np.arange(k_start, k_end + 1) == 2 * i)
        R_lin[~mask_quad] -= 2.0 * old_ai * a[j_off[~mask_quad]]
        R_quad = b[2 * i] - old_ai * old_ai

        # c_k = 2*a[k-i] for k != 2i
        c_vals = np.zeros(span)
        c_vals[~mask_quad] = 2.0 * a[j_off[~mask_quad]]

        # M_const = max of b[k] for k outside [k_start, k_end]
        if k_start == 0 and k_end == n_total - 1:
            M_const = 0.0
        else:
            mask_out = np.ones(n_total, dtype=bool)
            mask_out[k_start:k_end + 1] = False
            M_const = b[mask_out].max() if mask_out.any() else 0.0

        # Define h(x) = max(M_const, R_quad + x^2, max(R_lin + c_vals * x))
        def h_func(x):
            lin_max = np.max(R_lin + c_vals * x)
            quad_val = R_quad + x * x
            mc = M_const
            lm = lin_max
            qv = quad_val
            return max(mc, lm, qv)

        # Upper bound for x
        h0 = h_func(0.0)
        x_max = max(old_ai * 3.0, 3.0 / n, np.sqrt(max(0.0, h0 - R_quad)) + 0.5 / n)

        # Golden section search
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo = 0.0
        hi = x_max
        c_pt = hi - gr * (hi - lo)
        d_pt = lo + gr * (hi - lo)
        hc = h_func(c_pt)
        hd = h_func(d_pt)
        for _ in range(50):
            if hc < hd:
                hi = d_pt
                d_pt = c_pt
                hd = hc
                c_pt = hi - gr * (hi - lo)
                hc = h_func(c_pt)
            else:
                lo = c_pt
                c_pt = d_pt
                hc = hd
                d_pt = lo + gr * (hi - lo)
                hd = h_func(d_pt)

        new_ai = max(0.0, (lo + hi) / 2.0)

        # Update a and b in-place
        diff = new_ai - old_ai
        if abs(diff) > 1e-18:
            a[i] = new_ai
            # Update b in-place
            for offset in range(span):
                k = k_start + offset
                if k == 2 * i:
                    b[k] += new_ai * new_ai - old_ai * old_ai
                else:
                    j_other = k - i
                    b[k] += 2.0 * diff * a[j_other]

        # Periodic full recompute to prevent drift
        if it % 500 == 499:
            b = fast_convolve(a)

        # Track best
        current_max = b.max()
        sa = a.sum()
        sc = 2.0 * n * current_max / (sa * sa) if sa > 1e-15 else float('inf')
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose and it % 50000 == 0 and it > 0:
            elapsed = time.monotonic() - t0
            print(f"    CD it={it} ({elapsed:.0f}s): score={best_score:.10f} max_b={current_max:.8e}")

    return best_a, best_score


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 5 * x + 0.5)
        a = np.maximum(a, 0)
    elif choice == 7:
        a = 1.0 / (1.0 + 3.0 * np.abs(x - 0.5))
    elif choice == 8:
        a = (1 + np.cos(2 * np.pi * x)) / 2.0
    else:
        a = np.sin(np.pi * x) ** 1.5
    a = np.maximum(a, 0)
    sval = a.sum()
    if sval < 1e-15:
        return np.ones(n) / n
    return a / sval


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def perturb_sequence(a, rng, strength=0.03):
    """Create a perturbed version with structured frequency components."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq1 = rng.integers(1, 8)
    freq2 = rng.integers(2, 12)
    p1 = rng.uniform(0, 2 * np.pi)
    p2 = rng.uniform(0, 2 * np.pi)
    pert = strength * (np.cos(freq1 * x + p1) + 0.5 * np.sin(freq2 * x + p2))
    a_pert = np.maximum(a + pert, 0)
    sval = a_pert.sum()
    if sval < 1e-15:
        return a.copy()
    return a_pert / sval


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses: MW init -> LP alternating -> Coordinate descent -> Multi-length -> Polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}, budget: {budget_s:.0f}s")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Init: n={len(best_seq)}, score={best_score:.10f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        sval = a.sum()
        if sval < 0.01:
            return
        a = a / sval
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # === Phase 1: Quick MW ===
    print(f"\n=== Phase 1: MW n={n_best} (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.02, 30.0)
    if t1 > 5:
        a_mw, _ = multiplicative_weights(best_seq, t1, n_steps=200000, eta0=0.05)
        try_update(a_mw, "mw_1")

    # === Phase 2: LP alternating (main workhorse) ===
    print(f"\n=== Phase 2: LP n={n_best} (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.30, 500.0)
    if t2 > 60:
        print(f"  Budget: {t2:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t2, solver, n_max_iters=500000)
        try_update(a_lp, "lp_1")

    # === Phase 3: Coordinate descent (key refinement) ===
    print(f"\n=== Phase 3: CD n={n_best} (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.30, 400.0)
    if t3 > 30:
        print(f"  Budget: {t3:.0f}s")
        a_cd, _ = coordinate_descent(best_seq, t3, verbose=True)
        try_update(a_cd, "cd_1")

    # === Phase 4: Second LP pass from CD result ===
    print(f"\n=== Phase 4: LP from CD (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t4 = min(rem * 0.15, 200.0)
        a_lp2, _ = nesterov_lp_refine(best_seq, t4, solver, n_max_iters=200000)
        try_update(a_lp2, "lp_2")
        # Quick CD after second LP
        rem = deadline - time.monotonic()
        if rem > 20:
            t4b = min(rem * 0.3, 60.0)
            a_cd2, _ = coordinate_descent(best_seq, t4b, verbose=False)
            try_update(a_cd2, "cd_2")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    lengths_to_try = []
    if n_best != 1500:
        lengths_to_try.append(1500)
    if n_best != 1000:
        lengths_to_try.append(1000)
    if n_best != 1800:
        lengths_to_try.append(1800)

    for ln in lengths_to_try:
        rem = deadline - time.monotonic()
        if rem < 120:
            break
        t_ln = min(rem * 0.12, 150.0)
        if t_ln < 40:
            continue
        print(f"  Trying n={ln}, budget={t_ln:.0f}s")
        init_ln = interpolate_sequence(best_seq, ln)
        # Quick MW
        t_mw = t_ln * 0.1
        a_ln_mw, _ = multiplicative_weights(init_ln, t_mw, n_steps=50000, eta0=0.05)
        try_update(a_ln_mw, f"mw_n{ln}")
        # LP
        rem = deadline - time.monotonic()
        t_lp = min(rem * 0.5, t_ln * 0.5)
        if t_lp > 30:
            a_ln_lp, _ = nesterov_lp_refine(a_ln_mw, t_lp, solver, n_max_iters=100000)
            try_update(a_ln_lp, f"lp_n{ln}")
        # CD
        rem = deadline - time.monotonic()
        t_cd = min(rem * 0.4, t_ln * 0.4)
        if t_cd > 15:
            a_ln_cd, _ = coordinate_descent(a_ln_lp if a_ln_lp is not None else a_ln_mw, t_cd, verbose=False)
            try_update(a_ln_cd, f"cd_n{ln}")

    # === Phase 6: Perturbation + re-optimize ===
    print(f"\n=== Phase 6: Perturbation (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        for trial in range(3):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            a_pert = perturb_sequence(best_seq, rng, strength=0.01 + 0.005 * trial)
            t_sub = min(rem * 0.3, 40.0)
            a_opt, _ = coordinate_descent(a_pert, t_sub, verbose=False)
            try_update(a_opt, f"pert_cd{trial}")

    # === Phase 7: Final CD polish (long) ===
    print(f"\n=== Phase 7: Final CD (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t7 = min(rem * 0.5, 120.0)
        a_fin_cd, _ = coordinate_descent(best_seq, t7, verbose=True)
        try_update(a_fin_cd, "final_cd")

    # === Phase 8: Final LP + CD ===
    print(f"\n=== Phase 8: Final LP+CD (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t8_lp = min(rem * 0.4, 50.0)
        a_fin_lp, _ = nesterov_lp_refine(best_seq, t8_lp, solver, n_max_iters=50000)
        try_update(a_fin_lp, "final_lp")
        rem = deadline - time.monotonic()
        if rem > 10:
            a_fin_cd2, _ = coordinate_descent(best_seq, min(rem * 0.5, 30.0), verbose=False)
            try_update(a_fin_cd2, "final_cd2")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.10f} n={len(best_seq)}")
    else:
        best_seq = np.ones(1200) / 1200
        print("WARNING: No valid sequence found, using uniform.")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
