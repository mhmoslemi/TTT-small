Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS-B on LogSumExp surrogate in log-space with multi-temperature annealing
2. Multigrid-like refinement (coarse to fine)
3. Active-set focused refinement

5. LP-based alternation for escape from local minima

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time
import math

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def softmax(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def lse_obj_grad(z, T, top_k=None):
    """LogSumExp objective and gradient w.r.t. z."""
    n = len(z)
    a = softmax(z)
    b = fast_convolve(a)
    
    if top_k is not None and top_k < len(b):
        idx = np.argpartition(b, -top_k)[-top_k:]
        b_top = b[idx]
        max_top = b_top.max()
        exp_top = np.exp((b_top - max_top) / T)
        w = np.zeros(len(b))
        w[idx] = exp_top
        lse = max_top + T * np.log(exp_top.sum())
        w /= exp_top.sum()
    else:
        max_b = b.max()
        exp_all = np.exp((b - max_b) / T)
        w = exp_all
        lse = max_b + T * np.log(exp_all.sum())
        w /= exp_all.sum()
    
    obj = 2.0 * n * lse
    
    # Gradient w.r.t. a (approximately, ignoring diagonal correction)
    g_a_raw = np.correlate(w, a, mode='full')
    g_a = g_a_raw[n - 1:2 * n - 1] * 2.0
    
    # Gradient w.r.t. z
    S = np.dot(g_a, a)
    grad_z = a * (g_a - S)
    
    return obj, grad_z

    """Run L-BFGS-B on LogSumExp objective."""
    def fun_g(z):
        obj, grad = lse_obj_grad(z, T, top_k)
        # Shift to keep z bounded
        grad = grad - grad.mean()
        return obj, grad
    
    res = minimize(fun_g, z0, method='L-BFGS-B', jac=True,
                   options={'maxiter': maxiter, 'ftol': tol, 'gtol': 1e-8, 'maxfun': 10000})
    return res.x, res.fun, res

def lbfgs_anneal(a0, T_stages, time_limit, top_k_frac=0.1):
    """Multi-temperature L-BFGS-B annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z -= z.mean()
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    t0 = time.monotonic()
    
    for stage, T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        top_k = max(10, int(top_k_frac * (2 * n - 1)))
        maxiter = max(100, int(5000 * remaining / time_limit * 10))
        
        z_new, _, _ = lse_lbfgs(z, T, top_k, maxiter=maxiter)
        z = z_new
        z -= z.mean()
        
        a = softmax(z)
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        elapsed = time.monotonic() - t0
        print(f"    LSE-LBFGS stage {stage} (T={T:.3f}) ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

2. LP-based best-response alternation for global escape

5. MRF-inspired coordinate refinement (local field balancing)

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.sparse import issparse
import time
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fft_correlation(w, a):
    """Compute g[m] = sum_k w[k] * a[k-m] for m=0..n-1 via FFT."""
    n = len(a)
    L = len(w)
    P = 1 << (L + n - 1).bit_length()
    W = rfft(w, P)
    A_flip = rfft(a[::-1], P)
    conv = irfft(W * A_flip, P)
    return conv[n - 1:2 * n - 1]


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def softmax(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def lse_obj_grad(z, T, top_k=None):
    """LogSumExp objective and gradient w.r.t. z. O(n log n)."""
    n = len(z)
    a = softmax(z)
    b = fast_convolve(a)

    if top_k is not None and top_k < len(b):
        idx = np.argpartition(b, -top_k)[-top_k:]
        b_top = b[idx]
        max_top = b_top.max()
        exp_top = np.exp((b_top - max_top) / T)
        w = np.zeros(len(b))
        w[idx] = exp_top
        lse = max_top + T * np.log(exp_top.sum())
        w /= exp_top.sum()
    else:
        max_b = b.max()
        exp_all = np.exp((b - max_b) / T)
        w = exp_all
        lse = max_b + T * np.log(exp_all.sum())
        w /= exp_all.sum()

    obj = 2.0 * n * lse

    # FFT-based gradient: g_a[m] = 2 * sum_k w[k] * a[k-m]
    g_a = 2.0 * fft_correlation(w, a)

    # Gradient w.r.t. z via softmax Jacobian
    S = np.dot(g_a, a)
    grad_z = a * (g_a - S)

    return obj, grad_z


def lbfgs_stage(z, T, top_k, maxiter, tol=1e-12):
    """Run one L-BFGS-B stage at temperature T."""
    n = len(z)

    def fun_g(z_in):
        obj, grad = lse_obj_grad(z_in, T, top_k)
        # Center gradient (doesn't affect softmax)
        grad = grad - grad.mean()
        return obj, grad

    try:
        res = minimize(fun_g, z.copy(), method='L-BFGS-B', jac=True,
                       options={'maxiter': maxiter, 'ftol': tol,
                                'gtol': 1e-10, 'maxfun': maxiter * 3})
        if res.x is not None:
            return res.x, res.fun
    except Exception:
        pass
    return z, 2.0 * n * fast_convolve(softmax(z)).max()


def lbfgs_anneal(a0, T_stages, time_limit, top_k_frac=0.1, maxiter_per_stage=3000):
    """Multi-temperature L-BFGS-B annealing. O(n log n) per eval."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z -= z.mean()
    best_a = a0.copy()
    b0 = fast_convolve(a0)
    best_score = 2.0 * n * b0.max()
    t0 = time.monotonic()

    for stage, T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        top_k = max(10, int(top_k_frac * (2 * n - 1)))
        # Scale maxiter based on remaining time and stage
        frac_remaining = remaining / time_limit
        miter = max(200, int(maxiter_per_stage * (0.5 + 0.5 * frac_remaining)))

        z, _ = lbfgs_stage(z, T, top_k, miter)
        z -= z.mean()

        a = softmax(z)
        b = fast_convolve(a)
        score = 2.0 * n * b.max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        elapsed = time.monotonic() - t0
        if stage % 3 == 0 or stage == len(T_stages) - 1:
            print(f"      LSE-LBFGS st{stage} T={T:.3f} ({elapsed:.0f}s rem={time_limit-elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def build_conv_matrix_sparse(a_fixed):
    """Build sparse Toeplitz matrix T such that T @ b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    # Use scipy sparse
    from scipy.sparse import csr_matrix
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for i in range(n):
            k = i + j
            if k < size:
                rows.append(k)
                cols.append(j)
                vals.append(a_fixed[i])
    T = csr_matrix((vals, (rows, cols)), shape=(size, n))
    return T


def solve_lp_best_response(a_fixed, time_budget=5.0):
    """Given fixed a (sum=1), find b minimizing max_k conv(a,b)[k] via LP."""
    import cvxpy as cp
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix_sparse(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers = [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]
    best_b = None
    t0 = time.monotonic()
    for s in solvers:
        if time.monotonic() - t0 > time_budget:
            break
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                b_val[np.isnan(b_val)] = 0
                s_sum = b_val.sum()
                if s_sum > 1e-15:
                    b_val = b_val / s_sum
                    return b_val
        except Exception:
            continue
    return None


def lp_alternate(a0, time_limit, n_max_iters=5000):
    """LP best-response alternation with Nesterov averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    B_prev = None

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new = solve_lp_best_response(y, time_budget=3.0)
        if b_new is None:
            break

        # Nesterov dual averaging
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP-alt iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def mrf_coordinate_refine(a0, time_limit, n_steps=50000, eta0=0.05):
    """MRF-inspired coordinate refinement: balance local fields."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Active set: top peaks
        threshold = max_b * (1.0 - max(1e-4, 5e-3 / (1 + step / 100.0)))
        active_mask = b >= threshold
        if not active_mask.any():
            active_mask[np.argmax(b)] = True
        n_active = active_mask.sum()

        # Compute local field E[i] = sum_{k active} w_k * db_k/da_i
        # Using uniform weights over active set
        w = active_mask.astype(np.float64) / n_active
        g = 2.0 * fft_correlation(w, a)

        # Normalized field
        mean_g = g.mean()
        std_g = g.std() + 1e-15
        field = (g - mean_g) / std_g

        # Multiplicative update: reduce a where field is high
        log_update = -eta * field
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 30:
            eta *= 0.85
            no_improve = 0
            if eta < 1e-5:
                break

        if step % 10000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"      MRF-refine {elapsed:.0f}s: score={best_score:.8f} eta={eta:.5f}")

    return best_a, best_score


def structured_init(n, rng):
    """Create structured initial sequence."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.5 + 0.25 * np.cos(2 * np.pi * 7 * x) + 0.15 * np.cos(2 * np.pi * 13 * x + 0.5)
        a = np.maximum(a, 0)
    elif choice == 7:
        a = 1.0 / (1.0 + np.exp(-10 * (x - 0.5)))
    elif choice == 8:
        a = np.abs(np.sin(np.pi * x * 5)) ** 0.5
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def downsample(a, factor):
    """Downsample by block averaging."""
    n = len(a)
    new_n = n // factor
    a_trimmed = a[:new_n * factor]
    a_blocks = a_trimmed.reshape(new_n, factor)
    a_coarse = a_blocks.mean(axis=1)
    s = a_coarse.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_coarse / s


def upsample(a, factor):
    """Upsample by interpolation."""
    return interpolate_sequence(a, len(a) * factor)


def multigrid_refine(a0, time_limit, factors=[4, 2], T_coarse=[5.0, 2.0, 1.0, 0.5],
                     T_fine=[1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005]):
    """Coarse-to-fine multigrid L-BFGS refinement."""
    n = len(a0)
    t0 = time.monotonic()
    a = a0.copy()

    # Coarse levels
    for factor in factors:
        if time.monotonic() - t0 > time_limit * 0.4:
            break
        n_coarse = n // factor
        if n_coarse < 50:
            continue
        a_coarse = downsample(a, factor)
        t_coarse = min((time_limit * 0.15), time_limit - (time.monotonic() - t0) * 0.3)
        if t_coarse < 5:
            continue
        a_c_opt, _ = lbfgs_anneal(a_coarse, T_coarse, t_coarse,
                                   top_k_frac=0.2, maxiter_per_stage=2000)
        a_up = upsample(a_c_opt, factor)
        a_up = np.maximum(a_up, 0)
        s = a_up.sum()
        if s > 1e-15:
            a_up = a_up / s
        # Blend coarse solution with current
        a = 0.5 * a + 0.5 * a_up
        a = np.maximum(a, 0)
        a = a / a.sum()
        elapsed = time.monotonic() - t0
        score = 2.0 * n * fast_convolve(a).max()
        print(f"      Multigrid factor={factor} ({elapsed:.0f}s): score={score:.8f}")

    # Fine level
    remaining = time_limit - (time.monotonic() - t0)
    if remaining > 10:
        a_opt, _ = lbfgs_anneal(a, T_fine, remaining,
                                 top_k_frac=0.1, maxiter_per_stage=3000)
        return a_opt

    return a
