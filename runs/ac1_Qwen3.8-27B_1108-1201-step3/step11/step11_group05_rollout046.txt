        # Compute subgradient at y
        b = fast_convolve(y)
        max_b = b.max()
        
        # Active set: where b is close to max
        threshold = max_b * (1.0 - 1e-6)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        w = np.zeros(len(b))
        w[active] = 1.0 / len(active)
        
        # Subgradient
        g = compute_subgrad_active(y, w)
        
        # Projected gradient step
        eta = eta0 / (1.0 + step * 0.001)
        a_new = project_simplex(y - eta * g)
        
        a_prev = a
        a = a_new
        
        # Check score
        if step % 10 == 0:
            sc = score_of(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if step % 5000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            cur_score = score_of(a)
            print(f"    NesterovSubGrad step={step} ({elapsed:.0f}s): best={best_score:.8f} cur={cur_score:.8f}")
    
    return best_a, best_score


def adam_logsumexp(a0, time_limit, T_stages, lr_init=0.003):
    """Adam optimization on LogSumExp-smoothed objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    # Work in log-space for non-negativity constraint
    z = np.log(np.maximum(a, 1e-30))
    
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step_count = 0
    
    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        
        steps_in_stage = 3000
        for step in range(steps_in_stage):
            if time.monotonic() - t0 > time_limit:
                break
            step_count += 1
            
            # Convert z to a
            z_shift = z - z.max()
            exp_z = np.exp(z_shift)
            a = exp_z / exp_z.sum()
            
            # Compute LSE objective and gradient
            b = fast_convolve(a)
            max_b = b.max()
            
            # LSE approximation
            exp_b = np.exp((b - max_b) / T)
            w = exp_b / exp_b.sum()
            lse = max_b + T * np.log(exp_b.sum())
            
            score_approx = 2.0 * n * lse
            
            # Gradient w.r.t. a
            g_a = compute_subgrad_active(a, w)
            g_a = g_a * (2.0 * n / T)  # Scale for LSE
            
            # Convert to z-space gradient
            S = np.dot(g_a, a)
            g_z = a * (g_a - S)
            
            # Adam update
            lr = lr_init / (1.0 + 0.01 * step)
            m = beta1 * m + (1.0 - beta1) * g_z
            v = beta2 * v + (1.0 - beta2) * g_z ** 2
            m_hat = m / (1.0 - beta1 ** (step_count + 1))
            v_hat = v / (1.0 - beta2 ** (step_count + 1))
            z_new = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Center z (doesn't change softmax)
            z = z_new - z_new.mean()
            
            # Check actual score periodically
            if step_count % 500 == 0:
                sc = score_of(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
    
    return best_a, best_score


def structured_init(n, rng):
    """Create structured initial sequences."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = np.exp(-3 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x + 0.5)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 2 + 0.05
    elif choice == 5:
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.6 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.5 + 0.2 * np.cos(2 * np.pi * 5 * x) + 0.15 * np.cos(2 * np.pi * 11 * x + 1.0)
        a = np.maximum(a, 0)
    elif choice == 7:
        a = np.ones(n) * (1.0 + 0.1 * np.sin(2 * np.pi * 2 * x))
    elif choice == 8:
        a = 1.0 / (1.0 + np.exp(-10 * (x - 0.5)))  # sigmoid
    else:
        a = np.ones(n) / n
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0)
    ss = a_interp.sum()
    if ss > 1e-15:
        a_interp = a_interp / ss
    return a_interp


def multip_weights_polish(a0, time_limit, n_steps=200000):
    """
    Multiplicative weights (mirror descent) polish.
    At each step, reduce the values of a that contribute to the peak.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    eta = 0.02
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Focus on top peaks
        n_peaks = max(3, len(b) // 50)
        idx = np.argpartition(b, -n_peaks)[-n_peaks:]
        # Weight by height (softmax-like)
        b_peaks = b[idx] - b[idx].max()
        w_peaks = np.exp(b_peaks / (0.1 * max_b + 1e-10))
        w_peaks /= w_peaks.sum()
        
        w = np.zeros(len(b))
        w[idx] = w_peaks
        
        # Gradient
        g = compute_subgrad_active(a, w)
        
        # Multiplicative update
        a_new = a * np.exp(-eta * g)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        sc = score_of(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Reduce step size occasionally
            if step % 100 == 0:
                eta *= 0.95
    
    return best_a, best_score
