import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def lse_obj_grad(a, T):
    """LSE-smoothed objective and gradient. a is normalized (sum=1)."""
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    shifted = (c - c_max) / T
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_c = np.exp(shifted)
    exp_sum = exp_c.sum()
    lse = c_max + T * np.log(exp_sum)
    w = exp_c / exp_sum
    
    # Gradient: grad_a[m] = 2 * sum_j a[j] * w[m+j]
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w, fft_size)
    A_f = rfft(a, fft_size)
    corr = irfft(W * np.conj(A_f), fft_size)[:n]
    grad_a = 2.0 * corr
    
    obj = 2.0 * n * lse
    return obj, grad_a

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    c = fast_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:

    return obj, grad_a


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def evaluate_local(a):
    """Evaluate the objective locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    c = fast_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def fista_lse(a0, time_limit, T_init=0.5, T_min=0.005, T_halve_every=5000):
    """FISTA with LSE smoothing and geometric temperature decay."""
    n = len(a0)
    x = normalize(a0.copy())
    y = x.copy()
    x_prev = x.copy()
    t_mom = 1.0
    best_x = x.copy()
    best_score = evaluate_local(x)
    T = T_init
    L = 5.0

    t0 = time.monotonic()
    it = 0
    last_eval = 0

    while time.monotonic() - t0 < time_limit:
        _, grad_y = lse_obj_grad(y, T)

        # FISTA update
        z = y - grad_y / L
        x_new = project_simplex(z)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y_new = x_new + ((t_mom - 1.0) / t_new) * (x_new - x)
        y_new = np.maximum(y_new, 0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = x_new.copy()
        else:
            y_new /= s_y

        x_prev = x
        x = x_new
        y = y_new
        t_mom = t_new
        it += 1

        # Periodic evaluation and temperature update
        if it - last_eval >= 200:
            last_eval = it
            sc = evaluate_local(x)
            if sc < best_score:
                best_score = sc
                best_x = x.copy()
            T = max(T_min, T * (T_min / T_init) ** (1.0 / T_halve_every))

            if it % 5000 == 0:
                print(f"    FISTA it={it}, T={T:.6f}, score={best_score:.8f}")

    sc = evaluate_local(x)
    if sc < best_score:
        best_score = sc
        best_x = x.copy()

    return best_x, best_score


def adam_lse(a0, time_limit, lr=0.0005, T_init=0.1, T_decay=0.9995):
    """Adam optimizer with LSE smoothing."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    T = T_init

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    t = 0

    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        _, grad = lse_obj_grad(a, T)
        t += 1

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)
        a = a_new

        if t % 100 == 0:
            sc = evaluate_local(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            T = max(0.002, T * T_decay)
            if t % 2000 == 0:
                print(f"    Adam t={t}, T={T:.6f}, score={best_score:.8f}")

    sc = evaluate_local(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_a, best_score


def active_set_polish(a0, time_limit, eta_init=0.05):
    """Multiplicative weights using active set (near-max convolution values)."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    eta = eta_init
    t0 = time.monotonic()
    step = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        c_max = c.max()

        # Active set: within 0.05% of max
        threshold = c_max - c_max * 5e-5
        active_mask = c >= threshold
        active_count = active_mask.sum()
        if active_count == 0:
            active_mask = c >= c_max - 1e-15
            active_count = max(1, active_mask.sum())

        # Gradient via FFT: grad[m] = (2/|S|) * sum_{k in S} a[k-m] * I[0<=k-m<n]
        # This is cross-correlation of active_mask and a
        fft_size = 1 << (3 * n - 2).bit_length()
        AM = rfft(active_mask.astype(np.float64), fft_size)
        A_f = rfft(a, fft_size)
        corr = irfft(AM * np.conj(A_f), fft_size)[:n]
        grad = 2.0 * corr / active_count

        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = project_simplex(a_new)

        sc = evaluate_local(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            eta *= 0.7
            if eta < 1e-12:
                break

        step += 1
        if step % 200 == 0:
            print(f"    AP step={step}, eta={eta:.8f}, |S|={active_count}, score={best_score:.8f}")

    return best_a, best_score


def diverse_init(n, rng):
    """Generate diverse initial sequences."""
    candidates = []
    x = np.linspace(0, 1, n)

    # Uniform
    candidates.append(np.ones(n) / n)

    # Gaussian bump
    a = np.exp(-4 * (x - 0.5) ** 2)
    candidates.append(normalize(a))

    # Sine
    a = np.sin(np.pi * x) ** 2
    candidates.append(normalize(a))

    # Two bumps
    a = np.exp(-6 * (x - 0.25) ** 2) + np.exp(-6 * (x - 0.75) ** 2)
    candidates.append(normalize(a))

    # Cosine wave
    a = 1.0 + 0.5 * np.cos(2 * np.pi * 3 * x)
    candidates.append(normalize(a))

    # Random smooth (low frequency)
    coeffs = rng.normal(0, 1, 8)
    a = np.ones(n) * 0.5
    for i, c in enumerate(coeffs):
        a += c * 0.3 * np.cos(2 * np.pi * (i + 1) * x + rng.uniform(0, 2 * np.pi))
    a = np.maximum(a, 0)
    candidates.append(normalize(a))

    # Edge-peaked
    a = np.exp(-3 * x) + np.exp(-3 * (1 - x))
    candidates.append(normalize(a))

    # Power law
    a = (x * (1 - x)) ** 0.3
    candidates.append(normalize(a))

    return candidates


def structured_perturb(a, rng, method='random'):
    """Apply structured perturbation to sequence."""
    n = len(a)
    a = normalize(a.copy())

    if method == 'reverse_mix':
        t = rng.uniform(0.01, 0.1)
        a = (1 - t) * a + t * a[::-1]
    elif method == 'block_shift':
        bs = rng.integers(10, 50)
        shift = rng.integers(1, bs)
        a_new = np.zeros(n)
        a_new[shift:] = a[:-shift]
        a_new[:shift] = a[n - shift:]
        a = normalize(0.9 * a + 0.1 * a_new)
    elif method == 'noise':
        noise = rng.normal(0, 0.02, n)
        a = normalize(a + noise)
    elif method == 'lowpass_perturb':
        freq = rng.integers(1, 6)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.linspace(0, 1, n)
        noise = 0.03 * np.cos(2 * np.pi * freq * x + phase)
        a = normalize(a + noise)
    elif method == 'random_block':
        bs = rng.integers(10, 100)
        i = rng.integers(0, n - bs)
        a[i:i + bs] *= rng.uniform(0.5, 1.5)
        a = normalize(a)
    else:
        a = normalize(a + rng.normal(0, 0.01, n))

    return normalize(a)


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    return normalize(f_interp)


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))².
    Uses FISTA with LSE, Adam, active-set polish, and multi-restart.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # Try to load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        candidates = diverse_init(n_init, rng)
        scores = [evaluate_local(c) for c in candidates]
        best_idx = np.argmin(scores)
        best_seq = candidates[best_idx].copy()
        best_score = scores[best_idx]
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)
    print(f"Starting: n={n_best}, score={best_score:.8f}")

    # === Phase 1: FISTA with LSE (coarse to fine) ===
    print(f"\n=== Phase 1: FISTA LSE (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_f, _ = fista_lse(best_seq, t_alloc, T_init=0.3, T_min=0.005, T_halve_every=8000)
        local_update(a_f, "fista_1")

    # === Phase 2: Adam fine-tuning ===
    print(f"\n=== Phase 2: Adam LSE (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_ad, _ = adam_lse(best_seq, t_alloc, lr=0.0003, T_init=0.08, T_decay=0.9998)
        local_update(a_ad, "adam_1")

    # === Phase 3: Active set polish ===
    print(f"\n=== Phase 3: Active Set Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_ap, _ = active_set_polish(best_seq, t_alloc, eta_init=0.03)
        local_update(a_ap, "active_set")

    # === Phase 4: Multi-restart with structured perturbations ===
    print(f"\n=== Phase 4: Multi-Restart (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        print(f"  Budget: {rem:.0f}s")
        methods = ['reverse_mix', 'block_shift', 'noise', 'lowpass_perturb', 'random_block']
        n_restarts = min(8, int(rem / 30))
        for r in range(n_restarts):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 20:
                break
            method = methods[r % len(methods)]
            perturbed = structured_perturb(best_seq, rng, method)
            t_sub = min(rem_sub * 0.4, 120.0)
            # Quick FISTA on perturbed
            a_r, _ = fista_lse(perturbed, t_sub, T_init=0.1, T_min=0.005, T_halve_every=5000)
            local_update(a_r, f"restart_{method}")
            rem_sub2 = deadline - time.monotonic()
            if rem_sub2 > 20:
                a_r2, _ = active_set_polish(a_r, min(rem_sub2 * 0.3, 30.0), eta_init=0.02)
                local_update(a_r2, f"restart_{method}_ap")

    # === Phase 5: Try different lengths ===
    print(f"\n=== Phase 5: Length Exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        for n_try in [n_best - 100, n_best + 100, n_best + 200, n_best - 200]:
            if n_try < 500 or n_try > 3000:
                continue
            rem_sub = deadline - time.monotonic()
            if rem_sub < 30:
                break
            init = interpolate_sequence(best_seq, n_try)
            t_sub = min(rem_sub * 0.35, 80.0)
            a_try, _ = fista_lse(init, t_sub, T_init=0.15, T_min=0.005, T_halve_every=5000)
            local_update(a_try, f"len_{n_try}")
            rem_sub2 = deadline - time.monotonic()
            if rem_sub2 > 15:
                a_try2, _ = active_set_polish(a_try, min(rem_sub2 * 0.4, 25.0), eta_init=0.02)
                local_update(a_try2, f"len_{n_try}_ap")

    # === Phase 6: Final fine polish ===
    print(f"\n=== Phase 6: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_sub = min(rem * 0.4, 60.0)
        a_fin, _ = adam_lse(best_seq, t_sub, lr=0.0001, T_init=0.03, T_decay=0.9999)
        local_update(a_fin, "final_adam")
        rem2 = deadline - time.monotonic()
        if rem2 > 15:
            a_fin2, _ = active_set_polish(best_seq, min(rem2 * 0.5, 30.0), eta_init=0.01)
            local_update(a_fin2, "final_ap")

    # Final FISTA run with very low temperature
    rem = deadline - time.monotonic()
    if rem > 20:
        a_final, _ = fista_lse(best_seq, min(rem * 0.6, 60.0), T_init=0.02, T_min=0.001, T_halve_every=3000)
        local_update(a_final, "final_fista")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
