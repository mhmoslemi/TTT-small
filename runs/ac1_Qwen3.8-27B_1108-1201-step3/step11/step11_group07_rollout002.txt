def peak_suppression(a, k_star, lr):
    n = len(a)
    a_new = a.copy()
    # Find the "partner" of each position
    partner = np.where((k_star - np.arange(n) >= 0) & (k_star - np.arange(n) < n),
                       k_star - np.arange(n), -1)
    # Contribution to peak from position j: a_j * a_{partner[j]}
    contrib = np.zeros(n)
    valid = partner >= 0
    contrib[valid] = a[valid] * a[partner[valid]]
    # Reduce positions with high contribution
    # Use soft thresholding: reduce proportionally to contribution
    threshold = np.percentile(contrib[valid], 80)  # top 20%
    mask = contrib > threshold
    a_new[mask] *= (1 - lr)
    # Redistribute to positions with low contribution
    low_mask = contrib < np.percentile(contrib[valid], 20)
    total_reduced = a[mask].sum() - a_new[mask].sum()
    a_new[low_mask] += total_reduced / low_mask.sum()
    return a_new

def compute_subgrad(a, K=5, soft_T=0.0):
    """Compute subgradient of max(c) with respect to a.
    Uses top-K active set with optional soft weighting.
    Returns (g, c_max, active_weights) where g is the subgradient
    projected onto the tangent space of the simplex.
    """
    n = len(a)
    c = fast_convolve(a)  # length 2n-1
    c_max = c.max()

        # Soft active set: weight = exp((c_k - c_max) / T)
        w = np.exp((c - c_max) / soft_T)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros_like(c)
            w[c.argmax()] = 1.0
            w_sum = 1.0
        else:
            w = w / w_sum
    else:

        idx = np.argsort(c)[::-1][:K]
        w = np.zeros(len(c))
        w[idx] = 1.0 / K
        w_sum = 1.0
    
    # Compute cross-correlation: cross[j] = sum_k w[k] * a[k-j] for valid k-j
    # Using FFT
    fft_size = 1 << (3 * n).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # cross[j] = sum_k w[k] * a[k-j]
    # This is: ifft(fft(w) * conj(fft(a))) ... let me verify
    # circular cross-correlation: (w cross-corr a)[j] = sum_k w[k+j] * a[k]
    # = ifft(fft(w) * conj(fft(a))) [j]
    # But I want sum_k w[k] * a[k-j] = sum_m w[m+j] * a[m] (sub m = k-j)
    # = (w cross-corr a)[j] (with the convention above)
    # Hmm, let me be more careful.
    # (w cross-corr a)[j] = sum_k w[k] * conj(a[k-j]) = sum_k w[k] * a[k-j] (real)
    # In FFT: ifft(fft(w) * conj(fft(a)))... 
    # Actually, the standard formula is:
    # cross[j] = sum_k w[k+j] * a[k] = ifft(conj(fft(w)) * fft(a))[j]
    # Or equivalently: cross[j] = sum_k w[k] * a[k-j] = ifft(fft(w) * conj(fft(a)))[j]
    # Let me verify with a simple example.
    # w = [1, 0, 0], a = [a0, a1, a2]
    # sum_k w[k] * a[k-j] = a[-j] = a[3-j] (circular) 
    # For j=0: a[0], j=1: a[2], j=2: a[1]
    # FFT of w = [1,1,1], conj(FFT(a)) = conj(a0+a1+a2, ...)
    # This is getting confusing. Let me just use the direct formula.
    
    # I want: cross[j] = sum_{k=0}^{2n-2} w[k] * a[k-j] where 0 <= k-j < n
    # = sum_{m=0}^{n-1} w[m+j] * a[m] where 0 <= m+j < 2n-1
    # This is a "sliding dot product" or cross-correlation.
    # 
    # Using FFT: Let W_full = w (length 2n-1, zero-padded to fft_size)
    #            A_full = a (length n, zero-padded to fft_size)
    # Then: cross = ifft(fft(W_full) * conj(fft(A_full)))
    # Let's verify: ifft(fft(W) * conj(fft(A)))[j] = sum_k W[k] * A[(k-j) mod N]
    # For our case (with sufficient zero padding): = sum_k W[k] * A[k-j] (for valid indices)
    # = sum_k w[k] * a[k-j] (since A[k-j] = a[k-j] if 0 <= k-j < n, else 0)
    # Yes! This is exactly what we want.
    
    cross = irfft(W * np.conj(A), fft_size)[:n]
    
    # Second term: -a_j * w[2j] (the diagonal correction)
    diag = np.zeros(n)
    valid_diag = 2 * np.arange(n) < len(w)
    diag[valid_diag] = a[valid_diag] * w[2 * np.arange(n)[valid_diag]]
    
    # Full subgradient of max(c):
    # g_j = 2 * cross[j] - diag[j]
    # (Note: the factor of 2 comes from the chain rule, as derived earlier)
    g = 2.0 * cross - diag
    
    # Project onto tangent space of simplex: subtract mean
    g = g - g.mean()

"""

1. Exact subgradient method with multiplicative (mirror) descent on probability simplex
2. Adaptive active set (hard top-K and soft exponential weighting)

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(a)) / len(a)
    return a / s


def evaluate_local(a):
    """Evaluate the objective locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()

        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def compute_subgrad(a, K=5, soft_T=0.0):
    """Compute subgradient of max(conv(a,a)) w.r.t. a, projected to simplex tangent.
    Returns (g, c_max, peak_indices).
    """
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()

    if soft_T > 1e-15:
        w = np.exp((c - c_max) / soft_T)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros_like(c)
            w[c.argmax()] = 1.0
        else:
            w = w / w_sum
    else:
        if K >= len(c):
            w = np.ones(len(c)) / len(c)
        else:
            idx = np.argpartition(c, -K)[-K:]
            w = np.zeros(len(c))
            w[idx] = 1.0 / K

    # Cross-correlation: cross[j] = sum_k w[k] * a[k-j] for 0 <= k-j < n
    fft_size = 1 << (3 * n).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    cross = irfft(W * np.conj(A), fft_size)[:n]

    # Diagonal correction: -a_j * w[2j]
    j_idx = np.arange(n)
    valid_diag = 2 * j_idx < len(w)
    diag = np.zeros(n)
    diag[valid_diag] = a[valid_diag] * w[2 * j_idx[valid_diag]]

    # Full subgradient: g_j = 2*cross[j] - diag[j]
    g = 2.0 * cross - diag

    # Project to tangent space of simplex
    g = g - g.mean()

    peak_idx = np.argmax(c)
    return g, float(c_max), int(peak_idx)


def mirror_step(a, g, eta):

    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()

    return a_new / s

def simulated_annealing_subgrad(a_init, time_limit, T_start=0.05, T_end=0.001,
                                 eta0=0.2, K=3, verbose=False):
    """Simulated annealing with subgradient-based proposals."""
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(789)
    current_score = best_score
    iter_count = 0
    accepted = 0
    total = 0

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        # Compute temperature (exponential cooling)
        progress = (time.monotonic() - t0) / time_limit
        T = T_start * (T_end / T_start) ** progress

        iter_count += 1
        total += 1

        # Get subgradient direction
        g, c_max, _ = compute_subgrad(a, K=K, soft_T=c_max * 0.01)

        # Add noise proportional to temperature
        noise = rng.normal(0, T * 0.5, n)
        g_noisy = g + noise

        # Mirror step
        a_new = mirror_step(a, g_noisy, eta0)
        new_score = evaluate_local(a_new)

        # Metropolis acceptance
        if new_score < current_score:
            a = a_new
            current_score = new_score
            accepted += 1
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
        else:
            delta = new_score - current_score
            if delta < T * 10:  # Only accept if not too bad
                accept_prob = np.exp(-delta / max(T, 1e-10))
                if rng.random() < accept_prob:
                    a = a_new
                    current_score = new_score
                    accepted += 1

        # Occasionally make a large exploratory move
        if iter_count % 200 == 0:

            sc_pert = evaluate_local(a_pert)
            if sc_pert < best_score:
                best_score = sc_pert
                best_a = a_pert.copy()
                a = a_pert
                current_score = sc_pert
            # Interpolation with best

            sc_interp = evaluate_local(a_interp)
            if sc_interp < best_score:
                best_score = sc_interp
                best_a = a_interp.copy()
                a = a_interp
                current_score = sc_interp

        if verbose and iter_count % 500 == 0:
            print(f"    SA iter={iter_count}: best={best_score:.8f}, T={T:.6f}, accept_rate={accepted/max(total,1):.3f}")

    return best_a, best_score


def peak_pairing_refinement(a_init, time_limit, verbose=False):
    """Target peak reduction by identifying and suppressing contributing pairs."""
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()

        if remaining < 1:
            break

        c = fast_convolve(a)
        c_max = c.max()
        k_star = int(np.argmax(c))

        # Find positions contributing most to c[k_star]
        # c[k_star] = sum_j a_j * a[k_star - j]
        contrib = np.zeros(n)
        for j in range(n):
            kj = k_star - j
            if 0 <= kj < n:
                contrib[j] = a[j] * a[kj]

        # Target top contributors for reduction
        top_frac = 0.15
        n_top = max(10, int(n * top_frac))
        top_idx = np.argpartition(contrib, -n_top)[-n_top:]

        # Reduce top contributors and increase bottom contributors
        bottom_frac = 0.15
        n_bottom = max(10, int(n * bottom_frac))
        bottom_idx = np.argpartition(contrib, n_bottom)[:n_bottom]

        # Apply soft reduction
        for lr in [0.05, 0.03, 0.01]:
            a_new = a.copy()
            reduce_amount = lr * a[top_idx].mean()
            a_new[top_idx] -= reduce_amount
            a_new[top_idx] = np.maximum(a_new[top_idx], 0)
            # Redistribute to bottom
            total_reduced = (a[top_idx] - a_new[top_idx]).sum()
            a_new[bottom_idx] += total_reduced / max(len(bottom_idx), 1)
            a_new = normalize(a_new)
            sc = evaluate_local(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
            break

        # Also try: shift mass from peak-contributing region to non-contributing region
        if remaining > 5:
            # Identify the "hot zone" around k_star/2 (where both a_j and a_{k_star-j} are large)
            center = k_star / 2.0
            width = n * 0.1
            hot_mask = np.abs(np.arange(n) - center) < width
            cold_mask = ~hot_mask

            for shift in [0.005, 0.002]:
                a_new = a.copy()
                mass_to_shift = shift * a[hot_mask].sum() * 0.1
                a_new[hot_mask] -= mass_to_shift / max(hot_mask.sum(), 1)
                a_new[hot_mask] = np.maximum(a_new[hot_mask], 0)
                a_new[cold_mask] += mass_to_shift / max(cold_mask.sum(), 1)
                a_new = normalize(a_new)
                sc = evaluate_local(a_new)
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    a = a_new.copy()
                break

    return best_a, best_score


def coarse_to_fine(a_coarse, n_target, time_limit_per_scale=30):
    """Upsample coarse solution to target length and refine."""
    n_coarse = len(a_coarse)
    a_fine = np.interp(
        np.linspace(0, 1, n_target),
        np.linspace(0, 1, n_coarse),
        a_coarse
    )
    a_fine = normalize(a_fine)
    a_fine, _ = subgrad_mirror_descent(a_fine, time_limit_per_scale, eta0=0.3, K_start=1, K_max=50)
    return a_fine


def diverse_init(n, rng):
    """Generate diverse initial sequences of length n."""
    candidates = []
    x = np.linspace(0, 1, n)

    # Various parametric shapes
    candidates.append(normalize(np.exp(-6 * (x - 0.5) ** 2)))
    candidates.append(normalize(0.5 * (1 + np.cos(2 * np.pi * x))))
    candidates.append(normalize(np.sin(np.pi * x) ** 2))
    candidates.append(normalize((x * (1 - x)) ** 0.3))
    candidates.append(normalize(np.exp(-8 * (x - 0.3) ** 2) + np.exp(-8 * (x - 0.7) ** 2)))
    candidates.append(normalize(0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.15 * np.cos(10 * np.pi * x)))
    candidates.append(normalize(np.ones(n)))

    # Random smooth
    for _ in range(3):
        coeffs = rng.normal(0, 1, 15)
        a = np.zeros(n)
        for i in range(min(15, len(coeffs))):
            freq = (i + 1) * rng.uniform(0.5, 2.0)
            phase = rng.uniform(0, 2 * np.pi)
            a += coeffs[i] * np.cos(2 * np.pi * freq * x + phase)
        a = np.maximum(a + np.abs(a).max() * 0.5 + 0.1, 0)
        candidates.append(normalize(a))

    return candidates


def multi_start_search(n, time_limit, rng, verbose=False):
    """Try multiple initializations and keep the best."""
    inits = diverse_init(n, rng)
    best_a = None
    best_score = float('inf')

    n_starts = min(len(inits), max(3, int(time_limit / 30)))
    t_per_start = time_limit / max(n_starts, 1)

    for i in range(n_starts):
        if time.monotonic() - start_time_multi > time_limit:
            break
        a0 = inits[i % len(inits)]
        a, sc = subgrad_mirror_descent(a0, t_per_start, eta0=0.3, K_start=1, K_max=50)
        if sc < best_score:
            best_score = sc
            best_a = a
            if verbose:
                print(f"    MultiStart: init {i} gives {sc:.8f}")

    return best_a, best_score


def adaptive_polish(a_init, time_limit, verbose=False):
    """Aggressive final polish with very targeted subgradient steps."""
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()

    # Use very small K and small steps for precision
    strategies = [
        (1, 0.05, 0.0),    # K=1, small eta, hard
        (2, 0.03, 0.0),    # K=2, smaller eta, hard
        (3, 0.02, 0.0),    # K=3
        (5, 0.01, 0.0),    # K=5
        (1, 0.01, 0.001),  # K=1, soft
        (3, 0.005, 0.001), # K=3, soft
    ]

    for K, eta, soft_T in strategies:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        a, sc = subgrad_mirror_descent(a, remaining * 0.4, eta0=eta, K_start=K, K_max=K,
                                        soft_T_init=soft_T)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            if verbose:
                print(f"    Polish K={K}, eta={eta}: {sc:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimum peak autocorrelation of non-negative sequences.
    Uses exact subgradient + mirror descent, simulated annealing, peak pairing,
    and multi-length exploration.
    """
    global start_time_multi
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"No parent available: {e}")

    if parent is None:
        n_init = 1200
        inits = diverse_init(n_init, rng)
        scores = [evaluate_local(c) for c in inits]
        best_idx = np.argmin(scores)
        best_seq = inits[best_idx].copy()
        best_score = scores[best_idx]
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)
    print(f"Starting: n={n_best}, score={best_score:.8f}, budget={budget_s:.0f}s")

    # === Phase 1: Quick exploration of different lengths (15%) ===
    print(f"\n=== Phase 1: Length Exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_phase1 = min(rem * 0.12, 150.0)
    if t_phase1 > 30:
        print(f"  Budget: {t_phase1:.0f}s")
        start_time_multi = time.monotonic()
        # Try the parent length and nearby lengths
        lengths_to_try = [n_best]
        for dl in [100, 200, -100, 300]:
            nl = n_best + dl
            if 200 <= nl <= 3000:
                lengths_to_try.append(nl)

        for nl in lengths_to_try:
            sub_rem = deadline - time.monotonic()
            if sub_rem < 20:
                break
            if nl == n_best:
                # Skip, we already have this
                continue
            # Interpolate parent to new length
            x_old = np.linspace(0, 1, n_best)
            x_new = np.linspace(0, 1, nl)
            a_interp = np.interp(x_new, x_old, best_seq)
            a_interp = normalize(a_interp)
            t_sub = min(sub_rem * 0.4, 40.0)
            a_try, _ = subgrad_mirror_descent(a_interp, t_sub, eta0=0.3, K_start=1, K_max=50)
            try_update(a_try, f"len_{nl}")

    # === Phase 2: Main subgradient mirror descent (35%) ===
    print(f"\n=== Phase 2: Subgradient Mirror Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_phase2 = min(rem * 0.35, 500.0)
    if t_phase2 > 20:
        print(f"  Budget: {t_phase2:.0f}s")
        # Run multiple strategies with different hyperparameters
        strategies = [
            (0.3, 1, 50, 0.0, "hard_K1"),
            (0.2, 3, 30, 0.0, "hard_K3"),
            (0.15, 5, 20, 0.0, "hard_K5"),
        ]
        for eta, K, K_max, sT, label in strategies:
            sub_rem = deadline - time.monotonic()
            if sub_rem < 15:
                break
            t_sub = min(sub_rem * 0.5, t_phase2 / len(strategies) + 30)
            a_res, _ = subgrad_mirror_descent(best_seq, t_sub, eta0=eta,
                                               K_start=K, K_max=K_max, soft_T_init=sT)
            try_update(a_res, f"SMD_{label}")

    # === Phase 3: Simulated annealing (25%) ===
    print(f"\n=== Phase 3: Simulated Annealing (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_phase3 = min(rem * 0.25, 350.0)
    if t_phase3 > 20:
        print(f"  Budget: {t_phase3:.0f}s")
        a_sa, _ = simulated_annealing_subgrad(best_seq, t_phase3,
                                               T_start=0.03, T_end=0.0005,
                                               eta0=0.15, K=3)
        try_update(a_sa, "sim_anneal")

        # Second SA run with different parameters
        sub_rem = deadline - time.monotonic()
        if sub_rem > 30:
            a_sa2, _ = simulated_annealing_subgrad(best_seq, sub_rem * 0.5,
                                                    T_start=0.01, T_end=0.0002,
                                                    eta0=0.08, K=1)
            try_update(a_sa2, "sim_anneal_fine")

    # === Phase 4: Peak pairing refinement (15%) ===
    print(f"\n=== Phase 4: Peak Pairing (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_phase4 = min(rem * 0.15, 200.0)
    if t_phase4 > 15:
        print(f"  Budget: {t_phase4:.0f}s")
        a_pp, _ = peak_pairing_refinement(best_seq, t_phase4)
        try_update(a_pp, "peak_pairing")

    # === Phase 5: Adaptive final polish (15%) ===
    print(f"\n=== Phase 5: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_phase5 = min(rem * 0.18, 250.0)
    if t_phase5 > 10:
        print(f"  Budget: {t_phase5:.0f}s")
        a_polish, _ = adaptive_polish(best_seq, t_phase5)
        try_update(a_polish, "polish")

    # === Phase 6: Coarse-to-fine refinement ===
    print(f"\n=== Phase 6: Coarse-to-Fine (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        # Downsample, optimize, then upsample
        n_coarse = max(100, n_best // 6)
        x_coarse = np.linspace(0, 1, n_coarse)
        a_coarse = np.interp(x_coarse, np.linspace(0, 1, n_best), best_seq)
        a_coarse = normalize(a_coarse)
        a_coarse, _ = subgrad_mirror_descent(a_coarse, min(rem * 0.2, 30), eta0=0.5, K_start=1, K_max=10)
        # Upsample back
        a_fine = np.interp(np.linspace(0, 1, n_best), x_coarse, a_coarse)
        a_fine = normalize(a_fine)
        a_fine, _ = subgrad_mirror_descent(a_fine, min(rem * 0.3, 40), eta0=0.2, K_start=1, K_max=30)
        try_update(a_fine, "coarse_fine")

    # === Phase 7: Final micro-optimization ===
    print(f"\n=== Phase 7: Micro-Opt (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 10:
        # Very small steps, K=1
        a_micro, _ = subgrad_mirror_descent(best_seq, min(rem * 0.8, 60),
                                             eta0=0.02, K_start=1, K_max=1)
        try_update(a_micro, "micro")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
