import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.sparse import lil_matrix, csc_matrix

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_q_sparse(k, n):
    """Build sparse Q matrix for conv(a,a)[k] = a^T Q_k a."""
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)

def evaluate_local(a):
    """Evaluate sequence score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def get_best_solver():
    """Find the best available solver."""
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.XPRESS, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def solve_active_qcqp(a, active_set, solver_list, max_iters=1):
    """Solve min t s.t. a^T Q_k a <= t for k in active_set, sum(a)=1, a>=0"""
    n = len(a)
    a_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [cp.sum(a_var) == 1]
    
    for k in active_set:
        Q = build_q_sparse(int(k), n)
        constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    for s in solver_list:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.asarray(a_var.value, dtype=np.float64)
                a_val = np.maximum(a_val, 0)
                s_sum = a_val.sum()
                if s_sum < 1e-15:
                    return None, None
                a_val = a_val / s_sum
                return a_val, float(max(0, t_var.value))
        except Exception:
            continue
    return None, None

    best_score = evaluate_local(a)

        b_new = fast_convolve(a_new)
        max_b = b_new.max()

        violated = []
        for k in range(len(b_new)):

                violated.append((k, b_new[k]))

        # Add the most violated constraint(s)
        violated.sort(key=lambda x: -x[1])

        print(f"    CP iter {cp_iter}: |active|={len(active_set)}, t={t_val:.10f}, score={score_new:.10f}")
    
    return best_a, best_score

def projected_gradient_refine(a0, solver_list, time_limit, n_steps=100000):
    """
    Projected gradient descent on the objective (with exact projection to simplex).
    Uses subgradient of the max function.
    """
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    # Adaptive step size
    step = 0.01
    no_improve = 0
    
    for step_i in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Find active positions (near max)
        threshold = max_b * (1 - 1e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        # Compute subgradient: average of gradients of active Q_k
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]  # gradient of a^T Q_k a = 2 * (Q_k a)
        g /= len(active)
        
        # Projected gradient step on simplex
        a_new = a - step * g
        # Project to simplex: find tau such that a_new - tau >= 0 and sum = 1
        # Use the standard water-filling algorithm
        a_new_sorted = np.sort(a_new)[::-1]
        css = np.cumsum(a_new_sorted)
        rho = np.where(a_new_sorted * np.arange(1, n+1) > (css - 1))[0]
        if len(rho) == 0:
            a_new = a.copy()
        else:
            rho = rho[-1]
            tau = (css[rho] - 1) / (rho + 1)
            a_new = np.maximum(a_new - tau, 0)
        
        score_new = evaluate_local(a_new)
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            # Maybe increase step
        else:
            no_improve += 1
            if no_improve >= 50:
                step *= 0.7
                no_improve = 0
                if step < 1e-10:
                    break
    
    return best_a, best_score

def lse_smoothed_grad(a, n, temp):
    """Compute LogSumExp-smoothed objective and subgradient."""
    b = fast_convolve(a)
    max_b = b.max()
    log_terms = (b - max_b) / temp
    log_terms -= log_terms.max()
    exp_terms = np.exp(log_terms)
    w = exp_terms / exp_terms.sum()
    lse_val = max_b + temp * np.log(exp_terms.sum())
    
    # Gradient w.r.t. a
    g = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += 2.0 * w[k] * a[k - idx]
    return lse_val, g

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
