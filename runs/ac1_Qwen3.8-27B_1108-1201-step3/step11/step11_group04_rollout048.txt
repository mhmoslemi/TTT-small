"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast spectral flattening for global improvement
2. Nesterov-accelerated LP alternating with solver selection
3. Top-k LogSumExp L-BFGS with adaptive temperature
4. Active-set Newton refinement using Hessian structure

6. Multi-n exploration and diversification

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.CLP, cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            t0 = time.monotonic()
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver, timeout=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.SCS, cp.PDLP, cp.CLARABEL, cp.ECOS, cp.GUROBI, cp.MOSEK]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            if timeout is not None and time.monotonic() > timeout:
                return None, None
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_val = b_val.sum()
                if s_val < 1e-15:
                    return None, None
                b_val = b_val / s_val
                return b_val, float(max(0, M_var.value))
        except Exception:
            continue
    return None, None


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def topk_obj_grad(z, n, k_top):
    """
    Compute top-k sum objective and gradient w.r.t. z.
    This is a piecewise-linear approximation to the max.
    """
    a = softmax_from_z(z)
    b = fast_convolve(a)
    
    # Get top k values
    if k_top >= len(b):
        idx = np.argsort(b)[::-1]
    else:
        part = np.argpartition(b, -k_top)[-k_top:]
        idx = part[np.argsort(b[part])[::-1]]
    
    top_vals = b[idx]
    obj = 2.0 * n * np.sum(top_vals)
    
    # Gradient w.r.t. a
    grad_a = np.zeros(n)
    for kk in idx:
        js = max(0, kk - n + 1)
        je = min(n - 1, kk)
        idx_j = np.arange(js, je + 1)
        grad_a[idx_j] += 2.0 * a[kk - idx_j]
    
    grad_a /= len(idx)
    
    # Gradient w.r.t. z (softmax parameterization)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def lse_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for kk in range(len(b)):
        if w[kk] < 1e-15:
            continue
        js = max(0, kk - n + 1)
        je = min(n - 1, kk)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[kk] * a[kk - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def spectral_flatten(a, n_iters=100, blend=0.4, deadline=None):
    """Iterative spectral flattening (Gerchberg-Saxton style)."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)

        mags = np.abs(F)
        non_dc = mags[1:]
        if non_dc.min() < 1e-15:
            target = np.mean(non_dc)
        else:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        phases = np.angle(F)
        F_new = np.zeros_like(F)
        F_new[0] = F[0]
        for kk in range(1, L // 2 + 1):
            new_mag = (1 - blend) * mags[kk] + blend * target
            F_new[kk] = new_mag * np.exp(1j * phases[kk])
            if kk < L - kk:
                F_new[L - kk] = (1 - blend) * mags[L - kk] + blend * target
                F_new[L - kk] *= np.exp(1j * phases[L - kk])

        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new

        sc = 2.0 * n * fast_convolve(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a


def hessian_matvec(v, active_set, n):
    """Compute H @ v where H = sum_{k in active_set} 2*Q_k."""
    result = np.zeros(n)
    for kk in active_set:
        for i in range(n):
            j = kk - i
            if 0 <= j < n:
                if i == j:
                    result[i] += 1.0 * v[i]
                else:
                    result[i] += 2.0 * v[j]
    return result


def newton_active_set(a, active_set, time_limit, n_max_iters=5000):
    """
    Newton refinement on the active set of peaks.
    Minimizes sum_{k in active_set} conv(a,a)[k] subject to a>=0, sum(a)=1.
    """
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute gradient of sum of active quadratic forms
        grad = np.zeros(n)
        for kk in active_set:
            for i in range(n):
                j = kk - i
                if 0 <= j < n:
                    if i == j:
                        grad[i] += a[i]
                    else:
                        grad[i] += 2.0 * a[j]
        grad /= len(active_set)
        
        # Project gradient to tangent space (sum=0)
        grad -= grad.mean()
        
        # CG solve for Newton direction: H * d = -grad
        # H is the Hessian of the sum of quadratic forms = 2 * sum Q_k
        # H @ v computed via hessian_matvec
        r = -grad.copy()
        p = r.copy()
        rs_old = r @ r
        d = np.zeros(n)
        
        for cg_it in range(min(100, n)):
            Hp = hessian_matvec(p, active_set, n)
            # Project to tangent space
            Hp -= Hp.mean()
            alpha = rs_old / max(1e-30, p @ Hp)
            d += alpha * p
            r_new = r - alpha * Hp
            r_new -= r_new.mean()
            rs_new = r_new @ r_new
            if np.sqrt(rs_new) < 1e-12:
                break
            p = r_new + (rs_new / rs_old) * p
            p -= p.mean()
            rs_old = rs_new
        
        # Project d to tangent space
        d -= d.mean()
        
        # Line search: find t such that a + t*d is feasible and improves
        best_t = 0.0
        best_val = 2.0 * n * fast_convolve(a).max()
        for t_frac in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            t = t_frac * np.max(np.abs(d)) * 0.1
            a_trial = a + t * d
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            val = 2.0 * n * fast_convolve(a_trial).max()
            if val < best_val:
                best_val = val
                best_t = t
        
        if best_t > 0:
            a_new = a + best_t * d
            a_new = np.maximum(a_new, 0)
            st = a_new.sum()
            if st > 1e-15:
                a_new = a_new / st
                if best_val < best_score - 1e-15:
                    best_score = best_val
                    best_a = a_new.copy()
                    a = a_new
                else:
                    a = a_new
        
        # Check convergence
        if np.max(np.abs(d)) * np.max(np.abs(a)) < 1e-14:
            break
    
    return best_a


def peak_shave_adaptive(a0, time_limit, n_steps=100000):
    """Peak shaving with adaptive step size and backtracking."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return None, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 0.03
    no_improve = 0
    max_no_improve = 40

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Find top peaks
        threshold = max_b * (1.0 - 2e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        # Cap active set size for efficiency
        if len(active) > 50:
            part = np.argpartition(b, -50)[-50:]
            active = part

        # Compute subgradient
        g = np.zeros(n)
        for kk in active:
            js = max(0, kk - n + 1)
            je = min(n - 1, kk)
            idx = np.arange(js, je + 1)
            g[idx] += a[kk - idx]
        g /= len(active)

        # Multiplicative update with backtracking
        improved = False
        for scale_factor in [1.0, 0.5, 0.25, 0.1]:
            eta_eff = eta * scale_factor
            log_scale = -eta_eff * g
            log_scale -= log_scale.max()
            a_new = a * np.exp(log_scale)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            score = 2.0 * n * fast_convolve(a_new).max()
            if score < best_score - 1e-15:
                best_score = score
                best_a = a_new.copy()
                a = a_new
                improved = True
                break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                eta *= 0.7
                no_improve = 0
                if eta < 0.001:
                    break

    return best_a, best_score


def lp_alternating(a0, time_limit, solver, n_max_iters=500000):

            if avg_lp_time > 3.0:
                break

        # Nesterov dual averaging
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Stall detection
        if (it + 1) % 200 == 0:

                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                stall_count = 0

    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def perturbed_sequence(a, rng, strength=0.03):
    """Create a perturbed version of sequence a."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 10)
    phase = rng.uniform(0, 2 * np.pi)

    a_pert = np.maximum(a_pert, 0)

    if s < 1e-15:
        return a.copy()

    rem = deadline - time.monotonic()
    t_lbfgs = min(rem * 0.22, 350.0)
    if t_lbfgs > 30:
        print(f"  Budget: {t_lbfgs:.0f}s")
        a_lb = lbgfs_refine_improved(best_seq, t_lbfgs,
                                     deadline=time.monotonic() + t_lbfgs)
        local_update(a_lb, "lbgfs_improved")

    # === Phase 4: Active-Set Newton (150s) ===
    print(f"=== Phase 4: Active-Set Newton (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 50:
        t_newton = min(rem * 0.10, 150.0)
        # Identify active peaks
        b = fast_convolve(best_seq)
        max_b = b.max()
        threshold = max_b * (1.0 - 1e-6)
        active_set = list(np.where(b >= threshold)[0])
        if not active_set:
            active_set = [int(np.argmax(b))]
        print(f"  Active set size: {len(active_set)}, budget={t_newton:.0f}s")
        a_newton = newton_active_set(best_seq, active_set, t_newton, n_max_iters=2000)
        local_update(a_newton, "newton")
        
        # Try with slightly larger active set
        threshold2 = max_b * (1.0 - 5e-4)
        active_set2 = list(np.where(b >= threshold2)[0])
        if len(active_set2) < 30:
            t_newton2 = min(deadline - time.monotonic(), 30.0)
            if t_newton2 > 10:
                a_newton2 = newton_active_set(best_seq, active_set2, t_newton2, n_max_iters=1000)
                local_update(a_newton2, "newton_wide")

    # === Phase 5: Peak Shaving Adaptive (100s) ===
    print(f"=== Phase 5: Peak Shaving (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_peak = min(rem * 0.07, 100.0)
        a_pk, _ = peak_shave_adaptive(best_seq, t_peak, n_steps=100000)
        local_update(a_pk, "peak_shave")

    # === Phase 6: Second LP round (150s) ===
    print(f"=== Phase 6: LP Round 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t_lp2 = min(rem * 0.10, 150.0)
        a_lp2, _ = lp_alternating(best_seq, t_lp2, solver)
        local_update(a_lp2, "lp_2")

    # === Phase 7: Perturbation + re-optimize ===
    print(f"=== Phase 7: Perturbation (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        for trial in range(3):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 40:
                break
            strength = 0.01 + 0.01 * trial
            a_pert = perturbed_sequence(best_seq, rng, strength=strength)
            a_sp = spectral_flatten(a_pert, n_iters=20, blend=0.3)
            t_sub = min(rem_sub * 0.2, 50.0)
            if t_sub > 15:
                # Quick LP
                a_opt, _ = lp_alternating(a_sp, t_sub * 0.6, solver)
                local_update(a_opt, f"pert_lp{trial}")
                # Quick L-BFGS
                rem_sub2 = deadline - time.monotonic()
                if rem_sub2 > 20:
                    a_lb_p = lbgfs_refine_improved(a_opt, min(rem_sub2, 30.0),
                                                   deadline=time.monotonic() + min(rem_sub2, 30.0))
                    local_update(a_lb_p, f"pert_lb{trial}")

    # === Phase 8: Try different n ===
    print(f"=== Phase 8: Different n (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    for n_try in [1000, 1500, 2000]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        if n_try == n_best:
            continue
        t_n = min(rem * 0.12, 120.0)
        if t_n < 30:
            continue
        print(f"  Trying n={n_try}, budget={t_n:.0f}s")
        init_n = interpolate_sequence(best_seq, n_try)
        a_sp_n = spectral_flatten(init_n, n_iters=30, blend=0.3)
        a_n, _ = lp_alternating(a_sp_n, t_n * 0.5, solver)
        local_update(a_n, f"lp_n{n_try}")
        rem = deadline - time.monotonic()
        if best_seq is not None and len(best_seq) == n_try and rem > 30:
            a_n_lb = lbgfs_refine_improved(best_seq, min(rem, 60.0),
                                           deadline=time.monotonic() + min(rem, 60.0))
            local_update(a_n_lb, f"lbgfs_n{n_try}")

    # === Phase 9: Final Polish ===
    print(f"=== Phase 9: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        # L-BFGS final with very low temps
        t_fin = min(rem * 0.3, 60.0)
        if t_fin > 10:
            a_fin = lbgfs_refine_improved(best_seq, t_fin,
                                          deadline=time.monotonic() + t_fin)
            local_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 15:
        # Peak shave final
        t_ps = min(rem * 0.3, 30.0)
        if t_ps > 5:
            a_ps, _ = peak_shave_adaptive(best_seq, t_ps, n_steps=50000)
            local_update(a_ps, "final_peak")

    rem = deadline - time.monotonic()
    if rem > 15:
        # Final Newton
        b_fin = fast_convolve(best_seq)
        max_b_fin = b_fin.max()
        thresh_fin = max_b_fin * (1.0 - 1e-7)
        active_fin = list(np.where(b_fin >= thresh_fin)[0])
        if not active_fin:
            active_fin = [int(np.argmax(b_fin))]
        t_fn = min(rem * 0.4, 30.0)
        if t_fn > 5:
            a_fn = newton_active_set(best_seq, active_fin, t_fn, n_max_iters=1000)
            local_update(a_fn, "final_newton")

    rem = deadline - time.monotonic()
    if rem > 10:
        # Final LP
        t_flp = min(rem * 0.5, 20.0)
        if t_flp > 5:
            a_flp, _ = lp_alternating(best_seq, t_flp, solver)
            local_update(a_flp, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_start
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}) at {elapsed:.1f}s")
        test_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        try:
            eval_score = evaluate_sequence(test_seq)
            print(f"Verified with evaluate_sequence: {eval_score:.8f}")
        except Exception as e:
            print(f"Verification error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
