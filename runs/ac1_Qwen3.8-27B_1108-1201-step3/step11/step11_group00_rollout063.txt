"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. FFT-based multiplicative weights (mirror descent) with temperature annealing
2. Multi-length exploration via interpolation
3. Targeted peak shaving coordinate descent for final polish
Key: exploit non-symmetric structures where center of convolution is reduced.
"""

import numpy as np
import time


def fast_conv(a):
    """Compute np.convolve(a, a) using FFT. O(n log n)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def smooth_max_grad(a, T):
    """
    Compute (1/T)*log(sum_k exp(T*c[k])) and gradient w.r.t. a.
    c[k] = convolve(a,a)[k]. Gradient: grad[i] = 2*sum_m a[m]*w[i+m].
    Returns (value, gradient, weights).
    """
    n = len(a)
    c = fast_conv(a)
    Tc = T * c
    maxTc = Tc.max()
    expTc = np.exp(Tc - maxTc)
    log_sum = maxTc + np.log(expTc.sum())
    val = log_sum / T
    w = expTc / expTc.sum()
    # Cross-correlation via FFT: h[i] = sum_m a[m]*w[i+m]
    N = 1 << (3 * n - 2 - 1).bit_length()
    A_f = np.fft.rfft(a, N)
    W_f = np.fft.rfft(w, N)
    h_full = np.fft.irfft(np.conj(A_f) * W_f, N)
    h = h_full[0:n]
    grad = 2.0 * h
    return val, grad, w


def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s


def score(a):
    """Compute exact objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * len(a) * fast_conv(a).max() / (s * s)


def score_normalized(a):
    """Score assuming sum=1: 2*n*max(conv(a,a))."""
    c = fast_conv(a)
    return 2.0 * len(a) * c.max()


def mirror_descent(a0, time_limit, T_start=200.0, T_end=5.0, lr_scale=1.0,
                   use_avg=True, avg_start=100):
    """
    Multiplicative weights (mirror descent) with temperature annealing.
    a_new = normalize(a * exp(-eta_t * grad)), eta_t = lr_scale/sqrt(t).
    Tracks running average for improved solutions.
    """
    n = len(a0)
    a = normalize(a0.copy())
    t0 = time.monotonic()
    best_a = a.copy()
    best_sc = score(a)
    avg = np.zeros(n)
    cnt = 0
    best_avg_sc = best_sc
    best_avg_a = a.copy()

    for t in range(1, 50_000_000):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        frac = min(1.0, elapsed / max(time_limit, 1e-9))
        T = T_start * (T_end / T_start) ** frac
        eta = lr_scale / np.sqrt(t)

        val, grad, w = smooth_max_grad(a, T)
        a = normalize(a * np.exp(-eta * grad))
        avg += a
        cnt += 1

        if t % 200 == 0:
            sc = score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
            if use_avg and cnt >= avg_start:
                sc_avg = score(avg / cnt)
                if sc_avg < best_avg_sc:
                    best_avg_sc = sc_avg
                    best_avg_a = (avg / cnt).copy()

    if use_avg and cnt >= avg_start:
        sc_avg = score(avg / cnt)
        if sc_avg < best_sc:
            best_sc = sc_avg
            best_a = (avg / cnt).copy()

    return best_a, best_sc


def peak_shave(a0, time_limit, seed=42):
    """
    Coordinate descent: move mass from high-contributing positions to
    low-impact positions to reduce the convolution peak.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_sc = score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        if stall > 800:
            break
        c = fast_conv(a)
        top_k = np.argsort(c)[::-1][:12]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            src = idx[np.argmax(contrib)]

            # Collect peak areas
            peak_area = set()
            for kp in top_k:
                js2 = max(0, kp - n + 1)
                je2 = min(n - 1, kp)
                peak_area.update(range(js2, je2 + 1))

            candidates = [j for j in range(n) if j not in peak_area]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            if not candidates:
                continue
            candidates.sort(key=lambda j: a[j])
            dst = candidates[rng.integers(max(1, len(candidates) // 5))]

            for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.06]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_try = a.copy()
                a_try[src] -= amt
                a_try[dst] += amt
                a_try = normalize(a_try)
                sc = score(a_try)
                if sc < best_sc - 1e-12:
                    a = a_try
                    best_sc = sc
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a_try = a.copy()
                a_try[i] -= amt
                a_try[j] += amt
                a_try = normalize(a_try)
                sc = score(a_try)
                if sc < best_sc:
                    a = a_try
                    best_sc = sc
                    best_a = a.copy()
                    stall = 0

    return best_a, best_sc


def interpolate(a, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a_norm = a / a.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    return normalize(a_new)
