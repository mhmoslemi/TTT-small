import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def nesterov_mw(a0, time_limit, top_k=10, eta0=0.05):
    """
    Nesterov-accelerated multiplicative weights for minimizing max(conv(a,a))
    with the constraint sum(a)=1, a>=0.
    Uses top-k peak smoothing for better gradients.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    a_prev = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    t0 = time.monotonic()
    eta = eta0
    
    for it in range(10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov extrapolation (in the primal, as geometric mean)
        # y = a^2 / a_prev (geometric Nesterov for multiplicative weights)
        # Or simply: y = a + beta*(a - a_prev) in linear space
        beta = (1 + np.sqrt(1 + 4.0 * it * it)) / (2.0 * (it + 1))
        # For large it, beta ≈ 1 - 1/(2*it)
        y = a + beta * (a - a_prev)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = a.sum()
        y = y / sy
        
        a_prev = a.copy()
        
        # Compute convolution and find top-k peaks
        c = fast_convolve(y)
        
        # Top-k peaks for smoother gradient
        if top_k > 1 and len(c) > top_k:
            # Use softmax weighting over top peaks
            top_vals = np.sort(c)[::-1][:top_k]
            threshold = top_vals[-1] * 0.999
            active = np.where(c >= threshold)[0]
            if len(active) > 50:
                # Just use the top 50
                top_idx = np.argsort(c)[::-1][:50]
                weights = c[top_idx]
                weights = weights / weights.sum()
            else:
                top_idx = active
                weights = c[active]
                weights = weights / weights.sum()
            
            grad = np.zeros(n)
            for k, w in zip(top_idx, weights):
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad[idx] += w * y[k - idx]
        else:
            k_star = np.argmax(c)
            grad = np.zeros(n)
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            idx = np.arange(js, je + 1)
            grad[idx] = y[k_star - idx]
        
        # Multiplicative update
        log_mult = -eta * grad
        max_lm = log_mult.max()
        if max_lm > 700:  # overflow protection
            log_mult = log_mult - max_lm
        a_new = a * np.exp(log_mult)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        # Evaluate
        c_new = fast_convolve(a_new)
        score = 2.0 * n * c_new.max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        
        # Decay eta slowly
        if it % 5000 == 0 and it > 0:
            eta *= 0.995
        
        if it % 50000 == 0 and it > 0:
            elapsed = time.monotonic() - t0
            print(f"    NMW iter={it} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Ultra-fast Nesterov-accelerated multiplicative weights with top-K peak smoothing
2. FFT-based convolution (O(n log n) per iteration)
3. Multi-length exploration with structured asymmetric initializations
4. LogSumExp-smoothed gradient for better convergence
5. Final LP polish with Nesterov dual averaging
Key idea: exploit non-palindromic structure to reduce convolution peak below
the ||a||_2^2 barrier, using millions of fast iterations instead of expensive LPs.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Fast local evaluation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def structured_init(n, rng, mode=0):
    """Create diverse structured initial sequences emphasizing asymmetry."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        # Exponential ramp (lopsided)
        a = np.exp(2.0 * x)
    elif mode == 1:
        # Inverse exponential ramp
        a = np.exp(-2.0 * x)
    elif mode == 2:
        # Sigmoid centered off-center
        a = 1.0 / (1.0 + np.exp(-8.0 * (x - 0.3)))
    elif mode == 3:
        # Sigmoid centered other side
        a = 1.0 / (1.0 + np.exp(-8.0 * (x - 0.7)))
    elif mode == 4:
        # Polynomial ramp
        a = (1.0 + 3.0 * x) ** 2
    elif mode == 5:
        # Two-bump asymmetric
        a = np.exp(-20.0 * (x - 0.2) ** 2) + 0.5 * np.exp(-20.0 * (x - 0.8) ** 2)
    elif mode == 6:
        # Gaussian centered off-center
        a = np.exp(-4.0 * (x - 0.25) ** 2)
    elif mode == 7:
        # Cosine window off-center
        a = np.cos(np.pi * (x - 0.3)) ** 2
        a = np.maximum(a, 0)
    elif mode == 8:
        # Power law
        a = (x + 0.1) ** 3
    elif mode == 9:
        # Near-uniform with slight asymmetry
        a = 1.0 + 0.3 * x
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def nesterov_mw_fast(a0, time_limit, top_k=5, eta0=0.03, lse_temp=None):
    """
    Ultra-fast Nesterov-accelerated multiplicative weights.
    Each iteration: one FFT (O(n log n)) + O(n) updates.
    Uses top-K peak or LSE smoothing for gradient.
    """
    n = len(a0)
    a = normalize(a0.copy())
    a_prev = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0
    it = 0

    # Precompute FFT size
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    while time.monotonic() - t0 < time_limit:
        # Nesterov extrapolation
        if it == 0:
            y = a.copy()
        else:
            beta = (it + 1) / (it + 2)
            y = a + beta * (a - a_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y = y / sy

        a_prev = a.copy()

        # Compute convolution
        A = rfft(y, fft_size)
        c = irfft(A * A, fft_size)[:size]

        # Compute gradient using top-K or LSE
        if lse_temp is not None and lse_temp > 0:
            # LSE smoothing
            c_max = c.max()
            scaled = (c - c_max) / lse_temp
            scaled_max = scaled.max()
            exp_c = np.exp(scaled - scaled_max)
            w = exp_c / exp_c.sum()
            # Gradient: grad_j = sum_k w[k] * y[k-j] for valid k-j
            # This is cross-correlation of w and y
            grad = np.zeros(n)
            for k in range(size):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad[idx] += w[k] * y[k - idx]
        else:
            # Top-K peaks
            if top_k >= size:
                top_idx = np.argsort(c)[::-1]
            else:
                top_idx = np.argsort(c)[::-1][:min(top_k, size)]
            weights = c[top_idx]
            if weights.sum() < 1e-30:
                weights = np.ones(len(top_idx))
            weights = weights / weights.sum()

            grad = np.zeros(n)
            for k, wv in zip(top_idx, weights):
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad[idx] += wv * y[k - idx]

        # Multiplicative update (mirror descent step)
        log_mult = -eta * grad
        max_lm = log_mult.max()
        if max_lm > 500:
            log_mult -= max_lm
        elif max_lm < -500:
            log_mult += 500
        a_new = a * np.exp(log_mult)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        # Evaluate
        c_new = fast_convolve(a_new)
        score = 2.0 * n * c_new.max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        a = a_new
        it += 1

        # Adaptive eta decay
        if it % 10000 == 0 and it > 0:
            eta *= 0.99
            if lse_temp is not None and lse_temp > 0.01:
                lse_temp *= 0.95

        if it % 200000 == 0 and it > 0:
            elapsed = time.monotonic() - t0
            print(f"      NMW it={it} ({elapsed:.0f}s): score={best_score:.8f} eta={eta:.6f}")

    return best_a, best_score


def nesterov_mw_single_peak(a0, time_limit, eta0=0.02):
    """
    Fastest variant: single peak subgradient (no top-K).
    O(n log n) per iteration, minimal constant.
    """
    n = len(a0)
    a = normalize(a0.copy())
    a_prev = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    it = 0

    while time.monotonic() - t0 < time_limit:
        if it == 0:
            y = a.copy()
        else:
            beta = (it + 1) / (it + 2)
            y = a + beta * (a - a_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y = y / sy

        a_prev = a.copy()

        # FFT convolution
        A = rfft(y, fft_size)
        c = irfft(A * A, fft_size)[:size]
        k_star = np.argmax(c)

        # Single-peak subgradient
        grad = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        grad[idx] = y[k_star - idx]

        # Multiplicative update
        log_mult = -eta * grad
        max_lm = log_mult.max()
        if max_lm > 500:
            log_mult -= max_lm
        a_new = a * np.exp(log_mult)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        # Evaluate (can reuse c if k_star is still the max, but checking is cheap)
        c_new = fast_convolve(a_new)
        score = 2.0 * n * c_new.max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        a = a_new
        it += 1

        if it % 50000 == 0 and it > 0:
            eta *= 0.98
            elapsed = time.monotonic() - t0
            print(f"      NMW-SP it={it} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def build_conv_matrix_sparse(a):
    """Build sparse matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    total_entries = 0
    for j in range(n):
        total_entries += min(j + n, size) - j
    rows = np.zeros(total_entries, dtype=np.int32)
    cols = np.zeros(total_entries, dtype=np.int32)
    vals = np.zeros(total_entries, dtype=np.float64)
    count = 0
    for j in range(n):
        for k in range(j, min(j + n, size)):
            rows[count] = k
            cols[count] = j
            vals[count] = a[k - j]
            count += 1
    return csr_matrix((vals[:count], (rows[:count], cols[:count])), shape=(size, n))


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return None


def solve_lp_best_response(a_fixed, solver, T_mat):
    """LP: find b minimizing max_k conv(a,b)_k subject to sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = [s for s in [solver, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS] if s is not None]
    for s in solvers_to_try:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                ssum = b_val.sum()
                if ssum < 1e-15:
                    return None, None
                return b_val / ssum, float(max(0, M_var.value))
        except Exception:
            continue
    return None, None


def lp_polish(a0, time_limit, solver, n_max_iters=200):
    """LP-based Nesterov dual averaging polish (few iterations)."""
    n = len(a0)
    a = normalize(a0.copy())
    y = a.copy()
    T_mat = build_conv_matrix_sparse(y)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lp_times = []

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver, T_mat)
        lp_times.append(time.monotonic() - lp_t0)
        if b_new is None:
            break
        if len(lp_times) > 5:
            avg_lp = np.mean(lp_times[-5:])
            if avg_lp > 5.0:
                print(f"      LP too slow ({avg_lp:.2f}s), stopping.")
                break

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = a.sum()
        y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        T_mat = build_conv_matrix_sparse(y)

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def perturb_sequence(a, rng, strength=0.02):
    """Structured low-frequency perturbation."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq1 = rng.integers(1, 6)
    freq2 = rng.integers(2, 10)
    p1 = rng.uniform(0, 2 * np.pi)
    p2 = rng.uniform(0, 2 * np.pi)
    pert = strength * (np.cos(freq1 * x + p1) + 0.5 * np.sin(freq2 * x + p2))
    a_pert = np.maximum(a + pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s

            a4b, _ = nesterov_mw_fast(best_a, t3b, top_k=10, eta0=0.01)
            try_update(a4b, "nmw_refine_k10")
        rem = deadline - time.monotonic()
        if rem > 30:
            t3c = min(rem * 0.3, t3 * 0.3)
            a4c, _ = nesterov_mw_single_peak(best_a, t3c, eta0=0.015)
            try_update(a4c, "nmw_refine_sp")

    # === Phase 4: LP polish ===
    print(f"\n=== Phase 4: LP polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30 and solver is not None:
        t4 = min(rem * 0.2, 180.0)
        print(f"  Budget={t4:.0f}s")
        a5, _ = lp_polish(best_a, t4, solver, n_max_iters=100)
        try_update(a5, "lp_polish")
    elif rem > 20 and solver is None:
        # No LP solver, do more NMW
        t4 = min(rem * 0.3, 120.0)
        a5, _ = nesterov_mw_fast(best_a, t4, top_k=5, eta0=0.01)
        try_update(a5, "nmw_extra")

    # === Phase 5: Perturbation escape ===
    print(f"\n=== Phase 5: Perturbation escape (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        for trial in range(5):
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            strength = 0.01 + 0.015 * trial
            a_pert = perturb_sequence(best_a, rng, strength=strength)
            t_sub = min(rem * 0.2, 60.0)
            if t_sub < 10:
                break
            a6, _ = nesterov_mw_fast(a_pert, t_sub, top_k=5, eta0=0.02)
            try_update(a6, f"pert{trial}")

    # === Phase 6: Final fast polish ===
    print(f"\n=== Phase 6: Final polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t6 = min(rem * 0.5, 60.0)
        a7, _ = nesterov_mw_single_peak(best_a, t6, eta0=0.005)
        try_update(a7, "final_sp")

    # Final verification
    if best_a is not None:
        fs = evaluate_local(best_a)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} n={len(best_a)}")
    else:
        best_a = np.ones(1200) / 1200
        best_score = evaluate_local(best_a)
        print(f"\nFINAL (fallback): score={best_score:.8f} n={len(best_a)}")

    return [float(max(0.0, min(1000.0, x))) for x in best_a]
