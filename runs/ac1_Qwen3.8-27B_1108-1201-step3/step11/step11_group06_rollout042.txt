The min-max problem is:
```
min_a max_k (a*a)[k]  s.t. a >= 0, sum a = 1
```

The "value" function is `V(a) = max_k (a*a)[k]`. The LP for the best response to a fixed `y` is:
```
min_b max_k (y*b)[k]  s.t. b >= 0, sum b = 1
```

Note: `(y*b)[k] = sum_j y[k-j] * b[j]` (cross-correlation, not auto-correlation).

The Nesterov scheme maintains two sequences:
- `y_t` (the "dual" or "ghost" sequence, used in the LP)
- `a_t` (the "primal" sequence, the output)

The update is:
```
b_{t+1} = argmin LP(y_t)
y_{t+1} = (1 - α_{t+1}) * y_t + α_{t+1} * b_{t+1}
a_{t+1} = (sum_{s=1}^{t+1} α_s * b_s) / (sum_{s=1}^{t+1} α_s)
```

With `α_t = 2/(t+2)`.

The convergence guarantee is: `V(a_t) - V* ≤ C / t` where C depends on the problem.

def get_lp_solver():
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name

    return cp.ECOS


def solve_active_lp(y, active_idx, solver, b_init=None):
    n = len(y)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build the constraint matrix for active indices only
    T = build_active_conv_matrix(y, active_idx)
    # Extract only the active rows
    T_active = T[active_idx, :]
    
    constraints = [T_active @ b_var <= M_var, cp.sum(b_var) == 1]
    
    if b_init is not None:
        b_var.value = b_init
        # M_var.value = max(fast_cross_convolve(y, b_init))
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    for s in [solver, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)

                    return None, None
                b_val = b_val / s_val
                return b_val, float(max(0, M_var.value))

            continue
    return None, None


def multiplicative_weights(a0, time_limit, n_steps=500000, eta0=0.05):

def nesterov_lp_active(a0, time_limit, solver, K_init=50, K_max=None, 
                        n_max_iters=1000000):

    n = len(a0)
    if K_max is None:
        K_max = 2 * n - 1
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    b_prev = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(12345)
    
    K = K_init
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute convolution to find active set
        c = fast_cross_convolve(y, b_prev)

        if K < 2 * n - 1:
            top_idx = np.argpartition(c, -K)[-K:]
        else:
            top_idx = np.arange(2 * n - 1)

        b_new, M_b = solve_active_lp(y, top_idx, solver, b_prev)
        if b_new is None:
            b_new = b_prev
            continue
        
        # Nesterov update
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = b_prev.copy()
        else:
            y = y / sy
        
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        b_prev = b_new

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

            K = min(K + 20, K_max)
        
        # Stall detection and escape

                stall_count += 1

                b_prev = y.copy()
                stall_count = 0
                K = max(K_init, K - 20)  # Reset K to smaller
            if (it + 1) % 1000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} K={K}")
    
    return best_a, best_score


def peak_shave(a, n_shaves=10):
    """Peak shaving heuristic: reduce the highest convolution peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    
    for _ in range(n_shaves):
        b = fast_convolve(a)
        k_star = np.argmax(b)
        peak_val = b[k_star]
        
        if peak_val <= b.mean() * 1.5:
            break
        
        # Find the indices contributing to the peak
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        contributions = a[idx] * a[k_star - idx]
        
        # Find the largest contributing pair
        max_contrib_idx = np.argmax(contributions)
        i = idx[max_contrib_idx]
        j = k_star - i
        
        # Reduce the larger of a[i], a[j] and add to a random position
        if a[i] >= a[j]:
            reduce_amt = 0.01 * a[i]
            a[i] -= reduce_amt
        else:
            reduce_amt = 0.01 * a[j]
            a[j] -= reduce_amt
        
        # Add to a random position away from the peak
        candidates = np.where(b < peak_val * 0.5)[0]
        if len(candidates) > 0:
            # Find a position where we can add mass
            # (i.e., a position where the convolution is low)
            target = rng.randint(0, n)  # Use global rng
            a[target] += reduce_amt
        
        a = np.maximum(a, 0)
        a = a / a.sum()
    
    return a


def lse_gradient(a, T):
    """Compute LSE objective value and gradient."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T  # shift for numerical stability
    scaled = scaled - scaled.max()
    exp_scaled = np.exp(scaled)
    lse_val = max_b + T * (0 + np.log(exp_scaled.sum() + 1e-300))
    obj = 2.0 * n * lse_val
    
    w = exp_scaled / exp_scaled.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    
    return obj, grad_a

    s = a.sum()
    if s < 1e-15:

    stall_count = 0
    rng = np.random.default_rng(54321)
    K = K_init
    lp_times = []
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        c = fast_cross_convolve(y, b_prev)
        use_full = (full_lp_every > 0 and (it + 1) % full_lp_every == 0)
        if use_full or K >= 2 * n - 1:
            top_idx = np.arange(2 * n - 1)
        elif K < 2 * n - 1:
            K_use = min(K, 2 * n - 1)
            top_idx = np.argpartition(c, -K_use)[-K_use:]
        else:
            top_idx = np.arange(2 * n - 1)
        lp_start = time.monotonic()
        b_new, M_b = solve_active_lp(y, top_idx, solver, b_prev)
        lp_time = time.monotonic() - lp_start
        lp_times.append(lp_time)
        if b_new is None:
            b_new = b_prev.copy()
        if len(lp_times) > 50:
            avg_lp = np.mean(lp_times[-50:])
            if avg_lp > 2.0 and K > K_init:
                K = max(K_init, K - 10)
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = b_prev.copy()
        else:
            y = y / sy
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        b_prev = b_new
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if not use_full:
            score_b = 2.0 * n * fast_convolve(b_new).max()
            b_sum = b_new.sum()
            if b_sum > 1e-15:
                score_b_norm = 2.0 * n * fast_convolve(b_new / b_new.sum()).max()
                if score_b_norm < best_score:
                    best_score = score_b_norm
                    best_a = (b_new / b_sum).copy()
        if (it + 1) % 200 == 0:
            K = min(K + 15, K_max)
        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            if stall_count >= 4:
                x_arr = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 12)
                phase = rng.uniform(0, 2 * np.pi)
                noise = 0.015 * np.cos(freq * x_arr + phase)
                y = np.maximum(y + noise, 0)
                y = y / y.sum()
                b_prev = y.copy()
                stall_count = 0
                K = max(K_init, K - 30)
            if (it + 1) % 2000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} K={K} avg_lp={np.mean(lp_times[-200:]):.4f}s")
    return best_a, best_score


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def lse_obj_grad(a, T):
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T
    scaled = scaled - scaled.max()
    exp_scaled = np.exp(scaled)
    lse_val = max_b + T * np.log(exp_scaled.sum() + 1e-300)
    obj = 2.0 * n * lse_val
    w = exp_scaled / exp_scaled.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    return obj, grad_a


class LSEObjective:
    def __init__(self, T):
        self.T = T

    def value(self, z):
        a = softmax_from_z(z)
        obj, _ = lse_obj_grad(a, self.T)
        return obj

    def gradient(self, z):
        a = softmax_from_z(z)
        _, g_a = lse_obj_grad(a, self.T)
        S = np.dot(g_a, a)
        return a * (g_a - S)


def lse_lbgfs(a0, time_limit, T_stages):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    z = np.log(np.maximum(a, 1e-300))
    t0 = time.monotonic()
    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        func = LSEObjective(T)
        try:
            res = minimize(
                fun=func.value, x0=z, jac=func.gradient, method='L-BFGS-B',
                options={'maxiter': 400, 'ftol': 1e-25,
                         'gtol': 1e-20, 'maxfun': 15000, 'disp': False}
            )
            if res.x is not None:
                z = res.x
        except Exception:
            pass
    return softmax_from_z(z)


def lse_fista(a0, time_limit, T):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    t0 = time.monotonic()
    t_mom = 1.0
    y = a.copy()
    x_prev = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    step = T / (2.0 * n * 0.15)
    for it in range(500000):
        if time.monotonic() - t0 > time_limit:
            break
        obj, grad = lse_obj_grad(y, T)
        x_new = y - step * grad
        x_new = np.maximum(x_new, 0)
        s_new = x_new.sum()
        if s_new < 1e-15:
            x_new = x_prev.copy()
        else:
            x_new = x_new / s_new
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y = x_new + (t_mom / t_new) * (x_new - x_prev)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = x_new.copy()
        else:
            y = y / sy
        t_mom = t_new
        x_prev = x_new
        if (it + 1) % 500 == 0:
            sc = 2.0 * n * fast_convolve(x_new).max()
            if sc < best_score:
                best_score = sc
                best_a = x_new.copy()
    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def make_palindromic(a):
    n = len(a)
    a_pal = (a + a[::-1]) / 2.0
    a_pal = np.maximum(a_pal, 0)
    s = a_pal.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a_pal / s


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 5 * x + 0.5)
        a = np.maximum(a, 0)
    else:
        a = 1.0 / (1.0 + 3.0 * np.abs(x - 0.5))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def peak_shave(a, n_shaves=20, strength=0.008):
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    rng_local = np.random.default_rng(999)
    for _ in range(n_shaves):
        b = fast_convolve(a)
        k_star = np.argmax(b)
        peak_val = b[k_star]
        if peak_val <= b.mean() * 1.3:
            break
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        contributions = a[idx] * a[k_star - idx]
        max_c_idx = np.argmax(contributions)
        i = idx[max_c_idx]
        j = k_star - i
        if a[i] >= a[j]:
            reduce_amt = strength * a[i]
            a[i] -= reduce_amt
        else:
            reduce_amt = strength * a[j]
            a[j] -= reduce_amt
        target = rng_local.integers(0, n)
        a[target] += reduce_amt
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
    return a


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}, budget: {budget_s:.0f}s")

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        a_sum = a.sum()
        if a_sum < 0.01:
            return
        a_norm = a / a_sum
        sc = evaluate_local(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a_norm)})")

    # Phase 1: Quick MW on parent
    print(f"\n=== Phase 1: MW polish n={len(best_seq)} score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.04, 50.0)
    if t1 > 10:
        a_mw, _ = multiplicative_weights(best_seq, t1, n_steps=500000, eta0=0.05)
        try_update(a_mw, "mw_init")

    # Phase 2: Active-set LP with Nesterov (main workhorse) on parent length
    print(f"\n=== Phase 2: Active-set LP n={len(best_seq)} score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.35, 550.0)
    if t2 > 60:
        print(f"  Budget: {t2:.0f}s, K_init=40, K_max=2n-1={2*len(best_seq)-1}")
        a_lp, _ = nesterov_lp_active(best_seq, t2, solver, K_init=40, K_max=None,
                                     n_max_iters=5000000, full_lp_every=300)
        try_update(a_lp, "lp_main")

    # Phase 3: Palindromic LP
    print(f"\n=== Phase 3: Palindromic LP score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        t3 = min(rem * 0.1, 150.0)
        a_pal = make_palindromic(best_seq)
        a_pal_lr, _ = nesterov_lp_active(a_pal, t3, solver, K_init=40, K_max=None,
                                         n_max_iters=2000000, full_lp_every=300)
        try_update(a_pal_lr, "pal_lp")

    # Phase 4: Multi-length exploration
    print(f"\n=== Phase 4: Multi-length score={best_score:.8f} ===")
    lengths_to_try = [1500, 2000, 1000]
    for ln in lengths_to_try:
        rem = deadline - time.monotonic()
        if rem < 120:
            break
        if len(best_seq) == ln:
            continue
        t_ln = min(rem * 0.18, 250.0)
        if t_ln < 40:
            continue
        print(f"  Trying n={ln}, budget={t_ln:.0f}s")
        init_ln = interpolate_sequence(best_seq, ln)
        # Quick MW
        a_ln_mw, _ = multiplicative_weights(init_ln, t_ln * 0.15, n_steps=100000, eta0=0.05)
        try_update(a_ln_mw, f"mw_n{ln}")
        # Active-set LP
        rem = deadline - time.monotonic()
        t_lp = min(rem * 0.6, t_ln * 0.75)
        if t_lp > 30:
            a_ln_lp, _ = nesterov_lp_active(a_ln_mw, t_lp, solver, K_init=40, K_max=None,
                                            n_max_iters=3000000, full_lp_every=300)
            try_update(a_ln_lp, f"lp_n{ln}")

    # Phase 5: LSE L-BFGS polish on best
    print(f"\n=== Phase 5: LSE L-BFGS score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t5 = min(rem * 0.2, 300.0)
        T_stages = [20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
        print(f"  {len(T_stages)} stages, budget={t5:.0f}s")
        a_lse = lse_lbgfs(best_seq, t5, T_stages)
        try_update(a_lse, "lse_bfgs")

    # Phase 6: FISTA LSE polish
    print(f"\n=== Phase 6: FISTA LSE score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t6 = min(rem * 0.15, 120.0)
        a_fista, _ = lse_fista(best_seq, t6, T=0.5)
        try_update(a_fista, "fista")

    # Phase 7: Peak shave + re-optimize
    print(f"\n=== Phase 7: Peak shave + LP score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        a_shaved = peak_shave(best_seq, n_shaves=30, strength=0.005)
        try_update(a_shaved, "shaved")
        t7 = min(rem * 0.3, 100.0)
        if t7 > 20:
            a_shaved_lp, _ = nesterov_lp_active(a_shaved, t7, solver, K_init=40, K_max=None,
                                                n_max_iters=1000000, full_lp_every=200)
            try_update(a_shaved_lp, "shaved_lp")

    # Phase 8: Final LP polish on absolute best
    print(f"\n=== Phase 8: Final LP score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t8 = min(rem * 0.4, 150.0)
        a_fin, _ = nesterov_lp_active(best_seq, t8, solver, K_init=50, K_max=None,
                                      n_max_iters=2000000, full_lp_every=200)
        try_update(a_fin, "final_lp")

    # Phase 9: Final MW polish
    print(f"\n=== Phase 9: Final MW score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t9 = min(rem * 0.5, 40.0)
        a_fin_mw, _ = multiplicative_weights(best_seq, t9, n_steps=200000, eta0=0.02)
        try_update(a_fin_mw, "final_mw")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} n={len(best_seq)}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
