Let me also think about whether there are any issues with the Nesterov method:

The Nesterov method for minimizing a convex function f over a convex set C works as:
- y_k = a_k + (t_{k-1}/t_k) * (a_k - a_{k-1})  [extrapolation]
- a_{k+1} = prox_{alpha_k * f}(y_k)  [proximal step]
- t_{k+1} = (1 + sqrt(1 + 4*t_k^2)) / 2

In our case, the "function" is f(b) = max_k (y*b)[k], and the "set" is the probability simplex. The proximal step is:
a_{k+1} = argmin_b { alpha * f(b) + (1/2)||b - y_k||^2 } : b in simplex }

But we're approximating this with: a_{k+1} = (1-alpha) * a_k + alpha * b_LP where b_LP = argmin max_k (y*b)[k] s.t. b in simplex.

This is the "mirror descent" or "follow the regularized leader" approach, which is valid for convex-concave games. The Nesterov acceleration should give O(1/k^2) convergence.

The step size alpha_k = 2/(t_k + 1) where t_k = (1 + sqrt(1 + 4*(k-1)^2)) / 2. For large k, alpha_k ≈ 2/k.

One issue: the LP solution b_LP depends on y (the extrapolated point), which changes each iteration. This is the standard "Nesterov accelerated gradient" applied to theLP-based best response.

        y_new = a_new + momentum * (a_new - a)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc = score_normalized(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > perturb_stuck:
            noise_scale = min(0.02, 0.002 * (1 + no_improve // perturb_stuck))
            noise = rng.standard_normal(n) * noise_scale
            a_pert = a + noise
            a_pert = np.maximum(a_pert, 0)
            sp = a_pert.sum()
            if sp > 1e-15:
                a_pert = a_pert / sp
                sc_pert = score_normalized(a_pert)
                if sc_pert < best_score:
                    best_score = sc_pert
                    best_a = a_pert.copy()
                y = a_pert.copy()
                a = a_pert.copy()
                no_improve = 0
            center = n - 1
            active_set = list(range(max(0, center - radius), min(2 * n - 1, center + radius + 1)))

        if (it + 1) % 1000 == 0:
            print(f"    NLP iter={it+1} ({elapsed:.1f}s): score={best_score:.8f} "
                  f"|act|={len(active_set)} M={M_lp:.6f} no_imp={no_improve}")

        if no_improve > 5000 and it > 2000:
            break

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    """Compute LSE-smoothed objective and gradient in log-space."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        shifted = Tb - 700
    else:
        shifted = Tb - max_Tb
    exp_Tb = np.exp(shifted)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = (max_Tb if max_Tb <= 700 else 700) + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_logsumexp(a0, time_limit, T_start=50.0, T_end=0.02, n_stages=20):
    """LSE-smoothed L-BFGS optimization in log-space with multi-stage annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        max_iter_stage = int(min(50, max(5, remaining * 30 / (len(temps) - si))))

        def func_staged(zv):
            obj, _ = lse_objective_grad(zv, stage_T, n)
            return obj

        def jac_staged(zv):
            _, grad = lse_objective_grad(zv, stage_T, n)
            return grad

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': max_iter_stage, 'ftol': 1e-22,
                         'gtol': 1e-18, 'maxfun': max_iter_stage * 500,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

        a_check = softmax_from_z(z)
        sc = score_normalized(a_check)
        if (si + 1) % 5 == 0 or si == n_stages - 1:
            print(f"    LSE stage={si+1}/{n_stages} T={stage_T:.3f}: score={sc:.8f}")

    return softmax_from_z(z)


def interpolate_sequence(f, new_n):
    """Resample a sequence to a new length while preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate a structured initial sequence."""
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-3 * x ** 2)
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x)
    elif choice == 4:
        a = np.ones(n) * (1.0 + 0.15 * np.cos(2 * np.pi * 2 * np.arange(n) / n))
    else:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.2 * np.cos(2 * x) + 0.1 * np.cos(5 * x) + 0.05 * np.cos(11 * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function: find the best sequence minimizing the objective."""
    global _global_deadline
    rng = np.random.default_rng(seed)
    _global_deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    total_budget = float(budget_s) - 30.0

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}, total budget: {total_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    n_base = 1200

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            remaining = _global_deadline - time.monotonic()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, {remaining:.0f}s left)")

    # === Phase 1: Nesterov LP with constraint generation (50% budget) ===
    print(f"\n=== Phase 1: Nesterov LP active-set at n={n_base} ===")
    t1 = 0.50 * total_budget
    remaining = _global_deadline - time.monotonic()
    t1 = min(t1, remaining)
    a_out, sc_out = nesterov_lp_active(best_seq, t1, solver, max_active=200)
    try_update(a_out, "nesterov_lp_p1")
    print(f"  Phase 1 done: best={best_score:.8f}")

    # === Phase 2: LSE L-BFGS polishing (15% budget) ===
    print(f"\n=== Phase 2: LSE L-BFGS polish ===")
    remaining = _global_deadline - time.monotonic()
    t2 = min(0.15 * total_budget, remaining * 0.6)
    if t2 > 10:
        a_lse = lbgfs_logsumexp(best_seq, t2, T_start=50.0, T_end=0.02, n_stages=20)
        try_update(a_lse, "lse_polish_p2")
    print(f"  Phase 2 done: best={best_score:.8f}")

    # === Phase 3: Second Nesterov LP round (15% budget) ===
    print(f"\n=== Phase 3: Nesterov LP round 2 ===")
    remaining = _global_deadline - time.monotonic()
    t3 = min(0.15 * total_budget, remaining * 0.6)
    if t3 > 10:
        a_out2, sc_out2 = nesterov_lp_active(best_seq, t3, solver, max_active=150,
                                             perturb_stuck=1500)
        try_update(a_out2, "nesterov_lp_p3")
    print(f"  Phase 3 done: best={best_score:.8f}")

    # === Phase 4: LSE fine-tune (10% budget) ===
    print(f"\n=== Phase 4: LSE fine-tune ===")
    remaining = _global_deadline - time.monotonic()
    t4 = min(0.10 * total_budget, remaining * 0.5)
    if t4 > 8:
        a_lse2 = lbgfs_logsumexp(best_seq, t4, T_start=10.0, T_end=0.01, n_stages=15)
        try_update(a_lse2, "lse_fine_p4")
    print(f"  Phase 4 done: best={best_score:.8f}")

    # === Phase 5: Length exploration (10% budget) ===
    print(f"\n=== Phase 5: Length exploration ===")
    remaining = _global_deadline - time.monotonic()
    t5_total = min(0.10 * total_budget, remaining * 0.7)
    if t5_total > 30:
        lengths_to_try = [n_base + 100, n_base - 100, n_base + 200, n_base + 50]
        for n_try in lengths_to_try:
            if n_try <= 0:
                continue
            remaining = _global_deadline - time.monotonic()
            if remaining < 15:
                break
            t5 = min(t5_total / len(lengths_to_try), remaining * 0.4)
            if t5 < 5:
                break
            print(f"  Trying n={n_try}, budget={t5:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            a_try, sc_try = nesterov_lp_active(init_try, t5, solver, max_active=200)
            try_update(a_try, f"len_{n_try}")
    print(f"  Phase 5 done: best={best_score:.8f}")

    # === Phase 6: Final Nesterov polish (remaining time) ===
    remaining = _global_deadline - time.monotonic()
    if remaining > 10:
        print(f"\n=== Phase 6: Final Nesterov polish ({remaining:.0f}s) ===")
        a_final, sc_final = nesterov_lp_active(best_seq, remaining * 0.8, solver,
                                              max_active=150, perturb_stuck=1000)
        try_update(a_final, "final_nesterov")
    print(f"  Phase 6 done: best={best_score:.8f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(x) for x in best_seq]
