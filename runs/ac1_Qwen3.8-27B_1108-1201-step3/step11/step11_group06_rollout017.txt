    But `(T^T y)_j = sum_k T[k,j] y_k = sum_k a[k-j] y_k = conv(a, y)[j]`.
    
    So the dual is: `max_y min_j conv(a, y)[j]` s.t. `sum(y) = 1, y >= 0`.
    
    This is a max-min problem! We want to find `y` that maximizes the minimum of `conv(a, y)`.
    
    This is the "bottleneck" problem. It's equivalent to:
    ```
    max t
    s.t. conv(a, y)[j] >= t for all j
         sum(y) = 1
         y >= 0
    ```
    
    This is also a LP! And the constraint `conv(a, y)[j] >= t` means `sum_k a[k-j] y_k >= t`, which is linear in `(y, t)`.
    
    So both the primal and dual are LPs. The primal minimizes the max, the dual maximizes the min.
    
    The dual might be easier to solve if it has fewer constraints. The primal has 2n-1 + 1 + n = 3n-1 constraints and n+1 variables. The dual has n + 1 + (2n-1) = 3n-1 constraints and (2n-1) + 1 = 2n variables. For large n, the dual has more variables but the same number of constraints. So the primal is preferable (fewer variables).

def solve_inner_lp_hiGHS(a, T, method_ipm=True):

def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    ss = exp_z.sum()
    if ss < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / ss


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))

        j_end = min(n, k + 1)
        if j_end > j_start:
            idx = k - np.arange(j_start, j_end)
            grad_a[j_start:j_end] += w[k] * a[idx]
    grad_a *= 2.0
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


class LSEFunc:
    def __init__(self, n, T, deadline):
        self.n = n
        self.T = T
        self.deadline = deadline

    def __call__(self, z):
        if time.monotonic() > self.deadline:
            return 1e15
        obj, _ = lse_obj_and_grad(z, self.n, self.T, self.deadline)
        return obj


class LSEJac:
    def __init__(self, n, T, deadline):
        self.n = n
        self.T = T
        self.deadline = deadline

    def __call__(self, z):
        if time.monotonic() > self.deadline:
            return np.zeros(len(z))
        _, g = lse_obj_and_grad(z, self.n, self.T, self.deadline)
        return g


def lbgfs_refine(a0, time_limit, T_stages=None, maxiter_per_stage=400):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    if T_stages is None:
        T_stages = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        func = LSEFunc(n, stage_T, deadline)
        jac = LSEJac(n, stage_T, deadline)
        stage_time = min(remaining * 0.8, time_limit / len(T_stages) * 1.5)
        stage_deadline = time.monotonic() + stage_time
        func.deadline = stage_deadline
        jac.deadline = stage_deadline
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-30,
                         'gtol': 1e-20, 'maxfun': maxiter_per_stage * 100,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-6 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.05
    elif choice == 5:
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 7 * x + 0.3)
        a = np.maximum(a, 0)
    elif choice == 7:
        a = np.exp(-2 * (x - 0.5) ** 2) + 0.3 * np.exp(-20 * (x - 0.1) ** 2) + 0.3 * np.exp(-20 * (x - 0.9) ** 2)
    elif choice == 8:
        # Chirp-like: magnitude of a complex chirp projected to non-negative
        t = np.arange(n)
        chirp = np.exp(1j * np.pi * t * t / n)
        a = np.abs(np.fft.fft(chirp)[:n].real) + 0.1
    else:
        a = 1.0 / (1.0 + 5.0 * np.abs(x - 0.5))
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Fast local evaluation (assumes a >= 0, sum > 0)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def perturb_sequence(a, rng, strength=0.02):
    """Create a perturbed version with structured frequency components."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq1 = rng.integers(1, 12)
    freq2 = rng.integers(2, 20)
    p1 = rng.uniform(0, 2 * np.pi)
    p2 = rng.uniform(0, 2 * np.pi)
    pert = strength * (np.cos(freq1 * x + p1) + 0.5 * np.sin(freq2 * x + p2))
    a_pert = np.maximum(a + pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def peak_shave_lp(a0, time_limit, n_max_iters=20000):
    """Peak shaving using LP with peak-constrained objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        k_star = int(b.argmax())
        peak_val = b.max()

        # Solve LP: min max_k conv(y, a)[k] s.t. y in simplex
        # But focus on reducing the peak by constraining near the peak
        T = build_toeplitz_dense(a)
        y_new = solve_inner_lp_hiGHS(a, T, method_ipm=True)
        if y_new is None:
            continue

        # Blend
        alpha = 1.0 / np.sqrt(it + 1)
        a_new = (1.0 - alpha) * a + alpha * y_new
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new

        if (it + 1) % 500 == 0:
            print(f"    PeakShave iter={it+1}: score={best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses fast HiGHS LP, MW, LSE L-BFGS, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    best_seq = None
    best_score = float('inf')

    print(f"=== Starting optimization, budget: {budget_s:.0f}s ===")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: LP on parent ===
    print(f"\n=== Phase 1: HiGHS LP n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.20, 300.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        a_lp, _ = nesterov_lp_fast(best_seq, t1, n_max_iters=200000, print_every=200)
        try_update(a_lp, "lp_1")

    # === Phase 2: MW refinement ===
    print(f"\n=== Phase 2: MW (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.08, 100.0)
    if t2 > 15:
        a_mw, _ = multiplicative_weights(best_seq, t2, eta0=0.03)
        try_update(a_mw, "mw_1")

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length (score={best_score:.8f}) ===")
    lengths_to_try = [1500, 2000, 2500]
    for ln in lengths_to_try:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        if len(best_seq) == ln:
            continue
        t_ln = min(rem * 0.18, 250.0)
        if t_ln < 30:
            continue
        print(f"  Trying n={ln}, budget={t_ln:.0f}s")
        init_ln = interpolate_sequence(best_seq, ln)

        # Quick MW
        t_mw = t_ln * 0.15
        a_ln_mw, _ = multiplicative_weights(init_ln, t_mw, eta0=0.04)
        try_update(a_ln_mw, f"mw_n{ln}")

        # LP
        rem = deadline - time.monotonic()
        t_lp = min(rem * 0.4, t_ln * 0.7)
        if t_lp > 30:
            start_a = a_ln_mw if a_ln_mw is not None else init_ln
            a_ln_lp, _ = nesterov_lp_fast(start_a, t_lp, n_max_iters=100000, print_every=300)
            try_update(a_ln_lp, f"lp_n{ln}")

    # === Phase 4: Back to best length, more LP ===
    print(f"\n=== Phase 4: Extended LP n={len(best_seq)} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        t4 = min(rem * 0.20, 250.0)
        a_lp2, _ = nesterov_lp_fast(best_seq, t4, n_max_iters=200000, print_every=200)
        try_update(a_lp2, "lp_2")

    # === Phase 5: LSE L-BFGS polish ===
    print(f"\n=== Phase 5: LSE L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t5 = min(rem * 0.20, 250.0)
        T_stages = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
        print(f"  {len(T_stages)} stages, budget={t5:.0f}s")
        a_lse = lbgfs_refine(best_seq, t5, T_stages, maxiter_per_stage=300)
        try_update(a_lse, "lse")

    # === Phase 6: Perturbation escapes ===
    print(f"\n=== Phase 6: Perturbation escape (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        for trial in range(5):
            rem = deadline - time.monotonic()
            if rem < 25:
                break
            a_pert = perturb_sequence(best_seq, rng, strength=0.01 + 0.008 * trial)
            t_sub = min(rem * 0.2, 50.0)
            a_opt, _ = nesterov_lp_fast(a_pert, t_sub, n_max_iters=50000, print_every=0)
            try_update(a_opt, f"pert{trial}")

    # === Phase 7: Final LP polish ===
    print(f"\n=== Phase 7: Final LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t7 = min(rem * 0.35, 150.0)
        a_fin, _ = nesterov_lp_fast(best_seq, t7, n_max_iters=100000, print_every=200)
        try_update(a_fin, "final_lp")

    # === Phase 8: Final MW + LSE ===
    rem = deadline - time.monotonic()
    if rem > 20:
        t8 = min(rem * 0.3, 40.0)
        a_mw_fin, _ = multiplicative_weights(best_seq, t8, eta0=0.01)
        try_update(a_mw_fin, "final_mw")
        rem = deadline - time.monotonic()
        if rem > 15:
            t9 = min(rem * 0.6, 30.0)
            a_lse2 = lbgfs_refine(best_seq, t9,
                                  T_stages=[0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001],
                                  maxiter_per_stage=200)
            try_update(a_lse2, "final_lse")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} n={len(best_seq)}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
