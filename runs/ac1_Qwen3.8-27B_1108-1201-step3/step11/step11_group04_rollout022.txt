"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-n exploration with interpolation
2. Aggressive spectral peak equalization (targeting time-domain max)
3. Nesterov LP alternating with warm-starting

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS, cp.MOSEK, cp.GUROBI]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver, warm_start=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    if warm_start is not None:
        b_var.value = warm_start

    solvers_to_try = [solver, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    solved = False
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solved = True
                break
        except Exception:
            continue

    if not solved:
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(max(0, M_var.value))


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_obj(z, n, T_stage, deadline):
    """Objective for L-BFGS."""
    if time.monotonic() > deadline:
        return 1e15
    obj, _ = logsumexp_obj_grad(z, n, T_stage)
    return obj


def lbgfs_jac(z, n, T_stage, deadline):
    """Gradient for L-BFGS."""
    if time.monotonic() > deadline:
        return np.zeros_like(z)
    _, g = logsumexp_obj_grad(z, n, T_stage)
    return g


def lbgfs_hessp(z, n, T_stage, deadline, p):
    """Approximate Hessian-vector product for trust-constr."""
    if time.monotonic() > deadline:
        return np.zeros_like(p)
    # Use finite differences as fallback
    eps = 1e-6
    obj_plus = lbgfs_obj(z + eps * p, n, T_stage, deadline)
    obj_minus = lbgfs_obj(z - eps * p, n, T_stage, deadline)
    obj_here = lbgfs_obj(z, n, T_stage, deadline)
    return (obj_plus - 2 * obj_here + obj_minus) / (eps * eps)


def spectral_flatten(a, n_iters=100, blend=0.4, deadline=None):
    """
    Iterative spectral flattening targeting time-domain peak reduction.
    Uses a modified Gerchberg-Saxton that targets the autocorrelation peaks.
    """
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_score = fast_convolve(a).max()

    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)

        # Target: make |F|^2 (which is FFT of a*a) as flat as possible in time domain
        # This means |F| should be as flat as possible
        mags = np.abs(F)
        non_dc = mags[1:L // 2 + 1]
        if non_dc.min() < 1e-15:
            target = np.mean(non_dc)
        else:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        phases = np.angle(F)
        F_new = F.copy()
        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend) * mags[k] + blend * target
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])

        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        score = fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def spectral_peak_target(a, n_iters=200, deadline=None):
    """
    Directly target the maximum peak of a*a by adjusting the spectrum.
    At each step, identify the time-domain peak and reduce the spectral
    components that contribute most to it.
    """
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_score = fast_convolve(a).max()

    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        b = fast_convolve(a)
        max_b = b.max()
        peak_idx = np.argmax(b)

        # Compute current spectrum
        F = np.fft.fft(np.concatenate([a, np.zeros(L - n)]))

        # The peak at index peak_idx in time domain corresponds to
        # reducing the "coherence" of the spectrum at that frequency
        # Use a soft threshold: reduce magnitudes that align with the peak
        freq_of_peak = peak_idx % L
        # Reduce magnitudes near the peak frequency in the power spectrum
        alpha = 0.05 + 0.05 * (1 - i / max(1, n_iters))

        mags = np.abs(F)
        phases = np.angle(F)

        # Compute the "impulse response" contribution of each frequency bin
        # to the peak position
        # b[peak_idx] = sum_k |F[k]|^2 * exp(-i*2*pi*k*peak_idx/L) / L
        # To reduce b[peak_idx], reduce |F[k]| where the phase aligns

        # Simple approach: scale down all mags slightly proportional to
        # their contribution to the peak
        contrib = mags ** 2 * np.cos(2 * np.pi * freq_of_peak * np.arange(L) / L)
        contrib = np.maximum(contrib, 0)
        total_contrib = contrib.sum()
        if total_contrib > 1e-15:
            scale = contrib / total_contrib
            mags_new = mags * (1 - alpha * scale / max(mags.max(), 1e-15) * 0.1)
        else:
            mags_new = mags

        # Also flatten toward geometric mean
        non_dc = mags_new[1:L // 2 + 1]
        if non_dc.min() > 1e-15:
            target = np.exp(np.mean(np.log(non_dc)))
            for k in range(1, L // 2 + 1):
                mags_new[k] = 0.95 * mags_new[k] + 0.05 * target
                if k < L - k:
                    mags_new[L - k] = mags_new[k]

        F_new = mags_new * np.exp(1j * phases)
        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        score = fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def peak_shave_adaptive(a, time_limit, n_steps=500000):
    """
    Peak shaving with adaptive learning rate and top-K peak targeting.
    Uses smooth weights instead of hard active set.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return None, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 0.03
    no_improve = 0
    scale = 1.0
    momentum = np.zeros(n)
    beta = 0.9

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Smooth weights: Gaussian kernel around the max
        T_smooth = max_b * 0.01
        w = np.exp((b - max_b) / T_smooth)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(b))
            w[np.argmax(b)] = 1.0
        else:
            w = w / w_sum

        # Compute subgradient (weighted average of peak gradients)
        g = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += w[k] * a[k - idx]

        # Gradient with momentum
        momentum = beta * momentum + (1 - beta) * g
        g_use = momentum

        eta_eff = eta * scale
        log_scale = -eta_eff * g_use
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 20:
            scale *= 0.75
            no_improve = 0
            momentum *= 0.5
            if scale < 0.003:
                # Try a random restart from current best
                noise = np.random.default_rng(step).normal(0, 0.02, n)
                a = best_a + noise
                a = np.maximum(a, 0)
                s2 = a.sum()
                if s2 > 1e-15:
                    a = a / s2
                scale = 0.5
                if time.monotonic() - t0 > time_limit * 0.9:
                    break

    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=500000):
    """Nesterov dual averaging LP alternating with adaptive parameters."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(54321)
    lp_times = []
    warm_b = a.copy()

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver, warm_start=warm_b)
        lp_times.append(time.monotonic() - lp_t0)

        if b_new is None:
            continue
        warm_b = b_new.copy()

        if len(lp_times) >= 30:
            avg_lp_time = np.mean(lp_times[-20:])
            if avg_lp_time > 3.0:
                print(f"    LP avg={avg_lp_time:.2f}s/iter, adapting at iter {it+1}")
                # Reduce problem: skip some iterations
                if avg_lp_time > 5.0:
                    break

        # Nesterov dual averaging with tuned alpha
        alpha = 2.0 / ((it + 2) ** 1.0)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        a = ((it + 1) * a + (it + 2) * b_new) / (2 * (it + 1) + it + 2)
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Stall detection
        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-12:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 5:
                # Perturb in frequency domain
                F = np.fft.fft(y)
                mags = np.abs(F)
                mags[1:] *= (1 + 0.01 * rng.standard_normal(len(mags) - 1))
                mags = np.maximum(mags, 0)
                phases = np.angle(F)
                F_new = mags * np.exp(1j * phases)
                y = np.fft.ifft(F_new).real
                y = np.maximum(y, 0)
                sy2 = y.sum()
                if sy2 > 1e-15:
                    y = y / sy2
                stall_count = 0

            if (it + 1) % 2000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def lbgfs_refine_multi(a0, time_limit, deadline=None):
    """
    Multi-stage LogSumExp L-BFGS with trust-constr fallback.
    Very aggressive temperature annealing.
    """
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    if deadline is None:
        deadline = t0 + time_limit

    T_stages = [50.0, 20.0, 10.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5,
                0.4, 0.3, 0.25, 0.2, 0.15, 0.12, 0.1, 0.08, 0.06, 0.05,
                0.04, 0.03, 0.025, 0.02, 0.015, 0.012, 0.01, 0.008,
                0.006, 0.005, 0.004, 0.003, 0.002, 0.0015, 0.001, 0.0008, 0.0005]

    best_a = a0.copy()
    best_score = evaluate_local(a0)

    for stage_i, stage_T in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break

        # Allocate time: earlier stages get less time, later stages get more
        stages_remaining = len(T_stages) - stage_i
        time_per_stage = min(remaining / stages_remaining, 30.0)
        if time_per_stage < 1.0:
            break

        def func(z_in):
            return lbgfs_obj(z_in, n, stage_T, deadline)

        def jac(z_in):
            return lbgfs_jac(z_in, n, stage_T, deadline)

        use_lbfgs = stage_T > 0.01
        if use_lbfgs:
            try:
                res = minimize(
                    fun=func, x0=z, jac=jac, method='L-BFGS-B',
                    options={'maxiter': 500, 'ftol': 1e-30,
                             'gtol': 1e-20, 'maxfun': 5000, 'disp': False}
                )
                z = res.x
            except Exception:
                pass
        else:
            # For very low temperatures, use fewer iterations (essentially max)
            try:
                res = minimize(
                    fun=func, x0=z, jac=jac, method='L-BFGS-B',
                    options={'maxiter': 200, 'ftol': 1e-30,
                             'gtol': 1e-20, 'maxfun': 2000, 'disp': False}
                )
                z = res.x
            except Exception:
                pass

        a_curr = softmax_from_z(z)
        sc = evaluate_local(a_curr)
        if sc < best_score:
            best_score = sc
            best_a = a_curr.copy()

        if remaining < 10:
            break

    return best_a


def peak_quadratic_refine(a0, time_limit, n_peaks=5):
    """
    Target specific peaks using quadratic sub-problems.
    For each active peak, solve a QP to minimize it while keeping others bounded.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for outer in range(200):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Find top peaks
        sorted_idx = np.argsort(b)[::-1]
        active = sorted_idx[:n_peaks]

        # Try to reduce the worst peak
        worst_k = active[0]
        # Gradient of B_k w.r.t. a: d/d a[j] sum_i a[i]*a[k-i] = a[k-j] (summed over valid i)
        # Actually: B_k = sum_j a[j]*a[k-j], dB_k/da[j] = a[k-j] + a[j] (if both terms valid)
        # More carefully: B_k = sum_i a[i]*a[k-i]
        # dB_k/da[j] = a[k-j] (from the a[i] term when i=j) + a[j] (from the a[k-i] term when k-i=j, i.e., i=k-j)
        # But we need to be careful about boundaries

        grad = np.zeros(n)
        for j in range(n):
            # Contribution from a[j] in the "first" position: a[k-j] if 0 <= k-j < n
            val = 0.0
            if 0 <= k - j < n:
                val += a[k - j]
            # Contribution from a[j] in the "second" position: a[j] when k-j = j... 
            # Actually B_k = sum_i a[i]*a[k-i], so da[j] appears as:
            # - a[j]*a[k-j] when i=j (if k-j valid)
            # - a[k-j]*a[j] when k-i=j, i.e., i=k-j (if k-j valid and i != j)
            # So: dB_k/da[j] = a[k-j] + (a[j] if k-j != j and 0 <= k-j < n else 0)
            # Simplified: dB_k/da[j] = a[k-j] + a[j] * [0 <= k-j < n and k != 2j]
            # Actually for the diagonal (k=2j), both terms give a[j], so it's 2*a[j]
            if 0 <= (k - j) and (k - j) < n:
                if k - j == j:
                    val += a[j]  # diagonal: both terms contribute a[j]
                else:
                    val += a[j]  # off-diagonal: a[j] from the second position
            grad[j] = val

        # Normalized gradient direction
        g_norm = np.sqrt(np.sum(grad ** 2))
        if g_norm < 1e-15:
            break
        grad /= g_norm

        # Line search: reduce a in the direction of the gradient
        # a_new = a - step * grad, projected to non-negative simplex
        for step_try in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            a_new = a - step_try * grad
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            score = 2.0 * n * fast_convolve(a_new).max()
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
                a = a_new
                break
            elif score < best_score + 0.01 * max_b:
                # Accept small increases if the worst peak decreased
                if fast_convolve(a_new)[worst_k] < b[worst_k] * 0.99:
                    a = a_new
                    break

    return best_a


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng, style=None):
    """Create a structured initial sequence with various styles."""
    if style is None:
        style = rng.integers(0, 12)
    x = np.linspace(0, 1, n)
    if style == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif style == 1:
        m = rng.integers(1, 5)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * m * x)
        a = np.maximum(a, 0)
    elif style == 2:
        a = np.exp(-3 * x * (1 - x))
    elif style == 3:
        # Asymmetric: large on left, small on right
        a = np.exp(-2 * x) + 0.1
    elif style == 4:
        # Two-lobe
        c = rng.uniform(0.2, 0.8)
        a = np.exp(-5 * (x - c) ** 2) + 0.3 * np.exp(-5 * (x - (1-c)) ** 2)
    elif style == 5:
        # Chirp-like magnitude
        m = rng.integers(2, 6)
        a = np.abs(np.sin(np.pi * m * x)) ** 0.5
        a = np.maximum(a, 0.01)
    elif style == 6:
        # Triangle
        a = 1.0 - np.abs(2 * x - 1)
    elif style == 7:
        # Random smooth
        freqs = rng.integers(1, 4, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        a = 1.0
        for i in range(3):
            a += 0.2 * np.cos(2 * np.pi * freqs[i] * x + phases[i])
        a = np.maximum(a, 0)
    elif style == 8:
        # Power law
        a = x ** rng.uniform(0.3, 2.0) + 0.05
    elif style == 9:
        # Bump
        c = rng.uniform(0.2, 0.8)
        w = rng.uniform(0.1, 0.4)
        a = np.exp(-(x - c) ** 2 / (2 * w ** 2)) + 0.05
    elif style == 10:
        # Cosine window with harmonic
        m = rng.integers(1, 4)
        a = 0.5 * (1 + np.cos(np.pi * x)) + 0.3 * np.cos(2 * np.pi * m * x)
        a = np.maximum(a, 0)
    else:
        # Ramp with noise
        a = 1.0 - x + 0.1 * rng.standard_normal(n)
        a = np.maximum(a, 0)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def perturbed_sequence(a, rng, strength=0.03):
    """Create a perturbed version of sequence a."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    a_pert = a.copy()
    for _ in range(3):
        freq = rng.integers(1, 10)
        phase = rng.uniform(0, 2 * np.pi)
        a_pert += strength * np.cos(freq * x + phase)
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def diversify_select(parent, n, rng, budget_s=20):
    """Try several perturbations and spectral variants, return the best."""
    candidates = [parent.copy()]
    for strength in [0.01, 0.03, 0.05, 0.1, 0.15, 0.2]:
        candidates.append(perturbed_sequence(parent, rng, strength=strength))
    # Different structured inits at same n
    for style in [3, 5, 8, 10]:
        candidates.append(structured_init(n, rng, style=style))
    # Spectral flatten of parent
    sf = spectral_flatten(parent, n_iters=20, blend=0.3, deadline=time.monotonic() + 5)
    candidates.append(sf)

    best_c = parent.copy()
    best_sc = evaluate_local(parent)

    for c in candidates:
        sc = evaluate_local(c)
        if sc < best_sc:
            best_sc = sc
            best_c = c.copy()

    return best_c, best_sc


def quick_optimize(a, solver, time_limit=30):
    """Quick optimization pipeline: spectral + LP + shave."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)

    deadline = time.monotonic() + time_limit

    # Spectral
    if time.monotonic() < deadline - 5:
        a_sp = spectral_flatten(a, n_iters=30, blend=0.3, deadline=deadline)
        sc = evaluate_local(a_sp)
        if sc < best_score:
            best_score = sc
            best_a = a_sp.copy()
        a = a_sp

    # LP
    t_lp = max(5, (deadline - time.monotonic()) * 0.6)
    if t_lp > 5:
        a_lp, _ = nesterov_lp_refine(a, t_lp, solver, n_max_iters=100000)
        sc = evaluate_local(a_lp)
        if sc < best_score:
            best_score = sc
            best_a = a_lp.copy()
            a = a_lp

    # Peak shave
    t_ps = max(3, deadline - time.monotonic() - 2)
    if t_ps > 3:
        a_ps, _ = peak_shave_adaptive(a, t_ps, n_steps=50000)
        if a_ps is not None:
            sc = evaluate_local(a_ps)
            if sc < best_score:
                best_score = sc
                best_a = a_ps.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses spectral methods, Nesterov LP, LogSumExp L-BFGS, peak shaving,
    and multi-n exploration with adaptive time allocation.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    rem = deadline - time.monotonic()
    print(f"Total budget: {rem:.0f}s, starting score: {best_score:.8f}")

    # === Phase 0: Quick diversification ===
    print(f"\n=== Phase 0: Diversification ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_div = min(rem * 0.015, 20.0)
        best_div, sc_div = diversify_select(best_seq, n_best, rng, budget_s=t_div)
        local_update(best_div, "diversified")

    # === Phase 1: Multi-n exploration with quick optimization ===
    print(f"=== Phase 1: Multi-n exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    n_candidates = [n_best]
    # Add nearby n values
    for n_try in [800, 1000, 1200, 1500, 1800, 2000]:
        if n_try != n_best and 500 <= n_try <= 3000:
            n_candidates.append(n_try)

    n_candidates = sorted(set(n_candidates))
    print(f"  n candidates: {n_candidates}")

    # Allocate time proportional to how promising each n is
    for n_try in n_candidates:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        if n_try == n_best:
            continue  # Already have a good solution for this n
        t_n = min(rem * 0.08, 40.0)
        if t_n < 15:
            continue
        print(f"  Trying n={n_try}, budget={t_n:.0f}s")
        init_n = interpolate_sequence(best_seq, n_try)
        a_opt = quick_optimize(init_n, solver, time_limit=t_n)
        local_update(a_opt, f"n{n_try}_quick")

    # === Phase 2: Nesterov LP on best (main workhorse) ===
    print(f"\n=== Phase 2: Nesterov LP n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_lp = min(rem * 0.20, 300.0)
    if t_lp > 30:
        print(f"  Budget: {t_lp:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_lp, solver, n_max_iters=500000)
        local_update(a_lp, "nesterov_lp_1")

    # === Phase 3: Ultra-fine LogSumExp L-BFGS ===
    print(f"\n=== Phase 3: L-BFGS LogSumExp (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_lbfgs = min(rem * 0.25, 400.0)
    if t_lbfgs > 30:
        print(f"  Budget: {t_lbfgs:.0f}s")
        a_lb = lbgfs_refine_multi(best_seq, t_lbfgs, deadline=time.monotonic() + t_lbfgs)
        local_update(a_lb, "lbgfs_fine")

    # === Phase 4: Peak shaving ===
    print(f"\n=== Phase 4: Peak Shaving (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_peak = min(rem * 0.10, 120.0)
    if t_peak > 10:
        a_pk, _ = peak_shave_adaptive(best_seq, t_peak, n_steps=300000)
        local_update(a_pk, "peak_shave")

    # === Phase 5: Peak quadratic refine ===
    print(f"\n=== Phase 5: Peak Quadratic (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_pq = min(rem * 0.05, 60.0)
    if t_pq > 10:
        a_pq = peak_quadratic_refine(best_seq, t_pq, n_peaks=3)
        local_update(a_pq, "peak_quad")

    # === Phase 6: Second LP round ===
    print(f"\n=== Phase 6: Nesterov LP Round 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_lp2 = min(rem * 0.15, 200.0)
    if t_lp2 > 20:
        print(f"  Budget: {t_lp2:.0f}s")
        a_lp2, _ = nesterov_lp_refine(best_seq, t_lp2, solver, n_max_iters=300000)
        local_update(a_lp2, "nesterov_lp_2")

    # === Phase 7: Diversification + re-optimize ===
    print(f"\n=== Phase 7: Diversification (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        n_trials = 3
        t_per_trial = min(rem / (n_trials + 1), 60.0)
        for trial in range(n_trials):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 20:
                break
            strength = 0.02 + 0.02 * trial
            a_pert = perturbed_sequence(best_seq, rng, strength=strength)
            t_sub = min(t_per_trial, rem_sub - 5)
            if t_sub > 10:
                a_opt = quick_optimize(a_pert, solver, time_limit=t_sub)
                local_update(a_opt, f"diversify_{trial}")

    # === Phase 8: Try different n with full pipeline ===
    print(f"\n=== Phase 8: Different n full (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        # Try the best n + one alternative
        for n_alt in [1500, 1000, 2000]:
            rem_sub = deadline - time.monotonic()
            if rem_sub < 60:
                break
            if n_alt == len(best_seq):
                continue
            t_alt = min(rem_sub * 0.3, 80.0)
            if t_alt < 30:
                continue
            print(f"  Full optimize n={n_alt}, budget={t_alt:.0f}s")
            init_alt = interpolate_sequence(best_seq, n_alt)
            a_opt = quick_optimize(init_alt, solver, time_limit=t_alt)
            local_update(a_opt, f"full_n{n_alt}")

    # === Phase 9: Final polish ===
    print(f"\n=== Phase 9: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        # Ultra-fine L-BFGS
        t_fin = min(rem * 0.4, 80.0)
        if t_fin > 10:
            a_fin = lbgfs_refine_multi(best_seq, t_fin, deadline=time.monotonic() + t_fin)
            local_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 20:
        # Final peak shave
        t_ps = min(rem * 0.4, 40.0)
        if t_ps > 5:
            a_ps, _ = peak_shave_adaptive(best_seq, t_ps, n_steps=100000)
            local_update(a_ps, "final_peak")

    rem = deadline - time.monotonic()
    if rem > 15:
        # Final LP
        t_flp = min(rem * 0.5, 30.0)
        if t_flp > 5:
            a_flp, _ = nesterov_lp_refine(best_seq, t_flp, solver, n_max_iters=50000)
            local_update(a_flp, "final_lp")

    rem = deadline - time.monotonic()
    if rem > 10:
        # One more spectral pass
        a_sp = spectral_flatten(best_seq, n_iters=50, blend=0.2,
                                deadline=time.monotonic() + rem * 0.5)
        local_update(a_sp, "final_spectral")

    # Final verification and output
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_start
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}) at {elapsed:.1f}s")
        try:
            test_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
            eval_score = evaluate_sequence(test_seq)
            print(f"Verified with evaluate_sequence: {eval_score:.8f}")
            if eval_score < best_score:
                best_score = eval_score
        except Exception as e:
            print(f"Verification error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
