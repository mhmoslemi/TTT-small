```python
        else:
            a_new = a_new / sa
        
        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy
        
        a = a_new
        y = y_new
        
        sc = score_normalized(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        
        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            per_iter = elapsed / (it + 1)
            print(f"    NLP iter={it+1} ({elapsed:.0f}s, {per_iter*1000:.0f}ms/it): "
                  f"score={best_score:.8f} M_lp={M_lp:.6f}")
    
    return best_a, best_score
```
