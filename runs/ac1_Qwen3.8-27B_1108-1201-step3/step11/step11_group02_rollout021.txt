def evaluate_local(a):
    """Evaluate sequence score (scale-invariant, assumes non-negative)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)

        return float('inf')
    n = len(a)
    b = fast_convolve(a)

def softmax_fn(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def lse_obj_grad_z(z, T):
    """
    Compute smoothed objective and gradient in z-space (softmax parametrization).
    f_T(a) = 2n * T * log(sum_k exp(b_k / T)) where b = a*a
    Gradient in z-space: 4n * a * (c - E_w[b])
    where c_m = sum_j a_j * w_{m+j}, w_k = exp(b_k/T)/sum(exp(b/T))
    """
    n = len(z)
    a = softmax_fn(z)
    b = fast_convolve(a)

    # Compute LSE and weights
    max_b = b.max()
    exp_b = np.exp((b - max_b) / T)
    sum_exp = exp_b.sum()
    w = exp_b / sum_exp

    # Objective: 2n * (max_b + T * log(sum_exp))
    obj = 2.0 * n * (max_b + T * np.log(sum_exp))

    # Compute c_m = sum_j a_j * w_{m+j} via FFT (cross-correlation)
    # c is the cross-correlation of w (length 2n-1) with a (length n)
    # (w cross a)_m = sum_j w_{m+j} * a_j for m = 0,...,n-1
    fft_size = 1 << (3 * n - 2 + 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(np.conj(W) * A, fft_size)[:n]

    # E_w[b] = sum_k w_k * b_k
    EB = np.dot(w, b)

    # grad_z = 4n * a * (c - EB)
    grad_z = 4.0 * n * a * (c - EB)

    return obj, grad_z

def build_sparse_conv_matrix(a):

                    a_trial = a_trial / a_trial.sum()
                    score = evaluate_local(a_trial)
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            if stall > 200:
                # Multi-coordinate reduction
                peak_k = int(np.argmax(b))
                js = max(0, peak_k - n + 1)
                je = min(n - 1, peak_k)
                idx = np.arange(js, je + 1)
                contribs = a[idx] * a[peak_k - idx]
                top_positions = idx[np.argsort(contribs)[::-1][:6]]
                a_trial = a.copy()
                for pos in top_positions:
                    if a_trial[pos] > 1e-14:
                        a_trial[pos] *= 0.997
                a_trial = np.maximum(a_trial, 0)
                mask = np.ones(n, dtype=bool)
                mask[top_positions] = False
                diff = a.sum() - a_trial.sum()
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum > 1e-15 and diff > 0:
                    a_trial[mask] = others + diff * others / others_sum
                a_trial = a_trial / a_trial.sum()
                score = evaluate_local(a_trial)
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
                elif stall > 1000:
                    # Random perturbation
                    noise = rng.standard_normal(n) * 0.002
                    a_trial = a + noise
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    score = evaluate_local(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                    stall = 0
        else:
            stall = 0

        if (step + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc {elapsed:.0f}s: score={best_score:.10f}")

    return best_a, best_score


def peak_reduction_heuristic(a0, time_limit):
    """
    Iteratively reduce the dominant peak by adjusting contributing coordinates.
    Uses a "water-filling" approach: reduce peak contributors, add to non-contributors.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()

    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))

        # Get contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        # Identify non-contributing coordinates (don't appear in the peak sum)
        contributing_set = set(idx.tolist()) | set((peak_k - idx).tolist())
        non_contrib_mask = np.ones(n, dtype=bool)
        for pos in contributing_set:
            if 0 <= pos < n:
                non_contrib_mask[pos] = False

        n_non_contrib = non_contrib_mask.sum()
        if n_non_contrib < 3:
            break

        # Try reducing top contributor and adding to a random non-contributor
        top_pos = idx[np.argmax(contribs)]
        nc_indices = np.where(non_contrib_mask)[0]

        best_factor = None
        best_nc = None
        best_trial_score = best_score

        for factor in [0.999, 0.998, 0.995, 0.99]:
            a_trial = a.copy()
            a_trial[top_pos] *= factor
            diff = a[top_pos] - a_trial[top_pos]
            # Add to the non-contributor with smallest current value
            # (to minimize impact on other peaks)
            nc_vals = a_trial[nc_indices]
            min_nc_idx = nc_indices[np.argmin(nc_vals)]
            a_trial[min_nc_idx] += diff
            score = evaluate_local(a_trial)
            if score < best_trial_score:
                best_trial_score = score
                best_factor = factor
                best_nc = min_nc_idx

        if best_factor is not None:
            a_trial = a.copy()
            a_trial[top_pos] *= best_factor
            diff = a[top_pos] - a_trial[top_pos]
            a_trial[best_nc] += diff
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            a = a_trial
            if best_trial_score < best_score:
                best_score = best_trial_score
                best_a = a.copy()

    return best_a


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
