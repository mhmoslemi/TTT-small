import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.sparse import lil_matrix, csc_matrix, csr_matrix
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_q_sparse(k, n):
    """Build sparse Q matrix such that a^T Q a = (a*a)[k]."""
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def evaluate_local(a):
    """Evaluate the objective score locally (fast, normalized sum=1)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a / s)
    return float(2.0 * n * b.max())


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    solved = False
    b_val = None
    M_val = None
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                M_val = float(max(0, M_var.value))
                solved = True
                break
        except Exception:
            continue
    if not solved or b_val is None:
        return None, None
    b_val = np.maximum(b_val, 0)
    s_sum = b_val.sum()
    if s_sum < 1e-15:
        return None, None
    b_val = b_val / s_sum
    return b_val, M_val


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=500000):
    """Nesterov dual averaging LP alternating with adaptive restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(123)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        conv_a = fast_convolve(a)
        score_a = 2.0 * n * conv_a.max()
        conv_y = fast_convolve(y)
        score_y = 2.0 * n * conv_y.max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 10:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 5)
                noise = 0.02 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                sy2 = y.sum()
                if sy2 > 1e-15:
                    y = y / sy2
                stall_count = 0

            if (it + 1) % 1000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def solve_active_socp_iterative(a_init, solver, time_limit, max_active_set=80, verbose=True):
    """
    Iterative active-set SOCP:
    - Start with active set = {argmax of conv}
    - At each step: solve SOCP with active constraints, then check all inactive
      constraints. Add the most violated one if any violation exceeds tolerance.
    - Remove constraints that are no longer tight.
    - Converges when no inactive constraint is violated.
    """
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    t0 = time.monotonic()
    best_a = a.copy()
    best_score = evaluate_local(a)

    # Initial active set: top positions
    b = fast_convolve(a)
    # Only consider first half (k=0..n-1) due to symmetry
    b_half = b[:n]
    active = np.array([np.argmax(b_half)])
    tolerance = 1e-10

    iteration = 0
    max_iters = 200

    while time.monotonic() - t0 < time_limit and iteration < max_iters:
        iteration += 1
        iteration_start = time.monotonic()

        if len(active) > max_active_set:
            # Keep only the most active constraints
            b_cur = fast_convolve(a)
            active_scores = b_cur[:n][active]
            keep = np.argsort(active_scores)[-max_active_set:]
            active = active[keep]

        # Solve SOCP with current active set
        a_var = cp.Variable(n, nonneg=True)
        t_var = cp.Variable()
        constraints = [cp.sum(a_var) == 1]

        for k in active:
            int_k = int(k)
            Q = build_q_sparse(int_k, n)
            constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)

        prob = cp.Problem(cp.Minimize(t_var), constraints)

        solvers_to_try = [cp.MOSEK, cp.GUROBI, solver, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
        solved = False
        a_new = None
        for sv in solvers_to_try:
            if sv is None:
                continue
            try:
                prob.solve(solver=sv)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    a_new = np.asarray(a_var.value, dtype=np.float64)
                    a_new = np.maximum(a_new, 0)
                    s_new = a_new.sum()
                    if s_new < 1e-15:
                        a_new = None
                        continue
                    a_new = a_new / s_new
                    solved = True
                    break
            except Exception:
                continue

        if not solved or a_new is None:
            # Try with fewer active constraints
            if len(active) > 3:
                b_cur = fast_convolve(a)
                active_scores = b_cur[:n][active]
                keep = np.argsort(active_scores)[-max(3, len(active) // 2):]
                active = active[keep]
                continue
            break

        # Check score
        score_new = evaluate_local(a_new)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            if verbose:
                print(f"    ActiveSOCP iter={iteration}: new best={best_score:.10f} (active={len(active)})")
        a = a_new

        # Check inactive constraints for violation
        b_new = fast_convolve(a)
        max_val = b_new[:n].max()
        
        # Find most violated inactive constraint
        violation_max = -1
        worst_k = -1
        for k in range(n):
            if k in active:
                continue
            val = b_new[k]
            viol = val - (max_val + tolerance)
            if viol > violation_max:
                violation_max = viol
                worst_k = k

        if violation_max > 1e-12:
            # Add the most violated constraint
            active = np.append(active, worst_k)
            if verbose:
                print(f"    ActiveSOCP iter={iteration}: added k={worst_k}, violation={violation_max:.2e}, |active|={len(active)}")
        else:
            # Check if we can remove any active constraints (those no longer tight)
            active_tight = []
            for k in active:
                if b_new[k] >= max_val - 1e-8 * max_val:
                    active_tight.append(k)
            if len(active_tight) < len(active):
                if verbose:
                    print(f"    ActiveSOCP iter={iteration}: removing {len(active)-len(active_tight)} inactive constraints")
                active = np.array(active_tight)
                if len(active) == 0:
                    active = np.array([np.argmax(b_new[:n])])
            # Converged: no violation
            if verbose:
                print(f"    ActiveSOCP CONVERGED at iter={iteration}: score={best_score:.10f}, |active|={len(active)}")
            break

        # Time check per iteration
        if time.monotonic() - iteration_start > time_limit * 0.8 / max(1, iteration):
            if verbose:
                print(f"    ActiveSOCP: iterations getting slow, stopping at iter={iteration}")
            break

    return best_a, best_score


def try_full_socp(a_init, solver, time_limit):
    """Attempt to solve the full SOCP with all n constraints (first half)."""
    n = len(a_init)
    t0 = time.monotonic()

    a_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [cp.sum(a_var) == 1]

    # Add constraints for k = 0, ..., n-1 (first half due to symmetry)
    for k in range(n):
        Q = build_q_sparse(k, n)
        constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)

    prob = cp.Problem(cp.Minimize(t_var), constraints)

    solvers_to_try = [cp.MOSEK, cp.GUROBI, solver, cp.CLARABEL, cp.PDLP]
    for sv in solvers_to_try:
        if sv is None:
            continue
        if time.monotonic() - t0 > time_limit:
            break
        try:
            prob.solve(solver=sv, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.asarray(a_var.value, dtype=np.float64)
                a_val = np.maximum(a_val, 0)
                s_val = a_val.sum()
                if s_val < 1e-15:
                    return None
                a_val = a_val / s_val
                print(f"    Full SOCP solved by {sv} in {time.monotonic()-t0:.1f}s")
                return a_val
        except Exception as e:
            if verbose:
                print(f"    Full SOCP {sv} failed: {e}")
            continue
    return None

def gradient_substep(a, k_active, step_size):
    """One projected gradient step for reducing max over active positions."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b[:n].max()
    
    # Subgradient of max: average Q_k a over active k
    g = np.zeros(n)
    count = 0
    for k in k_active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            g[j] += a[k - j]
        count += 1
    g /= count
    
    # Gradient step with projection onto simplex
    a_new = a - step_size * g
    # Project onto simplex (sum=1, a>=0)
    a_new = project_onto_simplex(a_new)
    return a_new


def project_onto_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def subgradient_polish(a0, time_limit, n_steps=100000):
    """Polynomial-decay step-size subgradient method on max function."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b[:n].max()
        threshold = max_b * (1.0 - 1e-6)
        active = np.where(b[:n] >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b[:n])])

        # Subgradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                g[j] += a[k - j]
        g /= len(active)

        # Step size: O(1/sqrt(t))
        eta = 1.0 / np.sqrt(step + 1) * 0.5
        a_new = project_onto_simplex(a - eta * g)
        score = evaluate_local(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses Nesterov LP alternating, iterative active-set SOCP with constraint
    addition, full SOCP, and multi-n exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        x = np.linspace(0, 1, n_init)
        best_seq = np.cos(np.pi * x / 2) ** 2
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")

    def update_best(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # === Phase 1: LP Alternating (initial convergence) ===
    print(f"\n=== Phase 1: LP Alternating (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 480.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver, n_max_iters=500000)
        update_best(a_lp, "lp_phase1")

    # === Phase 2: Iterative Active-Set SOCP (key improvement) ===
    print(f"\n=== Phase 2: Iterative Active-Set SOCP (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 550.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_as, sc_as = solve_active_socp_iterative(best_seq, solver, t_alloc, max_active_set=60, verbose=True)
        update_best(a_as, "active_socp")

    # === Phase 3: Full SOCP attempt ===
    print(f"\n=== Phase 3: Full SOCP (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_full = try_full_socp(best_seq, solver, t_alloc)
        update_best(a_full, "full_socp")

    # === Phase 4: Active-set SOCP round 2 (from best) ===
    print(f"\n=== Phase 4: Active-Set SOCP Round 2 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_as2, _ = solve_active_socp_iterative(best_seq, solver, t_alloc, max_active_set=80, verbose=True)
        update_best(a_as2, "active_socp2")

    # === Phase 5: Try n=1199 ===
    print(f"\n=== Phase 5: n=1199 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_1199 = min(rem * 0.15, 180.0)
        print(f"  Budget: {t_1199:.0f}s")
        init_1199 = interpolate_sequence(best_seq, 1199)
        # Quick LP + active set
        t_lp = t_1199 * 0.5
        t_as = t_1199 * 0.5
        a_1199_lp, _ = nesterov_lp_refine(init_1199, t_lp, solver, n_max_iters=100000)
        update_best(a_1199_lp, "lp_n1199")
        rem = deadline - time.monotonic()
        if rem > 15:
            a_1199_as, _ = solve_active_socp_iterative(
                a_1199_lp if a_1199_lp is not None else init_1199,
                solver, t_as, max_active_set=50, verbose=False)
            update_best(a_1199_as, "as_n1199")

    # === Phase 6: Try n=1201 ===
    print(f"\n=== Phase 6: n=1201 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_1201 = min(rem * 0.15, 180.0)
        print(f"  Budget: {t_1201:.0f}s")
        # Interpolate from the current best (which might be n=1199 or 1200)
        ref_seq = best_seq if best_seq is not None else init_1199
        init_1201 = interpolate_sequence(ref_seq, 1201)
        t_lp = t_1201 * 0.5
        t_as = t_1201 * 0.5
        a_1201_lp, _ = nesterov_lp_refine(init_1201, t_lp, solver, n_max_iters=100000)
        update_best(a_1201_lp, "lp_n1201")
        rem = deadline - time.monotonic()
        if rem > 15:
            a_1201_as, _ = solve_active_socp_iterative(
                a_1201_lp if a_1201_lp is not None else init_1201,
                solver, t_as, max_active_set=50, verbose=False)
            update_best(a_1201_as, "as_n1201")

    # === Phase 7: Final subgradient polish ===
    print(f"\n=== Phase 7: Final Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_polish = min(rem * 0.3, 60.0)
        a_sg, _ = subgradient_polish(best_seq, t_polish, n_steps=50000)
        update_best(a_sg, "subgrad")
        rem = deadline - time.monotonic()
        if rem > 15:
            a_final, _ = solve_active_socp_iterative(
                best_seq, solver, min(rem * 0.8, 60.0), max_active_set=40, verbose=True)
            update_best(a_final, "final_as")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")

    # Sanitize output
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    # Verify sum > 0.01
    if sum(result) < 0.01:
        result = [1.0 / 1200] * 1200
    return result
