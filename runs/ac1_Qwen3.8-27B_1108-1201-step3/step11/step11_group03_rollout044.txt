2. Progressive active-set QCQP polish (exact convex optimization on active constraints)

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.sparse import lil_matrix, csc_matrix, csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def get_best_lp_solver():
    """Find the best working LP solver."""
    candidates = [cp.CLARABEL, cp.GUROBI, cp.MOSEK, cp.PDLP, cp.ECOS, cp.SCS]
    for solver_name in candidates:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            t0 = time.monotonic()
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                t = time.monotonic() - t0
                print(f"  LP solver {solver_name}: {t*1000:.1f}ms (test)")
                return solver_name
        except Exception as e:
            print(f"  LP solver {solver_name} unavailable: {e}")
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k] s.t. b>=0, sum(b)=1."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solvers_to_try = [solver, cp.CLARABEL, cp.GUROBI, cp.MOSEK, cp.ECOS, cp.SCS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_sum = b_val.sum()
                if s_sum < 1e-15:
                    return None, None
                b_val = b_val / s_sum
                return b_val, float(max(0, M_var.value))
        except Exception:
            continue
    return None, None


def build_q_sparse(k, n):
    """Build sparse Q matrix for conv(a,a)[k] = a^T Q a."""
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)


def nesterov_fw(a0, time_limit, solver, max_iters=1000000):
    """
    Nesterov-accelerated Frank-Wolfe for min max_k a^T Q_k a over simplex.
    Uses LP best-response for the linear minimization step.
    Returns (best_a, best_score).
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    # Nesterov state
    y = a.copy()  # lookahead point
    x_avg = a.copy()  # running average
    
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score
    rng = np.random.default_rng(12345)
    lp_count = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # LP best response at y
        b_new, M_b = solve_lp_best_response(y, solver)
        lp_count += 1
        if b_new is None:
            continue
        
        # Nesterov FW update
        alpha = 2.0 / (it + 2)
        
        # Update y (lookahead)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        
        # Update x_avg (running average)
        x_avg = (it * x_avg + b_new) / (it + 1)
        x_avg = np.maximum(x_avg, 0)
        sx = x_avg.sum()
        if sx < 1e-15:
            x_avg = y.copy()
        else:
            x_avg = x_avg / sx
        
        # Also try the "current point" update
        gamma = 2.0 / (it + 2)  # standard FW step size
        a = (1.0 - gamma) * a + gamma * b_new
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        # Evaluate all candidates
        for candidate in [a, y, x_avg]:
            score = 2.0 * n * fast_convolve(candidate).max()
            if score < best_score:
                best_score = score
                best_a = candidate.copy()
        
        # Stall detection and restart
        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            
            if stall_count >= 10:
                # Perturb and restart
                noise_scale = 0.05 / (1 + stall_count * 0.1)
                noise = noise_scale * rng.standard_normal(n)
                y = y + noise
                y = np.maximum(y, 0)
                sy2 = y.sum()
                if sy2 > 1e-15:
                    y = y / sy2
                stall_count = 0
            
            if (it + 1) % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"    FW iter={it+1} ({elapsed:.0f}s): score={best_score:.10f} evals={lp_count}")
    
    return best_a, best_score


def solve_active_qcqp(a, solver, rel_tol=1e-5, max_active=30, time_limit=60):
    """
    Solve the active-set QCQP:
    min t s.t. a^T Q_k a <= t for k in active, sum(a)=1, a>=0
    """
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    
    threshold = max_b * (1.0 - rel_tol)
    active = np.where(b >= threshold)[0]
    
    if len(active) > max_active:
        # Keep only the most binding constraints
        active = active[np.argsort(b[active])[-max_active:]]
    
    if len(active) == 0:
        active = np.array([np.argmax(b)])
    
    a_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [cp.sum(a_var) == 1]
    
    for k in active:
        Q = build_q_sparse(int(k), n)
        constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    solvers_to_try = [cp.MOSEK, cp.GUROBI, solver, cp.CLARABEL, cp.ECOS, cp.SCS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            t_start = time.monotonic()
            prob.solve(solver=s)
            t_elapsed = time.monotonic() - t_start
            if time.monotonic() - t_start > time_limit:
                continue
            if prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.asarray(a_var.value, dtype=np.float64)
                a_val = np.maximum(a_val, 0)
                s_sum = a_val.sum()
                if s_sum < 1e-15:
                    return None, None
                a_val = a_val / s_sum
                return a_val, float(max(0, t_var.value))
        except Exception:
            continue
    
    return None, None


def solve_full_qcqp(a, solver, time_limit=120):
    """
    Attempt to solve the full convex QCQP with ALL 2n-1 constraints.
    This is the exact problem but may be slow.
    """
    n = len(a)
    a_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [cp.sum(a_var) == 1]
    
    # Build all Q_k matrices
    for k in range(2 * n - 1):
        Q = build_q_sparse(k, n)
        constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    solvers_to_try = [cp.MOSEK, cp.GUROBI]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            t_start = time.monotonic()
            prob.solve(solver=s)
            if time.monotonic() - t_start > time_limit:
                print(f"    Full QCQP {s}: timeout")
                continue
            if prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.asarray(a_var.value, dtype=np.float64)
                a_val = np.maximum(a_val, 0)
                s_sum = a_val.sum()
                if s_sum < 1e-15:
                    return None
                return a_val / s_sum
        except Exception as e:
            print(f"    Full QCQP {s} failed: {e}")
            continue
    
    return None


def logsumexp_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse
    
    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    
    S_val = np.dot(grad_a, a)
    grad_z = a * (grad_a - S_val)
    return obj, grad_z


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    
    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        deadline = t0 + time_limit
        
        def lse_func(z_local):
            if time.monotonic() > deadline:
                return 1e15
            obj, _ = logsumexp_obj_grad(z_local, n, stage_T)
            return obj
        
        def lse_jac(z_local):
            if time.monotonic() > deadline:
                return np.zeros_like(z_local)
            _, g = logsumexp_obj_grad(z_local, n, stage_T)
            return g
        
        try:
            res = minimize(
                fun=lse_func, x0=z, jac=lse_jac,
                method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': maxiter_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)


def projected_subgradient(a0, time_limit, n_steps=100000):
    """Projected subgradient method for the convex objective."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    eta0 = 0.1
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Active set (top few)
        threshold = max_b * (1.0 - 1e-3)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        # Subgradient: average gradient over active constraints
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)
        
        # Step size (decreasing)
        eta = eta0 / (1.0 + step * 0.01)
        
        # Gradient step + projection to simplex
        a_new = a - eta * g
        # Project to simplex (sum=1, a>=0)
        # Simple projection: shift and clip
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 50:
            eta0 *= 0.7
            no_improve = 0
            if eta0 < 0.001:
                break
    
    return best_a, best_score
