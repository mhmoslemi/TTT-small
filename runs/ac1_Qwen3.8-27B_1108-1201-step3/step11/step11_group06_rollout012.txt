```python
def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast multiplicative weights for initial refinement
2. LP alternating with Nesterov acceleration (reduced time)
3. Extended LogSumExp L-BFGS with fine temperature cooling (main phase)

5. Final LSE polish with ultra-fine temperatures
Key differences from previous: more time on LSE, better temperature schedule,
wider n exploration, and more aggressive final polishing.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
import cvxpy as cp


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix_sparse(a):
    n = len(a)
    size = 2 * n - 1
    # T[k,j] = a[k-j] if 0 <= k-j < n, else 0
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(j, min(j + n, size)):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(size, n))


def get_lp_solver():
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver, T_mat):
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    for s in [solver, cp.CLARABEL, cp.ECOS, cp.SCS]:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_val = b_val.sum()
                if s_val < 1e-15:
                    return None, None
                b_val = b_val / s_val
                return b_val, float(max(0, M_var.value))
        except Exception:
            continue
    return None, None


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def compute_lse_obj(z, n, T):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T
    max_scaled = scaled.max()
    exp_terms = np.exp(scaled - max_scaled)
    lse_val = max_b + T * (max_scaled + np.log(exp_terms.sum() + 1e-300))
    obj = 2.0 * n * lse_val
    return obj


def compute_lse_grad(z, n, T):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T
    max_scaled = scaled.max()
    exp_terms = np.exp(scaled - max_scaled)
    w = exp_terms / exp_terms.sum()
    # grad of lse w.r.t. a: sum_k w[k] * grad_k where grad_k[j] = 2*a[k-j] (approx)
    grad_a = np.zeros(n)
    # Use convolution: grad_a ≈ 2 * correlate(w, a) but only for valid lags
    # Efficient computation: for each k with significant w[k], add w[k]*2*a_rev shifted
    for k in range(len(b)):
        if w[k] < 1e-12:
