Key improvements over previous: better LSE schedule, direct subgradient phase,

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        s = sum(a_list)
        if s < 0.01:
            return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def project_simplex(v):
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            _lp_solver = solver_name
            return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp(a0, time_limit, solver=None, max_iters=1000000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_logsumexp(a0, time_limit, T_start=100.0, T_end=0.5, n_stages=25, max_iters_per_stage=100):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        if stage_T < 5.0:
            iters = max_iters_per_stage * 3
        elif stage_T < 20.0:
            iters = max_iters_per_stage * 2
        else:
            iters = max_iters_per_stage

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, stage_T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, stage_T, n)
            return grad

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-30,
                         'gtol': 1e-18, 'maxfun': iters * 500,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(res.x)
            sc = score_normalized(a_check)
            if (si + 1) % 5 == 0 or si == n_stages - 1:
                print(f"    LSE stage={si+1}/{n_stages} T={stage_T:.2f}: exact_score={sc:.8f}")
        except Exception:
            pass

    return softmax_from_z(z)


def compute_subgradient(a, epsilon=0.01):
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    threshold = max_b * (1.0 - epsilon)
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])

    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            g[j] += a[k - j]
    g /= len(active)
    return g, b, active


def subgradient_momentum(a0, time_limit, n_steps=200000, momentum=0.9, eta_start=0.1, eta_end=0.001):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    v = np.zeros(n)
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        g, b, active = compute_subgradient(a, epsilon=0.02)
        eta = eta_start * (eta_end / eta_start) ** (step / max(1, n_steps))
        v = momentum * v - eta * g
        a_new = a + v
        a_new = project_simplex(a_new)

        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

        if (step + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGradMOM step={step+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, n_sweeps=500):
    n = len(a0)
    z = z_from_a(a0).copy()
    best_a = softmax_from_z(z).copy()
    best_score = score_normalized(best_a)
    t0 = time.monotonic()

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            a = softmax_from_z(z)
            b = fast_convolve(a)

            z_j_try = z[j] + 0.1
            a_try = softmax_from_z(np.concatenate([z[:j], [z_j_try], z[j+1:]]))
            sc_try = score_normalized(a_try)

            z_j_try = z[j] - 0.1
            a_try2 = softmax_from_z(np.concatenate([z[:j], [z_j_try], z[j+1:]]))
            sc_try2 = score_normalized(a_try2)

            if sc_try < best_score:
                z[j] += 0.1
                best_score = sc_try
                best_a = softmax_from_z(z).copy()
            elif sc_try2 < best_score:
                z[j] -= 0.1
                best_score = sc_try2
                best_a = softmax_from_z(z).copy()
            else:
                z[j] += 0.05
                a_try3 = softmax_from_z(z)
                sc3 = score_normalized(a_try3)
                if sc3 < best_score:
                    best_score = sc3
                    best_a = a_try3.copy()
                else:
                    z[j] -= 0.15
                    z[j] += 0.1
                    a_try4 = softmax_from_z(z)
                    sc4 = score_normalized(a_try4)
                    if sc4 < best_score:
                        best_score = sc4
                        best_a = a_try4.copy()

        if (sweep + 1) % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDescent sweep={sweep+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        xx = np.linspace(-1, 1, n)
        a = np.exp(-2 * xx ** 2)
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif choice == 4:
        a = np.ones(n) * (1.0 + 0.2 * np.cos(2 * np.pi * 3 * np.arange(n) / n))
    elif choice == 5:
        xx = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(3 * xx) + 0.15 * np.cos(7 * xx)
    elif choice == 6:
        a = np.ones(n) * (1.0 + 0.1 * np.sin(2 * np.pi * 5 * x))
    else:
        a = 0.7 + 0.3 * np.cos(2 * np.pi * 2 * x) * np.cos(2 * np.pi * 3 * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def multi_restart_lse(a0, time_limit, n_restarts=3, T_start=20.0, T_end=0.3, n_stages=15):
    n = len(a0)
    best_a = a0.copy()
    best_score = score_normalized(a0)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for restart in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 30:
            break

        if restart == 0:
            start_a = best_a.copy()
        else:
            z = z_from_a(best_a)
            noise_scale = 0.1 * (n_restarts - restart) / n_restarts
            z = z + noise_scale * rng.standard_normal(n)
            start_a = softmax_from_z(z)

        per_restart_time = remaining / (n_restarts - restart)
        a_res = lbgfs_logsumexp(start_a, per_restart_time, T_start=T_start,
                                 T_end=T_end, n_stages=n_stages, max_iters_per_stage=50)
        sc = score_normalized(a_res)
        if sc < best_score:
            best_score = sc
            best_a = a_res.copy()
            print(f"    MultiRestart LSE restart={restart}: score={best_score:.8f}")

    return best_a, best_score

            best_seq = p / p.sum()

            return

        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    t1 = 0.40 * total_budget

    # === Phase 4: Coordinate descent polish (10% of budget) ===

    remaining = global_deadline - time.monotonic()
    if remaining > 30:
        n_candidates = [n_base + 50, n_base - 50, n_base + 100, n_base - 100, n_base + 150]
        n_candidates = sorted(set([nc for nc in n_candidates if nc >= 200]))
        per_n_time = remaining / (len(n_candidates) + 1)

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
