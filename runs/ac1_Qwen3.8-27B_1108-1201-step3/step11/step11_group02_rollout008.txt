def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_local(a):
    """Compute score locally (scale-invariant)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:

    return cp.ECOS


def solve_best_response_lp(a, solver):
    """Given fixed a (sum=1, nonneg), find b minimizing max conv(a,b)."""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    
    # Build constraint matrix: conv(a, b)[k] = sum_j a[k-j]*b[j] for valid k-j
    # T[k, j] = a[k-j] if 0 <= k-j < n
    row_ind = []
    col_ind = []
    data = []
    for j in range(n):
        for k in range(j, j + n):
            row_ind.append(k)
            col_ind.append(j)
            data.append(a[k - j])
    
    T = csr_matrix((data, (row_ind, col_ind)), shape=(size, n))
    ones = np.ones(size)
    
    constraints = [T @ b_var <= t_var * ones, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s > 1e-15:
                b_val /= s
                return b_val

        pass

    return np.ones(n) / n

def logsumexp_grad(a, T):

    obj = (2n/T) * LogSumExp(conv(a,a)/T)

    # Wait, that's only if the convolution is a*a (same sequence).
    # Actually: d/da[i] [sum_j a[j] a[k-j]] = a[k-i] (from j=i term) + sum_j a[j] * delta_{k-j,i}
    # = a[k-i] + a[k-i] = 2*a[k-i] (when 0 <= k-i < n)
    
    # So grad[i] = (4n/T) * sum_{k: 0<=k-i<n} w[k] * a[k-i]
    # Let m = k-i, then k = m+i, and we need 0 <= m < n and 0 <= k < 2n-1
    # So m ranges from max(0, -i) to min(n-1, 2n-2-i), which is just 0 to min(n-1, 2n-2-i)

    grad = np.zeros(n)

    obj = (2.0 * n / T) * (max_b / T + np.log(exp_b.sum()))
    return obj, grad

        # Estimate max iterations from time budget
        # Each iteration: one func + one jac evaluation, roughly
        # Let's just set a reasonable maxiter
        maxiter = max(50, int(time_per_stage * 50))  # rough estimate
        
        try:
            res = minimize(fun=func, x0=a, jac=jac, method='L-BFGS-B',
                           options={'maxiter': maxiter, 'ftol': 1e-18,
                                    'gtol': 1e-14, 'maxfun': maxiter * 100,
                                    'disp': False})
            a = np.maximum(res.x, 0)
            sa = a.sum()
            if sa > 1e-15:
                a = a / sa
        except Exception as e:
            print(f"    L-BFGS error at T={T}: {e}")
        
        current_score = score_local(a)
        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()
        
        elapsed = time.monotonic() - t0
        print(f"    LSE T={T:.2f} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score


def coordinate_descent(a0, time_limit):
    """Coordinate descent focusing on peak-contributing elements."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_score = score_local(a)
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        max_b = b.max()
        # Find top contributing coordinates to the max
        peak_k = int(np.argmax(b))
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Try reducing top contributors
        top_m = min(5, len(idx))
        top_positions = idx[np.argsort(contribs)[::-1][:top_m]]
        
        improved = False
        for pos in top_positions:
            if a[pos] < 1e-12:
                continue
            for factor in [0.999, 0.997, 0.995, 0.99, 0.98, 0.97, 0.95]:
                a_trial = a.copy()
                a_trial[pos] *= factor
                # Redistribute
                diff = a[pos] - a_trial[pos]
                if diff < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    break
                a_trial[mask] += diff * a[mask] / a[mask].sum()
                score = score_local(a_trial)
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall > 500:
                # Try simultaneous reduction
                a_trial = a.copy()
                for pos in top_positions[:3]:
                    if a_trial[pos] > 1e-12:
                        a_trial[pos] *= 0.999
                a_trial = a_trial / a_trial.sum()
                score = score_local(a_trial)
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
                elif stall > 2000:
                    break
        else:
            stall = 0
    
    return best_a, best_score


def structured_init(n, rng):
    """Create structured initial sequence."""
    x = np.linspace(0, 1, n)
    choice = rng.integers(0, 8)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = np.exp(-3 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = np.exp(-2.0 * (x - 0.5) ** 2 / 0.2)
    elif choice == 5:
        a = np.exp(-8 * (x - 0.3) ** 2) + np.exp(-8 * (x - 0.7) ** 2)
    elif choice == 6:
        # Flat with slight edge reduction
        a = np.ones(n) * (1 + 0.1 * np.cos(np.pi * x))
    else:
        a = 1.0 / (1.0 + np.exp(-5 * (x - 0.5)))  # Sigmoid-like
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def fast_lp_solve(a, solver, prev_b=None):
    """Fast LP solve with optional warm start."""
    n = len(a)
    size = 2 * n - 1
    
    # Build sparse matrix more efficiently
    row_ind = np.zeros(n * n, dtype=np.int32)
    col_ind = np.zeros(n * n, dtype=np.int32)
    data = np.zeros(n * n, dtype=np.float64)
    
    pos = 0
    for j in range(n):
        k_start = j
        k_end = min(j + n, size)
        length = k_end - k_start
        row_ind[pos:pos+length] = np.arange(k_start, k_end)
        col_ind[pos:pos+length] = j
        data[pos:pos+length] = a[:length]
        pos += length
    
    T = csr_matrix((data[:pos], (row_ind[:pos], col_ind[:pos])), shape=(size, n))
    
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    ones = np.ones(size)
    
    constraints = [T @ b_var <= t_var * ones, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s > 1e-15:
                b_val /= s
                return b_val
    except Exception:
        pass
    
    if prev_b is not None:
        return prev_b.copy()
    return np.ones(n) / n


def refined_nesterov_lp(a0, time_limit, solver):
    """
    Improved Nesterov LP with:
    - Efficient matrix construction
    - Better restart strategy (based on duality gap proxy)
    - Adaptive exploration
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    y = a.copy()
    best_a = a.copy()
    best_score = score_local(a)
    
    # FTRL weighted average
    avg_a = a.copy()
    weight_sum = 1.0

    rng = np.random.default_rng(777)
    iter_count = 0

    col_template = np.zeros(n * n, dtype=np.int32)
    pos = 0
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        length = k_end - k_start
        row_template[pos:pos+length] = np.arange(k_start, k_end)
        col_template[pos:pos+length] = j
        pos += length
    row_template = row_template[:pos]
    col_template = col_template[:pos]
    
    for it in range(2000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Build T with current y
        data = np.zeros(pos, dtype=np.float64)
        dpos = 0
        for j in range(n):
            k_start = j
            k_end = min(j + n, 2 * n - 1)
            length = k_end - k_start
            data[dpos:dpos+length] = y[:length]
            dpos += length
        
        T = csr_matrix((data, (row_template, col_template)), shape=(2*n-1, n))
        
        # Solve LP
        b_var = cp.Variable(n, nonneg=True)
        t_var = cp.Variable()
        constraints = [T @ b_var <= t_var, cp.sum(b_var) == 1]
        # Note: t_var * ones is equivalent to just t_var (scalar broadcast)
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        
        b_new = None
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_new = np.asarray(b_var.value, dtype=np.float64)
                b_new = np.maximum(b_new, 0)
                sn = b_new.sum()
                if sn > 1e-15:
                    b_new /= sn
                else:
                    b_new = None
        except Exception:
            b_new = None
        
        if b_new is None:
            b_new = np.ones(n) / n
        
        iter_count += 1

        alpha = 2.0 / (np.sqrt(iter_count) + 1.0)

        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy
        
        # FTRL update
        avg_a = (weight_sum * avg_a + alpha * (iter_count + 1) * b_new) / (weight_sum + alpha * (iter_count + 1))
        weight_sum += alpha * (iter_count + 1)
        avg_a = np.maximum(avg_a, 0)
        sa = avg_a.sum()
        if sa < 1e-15:
            avg_a = y.copy()
            sa = 1.0
        avg_a = avg_a / sa
        
        # Evaluate
        sc_a = score_local(avg_a)
        sc_y = score_local(y)
        
        if sc_a < best_score:
            best_score = sc_a
            best_a = avg_a.copy()
            stall = 0
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            stall = 0
        
        # Monotone Nesterov: if new score is worse, don't move
        # (simplified: just check periodically)

            print(f"    RNL iter={it+1} ({elapsed:.0f}s/{time_limit:.0f}s): score={best_score:.8f}, stall={stall}")
