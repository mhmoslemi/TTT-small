from scipy.fft import rfft, irfft


def fast_auto_conv(a):

def fast_cross_corr(w, a):

    n_w = len(w)
    n_a = len(a)
    size = n_w + n_a - 1
    N = 1 << (size - 1).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    result = irfft(W * np.conj(A), N)[:n_a]
    return result


def softmax_from_z(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def z_from_a(a):
    """Log of a (with floor for numerical stability)."""
    return np.log(np.maximum(a, 1e-300))


def evaluate_local(a):
    """Compute J(a) = 2n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_auto_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix_row(a, k):
    """Row k of the convolution matrix T (such that (T@b)[k] = conv(a,b)[k])."""
    n = len(a)
    row = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        row[j] = a[k - j]
    return row


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for k in range(size):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            T[k, j] = a[k - j]
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s < 1e-15:
                return None
            return b_val / s
    except Exception:
        pass
    return None


def dual_adam_optimize(a0, time_limit, T_schedule, lr_init, lr_decay):
    """
    Dual Adam optimization in log-space.
    Maintains z_a (for a) and z_b (for b), does saddle point optimization.
    Minimizes LogSumExp(conv(a,b)) w.r.t. a, maximizes w.r.t. b.
    """
    n = len(a0)
    z_a = z_from_a(a0).copy()
    z_b = z_from_a(a0).copy()  # Initialize b same as a
    
    m_a = np.zeros(n)
    v_a = np.zeros(n)
    m_b = np.zeros(n)
    v_b = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    best_a = a0.copy()
    best_score = evaluate_local(a0)
    
    t0 = time.monotonic()
    deadline = t0 + time_limit
    step = 0
    
    num_stages = len(T_schedule)
    
    for stage_idx, T in enumerate(T_schedule):
        if time.monotonic() > deadline:
            break
        
        # Number of steps per stage
        remaining_stages = num_stages - stage_idx
        steps_per_stage = max(1000, int((deadline - time.monotonic()) * 2000 / remaining_stages))
        
        for _ in range(steps_per_stage):
            if time.monotonic() > deadline:
                break
            step += 1
            
            a = softmax_from_z(z_a)
            b = softmax_from_z(z_b)
            
            # Compute cross-convolution
            c = fast_cross_conv(a, b)  # length 2n-1
            
            # LogSumExp weights
            max_c = c.max()
            exp_c = np.exp((c - max_c) / T)
            w = exp_c / exp_c.sum()
            
            # Check score periodically
            if step % 500 == 0:
                a_score = evaluate_local(a)
                if a_score < best_score:
                    best_score = a_score
                    best_a = a.copy()
            
            # Gradients
            # grad_a[i] = (1/T) * corr(w, b)[i]
            # grad_b[i] = (1/T) * corr(w, a)[i]
            grad_a = fast_cross_corr(w, b) / T
            grad_b = fast_cross_corr(w, a) / T
            
            # Transform to z-space
            S_a = np.dot(grad_a, a)
            grad_z_a = a * (grad_a - S_a)
            
            S_b = np.dot(grad_b, b)
            grad_z_b = b * (grad_b - S_b)
            
            # Adam updates
            lr = lr_init * (lr_decay ** stage_idx)
            
            # a: minimize (descent)
            m_a = beta1 * m_a + (1.0 - beta1) * grad_z_a
            v_a = beta2 * v_a + (1.0 - beta2) * grad_z_a ** 2
            m_a_hat = m_a / (1 - beta1 ** (step + 1))
            v_a_hat = v_a / (1 - beta2 ** (step + 1))
            z_a = z_a - lr * m_a_hat / (np.sqrt(v_a_hat) + eps)
            z_a -= z_a.mean()
            
            # b: maximize (ascent)
            m_b = beta1 * m_b + (1.0 - beta1) * grad_z_b
            v_b = beta2 * v_b + (1.0 - beta2) * grad_z_b ** 2
            m_b_hat = m_b / (1 - beta1 ** (step + 1))
            v_b_hat = v_b / (1 - beta2 ** (step + 1))
            z_b = z_b + lr * m_b_hat / (np.sqrt(v_b_hat) + eps)
            z_b -= z_b.mean()
            
            if step % 10000 == 0:
                elapsed = time.monotonic() - t0
                cur_score = evaluate_local(softmax_from_z(z_a))
                print(f"    DualAdam stage {stage_idx} (T={T:.2e}) step {step} ({elapsed:.0f}s): best={best_score:.8f} cur={cur_score:.8f}")
    
    # Final score check
    a_final = softmax_from_z(z_a)
    final_score = evaluate_local(a_final)
    if final_score < best_score:
        best_score = final_score
        best_a = a_final.copy()
    
    return best_a, best_score

    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP: given a (as y), find best response b
        b_new = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        
        # Nesterov step
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa > 1e-15:
            a = a / sa
        
        # Check scores
        score_a = evaluate_local(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        score_y = evaluate_local(y)
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if (it + 1) % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {it+1} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score


def structured_init_asymmetric(n, rng):
    """Create diverse asymmetric initial sequences."""
    choice = rng.integers(0, 12)
    x = np.linspace(0, 1, n)
    
    if choice == 0:
        # Exponential decay
        a = np.exp(-2.0 * x)
    elif choice == 1:
        # Exponential rise
        a = np.exp(2.0 * x)
    elif choice == 2:
        # Asymmetric Gaussian (peak at 0.3)
        a = np.exp(-4.0 * (x - 0.3) ** 2)
    elif choice == 3:
        # Asymmetric Gaussian (peak at 0.7)
        a = np.exp(-4.0 * (x - 0.7) ** 2)
    elif choice == 4:
        # Linear ramp (ascending)
        a = 0.5 + x
    elif choice == 5:
        # Linear ramp (descending)
        a = 1.5 - x
    elif choice == 6:
        # Sigmoid (steep rise)
        a = 1.0 / (1.0 + np.exp(-10.0 * (x - 0.5)))
    elif choice == 7:
        # Sigmoid (steep fall)
        a = 1.0 / (1.0 + np.exp(10.0 * (x - 0.5)))
    elif choice == 8:
        # Power law
        a = x ** 0.5 + 0.1
    elif choice == 9:
        # Polynomial tilt
        a = 1.0 + 0.5 * x - 0.3 * x ** 2
        a = np.maximum(a, 0)
    elif choice == 10:
        # Asymmetric bump + noise
        a = np.exp(-3.0 * (x - 0.4) ** 2) + 0.3 * np.exp(-3.0 * (x - 0.7) ** 2)
    else:
        # Random smooth
        freq = rng.integers(1, 6)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * freq * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
