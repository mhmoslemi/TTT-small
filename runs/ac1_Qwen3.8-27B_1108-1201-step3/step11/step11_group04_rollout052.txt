def mirror_descent_best_response(a, b_init, n_md_steps=100, eta=0.1, n_top=5):
    """
    Approximate best response via mirror descent (multiplicative weights).
    Given fixed a, find b minimizing max_k conv(a,b)_k.
    """
    n = len(a)
    b = np.maximum(b_init.astype(np.float64).copy(), 0)
    s = b.sum()
    if s < 1e-15:
        b = np.ones(n) / n
    else:
        b = b / s

    for step in range(n_md_steps):
        c = fast_convolve_ab(a, b)
        max_c = c.max()
        
        # Find top peaks
        threshold = max_c * (1.0 - 5e-4)
        active = np.where(c >= threshold)[0]
        if len(active) > n_top:
            idx = np.argsort(c[active])[-n_top:]
            active = active[idx]
        
        # Compute subgradient (weighted combination of active peak rows)
        weights = c[active] - max_c * (1.0 - 5e-4)
        weights = np.maximum(weights, 0)
        w_sum = weights.sum()
        if w_sum < 1e-30:
            weights = np.ones(len(active)) / len(active)
        else:
            weights = weights / w_sum
        
        g = np.zeros(n)
        for i, k in enumerate(active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += weights[i] * a[k - idx]
        
        # Multiplicative update
        log_scale = -eta * g
        log_scale -= log_scale.max()
        b_new = b * np.exp(log_scale)
        b_new = np.maximum(b_new, 0)
        s_new = b_new.sum()
        if s_new < 1e-15:
            b_new = b.copy()
        else:
            b_new = b_new / s_new
        b = b_new

    return b


def solve_lp_best_response(a_fixed, solver, b_init=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build constraint matrix (sparse-friendly)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solvers_to_try = [solver, cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.GUROBI, cp.MOSEK]
    solved = False
    for s in solvers_to_try:

            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solved = True
                break
        except Exception:
            continue

    if not solved:
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(max(0, M_var.value))


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage, deadline=None):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    if deadline is None:
        deadline = t0 + time_limit
    n_stages = len(T_stages)
    iters_per_stage = max(30, maxiter_per_stage // n_stages)

    for stage_i, stage_T in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 3:
            break

        stages_left = n_stages - stage_i
        avail_time = remaining
        iters = max(20, min(iters_per_stage * 3, int(avail_time * 50 / max(1, stages_left))))

        def func(z_in):
            if time.monotonic() > deadline:
                return 1e15
            obj, _ = logsumexp_obj_grad(z_in, n, stage_T)
            return obj

        def jac(z_in):
            if time.monotonic() > deadline:
                return np.zeros_like(z_in)
            _, g = logsumexp_obj_grad(z_in, n, stage_T)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-25,
                         'gtol': 1e-18, 'maxfun': iters * 500, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def spectral_flatten(a, n_iters=50, blend=0.4, deadline=None):
    """
    Iterative spectral flattening (Gerchberg-Saxton style).
    """
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)

        mags = np.abs(F)
        non_dc = mags[1:]
        if non_dc.min() < 1e-15:
            target = np.mean(non_dc)
        else:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        phases = np.angle(F)
        F_new = F.copy()
        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend) * mags[k] + blend * target
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])

        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new

    return a

            if (it + 1) % 2000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP-Nesterov iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def multiplicative_subgradient(a0, time_limit, n_steps=300000, eta_base=0.03):
    """Mirror descent (multiplicative weights) directly on a."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    no_improve = 0
    scale = 1.0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        threshold = max_b * (1.0 - 2e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        eta = eta_base * scale
        log_scale = -eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 40:
            scale *= 0.7
            no_improve = 0
            if scale < 0.005:
                break

    return best_a, best_score


def structured_init(n, rng):
    """Create a structured initial sequence."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 6:
        center = n // 2
        t = (np.arange(n) - center) / center
        a = np.sinc(t)
        a = np.maximum(a, 0)
    else:
        a = 1.0 - np.abs(2 * x - 1)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def perturbed_sequence(a, rng, strength=0.03):
    """Create a perturbed version of sequence a."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 8)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Primary: Fast mirror-descent Nesterov alternating.
    Polish: LP + L-BFGS LogSumExp.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    rem = deadline - time.monotonic()
    print(f"Total budget: {rem:.0f}s, starting score: {best_score:.8f}")

    # === Phase 0: Spectral Flattening (15s) ===
    print(f"\n=== Phase 0: Spectral Flattening (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_spec = min(rem * 0.01, 15.0)
        a_spec = spectral_flatten(best_seq, n_iters=100, blend=0.25, deadline=time.monotonic() + t_spec)
        local_update(a_spec, "spectral_0")
        a_spec2 = spectral_flatten(best_seq, n_iters=50, blend=0.5, deadline=time.monotonic() + min(t_spec, 8))
        local_update(a_spec2, "spectral_1")

    # === Phase 1: Fast Mirror Descent Nesterov (main workhorse, ~700s) ===
    print(f"=== Phase 1: MD-Nesterov (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_md = min(rem * 0.40, 700.0)
    if t_md > 30:
        print(f"  Budget: {t_md:.0f}s")
        a_md, _ = nesterov_mirror_descent(best_seq, t_md, n_md_steps=60, eta_md=0.06, n_top=5)
        local_update(a_md, "md_nesterov_1")
        
        # Second round with different parameters
        rem = deadline - time.monotonic()
        if rem > 60:
            t_md2 = min(rem * 0.15, 200.0)
            a_md2, _ = nesterov_mirror_descent(best_seq, t_md2, n_md_steps=100, eta_md=0.04, n_top=3)
            local_update(a_md2, "md_nesterov_2")

    # === Phase 2: LP Polish (~200s) ===
    print(f"=== Phase 2: LP Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_lp = min(rem * 0.12, 200.0)
    if t_lp > 30:
        print(f"  Budget: {t_lp:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_lp, solver, n_max_iters=500000)
        local_update(a_lp, "lp_polish_1")

    # === Phase 3: L-BFGS LogSumExp (~300s) ===
    print(f"=== Phase 3: L-BFGS LogSumExp (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_lbfgs = min(rem * 0.20, 300.0)
    if t_lbfgs > 30:
        T_stages = [5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1,
                    0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001]
        print(f"  T_stages: {len(T_stages)}, budget={t_lbfgs:.0f}s")
        a_lb = lbgfs_refine(best_seq, t_lbfgs, T_stages, maxiter_per_stage=15000,
                            deadline=time.monotonic() + t_lbfgs)
        local_update(a_lb, "lbgfs_fine")

        # Second L-BFGS pass with even lower temperatures
        rem = deadline - time.monotonic()
        if rem > 30:
            T_stages2 = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
            t_lbfgs2 = min(rem * 0.08, 100.0)
            a_lb2 = lbgfs_refine(best_seq, t_lbfgs2, T_stages2, maxiter_per_stage=10000,
                                 deadline=time.monotonic() + t_lbfgs2)
            local_update(a_lb2, "lbgfs_ultra_fine")

    # === Phase 4: Multiplicative subgradient directly (~100s) ===
    print(f"=== Phase 4: Multiplicative Subgradient (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_ms = min(rem * 0.06, 100.0)
    if t_ms > 10:
        a_ms, _ = multiplicative_subgradient(best_seq, t_ms, n_steps=200000, eta_base=0.02)
        local_update(a_ms, "mult_subgrad")

    # === Phase 5: Try different n (~100s) ===
    print(f"=== Phase 5: Different n (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        # Try n=1000
        t_n1000 = min(rem * 0.10, 100.0)
        if t_n1000 > 30:
            print(f"  Trying n=1000, budget={t_n1000:.0f}s")
            init_1000 = interpolate_sequence(best_seq, 1000)
            a_sp1000 = spectral_flatten(init_1000, n_iters=30, blend=0.3)
            a_1000, _ = nesterov_mirror_descent(a_sp1000, t_n1000 * 0.6, n_md_steps=60, eta_md=0.06)
            local_update(a_1000, "md_n1000")

    rem = deadline - time.monotonic()
    if rem > 60:
        # Try n=1500
        t_n1500 = min(rem * 0.12, 120.0)
        if t_n1500 > 30:
            print(f"  Trying n=1500, budget={t_n1500:.0f}s")
            init_1500 = interpolate_sequence(best_seq, 1500)
            a_sp1500 = spectral_flatten(init_1500, n_iters=30, blend=0.3)
            a_1500, _ = nesterov_mirror_descent(a_sp1500, t_n1500 * 0.6, n_md_steps=60, eta_md=0.06)
            local_update(a_1500, "md_n1500")

    # === Phase 6: Perturbation escape + re-optimize ===
    print(f"=== Phase 6: Perturbation Escape (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        for trial in range(3):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 30:
                break
            a_pert = perturbed_sequence(best_seq, rng, strength=0.01 + 0.008 * trial)
            a_sp = spectral_flatten(a_pert, n_iters=15, blend=0.3)
            t_sub = min(rem_sub * 0.20, 50.0)
            if t_sub > 10:
                a_opt, _ = nesterov_mirror_descent(a_sp, t_sub, n_md_steps=60, eta_md=0.05)
                local_update(a_opt, f"pert_trial{trial}")

    # === Phase 7: Final polish ===
    print(f"=== Phase 7: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.4, 50.0)
        if t_fin > 10:
            T_stages_fin = [2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
            a_fin = lbgfs_refine(best_seq, t_fin, T_stages_fin, maxiter_per_stage=5000,
                                 deadline=time.monotonic() + t_fin)
            local_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_flp = min(rem * 0.5, 30.0)
        if t_flp > 5:
            a_flp, _ = nesterov_lp_refine(best_seq, t_flp, solver, n_max_iters=20000)
            local_update(a_flp, "final_lp")

    # Final verification and output
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_start
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}) at {elapsed:.1f}s")
        test_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        try:
            eval_score = evaluate_sequence(test_seq)
            print(f"Verified with evaluate_sequence: {eval_score:.8f}")
        except Exception as e:
            print(f"Verification error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
