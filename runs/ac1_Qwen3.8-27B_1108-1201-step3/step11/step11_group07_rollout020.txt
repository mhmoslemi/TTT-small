import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate score locally (fast version)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def lse_compute(a, T):
    """Compute LSE objective value and softmax weights."""
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    w = exp_c / exp_c.sum()
    lse = c_max + T * np.log(exp_c.sum())
    obj = 2.0 * n * lse
    return obj, w, c


def lse_gradient_corrected(a, T):
    """
    Compute the CORRECT gradient of 2n*LSE(conv(a,a), T) w.r.t. a.
    
    Key fix: The gradient is 4n * cross_correlation(w, a) with NO diagonal correction.
    The previous code incorrectly subtracted a[m]*w[2m].
    
    Returns gradient (unnormalized, for use with L-BFGS in log-param space).
    """
    n = len(a)
    _, w, _ = lse_compute(a, T)
    
    # b[m] = sum_k w[k]*a[k-m] for m=0..n-1
    # This is the cross-correlation of w and a
    N = 1 << (len(w) + n - 1).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    R = irfft(W * np.conj(A), N)
    b = R[:n]
    
    # Correct gradient: 4n * b[m] (no diagonal correction needed)
    grad = 4.0 * n * b
    return grad


def lse_obj_grad_z(z, n, T):
    """Compute LSE objective and gradient in log-parameterization.
    z are log-probabilities, a = softmax(z).
    Uses corrected gradient.
    """
    a = softmax_from_z(z)
    grad_a = lse_gradient_corrected(a, T)
    obj, _, _ = lse_compute(a, T)
    
    # Transform gradient to z-space: grad_z = a * (grad_a - sum(grad_a * a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return obj, grad_z


def lse_optimize(a0, time_limit, T_stages, maxiter_per_stage=5000):
    """L-BFGS optimization with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_z = z.copy()
    best_obj = float('inf')

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        def func(z_in):
            obj, _ = lse_obj_grad_z(z_in, n, stage_T)
            return obj

        def jac(z_in):
            _, g = lse_obj_grad_z(z_in, n, stage_T)
            return g

        iters = min(maxiter_per_stage, max(20, int(remaining * 50)))
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-20,
                         'gtol': 1e-18, 'maxfun': iters * 100, 'disp': False}
            )
            z = res.x
            obj_val = func(z)
            a_val = softmax_from_z(z)
            sc = evaluate_local(a_val)
            if sc < best_obj:
                best_obj = sc
                best_z = z.copy()
        except Exception:
            pass

    return softmax_from_z(best_z)


def spectral_refine(a0, time_limit, n_iters=5000):
    """Refine by flattening power spectrum to reduce autocorrelation peaks."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    t0 = time.monotonic()
    lr = 0.003

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        A = rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()

        # Direction to flatten P
        r = (P - P_mean) / (P_mean + 1e-15)

        H = np.zeros(fft_size // 2 + 1, dtype=np.complex128)
        mask = P > 1e-10
        H[mask] = -r[mask] * P_mean * np.conj(A[mask]) / (P[mask] + 1e-10)
        H[0] = 0.0
        h = irfft(H, fft_size)[:n]
        h -= h.mean()
        h_norm = np.linalg.norm(h)
        if h_norm < 1e-15:
            break
        h = h / h_norm

        improved = False
        for scale in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            a_new = a + scale * h
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new = a_new / s
            sc = evaluate_local(a_new)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
                improved = True
            break

        if not improved and it % 200 == 0:
            lr *= 0.85

    return best_a, best_score


def peak_directed_refine(a0, time_limit, n_steps=5000):
    """Direct peak reduction using subgradient at active peaks."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    eta = 0.05

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        c_max = c.max()

        # Active set: peaks within 1% of max
        threshold = c_max * 0.999
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])

        # Subgradient: average of dc[k]/da over active peaks
        # dc[k]/da[m] = 2*a[k-m] (for valid k-m)
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for m in range(js, je + 1):
                g[m] += 2.0 * a[k - m]
        g /= len(active)

        # Projected gradient descent on simplex
        # Use multiplicative update for non-negativity
        log_alpha = -eta * g
        log_alpha -= log_alpha.max()
        a_new = a * np.exp(log_alpha)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s
        sc = evaluate_local(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            eta *= 0.95
            if eta < 1e-8:
                break

    return best_a, best_score

    candidates.append(normalize(a))

    # Random smooth (Fourier synthesis)
    coeffs = rng.normal(0, 1, 15)
    phases = rng.uniform(0, 2 * np.pi, 15)
    a = np.zeros(n)
    for i in range(len(coeffs)):
        a += coeffs[i] * np.cos(2 * np.pi * (i + 1) * x + phases[i])
    a = np.maximum(a + 1.0, 0)
    candidates.append(normalize(a))

    # Uniform
    candidates.append(np.ones(n) / n)

    # Edge-peaked
    a = np.exp(-5 * x) + np.exp(-5 * (1 - x))
    candidates.append(normalize(a))

    # Flat-top window
    a = np.ones(n) / n
    w = 0.2
    a[:int(w*n)] = 0.3
    a[-int(w*n):] = 0.3
    candidates.append(normalize(a))

    return candidates


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    return normalize(f_interp)


def perturb_sequence(a, rng, method='gaussian', strength=0.03):
    """Perturb a sequence using various methods."""
    n = len(a)
    if method == 'gaussian':
        noise = rng.normal(0, strength, n)
        a_new = a + noise
    elif method == 'low_freq':
        freq = rng.integers(1, 10)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.linspace(0, 2 * np.pi, n)
        noise = strength * np.cos(freq * x + phase)
        a_new = a + noise
    elif method == 'spec_noise':
        # Add noise in frequency domain
        fft_size = 1 << (2 * n - 1).bit_length()
        A = rfft(a, fft_size)
        mask = np.zeros(fft_size // 2 + 1, dtype=bool)
        mask[rng.integers(0, fft_size // 2 + 1, rng.integers(10, 50))] = True
        noise_spec = rng.normal(0, strength, sum(mask)) + 1j * rng.normal(0, strength, sum(mask))
        A[mask] += noise_spec
        a_new = irfft(A, fft_size)[:n]
    elif method == 'block_swap':
        a_new = a.copy()
        block_size = rng.integers(max(10, n // 50), max(20, n // 20))
        i = rng.integers(0, max(1, n - block_size))
        j = rng.integers(0, max(1, n - block_size))
        if abs(i - j) > block_size:
            a_new[i:i+block_size], a_new[j:j+block_size] = a[j:j+block_size].copy(), a[i:i+block_size].copy()
    elif method == 'mirror':
        # Reflect a segment
        a_new = a.copy()
        seg_len = rng.integers(n // 10, n // 3)
        start = rng.integers(0, n - seg_len)
        a_new[start:start+seg_len] = a_new[start:start+seg_len][::-1]
    else:
        a_new = a + rng.normal(0, strength, n)

    return normalize(a_new)


def population_search(a_init, time_limit, pop_size=12, rng=None):
    """Population-based search with adaptive mutations."""
    if rng is None:
        rng = np.random.default_rng(789)

    n = len(a_init)
    population = [normalize(a_init.copy())]
    methods = ['gaussian', 'low_freq', 'spec_noise', 'block_swap', 'mirror']
    for i in range(pop_size - 1):
        m = methods[i % len(methods)]
        strength = 0.02 + 0.02 * (i % 3)
        population.append(perturb_sequence(a_init, rng, m, strength))

    scores = [evaluate_local(p) for p in population]
    best_idx = np.argmin(scores)
    best_a = population[best_idx].copy()
    best_score = scores[best_idx]

    t0 = time.monotonic()
    gen = 0
    adaptive_strength = 0.03

    while time.monotonic() - t0 < time_limit:
        gen += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        order = np.argsort(scores)
        population = [population[i] for i in order]
        scores = [scores[i] for i in order]

        if scores[0] < best_score:
            best_score = scores[0]
            best_a = population[0].copy()

        keep = max(3, pop_size // 3)
        elites = population[:keep]

        new_pop = list(elites)

        for i in range(pop_size - len(new_pop)):
            parent = elites[rng.integers(0, keep)]
            method = methods[rng.integers(0, len(methods))]
            strength = adaptive_strength * (1 + 0.3 * np.sin(gen * 0.05))
            child = perturb_sequence(parent, rng, method, strength)
            new_pop.append(child)

        if pop_size > len(new_pop) and len(elites) >= 2:
            p1, p2 = elites[0], elites[rng.integers(1, len(elites))]
            t = rng.uniform(0.2, 0.8)
            child = normalize(t * p1 + (1 - t) * p2)
            new_pop.append(child)

        population = new_pop[:pop_size]
        scores = [evaluate_local(p) for p in population]

        if gen % 30 == 0:
            if scores[0] >= best_score - 1e-12:
                adaptive_strength *= 0.9
            else:
                adaptive_strength = min(0.08, adaptive_strength * 1.1)

        if gen % 100 == 0:
            print(f"    PopSearch gen={gen}: best={best_score:.8f} strength={adaptive_strength:.4f}")

    return best_a, best_score


def multi_start_lse(a0, time_limit, n_starts=3, n=0):
    """Run LSE optimization from multiple random starts."""
    if n == 0:
        n = len(a0)
    rng = np.random.default_rng(1234)
    
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    T_stages_short = [10, 5, 2, 1, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
    
    per_start = (time_limit - 10) / n_starts
    
    for s in range(n_starts):
        if time.monotonic() - t0 > time_limit - 10:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 20:
            break
        
        if s == 0:
            start_a = a.copy()
        else:
            start_a = perturb_sequence(a, rng, 'gaussian', strength=0.05)
        
        rem = remaining - 10
        alloc = min(per_start, rem)
        if alloc < 10:
            break
        
        result = lse_optimize(start_a, alloc, T_stages_short, maxiter_per_stage=2000)
        sc = evaluate_local(result)
        if sc < best_score:
            best_score = sc
            best_a = result.copy()
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses corrected LSE gradient, spectral shaping, population search,
    peak-directed refinement, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        candidates = diverse_init(n_init, rng)
        scores = [evaluate_local(c) for c in candidates]
        best_idx = np.argmin(scores)
        best_seq = candidates[best_idx].copy()
        best_score = scores[best_idx]
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)
    print(f"Starting: n={n_best}, score={best_score:.8f}, target=1.5027")

    # === Phase 1: Corrected LSE gradient descent (main workhorse) ===
    print(f"\n=== Phase 1: Corrected LSE Gradient (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 400.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        T_stages = [20, 10, 5, 3, 2, 1.5, 1, 0.7, 0.5, 0.3, 0.2, 0.15,
                    0.1, 0.07, 0.05, 0.03, 0.02, 0.015, 0.01, 0.007, 0.005, 0.003, 0.002]
        a_lb = lse_optimize(best_seq, t_alloc, T_stages, maxiter_per_stage=5000)
        local_update(a_lb, "lse_phase1")

    # === Phase 2: Spectral refinement ===
    print(f"\n=== Phase 2: Spectral Refine (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_spec, _ = spectral_refine(best_seq, t_alloc, n_iters=5000)
        local_update(a_spec, "spectral_1")

    # === Phase 3: Peak-directed refinement ===
    print(f"\n=== Phase 3: Peak-Directed (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_pk, _ = peak_directed_refine(best_seq, t_alloc, n_steps=10000)
        local_update(a_pk, "peak_directed")

    # === Phase 4: Population search ===
    print(f"\n=== Phase 4: Population Search (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_pop, _ = population_search(best_seq, t_alloc, pop_size=14, rng=rng)
        local_update(a_pop, "population")

    # === Phase 5: Second LSE pass (finer temperatures) ===
    print(f"\n=== Phase 5: Fine LSE Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        T_fine = [1, 0.5, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02,
                  0.015, 0.01, 0.007, 0.005, 0.003, 0.002, 0.001]
        print(f"  Budget: {t_alloc:.0f}s")
        a_lb2 = lse_optimize(best_seq, t_alloc, T_fine, maxiter_per_stage=3000)
        local_update(a_lb2, "lse_fine")

    # === Phase 6: Try longer sequence ===
    print(f"\n=== Phase 6: Length Exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        for n_try in [n_best + 200, n_best + 400, n_best - 100]:
            if n_try < 500 or n_try > 3000:
                continue
            rem_sub = deadline - time.monotonic()
            if rem_sub < 40:
                break
            init = interpolate_sequence(best_seq, n_try)
            t_sub = min(rem_sub * 0.25, 100.0)
            T_short = [10, 5, 2, 1, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
            a_try = lse_optimize(init, t_sub, T_short, maxiter_per_stage=2000)
            local_update(a_try, f"len_{n_try}")

    # === Phase 7: Multi-start from perturbed versions ===
    print(f"\n=== Phase 7: Multi-Start (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_alloc = min(rem * 0.25, 200.0)
        a_ms, _ = multi_start_lse(best_seq, t_alloc, n_starts=3)
        local_update(a_ms, "multi_start")

    # === Phase 8: Final spectral + LSE combo ===
    print(f"\n=== Phase 8: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t1 = min(rem * 0.4, 60.0)
        if t1 > 10:
            a_s, _ = spectral_refine(best_seq, t1, n_iters=3000)
            local_update(a_s, "spec_final")
        rem = deadline - time.monotonic()
        if rem > 15:
            T_final = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001]
            t2 = min(rem * 0.6, 80.0)
            if t2 > 10:
                a_f = lse_optimize(best_seq, t2, T_final, maxiter_per_stage=2000)
                local_update(a_f, "final_lse")

    # === Phase 9: Final peak-directed polish ===
    rem = deadline - time.monotonic()
    if rem > 10:
        a_pf, _ = peak_directed_refine(best_seq, min(rem * 0.5, 30.0), n_steps=5000)
        local_update(a_pf, "final_peak")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
