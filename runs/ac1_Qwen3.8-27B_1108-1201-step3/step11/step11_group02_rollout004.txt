                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others_sum
                    a_trial = a_trial / a_trial.sum()
                    score = evaluate_local(a_trial)
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            if stall > 200:
                # Try increasing a small element
                peak_k = int(np.argmax(b))
                js = max(0, peak_k - n + 1)
                je = min(n - 1, peak_k)
                idx = np.arange(js, je + 1)
                contribs = a[idx] * a[peak_k - idx]
                # Find positions with low contribution but non-zero value
                low_positions = idx[np.argsort(contribs)[:3]]
                for pos in low_positions:
                    if a[pos] < 1e-12:
                        continue
                    for factor in [1.01, 1.02, 1.05]:
                        a_trial = a.copy()
                        a_trial[pos] *= factor
                        a_trial = a_trial / a_trial.sum()
                        score = evaluate_local(a_trial)
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
                            break
                    if stall == 0:
                        break
                if stall > 1500:
                    break
        else:
            stall = 0

        if (step + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc {elapsed:.0f}s: score={best_score:.10f}")

    return best_a, best_score


def structured_init(n, rng, style=None):
    """Create structured initial sequence."""
    if style is None:
        style = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if style == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif style == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif style == 2:
        a = np.exp(-3 * x * (1 - x))
    elif style == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x)
        a = np.maximum(a, 0)
    elif style == 4:
        a = np.exp(-3.0 * (x - 0.5) ** 2 / 0.15)
    elif style == 5:
        a = np.exp(-8 * (x - 0.3) ** 2) + np.exp(-8 * (x - 0.7) ** 2)
    elif style == 6:
        # Raised cosine
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    else:
        # Bell with wider tails
        a = 1.0 / (1.0 + 4.0 * (x - 0.5) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def random_perturb(a, strength, rng):
    """Apply random perturbation and renormalize."""
    n = len(a)
    noise = rng.standard_normal(n) * strength
    a_new = a + noise
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def mirror_descent_subgrad(a0, time_limit, eta0=0.03, print_interval=5000):
    """Mirror descent (exponential weights) with subgradient."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_val = fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    log_a = np.log(np.maximum(a, 1e-300))

    for it in range(500000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        if max_b < best_val - 1e-14:
            best_val = max_b
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        threshold = max_b * 0.99
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = compute_subgradient_fft(a, active)

        eta = eta0 / (1.0 + 0.001 * it)
        if no_improve > 300:
            eta *= 0.7

        # Mirror descent step in log space
        g_shifted = g - g.min()
        log_a = log_a - eta * g_shifted
        log_a -= log_a.max()
        a = np.exp(log_a)
        a = a / a.sum()

        if (it + 1) % print_interval == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter={it + 1} ({elapsed:.0f}s): best={best_val:.10f}")

    return best_a, best_val


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Combines subgradient, LogSumExp L-BFGS, power-mean L-BFGS,
    random restarts, coordinate descent, and multi-n exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng, style=0)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    n_best = len(best_seq)

    # === Phase 1: Subgradient with momentum (25% time) ===
    print(f"\n=== Phase 1: Subgradient Momentum (score={best_score:.10f}, n={n_best}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_sg, _ = subgradient_momentum(best_seq, t_alloc, eta0=0.04, beta=0.92)
        try_update(a_sg, "SG_p1")

    # === Phase 2: LogSumExp L-BFGS (30% time) ===
    print(f"\n=== Phase 2: LogSumExp L-BFGS (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 20:
        T_stages = [30.0, 15.0, 8.0, 4.0, 2.0, 1.0, 0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01]
        print(f"  Budget: {t_alloc:.0f}s")
        a_lse, _ = logsumexp_lbgfs(best_seq, t_alloc, T_stages=T_stages, maxiter_per_stage=5000)
        try_update(a_lse, "LSE_p2")

    # === Phase 3: Power Mean L-BFGS (10% time) ===
    print(f"\n=== Phase 3: Power Mean L-BFGS (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_pm, _ = power_mean_lbgfs(best_seq, t_alloc,
                                    p_stages=[8.0, 5.0, 3.0, 2.0, 1.5, 1.2, 1.1],
                                    maxiter_per_stage=3000)
        try_update(a_pm, "PM_p3")

    # === Phase 4: Random restarts with local optimization (15% time) ===
    print(f"\n=== Phase 4: Random Restarts (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 30:
        n_restarts = min(5, max(2, int(t_alloc / 40)))
        per_restart = t_alloc / n_restarts
        print(f"  {n_restarts} restarts, {per_restart:.0f}s each")
        for ri in range(n_restarts):
            if time.monotonic() > deadline - 10:
                break
            # Diverse initialization
            style = rng.integers(0, 8)
            a_init = structured_init(n_best, rng, style=style)
            # Perturb from best
            strength = 0.05 + 0.03 * ri
            a_init = 0.5 * best_seq + 0.5 * a_init
            a_init = random_perturb(a_init, strength, rng)

            # Quick local optimization
            t_sub = min(per_restart, deadline - time.monotonic() - 5)
            if t_sub < 10:
                continue
            # Subgradient for first half
            t_sg = t_sub * 0.4
            a_sub, _ = subgradient_momentum(a_init, t_sg, eta0=0.05, beta=0.9)
            try_update(a_sub, f"restart_{ri}_sg")
            # LSE for second half
            t_lse = min(t_sub * 0.6, deadline - time.monotonic() - 5)
            if t_lse > 5:
                T_short = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05]
                a_sub2, _ = logsumexp_lbgfs(a_sub, t_lse, T_stages=T_short, maxiter_per_stage=2000)
                try_update(a_sub2, f"restart_{ri}_lse")

    # === Phase 5: Multi-n exploration (10% time) ===
    print(f"\n=== Phase 5: Multi-n Exploration (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 30:
        n_best = len(best_seq)
        candidate_ns = sorted(set([
            max(400, n_best - 300),
            max(400, n_best - 150),
            max(400, n_best + 150),
            max(400, n_best + 300),
            max(400, n_best // 2),
            n_best * 2,
        ]))
        candidate_ns = [x for x in candidate_ns if x != n_best]
        if candidate_ns:
            per_n = t_alloc / len(candidate_ns)
            print(f"  Trying n values: {candidate_ns}, {per_n:.0f}s each")
            for n_try in candidate_ns:
                if time.monotonic() > deadline - 10:
                    break
                a_interp = interpolate_sequence(best_seq, n_try)
                t_sub = min(per_n, deadline - time.monotonic() - 5)
                if t_sub < 8:
                    continue
                print(f"  n={n_try}, budget={t_sub:.0f}s")
                T_short = [10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05]
                a_sub, _ = logsumexp_lbgfs(a_interp, t_sub, T_stages=T_short, maxiter_per_stage=2000)
                try_update(a_sub, f"multi_n_{n_try}")

    # === Phase 6: Deep subgradient refinement (10% time) ===
    print(f"\n=== Phase 6: Deep Subgradient (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_deep, _ = subgradient_momentum(best_seq, t_alloc, eta0=0.02, beta=0.95,
                                         active_frac=0.999, print_interval=10000)
        try_update(a_deep, "deep_sg")

    # === Phase 7: Coordinate Descent Polish (5% time) ===
    print(f"\n=== Phase 7: Coord Descent (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.05, 80.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        a_cd, _ = coordinate_descent_refine(best_seq, t_alloc, n_steps=300000)
        try_update(a_cd, "coord_desc")

    # === Phase 8: Final very-low-T LogSumExp (5% time) ===
    print(f"\n=== Phase 8: Final Low-T LSE (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.05, 60.0)
    if t_alloc > 5:
        T_final = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
        print(f"  Budget: {t_alloc:.0f}s, T_stages={T_final}")
        a_fin, _ = logsumexp_lbgfs(best_seq, t_alloc, T_stages=T_final, maxiter_per_stage=5000)
        try_update(a_fin, "final_lse")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
