import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v, target_sum=1.0):
    """Project v onto the simplex {x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    # Scale to handle target_sum != 1
    u = v / target_sum if target_sum > 0 else v
    
    # Standard simplex projection (sum = 1)
    u_sorted = np.sort(u)[::-1]
    cssv = np.cumsum(u_sorted)
    rho = np.nonzero(u_sorted * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(u - theta, 0)
    return w * target_sum


def active_set_gradient(a, b, n, threshold_frac=0.01):
    """Compute subgradient of max(b) w.r.t. a using active set.
    
    For soft active set: weight each b[k] by how close it is to max.
    
    Returns gradient g[j] = sum_k w_k * 2*a[k-j] (for valid k-j)
    where w_k are weights (softmax of (b - max_b)/T).
    """
    b_max = b.max()
    # Softmax weights for smoothness
    T = 0.01 * b_max  # Temperature
    if T < 1e-15:
        T = 1e-15
    exp_b = np.exp((b - b_max) / T)
    w = exp_b / exp_b.sum()
    
    # g[j] = sum_k w[k] * 2*a[k-j] for 0 <= k-j < n
    # This is a cross-correlation: g = 2 * (w * a) (cross-correlation)
    # g[j] = sum_k w[k] * a[k-j] (with a[m] = 0 for m outside [0,n))
    # = (w * a)[j] where * is cross-correlation
    
    # Using FFT: cross-correlation of w and a
    # g[j] = sum_k w[k] * a[k-j] = sum_m w[m+j] * a[m]
    # This is the correlation of w and a
    # = (IFFT(conj(FFT(w)) * FFT(a)))[j]
    
    fft_size = 1 << (len(w) + n - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # Cross-correlation: R[j] = sum_k w[k+j] * a[k]
    # In frequency: FFT(R) = conj(FFT(w)) * FFT(a)
    R = irfft(np.conj(W) * A, fft_size)[:n]
    
    return 2.0 * R


def exact_line_search(a, direction, b, active_indices, max_steps=20):
    """Exact quadratic line search for the max of active b[k] values.
    
    We search for t that minimizes max_{k in active} b[k](a + t*direction)
    where b[k](a+t*d) = b[k] + 2t*(d^T C_k a) + t^2*(d^T C_k d)
    
    For each active k, the optimal t is - (d^T C_k a) / (d^T C_k d)
    (if d^T C_k d > 0)
    
    We use a bisection approach on the max of the active quadratics.
    """
    n = len(a)
    d = direction
    
    # Compute d^T C_k a and d^T C_k d for each active k
    # d^T C_k a = sum_j d[j] * a[k-j] (for valid k-j, times 2 for symmetry... 
    # actually: (d^T C_k a) = sum_i d[i] * (C_k a)[i] = sum_i d[i] * a[k-i]
    # But C_k a has (C_k a)[i] = a[k-i] + a[i-k] ... 
    # Let me just compute it directly.
    
    # Actually, for the quadratic form:
    # b[k](x) = x^T C_k x where (C_k)_{i,j} = 1 if j = k-i (mod appropriate)
    # So x^T C_k x = sum_i x[i] * x[k-i] (for valid indices)
    
    # (a + t*d)^T C_k (a + t*d) = a^T C_k a + 2t * d^T C_k a + t^2 * d^T C_k d
    # = b[k] + 2t * (sum_i d[i]*a[k-i]) + t^2 * (sum_i d[i]*d[k-i])
    
    # Let me compute linear[k] = sum_i d[i]*a[k-i] and quad[k] = sum_i d[i]*d[k-i]
    # for active k values.
    
    # linear[k] = (d cross-correlate a)[k] = sum_i d[i] * a[k-i]
    # quad[k] = (d cross-correlate d)[k] = sum_i d[i] * d[k-i]
    
    # Efficient computation via FFT
    if len(active_indices) > 0:
        fft_size = 1 << (2 * n - 1).bit_length()
        D = rfft(d, fft_size)
        A = rfft(a, fft_size)
        # linear[k] = sum_i d[i] * a[k-i]
        # = (d * a)[k] (convolution)
        linear = irfft(D * A, fft_size)[:2*n-1]
        # quad[k] = sum_i d[i] * d[k-i] = (d * d)[k]
        quad = irfft(D * D, fft_size)[:2*n-1]
    else:
        linear = np.zeros(2*n-1)
        quad = np.zeros(2*n-1)
    
    # For each active k: f_k(t) = b[k] + 2*linear[k]*t + quad[k]*t^2
    # We want to minimize max_k f_k(t) over t in [0, t_max]
    
    t_max = 1.0  # Initial upper bound
    
    # Find the t that minimizes the max
    # Use golden section or bisection
    
    # First, find a good t_max
    best_t = 0.0
    best_val = max(b[active_indices])
    
    for t_try in [0.1, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        val = b[active_indices] + 2*linear[active_indices]*t_try + quad[active_indices]*t_try**2
        v = val.max()
        if v < best_val:
            best_val = v
            best_t = t_try
    
    # Refine with bisection/golden section around best_t
    if best_t > 0:
        t_lo = max(0, best_t * 0.5)
        t_hi = best_t * 1.5
        for _ in range(30):
            t_mid1 = t_lo + 0.382 * (t_hi - t_lo)
            t_mid2 = t_lo + 0.618 * (t_hi - t_lo)
            
            v1 = (b[active_indices] + 2*linear[active_indices]*t_mid1 + quad[active_indices]*t_mid1**2).max()
            v2 = (b[active_indices] + 2*linear[active_indices]*t_mid2 + quad[active_indices]*t_mid2**2).max()
            
            if v1 < v2:
                t_hi = t_mid2
            else:
                t_lo = t_mid1
        
        best_t = (t_lo + t_hi) / 2
        best_val = (b[active_indices] + 2*linear[active_indices]*best_t + quad[active_indices]*best_t**2).max()
    
    if best_t <= 0:
        # Try negative direction? No, we're doing gradient descent, so t should be positive.
        # If the direction doesn't help, return 0
        return 0.0
    
    return best_t, best_val if isinstance(best_val, float) else 0.0


def gradient_descent_with_line_search(a0, time_limit, max_iters=50000, tol=1e-12):
    """Main optimizer: gradient descent with exact quadratic line search."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    # Track active set
    active_threshold = 0.01  # Fraction of max_b to consider active
    
    consecutive_no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        b_max = b.max()
        
        # Identify active set (indices where b is close to max)
        threshold = b_max * (1.0 - active_threshold)
        active = np.where(b >= threshold)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        # Compute gradient
        g = active_set_gradient(a, b, n, threshold_frac=active_threshold)
        
        # Project gradient to be tangent to simplex (remove mean component)
        g -= g.mean()
        
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        
        # Direction = -gradient (normalized)
        direction = -g / g_norm
        
        # Line search
        result = exact_line_search(a, direction, b, active)
        if isinstance(result, tuple):
            t_opt, _ = result
        else:
            t_opt = result
        
        if t_opt < 1e-10:
            consecutive_no_improve += 1
            if consecutive_no_improve > 100:
                active_threshold *= 0.8
                consecutive_no_improve = 0
                if active_threshold < 1e-6:
                    break
            continue
        
        # Take step
        a_new = a + t_opt * direction
        a_new = project_simplex(a_new, 1.0)
        
        sc = evaluate_local(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
            consecutive_no_improve = 0
            active_threshold = min(0.05, active_threshold * 1.1)
        else:
            # Try smaller step
            for factor in [0.5, 0.1, 0.01]:
                a_try = a + t_opt * factor * direction
                a_try = project_simplex(a_try, 1.0)
                sc_try = evaluate_local(a_try)
                if sc_try < best_score:
                    best_score = sc_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    consecutive_no_improve = 0
                    break
            else:
                consecutive_no_improve += 1
                active_threshold *= 0.9
                if consecutive_no_improve > 200:
                    break
        
        if it % 1000 == 0:
            print(f"      GD iter={it}: score={best_score:.8f}, n_active={len(active)}, thresh={active_threshold:.4f}")
    
    return best_a, best_score


def lp_direction(a, b, time_budget=5.0):
    """Solve LP to find a direction that reduces the max while maintaining constraints.
    
    Linearized: find d that maximizes sum(d) subject to
    2 * (C_k a)^T d <= 0 for active k (linearization of b[k](a+td) <= b[k])
    sum(d) = 1 (we'll scale)
    a + d >= 0 (for some step size, approximately a >= 0 is maintained)
    """
    n = len(a)
    b_max = b.max()
    threshold = b_max * 0.999
    active = np.where(b >= threshold)[0]
    
    if len(active) == 0:
        return None
    
    # Build constraint matrix A_ub such that A_ub @ d <= b_ub
    # For each active k: 2 * sum_i d[i] * a[k-i] <= 0 (linearized)
    # (C_k a)[i] = a[k-i] (for valid k-i)
    # So constraint: 2 * sum_i d[i] * a[k-i] <= 0
    
    # Actually, the linearization of b[k](a + t*d) <= b[k] at t=0:
    # 2 * d^T C_k a <= 0
    # d^T C_k a = sum_i d[i] * (C_k a)[i]
    # (C_k a)[i] = sum_j C_k[i,j] * a[j] = a[k-i] (if valid) + a[i-k] (if valid)
    # Wait, C_k[i,j] = 1 if j = k-i (with appropriate bounds)
    # So (C_k a)[i] = a[k-i] if 0 <= k-i < n
    
    # Hmm, this is the same as the gradient computation.
    # (C_k a)[i] = a[k-i] for 0<=k-i<n, else 0
    
    # So the constraint is: 2 * sum_i d[i] * a[k-i] <= 0 for each active k
    # i.e., (conv(d, a_reversed))[k] <= 0 ... this is getting complicated.
    
    # Let me just build the matrix directly.
    # Row for active index k: A[k, i] = 2 * a[k-i] if 0 <= k-i < n, else 0
    
    m = len(active)
    A_ub = np.zeros((m, n))
    for row, k in enumerate(active):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A_ub[row, i] = 2.0 * a[j]
            # Also the symmetric part: if 0 <= i-k < n, add a[i-k]
            j2 = i - k
            if 0 <= j2 < n:
                A_ub[row, i] += 2.0 * a[j2]
    
    b_ub = np.zeros(m)
    
    # Objective: minimize -sum(d) (to maximize sum(d))
    c = -np.ones(n)
    
    # Bounds: a[i] + t*d[i] >= 0 for small t. For the direction, we just need d[i] >= -a[i]/t_max
    # For simplicity, use d[i] >= -a[i] (allowing at most halving)
    bounds = [(-a[i], None) for i in range(n)]
    
    # Add equality constraint: sum(d) = 1 (we'll scale the direction)
    A_eq = np.array([[1.0] * n])
    b_eq = np.array([1.0])
    
    try:
        from scipy.optimize import linprog
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                      bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception as e:
        pass
    
    return None


def lp_refine(a, time_limit, max_rounds=100):
    """Refine using LP-based direction finding."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    for round_num in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        b_max = b.max()
        
        direction = lp_direction(a, b)
        if direction is None:
            break
        
        # Take a step in the LP direction
        for t in [0.5, 0.2, 0.1, 0.05, 0.01]:
            a_new = a + t * direction
            a_new = project_simplex(a_new, 1.0)
            sc = evaluate_local(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
                break
        
        if round_num % 20 == 0:
            print(f"      LP round={round_num}: score={best_score:.8f}")
    
    return best_a, best_score


def parametric_init(n, rng):
    """Generate diverse parametric initial sequences."""
    candidates = []
    x = np.linspace(0, 1, n)
    
    # Uniform
    candidates.append(np.ones(n) / n)
    
    # Linear ramp (increasing)
    a = np.arange(1, n + 1, dtype=np.float64)
    candidates.append(a / a.sum())
    
    # Quadratic ramp
    a = (np.arange(1, n + 1, dtype=np.float64)) ** 2
    candidates.append(a / a.sum())
    
    # Cubic ramp
    a = (np.arange(1, n + 1, dtype=np.float64)) ** 3
    candidates.append(a / a.sum())
    
    # U-shape (parabolic)
    for t in [0.5, 1.0, 2.0, 5.0]:
        a = 1.0 + t * (2 * x - 1) ** 2
        candidates.append(a / a.sum())
    
    # Inverted U (Gaussian bump)
    for sigma in [0.1, 0.2, 0.3, 0.5]:
        a = np.exp(-((x - 0.5) ** 2) / (2 * sigma ** 2))
        candidates.append(a / a.sum())
    
    # Edge-peaked (exponential)
    for t in [1, 2, 5, 10]:
        a = np.exp(t * (2 * x - 1) ** 2)
        candidates.append(a / a.sum())
    
    # Asymmetric ramp with curvature
    for p in [0.5, 1.0, 2.0, 3.0]:
        a = x ** p
        candidates.append(a / a.sum())
    
    # Two-stage: ramp up then ramp down (asymmetric)
    for split in [0.3, 0.4, 0.5, 0.6, 0.7]:
        a = np.zeros(n)
        mask1 = x < split
        mask2 = x >= split
        a[mask1] = (x[mask1] / split) ** 2
        a[mask2] = ((1 - x[mask2]) / (1 - split)) ** 2
        candidates.append(a / a.sum())
    
    # Random smooth (low-frequency perturbation of uniform)
    for _ in range(5):
        coeffs = rng.normal(0, 1, 8)
        a = np.ones(n) * 0.5
        for i, c in enumerate(coeffs):
            a += c * np.cos(2 * np.pi * (i + 1) * x + rng.uniform(0, 2 * np.pi)) * 0.1
        a = np.maximum(a, 0.01)
        candidates.append(a / a.sum())
    
    # Sine-based
    for freq in [1, 2, 3, 5]:
        a = 1.0 + 0.5 * np.sin(2 * np.pi * freq * x)
        a = np.maximum(a, 0.01)
        candidates.append(a / a.sum())
    
    # Cosine tapered (like a window function)
    for alpha in [0.5, 1.0, 2.0]:
        a = np.cos(np.pi * (x - 0.5)) ** (2 * alpha)
        a = np.maximum(a, 0.001)
        candidates.append(a / a.sum())
    
    # Power-law edges
    for p in [0.3, 0.5, 1.0, 2.0, 5.0]:
        a = (x * (1 - x)) ** p
        a = np.maximum(a, 1e-10)
        candidates.append(a / a.sum())
    
    # "Hamming-like" window
    a = 0.54 + 0.46 * np.cos(2 * np.pi * x)
    a = np.maximum(a, 0.01)
    candidates.append(a / a.sum())
    
    # "Blackman-like" window
    a = 0.42 + 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
    a = np.maximum(a, 0.001)
    candidates.append(a / a.sum())
    
    return candidates


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    s = f_interp.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / s


def perturbed(a, rng, strength=0.01, method='gaussian'):
    """Perturb a sequence."""
    n = len(a)
    if method == 'gaussian':
        noise = rng.normal(0, strength, n)
        a_new = a + noise
    elif method == 'sin':
        freq = rng.integers(1, 20)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.linspace(0, 2 * np.pi, n)
        noise = strength * np.sin(freq * x + phase)
        a_new = a + noise
    elif method == 'drift':
        # Add a linear drift
        slope = rng.uniform(-strength, strength)
        x = np.linspace(0, 1, n)
        a_new = a + slope * (x - 0.5)
    elif method == 'edge_push':
        w = rng.uniform(0, strength)
        a_new = a.copy()
        a_new[:n//10] += w
        a_new[-n//10:] += w
        a_new[n//10:-n//10] -= 2 * w * (n//10) / max(1, n - 2*(n//10))
    else:
        a_new = a + rng.normal(0, strength, n)
    
    return project_simplex(a_new, 1.0)


def multi_phase_optimize(a_init, time_limit, rng, label=""):
    """Multi-phase optimization for a single sequence."""
    n = len(a_init)
    best_a = normalize(a_init.copy())
    best_score = evaluate_local(best_a)
    t0 = time.monotonic()
    
    def remaining():
        return max(0, time_limit - (time.monotonic() - t0))
    
    # Phase 1: Gradient descent with line search
    t1 = min(remaining() * 0.4, 400.0)
    if t1 > 10:
        a1, s1 = gradient_descent_with_line_search(best_a, t1)
        if s1 < best_score:
            best_score = s1
            best_a = a1
    
    # Phase 2: LP-based refinement
    t2 = min(remaining() * 0.2, 200.0)
    if t2 > 5:
        a2, s2 = lp_refine(best_a, t2)
        if s2 < best_score:
            best_score = s2
            best_a = a2
    
    # Phase 3: Gradient descent again (from LP result)
    t3 = min(remaining() * 0.3, 300.0)
    if t3 > 10:
        a3, s3 = gradient_descent_with_line_search(best_a, t3)
        if s3 < best_score:
            best_score = s3
            best_a = a3
    
    # Phase 4: Perturbation + re-optimize (escape local minima)
    if remaining() > 30:
        methods = ['gaussian', 'sin', 'drift', 'edge_push']
        for method in methods:
            if remaining() < 15:
                break
            a_pert = perturbed(best_a, rng, strength=0.005, method=method)
            t4 = min(remaining() * 0.2, 60.0)
            if t4 > 5:
                a4, s4 = gradient_descent_with_line_search(a_pert, t4)
                if s4 < best_score:
                    best_score = s4
                    best_a = a4
    
    return best_a, best_score
