    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def pnorm_gradient(a, p):
    """
    Compute gradient of (sum_k b[k]^p)^(1/p) where b = conv(a,a).
    grad_j = 2 * (sum b^p)^(1/p - 1) * sum_m b[j+m]^(p-1) * a[m]
    """
    n = len(a)
    b = fast_convolve(a)
    b_pos = np.maximum(b, 1e-30)
    
    if p == 1:
        # subgradient: just the sign, but since b >= 0 for a >= 0, it's 1
        # effectively gradient of sum(b) = sum of derivatives
        # grad_j = 2 * sum_m a[m] = 2 * sum(a)
        # This is constant, not useful. Use p >= 2.
        pass
    
    b_p_minus_1 = b_pos ** (p - 1)
    norm_factor = (b_pos ** p).sum() ** (1.0 / p - 1.0)
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    b_ext = np.zeros(fft_size)
    b_ext[:size] = b_p_minus_1
    a_ext = np.zeros(fft_size)
    a_ext[:n] = a
    
    Bf = fft_full(b_ext, fft_size)
    Af = fft_full(a_ext, fft_size)
    c = np.real(ifft_full(Bf * np.conj(Af), fft_size))
    
    grad = 2.0 * norm_factor * c[:n]
    return grad, b.max()


def simplex_project(v, epsilon=0.0):
    """Project v onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    v = np.maximum(v, epsilon)
    s = v.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return v / s


def evaluate_local(a):
    """Fast local evaluation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def pnorm_descent(a0, time_budget, p_schedule, momentum_decay=0.9, lr_init=0.1):
    """
    p-norm gradient descent with momentum and adaptive learning rate.
    p_schedule: list of (p, n_steps) tuples.
    """
    n = len(a0)
    a = simplex_project(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    velocity = np.zeros(n)
    
    for p, n_steps in p_schedule:
        if time.monotonic() - t0 > time_budget:
            break
        phi = time.monotonic() - t0
        
        for step in range(n_steps):
            if time.monotonic() - t0 > time_budget:
                break
            
            grad, max_b = pnorm_gradient(a, p)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            
            # Adaptive learning rate
            lr = lr_init / (1.0 + step * 0.01)
            
            # Momentum update
            velocity = momentum_decay * velocity - lr * grad
            a_new = a + velocity
            a_new = simplex_project(a_new)
            
            # Check improvement
            score = evaluate_local(a_new)
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
                a = a_new
            
            if step % 500 == 0 and step > 0:
                elapsed = time.monotonic() - t0
                print(f"    p={p} step={step} ({elapsed:.0f}s): score={best_score:.8f}")
        
        # Update a to best for next p
        a = best_a.copy()
        velocity *= 0.5  # dampen momentum when changing p
    
    return best_a, best_score


def build_conv_matrix_sparse(a):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    rows = np.zeros(n * n, dtype=np.int32)
    cols = np.zeros(n * n, dtype=np.int32)
    vals = np.zeros(n * n, dtype=np.float64)
    count = 0
    for j in range(n):
        for k in range(j, min(j + n, size)):
            rows[count] = k
            cols[count] = j
            vals[count] = a[k - j]
            count += 1
    return csr_matrix((vals[:count], (rows[:count], cols[:count])), shape=(size, n))


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver, T_mat):
    """Given fixed a, find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = [solver, cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                break
        except Exception:
            continue
    b_val = b_var.value
    if b_val is None:
        return None, None
    b_val = np.asarray(b_val, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(max(0, M_var.value if M_var.value is not None else 0))


def lp_alternating(a0, time_budget, solver, n_max_iters=500000, pnorm_interleave=False):
    """LP alternating with Nesterov step size and optional p-norm interleaving."""
    n = len(a0)
    a = simplex_project(a0.copy())
    y = a.copy()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score
    rng = np.random.default_rng(42)
    T_mat = build_conv_matrix_sparse(y)
    lp_times = []
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_budget:
            break
        
        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver, T_mat)
        lp_times.append(time.monotonic() - lp_t0)
        
        if b_new is None:
            continue
        
        if len(lp_times) > 20:
            avg_lp = np.mean(lp_times[-20:])
            if avg_lp > 10.0:
                print(f"    LP too slow ({avg_lp:.2f}s/iter), stopping at iter {it+1}")
                break
        
        # Nesterov step size
        alpha = 2.0 / (it + 2)
        
        y_old = y.copy()
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = y_old
        else:
            y = y / sy
        
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        T_mat = build_conv_matrix_sparse(y)
        
        score_a = evaluate_local(a)
        score_y = evaluate_local(y)
        
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if (it + 1) % 50 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            
            if stall_count >= 5:
                # Perturbation to escape
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 10)
                phase = rng.uniform(0, 2 * np.pi)
                noise = 0.03 * np.cos(freq * x + phase)
                y = np.maximum(y + noise, 0)
                ss = y.sum()
                if ss > 1e-15:
                    y = y / ss
                else:
                    y = best_a.copy()
                T_mat = build_conv_matrix_sparse(y)
                stall_count = 0
                
            if (it + 1) % 200 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def lse_objective_and_grad(z, n, T_temp):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T_temp
    scaled_shifted = scaled - scaled.max()
    exp_terms = np.exp(scaled_shifted)
    lse = max_b + T_temp * (scaled.max() + np.log(exp_terms.sum() + 1e-300))
    
    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        # grad_a[j] += w[k] * 2 * a[k-j] * [0<=k-j<n]
        js = 0
        je = min(n - 1, k)
        if js <= je:
            idx = np.arange(js, je + 1)
            grad_a[idx] += w[k] * 2.0 * a[k - idx]
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    obj = 2.0 * n * lse
    return obj, 2.0 * n * grad_z


def lbgfs_refine(a0, time_budget, T_stages):
    """L-BFGS with LogSumExp smoothing."""
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    deadline = t0 + time_budget
    
    for stage_T in T_stages:
        if time.monotonic() - t0 > time_budget:
            break
        remaining = time_budget - (time.monotonic() - t0)
        if remaining < 10:
            break
        
        n = len(a0)
        
        def func(z_local):
            if time.monotonic() > deadline:
                return 1e15
            obj, _ = lse_objective_and_grad(z_local, n, stage_T)
            return obj
        
        def jac(z_local):
            if time.monotonic() > deadline:
                return np.zeros(len(z_local))
            _, g = lse_objective_and_grad(z_local, n, stage_T)
            return g
        
        maxiter = max(100, int(remaining / (n * 0.01)))
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': min(maxiter, 2000), 'ftol': 1e-25,
                         'gtol': 1e-18, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    return softmax_from_z(z)


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 5 * x + 0.5)
        a = np.maximum(a, 0)
    else:
        a = 1.0 / (1.0 + 3.0 * np.abs(x - 0.5))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def perturb_sequence(a, rng, strength=0.03):
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq1 = rng.integers(1, 8)
    freq2 = rng.integers(2, 12)
    p1 = rng.uniform(0, 2 * np.pi)
    p2 = rng.uniform(0, 2 * np.pi)
    pert = strength * (np.cos(freq1 * x + p1) + 0.5 * np.sin(freq2 * x + p2))
    a_pert = np.maximum(a + pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. FFT-based p-norm gradient descent (fast, O(n log n) per step)
    2. LP alternating (Frank-Wolfe) with Nesterov
    3. p-norm gradient descent refinement
    4. LogSumExp + L-BFGS
    5. Multi-length exploration + perturbation escape
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}, budget: {budget_s:.0f}s")
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Init: n={len(best_seq)}, score={best_score:.8f}")
    
    n_best = len(best_seq)
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
    
    # === Phase 1: p-norm gradient descent (FAST) ===
    print(f"\n=== Phase 1: p-norm GD n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.12, 180.0)
    if t1 > 20:
        p_schedule = [
            (2, 2000),
            (4, 2000),
            (8, 2000),
            (16, 2000),
            (32, 2000),
            (64, 2000),
            (128, 2000),
            (256, 2000),
            (512, 2000),
        ]
        a_pn, _ = pnorm_descent(best_seq, t1, p_schedule, momentum_decay=0.9, lr_init=0.05)
        try_update(a_pn, "pnorm_1")
    
    # === Phase 2: LP alternating (main workhorse) ===
    print(f"\n=== Phase 2: LP n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.30, 500.0)
    if t2 > 60:
        print(f"  Budget: {t2:.0f}s")
