        return sTs / sTy

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT. O(n log n)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_corr(w, a):
    """Compute cross-correlation sum_k w[k]*a[k-i] for i=0..n-1 via FFT."""
    n = len(a)
    # g_i = sum_k w[k] * a[k-i] = conv(w, a_reversed)[i + n - 1]
    a_rev = a[::-1]
    # conv(w, a_rev) has length (2n-1) + n - 1 = 3n-2

    conv = irfft(W * A, fft_size)
    return conv[n-1 : 2*n-1]  # g_0, ..., g_{n-1}


def project_simplex(v):
    """Efficient Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v

    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:

    rho = rho[-1]

def evaluate_local(a):
    """Fast local evaluation (assumes a >= 0, sum > 0)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def softmax_weights(b, top_k_idx, temperature):
    """Compute softmax weights over top-K active constraints."""
    b_top = b[top_k_idx]
    max_b = b_top.max()
    scaled = (b_top - max_b) / temperature
    exp_s = np.exp(scaled - scaled.max())
    return exp_s / exp_s.sum()


def nesterov_subgradient(a0, time_limit, step0=0.005, K_start=30, K_end=150):
    """
    Nesterov-accelerated subgradient method for min max(conv(a,a))
    subject to a in simplex.
    Uses FFT for O(n log n) iterations.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    y = a.copy()  # Nesterov look-ahead point
    best_a = a.copy()
    best_score = evaluate_local(a)
    
    t0 = time.monotonic()
    step = step0
    t_nest = 1.0
    prev_y = a.copy()
    iter_count = 0
    stall_count = 0
    best_stall_score = best_score
    
    # Precompute reversed a (will update each iteration)
    while True:
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        
        # Compute convolution at y
        b = fast_convolve(y)
        max_b = b.max()
        
        # Adaptive K
        progress = min(1.0, iter_count / 50000.0)
        K = int(K_start + (K_end - K_start) * progress)
        K = min(K, 2 * n - 1)
        
        # Top-K active constraints
        # Use partial sort for efficiency
        if K >= 2 * n - 1:
            top_k_idx = np.arange(2 * n - 1)
        else:
            top_k_idx = np.argpartition(b, -K)[-K:]
        
        # Softmax weights
        temp = max(0.05 * max_b, 1e-10)
        w = softmax_weights(b, top_k_idx, temp)
        
        # Build weight vector
        w_vec = np.zeros(2 * n - 1)
        w_vec[top_k_idx] = w
        
        # Compute subgradient via FFT cross-correlation
        a_rev = y[::-1]
        size = (2 * n - 1) + n - 1  # = 3n - 2
        fft_size = 1 << (size - 1).bit_length()
        W = rfft(w_vec, fft_size)
        A = rfft(a_rev, fft_size)
        conv_wa = irfft(W * A, fft_size)
        g = 2.0 * conv_wa[n-1 : 2*n-1]
        
        # Ensure g has correct shape
        g = np.asarray(g, dtype=np.float64)
        
        # Nesterov step
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        momentum = t_nest / t_new
        
        # Gradient step at y
        y_new = y - step * g
        a_new = project_simplex(y_new)
        
        # Nesterov update
        y_next = a_new + momentum * (a_new - a)
        y_next = project_simplex(y_next)
        
        # Update
        a = a_new
        y = y_next
        t_nest = t_new
        iter_count += 1
        
        # Track best
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Adaptive step size
        if stall_count > 5000:
            step *= 0.9
            stall_count = 0
        elif stall_count == 0 and step < step0:
            step = min(step * 1.05, step0)
        
        if iter_count % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NSG it={iter_count} ({elapsed:.0f}s): score={best_score:.8f} step={step:.6f} K={K}")
    
    return best_a, best_score


def multiplicative_weights(a0, time_limit, eta0=0.03):
    """
    Fast multiplicative weights for the zero-sum game formulation.
    Player 1: choose distribution over positions (dual variables)
    Player 2: choose sequence a
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    # Dual variable: distribution over convolution positions
    m = 2 * n - 1
    mu = np.ones(m) / m
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0
    
    for it in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute conv(a, a)
        b = fast_convolve(a)
        max_b = b.max()
        
        # Update mu: proportional to b (the "payoff" for each position)
        # mu_k ∝ mu_k * exp(eta * b_k / max_b) -- but this might overflow
        # Use log-update
        log_mu = np.log(np.maximum(mu, 1e-300))
        log_mu += eta * b / max_b
        log_mu -= log_mu.max()
        mu_new = np.exp(log_mu)
        mu_new /= mu_new.sum()
        
        # Given mu, find best response a:
        # minimize sum_k mu_k * conv(a,a)_k subject to a in simplex
        # = minimize a^T (sum_k mu_k Q_k) a
        # where Q_k is the shift matrix for lag k
        # The gradient at a is: 2 * sum_k mu_k * (shift correlation)
        # = 2 * conv(mu, a_rev)[n-1 : 2n-1] (as before)
        
        # Actually, for the MW update, we update a multiplicatively:
        # a_i_new ∝ a_i * exp(-eta * g_i)
        # where g_i = d/da_i [sum_k mu_k * conv(a,a)_k]
        
        # Compute g via FFT
        a_rev = a[::-1]
        size = m + n - 1
        fft_size = 1 << (size - 1).bit_length()
        MU = rfft(mu, fft_size)
        A = rfft(a_rev, fft_size)
        conv_mua = irfft(MU * A, fft_size)
        g = 2.0 * conv_mua[n-1 : 2*n-1]
        
        # Multiplicative update
        log_update = -eta * g
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        # Check improvement
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            mu = mu_new
            eta = min(eta * 1.01, 0.3)
        else:
            eta *= 0.9
            if eta < 1e-8:
                break
        
        if it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MW it={it} ({elapsed:.0f}s): score={best_score:.8f} eta={eta:.6f}")
    
    return best_a, best_score


def lse_logsumexp_obj_grad(z, n, T):
    """LogSumExp objective and gradient in log-space."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T
    scaled_shifted = scaled - scaled.max()
    exp_terms = np.exp(scaled_shifted)
    lse = max_b + T * (scaled.max() + np.log(exp_terms.sum() + 1e-300))
    obj = 2.0 * n * lse
    
    # Gradient
    w = exp_terms / exp_terms.sum()
    # g_i = 2 * sum_k w_k * a_{k-i} (for valid indices)
    w_vec = np.zeros(2 * n - 1)
    w_vec[:] = w
    a_rev = a[::-1]
    size = (2 * n - 1) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w_vec, fft_size)
    A = rfft(a_rev, fft_size)
    conv_wa = irfft(W * A, fft_size)
    grad_a = 2.0 * conv_wa[n-1 : 2*n-1]
    
    # Chain rule: grad_z = a * (grad_a - sum(grad_a * a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return obj, 2.0 * n * grad_z


class LSEFunc:
    def __init__(self, n, T, deadline):
        self.n = n
        self.T = T
        self.deadline = deadline

    def __call__(self, z):
        if time.monotonic() > self.deadline:
            return 1e15
        obj, _ = lse_logsumexp_obj_grad(z, self.n, self.T)
        return obj


class LSEJac:
    def __init__(self, n, T, deadline):
        self.n = n
        self.T = T
        self.deadline = deadline

    def __call__(self, z):
        if time.monotonic() > self.deadline:
            return np.zeros(len(z))
        _, g = lse_logsumexp_obj_grad(z, self.n, self.T)
        return g


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage=500):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()
    deadline = t0 + time_limit
    
    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        func = LSEFunc(n, stage_T, deadline)
        jac = LSEJac(n, stage_T, deadline)
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-14, 'maxfun': maxiter_per_stage * 200,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def interpolate_sequence(f, new_n):
    """Linear interpolation to change sequence length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 5 * x + 0.5)
        a = np.maximum(a, 0)
    elif choice == 7:
        a = 1.0 / (1.0 + 3.0 * np.abs(x - 0.5))
    elif choice == 8:
        # Skewed exponential
        a = np.exp(-3 * x) + 0.3 * np.exp(-3 * (1 - x))
    else:
        # Two-peak with gap
        a = np.exp(-8 * (x - 0.25) ** 2) + np.exp(-8 * (x - 0.75) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
