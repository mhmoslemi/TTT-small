def interpolate_seq(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Power-Mean Smoothed Log-Space L-BFGS for minimizing 2n*max(conv(a,a))/(sum(a))^2.

Key approach:
- Parameterize a = softmax(z) to handle the simplex constraint {a>=0, sum(a)=1}
- Use power-mean smoothing f_p(a) = 2n*(sum_k conv(a,a)_k^p / (2n-1))^(1/p)
  which converges to the maxnorm as p -> infinity but is smooth for finite p
- Compute gradients via FFT cross-correlation (O(n log n) per evaluation)
- Anneal p from small (smooth, L2-like) to large (sharp, max-like)
- Multi-resolution exploration across different sequence lengths
- Perturbation-based multi-start to escape local optima
- Coordinate descent polish on peak-contributing coordinates
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

# Global state to avoid closures
_G_PM_P = 2
_G_PM_N = 1200
_G_PM_T0 = 0.0
_G_PM_TLIMIT = 100.0
_G_PM_BEST_SCORE = float('inf')
_G_PM_BEST_A = None
_G_BEST_SEQ = None
_G_BEST_SCORE = float('inf')


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Scale-invariant objective: 2n * max(conv(a,a)) / (sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a = a / s
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def softmax(z):
    """Numerically stable softmax."""
    z = np.asarray(z, dtype=np.float64)
    z_shift = z - z.max()
    exp_z = np.exp(z_shift)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def power_mean_obj(z, p, n):
    """
    Power-mean smoothed objective in log-space.
    f_p(a) = 2n * (sum_k conv(a,a)_k^p / (2n-1))^(1/p)
    where a = softmax(z).
    """
    a = softmax(z)
    conv = fast_convolve(a)
    log_conv = np.log(np.maximum(conv, 1e-300))
    p_log_conv = p * log_conv
    max_val = p_log_conv.max()
    log_S = max_val + np.log(np.exp(p_log_conv - max_val).sum()) - np.log(2 * n - 1)
    obj = 2.0 * n * np.exp(log_S / p)
    return obj


def power_mean_grad(z, p, n):
    """
    Gradient of power-mean smoothed objective in z-space (log-space).
    
    f_p = 2n * S^(1/p) where S = sum(conv^p)/(2n-1)
    df/da_j = 2n * S^(1/p-1) / (2n-1) * sum_k conv[k]^(p-1) * a[k-j]
    df/dz_m = a_m * (df/da_m - sum_j df/da_j * a_j)
    """
    a = softmax(z)
    conv = fast_convolve(a)
    log_conv = np.log(np.maximum(conv, 1e-300))
    
    # log_S for scaling
    p_log_conv = p * log_conv
    max_p = p_log_conv.max()
    log_S = max_p + np.log(np.exp(p_log_conv - max_p).sum()) - np.log(2 * n - 1)
    
    # weights[k] = conv[k]^(p-1), computed in log-space
    log_w = (p - 1) * log_conv
    max_lw = log_w.max()
    weights_scaled = np.exp(log_w - max_lw)
    
    # g_a[j] = sum_i weights[j+i] * a[i]
    # Computed as: conv(a, weights_flipped)[n-1 : 2n-1][::-1]
    weights_flipped = weights_scaled[::-1]
    fft_size = 1 << (3 * n - 2).bit_length()
    A_fft = rfft(a, fft_size)
    W_fft = rfft(weights_flipped, fft_size)
    conv_aw = irfft(A_fft * W_fft, fft_size)
    g_a = conv_aw[n - 1: 2 * n - 1][::-1].copy()
    
    # Scale: 2n * S^(1/p-1) / (2n-1) * exp(max_lw)
    # S^(1/p-1) = exp(log_S * (1/p - 1))
    scale = 2.0 * n * np.exp(log_S * (1.0 / p - 1.0)) / (2 * n - 1) * np.exp(max_lw)
    g_a *= scale
    
    # Transform to z-space (softmax parameterization)
    ga_dot_a = np.dot(g_a, a)
    g_z = a * (g_a - ga_dot_a)
    return g_z


def _pm_func(z):
    """Top-level function for L-BFGS objective."""
    if time.monotonic() - _G_PM_T0 > _G_PM_TLIMIT - 1:
        return 1e15
    obj = power_mean_obj(z, _G_PM_P, _G_PM_N)
    # Track best via actual score
    a_trial = softmax(z)
    sc = compute_score(a_trial)
    global _G_PM_BEST_SCORE, _G_PM_BEST_A
    if sc < _G_PM_BEST_SCORE:
        _G_PM_BEST_SCORE = sc
        _G_PM_BEST_A = a_trial.copy()
    return obj


def _pm_jac(z):

def interpolate_seq(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create a structured initial sequence normalized to sum=1."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = np.exp(-3 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = np.exp(-2.0 * (x - 0.5) ** 2 / 0.15)
    elif choice == 5:
        a = np.exp(-8 * (x - 0.3) ** 2) + np.exp(-8 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.6 + 0.4 * np.cos(2 * np.pi * 5 * x) * np.cos(np.pi * x / 2)
        a = np.maximum(a, 0)
    elif choice == 7:
        a = np.sin(np.pi * x) * (0.5 + 0.5 * np.cos(4 * np.pi * x))
        a = np.maximum(a, 0)
    elif choice == 8:
        # Flat with edge suppression
        a = np.ones(n)
        edge = int(n * 0.1)
        a[:edge] *= np.linspace(0.3, 1, edge)
        a[-edge:] *= np.linspace(1, 0.3, edge)
    else:
        # Wide Gaussian
        a = np.exp(-1.5 * (x - 0.5) ** 2 / 0.3)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def coord_descent_polish(a0, time_limit, max_steps=200000):
    """
    Coordinate descent focused on peak-reducing moves.
    Tries reducing top contributors to max autocorrelation and redistributing.
    """
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    stall = 0
    
    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))
        
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        improved = False
        top_m = min(8, len(idx))
        top_positions = idx[np.argsort(contribs)[::-1][:top_m]]
        
        for pos in top_positions:
            if a[pos] < 1e-12:
                continue
            for factor in [0.9995, 0.999, 0.998, 0.995, 0.99, 0.98, 0.97, 0.95]:
                a_trial = a.copy()
                a_trial[pos] *= factor
                diff = a[pos] - a_trial[pos]
                if diff < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    break
                a_trial[mask] += diff * a_trial[mask] / others_sum
                a_trial = a_trial / a_trial.sum()
                sc = compute_score(a_trial)
                if sc < best_score - 1e-15:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall % 200 == 0 and stall > 500:
                rng = np.random.default_rng(step * 31 + 7)
                a_trial = a + rng.standard_normal(n) * (0.002 * np.sqrt(stall / 100))
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = compute_score(a_trial)
                if sc < best_score:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    stall = 0
            if stall > 3000:
                break
        else:
            stall = 0
        
        if (step + 1) % 20000 == 0:
            print(f"    CoordDesc ({time.monotonic()-t0:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score


def try_update_global(a, label):
    """Try to update the global best sequence."""
    global _G_BEST_SEQ, _G_BEST_SCORE
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return
    a = a / s
    sc = compute_score(a)
    if np.isfinite(sc) and sc < _G_BEST_SCORE:
        _G_BEST_SCORE = sc
        _G_BEST_SEQ = a.copy()
        print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}) ***")
