"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Accelerated LP-alternating with Nesterov momentum and pre-computed matrices
2. LogSumExp Adam in log-space with adaptive temperature and top-k focusing
3. Peak-flattening heuristic for targeted convolution peak reduction
4. Multi-length exploration (n=800, 1200, 1400, 1600)
5. Symmetric sequence exploitation for reduced search space
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate the objective locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for k in range(size):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            T[k, i] = a[k - i]
    return T


def get_lp_solver():
    """Find the fastest working LP solver."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.PDLP, cp.CLARABEL, cp.SCS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
                break
        except Exception:
            continue
    else:
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    b_val[np.isnan(b_val)] = 0
    s_val = b_val.sum()
    if s_val < 1e-15:
        return None, None
    b_val = b_val / s_val
    return b_val, float(max(0.0, M_var.value)) if M_var.value else None


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def adam_logsumexp(a0, time_limit, T_stages, lr_init, lr_decay, top_k_frac=0.1):
    """Adam optimization in log-space with temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit

    best_a = a0.copy()
    best_score = evaluate_local(a0)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step_count = 0

    total_stages = len(T_stages)
    steps_per_stage = max(3000, int(time_limit * 30000 / total_stages))

    for stage_idx, T_stage in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 1:
            break

        top_k = max(5, int(top_k_frac * (2 * n - 1)))

        for step in range(steps_per_stage):
            if time.monotonic() > deadline:
                break
            if (step + 1) % 10000 == 0 and time.monotonic() - t0 > 10:
                if time.monotonic() > deadline:
                    break

            a = softmax_from_z(z)
            b = fast_convolve(a)
            max_b = b.max()
            score = 2.0 * n * max_b
            step_count += 1

            if score < best_score:
                best_score = score
                best_a = a.copy()

            # Compute weight vector for gradient
            if T_stage < 0.01:
                threshold = max_b * (1.0 - 1e-4)
                active = np.where(b >= threshold)[0]
                if len(active) == 0:
                    active = np.array([np.argmax(b)])
                w = np.zeros(len(b))
                w[active] = 1.0 / len(active)
            else:
                if top_k < len(b):
                    idx = np.argpartition(b, -top_k)[-top_k:]
                    b_top = b[idx]
                    max_top = b_top.max()
                    exp_top = np.exp((b_top - max_top) / T_stage)
                    w_top = exp_top / exp_top.sum()
                    w = np.zeros(len(b))
                    w[idx] = w_top
                else:
                    max_b_all = b.max()
                    exp_all = np.exp((b - max_b_all) / T_stage)
                    w = exp_all / exp_all.sum()

            # Gradient in a-space: g_a[i] = 2 * sum_k w[k] * a[k-i] (valid range)
            g_a_raw = np.correlate(w, a, mode='full')
            g_a = g_a_raw[n - 1:2 * n - 1] * 2.0

            # Gradient in z-space (softmax Jacobian)
            S = np.dot(g_a, a)
            grad_z = a * (g_a - S)

            # Adam update with decaying LR within stage
            step_in_stage = step
            lr = lr_init * (lr_decay ** stage_idx) / (1.0 + 0.05 * step_in_stage / max(1, steps_per_stage))
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** (step + 1))
            v_hat = v / (1 - beta2 ** (step + 1))
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            z -= z.mean()

            if (step + 1) % 50000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    Adam stage {stage_idx} (T={T_stage:.3f}) step {step+1} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver, max_iters=500000):
    """Nesterov accelerated LP alternating method."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(42)
    lp_times = []

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver)
        lp_times.append(time.monotonic() - lp_t0)

        if b_new is None:
            continue

        if len(lp_times) > 30:
            avg_lp_time = np.mean(lp_times[-30:])
            if avg_lp_time > 10.0:
                print(f"    LP too slow ({avg_lp_time:.2f}s/iter), stopping at iter {it+1}")
                break

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = evaluate_local(a)
        score_y = evaluate_local(y)

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 30:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 12)
                noise = 0.02 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0

            if (it + 1) % 500 == 0:
                elapsed = time.monotonic() - t0
                avg = np.mean(lp_times[-20:]) * 1000 if len(lp_times) >= 20 else 0
                print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} avg_lp={avg:.1f}ms")

    return best_a, best_score


def peak_flattening(a0, time_limit, n_iters=10000):
    """Targeted peak reduction heuristic."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    lr = 0.01

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        score = 2.0 * n * max_b

        # Find top peaks
        k_peaks = min(10, len(b))
        top_idx = np.argpartition(b, -k_peaks)[-k_peaks:]
        threshold = b[top_idx[0]] * 0.95
        active = np.where(b >= threshold)[0]

        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # Compute gradient from active peaks
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        g *= 2.0

        # Multiplicative update (mirror descent)
        a_new = a * np.exp(-lr * g)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score_new = evaluate_local(a_new)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
        else:
            lr *= 0.85
            if lr < 1e-6:
                break

        if it % 5000 == 0 and it > 0:
            print(f"    PeakFlat iter={it} ({time.monotonic()-t0:.0f}s): score={best_score:.8f} lr={lr:.6f}")

    return best_a, best_score


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.5 + 0.25 * np.cos(2 * np.pi * 7 * x) + 0.15 * np.cos(2 * np.pi * 13 * x + 0.5)
        a = np.maximum(a, 0)
    elif choice == 7:
        a = np.ones(n) / n
    elif choice == 8:
        a = np.exp(-2 * (x - 0.5) ** 2) + 0.3
    else:
        a = 0.4 + 0.3 * np.sin(4 * np.pi * x + 1.0)
        a = np.maximum(a, 0.01)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def perturbed_sequence(a, rng, strength=0.02):
    """Create a perturbed version."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 15)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses accelerated LP-alternating, Adam LogSumExp, peak flattening,
    and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    t_global = time.monotonic()

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Budget: {budget_s:.0f}s")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}) ***")

    # === Phase 1: LP Alternating (main workhorse) ===
    print(f"\n=== Phase 1: Nesterov LP n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver, max_iters=500000)
        local_update(a_lp, "lp_1")

    # === Phase 2: Adam LogSumExp refinement ===
    print(f"\n=== Phase 2: Adam Refinement (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        T_stages = [5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01]
        a_adam, _ = adam_logsumexp(best_seq, t_alloc, T_stages,
                                   lr_init=0.004, lr_decay=0.82, top_k_frac=0.15)
        local_update(a_adam, "adam_1")

    # === Phase 3: Peak flattening ===
    print(f"\n=== Phase 3: Peak Flattening (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_pf, _ = peak_flattening(best_seq, t_alloc)
        local_update(a_pf, "peak_flat_1")

    # === Phase 4: Second LP round ===
    print(f"\n=== Phase 4: LP Round 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 40:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, solver, max_iters=300000)

    rem = deadline - time.monotonic()

        t_n1400 = min(rem * 0.20, 300.0)

        t_adam = t_n1400 * 0.4

                                     [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02],
                                     lr_init=0.003, lr_decay=0.85, top_k_frac=0.15)
        local_update(a_1400a, "adam_n1400")

        # Then LP

            local_update(a_1400, "lp_n1400")

    # === Phase 6: Try n=1000 ===
    print(f"\n=== Phase 6: Try n=1000 (score={best_score:.8f}) ===")

    if rem > 150:
        t_n1000 = min(rem * 0.15, 200.0)

        print(f"  Budget: {t_n1600:.0f}s")
        init_1600 = interpolate_sequence(best_seq, 1600)
        a_1600a, _ = adam_logsumexp(init_1600, t_n1600 * 0.5,
                                     [3.0, 1.5, 0.7, 0.3, 0.1, 0.05, 0.02],
                                     lr_init=0.003, lr_decay=0.85, top_k_frac=0.15)
        local_update(a_1600a, "adam_n1600")

        rem = deadline - time.monotonic()
        if rem > 40:

            strength = 0.005 + 0.005 * trial
            a_pert = perturbed_sequence(best_seq, rng, strength=strength)

                                       lr_init=0.003, lr_decay=0.85, top_k_frac=0.1)
            local_update(a_opt, f"pert_t{trial}")

    rem = deadline - time.monotonic()

    # Final verification
