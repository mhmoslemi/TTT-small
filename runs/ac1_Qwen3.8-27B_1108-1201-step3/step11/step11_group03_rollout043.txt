def interpolate_to_n(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def lse_polish(a0, time_limit, deadline):
    """L-BFGS with LogSumExp smoothing."""
    n = len(a0)
    
    def softmax_z(z):
        z2 = z - z.max()
        e = np.exp(z2)
        s = e.sum()
        if s < 1e-300:
            return np.ones(n) / n
        return e / s
    
    def compute_obj_grad(z, T_temp):
        a = softmax_z(z)
        b = fast_convolve(a)
        max_b = b.max()
        exp_t = np.exp((b - max_b) / T_temp)
        lse = max_b + T_temp * np.log(exp_t.sum())
        obj = 2.0 * n * lse
        
        w = exp_t / exp_t.sum()
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-14:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
        S_val = np.dot(grad_a, a)
        grad_z = a * (grad_a - S_val)
        return obj, 2.0 * n * grad_z
    
    def lse_func(z, T_temp):
        if time.monotonic() > deadline:
            return 1e15
        o, _ = compute_obj_grad(z, T_temp)
        return o
    
    def lse_jac(z, T_temp):
        if time.monotonic() > deadline:
            return np.zeros(n)
        _, g = compute_obj_grad(z, T_temp)
        return g
    
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()
    
    T_stages = [2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
    for T_temp in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        maxiter = max(20, min(100, int(remaining * 3)))
        try:
            res = minimize(
                fun=lse_func, x0=z, jac=lse_jac,
                args=(T_temp,),
                method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': maxiter * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    return softmax_z(z)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-initialization (parent, cosine, flat, triangular, gaussian)
2. Fast Nesterov-accelerated LP alternating (PDLP/Clarabel primary)
3. Iterative active-set QCQP with grow-and-shrink
4. Wide n-sweep (1100-1300) with interpolation transfer
5. Refined n-sweep around best found
6. L-BFGS LogSumExp final polish
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.sparse import lil_matrix, csc_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate objective locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Clip non-negative and normalize to sum 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T where T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return csc_matrix(T)


def build_q_sparse(k, n):
    """Build sparse Q_k where a^T Q_k a = (conv(a,a))[k]."""
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)


def find_lp_solver():
    """Find the fastest working LP solver."""
    for s in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS, cp.GUROBI, cp.MOSEK]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.ECOS


def solve_lp_response(a_fixed, solver):
    """Given a, find b minimizing max_k (conv(a,b))[k] s.t. sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS, cp.GUROBI, cp.MOSEK]
    seen = set()
    for s in solvers_to_try:
        if s in seen:
            continue
        seen.add(s)
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_sum = b_val.sum()
                if s_sum < 1e-15:
                    return None, None
                b_val = b_val / s_sum
                return b_val, max(0.0, float(M_var.value))
        except Exception:
            continue
    return None, None


def solve_qcqp_active(a0, active_set, solver_qcqp):
    """Solve min t s.t. a^T Q_k a <= t for k in active_set, sum(a)=1, a>=0."""
    n = len(a0)
    a_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [cp.sum(a_var) == 1]
    for k in active_set:
        Q = build_q_sparse(int(k), n)
        constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), constraints)

    solvers = [solver_qcqp, cp.CLARABEL, cp.MOSEK, cp.GUROBI, cp.ECOS, cp.SCS]
    seen = set()
    for s in solvers:
        if s in seen:
            continue
        seen.add(s)
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.asarray(a_var.value, dtype=np.float64)
                a_val = np.maximum(a_val, 0)
                s_sum = a_val.sum()
                if s_sum < 1e-15:
                    return None, None
                a_val = a_val / s_sum
                return a_val, max(0.0, float(t_var.value))
        except Exception:
            continue
    return None, None


def nesterov_lp(a0, time_limit, solver, max_iters=200000):
    """Nesterov-accelerated LP alternating."""
    n = len(a0)
    y = normalize(a0)
    best_a = y.copy()
    best_score = 2.0 * n * fast_convolve(y).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall = 0
    iter_count = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, _ = solve_lp_response(y, solver)
        if b_new is None:
            continue

        iter_count += 1
        t_param = float(iter_count)
        alpha = 2.0 / (t_param + 2.0)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = best_a.copy()
        else:
            y = y / sy

        score = 2.0 * n * fast_convolve(y).max()
        if score < best_score:
            best_score = score
            best_a = y.copy()

        if (it + 1) % 50 == 0:
            if best_score < prev_best - 1e-12:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            if stall >= 25:
                y = best_a.copy()
                stall = 0
                iter_count = 0
            elif (it + 1) % 2000 == 0:
                print(f"      NLP iter={it+1} ({time.monotonic()-t0:.0f}s) score={best_score:.10f}")

    return best_a, best_score


def grow_shrink_qcqp(a0, time_limit, solver_qcqp):
    """Iteratively grow the active set and solve QCQP for polishing."""
    a = normalize(a0)
    n = len(a)
    t0 = time.monotonic()

    b = fast_convolve(a)
    max_b = b.max()
    best_a = a.copy()
    best_score = 2.0 * n * max_b

    sorted_idx = np.argsort(b)[::-1]
    active = [int(sorted_idx[0])]

    for it in range(30):
        if time.monotonic() - t0 > time_limit:
            break

        a_sol, t_val = solve_qcqp_active(a, active, solver_qcqp)
        if a_sol is None:
            break

        b_sol = fast_convolve(a_sol)
        max_b_sol = b_sol.max()
        score_sol = 2.0 * n * max_b_sol

        if score_sol < best_score:
            best_score = score_sol
            best_a = a_sol.copy()

        current_max = max_b_sol
        violators = np.where(b_sol > current_max + 1e-10)[0]
        if len(violators) > 0:
            worst = int(np.argmax(b_sol))
            if worst not in active:
                active.append(worst)
            if len(active) > 15:
                active_rank = np.argsort(b_sol)[::-1][:15]
                active = [int(k) for k in active_rank]
        else:
            near = np.where(b_sol > current_max * (1.0 - 5e-5))[0]
            new_ones = [int(k) for k in near if k not in active]
            if new_ones:
                active.extend(new_ones[:3])
            else:
                break

        if len(active) > 15:
            active_rank = np.argsort(b_sol)[::-1][:15]
            active = [int(k) for k in active_rank]

    return best_a, best_score


def interpolate_to_n(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def softmax_from_z(z):
    """Convert log-domain to probability vector."""
    z2 = z - z.max()
    e = np.exp(z2)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def lse_objective(z, n, T_temp, deadline):
    """LogSumExp objective for L-BFGS."""
    if time.monotonic() > deadline:
        return 1e15
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_t = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_t.sum())
    return 2.0 * n * lse


def lse_gradient(z, n, T_temp, deadline):
    """LogSumExp gradient for L-BFGS."""
    if time.monotonic() > deadline:
        return np.zeros(len(z))
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_t = np.exp((b - max_b) / T_temp)
    w = exp_t / exp_t.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-14:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    S_val = np.dot(grad_a, a)
    return 2.0 * n * a * (grad_a - S_val)


def lse_polish(a0, time_limit, deadline):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()

    T_stages = [2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
    for T_temp in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        maxiter = max(15, min(80, int(remaining * 4)))
        try:
            res = minimize(
                fun=lse_objective, x0=z, jac=lse_gradient,
                args=(n, T_temp, deadline),
                method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': maxiter * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)
