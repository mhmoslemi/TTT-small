def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:

    return e / s

    return cp.PDLP


def solve_lp_best_response(a_fixed, solver, warm_start=None):
    """Solve LP: min M s.t. T(a_fixed)@b <= M, sum(b)=1, b>=0. Returns (b, M)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, max_iters=100000):
    """
    Nesterov-accelerated LP best-response with:
    - Finite-horizon aware schedule
    - Frequent running average checks
    - Restart-from-average strategy
    - Warm-starting for LP
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Running average
    avg_a = np.zeros(n)
    n_avg = 0
    warm_b = None
    
    # Track restart
    last_restart = 0
    restart_interval = 500
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP
        b_star, M_val = solve_lp_best_response(y, solver, warm_b)
        if b_star is None:
            continue
        warm_b = b_star
        
        # Nesterov step
        alpha_t = 2.0 / (t + 1.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s_new = y_new.sum()
        if s_new < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / s_new
        
        # Nesterov extrapolation
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 1.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new
        
        y = y_new
        a = a_new
        
        # Update running average
        avg_a += a
        n_avg += 1
        
        # Check scores frequently
        if n_avg <= 100 or n_avg % 5 == 0:
            score_a = 2.0 * n * fast_autocorr(a).max()
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()
        
        if n_avg % 3 == 0:
            avg_check = avg_a / n_avg
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
        
        # Restart strategy: every restart_interval, reset from running average
        if t - last_restart >= restart_interval and n_avg > restart_interval:
            avg_check = avg_a / n_avg
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < 2.0 * n * fast_autocorr(a).max():
                a = avg_check.copy()
                y = avg_check.copy()
                avg_a = np.zeros(n)
                n_avg = 0
                last_restart = t
                warm_b = None
        
        if t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}, LP_M={M_val:.8f}, avg_score={2.0*n*fast_autocorr(avg_a/max(1,n_avg)).max():.6f}")
    
    # Final check of running average
    if n_avg > 0:
        avg_final = avg_a / n_avg
        score_final = 2.0 * n * fast_autocorr(avg_final).max()
        if score_final < best_score:
            best_score = score_final
            best_a = avg_final.copy()
    
    return best_a, best_score


def hedge_optimize(a0, time_limit, solver, max_iters=10000):
    """
    Exponential weights (Hedge) method for the zero-sum game.
    Uses LP best responses as reward signals.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Log-weights
    log_w = np.zeros(n)
    
    # Running average
    avg_a = np.zeros(n)
    n_avg = 0
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Learn step size: eta ~ 2*sqrt(log(n) / T)
        eta = 2.0 * np.sqrt(np.log(n) / t)
        
        # Solve LP to get best response (this gives us the "opponent's" optimal play)
        b_star, M_val = solve_lp_best_response(a, solver)
        if b_star is None:
            continue
        
        # The reward for position i is the contribution to the worst peak
        # Use the LP dual: the constraint (T@a)_k <= M tells us which peaks are tight
        T = build_conv_matrix(a)
        peaks = T @ b_star  # This is convolve(a, b_star)
        max_peak = peaks.max()
        
        # Gradient: for each position i, how much does it contribute to the max peak(s)?
        # Simplified: use the peak values as proxy rewards
        # Actually, the "reward" for the Hedge algorithm should be the payoff of playing position i
        # In our formulation, playing position i with weight a[i] contributes a[i]*a[k-i] to peak k
        # The "cost" of position i is how much it contributes to the maximum peak
        
        # Compute per-position cost using the tight constraints
        tight_mask = peaks >= max_peak - 1e-10
        # Cost of position i = sum over tight k of a[i]*a[k-i] (normalized)
        costs = np.zeros(n)
        for k in np.where(tight_mask)[0]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            costs[idx] += a[idx] * a[k - idx]
        
        # Update log-weights (Hedge): decrease weight for high-cost positions
        log_w -= eta * costs
        log_w -= log_w.max()  # for numerical stability
        w = np.exp(log_w)
        s_w = w.sum()
        if s_w < 1e-15:
            a = np.ones(n) / n
        else:
            a = w / s_w
        
        # Track best
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        # Running average
        avg_a += a
        n_avg += 1
        if n_avg % 10 == 0:
            avg_check = avg_a / n_avg
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
        
        if t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Hedge t={t} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score


def projected_subgradient(a0, time_limit, n_stages=30):
    """
    Projected subgradient method on the simplex for minimizing max_k (a*a)_k.
    Uses smooth approximation via LogSumExp with annealing.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    temps = np.geomspace(200.0, 0.1, n_stages)
    
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        iters = max(20, min(200, int(50 * np.sqrt(remaining / (time_limit - (t0)) if time_limit > 0 else 1))))
        
        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break
            
            b = fast_autocorr(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            
            # Gradient of LSE w.r.t. a: grad[i] = 2 * sum_k w[k] * a[k-i] (with bounds)
            # Use FFT cross-correlation
            conv_len = n + (2 * n - 1) - 1
            fft_size = 1 << (conv_len - 1).bit_length()
            A_rev = rfft(a[::-1].copy(), fft_size)
            W = rfft(w, fft_size)
            conv = irfft(A_rev * W, fft_size)
            grad = 2.0 * conv[n - 1:n - 1 + n]
            
            # Step size: polyak-like
            T_iter = T_val
            obj_val = (max_Tb + np.log(exp_Tb.sum())) / T_val
            # Use fixed step size for simplicity
            step = 1.0 / (T_val * T_val * n)
            
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a = a_new / s_new
            else:
                a = np.ones(n) / n
            
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
    
    return best_a, best_score


def peak_shaving(a0, time_limit, seed=123):
    """
    Coordinate-descent peak shaving: reduce top autocorrelation peaks
    by moving mass from high-contributing to low-contributing positions.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:min(12, 2 * n - 1)]
        
        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            
            src_local = np.argmax(contrib)
            src = idx[src_local]
            
            # Find target: low-value position far from peaks
            candidates = [j for j in range(n) if j != src]
            candidates.sort(key=lambda j: a[j])
            dst_pool = candidates[:max(1, len(candidates) // 4)]
            dst = dst_pool[rng.integers(len(dst_pool))]
            
            for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            if stall > 300:
                break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng, mode=0):
    """Generate diverse structured initializations."""
    if mode == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif mode == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 1.2 + 0.3 * x
    elif mode == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.7 + 0.25 * np.cos(3 * x) + 0.05 * np.cos(7 * x)
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * (x - 0.25) ** 2) + 0.4 * np.exp(-4 * (x - 0.75) ** 2) + 0.1
    elif mode == 6:
        x = np.linspace(0, 1, n)
        a = 1.0 / (x + 0.5)
    else:
        a = rng.random(n) + 0.1
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization:
    1. LP Nesterov with restarts (main workhorse, ~55% time)
    2. Hedge/exponential weights for diversity (~15% time)
    3. Length exploration (~10% time)
    4. Projected subgradient + peak shaving polish (~20% time)
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)
    
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")
    
    # Get parent sequence
    best_seq = None
    best_score = float('inf')
    parent = None
    
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Determine starting length
    if parent is not None:
        n_base = len(parent)
    else:
        n_base = 1600
        best_seq = structured_init(n_base, rng, 0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    
    # Phase 1: LP Nesterov at base length (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global_start)
    t1 = remaining * 0.50
    print(f"\n=== Phase 1: LP Nesterov at n={n_base}, budget={t1:.0f}s ===")
    if t1 > 30:
        a_out, sc_out = nesterov_lp_optimize(best_seq, t1, solver)
        try_update(a_out, f"lp_n{n_base}")
    
    # Phase 2: Hedge method for diversity
    remaining = time_budget - (time.monotonic() - t_global_start)
    t2 = remaining * 0.18
    print(f"\n=== Phase 2: Hedge at n={n_base}, budget={t2:.0f}s ===")
    if t2 > 30:
        # Start Hedge from a slightly perturbed version of best
        init_hedge = best_seq + 0.01 * rng.standard_normal(len(best_seq))
        init_hedge = np.maximum(init_hedge, 0)
        init_hedge = init_hedge / init_hedge.sum()
        a_out, _ = hedge_optimize(init_hedge, t2, solver)
        try_update(a_out, f"hedge_n{n_base}")
    
    # Phase 3: Length exploration
    remaining = time_budget - (time.monotonic() - t_global_start)
    n_long = min(n_base + 300, 2200)
    print(f"\n=== Phase 3: LP Nesterov at n={n_long}, budget={remaining*0.15:.0f}s ===")
    t3 = remaining * 0.15
    if t3 > 30:
        init_long = interpolate_sequence(best_seq, n_long)
        a_out, _ = nesterov_lp_optimize(init_long, t3, solver)
        try_update(a_out, f"lp_n{n_long}")
    
    # Phase 4: Projected subgradient polish
    remaining = time_budget - (time.monotonic() - t_global_start)
    t4 = remaining * 0.20
    print(f"\n=== Phase 4: Projected subgradient at n={n_base}, budget={t4:.0f}s ===")
    if t4 > 10:
        a_out, _ = projected_subgradient(best_seq, t4, n_stages=25)
        try_update(a_out, f"psg_n{n_base}")
    
    # Phase 5: Peak shaving
    remaining = time_budget - (time.monotonic() - t_global_start)
    t5 = remaining * 0.30
    print(f"\n=== Phase 5: Peak shaving, budget={t5:.0f}s ===")
    if t5 > 5:
        a_out, _ = peak_shaving(best_seq, t5, seed=456)
        try_update(a_out, "peak_shave_1")
    
    # Phase 6: Final LP Nesterov refinement on best
    remaining = time_budget - (time.monotonic() - t_global_start)
    t6 = remaining * 0.50
    print(f"\n=== Phase 6: Final LP Nesterov at n={len(best_seq)}, budget={t6:.0f}s ===")
    if t6 > 20:
        a_out, _ = nesterov_lp_optimize(best_seq, t6, solver)
        try_update(a_out, f"lp_final_n{len(best_seq)}")
    
    # Phase 7: One more peak shave
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 10:
        t7 = min(remaining * 0.6, 60.0)
        if t7 > 5:
            a_out, _ = peak_shaving(best_seq, t7, seed=789)
            try_update(a_out, "peak_shave_final")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
    
    return [float(x) for x in best_seq]
