import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft

def fast_convolve(a):
    """Compute linear convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def evaluate_safe(a):
    """Safely evaluate a sequence using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_from_z(z):
    """Convert log-parameters to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def lse_objective_and_grad(z, T, n):
    """
    Compute LSE(T * conv(a,a)) and its gradient w.r.t. z,
    where a = softmax(z) and n = len(a).

    b = fast_convolve(a)

    Tb = T * b

    grad_z = a * (grad_a - S)

def lse_adam_optimize(a0, time_limit, T_start=500.0, T_end=0.1, n_stages=25, 
                      lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    """
    Optimize using LSE-smoothed objective with Adam optimizer.
    Works in z-space (log-parameters) with softmax constraint.
    """
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-30)).copy()
    
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, T in enumerate(temps):
        # Reset Adam moments at each stage (or keep them? Keeping might be better)
        m = np.zeros(n)
        v = np.zeros(n)
        step = 0
        
        while time.monotonic() - t0 < time_limit:
            step += 1
            obj, grad = lse_objective_and_grad(z, T, n)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** step)
            v_hat = v / (1 - beta2 ** step)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Evaluate
            a = softmax_from_z(z)
            sc = score_normalized(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            
            if step % 100 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LSE-Adam stage={si+1}/{n_stages} T={T:.2f} step={step} "
                      f"({elapsed:.0f}s): score={best_score:.8f}")
            
            # If convergence detected, break to next stage
            if step > 50 and np.abs(grad).max() < 1e-10:
                break
    
    return best_a, best_score


def peak_targeted_search(a0, time_limit, n_steps=100000, eta_start=0.1, eta_end=0.001):
    """
    Subgradient descent targeting the top peaks in the convolution.
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    z = np.log(np.maximum(a, 1e-30)).copy()
    
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        a = softmax_from_z(z)
        b = fast_convolve(a)
        max_b = b.max()
        
        # Find active peaks (top K in value)
        threshold = max_b * 0.9
        active = np.where(b >= threshold)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        # Compute subgradient: grad_a[l] = (2/n_active) * sum_{k in active} 2*a[k-l]
        # = (4/n_active) * sum_{k in active} a[k-l]
        grad_a = np.zeros(n)
        # For each active k, add a[k-l] for valid l
        # grad_a[l] += a[k-l] for each active k where 0<=k-l<n
        # This is: grad_a = (4/|active|) * sum_k active a[k-l]
        # = (4/|active|) * the cross-correlation of indicator(active) with a
        
        # Use FFT for efficiency
        ind = np.zeros(2 * n - 1)
        ind[active] = 1.0
        L = 1 << (3 * n).bit_length()
        I_fft = fft(ind, L)
        A_fft = fft(a, L)
        corr = ifft(I_fft * A_fft).real[:n]
        grad_a = (4.0 / len(active)) * corr
        
        # Convert to z-gradient
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        
        eta = eta_start * (eta_end / eta_start) ** (step / max(1, n_steps))
        z = z - eta * grad_z
        
        a_new = softmax_from_z(z)
        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        if (step + 1) % 5000 == 0:
            print(f"    PeakTarget step={step+1} ({time.monotonic()-t0:.0f}s): "
                  f"score={best_score:.8f} max_b={max_b:.6e} |active|={len(active)}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def cosine_window_init(n):
    """Cosine window initialization (good for this problem)."""
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * (x - 0.5)) ** 2
    a = np.maximum(a, 0)

def gaussian_init(n, sigma_frac=0.4):
    """Gaussian window initialization."""
    x = np.linspace(0, 1, n)
    center = 0.5
    sigma = sigma_frac / np.sqrt(12)
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def random_smooth_init(n, rng):
    """Random smooth initialization via random Fourier series."""
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for k in range(1, 8):
        freq = rng.integers(1, 5) * k
        amp = rng.uniform(-0.1, 0.1)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.01)
    s = a.sum()
    return a / s


def fourier_flatten(a0, time_limit, n_iters=5000, alpha=0.7):
    """
    Fourier-domain flattening: iteratively flatten the power spectrum
    while preserving non-negativity.
    """
    n = len(a0)
    a = a0.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute power spectrum
        A = fft(a, 2 * n)
        P = np.abs(A) ** 2
        
        # Flatten: P_new = P^alpha (pushes toward flatness)
        P_min = P[P > 0].min()
        P_flat = P ** alpha
        P_flat = P_flat / P_flat.sum() * P.sum()
        
        # Spectral factorization (simple: take sqrt of magnitude, keep phase of A)
        A_new = np.sqrt(P_flat) * np.exp(1j * np.angle(A))
        
        # Inverse FFT
        a_new = ifft(A_new).real[:n]
        
        # Project to non-negative and normalize
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Keep old a (don't accept worsening)
            pass
        
        if (it + 1) % 500 == 0:
            print(f"    FourierFlatten iter={it+1} ({time.monotonic()-t0:.0f}s): "
                  f"score={best_score:.8f}")
    
    return best_a, best_score

import numpy as np
import time
from scipy.fft import fft, ifft, rfft, irfft


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_direct(a):
    n = len(a)
    b = fast_convolve(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * n * b.max() / (s * s)


def score_normalized(a):
    """Assumes sum(a) = 1."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def eval_safe(a):

def softmax_from_z(z):
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def lse_obj_grad(z, T):
    n = len(z)
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    lse_val = max_Tb + np.log(exp_Tb.sum())
    obj = 2.0 * n * lse_val
    
    # Gradient via FFT cross-correlation
    L = 1 << (3 * n - 2).bit_length()
    W_fft = fft(w, L)
    a_fft = fft(a, L)
    C = ifft(W_fft * np.conj(a_fft)).real[:n]
    grad_a = 4.0 * n * C
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z
