"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated LSE (log-sum-exp) smoothed objective with aggressive geometric temperature annealing via L-BFGS
2. Peak-targeting gradient descent: directly attacks the worst autocorrelation peaks with steepest descent
3. Spectral-domain refinement to flatten power spectrum
4. Multi-start with diverse initializations and cross-length interpolation
5. Simulated annealing-style restarts to escape local minima
6. Log-parameterization (softmax) to maintain simplex constraint automatically
Key improvement over previous: more aggressive cooling schedule, peak-targeting phase,
and better exploitation of the full time budget with multiple independent attempts.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def normalize(a):
    """Normalize to probability vector."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def evaluate_local(a):
    """Evaluate score locally (fast, no security checks)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def lse_obj_grad_prob(a, T):
    """
    LSE-smoothed objective and gradient for probability vector a (sum=1, a>=0).
    Objective: 2n * LSE(conv(a,a), T)
    """
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    shifted = (c - c_max) / T
    # Clamp for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_c = np.exp(shifted)
    exp_sum = exp_c.sum()
    lse = c_max + T * np.log(exp_sum)
    obj = 2.0 * n * lse

    # Softmax weights
    w = exp_c / exp_sum

    # Compute b_j = sum_i a[i] * w[i+j] for j=0..n-1
    # Using: b[j] = (flip(a) * w)[j + n - 1]
    a_flip = a[::-1]
    conv_len = n + (2 * n - 1) - 1
    N = 1 << (conv_len - 1).bit_length()
    Ar = rfft(a_flip, N)
    W = rfft(w, N)
    conv = irfft(Ar * W, N)
    b = conv[n - 1: 2 * n - 1]

    # Gradient: g_j = (2n/T) * (2*b_j - a_j * w[2j] * (2j < 2n-1))
    diag_idx = 2 * np.arange(n)
    diag_valid = diag_idx < (2 * n - 1)
    correction = np.zeros(n)
    correction[diag_valid] = a[diag_valid] * w[diag_idx[diag_valid]]
    g = (2.0 * n / T) * (2.0 * b - correction)

    return obj, g


def softmax_from_z(z):
    """Convert log-parameters to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-parameters."""
    return np.log(np.maximum(a, 1e-300))


def lse_obj_grad_z(z, T):
    """LSE objective and gradient in log-parameterization."""
    a = softmax_from_z(z)
    obj, grad_a = lse_obj_grad_prob(a, T)
    # Transform: grad_z = a * (grad_a - dot(grad_a, a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_lbgfs_optimize(a0, time_limit, T_stages, iters_per_stage=2000):
    """L-BFGS optimization with LSE smoothing and temperature annealing in log-space."""
    n = len(a0)
    z = z_from_a(a0.copy())
    t0 = time.monotonic()
    best_z = z.copy()
    best_obj = float('inf')

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func(z_in):
            o, _ = lse_obj_grad_z(z_in, stage_T)
            return o

        def jac(z_in):
            _, g = lse_obj_grad_z(z_in, stage_T)
            return g

        iters = min(iters_per_stage, max(20, int(remaining * 10)))
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                bounds=[(None, None)] * n,
                options={'maxiter': iters, 'ftol': 1e-16, 'gtol': 1e-14,
                         'maxfun': iters * 50, 'disp': False}
            )
            z = res.x
            obj_now, _ = lse_obj_grad_z(z, stage_T)
            if obj_now < best_obj:
                best_obj = obj_now
                best_z = z.copy()
        except Exception:
            pass

    a_out = softmax_from_z(best_z)
    return a_out


def peak_targeting_refine(a0, time_limit, n_steps=5000):
    """
    Directly target the worst autocorrelation peaks using steepest descent.
    At each step, identify the active peaks (near-max) and compute the
    subgradient direction to reduce them.
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    lr = 0.02

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        c_max = c.max()

        # Active set: peaks within threshold of max
        threshold = c_max * (1.0 - 1e-4)
        active = np.where(c >= threshold)[0]

        # Compute subgradient: average over active peaks
        # g_j = (1/|S|) * sum_{k in S} dc_k/da_j
        # dc_k/da_j = 2*a_{k-j} (if 2j != k) or a_j (if 2j == k), for valid k-j
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for m in range(js, je + 1):
                if 2 * m == k:
                    g[m] += a[m]
                else:
                    g[m] += 2.0 * a[k - m]
        g /= len(active)

        # Project gradient to simplex: subtract mean
        g -= g.mean()

        # Step with backtracking line search
        alpha = lr
        for _ in range(20):
            a_new = a - alpha * g
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                alpha *= 0.5
                continue
            a_new = a_new / s
            sc = evaluate_local(a_new)
            if sc < best_score * (1 - 1e-10) or sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
                break
            alpha *= 0.5
            if alpha < 1e-10:
                break

        # Decay learning rate
        if step % 200 == 199:
            lr *= 0.8

    return best_a, best_score


def spectral_flatten(a0, time_limit, n_iters=3000):
    """
    Flatten the power spectrum to reduce autocorrelation peaks.
    Works in frequency domain: minimize variance of |A(k)|^2.
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    t0 = time.monotonic()
    lr = 0.003

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        A = rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()

        # Gradient direction: reduce P variance
        # Want to move P toward P_mean
        # h = IFFT( -(P - P_mean) * conj(A) / (P + eps) )
        H = np.zeros(fft_size // 2 + 1, dtype=np.complex128)
        valid = P > 1e-15
        H[valid] = -(P[valid] - P_mean[valid]) * np.conj(A[valid]) / (P[valid] + 1e-10)
        H[0] = 0.0  # Preserve DC
        h = irfft(H, fft_size)[:n]
        h -= h.mean()
        h_norm = np.linalg.norm(h)
        if h_norm < 1e-15:
            break
        h /= h_norm

        # Line search
        for scale in [lr, lr * 0.4, lr * 0.15]:
            a_new = a + scale * h
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new /= s
            sc = evaluate_local(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
            break

        if it % 500 == 499:
            lr *= 0.8

    return best_a, best_score


def diverse_init(n, rng):

    # Sine squared

    # Uniform
    candidates.append(np.ones(n) / n)
    # Two bumps

    # Edge-peaked
    candidates.append(normalize(np.exp(-4 * x) + np.exp(-4 * (1 - x)) + 0.3))
    # Power law
    candidates.append(normalize((x * (1 - x)) ** 0.3 + 0.1))
    # Modulated cosine
    a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.15 * np.cos(2 * np.pi * 5 * x)
    candidates.append(normalize(np.maximum(a, 0)))
    # Random smooth
    coeffs = rng.normal(0, 1, 8)
    a = np.zeros(n)
    for i, c in enumerate(coeffs):
        a += c * np.cos(2 * np.pi * (i + 1) * x + rng.uniform(0, 2 * np.pi))
    candidates.append(normalize(np.maximum(a + 0.5, 0)))
    # Bump with taper
    candidates.append(normalize(np.exp(-4 * (x - 0.5) ** 2) * (0.5 + 0.5 * np.cos(2 * np.pi * x))))

    return candidates


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    return normalize(np.maximum(f_interp, 0))


def perturb_sequence(a, rng, strength=0.01):
    """Small random perturbation."""
    n = len(a)
    method = rng.integers(0, 4)
    if method == 0:
        # Gaussian noise
        a_new = a + rng.normal(0, strength, n)
    elif method == 1:
        # Low-frequency perturbation
        freq = rng.integers(1, 6)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.linspace(0, 2 * np.pi, n)
        a_new = a + strength * np.cos(freq * x + phase)
    elif method == 2:
        # Block swap
        a_new = a.copy()
        bs = rng.integers(max(5, n // 50), max(10, n // 10))
        i = rng.integers(0, n - bs)
        j = rng.integers(0, n - bs)
        if abs(i - j) > bs:
            a_new[i:i+bs], a_new[j:j+bs] = a_new[j:j+bs].copy(), a_new[i:i+bs].copy()
    else:
        # Random mask perturbation
        mask = rng.random(n) < 0.1
        a_new = a + strength * rng.normal(0, 1, n) * mask
    return normalize(a_new)


def full_optimization_round(a0, time_limit, rng, label=""):
    """Run a full optimization round: LSE+L-BFGS, peak targeting, spectral."""
    best_a = a0.copy()
    best_score = evaluate_local(a0)
    t0 = time.monotonic()

    def update(a, tag):
        nonlocal best_a, best_score
        if a is None:
            return
        a = normalize(a)
        sc = evaluate_local(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"    [{label}:{tag}] {sc:.8f}")

    # Phase A: LSE + L-BFGS
    rem = time_limit - (time.monotonic() - t0)
    if rem > 10:
        T_stages = [10, 5, 3, 2, 1.5, 1, 0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.05, 0.03, 0.02, 0.015, 0.01, 0.007, 0.005]
        t_alloc = rem * 0.5
        a_lb = lse_lbgfs_optimize(best_a, t_alloc, T_stages, iters_per_stage=1500)
        update(a_lb, "lbgfs")

    # Phase B: Peak targeting
    rem = time_limit - (time.monotonic() - t0)
    if rem > 10:
        a_pt, _ = peak_targeting_refine(best_a, min(rem * 0.25, 60), n_steps=10000)
        update(a_pt, "peak")

    # Phase C: Spectral flatten
    rem = time_limit - (time.monotonic() - t0)
    if rem > 10:
        a_sp, _ = spectral_flatten(best_a, min(rem * 0.2, 40), n_iters=5000)
        update(a_sp, "spectral")

    # Phase D: Fine LSE polish
    rem = time_limit - (time.monotonic() - t0)
    if rem > 5:
        T_fine = [1, 0.5, 0.3, 0.2, 0.1, 0.07, 0.05, 0.03, 0.02, 0.015, 0.01, 0.007, 0.005]
        a_fine = lse_lbgfs_optimize(best_a, rem * 0.7, T_fine, iters_per_stage=1000)
        update(a_fine, "fine")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses LSE+L-BFGS with aggressive cooling, peak-targeting, spectral shaping,
    and multi-start diversity.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Try to load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        candidates = diverse_init(n_init, rng)
        scores = [evaluate_local(c) for c in candidates]
        best_idx = np.argmin(scores)
        best_seq = candidates[best_idx].copy()
        best_score = scores[best_idx]
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    n_best = len(best_seq)
    print(f"Starting: n={n_best}, score={best_score:.8f}")

    # === Phase 1: Main LSE optimization on parent ===
    print(f"\n=== Phase 1: Main LSE (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.35, 500.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        a1, s1 = full_optimization_round(best_seq, t1, rng, "P1")
        try_update(a1, "phase1")

    # === Phase 2: Multi-start from perturbed versions ===
    print(f"\n=== Phase 2: Multi-start (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    n_starts = max(2, int(rem / 80))
    n_starts = min(n_starts, 6)
    for i in range(n_starts):
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        # Create perturbed start
        strength = 0.01 + 0.005 * i
        a_start = perturb_sequence(best_seq, rng, strength)
        # Also try interpolation to different lengths occasionally
        if i == n_starts // 2 and n_starts > 3:
            n_alt = n_best + 100
            a_start = interpolate_sequence(best_seq, n_alt)
            a_start = perturb_sequence(a_start, rng, 0.005)

        t_sub = min(rem * 0.4, 80.0)
        print(f"  Start {i+1}/{n_starts}: score={evaluate_local(a_start):.8f}, n={len(a_start)}, budget={t_sub:.0f}s")
        a_try, s_try = full_optimization_round(a_start, t_sub, rng, f"P2_{i}")
        try_update(a_try, f"multi_{i}")

    # === Phase 3: Aggressive peak targeting on best ===
    print(f"\n=== Phase 3: Peak Targeting (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t3 = min(rem * 0.2, 100.0)
        a_pt, s_pt = peak_targeting_refine(best_seq, t3, n_steps=20000)
        try_update(a_pt, "peak_target")

    # === Phase 4: Spectral refinement ===
    print(f"\n=== Phase 4: Spectral (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t4 = min(rem * 0.15, 60.0)
        a_sp, s_sp = spectral_flatten(best_seq, t4, n_iters=8000)
        try_update(a_sp, "spectral")

    # === Phase 5: Very fine LSE polish ===
    print(f"\n=== Phase 5: Fine Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        T_very_fine = [0.5, 0.3, 0.2, 0.15, 0.1, 0.08, 0.06, 0.04, 0.03, 0.02, 0.015, 0.01, 0.007, 0.005, 0.003, 0.002]
        a_fin = lse_lbgfs_optimize(best_seq, min(rem * 0.5, 80.0), T_very_fine, iters_per_stage=2000)
        try_update(a_fin, "fine_polish")

    # === Phase 6: One more peak targeting pass ===
    rem = deadline - time.monotonic()
    if rem > 10:
        a_pt2, _ = peak_targeting_refine(best_seq, min(rem * 0.4, 30.0), n_steps=10000)
        try_update(a_pt2, "peak_final")

    # === Phase 7: Length exploration ===
    print(f"\n=== Phase 7: Length Exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        for n_try in [n_best + 200, n_best - 200, n_best + 400]:
            if n_try < 200 or n_try > 4000:
                continue
            rem_sub = deadline - time.monotonic()
            if rem_sub < 30:
                break
            init = interpolate_sequence(best_seq, n_try)
            t_sub = min(rem_sub * 0.3, 50.0)
            T_short = [5, 2, 1, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.01]
            a_len = lse_lbgfs_optimize(init, t_sub, T_short, iters_per_stage=1500)
            try_update(a_len, f"len_{n_try}")

    # === Final: one last LSE pass ===
    rem = deadline - time.monotonic()
    if rem > 10:
        T_last = [0.3, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001]
        a_last = lse_lbgfs_optimize(best_seq, min(rem * 0.6, 20.0), T_last, iters_per_stage=1000)
        try_update(a_last, "final")

    # Final verification with the actual evaluate_sequence
    if best_seq is not None:
        final_check = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
        print(f"\nFINAL: local_score={evaluate_local(best_seq):.8f}, actual_score={final_check:.8f}, n={len(best_seq)}")
        if np.isfinite(final_check) and final_check < best_score:
            best_score = final_check

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
