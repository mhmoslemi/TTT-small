def fast_best_response(y, n_active=10, eta=0.1, n_inner=5):
    """Fast approximate LP best response using multiplicative weights."""
    n = len(y)
    b = np.ones(n) / n
    for _ in range(n_inner):
        # Compute convolution
        v = np.convolve(y, b)  # length 2n-1
        M = v.max()
        # Find top n_active peaks
        top_k = np.argsort(v)[::-1][:n_active]
        # Compute "pressure" on each b_j
        pressure = np.zeros(n)
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            pressure[js:je+1] += 1
        # Multiplicative update
        b = b * np.exp(-eta * pressure / pressure.max())
        b = np.maximum(b, 0)
        b = b / b.sum()
    return b
```

    A_rev = rfft(a_rev, fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    grad = 2.0 * conv[n - 1:n - 1 + n]
    return lse_val, grad, w


def mulw_best_response(y, n_inner=8, n_top=15):
    """
    Fast approximate best response using multiplicative weights.
    Solves: min_b max_k (y*b)_k s.t. sum b = 1, b >= 0
    """
    n = len(y)
    b = np.ones(n) / n
    for _ in range(n_inner):
        # Compute convolution y * b
        fft_size = 1 << (2 * n - 2).bit_length()
        Y = rfft(y, fft_size)
        B = rfft(b, fft_size)
        v = irfft(Y * B, fft_size)[:2*n-1]
        M = v.max()
        if M < 1e-15:
            break
        # Find top peaks
        k = n_top
        idx = np.argpartition(v, -k)[-k:]
        # Compute pressure on each position
        pressure = np.zeros(n)
        for ki in idx:
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            pressure[js:je+1] += 1.0
        # Normalized pressure
        pmax = pressure.max()
        if pmax < 1e-15:
            break
        # Multiplicative update with adaptive eta
        eta = 0.3
        b = b * np.exp(-eta * pressure / pmax)
        b = np.maximum(b, 1e-30)
        b = b / b.sum()
    return b


def nesterov_game(a0, time_limit, n_inner=8, n_top=15):
    """
    Nesterov-accelerated game iteration with fast approximate best-response.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    y = a.copy()  # Extrapolation point
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Running average
    avg_a = np.zeros(n)
    n_avg = 0
    
    t = 0
    while time.monotonic() - t0 < time_limit:
        t += 1
        # Fast best response
        b_star = mulw_best_response(y, n_inner=n_inner, n_top=n_top)
        
        # Nesterov step
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / s
        
        if t <= 2:
            a_new = y_new.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s
        
        y = y_new
        a = a_new
        
        # Check score
        if t % 10 == 0 or t <= 5:
            score_a = 2.0 * n * fast_autocorr(a).max()
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()
        
        # Running average (every 50 iterations)
        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
        
        if t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NG t={t} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score


def lse_adam_optimize(a0, time_limit, T_start=50.0, T_end=0.1, n_stages=15):
    """
    LogSumExp optimization with Adam optimizer and temperature annealing.
    Works directly on the simplex (a >= 0, sum a = 1).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        # Adam state
        m = np.zeros(n)
        v = np.zeros(n)
        beta1, beta2, eps = 0.8, 0.999, 1e-8
        lr = 0.01 / max(1, np.sqrt(si + 1))
        adam_t = 0
        
        iters_this_stage = 0
        while time.monotonic() - t0 < time_limit:
            adam_t += 1
            iters_this_stage += 1
            
            val, grad, w = compute_lse_grad(a, stage_T)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** adam_t)
            v_hat = v / (1 - beta2 ** adam_t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            
            a_new = a - step
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / s
            
            a = a_new
            
            # Check score periodically
            if iters_this_stage % 20 == 0:
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
            
            # Learning rate decay within stage
            if iters_this_stage % 100 == 0:
                lr *= 0.9
            
            if iters_this_stage > 500 and time.monotonic() - t0 > time_limit * 0.95:
                break
    
    return best_a, best_score


def lse_lbfgs_optimize(a0, time_limit, T_start=100.0, T_end=0.05, n_stages=12):
    """
    LogSumExp optimization with L-BFGS-B on z-parameterization.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a_init = a_init / a_init.sum()
    z = np.log(np.maximum(a_init, 1e-30))
    
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        # Scale iterations based on remaining time
        frac_remaining = remaining / time_limit
        iters = max(10, int(40 * frac_remaining + 20))
        
        def lse_obj(z_in):
            a_tmp = softmax_fn(z_in)
            val, _, _ = compute_lse_grad(a_tmp, stage_T)
            return val
        
        def lse_grad_z(z_in):
            a_tmp = softmax_fn(z_in)
            _, grad_a, _ = compute_lse_grad(a_tmp, stage_T)
            S = np.dot(grad_a, a_tmp)
            return a_tmp * (grad_a - S)
        
        try:
            res = minimize(
                fun=lse_obj, x0=z, jac=lse_grad_z, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': iters * 100, 'disp': False}
            )
            z = res.x
            a = softmax_fn(z)
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
        except Exception as e:
            pass
    
    return best_a, best_score


def peak_shaving(a0, time_limit, seed=123):
    """
    Targeted peak shaving: reduce top autocorrelation peaks by moving mass.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:10]
        
        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            
            # Top contributors
            top_contrib = np.argsort(contrib)[::-1][:5]
            
            for tc in top_contrib:
                src = idx[tc]
                src_val = a[src]
                if src_val < 1e-10:
                    continue
                
                # Find destination: position with low autocorrelation contribution
                # to the top peaks
                peak_areas = set()
                for k2 in top_k:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    peak_areas.update(range(js2, je2 + 1))
                
                candidates = [j for j in range(n) if j not in peak_areas and a[j] < a.mean()]
                if not candidates:
                    candidates = list(range(n))
                
                # Pick random candidate
                dst = candidates[rng.integers(len(candidates))]
                
                # Try different fractions
                for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbations
            for _ in range(5):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.02)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
                            break
            
            if stall > 300:
                break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 2 + 0.1
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.7 + 0.2 * np.cos(3 * x) + 0.1 * np.cos(7 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * (x - 0.3) ** 2) + 0.5 * np.exp(-5 * (x - 0.7) ** 2)
    elif choice == 6:
        # Flat-top window
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        # Slight taper at edges
        taper = n // 20
        a[:taper] = np.linspace(0.5, 1.0, taper)
        a[-taper:] = np.linspace(1.0, 0.5, taper)
    else:
        # Flat with small sinusoidal modulation
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.1 * np.cos(2 * np.pi * 5 * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def mulw_best_response(y, b_init=None, n_inner=5, T_soft=5.0):
    """
    Fast approximate LP best response using multiplicative weights.
    min_b max_k (y*b)_k s.t. sum b = 1, b >= 0
    """
    n = len(y)
    if b_init is None:
        b = np.ones(n) / n
    else:
        b = b_init.copy()
    
    for it in range(n_inner):
        # Compute convolution v = y * b
        fft_size = 1 << (2 * n - 2).bit_length()
        Y = rfft(y, fft_size)
        B = rfft(b, fft_size)
        v = irfft(Y * B, fft_size)[:2*n-1]
        
        # Soft weights on peaks
        v_shifted = v - v.max()
        w_peaks = np.exp(T_soft * v_shifted)
        w_peaks /= w_peaks.sum()
        
        # Pressure: pressure[j] = sum_k w_peaks[k] * 1[j in S_k]
        # S_k = {j : max(0,k-n+1) <= j <= min(n-1,k)}
        # For fixed j: k ranges from j to j+n-1 (bounded by [0, 2n-2])
        # pressure[j] = sum_{k=j}^{min(j+n-1, 2n-2)} w_peaks[k]
        # = prefix_sum[j+n] - prefix_sum[j] (with appropriate bounds)
        cumw = np.concatenate([[0.0], np.cumsum(w_peaks)])
        k_ends = np.minimum(np.arange(n) + n, 2 * n - 1)
        pressure = cumw[k_ends] - cumw[np.arange(n)]
        
        # Multiplicative update
        pmax = pressure.max()
        if pmax < 1e-15:
            break
        eta = 0.5  # learning rate
        b = b * np.exp(-eta * pressure / pmax)
        b = np.maximum(b, 1e-30)
        b = b / b.sum()
    
    return b

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast Nesterov-accelerated game iteration with FFT+prefix-sum multiplicative-weights
   approximate best-response (O(n log n) per iteration, main workhorse)
2. FFT-based LogSumExp gradient optimization with Adam and multi-stage annealing
3. L-BFGS refinement on softmax parameterization
4. Targeted peak-shaving coordinate descent
5. Multi-length exploration via interpolation (1600 -> 2000 -> 2400)
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_autocorr(a):
    """Full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv(x, y):
    """Full linear convolution x*y using FFT."""
    nx, ny = len(x), len(y)
    size = nx + ny - 1
    fft_size = 1 << (size - 1).bit_length()
    X = rfft(np.asarray(x, dtype=np.float64), fft_size)
    Y = rfft(np.asarray(y, dtype=np.float64), fft_size)
    return irfft(X * Y, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def compute_lse_grad(a, T):
    """
    Compute (1/T)*log(sum exp(T*b)) and gradient w.r.t. a.
    b = conv(a,a), using FFT for O(n log n).
    Returns (lse_value, gradient, weights).
    """
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    lse_val = (max_Tb + np.log(exp_Tb.sum())) / T
    w = exp_Tb / exp_Tb.sum()

    # Gradient: grad[i] = 2 * sum_j w[i+j] * a[j]
    a_rev = a[::-1].copy()
    conv_len = 3 * n - 2
    fft_size = 1 << (conv_len - 1).bit_length()
    A_rev = rfft(a_rev, fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    grad = 2.0 * conv[n - 1:n - 1 + n]
    return lse_val, grad, w


def mulw_best_response(y, n_inner=6, T_soft=5.0, eta=0.5):
    """
    Fast approximate LP best response using multiplicative weights with
    full soft-attention over all peaks (not just top-k).
    Solves: min_b max_k (y*b)_k s.t. sum b = 1, b >= 0
    Uses prefix-sum for O(n) pressure computation.
    """
    n = len(y)
    b = np.ones(n) / n
    fft_size = 1 << (2 * n - 2).bit_length()

    for _ in range(n_inner):
        Y = rfft(y, fft_size)
        B = rfft(b, fft_size)
        v = irfft(Y * B, fft_size)[:2 * n - 1]

        # Soft weights on all peaks
        v_shifted = v - v.max()
        w_peaks = np.exp(T_soft * v_shifted)
        w_peaks /= w_peaks.sum()

        # Pressure via prefix sum: pressure[j] = sum_{k=j}^{min(j+n-1, 2n-2)} w_peaks[k]
        cumw = np.concatenate([[0.0], np.cumsum(w_peaks)])
        k_ends = np.minimum(np.arange(n) + n, 2 * n - 1)
        pressure = cumw[k_ends] - cumw[np.arange(n)]

        pmax = pressure.max()
        if pmax < 1e-15:
            break

        b = b * np.exp(-eta * pressure / pmax)
        b = np.maximum(b, 1e-30)
        b = b / b.sum()

    return b


def nesterov_game(a0, time_limit, n_inner=6, T_soft=5.0, eta=0.5):
    """
    Nesterov-accelerated game iteration with fast MW best-response.
    Tracks running average for improved convergence.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0
    t = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        b_star = mulw_best_response(y, n_inner=n_inner, T_soft=T_soft, eta=eta)

        # Nesterov step
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / s

        if t <= 2:
            a_new = y_new.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s

        y = y_new
        a = a_new

        # Periodic score check
        if t % 20 == 0 or t <= 5:
            score_a = 2.0 * n * fast_autocorr(a).max()
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 20

            score_y = 2.0 * n * fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()
                no_improve = 0

        # Running average
        avg_a += a
        n_avg += 1
        if n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
                no_improve = 0

        # Restart if stalled for too long
        if no_improve > 500 and t > 1000:
            a = best_a.copy()
            y = best_a.copy()
            avg_a = np.zeros(n)
            n_avg = 0
            no_improve = 0

        if t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NG t={t} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def lse_adam_optimize(a0, time_limit, T_start=30.0, T_end=0.05, n_stages=15):
    """
    LogSumExp optimization with Adam on the simplex.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        m = np.zeros(n)
        v = np.zeros(n)
        beta1, beta2, eps = 0.85, 0.999, 1e-8
        lr = 5.0 / (si + 2)
        adam_t = 0
        iters = 0

        while time.monotonic() - t0 < time_limit:
            adam_t += 1
            iters += 1

            val, grad, _ = compute_lse_grad(a, stage_T)

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** adam_t)
            v_hat = v / (1.0 - beta2 ** adam_t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            a_new = a - step
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / s
            a = a_new

            if iters % 50 == 0:
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()

            if iters % 200 == 0:
                lr *= 0.95

            if iters > 2000:
                break

    return best_a, best_score


def lse_lbfgs_optimize(a0, time_limit, T_start=50.0, T_end=0.05, n_stages=10):
    """LSE optimization with L-BFGS-B on z-parameterization."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a_init = a_init / a_init.sum()
    z = np.log(np.maximum(a_init, 1e-30))

    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        frac = remaining / time_limit
        iters = max(10, int(30 * frac + 15))

        # Use closure-free approach: define inner functions that capture stage_T
        # via default argument trick - actually let's just inline
        z_start = z.copy()

        def _obj(z_in):
            a_tmp = softmax_fn(z_in)
            val, _, _ = compute_lse_grad(a_tmp, stage_T)
            return val

        def _jac(z_in):
            a_tmp = softmax_fn(z_in)
            _, grad_a, _ = compute_lse_grad(a_tmp, stage_T)
            S = np.dot(grad_a, a_tmp)
            return a_tmp * (grad_a - S)

        try:
            res = minimize(
                fun=_obj, x0=z_start, jac=_jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': iters * 50, 'disp': False}
            )
            z = res.x
            a = softmax_fn(z)
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
        except Exception:
            pass

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=123):
    """Targeted peak shaving via coordinate descent."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:10]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            top_contrib = np.argsort(contrib)[::-1][:5]

            for tc in top_contrib:
                src = idx[tc]
                src_val = a[src]
                if src_val < 1e-10:
                    continue

                # Find safe destination
                peak_mask = np.zeros(n, dtype=bool)
                for k2 in top_k:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    peak_mask[js2:je2 + 1] = True

                candidates = np.where(~peak_mask)[0]
                if len(candidates) == 0:
                    candidates = np.where(np.arange(n) != src)[0]
                # Prefer low-value positions
                cand_vals = a[candidates]
                if len(candidates) > 0:
                    threshold = np.percentile(cand_vals, 50)
                    low_cands = candidates[cand_vals <= threshold]
                    if len(low_cands) > 0:
                        candidates = low_cands

                dst = candidates[rng.integers(len(candidates))]

                for frac in [0.003, 0.007, 0.015, 0.03, 0.06]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            for _ in range(3):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.02)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
                            break
            if stall > 400:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1.0 - x) ** 2 + 0.1
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.7 + 0.2 * np.cos(3 * x) + 0.1 * np.cos(7 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * (x - 0.3) ** 2) + 0.5 * np.exp(-5 * (x - 0.7) ** 2)
    elif choice == 6:
        a = np.ones(n)
        taper = max(1, n // 20)
        a[:taper] = np.linspace(0.5, 1.0, taper)
        a[-taper:] = np.linspace(1.0, 0.5, taper)
    else:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.1 * np.cos(2.0 * np.pi * 5.0 * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
