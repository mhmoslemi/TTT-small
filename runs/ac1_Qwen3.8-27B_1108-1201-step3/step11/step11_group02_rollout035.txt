Specifically, the theory says: `eta_t = sqrt(2 * D^2 / (L^2 * t))` where `D` is the diameter of the domain (for entropy mirror, `D = log(n)`) and `L` is the Lipschitz constant.

The Lipschitz constant of `F` w.r.t. the L1 norm: `|F(a) - F(b)| <= L * ||a - b||_1`.

`|conv(a,a)[k] - conv(b,b)[k]| = |sum_j (a[j]-b[j])*a[k-j] + b[j]*(a[k-j]-b[k-j])|`
`<= ||a-b||_1 * max_j |a[k-j]| + ||b||_1 * max_j |a[k-j]-b[k-j]|`
`<= ||a-b||_1 * ||a||_inf + ||a-b||_inf`
`<= ||a-b||_1 * (||a||_inf + ||a||_inf)` (roughly)

"""

6. Exploits the structure: subgradient of max(conv(a,a)) at peak k is 2*a_shifted

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence using local computation (scale-invariant)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def compute_subgradient(a, b, active_frac=0.998):
    """Compute smooth subgradient using weighted active set.
    Returns gradient vector g of length n."""
    n = len(a)
    max_b = b.max()
    if max_b < 1e-300:
        return np.zeros(n)
    threshold = max_b * active_frac
    weights = np.maximum(b - threshold * 0.999, 0) ** 2
    wsum = weights.sum()
    if wsum < 1e-300:
        # Just use the argmax
        k = int(np.argmax(b))
        g = np.zeros(n)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        g[js:je+1] = 2.0 * a[k-je:k-js+1][::-1]
        return g
    weights /= wsum
    g = np.zeros(n)
    active = np.where(weights > 1e-12)[0]
    for k in active:
        w = weights[k]
        if w < 1e-12:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        g[js:je+1] += 2.0 * w * a[k-je:k-js+1][::-1]
    return g


def subgradient_hard(a, b, n_active_max=10):
    """Compute subgradient using top-k active peaks (hard)."""
    n = len(a)
    max_b = b.max()
    threshold = max_b * 0.9995
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    if len(active) > n_active_max:
        # Keep top n_active_max
        b_active = b[active]
        top = np.argsort(b_active)[::-1][:n_active_max]
        active = active[top]
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        g[js:je+1] += 2.0 * a[k-je:k-js+1][::-1]
    g /= len(active)
    return g


def nesterov_mir_descent(a0, time_limit, eta_scale=1.0, restart_patience=200):
    """
    Nesterov-accelerated mirror descent with analytical subgradients.
    Very fast - no LP required.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    y = a.copy()  # Nesterov lookahead
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    stall = 0
    prev_best = best_score
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        # Compute convolution at lookahead point
        b = fast_convolve(y)
        max_b = b.max()

        # Compute subgradient
        g = subgradient_hard(y, b, n_active_max=8)

        # Nesterov step size
        alpha = 2.0 / (np.sqrt(it + 1) + 1.0)
        eta = eta_scale * alpha / max(max_b, 1e-10)
        eta = min(eta, 1.0)  # Cap step size

        # Mirror descent step: a_new[i] = y[i] * exp(-eta * g[i])
        # Work in log-space for stability
        log_y = np.log(np.maximum(y, 1e-300))
        log_a_new = log_y - eta * g
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-300:
            a_new = y.copy()
            a_new_sum = 1.0
        a_new = a_new / a_new_sum

        # Update Nesterov lookahead
        y = (1.0 - alpha) * y + alpha * a_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-300:
            y = a_new.copy()
            sy = 1.0
        y = y / sy

        # Track best
        score_new = 2.0 * n * fast_convolve(a_new).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            stall = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0

        a = a_new
        it += 1

        # Periodic check for restart
        if it % 100 == 0:
            if best_score < prev_best - 1e-10:
                prev_best = best_score
            else:
                stall += 1
            if stall >= restart_patience:
                # Restart with perturbation
                noise_scale = 0.02 * np.sqrt(min(stall, 50) / restart_patience)
                y = y + rng.standard_normal(n) * noise_scale
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                a = y.copy()
                stall = 0
                restart_patience = min(500, restart_patience + 50)

        if (it + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NMD iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score

def build_sparse_conv_matrix(n):

    # Actually we need T[k,j] = a[k-j] for valid k-j
    # Row k, column j: value is a[k-j] if 0 <= k-j < n, else 0 (not in matrix)
    # With our structure: for column j, rows j to j+n-1 have values a[0], a[1], ..., a[n-1]
    # So data for column j is a_fixed
    data = np.repeat(a_fixed, n)
    T = csr_matrix((data, (row_idx, col_idx)), shape=(size, n))

    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [T @ b_var <= t_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver)
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            for alt in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
                if alt == solver:
                    continue
                try:
                    prob.solve(solver=alt)
                    if prob.status in ('optimal', 'optimal_inaccurate'):
                        break
                except Exception:
                    continue
    except Exception:
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    actual_conv = T @ b_val
    M_val = float(np.max(actual_conv))
    return b_val, M_val


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=50000):
    """Nesterov LP alternating method."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score
    rng = np.random.default_rng(12345)

    row_idx, col_idx = build_sparse_conv_matrix(n)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        T_sparse = build_sparse_conv_matrix(n)
        # Rebuild with current y values
        data = np.repeat(y, n)
        T = csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))

        b_new, M_b = solve_lp_best_response(y, solver, row_idx, col_idx)
        if b_new is None:
            continue

        alpha = 2.0 / (np.sqrt(it + 1) + 1.0)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
            sa = 1.0
        a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0

        if (it + 1) % 30 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score

            if stall >= 10:
                b_cur = fast_convolve(y)
                peak_idx = int(np.argmax(b_cur))
                js = max(0, peak_idx - n + 1)
                je = min(n - 1, peak_idx)
                idx = np.arange(js, je + 1)
                contribs = y[idx] * y[peak_idx - idx]
                n_reduce = max(3, len(idx) // 8)
                top_pos = idx[np.argsort(contribs)[::-1][:n_reduce]]
                for pos in top_pos:
                    if y[pos] > 1e-10:
                        y[pos] *= 0.95
                noise = rng.standard_normal(n) * 0.01
                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                a = y.copy()
                stall = 0

        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def logsumexp_smoothed_grad(a, T):
    """Compute LSE smoothed objective and gradient (vectorized via FFT)."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp(b / T - max_b / T)
    w = exp_b / exp_b.sum()

    # Gradient: grad_a[i] = (2n/T) * 2 * sum_j w[i+j] * a[j] for valid j
    # This is a cross-correlation of w and a, scaled
    # Use FFT for speed
    N = 2 * n - 1
    fft_size = 1 << (N - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # cross-correlation: sum_j w[i+j] * a[j]
    # = ifft(fft(w) * conj(fft(a_extended)))... 
    # Actually: (w cross a)[i] = sum_j w[i+j] * a[j]
    # This is the same as convolve(reversed_w, a) at appropriate offset
    # Or: ifft(conj(W) * A) gives circular cross-correlation
    # Let's just do it directly for correctness
    grad_a = np.zeros(n)
    for i in range(n):
        valid_j = min(n, N - i)
        if valid_j > 0:
            grad_a[i] = 2.0 * w[i:i + valid_j] @ a[:valid_j]
    grad_a *= (2.0 * n) / T

    logsumexp_val = max_b / T + np.log(exp_b.sum())
    obj = (2.0 * n / T) * logsumexp_val
    return obj, grad_a


def lse_lbgfs(a0, time_limit, T_stages=None, maxiter_per_stage=3000):
    """L-BFGS on LogSumExp smoothed objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [15.0, 8.0, 5.0, 3.0, 2.0, 1.2, 0.7, 0.4, 0.2, 0.1]

    best_a = a.copy()
    best_score = evaluate_local(a)
    n_stages = len(T_stages)
    iters_per_stage = max(20, maxiter_per_stage // n_stages)

    for T in T_stages:
        if time.monotonic() - t0 > time_limit - 5:
            break

        def func(z_inner):
            a_trial = np.maximum(z_inner, 0)
            st = a_trial.sum()
            if st < 1e-15:
                return 1e15
            a_trial = a_trial / st
            obj, _ = logsumexp_smoothed_grad(a_trial, T)
            return obj

        def jac(z_inner):
            a_trial = np.maximum(z_inner, 0)
            st = a_trial.sum()
            if st < 1e-15:
                return np.zeros_like(z_inner)
            a_trial = a_trial / st
            _, grad = logsumexp_smoothed_grad(a_trial, T)
            return grad

        try:
            res = minimize(fun=func, x0=a, jac=jac, method='L-BFGS-B',
                           options={'maxiter': iters_per_stage, 'ftol': 1e-15,
                                    'gtol': 1e-12, 'maxfun': iters_per_stage * 100,
                                    'disp': False})
            a_candidate = np.maximum(res.x, 0)
            sa = a_candidate.sum()
            if sa > 1e-15:
                a_candidate = a_candidate / sa
                sc = evaluate_local(a_candidate)
                if sc < best_score:
                    best_score = sc
                    best_a = a_candidate.copy()
                a = a_candidate
        except Exception:
            pass

        elapsed = time.monotonic() - t0
        if elapsed < time_limit - 5:
            print(f"    LSE-LB T={T:.2f} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def coordinate_descent_refine(a0, time_limit, n_steps=50000):
    """Targeted coordinate descent on peak-contributing elements."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall = 0
    rng = np.random.default_rng(99)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.9998
        peaks = np.where(b >= threshold)[0]
        if len(peaks) == 0:
            peaks = np.array([int(np.argmax(b))])

        improved = False
        for peak_k in peaks[:4]:
            if improved:
                break
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_m = min(10, len(idx))
            top_positions = idx[np.argsort(contribs)[::-1][:top_m]]

            for pos in top_positions:
                if a[pos] < 1e-12:
                    continue
                for factor in [0.9995, 0.999, 0.998, 0.995, 0.99, 0.98]:
                    a_trial = a.copy()
                    old_val = a_trial[pos]
                    a_trial[pos] *= factor
                    diff = old_val - a_trial[pos]
                    if diff < 1e-16:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    a_trial[mask] += diff * a_trial[mask] / max(a_trial[mask].sum(), 1e-300)
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    score = evaluate_local(a_trial)
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            if stall > 300:
                # Try increasing a random element that doesn't contribute to peak
                peak_k = int(np.argmax(b))
                js = max(0, peak_k - n + 1)
                je = min(n - 1, peak_k)
                peak_idx = set(range(js, je + 1))
                candidates = [i for i in range(n) if i not in peak_idx and a[i] > 1e-6]
                if candidates:
                    pos = rng.choice(candidates)
                    a_trial = a.copy()
                    a_trial[pos] *= 1.01
                    a_trial = a_trial / a_trial.sum()
                    score = evaluate_local(a_trial)
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                elif stall > 2000:
                    break
        else:
            stall = 0

        if (step + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc {elapsed:.0f}s: score={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create structured initial sequence."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = np.exp(-3 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = np.exp(-2.0 * (x - 0.5) ** 2 / 0.2)
    elif choice == 5:
        a = np.exp(-8 * (x - 0.3) ** 2) + np.exp(-8 * (x - 0.7) ** 2)
    elif choice == 6:
        # Edge-concentrated
        a = np.exp(-5 * x) + np.exp(-5 * (1 - x))
    else:
        # Asymmetric
        a = np.exp(-3 * x) * (1 + 0.5 * np.cos(4 * np.pi * x))
        a = np.maximum(a, 0)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def momentum_mir_descent(a0, time_limit, momentum=0.95, eta0=0.1):
    """
    Mirror descent with heavy-ball momentum in log-space.
    Different from Nesterov: uses fixed momentum coefficient.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    v = np.zeros(n)  # velocity in log-space
    log_a = np.log(a)
    it = 0
    stall = 0
    prev_best = best_score
    rng = np.random.default_rng(200)
    current_eta = eta0

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        max_b = b.max()
        g = subgradient_hard(a, b, n_active_max=6)

        # Adapt eta based on improvement
        v = momentum * v - current_eta * g
        log_new = log_a + v
        log_new -= log_new.max()
        a_new = np.exp(log_new)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-300:
            a_new = a.copy()
            a_new_sum = 1.0
        a_new = a_new / a_new_sum

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            log_a = np.log(a)
            stall = 0
            # Slightly increase eta if improving
            current_eta = min(current_eta * 1.001, 1.0)
        else:
            # Revert and reduce eta
            current_eta = max(current_eta * 0.95, 1e-4)
            v *= 0.5  # Dampen velocity
            stall += 1

        it += 1

        if it % 200 == 0:
            if best_score < prev_best - 1e-10:
                prev_best = best_score
            else:
                stall += 1
            if stall >= 500:
                # Big restart
                a = a + rng.standard_normal(n) * 0.02
                a = np.maximum(a, 0)
                a = a / a.sum()
                log_a = np.log(a)
                v = np.zeros(n)
                current_eta = eta0
                stall = 0

        if (it + 1) % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MomMD iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Fast Nesterov mirror descent (hard + soft subgradients)
    2. Momentum mirror descent
    3. LP refinement
    4. Multi-n exploration
    5. LogSumExp L-BFGS
    6. Coordinate descent polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}, Budget: {budget_s:.0f}s")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1a: Fast Nesterov MD (hard subgradient) ===
    print(f"\n=== Phase 1a: Nesterov MD Hard (score={best_score:.8f}, n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a1, _ = nesterov_mir_descent(best_seq, t_alloc, eta_scale=1.5, restart_patience=150)
        try_update(a1, "NMD_hard_1a")

    # === Phase 1b: Fast Nesterov MD (soft subgradient) ===
    print(f"\n=== Phase 1b: Nesterov MD Soft (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 350.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a2, _ = nesterov_mir_descent_soft(best_seq, t_alloc, eta_scale=1.2, active_frac=0.995)
        try_update(a2, "NMD_soft_1b")

    # === Phase 1c: Momentum Mirror Descent ===
    print(f"\n=== Phase 1c: Momentum MD (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a3, _ = momentum_mir_descent(best_seq, t_alloc, momentum=0.93, eta0=0.08)
        try_update(a3, "MomMD_1c")

    # === Phase 2: LP Refinement ===
    print(f"\n=== Phase 2: LP Refinement (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 350.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=30000)
        try_update(a_lp, "LP_p2")

    # === Phase 3: Multi-n Exploration ===
    print(f"\n=== Phase 3: Multi-n Exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 40:
        n_best = len(best_seq)
        candidate_ns = []
        for delta in [-300, -100, 100, 300, 500]:
            n_try = n_best + delta
            if 400 <= n_try <= 4000:
                candidate_ns.append(n_try)
        # Also add some novel values
        for n_try in [800, 1000, 1500, 2000]:
            if n_try not in candidate_ns and 400 <= n_try <= 4000:
                candidate_ns.append(n_try)
        candidate_ns = sorted(set(candidate_ns))
        candidate_ns = [n for n in candidate_ns if n != n_best][:5]
        per_n = t_alloc / max(1, len(candidate_ns))
        print(f"  Trying n values: {candidate_ns}, {per_n:.0f}s each")
        for n_try in candidate_ns:
            if time.monotonic() > deadline - 20:
                break
            t_sub = min(per_n, deadline - time.monotonic() - 10)
            if t_sub < 15:
                continue
            print(f"  n={n_try}, budget={t_sub:.0f}s")
            a_interp = interpolate_sequence(best_seq, n_try)
            # Quick Nesterov MD
            a_sub, _ = nesterov_mir_descent(a_interp, t_sub, eta_scale=1.5, restart_patience=100)
            try_update(a_sub, f"multi_n_{n_try}")

    # === Phase 4: LogSumExp L-BFGS ===
    print(f"\n=== Phase 4: LSE L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 15:
        T_stages = [10.0, 6.0, 4.0, 2.5, 1.5, 1.0, 0.6, 0.3, 0.15]
        print(f"  Budget: {t_alloc:.0f}s")
        a_lb = lse_lbgfs(best_seq, t_alloc, T_stages=T_stages, maxiter_per_stage=3000)
        try_update(a_lb, "LSE_LBFGS")

    # === Phase 5: Final Nesterov MD (short) ===
    print(f"\n=== Phase 5: Final NMD (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 15:
        a_fin, _ = nesterov_mir_descent(best_seq, t_alloc, eta_scale=0.8, restart_patience=300)
        try_update(a_fin, "final_NMD")

    # === Phase 6: Coordinate Descent Polish ===
    print(f"\n=== Phase 6: Coord Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.05, 60.0)
    if t_alloc > 10:
        a_cd, _ = coordinate_descent_refine(best_seq, t_alloc, n_steps=30000)
        try_update(a_cd, "coord_desc")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
