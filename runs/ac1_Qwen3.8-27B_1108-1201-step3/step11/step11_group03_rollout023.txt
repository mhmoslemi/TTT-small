The LP best response solves: min M s.t. (T b)_k <= M, sum(b)=1, b>=0

- The LP best response is what the "row player" (choosing b) does to minimize the worst-case loss against the "column player" (choosing k deterministically).
- The QP best response with a specific lambda is what the row player does to minimize the expected loss against the column player's mixed strategy lambda.

- LP approach: maximize min_b max_k (a*b)_k (over a in simplex) -- this is the value of the zero-sum game
- MW approach: max_lambda min_a a^T Q_lambda a -- this is the dual of the same problem

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a dual multiplicative-weights + QP
approach. The key insight is that the problem can be reformulated via Sion's
minimax theorem as: max_lambda min_a a^T Q_lambda a, where Q_lambda is a PSD
matrix determined by the dual variable lambda (a distribution over convolution
positions). The outer max is solved by multiplicative weights (mirror descent)
with Nesterov acceleration, and the inner min is a standard convex QP solved
exactly by an active-set or interior-point solver. This differs fundamentally
from the LP-alternating approach: the QP best response minimizes a *weighted*
correlation (determined by lambda) rather than the *max* correlation, leading
to different intermediate iterates and potentially faster convergence when
combined with the MW update on the dual.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence score locally (fast, sum-normalized)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def build_q_lambda(lambda_dist, n):
    """Build dense Q_lambda matrix (n x n) from dual distribution.
    Q[i,j] = 0.5 * lambda[i+j] for i != j, Q[i,i] = lambda[2i]
    """
    Q = np.zeros((n, n))
    for i in range(n):
        for j in range(i + 1, n):
            k = i + j
            if k < len(lambda_dist):
                Q[i, j] = 0.5 * lambda_dist[k]
                Q[j, i] = 0.5 * lambda_dist[k]
        k = 2 * i
        if k < len(lambda_dist):
            Q[i, i] = lambda_dist[k]
    return Q


def build_q_lambda_fast(lambda_dist, n):
    """Build dense Q_lambda using vectorized operations."""
    # Q[i,j] = 0.5 * lambda[i+j] for i!=j, Q[i,i] = lambda[2i]
    idx_i, idx_j = np.indices((n, n))
    k = idx_i + idx_j
    valid = k < len(lambda_dist)
    Q = np.zeros((n, n))
    Q[valid] = 0.5 * lambda_dist[k[valid]]
    # Fix diagonal: Q[i,i] should be lambda[2i], not 0.5*lambda[2i]
    diag_k = 2 * np.arange(n)
    diag_valid = diag_k < len(lambda_dist)
    Q[np.arange(n)[diag_valid], np.arange(n)[diag_valid]] = lambda_dist[diag_k[diag_valid]]
    return Q


def get_solver():
    """Find a working QP solver."""
    for solver_name in [cp.MOSEK, cp.CLARABEL, cp.GUROBI, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(10)
            prob = cp.Problem(cp.Minimize(cp.quad_form(x, np.eye(10), assume_PSD=True)),
                              [cp.sum(x) == 1, x >= 0])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_qp(q_matrix, n, solver):
    """Solve min a^T Q a s.t. sum(a)=1, a>=0. Returns a or None."""
    a_var = cp.Variable(n, nonneg=True)
    obj = cp.quad_form(a_var, q_matrix, assume_PSD=True)
    constraints = [cp.sum(a_var) == 1]
    prob = cp.Problem(cp.Minimize(obj), constraints)
    
    solvers_to_try = [cp.MOSEK, cp.CLARABEL, solver, cp.GUROBI, cp.ECOS, cp.SCS, cp.PDLP]
    seen = set()
    for s in solvers_to_try:
        if s is None or s in seen:
            continue
        seen.add(s)
        try:
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.asarray(a_var.value, dtype=np.float64)
                a_val = np.maximum(a_val, 0)
                s_sum = a_val.sum()
                if s_sum < 1e-15:
                    return None
                return a_val / s_sum
        except Exception:
            continue
    return None


def solve_qp_rank1_approx(v, n):
    """If Q ≈ vv^T (lambda concentrated), solve analytically: a = v+/sum(v+)."""
    v_pos = np.maximum(v, 0)
    s = v_pos.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return v_pos / s


def mw_qp_loop(a_init, time_limit, solver, n_max_iters=50000, eta_scale=1.0):
    """
    Multiplicative weights + QP loop with Nesterov acceleration.
    a_init: initial sequence (normalized to sum=1)
    Returns best sequence found.
    """
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    # Initialize lambda from the convolution
    b_init = fast_convolve(a)
    eta_initial = eta_scale * (2 * n - 1)
    log_lambda = eta_initial * (b_init - b_init.max())
    log_lambda -= log_lambda.max()
    lambda_dist = np.exp(log_lambda)
    lambda_dist /= lambda_dist.sum()
    
    # Nesterov state
    y = lambda_dist.copy()  # lookahead in lambda space
    a_nesterov = a.copy()  # averaged a
    best_a = a.copy()
    best_score = evaluate_local(a)
    sum_alpha = 0.0
    
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score
    rng = np.random.default_rng(123)
    qp_times = []
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov step size
        alpha = 2.0 / (it + 2)
        
        # Form Q_y and solve QP
        qp_t0 = time.monotonic()
        Q_y = build_q_lambda_fast(y, n)
        a_new = solve_qp(Q_y, n, solver)
        qp_times.append(time.monotonic() - qp_t0)
        
        if a_new is None:
            continue
        
        # Check if lambda is very concentrated → use rank-1 approximation
        lambda_entropy = -np.sum(y * np.log(y + 1e-300))
        max_entropy = np.log(2 * n - 1)
        
        if lambda_entropy < 0.1 * max_entropy and len(qp_times) > 5:
            # Very concentrated: Q ≈ v v^T where v_i = 2*a[argmax(y) - i]
            k_argmax = int(np.argmax(y))
            v = np.zeros(n)
            idx = np.arange(n)
            partners = k_argmax - idx
            valid = (partners >= 0) & (partners < n)
            v[valid] = 2.0 * a_new[partners[valid]]
            # Solve analytically
            a_analytic = solve_qp_rank1_approx(v, n)
            # Verify it's actually better
            if evaluate_local(a_analytic) <= evaluate_local(a_new) + 1e-10:
                a_new = a_analytic
        
        b_new = fast_convolve(a_new)
        
        # MW update: lambda ∝ y * exp(eta * b_new)
        eta_current = eta_scale * (2 * n - 1)
        log_update = eta_current * (b_new - b_new.max())
        log_new_lambda = np.log(y + 1e-300) + log_update
        log_new_lambda -= log_new_lambda.max()
        lambda_new = np.exp(log_new_lambda)
        lambda_new /= lambda_new.sum()
        
        # Nesterov combination
        y = (1.0 - alpha) * y + alpha * lambda_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-300:
            y = lambda_dist.copy()
        else:
            y = y / sy
        
        # Nesterov average for a
        sum_alpha += alpha
        a_nesterov = a_nesterov + alpha * (a_new - a_nesterov) / sum_alpha * (it + 1)
        # Actually, standard Nesterov: x_avg = (1/(t+1)) * sum_{k=1}^{t+1} alpha_k * x_k
        # Let's use the simpler form: a_nesterov = ((it)*a_nesterov + alpha*(it+1)*a_new) / (it+1+alpha*(it+1))
        # Simplified: just use a_new directly for tracking
        a = a_new
        
        score = evaluate_local(a)
        score_nest = evaluate_local(a_nesterov)
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
            stall_count = 0
        elif score_nest < best_score:
            best_score = score_nest
            best_a = a_nesterov.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Adaptive restart
        if (it + 1) % 50 == 0 and stall_count >= 200:
            if best_score < prev_best - 1e-10:
                stall_count = 0
                prev_best = best_score
            else:
                # Restart with perturbation
                noise = 0.02 * rng.standard_normal(n)
                a_pert = a + noise
                a_pert = np.maximum(a_pert, 0)
                sp = a_pert.sum()
                if sp > 1e-15:
                    a_pert = a_pert / sp
                    y_pert = fast_convolve(a_pert)
                    log_l = eta_current * (y_pert - y_pert.max())
                    log_l -= log_l.max()
                    y = np.exp(log_l)
                    y /= y.sum()
                    a = a_pert
                    a_nesterov = a_pert.copy()
                stall_count = 0
                prev_best = best_score
        
        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            avg_qp = np.mean(qp_times[-50:]) * 1000 if len(qp_times) >= 50 else 0
            print(f"    MW-QP iter={it+1} ({elapsed:.0f}s): score={best_score:.10f} avg_qp={avg_qp:.1f}ms ent={lambda_entropy:.2f}")
        
        if len(qp_times) > 50 and np.mean(qp_times[-50:]) > 3.0:
            print(f"    QP too slow ({np.mean(qp_times[-50:]):.2f}s), stopping at iter {it+1}")
            break
    
    return best_a, best_score


def active_set_qcqp(a, solver, rel_tol=1e-5, max_active=20):

    active = np.where(b >= threshold)[0]

    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    solvers_to_try = [cp.MOSEK, cp.CLARABEL, solver, cp.GUROBI, cp.ECOS, cp.SCS]
    seen = set()

        if s is None or s in seen:
            continue
        seen.add(s)

            if prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.asarray(a_var.value, dtype=np.float64)
                a_val = np.maximum(a_val, 0)
                s_sum = a_val.sum()
                if s_sum < 1e-15:

        except Exception:

    return None


def _build_single_qk(k, n):
    """Build Q_k matrix (n x n PSD) for the k-th convolution position."""
    Q = np.zeros((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return Q


def logsumexp_obj_grad(z, n, T_temp, deadline):
    """Compute LogSumExp-smoothed objective and gradient w.r.t. z (log-domain a)."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse
    
    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-14:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    
    S_val = np.dot(grad_a, a)
    grad_z = a * (grad_a - S_val)
    return obj, 2.0 * n * grad_z


def lse_func(z, n, T_temp, deadline):
    if time.monotonic() > deadline:
        return 1e15
    obj, _ = logsumexp_obj_grad(z, n, T_temp, deadline)
    return obj


def lse_jac(z, n, T_temp, deadline):
    if time.monotonic() > deadline:
        return np.zeros_like(z)
    _, g = logsumexp_obj_grad(z, n, T_temp, deadline)
    return g


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    iters_per_stage = max(20, maxiter_per_stage // len(T_stages))
    
    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        try:
            res = minimize(
                fun=lse_func, x0=z, jac=lse_jac,
                args=(n, stage_T, deadline),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-22,
                         'gtol': 1e-16, 'maxfun': iters_per_stage * 500, 'disp': False}
            )
            z = res.x
            # Renormalize to prevent drift
            z -= z.max()
            exp_z = np.exp(z)
            z = np.log(exp_z / exp_z.sum())
        except Exception:
            pass
    
    z_shifted = z - z.max()
    a = np.exp(z_shifted)
    a = a / a.sum()
    return a


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def multiplicative_subgradient(a0, time_limit, n_steps=100000):
    """Mirror descent on primal targeting active peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    eta_base = 0.02
    scale = 1.0
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * (1.0 - 2e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        
        eta = eta_base * scale
        log_scale = -eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        score = evaluate_local(a_new)
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 50:
            scale *= 0.85
            no_improve = 0
            if scale < 0.003:
                break
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):

    Multi-phase optimization using MW+QP (dual multiplicative weights with exact

    """

    best_score = float('inf')

    solver = get_solver()
    print(f"Using QP solver: {solver}")

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        n_init = 1200

        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")
    
    n_best = len(best_seq)
    
    def update_best(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:

    print(f"\n=== Phase 1: MW+QP n={n_best} (score={best_score:.10f}) ===")
