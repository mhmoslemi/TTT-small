def lse_objective_grad(z_in, T_val, n):
    """Compute LSE objective and gradient with respect to z (log-softmax param)."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp
    
    # Gradient: d(obj)/d(a_i) = (2n/T) * sum_k w_k * (db_k/da_i)
    # db_k/da_i = a_{k-i} (for valid indices)
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    
    # Convert to z-gradient: grad_z = a * (grad_a - (a . grad_a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)

        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    return normalize(f_interp)

from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """FFT-based linear convolution."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Safely evaluate the sequence using the provided evaluator."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if not a_list:
            return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_from_z(z):
    """Convert log-space parameters to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def normalize_vec(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


_T_structure_cache = {}


def get_T_structure(n):
    """Get precomputed row/col indices for the convolution matrix."""
    if n in _T_structure_cache:
        return _T_structure_cache[n]
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
    _T_structure_cache[n] = (np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32))
    return _T_structure_cache[n]


def build_sparse_T(a, n):
    """Build sparse convolution matrix for fixed a."""
    rows, cols = get_T_structure(n)
    vals = a[cols].astype(np.float64)
    return sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


_lp_solver = None


def get_lp_solver():
    """Get a working LP solver."""
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Solve LP: min M s.t. (T_a @ b)_k <= M, sum(b)=1, b>=0."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_sparse_T(a_fixed, n)
    conv_expr = T @ b_var
    constraints = [conv_expr <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


_global_deadline = 0.0


def lp_alternating(a0, time_limit, solver=None, alpha_start=0.8, alpha_end=0.05, verbose=True):
    """LP-based alternating minimization with geometrically decreasing step."""
    n = len(a0)
    a = normalize_vec(a0)
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    prev_best = best_score
    stall_counter = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() > _global_deadline:
            break

        # Adaptive step: geometric decay based on elapsed fraction
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        alpha = alpha_start * (alpha_end / alpha_start) ** elapsed_frac

        b_new, M_lp = solve_lp_best_response(a, solver)
        if b_new is None:
            stall_counter += 1
            if stall_counter > 100:
                break
            continue
        stall_counter = 0

        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = normalize_vec(a_new)
        sc_new = score_normalized(a_new)
        a = a_new

        if sc_new < best_score - 1e-12:
            best_score = sc_new
            best_a = a_new.copy()
            if sc_new < prev_best - 1e-10:
                prev_best = sc_new
        else:
            # Accept non-improving moves occasionally (exploration)
            if alpha > 0.2 and np.random.random() < 0.1:
                best_a = a_new.copy()

        if verbose and (time.monotonic() - t0) % 30 < 0.1:
            elapsed = time.monotonic() - t0
            print(f"    LP alt ({elapsed:.0f}s): best={best_score:.8f} cur={sc_new:.8f} alpha={alpha:.4f}")

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    """Compute LSE objective and gradient w.r.t. z."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_logsumexp(a0, time_limit, T_start=80.0, T_end=0.05, n_stages=20, iters_per_stage=40):
    """LSE L-BFGS multi-stage annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        if time.monotonic() > _global_deadline:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, stage_T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, stage_T, n)
            return grad

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 200,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)
            if (si + 1) % 5 == 0 or si == n_stages - 1:
                print(f"    LSE stage={si+1}/{n_stages} T={stage_T:.3f}: score={sc:.8f}")
        except Exception:
            pass

    return softmax_from_z(z)


def perturb_search(a0, time_limit, rng, solver=None):
    """Random perturbations followed by short LP optimization."""
    n = len(a0)
    best_a = a0.copy()
    best_score = score_normalized(a0)
    t0 = time.monotonic()
    attempt = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() > _global_deadline - 30:
            break
        attempt += 1
        # Adaptive noise scale: start large, decrease
        frac = (time.monotonic() - t0) / time_limit
        scale = 0.02 * (1.0 - 0.7 * frac) + 0.005
        noise = rng.normal(0, scale, size=n)
        a_pert = normalize_vec(a0 + noise)

        sc = score_normalized(a_pert)
        if sc < best_score:
            best_a = a_pert.copy()
            best_score = sc
            print(f"    Perturb {attempt}: new best score={sc:.8f}")

        # Short LP from perturbed point
        remaining = time_limit - (time.monotonic() - t0)
        t_sub = min(15.0, remaining * 0.3)
        if t_sub > 3:
            a_opt, sc_opt = lp_alternating(a_pert, t_sub, solver=solver,
                                            alpha_start=0.5, alpha_end=0.1, verbose=False)
            if sc_opt < best_score:
                best_a = a_opt.copy()
                best_score = sc_opt
                print(f"    Perturb {attempt} LP: new best score={sc_opt:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize_vec(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    return normalize_vec(f_interp)


def structured_init(n, rng):
    """Create a structured initial sequence."""
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-2 * x ** 2)
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif choice == 4:
        a = np.ones(n) * (1.0 + 0.2 * np.cos(2 * np.pi * 3 * np.arange(n) / n))
    else:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(3 * x) + 0.15 * np.cos(7 * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def try_update_seq(best_state, a, label, deadline):
    """Try to update the best sequence. Returns (best_seq, best_score)."""
    best_seq, best_score = best_state
    if a is None:
        return best_seq, best_score
    a = normalize_vec(a)
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
    elif np.isfinite(sc):
        print(f"  [{label}]: score={sc:.8f} (no improvement, current best={best_score:.8f})")
    return best_seq, best_score

    return [float(x) for x in best_seq]
