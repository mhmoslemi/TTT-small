def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):

def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def compute_lse_grad(a, T):
    """
    Compute (value, gradient) of (1/T)*log(sum exp(T*(a*a)_k)) w.r.t. a.
    """
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    val = (max_Tb + np.log(exp_Tb.sum())) / T
    w = exp_Tb / exp_Tb.sum()
    
    # grad[i] = 2 * sum_k w[k] * a[k-i] (cross-correlation)
    # Use FFT: conv(a_reversed, w) gives the result
    conv_len = 2 * n - 1 + n - 1  # length of conv(a[::-1], w)
    fft_size = 1 << (conv_len - 1).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    grad = 2.0 * conv[n - 1:n - 1 + n]
    return val, grad, w

def adam_lse_optimize(a0, time_limit, T_start=200.0, T_end=1.0, n_stages=15):
    """
    Adam-optimizer-based LogSumExp smoothing with multi-stage annealing.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    
    # Work in log-space: a = softmax(z)
    z = np.log(np.maximum(a, 1e-30))
    
    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = 0.01
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        # Learning rate schedule: start large, decrease
        lr_base = 0.05 * (n_stages - si) / n_stages + 0.005
        
        max_iters = int(min(200, max(20, remaining * 30 / (n_stages - si))))
        
        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                break
            
            a = softmax_fn(z)
            val, grad_a, w = compute_lse_grad(a, stage_T)
            
            # Transform gradient to z-space
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)
            
            # Adam update
            t += 1
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            z -= lr_base * m_hat / (np.sqrt(v_hat) + eps)
            
            # Periodic check
            if it % 5 == 0:
                a = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length using PCHIP-like approach."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    # Use cubic interpolation via scipy
    from scipy.interpolate import PchipInterpolator
    try:
        interp = PchipInterpolator(old_x, f_norm)
        f_interp = interp(new_x)
    except Exception:
        f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def structured_init(n, rng):

    elif choice == 4:

    elif choice == 5:

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:

import numpy as np

from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence using the provided evaluate_sequence."""

def topk_mean_grad(a, K):
    """
    Compute gradient of (1/K)*sum_{k in topK} (a*a)_k w.r.t. a.
    Returns (obj_value, gradient, top_k_indices).
    """
    n = len(a)
    b = fast_autocorr(a)
    top_idx = np.argsort(b)[::-1][:K]
    obj_val = b[top_idx].mean() / K * K  # = mean of top K
    obj_val = b[top_idx].mean()
    
    # grad[i] = (2/K) * sum_{k in topK} a[k-i] (when 0 <= k-i < n)
    grad = np.zeros(n)
    for k in top_idx:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * a[k - idx]
    grad /= K
    return obj_val, grad, top_idx

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length using PCHIP."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    try:
        interp = PchipInterpolator(old_x, f_norm)
        f_interp = interp(new_x)
    except Exception:
        f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        a = 1.0 / (1 + 10 * (x - 0.5) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def nesterov_entropy_optimize(a0, time_limit, beta=3.0, max_iters=50000):
    """
    Nesterov-accelerated iteration using entropy-based best response.
    The best response to current a is b ~ exp(-beta * (a*a)), which
    approximates the LP solution for the zero-sum game.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    y = a.copy()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    avg_a = np.zeros(n)
    n_avg = 0
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Best response: b ~ exp(-beta * (y*y))
        b_curr = fast_autocorr(y)
        b_log = -beta * (b_curr - b_curr.max())
        b_star = np.exp(b_log)
        s = b_star.sum()
        if s < 1e-15:
            b_star = np.ones(n) / n
        else:
            b_star = b_star / s
        
        # Nesterov update
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s
        
        momentum = (t - 1.0) / (t + 2.0)
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new / s
        
        y = y_new
        a = a_new
        
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        avg_a += a
        n_avg += 1
        if n_avg >= 50 and n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
        
        if t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization:
    1. Top-K mean gradient descent at parent length
    2. Nesterov entropy iteration
    3. Smart peak shaving
    4. Multi-length exploration (interpolate + optimize)
    5. Final polishing
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None
    
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    n_base = len(best_seq)
    print(f"Starting: n={n_base}, score={best_score:.6f}, budget={time_budget:.0f}s")
    
    # Phase 1: Top-K gradient descent at base length
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: Top-K GD at n={n_base} (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.20, 350.0)
    if t1 > 30:
        a_out, _ = topk_grad_descent(best_seq, t1, K_start=3, K_end=40, n_stages=12, seed=1)
        try_update(a_out, f"topk_n{n_base}")
    
    # Phase 2: Nesterov entropy at base length
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: Nesterov entropy at n={n_base} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.20, 350.0)
    if t2 > 30:
        a_out, _ = nesterov_entropy_optimize(best_seq, t2, beta=5.0)
        try_update(a_out, f"nesterov_n{n_base}")
    
    # Phase 3: Smart peak shaving at base length
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: Peak shaving at n={n_base} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 10:
        a_out, _ = smart_peak_shaving(best_seq, t3, seed=100)
        try_update(a_out, f"shave_n{n_base}")
    
    # Phase 4: LSE gradient descent
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: LSE GD at n={n_base} (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 20:
        a_out, _ = lse_grad_descent(best_seq, t4, T_start=30.0, T_end=1.0, n_stages=12, seed=5)
        try_update(a_out, f"lse_n{n_base}")
    
    # Phase 5: Multi-length exploration
    lengths_to_try = [
        int(n_base * 1.25),  # 2000
        int(n_base * 1.5),   # 2400
        int(n_base * 2.0),   # 3200
    ]
    for li, new_n in enumerate(lengths_to_try):
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining < 80:
            break
        frac = [0.12, 0.10, 0.08][min(li, 2)]
        t_phase = min(remaining * frac, 200.0)
        if t_phase < 20:
            continue
        
        print(f"\n=== Phase 5.{li+1}: Optimize at n={new_n} (budget={t_phase:.0f}s) ===")
        init_new = interpolate_sequence(best_seq, new_n)
        
        # Top-K GD
        t5a = min(t_phase * 0.5, 100.0)
        if t5a > 10:
            a_out, _ = topk_grad_descent(init_new, t5a, K_start=3, K_end=30, n_stages=8, seed=10+li)
            try_update(a_out, f"topk_n{new_n}")
        
        # Peak shaving
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining < 20:
            continue
        t5b = min(remaining * 0.05, 60.0)
        if t5b > 10:
            a_out, _ = smart_peak_shaving(best_seq, t5b, seed=200+li)
            try_update(a_out, f"shave_n{new_n}")
    
    # Phase 6: Final refinement on best
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: Final refinement (remaining={remaining:.0f}s) ===")
    t6 = min(remaining * 0.25, 200.0)
    if t6 > 10:
        # Top-K with very small K for precision
        a_out, _ = topk_grad_descent(best_seq, t6 * 0.5, K_start=3, K_end=10, n_stages=8, seed=300)
        try_update(a_out, "final_topk_smallK")
    
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 20:
        t6b = min(remaining * 0.3, 100.0)
        if t6b > 10:
            a_out, _ = smart_peak_shaving(best_seq, t6b, seed=500)
            try_update(a_out, "final_shave")
    
    # Phase 7: One last top-K pass
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 15:
        t7 = min(remaining * 0.4, 80.0)
        if t7 > 10:
            a_out, _ = topk_grad_descent(best_seq, t7, K_start=2, K_end=15, n_stages=6, seed=700)
            try_update(a_out, "last_topk")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
    
    return [float(x) for x in best_seq]
