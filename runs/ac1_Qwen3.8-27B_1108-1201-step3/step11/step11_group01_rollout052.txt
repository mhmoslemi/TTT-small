def fast_convolve(a):

    a = np.asarray(a, dtype=np.float64)

    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

    if s < 1e-30:

    """Compute subgradient of max_k conv(a,a)_k at point a using given active set."""
    n = len(a)
    g = np.zeros(n)
    for k in active_indices:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += 2.0 * a[k - idx]
    g /= len(active_indices)
    return g

def project_simplex(v):
    """Project vector onto probability simplex."""
    n = len(v)
    if v.sum() < 1e-30:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def lse_smoothed_objective(a, T):
    """LogSumExp smoothed version of max_k conv(a,a)_k."""
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp((b - max_b) / T)
    log_sum_exp = max_b + T * np.log(exp_b.sum())
    return log_sum_exp

def lse_grad(a, T):
    """Gradient of LogSumExp smoothed objective."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp((b - max_b) / T)
    w = exp_b / exp_b.sum()

        s = exp_z.sum()
        if s < 1e-300:
            return np.ones(len(z_val)) / len(z_val)
        return exp_z / s
    
    def lse_objective(z_val):
        a_val = softmax_z(z_val)
        return lse_smoothed_objective(a_val, T_staged)
    
    def lse_objective_grad(z_val):
        a_val = softmax_z(z_val)
        b = fast_convolve(a_val)
        max_b = b.max()
        exp_b = np.exp((b - max_b) / T_staged)
        w = exp_b / exp_b.sum()
        log_sum_exp = max_b + T_staged * np.log(exp_b.sum())
        
        # Gradient w.r.t. a
        g_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g_a[idx] += 2.0 * w[k] * a_val[k - idx]
        
        # Chain rule: d/dz_i = sum_j (da_j/dz_i) * g_a_j
        # da_j/dz_i = a_j * (delta_{j,i} - a_i)
        S = np.dot(g_a, a_val)
        g_z = a_val * (g_a - S)
        
        return log_sum_exp, g_z
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, T_staged in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        def func_staged(z_val):
            return lse_objective(z_val)
        
        def jac_staged(z_val):
            return lse_objective_grad(z_val)
        
        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': 50, 'ftol': 1e-20, 'gtol': 1e-18,
                         'maxfun': 10000, 'disp': False}
            )
            z = res.x
            a_check = softmax_z(z)
            sc = score_normalized(a_check)
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"      LSE stage={si+1}/{n_stages} T={T_staged:.2f}: score={sc:.8f} ({elapsed:.0f}s)")
        except Exception as e:
            if verbose:
                print(f"      LSE stage {si+1} failed: {e}")
    
    return softmax_z(z)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, maintaining normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    s = f_interp.sum()
    if s < 1e-30:
        return np.ones(new_n) / new_n
    return f_interp / s

def structured_init(n, rng, mode=0):
    """Generate structured initialization."""
    if mode == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif mode == 1:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * x * 3)
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * x * 7) + 0.2 * np.cos(2 * np.pi * x * 13)
    elif mode == 3:
        a = np.ones(n)
        a[rng.integers(0, n, size=n//10)] *= 1.5
    else:
        a = np.ones(n) * (1.0 + 0.3 * rng.random(n))
    
    a = np.maximum(a, 0.01)
    return a / a.sum()

def coordinate_descent_polish(a0, time_limit, n_iter=100000, verbose=True):
    """Coordinate descent: adjust one element at a time to reduce max."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Pick a random coordinate (or the one that seems most impactful)
        i = rng.integers(0, n)
        
        # Compute current max
        b = fast_convolve(a)
        max_b = b.max()
        
        # Try small perturbations of a_i
        best_val = a[i]
        best_local = best_score
        
        # Binary search for optimal a[i] (given others fixed)
        # The max changes as a_i changes
        for delta_frac in [0.1, -0.1, 0.05, -0.05, 0.01, -0.01]:
            test_val = max(0, a[i] * (1 + delta_frac))
            a_test = a.copy()
            a_test[i] = test_val
            a_test = normalize(a_test)
            b_test = fast_convolve(a_test)
            max_test = b_test.max()
            if max_test * 2 * n < best_local:
                best_local = max_test * 2 * n
                best_val = test_val
        
        if best_val != a[i]:
            a[i] = best_val
            a = normalize(a)
            sc = score_normalized(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if verbose and (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      CD iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score
