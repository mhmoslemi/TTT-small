def make_T(a):

def get_solv():

def lp_br(y, solv=None):
    """Solve LP best response given fixed y."""
    if solv is None: solv = get_solv()
    n = len(y)
    b = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    T = make_T(y)
    p = cp.Problem(cp.Minimize(M), [T@b <= M, cp.sum(b)==1])
    try:
        p.solve(solver=solv)
    except:
        p = cp.Problem(cp.Minimize(M), [T@b <= M, cp.sum(b)==1])
        p.solve(solver=cp.SCS)
    if p.status not in ('optimal','optimal_inaccurate'):
        return None, None
    bv = np.maximum(np.asarray(b.value, dtype=np.float64), 0)
    s = bv.sum()
    if s < 1e-15: return None, None
    return bv/s, float(M.value)

def subgrad_multi(a0, tlim, npeaks=5, momentum=0.9):
    """Multi-peak subgradient in log-space with momentum."""
    n = len(a0)
    a = norm1(np.asarray(a0, dtype=np.float64))
    z = log_a(a)
    v = np.zeros(n)  # momentum velocity
    best = a.copy()
    bs = score(a)
    t0 = time.monotonic()
    step = 0.1
    for it in range(10**7):
        if time.monotonic() >= _deadline or time.monotonic()-t0 > tlim: break
        b = fast_conv(a)
        # Find top peaks
        thresh = b.max() * (1 - 0.02)
        peaks = np.where(b >= thresh)[0]
        if len(peaks) == 0: peaks = np.array([int(np.argmax(b))])
        npe = min(npeaks, len(peaks))
        peaks = peaks[np.argsort(b[peaks])[-npe:]]
        # Compute gradient
        g = np.zeros(n)
        for k in peaks:
            # (a*a)[k] = sum_i a[i]*a[k-i], so d/d a[j] = sum over valid i: a[k-j] (if j in [0,n-1] and k-j in [0,n-1])
            js = max(0, k-n+1)
            je = min(n-1, k)
            g[js:je+1] += a[k - np.arange(js, je+1)]
        g /= npe

        sa = score(a)

        # Adaptive step
        if it % 100 == 0:

    return best, bs

def interp_len(f, new_n):

    fi = np.interp(nx, ox, fn)
    return norm1(fi)
