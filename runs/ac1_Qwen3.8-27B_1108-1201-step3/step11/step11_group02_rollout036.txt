Let me think about one more thing: **the LogSumExp approach**.

The LogSumExp(T * x) is a smooth approximation to max(x). As T → 0, it converges to max. The gradient of LogSumExp(T * conv(a,a)) is:
∇_a LogSumExp(conv(a,a)/T) = (1/T) * sum_k exp(conv(a,a)[k]/T - max/T) * ∂/∂a conv(a,a)[k]
= (1/T) * sum_k w[k] * 2 * a_cross[k]

where w[k] = exp(conv[a][k]/T - max/T) / sum_j exp(conv[a][j]/T - max/T)

This is a weighted average of the subgradients, with weights proportional to exp(conv/T). When T is small, the weights concentrate on the peaks, giving approximately the subgradient of max.

The L-BFGS method on this smooth objective should converge to a local minimum of the smoothed problem, which is close to a local minimum of the original problem when T is small.

The challenge is computing the gradient efficiently. The gradient at coordinate i is:
g[i] = (2/T) * sum_k w[k] * a[k-i] (with a[k-i] = 0 if out of range)

This is (2/T) times the cross-correlation of w with a. I can compute this using FFT:
- Pad w and a to the same length
- Compute the cross-correlation using FFT

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence (scale-invariant score)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_sparse_conv_matrix(a):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    row_ind = np.repeat(np.arange(n), n)
    col_ind = np.tile(np.arange(n), n)
    row_offset = np.repeat(np.arange(n), n)
    row_ind = row_ind + row_offset
    data = np.tile(a, n)
    return csr_matrix((data, (row_ind, col_ind)), shape=(size, n))


def get_lp_solver():
    global _solver_cache
    if hasattr(get_lp_solver, 'cached'):

def solve_best_response(a_fixed, T_sparse, solver):
    """LP: given a (sum=1), find b minimizing max conv(a,b) s.t. b>=0, sum(b)=1."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [T_sparse @ b_var <= t_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver)
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            for alt in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
                if alt == solver:
                    continue
                try:
                    prob.solve(solver=alt)
                    if prob.status in ('optimal', 'optimal_inaccurate'):
                        break
                except Exception:
                    continue
    except Exception:
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def compute_subgradient(a, b, threshold_frac=0.995):
    """Compute subgradient of max(conv(a,a)) at point a given b=conv(a,a)."""
    n = len(a)
    max_b = b.max()
    if max_b <= 0:
        return np.zeros(n)
    threshold = threshold_frac * max_b
    peaks = np.where(b >= threshold)[0]
    if len(peaks) == 0:
        peaks = np.array([np.argmax(b)])
    
    g = np.zeros(n)
    for k in peaks:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[js:je+1] += a[k - idx]
    g /= max(1, len(peaks))
    return g


def project_onto_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.maximum(v, 0)
    s = v.sum()
    if s < 1e-15:
        return np.full(n, S / n)
    return v * S / s

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def lse_objective_and_grad(a, T_val):
    """
    Compute LogSumExp smoothed objective and gradient.
    Objective: (2n/T) * LogSumExp(conv(a,a)/T)
    """
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp(b / T_val - max_b / T_val)
    w = exp_b / exp_b.sum()
    
    # grad_a[i] = (4n/T) * sum_j w[i+j] * a[j]
    g = np.zeros(n)
    for i in range(n):
        valid = min(n, 2 * n - 1 - i)
        if valid > 0:
            g[i] = np.dot(w[i:i + valid], a[:valid])
    g *= (4.0 * n) / T_val
    
    logsumexp_val = max_b / T_val + np.log(exp_b.sum())
    obj = (2.0 * n / T_val) * logsumexp_val
    return obj, g


def lse_lbgfs(a0, time_limit, T_stages=None, maxiter_per_stage=1500):
    """L-BFGS on LogSumExp smoothed objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [20.0, 12.0, 8.0, 5.0, 3.0, 2.0, 1.2, 0.7, 0.4, 0.2, 0.1]

    n_stages = len(T_stages)
    iters_per_stage = max(20, maxiter_per_stage // n_stages)
    best_a = a.copy()
    best_score = evaluate_local(a)

    for T_val in T_stages:
        if time.monotonic() - t0 > time_limit - 3:
            break

        def func(z_inner):
            if time.monotonic() - t0 > time_limit - 1:
                return 1e15
            a_trial = np.maximum(z_inner, 0)
            st = a_trial.sum()
            if st < 1e-15:
                return 1e15
            a_trial = a_trial / st
            obj, _ = lse_objective_and_grad(a_trial, T_val)
            return obj

        def jac(z_inner):
            if time.monotonic() - t0 > time_limit - 1:
                return np.zeros_like(z_inner)
            a_trial = np.maximum(z_inner, 0)
            st = a_trial.sum()
            if st < 1e-15:
                return np.zeros_like(z_inner)
            a_trial = a_trial / st
            _, grad = lse_objective_and_grad(a_trial, T_val)
            return grad

        try:
            res = minimize(fun=func, x0=a, jac=jac, method='L-BFGS-B',
                           options={'maxiter': iters_per_stage, 'ftol': 1e-14,
                                    'gtol': 1e-10, 'maxfun': iters_per_stage * 100,
                                    'disp': False})
            a = np.maximum(res.x, 0)
            sa = a.sum()
            if sa > 1e-15:
                a = a / sa
            sc = evaluate_local(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        except Exception:
            pass

    return best_a, best_score

def fast_subgradient_lp(a0, time_limit, solver, n_max_iters=200000):

    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a1, _ = fast_subgradient_lp(best_seq, t_alloc, solver)
        try_update(a1, "SG_LP_p1")

    # === Phase 2: Multi-n exploration ===
    print(f"\n=== Phase 2: Multi-n Exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 50:
        n_best = len(best_seq)
        # Generate candidate n values
        candidate_ns = []
        for delta in [-400, -200, 200, 400, -600, 600]:
            n_try = n_best + delta
            if 400 <= n_try <= 5000:
                candidate_ns.append(n_try)
        # Also try halving and doubling
        if n_best // 2 >= 400:
            candidate_ns.append(n_best // 2)
        if n_best * 2 <= 5000:
            candidate_ns.append(n_best * 2)
        # Remove duplicates and n_best
        candidate_ns = sorted(set(x for x in candidate_ns if x != n_best))
        candidate_ns = candidate_ns[:6]  # Limit to 6

        per_n = t_alloc / max(1, len(candidate_ns))
        print(f"  Trying n values: {candidate_ns}, ~{per_n:.0f}s each")
        for n_try in candidate_ns:
            if time.monotonic() > deadline - 20:
                break
            t_sub = min(per_n, deadline - time.monotonic() - 10)
            if t_sub < 15:
                continue
            print(f"  n={n_try}, budget={t_sub:.0f}s")
            # Interpolate
            a_interp = interpolate_sequence(best_seq, n_try)
            # Also try a fresh structured init
            a_struct = structured_init(n_try, rng)
            # Pick the better starting point
            sc_interp = evaluate_local(a_interp)
            sc_struct = evaluate_local(a_struct)
            if sc_struct < sc_interp:
                a_start = a_struct
            else:
                a_start = a_interp
            a_sub, _ = fast_subgradient_lp(a_start, t_sub, solver)
            try_update(a_sub, f"multi_n_{n_try}")

    # === Phase 3: Deep LP refinement on best ===
    print(f"\n=== Phase 3: Deep LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_deep, _ = fast_subgradient_lp(best_seq, t_alloc, solver)
        try_update(a_deep, "deep_lp")

    # === Phase 4: LogSumExp L-BFGS ===
    print(f"\n=== Phase 4: LSE L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 15:
        T_stages = [15.0, 10.0, 6.0, 4.0, 2.5, 1.5, 1.0, 0.6, 0.35, 0.2, 0.1]
        print(f"  Budget: {t_alloc:.0f}s")
        a_lb, _ = lse_lbgfs(best_seq, t_alloc, T_stages=T_stages,
                            maxiter_per_stage=3000)
        try_update(a_lb, "lse_lbgfs")

    # === Phase 5: Coordinate Descent Polish ===
    print(f"\n=== Phase 5: Coord Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 10:
        a_cd, _ = coord_descent_refine(best_seq, t_alloc, max_steps=100000)
        try_update(a_cd, "coord_desc")

    # === Phase 6: Final quick SG+LP ===
    print(f"\n=== Phase 6: Final SG+LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.05, 60.0)
    if t_alloc > 5:
        a_fin, _ = fast_subgradient_lp(best_seq, t_alloc, solver)
        try_update(a_fin, "final_sg_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
