def fast_convolve(a):
    """Compute linear autocorrelation using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-300:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Inverse of softmax (log of a, with clipping)."""
    return np.log(np.maximum(a, 1e-300))


def compute_lse_gradient(a, T, n):
    """
    Compute gradient of 2n * LSE(T * conv(a,a)) w.r.t. a.
    a is on the simplex (sum=1, a>=0).
    Returns (objective_value, gradient).
    """
    C = fast_convolve(a)  # length 2n-1
    TC = T * C
    max_TC = TC.max()
    exp_TC = np.exp(TC - max_TC)
    w = exp_TC / exp_TC.sum()
    lse_val = max_TC + np.log(exp_TC.sum())
    obj = 2.0 * n * lse_val
    
    # Gradient: grad_i = 4nT * sum_k w_k * a_{k-i}
    # sum_k w_k * a_{k-i} is the cross-correlation of w and a
    # w has length 2n-1, a has length n
    # We want (w * a)_i = sum_k w_k * a_{k-i} for i=0,...,n-1
    # This equals conv(w, a) evaluated at positions i + (2n-2) - (n-1) ... 
    # Actually: conv(w, a)_m = sum_k w_k * a_{m-k}
    # We want sum_k w_k * a_{k-i} = conv(w, a)_{?}
    # conv(w, a)_m = sum_k w_k * a_{m-k}, valid when 0 <= m-k <= n-1
    # We want: sum_k w_k * a_{k-i} where 0 <= k-i <= n-1, i.e., i <= k <= i+n-1
    # Let m = some function... 
    # Actually, let's use: sum_k w_k * a_{k-i} = sum_j w_{j+i} * a_j (substituting j = k-i)
    # = (correlation of w and a at lag i)
    # This is the cross-correlation, computable as:
    # ifft(fft(w_padded) * conj(fft(a_padded)))
    
    # Let's use a different approach:
    # grad_i = 4nT * sum_k w_k * a_{k-i}
    # = 4nT * sum_j w_{j+i} * a_j (letting j = k-i, so k = j+i)
    # This is the cross-correlation of w and a at lag i.
    # Cross-correlation can be computed via:
    # xcorr_i = sum_j w_{j+i} * a_j = conv(w, reverse(a))_{...}
    # Or: xcorr = ifft(fft(w_padded) * conj(fft(a_padded)))
    
    fft_size = 1 << (3 * n - 2 - 1).bit_length()  # need size >= (2n-1) + n - 1 = 3n-2
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    xcorr = irfft(W * np.conj(A), fft_size)
    # xcorr[m] = sum_k w_k * a_{k-m} (for valid indices)
    # We want grad_i proportional to xcorr[i] for i = 0, ..., n-1
    # Wait: xcorr[m] = sum_k w_k * a_{k-m}
    # We want sum_k w_k * a_{k-i} which is xcorr[i]
    
    grad = 4.0 * n * T * xcorr[:n]
    
    return obj, grad


def lse_fista(a0, time_limit, T_start=50.0, T_end=0.5, n_stages=20):
    """
    FISTA optimization of LSE-smoothed objective with temperature annealing.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        
        # FISTA parameters
        a_fista = a.copy()
        y = a.copy()
        t_fista = 1.0
        eta = 1.0 / (8.0 * n * T + 1.0)  # Conservative step size
        max_iters = int(remaining * 5000)  # ~5000 iters per second target
        
        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Compute gradient at y
            obj_y, grad_y = compute_lse_gradient(y, T, n)
            
            # FISTA update
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_fista * t_fista)) / 2.0
            momentum = (t_fista - 1.0) / t_new
            t_fista = t_new
            
            a_old = a_fista.copy()
            z = y - eta * grad_y
            a_fista = project_simplex(z)
            y = a_fista + momentum * (a_fista - a_old)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a_fista.copy()
            else:
                y = y / sy
            
            # Check improvement
            if (it + 1) % 100 == 0:
                sc = score_normalized(a_fista)
                if sc < best_score:
                    best_score = sc
                    best_a = a_fista.copy()
                    a = a_fista.copy()
        
        # Update best
        sc = score_normalized(a_fista)
        if sc < best_score:
            best_score = sc
            best_a = a_fista.copy()
            a = a_fista.copy()
        
        if (si + 1) % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LSE-FISTA stage={si+1}/{n_stages} T={T:.1f}: score={best_score:.8f} ({elapsed:.0f}s)")
    
    return best_a, best_score


def build_conv_matrix(a_fixed):
    """Build the convolution matrix T such that T @ b = conv(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None, prev_b=None):
    """Solve the LP best response problem."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    
    # Warm start
    if prev_b is not None:
        b_var.value = prev_b
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp(a0, time_limit, solver=None, max_iters=500000):
    """Nesterov-accelerated LP best response (fictitious play)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0
    prev_b = None
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_lp = solve_lp_best_response(y, solver, prev_b)
        if b_new is None:
            continue
        prev_b = b_new

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        if (it + 1) % 100 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 15:
                print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
                last_print = elapsed

    return best_a, best_score
