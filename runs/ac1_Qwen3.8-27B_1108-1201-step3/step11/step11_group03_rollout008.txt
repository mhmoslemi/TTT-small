import time

def get_lp_solver():
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver, deadline=None):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = [solver, cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    for s in solvers_to_try:
        if s is None:
            continue
        if deadline is not None and time.monotonic() > deadline:
            return None, None
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_sum = b_val.sum()
                if s_sum < 1e-15:
                    return None, None
                b_val = b_val / s_sum
                return b_val, float(max(0, M_var.value))
        except Exception:
            continue
    return None, None


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=200000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(123)
    lp_times = []
    deadline = t0 + time_limit

    for it in range(n_max_iters):
        if time.monotonic() > deadline:
            break
        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver)
        lp_times.append(time.monotonic() - lp_t0)
        if b_new is None:
            if len(lp_times) > 20 and np.mean(lp_times[-20:]) > 3.0:
                break
            continue
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            if stall_count >= 10:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 7)
                amp = rng.uniform(0.02, 0.05)
                noise = amp * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                sy2 = y.sum()
                if sy2 > 1e-15:
                    y = y / sy2
                stall_count = 0
            if (it + 1) % 2000 == 0:
                elapsed = time.monotonic() - t0
                avg_lp = np.mean(lp_times[-50:]) * 1000 if len(lp_times) >= 50 else 0
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.10f} avg_lp={avg_lp:.1f}ms")
        if len(lp_times) > 100 and np.mean(lp_times[-100:]) > 4.0:
            print(f"    LP too slow ({np.mean(lp_times[-100:]):.2f}s/iter), stopping at iter {it+1}")
            break
    return best_a, best_score


def solve_qcqp_constraint_gen(a_init, solver, deadline, max_active=80, max_rounds=15, tol=1e-6):
    """
    Constraint generation QCQP: iteratively solve with active constraints,
    check for violations, add violated constraints, repeat.
    """
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return None
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    for round_num in range(max_rounds):
        if time.monotonic() > deadline:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Identify active set: constraints close to the max
        threshold = max_b * (1.0 - 1e-4)
        active = set(np.where(b >= threshold)[0].tolist())

        # Also add the top constraints by value
        top_k = min(20, 2 * n - 1)
        top_idx = np.argsort(b)[-top_k:]
        for idx in top_idx:
            active.add(int(idx))

        if len(active) > max_active:
            active_list = list(active)
            active_list.sort(key=lambda k: -b[k])
            active = set(active_list[:max_active])

        # Solve QCQP with these constraints
        a_var = cp.Variable(n, nonneg=True)
        t_var = cp.Variable()
        constraints = [cp.sum(a_var) == 1]

        for k in sorted(active):
            Q = build_q_sparse(int(k), n)
            constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)

        prob = cp.Problem(cp.Minimize(t_var), constraints)
        solvers_to_try = [cp.MOSEK, cp.GUROBI, solver, cp.CLARABEL, cp.SCS, cp.ECOS]
        a_sol = None
        for sv in solvers_to_try:
            if sv is None:
                continue
            if time.monotonic() > deadline:
                break
            try:
                prob.solve(solver=sv)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    a_sol = np.asarray(a_var.value, dtype=np.float64)
                    a_sol = np.maximum(a_sol, 0)
                    ss = a_sol.sum()
                    if ss < 1e-15:
                        a_sol = None
                        continue
                    a_sol = a_sol / ss
                    break
            except Exception:
                continue

        if a_sol is None:
            break

        # Check for violated constraints
        b_sol = fast_convolve(a_sol)
        t_val = float(t_var.value) if t_var.value is not None else b_sol.max()
        violations = np.where(b_sol > t_val + tol)[0]

        score_sol = 2.0 * n * b_sol.max()
        if score_sol < best_score:
            best_score = score_sol
            best_a = a_sol.copy()
        a = a_sol

        if len(violations) == 0:
            print(f"    QCQP CG round {round_num+1}: converged, active={len(active)}, score={best_score:.10f}")
            break

        # Add most violated constraints
        violation_amounts = b_sol[violations] - t_val
        worst = violations[np.argsort(violation_amounts)[-10:]]
        for w in worst:
            active.add(int(w))
        if len(active) > max_active:
            active_list = list(active)
            active_list.sort(key=lambda k: -b_sol[k])
            active = set(active_list[:max_active])
        print(f"    QCQP CG round {round_num+1}: {len(violations)} violations, active={len(active)}, score={best_score:.10f}")

    return best_a


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_grad(z, n, T_temp):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse
    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    S_val = np.dot(grad_a, a)
    grad_z = a * (grad_a - S_val)
    return obj, 2.0 * n * grad_z


def lse_func(z, n, T_temp, deadline):
    if time.monotonic() > deadline:
        return 1e15
    obj, _ = logsumexp_obj_grad(z, n, T_temp)
    return obj


def lse_jac(z, n, T_temp, deadline):
    if time.monotonic() > deadline:
        return np.zeros_like(z)
    _, g = logsumexp_obj_grad(z, n, T_temp)
    return g


def lbgfs_refine(a0, time_limit, T_stages=None):
    if T_stages is None:
        T_stages = [2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    n_stages = len(T_stages)
    iters_per_stage = max(30, 6000 // n_stages)

    for stage_T in T_stages:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 3:
            break
        try:
            res = minimize(
                fun=lse_func, x0=z, jac=lse_jac,

                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': iters_per_stage * 500, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    return softmax_from_z(z)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def multiplicative_subgradient(a0, time_limit, n_steps=50000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta_base = 0.03
    no_improve = 0
    scale = 1.0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * (1.0 - 3e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([0])
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        eta = eta_base * scale
        log_scale = -eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= 30:
            scale *= 0.8
            no_improve = 0
            if scale < 0.005:
                break
    return best_a, best_score

    # === Phase 3: LP alternating round 2 ===
    print(f"\n=== Phase 3: LP Round 2 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 150.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, solver, n_max_iters=100000)
        update_best(a_lp2, "lp_phase3")

    # === Phase 4: L-BFGS LogSumExp ===
    print(f"\n=== Phase 4: L-BFGS Low-Temp (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_lbgfs = min(rem * 0.10, 130.0)
    if t_lbgfs > 15:
        T_stages = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
        a_lb = lbgfs_refine(best_seq, t_lbgfs, T_stages)
        update_best(a_lb, "lbgfs")

    # === Phase 5: Final QCQP + multiplicative ===
    print(f"\n=== Phase 5: Final Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        a_ms, _ = multiplicative_subgradient(best_seq, min(rem * 0.15, 25.0), n_steps=30000)
        update_best(a_ms, "final_mult")
        rem = deadline - time.monotonic()
        if rem > 20:
            a_cg3 = solve_qcqp_constraint_gen(best_seq, solver, deadline, max_active=50, max_rounds=8, tol=1e-6)
            update_best(a_cg3, "qcqp_final")

    # === Phase 6: Try n=1199 ===
    print(f"\n=== Phase 6: Try n=1199 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        t_1199 = min(rem * 0.14, 200.0)
        print(f"  Budget: {t_1199:.0f}s")
        init_1199 = interpolate_sequence(best_seq, 1199)
        a_1199, _ = nesterov_lp_refine(init_1199, t_1199 * 0.55, solver, n_max_iters=100000)
        update_best(a_1199, "lp_n1199")
        rem = deadline - time.monotonic()
        if len(best_seq) == 1199 and rem > 20:
            a_q = solve_qcqp_constraint_gen(best_seq, solver, deadline, max_active=60, max_rounds=8, tol=1e-5)
            update_best(a_q, "qcqp_n1199")

    # === Phase 7: Try n=1201 ===
    print(f"\n=== Phase 7: Try n=1201 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        t_1201 = min(rem * 0.14, 200.0)
        print(f"  Budget: {t_1201:.0f}s")
        # Re-interpolate from original n=1200 best
        if len(best_seq) != 1201:
            base_for_interp = best_seq
            if len(base_for_interp) == 1200:
                init_1201 = interpolate_sequence(base_for_interp, 1201)
            else:
                # Interpolate back to 1200 first, then to 1201
                base_1200 = interpolate_sequence(best_seq, 1200)
                init_1201 = interpolate_sequence(base_1200, 1201)
        else:
            init_1201 = best_seq.copy()
        a_1201, _ = nesterov_lp_refine(init_1201, t_1201 * 0.55, solver, n_max_iters=100000)
        update_best(a_1201, "lp_n1201")
        rem = deadline - time.monotonic()
        if len(best_seq) == 1201 and rem > 20:
            a_q = solve_qcqp_constraint_gen(best_seq, solver, deadline, max_active=60, max_rounds=8, tol=1e-5)
            update_best(a_q, "qcqp_n1201")

    # === Phase 8: Final L-BFGS on best ===
    print(f"\n=== Phase 8: Final L-BFGS (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        T_stages_final = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
        a_lb_final = lbgfs_refine(best_seq, min(rem * 0.5, 60.0), T_stages_final)
        update_best(a_lb_final, "lbgfs_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
