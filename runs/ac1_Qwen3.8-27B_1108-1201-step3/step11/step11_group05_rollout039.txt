import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_sparse(a):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    T = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS, cp.GLOP, cp.GLPK]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def solve_lp_best_response(a_fixed, solver, b_warm=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_sparse(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]
    seen = set()
    for s in solvers_to_try:
        if s is None or s in seen:
            continue
        seen.add(s)
        try:
            kwargs = {}
            if b_warm is not None and s == cp.CLARABEL:
                kwargs['warm_start'] = True
            prob.solve(solver=s, **kwargs)
            if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
                break
        except Exception:
            continue
    else:
        return None

    b_val = np.asarray(b_var.value, dtype=np.float64).flatten()
    b_val = np.maximum(b_val, 0)
    b_val[np.isnan(b_val)] = 0
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def get_active_set(c, rel_tol=1e-4):
    """Get indices where c is within rel_tol of max."""
    max_c = c.max()
    threshold = max_c * (1.0 - rel_tol)
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    return active


def subgradient_active(a, active_set):
    """Compute subgradient of max_k conv(a,a)[k] at given active set."""
    n = len(a)
    g = np.zeros(n)
    for k in active_set:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        indices = np.arange(i_start, i_end + 1)
        g[indices] += a[k - indices]
    g /= len(active_set)
    return g


def multiplicative_weights_step(a, g, eta):
    """One step of multiplicative weights (mirror descent on simplex)."""
    log_update = -eta * g
    log_update -= log_update.max()
    a_new = a * np.exp(log_update)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-300:
        return a.copy()
    return a_new / s


def multiplicative_polish(a0, time_limit, eta=0.05, rel_tol=1e-3):
    """Multiplicative weights method targeting active peaks."""
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    no_improve = 0
    eta_cur = eta

    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        active = get_active_set(c, rel_tol)
        g = subgradient_active(a, active)
        a_new = multiplicative_weights_step(a, g, eta_cur)
        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 30:
                eta_cur *= 0.8
                no_improve = 0
                if eta_cur < 1e-5:
                    break
        if (int((time.monotonic() - t0) * 10)) % 50 == 0:
            pass
    return best_a, best_score


def adam_logsumexp(a0, time_limit, temperatures, lr_init=0.01):
    """Adam optimization in log-space with LogSumExp smoothing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    z -= z.mean()
    best_a = a0.copy()
    best_score = compute_score(a0)
    t0 = time.monotonic()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step_count = 0
    n_temps = len(temperatures)

    for T_idx, T in enumerate(temperatures):
        if time.monotonic() - t0 >= time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        stage_budget = remaining / (n_temps - T_idx)
        stage_steps = max(500, int(stage_budget * 800))

        for _ in range(stage_steps):
            if time.monotonic() - t0 >= time_limit:
                break
            step_count += 1

            a = np.exp(z - z.max())
            a /= a.sum()
            c = fast_convolve(a)
            max_c = c.max()
            score = 2.0 * n * max_c

            if score < best_score:
                best_score = score
                best_a = a.copy()

            if T < 0.02:
                active = get_active_set(c, 1e-4)
                w = np.zeros(len(c))
                w[active] = 1.0 / len(active)
            else:
                shifted = (c - max_c) / T
                w = np.exp(shifted - shifted.max())
                w /= w.sum()

            g = np.correlate(w, a, mode='full')[n - 1:2 * n - 1]
            S = np.dot(g, a)
            grad_z = a * (g - S)

            progress = step_count / max(1, stage_steps)
            lr = lr_init * (1.0 - 0.5 * progress)
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * grad_z ** 2
            m_hat = m / (1.0 - beta1 ** (step_count + 1))
            v_hat = v / (1.0 - beta2 ** (step_count + 1))
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            z -= z.mean()

    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver, max_iters=500000):
    """Nesterov dual averaging LP alternating method."""
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    b_prev = None
    lp_times = []
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(12345)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        lp_t0 = time.monotonic()
        b_new = solve_lp_best_response(y, solver, b_warm=b_prev)
        lp_time = time.monotonic() - lp_t0
        lp_times.append(lp_time)

        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = a.copy()
        else:
            y = y / s_y

        a = ((it + 1) * a + (it + 2) * alpha * b_new) / ((it + 1) + (it + 2) * alpha)
        a = np.maximum(a, 0)
        s_a = a.sum()
        if s_a < 1e-15:
            a = y.copy()
        else:
            a = a / s_a

        score_a = compute_score(a)
        score_y = compute_score(y)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        b_prev = b_new

        if (it + 1) % 25 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 15:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 6)
                noise = 0.02 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = np.maximum(y + noise, 0)
                y /= y.sum()
                stall_count = 0

            if (it + 1) % 100 == 0:
                avg_lp = np.mean(lp_times[-20:]) if len(lp_times) >= 20 else lp_time
                elapsed = time.monotonic() - t0
                print(f"    LP iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} avg_lp={avg_lp*1000:.0f}ms")

        if len(lp_times) > 30 and np.mean(lp_times[-30:]) > 3.0:
            print(f"    LP too slow, stopping at iter {it+1}")
            break

    return best_a, best_score


def structured_init(n, rng):
    """Create various structured initial sequences."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = x ** 0.5
    elif choice == 2:
        a = np.exp(-3 * x)
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 2 + 0.1 * x
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def perturb_sequence(a, rng, strength=0.02):
    """Create a perturbed version of sequence a."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 8)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    a_pert = np.maximum(a + perturbation, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s

    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses LP alternating, Adam LogSumExp, multiplicative weights,
    and multi-length exploration with adaptive time allocation.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Budget: {budget_s:.0f}s")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = compute_score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = compute_score(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = compute_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}) ***")

    # === Phase 1: LP Alternating on original n (main workhorse) ===
    print(f"\n=== Phase 1: Nesterov LP n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.40, 600.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver, max_iters=500000)
        local_update(a_lp, "lp_phase1")

    # === Phase 2: Adam LogSumExp fine-tuning ===
    print(f"\n=== Phase 2: Adam LogSumExp n={len(best_seq)} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        temps = [5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]
        a_adam, _ = adam_logsumexp(best_seq, t_alloc, temps, lr_init=0.008)
        local_update(a_adam, "adam_phase2")

    # === Phase 3: Multiplicative weights polish ===
    print(f"\n=== Phase 3: MultiWeights Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_mw, _ = multiplicative_polish(best_seq, t_alloc, eta=0.04, rel_tol=1e-3)
        local_update(a_mw, "mw_phase3")

    # === Phase 4: Interpolate to n=1500 and optimize ===
    print(f"\n=== Phase 4: Interpolate to n=1500 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 200:
        n_new = 1500
        t_phase4 = min(rem * 0.30, 400.0)
        print(f"  Budget: {t_phase4:.0f}s, interpolating from n={len(best_seq)} to n={n_new}")
