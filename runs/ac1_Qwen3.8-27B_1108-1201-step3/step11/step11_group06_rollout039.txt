def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Fast local evaluation (assumes a >= 0, sum > 0)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()

        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix_sparse(a):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    rows = np.zeros(n * n, dtype=np.int32)
    cols = np.zeros(n * n, dtype=np.int32)
    vals = np.zeros(n * n, dtype=np.float64)
    count = 0
    for j in range(n):
        for k in range(j, min(j + n, size)):
            rows[count] = k
            cols[count] = j
            vals[count] = a[k - j]
            count += 1
    return csr_matrix((vals[:count], (rows[:count], cols[:count])), shape=(size, n))

def flat_convolution_sequence(n):
    """Generate sequence with approximately flat convolution (front half)."""
    t = 1.0
    a = np.zeros(n)
    a[0] = np.sqrt(t)
    for k in range(1, n):
        s = 0.0
        for i in range(1, k):
            s += a[i] * a[k - i]
        a[k] = (t - s) / (2.0 * np.sqrt(t))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def init_flat_conv_fast(n):
    """Faster flat-convolution init using FFT-based convolution check."""

        s = np.dot(a[1:k], a[k-1:0:-1]) if k > 1 else 0.0

    return a / s

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += wi * a[k - idx]

def lse_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z for softmax parameterization."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        a = np.ones(n) / n
    else:
        a = exp_z / s
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T_temp
    scaled_shifted = scaled - scaled.max()
    exp_terms = np.exp(scaled_shifted)
    exp_sum = exp_terms.sum() + 1e-300
    lse = max_b + T_temp * (scaled.max() + np.log(exp_sum))
    obj = 2.0 * n * lse
    w = exp_terms / (exp_terms.sum() + 1e-300)
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
    
    # === Phase 0: Diverse initialization ===
    n_best = len(best_seq) if best_seq is not None else 1200
    print(f"\n=== Phase 0: Diverse Init (n={n_best}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        for idx in range(6):
            a_init = diverse_initialization(n_best, rng, idx + 10)  # start at 10 to avoid parent
            sc = evaluate_local(a_init)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_init.copy()
                print(f"  Init[{idx}]: score={sc:.8f}")
    
    # Also try flat-convolution init
    a_flat = init_flat_conv_fast(n_best)
    sc = evaluate_local(a_flat)
    print(f"  Flat-conv init: score={sc:.8f}")
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_flat.copy()
    
    n_best = len(best_seq)
    print(f"  Best after init: score={best_score:.8f}, n={n_best}")
    
    # === Phase 1: Fast MW ===
    print(f"\n=== Phase 1: MW (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.04, 50.0)
    if t1 > 10:
        a_mw, _ = multiplicative_weights(best_seq, t1, n_steps=500000, eta0=0.05, active_frac=0.03)
        try_update(a_mw, "mw_1")
    
    # === Phase 2: Proximal Gradient (new) ===
    print(f"\n=== Phase 2: Proximal Gradient (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.06, 80.0)
    if t2 > 20:
        a_pg, _ = proximal_gradient(best_seq, t2, max_iter=50000)
        try_update(a_pg, "pg_1")
    
    # === Phase 3: LP Alternating (main workhorse) ===
    print(f"\n=== Phase 3: LP Alternating (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.35, 500.0)
    if t3 > 60:
        print(f"  Budget: {t3:.0f}s")
        a_lp, _ = lp_alternating_warm(best_seq, t3, solver, n_max_iters=500000)
        try_update(a_lp, "lp_1")
    
    # === Phase 4: Peak Deflation ===
    print(f"\n=== Phase 4: Peak Deflation (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        a_def, _ = peak_deflation(best_seq, n_iter=100, strength=0.08)
        try_update(a_def, "deflate_1")
        # Re-run LP briefly after deflation
        rem = deadline - time.monotonic()
        if rem > 60:
            t4b = min(rem * 0.15, 120.0)
            a_lp2, _ = lp_alternating_warm(best_seq, t4b, solver, n_max_iters=200000)
            try_update(a_lp2, "lp_2")
    
    # === Phase 5: LSE L-BFGS ===
    print(f"\n=== Phase 5: LSE L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t5 = min(rem * 0.2, 280.0)
        T_stages = [20.0, 10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
        print(f"  {len(T_stages)} stages, budget={t5:.0f}s")
        a_lse = lse_lbfgs_refine(best_seq, t5, T_stages, maxiter_per_stage=500)
        try_update(a_lse, "lse_1")
    
    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-length (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    lengths_to_try = [800, 1000, 1500, 1800]
    for ln in lengths_to_try:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        if abs(len(best_seq) - ln) < 50:
            continue
        t_ln = min(rem * 0.12, 150.0)
        if t_ln < 30:
            continue
        print(f"  Trying n={ln}, budget={t_ln:.0f}s")
        init_ln = interpolate_sequence(best_seq, ln)
        # Quick MW
        a_ln_mw, _ = multiplicative_weights(init_ln, t_ln * 0.2, n_steps=100000, eta0=0.05, active_frac=0.03)
        try_update(a_ln_mw, f"mw_n{ln}")
        # LP
        rem = deadline - time.monotonic()
        t_lp = min(rem * 0.4, t_ln * 0.6)
        if t_lp > 30:
            start = a_ln_mw if a_ln_mw is not None else init_ln
            a_ln_lp, _ = lp_alternating_warm(start, t_lp, solver, n_max_iters=200000)
            try_update(a_ln_lp, f"lp_n{ln}")
    
    # === Phase 7: Perturbation + LP ===
    print(f"\n=== Phase 7: Perturbation escape (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        for trial in range(3):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            n = len(best_seq)
            x = np.linspace(0, 2 * np.pi, n)
            freq = rng.integers(1, 8)
            phase = rng.uniform(0, 2 * np.pi)
            strength = 0.01 + 0.008 * trial
            pert = strength * np.cos(freq * x + phase)
            a_pert = normalize(best_seq + pert)
            t_sub = min(rem * 0.25, 80.0)
            a_opt, _ = lp_alternating_warm(a_pert, t_sub, solver, n_max_iters=100000)
            try_update(a_opt, f"pert{trial}")
    
    # === Phase 8: Final LP polish ===
    print(f"\n=== Phase 8: Final LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t8 = min(rem * 0.3, 100.0)
        a_fin, _ = lp_alternating_warm(best_seq, t8, solver, n_max_iters=200000)
        try_update(a_fin, "final_lp")
    
    # === Phase 9: Final MW + deflation ===
    print(f"\n=== Phase 9: Final polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t9 = min(rem * 0.3, 40.0)
        a_mw_fin, _ = multiplicative_weights(best_seq, t9, n_steps=200000, eta0=0.02, active_frac=0.01)
        try_update(a_mw_fin, "final_mw")
        a_def_fin, _ = peak_deflation(best_seq, n_iter=50, strength=0.03)
        try_update(a_def_fin, "final_def")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} n={len(best_seq)}")
    
    if best_seq is None:
        best_seq = np.ones(1200) / 1200
    
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
