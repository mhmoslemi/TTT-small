def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:

        # any other non-integer/non-float types (like strings or complex numbers).

            return np.inf

        if np.isnan(x) or np.isinf(x):
            return np.inf

    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_sparse_conv_matrix(a):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)

    col_ind = []

        col_ind.extend([i] * n)
        data.extend(a)
    T = csr_matrix((data, (np.array(row_ind), np.array(col_ind))), shape=(size, n))
    return T

def softmax_safe(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def logsumexp_obj_and_grad(a, T):
    """
    Compute LogSumExp smoothed objective and gradient.
    Objective: (2n/T) * LogSumExp(conv(a,a)/T)
    Gradient wrt a computed via cross-correlation.
    """
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp(b / T - max_b / T)
    w = exp_b / exp_b.sum()

    # grad_a[i] = (2n/T) * sum_k w[k] * 2*a[k-i] (for valid k-i)
    # = (4n/T) * sum_{j=0}^{n-1} w[i+j] * a[j]
    grad_a = np.zeros(n)
    for i in range(n):
        # Sum over j: w[i+j] * a[j], where i+j < 2n-1
        valid_j = min(n, 2 * n - 1 - i)
        if valid_j > 0:
            grad_a[i] = 2.0 * w[i:i + valid_j] @ a[:valid_j]
    grad_a *= (2.0 * n) / T

    # Objective value
    logsumexp_val = max_b / T + np.log(exp_b.sum())
    obj = (2.0 * n / T) * logsumexp_val

    return obj, grad_a

def mirror_descent_adam(a0, time_limit, eta0=0.05):
    """
    Mirror descent with Adam-like adaptive step size.
    Uses per-coordinate exponential moving averages for step adaptation.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = visualize_score(a)
    t0 = time.monotonic()

    # Adam state
    m_g = np.zeros(n)  # First moment
    v_g = np.zeros(n)  # Second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_count = 0
    no_improve = 0

    for it in range(2000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Compute subgradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= max(1, len(active))

        # Adam update
        t_count += 1
        m_g = beta1 * m_g + (1 - beta1) * g
        v_g = beta2 * v_g + (1 - beta2) * g * g
        m_hat = m_g / (1 - beta1 ** t_count)
        v_hat = v_g / (1 - beta2 ** t_count)
        adaptive_eta = eta0 / (np.sqrt(v_hat + eps) + eps)
        # Adaptive eta should be moderate
        adaptive_eta = np.clip(adaptive_eta, 1e-6, 2.0)
        # Use scalar average for stability
        eta_eff = float(np.mean(adaptive_eta))
        eta_eff = max(1e-5, min(eta_eff, 1.0))

        # Mirror descent step
        log_a = np.log(np.maximum(a, 1e-300))
        g_shifted = g - g.min()
        log_a_new = log_a - eta_eff * g_shifted
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-300:
            a_new = a.copy()
            a_new_sum = 1.0
        a_new = a_new / a_new_sum
        a = a_new

        score = visualize_score(a)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 500:
            rng = np.random.default_rng(it * 13 + 7)
            noise = rng.standard_normal(n) * 0.01
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            no_improve = 0
            m_g = np.zeros(n)
            v_g = np.zeros(n)

        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD-Adam iter={it+1} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create structured initial sequence."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = np.exp(-3 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = np.exp(-2.0 * (x - 0.5) ** 2 / 0.2)
    else:
        # Bimodal
        a = np.exp(-8 * (x - 0.3) ** 2) + np.exp(-8 * (x - 0.7) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
