def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)

def softmax_from_z(z):

def build_conv_matrix(a_fixed):

    size = 2 * n - 1

    # (T @ b)[k] = sum_j a_fixed[k-j] * b[j]
    T = np.zeros((size, n))

    return T

def lse_objective_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

            je = min(n - 1, k)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

import numpy as np

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:

        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_restarts(a0, time_limit, solver=None, max_iters=1000000,
                         restart_interval=400, restart_prob=0.25, pert_scale=0.15):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0
    rng = np.random.default_rng(99999)
    stall_counter = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            stall_counter = 0
        else:
            stall_counter += 1

        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            stall_counter = 0

        if stall_counter >= restart_interval and rng.random() < restart_prob:
            scale = pert_scale * (1.0 + 0.5 * rng.random())
            perturbation = rng.normal(0, scale, size=n)
            a_perturbed = a + perturbation
            a_perturbed = np.maximum(a_perturbed, 0)
            sp = a_perturbed.sum()
            if sp > 1e-15:
                a_perturbed = a_perturbed / sp
                a = a_perturbed
                y = a_perturbed.copy()
            stall_counter = 0

        if (it + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 20:
                print(f"    NL-restart iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} stall={stall_counter}")
                last_print = elapsed

    return best_a, best_score

def lse_objective_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_func(z_val, T_val, n):
    obj, _ = lse_objective_grad(z_val, T_val, n)
    return obj


def lse_jac(z_val, T_val, n):
    _, grad = lse_objective_grad(z_val, T_val, n)
    return grad


def lbgfs_logsumexp(a0, time_limit, T_start=100.0, T_end=0.005, n_stages=25):

    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_sc = score_normalized(a0)

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        try:
            res = minimize(
                fun=lse_func, x0=z, jac=lse_jac,
                args=(float(stage_T), n), method='L-BFGS-B',
                options={'maxiter': 80, 'ftol': 1e-25,
                         'gtol': 1e-18, 'maxfun': 20000,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()
            if (si + 1) % 5 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LSE stage={si+1}/{n_stages} T={stage_T:.4f} ({elapsed:.0f}s): score={best_sc:.8f}")
        except Exception:
            pass

    return best_a


def subgradient_polish(a0, time_limit, n_steps=300000, eta_start=0.3, eta_end=0.003):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        active = np.where(b >= max_b * 0.85)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        S = np.dot(g, a)
        grad_z = a * (g - S)

        eta = eta_start * (eta_end / eta_start) ** (step / max(1, n_steps))
        z_new = z - eta * grad_z
        a_new = softmax_from_z(z_new)

        z = z_new
        a = a_new
        sc_new = score_normalized(a)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a.copy()

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

            a_try, _ = nesterov_lp_restarts(init_try, t_try, solver=solver,
                                             restart_interval=300, restart_prob=0.3, pert_scale=0.12)
            try_update(a_try, f"len_{n_try}")
    print(f"  Phase 4 done: best={best_score:.8f}")

    # === Phase 5: LSE fine polish (10% of budget) ===
    print(f"\n=== Phase 5: Fine LSE polish ===")
    remaining = _global_deadline - time.monotonic()
    t5 = min(0.10 * (budget_s - 25), remaining * 0.7)
    if t5 > 5:
        a_lse2 = lbgfs_logsumexp(best_seq, t5, T_start=10.0, T_end=0.002, n_stages=15)
        try_update(a_lse2, "phase5_fine_lse")
    print(f"  Phase 5 done: best={best_score:.8f}")

    # === Phase 6: Perturbation search + short optimize (5% of budget) ===
    print(f"\n=== Phase 6: Perturbation search ===")
    remaining = _global_deadline - time.monotonic()
    t6 = min(0.05 * (budget_s - 25), remaining * 0.7)
    if t6 > 10:
        a_pert, _ = perturbation_search(best_seq, t6, n_perturbations=30,
                                         pert_scale=0.08, solver=solver)
        try_update(a_pert, "phase6_perturb")
    print(f"  Phase 6 done: best={best_score:.8f}")

    # === Phase 7: Final Nesterov + LSE combo from best ===
    print(f"\n=== Phase 7: Final combo ===")
    remaining = _global_deadline - time.monotonic()
    if remaining > 15:
        t7a = min(remaining * 0.5, 60.0)
        t7b = min(remaining * 0.4, 40.0)
        if t7a > 5:
            a_f1, _ = nesterov_lp_clean(best_seq, t7a, solver=solver)
            try_update(a_f1, "phase7a_nesterov")
        remaining = _global_deadline - time.monotonic()
        if t7b > 5 and remaining > 5:
            a_f2 = lbgfs_logsumexp(best_seq, t7b, T_start=5.0, T_end=0.001, n_stages=10)
            try_update(a_f2, "phase7b_lse")
    print(f"  Phase 7 done: best={best_score:.8f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")

    return [float(x) for x in best_seq]
