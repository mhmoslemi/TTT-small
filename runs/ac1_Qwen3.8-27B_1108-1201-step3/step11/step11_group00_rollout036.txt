def normalize_nonneg(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


class LPBestResponse:
    def __init__(self, n):
        self.n = n
        # Pre-allocate
        self.A_ub = np.zeros((2*n - 1, n + 1))
        self.A_ub[:, -1] = -1.0
        self.b_ub = np.zeros(2 * n - 1)
        self.A_eq = np.zeros((1, n + 1))
        self.A_eq[0, :n] = 1.0
        self.b_eq = np.array([1.0])
        self.c = np.zeros(n + 1)
        self.c[-1] = 1.0
        self.bounds = [(0, None)] * (n + 1)
    
    def solve(self, y):
        n = self.n
        # Update A_ub in place
        for j in range(n):
            self.A_ub[j:j+n, j] = y
        res = linprog(self.c, A_ub=self.A_ub, b_ub=self.b_ub,
                      A_eq=self.A_eq, b_eq=self.b_eq,
                      bounds=self.bounds, method='highs',
                      options={'presolve': True})

            return None, None

        b_val = np.maximum(x[:n], 0)
        s = b_val.sum()
        if s < 1e-15:

def nesterov_lp(a0, time_limit, verbose=True):
    """Nesterov-accelerated LP best-response minimization."""
    n = len(a0)
    a = normalize_nonneg(a0)
    lp_solver = LPBestResponse(n)
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    
    # Nesterov state
    y = a.copy()  # extrapolation point
    a_prev = a.copy()
    
    # Running average
    avg_a = np.zeros(n)
    n_avg = 0
    
    t0 = time.monotonic()
    iteration = 0
    lp_time_total = 0.0
    stall_count = 0
    last_best_iter = 0
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        
        # Solve LP
        t_lp_start = time.monotonic()
        b_star, M_val = lp_solver.solve(y)
        lp_time_total += time.monotonic() - t_lp_start
        
        if b_star is None:
            continue
        
        # Nesterov update
        t_k = float(iteration)
        alpha_t = 2.0 / (t_k + 2.0)
        
        # Update y
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / s_y
        
        # Update a with Nesterov momentum
        if iteration == 1:
            a_new = b_star.copy()
        else:
            momentum = (t_k - 1.0) / (t_k + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_a
        
        y = y_new
        a_prev = a
        a = a_new
        
        # Running average (exponentially weighted)
        avg_a = 0.95 * avg_a + 0.05 * a if n_avg > 0 else a.copy()
        n_avg += 1
        
        # Check scores
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_best_iter = iteration
            stall_count = 0
        
        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_best_iter = iteration
            stall_count = 0
        
        # Check running average periodically

            avg_check = normalize_nonneg(avg_a)
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
                last_best_iter = iteration
                stall_count = 0
        
        if n_avg % 20 == 0:

        if verbose and iteration % 500 == 0:
            elapsed = time.monotonic() - t0
            avg_lp_time = lp_time_total / iteration * 1000
            print(f"    NL t={iteration} ({elapsed:.0f}s): best={best_score:.6f}, "

        if stall_count > 5000:

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length using cubic interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    
    # Use cubic spline for smoother interpolation
    from scipy.interpolate import CubicSpline
    try:
        cs = CubicSpline(old_x, f_norm, bc_type='natural')
        f_interp = cs(new_x)
    except Exception:
        f_interp = np.interp(new_x, old_x, f_norm)
    
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def peak_shaving(a0, time_limit, seed=42):
    """
    Multi-scale peak shaving: identify top autocorrelation peaks and
    systematically move mass to reduce them.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    total_moves = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        
        # Find top peaks
        n_peaks = min(12, len(b))
        top_k = np.argsort(b)[::-1][:n_peaks]
        
        improved = False
        
        for k in top_k:
            # Find contributing pairs for this lag
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            
            # Find the highest-contributing pair
            src_local = np.argmax(contrib)
            src = int(idx[src_local])
            dst_in_pair = int(k - src)
            
            if a[src] < 1e-12:
                continue
            
            # Find target: position that doesn't contribute to any top peak
            peak_positions = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_positions.update(range(js2, je2 + 1))
            
            # Candidates: positions NOT in peak areas, or positions with very low value
            if len(peak_positions) < n:
                candidates = [j for j in range(n) if j not in peak_positions]
            else:
                candidates = [j for j in range(n) if j != src]
            
            if not candidates:
                candidates = list(range(n))
            
            # Pick from lowest-value candidates
            candidates_sorted = sorted(candidates, key=lambda j: a[j])
            dst_pool = candidates_sorted[:max(1, len(candidates_sorted) // 4)]
            
            # Try multiple fractions
            fracs = [0.001, 0.003, 0.007, 0.015, 0.03, 0.05, 0.08]
            for frac in fracs:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                trial = a.copy()
                trial[src] -= amt
                # Try multiple destinations
                for dst in dst_pool[min(5, len(dst_pool))]:
                    trial2 = trial.copy()
                    trial2[dst] += amt
                    if trial2[src] < -1e-15:
                        continue
                    trial2 = np.maximum(trial2, 0)
                    ss = trial2.sum()
                    if ss < 1e-15:
                        continue
                    trial2 = trial2 / ss
                    score = 2.0 * n * fast_autocorr(trial2).max()
                    if score < best_score - 1e-14:
                        a = trial2
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        total_moves += 1
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Random perturbation to escape local minima
            if stall < 500:
                n_random = rng.integers(1, 5)
                for _ in range(n_random):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i == j or a[i] < 1e-12:
                        continue
                    frac = float(rng.uniform(0.001, 0.02))
                    amt = a[i] * frac
                    trial = a.copy()
                    trial[i] -= amt
                    trial[j] += amt
                    trial = np.maximum(trial, 0)
                    ss = trial.sum()
                    if ss > 1e-15:
                        trial = trial / ss
                        score = 2.0 * n * fast_autocorr(trial).max()
                        if score < best_score - 1e-14:
                            a = trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            total_moves += 1
                            break
            
            if not improved:
                stall += 1
                if stall > 1000:
                    break
        
        # Progress report
        if total_moves % 200 == 0 and total_moves > 0:
            elapsed = time.monotonic() - t0
            print(f"    PS ({elapsed:.0f}s): score={best_score:.6f}, moves={total_moves}")
    
    return best_a, best_score


def gradient_polish(a0, time_limit):

    Gradient-based polishing using subgradient of the max autocorrelation.
    Uses the top few peaks for a smoother gradient signal.

                # Plus a[i] from j=k-i term (if k-i != i and 0<=k-i<n)
                # Simplest: just compute numerically or use the fact that

        # Actually, let me use the FFT cross-correlation approach for the gradient
        # grad_a[i] = 2 * sum_k weights[k] * a[k-i] (where valid)
        # This is 2 * cross_correlation(a, weights)
        # = 2 * convolve(a_reversed, weights) evaluated at appropriate indices
        
        # Simpler: just compute directly for the top peaks
        grad = np.zeros(n)
        for k in top_indices:
            wk = weights[k]
            if wk < 1e-10:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for i in range(js, je + 1):
                # Contribution to grad[i] from peak k:
                # ∂b[k]/∂a[i] = a[k-i] (term where j=i) + a[i] (term where j=k-i, if k-i != i)
                # For simplicity and symmetry, use: 2*a[k-i] (approximately)
                # More precisely: if k-i == i, it's 2*a[i], else a[k-i] + a[i]
                # But since we're just doing subgradient descent, approximate is fine
                if 0 <= k - i < n:
                    if k - i == i:

        if g_norm < 1e-15:

        a_new = np.maximum(a_new, 0)

import numpy as np

def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence using the provided evaluate_sequence."""

        return float(sc)

def normalize_nonneg(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def softmax_fn(z):
    """Numerically stable softmax."""
    z = np.asarray(z, dtype=np.float64)
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:

class LPBestResponse:
    """Efficient LP solver for best-response, reusing pre-allocated matrices."""

    def __init__(self, n):
        self.n = n
        self.A_ub = np.zeros((2 * n - 1, n + 1))
        self.A_ub[:, -1] = -1.0
        self.b_ub = np.zeros(2 * n - 1)
        self.A_eq = np.zeros((1, n + 1))
        self.A_eq[0, :n] = 1.0
        self.b_eq = np.array([1.0])
        self.c = np.zeros(n + 1)
        self.c[-1] = 1.0
        self.bounds = [(0, None)] * (n + 1)

    def solve(self, y):
        """Solve: min M s.t. (T_y @ b) <= M, sum(b)=1, b>=0, M>=0."""
        n = self.n
        for j in range(n):
            self.A_ub[j:j + n, j] = y
        res = linprog(self.c, A_ub=self.A_ub, b_ub=self.b_ub,
                      A_eq=self.A_eq, b_eq=self.b_eq,
                      bounds=self.bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.status != 0 or res.x is None:
            return None, None
        x = res.x
        b_val = np.maximum(x[:n], 0)
        s = b_val.sum()
        if s < 1e-15:
            return None, None
        b_val = b_val / s
        M_val = max(0.0, float(x[-1]))
        return b_val, M_val

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length using cubic spline."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    try:
        cs = CubicSpline(old_x, f_norm, bc_type='natural')
        f_interp = cs(new_x)
    except Exception:
        f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def gradient_polish(a0, time_limit):
    """
    FFT-based subgradient polishing for the max autocorrelation.
    Uses exponential weighting on top peaks for smooth subgradient.
    """
    n = len(a0)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    learning_rate = 1.0
    top_k = 10
    stall = 0
    convergence_factor = 0.995

    # Pre-allocate FFT sizes
    conv_len = 3 * n - 2
    fft_size = 1 << (conv_len - 1).bit_length()
    if conv_len > fft_size:
        fft_size *= 2

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)

        # Select top peaks
        top_indices = np.argsort(b)[::-1][:top_k]
        top_vals = b[top_indices]

        # Exponential weights emphasizing highest peaks
        beta = 3.0
        w_tmp = np.exp(beta * top_vals - beta * top_vals.max())
        w_tmp = w_tmp / w_tmp.sum()

        # Build weight vector
        weights = np.zeros(2 * n - 1)
        for i in range(len(top_indices)):
            weights[int(top_indices[i])] = w_tmp[i]

        # Compute gradient via FFT cross-correlation
        # grad[i] = 2 * sum_k weights[k] * a[k-i] (where valid)
        A_rev = rfft(a[::-1], fft_size)
        W = rfft(weights, fft_size)
        conv = irfft(A_rev * W, fft_size)
        grad = 2.0 * conv[n - 1: n - 1 + n]

        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break

        # Projected gradient step with adaptive step size
        lr = learning_rate / g_norm
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss

        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            stall = 0
            learning_rate *= convergence_factor
        else:
            stall += 1
            learning_rate *= 0.98
            if stall > 300 or learning_rate < 1e-6:
                break

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=42):
    """
    Multi-scale peak shaving: identify top autocorrelation peaks and
    systematically move mass from high-contributing positions to low-value
    positions that don't feed into any top peak.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    total_moves = 0

    fracs = [0.001, 0.002, 0.005, 0.01, 0.02, 0.04, 0.07, 0.1]

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)

        # Find top peaks
        n_peaks = min(10, 2 * n - 1)
        top_k = np.argsort(b)[::-1][:n_peaks]

        improved = False

        for k in top_k:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]

            # Find highest contributor
            src_local = int(np.argmax(contrib))
            src = int(idx[src_local])
            if a[src] < 1e-12:
                continue

            # Build set of positions that feed into top peaks
            peak_positions = set()
            for k2 in top_k:
                k2 = int(k2)
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_positions.update(range(js2, je2 + 1))

            # Candidates: positions NOT in any peak area
            if len(peak_positions) < n:
                candidates = [j for j in range(n) if j not in peak_positions]
            else:
                candidates = [j for j in range(n) if j != src]

            if not candidates:
                candidates = [j for j in range(n) if j != src]

            # Sort by value (prefer low-value targets)
            candidates.sort(key=lambda j: a[j])
            dst_pool = candidates[:max(2, len(candidates) // 5)]
            dst_pool = dst_pool[:6]  # Limit to 6 candidates for speed

            for frac in fracs:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                if a[src] - amt < -1e-15:
                    continue
                for dst in dst_pool:
                    trial = a.copy()
                    trial[src] -= amt
                    trial[dst] += amt
                    trial = np.maximum(trial, 0)
                    ss = trial.sum()
                    if ss < 1e-15:
                        continue
                    trial = trial / ss
                    score = 2.0 * n * fast_autocorr(trial).max()
                    if score < best_score - 1e-14:
                        a = trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        total_moves += 1
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Random perturbation for exploration
            if stall < 800:
                n_attempts = int(rng.integers(1, 4))
                for _ in range(n_attempts):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i == j or a[i] < 1e-12:
                        continue
                    frac = float(rng.uniform(0.001, 0.03))
                    amt = a[i] * frac
                    trial = a.copy()
                    trial[i] -= amt
                    trial[j] += amt
                    trial = np.maximum(trial, 0)
                    ss = trial.sum()
                    if ss > 1e-15:
                        trial = trial / ss
                        score = 2.0 * n * fast_autocorr(trial).max()
                        if score < best_score - 1e-14:
                            a = trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            total_moves += 1
                            break

            if not improved:
                stall += 1
                if stall > 1500:
                    break

        if total_moves > 0 and total_moves % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PS ({elapsed:.0f}s): score={best_score:.6f}, moves={total_moves}")

    return best_a, best_score


def structured_init(n, seed=0):
    """Generate a diverse structured initialization."""
    rng = np.random.default_rng(seed)
    choice = int(rng.integers(0, 5))
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * (x - 0.5) ** 2)
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.exp(-8 * (x - 0.3) ** 2) + np.exp(-8 * (x - 0.7) ** 2)
    elif choice == 2:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 3:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.7 + 0.2 * np.cos(3 * x) + 0.1 * np.cos(7 * x)
        a = np.maximum(a, 0)
    else:
        a = np.ones(n) + 0.2 * rng.standard_normal(n)
        a = np.maximum(a, 0.01)
    return normalize_nonneg(a)


def try_update_state(state, a, label, t_start):
    """
    Update the best sequence state dict.
    state: dict with keys 'seq' (array or None) and 'score' (float)
    """
    if a is None:
        return
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < state['score']:
        state['score'] = sc
        state['seq'] = a.copy()
        elapsed = time.monotonic() - t_start
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Uses Nesterov-accelerated LP best-response as the
    primary optimizer, with multi-length exploration, gradient polishing,
    and peak shaving for final refinement.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    print(f"Time budget: {time_budget:.0f}s")

    # Load parent sequence
    state = {'seq': None, 'score': float('inf')}
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            state['seq'] = parent.copy()
            state['score'] = p_score
    except Exception as e:
        print(f"Warning: could not load parent: {e}")

    if state['seq'] is None:
        n_default = 1600
        state['seq'] = structured_init(n_default, 0)
        state['score'] = evaluate_safe(state['seq'])
        print(f"Default init: n={n_default}, score={state['score']:.6f}")

    # === PHASE 1: LP Nesterov at parent length ===
    n1 = len(state['seq'])
    remaining = time_budget - (time.monotonic() - t_start)
    t1 = min(remaining * 0.30, 500.0)
    print(f"\n=== Phase 1: LP Nesterov at n={n1} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = nesterov_lp(state['seq'], t1)
        try_update_state(state, a_out, f"lp_n{n1}", t_start)

    # === PHASE 2: LP at 1.25x length ===
    n2 = int(round(n1 * 1.25))
    remaining = time_budget - (time.monotonic() - t_start)
    t2 = min(remaining * 0.22, 380.0)
    print(f"\n=== Phase 2: LP Nesterov at n={n2} (budget={t2:.0f}s) ===")
    if t2 > 30:
        init2 = interpolate_sequence(state['seq'], n2)
        a_out, _ = nesterov_lp(init2, t2)
        try_update_state(state, a_out, f"lp_n{n2}", t_start)

    # === PHASE 3: LP at 1.5x length ===
    n3 = int(round(n1 * 1.5))
    remaining = time_budget - (time.monotonic() - t_start)
    t3 = min(remaining * 0.18, 300.0)
    print(f"\n=== Phase 3: LP Nesterov at n={n3} (budget={t3:.0f}s) ===")
    if t3 > 30:
        init3 = interpolate_sequence(state['seq'], n3)
        a_out, _ = nesterov_lp(init3, t3)
        try_update_state(state, a_out, f"lp_n{n3}", t_start)

    # === PHASE 4: LP refinement at best length ===
    best_n = len(state['seq'])
    remaining = time_budget - (time.monotonic() - t_start)
    t4 = min(remaining * 0.20, 350.0)
    print(f"\n=== Phase 4: LP refinement at n={best_n} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = nesterov_lp(state['seq'], t4)
        try_update_state(state, a_out, f"lp_refine_n{best_n}", t_start)

    # === PHASE 5: FFT gradient polishing ===
    remaining = time_budget - (time.monotonic() - t_start)
    t5 = min(remaining * 0.12, 180.0)
    print(f"\n=== Phase 5: Gradient polish at n={len(state['seq'])} (budget={t5:.0f}s) ===")
    if t5 > 10:
        a_out, _ = gradient_polish(state['seq'], t5)
        try_update_state(state, a_out, "grad_polish", t_start)

    # === PHASE 6: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_start)
    t6 = min(remaining * 0.25, 300.0)
    print(f"\n=== Phase 6: Peak shaving at n={len(state['seq'])} (budget={t6:.0f}s) ===")
    if t6 > 10:
        a_out, _ = peak_shaving(state['seq'], t6, seed=123)
        try_update_state(state, a_out, "peak_shave_1", t_start)

    # === PHASE 7: Final gradient + peak shave combo ===
    remaining = time_budget - (time.monotonic() - t_start)
    if remaining > 30:
        t7_grad = min(remaining * 0.3, 100.0)
        t7_peak = min(remaining * 0.6, 200.0)
        print(f"\n=== Phase 7: Final polish (grad={t7_grad:.0f}s, peak={t7_peak:.0f}s) ===")
        if t7_grad > 5:
            a_out, _ = gradient_polish(state['seq'], t7_grad)
            try_update_state(state, a_out, "final_grad", t_start)
        remaining2 = time_budget - (time.monotonic() - t_start)
        if remaining2 > 10:
            t7p = min(remaining2, 150.0)
            a_out, _ = peak_shaving(state['seq'], t7p, seed=789)
            try_update_state(state, a_out, "final_peak", t_start)

    # Final verification
    if state['seq'] is not None:
        fs = evaluate_safe(state['seq'])
        if np.isfinite(fs):
            state['score'] = fs
        print(f"\nFINAL SCORE: {state['score']:.6f} (n={len(state['seq'])})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in state['seq']]
