"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated LP best-response with adaptive restarts (primary method)
2. FFT-based LogSumExp gradient optimization with adaptive annealing
3. Peak-shaving coordinate descent for local refinement

Key improvements: restart heuristics for Nesterov, longer LP phases,
exploration of longer sequences, and better time allocation.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def lse_obj_grad_z(z, T):
    """Objective and gradient for z-parameterization (a = softmax(z))."""
    a = softmax_fn(z)
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    obj_val = (max_Tb + np.log(exp_Tb.sum())) / T

    w = exp_Tb / exp_Tb.sum()

    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    if conv_len > fft_size:
        fft_size *= 2
    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    grad_a = 2.0 * conv[n - 1:n - 1 + n]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj_val, grad_z


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    for solver_name in [cp.SCS, cp.ECOS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            t0 = time.monotonic()
            prob.solve(solver=solver_name)
            elapsed = time.monotonic() - t0
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  Solver {solver_name}: {elapsed:.3f}s for test problem")
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a_fixed):
    """Build the (2n-1) x n convolution matrix T where (T@b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0. Returns (b, M)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.ECOS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """
    Nesterov-accelerated LP best-response with adaptive restarts.
    Tracks running average which often outperforms last iterate.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    # Nesterov state
    t_n = 0
    y = a.copy()
    a_nest = a.copy()

    # Running average (exponential and uniform)
    avg_a = np.zeros(n)
    n_avg = 0
    exp_avg_a = a.copy()
    exp_alpha = 0.05

    # Restart tracking
    consecutive_no_improve = 0
    last_best_score = best_score
    M_history = []
    total_iters = 0

    for epoch in range(200):
        if time.monotonic() - t0 > time_limit:
            break

        # Reset Nesterov counter for this "round"
        # We keep y as the extrapolation point
        t_n = 0
        rounds_per_epoch = 30

        for _ in range(rounds_per_epoch):
            if time.monotonic() - t0 > time_limit:
                break

            b_star, M_val = solve_lp_best_response(y, solver)
            if b_star is None:
                break

            total_iters += 1
            t_n += 1

            alpha_t = 2.0 / (t_n + 2.0)
            y_new = (1 - alpha_t) * y + alpha_t * b_star
            y_new = np.maximum(y_new, 0)
            s_y = y_new.sum()
            if s_y > 1e-15:
                y_new = y_new / s_y

            if t_n == 1:
                a_new = b_star.copy()
            else:
                momentum = (t_n - 1.0) / (t_n + 2.0)
                a_new = y_new + momentum * (y_new - y)
                a_new = np.maximum(a_new, 0)
                s_a = a_new.sum()
                if s_a < 1e-15:
                    a_new = y_new.copy()
                else:
                    a_new = a_new / s_a

            y = y_new
            a_nest = a_new

            # Check Nesterov iterate
            score = 2.0 * n * fast_autocorr(a_nest).max()
            if score < best_score:
                best_score = score
                best_a = a_nest.copy()
                consecutive_no_improve = 0
            else:
                consecutive_no_improve += 1

            # Exponential running average
            exp_avg_a = (1 - exp_alpha) * exp_avg_a + exp_alpha * a_nest
            exp_avg_a = np.maximum(exp_avg_a, 0)
            s_exp = exp_avg_a.sum()

                exp_avg_a = exp_avg_a / s_exp

                    best_score = score_exp
                    best_a = exp_avg_a.copy()
                    consecutive_no_improve = 0

            n_avg += 1
            if n_avg % 20 == 0:
                avg_check = avg_a / n_avg
                score_avg = 2.0 * n * fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    consecutive_no_improve = 0

            # Track M values

                M_history.pop(0)

        # End of epoch: check for restart
        if len(M_history) >= 20:

            M_range = max(M_recent) - min(M_recent)

        if consecutive_no_improve > 50:
            # Restart from best
            y = best_a.copy()
            a_nest = best_a.copy()
            exp_avg_a = best_a.copy()
            M_history = []
            consecutive_no_improve = 0
            if verbose:
                elapsed = time.monotonic() - t0

        if verbose and total_iters % 100 == 0:

    return best_a, best_score, total_iters


def logsumexp_optimize(a0, time_limit, T_start=200.0, T_end=0.05, n_stages=20):
    """LogSumExp-smoothed optimization with L-BFGS and multi-stage annealing."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a_init = a_init / a_init.sum()
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        iters_per_stage = max(20, min(100, int(50 + 30 * (si / n_stages))))

        def func(z_in):
            val, _ = lse_obj_grad_z(z_in, stage_T)
            return val

        def jac(z_in):
            _, g = lse_obj_grad_z(z_in, stage_T)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': iters_per_stage * 300,
                         'disp': False}
            )
            z = res.x
            a = softmax_fn(z)
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
        except Exception:
            pass

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=123):
    """Coordinate-descent peak shaving."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:6]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            src_local = np.argmax(contrib)
            src = idx[src_local]

            peak_areas = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_areas.update(range(js2, je2 + 1))

            candidates = [j for j in range(n) if j not in peak_areas]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            candidates.sort(key=lambda j: a[j])
            dst_pool = candidates[:max(1, len(candidates) // 3)]
            dst = dst_pool[rng.integers(len(dst_pool))]

            for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > 150:
                break

    return best_a, best_score


def interpolate_pchip(f, new_n):
    """PCHIP-style interpolation (monotone cubic) to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)

    # Use np.interp (linear) as a safe fallback, then smooth
    f_interp = np.interp(new_x, old_x, f_norm)

    # Apply a light smoothing to avoid interpolation artifacts
    if new_n > 10:
        kernel_size = min(5, new_n // 2 * 2 + 1)
        kernel = np.ones(kernel_size) / kernel_size
        f_smooth = np.convolve(f_interp, kernel, mode='same')
        # Blend original and smoothed
        f_interp = 0.7 * f_interp + 0.3 * f_smooth

    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 1.5
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.6 + 0.3 * np.cos(3 * x) + 0.1 * np.cos(7 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * (1 + 0.2 * np.sin(4 * np.pi * x))
    else:
        a = np.ones(n)
        a = a + 0.3 * np.sin(2 * np.pi * np.arange(n) / n * 3)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

        print(f"\n=== Phase 6: Peak shaving (time={t6:.0f}s) ===")
        a_out, _ = peak_shaving(best_seq, t6, seed=456)
        try_update(a_out, "peak_shave_1")

    # Phase 7: Final LSE polish with very low temperature
    remaining = time_budget - (time.monotonic() - t_global_start)
    t7 = min(remaining * 0.15, 120.0)
    if t7 > 10:
        print(f"\n=== Phase 7: Final LSE polish (time={t7:.0f}s) ===")
        a_out, _ = logsumexp_optimize(best_seq, t7, T_start=10.0, T_end=0.01, n_stages=15)
        try_update(a_out, "final_lse")

    # Phase 8: Another peak shave with different seed
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 10:
        t8 = min(remaining * 0.4, 80.0)
        if t8 > 5:
            print(f"\n=== Phase 8: Second peak shave (time={t8:.0f}s) ===")
            a_out, _ = peak_shaving(best_seq, t8, seed=789)
            try_update(a_out, "peak_shave_2")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
