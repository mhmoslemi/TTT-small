**Approach: Frank-Wolfe (Conditional Gradient) Method**
The Frank-Wolfe method for convex optimization over the simplex:
- At each iteration, find the vertex of the simplex that minimizes the linearization of the objective
- The linearization at a is: `<grad f(a), v> = <grad f(a), e_i>` where e_i is a basis vector
- So we just pick the coordinate with the smallest gradient component!
- Update: a_new = (1-t) * a + t * e_i

This is extremely simple and doesn't require any LP solves! The step size t can be adaptive.

For our problem, the subgradient of `f(a) = max_k (a^T Q_k a)` at a is `2 * Q_{k*} * a` where k* is an argmax. So the Frank-Wolfe step is:
- Compute g = 2 * Q_{k*} * a (the subgradient)
- Find i* = argmin_j g[j] (the vertex with minimum linearized objective)
- Update: a = (1-t) * a + t * e_{i*}

This is incredibly fast! Each iteration is O(n) (compute the subgradient and find the min). We can run millions of iterations in a second.

The convergence rate of Frank-Wolfe is O(1/t), which is slower than Nesterov's O(1/sqrt(t)) or the LP alternating, but the per-iteration cost is much lower.

- y = (1 - alpha) * y + alpha * a
- Find s = argmin_{v in simplex} <grad f(y), v> = e_{argmin grad f(y)}
- a_new = (1 - gamma) * a + gamma * s
- where gamma is chosen to minimize f (line search or fixed formula)

With L = 600, the optimal step size for gradient descent is 1/L = 1/600. For Nesterov, the step size is still `2/(t+2)` (it doesn't depend on L directly; the function value uses `f(y_t) + (1/L) * ||grad f(y_t)||^2`).

The mirror descent with exact oracle (multiplicative weights) is:

```
a_{t+1} = argmin_{a in simplex} <g_t, a> + (1/eta) * D_KL(a, a_t)
```
where g_t is the subgradient of f at a_t, and D_KL is the KL divergence.

The solution is: `a_{t+1}[j] = a_t[j] * exp(-eta * g_t[j]) / sum_k a_t[k] * exp(-eta * g_t[k])`

This is exactly the multiplicative weights update! And the subgradient g_t = 2 * Q_{k*} * a_t where k* is an argmax.

The convergence rate of mirror descent is O(sqrt(L^2 * D / t)) where D is the diameter of the domain. For the simplex with KL divergence, D = log(n). So the rate is O(sqrt(L^2 * log(n) / t)).

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast multiplicative weights warm-up (no solver, very fast)
2. LP alternating with Nesterov acceleration (main workhorse)
3. Low-temperature LogSumExp L-BFGS for final precision
4. Multi-n exploration (n-2 to n+2)
5. Multiple diverse starting points to avoid local minima
The problem is non-convex (max of indefinite quadratic forms), so we use
heuristic methods with multiple restarts and local refinement.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def get_lp_solver():
    """Find a working LP solver, benchmarking small problems."""
    n_test = 50
    a_test = np.random.default_rng(0).random(n_test)
    a_test = a_test / a_test.sum()
    T_test = build_conv_matrix(a_test)

    solver_times = {}
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS, cp.GUROBI, cp.MOSEK]:
        try:
            b_var = cp.Variable(n_test, nonneg=True)
            M_var = cp.Variable()
            constraints = [T_test @ b_var <= M_var, cp.sum(b_var) == 1]
            prob = cp.Problem(cp.Minimize(M_var), constraints)
            t0 = time.monotonic()
            prob.solve(solver=solver_name)
            t1 = time.monotonic()
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solver_times[solver_name] = t1 - t0
        except Exception:
            continue

    if not solver_times:
        return cp.ECOS
    best = min(solver_times, key=solver_times.get)
    print(f"  LP solver benchmark: { {k: f'{v*1000:.1f}ms' for k, v in solver_times.items()} }")
    return best


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP, cp.GUROBI, cp.MOSEK]
    solved = False
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solved = True
                break
        except Exception:
            continue

    if not solved:
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s_sum = b_val.sum()
    if s_sum < 1e-15:
        return None, None
    b_val = b_val / s_sum
    return b_val, float(max(0, M_var.value))


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters):
    """Nesterov dual averaging LP alternating with adaptive restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(42)
    lp_times = []
    best_iter = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver)
        lp_times.append(time.monotonic() - lp_t0)

        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            best_iter = it + 1
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            best_iter = it + 1

        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 6:
                # Adaptive restart: inject structured noise
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 8)
                noise = 0.05 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                sy2 = y.sum()
                if sy2 > 1e-15:
                    y = y / sy2
                a = y.copy()
                stall_count = 0

            if (it + 1) % 500 == 0:
                elapsed = time.monotonic() - t0
                avg_lp = np.mean(lp_times[-50:]) * 1000 if len(lp_times) >= 50 else 0
                print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.10f} avg_lp={avg_lp:.1f}ms stall={stall_count}")

        if len(lp_times) > 100 and np.mean(lp_times[-100:]) > 5.0:
            print(f"    LP too slow ({np.mean(lp_times[-100:]):.2f}s/iter), stopping at iter {it+1}")
            break

    return best_a, best_score


def multiplicative_weights_refine(a0, time_limit, n_steps=50000):
    """Multiplicative weights / mirror descent for fast warm-up."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 0.05
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Use top active peaks (wider band for multiplicative weights)
        threshold = max_b * (1.0 - 5e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        log_scale = -eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 50:
            eta *= 0.8
            no_improve = 0
            if eta < 0.002:
                break

    return best_a, best_score


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def lse_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S_val = np.dot(grad_a, a)
    grad_z = a * (grad_a - S_val)
    return obj, 2.0 * n * grad_z


def lse_func_wrap(z, n, T_temp, deadline):
    """Top-level LSE objective for L-BFGS."""
    if time.monotonic() > deadline:
        return 1e15
    obj, _ = lse_obj_grad(z, n, T_temp)
    return obj


def lse_jac_wrap(z, n, T_temp, deadline):
    """Top-level LSE gradient for L-BFGS."""
    if time.monotonic() > deadline:
        return np.zeros_like(z)
    _, g = lse_obj_grad(z, n, T_temp)
    return g


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    n_stages = len(T_stages)
    iters_per_stage = max(20, maxiter_per_stage // n_stages)

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        try:
            res = minimize(
                fun=lse_func_wrap, x0=z, jac=lse_jac_wrap,
                args=(n, stage_T, deadline),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-25,
                         'gtol': 1e-18, 'maxfun': iters_per_stage * 500, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def generate_diverse_inits(n, rng):
    """Generate diverse initialization vectors of length n."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine (smooth bell)
    x = np.linspace(0, 1, n)
    c1 = np.cos(np.pi * x / 2) ** 2
    c1 = np.maximum(c1, 0)
    inits.append(c1 / c1.sum())

    # Symmetric bell (centered)
    c2 = np.exp(-5 * (x - 0.5) ** 2)
    inits.append(c2 / c2.sum())

    # Edge-peaked
    c3 = np.exp(-10 * x * (1 - x) * 4)
    c3 = np.maximum(c3, 1e-10)
    inits.append(c3 / c3.sum())

    # Random smooth
    freq = rng.integers(2, 6)
    c4 = 1.0 + 0.3 * np.cos(2 * np.pi * freq * x + rng.uniform(0, 2 * np.pi))
    c4 = np.maximum(c4, 0.01)
    inits.append(c4 / c4.sum())

    # Slight asymmetric
    c5 = 1.0 + 0.2 * np.sin(np.pi * x + 0.5)
    c5 = np.maximum(c5, 0.01)
    inits.append(c5 / c5.sum())

    return inits

            print(f"  NEW BEST (final LP): {best_score:.10f}")

        rem = deadline - time.monotonic()
        if rem > 10:
            T_final = [0.005, 0.002, 0.001, 0.0005]
            a_lb_f = lbgfs_refine(best_seq, min(rem * 0.6, 50.0), T_final, maxiter_per_stage=3000)
            sc = evaluate_local(a_lb_f)
            if sc < best_score:
                best_score = sc
                best_seq = a_lb_f
                print(f"  NEW BEST (final L-BFGS): {best_score:.10f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
        print(f"Total time: {time.monotonic() - t_start:.1f}s")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
```
