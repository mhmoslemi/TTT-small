import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS, cp.GUROBI, cp.MOSEK]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver, timeout=10.0):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]
    for s in solvers_to_try:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                break
        except Exception:
            prob = cp.Problem(cp.Minimize(M_var), constraints)
            b_var = cp.Variable(n, nonneg=True)
            M_var = cp.Variable()
            constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
            prob = cp.Problem(cp.Minimize(M_var), constraints)
            continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(max(0, M_var.value))


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def spectral_era(a, n_iters=100, blend=1.0, deadline=None):
    """
    Fienup Error Reduction Algorithm for flat spectrum.
    Projects alternatingly onto time-domain (non-neg, support) 
    and frequency-domain (flat magnitude) constraints.
    """
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        # Time-domain projection: non-negativity + normalization
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        # Frequency-domain projection: flatten magnitude
        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)

        mags = np.abs(F)
        non_dc_mags = mags[1:]
        # Use geometric mean as target (better for flatness)
        valid = non_dc_mags > 1e-15
        if valid.sum() > 0:
            target = np.exp(np.mean(np.log(non_dc_mags[valid])))
        else:
            target = np.mean(non_dc_mags)

        # Blend toward target
        phases = np.angle(F)
        F_new = F.copy()
        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend) * mags[k] + blend * target
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])

        a = np.fft.ifft(F_new).real[:n]

    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a


def scf_iteration(a0, time_limit, T_start=None, T_end=0.005, n_max_iters=5000,
                  damp=0.5, deadline=None):
    """
    Self-Consistent Field iteration exploiting Hankel structure.
    At each step: compute autocorrelation, form weighted Hankel matrix,
    solve linear system, project onto feasible set.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()

    if deadline is None:
        deadline = t0 + time_limit
    if T_start is None:
        T_start = fast_convolve(a).max() * 0.5

    for it in range(n_max_iters):
        if time.monotonic() > deadline:
            break

        # Adaptive temperature: anneal from T_start to T_end
        progress = min(1.0, it / max(1, n_max_iters * 0.8))
        T = T_start * (T_end / T_start) ** progress
        T = max(T, 1e-6)

        # Compute autocorrelation
        b = fast_convolve(a)
        max_b = b.max()

        # Compute softmax weights over autocorrelation
        b_shifted = b - max_b
        exp_b = np.exp(b_shifted / T)
        w = exp_b / exp_b.sum()

        # Build Hankel matrix: C[j,l] = w[j+l]
        # C_lambda is n x n with C[j,l] = w[j+l]
        indices = np.arange(n)
        C = w[np.ix_(indices + indices, indices + indices)]

        # Add small regularization for numerical stability
        C_reg = C + 1e-10 * np.eye(n)

        # Solve C_reg @ x = 1
        try:
            x = np.linalg.solve(C_reg, np.ones(n))
        except np.linalg.LinAlgError:
            x = np.linalg.lstsq(C_reg, np.ones(n), rcond=None)[0]

        # Project: clip to non-negative, normalize
        x = np.maximum(x, 0)
        sx = x.sum()
        if sx < 1e-15:
            continue
        x = x / sx

        # Damped update
        a_new = (1 - damp) * a + damp * x
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            continue
        a_new = a_new / sa

        score = evaluate_local(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            # Still update a for the iteration to progress
            a = a_new

        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SCF iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} T={T:.6f}")

    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=1000000):

    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)

    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(54321)
    lp_times = []

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver)

        if b_new is None:
            continue

        if len(lp_times) >= 20:
            avg_lp_time = np.mean(lp_times[-20:])
            if avg_lp_time > 5.0 and len(lp_times) > 50:
                print(f"    LP too slow ({avg_lp_time:.2f}s/iter), stopping at iter {it+1}")
                break

        # Nesterov dual averaging
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = evaluate_local(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = evaluate_local(y)
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Stall detection and restart
        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-12:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 5:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 10)
                noise = 0.02 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                a = a + 0.5 * noise
                a = np.maximum(a, 0)
                sa = a.sum()
                if sa > 1e-15:
                    a = a / sa
                stall_count = 0

            if (it + 1) % 2000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def peak_shave(a0, time_limit, n_steps=200000):
    """
    Directly target the maximum of the autocorrelation using multiplicative updates.
    Focus on the top few peaks to reduce them fastest.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return None, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()

    eta = 0.015
    scale = 1.0
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Find top peaks
        threshold = max_b * (1.0 - 2e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        # Limit to top 20 peaks for efficiency
        if len(active) > 20:
            top_idx = np.argsort(b[active])[-20:]
            active = active[top_idx]

        # Compute subgradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        eta_eff = eta * scale
        log_scale = -eta_eff * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = evaluate_local(a_new)
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 40:
            scale *= 0.7
            no_improve = 0
            if scale < 0.003:
                break

    return best_a, best_score


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage=5000, deadline=None):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    if deadline is None:
        deadline = t0 + time_limit
    n_stages = len(T_stages)

    for stage_i, stage_T in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 2:
            break

        stages_left = n_stages - stage_i
        iters = max(20, min(maxiter_per_stage, int(remaining * 30 / max(1, stages_left))))

        def _func(z_in):
            if time.monotonic() > deadline:
                return 1e15
            a_tmp = _softmax(z_in)
            b_tmp = fast_convolve(a_tmp)
            max_b_tmp = b_tmp.max()
            exp_terms = np.exp((b_tmp - max_b_tmp) / stage_T)
            lse = max_b_tmp + stage_T * np.log(exp_terms.sum())
            return 2.0 * n * lse

        def _jac(z_in):
            if time.monotonic() > deadline:
                return np.zeros_like(z_in)
            a_tmp = _softmax(z_in)
            b_tmp = fast_convolve(a_tmp)
            max_b_tmp = b_tmp.max()
            exp_terms = np.exp((b_tmp - max_b_tmp) / stage_T)
            w = exp_terms / exp_terms.sum()
            grad_a = np.zeros(n)
            for k in range(len(b_tmp)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += w[k] * a_tmp[k - idx]
            grad_a *= 2.0
            S = np.dot(grad_a, a_tmp)
            grad_z = a_tmp * (grad_a - S)
            return 2.0 * n * grad_z

        try:
            res = minimize(
                fun=_func, x0=z, jac=_jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': iters * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return _softmax(z)


def _softmax(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def structured_init(n, rng):
    """Create a structured initial sequence."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 5 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    else:
        a = 1.0 - np.abs(2 * x - 1)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses SCF iteration, Nesterov LP alternating, spectral ERA,
    peak shaving, and L-BFGS LogSumExp polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    rem = deadline - time.monotonic()
    print(f"Total budget: {rem:.0f}s, starting score: {best_score:.8f}")

    # === Phase 0: Spectral ERA initialization (30s) ===
    print(f"\n=== Phase 0: Spectral ERA Init ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_init = min(rem * 0.03, 30.0)
        a_era = spectral_era(best_seq, n_iters=100, blend=0.8,
                             deadline=time.monotonic() + t_init)
        local_update(a_era, "era")
        a_era2 = spectral_era(best_seq, n_iters=50, blend=1.0,
                              deadline=time.monotonic() + min(t_init, 15))
        local_update(a_era2, "era_full")

    # === Phase 1: SCF Iteration (120s) ===
    print(f"=== Phase 1: SCF Iteration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_scf = min(rem * 0.08, 120.0)
    if t_scf > 15:
        print(f"  Budget: {t_scf:.0f}s")
        a_scf, _ = scf_iteration(best_seq, t_scf, T_start=None, T_end=0.003,
                                 n_max_iters=3000, damp=0.4)
        local_update(a_scf, "scf_1")

    # === Phase 2: Nesterov LP (main workhorse, 400s) ===
    print(f"=== Phase 2: Nesterov LP n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_lp = min(rem * 0.25, 400.0)
    if t_lp > 30:
        print(f"  Budget: {t_lp:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_lp, solver, n_max_iters=500000)
        local_update(a_lp, "nesterov_lp_1")

    # === Phase 3: SCF fine-tune (80s) ===
    print(f"=== Phase 3: SCF Fine-tune (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_scf2 = min(rem * 0.05, 80.0)
    if t_scf2 > 10:
        a_scf2, _ = scf_iteration(best_seq, t_scf2, T_start=None, T_end=0.001,
                                  n_max_iters=2000, damp=0.3)
        local_update(a_scf2, "scf_fine")

    # === Phase 4: Nesterov LP round 2 (300s) ===
    print(f"=== Phase 4: Nesterov LP R2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_lp2 = min(rem * 0.18, 300.0)
    if t_lp2 > 20:
        print(f"  Budget: {t_lp2:.0f}s")
        a_lp2, _ = nesterov_lp_refine(best_seq, t_lp2, solver, n_max_iters=400000)
        local_update(a_lp2, "nesterov_lp_2")

    # === Phase 5: Peak Shaving (80s) ===
    print(f"=== Phase 5: Peak Shaving (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_peak = min(rem * 0.05, 80.0)
    if t_peak > 10:
        a_pk, _ = peak_shave(best_seq, t_peak, n_steps=100000)
        local_update(a_pk, "peak_shave")

    # === Phase 6: L-BFGS LogSumExp (200s) ===
    print(f"=== Phase 6: L-BFGS LogSumExp (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_lbfgs = min(rem * 0.12, 200.0)
    if t_lbfgs > 20:
        T_stages = [5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1,
                    0.05, 0.03, 0.02, 0.01, 0.007, 0.005, 0.003, 0.002]
        print(f"  T_stages: {len(T_stages)}, budget={t_lbfgs:.0f}s")
        a_lb = lbgfs_refine(best_seq, t_lbfgs, T_stages,
                            maxiter_per_stage=3000, deadline=time.monotonic() + t_lbfgs)
        local_update(a_lb, "lbgfs")

    # === Phase 7: Nesterov LP round 3 (200s) ===
    print(f"=== Phase 7: Nesterov LP R3 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_lp3 = min(rem * 0.12, 200.0)
    if t_lp3 > 15:
        print(f"  Budget: {t_lp3:.0f}s")
        a_lp3, _ = nesterov_lp_refine(best_seq, t_lp3, solver, n_max_iters=300000)
        local_update(a_lp3, "nesterov_lp_3")

    # === Phase 8: Perturbation + quick optimize (100s) ===
    print(f"=== Phase 8: Perturbation Escape (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        for trial in range(3):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 30:
                break
            x = np.linspace(0, 2 * np.pi, len(best_seq))
            freq = rng.integers(1, 12)
            noise = (0.01 + 0.005 * trial) * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
            a_pert = best_seq + noise
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            # Quick LP
            t_sub = min(rem_sub * 0.25, 40.0)
            if t_sub > 10:
                a_opt, _ = nesterov_lp_refine(a_pert, t_sub, solver, n_max_iters=100000)
                local_update(a_opt, f"pert{trial}")

    # === Phase 9: Try different n ===
    print(f"=== Phase 9: Try different n (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        # Try n-100
        n_alt = max(500, n_best - 100)
        t_alt = min(rem * 0.15, 100.0)
        if t_alt > 30 and n_alt != n_best:
            print(f"  Trying n={n_alt}, budget={t_alt:.0f}s")
            init_alt = interpolate_sequence(best_seq, n_alt)
            a_sp = spectral_era(init_alt, n_iters=30, blend=0.7,
                                deadline=time.monotonic() + 20)
            a_opt, _ = nesterov_lp_refine(a_sp, t_alt * 0.7, solver, n_max_iters=100000)
            local_update(a_opt, f"lp_n{n_alt}")

    rem = deadline - time.monotonic()
    if rem > 80:
        # Try n+100
        n_alt2 = n_best + 100
        t_alt2 = min(rem * 0.15, 80.0)
        if t_alt2 > 20:
            print(f"  Trying n={n_alt2}, budget={t_alt2:.0f}s")
            init_alt2 = interpolate_sequence(best_seq, n_alt2)
            a_sp2 = spectral_era(init_alt2, n_iters=20, blend=0.7,
                                 deadline=time.monotonic() + 15)
            a_opt2, _ = nesterov_lp_refine(a_sp2, t_alt2 * 0.7, solver, n_max_iters=80000)
            local_update(a_opt2, f"lp_n{n_alt2}")

    # === Phase 10: Final polish ===
    print(f"=== Phase 10: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        # Quick SCF
        t_fin_scf = min(rem * 0.2, 30.0)
        if t_fin_scf > 5:
            a_fin_scf, _ = scf_iteration(best_seq, t_fin_scf, T_start=None,
                                         T_end=0.0005, n_max_iters=1000, damp=0.3)
            local_update(a_fin_scf, "final_scf")

    rem = deadline - time.monotonic()
    if rem > 15:
        # Quick peak shave
        t_ps = min(rem * 0.3, 20.0)
        if t_ps > 5:
            a_ps, _ = peak_shave(best_seq, t_ps, n_steps=30000)
            local_update(a_ps, "final_peak")

    rem = deadline - time.monotonic()
    if rem > 15:
        # Final LP
        t_flp = min(rem * 0.4, 20.0)
        if t_flp > 5:
            a_flp, _ = nesterov_lp_refine(best_seq, t_flp, solver, n_max_iters=30000)
            local_update(a_flp, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_start
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}) at {elapsed:.1f}s")
        test_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        try:
            eval_score = evaluate_sequence(test_seq)
            print(f"Verified with evaluate_sequence: {eval_score:.8f}")
        except Exception as e:
            print(f"Verification error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
