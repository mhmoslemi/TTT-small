    """Build rows of convolution matrix T for specified lag indices W.
    Returns matrix of shape (len(W), n) where row i is T[W[i], :]."""
    n = len(a)
    rows = np.zeros((len(W), n))
    for i, k in enumerate(W):
        m_start = max(0, k - n + 1)
        m_end = min(k, n - 1)
        if m_start <= m_end:
            idx = k - np.arange(m_start, m_end + 1)
            rows[i, m_start:m_end + 1] = a[idx]
    return rows


def solve_best_response_lp(a, max_ws=80, max_inner=40, tol=1e-11):
    """Solve min M s.t. (a*b)[k]<=M for all k, sum(b)=1, b>=0 via cutting plane."""
    n = len(a)
    W = [n - 1]
    b_cur = np.ones(n) / n

    for _ in range(max_inner):
        rows = build_conv_rows(a, W)
        nW = len(W)
        A_ub = np.hstack([rows, -np.ones((nW, 1))])
        b_ub = np.zeros(nW)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        c = np.zeros(n + 1)
        c[n] = 1.0
        bounds = [(0, None)] * n + [(None, None)]
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 3.0})
            if res.status != 0:
                return None, None
        except Exception:
            return None, None

        b_opt = np.maximum(res.x[:n], 0.0)
        s = b_opt.sum()
        if s < 1e-15:
            return None, None
        b_opt /= s
        M_opt = float(res.x[n])

        conv = cross_convolve_fft(a, b_opt)
        max_k = int(np.argmax(conv))
        if conv[max_k] <= M_opt + tol:
            return b_opt, float(conv[max_k])

        if max_k not in W:
            W.append(max_k)
        if len(W) > max_ws:
            slacks = M_opt - conv
            W_sorted = sorted(W, key=lambda k: slacks[k])
            W = W_sorted[:max_ws]
        b_cur = b_opt

    conv = cross_convolve_fft(a, b_cur)
    return b_cur, float(conv.max())


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_grad(z, n, T):
    """Compute LogSumExp smoothed objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T)
    lse = max_b + T * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def lbgfs_refine(a0, time_limit, T_stages, deadline):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    n_stages = len(T_stages)

    for stage_i, stage_T in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 3:
            break
        stages_left = n_stages - stage_i
        iters = max(20, min(3000, int(remaining * 40 / stages_left)))

        def func_local(z_in):
            if time.monotonic() > deadline:
                return 1e15
            obj, _ = logsumexp_obj_grad(z_in, n, stage_T)
            return obj

        def jac_local(z_in):
            if time.monotonic() > deadline:
                return np.zeros_like(z_in)
            _, g = logsumexp_obj_grad(z_in, n, stage_T)
            return g

        try:
            res = minimize(fun=func_local, x0=z, jac=jac_local, method='L-BFGS-B',
                           options={'maxiter': iters, 'ftol': 1e-25,
                                    'gtol': 1e-18, 'maxfun': iters * 500, 'disp': False})
            z = res.x
        except Exception:
            pass
    return softmax_from_z(z)


def peak_shave(a0, time_limit, n_steps=200000):
    """Direct peak shaving via multiplicative updates targeting argmax."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = 0.03
    scale = 1.0
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * (1.0 - 2e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        eta_eff = eta * scale
        log_scale = -eta_eff * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= 40:
            scale *= 0.7
            no_improve = 0
            if scale < 0.003:
                break
    return best_a, best_score


def spectral_flatten(a, n_iters=50, blend=0.3, deadline=None):
    """Gerchberg-Saxton style spectral flattening."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)
        mags = np.abs(F)
        non_dc = mags[1:]
        if non_dc.min() < 1e-15:
            target = np.mean(non_dc)
        else:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        phases = np.angle(F)
        F_new = F.copy()
        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend) * mags[k] + blend * target
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])
        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new
    return a


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def nesterov_lp_refine(a0, time_limit, deadline=None):
    """Nesterov dual averaging LP alternating. Main optimization workhorse."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    if deadline is None:
        deadline = t0 + time_limit
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(777)

    it = 0
    while time.monotonic() < deadline:
        it += 1
        b_new, M_b = solve_best_response_lp(y)
        if b_new is None:
            b_new, M_b = solve_best_response_lp(y, max_ws=20, max_inner=20)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-12:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 10:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 6)
                noise = 0.02 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                stall_count = 0

            if (it + 1) % 2000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def multiplicative_subgrad(a0, time_limit, n_steps=1000000):
    """Mirror descent (multiplicative weights) on active peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = 0.05
    scale = 1.0
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * (1.0 - 5e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        eta_eff = eta * scale
        log_scale = -eta_eff * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= 80:
            scale *= 0.7
            no_improve = 0
            if scale < 0.003:
                break
    return best_a, best_score


def evaluate_local(a):
    """Fast local evaluation of score."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def propose_candidate(seed=42, budget_s=1800.0):
    """Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    rem = deadline - time.monotonic()
    print(f"Total budget: {rem:.0f}s, starting score: {best_score:.8f}")

    # === Phase 1: Spectral initialization and multi-n exploration ===
    print("\n=== Phase 1: Initialization & Multi-n (30s) ===")
    t_p1 = min(rem * 0.02, 30.0)
    n_candidates = [800, 1000, 1200, 1500, 2000]
    if parent is not None:
        n_candidates = [len(parent)] + [n for n in n_candidates if n != len(parent)]

    inits = {}
    for n_try in n_candidates[:4]:
        if time.monotonic() > time.monotonic() + t_p1:
            break
        if parent is not None and n_try != len(parent):
            init = interpolate_sequence(parent, n_try)
        else:
            # Structured init
            x = np.linspace(0, 1, n_try)
            choice = rng.integers(0, 4)
            if choice == 0:
                init = np.cos(np.pi * x / 2) ** 2
            elif choice == 1:
                init = np.sin(np.pi * x) ** 2
            elif choice == 2:
                init = np.exp(-4 * x * (1 - x))
            else:
                init = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
            init = np.maximum(init, 0)
            s = init.sum()
            if s < 1e-15:
                init = np.ones(n_try) / n_try
            else:
                init = init / s

        # Quick spectral flatten
        init_sp = spectral_flatten(init, n_iters=20, blend=0.3)
        sc = evaluate_local(init_sp)
        inits[n_try] = init_sp
        print(f"  n={n_try}: init score={sc:.8f}")
        local_update(init_sp, f"init_n{n_try}")

    # Pick the best initial for main optimization
    if inits:
        best_n = min(inits.keys(), key=lambda k: evaluate_local(inits[k]))
        main_init = inits[best_n].copy()
    else:
        best_n = len(parent) if parent is not None else 1200
        main_init = parent.copy() if parent is not None else np.ones(1200) / 1200

    print(f"  Best init: n={best_n}, score={evaluate_local(main_init):.8f}")

    # === Phase 2: Nesterov LP (main workhorse, ~50% budget) ===
    print(f"\n=== Phase 2: Nesterov LP n={best_n} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_p2 = min(rem * 0.45, 700.0)
    if t_p2 > 30:
        print(f"  Budget: {t_p2:.0f}s")
        a_lp, _ = nesterov_lp_refine(main_init, t_p2, deadline=time.monotonic() + t_p2)
        local_update(a_lp, "nesterov_lp_1")

    # === Phase 3: Continue LP or try another n ===
    print(f"\n=== Phase 3: Additional optimization (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_p3 = min(rem * 0.25, 400.0)
    if t_p3 > 30:
        # If we have room, try LP on a different n
        other_ns = [n for n in inits if n != len(best_seq)]
        if other_ns:
            target_n = other_ns[0]
            target_init = interpolate_sequence(best_seq, target_n)
            target_init = spectral_flatten(target_init, n_iters=15, blend=0.3)
            t_sub = min(t_p3 * 0.5, 150.0)
            print(f"  Trying n={target_n}, budget={t_sub:.0f}s")
            a_other, _ = nesterov_lp_refine(target_init, t_sub, deadline=time.monotonic() + t_sub)
            local_update(a_other, f"lp_n{target_n}")

        # Continue LP on best
        rem = deadline - time.monotonic()
        t_sub2 = min(rem * 0.5, 200.0)
        if t_sub2 > 20:
            print(f"  Continuing LP n={len(best_seq)}, budget={t_sub2:.0f}s")
            a_lp2, _ = nesterov_lp_refine(best_seq, t_sub2, deadline=time.monotonic() + t_sub2)
            local_update(a_lp2, "nesterov_lp_2")

    # === Phase 4: L-BFGS LogSumExp polish ===
    print(f"\n=== Phase 4: L-BFGS Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_p4 = min(rem * 0.25, 300.0)
    if t_p4 > 20:
        T_stages = [5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1,
                    0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.001]
        print(f"  T_stages: {len(T_stages)} stages, budget={t_p4:.0f}s")
        a_lb = lbgfs_refine(best_seq, t_p4, T_stages, deadline=time.monotonic() + t_p4)
        local_update(a_lb, "lbgfs_polish")

    # === Phase 5: Peak shaving ===
    print(f"\n=== Phase 5: Peak Shaving (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_p5 = min(rem * 0.10, 100.0)
    if t_p5 > 10:
        a_ps, _ = peak_shave(best_seq, t_p5, n_steps=300000)
        local_update(a_ps, "peak_shave")

    # === Phase 6: Multiplicative subgradient ===
    print(f"\n=== Phase 6: Multiplicative Subgrad (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_p6 = min(rem * 0.10, 80.0)
    if t_p6 > 10:
        a_ms, _ = multiplicative_subgrad(best_seq, t_p6, n_steps=500000)
        local_update(a_ms, "mult_subgrad")

    # === Phase 7: Final LP polish ===
    print(f"\n=== Phase 7: Final LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_p7 = min(rem * 0.15, 120.0)
    if t_p7 > 10:
        a_fin, _ = nesterov_lp_refine(best_seq, t_p7, deadline=time.monotonic() + t_p7)
        local_update(a_fin, "final_lp")

    # === Phase 8: Final L-BFGS ===
    print(f"\n=== Phase 8: Final L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        T_fin = [2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
        a_fin2 = lbgfs_refine(best_seq, min(rem, 60.0), T_fin, deadline=time.monotonic() + min(rem, 60.0))
        local_update(a_fin2, "final_lbgfs2")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
        elapsed = time.monotonic() - t_start
        print(f"\nFINAL SCORE (local): {best_score:.8f} (n={len(best_seq)}) at {elapsed:.1f}s")
        test_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        try:
            eval_score = evaluate_sequence(test_seq)
            print(f"Verified with evaluate_sequence: {eval_score:.8f}")
        except Exception as e:
            print(f"Verification error: {e}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
