        b = 1.0 / (S + 1e-10)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated LP best-response with Polyak averaging
2. FFT-based LogSumExp gradient optimization with aggressive annealing
3. Convex combination of LP solutions for improved convergence
4. Multi-length exploration with adaptive time allocation
5. Final coordinate-descent polish

import numpy as np

import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def fft_convolve(a, b):
    """Full convolution of a and b using FFT."""
    n1, n2 = len(a), len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def softmax_fn(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def get_lp_solver():
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

def solve_lp_best_response(a_fixed, solver):
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def nesterov_lp_optimize(a0, time_limit, solver, max_iters=50000):
    """
    Nesterov-accelerated LP best-response with Polyak averaging.
    Tracks:
    - Best primal iterate
    - Polyak average of primal iterates
    - Average of LP solutions (dual iterates)
    - Best convex combination found so far
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    y = a.copy()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Polyak averages
    polyak_a = np.zeros(n)
    polyak_count = 0
    b_avg = np.zeros(n)  # Average of LP solutions
    b_count = 0
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue
        
        # Accumulate LP solutions
        b_avg += b_star
        b_count += 1
        
        # Nesterov step
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_sum = y_new.sum()
        if y_sum < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / y_sum
        
        if t <= 2:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new
        
        y = y_new
        a = a_new
        
        # Polyak average
        polyak_a += a
        polyak_count += 1
        
        # Check multiple candidates
        candidates = [a_new, y_new]
        if b_count >= 5:
            candidates.append(b_avg / b_count)
        if polyak_count >= 5:
            candidates.append(polyak_a / polyak_count)
        
        for cand in candidates:
            cand = np.maximum(np.asarray(cand, dtype=np.float64), 0)
            sc = cand.sum()
            if sc < 1e-15:
                continue
            cand = cand / sc
            score = 2.0 * n * fast_autocorr(cand).max()
            if score < best_score:
                best_score = score
                best_a = cand.copy()
        
        if t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}, LP_M={M_val:.8f}")
    
    return best_a, best_score

def logsumexp_optimize(a0, time_limit, T_start=100.0, T_end=0.05, n_stages=25):
    """LogSumExp-smoothed optimization with aggressive multi-stage annealing."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        iters_per_stage = max(20, min(100, int(50 + 30 * (si / n_stages))))
        
        def obj_fn(z_in):
            a_s = softmax_fn(z_in)
            b = fast_autocorr(a_s)
            Tb = stage_T * b
            return (Tb.max() + np.log(np.exp(Tb - Tb.max()).sum())) / stage_T
        
        def grad_fn(z_in):
            a_s = softmax_fn(z_in)
            b = fast_autocorr(a_s)
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            
            conv_len = n + (2 * n - 1) - 1
            fft_size = 1 << (conv_len - 1).bit_length()
            if conv_len > fft_size:
                fft_size *= 2
            A_rev = rfft(a_s[::-1].copy(), fft_size)
            W = rfft(w, fft_size)
            conv = irfft(A_rev * W, fft_size)
            grad_a = 2.0 * conv[n - 1:n - 1 + n]
            
            S = np.dot(grad_a, a_s)
            grad_z = a_s * (grad_a - S)
            return grad_z
        
        try:
            res = minimize(
                fun=obj_fn, x0=z, jac=grad_fn, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': iters_per_stage * 300,
                         'disp': False}
            )
            z = res.x
            a_s = softmax_fn(z)
            score = 2.0 * n * fast_autocorr(a_s).max()
            if score < best_score:
                best_score = score
                best_a = a_s.copy()
        except Exception:
            pass
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
