```python
def compute_grad_fft(a, T):
    n = len(a)
    # Step 1: conv(a, a)
    size_c = 2*n - 1
    fft_size = 1 << (size_c - 1).bit_length()
    A = rfft(a, fft_size)
    c = irfft(A * A, fft_size)[:size_c]
    
    # Step 2: softmax
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    W = exp_c / exp_c.sum()
    
    # Step 3: conv(W, a)
    size_3 = size_c + n - 1  # 3n-2
    fft_size_3 = 1 << (size_3 - 1).bit_length()
    Wf = rfft(W, fft_size_3)
    Af = rfft(a, fft_size_3)
    conv_W_a = irfft(Wf * Af, fft_size_3)[:size_3]
    
    # Step 4: conv(W, a[::-1])
    a_rev = a[::-1]
    Af_rev = rfft(a_rev, fft_size_3)
    conv_W_a_rev = irfft(Wf * Af_rev, fft_size_3)[:size_3]
    
    # Step 5: G[i] = (2n/T) * (conv_W_a[i] + conv_W_a_rev[n-1-i])
    G = (2.0 * n / T) * (conv_W_a[:n] + conv_W_a_rev[n-1::-1])
    
    return G, W, c
```

```python
"""
Fast Smoothed Mirror Descent (FSMD) for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

Key innovations:
1. FFT-based O(n log n) gradient computation for LogSumExp-smoothed objective
   (vs. O(n^2) LP solve in previous approach)
2. Mirror descent (exponential family) updates for the simplex constraint
3. Very fine temperature annealing schedule (many small-T stages)
4. Periodic "peak surgery" to escape local optima
5. Efficient coordinate descent polishing on the exact objective
6. Multi-resolution exploration
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def _fast_conv(a):
    n = len(a)
    size = 2*n - 1
    fs = 1 << (size - 1).bit_length()
    A = rfft(a, fs)
    return irfft(A * A, fs)[:size]

def _score(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    n = len(a)
    c = _fast_conv(a)
    return 2.0 * n * c.max() / (s * s)

def _grad_smoothed(a, T):
    """Gradient of (2n/T)*LogSumExp(conv(a,a)/T) via FFT."""
    n = len(a)
    size_c = 2*n - 1
    fs_c = 1 << (size_c - 1).bit_length()
    A = rfft(a, fs_c)
    c = irfft(A*A, fs_c)[:size_c]
    
    cm = c.max()
    ec = np.exp((c - cm) / T)
    W = ec / ec.sum()
    
    size_3 = size_c + n - 1
    fs_3 = 1 << (size_3 - 1).bit_length()
    Wf = rfft(W, fs_3)
    Af = rfft(a, fs_3)
    conv1 = irfft(Wf * Af, fs_3)[:n]  # conv(W, a)[:n]
    
    a_rev = a[::-1]
    Afr = rfft(a_rev, fs_3)
    conv2 = irfft(Wf * Afr, fs_3)[n-1::-1]  # conv(W, a_rev)[n-1::-1]
    
    G = (2.0 * n / T) * (conv1 + conv2)
    return G, W, c

def _mirror_update(a, G, eta):
    log_a = np.log(np.maximum(a, 1e-300))
    log_new = log_a - eta * G
    log_new -= log_new.max()
    a_new = np.exp(log_new)
    s = a_new.sum()
    if s < 1e-300:
        return a.copy()
    return a_new / s

def _peak_surgery(a, strength=0.05):
    """Reduce top contributors to the dominant peak."""
    n = len(a)
    c = _fast_conv(a)
    k = int(np.argmax(c))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    idx = np.arange(js, je + 1)
    contribs = a[idx] * a[k - idx]
    m = max(3, len(idx) // 15)
    top = idx[np.argsort(contribs)[::-1][:m]]
    a_new = a.copy()
    for pos in top:
        if a_new[pos] > 1e-10:
            a_new[pos] *= (1 - strength)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    return a_new

def _fsmd(a0, time_limit, T_schedule=None, eta_scale=0.4, peak_every=200):
    """Fast Smoothed Mirror Descent."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a /= s
    best_a = a.copy()
    best_score = _score(a)
    t0 = time.monotonic()
    
    if T_schedule is None:
        T_schedule = [10, 6, 4, 2.5, 1.5, 1.0, 0.6, 0.4, 0.25, 0.15, 0.1, 0.06, 0.04, 0.025]
    
    rng = np.random.default_rng(777)
    
    for Ti, T in enumerate(T_schedule):
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        # Adaptive number of iterations based on remaining time
        iters_this_T = 3000 + 2000 * Ti  # More iterations at finer T
        
        for it in range(iters_this_T):
            if time.monotonic() - t0 > time_limit - 5:
                break
            
            G, W, c = _grad_smoothed(a, T)
            eta = eta_scale * T
            
            a = _mirror_update(a, G, eta)
            
            # Periodic peak surgery
            if (it + 1) % peak_every == 0:
                a_surg = _peak_surgery(a, strength=0.03 + 0.02 * Ti / len(T_schedule))
                if _score(a_surg) < _score(a):
                    a = a_surg
            
            # Check improvement
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        # Print progress
        if (time.monotonic() - t0) < time_limit:
            print(f"  FSMD T={T:.3f} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def _interpolate(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0)
    ss = a_new.sum()
    if ss > 1e-15:
        a_new /= ss
    return a_new

        per_n = t_alloc / max(1, len(n_candidates))
        for n_try in n_candidates:
            if time.monotonic() > deadline - 10:
                break
            if n_try == n_best:
                continue
            a_interp = _interpolate(best_seq, n_try)
            t_sub = min(per_n, deadline - time.monotonic() - 5)
            if t_sub < 15:
                continue
            print(f"  n={n_try}, budget={t_sub:.0f}s")
            T_sched = [5, 3, 1.5, 0.8, 0.4, 0.2, 0.1, 0.05]
            a_sub, _ = _fsmd(a_interp, t_sub, T_schedule=T_sched, eta_scale=0.4)
            try_update(a_sub, f"multi_n_{n_try}")
    
    # Phase 4: Coordinate descent polish
    print(f"\n=== Phase 4: Coord Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 180.0)
    if t_alloc > 10:
        a_cd, sc_cd = _coord_descent(best_seq, t_alloc)
        try_update(a_cd, "coord_desc")
    
    # Phase 5: Final ultra-fine FSMD
    print(f"\n=== Phase 5: Final polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = rem - 5
    if t_alloc > 10:
        T_sched_final = [0.05, 0.03, 0.02, 0.012, 0.008, 0.005, 0.003]
        a_fin, _ = _fsmd(best_seq, t_alloc, T_schedule=T_sched_final, eta_scale=0.6)
        try_update(a_fin, "final")
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
```
