"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based linearized subgradient method (main engine)
2. FISTA with log-sum-exp smoothing (secondary)
3. Multi-length exploration and random restarts
4. Active-set equioscillation refinement

Key innovation: At each iteration, solve a small LP to find the best
linearized direction that reduces the max autocorrelation peak, then
take a damped step. This is much faster than the previous approach
of solving an LP for the best response at each iteration.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_local(a):
    """Evaluate the objective locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def solve_direction_lp(a, b, active_idx, step_bound, solver):
    """
    Solve LP to find best linearized direction.
    
    Minimize t
    Subject to:
        sum_i (2*a[i]) * delta[k-i] <= t - b[k]  for k in active_idx
        -a[i] <= delta[i] (non-negativity of a + delta)
        sum(delta) = 0
        |delta[i]| <= step_bound
    """
    n = len(a)
    delta = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        cp.sum(delta) == 0,
        delta >= -a,  # a + delta >= 0
        delta <= step_bound,
        delta >= -step_bound,
    ]
    
    # Active constraints: b[k] + 2*(a*delta)[k] <= t
    # (a*delta)[k] = sum_i a[i]*delta[k-i] for valid i
    # = sum_{i: max(0,k-n+1) <= i <= min(n-1,k)} a[i] * delta[k-i]
    # Let j = k-i, then i = k-j, so:
    # = sum_{j: max(0,k-n+1) <= k-j <= min(n-1,k)} a[k-j] * delta[j]
    # = sum_{j: max(0, k-n+1) <= j <= min(n-1, k)} a[k-j] * delta[j]
    
    for k in active_idx:
        # Build the row: c[i] = 2*a[k-i] for valid i
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        # For delta[j], j ranges from js to je, coefficient is 2*a[k-j]
        idx = np.arange(js, je + 1)
        coeff = 2.0 * a[k - idx]
        constraints.append(coeff @ delta <= t - b[k])
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            return delta.value.copy(), float(t.value)
    except Exception:
        pass
    
    # Try other solvers
    for s in [cp.PDLP, cp.CLARABEL, cp.SCS]:
        if s == solver:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return delta.value.copy(), float(t.value)
        except Exception:
            continue
    
    return None, None

    best_a = a.copy()

    t0 = time.monotonic()

    tol = 0.1  # fraction of max that defines "active"
    no_improve = 0
    rng = np.random.default_rng(12345)
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        b_max = b.max()
        
        # Identify active set (top peaks)
        threshold = b_max * (1.0 - tol)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        # Cap at top_k
        if len(active) > top_k:
            idx = np.argpartition(b, -top_k)[-top_k:]
            active = idx
        
        # Solve LP for direction
        delta, t_val = solve_direction_lp(a, b, active, step, solver)
        if delta is None:
            # Try with wider step bound
            delta, t_val = solve_direction_lp(a, b, active, step * 5, solver)
        
        if delta is None:
            # Stuck, try perturbation
            a_pert = a + 0.01 * (np.random.randn(n) - 0.5)
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            if eval_local(a_pert) < best_score:
                a = a_pert
                best_a = a.copy()
                best_score = eval_local(a)
            continue
        
        # Take damped step with line search
        improved = False
        for alpha in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
            a_new = a + alpha * delta
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new
            score = eval_local(a_new)
            if score < best_score - 1e-15:
                best_score = score
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
        
        if not improved:
            no_improve += 1
            if no_improve >= 10:
                # Reduce step size
                step *= 0.8
                no_improve = 0
            if no_improve >= 50:
                # Perturb
                a_pert = a + 0.02 * np.random.randn(n)
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / a_pert.sum()
                a = a_pert
                no_improve = 0
        else:
            no_improve = 0
            step = max(step * step_decay, 0.01)
        
        it += 1
        if it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Linearized it={it} ({elapsed:.0f}s): score={best_score:.8f} step={step:.4f} |K|={len(active)}")
        
        # gradually tighten tolerance
        tol = max(0.001, tol * 0.999)
    
    return best_a, best_score

def softmax(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s
