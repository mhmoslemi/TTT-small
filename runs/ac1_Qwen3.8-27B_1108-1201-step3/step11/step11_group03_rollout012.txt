import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.sparse import lil_matrix, csc_matrix
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def build_q_sparse(k, n):
    """Build sparse Q matrix for conv(a,a)[k] = a^T Q_k a."""
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_sum = b_val.sum()
                if s_sum < 1e-15:
                    return None, None
                b_val = b_val / s_sum
                return b_val, float(max(0, M_var.value))
        except Exception:
            continue
    return None, None


def evaluate_local(a):
    """Evaluate sequence score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=500000):
    """Nesterov dual averaging LP alternating with adaptive restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(123)
    lp_times = []

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver)
        lp_time = time.monotonic() - lp_t0
        lp_times.append(lp_time)

        if b_new is None:
            continue

        # Nesterov acceleration
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 6:
                # Restart Nesterov from best
                y = best_a.copy()
                a = best_a.copy()
                # Add small perturbation
                noise = 0.01 * rng.standard_normal(n)
                y = y + noise
                y = np.maximum(y, 0)
                sy2 = y.sum()
                if sy2 > 1e-15:
                    y = y / sy2
                stall_count = 0

            if (it + 1) % 1000 == 0:
                elapsed = time.monotonic() - t0
                avg_lp = np.mean(lp_times[-50:]) * 1000 if len(lp_times) >= 50 else 0
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.10f} avg_lp={avg_lp:.1f}ms")

        if len(lp_times) > 50 and np.mean(lp_times[-50:]) > 3.0:
            print(f"    LP too slow ({np.mean(lp_times[-50:]):.2f}s/iter), stopping at iter {it+1}")
            break

    return best_a, best_score


def solve_active_qcqp(a, solver, rel_tol=1e-4, max_active=20, max_rounds=5):
    """
    Active-set method for the convex QCQP:
    min t s.t. a^T Q_k a <= t for all active k, sum(a)=1, a>=0
    Iteratively adds violated constraints.
    """
    n = len(a)
    a_curr = np.maximum(np.asarray(a, dtype=np.float64), 0).copy()
    s = a_curr.sum()
    if s < 1e-15:
        a_curr = np.ones(n) / n
    a_curr = a_curr / s

    best_a = a_curr.copy()
    best_t = fast_convolve(a_curr).max()

    for round_idx in range(max_rounds):
        b = fast_convolve(a_curr)
        max_b = b.max()

        threshold = max_b * (1.0 - rel_tol)
        active = np.where(b >= threshold)[0]

        if len(active) > max_active:
            active = active[np.argsort(b[active])[-max_active:]]

        if len(active) == 0:
            active = np.array([np.argmax(b)])

        a_var = cp.Variable(n, nonneg=True)
        t_var = cp.Variable()
        constraints = [cp.sum(a_var) == 1]

        for k in active:
            Q = build_q_sparse(int(k), n)
            constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)

        prob = cp.Problem(cp.Minimize(t_var), constraints)

        solved = False
        solvers_to_try = [cp.MOSEK, cp.GUROBI, solver, cp.CLARABEL, cp.ECOS, cp.SCS]
        for s_solver in solvers_to_try:
            if s_solver is None:
                continue
            try:
                prob.solve(solver=s_solver)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    a_val = np.asarray(a_var.value, dtype=np.float64)
                    a_val = np.maximum(a_val, 0)
                    s_sum = a_val.sum()
                    if s_sum < 1e-15:
                        break
                    a_val = a_val / s_sum
                    t_val = float(max(0, t_var.value))
                    solved = True
                    break
            except Exception as e:
                continue

        if not solved:
            break

        # Check for violated constraints
        b_new = fast_convolve(a_val)
        t_opt = t_val
        violations = b_new - t_opt
        max_violation = violations.max()

        if max_violation < 1e-8:
            # No violations, we have the optimum
            if 2.0 * n * t_val < 2.0 * n * best_t:
                best_t = t_val
                best_a = a_val.copy()
            break

        # Update active set: add most violated constraints
        new_active = np.where(violations > 1e-10)[0]
        active = np.unique(np.concatenate([active, new_active]))
        if len(active) > max_active:
            # Keep the most important constraints
            b_vals = b_new[active]
            v_vals = violations[active]
            # Rank by max of b value and violation
            importance = np.maximum(b_vals / max_b, v_vals / max(max_violation, 1e-15))
            top_idx = np.argsort(importance)[-max_active:]
            active = active[top_idx]

        a_curr = a_val.copy()
        rel_tol = max(rel_tol * 0.5, 1e-7)

    return best_a


def full_qcqp_solve(a, solver, n_constraints=None):
    """Try to solve the full QCQP with all constraints."""
    n = len(a)
    a_curr = np.maximum(np.asarray(a, dtype=np.float64), 0).copy()
    s = a_curr.sum()
    if s < 1e-15:
        a_curr = np.ones(n) / n

        n_constraints = 2 * n - 1

    # Only use top n_constraints most important

        top_k = np.argsort(b)[-n_constraints:]
    else:
        top_k = np.arange(len(b))

    a_var = cp.Variable(n, nonneg=True)

def multiplicative_weights_refine(a0, time_limit, n_steps=50000):
    """Mirror descent / multiplicative weights on the simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 0.05
    no_improve = 0
    scale = 1.0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Use top-k active constraints
        threshold = max_b * (1.0 - 1e-3)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        eff_eta = eta * scale
        log_scale = -eff_eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 50:
            scale *= 0.7
            no_improve = 0
            if scale < 0.001:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.sparse import lil_matrix, csc_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def build_q_sparse(k, n):
    """Build sparse Q matrix for conv(a,a)[k] = a^T Q_k a."""
    Q = lil_matrix((n, n))
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        partner = k - j
        if j == partner:
            Q[j, j] = 1.0
        else:
            Q[j, partner] = 0.5
            Q[partner, j] = 0.5
    return csc_matrix(Q)


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)

    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

    n = len(a0)

    iterations_since_restart = 0

        lp_times.append(time.monotonic() - lp_t0)

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:

            sy = 1.0
        y = y / sy

        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        iterations_since_restart += 1

        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 5:
                # Restart from best with perturbation
                y = best_a.copy()
                a = best_a.copy()
                noise_scale = max(0.005, 0.02 / (1 + stall_count))
                noise = noise_scale * rng.standard_normal(n)
                y = y + noise
                y = np.maximum(y, 0)
                sy2 = y.sum()
                if sy2 > 1e-15:
                    y = y / sy2
                stall_count = 0
                iterations_since_restart = 0

            if (it + 1) % 500 == 0:
                elapsed = time.monotonic() - t0
                avg_lp = np.mean(lp_times[-50:]) * 1000 if len(lp_times) >= 50 else 0
                print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.10f} avg_lp={avg_lp:.1f}ms stalls={stall_count}")

        if len(lp_times) > 30 and np.mean(lp_times[-30:]) > 4.0:
            print(f"    LP too slow ({np.mean(lp_times[-30:]):.2f}s/iter), stopping at iter {it+1}")
            break

    return best_a, best_score


def solve_active_qcqp(a, solver, rel_tol=1e-4, max_active=20, max_rounds=10):
    """
    Active-set method for the convex QCQP with iterative constraint addition.
    min t s.t. a^T Q_k a <= t for k in active, sum(a)=1, a>=0
    """
    n = len(a)
    a_curr = np.maximum(np.asarray(a, dtype=np.float64), 0).copy()
    s = a_curr.sum()
    if s < 1e-15:
        a_curr = np.ones(n) / n
    a_curr = a_curr / s

    best_a = a_curr.copy()
    best_t = fast_convolve(a_curr).max()

    for round_idx in range(max_rounds):
        b = fast_convolve(a_curr)
        max_b = b.max()

        # Identify active constraints
        threshold = max_b * (1.0 - rel_tol)
        active = np.where(b >= threshold)[0]

        if len(active) > max_active:
            active = active[np.argsort(b[active])[-max_active:]]

        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # Build and solve QCQP
        a_var = cp.Variable(n, nonneg=True)
        t_var = cp.Variable()
        constraints = [cp.sum(a_var) == 1]

        for k in active:
            Q = build_q_sparse(int(k), n)
            constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)

        prob = cp.Problem(cp.Minimize(t_var), constraints)

        solved = False
        solvers_to_try = [cp.MOSEK, cp.GUROBI, solver, cp.CLARABEL, cp.ECOS, cp.SCS]
        for s_solver in solvers_to_try:
            if s_solver is None:
                continue
            try:
                prob.solve(solver=s_solver)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    a_val = np.asarray(a_var.value, dtype=np.float64)
                    a_val = np.maximum(a_val, 0)
                    s_sum = a_val.sum()
                    if s_sum < 1e-15:
                        break
                    a_val = a_val / s_sum
                    t_val = float(max(0, t_var.value))
                    solved = True
                    break
            except Exception:
                continue

        if not solved:
            break

        # Check for violated constraints
        b_new = fast_convolve(a_val)
        max_b_new = b_new.max()
        t_opt = max(t_val, max_b_new)

        if 2.0 * n * max_b_new < 2.0 * n * best_t - 1e-12:
            best_t = max_b_new
            best_a = a_val.copy()

        # Check violations using the solved t
        violations = b_new - t_opt
        max_violation = violations.max()

        if max_violation < 1e-9:
            break

        # Add violated constraints
        new_active = np.where(violations > 1e-10)[0]
        combined = np.unique(np.concatenate([active, new_active]))
        if len(combined) > max_active:
            # Keep most important by b value
            b_at_combined = b_new[combined]
            top_idx = np.argsort(b_at_combined)[-max_active:]
            combined = combined[top_idx]
        active = combined

        a_curr = a_val.copy()
        # Tighten tolerance for next round
        rel_tol = max(rel_tol * 0.5, 1e-8)

    return best_a


def full_qcqp_top_k(a, solver, k_constraints=100):
    """Solve QCQP with top-k most important constraints."""
    n = len(a)
    a_curr = np.maximum(np.asarray(a, dtype=np.float64), 0).copy()
    s = a_curr.sum()
    if s < 1e-15:
        a_curr = np.ones(n) / n
    a_curr = a_curr / s

    b = fast_convolve(a_curr)
    if k_constraints < len(b):
        top_k = np.argsort(b)[-k_constraints:]
    else:
        top_k = np.arange(len(b))

    a_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [cp.sum(a_var) == 1]

    for k in top_k:
        Q = build_q_sparse(int(k), n)
        constraints.append(cp.quad_form(a_var, Q, assume_PSD=True) <= t_var)

    prob = cp.Problem(cp.Minimize(t_var), constraints)

    solvers_to_try = [cp.MOSEK, cp.GUROBI, solver, cp.CLARABEL]
    for s_solver in solvers_to_try:
        if s_solver is None:
            continue
        try:
            prob.solve(solver=s_solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                a_val = np.asarray(a_var.value, dtype=np.float64)
                a_val = np.maximum(a_val, 0)
                s_sum = a_val.sum()
                if s_sum < 1e-15:
                    return None
                a_val = a_val / s_sum
                return a_val
        except Exception:
            continue
    return None


def multiplicative_weights_refine(a0, time_limit, n_steps=100000):
    """Mirror descent / multiplicative weights on the simplex targeting active peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 0.04
    no_improve = 0
    scale = 1.0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Use top active constraints
        threshold = max_b * (1.0 - 5e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        eff_eta = eta * scale
        log_update = -eff_eta * g
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 40:
            scale *= 0.7
            no_improve = 0
            if scale < 0.001:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    
    Phases:
    1. Nesterov LP alternating (primary optimization)
    2. Active-set QCQP polishing
    3. Second LP round from polished solution
    4. Full top-k QCQP attempt
    5. Multiplicative weights subgradient
    6. Final QCQP
    7. Multi-n exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        x = np.linspace(0, 1, n_init)
        best_seq = np.cos(np.pi * x / 2) ** 2
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")

    n_base = len(best_seq)

    def update_best(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score - 1e-14:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # === Phase 1: LP alternating (main workhorse) ===
    print(f"\n=== Phase 1: LP Alternating n={n_base} (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.45, 650.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver, n_max_iters=500000)
        update_best(a_lp, "lp_p1")

    # === Phase 2: Active-set QCQP ===
    print(f"\n=== Phase 2: Active QCQP (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t2 = min(rem * 0.12, 150.0)
        t2_start = time.monotonic()
        a_q1 = solve_active_qcqp(best_seq, solver, rel_tol=2e-3, max_active=12, max_rounds=8)
        update_best(a_q1, "qcqp_loose")

        a_q2 = solve_active_qcqp(best_seq, solver, rel_tol=1e-5, max_active=8, max_rounds=10)
        update_best(a_q2, "qcqp_tight")
        print(f"  QCQP time: {time.monotonic() - t2_start:.1f}s")

    # === Phase 3: LP round 2 ===
    print(f"\n=== Phase 3: LP Round 2 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.18, 220.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, solver, n_max_iters=300000)
        update_best(a_lp2, "lp_p3")

    # === Phase 4: Full top-k QCQP ===
    print(f"\n=== Phase 4: Full Top-k QCQP (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t4_start = time.monotonic()
        a_full = full_qcqp_top_k(best_seq, solver, k_constraints=80)
        update_best(a_full, "full_qcqp_80")

        rem = deadline - time.monotonic()
        if rem > 30:
            a_full2 = full_qcqp_top_k(best_seq, solver, k_constraints=50)
            update_best(a_full2, "full_qcqp_50")
        print(f"  Full QCQP time: {time.monotonic() - t4_start:.1f}s")

    # === Phase 5: Multiplicative weights ===
    print(f"\n=== Phase 5: Mult Weights (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        a_ms, _ = multiplicative_weights_refine(best_seq, min(rem * 0.08, 25.0), n_steps=200000)
        update_best(a_ms, "mult_w")

    # === Phase 6: Final tight QCQP ===
    print(f"\n=== Phase 6: Final QCQP (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        a_qf = solve_active_qcqp(best_seq, solver, rel_tol=1e-7, max_active=6, max_rounds=8)
        update_best(a_qf, "qcqp_final")

    # === Phase 7: Multi-n exploration ===
    print(f"\n=== Phase 7: Multi-n Exploration (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    n_base = len(best_seq)

    n_list = [n_base - 1, n_base + 1, n_base - 2, n_base + 2, n_base - 3, n_base + 3]
    for n_try in n_list:
        rem = deadline - time.monotonic()
        if rem < 35:
            break
        if n_try < 100:
            continue

        t_n = min(rem * 0.20, 50.0)
        if t_n < 12:
            continue

        print(f"  n={n_try}, budget={t_n:.0f}s, remaining={rem:.0f}s")
        init_n = interpolate_sequence(best_seq, n_try)
        a_n, _ = nesterov_lp_refine(init_n, t_n * 0.65, solver, n_max_iters=100000)
        update_best(a_n, f"lp_n{n_try}")

        rem = deadline - time.monotonic()
        if best_seq is not None and len(best_seq) == n_try and rem > 15:
            a_qn = solve_active_qcqp(best_seq, solver, rel_tol=1e-5, max_active=10, max_rounds=6)
            update_best(a_qn, f"qcqp_n{n_try}")

    # Final QCQP on overall best
    rem = deadline - time.monotonic()
    if rem > 15 and best_seq is not None:
        a_final = solve_active_qcqp(best_seq, solver, rel_tol=1e-8, max_active=5, max_rounds=5)
        update_best(a_final, "absolute_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
