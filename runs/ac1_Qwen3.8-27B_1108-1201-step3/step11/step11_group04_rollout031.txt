```python
def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 1 and (v >= 0).all():
        return v / v.sum() if v.sum() > 0 else np.ones(n)/n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.max(np.where(u * np.arange(1, n+1) > (cssv - 1), np.arange(1, n+1)))
    theta = (cssv[rho-1] - 1) / rho
    return np.maximum(v - theta, 0)
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based subgradient with Nesterov acceleration (fast, exact max)
2. Multi-restart L-BFGS with adaptive LogSumExp temperature
3. Spectral domain peak equalization

5. Multi-length exploration
Key improvements: faster per-iteration gradient via FFT cross-correlation,
better temperature scheduling, hybrid smooth+exact phases, and
adaptive peak weighting to escape local minima.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    s = v.sum()
    if s < 1e-300:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1) / rho_val
    return np.maximum(v - theta, 0)


def softmax_safe(z):
    """Numerically stable softmax."""
    z = np.asarray(z, dtype=np.float64)
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def logsumexp_obj_grad_z(z, T, n):
    """Compute LogSumExp objective and gradient w.r.t. z (log-domain)."""
    a = softmax_safe(z)
    b = fast_convolve(a)
    max_b = b.max()
    shifted = (b - max_b) / T
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    if sum_exp < 1e-300:
        sum_exp = 1.0
    lse = max_b + T * np.log(sum_exp)
    obj = 2.0 * n * lse

    # Weights for each lag
    w = exp_shifted / sum_exp

    # Gradient of LSE w.r.t. a: grad_a[m] = 2 * sum_k w[k] * a[k-m]
    # = 2 * (a cross-correlate w_padded)[m]
    fft_size = 1 << (2 * n - 1).bit_length()
    w_padded = np.zeros(fft_size)
    w_padded[:len(b)] = w
    A = rfft(a, fft_size)
    W = rfft(w_padded, fft_size)
    grad_a = irfft(np.conj(A) * W, fft_size)[:n] * 2.0

    # Convert to z-gradient via softmax Jacobian
    dot = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot)

    return obj, grad_z


def exact_subgradient(a, rel_tol=1e-8, n_top=20):
    """
    Compute exact subgradient of max_k conv(a,a)_k using FFT cross-correlation.
    Uses top n_top peaks weighted by their distance from the median of active peaks.
    """
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()

    # Find active peaks
    threshold = max_b * (1.0 - rel_tol)
    active = np.where(b >= threshold)[0]
    if len(active) > n_top:
        # Keep only the top n_top
        idx = np.argsort(b[active])[::-1][:n_top]
        active = active[idx]
    if len(active) == 0:
        active = np.array([np.argmax(b)])

    # Weights: give more weight to higher peaks
    active_vals = b[active]
    min_active = active_vals.min()
    span = max_b - min_active
    if span < 1e-300:
        weights = np.ones(len(active)) / len(active)
    else:
        weights = (active_vals - min_active) / span
        weights = weights / weights.sum()

    # Compute gradient via FFT cross-correlation
    fft_size = 1 << (2 * n - 1).bit_length()
    w_full = np.zeros(fft_size)
    w_full[active] = 2.0 * weights
    A = rfft(a, fft_size)
    W = rfft(w_full, fft_size)
    grad = irfft(np.conj(A) * W, fft_size)[:n]

    return grad, float(max_b)


def nesterov_subgradient(a0, time_limit, rel_tol=1e-8, n_top=20, verbose=True):
    """
    Nesterov accelerated subgradient method for min max_k conv(a,a)_k on simplex.
    Uses FFT-based gradient computation for speed.
    """
    n = len(a0)
    a = project_simplex(a0.astype(np.float64))
    y = a.copy()
    v = np.zeros(n)  # momentum

    best_a = a.copy()
    best_val = fast_convolve(a).max()

    t0 = time.monotonic()
    step_size = 0.05
    momentum = 0.7
    no_improve = 0
    stall_count = 0
    rng = np.random.default_rng(99)

    for t in range(5000000):
        if time.monotonic() - t0 > time_limit:
            break

        g, max_val, _ = exact_subgradient(y, rel_tol=rel_tol, n_top=n_top)

        # Adaptive step size
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-300:
            break
        step = step_size / (1.0 + 0.002 * t)

        # Momentum update
        v = momentum * v - step * g
        y_new = y + v
        y_new = project_simplex(y_new)

        # Evaluate
        val_new = fast_convolve(y_new).max()
        if val_new < best_val - 1e-14:
            best_val = val_new
            best_a = y_new.copy()
            a = y_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        y = y_new

        # Adaptive learning rate
        if no_improve >= 200:

            momentum = max(0.5, momentum * 0.95)
            no_improve = 0
            stall_count += 1

            # Perturb to escape local minima
            if stall_count >= 5:
                noise = 0.02 * rng.standard_normal(n)
                y = y + noise
                y = project_simplex(y)
                a = y.copy()
                stall_count = 0

            if step_size < 0.001:

        if verbose and (t + 1) % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NSub t={t+1} ({elapsed:.0f}s): val={best_val:.10f} step={step_size:.6f}")

def spectral_equalize(a0, n_iters=100, blend=0.3, deadline=None):
    """
    Spectral domain peak equalization.
    Adjusts |F(a)|^2 to reduce autocorrelation peaks.
    """
    n = len(a0)
    L = 1 << (2 * n - 1).bit_length()
    a = project_simplex(a0.astype(np.float64))
    best_a = a.copy()
    best_val = fast_convolve(a).max()

    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        F = np.fft.fft(a, L)
        power = np.abs(F) ** 2
        autocorr = np.fft.ifft(power).real[:2*n-1]
        max_ac = autocorr.max()

        # Identify peak locations in autocorrelation
        peak_locs = np.where(autocorr > max_ac * 0.95)[0]

        # In frequency domain, reduce power at frequencies that contribute to peaks
        # The contribution of frequency k to autocorrelation lag m is |F_k|^2 * exp(2*pi*i*k*m/L)
        # To reduce peaks, we want to make |F| more uniform
        # Use geometric mean as target
        non_dc_power = power[1:]
        log_target = np.mean(np.log(np.maximum(non_dc_power, 1e-300)))
        target = np.exp(log_target)

        # Blend power spectrum toward target
        new_power = power.copy()
        for k in range(1, L):
            # Reduce power at high-power frequencies more aggressively
            excess = max(0, power[k] - target)
            new_power[k] = power[k] - blend * excess

        # Recover a
        new_mag = np.sqrt(np.maximum(new_power, 0))
        phases = np.angle(F)
        F_new = new_mag * np.exp(1j * phases)
        F_new[0] = F[0]  # Keep DC (preserves sum)

        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-300:
            a_new = a_new / s
            val = fast_convolve(a_new).max()
            if val < best_val:
                best_val = val
                best_a = a_new.copy()
            a = a_new
        else:
            break

    return best_a, best_val * 2 * n


def peak_shave(a0, time_limit, n_top=5, verbose=True):
    """
    Targeted peak shaving: directly reduce the top autocorrelation peaks
    using adaptive multiplicative updates.
    """
    n = len(a0)
    a = project_simplex(a0.astype(np.float64))
    best_a = a.copy()
    best_val = fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 0.08
    scale = 1.0
    no_improve = 0

    for step in range(500000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Find top peaks
        sorted_idx = np.argsort(b)[::-1]
        active = sorted_idx[:n_top]

        # Compute subgradient from active peaks
        g = np.zeros(n)
        for k in active:
            for j in range(n):
                kj = k - j
                if 0 <= kj < n:

        a_new = np.maximum(a_new, 0)
        s = a_new.sum()

            val = fast_convolve(a_new).max()

                best_val = val
                best_a = a_new.copy()

                no_improve = 0
            else:
                no_improve += 1
        else:

        if no_improve >= 100:

            if scale < 0.01:
                break

        if verbose and (step + 1) % 50000 == 0:
            elapsed = time.monotonic() - t0

def evaluate_local(a):
    """Compute the objective locally (sum-normalized)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-300:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n, endpoint=False)
    new_x = np.linspace(0, 1, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-300:
        f_interp = f_interp / ss
    return f_interp


def perturb_sequence(a, rng, strength=0.03):
    """Add structured perturbation to sequence."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 10)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-300:
        return a.copy()
    return a_pert / s

    print(f"\n=== Phase 3: Nesterov Subgradient (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_ns = min(rem * 0.25, 400.0)
    if t_ns > 30:
        print(f"  Budget: {t_ns:.0f}s")
        a_ns, ns_score = nesterov_subgradient(best_seq, t_ns, verbose=True)
        local_update(a_ns, "nesterov_sub")

        # Run again with different parameters
        rem = deadline - time.monotonic()
        if rem > 100:
            t_ns2 = min(rem * 0.15, 150.0)
            if t_ns2 > 30:
                print(f"  Nesterov round 2: budget={t_ns2:.0f}s")
                a_ns2, _ = nesterov_subgradient(best_seq, t_ns2, rel_tol=1e-6,
                                                 n_top=10, verbose=False)
                local_update(a_ns2, "nesterov_sub2")

    # === Phase 4: Peak Shaving (100s) ===
    print(f"\n=== Phase 4: Peak Shaving (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_ps = min(rem * 0.06, 100.0)
    if t_ps > 10:
        print(f"  Budget: {t_ps:.0f}s")
        a_pk, pk_score = peak_shave(best_seq, t_ps, n_top=5, verbose=True)
        local_update(a_pk, "peak_shave")

    # === Phase 5: L-BFGS fine-tune on best (200s) ===
    print(f"\n=== Phase 5: L-BFGS Fine (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_fine = min(rem * 0.15, 200.0)
    if t_fine > 30:
        print(f"  Budget: {t_fine:.0f}s")
        a_fine, _ = lbgfs_refine(best_seq, t_fine, verbose=True,
                                  deadline=time.monotonic() + t_fine)
        local_update(a_fine, "lbgfs_fine")

    # === Phase 6: Try different n ===
    print(f"\n=== Phase 6: Multi-length (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 120:
        for target_n in [1000, 1500]:
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t_n = min(rem * 0.12, 100.0)
            if t_n < 30:
                continue
            print(f"  Trying n={target_n}, budget={t_n:.0f}s")
            init_n = interpolate_seq(best_seq, target_n)
            # Quick spectral
            a_sn, _ = spectral_equalize(init_n, n_iters=30, blend=0.3,
                                         deadline=time.monotonic() + min(t_n, 10))
            # Quick subgradient
            a_opt, _ = nesterov_subgradient(a_sn, t_n * 0.6, verbose=False)
            local_update(a_opt, f"n{target_n}")

    # === Phase 7: Final polish (100s) ===
    print(f"\n=== Phase 7: Final Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.5, 60.0)
        if t_fin > 10:
            a_fin, _ = lbgfs_refine(best_seq, t_fin, verbose=False,
                                     deadline=time.monotonic() + t_fin)
            local_update(a_fin, "final_lbfgs")

        rem = deadline - time.monotonic()
        if rem > 20:
            t_ps2 = min(rem * 0.5, 30.0)
            if t_ps2 > 5:
                a_ps2, _ = peak_shave(best_seq, t_ps2, n_top=3, verbose=False)
                local_update(a_ps2, "final_peak")

        rem = deadline - time.monotonic()
        if rem > 15:
            a_ns3, _ = nesterov_subgradient(best_seq, min(rem, 20.0),
                                             verbose=False, n_top=5)
            local_update(a_ns3, "final_ns")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_start
        print(f"\nFINAL: score={best_score:.10f} (n={len(best_seq)}) at {elapsed:.1f}s")
        test_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        try:
            eval_score = evaluate_sequence(test_seq)
            print(f"Verified: {eval_score:.10f}")
        except Exception as e:
            print(f"Verification: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
