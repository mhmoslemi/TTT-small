def fast_convolve(a):

    return irfft(A * A, fft_size)[:size]


def proj_simplex(v):
    """Project vector onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

        if time.monotonic() - t0 > time_limit:
            break

        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver)
        lp_times.append(time.monotonic() - lp_t0)

        if b_new is None:
            continue

        if len(lp_times) > 100:
            avg_lp_time = np.mean(lp_times[-50:])
            if avg_lp_time > 3.0:
                print(f"    LP too slow ({avg_lp_time:.2f}s/iter), stopping at iter {it+1}")
                break

        # Nesterov dual averaging
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        # Primal averaging (weighted average of iterates)
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        # Evaluate both a and y
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Periodic verification and restart
        if verify_interval > 0 and (it + 1) % verify_interval == 0:
            if best_score < prev_best - 1e-8:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 5:
                # Perturb y to escape local minimum
                strength = 0.02 * math.sqrt(stall_count)
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 15)
                phase = rng.uniform(0, 2 * np.pi)
                noise = strength * np.cos(freq * x + phase)
                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                stall_count = 0

        if (it + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            avg_t = np.mean(lp_times[-100:]) if len(lp_times) >= 100 else 0
            print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} avg_lp={avg_t*1000:.1f}ms")

    return best_a, best_score
