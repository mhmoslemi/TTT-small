"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_with(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS, cp.SCP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS, cp.GUROBI, cp.MOSEK]
    solved = False
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solved = True
                break
        except Exception:
            continue

    if not solved:
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(max(0, M_var.value))


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    shifted = (b - max_b) / T_temp
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_terms = np.exp(shifted)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    # Use the convolution structure: grad_a[j] = sum_k w[k] * (a[k-j] + a[j-k]) / |K|
    # More efficiently: grad_a = (w * a) convolved, but let's use the direct approach
    # b[k] = sum_j a_j a_{k-j}, so db[k]/da[m] = a[k-m] + a[m-k] (with boundary handling)
    # grad_a[m] = sum_k w[k] * (a[k-m] + a[m-k]) / sum(w)
    # This is: grad_a = w_conv(a) + w_conv(reversed_a) where w_conv means cross-correlation
    # Simpler: use the FFT-based approach
    # Actually, let's just do it directly using the Toeplitz structure
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def spectral_flatten(a, n_iters=50, blend=0.4, deadline=None):
    """Gerchberg-Saxton spectral flattening."""
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)

        mags = np.abs(F)
        non_dc = mags[1:]
        non_dc_pos = non_dc[non_dc > 1e-15]
        if len(non_dc_pos) > 0:
            target = np.exp(np.mean(np.log(non_dc_pos)))
        else:
            target = np.mean(non_dc)

        phases = np.angle(F)
        F_new = F.copy()
        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend) * mags[k] + blend * target
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])

        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new

    return a


def peak_shave_exact(a, time_limit, n_steps=200000):
    """
    Exact subgradient multiplicative weights targeting the active set.
    Uses the exact set of near-maximal autocorrelation values.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return None, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 0.015
    scale = 1.0
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Active set: peaks within 0.05% of max
        threshold = max_b * (1.0 - 5e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # Weighted subgradient: weight by how close to max
        weights = b[active] - max_b * (1.0 - 5e-4)
        weights = np.maximum(weights, 1e-10)
        weights /= weights.sum()

        g = np.zeros(n)
        for i, k in enumerate(active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += weights[i] * a[k - idx]

        eta_eff = eta * scale
        log_scale = -eta_eff * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 40:
            scale *= 0.65
            no_improve = 0
            if scale < 0.003:
                break

    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=1000000):
    """Nesterov dual averaging LP alternating with adaptive restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(12345)
    lp_times = []

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver)
        lp_time = time.monotonic() - lp_t0
        lp_times.append(lp_time)

        if b_new is None:
            continue

        if len(lp_times) >= 30:
            avg_lp_time = np.mean(lp_times[-30:])
            if avg_lp_time > 8.0:
                print(f"    LP too slow ({avg_lp_time:.2f}s/iter), stopping at iter {it+1}")
                break

        # Nesterov dual averaging with adaptive step
        alpha = 2.0 / (it + 2)

        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Stall detection and restart
        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-12:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 10:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 9)
                noise = 0.02 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                stall_count = 0

            if (it + 1) % 2000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage, deadline=None):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    if deadline is None:
        deadline = t0 + time_limit
    n_stages = len(T_stages)

    def func(z_in):
        if time.monotonic() > deadline:
            return 1e15
        obj, _ = logsumexp_obj_grad(z_in, n, T_stages[0])
        return obj

    best_z = z.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()

    for stage_i, stage_T in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 3:
            break

        stages_left = n_stages - stage_i
        # More iterations at lower temperatures
        if stage_T > 1.0:
            iters = max(20, min(maxiter_per_stage // n_stages, 100))
        elif stage_T > 0.01:
            iters = max(30, min(maxiter_per_stage // (n_stages // 2), 300))
        else:
            iters = max(50, min(maxiter_per_stage, 500))

        # Adjust based on remaining time
        time_per_iter_est = 0.02 * n / 1200  # rough estimate
        max_iters_by_time = int(remaining / max(time_per_iter_est, 0.005) / stages_left)
        iters = min(iters, max(20, max_iters_by_time))

        def func_stage(z_in):
            return lbgfs_objective(z_in, n, stage_T, deadline)

        def jac_stage(z_in):
            return lbgfs_gradient(z_in, n, stage_T, deadline)

        try:
            res = minimize(
                fun=func_stage, x0=z, jac=jac_stage, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-30,
                         'gtol': 1e-20, 'maxfun': iters * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        a_candidate = softmax_from_z(z)
        sc = 2.0 * n * fast_convolve(a_candidate).max()
        if sc < best_score:
            best_score = sc
            best_z = z.copy()

    return softmax_from_z(best_z)


def lbgfs_objective(z_in, n, T_stage, deadline):
    if time.monotonic() > deadline:
        return 1e15
    obj, _ = logsumexp_obj_grad(z_in, n, T_stage)
    return obj


def lbgfs_gradient(z_in, n, T_stage, deadline):
    if time.monotonic() > deadline:
        return np.zeros_like(z_in)
    _, g = logsumexp_obj_grad(z_in, n, T_stage)
    return g


def structured_init(n, rng):
    """Create a structured initial sequence."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 6:
        center = n // 2
        t = (np.arange(n) - center) / center
        a = np.sinc(t)
        a = np.maximum(a, 0)
    elif choice == 7:
        a = 1.0 - np.abs(2 * x - 1)
    elif choice == 8:
        # Asymmetric ramp (designed to reduce center peak)
        a = x ** 0.7 + 0.05
    else:
        # Bimodal asymmetric
        a = np.exp(-50 * (x - 0.2) ** 2) + 0.6 * np.exp(-50 * (x - 0.8) ** 2) + 0.05

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def perturbed_sequence(a, rng, strength=0.03):
    """Create a perturbed version of sequence a."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 9)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s

    t0 = time.monotonic()

    eta_base = 0.03
    scale = 1.0
    no_improve = 0
    b_for_lse = 1.0 / (2 * n)  # approximate target

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Use LogSumExp-style weights for smoother gradients
        T = max(0.001, (max_b - b_for_lse) * 0.5)
        shifted = (b - max_b) / T
        shifted = np.clip(shifted, -50, 0)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            active = np.argmax(b)
            g = np.zeros(n)
            js = max(0, active - n + 1)
            je = min(n - 1, active)
            idx = np.arange(js, je + 1)
            g[idx] = a[active - idx]
        else:
            w = w / w_sum
            g = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-12:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                g[idx] += w[k] * a[k - idx]

        eta = eta_base * scale
        log_scale_vec = -eta * g
        log_scale_vec -= log_scale_vec.max()
        a_new = a * np.exp(log_scale_vec)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 40:
            scale *= 0.65
            no_improve = 0
            if scale < 0.003:
                break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Key improvements: extended L-BFGS temperature annealing, exact subgradient
    peak shaving, adaptive time allocation, and multi-n exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    rem = deadline - time.monotonic()
    print(f"Total budget: {rem:.0f}s, starting score: {best_score:.8f}")

    # === Phase 0: Quick diversification (10s) ===
    print(f"\n=== Phase 0: Diversification ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_div = min(rem * 0.008, 10.0)
        candidates = [best_seq.copy()]
        for strength in [0.02, 0.05, 0.08]:
            candidates.append(perturbed_sequence(best_seq, rng, strength=strength))
        candidates.append(spectral_flatten(best_seq, n_iters=5, blend=0.3))
        best_c = best_seq.copy()
        best_sc = best_score
        for c in candidates:
            sc = evaluate_local(c)
            if sc < best_sc:
                best_sc = sc
                best_c = c.copy()
        local_update(best_c, "diversified")

    # === Phase 1: Spectral Flattening (30s) ===
    print(f"=== Phase 1: Spectral Flattening (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_spec = min(rem * 0.02, 30.0)
    if t_spec > 5:
        a_spec = spectral_flatten(best_seq, n_iters=300, blend=0.25, deadline=time.monotonic() + t_spec)
        local_update(a_spec, "spectral")
        a_spec2 = spectral_flatten(best_seq, n_iters=100, blend=0.5, deadline=time.monotonic() + min(t_spec * 0.5, 15))
        local_update(a_spec2, "spectral_strong")

    # === Phase 2: Nesterov LP (main workhorse, 350s) ===
    print(f"=== Phase 2: Nesterov LP n={len(best_seq)} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_lp = min(rem * 0.22, 350.0)
    if t_lp > 30:
        print(f"  Budget: {t_lp:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_lp, solver, n_max_iters=1000000)
        local_update(a_lp, "nesterov_lp_1")

    # === Phase 3: Aggressive LogSumExp L-BFGS (700s) ===
    print(f"=== Phase 3: L-BFGS LogSumExp (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_lbfgs = min(rem * 0.38, 700.0)
    if t_lbfgs > 30:
        # Extended temperature schedule with more low-temperature stages
        T_stages = [10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1,
                    0.05, 0.03, 0.02, 0.015, 0.01, 0.007, 0.005,
                    0.004, 0.003, 0.002, 0.0015, 0.001, 0.0008, 0.0005]
        print(f"  T_stages: {len(T_stages)} stages, budget={t_lbfgs:.0f}s")
        a_lb = lbgfs_refine(best_seq, t_lbfgs, T_stages, maxiter_per_stage=30000,
                            deadline=time.monotonic() + t_lbfgs)
        local_update(a_lb, "lbgfs_extended")

    # === Phase 4: Exact Peak Shaving (80s) ===
    print(f"=== Phase 4: Peak Shaving (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_peak = min(rem * 0.05, 80.0)
    if t_peak > 5:
        a_pk, _ = peak_shave_exact(best_seq, t_peak, n_steps=300000)
        local_update(a_pk, "peak_exact")

    # === Phase 5: Second Nesterov LP round (150s) ===
    print(f"=== Phase 5: Nesterov LP Round 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_lp2 = min(rem * 0.10, 150.0)
    if t_lp2 > 20:
        print(f"  Budget: {t_lp2:.0f}s")
        a_lp2, _ = nesterov_lp_refine(best_seq, t_lp2, solver, n_max_iters=500000)
        local_update(a_lp2, "nesterov_lp_2")

    # === Phase 6: Post-LP L-BFGS refinement (150s) ===
    print(f"=== Phase 6: Post-LP L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_lbfgs2 = min(rem * 0.10, 150.0)
    if t_lbfgs2 > 15:
        T_stages2 = [0.5, 0.3, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001, 0.0005]
        a_lb2 = lbgfs_refine(best_seq, t_lbfgs2, T_stages2, maxiter_per_stage=20000,
                             deadline=time.monotonic() + t_lbfgs2)
        local_update(a_lb2, "lbgfs_post_lp")

    # === Phase 7: Try different n values (120s) ===
    print(f"=== Phase 7: Try different n (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        # Try n = 800 (smaller, might be easier to optimize)
        t_n800 = min(rem * 0.15, 60.0)
        if t_n800 > 20:
            print(f"  Trying n=800, budget={t_n800:.0f}s")
            init_800 = interpolate_sequence(best_seq, 800)
            a_sp800 = spectral_flatten(init_800, n_iters=20, blend=0.3)
            a_800, _ = nesterov_lp_refine(a_sp800, t_n800 * 0.6, solver, n_max_iters=100000)
            local_update(a_800, "lp_n800")
            # Quick L-BFGS on n=800
            rem = deadline - time.monotonic()
            if best_seq is not None and len(best_seq) == 800 and rem > 20:
                T_st800 = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
                a_800b = lbgfs_refine(best_seq, min(rem, 40.0), T_st800,
                                       maxiter_per_stage=10000, deadline=time.monotonic() + min(rem, 40.0))
                local_update(a_800b, "lbgfs_n800")

    rem = deadline - time.monotonic()
    if rem > 50:
        # Try n = 2000 (larger, better theoretical bound)
        t_n2000 = min(rem * 0.3, 80.0)
        if t_n2000 > 20:
            print(f"  Trying n=2000, budget={t_n2000:.0f}s")
            init_2000 = interpolate_sequence(best_seq, 2000)
            a_sp2000 = spectral_flatten(init_2000, n_iters=20, blend=0.3)
            a_2000, _ = nesterov_lp_refine(a_sp2000, t_n2000 * 0.5, solver, n_max_iters=50000)
            local_update(a_2000, "lp_n2000")
            rem = deadline - time.monotonic()
            if best_seq is not None and len(best_seq) == 2000 and rem > 15:
                T_st2000 = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
                a_2000b = lbgfs_refine(best_seq, min(rem, 30.0), T_st2000,
                                        maxiter_per_stage=5000, deadline=time.monotonic() + min(rem, 30.0))
                local_update(a_2000b, "lbgfs_n2000")

    # === Phase 8: Perturbation escape + re-optimize ===
    print(f"=== Phase 8: Perturbation Escape (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        for trial in range(3):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 25:
                break
            a_pert = perturbed_sequence(best_seq, rng, strength=0.01 + 0.008 * trial)
            a_sp = spectral_flatten(a_pert, n_iters=15, blend=0.3)
            t_sub = min(rem_sub * 0.15, 40.0)
            if t_sub > 10:
                a_opt, _ = nesterov_lp_refine(a_sp, t_sub, solver, n_max_iters=100000)
                local_update(a_opt, f"pert_trial{trial}")

    # === Phase 9: Final polish ===
    print(f"=== Phase 9: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        # Very fine L-BFGS
        t_fin = min(rem * 0.3, 50.0)
        if t_fin > 5:
            T_stages_fin = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001, 0.0005]
            a_fin = lbgfs_refine(best_seq, t_fin, T_stages_fin, maxiter_per_stage=10000,
                                 deadline=time.monotonic() + t_fin)
            local_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 10:
        # Final peak shave
        t_ps = min(rem * 0.3, 25.0)
        if t_ps > 3:
            a_ps, _ = peak_shave_exact(best_seq, t_ps, n_steps=100000)
            local_update(a_ps, "final_peak")

    rem = deadline - time.monotonic()
    if rem > 10:
        # Final multiplicative subgradient
        t_ms = min(rem * 0.4, 30.0)
        if t_ms > 3:

    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_start
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}) at {elapsed:.1f}s")
        test_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        try:
            eval_score = evaluate_sequence(test_seq)
            print(f"Verified with evaluate_sequence: {eval_score:.8f}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
