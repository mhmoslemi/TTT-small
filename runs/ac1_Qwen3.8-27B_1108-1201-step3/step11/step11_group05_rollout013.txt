def fast_convolve(a, fft_size=None):
    """Linear auto-convolution via FFT with optional precomputed size."""
    n = len(a)
    if fft_size is None:
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:2 * n - 1]


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = [solver, cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
                break
        except Exception:
            continue
    else:
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    b_val[np.isnan(b_val)] = 0
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def softmax_from_z(z):
    """Convert log-domain z to probability vector (sum = 1)."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

    return np.log(np.maximum(a, 1e-300))


def evaluate_local(a):
    """Evaluate sequence score locally (fast, assumes a >= 0)."""
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def iceshard_init(n, rng, mode=0):
    """Create structured initialization based on crystal-shard-like symmetric patterns."""
    if mode == 0:
        # Raised cosine (smooth, symmetric)
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif mode == 1:
        # Hamming window
        x = np.linspace(0, 1, n)
        a = 0.54 - 0.46 * np.cos(2 * np.pi * x)
    elif mode == 2:
        # Sum of shifted Gaussians (shard-like multi-peak)
        x = np.linspace(0, 1, n)
        a = np.exp(-8*(x-0.25)**2) + np.exp(-8*(x-0.5)**2) + np.exp(-8*(x-0.75)**2)
        a = a + 0.1
    elif mode == 3:
        # Modulated carrier (ice shard diffraction pattern)
        x = np.linspace(0, 1, n)
        env = np.sin(np.pi * x) ** 1.5
        carrier = 0.7 + 0.3 * np.cos(2 * np.pi * 5 * x)
        a = env * carrier
        a = np.maximum(a, 0)
    elif mode == 4:
        # Flat with edge enhancement
        x = np.linspace(0, 1, n)
        a = 0.6 + 0.4 * (2 * x - 1) ** 2
    elif mode == 5:
        # Two-lobe shard
        x = np.linspace(0, 1, n)
        a = np.exp(-6*(x-0.3)**2) + 0.8*np.exp(-6*(x-0.7)**2) + 0.2
    elif mode == 6:
        # Spectral flatness attempt via inverse DFT of random phase
        fft_size = 1 << (n - 1).bit_length()
        for _ in range(10):
            phase = rng.uniform(0, 2 * np.pi, fft_size)
            mag = 0.5 + 0.5 * rng.uniform(0.7, 1.3, fft_size)
            spec = mag * np.exp(1j * phase)
            half = fft_size // 2
            hs = np.zeros(fft_size, dtype=complex)
            hs[:half+1] = spec[:half+1]
            hs[half+1:] = np.conj(hs[1:half+1][::-1])
            a_try = irfft(hs, fft_size)[:n]
            a_try = np.maximum(a_try, 0)
            if a_try.sum() > 0.1:
                a = a_try
                break
            a = np.ones(n)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def adam_logsumexp_optimize(a0, time_limit, T_schedule, lr_init, lr_decay_per_stage,
                            top_k_start, top_k_end, solver=None,
                            lp_every=8000, lp_duration=30.0, verbose=True):
    """
    Adam-based LogSumExp optimization in log-space with:
    - Temperature annealing
    - Adaptive top-k focusing
    - Periodic LP corrections
    """
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    step_count = 0
    
    # Pre-compute FFT size
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    n_stages = len(T_schedule)
    stage_time = time_limit / n_stages
    
    for stage_idx, T_stage in enumerate(T_schedule):
        if time.monotonic() > deadline:
            break
        stage_deadline = min(t0 + (stage_idx + 1) * stage_time, deadline)
        
        # Interpolate top_k
        frac = stage_idx / max(1, n_stages - 1)
        top_k = int(top_k_start + frac * (top_k_end - top_k_start))
        top_k = max(3, min(top_k, size))
        
        step_in_stage = 0
        while time.monotonic() < stage_deadline:
            if step_in_stage % 1000 == 0 and time.monotonic() > deadline:
                break
            
            a = softmax_from_z(z)
            b = fast_convolve(a, fft_size)
            max_b = b.max()
            score = 2.0 * n * max_b
            step_count += 1
            step_in_stage += 1
            
            if score < best_score:
                best_score = score
                best_a = a.copy()
            
            # Compute weights (top-k LogSumExp)
            if T_stage <= 0.001:
                # Pure subgradient on active set
                threshold = max_b * (1.0 - 1e-6)
                active = np.where(b >= threshold)[0]
                if len(active) == 0:
                    active = np.array([np.argmax(b)])
                w = np.zeros(size)
                w[active] = 1.0 / len(active)
            elif top_k < size:
                idx = np.argpartition(b, -top_k)[-top_k:]
                b_top = b[idx]
                max_top = b_top.max()
                exp_top = np.exp((b_top - max_top) / T_stage)
                w сум = exp_top.sum()
                w_top = exp_top / w_sum
                w = np.zeros(size)
                w[idx] = w_top
            else:
                max_b_all = b.max()
                exp_all = np.exp((b - max_b_all) / T_stage)
                w = exp_all / exp_all.sum()
            
            # Gradient in a-space: g_a[i] = 2 * sum_k w[k] * a[k-i]
            # = 2 * correlate(w, a, 'full')[:n]
            g_a = 2.0 * np.correlate(w, a, mode='full')[:n]
            
            # Gradient in z-space: grad_z = a * (g_a - dot(g_a, a))
            S = np.dot(g_a, a)
            grad_z = a * (g_a - S)
            
            # Adam update
            lr = lr_init * (lr_decay_per_stage ** stage_idx)
            # Slight decay within stage
            lr *= (1.0 - 0.3 * step_in_stage / max(1, int(stage_deadline - time.monotonic()) if time.monotonic() < stage_deadline else 1))
            
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** (step_count + 1))
            v_hat = v / (1 - beta2 ** (step_count + 1))
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            z -= z.mean()
            
            # Periodic LP correction
            if (solver is not None and step_count % lp_every == 0 and
                    time.monotonic() < deadline - lp_duration):
                lp_t0 = time.monotonic()
                b_target = solve_lp_best_response(a, solver)
                if b_target is not None:
                    # Blend towards LP solution
                    t_blend = 0.3
                    z_target = z_from_a(b_target)
                    z = (1.0 - t_blend) * z + t_blend * (z_target - z_target.mean() + z.mean())
                    z -= z.mean()
                if verbose:
                    lp_time = time.monotonic() - lp_t0
                    print(f"    [LP correction at step {step_count}, {lp_time:.1f}s] best={best_score:.8f}")
            
            if verbose and step_in_stage % 20000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    Adam stage {stage_idx} (T={T_stage:.3f}) step {step_in_stage} "
                      f"({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def nesterov_lp_alternate(a0, time_limit, solver, max_iters=500000):
    """Nesterov dual averaging LP alternating."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score
    rng = np.random.default_rng(12345)
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy
        
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if (it + 1) % 50 == 0:
            if best_score < prev_best - 1e-12:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            
            if stall_count >= 15:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 10)
                noise = 0.02 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0
            
            if (it + 1) % 200 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score


def mrf_directional_refinement(a0, time_limit, n_iterations=50000):
    """
    MRF-style Directional Refinement:
    - Identifies active peaks
    - For each peak, computes the exact gradient direction 2*a[k-j]
    - Projects onto sum=0 constraint
    - Takes a projected step with backtracking line search
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0, float('inf')
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Adaptive step size
    eta = 0.05
    no_improve = 0
    
    for it in range(n_iterations):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a, fft_size)
        max_b = b.max()
        score = 2.0 * n * max_b
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:
            eta *= 0.8
            no_improve = 0
            if eta < 1e-6:
                break
        
        # Find active set (peaks within small fraction of max)
        threshold = max_b * (1.0 - max(1e-5, 5.0 / (n * (1 + it / 1000))))
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        # Compute subgradient: g[j] = (1/|A|) * sum_{k in A} 2*a[k-j]
        # This is (2/|A|) * sum_{k in A} a[k-j]
        # = (2/|A|) * correlate(indicator_A, a, 'full')[:n]
        ind_A = np.zeros(size)
        ind_A[active] = 1.0 / len(active)
        g = 2.0 * np.correlate(ind_A, a, mode='full')[:n]
        
        # Project onto sum=0 (tangent to simplex)
        g_mean = g.mean()
        g_tangent = g - g_mean
        
        # Try step: a_new = a - eta * g_tangent, projected to non-negative simplex
        a_trial = a - eta * g_tangent
        a_trial = np.maximum(a_trial, 0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            continue
        a_trial = a_trial / s_trial
        
        # Backtracking line search
        t = 1.0
        accepted = False
        for _ in range(10):
            a_bt = a - t * eta * g_tangent
            a_bt = np.maximum(a_bt, 0)
            s_bt = a_bt.sum()
            if s_bt < 1e-15:
                t *= 0.5
                continue
            a_bt = a_bt / s_bt
            b_bt = fast_convolve(a_bt, fft_size)
            score_bt = 2.0 * n * b_bt.max()
            if score_bt < score - 1e-14:
                a = a_bt
                accepted = True
                break
            t *= 0.5
        
        if not accepted:
            # Very small step
            a = a - 0.01 * eta * g_tangent
            a = np.maximum(a, 0)
            s = a.sum()
            if s < 1e-15:
                break
            a = a / s
        
        if it % 10000 == 0 and it > 0:
            elapsed = time.monotonic() - t0
            print(f"    MRF_dir_refine {elapsed:.0f}s: score={best_score:.8f} eta={eta:.6f}")
    
    return best_a, best_score


def multiplicative_weights_refine(a0, time_limit, n_steps=100000):
    """Mirror descent (multiplicative weights) targeting active peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0, float('inf')
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    eta = 0.02
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a, fft_size)
        max_b = b.max()
        
        threshold = max_b * (1.0 - max(1e-5, 3.0 / (n * (1 + step / 500))))
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([0])
        
        # Gradient: g[j] = (1/|A|) * sum_{k in A} 2*a[k-j]
        ind_A = np.zeros(size)
        ind_A[active] = 1.0 / len(active)
        g = 2.0 * np.correlate(ind_A, a, mode='full')[:n]
        
        # Multiplicative update: a_new[j] = a[j] * exp(-eta * g[j])
        log_update = -eta * g
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        score = 2.0 * n * fast_convolve(a_new, fft_size).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 100:
            eta *= 0.7
            no_improve = 0
            if eta < 1e-5:
                break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses IceShard initialization, Adam LogSumExp, LP alternating,
    MRF directional refinement, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    
    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Budget: {budget_s:.0f}s")
    
    # Try to load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        # Try multiple IceShard initializations
        n_init = 1200
        for mode in range(7):
            a_try = iceshard_init(n_init, rng, mode=mode)
            sc = evaluate_local(a_try)
            if sc < best_score:
                best_score = sc
                best_seq = a_try.copy()
                print(f"  IceShard mode {mode}: score={sc:.8f}")
        print(f"Best init: n={len(best_seq)}, score={best_score:.8f}")
    
    n_best = len(best_seq)
    
    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}) ***")
            return True
        return False
    
    # === Phase 1: Quick Adam warm-up from best init ===
    print(f"\n=== Phase 1: Adam Warm-up n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()

    if t_alloc > 15:

        T_stages = [20.0, 10.0, 5.0, 3.0, 2.0, 1.5]
        a_res, _ = adam_logsumexp_optimize(
            best_seq, t_alloc, T_stages,
            lr_init=0.008, lr_decay_per_stage=0.85,
            top_k_start=200, top_k_end=50,
            solver=None, verbose=True
        )
        local_update(a_res, "adam_warm")
    
    # === Phase 2: Main Adam with LP corrections ===
    print(f"\n=== Phase 2: Main Adam+LP n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 500.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        T_stages = [5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02]
        a_res, _ = adam_logsumexp_optimize(
            best_seq, t_alloc, T_stages,
            lr_init=0.005, lr_decay_per_stage=0.9,
            top_k_start=100, top_k_end=15,
            solver=solver, lp_every=10000, lp_duration=30.0,
            verbose=True
        )
        local_update(a_res, "main_adam_lp")

    print(f"\n=== Phase 3: Nesterov LP n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")

            best_seq, t_alloc, T_stages,

            top_k_start=30, top_k_end=5,
            solver=None, verbose=True
        )
        local_update(a_res, "fine_adam")

    print(f"\n=== Phase 5: MRF Directional Refine (score={best_score:.8f}) ===")

    t_alloc = min(rem * 0.12, 150.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        a_res, _ = mrf_directional_refinement(best_seq, t_alloc)
        local_update(a_res, "mrf_dir")

    print(f"\n=== Phase 6: MultiWe
