def _compute_score(a):
    """Compute 2*n*max(convolve(a,a)) for a sequence with sum=1."""
    n = len(a)
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b)

def _lse_gradient(a, epsilon):
    """
    Compute gradient of 2n * epsilon * log(sum exp(b_k/epsilon)) w.r.t. a.
    Returns (gradient, smooth_objective_value).
    """
    n = len(a)
    b = np.convolve(a, a)  # length 2n-1
    # Log-sum-exp for numerical stability
    b_max = np.max(b)
    log_sum_exp = b_max/epsilon + np.log(np.sum(np.exp((b - b_max)/epsilon)))
    F = 2.0 * n * epsilon * log_sum_exp
    p = np.exp((b - b_max) / epsilon)
    p /= np.sum(p)
    # Gradient: 4n * correlate(p, a, 'valid')
    r = np.correlate(p, a, mode='valid')
    grad = 4.0 * n * r
    return grad, F

"""
Multi-phase optimizer for minimizing the normalized peak autocorrelation
of a non-negative sequence. Uses Fourier-parametric initialization,
Adam on log-parameterization with log-sum-exp smoothing, and a final
coordinate-descent polish.
"""

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def _compute_score(a):
    """Compute 2*n*max(convolve(a,a)) for a sequence with sum=1."""
    n = len(a)
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b)


def _softmax(theta):
    """Numerically stable softmax."""
    theta_shifted = theta - np.max(theta)
    exp_theta = np.exp(theta_shifted)
    return exp_theta / np.sum(exp_theta)


def _lse_smooth_objective_and_grad(a, epsilon):
    """
    Compute the log-sum-exp smoothed objective and its gradient.
    
    F(a) = 2n * epsilon * log(sum_k exp(b_k / epsilon))
    grad_i = 4n * sum_j a_j * p_{i+j}
    
    where p_k = exp(b_k/epsilon) / sum_j exp(b_j/epsilon)
    """
    n = len(a)
    b = np.convolve(a, a)  # length 2n-1
    b_max = np.max(b)
    shifted = (b - b_max) / epsilon
    log_sum = b_max / epsilon + np.log(np.sum(np.exp(shifted)))
    F = 2.0 * n * epsilon * log_sum
    p = np.exp(shifted)
    p /= np.sum(p)
    # Gradient: 4n * correlate(p, a, 'valid')
    r = np.correlate(p, a, mode='valid')
    grad = 4.0 * n * r
    return F, grad


def _adam_update(params, grad, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """In-place Adam update."""
    m[:] = beta1 * m + (1.0 - beta1) * grad
    v[:] = beta2 * v + (1.0 - beta2) * (grad * grad)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    params[:] -= lr * m_hat / (np.sqrt(v_hat) + eps)


def _make_fourier_seq(n, amps):
    """
    Create a sequence from cosine amplitudes.
    a_i = 1 + sum_k amps[k] * cos(2*pi*(k+1)*i/n)
    """
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k, A in enumerate(amps):
        if abs(A) > 1e-15:
            a += A * np.cos(2.0 * np.pi * (k + 1) * i / n)
    a = np.maximum(a, 1e-10)
    return a / np.sum(a)


def _fourier_objective(amps, n):
    """Objective for Fourier-parameter optimization."""
    a = _make_fourier_seq(n, amps)
    return _compute_score(a)


def _adam_optimize(theta, deadline, rng, epsilon_init, lr_init=0.05,
                   max_iterations=None):

    theta is modified in place. Returns (best_a, best_score).
    """

    b_init = np.convolve(_softmax(theta), _softmax(theta))
    b_scale = np.max(b_init)
    epsilon = epsilon_init * b_scale
    lr = lr_init
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    a = _softmax(theta)
    best_a = a.copy()
    best_score = _compute_score(best_a)
    
    no_improve = 0
    patience = 300
    
    while time.monotonic() < deadline:
        t += 1
        
        a = _softmax(theta)
        
        # Compute gradient in a-space
        _, grad_a = _lse_smooth_objective_and_grad(a, epsilon)
        
        # Transform to theta-space
        dot = np.dot(a, grad_a)
        grad_theta = a * (grad_a - dot)
        
        # Adam step
        _adam_update(theta, grad_theta, m, v, t, lr)
        
        # Evaluate
        a = _softmax(theta)
        score = _compute_score(a)
        
        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:

def _coordinate_polish(a, deadline, rng):
    """
    Coordinate descent polish: try adjusting individual elements.
    For each coordinate, try increasing or decreasing it slightly
    and adjust the rest proportionally.
    """
    n = len(a)
    best_a = a.copy()
    best_score = _compute_score(best_a)
    
    # Precompute for efficiency
    for iteration in range(200):
        if time.monotonic() >= deadline:
            break
        
        improved = False
        # Random subset of coordinates per iteration
        idx = rng.choice(n, size=min(500, n), replace=False)
        
        for i in idx:
            # Try small perturbations
            for delta_frac in [0.01, -0.01, 0.05, -0.05, 0.1, -0.1]:
                a_new = best_a.copy()
                a_new[i] = max(0, a_new[i] * (1 + delta_frac))
                # Renormalize others proportionally
                old_sum = np.sum(best_a)
                new_i = a_new[i]
                rest_sum = old_sum - new_i
                if rest_sum <= 0:
                    continue
                a_new[np.arange(n) != i] = best_a[np.arange(n) != i] * (rest_sum / (old_sum - best_a[i]))
                a_new = a_new / np.sum(a_new)
                
                score = _compute_score(a_new)
                if score < best_score - 1e-12:
                    best_score = score
                    best_a = a_new.copy()
                    improved = True
                    break
        
        if not improved and iteration > 20:
            break
    
    return best_a, best_score
