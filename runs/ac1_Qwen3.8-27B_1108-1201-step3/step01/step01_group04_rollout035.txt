"""
Optimizes a non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses combined smooth gradient (FFT-based), subgradient, and LP methods.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize_a(a, S_target=None):
    if S_target is None:
        S_target = float(np.sqrt(2.0 * len(a)))
    S = a.sum()
    if S < 1e-12:
        return a
    return a * (S_target / S)


def get_grad_smooth(a):
    """Gradient of sum(conv(a,a)^2) with respect to a. Using FFT."""
    n = len(a)
    c = np.convolve(a, a)
    N = 1
    while N < n + len(c) - 1:
        N *= 2
    A = np.fft.rfft(a, N)
    C = np.fft.rfft(c, N)
    cross = np.fft.irfft(np.conj(A) * C, N)[:n]
    return 4.0 * cross


def get_grad_peak(a):
    """Subgradient of max(conv(a,a)) with respect to a."""
    n = len(a)
    c = np.convolve(a, a)
    k_star = int(c.argmax())
    grad = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    for j in range(lo, hi + 1):
        grad[j] = 2.0 * a[k_star - j]
    return grad, c


def solve_lp_minpeak(a, S):
    """
    Solve LP: min M s.t. conv(a,g) <= M, sum(g) = S, g >= 0.
    Returns g (the optimal direction) or None.
    """
    n = len(a)
    if n > 4096:
        return None
    
    # Build constraint matrix A: A[k, j] = a[k-j] if 0 <= k-j < n
    A_mat = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A_mat[k, j] = a[k - j]
    
    # c_obj: minimize M (last variable)
    c_obj = np.append(np.zeros(n), 1.0)
    
    # A_ub: conv(a,g) - M <= 0 → A_mat @ g - M <= 0
    A_ub = np.hstack([A_mat, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    # A_eq: sum(g) = S
    A_eq = np.append(np.ones(n), 0.0).reshape(1, -1)
    b_eq = np.array([S])
    
    # Bounds: g >= 0, M >= 0
    bounds = [(0, None)] * (n + 1)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            return res.x[:n]
    except Exception:
        pass
    return None


def optimize_single(a0, deadline, verbose=False):
    """Optimize a single sequence. Returns (best_seq, best_score)."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    S_target = float(np.sqrt(2.0 * n))
    a = normalize_a(a, S_target)
    score = compute_obj(a)
    best = a.copy()
    best_score = score
    
    if verbose:
        print(f"  Start: n={n}, score={score:.6f}")
    
    max_iters = 500
    smooth_steps = 20
    peak_steps = 5
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 10.0:
            break
        
        # Phase A: Smooth gradient descent
        lr = 0.05 * (1.0 + np.cos(np.pi * it / max_iters)) + 0.005
        for _ in range(smooth_steps):
            if time.monotonic() > deadline - 10.0:
                break
            grad = get_grad_smooth(a)
            grad -= grad.mean()  # project onto fixed-sum subspace
            a = a - lr * grad
            a = np.maximum(a, 0.0)
            a = normalize_a(a, S_target)
        
        # Phase B: Peak subgradient step
        for _ in range(peak_steps):
            if time.monotonic() > deadline - 10.0:

            grad, c = get_grad_peak(a)
            grad -= grad.mean()
            a = a - lr * grad
            a = np.maximum(a, 0.0)
            a = normalize_a(a, S_target)

        s = compute_obj(a)
        if s < best_score:
            best_score = s
            best = a.copy()
            if verbose:
