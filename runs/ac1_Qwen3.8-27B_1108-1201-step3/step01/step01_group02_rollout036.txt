```python
"""
Multi-phase optimization for minimizing peak auto-convolution of non-negative sequences.
Uses LP-based refinement with line search, multi-start diversity, and fine polishing.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time

def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)

def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s

def build_cross_conv_matrix(a):
    n = len(a)
    n_rows = 2 * n - 1
    row_idx = []
    col_idx = []
    data = []
    for k in range(n_rows):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            if 0 <= j < n:
                row_idx.append(k)
                col_idx.append(j)
                data.append(a[i])
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))

def solve_lp_for_direction(a, M, timeout=30.0):
    n = len(a)
    A = build_cross_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None

def lp_refine(a, time_limit_s, verbose=True, alphas=None):
    if alphas is None:
        alphas = [0.01, 0.02, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.5]
    
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    if verbose:
        print(f"    LP start: score={best_score:.8f}, n={n}")
    
    stagnant = 0
    for it in range(5000):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s - 5.0:
            break
        
        conv = np.convolve(a, a)
        M = conv.max()
        score = 2.0 * n * M / (a.sum() ** 2)
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it < 10 or it % 100 == 0):
                print(f"    LP iter {it:5d}: score={score:.8f}")
        else:
            stagnant += 1
        
        if stagnant > 50:
            if verbose:
                print(f"    LP stagnant at iter {it}, best={best_score:.8f}")
            break
        
        # Solve LP
        g = solve_lp_for_direction(a, M, timeout=20.0)
        if g is None:
            break
        
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-10:
            if verbose:
                print(f"    LP no improvement (sum_g={sum_g:.10f})")
            break
        
        g = g / sum_g
        
        # Line search
        best_alpha = alphas[0]
        best_alpha_score = float('inf')
        for alpha in alphas:
            a_try = (1.0 - alpha) * a + alpha * g
            a_try = normalize_nonneg(a_try)
            s_try = score_sequence(a_try)
            if s_try < best_alpha_score:
                best_alpha_score = s_try
                best_alpha = alpha
        
        a = (1.0 - best_alpha) * a + best_alpha * g
        a = normalize_nonneg(a)
    
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    if verbose:
        print(f"    LP done: best={best_score:.8f}, iters={it+1}, time={time.monotonic()-start_time:.1f}s")
    return best_a, best_score

def local_search(a, time_limit_s, rng, verbose=True):
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    if verbose:
        print(f"    LS start: score={best_score:.8f}")
    
    accept_count = 0
    while time.monotonic() - start_time < time_limit_s:
        # Random segment scaling
        width = max(2, int(rng.integers(n // 500, n // 4)))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)
        
        candidate = a.copy()
        noise = rng.normal(0.0, 0.05, size=stop - start)
        candidate[start:stop] *= np.exp(noise)
        
        # Sometimes symmetrize
        if rng.random() < 0.3:
            candidate = 0.5 * (candidate + candidate[::-1])
        
        candidate = normalize_nonneg(candidate)
        score = score_sequence(candidate)
        
        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1

"""
Multi-phase optimization to minimize peak auto-convolution of non-negative sequences.
Combines LP-based directional refinement with line search, multi-start diversity,
perturbation-based exploration, and fine polishing.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def build_cross_conv_matrix(a):
    n = len(a)
    n_rows = 2 * n - 1
    row_idx = np.empty(n_rows * n, dtype=np.int32)
    col_idx = np.empty(n_rows * n, dtype=np.int32)
    data = np.empty(n_rows * n, dtype=np.float64)
    count = 0
    for k in range(n_rows):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            if 0 <= j < n:
                row_idx[count] = k
                col_idx[count] = j
                data[count] = a[i]
                count += 1
    return csr_matrix((data[:count], (row_idx[:count], col_idx[:count])), shape=(n_rows, n))


def solve_lp_direction(a, M, timeout=20.0):
    n = len(a)
    A = build_cross_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def lp_refine(a, time_limit_s, verbose=True, alphas=None):
    if alphas is None:
        alphas = [0.01, 0.02, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.5]

    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)

    if verbose:
        print(f"    LP start: score={best_score:.8f}, n={n}")

    stagnant = 0
    it = 0
    for it in range(10000):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s - 5.0:
            break

        conv = np.convolve(a, a)
        M = conv.max()
        score = 2.0 * n * M / (a.sum() ** 2)

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it < 10 or it % 200 == 0):
                print(f"    LP iter {it:5d}: score={score:.8f}")
        else:
            stagnant += 1

        if stagnant > 60:
            if verbose:
                print(f"    LP stagnant at iter {it}, best={best_score:.8f}")
            break

        g = solve_lp_direction(a, M, timeout=15.0)
        if g is None:
            break

        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-10:
            if verbose:
                print(f"    LP no improvement (sum_g={sum_g:.10f})")
            break

        g = g / sum_g

        best_alpha = alphas[0]
        best_alpha_score = float('inf')
        for alpha in alphas:
            a_try = (1.0 - alpha) * a + alpha * g
            a_try = normalize_nonneg(a_try)
            s_try = score_sequence(a_try)
            if s_try < best_alpha_score:
                best_alpha_score = s_try
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g
        a = normalize_nonneg(a)

    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    if verbose:
        print(f"    LP done: best={best_score:.8f}, iters={it+1}, t={time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def local_search(a, time_limit_s, rng, verbose=True):
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)

    if verbose:
        print(f"    LS start: score={best_score:.8f}")

    accept_count = 0
    while time.monotonic() - start_time < time_limit_s:
        width = max(2, int(rng.integers(n // 500, n // 3)))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)

        candidate = a.copy()
        noise = rng.normal(0.0, 0.04, size=stop - start)
        candidate[start:stop] *= np.exp(noise)

        if rng.random() < 0.3:
            candidate = 0.5 * (candidate + candidate[::-1])

        candidate = normalize_nonneg(candidate)
        score = score_sequence(candidate)

        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1
            if verbose and accept_count % 200 == 0:
                print(f"    LS accept {accept_count}: score={best_score:.8f}")

    if verbose:
        print(f"    LS done: best={best_score:.8f}, accepts={accept_count}, t={time.monotonic()-start_time:.1f}s")
    return best_a, best_score
