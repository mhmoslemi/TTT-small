```python
import numpy as np
import time
from scipy.optimize import linprog

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score_fast(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def solve_lp_direction(a, M):
    """Solve LP to find direction g that maximizes sum(g) subject to a*g <= M and g >= 0."""
    n = len(a)
    # Build constraint matrix: (2n-1) x n
    # A[k, i] = a[k-i] if 0 <= k-i < n, else 0
    A = np.zeros((2*n - 1, n))
    for k in range(2*n - 1):
        for i in range(max(0, k-n+1), min(n, k+1)):
            A[k, i] = a[k-i]
    
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=(0, None), method='highs')
    
    if res.success:
        return res.x
    return None

def lp_iteration(a, time_limit, n_max_iters=100):
    """Iterative LP-based optimization."""
    a = a.copy()
    sum_a = np.sum(a)
    a_norm = a / sum_a  # normalize to sum = 1
    
    best_a = a_norm.copy()
    best_score = local_score_fast(a_norm)
    
    for it in range(n_max_iters):
        if time.monotonic() > time_limit:
            break
        conv = fft_convolve(a_norm, a_norm)
        M = np.max(conv)
        
        g = solve_lp_direction(a_norm, M)
        if g is None:
            break
        
        sum_g = np.sum(g)
        if sum_g < 1e-10:
            break
        
        g_norm = g / sum_g  # normalize g to sum = 1
        
        # Line search on t
        best_t = 0.0
        best_s = best_score
        for t in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
            candidate = (1-t) * a_norm + t * g_norm
            candidate = np.clip(candidate, 0.0, None)
            s = local_score_fast(candidate)
            if s < best_s:
                best_s = s
                best_t = t
        
        if best_t > 0:
            a_norm = (1-best_t) * a_norm + best_t * g_norm
            a_norm = np.clip(a_norm, 0.0, None)
            new_score = local_score_fast(a_norm)
            if new_score < best_score:
                best_score = new_score
                best_a = a_norm.copy()
        else:
            # No improvement, try smaller steps or break
            break
    
    return best_a, best_score
```

```python
"""
Iterative LP-based optimization for minimizing peak autocorrelation of
non-negative step functions. Uses linear programming to find descent directions,
adaptive line search, subgradient refinement, and local search polishing.
Multiple sequence lengths are explored with diverse initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the score (same formula as evaluate_sequence) efficiently."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * float(np.max(c)) / float(sum_a ** 2)


def build_lp_matrix(a):
    """Build the constraint matrix for the LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = a[k - j]
    return A


def solve_lp_direction(a, M, top_k=1):
    """
    Solve LP: maximize sum(g) subject to (a*g)_k <= M_k and g >= 0.
    M_k uses the current peak for all k if top_k=1, or top-k peaks.
    """
    n = len(a)
    conv = fft_convolve(a, a)
    
    # Set up right-hand side
    rhs = np.full(2 * n - 1, M)
    if top_k > 1:
        # Tighten constraints at the top-k peak positions
        peak_indices = np.argsort(conv)[-top_k:]
        for idx in peak_indices:
            rhs[idx] = conv[idx] * 0.99  # slightly tighter
    
    A = build_lp_matrix(a)
    c_obj = -np.ones(n)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=rhs, bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

            no_improve = 0

def subgradient_refine(a_init, time_limit, verbose=True, rng=None):
    """
    Subgradient-based refinement using the structure of the peak.
    Uses multiplicative updates to maintain non-negativity.
    """
    if rng is None:
        rng = np.random.default_rng(123)
    
    a = a_init.astype(np.float64).copy()
    a = np.clip(a, 1e-10, None)
    a = a / np.sum(a)
    n = len(a)
    
    best_a = a.copy()
    best_score = local_score(a)
    lr = 0.05
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a, a)
        peak_idx = int(np.argmax(conv))
        
        # Subgradient: g[j] = 2*a[peak_idx - j] if valid
        g = np.zeros(n)
        for j in range(n):
            idx = peak_idx - j
            if 0 <= idx < n:
                g[j] = 2.0 * a[idx]
        
        # Also consider second peak for more balanced descent
        sorted_peaks = np.argsort(conv)[::-1]
        if len(sorted_peaks) > 1:
            second_idx = int(sorted_peaks[1])
            if conv[second_idx] > 0.95 * conv[peak_idx]:
                for j in range(n):
                    idx = second_idx - j
                    if 0 <= idx < n:
                        g[j] += 0.5 * a[idx]
        
        # Multiplicative update
        candidate = a * np.exp(-lr * g / max(1.0, np.max(g)))
        candidate = np.clip(candidate, 1e-10, None)
        candidate = candidate / np.sum(candidate)
        
        s = local_score(candidate)
        if s < best_score:
            best_score = s
            best_a = candidate.copy()
            no_improve = 0
            lr = min(lr * 1.05, 0.5)
        elif s < best_score + 0.001:
            # Accept small deterioration for exploration
            a = candidate
            no_improve += 1
        else:
            a = candidate * (0.5 + 0.5 * (a / candidate))  # partial step
            a = np.clip(a, 1e-10, None)
            a = a / np.sum(a)
            no_improve += 1
            lr = max(lr * 0.9, 0.001)
        
        if no_improve > 200:
            break
    
    return best_a, best_score


def local_search_polish(a_init, time_limit, rng=None):
    """
    Simulated annealing local search with targeted peak reduction.
    """
    if rng is None:
        rng = np.random.default_rng(777)
    
    current = a_init.astype(np.float64).copy()
    current = np.clip(current, 1e-10, None)
    current = current / np.sum(current)
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temperature = 0.002
    cooling = 0.99985
    
    while time.monotonic() - t0 < time_limit:
        batch = min(50, max(1, int((time_limit - (time.monotonic() - t0)) * 100)))
        
        for _ in range(batch):
            if time.monotonic() - t0 >= time_limit:
                break
            candidate = current.copy()
            r = rng.random()
            
            if r < 0.4:
                # Targeted: reduce peak contributors
                conv = fft_convolve(current, current)
                peak_idx = int(np.argmax(conv))
                contribs = np.zeros(n)
                i_min = max(0, peak_idx - n + 1)
                i_max = min(n - 1, peak_idx)
                for i in range(i_min, i_max + 1):
                    contribs[i] = current[i] * current[peak_idx - i]
                top_k = max(3, n // 12)
                top_idx = np.argsort(contribs)[-top_k:]
                candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.02, size=len(top_idx)))
                other = np.setdiff1d(np.arange(n), top_idx)
                if len(other) > 0:
                    n_inc = min(10, len(other))
                    inc_idx = rng.choice(other, size=n_inc, replace=False)
                    candidate[inc_idx] *= np.exp(rng.normal(0.05, 0.02, size=n_inc))
            elif r < 0.6:
                # Block perturbation
                width = max(1, int(rng.integers(1, max(2, n // 8))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.04, size=width))
            elif r < 0.75:
                # Symmetrize partially
                current_sym = 0.5 * (current + current[::-1])
                candidate = current_sym * np.exp(rng.normal(0.0, 0.015, size=n))
            elif r < 0.85:
                # Swap-based: exchange mass between peak and non-peak regions
                conv = fft_convolve(current, current)
                peak_half = conv[:n]
                sort_idx = np.argsort(peak_half)[::-1]
                n_swap = max(2, n // 20)
                src = sort_idx[:n_swap]
                dst = sort_idx[-n_swap:]
                amount = rng.uniform(0.02, 0.08, size=n_swap)
                candidate[src] -= current[src] * amount
                candidate[dst] += current[dst] * amount
            else:
                # Small global noise
                candidate = current * np.exp(rng.normal(0.0, 0.015, size=n))
            
            candidate = np.clip(candidate, 1e-10, None)
            new_score = local_score(candidate)
            
            if new_score < current_score:
                current = candidate
                current_score = new_score
                if current_score < best_score:
                    best_score = current_score
                    best = current.copy()
            elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
                current = candidate
                current_score = new_score
            
            temperature *= cooling
        
        if verbose := (time.monotonic() - t0) % 30 < 0.1:
            print(f"    Polish: score={best_score:.6f}, t={time.monotonic()-t0:.1f}s")
    
    return best, best_score

    return np.clip(a, 0.01, None)


def make_optimized_init(n, rng):
    """Create an initial sequence based on known good patterns."""
    t = np.arange(n, dtype=np.float64)
    # Combination of low-frequency modulation
    a = 1.0
    a += 0.3 * np.cos(2 * np.pi * t / n * 2 + rng.uniform(0, 2*np.pi))
    a += 0.2 * np.cos(2 * np.pi * t / n * 5 + rng.uniform(0, 2*np.pi))
    a += 0.1 * np.cos(2 * np.pi * t / n * 13 + rng.uniform(0, 2*np.pi))
    a += 0.3 * np.sin(np.pi * t / n)  # envelope
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search for minimizing peak autocorrelation:
    Phase 1: LP-based optimization on n=256 (primary)
    Phase 2: LP-based optimization on other lengths
    Phase 3: Subgradient refinement
    Phase 4: Local search polish
    Returns best sequence found.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 25.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    
    # Check for provided sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s = local_score(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()
    
    # ==================== Phase 1: LP optimization on n=256 ====================
    print("\n=== Phase 1: LP optimization, n=256 ===")
    n1 = 256
    time_p1 = min(900.0, max(60.0, (deadline - time.monotonic()) * 0.55))
    
    if start_seq is not None and len(start_seq) == n1:
        a1 = start_seq.copy()
    else:
        a1 = make_optimized_init(n1, rng)
    
    a1 = np.clip(a1, 0.01, None)
    a1 = a1 / np.sum(a1)
    
    a_best, s_best = lp_optimize(a1, time_p1, verbose=True, rng=rng)
    actual = local_score(a_best)
    print(f"  Phase 1 result: n={n1}, score={actual:.6f}")
    if actual < best_score:
        best_score = actual
        best_seq = a_best.copy()
        print(f"  *** New best: {best_score:.6f} ***")
    
    # Try a second LP run with different initialization
    if time.monotonic() < deadline - 120:
        a1b = make_chirp_init(n1, 0.4 + 0.2 * rng.random(), rng)
        a1b = a1b / np.sum(a1b)
        time_p1b = min(200.0, (deadline - time.monotonic()) * 0.25)
        a_best_b, s_best_b = lp_optimize(a1b, time_p1b, verbose=False, rng=rng)
        actual_b = local_score(a_best_b)
        print(f"  phase1b: score={actual_b:.6f}")
        if actual_b < best_score:
            best_score = actual_b
            best_seq = a_best_b.copy()
            print(f"  *** New best: {best_score:.6f} ***")
    
    # ==================== Phase 2: Try other lengths ====================
    print("\n=== Phase 2: Other sequence lengths ===")
    for n2 in [320, 384, 512]:
        if time.monotonic() > deadline - 100:
            break
        time_p2 = min(120.0, (deadline - time.monotonic()) * 0.2)
        if time_p2 < 30:
            break
        
        a2 = make_optimized_init(n2, rng)
        a2 = a2 / np.sum(a2)
        a_best2, s_best2 = lp_optimize(a2, time_p2, verbose=False, rng=rng)
        actual2 = local_score(a_best2)
        print(f"  n={n2}: score={actual2:.6f}")
        if actual2 < best_score:
            best_score = actual2
            best_seq = a_best2.copy()
            print(f"  *** New best: {best_score:.6f} ***")
    
    # ==================== Phase 3: Subgradient refinement ====================
    print("\n=== Phase 3: Subgradient refinement ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        time_p3 = min(150.0, (deadline - time.monotonic()) * 0.3)
        a_sub, s_sub = subgradient_refine(best_seq, time_p3, verbose=True, rng=rng)
        actual_sub = local_score(a_sub)
        print(f"  Subgradient: score={actual_sub:.6f}")
        if actual_sub < best_score:
            best_score = actual_sub
            best_seq = a_sub.copy()
            print(f"  *** New best: {best_score:.6f} ***")
    
    # ==================== Phase 4: Local search polish ====================
    print("\n=== Phase 4: Local search polish ===")
    if best_seq is not None and time.monotonic() < deadline - 30:
        time_p4 = min(200.0, deadline - time.monotonic() - 15.0)
        print(f"  Time available for polish: {time_p4:.1f}s")
        a_pol, s_pol = local_search_polish(best_seq, time_p4, rng=rng)
        actual_pol = local_score(a_pol)
        print(f"  After polish: score={actual_pol:.6f}")
        if actual_pol < best_score:
            best_score = actual_pol
            best_seq = a_pol.copy()
            print(f"  *** New best: {best_score:.6f} ***")
    
    # ==================== Phase 5: One more LP pass on best ====================
    if best_seq is not None and time.monotonic() < deadline - 30:
        time_p5 = min(100.0, deadline - time.monotonic() - 20.0)
        if time_p5 > 30:
            print(f"\n=== Phase 5: Final LP pass (time={time_p5:.0f}s) ===")
            a_final, s_final = lp_optimize(best_seq, time_p5, verbose=False, rng=rng)
            actual_final = local_score(a_final)
            print(f"  Final LP: score={actual_final:.6f}")
            if actual_final < best_score:
                best_score = actual_final
                best_seq = a_final.copy()
                print(f"  *** New best: {best_score:.6f} ***")
    
    # ==================== Final verification ====================
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        # Scale up so sum > 0.01 (safety, though our sequences sum to 1*something)
        final_sum = sum(final_list)
        if final_sum < 1.0:
            scale = 2.0 / final_sum
            final_list = [x * scale for x in final_list]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)}, sum={sum(final_list):.4f} ===")
        return final_list
    
    # Fallback
    if start_seq is not None:
        fl = [float(max(0.0, min(1000.0, x))) for x in start_seq]
        return fl
    return [1.0] * 256
```
