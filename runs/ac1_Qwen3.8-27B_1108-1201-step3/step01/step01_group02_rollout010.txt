**Iterative flattening algorithm:**
1. Start with a (e.g., raised cosine).
2. Compute F = FFT(a, L) for L >= 2n-1.
3. Compute P = |F|^2 / sum(|F|^2) (normalized power spectrum).
4. The "excess" at frequency k is P[k] - 1/L.
5. To reduce the peak of the time-domain convolution, we need to reduce the peak of P.
6. Update: F_new[k] = F[k] / sqrt(P[k] + eps) (this equalizes the power spectrum).
7. a_new = IFFT(F_new). Clip to non-negative: a_new = max(0, a_new).
8. Normalize a_new.
9. Repeat.

This is similar to the "Parselither" algorithm or "greedy spectral flatness maximization."

def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def smooth_objective_and_grad(a, tau):
    """Compute smooth approximation of 2n*max(conv)/sum^2 and its gradient."""

    E = np.sum(np.exp(conv_shifted))

    log_sum_exp = tau * conv.max() + np.log(np.sum(np.exp(tau * (conv - conv.max()))))
    
    # Objective: 2n * log_sum_exp / tau / S^2
    obj = 2.0 * n * log_sum_exp / (tau * S**2)

def spectral_flatten(a, n_iters=100, eps=1e-10, verbose=False):
    """Iteratively flatten the power spectrum of a."""
    n = len(a)
    L = 1
    while L < 2*n-1:
        L *= 2
    F = np.fft.rfft(a, L)
    for it in range(n_iters):
        P = np.abs(F)**2
        P_max = P.max()
        if P_max < eps:
            break
        # Equalize: divide by sqrt(P + eps)
        F_new = F / np.sqrt(P + eps)
        a_new = np.fft.irfft(F_new, L)[:n]
        a_new = np.maximum(0, a_new)
        s = a_new.sum()
        if s < eps:
            break
        a_new = a_new / s
        # Check for improvement
        if verbose and it % 20 == 0:
            score = score_sequence(a_new)
            print(f"      SF iter {it}: score={score:.6f}")
        a = a_new
    return a


def smooth_optimize(a, time_limit_s=60.0, tau_start=10.0, tau_end=1000.0, n_anneal=10, verbose=True):
    """Optimize using smooth approximation with tau annealing."""
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    if verbose:
        print(f"    SmoothOpt start: score={best_score:.6f}, n={n}")
    
    for phase in range(n_anneal):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break
        tau = tau_start * (tau_end / tau_start) ** (phase / max(1, n_anneal-1))
        
        def obj_wrap(x):
            return smooth_objective_and_grad(x, tau)[0]
        
        def grad_wrap(x):
            return smooth_objective_and_grad(x, tau)[1]
        
        remaining = time_limit_s - elapsed
        if remaining < 5:
            break
        
        try:
            res = minimize(obj_wrap, a, jac=grad_wrap, method='L-BFGS-B',
                          bounds=[(0, None)]*n,
                          options={'maxiter': 200, 'maxfun': 500,
                                   'ftol': 1e-12, 'gtol': 1e-
8})
            if res.x is not None:
                a_new = normalize_nonneg(res.x)
                score = score_sequence(a_new)
                if score < best_score:
                    best_score = score
                    best_a = a_new.copy()
                    a = a_new.copy()
                    if verbose:
                        print(f"    SmoothOpt phase {phase} (tau={tau:.1f}): score={score:.6f}")
            else:
                a = normalize_nonneg(a)
        except Exception as e:
            if verbose:
                print(f"    SmoothOpt phase {phase} error: {e}")
        
        # Update a regardless (use the result from optimizer)
        a = normalize_nonneg(a)
    
    # Final evaluation
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    if verbose:
        print(f"    SmoothOpt done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def build_conv_matrix(a):
    n = len(a)
    n_rows = 2 * n - 1
    # For the constraint conv(a, g)_k <= M:
    # conv(a,g)_k = sum_j a[j] * g[k-j] (valid j)
    # = sum_j C[k,j] * g[j] where C[k,j] = a[k-j] if 0<=k-j<n else 0
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    # C[k, j] = a[k-j] if 0 <= k-j < n
    # For entry (k, j): k-j is the index into a, valid when 0<=k-j<n
    # So C[k,j] = a[k-j] if 0<=k-j<n, else 0
    # We need to build this sparsely.
    # For each j, the non-zero entries are at k = j, j+1, ..., j+n-1 (where k-j = 0,1,...,n-1)
    # C[j+m, j] = a[m] for m=0,...,n-1
    
    col_idx = []
    row_idx = []
    data = []
    for j in range(n):
        for m in range(n):
            k = j + m
            if k < 2*n-1:
                col_idx.append(j)
                row_idx.append(k)
                data.append(a[m])
    
    return csr_matrix((data, (row_idx, col_idx)), shape=(2*n-1, n))


def solve_lp_step(a, M, timeout=30.0):
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def lp_refine(a, time_limit_s=120.0, verbose=True):
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    stagnant = 0
    
    for it in range(500):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break
        conv = np.convolve(a, a)
        score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        if stagnant > 30:
            break
        M = conv.max()
        remaining = time_limit_s - elapsed
        if remaining < 5.0:
            break
        timeout = max(2.0, min(30.0, remaining * 0.3))
        g = solve_lp_step(a, M, timeout=timeout)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-10:
            break
        g = g / sum_g
        alpha = min(0.5, max(0.05, 1.0 / (1.0 + 0.2 * it)))
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = normalize_nonneg(a_new)
        new_score = score_sequence(a_new)
        if new_score <= score + 1e-14:
            a = a_new
        else:
            a = (1.0 - alpha*0.3) * a + (alpha*0.3) * g
            a = normalize_nonneg(a)
    
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    if verbose:
        print(f"    LP done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score
