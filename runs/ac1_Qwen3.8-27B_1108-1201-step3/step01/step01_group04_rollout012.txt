    # Use next power of 2 for FFT
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    A = np.fft.rfft(a, fft_size)
    C = np.fft.irfft(A * A, fft_size)[:size]

def _grad_hard_max(a):

    n = len(a)
    c = np.convolve(a, a)
    m_star = int(np.argmax(c))
    grad = np.zeros(n)

    hi = min(n - 1, m_star)

    return a * (target / S)

    best_score = _compute_obj(a)
    best = a.copy()
    S_t = float(np.sqrt(2.0 * n))

    # Phase 1: Gradient descent with momentum (hard max)
    v = np.zeros(n)
    beta = 0.85
    for it in range(max_grad):
        if time.monotonic() > deadline - 90.0:
            break
        lr = 0.12 * 0.5 * (1.0 + np.cos(np.pi * it / max_grad))
        lr = max(lr, 0.002)

        grad, _ = _grad_hard(a)
        grad -= grad.mean()

        v = beta * v + (1.0 - beta) * grad
        a = a - lr * v
        a = np.maximum(a, 0.0)
        a = _normalize(a, S_t)

        if (it + 1) % 500 == 0:
            s = _compute_obj(a)
            if s < best_score:
                best_score = s
                best = a.copy()

    s = _compute_obj(a)
    if s < best_score:
        best_score = s
        best = a.copy()

    # Phase 2: Soft gradient refinement (annealed tau)
    v = np.zeros(n)
    for it in range(max_soft):
        if time.monotonic() > deadline - 60.0:
            break
        tau = 10.0 + 50.0 * (it / max_soft)
        lr = 0.06 * 0.5 * (1.0 + np.cos(np.pi * it / max_soft))
        lr = max(lr, 0.002)

        grad = _grad_soft(a, tau)
        grad -= grad.mean()

        v = 0.85 * v + 0.15 * grad
        a = a - lr * v
        a = np.maximum(a, 0.0)
        a = _normalize(a, S_t)

        if (it + 1) % 300 == 0:
            s = _compute_obj(a)
            if s < best_score:
                best_score = s
                best = a.copy()

    s = _compute_obj(a)
    if s < best_score:
        best_score = s
        best = a.copy()

    # Phase 3: LP-based block coordinate descent
    if n <= 1500:
        for lp_it in range(max_lp):
            if time.monotonic() > deadline - 15.0:
                break

            c = np.convolve(a, a)
            M = float(np.max(c))
            S_cur = float(np.sum(a))

            t0_lp = time.monotonic()
            g = _solve_lp(a, M)
            lp_time = time.monotonic() - t0_lp

            if g is None:
                break

            g_sum = float(np.sum(g))
            if g_sum < S_cur * 0.8:
                # LP couldn't find a significantly better solution
                break

            # Try different blend factors (full replacement first)
            improved = False
            for t in [1.0, 0.9, 0.7, 0.5, 0.3]:
                a_new = (1.0 - t) * a + t * g
                a_new = np.maximum(a_new, 0.0)
                a_new = _normalize(a_new, S_t)
                s_new = _compute_obj(a_new)
                if s_new < best_score:
                    a = a_new
                    best = a.copy()
                    best_score = s_new
                    improved = True
                    break

            if not improved:
                # Try small perturbation escapes
                for attempt in range(3):
                    rng_local = np.random.default_rng(lp_it * 7 + attempt)
                    a_pert = a + 0.02 * rng_local.standard_normal(n)
                    a_pert = np.maximum(a_pert, 0.0)
                    a_pert = _normalize(a_pert, S_t)
                    s_pert = _compute_obj(a_pert)
                    if s_pert < best_score:
                        a = a_pert
                        best = a_pert.copy()
                        best_score = s_pert
                        improved = True
                        break
                if not improved:
                    # Small gradient step
                    grad, _ = _grad_hard(a)
                    grad -= grad.mean()
                    a = a - 0.015 * grad
                    a = np.maximum(a, 0.0)
                    a = _normalize(a, S_t)
                    s = _compute_obj(a)
                    if s < best_score:
                        best = a.copy()
                        best_score = s

    # Final check
    s = _compute_obj(a)
    if s < best_score:
        best_score = s
        best = a.copy()

    return best, best_score

    except (NameError, TypeError, Exception):

        starts.append(0.4 + 0.4 * np.sin(2.0 * np.pi * i * 5.0 / n + 0.5))
        starts.append(np.exp(-0.5 * ((i - n / 2.0) / (n * 0.22)) ** 2) + 0.1)
        starts.append(np.abs(np.sin(2.0 * np.pi * i * 2.0 / n)) * 0.4 + 0.3)

    warm_count = 0
    try:
        if gs.size >= 8:

        warm_count = 0

    # Prioritize shorter sequences for quick evaluation, but keep warm starts first
    starts = sorted(starts[warm_count:], key=len)

    print(f"Starting search: {len(full_starts)} initial sequences, budget={float(budget_s):.0f}s")
