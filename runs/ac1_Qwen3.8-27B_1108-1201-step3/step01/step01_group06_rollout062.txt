def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 1 << (2*n-1).bit_length()
    A = fft(a, size)
    c = np.real(ifft(A * A))
    return c[:2*n-1]

def objective_and_gradient(x, n, alpha, target_sum=None):
    """Compute the objective and gradient."""
    a = np.maximum(x, 0.0)
    S = a.sum()
    if S < 1e-10:

    ac = alpha * c
    ac_max = ac.max()
    exp_ac = np.exp(ac - ac_max)

    if sum_exp < 1e-300:
        return 1e10, np.zeros_like(x)
    
    smooth_val = (ac_max + np.log(sum_exp)) / alpha
