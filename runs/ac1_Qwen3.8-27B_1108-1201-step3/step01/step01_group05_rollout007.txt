import time
import numpy as np
from scipy.fft import fft, ifft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def convolve_self(p):
    """Compute full self-convolution of p."""
    n = len(p)
    L = next_fast_len(2*n - 1)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2*n-1]

def score_of(p):
    """Compute the evaluation score for a normalized sequence."""
    n = len(p)
    s = np.sum(p)
    if s < 1e-12:
        return np.inf
    c = convolve_self(p)
    return 2.0 * n * np.max(c) / (s * s)

def normalize(p):
    p = np.maximum(p, 0)
    s = np.sum(p)
    if s < 1e-12:
        return p / max(n, 1)  # fallback
    return p / s

def lp_best_response(p, M):
    """
    Solve: max sum(g) s.t. (p*g)[k] <= M for all k, g >= 0.
    Returns the optimal g.
    """
    n = len(p)
    # Build constraint matrix A (2n-1) x n
    # A[k, j] = p[k-j] if 0 <= k-j < n, else 0
    rows = []
    cols = []
    data = []
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            data.append(p[k - j])
    
    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs', options={'disp': False})
    
    if res.status != 0:
        return None
    return res.x

"""
Optimizes non-negative step function to minimize 2*n*max(conv(p,p))/sum(p)^2.
Uses iterative LP best-response with line search, combined with gradient polishing.

import time
import numpy as np
from scipy.fft import fft, ifft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def _conv_self(p):
    n = len(p)
    L = next_fast_len(2*n - 1)
    F = fft(p, L)
    return np.real(ifft(F * F))[:2*n-1]

def _score(p):
    n = len(p)
    s = p.sum()
    if s < 1e-10:
        return np.inf
    c = _conv_self(p)
    return float(2.0 * n * c.max() / (s*s))

def _norm(p):
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-300:
        return np.ones_like(p) / len(p)
    return p / s

def _build_lp_matrix(p):
    n = len(p)
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j0 = max(0, k - n + 1)
        j1 = min(k, n - 1)
        js = np.arange(j0, j1 + 1)
        rows.extend([k] * len(js))
        cols.extend(js.tolist())
        vals.extend((p[k - js]).tolist())
    return csr_matrix((np.array(vals), (np.array(rows), np.array(cols))), shape=(2*n-1, n))

def _lp_solve(p, M):
    n = len(p)
    A = _build_lp_matrix(p)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs',
                  options={'disp': False, 'presolve': True})
    if res.status != 0 or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.maximum(g, 0)
    if g.sum() < 1e-300:
        return None
    return g / g.sum()

def _lp_iterate(p, max_iters, time_limit, label=""):
    p = _norm(p.copy())
    best_p = p.copy()
    best_sc = _score(p)
    t_start = time.monotonic()
    t = 0.5  # mixing parameter
    
    for it in range(max_iters):
        if time.monotonic() - t_start > time_limit:
            break
        M = float(_conv_self(p).max())
        g = _lp_solve(p, M)
        if g is None:
            t *= 0.5
            continue
        # Line search over mixing parameter
        best_t = t
        best_trial_sc = best_sc
        for tt in [t, 0.8, 0.5, 0.3, 0.15, 0.07]:
            trial = _norm((1-tt)*p + tt*g)
            sc = _score(trial)
            if sc < best_trial_sc:

                best_t = tt

        sc = _score(p)

            best_sc = sc
            best_p = p.copy()
            t = best_t  # keep the step size that worked

        if (it+1) % 10 == 0:
            print(f"  {label} iter {it+1}: score={best_sc:.10f} M={M:.8f} t={t:.3f}", flush=True)
    
    return best_p, best_sc

"""
Minimizes 2*n*max(conv(p,p))/sum(p)^2 for non-negative p.
Uses iterative LP best-response method with line search.
Key insight: the LP finds the "best response" g that maximizes sum(g)
subject to the cross-correlation bound, and iterating converges to a
fixed point with flat-top autocorrelation.
"""

import time
import numpy as np
from scipy.fft import fft, ifft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _conv_self(p):
    """Full self-convolution via FFT."""
    n = len(p)
    if n == 0:
        return np.empty(0)
    L = next_fast_len(2 * n - 1)
    F = fft(p, L)
    return np.real(ifft(F * F))[:2 * n - 1]


def _score(p):
    """Compute evaluation score."""
    n = len(p)
    s = float(p.sum())
    if s < 1e-300 or n == 0:
        return np.inf
    c = _conv_self(p)
    return float(2.0 * n * float(c.max()) / (s * s))


def _norm(p):
    """Normalize to sum 1, clamp negatives."""
    p = np.asarray(p, dtype=np.float64)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if s < 1e-300:
        n = max(1, len(p))
        return np.full(n, 1.0 / n)
    return p / s


def _build_conv_matrix(p):
    """Build the (2n-1) x n convolution matrix for p."""
    n = len(p)
    i = np.arange(n, dtype=np.int64)[:, None]
    j = np.arange(n, dtype=np.int64)[None, :]
    rows = (i + j).ravel()
    cols = j.ravel()
    vals = np.broadcast_to(p, (n, n)).ravel()
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_best_response(p, M, time_hint=30.0):
    """Solve LP: max sum(g) s.t. conv_matrix(p) @ g <= M, g >= 0."""
    n = len(p)
    A = _build_conv_matrix(p)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'disp': False, 'time_limit': time_hint})
    except Exception:
        return None
    if res.status not in (0, 3) or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s < 1e-300:
        return None
    return g / s

def _make_gaussian(n, sigma_ratio=0.3):
    """Gaussian-shaped sequence of length n."""
    x = np.linspace(0, 1, n)
    mu = 0.5
    sigma = sigma_ratio
    p = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    return _norm(p)


def _make_raised_cosine(n, beta=1.0):
    """Raised cosine window."""
    x = np.linspace(-beta * np.pi, beta * np.pi, n)
    p = np.sinc(x / np.pi) * np.cos(x / (2.0 * beta)) / (1.0 - 2.0 * x / ((2.0 * beta * np.pi)) ** 2)
    p = np.where(np.abs(x) > 2 * beta * np.pi, 0.0, p)
    return _norm(np.abs(p))


def _make_beta_dist(n, alpha, beta_param):
    """Beta distribution shape."""
    x = np.linspace(1e-6, 1 - 1e-6, n)
    p = np.power(x, alpha) * np.power(1 - x, beta_param)
    return _norm(p)


def _interpolate_sequence(p, new_n):
    """Interpolate sequence to new length."""
    old_n = len(p)
    if old_n == new_n:
        return p.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    p_new = np.interp(x_new, x_old, p)
    return _norm(p_new)
