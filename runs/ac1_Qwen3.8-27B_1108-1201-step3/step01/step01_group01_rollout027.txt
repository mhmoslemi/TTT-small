```python
    n = len(a)
    N = 1 << (2*n - 2).bit_length()
    fa = np.fft.fft(a, N)
    return np.fft.ifft(fa * fa).real[:2*n-1]

    """Create LSE objective and gradient function for softmax parameterization."""
    def func_and_grad(x):

        xm = x - np.max(x)
        ex = np.exp(xm)
        S = np.sum(ex)
        a = ex / S

        return f, grad_x
```

```python
"""
LSE (Log-Sum-Exp) continuation method with softmax parameterization and L-BFGS
to minimize the peak autocorrelation of a non-negative sequence.
Uses adaptive alpha scheduling, multi-start, and final subgradient polishing.
"""

import numpy as np
from scipy.optimize import minimize
import time


def _alongest_power_of_2(m):
    """Return smallest power of 2 >= m."""
    return 1 << max(0, (m - 1).bit_length())


def _fft_autocorr(a, N):
    """Full autocorrelation via FFT. N >= 2*len(a)-1."""
    fa = np.fft.fft(a, N)
    return np.fft.ifft(fa * fa).real[:2 * len(a) - 1]


def _lse_fg(x, alpha, n, N1, N2):
    """LSE objective and gradient w.r.t. log-softmax parameters x."""
    # Softmax
    xm = x - np.max(x)
    ex = np.exp(xm)
    S = np.sum(ex)
    a = ex / S

    # Autocorrelation
    fa = np.fft.fft(a, N1)
    b = np.fft.ifft(fa * fa).real[:2 * n - 1]

    # LSE value
    bmax = np.max(b)
    ab = np.clip(alpha * (b - bmax), -500.0, 500.0)
    eab = np.exp(ab)
    Z = np.sum(eab)
    fval = bmax + np.log(Z) / alpha

    # Gradient w.r.t. a
    w = eab / Z
    fw = np.fft.fft(w, N2)
    fa2 = np.fft.fft(a, N2)
    cross = np.fft.ifft(np.conj(fw) * fa2).real[:n]
    grad_a = (2.0 / alpha) * cross

    # Gradient w.r.t. x (softmax chain rule)
    a_dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - a_dot)

    return fval, grad_x


def _optimize_lse(x0, n, alphas, deadline, max_iter=300, verbose=False):
    """
    Run LSE continuation optimization.
    Returns (best_a_normalized, best_max_b).
    """
    N1 = _alongest_power_of_2(2 * n - 1)
    N2 = _alongest_power_of_2(3 * n - 2)

    x = x0.copy().astype(np.float64)
    best_a = None
    best_max_b = np.inf

    for alpha in alphas:
        if time.monotonic() >= deadline:
            break

        # Track best within this alpha stage
        stage_best_a = None
        stage_best_max_b = np.inf

        def fg(xx):
            return _lse_fg(xx, alpha, n, N1, N2)

        def callback(xx):
            nonlocal stage_best_a, stage_best_max_b
            xm2 = xx - np.max(xx)
            a2 = np.exp(xm2) / np.sum(np.exp(xm2))
            b2 = _fft_autocorr(a2, N1)
            mb2 = np.max(b2)
            if mb2 < stage_best_max_b:
                stage_best_max_b = mb2
                stage_best_a = a2.copy()
            return False  # don't stop

        try:
            res = minimize(fg, x, method='L-BFGS', jac=True, callback=callback,
                           options={'maxiter': max_iter, 'gtol': 1e-10, 'ftol': 1e-12})
            x = res.x
        except Exception:
            pass

        # Final check
        xm = x - np.max(x)
        a = np.exp(xm) / np.sum(np.exp(xm))
        b = _fft_autocorr(a, N1)
        mb = np.max(b)

        if mb < stage_best_max_b:
            stage_best_a = a
            stage_best_max_b = mb

        if stage_best_a is not None and stage_best_max_b < best_max_b:
            best_a = stage_best_a.copy()
            best_max_b = stage_best_max_b

        if verbose:
            print(f"  alpha={alpha:.1f}: max_b={best_max_b:.10f}, score={2*n*best_max_b:.6f}")

    return best_a, best_max_b


def _subgradient_polish(a, n, deadline, eta0=0.01, max_iter=5000):
    """
    Subgradient polish on the actual max(b) objective.
    a is normalized (sum=1, a>=0).
    """
    N1 = _alongest_power_of_2(2 * n - 1)
    a = a.copy()
    best_a = a.copy()
    best_max_b = np.max(_fft_autocorr(a, N1))
    eta = eta0

    for it in range(max_iter):
        if time.monotonic() >= deadline:
            break

        b = _fft_autocorr(a, N1)
        bmax = np.max(b)
        if bmax < best_max_b:
            best_max_b = bmax
            best_a = a.copy()

        # Find top peaks
        k_max = np.argmax(b)
        # Gradient at the peak
        # grad_a[j] = 2 * a[k_max - j] for 0 <= k_max - j < n
        shift = k_max
        grad_a = np.zeros(n)
        lo = max(0, shift - n + 1)
        hi = min(n - 1, shift)
        for j in range(n):
            idx = shift - j
            if 0 <= idx < n:
                grad_a[j] += 2.0 * a[idx]

        # Center for sum constraint
        grad_a -= np.mean(grad_a)
        gnorm = np.linalg.norm(grad_a)
        if gnorm < 1e-15:
            break

        # Update
        a = a - eta * grad_a / gnorm

        # Project onto simplex {x >= 0, sum(x) = 1}
        # Using the water-filling algorithm
        v = np.sort(a)[::-1]
        cssv = np.cumsum(v)
        rho = np.max(np.nonzero(v * np.arange(1, n + 1) > (cssv - 1.0))[0]) + 1
        theta = (cssv[rho - 1] - 1.0) / rho
        a = np.maximum(a - theta, 0.0)

        # Decay eta
        eta *= 0.9999

    return best_a, best_max_b


def _make_two_level(n, eps):
    """Two-level anti-symmetric sequence, normalized."""
    a = np.ones(n)
    a[:n // 2] = 1.0 + eps
    a[n // 2:] = 1.0 - eps
    a = np.maximum(a, 0)
    S = np.sum(a)
    return a / S


def _make_smooth(n, eps):
    """Smooth cosine profile."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * np.cos(2.0 * np.pi * i / (n - 1))
    a = np.maximum(a, 0)
    S = np.sum(a)
    return a / S


def _project_simplex(v, target=1.0):
    """Project onto {x >= 0, sum(x) = target}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.full(n, target / n)
    rho = idx[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
```
