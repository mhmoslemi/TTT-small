    """
    Optimize a sequence to minimize the peak autocorrelation.
    Uses log-sum-exp smoothing with annealed temperature and Adam optimizer.
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n, np.inf
    a = a / S  # Normalize to sum = 1

    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.1  # Learning rate (will be adjusted)
    t = 0
    
    # Temperature schedule
    T = T_init
    
    best_a = a.copy()
    best_score = _compute_score(a)
    
    while time.monotonic() < deadline:
        t += 1
        # Anneal temperature
        T = T_init * (T_final / T_init) ** (t / n_iters_target)
        T = min(max(T, T_init), T_final)
        
        # Compute convolution
        b = np.convolve(a, a)
        b_max = np.max(b)
        cur_score = 2.0 * n * b_max

            best_score = cur_score
            best_a = a.copy()
        
        # Compute smooth gradient
        # log-sum-exp for stability
        b_shifted = b - b_max
        exp_Tb = np.exp(T * b_shifted)

        grad = _compute_grad(a, w)

        v_hat = v / (1 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project onto simplex {x >= 0, sum(x) = 1}
