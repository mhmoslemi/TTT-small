LP-based iterative optimization for minimizing the normalized peak autocorrelation
of non-negative step functions. Uses linear programming to find optimal update
directions, combined with gradient-based polishing and simulated annealing.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def build_constraint_A(f):
    """Build constraint matrix for LP. A[k,j] = f[k-j] if 0<=k-j<n."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A

def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. f*g <= M, g >= 0. Returns g or None."""
    n = len(f)
    A = build_constraint_A(f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)

def gradient_polish(f_init, time_limit, verbose=False):
    """Adam gradient descent on soft-max parameterization to minimize soft-max of convolution."""
    n = len(f_init)
    S = np.sqrt(2.0 * n)
    theta = np.log(np.clip(f_init, 1e-10, None) / np.sum(f_init))
    
    momentum1 = np.zeros(n)
    momentum2 = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 1.0
    step = 0
    best_theta = theta.copy()

    t0 = time.monotonic()
    tau = 5.0 / n
    
    while time.monotonic() - t0 < time_limit:
        # Compute a from theta
        t_shifted = theta - np.max(theta)
        e = np.exp(t_shifted)
        a = e / np.sum(e) * S
        
        # Compute soft-max of conv(a,a) and its gradient
        conv = fft_convolve(a, a)
        # Soft-max
        conv_max = np.max(conv)
        shifted_conv = (conv - conv_max) / tau
        exp_conv = np.exp(shifted_conv)
        w = exp_conv / np.sum(exp_conv)
        value = conv_max + tau * np.log(np.sum(exp_conv))
        
        # Gradient of soft-max(conv(a,a)) w.r.t. a
        # d/da[soft_max(conv(a,a))] = 2 * conv(a[::-1], w)[n-1 : 2n-1]
        rev_a = a[::-1]
        corr = fft_convolve(rev_a, w)
        g_a = 2.0 * corr[n-1:2*n-1]
        
        # Gradient w.r.t. theta (through softmax)
        inner = np.dot(a, g_a) / S
        g_theta = a * (g_a - inner)
        
        step += 1
        momentum1 = beta1 * momentum1 + (1 - beta1) * g_theta
        momentum2 = beta2 * momentum2 + (1 - beta2) * (g_theta ** 2)
        m1_hat = momentum1 / (1 - beta1 ** step)
        m2_hat = momentum2 / (1 - beta2 ** step)
        effective_lr = lr / np.sqrt(1 + 0.001 * step)
        theta -= effective_lr * m1_hat / (np.sqrt(m2_hat) + eps)
        
        # Anneal tau
        elapsed_frac = (time.monotonic() - t0) / time_limit
        tau = (5.0 / n) * (0.2 / 5.0) ** min(1.0, elapsed_frac)
        
        a_check = e / np.sum(e) * S  # Wait, I need to recompute
        t_shifted = theta - np.max(theta)
        e_new = np.exp(t_shifted)
        a_check = e_new / np.sum(e_new) * S
        score = local_score(a_check)
        
        if score < best_score:
            best_score = score
            best_theta = theta.copy()
        
        if verbose and step % 500 == 0:
            print(f"    Grad: step={step}, score={score:.6f}, best={best_score:.6f}, tau={tau:.4f}")
    
    t_shifted = best_theta - np.max(best_theta)
    a_best = np.exp(t_shifted) / np.sum(np.exp(t_shifted)) * S
    return a_best, local_score(a_best)

def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def build_constraint_A(f):
    """Build constraint matrix A for the LP. A[k,j] = f[k-j] if 0<=k-j<n."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. A@f conjugate with g <= M, g >= 0."""
    n = len(f)
    A = build_constraint_A(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                     method='highs', options={'presolve': True, 'time_limit': 30})
    if result.success:
        return result.x
    return None


def lp_optimize(f_init, time_limit, t_init=0.3, verbose=False):
    """LP-based iterative optimization with adaptive step size."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    sum_f = np.sum(f)
    best_f = f.copy()
    best_score = local_score(f)
    t0 = time.monotonic()
    t = t_init
    stagnant = 0
    iteration = 0
    rng_local = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(f, f)
        M = np.max(conv)

        g = solve_lp(f, M)
        if g is None:
            break

        sum_g = np.sum(g)
        if sum_g < 1e-10:
            break

        g_rescaled = g * (sum_f / sum_g)

        f_new = (1.0 - t) * f + t * g_rescaled
        f_new = np.clip(f_new, 0.0, None)
        sum_new = np.sum(f_new)
        if sum_new < 1e-10:
            t *= 0.5
            continue
        f_new = f_new * (sum_f / sum_new)

        score_new = local_score(f_new)
        iteration += 1

        if score_new < best_score - 1e-12:
            f = f_new
            best_score = score_new
            best_f = f.copy()
            stagnant = 0
            # Slightly increase step if consistently improving
            if stagnant == 0 and t < 0.8:
                t = min(0.8, t * 1.05)
        else:
            f = f_new  # Accept anyway to avoid getting stuck
            stagnant += 1
            if stagnant > 20:
                t *= 0.7
                stagnant = 0
            if stagnant > 80:
                # Add perturbation to escape
                f = f + 0.02 * rng_local.standard_normal(n)
                f = np.clip(f, 0.0, None)
                f = f * (sum_f / np.sum(f))
                t = 0.2
                stagnant = 0

        if verbose and iteration % 50 == 0:
            print(f"    LP iter={iteration}, t={t:.3f}, score={local_score(f):.6f}, "
                  f"best={best_score:.6f}, elapsed={time.monotonic()-t0:.1f}s")

    return best_f, best_score


def gradient_polish(f_init, time_limit, verbose=False):
    """Adam gradient descent on log-softmax parameterization."""
    n = len(f_init)
    S = np.sqrt(2.0 * n)
    f_safe = np.clip(f_init, 1e-10, None)
    theta = np.log(f_safe / np.sum(f_safe))

    momentum1 = np.zeros(n)
    momentum2 = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    base_lr = 0.8
    step_count = 0
    best_theta = theta.copy()
    best_score = local_score(f_init)
    t0 = time.monotonic()
    tau_start = 8.0 / n
    tau_end = 0.3 / n

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** elapsed_frac

        t_shifted = theta - np.max(theta)
        e = np.exp(t_shifted)
        a = e / np.sum(e) * S

        conv = fft_convolve(a, a)
        conv_max = np.max(conv)
        shifted_conv = (conv - conv_max) / tau
        exp_conv = np.exp(shifted_conv)
        sum_exp = np.sum(exp_conv)
        w = exp_conv / sum_exp

        rev_a = a[::-1]
        corr = fft_convolve(rev_a, w)
        g_a = 2.0 * corr[n - 1:2 * n - 1]

        inner = np.dot(a, g_a) / S
        g_theta = a * (g_a - inner)

        step_count += 1
        momentum1 = beta1 * momentum1 + (1.0 - beta1) * g_theta
        momentum2 = beta2 * momentum2 + (1.0 - beta2) * (g_theta ** 2)
        m1_hat = momentum1 / (1.0 - beta1 ** step_count)
        m2_hat = momentum2 / (1.0 - beta2 ** step_count)
        lr = base_lr / np.sqrt(1.0 + 0.01 * step_count)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)

        t_shifted = theta - np.max(theta)
        e_new = np.exp(t_shifted)
        a_check = e_new / np.sum(e_new) * S
        score = local_score(a_check)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_theta = theta.copy()

        if verbose and step_count % 200 == 0:
            print(f"    Grad step={step_count}, score={score:.6f}, "
                  f"best={best_score:.6f}, tau={tau:.5f}")

    t_shifted = best_theta - np.max(best_theta)
    a_best = np.exp(t_shifted) / np.sum(np.exp(t_shifted)) * S
    return a_best, local_score(a_best)


def sa_polish(f_init, time_limit, rng):
    """Simulated annealing for final polishing."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    sum_f = np.sum(f)
    best_f = f.copy()
    best_score = local_score(f)
    current_score = best_score
    t0a = time.monotonic()
    temperature = 0.003
    cooling = 0.99975
    iter_count = 0

    while time.monotonic() - t0a < time_limit:
        candidate = f.copy()
        r = rng.random()

        if r < 0.35:
            conv = fft_convolve(f, f)
            peak_idx = int(np.argmax(conv))
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = f[i] * f[j]
            top_k = max(3, n // 10)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.035, 0.012, size=top_k))
            other_idx = np.setdiff1d(np.arange(n), top_idx)
            if len(other_idx) > 0:
                n_inc = min(6, len(other_idx))
                inc_idx = rng.choice(other_idx, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.035, 0.012, size=n_inc))
        elif r < 0.6:
            width = max(1, int(rng.integers(1, max(2, n // 6))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.025, size=width))
        elif r < 0.8:
            for _ in range(2):
                width = max(1, int(rng.integers(1, max(2, n // 12))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.02, size=width))
        elif r < 0.9:
            # Asymmetric shift
            shift = int(rng.integers(1, max(2, n // 20)))
            candidate = np.roll(f, shift)
            candidate *= np.exp(rng.normal(0.0, 0.005, size=n))
        else:
            candidate *= np.exp(rng.normal(0.0, 0.006, size=n))

        candidate = np.clip(candidate, 0.0, None)
        cand_sum = np.sum(candidate)
        if cand_sum < 1e-10:
            continue
        candidate = candidate * (sum_f / cand_sum)

        new_score = local_score(candidate)
        iter_count += 1

        if new_score < current_score:
            f = candidate
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = candidate.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            f = candidate
            current_score = new_score

        temperature *= cooling

    return best_f, best_score


def make_chirp_init(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.04 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_init(n, rng, spread=0.3):
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multitone_init(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for _ in range(6):
        k = int(rng.integers(1, max(2, n // 3)))
        amp = rng.uniform(0.05, 0.2)
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_provided_init(provided_seq, n, rng):
    if len(provided_seq) == n:
        return provided_seq.copy()
    x_old = np.linspace(0, 1, len(provided_seq))
    x_new = np.linspace(0, 1, n)
    a = np.interp(x_new, x_old, provided_seq)
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase LP-based optimization with gradient and SA polishing.
    Phase 1: LP iterative optimization on multiple initializations/lengths.
    Phase 2: Gradient-based polishing (annealed soft-max + Adam).
    Phase 3: Simulated annealing for fine-tuning.
    Phase 4: Final LP refinement if time permits.
    """
    t_global = time.monotonic()
    deadline = t_global + float(budget_s) - 25.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check provided sequence
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s_prov = local_score(provided_seq)
        print(f"Provided sequence: n={len(provided_seq)}, score={s_prov:.6f}")
        if s_prov < best_score:
            best_score = s_prov
            best_seq = provided_seq.copy()

    # === Phase 1: LP-based optimization ===
    print("\n" + "=" * 60)
    print("Phase 1: LP-based iterative optimization")
    print("=" * 60)

    # Define search configurations: (n, init_func_name, time_frac)
    n_prov = len(provided_seq) if provided_seq is not None else 256

    search_configs = [
        (n_prov, 'provided', 0.30),
        (n_prov, 'chirp', 0.15),
        (n_prov, 'multitone', 0.10),
        (512, 'chirp', 0.15),
        (512, 'random', 0.08),
        (384, 'chirp', 0.08),
        (768, 'random', 0.07),
        (n_prov, 'random', 0.07),
    ]

    total_frac = sum(c[2] for c in search_configs)
    available_time = deadline - time.monotonic()

    for n, init_name, frac in search_configs:
        if time.monotonic() > deadline - 30:
            break

        time_alloc = max(15.0, frac / total_frac * available_time * 0.85)
        time_alloc = min(time_alloc, deadline - time.monotonic() - 10)
        if time_alloc < 10:
            break

        # Generate initialization
        if init_name == 'provided' and provided_seq is not None:
            a0 = make_provided_init(provided_seq, n, rng)
        elif init_name == 'chirp':
            alpha = 0.3 + 0.4 * rng.random()
            a0 = make_chirp_init(n, alpha, rng)
        elif init_name == 'multitone':
            a0 = make_multitone_init(n, rng)
        else:
            a0 = make_random_init(n, rng, 0.2 + 0.3 * rng.random())

        print(f"\n  n={n}, init='{init_name}', alloc={time_alloc:.0f}s")
        t_start = time.monotonic()

        f_best, score = lp_optimize(a0, time_alloc, t_init=0.3, verbose=False)
        elapsed = time.monotonic() - t_start
        print(f"  LP done: score={score:.6f}, time={elapsed:.1f}s")

        if score < best_score:
            best_score = score
            best_seq = f_best.copy()
            print(f"  *** New best: {best_score:.6f} (n={n}, init={init_name}) ***")

    # === Phase 2: Gradient polish ===
    print("\n" + "=" * 60)
    print("Phase 2: Gradient-based polishing")
    print("=" * 60)

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        grad_time = min(250.0, remaining * 0.45)
        if grad_time > 20:
            print(f"  Polishing n={len(best_seq)}, time={grad_time:.0f}s")
            a_grad, score_grad = gradient_polish(best_seq, grad_time, verbose=False)
            print(f"  Gradient polish result: {score_grad:.6f}")
            if score_grad < best_score:
                best_score = score_grad
                best_seq = a_grad.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # === Phase 3: SA polish ===
    print("\n" + "=" * 60)
    print("Phase 3: Simulated annealing polish")
    print("=" * 60)

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        sa_time = min(200.0, remaining * 0.5)
        if sa_time > 15:
            print(f"  SA polish, n={len(best_seq)}, time={sa_time:.0f}s")
            a_sa, score_sa = sa_polish(best_seq, sa_time, rng)
            print(f"  SA result: {score_sa:.6f}")
            if score_sa < best_score:
                best_score = score_sa
                best_seq = a_sa.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # === Phase 4: Final LP refinement ===
    print("\n" + "=" * 60)
    print("Phase 4: Final LP refinement")
    print("=" * 60)

    if best_seq is not None and time.monotonic() < deadline - 40:
        remaining = deadline - time.monotonic()
        final_lp_time = min(100.0, remaining * 0.7)
        if final_lp_time > 10:
            print(f"  Final LP, n={len(best_seq)}, time={final_lp_time:.0f}s")
            a_final, score_final = lp_optimize(best_seq, final_lp_time, t_init=0.2)
            print(f"  Final LP result: {score_final:.6f}")
            if score_final < best_score:
                best_score = score_final
                best_seq = a_final.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'=' * 60}")
        print(f"FINAL RESULT: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'=' * 60}")
        return final_list

    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]
    return [1.0] * 256
