"""
Softmax-smoothed gradient descent for minimizing the normalized autocorrelation
peak of a non-negative sequence. Uses FFT-based cross-correlation for efficient
gradient computation, adaptive temperature cooling, and multi-start strategy
with diverse initializations including the parent sequence and smooth profiles.
"""

import numpy as np
import time


def _fft_cross_correlate(w, a):
    """
    Compute g[j] = sum_{l=0}^{n-1} a[l] * w[l+j] for j = 0, ..., n-1
    using FFT. w has length 2n-1, a has length n.
    """
    n = len(a)
    # Need FFT size >= 2n-1
    L = 1
    while L < 2 * n - 1:
        L *= 2
    
    a_pad = np.zeros(L)
    a_pad[:n] = a
    w_pad = np.zeros(L)
    w_pad[:2*n-1] = w
    
    A = np.fft.rfft(a_pad)
    W = np.fft.rfft(w_pad)
    
    result = np.fft.irfft(np.conj(A) * W)
    return result[:n]


def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_max_b(a):
    """Compute max of convolve(a, a)."""
    n = len(a)
    b = np.convolve(a, a)
    return float(np.max(b))


def _compute_score(a):
    """Compute the evaluation score (with sum=1 normalization)."""
    n = len(a)
    b = np.convolve(a, a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    return float(2.0 * n * np.max(b) / (S * S))


def _make_two_level(n, alpha):
    """Two-level anti-symmetric: first half = alpha/m, second half = (1-alpha)/m."""
    m = n // 2
    a = np.zeros(n)
    a[:m] = alpha / m
    a[m:] = (1.0 - alpha) / (n - m)
    S = np.sum(a)
    if S > 1e-10:
        a /= S
    return a


def _make_smooth_profile(n, mode='cosine', eps=0.7):
    """Create a smooth profile with given shape."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)  # t in [0, 1]
    if mode == 'cosine':
        a = 1.0 + eps * np.cos(2 * np.pi * t)
    elif mode == 'sigmoid':
        a = 1.0 + eps * (2 / (1 + np.exp(-10 * (t - 0.5))) - 1)
    elif mode == 'tent':
        a = 1.0 + eps * (1 - 2 * np.abs(t - 0.5))
    elif mode == 'v_shape':
        a = 1.0 + eps * (2 * np.abs(t - 0.5) - 1)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S > 1e-10:
        a /= S
    return a


def _make_four_level(n, eps):
    """Four-level sequence: [L+eps, L+eps/3, L-eps/3, L-eps]."""
    a = np.ones(n)
    q = n // 4
    a[:q] = 1.0 + eps
    a[q:2*q] = 1.0 + eps / 3.0
    a[2*q:3*q] = 1.0 - eps / 3.0
    a[3*q:] = 1.0 - eps
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S > 1e-10:
        a /= S
    return a


def _make_random_smooth(n, rng, num_freqs=5, max_freq=None):
    """Generate a random smooth sequence using low-frequency Fourier components."""
    if max_freq is None:
        max_freq = max(2, n // 10)
    freqs = rng.integers(1, max_freq, size=min(num_freqs, max_freq - 1))
    a = np.ones(n)
    t = np.arange(n, dtype=np.float64) / n
    for f in freqs:
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(-0.3, 0.3)
        a += amp * np.cos(2 * np.pi * f * t + phase)
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S > 1e-10:
        a /= S
    return a


def _softmax_gradient_descent(a0, deadline, rng, n, tau_init=None, tau_min=None,

    Softmax-smoothed gradient descent to minimize max_b(a) = max_k sum_i a_i*a_{k-i}

    """

    else:

    best = a.copy()

    best_score = 2.0 * n * best_max_b
    
    tau = tau_init
    eta = eta_init
    iter_count = 0
    no_improve = 0
    
    # Precompute FFT size
    L = 1
    while L < 2 * n - 1:
        L *= 2
    
    while time.monotonic() < deadline:
        iter_count += 1
        
        # Compute convolution
        b = np.convolve(a, a)
        b_max = float(np.max(b))
        cur_score = 2.0 * n * b_max
        
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Compute softmax weights (numerically stable)
        b_shifted = b - b_max
        w = np.exp(b_shifted / tau)
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            # All weights are zero, use uniform
            w = np.ones(2 * n - 1) / (2 * n - 1)
            w_sum = 1.0
        else:
            w /= w_sum
        
        # Compute gradient via FFT cross-correlation
        # grad[j] = 2 * sum_{l=0}^{n-1} a[l] * w[l+j]
        a_pad = np.zeros(L)
        a_pad[:n] = a
        w_pad = np.zeros(L)
        w_pad[:2*n-1] = w
        
        A = np.fft.rfft(a_pad)
        W = np.fft.rfft(w_pad)
        grad = 2.0 * np.fft.irfft(np.conj(A) * W)[:n]

        grad -= np.mean(grad)

        a_new = _project_simplex(a_new, 1.0)

        if iter_count % 10000 == 0:

def _perturb_best(a, rng, scale=0.05, num_freqs=3):
    """Add smooth random perturbation to a sequence."""
    n = len(a)
    t = np.arange(n, dtype=np.float64) / n
    noise = np.zeros(n)
    for _ in range(num_freqs):
        f = rng.integers(1, max(2, n // 10))
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.5, 1.0)
        noise += amp * np.cos(2 * np.pi * f * t + phase)
    noise /= (np.max(np.abs(noise)) + 1e-15)
    a_new = a + scale * noise
    a_new = _project_simplex(a_new, 1.0)
    return a_new
