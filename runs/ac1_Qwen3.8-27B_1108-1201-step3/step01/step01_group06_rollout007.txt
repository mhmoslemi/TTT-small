"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Key insight: The sine-ramp a[j] = 1 - cos(pi*j/(n-1)) achieves objective ~1.0
(theoretical lower bound is 2n/(2n-1) ~ 1). We use this as the primary
initialization and refine with subgradient-based optimization (mirror descent
with momentum, adaptive step sizing, top-k subgradients, and periodic
perturbations). Multiple structured initializations are tried to avoid local
minima, and the globally best sequence is polished in the final time budget.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution of a (length 2*n-1) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, top_indices, c):
    """Weighted subgradient of max(c) using top-k active lags."""
    n = len(a)
    grad = np.zeros(n, dtype=np.float64)
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total <= 1e-15:
        return grad
    weights = top_vals / total
    for w_val, k in zip(weights, top_indices):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += w_val * 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad


def get_top_k(c, k):
    """Get indices of the k largest values in c."""
    n = len(c)
    k = min(k, n)
    if n <= k:
        return np.arange(n)
    return np.argpartition(c, -k)[-k:]


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2*n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        a = np.ones(n, dtype=np.float64)
        s = n
    return a * (S / s)


def run_mirror_descent(a_init, n, max_time, rng, top_k=30, eta_init=0.3,
                       beta_mom=0.85, perturb_interval=300, perturb_prob=0.15):
    """Mirror descent (multiplicative weights) with momentum and perturbations."""
    S = np.sqrt(2.0 * float(n))
    a = safe_normalize(a_init, n)
    c = fft_convolve_self(a)
    best_score = float(c.max())
    best_a = a.copy()

    start = time.monotonic()
    t = 0
    eta = eta_init
    momentum = np.zeros(n, dtype=np.float64)
    stall_count = 0
    stall_best = best_score

    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur = float(c.max())
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
            stall_count = 0
            stall_best = best_score
        else:
            stall_count += 1

        top_indices = get_top_k(c, top_k)
        grad = compute_subgradient(a, top_indices, c)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-13:
            break

        grad_unit = grad / gnorm
        momentum = beta_mom * momentum + (1.0 - beta_mom) * grad_unit

        a_new = a * np.exp(-eta * momentum)
        a_new = np.maximum(a_new, 1e-18)
        s = a_new.sum()
        if s <= 1e-15:
            a = np.full(n, S / n, dtype=np.float64)
        else:
            a = a_new * (S / s)

        eta = eta_init / (1.0 + 0.005 * t) ** 0.6

        if t % perturb_interval == 0 and rng.random() < perturb_prob:
            noise_scale = max(0.01, 0.1 / (1.0 + 0.01 * t)) * S / n
            noise = rng.normal(0.0, noise_scale, n)
            a = project_simplex(a + noise, S)

        if stall_count > 1000:
            noise_scale = 0.05 * S / n
            noise = rng.normal(0.0, noise_scale, n)
            a = project_simplex(a + noise, S)
            stall_count = 0

    return best_a, best_score


def run_adam(a_init, n, max_time, rng, top_k=30, lr_init=0.3):
    """Adam with simplex projection and cosine decay."""
    S = np.sqrt(2.0 * float(n))
    a = safe_normalize(a_init, n)
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    c = fft_convolve_self(a)
    best_score = float(c.max())
    best_a = a.copy()
    start = time.monotonic()
    t = 0
    total_iters_est = 20000

    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur = float(c.max())
        if cur < best_score:
            best_score = cur
            best_a = a.copy()

        top_indices = get_top_k(c, top_k)
        grad = compute_subgradient(a, top_indices, c)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-13:
            break
        grad = grad / gnorm

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        progress = float(t) / max(total_iters_est, 1)
        lr = lr_init * 0.5 * (1.0 + np.cos(np.pi * min(progress, 1.0)))
        a = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps), S)

        if t % 500 == 0 and rng.random() < 0.1:
            noise = rng.normal(0.0, 0.02 * S / n, n)
            a = project_simplex(a + noise, S)

    return best_a, best_score


def run_polish(a_init, n, max_time, rng, top_k=8):
    """Fine polishing with very small steps, focusing on the top lags."""
    if max_time < 1.0:
        return a_init, float(fft_convolve_self(a_init).max())
    S = np.sqrt(2.0 * float(n))
    a = safe_normalize(a_init, n)
    c = fft_convolve_self(a)
    best_score = float(c.max())
    best_a = a.copy()
    start = time.monotonic()
    t = 0
    eta = 0.05
    momentum = np.zeros(n, dtype=np.float64)
    beta_mom = 0.95

    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur = float(c.max())
        if cur < best_score:
            best_score = cur
            best_a = a.copy()

        top_indices = get_top_k(c, top_k)
        grad = compute_subgradient(a, top_indices, c)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-13:
            break
        grad_unit = grad / gnorm
        momentum = beta_mom * momentum + (1.0 - beta_mom) * grad_unit

        a_new = a * np.exp(-eta * momentum)
        a_new = np.maximum(a_new, 1e-18)
        s = a_new.sum()
        if s > 1e-15:
            a = a_new * (S / s)

        eta = 0.05 / (1.0 + 0.01 * t) ** 0.4

    return best_a, best_score


def generate_initializations(n, rng):
    """Generate diverse structured initializations of length n."""
    inits = []
    j = np.arange(n, dtype=np.float64)

    # Sine ramp (primary - achieves ~1.0)
    if n > 2:
        inits.append(("sine_ramp", 1.0 - np.cos(np.pi * j / (n - 1))))

    # Linear ramp
    inits.append(("linear_ramp", 2.0 * j / max(n - 1, 1)))

    # Asymmetric sine ramp (shifted)
    inits.append(("asym_sine", 1.0 - np.cos(np.pi * j / (n - 1) * 1.2)))

    # Quadratic bump (peaked at center)
    inits.append(("quadratic_bump", 1.0 + np.sin(np.pi * j / (n - 1)) ** 2))

    # Smooth edge-ramp with plateau
    inits.append(("plateau", np.clip(2.0 * j / (0.3 * n) - 1.0, 0, 2.0) *
                 np.clip(1.0 - 2.0 * (j - 0.7 * n) / (0.3 * n), 0, 1.0) + 0.5))

    # Random smooth
    inits.append(("rand_smooth",
                  1.0 + 0.5 * np.sin(2 * np.pi * 3.0 * j / n) +
                  0.3 * np.sin(2 * np.pi * 7.0 * j / n + 0.5) +
                  np.abs(0.1 * rng.standard_normal(n))))

    # Pure random
    inits.append(("random", rng.random(n) + 0.1))

    return inits
