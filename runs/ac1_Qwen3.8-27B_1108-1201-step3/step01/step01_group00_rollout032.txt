"""
LP-guided iterative refinement combined with gradient-based optimization
for minimizing the normalized peak autocorrelation of non-negative sequences.
The LP identifies optimal mass redistribution directions, while gradient
descent with annealed soft-max provides fine-tuning. Multiple sequence
lengths and initializations are explored within the time budget.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the score (same formula as evaluate_sequence)."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def build_lp_matrix(a):
    """Build constraint matrix A for the LP.
    A[k, j] = a[k-j] if 0 <= k-j < n, else 0.
    Shape: (2n-1, n)
    """
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            # A[k, j] = a[k-j] for j in [j_start, j_end]
            # When j = j_start: a[k - j_start]
            # When j = j_end: a[k - j_end]
            # So the values are a[k-j_end], a[k-j_end+1], ..., a[k-j_start]
            A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1]
    return A


def solve_lp(a, M, maxiter=10000):
    """Solve: max sum(g) s.t. conv(a,g) <= M, g >= 0.
    Returns (g, sum_g) or (None, 0).
    """
    n = len(a)
    A = build_lp_matrix(a)
    c_obj = -np.ones(n)
    b_vec = np.full(2 * n - 1, M)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_vec,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'maxiter': maxiter})
        if res.success:
            return res.x, -res.fun
        else:
            return None, 0.0
    except Exception:
        return None, 0.0


def lp_refine(a_init, max_iter=300, time_limit=60.0):
    """Iterative LP-based refinement.
    Returns (best_a, best_score).
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(a_init.copy(), 0.0, None)
    s = np.sum(a)
    if s < 0.01:
        return a_init, np.inf
    a = a * S / s
    
    t0 = time.monotonic()
    current_peak = np.max(fft_convolve(a, a))
    best_a = a.copy()
    best_peak = current_peak
    stall_count = 0
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        g, g_sum = solve_lp(a, current_peak)
        if g is None or g_sum < 1e-10:
            break
        
        g_rescaled = g * S / g_sum
        
        # Line search
        best_t = 0.0
        best_new_peak = current_peak
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            a_new = (1 - t) * a + t * g_rescaled
            a_new = np.clip(a_new, 0.0, None)
            sn = np.sum(a_new)
            if sn < 0.01:
                continue
            a_new = a_new * S / sn
            new_peak = np.max(fft_convolve(a_new, a_new))
            if new_peak < best_new_peak - 1e-10:
                best_new_peak = new_peak
                best_t = t
        
        if best_t == 0.0:
            stall_count += 1
            if stall_count > 20:
                break
            # Try a smaller step
            t = 0.01
            a_new = (1 - t) * a + t * g_rescaled
            a_new = np.clip(a_new, 0.0, None)
            sn = np.sum(a_new)
            if sn >= 0.01:
                a_new = a_new * S / sn
                a = a_new
                current_peak = np.max(fft_convolve(a, a))
            continue
        
        a = (1 - best_t) * a + best_t * g_rescaled
        a = np.clip(a, 0.0, None)
        s = np.sum(a)
        if s < 0.01:
            break
        a = a * S / s
        current_peak = np.max(fft_convolve(a, a))
        stall_count = 0
        
        if current_peak < best_peak:
            best_peak = current_peak
            best_a = a.copy()
            if it % 50 == 0:
                print(f"    LP iter {it}: peak={current_peak:.6f}, t={best_t}")
    
    best_score = 2.0 * n * best_peak / (S * S)
    return best_a, best_score


def softmax_normalize(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def grad_soft_max_conv(a, tau):
    """Gradient of soft-max(conv(a,a), tau) w.r.t. a."""
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    value = c_max + tau * np.log(sum_exp)
    
    rev_a = a[::-1]
    corr = fft_convolve(rev_a, w)
    g = 2.0 * corr[n - 1:2 * n - 1]
    return value, g, c


def gradient_optimize(n, a_init, time_limit, rng):
    """Adam gradient descent on softmax parameters with annealed temperature."""
    S = np.sqrt(2.0 * n)
    a_init = np.clip(a_init, 0.01, None)
    theta = np.log(a_init / np.sum(a_init) + 1e-10).astype(np.float64)
    
    momentum1 = np.zeros(n)
    momentum2 = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step = 0
    
    t0 = time.monotonic()
    best_theta = theta.copy()
    best_score = np.inf
    
    tau_start = 50.0 / n
    tau_end = 0.1 / n
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = (time.monotonic() - t0) / time_limit
        tau = tau_start * (tau_end / tau_start) ** min(1.0, elapsed_frac)
        
        a = softmax_normalize(theta) * S
        value, g_a, c = grad_soft_max_conv(a, tau)
        inner = np.dot(a, g_a)
        g_theta = a * (g_a - inner / S)
        
        step += 1
        momentum1 = beta1 * momentum1 + (1 - beta1) * g_theta
        momentum2 = beta2 * momentum2 + (1 - beta2) * (g_theta * g_theta)
        m1_hat = momentum1 / (1.0 - beta1 ** step)
        m2_hat = momentum2 / (1.0 - beta2 ** step)
        lr = 1.5 / np.sqrt(step + 1)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)
        
        if step % 50 == 0:
            a_check = softmax_normalize(theta) * S
            score = local_score(a_check)
            if score < best_score:
                best_score = score
                best_theta = theta.copy()
    
    a_best = softmax_normalize(best_theta) * S
    return a_best, best_score


def polish_annealing(a_init, time_limit, rng):
    """Simulated annealing polish with diverse perturbations."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    current = np.clip(a_init.copy(), 0.01, None)
    s = np.sum(current)
    current = current * S / s
    current_score = local_score(current)
    best = current.copy()
    best_score = current_score
    
    t0 = time.monotonic()
    temperature = 0.002
    cooling = 0.9998
    
    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()
        
        if r < 0.3:
            # Targeted peak reduction
            conv = fft_convolve(current, current)
            peak_idx = int(np.argmax(conv))
            contribs = np.zeros(n)
            for i in range(min(n, peak_idx + 1)):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            top_k = max(3, n // 12)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.03, 0.01, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(10, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.03, 0.01, size=n_inc))
        elif r < 0.5:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.025, size=width))
        elif r < 0.7:
            # Multi-block
            for _ in range(3):
                width = max(1, int(rng.integers(1, max(2, n // 15))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.02, size=width))
        elif r < 0.85:
            # Partial symmetrization
            current_sym = 0.5 * (current + current[::-1])
            candidate = current_sym * np.exp(rng.normal(0.0, 0.008, size=n))
        else:
            # Small global
            candidate = current * np.exp(rng.normal(0.0, 0.008, size=n))
        
        candidate = np.clip(candidate, 0.01, 1000.0)
        sn = np.sum(candidate)
        if sn >= 0.01:
            candidate = candidate * S / sn
        new_score = local_score(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            current = candidate
            current_score = new_score
        
        temperature *= cooling
    
    return best, best_score


def make_chirp_init(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / (n + 1))
    a = np.clip(a, 0.01, None)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multi_tone_init(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for _ in range(5):
        k = int(rng.integers(1, n // 2 + 1))
        amp = rng.uniform(0.05, 0.3)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_init(n, rng, spread=0.3):
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_barker_inspired_init(n, rng):
    """Create an initialization inspired by Barker/optimal sequences."""
    # Use a "raised cosine" window modulated by a chirp
    t = np.arange(n, dtype=np.float64)
    window = 0.5 * (1 + np.cos(np.pi * (t - n/2) / (n/2)))
    chirp = np.cos(np.pi * t * (t + 1) / n)
    a = 0.5 + 0.5 * window * (1 + 0.5 * chirp)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

"""
LP-guided iterative refinement combined with multiplicative weight updates
and Adam gradient descent for minimizing the normalized peak autocorrelation
of non-negative step functions. The LP identifies optimal mass redistribution
directions, multiplicative weights provide fast local updates, and gradient
descent with annealed soft-max gives fine-tuning. Multiple sequence lengths
and initialization strategies are explored within the 1800s time budget.

import numpy as np
from scipy.optimize import linprog
import time


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the score (same formula as evaluate_sequence)."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def build_lp_matrix(a):
    """Build constraint matrix A for the LP.
    A[k, j] = a[k-j] if 0 <= k-j < n, else 0.
    Shape: (2n-1, n)
    """
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1]
    return A


def solve_lp(a, M):
    """Solve: max sum(g) s.t. conv(a,g) <= M, g >= 0.
    Returns (g, sum_g) or (None, 0).
    """
    n = len(a)
    A = build_lp_matrix(a)
    c_obj = -np.ones(n)
    b_vec = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_vec,
                      bounds=[(0, None)] * n,
                      method='highs')
        if res.success:
            return res.x, float(-res.fun)
    except Exception:
        pass
    return None, 0.0


def lp_refine(a_init, max_iter=400, time_limit=60.0):
    """Iterative LP-based refinement. Returns (best_a, best_score)."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(a_init.copy(), 0.01, None)
    s = np.sum(a)
    if s < 0.01:
        return a_init, np.inf
    a = a * S / s

    t0 = time.monotonic()
    current_peak = np.max(fft_convolve(a, a))
    best_a = a.copy()
    best_peak = current_peak
    stall_count = 0

    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break

        g, g_sum = solve_lp(a, current_peak)
        if g is None or g_sum < 1e-10:
            break

        g_rescaled = g * S / g_sum

        # Line search for best blend fraction
        best_t = 0.0
        best_new_peak = current_peak
        for t in [0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7]:
            a_new = (1 - t) * a + t * g_rescaled
            a_new = np.clip(a_new, 0.01, None)
            sn = np.sum(a_new)
            if sn < 0.01:
                continue
            a_new = a_new * S / sn
            new_peak = np.max(fft_convolve(a_new, a_new))
            if new_peak < best_new_peak - 1e-9:
                best_new_peak = new_peak
                best_t = t

        if best_t == 0.0:
            stall_count += 1
            if stall_count > 30:
                break
            continue

        a = (1 - best_t) * a + best_t * g_rescaled
        a = np.clip(a, 0.01, None)
        s = np.sum(a)
        if s < 0.01:
            break
        a = a * S / s
        current_peak = np.max(fft_convolve(a, a))
        stall_count = 0

        if current_peak < best_peak:
            best_peak = current_peak
            best_a = a.copy()

    best_score = 2.0 * n * best_peak / (S * S)
    return best_a, best_score


def multiplicative_weight_refine(a_init, max_iter=2000, time_limit=30.0, eta=0.05):
    """Multiplicative weight updates targeting peak reduction. Returns (best_a, best_score)."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(a_init.copy(), 0.01, None)
    s = np.sum(a)
    a = a * S / s

    t0 = time.monotonic()
    current_peak = np.max(fft_convolve(a, a))
    best_a = a.copy()
    best_peak = current_peak
    eta_cur = eta

    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fft_convolve(a, a)
        peak_idx = np.where(conv > current_peak * 0.95)[0]

        # Compute subgradient: g[i] = sum_{k in peak_idx} a[k-i]
        subgrad = np.zeros(n)
        for k in peak_idx:
            for i in range(min(n, k + 1)):
                j = k - i
                if 0 <= j < n:
                    subgrad[i] += a[j]

        # Multiplicative weight update
        a = a * np.exp(-eta_cur * subgrad / max(1.0, np.max(subgrad)))
        a = np.clip(a, 1e-6, None)
        s = np.sum(a)
        if s < 0.01:
            break
        a = a * S / s

        current_peak = np.max(fft_convolve(a, a))
        if current_peak < best_peak:
            best_peak = current_peak
            best_a = a.copy()

        # Anneal eta
        eta_cur *= 0.9995

    best_score = 2.0 * n * best_peak / (S * S)
    return best_a, best_score


def softmax_normalize(theta):
    """Stable softmax."""
    t = theta - np.max(theta)
    e = np.exp(t)
    return e / np.sum(e)


def grad_soft_max_conv(a, tau):
    """Gradient of soft-max(conv(a,a), tau) w.r.t. a."""
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    value = c_max + tau * np.log(sum_exp)
    rev_a = a[::-1]
    corr = fft_convolve(rev_a, w)
    g = 2.0 * corr[n - 1:2 * n - 1]
    return value, g, c


def gradient_optimize(n, a_init, time_limit, rng):
    """Adam gradient descent on softmax parameters with annealed temperature."""
    S = np.sqrt(2.0 * n)
    a_init = np.clip(a_init, 0.01, None)
    theta = np.log(a_init / np.sum(a_init) + 1e-10).astype(np.float64)

    momentum1 = np.zeros(n)
    momentum2 = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step = 0

    t0 = time.monotonic()
    best_theta = theta.copy()
    best_score = np.inf

    tau_start = 80.0 / n
    tau_end = 0.05 / n

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** elapsed_frac

        a = softmax_normalize(theta) * S
        value, g_a, c = grad_soft_max_conv(a, tau)
        inner = np.dot(a, g_a)
        g_theta = a * (g_a - inner / S)

        step += 1
        momentum1 = beta1 * momentum1 + (1 - beta1) * g_theta
        momentum2 = beta2 * momentum2 + (1 - beta2) * (g_theta * g_theta)
        m1_hat = momentum1 / (1.0 - beta1 ** step)
        m2_hat = momentum2 / (1.0 - beta2 ** step)
        lr = 2.0 / np.sqrt(step + 1)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)

        if step % 30 == 0:
            a_check = softmax_normalize(theta) * S
            score = local_score(a_check)
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_theta = theta.copy()

    a_best = softmax_normalize(best_theta) * S
    if not np.isfinite(best_score):
        best_score = local_score(a_best)
    return a_best, best_score


def polish_annealing(a_init, time_limit, rng):

    n = len(a_init)
    S = np.sqrt(2.0 * n)
    current = np.clip(a_init.copy(), 0.01, None)
    s = np.sum(current)
    current = current * S / s
    current_score = local_score(current)
    best = current.copy()

    t0 = time.monotonic()
    temperature = 0.0015
    cooling = 0.99985

    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()

        if r < 0.25:
            # Targeted: reduce peak contributors
            conv = fft_convolve(current, current)
            peak_idx = int(np.argmax(conv))
            contribs = np.zeros(n)
            for i in range(min(n, peak_idx + 1)):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            top_k = max(3, n // 10)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.025, 0.01, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(12, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.025, 0.01, size=n_inc))
        elif r < 0.45:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 6))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.02, size=width))
        elif r < 0.6:
            # Multi-block
            for _ in range(4):
                width = max(1, int(rng.integers(1, max(2, n // 12))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.015, size=width))
        elif r < 0.75:
            # Partial symmetrization
            current_sym = 0.5 * (current + current[::-1])
            candidate = current_sym * np.exp(rng.normal(0.0, 0.006, size=n))
        elif r < 0.9:
            # Swap-based: exchange mass between two blocks
            w1 = max(1, int(rng.integers(2, max(3, n // 8))))
            w2 = max(1, int(rng.integers(2, max(3, n // 8))))
            s1 = int(rng.integers(0, n - w1))
            s2 = int(rng.integers(0, n - w2))
            delta = (current[s1:s1+w1].sum() - current[s2:s2+w2].sum()) * 0.1
            candidate[s1:s1+w1] -= delta / w1
            candidate[s2:s2+w2] += delta / w2
        else:
            # Small global
            candidate = current * np.exp(rng.normal(0.0, 0.006, size=n))

        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-14)):

def make_chirp_init(n, alpha, rng):

def make_windowed_chirp_init(n, rng):

        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
            s = local_score(start_seq)
            print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
            if np.isfinite(s) and s < best_score:
                best_score = s
                best_seq = start_seq.copy()
    except Exception:
        pass

    # Define search configurations: (length, num_inits, lp_time, grad_time)
    configs = []
    used_lengths = set()

    # Provided length first
    if start_seq is not None:
        n_prov = len(start_seq)
        configs.append((n_prov, 5, 80, 25))
        used_lengths.add(n_prov)

    # Other lengths (prioritize smaller for more iterations)
    for n in [256, 320, 384, 512, 640, 768, 1024, 1280, 1536, 2048]:
        if n in used_lengths:
            continue
        est_time = 100  # rough estimate per config
        if len(configs) * est_time > time_budget * 0.65:
            break
        n_inits = 3 if n <= 512 else 2
        lp_t = 60 if n <= 512 else 40
        grad_t = 20 if n <= 512 else 15
        configs.append((n, n_inits, lp_t, grad_t))
        used_lengths.add(n)

    print(f"\n=== Phase 1: LP + MW + Gradient optimization ===")
    print(f"Configs: {[(c[0], c[1]) for c in configs]}")

    init_generators = [
        lambda n, rng: (start_seq if start_seq is not None and len(start_seq) == n else make_chirp_init(n, 0.5, rng)),
        lambda n, rng: make_chirp_init(n, 0.3 + 0.3 * rng.random(), rng),
        lambda n, rng: make_multi_tone_init(n, rng),
        lambda n, rng: make_windowed_chirp_init(n, rng),
        lambda n, rng: make_symmetric_init(n, rng),
        lambda n, rng: make_random_init(n, rng, spread=0.2 + 0.3 * rng.random()),
    ]

    for cfg_i, (n, n_inits, lp_time, grad_time) in enumerate(configs):
        if time.monotonic() > deadline - 45:
            break

        remaining = deadline - time.monotonic()
        for init_i in range(n_inits):
            if time.monotonic() > deadline - 30:
                break

            # Generate initialization
            gen_idx = min(init_i, len(init_generators) - 1)
            try:
                a0 = init_generators[gen_idx](n, rng)
            except Exception:
                a0 = make_random_init(n, rng)
            a0 = np.clip(a0, 0.01, 1000.0)

            # LP refinement
            a_lp, lp_score = lp_refine(a0, max_iter=400, time_limit=lp_time)

            # Multiplicative weight refinement from LP result
            a_mw, mw_score = multiplicative_weight_refine(
                a_lp, max_iter=3000,
                time_limit=min(15.0, max(5.0, time.monotonic() < deadline - 30 and 15.0 or 5.0)),
                eta=0.03
            )

            # Gradient polish
            remaining_grad = min(grad_time, max(3.0, deadline - time.monotonic() - 30))
            a_grad, grad_score = gradient_optimize(n, a_mw, remaining_grad, rng)

            # Pick the best
            candidates = [
                (lp_score, a_lp),
                (mw_score, a_mw),
                (grad_score, a_grad),
            ]
            candidates = [(s, a) for s, a in candidates if np.isfinite(s)]
            if candidates:
                score_local, a_local = min(candidates, key=lambda x: x[0])
            else:
                score_local = np.inf
                a_local = a0

            print(f"  n={n}, init={init_i}: LP={lp_score:.6f}, MW={mw_score:.6f}, Grad={grad_score:.6f}")

            if score_local < best_score:
                best_score = score_local
                best_seq = a_local.copy()
                print(f"    *** New best: {best_score:.6f} ***")

        print(f"  Best after n={n}: {best_score:.6f}")

    # Phase 2: Polish the overall best with annealing
    print(f"\n=== Phase 2: Annealing polish ===")
    if best_seq is not None:
        polish_time = min(150.0, max(10.0, deadline - time.monotonic() - 30))
        print(f"  Score before polish: {best_score:.6f}")
        polished, polish_score = polish_annealing(best_seq, polish_time, rng)
        if polish_score < best_score:
            best_score = polish_score
            best_seq = polished.copy()
            print(f"  After polish: {best_score:.6f}")
        else:
            print(f"  Polish didn't improve (best trail={polish_score:.6f})")

    # Phase 3: LP on polished result
    if best_seq is not None and time.monotonic() < deadline - 40:
        print(f"\n=== Phase 3: LP refinement on polished result ===")
        n = len(best_seq)
        lp_time3 = min(80.0, max(10.0, deadline - time.monotonic() - 20))
        a_lp3, lp3_score = lp_refine(best_seq, max_iter=500, time_limit=lp_time3)
        if lp3_score < best_score:
            best_score = lp3_score
            best_seq = a_lp3.copy()
            print(f"  After LP: {best_score:.6f}")
        else:
            print(f"  LP didn't improve (score={lp3_score:.6f})")

        # Final annealing
        if time.monotonic() < deadline - 15:
            final_time = min(60.0, max(5.0, deadline - time.monotonic() - 10))
            polished2, ps2 = polish_annealing(best_seq, final_time, rng)
            if ps2 < best_score:
                best_score = ps2
                best_seq = polished2.copy()
                print(f"  After final polish: {best_score:.6f}")

    # Phase 4: Try one more LP iteration on the very best
    if best_seq is not None and time.monotonic() < deadline - 20:
        n = len(best_seq)
        remaining = min(40.0, max(5.0, deadline - time.monotonic() - 10))
        a_final, final_lp_score = lp_refine(best_seq, max_iter=200, time_limit=remaining)
        if final_lp_score < best_score:
            best_score = final_lp_score
            best_seq = a_final.copy()
            print(f"\n  After final LP: {best_score:.6f}")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    a = make_chirp_init(256, 0.4, np.random.default_rng(0))
    return [float(x) for x in a]
