```python
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix
import time

def fast_score(f):
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))

def solve_minmax_lp(f):
    """
    Solve: min t s.t. (f*g)[k] <= t for all k, sum(g)=1, g>=0
    Returns (g, t) or (None, None) on failure.
    """
    n = len(f)
    # Build sparse convolution matrix A (2n-1 x n)
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                rows.append(k)
                cols.append(j)
                vals.append(f[idx])
    
    A = coo_matrix((vals, (rows, cols)), shape=(2*n-1, n)).tocsr()
    
    # LP: min t
    # Variables: [g_0, ..., g_{n-1}, t]
    # Objective: [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Constraints:
    # For each k: sum_j A[k,j]*g[j] - t <= 0
    A_ub = A.tocsc()
    # Make it (2n-1) x (n+1) with last column = -1
    import scipy.sparse as sp
    A_ub_full = sp.hstack([A_ub, sp.csc_matrix((-np.ones(2*n-1), 1))]).tocsr()
    b_ub = np.zeros(2*n-1)
    
    # Equality: sum g = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: g >= 0, t >= 0
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub_full, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    
    if res.status == 0:
        g = np.maximum(res.x[:n], 0)
        t = max(res.x[-1], 0)
        return g, t
    return None, None
```

```python
LP-guided optimization to minimize peak linear autocorrelation of non-negative

multi-start with diverse initializations, and a final polish phase.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix, hstack, csc_matrix
import time

def fast_score(f):
    """Compute score = 2*n*max(f*f)/(sum f)^2."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))

def normalize(f):
    """Clip to [0, inf) and normalize to sum 1."""
    f = np.maximum(np.asarray(f, dtype=np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(len(f)) / len(f)
    return f / s

def build_conv_matrix(f):
    """Build sparse matrix A (2n-1 x n) where A[k,j] = f[k-j] if valid."""
    n = len(f)
    js = np.repeat(np.arange(n), n)
    f_idxs = np.tile(np.arange(n), n)
    ks = js + f_idxs
    vals = f[f_idxs]
    return coo_matrix((vals, (ks, js)), shape=(2*n-1, n)).tocsr()

def solve_minmax_lp(f, time_limit=5.0):
    """
    Solve: min t s.t. (f*g)[k] <= t for all k, sum(g)=1, g>=0.
    Returns (g, t) on success, (None, None) on failure.
    """
    n = len(f)
    A = build_conv_matrix(f)
    
    # Variable vector: [g_0, ..., g_{n-1}, t]
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    # Inequality constraints: A@g - t <= 0
    A_ub = hstack([A.tocsc(), csc_matrix((-np.ones(2*n-1), 1))]).tocsr()
    b_ub = np.zeros(2*n-1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.ones(1)
    
    bounds = [(0.0, None)] * (n+1)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True,
                               'dual_feasibility_tolerance': 1e-8,
                               'primal_feasibility_tolerance': 1e-8})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0.0)
            t = max(float(res.x[n]), 0.0)
            # Verify solution quality
            if g.sum() > 0:
                g = g / g.sum()
                actual_max = np.convolve(f, g).max()
                t = max(t, actual_max)
                return g, t
    except Exception:
        pass
    return None, None

def make_initial(n, method, rng=None):
    """Create initial sequence of length n using given method."""
    if rng is None:
        rng = np.random.default_rng(0)
    if method == 'uniform':
        f = np.ones(n)
    elif method == 'cosine':
        x = np.linspace(0, np.pi, n)
        f = 0.5 * (1 + np.cos(x))
    elif method == 'asym_cos':
        x = np.linspace(0, np.pi, n)
        f = 0.2 + 0.8 * (0.5 + 0.5*np.cos(x))**1.5
    elif method == 'linear_decay':
        f = np.linspace(1.0, 0.1, n)
    elif method == 'quad':
        x = np.linspace(0, 1, n)
        f = x*(1-x)*4 + 0.05
    elif method == 'bimodal':

        f = np.exp(-((x-0.2)/0.1)**2) + np.exp(-((x-0.8)/0.1)**2) + 0.01
    elif method == 'random':
        f = rng.random(n)
    elif method == 'heavy_tail':
        f = 1.0 / (np.arange(n) + 1)
    else:
        f = np.ones(n)
    return normalize(f)

def block_perturb(f, rng, max_block=None):
    """Perturb a random contiguous block."""
    f_new = f.copy()
    n = len(f)
    if max_block is None:
        max_block = max(5, n // 20)
    bs = int(rng.integers(1, min(max_block, n) + 1))
    start = int(rng.integers(0, n - bs + 1))
    f_new[start:start+bs] *= np.exp(rng.normal(0, 0.2, bs))
    return normalize(f_new)

def full_perturb(f, rng, scale=0.1):
    """Multiplicative noise on all components."""
    f_new = f * np.exp(rng.normal(0, scale, len(f)))
    return normalize(f_new)

    """
    rng = np.random.default_rng(seed)
    f = normalize(f_init.copy())
    best_f = f.copy()
    best_sc = fast_score(f)
    
    alpha = 0.4  # step size
    stall = 0
    it = 0
    
    print(f"    LP start: score={best_sc:.6f}", flush=True)

                    stall = 0
```

```python
"""
LP-guided optimization to minimize peak linear autocorrelation of non-negative
step functions. Uses direct min-max LP for direction finding, backtracking line
search, multi-start with diverse initializations, spectral smoothing, and a
final polish phase.

Strategy:
  1. Multi-start: optimize from the parent sequence and several structured/random
     initializations at different lengths.
  2. At each iteration, solve the LP: min t s.t. (f*g)[k] <= t, sum(g)=1, g>=0.
  3. Use the LP solution as a direction with backtracking line search.
  4. Periodically perturb to escape local minima.
  5. Final polish with intensive local search.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix, hstack, csc_matrix
import time


def fast_score(f):
    """Compute score = 2*n*max(convolve(f,f))/(sum(f))^2."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))


def normalize(f):
    """Clip to non-negative and normalize to sum 1."""
    f = np.maximum(np.asarray(f, dtype=np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(len(f)) / len(f)
    return f / s


def build_conv_matrix(f):
    """Build sparse (2n-1 x n) matrix A where A[k,j] = f[k-j] if 0<=k-j<n."""
    n = len(f)
    js = np.repeat(np.arange(n), n)
    f_idxs = np.tile(np.arange(n), n)
    ks = js + f_idxs
    vals = f[f_idxs]
    return coo_matrix((vals, (ks, js)), shape=(2 * n - 1, n)).tocsr()


def solve_minmax_lp(f, time_limit=5.0):
    """
    Solve: min t s.t. (f*g)[k] <= t for all k, sum(g)=1, g>=0.
    Returns (g, t) on success, (None, None) on failure.
    """
    n = len(f)
    A = build_conv_matrix(f)

    # Variable vector: [g_0, ..., g_{n-1}, t]
    c = np.zeros(n + 1)
    c[n] = 1.0

    # Inequality: A@g - t <= 0 for each of the 2n-1 lags
    A_ub = hstack([A.tocsc(), csc_matrix((-np.ones(2 * n - 1), 1))]).tocsr()
    b_ub = np.zeros(2 * n - 1)

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.ones(1)

    bounds = [(0.0, None)] * (n + 1)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0.0)
            if g.sum() > 1e-15:
                g = g / g.sum()
                t = max(float(res.x[n]), float(np.convolve(f, g).max()))
                return g, t
    except Exception:
        pass
    return None, None


def make_initial(n, method, rng=None):
    """Create a normalized non-negative initial sequence of length n."""
    if rng is None:
        rng = np.random.default_rng(0)
    if method == 'uniform':
        f = np.ones(n)
    elif method == 'cosine':
        x = np.linspace(0, np.pi, n)
        f = 0.5 * (1.0 + np.cos(x))
    elif method == 'asym_cos':
        x = np.linspace(0, np.pi, n)
        f = 0.15 + 0.85 * (0.5 + 0.5 * np.cos(x)) ** 1.3
    elif method == 'linear_decay':
        f = np.linspace(1.0, 0.05, n)
    elif method == 'quad_bump':
        x = np.linspace(0, 1, n)
        f = 4.0 * x * (1.0 - x) + 0.05
    elif method == 'bimodal':
        x = np.arange(n, dtype=float) / n
        f = np.exp(-((x - 0.2) / 0.08) ** 2) + np.exp(-((x - 0.75) / 0.1) ** 2) + 0.01
    elif method == 'heavy_tail':
        f = 1.0 / (np.arange(n) + 1.0) ** 0.7
    elif method == 'random':
        f = rng.random(n) + 0.01
    elif method == 'flat_center':
        f = np.ones(n)
        w = n // 4
        f[:w] = 0.3
        f[-w:] = 0.3
    elif method == 'ramp':
        x = np.arange(n) / n
        f = 0.2 + 0.8 * x
    else:
        f = np.ones(n)
    return normalize(f)


def block_perturb(f, rng, max_block=None, sigma=0.2):
    """Perturb a random contiguous block multiplicatively."""
    f_new = f.copy()
    n = len(f)
    if max_block is None:
        max_block = max(5, n // 20)
    bs = int(rng.integers(1, min(max_block, n) + 1))
    start = int(rng.integers(0, max(1, n - bs + 1)))
    f_new[start:start + bs] *= np.exp(rng.normal(0, sigma, bs))
    return normalize(f_new)


def full_perturb(f, rng, scale=0.1):
    """Multiplicative Gaussian noise on all components."""
    f_new = f * np.exp(rng.normal(0, scale, len(f)))
    return normalize(f_new)


def smooth_sequence(f, sigma=0.5):
    """Apply a mild Gaussian smoothing in the spatial domain."""
    n = len(f)
    kernel_width = max(1, int(2 * sigma))
    kernel = np.exp(-0.5 * (np.arange(-kernel_width, kernel_width + 1) / sigma) ** 2)
    kernel = kernel / kernel.sum()
    f_smooth = np.convolve(f, kernel, mode='same')
    return normalize(f_smooth)


def lp_optimize(f_init, deadline, seed=0, verbose=True):
    """
    LP-based local optimization from f_init until deadline.
    Returns (best_f, best_score).
    """
    rng = np.random.default_rng(seed)
    f = normalize(f_init.copy())
    best_f = f.copy()
    best_sc = fast_score(f)

    alpha = 0.4
    stall = 0
    no_improve_time = time.monotonic()
    it = 0

    if verbose:
        print(f"    LP start: score={best_sc:.6f}, n={len(f)}", flush=True)

    while time.monotonic() < deadline:
        it += 1

        # Check for prolonged stagnation (300s without improvement)
        if time.monotonic() - no_improve_time > 300:
            if verbose:
                print(f"    Stagnant for 300s, breaking. Best={best_sc:.6f}", flush=True)
            break

        # Solve LP
        g, t_opt = solve_minmax_lp(f, time_limit=3.0)

        improved = False
        if g is not None:
            g = normalize(g)
            # Optionally smooth the LP solution to reduce oscillations
            if len(f) >= 64 and it % 5 == 0:
                g_smooth = smooth_sequence(g, sigma=1.0)
                # Try both raw and smoothed
                candidates = [g, g_smooth]
            else:
                candidates = [g]

            for g_cand in candidates:
                g_cand = normalize(g_cand)
                for a in [alpha, alpha * 0.5, alpha * 0.25, 0.1, 0.05, 0.02]:
                    cand = normalize(f + a * (g_cand - f))
                    sc = fast_score(cand)
                    if sc < best_sc - 1e-9:
                        f = cand
                        best_sc = sc
                        best_f = f.copy()
                        stall = 0
                        no_improve_time = time.monotonic()
                        improved = True
                        if verbose and (it % 50 == 0 or sc < 1.52):
                            print(f"    it={it}: {sc:.6f}", flush=True)
                        break
                if improved:
                    break

            if improved:
                alpha = min(0.7, alpha * 1.01)
                continue

        # No improvement from LP - perturb to escape
        stall += 1
        if stall <= 5:
            f = block_perturb(f, rng, max_block=max(3, len(f) // 25), sigma=0.15)
        elif stall <= 30:
            f = block_perturb(f, rng, max_block=max(5, len(f) // 15), sigma=0.25)
        elif stall <= 80:
            f = full_perturb(f, rng, scale=0.12)
            alpha = 0.4
        elif stall <= 150:
            f = full_perturb(f, rng, scale=0.2)
            alpha = 0.5
        else:
            # Major restart: blend with a fresh initialization
            fresh = make_initial(len(f), 'random', rng)
            f = normalize(0.5 * f + 0.5 * fresh)
            alpha = 0.6
            stall = 80

    if verbose:
        print(f"    LP done: best={best_sc:.6f} after {it} iters", flush=True)
    return best_f, best_sc


def polish_sequence(f_init, deadline, seed=0):
    """Intensive local search: alternate LP steps and small perturbations."""
    rng = np.random.default_rng(seed)
    f = normalize(f_init.copy())
    best_f = f.copy()
    best_sc = fast_score(f)
    print(f"  Polish start: {best_sc:.6f}", flush=True)

    while time.monotonic() < deadline:
        # Try LP-based step
        g, t_opt = solve_minmax_lp(f, time_limit=2.0)
        improved = False
        if g is not None:
            g = normalize(g)
            for a in [0.25, 0.12, 0.06, 0.03, 0.01]:
                cand = normalize(f + a * (g - f))
                sc = fast_score(cand)
                if sc < best_sc - 1e-9:
                    best_sc = sc
                    best_f = cand.copy()
                    f = cand
                    improved = True
                    print(f"  Polish LP: {best_sc:.6f}", flush=True)
                    break
        if not improved:
            # Small perturbation
            f = block_perturb(f, rng, max_block=max(2, len(f) // 50), sigma=0.08)
            sc = fast_score(f)
            if sc < best_sc - 1e-9:
                best_sc = sc
                best_f = f.copy()
                print(f"  Polish pert: {best_sc:.6f}", flush=True)

    return best_f, best_sc
```
