        weights = np.full(top_k, 1.0 / top_k)
    
    grad = np.zeros(n, dtype=np.float64)
    for w, k in zip(weights, top_indices):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += w * 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad


def run_optimization(a_init, n, S, max_time, rng, 
                      T_start=5.0, T_end=0.1,
                      top_k_start=32, top_k_end=4,
                      lr_init=0.5):
    """
    Adam optimization with annealed log-sum-exp and top-k.
    """
    a = safe_normalize(a_init, n)
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    last_improve = 0
    no_improve_count = 0
    
    total_duration = max_time
    
    while time.monotonic() - start < max_time:
        t += 1
        progress = min(1.0, (t - last_improve) / max(1, t))  # animate based on total time spent
        # Actually, use wall-clock for annealing
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / total_duration)
        
        # Anneal T and top_k
        T = T_start * (T_end / T_start) ** frac  # exponential annealing
        top_k = max(2, int(round(top_k_start * (top_k_end / top_k_start) ** frac)))
        
        c = fft_convolve_self(a)
        cur = c.max()
        
        if cur < best_score - 1e-12:
            best_score = cur
            best_a = a.copy()
            last_improve = t
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        # Restart if stalled for too long
        if no_improve_count > 3000 and t > 500:
            noise_scale = 0.05 * S / np.sqrt(n)
            a = safe_normalize(a + rng.normal(0.0, noise_scale, n), n)
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            no_improve_count = 0
            # Keep best_a as is
        
        # Compute gradient
        grad = compute_gradient(a, c, top_k, T)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        
        # Adam step
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        # Learning rate with slow decay
        lr = lr_init / (1.0 + 0.0005 * t)
        
        a = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps), S)

def coordinate_refinement(a, n, S, max_time, rng):
    """
    Fast coordinate-level refinement using incremental convolution updates.
    """
    a = safe_normalize(a, n)
    c = fft_convolve_self(a)
    best_score = c.max()
    best_a = a.copy()
    start = time.monotonic()
    
    # We'll try random small perturbations of single coordinates
    # and use the incremental update formula:
    # If a[i] changes by delta, then c[k] changes by 2*delta*a[k-i] for 0<=k-i<n
    
    n_steps = 0
    base_sum = a.sum()
    
    while time.monotonic() - start < max_time:
        i = rng.integers(0, n)
        # Try a small change
        magnitude = rng.choice([0.01, 0.005, 0.001]) * S / n
        sign = rng.choice([-1.0, 1.0])
        delta = sign * magnitude
        
        old_val = a[i]
        new_val = old_val + delta
        if new_val < 0:
            new_val = 0.0
        actual_delta = new_val - old_val
        if abs(actual_delta) < 1e-15:
            n_steps += 1
            continue
        
        # Incremental update
        c_new = c.copy()
        lo = i
        hi = min(i + n - 1, 2 * n - 2)
        c_new[lo:hi + 1] += 2.0 * actual_delta * a[:hi - lo + 1]
        
        # Renormalize: we need sum(a) = S
        # a.sum() has changed by actual_delta
        # Scale factor: S / (S + actual_delta)
        scale = S / (base_sum + actual_delta)
        
        # The full a would be a * scale with a[i] = new_val * scale
        # But for the convolution, this is more complex.
        # For simplicity, let's just check the max of the incremental update
        # and accept if it improves, then renormalize periodically.
        
        trial_max = c_new.max()
        if trial_max < best_score - 1e-14:
            a = a.copy()
            a[i] = new_val
            c = c_new
            best_score = trial_max
            best_a = a.copy()
            base_sum += actual_delta
        
        n_steps += 1
        if n_steps % 10000 == 0:
            # Periodic renormalization
            a = safe_normalize(a, n)
            c = fft_convolve_self(a)
            if c.max() < best_score:
                best_score = c.max()
                best_a = a.copy()
            base_sum = a.sum()
    
    return best_a, best_score

"""
Minimize 2*n*max(convolve(a,a)) / sum(a)^2 over non-negative sequences.

Uses Adam optimization with annealed log-sum-exp smoothing,
top-k subgradient with decreasing k, and coordinate-level refinement.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def safe_normalize(a, n):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        a = np.ones(n, dtype=np.float64)
        s = float(n)
    return a * (S / s)


def compute_gradient(a, c, top_k, T):
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    if T > 0:
        top_vals = c[top_indices]
        top_vals_shifted = top_vals - top_vals[0]
        weights = np.exp(top_vals_shifted / T)
        weights /= weights.sum()
    else:
        weights = np.full(top_k, 1.0 / top_k)
    grad = np.zeros(n, dtype=np.float64)
    for w, k in zip(weights, top_indices):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += w * 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad
