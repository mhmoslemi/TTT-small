def stochastic_refine(a, time_limit_s=120.0, seed=0, verbose=True):
    """Stochastic local search with diverse perturbation strategies."""
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    if verbose:
        print(f"    Stoch start: score={best_score:.6f}")
    
    temp = 0.01
    accept_count = 0
    
    while time.monotonic() - start_time < time_limit_s:
        # Choose perturbation type
        ptype = rng.random()
        
        if ptype < 0.3:
            # Block perturbation
            width = max(2, int(rng.integers(5, max(6, n // 10))))
            start = int(rng.integers(0, max(1, n - width)))
            candidate = a.copy()
            candidate[start:start+width] *= rng.uniform(0.5, 1.5)
        
        elif ptype < 0.5:
            # Gaussian perturbation
            candidate = a * np.exp(rng.normal(0, 0.05, n))
        
        elif ptype < 0.7:
            # Swap-based: redistribute mass between two random positions
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            candidate = a.copy()
            amt = min(candidate[i], candidate[j]) * rng.uniform(0.1, 0.5)
            # Move mass from higher to lower to spread out
            if candidate[i] > candidate[j]:
                candidate[i] -= amt
                candidate[j] += amt
            else:
                candidate[j] -= amt
                candidate[i] += amt
        
        elif ptype < 0.85:
            # Symmetric perturbation
            i = int(rng.integers(0, n // 2))
            j = n - 1 - i
            candidate = a.copy()
            delta = rng.uniform(-0.1, 0.1) * max(a[i], a[j])
            candidate[i] += delta
            candidate[j] -= delta  # Keep symmetric-ish
        
        else:
            # Smooth perturbation (low-frequency)
            candidate = a.copy()
            freq = rng.uniform(0.5, 5.0)
            t = np.arange(n)
            modulation = 1.0 + 0.1 * np.sin(2 * np.pi * freq * t / n + rng.uniform(0, 2*np.pi))
            candidate *= modulation
        
        candidate = normalize_nonneg(candidate)
        new_score = score_sequence(candidate)
        
        # Metropolis acceptance
        delta = new_score - best_score
        if delta < 0 or rng.random() < np.exp(-delta / temp):
            a = candidate
            if new_score < best_score:
                best_score = new_score
                best_a = candidate.copy()
                accept_count += 1
                temp = max(0.001, temp * 0.99)
                if verbose and accept_count % 30 == 0:
                    print(f"    Stoch accept {accept_count}: score={best_score:.6f}")
        
        # Cool down
        temp = max(0.001, temp * 0.9999)
    
    if verbose:
        print(f"    Stoch done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")

Hybrid LP + line search + stochastic refinement to minimize peak auto-convolution.
Key improvements: proper line search, multiple M targets, diverse restarts,
symmetry exploitation, adaptive time allocation, and subgradient-based polishing.

import numpy as np
from scipy.sparse import csr_matrix, coo_matrix
from scipy.optimize import linprog

def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize_nonneg(a):
    a = np.maximum(0.0, a).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def enforce_symmetry(a):
    return 0.5 * (a + a[::-1])


def build_conv_matrix(a):
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = []
    row_idx = []
    data = []
    for k in range(n_rows):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            js = np.arange(j_min, j_max + 1)
            row_idx.extend([k] * len(js))
            col_idx.extend(js.tolist())
            data.extend(a[k - js].tolist())
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp(a, M, timeout=30.0):
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            g = np.maximum(0.0, res.x)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search(a, g, n_trials=25):

        a_new = normalize_nonneg((1.0 - alpha) * a + alpha * g)
        s = score_sequence(a_new)
        if s < best_score:
            best_score = s
            best_alpha = alpha

    # Refine around best
    lo = max(0.01, best_alpha - 0.05)
    hi = min(1.0, best_alpha + 0.05)

        s2 = score_sequence(normalize_nonneg((1.0 - m2) * a + m2 * g))

        else:
            lo = m1

def subgradient_polish(a, time_limit_s=60.0, verbose=False):
    """Polish using subgradient of the max-convolution function."""
    rng = np.random.default_rng(12345)

        it += 1
        conv = np.convolve(a, a)
        M = conv.max()
        score = 2.0 * n * M / (a.sum() ** 2)

            best_score = score
            best_a = a.copy()

        # Find active lags (within 1% of max)
        threshold = M * 0.99
        active = np.where(conv >= threshold)[0]

        # Compute subgradient: average of gradients at active lags
        # grad_k[j] = a[k-j] + a[j-k] (with out-of-bounds = 0)
        grad = np.zeros(n)
        for k in active:
            # Contribution from first factor: a[k-j] for j where 0<=k-j<n
            j_lo1 = max(0, k - n + 1)
            j_hi1 = min(n - 1, k)
            if j_lo1 <= j_hi1:

            # Contribution from second factor: a[j-k] for j where 0<=j-k<n
            j_lo2 = max(0, k)  # j-k >= 0 => j >= k
            j_hi2 = min(n - 1, k + n - 1)
            if j_lo2 <= j_hi2:
                grad[j_lo2:j_hi2+1] += a[np.arange(j_lo2, j_hi2+1) - k]

        grad /= len(active)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Step in negative subgradient direction
        a_new = a - step_size * grad / grad_norm
        a_new = normalize_nonneg(a_new)
        new_score = score_sequence(a_new)

            a = a_new

        candidate = normalize_nonneg(candidate)
        new_score = score_sequence(candidate)

        delta = new_score - best_score
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-6)):
            a = candidate
            if new_score < best_score:
                best_score = new_score
                best_a = candidate.copy()
                accept_count += 1
                temp = max(0.0005, temp * 0.995)

        temp = max(0.0005, temp * 0.9999)

    if verbose:
        print(f"    Stoch done: best={best_score:.6f}, {accept_count} accepts, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_gaussian(n, sigma_frac=0.3):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_uniform(n):
    return np.ones(n, dtype=np.float64)


def propose_candidate(seed=42, budget_s=1800):
    """
    Hybrid optimization combining LP refinement with line search, subgradient
    polishing, and stochastic local search. Uses diverse restarts and adaptive
    time allocation to escape local minima.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + max(10.0, budget_s - 25.0)

    best_score = float('inf')
    best_seq = None

    # Get starting point
    start_a = None
    try:
        hs1 = globals().get('height_sequence_1', None)
        if hs1 is not None:
            start_a = np.array(hs1, dtype=np.float64)
            start_a = normalize_nonneg(start_a)
            start_score = score_sequence(start_a)
            print(f"Loaded height_sequence_1: n={len(start_a)}, score={start_score:.6f}")
    except Exception:
        pass

    if start_a is None:
        start_a = initial_raised_cosine(500)
        print("No height_sequence_1 available, using raised cosine n=500")

    n0 = len(start_a)

    # === Phase 1: Intensive LP from best known ===
    print(f"\n{'='*60}")
    print(f"Phase 1: LP refinement from start (n={n0})")
    print(f"{'='*60}")
    remaining = deadline - time.monotonic()
    t_limit = min(400.0, remaining * 0.3)
    a_ref, score = lp_refine(start_a, time_limit_s=t_limit, verbose=True, seed=seed)
    if score < best_score:
        best_score = score
        best_seq = a_ref.copy()
        print(f"  Phase 1 result: {score:.6f}")

    # === Phase 2: Subgradient polish ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 40:
            print(f"\n{'='*60}")
            print(f"Phase 2: Subgradient polish")
            print(f"{'='*60}")
            t_limit = min(60.0, remaining * 0.15)
            a_ref, score = subgradient_polish(best_seq, time_limit_s=t_limit, verbose=True)
            if score < best_score:
                best_score = score
                best_seq = a_ref.copy()
                print(f"  Phase 2 result: {score:.6f}")

    # === Phase 3: Stochastic refinement ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 40:
            print(f"\n{'='*60}")
            print(f"Phase 3: Stochastic refinement")
            print(f"{'='*60}")
            t_limit = min(100.0, remaining * 0.2)
            a_ref, score = stochastic_refine(best_seq, time_limit_s=t_limit, seed=seed + 1, verbose=True)
            if score < best_score:
                best_score = score
                best_seq = a_ref.copy()
                print(f"  Phase 3 result: {score:.6f}")

    # === Phase 4: Diverse restarts ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 80:
            print(f"\n{'='*60}")
            print(f"Phase 4: Diverse restarts")
            print(f"{'='*60}")

            n_restarts = min(6, max(2, int(remaining / 80)))
            t_each = (remaining * 0.6) / n_restarts

            for r in range(n_restarts):
                if time.monotonic() > deadline - 30:
                    break

                n = len(best_seq)
                pert_type = r % 5

                if pert_type == 0:
                    # Gaussian noise
                    a_init = best_seq * np.exp(rng.normal(0, 0.08, n))
                elif pert_type == 1:
                    # Block swap
                    a_init = best_seq.copy()
                    w = max(10, n // 8)
                    i = int(rng.integers(0, n - 2 * w))
                    a_init[i:i+w], a_init[i+w:i+2*w] = a_init[i+w:i+2*w].copy(), a_init[i:i+w].copy()
                elif pert_type == 2:
                    # Smooth modulation
                    t = np.arange(n)
                    a_init = best_seq * (1 + 0.12 * np.sin(2 * np.pi * rng.uniform(1, 3) * t / n))
                elif pert_type == 3:
                    # Blend with different init
                    alt_init = initial_gaussian(n, sigma_frac=rng.uniform(0.2, 0.5))
                    a_init = 0.7 * best_seq + 0.3 * normalize_nonneg(alt_init)
                else:
                    # Invert and blend (exploits symmetry breaking)
                    a_inv = best_seq[::-1]
                    a_init = 0.6 * best_seq + 0.4 * a_inv

                a_init = normalize_nonneg(a_init)
                init_s = score_sequence(a_init)

                # LP refine from this perturbation
                a_ref, score = lp_refine(a_init, time_limit_s=t_each, verbose=False, seed=seed + 100 + r)
                if score < best_score:
                    best_score = score
                    best_seq = a_ref.copy()
                    print(f"  Restart {r} (type={pert_type}): NEW BEST {score:.6f}")
                else:
                    print(f"  Restart {r} (type={pert_type}): score={score:.6f} (init={init_s:.6f})")

    # === Phase 5: Try nearby lengths ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 50:
            print(f"\n{'='*60}")
            print(f"Phase 5: Length exploration")
            print(f"{'='*60}")

            n_best = len(best_seq)
            n_options = sorted(set([
                max(100, n_best - 50),
                max(100, n_best - 20),
                n_best + 20,
                min(1500, n_best + 50),
            ]))

            t_each = min(50.0, remaining * 0.12)
            for n_try in n_options:
                if n_try == n_best:
                    continue
                if time.monotonic() > deadline - 30:
                    break

                # Resample
                t_old = np.linspace(0, 1, n_best)
                t_new = np.linspace(0, 1, n_try)
                a_new = np.interp(t_new, t_old, best_seq)
                a_new = normalize_nonneg(a_new)
                init_s = score_sequence(a_new)

                a_ref, score = lp_refine(a_new, time_limit_s=t_each, verbose=False, seed=seed + 300)
                if score < best_score:
                    best_score = score
                    best_seq = a_ref.copy()
                    print(f"  n={n_try}: NEW BEST {score:.6f}")
                else:
                    print(f"  n={n_try}: score={score:.6f} (init={init_s:.6f})")

    # === Phase 6: Final intensive polish ===
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print(f"\n{'='*60}")
            print(f"Phase 6: Final polish (remaining={remaining:.0f}s)")
            print(f"{'='*60}")

            # LP
            t_lp = remaining * 0.5
            a_ref, score = lp_refine(best_seq, time_limit_s=t_lp, verbose=True, seed=seed + 999)
            if score < best_score:
                best_score = score
                best_seq = a_ref.copy()
                print(f"  Final LP: {score:.6f}")

            # Subgradient
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_ref, score = subgradient_polish(best_seq, time_limit_s=remaining * 0.4, verbose=True)
                if score < best_score:
                    best_score = score
                    best_seq = a_ref.copy()
                    print(f"  Final subgrad: {score:.6f}")

            # Quick stochastic
            remaining = deadline - time.monotonic()
            if remaining > 15:
                a_ref, score = stochastic_refine(best_seq, time_limit_s=remaining * 0.5, seed=seed + 2000, verbose=True)
                if score < best_score:
                    best_score = score
                    best_seq = a_ref.copy()
                    print(f"  Final stoch: {score:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = initial_raised_cosine(500)
        best_score = score_sequence(best_seq)

    # Final verification
    print(f"\n{'='*60}")
    print(f"FINAL: score={best_score:.6f}, n={len(best_seq)}")
    print(f"{'='*60}")
    verify_score = evaluate_sequence([float(x) for x in best_seq])
    print(f"Verified via evaluate_sequence: {verify_score:.6f}")
    print(f"Total time: {time.monotonic() - total_start:.1f}s")

    return [float(x) for x in best_seq]
