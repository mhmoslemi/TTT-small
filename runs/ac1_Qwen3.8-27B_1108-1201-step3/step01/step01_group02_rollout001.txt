"""
Hybrid LP + smooth gradient optimization for minimizing peak auto-convolution.
Key improvements: line search for step size, smooth gradient refinement,
multi-scale exploration (n=500, 1000, 2000), adaptive LP bounds,
and periodic restarts to escape local minima.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def build_conv_matrix(a):
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp(a, M, timeout=60.0):
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def line_search(a, g, num_iters=25):
    """Ternary search for best alpha in [0,1] for (1-alpha)*a + alpha*g"""
    best_alpha = 0.0
    best_score = score_sequence(a)
    lo, hi = 0.0, 1.0
    for _ in range(num_iters):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        a1 = normalize_nonneg((1 - m1) * a + m1 * g)
        a2 = normalize_nonneg((1 - m2) * a + m2 * g)
        s1 = score_sequence(a1)
        s2 = score_sequence(a2)
        if s1 < s2:
            hi = m2
            if s1 < best_score:
                best_score = s1
                best_alpha = m1
        else:
            lo = m1
            if s2 < best_score:
                best_score = s2
                best_alpha = m2
    return best_alpha, best_score


def smooth_max_gradient(a, beta):
    """Compute gradient of logsumexp(beta * conv(a,a)) / beta"""
    n = len(a)
    conv = np.convolve(a, a)
    scaled = beta * conv
    max_val = scaled.max()
    weights = np.exp(scaled - max_val)
    weights = weights / weights.sum()
    # grad[j] = sum_k weights[k] * d(conv[k])/d(a[j])
    # d(conv[k])/d(a[j]) = a[k-j] if 0<=k-j<n (from first term)
    #                     + a[k-j] if 0<=k-j<n (from second term, i=k-j)
    # = 2*a[k-j] if 0<=k-j<n, else 0
    # So grad[j] = 2 * sum_{k: 0<=k-j<n} weights[k] * a[k-j]
    # = 2 * cross_correlate(weights, a)[j]
    # Using np.correlate or direct computation
    # cross[j] = sum_{m=0}^{n-1} weights[j+m] * a[m] (with weights zero-padded to 2n-1)
    w_pad = np.zeros(2 * n - 1)
    w_pad[:] = weights
    grad = np.zeros(n)
    for j in range(n):
        valid_len = min(n, 2 * n - 1 - j)
        grad[j] = 2.0 * np.dot(w_pad[j:j + valid_len], a[:valid_len])
    return grad, (conv.max())


def smooth_gradient_refine(a, time_limit, verbose=True):
    """Projected gradient descent on smooth approximation of max(conv)"""
    start = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    beta = 50.0
    lr = 0.01
    stagnant = 0

    while time.monotonic() - start < time_limit:
        remaining = time_limit - (time.monotonic() - start)
        if remaining < 2.0:
            break

        grad, _ = smooth_max_gradient(a, beta)
        # Normalize gradient by the scale of a
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break
        grad = grad / grad_norm

        # Projected gradient step
        a_new = a - lr * grad
        a_new = np.maximum(0.0, a_new)
        a_new = normalize_nonneg(a_new)

        new_score = score_sequence(a_new)
        if new_score < best_score - 1e-14:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            stagnant = 0
            # Increase beta slowly
            if stagnant == 0 and beta < 200:
                beta *= 1.02
        else:
            stagnant += 1
            lr *= 0.9
            if stagnant > 100:
                # Restart with perturbation
                a = normalize_nonneg(a + 0.01 * np.random.default_rng(
                    int(time.time() % 10000)).random(n))
                stagnant = 0
                lr = 0.01

    return best_a, best_score

def local_search_refine(a, time_limit, rng=None, verbose=True):
    """Local search with multiplicative perturbations"""
    if rng is None:
        rng = np.random.default_rng(123)
    start = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    accept_count = 0
    step_size = 0.15

    if verbose:
        print(f"    LS start: score={best_score:.6f}")

    while time.monotonic() - start < time_limit:
        # Random segment perturbation
        width = max(2, int(rng.integers(n // 300, n // 6)))
        ss = int(rng.integers(0, max(1, n - width)))
        candidate = a.copy()
        candidate[ss:ss + width] *= np.exp(rng.normal(0.0, step_size, size=width))
        candidate = np.maximum(0, candidate)
        candidate = normalize_nonneg(candidate)

        score = score_sequence(candidate)
        if score < best_score - 1e-12:
            best_score = score
            best_a = candidate.copy()
            a = candidate
            accept_count += 1
            step_size = max(0.02, step_size * 0.98)
            if verbose and accept_count % 30 == 0:
                print(f"    LS accept {accept_count}: score={best_score:.6f}")
        else:
            step_size = min(0.3, step_size * 1.02)

    if verbose:
        print(f"    LS done: best={best_score:.6f}, {time.monotonic()-start:.1f}s")
    return best_a, best_score


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_gaussian(n, sigma_frac=0.25):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_blackman(n):
    t = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.ones(1, dtype=np.float64)
    x = t / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))


def interpolate_sequence(a, new_n):
    """Interpolate a sequence to a new length"""
    old_n = len(a)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    interpolated = np.interp(new_x, old_x, a)
    return np.maximum(0, interpolated)
