"""
Improved LP-based + subgradient optimization for minimizing peak auto-convolution
of non-negative sequences. Combines:
- Multi-start with diverse initializations
- LP-based direction finding with adaptive line search
- Subgradient polishing for local refinement
- Time-adaptive search budget allocation
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time

def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)

def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def init_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))

def init_hill(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 - np.cos(2.0 * np.pi * t / n))

def init_gaussian(n, sigma_frac=0.3):
    t = np.arange(n, dtype=np.float64)
    center = (n-1)/2.0
    sigma = sigma_frac * n
    return np.exp(-0.5*((t-center)/sigma)**2)

def init_double_bump(n, frac=0.25, width=0.1):
    t = np.arange(n, dtype=np.float64)
    c1 = frac * n
    c2 = (1-frac) * n
    sigma = width * n
    return np.exp(-0.5*((t-c1)/sigma)**2) + np.exp(-0.5*((t-c2)/sigma)**2)

def init_random_smooth(n, rng, n_bumps=3):
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = rng.integers(0, n)
        width = rng.integers(max(1, n//15), max(2, n//4))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5*((t-center)/width)**2)
    return np.maximum(0.01, a)  # Small floor to avoid zeros

def build_conv_matrix(a):
    n = len(a)
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        lo = max(0, k-n+1)
        hi = min(n, k+1)
        for j in range(lo, hi):
            rows.append(k)
            cols.append(j)
            data.append(a[k-j])
    return csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

def solve_lp_direction(a, M, time_limit=30.0):
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None

def lp_refine_advanced(a, time_limit, rng, verbose=True):
    """LP-based refinement with adaptive line search."""
    start = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    stagnant = 0
    
    for it in range(10000):
        elapsed = time.monotonic() - start
        if elapsed > time_limit:
            break
        
        conv = np.convolve(a, a)
        M = conv.max()
        score = 2.0 * n * M / (a.sum()**2)
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        if stagnant > 30:
            # Try random perturbation to escape
            if rng.random() < 0.3:
                a = normalize_nonneg(a * np.exp(rng.normal(0, 0.1, n)))
                stagnant = 10
            elif stagnant > 60:
                break
        
        remaining = time_limit - elapsed
        if remaining < 3.0:
            break
        
        lp_timeout = min(30.0, remaining * 0.4)
        b = solve_lp_direction(a, M, time_limit=lp_timeout)
        if b is None:
            break
        
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-8:
            if stagnant > 20:
                break
            stagnant += 5
            continue
        
        b_norm = b / sum_b
        
        # Line search along the LP direction
        best_t = 0.0
        best_line_score = score
        for t in [0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = normalize_nonneg((1-t)*a + t*b_norm)
            s_try = score_sequence(a_try)
            if s_try < best_line_score:
                best_line_score = s_try
                best_t = t
        
        if best_t > 0:
            a = normalize_nonneg((1-best_t)*a + best_t*b_norm)
        else:
            # Try subgradient step instead
            k_star = int(np.argmax(conv))
            g = np.zeros(n)
            valid = (k_star - np.arange(n) >= 0) & (k_star - np.arange(n) < n)
            g[valid] = 2.0 * a[k_star - np.arange(n)[valid]]
            # Remove mean (orthogonalize to constraint)
            g = g - g.mean()
            for step in [0.01, 0.05, 0.1]:

                s_try = score_sequence(a_try)
                if s_try < score:
                    a = a_try
                    break
    
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    if verbose:
        print(f"    LP+subgrad done: score={best_score:.6f}, iters={it+1}, time={time.monotonic()-start:.1f}s")
    return best_a, best_score

def subgradient_polish(a, time_limit, rng, verbose=True):
    """Pure subgradient method for final polishing."""
    start = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    accepted = 0
    
    while time.monotonic() - start < time_limit:
        conv = np.convolve(a, a)
        max_val = conv.max()
        # Find all lags where conv > 0.95 * max
        threshold = 0.95 * max_val
        active_lags = np.where(conv > threshold)[0]
        
        # Average subgradient over active lags
        g = np.zeros(n)
        for k in active_lags:
            valid = (k - np.arange(n) >= 0) & (k - np.arange(n) < n)
            g += 2.0 * a[k - np.arange(n)[valid]] * valid
        g /= len(active_lags)
        
        # Orthogonalize to sum constraint
        g = g - g.mean()
        
        # Try multiple step sizes
        score = 2.0 * n * max_val / (a.sum()**2)
        improved = False
        for step in [0.1, 0.05, 0.02, 0.01, 0.005]:
            a_try = normalize_nonneg(a - step * g)
            s_try = score_sequence(a_try)
            if s_try < best_score:
                a = a_try
                best_score = s_try
                best_a = a.copy()
                accepted += 1
                improved = True
                break
        
        if not improved:
            # Add small noise
            a = normalize_nonneg(a * np.exp(rng.normal(0, 0.05, n)))
    
    if verbose:
        print(f"    Subgrad polish: score={best_score:.6f}, accepted={accepted}, time={time.monotonic()-start:.1f}s")
    return best_a, best_score
