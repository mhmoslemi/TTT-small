def internal_score(p):
    """Score = 2n * max(conv(p,p)) / (sum(p))^2"""

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / float(k + 1)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s

def _conv_len(n):
    return int(next_fast_len(max(1, 2 * int(n) - 1)))


def _grad_len(n):
    return int(next_fast_len(max(1, 3 * int(n) - 2)))


def conv_sq(p):
    """Full self-convolution, length 2n-1."""
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = _conv_len(n)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1].astype(np.float64, copy=False)


def internal_score(p):
    """2n * max(conv(p,p)) / sum(p)^2."""
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.inf
    s = float(np.sum(p))
    if not np.isfinite(s) or s <= 1e-12:
        return np.inf
    c = conv_sq(p)
    cmax = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(cmax) or cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (s * s))


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / float(k + 1)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def safe_normalize(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if not np.isfinite(s) or p.size == 0 or s <= 1e-12:
        n = max(1, int(p.size))
        return np.full(n, 1.0 / n, dtype=np.float64)
    return p / s

def grad_at_lag(p, k):
    """Exact gradient of b[k] = sum_i p[i]*p[k-i] w.r.t. p."""
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    g = np.zeros(n, dtype=np.float64)
    j_min = max(0, k - n + 1)
    j_max = min(k, n - 1)
    if j_min <= j_max:
        idx = np.arange(j_min, j_max + 1)
        g[idx] = 2.0 * p[k - idx]
    return g

def make_uniform(n):
    return np.full(max(1, int(n)), 1.0 / max(1, int(n)), dtype=np.float64)


def make_geometric(n, s, reverse=False):
    n = max(2, int(n))
    x = np.linspace(0.0, 1.0, n)
    p = np.exp(-float(s) * x)
    if reverse:
        p = p[::-1]
    return safe_normalize(p)


def make_power(n, a, reverse=False):
    n = max(2, int(n))
    x = np.linspace(1e-5, 1.0, n)
    p = np.power(x, float(a))
    if reverse:
        p = p[::-1]
    return safe_normalize(np.maximum(p, 0.0))


def make_cup(n, a):
    n = max(2, int(n))
    x = np.linspace(0.0, 1.0, n)
    p = 1.0 + float(a) * (2.0 * x - 1.0) ** 2
    return safe_normalize(p)


def make_beta(n, alpha, beta):
    n = max(2, int(n))
    x = np.linspace(1e-5, 1.0 - 1e-5, n)
    p = np.power(x, float(alpha)) * np.power(1.0 - x, float(beta))
    return safe_normalize(np.maximum(p, 0.0))


def make_skew_gauss(n, a, mu, b):
    n = max(2, int(n))
    x = np.linspace(0.0, 1.0, n)
    p = np.exp(-float(a) * (x - float(mu)) ** 2 + float(b) * (x - 0.5))
    return safe_normalize(p)


def make_two_peak(n, mu1, sig1, w1, mu2, sig2, w2):
    n = max(2, int(n))
    x = np.linspace(0.0, 1.0, n)
    p = float(w1) * np.exp(-((x - mu1) ** 2) / (2.0 * max(1e-4, sig1) ** 2))
    p += float(w2) * np.exp(-((x - mu2) ** 2) / (2.0 * max(1e-4, sig2) ** 2))
    return safe_normalize(np.maximum(p, 0.0))


def make_random_smooth(n, rng, strength=1.0):
    n = max(2, int(n))
    x = np.linspace(0.0, 1.0, n)
    f1 = int(rng.integers(1, max(2, n // 80)))
    f2 = int(rng.integers(1, max(2, n // 40)))
    f3 = int(rng.integers(1, max(2, n // 16)))
    p = 1.0
    p += 0.5 * strength * np.sin(2.0 * np.pi * f1 * x + rng.uniform(0, 2*np.pi))
    p += 0.3 * strength * np.sin(2.0 * np.pi * f2 * x + rng.uniform(0, 2*np.pi))
    p += 0.15 * strength * np.sin(2.0 * np.pi * f3 * x + rng.uniform(0, 2*np.pi))
    p = np.maximum(p, 0.02)
    return safe_normalize(p)


def make_chebyshev_shape(n, theta, deg=10):
    """Chebyshev polynomial + endpoint log-taper parameterization."""
    n = max(2, int(n))
    x = np.linspace(0.0, 1.0, n)
    z = 2.0 * x - 1.0
    logp = np.zeros(n, dtype=np.float64)

    D = min(deg, int(theta.size) - 2)
    if D >= 1:
        T_nm = np.ones(n)
        T_n = z.copy()
        logp += theta[0] * T_n
        for d in range(2, D + 1):
            T_np = 2.0 * z * T_n - T_nm
            logp += theta[d - 1] * T_np
            T_nm = T_n
            T_n = T_np

    if theta.size > D:
        eps = 1e-4
        if theta.size > D + 1:
            logp += theta[D] * np.log(x + eps) + theta[D + 1] * np.log(1.0 - x + eps)
        else:
            logp += theta[D] * np.log(x + 1.0 - x + eps)

    if not np.all(np.isfinite(logp)):
        return make_uniform(n)
    m = float(np.max(logp))
    logp = logp - m
    p = np.exp(logp)
    return safe_normalize(p)

def generate_initials(n, rng, parent=None, extra=None):
    n = max(16, int(n))
    out = []
    if parent is not None:
        par = np.asarray(parent, dtype=np.float64)
        if par.size >= 8 and float(np.sum(par)) > 0:
            # Resize parent to n via interpolation
            par = np.where(np.isfinite(par), par, 0.0)
            par = np.maximum(par, 0.0)
            xs = np.linspace(0, 1, par.size)
            xn = np.linspace(0, 1, n)
            p_resized = np.interp(xn, xs, par)
            out.append(safe_normalize(p_resized))

    out.append(make_uniform(n))
    if extra is not None:
        for q in extra:
            if q is not None:
                q = np.asarray(q, dtype=np.float64)
                if q.size == n and float(np.sum(q)) > 0:
                    out.append(safe_normalize(q))

    for s in (0.3, 0.8, 1.5, 3.0, 6.0, 12.0):
        out.append(make_geometric(n, s, False))
        out.append(make_geometric(n, s, True))

    for a in (0.3, 0.8, 1.5, 3.0, 6.0):
        out.append(make_power(n, a, False))
        out.append(make_power(n, a, True))

    for a in (0.3, 0.8, 1.5, 3.0):
        out.append(make_cup(n, a))

    for alpha, beta in ((0.5, 2.0), (1.0, 3.0), (2.0, 0.5), (3.0, 1.0), (0.1, 4.0), (4.0, 0.1)):
        out.append(make_beta(n, alpha, beta))

    for a, mu, b in ((3.0, 0.3, 1.5), (3.0, 0.7, -1.5), (8.0, 0.3, 2.0), (8.0, 0.7, -2.0), (15.0, 0.25, 2.5), (15.0, 0.75, -2.5)):
        out.append(make_skew_gauss(n, a, mu, b))

    out.append(make_two_peak(n, 0.25, 0.06, 1.0, 0.75, 0.06, 1.0))
    out.append(make_two_peak(n, 0.15, 0.10, 1.5, 0.85, 0.05, 1.0))
    out.append(make_two_peak(n, 0.50, 0.12, 2.0, 0.05, 0.04, 1.0))
    out.append(make_two_peak(n, 0.50, 0.12, 2.0, 0.95, 0.04, 1.0))

    for _ in range(8):
        out.append(make_random_smooth(n, rng, 1.0))
    for _ in range(4):
        out.append(make_random_smooth(n, rng, 0.4))

    return out

def lp_best_response(p, best_score, budget_s=30.0):
    if not _HAVE_LP:
        return p, best_score
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n <= 0 or n > 800:
        return p, best_score
    if budget_s <= 0.5:
        return p, best_score

    c = conv_sq(p)
    M = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(M) or M <= 0:
        return p, best_score

    f = p + 1e-14
    fs = float(f.sum())
    if fs <= 0:
        return p, best_score
    f = f / fs

    try:
        rows = np.add.outer(np.arange(n), np.arange(n)).ravel().astype(np.int32)
        rows = rows - np.arange(n, dtype=np.int32)  # shift to get k-i instead of k+i
        # Actually np.convolve: b[k] = sum_i a[i]*a[k-i]
        # For constraint b[k] <= M: sum_i a[i]*a[k-i] <= M
        # If we fix a (as f) and optimize g: sum_i f[i]*g[k-i] <= M
        # This is: for each k, sum_j f[k-j]*g[j] <= M (valid j)
        # So the constraint matrix A has A[k,j] = f[k-j] if 0<=k-j<n, else 0
        
        n_ineq = 2 * n - 1
        # Build sparse matrix
        jj = np.repeat(np.arange(n), n)
        ii = np.add.outer(np.arange(n_ineq), np.arange(n)).ravel()
        # A[k, j] = f[k-j] if 0 <= k-j < n
        vals = np.empty(n_ineq * n, dtype=np.float64)
        for k in range(n_ineq):
            for j in range(n):
                kj = k - j
                if 0 <= kj < n:
                    vals[k * n + j] = f[kj]
                else:
                    vals[k * n + j] = 0.0
        A = csr_matrix((vals, (np.repeat(np.arange(n_ineq), n), jj)), shape=(n_ineq, n))
        
        res = linprog(
            -np.ones(n),
            A_ub=A,
            b_ub=np.full(n_ineq, M),
            bounds=[(0, None)] * n,
            method='highs',
            options={'disp': False, 'time_limit': max(1.0, budget_s * 0.8)}
        )
        g = getattr(res, 'x', None)
        if g is None:
            return p, best_score
        g = np.asarray(g, dtype=np.float64)
        g = np.where(np.isfinite(g), g, 0.0)
        g = np.maximum(g, 0.0)
        s = float(g.sum())
        if s <= 1e-12:
            return p, best_score
        g = g / s

        for t in (1.0, 0.7, 0.4, 0.2, 0.1):
            q = project_simplex((1 - t) * p + t * g)
            sc = internal_score(q)
            if np.isfinite(sc) and sc < best_score:
                return q, sc
    except Exception:
        pass
    return p, best_score

def optimize_local(p0, rng, time_limit, label=""):
    """
    Projected gradient descent with adaptive temperature and direct peak shaving.
    """
    p = project_simplex(np.asarray(p0, dtype=np.float64))
    n = int(p.size)
    if n == 0:
        return p, np.inf

    best_p = p.copy()
    best_score = internal_score(best_p)
    current_score = best_score
    prev_p = p.copy()

    end = time.monotonic() + max(0.0, float(time_limit))
    iter_no = 0
    since_improve = 0
    T_factor = 0.5  # Temperature relative to cmax

    while time.monotonic() < end:
        iter_no += 1

        # Compute convolution and identify active lags
        c = conv_sq(p)
        cmax = float(np.max(c))
        if not np.isfinite(cmax):
            break

        # Find active set: lags within 2% of max
        threshold = cmax * 0.98
        active_mask = c >= threshold
        active_lags = np.where(active_mask)[0]

        if active_lags.size <= 3:
            # Use exact gradient of active lags (weighted equally)
            g = np.zeros(n, dtype=np.float64)
            for k in active_lags:
                g += grad_at_lag(p, int(k))
            g /= float(active_lags.size)
        else:
            # Use smoothed max gradient
            g, _, _, _ = grad_smoothed_max(p, T_factor)

        if not np.all(np.isfinite(g)):
            g = np.zeros(n, dtype=np.float64)

        # Adaptive temperature: cool down if few active lags, warm up if stuck
        if active_lags.size <= 2:
            T_factor = max(0.02, T_factor * 0.95)
        elif since_improve > 200:
            T_factor = min(2.0, T_factor * 1.05)

        # Compute step size
        pmax = float(np.max(p))
        p_scale = max(1.0 / n, 0.03 * pmax, 1e-12)
        gabs = float(np.max(np.abs(g))) if g.size else 0.0
        if gabs > 1e-300:
            base_lr = p_scale / gabs
        else:
            base_lr = 0.1

        # Line search
        accepted = False
        lrs = (0.5 * base_lr, 0.2 * base_lr, 0.08 * base_lr, 0.03 * base_lr, 0.01 * base_lr, 0.003 * base_lr)

        for lr in lrs:
            if not np.isfinite(lr) or lr <= 0:
                continue
            lr = min(lr, 10.0)
            v = p - lr * g
            pt = project_simplex(v)
            sc = internal_score(pt)

            if np.isfinite(sc) and sc < current_score * (1 + 1e-4):
                # Accept if it improves best, or slightly worsens current (annealing)
                if sc < best_score:
                    best_score = sc
                    best_p = pt.copy()
                    since_improve = 0
                elif since_improve < 100 and sc < current_score:
                    since_improve += 1
                else:
                    since_improve += 1

                current_score = sc
                prev_p = p
                p = pt
                accepted = True
                break

        if not accepted:
            # Try direct peak shaving: target the argmax lag
            k_star = int(np.argmax(c))
            g_peak = grad_at_lag(p, k_star)
            
            # Find where to add mass (low gradient) and where to remove (high gradient)
            if n > 2 and np.all(np.isfinite(g_peak)) and np.max(g_peak) > np.min(g_peak) + 1e-15:
                # Transfer mass from high-g to low-g positions
                hi_idx = np.argsort(g_peak)[-min(5, n//10):]
                lo_idx = np.argsort(g_peak)[:min(5, n//10)]
                
                for frac in (0.1, 0.05, 0.02, 0.01):
                    moved = 0.0
                    pt = p.copy()
                    for hi in hi_idx:
                        transfer = frac * p[hi]
                        if transfer <= 0:
                            continue
                        # Distribute to low-gradient positions
                        for lo in lo_idx:
                            if lo != hi:
                                amt = transfer / max(1, len(lo_idx))
                                pt[hi] -= amt
                                pt[lo] += amt
                                moved += amt
                    if moved > 0:
                        pt = project_simplex(pt)
                        sc = internal_score(pt)
                        if np.isfinite(sc) and sc < best_score:
                            best_score = sc
                            best_p = pt.copy()
                            since_improve = 0
                            current_score = sc
                            prev_p = p
                            p = pt
                            accepted = True
                            break
                        elif np.isfinite(sc) and sc < current_score * 1.01:
                            current_score = sc
                            prev_p = p
                            p = pt
                            since_improve += 1
                            accepted = True
                            break

        if not accepted:
            # Random perturbation
            since_improve += 1
            if since_improve > 500:
                sigma = 0.5 / np.sqrt(n)
                pt = project_simplex(p + rng.normal(0, sigma, n))
                sc = internal_score(pt)
                if np.isfinite(sc) and sc < best_score * 1.05:
                    p = pt
                    current_score = sc
                    prev_p = pt
                    if sc < best_score:
                        best_score = sc
                        best_p = p.copy()
                    since_improve = 0
            elif since_improve > 200:
                # Pull back toward best
                p = project_simplex(0.7 * best_p + 0.3 * p + rng.normal(0, 0.1/np.sqrt(n), n))
                current_score = internal_score(p)
                prev_p = p
                since_improve = 50

        if iter_no % 2000 == 0:
            print("%s iter=%d best=%.10f cur=%.10f T=%.4f" % (label, iter_no, best_score, current_score, T_factor), flush=True)

    # Final
    if np.all(np.isfinite(best_p)):
        best_p = project_simplex(best_p)
    else:
        best_p = p

    return best_p, float(best_score) if np.isfinite(best_score) else np.inf

# Shape search

def search_shapes(n, rng, budget_s):
    """Global shape search using differential evolution + Nelder-Mead."""
    if budget_s <= 10:
        return None, np.inf

    deg = 10
    dim = deg + 2
    end = time.monotonic() + budget_s

    def shape_objective(theta):
        try:
            p = make_chebyshev_shape(n, theta, deg)
            sc = internal_score(p)
            return sc if np.isfinite(sc) else 1e10
        except Exception:
            return 1e10

    best_score = np.inf
    best_p = None
    best_theta = None

    # Initial evaluations
    starts = [np.zeros(dim)]
    for _ in range(5):
        starts.append(rng.uniform(-1.5, 1.5, dim))

    # Taper-focused starts
    t = np.zeros(dim)
    t[deg] = 2.0
    t[deg+1] = 2.0
    starts.append(t)

    t = np.zeros(dim)
    t[0] = 2.0
    t[1] = -1.0
    starts.append(t)

    t = np.zeros(dim)
    t[deg] = 4.0
    t[deg+1] = 4.0
    starts.append(t)

    for x0 in starts:
        sc = shape_objective(x0)
        if sc < best_score:
            best_score = sc
            best_theta = x0.copy()
            best_p = make_chebyshev_shape(n, x0, deg)

    if _HAVE_SCIPY_OPT and time.monotonic() < end - 60:
        bounds = [(-3, 3)] * deg + [(-1, 6), (-1, 6)]
        try:
            res = differential_evolution(
                shape_objective, bounds,
                maxiter=30, popsize=12, tol=0.3,
                seed=int(rng.integers(0, 2**31)),
                polish=False, init='latinhypercube'
            )
            if res is not None and np.isfinite(res.fun) and res.fun < best_score:
                best_score = res.fun
                best_theta = res.x.copy()
                best_p = make_chebyshev_shape(n, res.x, deg)
                starts.append(res.x)
        except Exception:
            pass

        for x0 in starts:
            if time.monotonic() > end - 5:
                break
            try:
                res = minimize(shape_objective, x0, method='Nelder-Mead',
                               options={'maxiter': 200, 'xatol': 1e-5, 'fatol': 1e-7})
                if res is not None and np.isfinite(res.fun) and res.fun < best_score:
                    best_score = res.fun
                    best_theta = res.x.copy()
                    best_p = make_chebyshev_shape(n, res.x, deg)
            except Exception:
                pass

    return best_p, float(best_score) if np.isfinite(best_score) else np.inf, best_theta

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for non-negative step function minimizing 2n*max(autocorr)/(sum)^2.
    Returns a list of non-negative floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 15.0)

    # Try to get parent sequence
    parent = None
    try:
        par = np.asarray(height_sequence_1, dtype=np.float64)
        par = np.where(np.isfinite(par), par, 0.0)
        par = np.maximum(par, 0.0)
        if par.ndim == 1 and par.size >= 32 and float(par.sum()) > 0:
            parent = par
            print("Using parent sequence, n=%d, score=%.8f" % (par.size, internal_score(par)), flush=True)
    except Exception:
        pass

    best_p = None
    best_score = np.inf

    def try_record(p, sc):
        nonlocal best_p, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64)
        if p.size == 0:
            return
        p = safe_normalize(p)
        if not np.isfinite(sc):
            sc = internal_score(p)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_p = p.copy()
            print("NEW BEST: score=%.10f n=%d (t=%.1fs)" % (sc, p.size, time.monotonic() - start_time), flush=True)

    # Determine which n values to try
    n_candidates = []
    if parent is not None:
        n_parent = int(parent.size)
        n_candidates.append(n_parent)
        # Try nearby sizes
        for delta in ([-100, -50, 50, 100, 200]):
            nn = n_parent + delta
            if 100 <= nn <= 5000:
                n_candidates.append(nn)
        # Also try some standard sizes
        for nn in (512, 768, 1024, 1536, 2048):
            if nn not in n_candidates:
                n_candidates.append(nn)
    else:
        n_candidates = [700, 1000, 1024, 768, 512, 1536]

    # Deduplicate and sort by proximity to 700
    n_candidates = sorted(set(n_candidates), key=lambda x: abs(x - 700))
    n_candidates = n_candidates[:6]  # Limit to 6 different sizes

    print("Trying n values: %s" % n_candidates, flush=True)

    # Phase 1: Quick evaluation of initial shapes for each n (30s)
    phase1_end = start_time + 30.0
    all_inits = []  # (score, p, n, label)

    for n in n_candidates:
        if time.monotonic() > phase1_end:
            break
        inits = generate_initials(n, rng, parent if parent is not None and abs(parent.size - n) < 200 else None)
        for p in inits:
            p = safe_normalize(p)
            sc = internal_score(p)
            if np.isfinite(sc):
                all_inits.append((sc, p, n, "init"))
    
    all_inits.sort(key=lambda x: x[0])
    for sc, p, n, _ in all_inits[:10]:
        try_record(p, sc)

    # Phase 2: Shape search for best n (up to 180s)
    shape_end = start_time + 210.0
    n_best_shape = all_inits[0][2] if all_inits else 700
    shape_p, shape_sc, shape_theta = search_shapes(n_best_shape, rng, max(5.0, shape_end - time.monotonic()))
    try_record(shape_p, shape_sc)

    # Also do shape search at a different n
    if len(n_candidates) > 1 and time.monotonic() < shape_end - 30:
        n2 = n_candidates[1]
        shape_p2, shape_sc2, _ = search_shapes(n2, rng, max(5.0, shape_end - time.monotonic() - 5))
        try_record(shape_p2, shape_sc2)

    # Phase 3: Local optimization from top candidates (majority of time)
    # Build job list
    jobs = []
    seen = set()
    for sc, p, n, lbl in all_inits[:8]:
        key = (n, round(float(p[0]), 6) if p.size > 0 else 0)
        if key not in seen:
            seen.add(key)
            jobs.append([p, n, lbl])
    if shape_p is not None:
        jobs.append([shape_p, n_best_shape, "shape"])
    
    # Also add parent (resized to each n)
    if parent is not None:
        for n in n_candidates[:3]:
            xs = np.linspace(0, 1, parent.size)
            xn = np.linspace(0, 1, n)
            p_parent = np.interp(xn, xs, parent)
            p_parent = safe_normalize(p_parent)
            jobs.append([p_parent, n, "parent"])

    # Sort by initial score
    job_scores = []
    for p, n, lbl in jobs:
        sc = internal_score(p)
        job_scores.append((sc, p, n, lbl))
    job_scores.sort(key=lambda x: x[0])

    # Allocate time to jobs
    phase3_end = deadline - 60.0  # Reserve 60s for final polish
    n_jobs = max(1, len(job_scores))
    # Give more time to better initial scores
    total_weight = sum(1.0 / (1 + 0.1 * i) for i in range(n_jobs))
    available = max(30.0, phase3_end - time.monotonic())
    
    j = 0
    while j < n_jobs and time.monotonic() < phase3_end:
        sc_init, p, n, lbl = job_scores
        if p is None or p.size == 0:
            j += 1
            continue
        remaining = phase3_end - time.monotonic()
        if remaining < 15:
            break
        # Time allocation: proportional to quality
        w = 1.0 / (1 + 0.1 * j)
        t_alloc = min(available * w / total_weight, max(30.0, remaining * 0.4))
        t_alloc = max(20.0, min(t_alloc, 400.0))

        p_best, sc_best = optimize_local(p, rng, t_alloc, lbl + ":")
        try_record(p_best, sc_best)
        
        # If we found something good, do a second pass
        if sc_best < best_score * 0.999 and time.monotonic() < phase3_end - 30:
            t2 = min(100.0, (phase3_end - time.monotonic()) * 0.3)
            if t2 > 15:
                p_best2, sc_best2 = optimize_local(p_best, rng, t2, lbl + " v2:")
                try_record(p_best2, sc_best2)

        j += 1

    # Phase 4: Final polish on the best (remaining time)
    if best_p is not None:
        remaining = deadline - 5.0 - time.monotonic()
        if remaining > 10:
            print("Final polish, remaining=%.1fs, current best=%.10f" % (remaining, best_score), flush=True)
            p_final, sc_final = optimize_local(best_p, rng, remaining, "FINAL:")
            try_record(p_final, sc_final)

    # Fallback
    if best_p is None:
        best_p = make_uniform(700)
        best_score = internal_score(best_p)

    # Ensure valid output
    best_p = np.asarray(best_p, dtype=np.float64)
    best_p = np.where(np.isfinite(best_p), best_p, 0.0)
    best_p = np.maximum(best_p, 0.0)
    best_p = np.minimum(best_p, 1000.0)
    s = float(best_p.sum())
    if s <= 1e-12:
        best_p = make_uniform(700)

    final_score = internal_score(best_p)
    print("\n=== FINAL RESULT ===")
    print("n=%d, score=%.10f, time=%.1fs" % (len(best_p), final_score, time.monotonic() - start_time), flush=True)

    return [float(x) for x in best_p]
