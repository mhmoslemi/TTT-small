def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

import numpy as np

def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def soft_max_and_weights(x, tau):
    """Compute soft-max value and corresponding softmax weights."""
    x_max = np.max(x)
    shifted = (x - x_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    value = x_max + tau * np.log(sum_exp)
    return value, weights


def grad_soft_max_conv(a, tau):
    """
    Compute value and gradient of soft_max(conv(a,a), tau) w.r.t. a.
    Gradient formula: g[j] = 2 * sum_m a[m] * w[m+j]
    Computed via: g[j] = 2 * conv(a[::-1], w)[j + n - 1]
    """
    n = len(a)
    c = fft_convolve(a, a)
    value, w = soft_max_and_weights(c, tau)
    rev_a = a[::-1]
    corr = fft_convolve(rev_a, w)
    g = 2.0 * corr[n - 1: 2 * n - 1]
    return value, g, c

# Approach and Rationale
This solution uses a multi-start gradient-based optimization algorithm. Inspired by the annealed softmax gradient descent method from the previous program, we maintain several candidates simultaneously across different sequence lengths. Each candidate is optimized using annealed softmax and rmsprop, and the best candidate is updated at each iteration. This parallel approach helps overcome local minima by restarting the optimization process for each candidate with modified target constraints, encouraging exploration. We incorporate normalization scaling and targeted perturbations to guide the optimization efficiently.

The key bottleneck addressed is that simple gradient descent might get stuck while optimization is expensive per evaluation. By funding backward scalings and alternating between varied population states, we ensure progress toward the global minimum while managing computational cost.

# Mathematical Specification
We optimize the score function:
\[
\text{score} = \frac{2 \cdot n \cdot \max(\text{convolve}(a, a))}{(\sum a)^2}
\]

- \(a_i \in [0, 1000]\)
- \(\sum a \geq 0.01\)
- \(n \in \{256, 384, 512, 768, 1024\}\)
