Or: h = conv(a, w_flipped)[2n-2 : n-2 : -1] which gives n elements from index 2n-2 down to n-1.

In code:

h = full[2*n-2 : n-2 : -1]  # n elements: indices 2n-2, 2n-3, ..., n-1
```

Let me verify: for n=3, this gives indices 4, 3, 2, which is [conv[4], conv[3], conv[2]] = [h[0], h[1], h[2]]. ✓

Great, so the gradient is 2*h.

Now let me code the full algorithm.

Actually, I realize I should also think about whether to use the log-sum-exp or a different smooth approximation. The log-sum-exp is good because:
1. It's a smooth upper bound on the max
2. The gradient is well-defined everywhere
3. The parameter β controls the sharpness

For β → ∞, it converges to the max. For β → 0, it converges to the log of the geometric mean (which is less useful).

A good strategy is to use a "continuation" approach:
1. Start with small β (smooth, easy to optimize)
2. Gradually increase β (sharper, closer to the true objective)
3. Use the current solution as the starting point for the next β

This is similar to the "homotopy" or "continuation" method.

Let me also think about the step size. With Adam, the step size is adaptive, so I don't need to worry too much. But I should make sure the learning rate is not too large (to avoid oscillations) and not too small (to avoid slow convergence).

A good starting point for the learning rate is:
lr = 0.1 * S / n

where S = sqrt(2n) is the target sum. This gives a step size of about 0.1 * sqrt(2n) / n = 0.1 * sqrt(2/n), which is O(1/sqrt(n)). For n=8192, this is about 0.0016, which seems reasonable.

Actually, with Adam, the effective step size is lr / sqrt(v_hat), where v_hat is the second moment of the gradient. If the gradient is of order 1 (which it should be, since it's a correlation of two O(1) vectors), then the effective step size is about lr. So lr = 0.1 should work.

```python
"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Uses log-sum-exp smooth approximation of max(self-conv) with Adam optimization,
combined with top-k subgradient and LP-based alternating minimization.
Multiple sequence lengths and initializations are explored.
"""

import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v.copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_gradient(a, beta):
    """Gradient of (1/beta) * log(sum exp(beta * b_k)) where b = conv(a,a)."""
    n = len(a)
    b = fft_convolve_self(a)
    # Stable softmax
    b_max = b.max()
    exp_b = np.exp(beta * (b - b_max))
    w = exp_b / exp_b.sum()  # length 2n-1, sums to 1
    
    # h[m] = sum_k w[k] * a[k-m] * [0 <= k-m < n]
    # = conv(a, w_flipped)[2n-2-m]
    w_flip = w[::-1]
    L = 1 << (3 * n - 2 - 1).bit_length()
    full = np.real(ifft(fft(a, L) * fft(w_flip, L)))[:3*n-2]
    h = full[2*n-2 : n-2 : -1]  # n elements
    grad = 2.0 * h
    return grad, b_max

def topk_subgradient(a, top_k):
    """Subgradient using top-k active lags (weighted by b[k])."""
    n = len(a)
    b = fft_convolve_self(a)
    top_idx = np.argsort(b)[::-1][:top_k]
    top_vals = b[top_idx]
    total = top_vals.sum()
    if total <= 1e-15:
        return np.zeros(n), b.max()
    weights = top_vals / total
    w_vec = np.zeros(2*n-1)
    w_vec[top_idx] = weights
    # Same computation as lse_gradient but with discrete weights
    w_flip = w_vec[::-1]
    L = 1 << (3 * n - 2 - 1).bit_length()
    full = np.real(ifft(fft(a, L) * fft(w_flip, L)))[:3*n-2]
    h = full[2*n-2 : n-2 : -1]
    grad = 2.0 * h
    return grad, b.max()

def run_lse_adam(a_init, n, S, max_time, rng, beta_init=10.0, beta_max=500.0, lr_init=0.1, beta_growth=1.05):
    """Adam with log-sum-exp smooth objective."""
    a = project_simplex(a_init.copy(), S)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    best_score = np.inf
    best_local = a.copy()
    start = time.monotonic()
    beta = beta_init
    lr = lr_init
    stall_count = 0
    stall_threshold = 5000
    last_improve = 0.0
    
    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score - 1e-10:
            best_score = cur
            best_local = a.copy()
            last_improve = time.monotonic()
            stall_count = 0
        else:
            stall_count += 1
        
        # Increase beta periodically
        if t % 200 == 0:
            beta = min(beta_max, beta * beta_growth)
        
        grad, _ = lse_gradient(a, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        a = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps), S)
        
        # Decay learning rate
        lr = lr_init * (0.8 ** (float(t) / 5000.0))
        
        # Add noise to escape local minima
        if t % 5000 == 0 and rng.random() < 0.2:
            a = project_simplex(a + rng.normal(0, 0.05 * S / np.sqrt(n), n), S)
    
    return best_local, best_score

def run_topk_adam(a_init, n, S, max_time, rng, top_k=10, lr_init=0.1):
    """Adam with top-k subgradient."""
    a = project_simplex(a_init.copy(), S)
    m = np.zeros(n)

def run_alternating_lp(a_init, n, S, max_time, top_k=5):
    """LP-based alternating minimization (simplified version).
    At each step, find the direction that reduces max_conv the most,
    using a linearized version of the constraints."""
    a = project_simplex(a_init.copy(), S)
    best_score = np.inf
    best_local = a.copy()
    start = time.monotonic()
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        M = c.max()
        if M < best_score - 1e-10:
            best_score = M
            best_local = a.copy()
        
        # Find top-k lags
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        
        # For each top lag k, the constraint is b[k] <= M.
        # The "excess" is b[k] - M (should be ~0 for active constraints).
        # To reduce b[k], we need to reduce a[m] for m where a[k-m] is large.
        # This is the same as the subgradient approach.
        
        # Use a "water-filling" step: reduce the sequence at positions
        # that contribute most to the peak lags.
        grad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
        
        # Project gradient direction
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        grad = grad / gnorm
        
        # Step
        step = 0.1 * S / n
        a_new = a - step * grad
        a = project_simplex(a_new, S)
    
    return best_local, best_score

def make_gaussian_init(n, center_frac=0.5, width_frac=0.3):
    """Create a Gaussian-shaped initialization."""
    t = np.linspace(0, 1, n)
    center = center_frac
    width = width_frac
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    # Add some structure
    a = a + 0.1 * np.sin(2 * np.pi * 3 * t) * np.exp(-0.5 * ((t - center) / (2*width)) ** 2)
    return np.maximum(a, 0)

def make_raised_cosine_init(n, alpha=0.5):
    """Raised cosine window."""
    t = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * (t - 0.5)))
    a = a ** (1 + alpha)
    return np.maximum(a, 1e-6)
