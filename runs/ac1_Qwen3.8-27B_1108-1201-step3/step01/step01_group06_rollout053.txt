def run_smoothed_mirror(a, n, S, max_time, rng, beta_init, beta_final, top_fallback=10):
    """Mirror descent with smoothed max subgradient."""
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * (S/n)
    else:
        a = a * (S / s)
    
    best_a = a.copy()
    best_score = np.inf
    start = time.monotonic()
    t = 0
    
    while time.monotonic() - start < max_time:
        t += 1
        conv = fft_convolve_self(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
        
        # Annealing schedule for beta
        progress = min(1.0, t / (max_time * 10))  # adjust
        beta = beta_init * (beta_final / beta_init) ** progress
        
        # Compute smoothed subgradient
        weights = np.exp(beta * (conv - conv.max()))
        wsum = weights.sum()
        if wsum < 1e-300:
            # Fallback to hard max
            peak_idx = np.argmax(conv)
            weights = np.zeros_like(conv)
            weights[peak_idx] = 1.0
        else:
            weights /= wsum
        
        # grad_i = 2 * sum_j a_j * w_{i+j}
        fsize = 1 << (3*n - 1).bit_length()
        A = fft(a, fsize)
        W = fft(weights, fsize)
        grad = 2.0 * np.real(ifft(A * np.conj(W)))[:n]
        
        # Mirror descent update (multiplicative weights)
        gmax = np.abs(grad).max()
        if gmax < 1e-15:
            break
        grad_normed = grad / gmax
        
        eta = 1.0 / (1.0 + t * 0.01)  # decreasing step size
        a_new = a * np.exp(-eta * grad_normed)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a = np.ones(n) * (S / n)
        else:
            a = a_new * (S / s_new)
        
        # Occasional perturbation
        if t % 100 == 0 and rng.random() < 0.3:
            noise = rng.random(n) - 0.5
            a = a + 0.01 * S / n * noise
            a = np.maximum(a, 0)
            a = a * (S / a.sum())
    
    return best_a, best_score

"""
Minimize 2*n*max(self-convolution) / (sum)^2 over non-negative sequences.

Key improvements over previous approach:
1. FFT-based cross-correlation for O(n log n) smoothed gradient computation
2. Softmax-temperature annealing for adaptive subgradient richness
3. Adam-adapted mirror descent (multiplicative weights with per-coordinate LRs)
4. Multiple diverse initializations including the given height_sequence_1
5. Periodic perturbation for escaping local minima
6. Final fine-tuning phase with very small steps and high beta (sharp gradient)
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def next_power_of_2(n):
    return 1 << (n - 1).bit_length()


def fft_convolve_self(a):
    """Full self-convolution of a (length 2*n-1) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = next_power_of_2(size)
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_cross_corr(a, w):
    """Linear cross-correlation (a ⋆ w)[m] = sum_j a[j]*w[j+m] for m=0..len(a)-1.
    a has length n, w has length 2n-1. Uses FFT with zero-padding."""
    n = len(a)
    fsize = next_power_of_2(n + len(w) - 1)
    A = fft(a, fsize)
    W = fft(w, fsize)
    return np.real(ifft(A * np.conj(W)))[:n]


def smooth_gradient(a, conv, beta):
    """Compute smoothed subgradient of max(conv) w.r.t. a using softmax weights.
    Returns grad where grad[m] = 2 * sum_j a[j] * softmax(conv)[m+j]."""
    n = len(a)
    conv_shifted = conv - conv.max()
    if beta > 1e6:
        # Hard max fallback
        peak = np.argmax(conv)
        weights = np.zeros(len(conv), dtype=np.float64)
        weights[peak] = 1.0
    else:
        weights = np.exp(beta * conv_shifted)
        wsum = weights.sum()
        if wsum < 1e-300:
            peak = np.argmax(conv)
            weights = np.zeros(len(conv), dtype=np.float64)
            weights[peak] = 1.0
        else:
            weights /= wsum
    corr = fft_cross_corr(a, weights)
    return 2.0 * corr


def run_adapted_mirror(a_init, n, max_time, rng, beta_start, beta_end,
                       lr_init, momentum=0.9, perturb_every=200, perturb_prob=0.2):
    """Adam-adapted mirror descent with smoothed max subgradient.
    Works on sequences normalized to sum=S=sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n, dtype=np.float64)
    else:
        a = a * (S / s)

    m = np.zeros(n, dtype=np.float64)  # first moment
    v = np.zeros(n, dtype=np.float64)  # second moment
    t = 0
    best_a = a.copy()
    best_score = fft_convolve_self(a).max()

    start = time.monotonic()
    total_iters = 0

    while time.monotonic() - start < max_time:
        t += 1
        total_iters += 1

        conv = fft_convolve_self(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # Anneal beta
        progress = min(1.0, t / 5000.0)
        beta = beta_start * (beta_end / beta_start) ** progress

        # Compute smoothed gradient
        grad = smooth_gradient(a, conv, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        # Adam update
        m = momentum * m + (1.0 - momentum) * grad
        v = 0.999 * v + 0.001 * (grad * grad)
        m_hat = m / (1.0 - momentum ** t)
        v_hat = v / (1.0 - 0.999 ** t)

        # Learning rate with decay
        lr = lr_init / (1.0 + 0.001 * t) ** 0.5

        # Mirror descent (multiplicative weights) update
        a_new = a * np.exp(-lr * m_hat / (np.sqrt(v_hat) + 1e-8))
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a = np.full(n, S / n, dtype=np.float64)
        else:
            a = a_new * (S / sn)

        # Periodic perturbation to escape local minima
        if t % perturb_every == 0 and rng.random() < perturb_prob:
            noise_scale = 0.05 * S / n * (1.0 + 0.5 * progress)
            noise = rng.standard_normal(n)
            a = a + noise_scale * noise
            a = np.maximum(a, 0.0)
            sn = a.sum()
            if sn > 1e-15:
                a = a * (S / sn)

    return best_a, best_score


def run_fine_tune(a_init, n, max_time, beta):
    """Very fine-tuning phase with small fixed steps and sharp gradient."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n, dtype=np.float64)
    else:
        a = a * (S / s)

    best_a = a.copy()
    best_score = fft_convolve_self(a).max()
    start = time.monotonic()
    lr = 0.1 * S / n
    m = np.zeros(n, dtype=np.float64)
    t = 0

    while time.monotonic() - start < max_time:
        t += 1
        conv = fft_convolve_self(a)
        cur = conv.max()
        if cur < best_score:
            best_score = cur
            best_a = a.copy()

        grad = smooth_gradient(a, conv, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        # Simple momentum (no adaptation, for stability)
        m = 0.9 * m + 0.1 * grad
        a_new = a * np.exp(-lr * m)
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a = np.full(n, S / n, dtype=np.float64)
        else:
            a = a_new * (S / sn)

        # Decay learning rate
        lr = 0.1 * S / n / (1.0 + 0.0005 * t)

    return best_a, best_score


def try_gaussian_init(n, rng, center_frac, width_frac):
    """Generate a Gaussian-bump initialization."""
    center = int(n * center_frac)
    width = max(1, int(n * width_frac))
    x = np.arange(n)
    a = np.exp(-0.5 * ((x - center) / width) ** 2)
    a = np.maximum(a, 0.0)
    return a


def try_two_hump_init(n, rng):
    """Two symmetric humps to reduce central convolution peak."""
    x = np.arange(n)
    w = n / 8.0
    c1 = n / 4.0
    c2 = 3.0 * n / 4.0
    a = np.exp(-0.5 * ((x - c1) / w) ** 2) + np.exp(-0.5 * ((x - c2) / w) ** 2)
    a = np.maximum(a, 1e-10)
    return a


def try_periodic_init(n, rng, period):

    a = 1.0 + 0.3 * np.sin(2.0 * np.pi * x / period) + 0.1 * np.cos(4.0 * np.pi * x / period)
    a = np.maximum(a, 0.0)
    return a
