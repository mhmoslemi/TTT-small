```python
"""
Multi-strategy optimization for minimizing normalized autocorrelation peak.
Uses mirror descent (multiplicative weights), Nesterov-accelerated subgradient,
L_p smoothing with increasing p, and diverse window-function starting points.
Time-divides among strategies and polishes the best result.
"""

import numpy as np
import time
import math


def _score(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = np.convolve(a, a)
    return 2.0 * n * float(np.max(b))


def _convolve(a):
    """Compute autocorrelation of a."""
    return np.convolve(a, a)


def _grad_peaks(a, peaks, weights):
    """
    Compute weighted gradient of sum_k w_k * b_k w.r.t. a.
    peaks: array of peak indices into b
    weights: array of weights for each peak
    """
    n = len(a)
    grad = np.zeros(n)
    for k, w in zip(peaks, weights):
        if w < 1e-12:
            continue
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            grad[idx] += w * 2.0 * a[k - idx]
    return grad


def _project_simplex(v, target=1.0):
    """Project v onto simplex {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_weights(b, peaks, temp):
    """Compute softmax weights over selected peaks."""
    vals = b[peaks]
    shifted = vals - np.max(vals)
    w = np.exp(shifted / max(temp, 1e-15))
    w /= np.sum(w) + 1e-30
    return w


def _make_hann(n):
    i = np.arange(n)
    a = 0.5 * (1 + np.cos(2 * np.pi * i / (n - 1)))
    s = np.sum(a)
    return a / s if s > 0 else np.ones(n) / n


def _make_blackman(n):
    i = np.arange(n)
    a = 0.42 - 0.5 * np.cos(2 * np.pi * i / (n - 1)) + 0.08 * np.cos(4 * np.pi * i / (n - 1))
    a = np.maximum(a, 0)
    s = np.sum(a)
    return a / s if s > 0 else np.ones(n) / n


def _make_flattop(n):
    """Flat-top window (designed to minimize sidelobes)."""
    i = np.arange(n)
    a = (0.21557895 - 0.41663158 * np.cos(2 * np.pi * i / (n - 1))
         - 0.277263158 * np.cos(4 * np.pi * i / (n - 1))
         - 0.083578947 * np.cos(6 * np.pi * i / (n - 1))
         - 0.006947368 * np.cos(8 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0)
    s = np.sum(a)
    return a / s if s > 0 else np.ones(n) / n


def _make_anti_sym(n, eps, mode='cos'):
    i = np.arange(n, dtype=np.float64)
    if mode == 'cos':
        a = 1.0 + eps * np.cos(np.pi * i / (n - 1))
    elif mode == 'step':
        a = np.ones(n)
        a[:n // 2] = 1.0 + eps
        a[n // 2:] = 1.0 - eps
    elif mode == 'smooth_step':
        t = i / (n - 1)
        a = 1.0 + eps * np.tanh(3.0 * (0.5 - t))
    a = np.maximum(a, 0)
    s = np.sum(a)
    return a / s if s > 0 else np.ones(n) / n


def _make_quadratic(n, spec):
    """Quadratic profile: a_i = 1 + spec*(i/(n-1) - 0.5)^2, normalized."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1) - 0.5
    a = 1.0 + spec * t * t
    a = np.maximum(a, 0)
    s = np.sum(a)
    return a / s if s > 0 else np.ones(n) / n


def _mirror_descent(a, deadline, rng, init_temp=1.0, eta_init=None, cool_eta=0.95,
                    cool_temp=0.995, min_eta=1e-10, M_init=20, M_final=5):
    """
    Mirror descent (multiplicative weights) to minimize max(b_k).
    Assumes sum(a) = 1, a >= 0.
    """
    n = len(a)
    if eta_init is None:
        eta_init = 0.5 / np.sqrt(n)

    a = np.asarray(a, dtype=np.float64).copy()
    s = np.sum(a)
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n

    a = np.maximum(a, 1e-15)
    a /= np.sum(a)

    best_a = a.copy()
    best_score = _score(best_a)
    eta = eta_init
    temp = init_temp
    no_improve = 0
    iter_count = 0
    total_iters = 50000  # for scheduling M

    while time.monotonic() < deadline:
        b = _convolve(a)
        b_max = np.max(b)
        cur_score = 2.0 * n * b_max

        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 500:
            eta *= cool_eta
            if eta < min_eta:
                break
            no_improve = 0

        # Schedule M
        frac = min(1.0, iter_count / total_iters)
        M = max(M_final, int(M_init * (1 - frac) + M_final * frac))

        # Select top M peaks
        if M >= len(b):
            peaks = np.arange(len(b))
        else:
            peaks = np.argpartition(b, -M)[-M:]
            peaks = peaks[np.argsort(b[peaks])[::-1]]

        # Softmax weights with temperature
        temp = max(temp, 1e-10 * b_max)
        w = _softmax_weights(b, peaks, temp)
        temp *= cool_temp

        # Compute gradient
        grad = _grad_peaks(a, peaks, w)

        # Center gradient (remove mean for sum constraint in mirror descent)
        grad_centered = grad - np.mean(grad)

        # Multiplicative update: a_i <- a_i * exp(-eta * grad_centered_i)
        update = np.exp(-eta * grad_centered)
        a = a * update
        a_sum = np.sum(a)
        if a_sum < 1e-30:
            a = best_a.copy()
            break
        a /= a_sum
        a = np.maximum(a, 1e-15)
        a /= np.sum(a)

        iter_count += 1

    return best_a, best_score


def _nesterov_subgrad(a, deadline, rng, eta_init=None, M_init=30, M_final=3,
                      momentum_init=0.3, momentum_max=0.9):
    """
    Nesterov-accelerated subgradient method for minimizing max(b_k).
    """
    n = len(a)
    if eta_init is None:
        eta_init = 0.3 / np.sqrt(n)

    a = np.asarray(a, dtype=np.float64).copy()
    s = np.sum(a)
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n

    y = a.copy()  # Nesterov lookahead point
    momentum = momentum_init
    best_a = a.copy()
    best_score = _score(best_a)
    eta = eta_init
    no_improve = 0
    iter_count = 0
    total_iters = 50000

    while time.monotonic() < deadline:
        b = _convolve(y)
        b_max = np.max(b)
        cur_score_y = 2.0 * n * b_max

        # Also check current a
        b_a = _convolve(a)
        cur_score_a = 2.0 * n * np.max(b_a)
        cur_score = min(cur_score_y, cur_score_a)

        if cur_score < best_score - 1e-14:
            best_score = cur_score
            if cur_score_a <= cur_score_y:
                best_a = a.copy()
            else:
                best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 400:
            eta *= 0.92
            if eta < 1e-10:
                break
            no_improve = 0

        # Schedule M
        frac = min(1.0, iter_count / total_iters)
        M = max(M_final, int(M_init * (1 - frac) + M_final * frac))

        # Select peaks from y's convolution
        if M >= len(b):
            peaks = np.arange(len(b))
        else:
            peaks = np.argpartition(b, -M)[-M:]
            peaks = peaks[np.argsort(b[peaks])[::-1]]

        # Use top peaks with softmax weighting
        w = _softmax_weights(b, peaks, np.max(b[peaks]) * 0.5)

        # Gradient at y
        grad = _grad_peaks(y, peaks, w)
        grad -= np.mean(grad)

        # Nesterov update
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # x_new = project(y - eta * grad)
        x_new = y - eta * grad
        x_new = _project_simplex(x_new, 1.0)

        # y_new = x_new + momentum * (x_new - a)
        momentum = min(momentum + 0.0001, momentum_max)
        a = x_new
        y = a + momentum * (a - (y - eta * grad / max(grad_norm, 1e-15)))
        # Project y back to simplex for numerical stability
        y = _project_simplex(y, 1.0)

        iter_count += 1

    return best_a, best_score


def _lp_smooth_optimize(a, deadline, rng, p_init=4.0, p_max=100.0, eta_init=None,
                        M_cap=50):
    """
    Optimize using L_p norm of b as a smooth surrogate for max(b).
    p increases over time to sharpen the approximation.
    """
    n = len(a)
    if eta_init is None:
        eta_init = 0.2 / np.sqrt(n)

    a = np.asarray(a, dtype=np.float64).copy()
    s = np.sum(a)
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n

    best_a = a.copy()
    best_score = _score(best_a)
    eta = eta_init
    p = p_init
    no_improve = 0
    iter_count = 0
    total_iters = 30000

    while time.monotonic() < deadline:
        b = _convolve(a)
        b_max = np.max(b)
        cur_score = 2.0 * n * b_max

        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            eta *= 0.93
            if eta < 1e-11:
                break
            no_improve = 0

        # Increase p over time
        frac = min(1.0, iter_count / total_iters)
        p = p_init + (p_max - p_init) * frac

        # For L_p gradient: grad_i = (p/p) * sum_k w_k * 2*a_{k-i} / (sum b^p)^{1/p}
        # where w_k = b_k^{p-1} / sum_j b_j^{p-1}
        # But we only compute over top M for efficiency
        M = min(M_cap, len(b))
        if M >= len(b):
            peaks = np.arange(len(b))
        else:
            # For L_p, weight by b^p, so top peaks dominate
            peaks = np.argpartition(b, -M)[-M:]

        b_p = b[peaks] ** (p - 1)
        w = b_p / (np.sum(b_p) + 1e-30)

        grad = _grad_peaks(a, peaks, w)
        grad -= np.mean(grad)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        a = a - eta * grad / grad_norm
        a = _project_simplex(a, 1.0)

        iter_count += 1

    return best_a, best_score


def _local_polish(a, deadline, rng):
    """
    Fine-grained local search: try small random perturbations and keep improvements.
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    s = np.sum(a)
    if s > 1e-10:
        a /= s
    best_a = a.copy()
    best_score = _score(best_a)
    step = 0.02
    no_improve = 0

    while time.monotonic() < deadline:
        # Random smooth perturbation
        n_pert = rng.integers(1, max(2, n // 10))
        idx = rng.choice(n, size=n_pert, replace=False)
        delta = rng.normal(0, step, size=n_pert)
        a_new = a.copy()
        a_new[idx] += delta
        a_new = _project_simplex(a_new, 1.0)

        new_score = _score(a_new)
        if new_score < best_score - 1e-12:
            best_a = a_new.copy()
            best_score = new_score
            a = a_new.copy()
            no_improve = 0
            # Slightly increase step
            step = min(step * 1.05, 0.1)
        else:
            no_improve += 1
            if no_improve > 100:
                step *= 0.9
                no_improve = 0
                if step < 0.001:
                    break

    return best_a, best_score


def _anneal_refine(a, deadline, rng):
    """
    Simulated annealing style: perturb with decreasing temperature,
    accept improvements and occasionally accept worse solutions.
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    s = np.sum(a)
    if s > 1e-10:
        a /= s
    best_a = a.copy()
    best_score = _score(best_a)
    T = 0.05  # initial temperature
    no_improve = 0
    cooldown = 200
    iter_count = 0

    while time.monotonic() < deadline:
        # Generate perturbation
        n_pert = rng.integers(max(5, n // 50), max(10, n // 20))
        idx = rng.choice(n, size=n_pert, replace=False)
        delta = rng.normal(0, 0.03, size=n_pert)
        a_new = a.copy()
        a_new[idx] += delta
        a_new = _project_simplex(a_new, 1.0)

        new_score = _score(a_new)
        diff = new_score - _score(a)

        # Metropolis acceptance
        if diff < 0 or rng.random() < np.exp(-diff / max(T, 1e-10)):
            a = a_new.copy()
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

        if no_improve > cooldown:
            T *= 0.8
            no_improve = 0
            if T < 1e-6:
                break

        iter_count += 1

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy search for minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses mirror descent, Nesterov subgradient, L_p smoothing, and local polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 10.0)

    # Get starting sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        parent = np.ones(6121)
    parent = np.clip(parent, 0.0, None)
    if parent.size == 0:
        parent = np.ones(6121)
    n = parent.size

    # Global best
    global_best = None
    global_best_score = np.inf

    def _update_best(a, s):
        nonlocal global_best, global_best_score
        if s < global_best_score:
            global_best = a.copy()
            global_best_score = s
            print(f"  NEW BEST: {global_best_score:.8f} (n={len(a)})")

    # Evaluate parent
    parent_s = np.sum(parent)
    if parent_s < 0.01:
        parent = np.ones(n)
        parent_s = n
    parent_norm = parent / parent_s
    parent_score = _score(parent_norm)
    _update_best(parent_norm, parent_score)
    print(f"Parent score: {parent_score:.8f}")

    # Generate diverse starting points
    starts = [parent_norm.copy()]

    # Window functions
    starts.append(_make_hann(n))
    starts.append(_make_blackman(n))
    starts.append(_make_flattop(n))

    # Anti-symmetric
    starts.append(_make_anti_sym(n, 0.3, 'cos'))
    starts.append(_make_anti_sym(n, 0.5, 'step'))
    starts.append(_make_anti_sym(n, 0.7, 'smooth_step'))
    starts.append(_make_anti_sym(n, 0.4, 'cos'))

    # Quadratic profiles
    starts.append(_make_quadratic(n, 2.0))
    starts.append(_make_quadratic(n, -1.0))  # inverted parabola
    starts.append(_make_quadratic(n, 5.0))

    # Random
    a_rand = rng.random(n)
    starts.append(a_rand / np.sum(a_rand))

    print(f"Generated {len(starts)} starting points")

    # Time allocation strategy:
    # Phase 1: Quick scan of all starts with mirror descent (60% of time)
    # Phase 2: Best 3 starts with Nesterov (20% of time)
    # Phase 3: L_p optimization on best (10% of time)
    # Phase 4: Polish (10% of time)

    total_time = max(1.0, deadline - time.monotonic())
    t_phase1 = 0.50 * total_time
    t_phase2 = 0.20 * total_time
    t_phase3 = 0.15 * total_time
    t_phase4 = 0.15 * total_time

    # === Phase 1: Mirror descent on all starts ===
    print(f"\n=== Phase 1: Mirror Descent ({t_phase1:.1f}s) ===")
    per_start_t1 = t_phase1 / len(starts)

    # Sort starts by initial score (best first)
    start_scores = [_score(s) for s in starts]
    order = np.argsort(start_scores)
    for i in order:
        if time.monotonic() >= deadline - (t_phase2 + t_phase3 + t_phase4):
            break
        s = starts[i]
        t_deadline = min(time.monotonic() + per_start_t1, deadline)
        if time.monotonic() >= t_deadline:
            break
        print(f"  Start {i} (init={start_scores[i]:.6f}), time={per_start_t1:.1f}s")
        best_a, best_s = _mirror_descent(s, t_deadline, rng,
                                         init_temp=1.0,
                                         eta_init=0.3 / np.sqrt(n),
                                         M_init=30, M_final=5)
        _update_best(best_a, best_s)

    # === Phase 2: Nesterov on top starts ===
    print(f"\n=== Phase 2: Nesterov Subgradient ({t_phase2:.1f}s) ===")
    # Take the globally best and a couple of diverse ones
    nesterov_starts = [global_best.copy()]
    # Add diverse starts (windows, anti-sym)
    for idx in [1, 2, 5]:  # Hann, Blackman, smooth_step
        if idx < len(starts):
            nesterov_starts.append(starts[idx].copy())

    per_start_t2 = t_phase2 / max(1, len(nesterov_starts))
    for i, s in enumerate(nesterov_starts):
        t_deadline = min(time.monotonic() + per_start_t2, deadline)
        if time.monotonic() >= deadline:
            break
        print(f"  Nesterov start {i} (init={_score(s):.6f}), time={per_start_t2:.1f}s")
        best_a, best_s = _nesterov_subgrad(s, t_deadline, rng,
                                           eta_init=0.2 / np.sqrt(n),
                                           M_init=30, M_final=3)
        _update_best(best_a, best_s)

    # === Phase 3: L_p smoothing on best ===
    print(f"\n=== Phase 3: L_p Smoothing ({t_phase3:.1f}s) ===")
    lp_deadline = min(time.monotonic() + t_phase3, deadline)
    if time.monotonic() < lp_deadline:
        best_a, best_s = _lp_smooth_optimize(global_best.copy(), lp_deadline, rng,
                                             p_init=4.0, p_max=200.0,
                                             eta_init=0.15 / np.sqrt(n))
        _update_best(best_a, best_s)

    # Also try L_p on a diverse start
    remaining = deadline - time.monotonic()
    if remaining > 10:
        diverse_start = _make_flattop(n)
        lp_deadline2 = min(time.monotonic() + remaining * 0.5, deadline)
        best_a, best_s = _lp_smooth_optimize(diverse_start, lp_deadline2, rng,
                                             p_init=2.0, p_max=50.0,
                                             eta_init=0.2 / np.sqrt(n))
        _update_best(best_a, best_s)

    # === Phase 4: Polish ===
    print(f"\n=== Phase 4: Polish ({t_phase4:.1f}s) ===")
    remaining = deadline - time.monotonic()
    if remaining > 5 and global_best is not None:
        # Split: half for local polish, half for annealing
        t_polish = remaining * 0.5
        t_anneal = remaining * 0.5

        # Local polish
        t_deadline_p = min(time.monotonic() + t_polish, deadline)
        best_a, best_s = _local_polish(global_best.copy(), t_deadline_p, rng)
        _update_best(best_a, best_s)

        # Annealing
        remaining2 = deadline - time.monotonic()
        if remaining2 > 3:
            best_a, best_s = _anneal_refine(global_best.copy(), deadline, rng)
            _update_best(best_a, best_s)

    # Final: one more mirror descent pass on the best
    remaining = deadline - time.monotonic()
    if remaining > 5 and global_best is not None:
        print(f"\n=== Final Mirror Descent Polish ({remaining:.1f}s) ===")
        best_a, best_s = _mirror_descent(global_best.copy(), deadline, rng,
                                         init_temp=0.1,
                                         eta_init=0.05 / np.sqrt(n),
                                         M_init=10, M_final=3)
        _update_best(best_a, best_s)

    # Final evaluation
    if global_best is None:
        result = [float(x) for x in starts[0]]
    else:
        result = [float(x) for x in global_best]

    final_score = float(evaluate_sequence(result))
    print(f"\nFINAL SCORE: {final_score:.8f}")
    print(f"Sequence length: {len(result)}")
    print(f"Sum: {sum(result):.6f}")
    print(f"Min: {min(result):.10f}")
    print(f"Max: {max(result):.10f}")
    return result
```
