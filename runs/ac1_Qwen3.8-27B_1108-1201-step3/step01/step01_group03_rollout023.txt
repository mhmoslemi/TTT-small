For the uniform case, (b*b)_k = (k+1)/n² for k < n, and (2n-1-k)/n² for k ≥ n. The max is n/n² = 1/n at k = n-1.

To get max(b*b) < 1/n, we need the middle lags to be below 1/n. This means we need to "spread out" the energy so that no single lag accumulates too much.

For larger n, we have more lags (2n-1 of them) to spread the energy across, so it might be easier to keep each one below 1/n. This suggests that larger n should give better (lower) scores.

Let me verify: if we could make (b*b)_k = 1/(2n-1) for all k (perfectly flat), then max(b*b) = 1/(2n-1) and score = 2n/(2n-1) ≈ 1. But this is impossible because sum_k (b*b)_k = (sum b)² = 1, and if all 2n-1 values are equal, each is 1/(2n-1). But can we actually achieve a flat convolution? Probably not exactly, but we can get close with large n.

def softmax_grad_to_w(score_grad_b, b):
    """Convert gradient w.r.t. b to gradient w.r.t. w (softmax parameterization)."""
    # b = softmax(w), so db_j/dw_i = b_j * (delta(j,i) - b_i)
    # dF/dw_i = sum_j dF/db_j * db_j/dw_i = b_i * (dF/db_i - sum_j dF/db_j * b_j)
    sum_db = np.dot(score_grad_b, b)
    return b * (score_grad_b - sum_db)


def optimize_smooth(b_init, T, maxiter=500, tol=1e-10):
    """Optimize the smooth score using L-BFGS in w-space."""
    n = len(b_init)
    w0 = np.log(b_init + 1e-15)  # Inverse softmax (approximately)
    
    def objective(w):
        b = np.exp(w - w.max())
        b = b / b.sum()
        score, grad_b = smooth_score_grad(b, T)
        grad_w = softmax_grad_to_w(grad_b, b)
        return score, grad_w
    
    from scipy.optimize import minimize
    res = minimize(objective, w0, method='L-BFGS-B',
                   options={'maxiter': maxiter, 'gtol': tol, 'ftol': 1e-12})
    w = res.x
    b = np.exp(w - w.max())
    b = b / b.sum()
    return b, res.fun, res


def build_conv_matrix(n, f):
    """Sparse A (2n-1 x n) for the LP: (f*g)[k] = sum_j A[k,j] * g[j]."""
    # A[k,j] = f[k-j] if 0 <= k-j < n, else 0
    # For efficiency, build using the structure
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        for j in range(max(0, k - n + 1), min(k + 1, n)):
            idx = k - j
            if 0 <= idx < n:
                rows.append(k)
                cols.append(j)
                data.append(f[idx])
    A = coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    return A


def solve_lp(f, M, time_limit=5.0):
    """Solve LP: max sum(g) s.t. A_g * g <= M, g >= 0.
    Here A_g[k,j] = f[k-j] (convolution matrix with f)."""
    n = len(f)
    # The constraint matrix for g: (f*g)[k] = sum_j f[k-j] * g[j]
    # So A[k,j] = f[k-j] if 0 <= k-j < n
    # This is the same as build_conv_matrix(n, f)
    A = build_conv_matrix(n, f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None


def lp_refine(f, budget_s, t_init=0.5, t_min=0.01, t_decay=0.99):
    """Iteratively refine f using LP steps."""
    t = t_init
    best = f.copy()
    best_sc = fast_score(f)
    start_time = time.monotonic()
    stall = 0
    
    while time.monotonic() - start_time < budget_s:
        conv = np.convolve(f, f)
        M = float(conv.max())
        g = solve_lp(f, M, time_limit=3.0)
        if g is None:
            break
        if np.linalg.norm(g - f) < 1e-10:
            # No movement, try perturbation
            rng = np.random.default_rng(time.monotonic() * 1e6)
            mask = rng.random(n) < 0.1
            f[mask] *= np.exp(rng.normal(0, 0.3, mask.sum()))
            f = np.maximum(f, 0)
            f = f / f.sum()
            continue
        f_new = (1 - t) * f + t * g
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-10:
            break
        f_new = f_new / s
        sc = fast_score(f_new)
        if sc < best_sc:
            best_sc = sc
            best = f_new.copy()
            f = f_new
            stall = 0
            t = max(t_min, t * 0.95)
        else:
            stall += 1
            t = max(t_min, t * 0.9)
            if stall > 30:
                # Revert and perturb
                f = best.copy()
                rng = np.random.default_rng(int(time.monotonic() * 1e9) % 2**31)

                start = int(rng.integers(0, n - w))
                f[start:start+w] *= np.exp(rng.normal(0, 0.5, w))
                f = np.maximum(f, 0)
                f = f / f.sum()
                stall = 0
    return best, best_sc

Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Phase 1: Smooth gradient descent (log-sum-exp) with multiple starts.
Phase 2: LP-based local refinement.
Phase 3: Diverse global search with periodic LP returns.

import numpy as np

from numpy.lib.stride_tricks import sliding_window_view


def fast_score(f):
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))


def compute_grad_convolution(b, T):
    """Compute smooth score and gradient for the self-convolution peak.
    Returns (score, grad) where grad is w.r.t. b (unconstrained)."""
    n = len(b)
    conv = np.convolve(b, b)
    conv_max = conv.max()
    exp_conv = np.exp(T * (conv - conv_max))
    sum_exp = exp_conv.sum()
    smax = conv_max + np.log(sum_exp) / T
    p = exp_conv / sum_exp
    # grad[i] = 2 * sum_j p[i+j] * b[j]
    windows = sliding_window_view(p, n)
    grad = 2.0 * (windows @ b)
    score = 2.0 * n * smax
    return score, grad


def softmax_from_w(w):
    w_shifted = w - w.max()
    b = np.exp(w_shifted)
    return b / b.sum()


def smooth_objective(w, T):
    """Objective and gradient in w-space."""
    b = softmax_from_w(w)
    score, grad_b = compute_grad_convolution(b, T)
    # Chain rule: dF/dw_i = b_i * (grad_b_i - sum_j grad_b_j * b_j)
    proj = np.dot(grad_b, b)
    grad_w = b * (grad_b - proj)
    return score, grad_w


def build_conv_matrix(n, f):
    """Sparse matrix A (2n-1 x n) where A[k,j] = f[k-j] if valid."""
    cols = np.repeat(np.arange(n), n)
    rows = cols + np.tile(np.arange(n), n)
    data = np.tile(f, n)
    return coo_matrix((data, (rows, cols)), shape=(2*n-1, n)).tocsr()


def solve_lp_constrained(f, M, time_limit=5.0):
    """Solve: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns normalized g or None."""
    n = len(f)
    A = build_conv_matrix(n, f)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None


def lp_refine(f, budget_s, verbose=False):
    """Refine f using LP steps. Returns (best_f, best_score)."""
    n = len(f)
    t = 0.5
    best = f.copy()
    best_sc = fast_score(f)
    start_time = time.monotonic()
    stall = 0
    
    while time.monotonic() - start_time < budget_s:
        conv = np.convolve(f, f)
        M = float(conv.max())
        g = solve_lp_constrained(f, M, time_limit=3.0)
        if g is None:
            break
        if np.linalg.norm(g - f) < 1e-12:
            rng = np.random.default_rng(int(time.monotonic()*1e9) % (2**31))
            w = int(rng.integers(1, max(2, n//10)))
            start = int(rng.integers(0, n-w))
            f[start:start+w] *= np.exp(rng.normal(0, 0.5, w))
            f = np.maximum(f, 0)
            f = f / f.sum()
            stall += 1
            if stall > 20:
                break
            continue
        f_new = (1-t) * f + t * g
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-10:
            break
        f_new = f_new / s
        sc = fast_score(f_new)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best = f_new.copy()
            f = f_new
            stall = 0
            t = max(0.01, t * 0.97)
        else:
            stall += 1
            t = max(0.01, t * 0.9)
            if stall > 40:
                f = best.copy()
                rng = np.random.default_rng(int(time.monotonic()*1e9) % (2**31))
                # Multiple small perturbations
                for _ in range(3):
                    w = int(rng.integers(1, max(2, n//15)))
                    start = int(rng.integers(0, n-w))
                    f[start:start+w] *= np.exp(rng.normal(0, 0.4, w))
                f = np.maximum(f, 0)
                f = f / f.sum()
                stall = 0
    return best, best_sc
