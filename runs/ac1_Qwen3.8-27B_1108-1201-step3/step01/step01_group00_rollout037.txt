Multi-start over sequence lengths and initialization types.

def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)

    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)

def coordinate_descent_polish(a_init, time_limit, rng):
    """
    Coordinate-descent style polish: try small transfers between pairs of
    elements that are (contributor, non-contributor) for the peak.
    """
    current = a_init.astype(np.float64).copy()
    current_score = score_fast(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(current, current)
        peak_idx = int(np.argmax(conv))
        
        # Get contributions
        contribs = np.zeros(n)
        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)
        for i in range(lo, hi + 1):
            j = peak_idx - i
            if 0 <= j < n:
                contribs[i] += current[i] * current[j]
        
        # Try several transfers
        n_trials = max(50, n // 4)
        for _ in range(n_trials):
            # Pick a high contributor
            if contribs.max() > 0:
                weights = contribs / contribs.sum()
                src = int(rng.choice(n, p=weights))
            else:
                src = int(rng.integers(n))
            
            dst = int(rng.integers(n))
            if src == dst:
                continue
            
            # Try different transfer amounts
            for frac in [0.02, 0.05, 0.08, 0.12]:
                candidate = current.copy()
                transfer = current[src] * frac
                if current[src] - transfer < 1e-6:
                    continue
                candidate[src] -= transfer
                candidate[dst] += transfer
                
                new_score = score_fast(candidate)
                if new_score < current_score:
                    current = candidate
                    current_score = new_score
                    if current_score < best_score:
                        best_score = current_score
                        best = current.copy()
                    break
                elif new_score < current_score * 1.001:
                    # Accept with small probability (simulated annealing flavor)
                    if rng.random() < 0.1:
                        current = candidate
                        current_score = new_score
                    break
    
    return best, best_score

    return np.clip(a, 0.01, None)


def make_random_init(n, rng, spread=0.3):
    """Create a random initial sequence."""
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multi_tone_init(n, rng):
    """Create a multi-tone initial sequence."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)
    n_freqs = rng.integers(3, 8)
    for _ in range(n_freqs):
        k = int(rng.integers(1, n // 3 + 1))
        amp = rng.uniform(0.05, 0.25)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_linear_asym_init(n, rng):
    """Create a linearly asymmetric initialization."""
    t = np.linspace(0, 1, n)
    slope = rng.uniform(-0.5, 0.5)
    a = 1.0 + slope * (t - 0.5)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)
