    """Optimize using smooth max with FFT-based gradients."""
    a = safe_normalize(a_init, n)
    L = 1 << (3*n - 2 - 1).bit_length()  # next power of 2 >= 3n-2
    
    best_score = np.inf
    best_a = a.copy()
    
    start = time.monotonic()
    
    for it in range(n_iter):
        if time.monotonic() - start > max_time:
            break
        
        # Increase p (anneal temperature)
        t = it / max(1, n_iter - 1)
        p = p_start * (p_end / p_start) ** t  # geometric annealing
        
        # Compute self-convolution
        A = fft(a, L)
        c = np.real(ifft(A * A))[:2*n-1]
        cur_max = c.max()
        
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
        
        # Compute softmax weights
        # Numerical stability: subtract max
        c_shifted = p * (c - c.max())
        weights_unnorm = np.exp(c_shifted)
        weights = weights_unnorm / weights_unnorm.sum()
        
        # Compute gradient
        Wf = fft(weights, L)
        Af = fft(a, L)
        grad = 2.0 * np.real(ifft(Wf * np.conj(Af)))[:n]
        
        # Gradient normalization for stable updates

        if gnorm < 1e-15:

Minimize 2*n*max(self_conv) / sum^2 using smooth-max optimization with FFT-based gradients.

Key ideas:
1. Normalize to sum = sqrt(2n), so objective = max(self_conv).
2. Use smooth approximation: F_p = (1/p) log sum exp(p*c_k) → max(c_k) as p→∞.
3. Compute the exact gradient of F_p via FFT in O(N log N).
4. Anneal p from low to high (exploration → exploitation).
5. Multiple initializations and lengths for diversity.
6. Adam optimizer with non-negativity projection.

from scipy.fft import fft, ifft


def safe_normalize(a, n):
    """Clamp to [0, n*1000] and rescale to sum = sqrt(2*n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        a = np.ones(n, dtype=np.float64)
        s = n
    return a * (S / s)


def fft_self_conv(a, L):
    """Compute linear self-convolution via FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    Af = fft(a, L)
    c = np.real(ifft(Af * Af))
    return c[:2*n-1]


def compute_smooth_gradient(a, c, p, L):
    """Compute gradient of smooth max objective via FFT."""
    n = len(a)
    # Softmax weights (numerically stable)
    c_shifted = p * (c - c.max())
    w = np.exp(c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return np.zeros(n)
    w = w / w_sum
    
    # Gradient: grad[m] = 2 * sum_k w[k] * a[k-m] (for valid k, m)
    # = 2 * crossCorr(w, a)[m]
    # Cross-correlation via FFT: ifft(fft(w) * conj(fft(a)))
    Wf = fft(w, L)
    Af = fft(a, L)
    corr = np.real(ifft(Wf * np.conj(Af)))
    return 2.0 * corr[:n]


def smooth_adam(a_init, n, S, max_time, p_start=1.0, p_end=200.0, 
                lr_init=1.0, top_len=None):
    """Adam optimization of smooth max with p-annealing."""
    a = safe_normalize(a_init, n)
    L = 1 << (3 * n - 2 - 1).bit_length()  # next power of 2 >= 3n-2
    
    m_adam = np.zeros(n, dtype=np.float64)
    v_adam = np.zeros(n, dtype=np.float64)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    
    best_score = np.inf
    best_a = a.copy()
    start = time.monotonic()
    it = 0
    
    # Phase durations (fractions of max_time)
    phase1_frac = 0.5  # exploration (low p)
    phase2_frac = 0.3  # transition
    # phase3: exploitation (high p)
    
    while time.monotonic() - start < max_time:
        it += 1
        t_frac = (time.monotonic() - start) / max_time
        
        # Anneal p: sigmoid-like schedule
        if t_frac < phase1_frac:
            p = p_start
        elif t_frac < phase1_frac + phase2_frac:
            frac = (t_frac - phase1_frac) / phase2_frac
            p = p_start * (p_end / p_start) ** (frac ** 2)  # quadratic ramp
        else:
            p = p_end
        
        # Compute self-convolution
        Af = fft(a, L)
        c = np.real(ifft(Af * Af))[:2*n-1]
        cur_max = c.max()
        
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
        
        # Compute gradient
        grad = compute_smooth_gradient(a, c, p, L)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm  # normalize for Adam
        
        # Adam update
        m_adam = beta1 * m_adam + (1 - beta1) * grad
        v_adam = beta2 * v_adam + (1 - beta2) * (grad ** 2)
        m_hat = m_adam / (1 - beta1 ** it)
        v_hat = v_adam / (1 - beta2 ** it)
        
        # Learning rate schedule
        lr = lr_init * S / n * (0.999 ** it)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project to non-negative, maintain sum
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a = np.full(n, S / n, dtype=np.float64)
        else:
            a = a_new * (S / s)
    
    return best_a, best_score


def smooth_mirror(a_init, n, S, max_time, p_start=1.0, p_end=200.0,
                  eta_init=0.5):
    """Mirror descent (multiplicative weights) with smooth max gradient."""
    a = safe_normalize(a_init, n)
    L = 1 << (3 * n - 2 - 1).bit_length()
    
    best_score = np.inf
    best_a = a.copy()
    start = time.monotonic()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        t_frac = (time.monotonic() - start) / max_time
        
        if t_frac < 0.4:
            p = p_start
        elif t_frac < 0.7:
            frac = (t_frac - 0.4) / 0.3
            p = p_start * (p_end / p_start) ** (frac ** 2)
        else:
            p = p_end
        
        Af = fft(a, L)
        c = np.real(ifft(Af * Af))[:2*n-1]
        cur_max = c.max()
        
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
        
        grad = compute_smooth_gradient(a, c, p, L)
        gmax = np.abs(grad).max()
        if gmax < 1e-15:
            break
        grad = grad / gmax
        
        # Multiplicative update
        eta = eta_init * np.exp(-0.001 * it)
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a = np.full(n, S / n, dtype=np.float64)
        else:
            a = a_new * (S / s)
    
    return best_a, best_score


def generate_initializations(rng, n):
    """Generate diverse initializations."""
    inits = []
    
    # Uniform
    inits.append(("uniform", np.ones(n)))
    
    # Random
    inits.append(("random", rng.random(n)))
    
    # Gaussian (peaked in center)
    x = np.arange(n)
    inits.append(("gaussian", np.exp(-((x - n/2) ** 2) / (2 * (n/8) ** 2))))
    
    # Sinusoidal
    inits.append(("sinusoidal", 1.0 + 0.5 * np.sin(2 * np.pi * 3 * x / n)))
    
    # Two-lobe (high-low-high-low pattern)
    inits.append(("two_lobe", np.where(x % 2 == 0, 2.0, 0.5)))
    
    # Frequency-modulated
    inits.append(("fm", 1.0 + 0.3 * np.sin(2 * np.pi * (x/n) * (1 + 2*x/n))))
    
    # Exponential ramp
    inits.append(("ramp", np.exp(-x / n) + 0.1))
    
    return inits
