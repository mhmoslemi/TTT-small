"""
Multi-phase optimization for minimizing the normalized autocorrelation peak
of a non-negative sequence. Combines smooth L-BFGS (with softmax parameterization
and log-sum-exp objective) with subgradient descent and coordinate descent.
"""

import numpy as np
from scipy.signal import fftconvolve, correlate
from scipy.optimize import minimize
import time


def _normalize(a):
    """Normalize to sum to 1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s > 1e-30:
        return a / s
    return np.ones(len(a)) / len(a)


def _true_score(a):
    """Compute the true (non-smooth) objective."""
    n = len(a)
    b = fftconvolve(a, a, mode='full')
    return 2.0 * n * np.max(b) / (np.sum(a))**2


def _softmax(z):
    """Stable softmax."""
    z_shifted = z - np.max(z)
    a = np.exp(z_shifted)
    return a / a.sum()


def _logsumexp_smooth_obj(z, T):
    """Compute smooth objective value (2n * T * logsumexp(b/T))."""
    a = _softmax(z)
    b = fftconvolve(a, a, mode='full')
    b_max = np.max(b)
    S = np.sum(np.exp((b - b_max) / T))
    return 2.0 * len(z) * (b_max + T * np.log(S))


def _logsumexp_smooth_grad(z, T):
    """Compute smooth gradient w.r.t. z."""
    n = len(z)
    a = _softmax(z)
    b = fftconvolve(a, a, mode='full')
    b_max = np.max(b)
    exp_b = np.exp((b - b_max) / T)
    S = np.sum(exp_b)
    w = exp_b / S  # softmax weights over b_k
    
    # g_j = sum_k w_k * 2 * a[k-j]
    # = 2 * correlate(w, a, 'full')[:n]
    g = 2.0 * correlate(w, a, mode='full')[:n]
    
    # grad_z_j = 2n * a_j * (g_j - sum_i g_i * a_i)
    g_dot_a = np.dot(g, a)
    grad_z = 2.0 * n * a * (g - g_dot_a)
    
    obj = 2.0 * n * (b_max + T * np.log(S))
    return obj, grad_z


def _subgradient_step(a, eta, M=10):
    """One subgradient step with top-M lags."""
    n = len(a)
    b = fftconvolve(a, a, mode='full')
    if M >= len(b):
        k_idx = np.arange(len(b))
        wts = np.ones(len(b)) / len(b)
    else:
        k_idx = np.argpartition(b, -M)[-M:]
        b_sel = b[k_idx]
        b_max = b_sel.max()
        wts = np.exp(-10.0 * (b_max - b_sel) / max(b_max, 1e-15))
        wts /= np.sum(wts)
    
    # Compute gradient using correlate
    g = 2.0 * correlate(wts, a, mode='full')[:n]
    
    # Center for sum constraint
    g -= np.mean(g)
    
    # Update
    a = a - eta * g / (np.linalg.norm(g) + 1e-30)
    a = _normalize(a)
    return a


def _subgradient_optimize(a, deadline, M=10, eta_init=0.1, mom_init=0.9, cooldown=0.999):
    """Subgradient optimization with momentum."""
    n = len(a)
    a = np.array(a, dtype=np.float64)
    a = _normalize(a)
    
    best = a.copy()
    best_score = _true_score(best)
    eta = eta_init / np.sqrt(n)
    mom = np.zeros(n)
    momentum = mom_init
    no_improve = 0
    
    while time.monotonic() < deadline:
        b = fftconvolve(a, a, mode='full')
        b_max = np.max(b)
        cur_score = 2.0 * n * b_max  # since sum(a)=1
        
        if cur_score < best_score - 1e-10:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                eta *= 0.9
                no_improve = 0
                if eta < 1e-12:
                    break
        
        # Subgradient step
        b = fftconvolve(a, a, mode='full')
        if M >= len(b):
            k_idx = np.arange(len(b))
            wts = np.ones(len(b)) / len(b)
        else:
            k_idx = np.argpartition(b, -M)[-M:]
            b_sel = b[k_idx]
            b_max_sel = b_sel.max()
            wts = np.exp(-10.0 * (b_max_sel - b_sel) / max(b_max_sel, 1e-15))
            wts /= np.sum(wts)
        
        g = 2.0 * correlate(wts, a, mode='full')[:n]
        g -= np.mean(g)
        
        # Momentum update
        mom = momentum * mom + g
        a = a - eta * mom
        a = _normalize(a)
    
    return best, best_score


def _coordinate_descent(a, deadline, n_iter=100):
    """Simple coordinate descent: optimize one element at a time."""
    n = len(a)
    a = np.array(a, dtype=np.float64)
    a = _normalize(a)
    best = a.copy()
    best_score = _true_score(best)
    
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(0, n, max(1, n // 200)):  # random subset
            # Optimize a[i] with others fixed
            # b_k = sum_j a[j]*a[k-j]
            # For a given i, b_k = (terms not involving a[i]) + a[i]*(a[k-i] + a[i-k]) [if valid]
            # + a[i]^2 if 2i = k... this gets complex.
            # Simpler: try a few values of a[i] and pick the best.
            old_ai = a[i]
            # Try scaling a[i] by factors
            for factor in [0.5, 0.7, 1.3, 1.5, 2.0]:
                a[i] = old_ai * factor
                a = _normalize(a)
                s = _true_score(a)
                if s < best_score:
                    best_score = s
                    best = a.copy()
                    improved = True
                a[i] = old_ai
            a = _normalize(a)
        
        if not improved:
            break
    
    return best, best_score


def _make_two_level(n, ratio=2.0):

    global_best = a_init.copy()
    global_best_score = _true_score(global_best)
    print(f"Initial score: {global_best_score:.6f}")
    
    # Create starting points
    starts = [
        a_init,
        _make_two_level(n, 2.0),
        _make_two_level(n, 1.5),
        _make_cosine_anti(n, 0.7),
        _make_cosine_anti(n, 0.9),
    ]
    
    # Add a random start
    a_rand = rng.random(n)
    starts.append(_normalize(a_rand))
    
    # Phase 1: Quick subgradient on each start (30s total)
    phase1_deadline = time.monotonic() + 30
    t_per = 30.0 / len(starts)
    
    for i, s in enumerate(starts):
        dl = min(time.monotonic() + t_per, phase1_deadline)
        if time.monotonic() >= dl:
            break
        best_s, score = _subgradient_optimize(s, dl, M=10, eta_init=0.5)
        if score < global_best_score:
            global_best = best_s.copy()
            global_best_score = score
            print(f"Phase 1, start {i}: {score:.6f}")
    
    print(f"After Phase 1: {global_best_score:.6f}")
    
    # Phase 2: Smooth L-BFGS (600s)
    phase2_deadline = time.monotonic() + 600
    if time.monotonic() < phase2_deadline:
        # Convert to log parameterization: z = log(a + eps)
        # But we use softmax, so z = log(a) is a valid parameter (not unique due to shift invariance)
        # Let's just use z = log(a + 1e-10) as the starting point
        
        best_start = global_best.copy()
        z_init = np.log(best_start + 1e-15)
        
        # Run L-BFGS with decreasing temperature
        T_values = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
        
        for T in T_values:
            if time.monotonic() >= phase2_deadline:
                break
            t_left = phase2_deadline - time.monotonic()
            if t_left < 5:
                break
            
            def fun(z):
                return _logsumexp_smooth_obj(z, T)
            
            def fun_and_grad(z):
                return _logsumexp_smooth_grad(z, T)
            
            try:
                res = minimize(fun_and_grad, z_init, method='L-BFGS-B',
                             options={'maxiter': 200, 'maxfun': 500,
                                    'ftol': 1e-10, 'gtol': 1e-6,
                                    'maxls': 50})
                a_new = _softmax(res.x)
                a_new = _normalize(a_new)
                score_new = _true_score(a_new)
                if score_new < global_best_score:
                    global_best = a_new.copy()
                    global_best_score = score_new
                    z_init = res.x.copy()
                    print(f"Phase 2, T={T}: {score_new:.6f}")
                else:
                    z_init = res.x.copy()  # Still update z for next T
                    print(f"Phase 2, T={T}: smooth={res.fun:.6f}, true={score_new:.6f} (no improvement)")
            except Exception as e:
                print(f"Phase 2, T={T}: error {e}")
                break
        
        print(f"After Phase 2: {global_best_score:.6f}")
