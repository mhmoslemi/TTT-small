"""
FFT-accelerated smoothed gradient optimization for minimizing the normalized
autocorrelation peak of a non-negative sequence. Uses log-sum-exp smoothing, 
momentum, adaptive temperature, and multi-start strategy.
"""

import numpy as np
import time


def _fft_convolve(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size <<= 1
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def _fft_cross_corr(p, a):
    """
    Compute cross-correlation: result[i] = sum_j p[i+j] * a[j] for i=0..len(a)-1.
    p has length >= len(a).
    """
    n = len(a)
    m = len(p)
    # Need FFT size >= n + m - 1
    fft_size = 1
    while fft_size < n + m - 1:
        fft_size <<= 1
    P = np.fft.rfft(p, fft_size)
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(P * np.conj(A), fft_size)[:n]


def _project_simplex(v, target=1.0):
    """Project vector onto {x >= 0, sum(x) = target} in O(n) expected time."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_score(a):
    """Compute the objective: 2*n*max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-15:
        return np.inf
    b = _fft_convolve(a)
    return 2.0 * n * np.max(b) / (S * S)

    S = np.sum(a)
    if S > 1e-15:
        a = a / S
    else:
        a = np.ones(n) / n
    
    b = _fft_convolve(a)
    best_a = a.copy()
    best_max_b = np.max(b)
    best_score = 2.0 * n * best_max_b

        cur_max_b = np.max(b)
        cur_score = 2.0 * n * cur_max_b

            best_a = a.copy()
            best_max_b = cur_max_b
            no_improve = 0
        else:

            eta = max(eta * 0.6, eta_min)
            no_improve = 0
            if eta <= eta_min * 1.1:
                break
        
        # Compute softmax weights
        # Use numerical stability: subtract max
        b_shifted = b - cur_max_b
        T_b = T * b_shifted
        # Clip to avoid overflow
        T_b = np.clip(T_b, -500, 500)
        p = np.exp(T_b)
        p_sum = np.sum(p)
        if p_sum < 1e-300:
            # All weights are ~0 except the max
            p = np.zeros(len(b))
            p[np.argmax(b)] = 1.0
        else:
            p /= p_sum
        
        # Compute gradient via cross-correlation
        grad = 2.0 * _fft_cross_corr(p, a)
        
        # Subtract mean for sum constraint
        grad -= np.mean(grad)
        
        # Normalize gradient (use norm for scale invariance)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Update with momentum
        v = momentum * v - eta * grad / grad_norm
        a_new = a + v
        a_new = _project_simplex(a_new, 1.0)
        
        # Check if the new point is valid
        S_new = np.sum(a_new)
        if S_new < 1e-15:
            a = best_a.copy()
            v = np.zeros(n)
            eta *= 0.5
            continue
        
        a = a_new
        total_steps += 1
        
        # Adapt temperature: increase over time
        progress = min(1.0, total_steps / 5000.0)
        T = T_start * (T_end / T_start) ** progress
        
        if verbose and total_steps % 500 == 0:
            actual_score = 2.0 * n * np.max(_fft_convolve(a))
            print(f"  {label} Step {total_steps}: actual={actual_score:.6f}, best={best_score:.6f}, T={T:.0f}, eta={eta:.2e}, no_improve={no_improve}")
    
    return best_a, best_score


def _make_antisymmetric(n, profile):
    """
    Create an anti-symmetric sequence from a profile.
    a[i] + a[n-1-i] = C for all i.
    profile[i] = a[i] for i < n/2.
    """
    half = n // 2
    a = np.zeros(n)
    a[:half] = profile
    a[half:] = np.sum(profile) / half - profile[::-1] if n % 2 == 0 else 0
    # Actually, let me do this differently.
    # Anti-symmetric: a[n-1-i] = C - a[i]
    # sum(a) = n*C/2 = 1 → C = 2/n
    C = 2.0 / n
    a[:half] = profile
    for i in range(half):
        a[n-1-i] = C - profile[i]
    if n % 2 == 1:
        a[half] = C / 2
    return np.maximum(a, 0.0)
