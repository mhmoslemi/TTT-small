def lp_perturbation(f, M, solver='glpk'):
    """Solve LP to find best perturbation direction."""
    from scipy.optimize import linprog
    n = len(f)
    # Variables: b_0, ..., b_{n-1} (n variables, all >= 0)
    # Objective: maximize sum(b) => minimize -sum(b)
    c = -np.ones(n)
    # Constraints: for each lag k (0 to 2n-2), conv(f, b)[k] <= M
    # conv(f, b)[k] = sum_i f[i] * b[k-i] (valid indices)
    # This is linear in b.
    # A_ub * b <= b_ub
    # Row for lag k: sum_i f[i] * b[k-i] = 0 (if k-i < 0 or k-i >= n) + sum_{valid i} f[i] * b[k-i]
    # The coefficient of b[j] in the constraint for lag k is f[k-j] (if 0 <= k-j < n).
    
    # So A_ub[k, j] = f[k-j] if 0 <= k-j < n, else 0.
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = f[idx]
    
    b_ub = np.full(2*n-1, M)
    
    # Solve: minimize c^T b subject to A b <= b_ub, b >= 0
    res = linprog(c, A_ub=A, b_ub=b_ub, method='highs')
    if res.success:
        return res.x
    return None

import numpy as np
import time
from scipy.optimize import linprog

def fft_convolve(a, b):
    """Fast linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_score(a):
    """Compute the score for a non-negative sequence (sum normalized to 1)."""
    a = np.maximum(a, 0.0)
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)

def score_normalized(a):
    """Score with sum(a)=1 normalization."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    a = a / s
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)

def softmax(x):
    """Stable softmax."""
    x = x - np.max(x)
    e = np.exp(x)
    return e / np.sum(e)

def soft_max_weights(b, tau):
    """Compute soft-max value and weights for a vector b."""
    b_max = np.max(b)
    shifted = (b - b_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    value = b_max + tau * np.log(sum_exp)
    return value, w

def gradient_descent_step(theta, tau, n, momentum1, momentum2, beta1, beta2, eps, lr, step):
    """One Adam step for the soft-max gradient descent."""
    a = softmax(theta)
    c = fft_convolve(a, a)
    value, w = soft_max_weights(c, tau)
    
    # Compute c[j] = sum_k w[k] * a[k-j] (cross-correlation)
    flip_a = a[::-1]
    corr = fft_convolve(w, flip_a)
    c_vec = corr[n-1:2*n-1]
    
    C = np.sum(a * c_vec)
    score = 2.0 * n * value
    
    # Gradient w.r.t. theta
    g = 4.0 * n * a * (c_vec - C)
    
    # Adam update
    momentum1 = beta1 * momentum1 + (1 - beta1) * g
    momentum2 = beta2 * momentum2 + (1 - beta2) * (g * g)
    m1_hat = momentum1 / (1.0 - beta1 ** step)
    m2_hat = momentum2 / (1.0 - beta2 ** step)
    theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)
    
    return theta, score, momentum1, momentum2

def lp_jump(f, M, max_iter=5000):
    """Solve LP to find perturbation direction. Returns g or None."""
    n = len(f)
    c_obj = -np.ones(n)
    # Build constraint matrix: for each lag k, sum_i f[i]*b[k-i] <= M

        # So k ranges from j to j+n-1

        # Coefficient of b[j] in conv(f,b)[k]: f[k-j] if 0 <= k-j < n
        # So A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # Let me redo this properly

    for k in range(2*n-1):

    b_ub = np.full(2*n-1, M)

def make_chirp(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    return np.clip(a, 0.01, None)

def make_ramp(n, alpha, rng):
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = 1.0 + alpha * t
    return np.clip(a, 0.01, None)

def make_window(n, window_type, rng):
    t = np.arange(n, dtype=np.float64) / (n - 1)
    if window_type == 'hann':
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
    elif window_type == 'hamming':
        a = 0.54 - 0.46 * np.cos(2 * np.pi * t)
    elif window_type == 'blackman':
        a = 0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t)
    else:
        a = np.ones(n)
    return np.clip(a, 0.01, None)

def make_multitone(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in rng.choice([1,2,3,4,5,7,8,11,13], size=4, replace=False):
        a += 0.15 * np.cos(2 * np.pi * k * t / n + rng.uniform(0, 2*np.pi))
    return np.clip(a, 0.01, None)

def make_random(n, rng, mu=1.0, sigma=0.3):
    a = mu + sigma * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

def optimize_single(n, theta_init, time_limit, rng, use_lp=False):
    """Run gradient descent with annealed soft-max. Optionally with LP jumps."""
    t0 = time.monotonic()
    step = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    lr_base = 5.0
    tau_start = 50.0 / n
    tau_end = 0.1 / n
    
    momentum1 = np.zeros(n)
    momentum2 = np.zeros(n)
    theta = theta_init.copy()
    
    best_theta = theta.copy()
    best_score = np.inf
    plateau = 0
    plateau_best = np.inf
    lp_counter = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** (elapsed_frac * 1.2)
        
        step += 1
        lr = lr_base / np.sqrt(1 + step * 0.01)
        
        theta, score, momentum1, momentum2 = gradient_descent_step(
            theta, tau, n, momentum1, momentum2, beta1, beta2, eps, lr, step
        )
        
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_theta = theta.copy()
            plateau = 0
            plateau_best = score
        else:
            plateau += 1
        
        # Occasional LP jump
        if use_lp and step % 5000 == 0 and time.monotonic() - t0 < time_limit - 5:
            a = softmax(theta)
            c = fft_convolve(a, a)
            M = np.max(c)
            g = lp_jump(a, M)
            if g is not None:
                g = g / np.sum(g)  # normalize
                theta_lp = np.log(g + 1e-10)
                # Blend
                t_blend = 0.3
                theta = (1 - t_blend) * theta + t_blend * theta_lp
                # Don't update best here, just use as new starting point
        
        # Random restart if stuck for too long
        if plateau > 20000:
            theta += rng.normal(0, 0.1, size=n)
            plateau = 0
        
        if step % 10000 == 0:
            a = softmax(theta)
            actual_score = score_normalized(a)
            print(f"    step={step}, approx_score={score:.6f}, actual={actual_score:.6f}, tau={tau:.2e}, t={time.monotonic()-t0:.1f}s")
    
    a_best = softmax(best_theta)
    final_score = score_normalized(a_best)
    return best_theta, final_score

def polish_local(a_init, time_limit, rng):
    """Simulated annealing local search."""
    current = a_init.astype(np.float64).copy()
    current = np.maximum(current, 0.01)
    n = len(current)
    current_score = score_normalized(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()
    temperature = 0.002
    cooling = 0.9995
    
    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()
        
        if r < 0.4:
            # Targeted peak reduction
            conv = fft_convolve(current, current)
            peak_idx = int(np.argmax(conv))
            # Identify contributing indices
            contribs = np.zeros(n)
            lo = max(0, peak_idx - n + 1)
            hi = min(n - 1, peak_idx)
            for i in range(lo, hi + 1):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            top_k = max(3, n // 10)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.02, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(10, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.05, 0.02, size=n_inc))
        elif r < 0.6:
            # Block perturbation
            width = max(1, int(n * rng.uniform(0.01, 0.1)))
            start = int(rng.integers(0, n - width))
            candidate[start:start+width] *= np.exp(rng.normal(0, 0.04, size=width))
        elif r < 0.75:
            # Symmetry breaking
            asym = 0.5 * (current - current[::-1])
            candidate = current + 0.1 * rng.random() * asym
            candidate = np.maximum(candidate, 0.01)
        elif r < 0.9:
            # Global small perturbation
            candidate = current * np.exp(rng.normal(0, 0.015, size=n))
        else:
            # Large jump with recovery
            candidate = current * np.exp(rng.normal(0, 0.08, size=n))
            candidate = 0.7 * candidate + 0.3 * current
        
        candidate = np.clip(candidate, 0.01, 1000.0)
        new_score = score_normalized(candidate)
        
        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-15)):
            current = candidate
            current_score = new_score
        
        temperature *= cooling
    
    return best, best_score

import numpy as np

from scipy.optimize import linprog


def fft_convolve(a, b):
    """Fast linear convolution using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the score for a non-negative sequence."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)


def score_normalized(a):
    """Score assuming we normalize to sum=1 internally."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.inf
    a = a / s
    n = len(a)
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c)


def softmax(x):
    """Numerically stable softmax."""
    x = x - np.max(x)
    e = np.exp(x)
    return e / np.sum(e)


def soft_max_weights(b, tau):
    """Compute soft-max value and weight vector for b with temperature tau."""
    b_max = np.max(b)
    shifted = (b - b_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    value = b_max + tau * np.log(sum_exp)
    return value, w


def grad_soft_max_conv(a, tau):
    """
    Compute soft-max of conv(a,a) and its gradient w.r.t. a.
    Returns (soft_max_value, grad_wrt_a, convolution).
    """
    n = len(a)
    c = fft_convolve(a, a)
    value, w = soft_max_weights(c, tau)
    # Gradient: d(soft_max)/da[j] = 2 * sum_k w[k] * a[k-j] (valid indices)
    flip_a = a[::-1]
    corr = fft_convolve(w, flip_a)
    g = 2.0 * corr[n - 1: 2 * n - 1]
    return value, g, c


def adam_step(theta, g, momentum1, momentum2, step, beta1, beta2, eps, lr):
    """One Adam optimizer step."""
    momentum1 = beta1 * momentum1 + (1 - beta1) * g
    momentum2 = beta2 * momentum2 + (1 - beta2) * (g * g)
    m1_hat = momentum1 / (1.0 - beta1 ** step)
    m2_hat = momentum2 / (1.0 - beta2 ** step)
    theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)
    return theta, momentum1, momentum2


def lp_perturbation(f, M, timeout=5.0):
    """
    Solve LP: maximize sum(b) s.t. conv(f,b)[k] <= M for all k, b >= 0.
    Returns the optimal b or None.
    """
    n = len(f)
    c_obj = -np.ones(n)
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_min = j
        k_max = min(2 * n - 2, j + n - 1)
        indices = np.arange(k_min, k_max + 1)
        A[indices, j] = f[n - 1 - (k_max - j)::-1] if k_max >= n else f[k_min - j:n - (k_min - j)]
    # Redo properly
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_lo = max(0, k - (n - 1))
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            A[k, j] = f[k - j]

    b_ub = np.full(2 * n - 1, M)
    t_start = time.monotonic()
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs',
                  options={'time_limit': timeout, 'maxiter': 100000})
    if time.monotonic() - t_start > timeout + 1:
        return None
    if res.success and np.sum(res.x) > 1e-10:
        return np.maximum(res.x, 0.0)
    return None


def make_chirp(n, alpha, rng):

def make_window(n, wtype, rng):
    """Standard window functions."""
    t = np.arange(n, dtype=np.float64) / max(1, n - 1)

def make_multitone(n, rng, n_tones=4):
    """Sum of random sinusoids."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    freqs = rng.choice(np.arange(1, n // 2 + 1), size=min(n_tones, n // 2), replace=False)
    for k in freqs:
        a += 0.12 * np.cos(2 * np.pi * k * t / n + rng.uniform(0, 2 * np.pi))
    return np.clip(a, 0.01, None)


def make_barker_lift(n, rng):
    """
    Lift a Barker-like ±1 sequence to non-negative.
    Use the periodic sequence with minimum autocorrelation peak.
    """
    # Generate a ±1 sequence with low peak autocorrelation
    # Use a m-sequence or random search for small n
    s = rng.choice([-1, 1], size=n)
    # Try to reduce peak autocorrelation via bit flips
    best_s = s.copy()
    best_peak = np.inf
    for _ in range(min(200, n * 5)):
        idx = rng.integers(0, n)
        s = best_s.copy()
        s[idx] *= -1
        ac = np.abs(fft_convolve(s, s))

            best_s = s.copy()
    # Lift to non-negative: a = s + 2 (values in {1, 3})
    a = best_s.astype(float) + 2.0
    return a
