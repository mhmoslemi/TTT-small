```python
    row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))

def solve_lp_step(a, M, timeout=30.0):
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None

def lp_refine(a, time_limit_s=600.0, max_iters=500, verbose=True):
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    if verbose:
        print(f"    LP start: score={best_score:.6f}, n={n}")
    stagnant = 0
    it = 0
    for it in range(max_iters):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break
        conv = np.convolve(a, a)
        score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 50 == 0 or it < 5):
                print(f"    LP iter {it:4d}: score={score:.6f}")
        else:
            stagnant += 1
        if stagnant > 40:
            if verbose:
                print(f"    LP stagnant at iter {it}, best={best_score:.6f}")
            break
        M = conv.max()
        remaining = time_limit_s - elapsed
        if remaining < 5.0:
            break
        timeout = max(3.0, min(60.0, remaining * 0.3))
        g = solve_lp_step(a, M, timeout=timeout)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-10:
            if verbose:
                print(f"    LP no improvement (sum_g={sum_g:.10f}) at iter {it}")
            break
        g = g / sum_g
        alpha = min(0.6, max(0.05, 1.5 / (1.0 + 0.3 * it)))
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = normalize_nonneg(a_new)
        new_score = score_sequence(a_new)
        if new_score <= score + 1e-14:
            a = a_new
        else:
            alpha2 = alpha * 0.2
            a_new = (1.0 - alpha2) * a + alpha2 * g
            a_new = normalize_nonneg(a_new)
            a = a_new
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    if verbose:
        print(f"    LP done: best={best_score:.6f} after {it+1} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score

def gs_refine(a, time_limit_s=120.0, alpha=0.3, verbose=True):
    """Gerchberg-Saxton style iteration to flatten auto-convolution."""
    start_time = time.monotonic()
    a = a.copy().astype(np.float64)
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    # Choose FFT length (power of 2 >= 2n-1)
    L = 1
    while L < 2 * n - 1:
        L *= 2
    
    if verbose:
        print(f"    GS start: score={best_score:.6f}, n={n}, FFT_L={L}")
    
    stagnant = 0
    it = 0
    for it in range(10000):
        if time.monotonic() - start_time > time_limit_s:
            break
        
        a_padded = np.zeros(L)
        a_padded[:n] = a
        A = np.fft.fft(a_padded)
        C = A * A
        c = np.fft.ifft(C).real
        
        # c should be the auto-convolution (zero-padded to length L)
        # The relevant part is c[0:2n-1]
        
        # Compress the peaks
        c_mean = c.mean()
        # Soft compression: reduce values above mean, increase below
        c_compressed = c * np.exp(-alpha * (c - c_mean) / max(c_mean, 1e-10))
        # Or simpler:
        # c_compressed = (1 - alpha) * c + alpha * c_mean
        
        # Get new spectrum
        C_new = np.fft.fft(c_compressed)
        
        # A_new = sqrt(C_new), but we need to be careful
        # Use: A_new = A * sqrt(C_new / C) to preserve phase structure
        ratio = C_new / (C + 1e-30 + 0j)
        A_new = A * np.sqrt(ratio)
        
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(0.0, a_new)
        
        new_score = score_sequence(a_new)
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_new.copy()
            stagnant = 0
            if verbose and (it % 100 == 0 or it < 10):
                print(f"    GS iter {it:5d}: score={new_score:.6f}")
        else:
            stagnant += 1
            if stagnant > 200:
                if verbose:
                    print(f"    GS stagnant at iter {it}, best={best_score:.6f}")
                break
        
        a = a_new
    
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    if verbose:
        print(f"    GS done: best={best_score:.6f} after {it+1} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score

def local_search_refine(a, time_limit_s=120.0, rng=None, verbose=True):
    if rng is None:
        rng = np.random.default_rng(0)
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    if verbose:
        print(f"    LS start: score={best_score:.6f}")
    accept_count = 0
    while time.monotonic() - start_time < time_limit_s:
        width = max(2, int(rng.integers(n // 200, n // 8)))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)
        candidate = a.copy()
        candidate[start:stop] *= np.exp(rng.normal(0.0, 0.1, size=stop - start))
        if rng.random() < 0.3:
            candidate = 0.5 * (candidate + candidate[::-1])
        candidate = normalize_nonneg(candidate)
        score = score_sequence(candidate)
        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1
            if verbose and accept_count % 50 == 0:
                print(f"    LS accept {accept_count}: score={best_score:.6f}")
    if verbose:
        print(f"    LS done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score

def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))

def initial_gaussian(n, sigma_frac=0.25):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)

def initial_blackman(n):
    t = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.ones(1, dtype=np.float64)
    x = t / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))

def random_smooth_init(n, rng, n_bumps=3):
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(1, n // 10), max(2, n // 3)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return initial_raised_cosine(n)
    return a / s

def two_bump_init(n, rng):
    """Initialize with two well-separated bumps."""
    a = np.zeros(n)
    t = np.arange(n, dtype=np.float64)
    center1 = float(n * rng.uniform(0.15, 0.35))
    center2 = float(n * rng.uniform(0.65, 0.85))
    width = n * 0.15
    a += np.exp(-0.5 * ((t - center1) / width) ** 2)
    a += np.exp(-0.5 * ((t - center2) / width) ** 2)
    return a / a.sum()

def comb_init(n, n_teeth=3):
    """Initialize with a comb-like pattern."""
    a = np.zeros(n)
    for i in range(n_teeth):
        center = (i + 0.5) * n / n_teeth
        width = n / (n_teeth * 4)
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    return a / a.sum()
```

Wait, I need to think more carefully about the GS method. Let me reconsider.

The issue with the GS method as I described it is that the complex square root can introduce phase discontinuities. Let me use a different approach:

Instead of A_new = sqrt(C_new), I'll use:
- Compute c = ifft(A^2).real (the auto-convolution)
- Modify c to be flatter: c_new = c / (1 + beta * c / c_max) or c_new = (1-alpha)*c + alpha*c_mean
- Compute the new spectrum: C_new = fft(c_new)
- Update A: A_new = A * (C_new / C)^(1/2) element-wise

Actually, (C_new / C)^(1/2) can have issues when C is near zero. Let me use a different approach.

Alternative: Use the "error reduction" GS:
1. a -> A = fft(a_padded)
2. c = ifft(A^2).real (auto-conv)
3. c_target = some flat target (e.g., c_mean * ones)
4. error = c - c_target
5. A_correction = fft(error) / (2 * A) (linearized correction)
6. A_new = A - A_correction
7. a_new = ifft(A_new).real[:n]
8. a_new = max(0, a_new)

This is like a Newton step for the constraint that the auto-conv equals the target.

Let me think about this more carefully. We want to find a such that ifft(fft(a)^2) is as flat as possible. Let's define:
F(a) = ifft(fft(a_padded)^2) - target

We want F(a) = 0. The linearization is:
F(a + da) ≈ F(a) + DF[a] * da

Where DF[a] * da = ifft(2 * A * fft(da_padded))

So we want: F(a) + ifft(2 * A * fft(da)) = 0
=> fft(da) = -F(a) * L / (2 * A)  (where L is the FFT length, for the scaling)
=> da = ifft(-fft(F(a)) / (2 * A))

Wait, let me be more careful with the FFT scaling.

Let L be the FFT length. If x is a vector of length L, and X = FFT(x), then:
IFFT(X * Y) = (1/L) * (x * y) in circular convolution? No, that's not right.

Actually: IFFT(FFT(x) * FFT(y)) = circular_convolution(x, y)

So if C = A^2 where A = FFT(a_padded), then:
c = IFFT(C) = circular_convolution(a_padded, a_padded)

For the perturbation: if a_padded -> a_padded + da_padded, then:
A -> A + da_A where da_A = FFT(da_padded)
C -> (A + da_A)^2 ≈ A^2 + 2*A*da_A
c -> c + IFFT(2*A*da_A) = c + 2*circular_convolution(a_padded, da_padded)

So the linearized model is:
c + 2*circ_conv(a_padded, da_padded) = target
=> circ_conv(a_padded, da_padded) = (target - c) / 2
=> FFT(circ_conv(a_padded, da_padded)) = FFT(a_padded) * FFT(da_padded) = A * FFT(da_padded)
=> A * FFT(da_padded) = FFT((target - c)/2)
=> FFT(da_padded) = FFT((target - c)/2) / A
=> da_padded = IFFT(FFT((target - c)/2) / A)

This is the Newton step! Let me implement this:

```python
def gs_newton_refine(a, time_limit_s=120.0, alpha=0.5, verbose=True):
    start_time = time.monotonic()
    a = a.copy().astype(np.float64)
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    L = 1
    while L < 2 * n - 1:
        L *= 2
    
    if verbose:
        print(f"    GS-Newton start: score={best_score:.6f}, n={n}, L={L}")
    
    stagnant = 0
    for it in range(5000):
        if time.monotonic() - start_time > time_limit_s:
            break
        
        a_padded = np.zeros(L)
        a_padded[:n] = a
        A = np.fft.fft(a_padded)
        C = A * A
        c = np.fft.ifft(C).real
        
        # Target: flat
        c_target = np.full_like(c, c.mean())
        
        # Newton step
        residual = (c_target - c) / 2.0
        R = np.fft.fft(residual)
        # da_padded = IFFT(R / A)
        # Handle zeros in A
        mask = np.abs(A) > 1e-10
        da_A = np.zeros_like(A)
        da_A[mask] = R[mask] / A[mask]
        da_padded = np.fft.ifft(da_A).real
        
        # Damped step
        a_new = a_padded + alpha * da_padded
        a_new = a_new[:n]
        a_new = np.maximum(0.0, a_new)
        
        new_score = score_sequence(a_new)
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_new.copy()
            stagnant = 0
            if verbose and (it % 100 == 0 or it < 10):
                print(f"    GS-Newton iter {it:5d}: score={new_score:.6f}")
        else:
            stagnant += 1
            # Reduce step size
            alpha *= 0.9
            if stagnant > 100:
                if verbose:
                    print(f"    GS-Newton stagnant, best={best_score:.6f}")
                break
        
        a = a_new
    
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    if verbose:
        print(f"    GS-Newton done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score
```

This Newton approach should converge faster and more reliably than the amplitude-modulation GS approach.

Let me also think about whether there are any other tricks:

1. **Alternating optimization**: Alternate between LP and GS-Newton steps. The LP finds the optimal response, and the GS-Newton flattens the spectrum.

2. **Simulated annealing**: Accept worse solutions with some probability to escape local minima.

3. **Objective-based regularization**: Instead of just minimizing the peak, minimize a "smoothed" peak like sum(exp(beta * conv[k])) for large beta. This gives a differentiable objective that we can optimize with gradient methods.

Let me think about option 3 more carefully. Define:
g(a) = sum_k exp(beta * conv(a,a)[k] / M) where M is some normalization

The gradient of g with respect to a[i] is:
dg/da[i] = (2*beta/M) * sum_k exp(beta*conv[k]/M) * a[k-i] (for valid k-i)

For large beta, this approximates (2*beta/M) * exp(beta*max_conv/M) * sum_{k:conv[k]=max} a[k-i]

So for large beta, the gradient is concentrated at the peaks, similar to the LP approach but smoother.

This could be optimized with simple gradient descent + projection to non-negative orthant + normalization.

Let me implement this as well:

```python
def soft_peak_refine(a, time_limit_s=120.0, beta=50.0, lr=0.01, verbose=True):
    """Minimize sum(exp(beta * conv[k])) as a smooth approximation to max(conv)."""
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    if verbose:
        print(f"    SoftPeak start: score={best_score:.6f}, beta={beta}, lr={lr}")
    
    stagnant = 0
    for it in range(10000):
        if time.monotonic() - start_time > time_limit_s:
            break
        
        conv = np.convolve(a, a)
        M = conv.max()
        if M < 1e-15:
            break
        
        # Compute soft peak and its gradient
        exp_conv = np.exp(beta * conv / M)
        # Gradient: d/d a[i] of sum_k exp_conv[k] = 2 * beta/M * sum_k exp_conv[k] * a[k-i]
        # This is 2*beta/M * convolve(exp_conv, a) (circular or linear?)
        # For linear convolution, the gradient of conv(a,a)[k] w.r.t. a[i] is a[k-i] (if valid)
        # So gradient of sum_k f(conv[k]) w.r.t. a[i] = sum_k f'(conv[k]) * a[k-i]
        # = convolve(f'(conv), a)[i] (in the appropriate sense)
        
        # Actually, let's compute it directly:
        # grad[i] = sum_k (2*beta/M) * exp_conv[k] * a[k-i] for all valid (k,i) pairs
        # This is 2*beta/M * convolve(exp_conv, a) but we need to be careful about the indexing
        
        # conv(a, a)[k] = sum_i a[i] * a[k-i]
        # d/d a[i] conv(a,a)[k] = a[k-i] + a[i] * d/d a[i] a[k-i]
        # Wait, conv(a,a)[k] = sum_j a[j] * a[k-j]
        # d/d a[i] = sum_j [delta(j=i) * a[k-j] + a[j] * delta(k-j=i)]
        # = a[k-i] + a[k-i] = 2*a[k-i] (if 0 <= k-i < n)
```

Hybrid optimization for minimum peak auto-convolution of non-negative sequences.
Combines LP-based refinement, Gerchberg-Saxton Newton spectral iteration,
soft-peak gradient descent, and local search with multiple diverse initializations.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def build_conv_matrix(a):
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp_step(a, M, timeout=30.0):
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def lp_refine(a, time_limit_s=600.0, max_iters=500, verbose=True):
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    if verbose:
        print(f"    LP start: score={best_score:.6f}, n={n}")
    stagnant = 0
    it = 0
    for it in range(max_iters):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break
        conv = np.convolve(a, a)
        score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 50 == 0 or it < 5):
                print(f"    LP iter {it:4d}: score={score:.6f}")
        else:
            stagnant += 1
        if stagnant > 40:
            if verbose:
                print(f"    LP stagnant at iter {it}, best={best_score:.6f}")
            break
        M = conv.max()
        remaining = time_limit_s - elapsed
        if remaining < 5.0:
            break
        timeout = max(3.0, min(60.0, remaining * 0.3))
        g = solve_lp_step(a, M, timeout=timeout)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-10:
            if verbose:
                print(f"    LP no improvement (sum_g={sum_g:.10f}) at iter {it}")
            break
        g = g / sum_g
        alpha = min(0.6, max(0.05, 1.5 / (1.0 + 0.3 * it)))
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = normalize_nonneg(a_new)
        new_score = score_sequence(a_new)
        if new_score <= score + 1e-14:
            a = a_new
        else:
            alpha2 = alpha * 0.2
            a_new = (1.0 - alpha2) * a + alpha2 * g
            a_new = normalize_nonneg(a_new)
            a = a_new
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    if verbose:
        print(f"    LP done: best={best_score:.6f} after {it+1} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def gs_newton_refine(a, time_limit_s=120.0, alpha=0.5, verbose=True):
    """Gerchberg-Saxton with Newton updates to flatten auto-convolution in frequency domain."""
    start_time = time.monotonic()
    a = a.copy().astype(np.float64)
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)

    L = 1
    while L < 2 * n - 1:
        L *= 2

    if verbose:
        print(f"    GS-Newton start: score={best_score:.6f}, n={n}, L={L}")

    stagnant = 0
    cur_alpha = alpha
    for it in range(10000):
        if time.monotonic() - start_time > time_limit_s:
            break

        a_padded = np.zeros(L)
        a_padded[:n] = a
        A = np.fft.fft(a_padded)
        C = A * A
        c = np.fft.ifft(C).real

        c_target = np.full_like(c, c.mean())
        residual = (c_target - c) / 2.0
        R = np.fft.fft(residual)

        mask = np.abs(A) > 1e-12
        da_A = np.zeros_like(A, dtype=np.complex128)
        da_A[mask] = R[mask] / A[mask]
        da_padded = np.fft.ifft(da_A).real

        a_new = a_padded + cur_alpha * da_padded
        a_new = a_new[:n]
        a_new = np.maximum(0.0, a_new)
        a_new = normalize_nonneg(a_new)

        new_score = score_sequence(a_new)
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_new.copy()
            stagnant = 0
            cur_alpha = min(1.0, cur_alpha * 1.05)
            a = a_new
            if verbose and (it % 200 == 0 or it < 10):
                print(f"    GS-Newton iter {it:5d}: score={new_score:.6f}, alpha={cur_alpha:.3f}")
        else:
            stagnant += 1
            cur_alpha *= 0.95
            if stagnant > 200:
                if verbose:
                    print(f"    GS-Newton stagnant, best={best_score:.6f}")
                break

    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    if verbose:
        print(f"    GS-Newton done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def soft_peak_refine(a, time_limit_s=120.0, beta_init=30.0, lr_init=0.05, verbose=True):
    """Minimize sum(exp(beta*conv[k])) as smooth approximation to max(conv)."""
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    beta = beta_init
    lr = lr_init

    if verbose:
        print(f"    SoftPeak start: score={best_score:.6f}, beta={beta}, lr={lr}")

    stagnant = 0
    for it in range(20000):
        if time.monotonic() - start_time > time_limit_s:
            break

        conv = np.convolve(a, a)
        M = conv.max()
        if M < 1e-15:
            break

        scaled = beta * conv / M
        np.clip(scaled, -50, 50, out=scaled)
        exp_conv = np.exp(scaled)

        corr_valid = np.correlate(exp_conv, a, mode='valid')
        grad = 2.0 * beta / M * corr_valid

        a_new = a - lr * grad
        a_new = np.maximum(0.0, a_new)
        a_new = normalize_nonneg(a_new)

        new_score = score_sequence(a_new)
        if new_score < best_score - 1e-14:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            stagnant = 0
            if verbose and (it % 500 == 0 or it < 10):
                print(f"    SoftPeak iter {it:5d}: score={new_score:.6f}, beta={beta:.1f}")
        else:
            stagnant += 1
            if stagnant % 50 == 0:
                lr *= 0.9
                beta *= 1.5
                if lr < 1e-8:
                    break
            if stagnant > 500:
                beta *= 3.0
                lr = lr_init * 0.1
                stagnant = 0
                if beta > 5000:
                    if verbose:
                        print(f"    SoftPeak done (beta too high), best={best_score:.6f}")
                    break

    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    if verbose:
        print(f"    SoftPeak done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def local_search_refine(a, time_limit_s=120.0, rng=None, verbose=True):
    if rng is None:
        rng = np.random.default_rng(0)
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    if verbose:
        print(f"    LS start: score={best_score:.6f}")
    accept_count = 0
    while time.monotonic() - start_time < time_limit_s:
        width = max(2, int(rng.integers(n // 200, n // 8)))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)
        candidate = a.copy()
        candidate[start:stop] *= np.exp(rng.normal(0.0, 0.1, size=stop - start))
        if rng.random() < 0.3:
            candidate = 0.5 * (candidate + candidate[::-1])
        candidate = normalize_nonneg(candidate)
        score = score_sequence(candidate)
        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1
            if verbose and accept_count % 50 == 0:
                print(f"    LS accept {accept_count}: score={best_score:.6f}")
    if verbose:
        print(f"    LS done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_gaussian(n, sigma_frac=0.25):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_blackman(n):
    t = np.arange(n, dtype=np.float64)
    if n == 1:
        return np.ones(1, dtype=np.float64)
    x = t / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))


def random_smooth_init(n, rng, n_bumps=3):
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(1, n // 10), max(2, n // 3)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return initial_raised_cosine(n)
    return a / s


def two_bump_init(n, rng):
    a = np.zeros(n)
    t = np.arange(n, dtype=np.float64)
    center1 = float(n * rng.uniform(0.15, 0.35))
    center2 = float(n * rng.uniform(0.65, 0.85))
    width = n * 0.15
    a += np.exp(-0.5 * ((t - center1) / width) ** 2)
    a += np.exp(-0.5 * ((t - center2) / width) ** 2)
    return a / a.sum()


def comb_init(n, n_teeth=4):
    a = np.zeros(n)
    for i in range(n_teeth):
        center = (i + 0.5) * n / n_teeth
        width = n / (n_teeth * 3.0)
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    return a / a.sum()


def wavelet_init(n, rng):
    """Initialize with a wavelet-like multi-scale structure."""
    a = np.zeros(n)
    t = np.arange(n, dtype=np.float64)
    # Add multiple frequency components
    for freq in [1, 2, 3, 5]:
        phase = rng.uniform(0, 2 * np.pi)
        a += np.cos(2 * np.pi * freq * t / n + phase)
    a = np.abs(a) ** 2
    return a / a.sum()

    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + max(10.0, budget_s - 20.0)

        gs_time = time_limit * 0.25
