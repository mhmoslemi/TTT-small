def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

Multi-phase optimization for minimizing normalized autocorrelation peak of
non-negative step functions. Combines improved gradient descent (Nesterov
momentum, adaptive soft-max), LP-based refinement (state-of-the-art active
set method), and targeted peak-reduction local search.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the score (same formula as evaluate_sequence)."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def normalize_to_sum(a, target_sum):
    """Normalize a to have the given sum, maintaining non-negativity."""
    s = np.sum(a)
    if s < 1e-15:
        return a
    return a * (target_sum / s)


def soft_max_grad_conv(a, tau):
    """
    Compute soft-max value and gradient of soft_max(conv(a,a), tau).
    Gradient: grad[j] = sum_k w_k * (a[j] + a[k-j]) * 1[0<=k-j<n]
             = a[j] * sum_{k=j}^{j+n-1} w_k + cross_corr(a, w)[j]
    where w is the soft-max weights on conv(a,a).
    """
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    value = c_max + tau * np.log(sum_exp)

    # Gradient computation:
    # grad[j] = a[j] * S[j] + X[j]
    # where S[j] = sum_{k: 0<=k-j<n} w_k = sum_{k=j}^{j+n-1} w_k (clamped)
    # and X[j] = sum_m a[m] * w[m+j] (cross-correlation)

    # S[j] via cumulative sum of w
    w_cumsum = np.zeros(len(w) + 1)
    w_cumsum[1:] = np.cumsum(w)
    S = np.zeros(n)
    for j in range(n):
        k_start = j
        k_end = j + n  # exclusive
        k_start = max(0, k_start)
        k_end = min(len(w), k_end)
        S[j] = w_cumsum[k_end] - w_cumsum[k_start]

    # X[j] = sum_m a[m] * w[m+j] for m=0..n-1, m+j in range of w
    # This is cross-correlation of a and w
    # X[j] = sum_{m=0}^{n-1} a[m] * w[m+j]
    # Use FFT: pad a and w, compute ifft(FFT(a) * conj(FFT(w)))
    size = 1
    while size < n + len(w) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w, size)
    cross = np.fft.irfft(fa * np.conj(fw), size)
    # cross[j] = sum_m a[m] * w[m+j]... let me verify
    # Actually ifft(FFT(a) * conj(FFT(w)))[j] = sum_m a[m] * w[m-j] (autocorrelation-like)
    # I need sum_m a[m] * w[m+j] which is cross-correlation with a different convention
    # cross_corr(j) = sum_m a[m] * conj(w[m+j])
    # = sum_m a[m] * w[m+j] (since w is real)
    # ifft(FFT(a) * conj(FFT(w)))[j] = sum_m a[m] * w[m-j+n] (circular)
    # For the linear version: cross_corr[j] = sum_{m} a[m] * w[m+j]
    # This equals ifft(FFT(w) * conj(FFT(a)))[n-1-j]... this is getting confusing.
    
    # Let me just use the direct formula for X:
    # X[j] = sum_{m=0}^{n-1} a[m] * w[m+j] where w has length 2n-1
    X = np.zeros(n)
    for j in range(n):
        # w[m+j] for m=0..n-1, valid when m+j < 2n-1, i.e., always true for m<n, j<n
        X[j] = np.dot(a, w[j:j+n])

    grad = a * S + X
    return value, grad, c


def gradient_descent(a_init, time_limit, rng, verbose=False):
    """
    Adam gradient descent with Nesterov momentum and annealed soft-max.
    Direct non-negative parameterization with projection.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = normalize_to_sum(np.clip(a_init, 0.01, 1000.0), S)
    
    best_a = a.copy()
    best_score = local_score(a)
    
    # Adam + Nesterov
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step = 0
    t0 = time.monotonic()
    
    tau_start = 50.0 / n
    tau_end = 0.1 / n
    
    # Learning rate schedule
    lr_init = 3.0
    lr_min = 0.05
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        # Cosine annealing
        lr = lr_min + 0.5 * (lr_init - lr_min) * (1 + np.cos(np.pi * elapsed_frac))
        
        # Temperature: slower annealing
        tau = tau_start * (tau_end / tau_start) ** (elapsed_frac ** 0.7)
        
        # Compute gradient of soft-max(conv(a,a))
        value, grad_c, c = soft_max_grad_conv(a, tau)
        
        # Total gradient includes normalization constraint
        # We minimize soft_max(conv(a,a)) subject to sum(a) = S, a >= 0
        # Using Lagrange multiplier: grad_total = grad_c - lambda * 1
        # lambda = sum(a * grad_c) / S
        lam = np.dot(a, grad_c) / S
        grad_total = grad_c - lam
        
        # Nesterov + Adam
        step += 1
        m = beta1 * m + (1 - beta1) * grad_total
        v = beta2 * v + (1 - beta2) * (grad_total * grad_total)
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project: clip to non-negative, rescale
        a_new = np.clip(a_new, 0.0, 1000.0)
        a_new = normalize_to_sum(a_new, S)
        
        score = local_score(a_new)
        
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and step % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GD step={step}, score={score:.6f}, best={best_score:.6f}, "
                  f"tau={tau:.2e}, lr={lr:.4f}, t={elapsed:.1f}s")
        
        # Adaptive: if stuck for a while, perturb
        if step % 2000 == 0 and score > best_score * 1.01:
            a = a * np.exp(rng.normal(0, 0.02, n))
            a = normalize_to_sum(np.clip(a, 0.01, 1000.0), S)
    
    return best_a, best_score


def build_conv_matrix(a):
    """Build matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a[:n]
    return A


def lp_refine(a_init, time_limit, rng, verbose=False):
    """
    LP-based refinement following the state of the art:
    At each iteration, solve LP to find g maximizing sum(g)
    subject to g >= 0 and conv(a, g) <= M (current peak).
    Update a = (1-t)*a + t*g, rescale.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = normalize_to_sum(np.clip(a_init, 0.01, 1000.0), S).copy()
    
    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    it = 0
    
    # Pre-compute structure (indices for the convolution matrix)
    # A[k, j] = a[k-j] for valid k, j
    # For fixed j: A[k, j] is non-zero for k in [j, j+n-1]
    
    while time.monotonic() - t0 < time_limit:
        conv_a = fft_convolve(a, a)
        current_peak = np.max(conv_a)
        
        it += 1
        
        # Build constraint matrix
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            A[j:j + n, j] = a
        
        # LP: max sum(g) s.t. A@g <= M, g >= 0
        c_obj = -np.ones(n)
        b_cons = np.full(2 * n - 1, current_peak)
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_cons,
                          bounds=[(0, None)] * n, method='highs',
                          options={'presolve': True, 'time_limit': 5.0})
        except Exception:
            break
        
        if not res.success or res.x is None:
            # Try with relaxed bound
            b_cons = np.full(2 * n - 1, current_peak * 1.05)
            try:
                res = linprog(c_obj, A_ub=A, b_ub=b_cons,
                              bounds=[(0, None)] * n, method='highs',
                              options={'presolve': True, 'time_limit': 5.0})
            except Exception:
                break
            if not res.success or res.x is None:
                break
        
        g0 = np.clip(res.x, 0, None)
        sum_g = np.sum(g0)
        
        if sum_g < 1e-10:
            break
        
        # Try multiple step sizes
        a_cand_best = a
        score_cand_best = local_score(a)
        
        for t in [0.05, 0.02, 0.01, 0.005, 0.001]:
            a_cand = (1.0 - t) * a + t * g0
            a_cand = np.clip(a_cand, 0.0, 1000.0)
            a_cand = normalize_to_sum(a_cand, S)
            sc = local_score(a_cand)
            if sc < score_cand_best:
                score_cand_best = sc
                a_cand_best = a_cand
        
        if score_cand_best < best_score:
            best_score = score_cand_best
            best_a = a_cand_best.copy()
            a = a_cand_best
        else:
            # No improvement, try smaller perturbation
            a = a * np.exp(rng.normal(0, 0.005, n))
            a = normalize_to_sum(np.clip(a, 0.01, 1000.0), S)
        
        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={it}, score={score_cand_best:.6f}, "
                  f"best={best_score:.6f}, t={elapsed:.1f}s")
    
    return best_a, best_score


def local_search(a_init, time_limit, rng, verbose=False):
    """
    Targeted local search with peak reduction.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = normalize_to_sum(np.clip(a_init, 0.01, 1000.0), S).copy()
    
    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    
    temperature = 0.002
    cooling = 0.99995
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        r = rng.random()
        candidate = a.copy()
        
        if r < 0.4:
            # Targeted: reduce top contributors to peak
            conv = fft_convolve(a, a)
            peak_idx = int(np.argmax(conv))
            # Contributors: a[i] * a[peak_idx - i] for valid i
            i_start = max(0, peak_idx - n + 1)
            i_end = min(n - 1, peak_idx)
            contrib = np.zeros(n)
            for i in range(i_start, i_end + 1):
                j = peak_idx - i
                contrib[i] = a[i] * a[j]
            
            top_k = max(2, min(n // 8, 32))
            top_idx = np.argsort(contrib)[-top_k:]
            # Reduce top contributors
            candidate[top_idx] *= np.exp(rng.normal(-0.03, 0.01, size=len(top_idx)))
            # Increase non-contributors
            others = np.setdiff1d(np.arange(n), top_idx)
            if len(others) > 0:
                n_inc = min(10, len(others))
                inc_idx = rng.choice(others, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.03, 0.01, size=n_inc))
                
        elif r < 0.55:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 12))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.025, size=width))
            
        elif r < 0.70:
            # Pair swap: move mass between two positions
            i, j = rng.choice(n, size=2, replace=False)
            amt = a[i] * rng.uniform(0.05, 0.2)
            candidate[i] -= amt
            candidate[j] += amt
            candidate = np.clip(candidate, 0.0, None)
            
        elif r < 0.80:
            # Symmetrize partially
            sym = 0.5 * (a + a[::-1])
            candidate = sym * np.exp(rng.normal(0.0, 0.008, size=n))
            
        elif r < 0.90:
            # Multi-block
            for _ in range(4):
                width = max(1, int(rng.integers(1, max(2, n // 20))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.02, size=width))
        else:
            # Global small perturbation
            candidate = a * np.exp(rng.normal(0.0, 0.008, size=n))
        
        candidate = np.clip(candidate, 0.0, 1000.0)
        candidate = normalize_to_sum(candidate, S)
        new_score = local_score(candidate)
        
        if new_score < best_score:
            best_score = new_score
            best_a = candidate.copy()
        
        if new_score < local_score(a):
            a = candidate
        elif rng.random() < np.exp(-(new_score - local_score(a)) / max(temperature, 1e-12)):
            a = candidate
        
        temperature *= cooling
        
        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LS iter={it}, best={best_score:.6f}, temp={temperature:.6f}, t={elapsed:.1f}s")
    
    return best_a, best_score


def make_init(n, rng, init_type='random'):
    """Create initial sequence of length n."""
    if init_type == 'uniform':
        return np.ones(n) + 0.05 * rng.standard_normal(n)
    elif init_type == 'chirp':
        t = np.arange(n, dtype=np.float64)
        alpha = 0.3 + 0.3 * rng.random()
        a = 1.0 + alpha * np.cos(np.pi * t * t / n)
        a += 0.03 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    elif init_type == 'multitone':
        t = np.arange(n, dtype=np.float64)
        a = 1.0
        n_tones = rng.integers(3, 8)
        for _ in range(n_tones):
            k = rng.integers(1, n // 3)
            amp = rng.uniform(0.05, 0.2)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * k * t / n + phase)
        a += 0.02 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    elif init_type == 'triangle':
        t = np.arange(n, dtype=np.float64)
        a = 1.0 - 0.3 * np.abs(2 * t / n - 1)
        a += 0.03 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)
    else:
        a = 1.0 + 0.2 * rng.standard_normal(n)
        return np.clip(a, 0.01, None)


def interpolate_sequence(a, new_n):
    """Interpolate sequence to new length."""
    if len(a) == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search:
    1. Gradient-based optimization over multiple initializations
    2. LP-based refinement (state of the art)
    3. Aggressive local search polishing
    4. Additional exploration at different lengths
    """
    t_global = time.monotonic()
    deadline = t_global + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    
    # Check provided sequence
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s = local_score(start_seq)
        print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()
    
    # === Phase 1: Gradient optimization ===
    print("\n=== Phase 1: Gradient optimization ===")
    n_main = 256
    configs = [
        (256, ['chirp', 'multitone', 'triangle', 'random']),
        (384, ['chirp', 'multitone']),
        (512, ['chirp', 'random']),
    ]
    
    for n, init_types in configs:
        if time.monotonic() > deadline - 300:
            break
        time_left = deadline - time.monotonic()
        time_per = time_left / (len(init_types) + 1.0)
        time_per = min(time_per, 120.0)
        
        for init_type in init_types:
            if time.monotonic() > deadline - 250:
                break
            
            if init_type == 'random' and best_seq is not None:
                a0 = interpolate_sequence(best_seq, n)
            else:
                a0 = make_init(n, rng, init_type)
            
            print(f"  n={n}, init='{init_type}'...", end=" ", flush=True)
            a_best, score = gradient_descent(a0, time_per, rng, verbose=False)
            actual_score = local_score(a_best)
            print(f"score={actual_score:.6f}")
            
            if actual_score < best_score:
                best_score = actual_score
                best_seq = a_best.copy()
                print(f"    *** New best: {best_score:.6f} ***")
    
    # === Phase 2: LP-based refinement ===
    print(f"\n=== Phase 2: LP refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_lp = len(best_seq)
        lp_time = min(600.0, max(60.0, deadline - time.monotonic() - 200.0))
        print(f"  LP on n={n_lp}, time_budget={lp_time:.0f}s")
        a_lp, lp_score = lp_refine(best_seq, lp_time, rng, verbose=True)
        if lp_score < best_score:
            best_score = lp_score
            best_seq = a_lp.copy()
            print(f"  *** LP improved to: {best_score:.6f} ***")
        else:
            print(f"  LP final: {lp_score:.6f} (no improvement)")
    
    # === Phase 3: Local search polishing ===
    print(f"\n=== Phase 3: Local search (best={best_score:.6f}) ===")
    if best_seq is not None:
        ls_time = min(300.0, max(30.0, deadline - time.monotonic() - 60.0))
        print(f"  LS time_budget={ls_time:.0f}s")
        a_ls, ls_score = local_search(best_seq, ls_time, rng, verbose=True)
        if ls_score < best_score:
            best_score = ls_score
            best_seq = a_ls.copy()
            print(f"  *** LS improved to: {best_score:.6f} ***")
        else:
            print(f"  LS final: {ls_score:.6f}")
    
    # === Phase 4: Try different lengths ===
    print(f"\n=== Phase 4: Multi-length exploration (best={best_score:.6f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        remaining = deadline - time.monotonic()
        for n_try in [192, 320, 448, 640]:
            if time.monotonic() > deadline - 40:
                break
            a0 = interpolate_sequence(best_seq, n_try)
            t_alloc = min(60.0, remaining * 0.4)
            remaining -= t_alloc
            a_best, sc = gradient_descent(a0, t_alloc * 0.5, rng)
            if t_alloc > 30:
                a_best, sc2 = lp_refine(a_best, t_alloc * 0.3, rng)
                sc = min(sc, sc2)
            if sc < best_score:
                best_score = sc
                best_seq = a_best.copy()
                print(f"  n={n_try}: *** New best: {best_score:.6f} ***")
            else:
                print(f"  n={n_try}: score={sc:.6f}")
    
    # === Phase 5: Final intensive polish ===
    print(f"\n=== Phase 5: Final polish (best={best_score:.6f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 20:
        final_time = deadline - time.monotonic() - 15
        if final_time > 10:
            a_final, final_score = local_search(best_seq, final_time, rng)
            if final_score < best_score:
                best_score = final_score
                best_seq = a_final.copy()
            print(f"  Final: {best_score:.6f}")
    
    # Verify
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score_verified = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score_verified:.6f}, n={len(final_list)} ===")
        return final_list
    
    # Fallback
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        return [float(max(0.0, min(1000.0, x))) for x in height_sequence_1]
    return [1.0] * 256
