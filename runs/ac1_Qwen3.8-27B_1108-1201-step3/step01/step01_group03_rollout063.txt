```python
"""
Optimization of non-negative step function to minimize autocorrelation peak.
Uses LP-based iterative method with line search, multiple starting points,
and adaptive time allocation.
"""
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
import time


def fast_score(f):
    """Compute 2*n*max(conv)/sum^2 efficiently."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))


def build_A(n, f):
    """Build sparse constraint matrix A for the LP.
    A[k, i] = f[k-i] if 0 <= k-i < n else 0, shape (2n-1, n)."""
    # For each column i, the non-zero rows are k = i, i+1, ..., i+n-1
    # with values f[0], f[1], ..., f[n-1]
    rows = []
    cols = []
    data = []
    for i in range(n):
        rows.extend(range(i, i + n))
        cols.extend([i] * n)
        data.extend(f)
    A = sp.coo_matrix((data, (rows, cols)), shape=(2*n-1, n)).tocsr()
    return A


def solve_lp(f, M):
    """Solve max sum(g) s.t. A@g <= M, g >= 0. Returns g or None."""
    n = len(f)
    A = build_A(n, f)
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    bounds = [(0, None)] * n
```

```python
def fast_score(f):

    # Column i has non-zeros at rows i, i+1, ..., i+n-1 with values f[0], f[1], ..., f[n-1]
    cols = np.repeat(np.arange(n), n)
    rows = np.tile(np.arange(n), n) + np.repeat(np.arange(n), n)
    data = np.tile(f, n)
    return sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
```

```python
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
import time


def fast_score(f):
    """Compute 2*n*max(conv)/sum^2 efficiently."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))


def build_A(n, f):
    """Sparse A (2n-1 x n) with A[k,i] = f[k-i] if valid."""
    cols = np.repeat(np.arange(n), n)
    rows = np.tile(np.arange(n), n) + np.repeat(np.arange(n), n)
    data = np.tile(f, n)
    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_primal(f, M, time_limit=4.0):
    """max sum(g) s.t. A@g <= M, g >= 0. Returns normalized g or None."""
    n = len(f)
    A = build_A(n, f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def solve_lp_dual(f, time_limit=4.0):
    """min M s.t. A@g <= M, sum(g)=1, g>=0. Returns (g, M) or None."""
    n = len(f)
    A = build_A(n, f)
    # Variables: g[0..n-1], M
    # Minimize M
    c = np.zeros(n + 1)
    c[n] = 1.0
    # Constraints: A@g - M <= 0, -A@g + M <= 0 -> A@g <= M and -A@g <= -M
    # Actually: A@g <= M means A@g - M <= 0
    # And we need sum(g) = 1 as equality
    # Let's use: A_ub includes (A, -I) for A@g - M <= 0
    # and equality sum(g) = 1
    A_ub = sp.hstack([A, -sp.eye(2 * n - 1)]).tocsr()
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0.0)
            s = g.sum()
            if s > 1e-10:
                return g / s, float(res.x[n])
    except Exception:
        pass
    return None


def line_search(f, g, n_pts=25):
    """Find best t for (1-t)*f + t*g minimizing score."""
    best_t = 0.0
    best_sc = fast_score(f)
    ts = np.linspace(0.02, 1.0, n_pts)
    for t in ts:
        fn = np.maximum((1 - t) * f + t * g, 0.0)
        s = fn.sum()
        if s < 1e-10:
            continue
        fn = fn / s
        sc = fast_score(fn)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    if best_t == 0.0:
        return f, best_sc
    fn = np.maximum((1 - best_t) * f + best_t * g, 0.0)
    return fn / fn.sum(), best_sc


def make_initial(n, rng, method):
    """Create diverse initial sequences."""
    if method == 'uniform':
        a = np.ones(n)
    elif method == 'cosine':
        a = 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))
    elif method == 'half_cosine':
        a = np.cos(np.linspace(0, np.pi / 2, n))
    elif method == 'squared_cos':
        a = np.sin(np.linspace(0, np.pi, n)) ** 2
    elif method == 'bilateral':
        c = n / 2.0
        a = np.exp(-np.abs(np.arange(n) - c) / (n / 6.0))
    elif method == 'random':
        a = rng.random(n) + 0.01
    elif method == 'tapered_up':
        # Higher at edges (to counteract triangular convolution shape)
        x = np.linspace(-1, 1, n)
        a = 1.0 + 0.8 * x**2
    elif method == 'tapered_down':
        x = np.linspace(-1, 1, n)
        a = 1.0 - 0.5 * x**2
    elif method == 'freq_flat':
        # Try to create flat spectrum via inverse DFT approach
        phases = rng.uniform(0, 2 * np.pi, n)
        magnitudes = np.ones(n)
        magnitudes[0] = n  # DC component
        spectrum = magnitudes * np.exp(1j * phases)
        a = np.real(np.ifft(spectrum))
        a = np.maximum(a, 0.01)
    elif method == 'block_structured':
        # Alternating blocks of different heights
        block_size = max(2, n // 16)
        a = np.zeros(n)
        for i in range(0, n, block_size):
            h = 0.5 + rng.random()
            a[i:i + block_size] = h
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def block_perturb(f, rng, strength=0.25):
    """Random multiplicative block perturbation."""
    n = len(f)
    cand = f.copy()
    w = max(2, int(n * rng.uniform(0.005, 0.08)))
    start = int(rng.integers(0, max(1, n - w)))
    cand[start:start + w] *= np.exp(rng.normal(0, strength, w))
    cand = np.maximum(cand, 0.0)
    s = cand.sum()
    if s < 1e-10:
        return f
    return cand / s


def element_perturb(f, rng, strength=0.3, n_elem=50):
    """Perturb random individual elements."""
    cand = f.copy()
    n = len(f)
    idx = rng.integers(0, n, min(n_elem, n))
    cand[idx] *= np.exp(rng.normal(0, strength, len(idx)))
    cand = np.maximum(cand, 0.0)
    s = cand.sum()
    if s < 1e-10:
        return f
    return cand / s


def lp_optimize(f, deadline, ls_every=4, use_dual_fraction=0.5, verbose=True, tag=""):
    """Main LP optimization loop with both primal and dual LPs."""
    best = f.copy()
    best_sc = fast_score(f)
    cur_sc = best_sc
    it = 0
    stall = 0
    rng = np.random.default_rng(abs(hash(f.tobytes())) % (2**31))
    
    while time.monotonic() < deadline:
        it += 1
        conv = np.convolve(f, f)
        M = conv.max()
        
        g = None
        # Choose between primal and dual LP
        if rng.random() < use_dual_fraction:
            result = solve_lp_dual(f, time_limit=3.0)
            if result is not None:
                g = result[0]
        else:
            # Try primal with slightly reduced M for aggression
            M_try = M * (0.95 + 0.1 * rng.random())
            g = solve_lp_primal(f, M_try, time_limit=3.0)
        
        if g is not None and np.linalg.norm(g - f) > 1e-8:
            if it % ls_every == 0 or it < 10:
                f_new, sc = line_search(f, g, n_pts=20)
            else:
                # Adaptive step size: larger early, smaller later
                t = 0.3 / (1.0 + 0.05 * it)
                t = max(0.05, min(0.8, t))
                f_new = np.maximum((1 - t) * f + t * g, 0.0)
                s = f_new.sum()
                if s < 1e-10:
                    f_new = f
                    sc = cur_sc
                else:
                    f_new = f_new / s
                    sc = fast_score(f_new)
            
            if sc < cur_sc - 1e-12:
                f = f_new
                cur_sc = sc
                if sc < best_sc:
                    best_sc = sc
                    best = f.copy()
                    stall = 0
                    if verbose and (it % 100 == 0 or best_sc < 1.52):
                        print(f"    {tag}it={it} score={sc:.6f} best={best_sc:.6f}")
                else:
                    stall += 1
            else:
                stall += 1
        else:
            stall += 1
        
        if stall > 25:
            # Escape: random perturbation
            if rng.random() < 0.5:
                f = block_perturb(f, rng, strength=0.3)
            else:
                f = element_perturb(f, rng, strength=0.4, n_elem=100)
            cur_sc = fast_score(f)
            stall = 0
    
    return best, best_sc

    best = np.maximum(parent, 0.0).copy()
    best_sc = fast_score(best)
    print(f"Initial: score={best_sc:.6f}, n={best.size}")
    
    # Phase 1: Intensive LP optimization from parent (35% of time)
    phase1_time = budget_s * 0.35
    phase1_end = min(time.monotonic() + phase1_time, deadline)
    print(f"Phase 1: LP from parent ({phase1_time:.0f}s budget)")
    b, s = lp_optimize(best.copy(), phase1_end, ls_every=3, tag="P1 ")
    if s < best_sc:
        best_sc, best = s, b
        print(f"  Phase 1 result: {best_sc:.6f}")
    
    # Phase 2: Multi-start with diverse initializations (35% of time)
    phase2_time = budget_s * 0.35
    phase2_end = min(time.monotonic() + phase2_time, deadline)
    print(f"Phase 2: Multi-start ({phase2_time:.0f}s budget)")
    
    methods = ['uniform', 'cosine', 'squared_cos', 'bilateral',
               'tapered_up', 'tapered_down', 'freq_flat', 'block_structured']
    
    # Determine which lengths to try
    n_parent = best.size
    lengths = sorted(set([n_parent, 512, 768, 1024]))
    lengths = [n for n in lengths if n <= 2048]
    
    configs = []
    for n in lengths:
        for m in methods:
            configs.append((n, m))
    
    # Prioritize: parent length first, then shorter (faster LP)
    configs.sort(key=lambda x: (x[0] != n_parent, x[0]))
    
    per_config = max(8.0, (phase2_end - time.monotonic()) / max(1, len(configs)))
    per_config = min(per_config, 120.0)
    
    for n, m in configs:
        if time.monotonic() > phase2_end - 10:
            break
        start = make_initial(n, rng, m)
        local_dl = min(phase2_end, time.monotonic() + per_config)
        tag = f"P2[{n},{m[:4]}] "
        b, s = lp_optimize(start, local_dl, ls_every=5,
                           use_dual_fraction=0.4, verbose=False, tag=tag)
        if s < best_sc:
            best_sc, best = s, b
            print(f"  NEW BEST! n={n}, m={m}, score={best_sc:.6f}")
    
    # Phase 3: Final intensive polish (remaining time)
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"Phase 3: Final polish ({remaining:.0f}s, best={best_sc:.6f})")
        # Try multiple polish runs with different parameters
        runs = 3
        per_run = remaining / runs
        for r in range(runs):
            if time.monotonic() > deadline - 5:
                break
            start = best.copy()
            if r > 0:
                start = block_perturb(start, rng, strength=0.1)
            local_dl = min(deadline, time.monotonic() + per_run)
            b, s = lp_optimize(start, local_dl, ls_every=2,
                               use_dual_fraction=0.6, verbose=True, tag=f"P3r{r} ")
            if s < best_sc:
                best_sc, best = s, b
                print(f"  Phase 3 run {r}: {best_sc:.6f}")
    
    # Final validation
    best = np.clip(np.asarray(best, dtype=np.float64), 0.0, 1000.0)
    if best.size == 0 or not np.isfinite(best).all() or best.sum() < 0.01:
        best = np.ones(512)
```
