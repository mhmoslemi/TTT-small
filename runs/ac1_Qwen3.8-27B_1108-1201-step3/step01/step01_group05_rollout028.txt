"""
Step-function search for minimum peak self-convolution score.
Uses projected gradient descent on the probability simplex with:
- Adaptive log-sum-exp temperature annealing
- Nesterov-accelerated momentum
- Backtracking line search (Armijo condition)
- Simulated annealing for escape from local minima
- Diverse structure-aware initializations (Gaussian mixtures, chirps, Beta, power-law)
- Multi-length exploration to find the best n
- Periodic structure-aware moves (pair swaps, random smooth perturbations)
"""

import time
import numpy as np
from scipy.fft import fft, ifft, next_fast_len


def _convolve_sq(p):
    """Full linear self-convolution of 1D vector p via FFT."""
    n = int(p.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = next_fast_len(2 * n - 1)
    F = fft(p, L)
    c = np.real(ifft(F * F))[:2 * n - 1]
    return c


def _score(p):
    """Compute 2*n*max_conv / (sum)^2 for non-negative p."""
    n = int(p.size)
    if n == 0:
        return np.inf
    s = float(np.sum(p))
    if s <= 1e-12:
        return np.inf
    c = _convolve_sq(p)
    cmax = float(np.max(c)) if c.size > 0 else 0.0
    if cmax < 0:
        cmax = 0.0
    return 2.0 * n * cmax / (s * s)


def _logsumexp_surrogate(c, T):
    """Stable logsumexp(c/T) * T = smoothed max of c."""
    cmax = float(np.max(c))
    if not np.isfinite(cmax) or cmax <= 0:
        return 0.0
    T = max(float(T), 1e-300)
    diff = (c - cmax) / T
    e = np.exp(diff)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0:
        return cmax
    return cmax + T * np.log(s)


def _gradient_smooth_max(p, T_factor):
    """
    Gradient of T*logsumexp(c/T) where c = p*p (linear convolution).
    Returns (gradient, convolution, T, surrogate_value).
    """
    n = int(p.size)
    if n == 0:
        return np.empty(0), np.empty(0), 1.0, 0.0

    L = next_fast_len(3 * n - 2)
    Fp = fft(p, L)
    c = np.real(ifft(Fp * Fp))[:2 * n - 1]
    cmax = float(np.max(c)) if c.size > 0 else 0.0
    if not np.isfinite(cmax) or cmax <= 0:
        return np.zeros(n), c, 1.0, 0.0

    T = max(T_factor * cmax, 1e-300)
    diff = (c - cmax) / T
    e = np.exp(diff)
    raw = float(e.sum())
    if not np.isfinite(raw) or raw <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        raw = 1.0
    else:
        w = e / raw

    # Gradient: g_j = 2 * sum_k w_k * p_{k-j} (valid range)
    # This is 2 * (w_reversed * p) evaluated at appropriate indices
    Lg = next_fast_len(3 * n - 2)
    Fw = fft(w[::-1], Lg)
    Fp2 = fft(p, Lg)
    h = np.real(ifft(Fw * Fp2))

    # h[2n-2 : n-2 : -1] gives the gradient for j=0,...,n-1
    if n == 1:
        G = np.array([h[0]], dtype=np.float64)
    else:
        G = h[2 * n - 2 : n - 2 : -1].copy()

    if G.size != n:
        tmp = np.zeros(n, dtype=np.float64)
        m = min(n, G.size)
        tmp[:m] = G[:m]
        G = tmp

    G = np.maximum(G, 0.0)
    grad = 2.0 * G
    surr = cmax + T * np.log(raw)
    return grad, c, T, surr


def _project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = int(v.size)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if not np.all(np.isfinite(v)):
        v = np.nan_to_num(v, nan=0.0, posinf=1.0, neginf=0.0)

    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)

    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)

    k = int(np.count_nonzero(cond)) - 1
    theta = (cssum[k] - 1.0) / float(k + 1)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def _safe_normalize(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if s <= 1e-12:
        return np.full(max(1, p.size), 1.0 / max(1, p.size), dtype=np.float64)
    return p / s


def _make_gaussian(n, mu, sigma):
    x = np.linspace(0.0, 1.0, n)
    p = np.exp(-0.5 * ((x - mu) / max(sigma, 1e-6)) ** 2)
    return _safe_normalize(p)


def _make_two_gauss(n, mu1, s1, w1, mu2, s2, w2):
    x = np.linspace(0.0, 1.0, n)
    p = w1 * np.exp(-0.5 * ((x - mu1) / max(s1, 1e-6)) ** 2)
    p += w2 * np.exp(-0.5 * ((x - mu2) / max(s2, 1e-6)) ** 2)
    return _safe_normalize(np.maximum(p, 0.0))


def _make_beta_dist(n, a, b):
    x = np.linspace(1e-6, 1.0 - 1e-6, n)
    p = x ** (a - 1.0) * (1.0 - x) ** (b - 1.0)
    return _safe_normalize(np.maximum(p, 0.0))


def _make_power(n, alpha, rev=False):
    x = np.linspace(1e-6, 1.0, n)
    p = x ** alpha
    if rev:
        p = p[::-1]
    return _safe_normalize(np.maximum(p, 0.0))


def _make_chirp(n, freq, amp=1.0):
    x = np.linspace(0.0, 1.0, n)
    p = 1.0 + amp * np.sin(2.0 * np.pi * freq * x)
    return _safe_normalize(np.maximum(p, 0.01))


def _make_random_smooth(n, rng, n_harmonics=5):
    x = np.linspace(0.0, 1.0, n)
    p = np.ones(n)
    for k in range(1, n_harmonics + 1):
        a = float(rng.normal(0, 0.5 / k))
        b = float(rng.normal(0, 0.5 / k))
        p += a * np.sin(2 * np.pi * k * x) + b * np.cos(2 * np.pi * k * x)
    return _safe_normalize(np.maximum(p, 0.01))


def _make_cup(n, depth):
    x = np.linspace(0.0, 1.0, n)
    p = 1.0 + depth * (2.0 * x - 1.0) ** 2
    return _safe_normalize(p)


def _make_cap(n, depth):
    x = np.linspace(0.0, 1.0, n)
    p = 1.0 - depth * (2.0 * x - 1.0) ** 2
    return _safe_normalize(np.maximum(p, 0.01))


def _make_rectangular_mix(n, k, rng):
    """Mix of k rectangular bumps at random positions."""
    p = np.ones(n) * 0.05
    for _ in range(k):
        center = int(rng.integers(0, n))
        width = int(rng.integers(max(1, n // (4 * k)), n // (2 * k)))
        w = float(rng.uniform(0.5, 2.0))
        for i in range(max(0, center - width), min(n, center + width + 1)):
            p[i] += w * np.exp(-0.5 * ((i - center) / max(width // 3, 1)) ** 2)
    return _safe_normalize(p)


def _generate_initials(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    out = []

    if parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.maximum(p, 0.0)
        if p.ndim == 1 and p.size == n and p.sum() > 0:
            out.append(_safe_normalize(p))

    # Uniform
    out.append(np.full(n, 1.0 / n))

    # Gaussian variants
    for mu in (0.2, 0.3, 0.5, 0.7, 0.8):
        for s in (0.05, 0.1, 0.15, 0.25, 0.4):
            out.append(_make_gaussian(n, mu, s))

    # Two Gaussian mixtures
    two_g = [
        (0.25, 0.06, 1.0, 0.75, 0.06, 1.0),
        (0.2, 0.08, 1.5, 0.8, 0.08, 1.0),
        (0.15, 0.05, 1.0, 0.5, 0.15, 0.5),
        (0.5, 0.12, 1.0, 0.05, 0.04, 1.5),
        (0.5, 0.12, 1.0, 0.95, 0.04, 1.5),
        (0.3, 0.05, 2.0, 0.7, 0.08, 1.0),
        (0.35, 0.04, 1.0, 0.65, 0.04, 1.0),
    ]
    for args in two_g:
        out.append(_make_two_gauss(n, *args))

    # Beta distributions
    beta_params = [
        (0.5, 2.0), (0.5, 5.0), (1.0, 3.0), (2.0, 5.0),
        (3.0, 1.0), (5.0, 2.0), (2.0, 0.5), (5.0, 0.5),
        (0.3, 0.3), (0.7, 0.7), (1.5, 1.5), (3.0, 3.0),
    ]
    for a, b in beta_params:
        out.append(_make_beta_dist(n, a, b))

    # Power laws
    for alpha in (-1.5, -0.8, -0.3, 0.3, 0.8, 1.5, 3.0):
        out.append(_make_power(n, alpha, False))
        out.append(_make_power(n, alpha, True))

    # Chirps
    for freq in (2, 3, 4, 5, 7, 10):
        for amp in (0.3, 0.6, 1.0, 1.5):
            out.append(_make_chirp(n, freq, amp))

    # Cups and caps
    for d in (0.5, 1.0, 2.0, 4.0, 8.0):
        out.append(_make_cup(n, d))
    for d in (0.3, 0.5, 0.7, 0.9):
        out.append(_make_cap(n, d))

    # Random smooth
    for _ in range(20):
        out.append(_make_random_smooth(n, rng, 5))
    for _ in range(10):
        out.append(_make_random_smooth(n, rng, 3))

    # Rectangular mixtures
    for _ in range(10):
        k = int(rng.integers(2, 6))
        out.append(_make_rectangular_mix(n, k, rng))

    return out


def _optimize_local(p0, rng, time_limit, verbose=False):
    """
    Projected gradient descent with momentum, adaptive T, SA acceptance,
    and structure-aware moves.
    """
    p = _project_simplex(np.asarray(p0, dtype=np.float64))
    n = int(p.size)
    if n == 0:
        return p, np.inf

    best_p = p.copy()
    best_score = _score(best_p)
    current_score = best_score
    v = np.zeros(n, dtype=np.float64)  # momentum
    momentum = 0.9

    T_factor = 0.15
    T_min_factor = 0.001

    end = time.monotonic() + max(0.0, float(time_limit))
    iter_no = 0
    since_improve = 0
    sa_temperature = 0.005
    c1 = 1e-4  # Armijo
    rho = 0.5  # backtrack factor

    while time.monotonic() < end:
        iter_no += 1

        # Anneal temperature
        T_factor = max(T_min_factor, 0.15 * np.exp(-iter_no / 800.0))
        sa_temperature = max(0.0001, 0.005 * np.exp(-iter_no / 500.0))

        g, c_cur, T, surr = _gradient_smooth_max(p, T_factor)
        if not np.all(np.isfinite(g)):
            g = np.zeros(n)

        gnorm = float(np.linalg.norm(g))
        if gnorm < 1e-15:
            since_improve += 1
            if since_improve > 200:
                break
            p = _project_simplex(p + rng.normal(0, 0.01 / n, n))
            current_score = _score(p)
            continue

        # Determine base learning rate
        pmax = float(np.max(p))
        pmin_pos = float(np.min(p[p > 0])) if np.any(p > 0) else 1.0 / n
        scale = max(1.0 / n, 0.01 * pmax)
        lr0 = min(2.0 * scale / gnorm, 10.0)

        # Backtracking line search with momentum
        lr = lr0
        accepted = False

        for backtrack in range(12):
            v = momentum * v + (1.0 - momentum) * (-lr * g)
            candidate = p + v
            candidate = _project_simplex(candidate)

            c_new = _convolve_sq(candidate)
            cmax_new = float(np.max(c_new)) if c_new.size > 0 else 0.0
            score_new = 2.0 * n * cmax_new
            surr_new = _logsumexp_surrogate(c_new, T)

            # Armijo condition on surrogate + check actual score
            improve_surr = surr - surr_new
            improve_score = current_score - score_new

            if improve_surr > c1 * lr * gnorm * gnorm or improve_score > 1e-12:
                # Accept
                # SA: sometimes accept worse moves
                accept = True
            elif np.random.random() < np.exp(-max(0, surr_new - surr) / max(sa_temperature, 1e-10)):
                accept = True
            else:
                accept = False

            if accept:
                if score_new < best_score:
                    best_score = score_new
                    best_p = candidate.copy()
                    since_improve = 0
                else:
                    since_improve += 1
                current_score = score_new
                p = candidate
                accepted = True
                break

            lr *= rho
            # Reset v for next attempt
            v = momentum * v

        if not accepted:
            since_improve += 1

            # Structure moves when stuck
            if since_improve > 100:
                move_type = int(rng.integers(0, 4))
                if move_type == 0:
                    # Random smooth perturbation
                    x = np.linspace(0, 1, n)
                    pert = np.zeros(n)
                    for k in range(1, 4):
                        pert += float(rng.normal(0, 0.3 / k)) * np.sin(2 * np.pi * k * x + rng.uniform(0, 2 * np.pi))
                    p = _project_simplex(p + 0.1 * pert)
                elif move_type == 1 and n > 1:
                    # Pair swap: transfer mass from high to low gradient
                    hi = int(np.argmax(g))
                    lo = int(np.argmin(g))
                    if hi != lo:
                        for frac in (0.3, 0.1, 0.03):
                            delta = frac * p[hi]
                            if delta <= 0:
                                continue
                            cand = p.copy()
                            cand[hi] -= delta
                            cand[lo] += delta
                            cand = _project_simplex(cand)
                            sc = _score(cand)
                            if sc < current_score:
                                p = cand
                                current_score = sc
                                if sc < best_score:
                                    best_score = sc
                                    best_p = p.copy()
                                    since_improve = 0
                                break
                elif move_type == 2:
                    # Blend with best
                    p = _project_simplex(0.7 * best_p + 0.3 * p + rng.normal(0, 0.05 / n, n))
                    current_score = _score(p)
                else:
                    # Larger random kick
                    p = _project_simplex(p + rng.normal(0, 0.5 / n, n))
                    current_score = _score(p)

            if since_improve > 500:
                # Major restart
                p = _project_simplex(0.6 * best_p + 0.4 * _make_random_smooth(n, rng, 5))
                current_score = _score(p)
                v = np.zeros(n)
                since_improve = 0
                T_factor = 0.1
                momentum = 0.8

            if since_improve > 5000:
                # Nudge momentum
                momentum = max(0.5, momentum - 0.1)
                T_factor = 0.08

        if verbose and iter_no % 5000 == 0:
            print("  iter=%d best=%.10f cur=%.10f T=%.5f since_imp=%d" %
                  (iter_no, best_score, current_score, T_factor, since_improve), flush=True)

        if best_score < 1.5000 and since_improve > 3000:
            break

    return best_p, best_score


def _try_different_lengths(best_p, best_score, rng, time_limit):
    """Try optimizing at different lengths and see if a different n gives better score."""
    if time_limit < 60:
        return best_p, best_score

    current_n = int(best_p.size)
    candidates = []
    for factor in (0.5, 0.75, 1.5, 2.0):
        new_n = int(current_n * factor)
        new_n = max(128, min(16384, new_n))
        if new_n != current_n:
            candidates.append(new_n)

    # Also try some fixed sizes
    for sn in (1024, 2048, 4096, 8192):
        if sn != current_n and sn not in candidates:
            candidates.append(sn)

    candidates = sorted(set(candidates))[:4]
    time_per = max(15.0, time_limit / max(1, len(candidates)))

    for new_n in candidates:
        if time.monotonic() > time_limit:
            break
        # Reshape best_p to new_n by interpolation
        old_n = int(best_p.size)
        x_old = np.linspace(0, 1, old_n)
        x_new = np.linspace(0, 1, new_n)
        p_resized = np.interp(x_new, x_old, best_p)
        p_resized = _safe_normalize(p_resized)

        sc = _score(p_resized)
        remaining = time_limit - time.monotonic()
        if remaining < 10:
            break

        p_opt, sc_opt = _optimize_local(p_resized, rng, min(time_per, remaining), verbose=False)
        if sc_opt < best_score:
            best_p = p_opt
            best_score = sc_opt
            print("  Length change: n=%d -> n=%d score=%.10f" % (current_n, new_n, best_score), flush=True)
            current_n = new_n

    return best_p, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative step function minimizing 2*n*max(convolve(a,a))/sum(a)^2.
    Uses projected gradient descent with adaptive temperature, momentum,
    simulated annealing, and diverse initialization.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 8.0)

    # Try to use parent sequence if available
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.where(np.isfinite(parent_arr), parent_arr, 0.0)
        parent_arr = np.maximum(parent_arr, 0.0)
        if parent_arr.ndim == 1 and parent_arr.size >= 16 and parent_arr.sum() > 0:
            parent = parent_arr
    except Exception:
        parent = None

    # Determine primary length
    if parent is not None:
        n_primary = int(parent.size)
    else:
        n_primary = 4096
    n_primary = max(256, min(n_primary, 8192))

    print("Starting search with n=%d, budget=%.0fs" % (n_primary, budget_s), flush=True)

    best_p = None
    best_score = np.inf

    # Phase 1: Generate and score initializations
    t_phase1_end = min(start_time + 30.0, deadline - 100.0)
    initials = _generate_initials(n_primary, rng, parent)
    print("Generated %d initializations" % len(initials), flush=True)

    scored = []
    for i, p in enumerate(initials):
        if time.monotonic() > t_phase1_end:
            break
        p = _safe_normalize(p)
        sc = _score(p)
        if np.isfinite(sc):
            scored.append((sc, p))

    scored.sort(key=lambda x: x[0])
    print("Best initial: score=%.10f (from %d scored)" % (scored[0][0], len(scored)), flush=True)

    for sc, p in scored:
        if sc < best_score:
            best_score = sc
            best_p = p.copy()

    # Phase 2: Optimize top initializations
    top_k = min(8, len(scored))
    remaining = deadline - time.monotonic()
    final_reserve = max(30.0, remaining * 0.15)
    available = max(0.0, remaining - final_reserve)
    time_per_init = max(10.0, available / max(1, top_k))

    print("Optimizing top %d initializations, time each=%.0fs" % (top_k, time_per_init), flush=True)

    for i in range(top_k):
        if time.monotonic() > deadline - final_reserve:
            break
        sc_init, p_init = scored[i]
        alloc = min(time_per_init, deadline - final_reserve - time.monotonic())
        if alloc < 5.0:
            break

        print("  Init %d: score=%.10f, allocating %.0fs" % (i, sc_init, alloc), flush=True)
        p_opt, sc_opt = _optimize_local(p_init, rng, alloc, verbose=(i == 0))

        if sc_opt < best_score:
            best_score = sc_opt
            best_p = p_opt.copy()
            print("  -> New best: %.10f" % best_score, flush=True)
        else:
            print("  -> Score: %.10f (best=%.10f)" % (sc_opt, best_score), flush=True)

    # Phase 3: Try different lengths
    if best_p is not None and time.monotonic() < deadline - 60.0:
        remaining = deadline - time.monotonic() - 30.0
        if remaining > 60:
            print("Trying different lengths (budget=%.0fs)" % remaining, flush=True)
            best_p, best_score = _try_different_lengths(
                best_p, best_score, rng,
                start_time + remaining + 30.0
            )

    # Phase 4: Final polish
    if best_p is not None and time.monotonic() < deadline - 5.0:
        remaining = deadline - time.monotonic() - 3.0
        if remaining > 5.0:
            print("Final polish (budget=%.0fs)" % remaining, flush=True)
            p_polished, sc_polished = _optimize_local(best_p, rng, remaining, verbose=False)
            if sc_polished < best_score:
                best_score = sc_polished
                best_p = p_polished

    # Ensure valid output
    if best_p is None:
        best_p = np.full(n_primary, 1.0 / n_primary)
        best_score = _score(best_p)

    best_p = np.asarray(best_p, dtype=np.float64)
    best_p = np.where(np.isfinite(best_p), best_p, 0.0)
    best_p = np.maximum(best_p, 0.0)
    best_p = np.minimum(best_p, 1000.0)
    s = float(best_p.sum())
    if s < 0.01:
        best_p = best_p * (0.02 / max(s, 1e-300))

    print("Final result: n=%d, score=%.10f" % (best_p.size, best_score), flush=True)

    return [float(x) for x in best_p]
