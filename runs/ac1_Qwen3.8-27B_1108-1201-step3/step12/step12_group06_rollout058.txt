"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated Adam on smooth LSE objective (fast, parallelizable)
2. Nesterov-accelerated LP best-response (strong convergence guarantee)
3. Multi-length exploration with interpolation
4. Peak-shaving post-processing
Key improvements: O(n log n) gradients via FFT, pre-compiled LP,
temperature annealing, and hybrid optimization pipeline.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Fast linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score assuming sum(a)=1: returns 2*n*max(b)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a_fixed):
    """Build the convolution matrix T where (T@b)[k] = (a_fixed*b)[k]."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


_lp_solver_name = None


def get_lp_solver():
    """Find the best available LP solver."""
    global _lp_solver_name
    if _lp_solver_name is not None:
        return _lp_solver_name
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CBC, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_name = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_name = cp.ECOS
    return _lp_solver_name


_lp_cache = {}


def solve_lp_best_response(a_fixed, solver=None):
    """Solve the LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0.
    Uses caching of the conv matrix for repeated calls with same n."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)

    # Cache the matrix construction
    if n not in _lp_cache:
        _lp_cache[n] = build_conv_matrix(a_fixed)
    # But the matrix depends on a_fixed, so we need to rebuild if a changes
    # Actually the matrix IS a_fixed, so we can't cache it across different a_fixed.
    # Let's just build it each time but use a faster method.
    
    T = build_conv_matrix(a_fixed)
    
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS, cp.CBC, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp(a0, time_limit, solver=None, max_iters=500000):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 20:
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} M_lp={M_lp:.6f}")
                last_print = elapsed

    return best_a, best_score


def lse_grad_fft(a, T):
    """Compute LSE objective and gradient w.r.t. a using FFT. O(n log n)."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp

    # Gradient: g[i] = 2 * sum_k w[k] * a[k-i]
    # Use FFT cross-correlation: g[i] = 2 * (w * a_flipped)[n-1+i]
    a_flipped = a[::-1].copy()
    conv_size = 3 * n - 2
    fft_size = 1 << (conv_size).bit_length()
    W = rfft(w, fft_size)
    Af = rfft(a_flipped, fft_size)
    conv_result = irfft(W * Af, fft_size)[:conv_size]
    g = 2.0 * conv_result[n - 1:2 * n - 1]

    return obj, g


def adam_lse_optimize(a0, time_limit, T_start=50.0, T_end=5.0, lr=0.05, n_iters=200000, restart_interval=20000):
    """Adam optimizer on LSE objective with temperature annealing and restarts."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-30)).copy()
    a = a0.copy()

    best_a = a.copy()
    best_score = score_normalized(a)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    t0 = time.monotonic()
    it = 0

    while it < n_iters:
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, it / n_iters)
        T = T_start * (T_end / T_start) ** progress

        obj, g = lse_grad_fft(a, T)

        # Gradient w.r.t. z (softmax parameterization)
        S = np.dot(g, a)
        grad_z = a * (g - S)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * grad_z ** 2
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))

        # Learning rate schedule: decay with sqrt
        lr_eff = lr / np.sqrt(1.0 + 0.01 * it)
        z = z - lr_eff * m_hat / (np.sqrt(v_hat) + eps)
        a = softmax_fn(z)

        sc = score_normalized(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        it += 1

        # Periodic restart: reset Adam state, slight perturbation
        if restart_interval > 0 and it % restart_interval == 0:
            m = np.zeros(n)
            v = np.zeros(n)
            # Small perturbation to escape local minima
            perturb = 0.01 * np.random.default_rng(it).normal(0, 1, n)
            z = z + perturb
            a = softmax_fn(z)

        if it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AdamLSE iter={it} ({elapsed:.0f}s): T={T:.1f} cur={sc:.8f} best={best_score:.8f}")

    return best_a, best_score

def peak_shave(a0, time_limit, n_steps=50000):
    """Directly reduce peaks of the convolution."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-30)).copy()
    a = a0.copy()

    best_a = a.copy()
    best_score = score_normalized(a)

    t0 = time.monotonic()
    lr = 0.02

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Find top peaks (top 10 or 1% whichever is larger)
        n_peaks = max(10, n // 100)
        peak_indices = np.argsort(b)[-n_peaks:]
        # Weight peaks by how much they exceed the mean
        threshold = b.mean() * 1.5
        above = peak_indices[b[peak_indices] > threshold]
        if len(above) == 0:
            above = peak_indices[:5]

        # Gradient: pull mass away from contributions to peaks
        g = np.zeros(n)
        for k in above:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = range(lo, hi + 1)
            for i in idx:
                g[i] += a[k - i]
        g /= len(above)

        S = np.dot(g, a)
        grad_z = a * (g - S)

        grad_norm = np.linalg.norm(grad_z)
        if grad_norm < 1e-15:
            break

        lr_eff = lr * (1.0 + step / n_steps) ** (-0.5)
        z = z - lr_eff * grad_z / (grad_norm + 1e-10)
        a = softmax_fn(z)

        sc = score_normalized(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if (step + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave step={step+1} ({elapsed:.0f}s): cur={sc:.8f} best={best_score:.8f}")

    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    # === Phase 2: Nesterov LP - main optimization (40% time) ===
    print(f"\n=== Phase 2: Nesterov LP (main) ===")
    remaining = deadline - time.monotonic()
    t2 = 0.40 * (deadline - t_start)
    if t2 > 60:
        print(f"  Running Nesterov LP for {t2:.0f}s at n={n_base}")
        a2, _ = nesterov_lp(best_seq, t2, solver=solver)
        try_update(a2, "nesterov_main")

    # === Phase 3: Adam LSE polish (20% time) ===
    print(f"\n=== Phase 3: Adam LSE polish ===")
    remaining = deadline - time.monotonic()
    t3 = min(0.20 * (deadline - t_start), remaining * 0.5)
    if t3 > 30:
        print(f"  Running Adam LSE polish for {t3:.0f}s")
        # Low temperature for fine-tuning
        a3, _ = adam_lse_optimize(best_seq, t3, T_start=20.0, T_end=3.0, lr=0.02, n_iters=100000, restart_interval=20000)
        try_update(a3, "adam_lse_polish1")

        remaining = deadline - time.monotonic()
        if remaining > 20:
            # Even lower temperature
            a3b, _ = adam_lse_optimize(best_seq, remaining * 0.5, T_start=10.0, T_end=2.0, lr=0.01, n_iters=50000, restart_interval=0)
            try_update(a3b, "adam_lse_polish2")

    # === Phase 4: Peak shaving (10% time) ===
    print(f"\n=== Phase 4: Peak shaving ===")
    remaining = deadline - time.monotonic()
    t4 = min(0.10 * (deadline - t_start), remaining * 0.4)
    if t4 > 15:
        print(f"  Running peak shave for {t4:.0f}s")
        a4, _ = peak_shave(best_seq, t4, n_steps=30000)
        try_update(a4, "peak_shave")

    # === Phase 5: Try larger n (10% time) ===
    print(f"\n=== Phase 5: Multi-length exploration ===")
    remaining = deadline - time.monotonic()
    t5_total = min(0.10 * (deadline - t_start), remaining * 0.6)
    if t5_total > 40:
        for n_try in [n_base + 300, n_base + 600]:
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            t5 = min(t5_total / 2.0, remaining * 0.45)
            if t5 < 20:
                break
            print(f"  Trying n={n_try}, budget={t5:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            # Quick Adam LSE
            a_try, _ = adam_lse_optimize(init_try, t5 * 0.6, T_start=30.0, T_end=8.0, lr=0.03, n_iters=50000, restart_interval=10000)
            try_update(a_try, f"len_{n_try}_adam")

            remaining = deadline - time.monotonic()
            if remaining > 15:
                # Quick Nesterov
                a_try2, _ = nesterov_lp(a_try, remaining * 0.5, solver=solver, max_iters=20000)
                try_update(a_try2, f"len_{n_try}_nesterov")

    # === Phase 6: Final Nesterov polish on best sequence (5% time) ===
    print(f"\n=== Phase 6: Final Nesterov polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"  Final Nesterov for {remaining * 0.7:.0f}s")
        a6, _ = nesterov_lp(best_seq, remaining * 0.7, solver=solver, max_iters=30000)
        try_update(a6, "final_nesterov")

        remaining = deadline - time.monotonic()
        if remaining > 10:
            a6b, _ = adam_lse_optimize(best_seq, remaining * 0.6, T_start=8.0, T_end=2.0, lr=0.01, n_iters=20000, restart_interval=0)
            try_update(a6b, "final_adam")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    total_time = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"Total time: {total_time:.0f}s")
    return [float(x) for x in best_seq]
