import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_autocorr(a):
    """Compute full self-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def subgradient(a, peaks):
    """
    Compute subgradient of max_{k in peaks} conv(a,a)_k.
    For a single peak k: grad_j = 2 * a[k-j] (if valid), 0 otherwise.
    For multiple peaks: average of the individual subgradients.
    Returns the gradient vector of length n.
    """
    n = len(a)
    K = len(peaks)
    grad = np.zeros(n)
    for k in peaks:
        # grad_j += 2 * a[k-j] / K for valid j
        # a[k-j] is valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            # a[k-j] for j in [js, je]
            # When j goes from js to je, k-j goes from k-js to k-je
            indices = k - np.arange(js, je + 1)
            grad[js:je+1] += 2.0 * a[indices] / K
    return grad

def simplex_project(v):
    """Project v onto the probability simplex {a >= 0, sum(a) = 1}."""

def fast_autocorr(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(v - theta, 0)


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def score_a(a):
    """Compute the objective score (assuming sum(a)=1)."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()


def build_conv_matrix_rows(a_fixed, row_indices):
    """
    Build rows of the convolution matrix T where T[k, j] = a_fixed[k-j] (if valid).
    Only builds the specified rows.
    """
    n = len(a_fixed)
    T = np.zeros((len(row_indices), n))
    for idx, k in enumerate(row_indices):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            T[idx, js:je+1] = a_fixed[k - je:k - js + 1][::-1]
    return T


def solve_lp_best_response(a_fixed, peaks, solver):
    """
    Solve: min M s.t. conv(a_fixed, b)_k <= M for k in peaks, sum(b)=1, b>=0.
    This is B's best response in the game, restricted to the given peaks.
    """
    n = len(a_fixed)
    K = len(peaks)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    T = build_conv_matrix_rows(a_fixed, peaks)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = float(M_var.value)
    return b_val, M_val
