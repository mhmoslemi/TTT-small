    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break
        
        # Number of inner iterations
        iters = max(100, min(5000, int(remaining * 30000.0 / (n_stages - si + 1))))
        
        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break
            
            b = fast_autocorr(a)
            Tb = T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            
            grad = compute_grad_weights(a, w)
            
            # Adaptive step size
            grad_inf = np.abs(grad).max()
            if grad_inf < 1e-12:
                break
            eta = 2.0 / (grad_inf * np.sqrt(T + 1.0))
            eta = min(eta, 2.0)
            
            # Multiplicative update
            log_step = -eta * grad
            log_step = np.clip(log_step, -50, 50)
            a = a * np.exp(log_step)
            a = np.maximum(a, 1e-30)
            a = a / a.sum()
            
            if it % 50 == 0:
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
        
        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if si % 10 == 0:
            print(f"      MD-LSE s{si}/{n_stages} T={T:.1f}: {best_score:.6f} ({time.monotonic()-t0:.0f}s)")
    
    return best_a, best_score

def lp_adam(a0, time_limit, p=32, lr=0.01, n_stages=20):
    """Adam optimization of Lp norm of convolution."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)/n
    else:
        a = a / s
    z = np.log(np.maximum(a, 1e-30))
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2*n-1).bit_length()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_adam = 0
    
    # Decreasing p over stages (start with moderate p, increase)
    p_vals = np.geomspace(min(p, 8), p, n_stages)
    
    for si, p_cur in enumerate(p_vals):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        
        iters = max(50, min(3000, int(remaining * 5000.0 / (n_stages - si + 1))))
        
        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break
            
            b = fast_autocorr(a)
            b_pos = np.maximum(b, 1e-30)
            b_p = b_pos ** p_cur
            sum_bp = b_p.sum()
            if sum_bp < 1e-30:
                break
            
            # weights for gradient
            w = b_pos ** (p_cur - 1) / (sum_bp ** (1.0 - 1.0/p_cur))
            
            grad_a = compute_grad_weights(a, w)
            
            # Convert to z-gradient
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)
            
            # Adam update
            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z**2
            m_hat = m / (1 - beta1**t_adam)
            v_hat = v / (1 - beta2**t_adam)
            
            lr_t = lr / np.sqrt(1 + 0.01 * it)
            z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            a = softmax_fn(z)
            
            if it % 50 == 0:
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
        
        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if si % 5 == 0:
            print(f"      Lp-Adam s{si}/{n_stages} p={p_cur:.1f}: {best_score:.6f} ({time.monotonic()-t0:.0f}s)")
    
    return best_a, best_score

def spectral_shaping(a, n_iter=500, alpha=0.3):
    n = len(a)
    a = np.maximum(a.copy(), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    fft_size = 1 << (2*n-1).bit_length()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    
    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A)**2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.05, 20.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    
    return best_a

def peak_shaving(a0, time_limit, seed=42):

    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:8]
        improved = False
        
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n-1, k)
            idx = np.arange(js, je+1)
            contrib = a[idx] * a[k - idx]
            src = idx[np.argmax(contrib)]
            
            peak_areas = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n-1, k2)
                peak_areas.update(range(js2, je2+1))
            
            candidates = [j for j in range(n) if j not in peak_areas]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            candidates_sorted = sorted(candidates, key=lambda j: a[j])
            dst_pool = candidates_sorted[:max(1, len(candidates_sorted)//3)]
            dst = dst_pool[rng.integers(len(dst_pool))]
            
            for frac in [0.003, 0.007, 0.015, 0.03, 0.05]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt

                if a_trial[i] >= 0:

                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            if stall > 300:
                break
    
    return best_a, best_score

We have:
- a of length n
- weights w of length 2n-1 (corresponding to C[0], ..., C[2n-2])
- We want: grad[j] = 2 * sum_{m=0}^{n-1} w[j+m] * a[m]

Using convolution:
- w_rev = w reversed, so w_rev[i] = w[2n-2-i]
- conv(k) = sum_i a[i] * w_rev[k-i] = sum_i a[i] * w[2n-2-(k-i)] = sum_i a[i] * w[2n-2-k+i]

We want: sum_m a[m] * w[j+m]
Set 2n-2-k+m = j+m => k = 2n-2-j

So grad[j] = 2 * conv(2n-2-j)

In numpy: conv is computed via IFFT(A * Wr) where A = FFT(a), Wr = FFT(w_rev).
The result has length fft_size (padded). The valid part is conv[0 : 2n-1].

grad = 2 * conv[2n-2, 2n-3, ..., n-1] = 2 * conv[n-1 : 2n-1][::-1]

In numpy slicing: `conv[2*n-2 : n-2 : -1]` gives elements at indices 2n-2, 2n-3, ..., n-1. That's n elements. ✓

Wait, `conv[2*n-2 : n-2 : -1]`:
- Start: 2n-2
- Stop: n-2 (exclusive, so last element is n-1)
- Step: -1
- Elements: 2n-2, 2n-3, ..., n-1 → n elements ✓
