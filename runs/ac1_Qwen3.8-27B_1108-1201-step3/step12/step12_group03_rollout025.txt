def peak_shaving(a0, time_limit, seed=42):
    """Coordinate descent peak shaving."""
    n = len(a0)

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)

    stall = 0

            je = min(n - 1, k)

                    best_dst = None

                    for dst in range(0, n, max(1, n // 50)):
                        if dst == src or dst == src_other:
                            continue
                        # Estimate: how much does dst contribute to top peaks?
                        peak_sum = 0
                        for k2 in top_peaks[:3]:
                            if 0 <= k2 - dst < n:
                                peak_sum += a[k2 - dst]
                        score_dst = peak_sum + a[dst] * 0.1
                        if score_dst < best_dst_score:
                            best_dst_score = score_dst
                            best_dst = dst

                    if best_dst is None:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    if src != src_other:
                        a_trial[src_other] = max(0, a_trial[src_other] - amt * 0.3)
                    a_trial[best_dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    a_trial_s = a_trial.sum()
                    if a_trial_s < 1e-15:
                        continue
                    a_trial = a_trial / a_trial_s
                    score = compute_score_normalized(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 20 == 0:
                # Random perturbation
                n_pert = rng.integers(2, 8)
                srcs = rng.choice(n, size=n_pert, replace=False)
                dsts = rng.choice(n, size=n_pert, replace=False)

                        stall = 0

                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
