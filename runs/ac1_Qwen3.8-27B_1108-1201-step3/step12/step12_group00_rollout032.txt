```python
def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T
```

```python
def interp_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

def solve_lp_best_response(a, M, solver, top_k_indices=None):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for k in top_k_indices, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)

    # Build constraint matrix for relevant rows
    if top_k_indices is not None:
        rows = np.array(top_k_indices)
    else:
        rows = np.arange(2 * n - 1)

    nrows = len(rows)
    if nrows == 0:
        return np.ones(n) / n

    # Build sparse matrix for selected rows
    data = []
    row_idx = []
    col_idx = []
    for ii, k in enumerate(rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            data.append(a[k - j])
            row_idx.append(ii)
            col_idx.append(j)

    T_sub = csr_matrix((data, (row_idx, col_idx)), shape=(nrows, n))
    constraint = T_sub @ b_var <= M
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [constraint])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp(a0, time_limit, solver, n_active=30, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_iters = 0

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        b_curr = fast_autocorr(y)
        M_curr = b_curr.max()
        if M_curr < 1e-15:
            break

        # Choose active constraints
        if t <= 5:
            active = None  # Use all constraints initially
        else:
            # Top n_active peaks plus some random ones for diversity
            top_idx = np.argsort(b_curr)[::-1][:n_active]
            # Add a few random indices to avoid missing constraints
            extra = np.random.randint(0, 2 * n - 1, size=max(1, n_active // 5))
            active = sorted(set(top_idx.tolist()) | set(extra.tolist()))

        b = solve_lp_best_response(y, M_curr, solver, top_k_indices=active)
        if b is None:
            b = y.copy()
        n_iters += 1

        b_sum = b.sum()
        if b_sum < 1e-15:
            b = y.copy()
            b_sum = b.sum()
        g = b * norm / b_sum

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        # Running average
        avg_a += a
        n_avg += 1
        if n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                sc = fast_autocorr(avg_check).max()
                if sc < best_score:
                    best_score = sc
                    best_a = avg_check.copy()

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def frank_wolfe(a0, time_limit, verbose=False):
    """Frank-Wolfe (conditional gradient) with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(y)
        k_star = int(np.argmax(b))
        M_curr = b[k_star]

        # Subgradient of max_k conv(y,y) w.r.t. y
        # grad[i] = 2*(y[k_star - i] + y[i - k_star]) with 0 for out-of-range
        grad = np.zeros(n)
        for i in range(n):
            v = 0.0
            j1 = k_star - i
            if 0 <= j1 < n:
                v += y[j1]
            j2 = i - k_star
            if 0 <= j2 < n:
                v += y[j2]
            grad[i] = 2.0 * v

        # Frank-Wolfe: move to vertex minimizing <grad, v>
        # Vertex: norm * e_j where j = argmin grad
        j_min = int(np.argmin(grad))
        v = np.zeros(n)
        v[j_min] = norm

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * v
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = v.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 200 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                sc = fast_autocorr(avg_check).max()
                if sc < best_score:
                    best_score = sc
                    best_a = avg_check.copy()

        if verbose and t % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lse_gradient_step(a, tau):
    """One gradient step on the LogSumExp surrogate of max(conv(a,a))."""
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    exp_b = np.exp(tau * (b - max_b))
    w = exp_b / exp_b.sum()

    # grad[i] = 2 * sum_k w[k] * (a[k-i] + a[i-k]) for valid indices
    # = 2 * (w * a_rev_partial + w_conv_a)
    # Use FFT: grad[i] = 2 * cross_corr(w, a)[i]
    # cross_corr(w, a)[i] = sum_k w[k] * a[k-i] (with a out-of-range = 0)
    # This equals (rev(w) * a)[i + (2n-2)]... let's just do it directly with FFT

    # Method: pad w to length 4n-2, compute convolution with a
    fft_size = 1 << (4 * n - 2).bit_length()
    # cross_corr[i] = sum_k w[k] * a[k-i]
    # = (a * rev(w))[i + (len(w)-1)] where * is linear convolution
    w_rev = w[::-1]
    A = rfft(a, fft_size)
    Wr = rfft(w_rev, fft_size)
    conv = irfft(A * Wr, fft_size)
    # conv[j] = sum_i a[i] * w_rev[j-i] = sum_i a[i] * w[len(w_rev)-1-(j-i)]
    # = sum_i a[i] * w[len(w)-1-j+i]
    # We want cross_corr[i] = sum_k w[k] * a[k-i]
    # Let j = k-i: cross_corr[i] = sum_j w[j+i] * a[j]
    # conv[i + 2n-2] = sum_j a[j] * w_rev[2n-2-i] ... hmm this is getting confusing.

    # Simpler: just compute grad directly
    # grad[i] = 2 * (sum_{k: 0<=k-i<n} w[k]*a[k-i] + sum_{k: 0<=i-k<n} w[k]*a[i-k])
    # First sum: let j = k-i, j from max(0,-i) to min(2n-2, n-1-i)
    #   = sum_j w[j+i] * a[j] for j where 0<=j<n and 0<=j+i<2n-1
    # Second sum: let j = i-k, j from max(0,i-(2n-2)) to min(n-1, i)
    #   = sum_j w[i-j] * a[j] for j where 0<=j<n and 0<=i-j<2n-1

    # Use FFT for first sum: cross_corr1[i] = sum_j a[j] * w[i+j]
    # This is (a * rev(w))[2n-2-i]... let me just use a direct FFT approach

    # Actually the simplest: use the fact that
    # sum_k w[k]*a[k-i] = (w * a_full)[2n-2-i] where a_full is a placed at position 0
    # and w * a_full is the linear convolution
    # But w has length 2n-1 and a has length n, so their convolution has length 3n-2

    # Let me just do it with a single FFT
    # Define: h[i] = sum_k w[k] * a[k-i] (cross-correlation of w and a)
    # h = ifft(fft(w) * conj(fft(a))) ... no, that's circular.

    # For linear: pad both to length >= (2n-1) + n - 1 = 3n-2
    L = 1 << (3 * n - 2).bit_length()
    W = rfft(w, L)
    A = rfft(a, L)
    # h[i] = sum_k w[k] * a[k-i] = sum_k w[k] * conj(a)[i-k]...
    # Actually h = ifft(W * conj(A)) gives circular cross-correlation
    # For linear, with zero-padding it's the same.
    h1 = irfft(W * np.conj(A), L)[:2*n-1]  # h1[i] = sum_k w[k]*a[k-i] (circular, but zero-padded)

    # h2[i] = sum_k w[k] * a[i-k]
    # h2 = ifft(conj(W) * A)... no
    # h2[i] = sum_k w[k] * a[i-k] = (w * a)[i] (linear convolution)
    h2 = irfft(W * A, L)[:2*n-1]

    # grad[i] = 2 * (h1[i] + h2[i]) for i = 0, ..., n-1
    # But we need to be careful: h1 and h2 have length 2n-1, we take the first n
    grad = 2.0 * (h1[:n] + h2[:n])

    return grad


def lse_anneal(a0, time_limit, tau_start=50.0, tau_end=0.5, n_stages=8, verbose=False):
    """Anneal the LSE temperature to minimize the max of conv(a,a)."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(tau_start, tau_end, n_stages)
    for si, tau in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break

        # Adaptive number of steps based on remaining time
        n_steps = max(30, min(300, int(remaining * (10 + si * 5))))
        step_base = norm / (tau * n)

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            step = step_base / (1.0 + 0.05 * it)
            grad = lse_gradient_step(a, tau)
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s
            a = a_new
            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

    return best_a, best_score


def coordinate_descent(a0, time_limit, verbose=False):
    """
    Active-set coordinate descent: repeatedly find the peak, then
    move mass from high-contribution coordinates to low-contribution ones.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_peaks = np.argsort(b)[::-1][:20]
        improved = False

        for k in top_peaks:
            # Find coordinates that contribute most to this peak
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            if len(contrib) == 0:
                continue
            src = idx[np.argmax(contrib)]
            max_contrib = contrib.max()
            if max_contrib < 1e-20:
                continue

            # Find target: coordinate with low contribution to ALL top peaks
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            # Candidates: outside peak mask, or inside but with low value
            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]

            # Pick a few random candidates and try moving mass
            n_try = min(10, len(candidates))
            try_idx = rng.choice(candidates, size=n_try, replace=False)

            for dst in try_idx:
                if dst == src:
                    continue
                # Try several fractions
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    amt = a[src] * frac
                    if amt < 1e-20:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Stuck: try random perturbations
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-8:
                frac = rng.uniform(0.02, 0.08)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Determine base length
    if parent is not None:
        n_base = len(parent)
    else:
        n_base = 1600
        best_seq = init_ramp(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # ===== Phase 1: LP Nesterov at base length (from parent) =====
    remaining = time_budget - (time.monotonic() - t_global)
    n1 = n_base
    print(f"\n=== Phase 1: LP Nesterov n={n1} ===")
    t1 = min(remaining * 0.30, 500.0)
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n1) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t1, solver, n_active=40, verbose=True)
        do_try_update(a_out, f"lp_n{n1}")

    # ===== Phase 2: LP Nesterov with ramp init at base length =====
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = n_base
    print(f"\n=== Phase 2: LP Nesterov n={n2} (ramp init) ===")
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 20:
        a_ramp = init_ramp(n2)
        a_norm = a_ramp * np.sqrt(2.0 * n2)
        a_out, _ = nesterov_lp(a_norm, t2, solver, n_active=40, verbose=True)
        do_try_update(a_out, f"lp_ramp_n{n2}")

    # ===== Phase 3: LP Nesterov at longer length =====
    n3 = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LP Nesterov n={n3} ===")
    t3 = min(remaining * 0.20, 350.0)
    if t3 > 30:
        init_long = interp_sequence(best_seq, n3)
        a_norm = init_long * np.sqrt(2.0 * n3) / init_long.sum()
        a_out, _ = nesterov_lp(a_norm, t3, solver, n_active=50, verbose=True)
        do_try_update(a_out, f"lp_n{n3}")

    # ===== Phase 4: Frank-Wolfe at best length =====
    n_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Frank-Wolfe n={n_best} ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = frank_wolfe(a_norm, t4, verbose=False)
        do_try_update(a_out, f"fw_n{n_best}")

    # ===== Phase 5: LSE anneal at best length =====
    n_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: LSE anneal n={n_best} ===")
    t5 = min(remaining * 0.12, 200.0)
    if t5 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_anneal(a_norm, t5, tau_start=40.0, tau_end=0.3, n_stages=8, verbose=True)
        do_try_update(a_out, f"lse_n{n_best}")

    # ===== Phase 6: Coordinate descent polish =====
    n_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Coord descent n={n_best} ===")
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 5:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent(a_norm, t6, verbose=False)
        do_try_update(a_out, f"cd_n{n_best}")

    # ===== Phase 7: LP Nesterov at even longer length =====
    n7 = n_base + 800
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: LP Nesterov n={n7} ===")
    t7 = min(remaining * 0.15, 250.0)
    if t7 > 30:
        init_long7 = interp_sequence(best_seq, n7)
        a_norm = init_long7 * np.sqrt(2.0 * n7) / init_long7.sum()
        a_out, _ = nesterov_lp(a_norm, t7, solver, n_active=50, verbose=True)
        do_try_update(a_out, f"lp_n{n7}")

    # ===== Phase 8: Final LSE + CD polish on best =====
    n_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Final polish n={n_best} ===")
    t8 = min(remaining * 0.5, 200.0)
    if t8 > 5:
        # LSE
        t8a = t8 * 0.5
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_anneal(a_norm, t8a, tau_start=10.0, tau_end=0.1, n_stages=5, verbose=False)
        do_try_update(a_out, "lse_final")
        # CD
        remaining2 = time_budget - (time.monotonic() - t_global)
        t8b = min(remaining2 * 0.5, 100.0)
        if t8b > 3:
            n_best2 = len(best_seq)
            a_norm2 = best_seq * np.sqrt(2.0 * n_best2) / best_seq.sum()
            a_out2, _ = coordinate_descent(a_norm2, t8b, verbose=False)
            do_try_update(a_out2, "cd_final")

    # ===== Final verification =====
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
```
