def fast_convolve(a):
    """Fast linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv(a, b):
    """Fast linear convolution of a and b via FFT."""
    n = len(a) + len(b) - 1
    fft_size = 1 << (n - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:n]


def score_normalized(a):
    """Score assuming sum(a)=1: returns 2*n*max(b)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:

        return float(evaluate_sequence(a_list))
    except Exception:

def softmax_from_z(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Invert softmax (log)."""
    return np.log(np.maximum(a, 1e-300))


def symmetric_constraint(a):
    """Enforce palindromic symmetry: a[i] = a[n-1-i]."""
    n = len(a)
    half = (n + 1) // 2
    a_sym = np.zeros(n)
    for i in range(half):
        a_sym[i] = a[i]
        a_sym[n - 1 - i] = a[i]
    ss = a_sym.sum()
    if ss > 1e-15:
        a_sym /= ss
    return a_sym


def build_conv_rows(a_fixed, row_indices):
    """Build rows of convolution matrix for given indices."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((len(row_indices), n))
    for ii, k in enumerate(row_indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            T[ii, j] = a_fixed[k - j]
    return T

def nesterov_lp(a0, time_limit, solver=None, max_iters=500000, use_cutting=True):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0
    lp_fail_count = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        if use_cutting:
            b_new, M_lp = solve_lp_cutting_plane(y, y, solver=solver)
        else:
            b_new, M_lp = solve_lp_full(y, solver=solver)

        if b_new is None:
            lp_fail_count += 1
            if lp_fail_count > 10:
                use_cutting = False
                lp_fail_count = 0
            continue

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * (it + 1) ** 2)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 10:
                print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} M_lp={M_lp:.6f}")
                last_print = elapsed

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    max_b = b.max()
    scaled = T_val * b
    max_s = scaled.max()
    if max_s > 700:
        scaled = scaled - max_s + 700
        max_s = 700
    exp_s = np.exp(scaled - max_s)
    w = exp_s / exp_s.sum()
    log_sum_exp = max_s + np.log(exp_s.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    # Gradient w.r.t. a
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_bfgs(a0, time_limit, T_start=80.0, T_min=0.3, n_stages=20, iters_per_stage=40):
    """Adaptive LogSumExp L-BFGS."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    T = T_start

    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        T_cur = T

        def func_staged(z_val, T_c=T_cur, n_c=n):
            obj, _ = lse_objective_grad(z_val, T_c, n_c)
            return obj

        def jac_staged(z_val, T_c=T_cur, n_c=n):
            _, grad = lse_objective_grad(z_val, T_c, n_c)
            return grad

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 500,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)

            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()

            # Adaptive temperature
            gap = (sc - best_score) / (best_score + 1e-15)
            if gap > 0.01:
                T = max(T, T * 1.3)
            else:
                T = max(T_min, T * 0.7)

            if (stage + 1) % 5 == 0:
                print(f"    LSE stage={stage+1}/{n_stages} T={T:.2f}: score={sc:.8f} best={best_score:.8f}")

        except Exception:
            T = max(T_min, T * 0.8)

    return best_a, best_score


def irls_momentum(a0, time_limit, n_steps=200000):
    """IRLS with Nesterov momentum and FFT-based gradient."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    # Nesterov momentum state
    y = a.copy()
    v = np.zeros(n)
    gamma = 0.9

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(y)
        max_b = b.max()

        # IRLS weights: emphasize peaks
        T_irls = 3.0 + 5.0 * (step / max(1, n_steps))
        scaled = T_irls * b
        max_s = scaled.max()
        exp_s = np.exp(scaled - max_s)
        w = exp_s / exp_s.sum()

        # Compute gradient using FFT
        # grad[k] = sum_m w[m] * a[m-k] for m-k in 0..n-1
        # This is cross-correlation of w and a
        fft_size = 1 << (2 * n - 2).bit_length()
        W = fft(w, fft_size)
        A = fft(y, fft_size)
        g_full = ifft(W * np.conj(A)).real[:n]

        # Project to tangent space of simplex
        S = np.dot(g_full, y)
        grad = y * (g_full - S)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Polyak step size
        cur_score = score_normalized(y)
        lower_bound = 2.0 * n / (2.0 * n - 1.0)
        gap = cur_score - lower_bound
        eta = gap / (2.0 * grad_norm ** 2)
        eta = min(eta, 0.3)

        # Nesterov update
        v = gamma * v - eta * grad
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / sa

        # Momentum extrapolation
        mu = (1.0 + np.sqrt(1.0 + 4.0 * step * step)) / 2.0
        beta = (mu - 1.0) / mu
        y_new = a_new + beta * (a_new - y)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        # Line search on a_new
        sc_new = score_normalized(a_new)
        if sc_new > cur_score * 1.005:
            # Backtrack
            for _ in range(8):
                eta *= 0.5
                v = gamma * v - eta * grad
                a_candidate = y + v
                a_candidate = np.maximum(a_candidate, 0)
                sc_c = a_candidate.sum()
                if sc_c < 1e-15:
                    continue
                a_candidate = a_candidate / sc_c
                sc_cand = score_normalized(a_candidate)
                if sc_cand < cur_score:
                    a_new = a_candidate
                    break
            else:
                a_new = y.copy()
                v = np.zeros(n)

        y = y_new
        a = a_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        if (step + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    IRLS step={step+1} ({elapsed:.0f}s): cur={cur_score:.8f} best={best_score:.8f}")

    return best_a, best_score


def kkt_polish(a0, time_limit, n_steps=50000):
    """KKT-based peak equalization with adaptive step."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / a.sum()
    z = z_from_a(a).copy()

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    avg_a = a.copy()
    avg_n = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b

        # Active peaks: top 1% or within 0.5% of max
        threshold = max_b * 0.998
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        if len(active) > 40:
            top_idx = np.argsort(b[active])[-40:]
            active = active[top_idx]

        # Average subgradient over active peaks using FFT
        # For each active peak k, gradient contribution to a[j] is a[k-j]
        # Total: g[j] = (1/|S|) * sum_{k in S} a[k-j]
        # This is (1/|S|) * cross_correlation(indicator_S, a)
        ind_S = np.zeros(2 * n - 1)
        ind_S[active] = 1.0

        fft_size = 1 << (2 * n - 2).bit_length()
        IND = fft(ind_S, fft_size)
        A = fft(a, fft_size)
        g_full = ifft(IND * np.conj(A)).real[:n] / len(active)

        # Project to tangent space
        S_dot = np.dot(g_full, a)
        grad = a * (g_full - S_dot)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Adaptive step
        lower_bound = 2.0 * n / (2.0 * n - 1.0)
        gap = cur_score - lower_bound
        eta = gap / (2.0 * grad_norm ** 2)
        eta = min(eta, 0.2)

        z_new = z - eta * grad
        a_new = softmax_from_z(z_new)

        # Line search
        sc_new = score_normalized(a_new)
        if sc_new > cur_score:
            for _ in range(6):
                eta *= 0.5
                z_new = z - eta * grad
                a_new = softmax_from_z(z_new)
                sc_new = score_normalized(a_new)
                if sc_new < cur_score:
                    break
            else:
                a_new = a.copy()

        z = z_new
        a = a_new

        # Running average
        avg_n = min(avg_n + 1, 3000)
        avg_a = (avg_a * (avg_n - 1) + a) / avg_n if avg_n > 1 else a.copy()

        if sc_new < best_score:
            best_score = sc_new
            best_a = a.copy()

        sc_avg = score_normalized(avg_a)
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = avg_a.copy()

        if (step + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    KKT step={step+1} ({elapsed:.0f}s): cur={cur_score:.8f} best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate structured initial sequences."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif choice == 3:
        a = np.ones(n) * (1.0 + 0.2 * np.cos(2 * np.pi * 3 * np.arange(n) / n))
    elif choice == 4:
        xx = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(3 * xx) + 0.15 * np.cos(7 * xx)
    else:
        a = np.sin(np.pi * x) ** 0.5
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def final_propose_candidate(seed=42, budget_s=1800.0):
    """Main search function."""
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    global_deadline = t_start + max(1.0, float(budget_s) - 30.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Total budget: {budget_s}s, effective deadline: {global_deadline - t_start:.0f}s")

    # Get parent sequence
    best_seq = None
    best_score = float('inf')
    n_base = 1200

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            remain = global_deadline - time.monotonic()
            print(f"  ** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, remaining={remain:.0f}s)")

    # === Phase 1: Nesterov LP with cutting plane - 45% time ===
    t1_frac = 0.45
    print(f"\n=== Phase 1: Nesterov LP cutting (score={best_score:.8f}) ===")
    t1 = min(t1_frac * (budget_s - 30), global_deadline - time.monotonic())
    a_out, _ = nesterov_lp(best_seq, t1, solver=solver, max_iters=1000000, use_cutting=True)
    try_update(a_out, "nlp_cutting_main")
    print(f"  Phase 1 done: best_score={best_score:.8f}")

    # === Phase 2: IRLS with momentum - 20% time ===
    print(f"\n=== Phase 2: IRLS momentum (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t2 = min(0.20 * (budget_s - 30), remaining * 0.5)
    if t2 > 30:
        a_irls, _ = irls_momentum(best_seq, t2, n_steps=500000)
        try_update(a_irls, "irls_momentum")

        remaining = global_deadline - time.monotonic()
        if remaining > 30:
            # Second IRLS pass from result
            a_irls2, _ = irls_momentum(a_irls, remaining * 0.5, n_steps=300000)
            try_update(a_irls2, "irls_pass2")
    print(f"  Phase 2 done: best_score={best_score:.8f}")

    # === Phase 3: LSE L-BFGS - 15% time ===
    print(f"\n=== Phase 3: LSE L-BFGS (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t3 = min(0.15 * (budget_s - 30), remaining * 0.5)
    if t3 > 30:
        a_lse, _ = lse_bfgs(best_seq, t3, T_start=60.0, T_min=0.2, n_stages=15, iters_per_stage=40)
        try_update(a_lse, "lse_bfgs")

        remaining = global_deadline - time.monotonic()
        if remaining > 20:
            a_lse2, _ = lse_bfgs(best_seq, remaining * 0.5, T_start=20.0, T_min=0.1, n_stages=12, iters_per_stage=35)
            try_update(a_lse2, "lse_lowT")
    print(f"  Phase 3 done: best_score={best_score:.8f}")

    # === Phase 4: KKT Polish - 10% time ===
    print(f"\n=== Phase 4: KKT Polish (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t4 = min(0.10 * (budget_s - 30), remaining * 0.5)
    if t4 > 20:
        a_kkt, _ = kkt_polish(best_seq, t4, n_steps=100000)
        try_update(a_kkt, "kkt_polish")
    print(f"  Phase 4 done: best_score={best_score:.8f}")

    # === Phase 5: Multi-length exploration - 8% time ===
    print(f"\n=== Phase 5: Multi-length (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t5_total = min(0.08 * (budget_s - 30), remaining * 0.6)
    if t5_total > 30:
        lengths_to_try = [n_base + 200, n_base + 500, n_base - 200]
        for n_try in lengths_to_try:
            remaining = global_deadline - time.monotonic()
            if remaining < 25:
                break
            t5 = min(t5_total / 3.0, remaining * 0.45)
            if t5 < 15:
                break
            print(f"  Trying n={n_try}, budget={t5:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            a_try, _ = nesterov_lp(init_try, t5 * 0.7, solver=solver, max_iters=200000, use_cutting=True)
            try_update(a_try, f"len_{n_try}")
            remaining = global_deadline - time.monotonic()
            if remaining > 15:
                a_try2, _ = irls_momentum(a_try, remaining * 0.4, n_steps=100000)
                try_update(a_try2, f"len_{n_try}_irls")
    print(f"  Phase 5 done: best_score={best_score:.8f}")

    # === Phase 6: Final combined polish - remaining time ===
    print(f"\n=== Phase 6: Final polish (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    if remaining > 30:
        # Short NLP
        a_f1, _ = nesterov_lp(best_seq, remaining * 0.3, solver=solver, max_iters=50000, use_cutting=True)
        try_update(a_f1, "final_nlp")

        remaining = global_deadline - time.monotonic()
        if remaining > 20:
            # KKT from current best
            a_f2, _ = kkt_polish(best_seq, remaining * 0.4, n_steps=50000)
            try_update(a_f2, "final_kkt")

        remaining = global_deadline - time.monotonic()
        if remaining > 15:
            # Symmetric variant
            a_sym = symmetric_constraint(best_seq)
            a_f3, _ = nesterov_lp(a_sym, remaining * 0.5, solver=solver, max_iters=30000, use_cutting=True)
            try_update(a_f3, "final_sym")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    elapsed = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"Time used: {elapsed:.0f}s")
    return [float(x) for x in best_seq]


def propose_candidate(seed=42, budget_s=1800.0):
    """Entry point for the search."""
    return final_propose_candidate(seed=seed, budget_s=budget_s)
