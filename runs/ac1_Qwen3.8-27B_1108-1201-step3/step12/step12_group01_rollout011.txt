def optimal_move_amount(a, i, j, active_set):
    """Find optimal δ to move from i to j, given active set."""
    b = fast_autocorr(a)
    M = b.max()
    
    # For each active k, compute the slope
    # f_k(δ) = b[k] + 2*δ*(a[k-j] - a[k-i])
    # Find δ that minimizes max over active k of f_k
    
    slopes = []
    intercepts = []
    for k in active_set:
        # a[k-i] is valid if 0 <= k-i <= n-1
        ai = a[k-i] if 0 <= k-i < len(a) else 0
        aj = a[k-j] if 0 <= k-j < len(a) else 0
        slopes.append(2*(aj - ai))
        intercepts.append(b[k])
    
    slopes = np.array(slopes)
    intercepts = np.array(intercepts)
    
    # Find kink points: where f_k1(δ) = f_k2(δ)
    # intercepts[k1] + slopes[k1]*δ = intercepts[k2] + slopes[k2]*δ
    # δ = (intercepts[k2] - intercepts[k1]) / (slopes[k1] - slopes[k2])
    
    max_δ = a[i]  # Can't move more than a[i]
    
    candidates = [0, max_δ]
    for p in range(len(slopes)):
        for q in range(p+1, len(slopes)):
            if abs(slopes[p] - slopes[q]) > 1e-15:
                δ = (intercepts[q] - intercepts[p]) / (slopes[p] - slopes[q])
                if 0 <= δ <= max_δ:
                    candidates.append(δ)
    
    # Evaluate g(δ) at each candidate
    best_δ = 0
    best_g = np.inf
    for δ in candidates:
        vals = intercepts + slopes * δ
        g = vals.max()
        if g < best_g:
            best_g = g
            best_δ = δ
    
    return best_δ, best_g

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP Nesterov zero-sum game (main optimizer)
2. Analytical peak shaving with exact 1D minimax (fine polish)
3. Multi-n exploration and interpolation
4. Coordinate descent polishing
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_best_response(a_fixed, solver):
    """Solve LP: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0.
    Returns (b_star, M_value)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.SCS, cp.ECOS, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.PDLP, cp.SCS, cp.ECOS, cp.CLARABEL]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def nesterov_lp(a0, time_limit, solver, max_iters=500000):
    """Nesterov-accelerated LP best-response."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    # Running average
    avg = np.zeros(n)
    n_avg = 0
    stall = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val = solve_best_response(y, solver)
        if b_star is None:
            continue

        alpha = 2.0 / (t + 2.0)
        y_new = (1 - alpha) * y + alpha * b_star
        y_new = np.maximum(y_new, 0)
        ys = y_new.sum()
        if ys < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / ys

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new

        y = y_new
        a = a_new

        # Track average
        avg += a
        n_avg += 1

        # Check periodically
        if t % 10 == 1:
            score_a = 2.0 * n * fast_autocorr(a).max()
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()
                stall = 0
            else:
                stall += 1

            if n_avg >= 100 and t % 50 == 0:
                avg_a = avg / n_avg
                score_avg = 2.0 * n * fast_autocorr(avg_a).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_a.copy()
                    stall = 0

        # Restart if stalled
        if stall > 500 and t > 100:
            a = best_a.copy()
            y = best_a.copy()
            avg = np.zeros(n)
            n_avg = 0
            stall = 0

        if t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def analytical_shave(a0, time_limit):
    """
    Analytical peak shaving: for each (src, dst) pair, compute the exact
    optimal move amount using 1D minimax.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0, float('inf')
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 1:
        if stall > 500:
            break

        b = fast_autocorr(a)
        M = b.max()

        # Active set: top peaks
        top_k = min(50, len(b))
        active_idx = np.argsort(b)[::-1][:top_k]
        # Only include those close to max
        active_idx = active_idx[b[active_idx] >= M * 0.999]
        if len(active_idx) == 0:
            active_idx = np.argsort(b)[::-1][:top_k]

        # Compute danger for each position
        danger = np.zeros(n)
        for k in active_idx:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            danger[idx] += a[idx] * a[k - idx]

        # Source candidates: high danger * high value
        src_score = danger * (a + 1e-10)
        src_candidates = np.argsort(src_score)[::-1][:50]

        # Dest candidates: low danger
        dst_candidates = np.argsort(danger)[:50]

        improved = False
        for src in src_candidates:
            if a[src] < 1e-12:
                continue
            for dst in dst_candidates:
                if dst == src:
                    continue

                # Compute the exact optimal δ
                # For each relevant k, b[k] changes by:
                # Δb[k] = 2δ*(a[k-dst] - a[k-src]) + δ²*(correction terms)
                # For small δ, first order is sufficient for finding the kink.
                
                # Use top 30 peaks for the minimax
                use_k = np.argsort(b)[::-1][:30]
                
                slopes = np.zeros(len(use_k))
                for idx2, k in enumerate(use_k):
                    ak_src = a[k - src] if 0 <= k - src < n else 0.0
                    ak_dst = a[k - dst] if 0 <= k - dst < n else 0.0
                    slopes[idx2] = 2.0 * (ak_dst - ak_src)
                
                intercepts = b[use_k]
                max_delta = a[src]

                # Find kink points
                candidates = [0.0, max_delta]
                for p in range(len(slopes)):
                    for q in range(p + 1, len(slopes)):
                        ds = slopes[p] - slopes[q]
                        if abs(ds) > 1e-15:
                            delta = (intercepts[q] - intercepts[p]) / ds
                            if 0 < delta < max_delta:
                                candidates.append(delta)

                # Also try some fractional moves
                candidates.append(max_delta * 0.5)
                candidates.append(max_delta * 0.1)
                candidates.append(max_delta * 0.01)

                best_delta = 0
                best_new_m = M
                for delta in candidates:
                    delta = max(0, min(max_delta, delta))
                    if delta < 1e-15:
                        continue
                    new_m = (intercepts + slopes * delta).max()
                    if new_m < best_new_m - 1e-14:
                        best_new_m = new_m
                        best_delta = delta

                if best_delta > 1e-12:
                    a_new = a.copy()
                    a_new[src] -= best_delta
                    a_new[dst] += best_delta
                    a_new = np.maximum(a_new, 0)
                    ss = a_new.sum()
                    if ss < 1e-15:
                        continue
                    a_new = a_new / ss
                    score = 2.0 * n * fast_autocorr(a_new).max()
                    if score < best_score - 1e-14:
                        a = a_new
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
            if improved:
                break

        if not improved:
            stall += 1
            # Small random perturbation to escape
            if stall > 50:
                rng = np.random.default_rng(stall)
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.01)
                    a_trial = a.copy()
                    a_trial[i] -= a_trial[i] * frac
                    a_trial[j] += a_trial[i] * frac
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st > 1e-10:
                        a_trial = a_trial / st
                        score = 2.0 * n * fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0

        if stall > 500:
            break

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit):
    """
    Coordinate descent: for each position i, find the optimal value
    given all others fixed, using the sum constraint.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0, float('inf')
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit - 1:
        b = fast_autocorr(a)
        M = b.max()
        active_idx = np.where(b >= M * 0.999)[0]
        if len(active_idx) > 50:
            active_idx = np.argsort(b)[::-1][:50]

        improved = False
        # Random order
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit - 1:
                break
            if a[i] < 1e-12:
                continue

            # Try moving small amounts from i to random destinations
            n_trials = 5
            for _ in range(n_trials):
                j = rng.integers(n)
                if j == i:
                    continue
                for frac in [0.01, 0.05, 0.1, 0.3]:
                    delta = a[i] * frac
                    if delta < 1e-14:
                        continue
                    a_trial = a.copy()
                    a_trial[i] -= delta
                    a_trial[j] += delta
                    if a_trial[i] < 0:
                        continue
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-10:
                        continue
                    a_trial = a_trial / st
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
