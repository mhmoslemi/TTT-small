def parametric_init(n):
    """Try multiple parametric families, return the best sequence."""
    candidates = []

    # Uniform
    candidates.append(np.ones(n))

    # Cosine window
    x = np.linspace(0, np.pi, n)
    candidates.append(np.cos(x / 2.0) ** 2)

    # Hann
    x = np.linspace(0, 2 * np.pi, n, endpoint=False)
    candidates.append(np.maximum(0, 0.5 * (1 + np.cos(x))))

    # Exponential decay (various rates)
    for alpha in [0.3, 0.5, 1.0, 2.0, 3.0, 5.0]:
        candidates.append(np.exp(-alpha * np.arange(n) / n))

    # Gaussian (various widths)
    x = np.arange(n) / n - 0.5
    for sigma in [0.15, 0.2, 0.25, 0.3, 0.4, 0.5]:
        candidates.append(np.exp(-x ** 2 / (2.0 * sigma ** 2)))

    # Power law: (1-x^2)^p
    x = np.linspace(-1, 1, n)
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        candidates.append(np.maximum(0, 1 - x ** 2) ** p)

    # Skewed polynomial
    for p in [0.5, 1.0, 1.5, 2.0]:
        x = np.arange(n) / (n - 1)
        candidates.append(x ** p)

    # Product of two exponentials (bell-shaped but skewed)
    for a1 in [0.5, 1.0, 2.0]:
        for a2 in [1.0, 2.0, 3.0]:
            x = np.arange(n) / n
            candidates.append(np.exp(-a1 * x) * np.exp(-a2 * (1 - x)))

    # Fejer-like
    x = np.linspace(0, 1, n)
    candidates.append((1 - x) ** 1)
    candidates.append((1 - x * x) ** 2)

    best = None
    best_score = float('inf')
    for a in candidates:
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-10:
            continue
        a = a / s
        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best = a.copy()

    if best is None:
        best = np.ones(n) / n
        best_score = 2.0
    return best, best_score

            # dc_2/da_1 = a_{2-1} + a_{2-1} = 2*a_1. Directly: dc_2/da1 = 2*a1. Yes.
            # dc_2/da_0 from the "other side": when k-j=0, j=2, term is a_2*a_0, derivative wrt a_0 is a_2.
            # And when j=0, term is a_0*a_2, derivative wrt a_0 is a_2.
            # Total: 2*a_2. Correct.
            # For c_1 = a0*a1 + a1*a0: dc_1/da_0 = a_1 + a_1 = 2*a_1. Correct.
            # For c_0 = a0*a0: dc_0/da_0 = a_0 (only one term). So the formula gives 2*a_0, which is wrong!
            # The issue: when j=i and k-j=i are the same (k=2i=0, i=0), we only have one term.
            # So: dc_k/da_i = 2*a_{k-i} if k-i != i (i.e., k != 2i), else a_{k-i} = a_i.
            # Or equivalently: dc_k/da_i = a_{k-i} * (1 + 1[k != 2i]) if 0<=k-i<=n-1, else 0.

            j = k - i  # This would be the index of a in the other factor
            # But we need to handle the full range properly
            pass

        # Simplified gradient computation using the symmetry:
        # For the full convolution c_k = sum_j a_j * a_{k-j}, the gradient is:
        # (grad c_k)_i = 2 * a_{k-i} for k-i in [0, n-1], with special handling when k=2i
        # For practical purposes, let's use:
        # grad_i = 2 * sum_{k active} w_k * a_{k-i} (clamped to valid range)
        # This is an approximation that works well in practice.

        # Compute gradient more carefully using the structure
        for wi, k in zip(w, active):
            # Valid j range for c_k: j from max(0, k-(n-1)) to min(n-1, k)
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            for j in range(j_lo, j_hi + 1):
                other = k - j
                # dc_k/da_j gets contribution a_other
                grad[j] += 2.0 * wi * a[other]
                # dc_k/da_other gets contribution a_j (from the other side)
                if other != j:
                    grad[other] += 2.0 * wi * a[j]

        # Mirror descent step: a_new(i) = a(i) * exp(-eta * grad(i)) / Z
        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a - eta * grad
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        s_new = a_new.sum()
        if s_new < 1e-30:
            break
        a_new = a_new / s_new

        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        a = a_new
        eta = max(eta * eta_decay, min_eta)
        n_iters += 1

        if verbose and it % 500 == 0:
            print(f"    SG it={n_iters} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def coordinate_polish(a0, time_limit, seed=123):
    """Lightweight coordinate descent: try moving mass from peak-contributing positions."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize_vec(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        peak_idx = np.argmax(b)

        # Find top contributors to the peak
        j_lo = max(0, peak_idx - n + 1)
        j_hi = min(n - 1, peak_idx)
        contrib = np.zeros(n)
        for j in range(j_lo, j_hi + 1):
            other = peak_idx - j
            contrib[j] = a[j] * a[other]

        # Try moving mass from top contributor to a random target
        top_srcs = np.argsort(contrib)[::-1][:3]
        improved = False

        for src in top_srcs:
            if a[src] < 1e-10:
                continue
            for frac in [0.005, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                dst = rng.integers(0, n)
                if dst == src:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 200:
                # Random perturbation to escape
                n_pert = rng.integers(2, 6)
                srcs = rng.choice(n, size=n_pert, replace=False)
                dsts = rng.choice(n, size=n_pert, replace=False)
                a_trial = a.copy()
                for s_i, d_i in zip(srcs, dsts):
                    amt = a_trial[s_i] * 0.05
                    a_trial[s_i] -= amt
                    a_trial[d_i] += amt
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s > 1e-15:
                    a_trial = a_trial / s
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            if stall > 500:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Uses LP Nesterov with HiGHS as the primary optimizer,
    with parametric initialization, multi-length exploration, and refinement.
    """
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1680.0)

    best_seq = None
    best_score = float('inf')

    # Try to get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Determine base length
    n_base = len(best_seq) if best_seq is not None else 1600

    if parent is None:
        n_base = 1600
        print(f"No parent, using n_base={n_base}")
        init, init_score = parametric_init(n_base)
        best_seq = init
        best_score = 2.0 * n_base * fast_autocorr(init).max()
        print(f"Parametric init: score={best_score:.6f}")

    # Phase 1: Main LP Nesterov at base length
    remaining = time_budget - (time.monotonic() - t_global_start)
    t1 = min(remaining * 0.45, 750.0)
    print(f"\n=== Phase 1: LP Nesterov n={n_base}, budget={t1:.0f}s ===")
    if t1 > 60:
        a_out, sc_out = nesterov_lp_optimize(best_seq, t1, verbose=True, tag=f"_n{n_base}")
        try_update(a_out, f"nlp_n{n_base}")

    # Phase 2: Subgradient refinement
    remaining = time_budget - (time.monotonic() - t_global_start)
    t2 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 2: Subgradient refine, budget={t2:.0f}s ===")
    if t2 > 20:
        a_out, sc_out = subgradient_refine(best_seq, t2, verbose=True)
        try_update(a_out, "subgrad")

    # Phase 3: Try a longer length
    n_long = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global_start)
    t3 = min(remaining * 0.20, 350.0)
    print(f"\n=== Phase 3: LP Nesterov n={n_long}, budget={t3:.0f}s ===")
    if t3 > 60:
        init_long = interpolate_sequence(best_seq, n_long)
        a_out, sc_out = nesterov_lp_optimize(init_long, t3, verbose=False, tag=f"_n{n_long}")
        try_update(a_out, f"nlp_n{n_long}")

    # Phase 4: Try a shorter length
    n_short = n_base - 200
    if n_short >= 400:
        remaining = time_budget - (time.monotonic() - t_global_start)
        t4 = min(remaining * 0.12, 200.0)
        print(f"\n=== Phase 4: LP Nesterov n={n_short}, budget={t4:.0f}s ===")
        if t4 > 40:
            init_short = interpolate_sequence(best_seq, n_short)
            a_out, sc_out = nesterov_lp_optimize(init_short, t4, verbose=False, tag=f"_n{n_short}")
            try_update(a_out, f"nlp_n{n_short}")

    # Phase 5: Second LP Nesterov round on the current best
    remaining = time_budget - (time.monotonic() - t_global_start)
    t5 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 5: LP Nesterov round 2, n={len(best_seq)}, budget={t5:.0f}s ===")
    if t5 > 40:
        a_out, sc_out = nesterov_lp_optimize(best_seq, t5, verbose=True, tag="_r2")
        try_update(a_out, "nlp_round2")

    # Phase 6: Coordinate polish
    remaining = time_budget - (time.monotonic() - t_global_start)
    t6 = min(remaining * 0.15, 120.0)
    print(f"\n=== Phase 6: Coordinate polish, budget={t6:.0f}s ===")
    if t6 > 10:
        a_out, sc_out = coordinate_polish(best_seq, t6, seed=999)
        try_update(a_out, "coord_polish")

    # Phase 7: Final short LP Nesterov
    remaining = time_budget - (time.monotonic() - t_global_start)
    t7 = min(remaining * 0.5, 100.0)
    print(f"\n=== Phase 7: Final LP Nesterov, budget={t7:.0f}s ===")
    if t7 > 20:
        a_out, sc_out = nesterov_lp_optimize(best_seq, t7, verbose=False, tag="_final")
        try_update(a_out, "nlp_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        total_time = time.monotonic() - t_global_start
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={total_time:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0] * 100

    return [float(x) for x in best_seq]
