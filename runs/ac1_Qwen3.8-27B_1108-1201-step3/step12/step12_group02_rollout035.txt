```python
def lp_direction(a, K=20):
    """Find the LP-optimal direction to reduce max(b)."""
    n = len(a)
    b = fast_autocorr(a)
    top_idx = np.argsort(b)[::-1][:K]
    
    # Build LP:
    # min t
    # s.t. g_k @ delta - t <= -b[k], k in top_idx
    #      sum(delta) = 0
    #      -a <= delta <= inf (i.e., a + delta >= 0)
    
    # Variables: [delta_0, ..., delta_{n-1}, t] (n+1 vars)
    
    # A_ub: (K + n) x (n+1)
    # b_ub: (K + n,)
    
    # For each k in top_idx:
    #   g_k[0]*delta_0 + ... + g_k[n-1]*delta_{n-1} - t <= -b[k]
    #   where g_k[j] = a[|k-j|] (with proper bounds)
    
    # For each j:
    #   -delta_j <= a[j]  (i.e., delta_j >= -a[j])
    
    # A_eq: [[1, 1, ..., 1, 0]] (sum delta = 0)
    # b_eq: [0]
    
    # c (objective): [0, 0, ..., 0, 1] (minimize t)
    
    # Build A_ub
    A_ub = np.zeros((K + n, n + 1))
    b_ub = np.zeros(K + n)
    
    for ii in range(K):
        k = int(top_idx[ii])
        # g_k[j] for j = 0..n-1
        # Part 1: a[k-j] for j in [max(0,k-n+1), min(n-1,k)]
        j1_start = max(0, k - n + 1)
        j1_end = min(n - 1, k)
        if j1_start <= j1_end:
            js = np.arange(j1_start, j1_end + 1)
            A_ub[ii, js] = a[k - js]
        
        # Part 2: a[j-k] for j in [max(0,k), min(n-1,k+n-1)]
        j2_start = max(0, k)
        j2_end = min(n - 1, k + n - 1)
        if j2_start <= j2_end:
            js = np.arange(j2_start, j2_end + 1)
            A_ub[ii, js] += a[js - k]
        
        A_ub[ii, n] = -1  # coefficient of t
        b_ub[ii] = -b[k]
    
    # Lower bound constraints: -delta_j <= a[j]
    for j in range(n):
        A_ub[K + j, j] = -1
        b_ub[K + j] = a[j]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1
    b_eq = np.array([0.0])
    
    c = np.zeros(n + 1)
    c[n] = 1  # minimize t
    
    # Solve
    from scipy.optimize import linprog
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  method='highs')
    
    if res.success:
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based subgradient optimization
combined with LSE-smoothed L-BFGS refinement.

Key innovations over previous approach:
1. LP-based direction finding: at each step, solve a linear program to find the
   optimal direction that simultaneously reduces the top-K autocorrelation peaks.
   This is more principled than heuristic subgradient methods.
2. Integrated LSE refinement within the main loop (not as a separate phase).
3. Adaptive step-size with backtracking line search.
4. Multi-length exploration with frequency-informed initialization.
5. Nesterov-style momentum for faster convergence.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
from scipy.sparse import lil_matrix
import time


def fa_fast_autocorr(a):
    """Full autocorrelation a*a using FFT. Returns array of length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fa_fast_score(a):
    """Compute 2n*max(b)/(sum(a))^2 quickly."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fa_fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)


def fa_cross_corr(w, a):
    """Compute cross-correlation: result[j] = sum_i w[j+i]*a[i]."""
    n = len(a)
    a_rev = a[::-1].copy()
    conv_len = n + len(w) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a_rev, fft_size)
    conv = irfft(W * A, fft_size)
    return conv[n - 1:n - 1 + n]


def fa_lse_obj_grad_z(z, T):
    """Objective and gradient in z-space where a = softmax(z)."""
    n = len(z)
    z_shift = z - z.max()
    e = np.exp(z_shift)
    a = e / e.sum()

    b = fa_fast_autocorr(a)
    Tb = T * b
    Tb_shift = Tb - Tb.max()
    exp_Tb = np.exp(Tb_shift)
    obj = (Tb.max() + np.log(exp_Tb.sum())) / T

    w = exp_Tb / exp_Tb.sum()
    cc = fa_cross_corr(w, a)
    grad_a = 2.0 * cc
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    return obj, grad_z


def fa_lp_direction(a, K, use_sparse=True):
    """
    Solve LP to find optimal direction to reduce top-K autocorrelation peaks.
    Returns (delta, t_value) where delta is the direction and t_value is the
    LP optimal value (predicted new max).
    """
    n = len(a)
    b = fa_fast_autocorr(a)
    top_idx = np.argsort(b)[::-1][:K]

    # Build sparse A_ub
    # Variables: [delta_0, ..., delta_{n-1}, t] (n+1 vars)
    # Constraints:
    #   For each k in top_idx: g_k @ delta - t <= -b[k]  (K constraints)
    #   -delta_j <= a[j] for j=0..n-1  (n constraints, i.e., delta_j >= -a[j])
    # Equality: sum(delta) = 0

    nnz_estimate = K * n + n  # rough estimate
    A_lil = lil_matrix((K + n, n + 1), dtype=np.float64)
    b_ub = np.zeros(K + n)

    for ii in range(K):
        k = int(top_idx[ii])
        # g_k[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
        j1_start = max(0, k - n + 1)
        j1_end = min(n - 1, k)
        if j1_start <= j1_end:
            js = np.arange(j1_start, j1_end + 1)
            A_lil[ii, js] = a[k - js]

        j2_start = max(0, k)
        j2_end = min(n - 1, k + n - 1)
        if j2_start <= j2_end:
            js = np.arange(j2_start, j2_end + 1)
            A_lil[ii, js] = A_lil[ii, js] + a[js - k]

        A_lil[ii, n] = -1.0
        b_ub[ii] = -b[k]

    # Lower bound: -delta_j <= a[j]
    idx = np.arange(n)
    A_lil[K:K + n, idx] = -1.0
    b_ub[K:K + n] = a

    A_eq = lil_matrix((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])

    c = np.zeros(n + 1)
    c[n] = 1.0

    # Convert to CSR for solver
    A_csr = A_lil.tocsr()
    Aeq_csr = A_eq.tocsr()

    res = linprog(c, A_ub=A_csr, b_ub=b_ub, A_eq=Aeq_csr, b_eq=b_eq,
                  method='highs', options={'presolve': True, 'dual_feasibility_tolerance': 1e-7})

    if res.success:
        delta = res.x[:n]
        return delta, float(res.fun), b, top_idx
    else:
        return None, None, b, top_idx


def fa_lp_optimize(a0, time_limit, K_start=5, K_end=30, K_ramp_frac=0.3,
                   alpha_start=1.0, alpha_min=0.1, seed=42, verbose=True):
    """
    LP-based subgradient optimization with adaptive K and step size.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fa_fast_score(a)
    t0 = time.monotonic()
    stall = 0
    last_best = best_score
    momentum = np.zeros(n)
    beta = 0.3  # momentum coefficient

    iters = 0
    while time.monotonic() - t0 < time_limit:
        iters += 1
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / (time_limit * K_ramp_frac))
        K = max(K_start, min(K_end, int(K_start + frac * (K_end - K_start))))
        alpha = max(alpha_min, alpha_start * (alpha_min / alpha_start) ** frac)

        # Compute LP direction
        delta, t_val, b, top_idx = fa_lp_direction(a, K)

        if delta is None:
            # Fallback: random perturbation
            delta = rng.standard_normal(n)
            delta -= delta.mean()
            alpha = 0.01

        # Nesterov momentum
        momentum = beta * momentum + (1 - beta) * delta
        direction = momentum

        # Normalize direction
        d_norm = np.linalg.norm(direction)
        if d_norm < 1e-15:
            direction = rng.standard_normal(n)
            direction -= direction.mean()
            d_norm = np.linalg.norm(direction)
        direction = direction / d_norm

        # Backtracking line search
        best_alpha = 0.01
        for trial_alpha in [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.1, 0.005, 0.001]:
            a_trial = a + trial_alpha * d_norm * direction
            a_trial = np.maximum(a_trial, 0)
            s_trial = a_trial.sum()
            if s_trial < 1e-15:
                continue
            a_trial = a_trial / s_trial
            sc = fa_fast_score(a_trial)
            if sc < best_score:
                best_alpha = trial_alpha
                break

        a_new = a + best_alpha * d_norm * direction
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a + 0.01 * rng.standard_normal(n)
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
        else:
            a_new = a_new / s_new

        a = a_new
        iters += 1

        # Check score
        if iters % 5 == 0 or iters < 20:
            sc = fa_fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stall = 0
            else:
                stall += 1

        # Stall handling
        if stall > 300:
            noise_scale = min(0.03, 0.005 * (stall / 100.0))
            a = a + noise_scale * rng.standard_normal(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            momentum = np.zeros(n)
            stall = 0

        if verbose and iters % 5000 == 0:
            print(f"    LP t={iters} ({elapsed:.0f}s): best={best_score:.6f}, K={K}, alpha={alpha:.4f}")

    return best_a, best_score, iters


def fa_lse_refine(a0, time_limit, T_start=50.0, T_end=0.2, n_stages=15, seed=42):
    """LSE-smoothed L-BFGS refinement."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s

    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = fa_fast_score(a_init)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        iters = max(10, min(60, int(20 + 50 * si / max(1, n_stages))))

        def func(z_in):
            val, _ = fa_lse_obj_grad_z(z_in, stage_T)
            return val

        def jac(z_in):
            _, g = fa_lse_obj_grad_z(z_in, stage_T)
            return g

        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': iters, 'ftol': 1e-18,
                                    'gtol': 1e-12, 'maxfun': iters * 300,
                                    'disp': False})
            z = res.x
            z_shift = z - z.max()
            a = np.exp(z_shift)
            a = a / a.sum()
            sc = fa_fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        except Exception:
            pass

    return best_a, best_score


def fa_interpolate(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = fa_fast_score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = fa_fast_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_default = 1600
        best_seq = fa_structured_init(n_default, variant=4, rng=rng)
        best_score = fa_fast_score(best_seq)
        print(f"Default init: n={n_default}, score={best_score:.6f}")

    n_parent = len(best_seq)
    print(f"\nStarting optimization at n={n_parent}, score={best_score:.6f}")

    # Phase 1: Alternating LP + LSE at parent length (main workhorse)
    print(f"\n=== Phase 1: Alternating LP+LSE at n={n_parent} ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase1 = min(remaining * 0.45, 700.0)
    if t_phase1 > 30:
        a_out, sc = fa_alternating_lp_lse(best_seq, t_phase1, seed=1, verbose=True)
        try_update(a_out, f"alt_lp_lse_n{n_parent}")

    # Phase 2: Direct LP optimization with different parameters
    print(f"\n=== Phase 2: Direct LP at n={n_parent} ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase2 = min(remaining * 0.15, 250.0)
    if t_phase2 > 20:
        a_out, sc, iters = fa_lp_optimize(best_seq, t_phase2,
                                           K_start=15, K_end=40,
                                           K_ramp_frac=0.5,
                                           alpha_start=0.3, alpha_min=0.05,
                                           seed=7, verbose=True)
        try_update(a_out, f"lp_direct_n{n_parent} ({iters} iters)")

    # Phase 3: LSE refinement
    print(f"\n=== Phase 3: LSE refinement at n={n_parent} ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase3 = min(remaining * 0.12, 200.0)
    if t_phase3 > 20:
        a_out, sc = fa_lse_refine(best_seq, t_phase3, T_start=30.0, T_end=0.1,
                                   n_stages=15, seed=11)
        try_update(a_out, f"lse_n{n_parent}")

    # Phase 4: Multi-length exploration
    print(f"\n=== Phase 4: Multi-length exploration ===")
    remaining = time_budget - (time.monotonic() - t_global)
    n_candidates = [int(n_parent * ratio) for ratio in [0.8, 1.2, 1.5, 1.3]]
    n_candidates = sorted(set([n for n in n_candidates if 500 <= n <= 5000]))

    for n_try in n_candidates[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        t_trial = min(remaining * 0.2, 300.0)
        if t_trial < 30:
            continue

        print(f"  Trying n={n_try} (budget={t_trial:.0f}s)")
        a_interp = fa_interpolate(best_seq, n_try)
        a_out = fa_coarse_fine_optimize(a_interp, n_try, t_trial, seed=n_try % 100)
        try_update(a_out, f"coarse_fine_n{n_try}")

    # Phase 5: Final polish
    print(f"\n=== Phase 5: Final polish ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase5 = min(remaining * 0.25, 300.0)
    if t_phase5 > 10 and best_seq is not None:
        a_out, sc = fa_final_polish(best_seq, t_phase5, seed=999)
        try_update(a_out, "final_polish")

    # Phase 6: One more LSE pass on the very best
    remaining = time_budget - (time.monotonic() - t_global)
    t_phase6 = min(remaining * 0.3, 150.0)
    if t_phase6 > 10 and best_seq is not None:
        a_out, sc = fa_lse_refine(best_seq, t_phase6, T_start=5.0, T_end=0.05,
                                   n_stages=12, seed=777)
        try_update(a_out, "final_lse")

    # Final verification
    if best_seq is not None:
        final_sc = fa_fast_score(best_seq)
        print(f"\nFINAL: score={final_sc:.6f}, n={len(best_seq)}, "
              f"max_b={fa_fast_autocorr(best_seq).max():.6e}")
    else:
        print("ERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
```
