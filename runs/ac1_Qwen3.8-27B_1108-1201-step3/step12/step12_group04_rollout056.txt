Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast 2-exchange local search guided by peak analysis

4. Adaptive step-size coordinate descent for final polish

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Full linear self-convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.3])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where (T@b)[k] = convolve(a_fixed, b)[k]."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

def lp_best_response(a_fixed, solver):
    """
    Solve LP: min M s.t. (a_fixed * b)[k] <= M for all k, sum(b)=1, b>=0.
    Returns (b, M).
    """
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints = constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:

def nesterov_lp(a0, time_limit, solver, max_iters=500):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_star, M_val = lp_best_response(y, solver)
        if b_star is None:
            continue
        
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        ss = y_new.sum()
        if ss < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / ss
        
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            ss = a_new.sum()
            if ss < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / ss
        
        y = y_new
        a = a_new
        
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP t={t} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def peak_guided_local_search(a0, time_limit, seed=42):
    """
    2-exchange local search guided by peak analysis.
    For each active peak k*, find pairs (i, j) where moving mass from i to j
    reduces the peak: a[k*-j] < a[k*-i].
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    max_stall = 50000
    
    # Precompute useful indices
    indices = np.arange(n)
    
    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a)
        c_max = c.max()
        
        if 2.0 * n * c_max < best_score:
            best_score = 2.0 * n * c_max
            best_a = a.copy()
            stall = 0
            if best_score < 1.503:
                print(f"    PGLS: best={best_score:.8f} (t={time.monotonic()-t0:.0f}s)")
        
        # Find active peaks (within 0.5% of max)
        threshold = 0.995 * c_max
        active_peaks = np.where(c >= threshold)[0]
        
        # For each active peak, compute the "score" for moving mass
        # For peak k*, the contribution of position i is a[i] * a[k*-i]
        # We want to move mass from positions with high contribution to positions with low contribution
        
        improved = False
        n_peaks = len(active_peaks)
        
        if n_peaks == 0:
            break
        
        # Strategy: for the dominant peak, find the top contributors and try to move them
        # Sort active peaks by value
        peak_order = active_peaks[np.argsort(c[active_peaks])[::-1]]
        
        # Limit to top few peaks for efficiency
        top_peaks = peak_order[:min(5, n_peaks)]
        
        for k_star in top_peaks:
            # Valid range for i in this peak
            i_min = max(0, k_star - n + 1)
            i_max = min(n - 1, k_star)
            if i_min > i_max:
                continue
            
            i_range = np.arange(i_min, i_max + 1)
            partners = k_star - i_range
            
            # Contribution of each position i to this peak
            contrib = a[i_range] * a[partners]
            
            # Get top contributors
            n_top = min(20, len(i_range))
            top_idx = np.argsort(contrib)[::-1][:n_top]
            
            for ti in top_idx:
                i = i_range[ti]
                partner_i = partners[ti]
                a_i = a[i]
                if a_i < 1e-10:
                    continue
                
                # Find good destinations: positions j where for ALL top peaks,
                # a[peak - j] is small (so moving mass there doesn't create new peaks)
                # Simplification: find j where a[k_star - j] is small
                # and j is not i
                
                # Score for destination j: want a[peak - j] to be small for all active peaks
                # For efficiency, precompute the "danger" of each position
                
                # Try a few candidate destinations
                # Cold positions: those with low "danger" score
                # danger(j) = max over active peaks k of a[|k - j|] (where valid)
                
                # For the current peak k_star, we want a[k_star - j] to be small
                # and j should be such that it doesn't contribute much to other peaks
                
                # Simple heuristic: try positions far from i and from peak partners
                # Or: try positions where a is small
                
                n_dest_tries = 5
                # Sort all positions by a[k_star - j] (ascending) - these are "cold" for this peak
                # But also need to check other peaks
                
                # For efficiency, precompute a "coldness" score for each position
                # cold(j) = -max over top_peaks of (a[peak-j] if valid, else 0)
                
                cold_scores = np.zeros(n)
                for kp in top_peaks:
                    j_min = max(0, kp - n + 1)
                    j_max = min(n - 1, kp)
                    if j_min <= j_max:
                        partners_j = kp - np.arange(j_min, j_max + 1)
                        # a[partners_j] is the contribution if we put mass at j
                        # We want this to be small
                        vals = a[partners_j]
                        cold_scores[j_min:j_max+1] = np.maximum(cold_scores[j_min:j_max+1], vals)
                
                # Sort by coldness (ascending = colder)
                # Exclude i itself
                cold_scores[i] = np.inf
                sorted_cold = np.argsort(cold_scores)
                
                # Try the coldest destinations
                for di in range(min(n_dest_tries, n)):
                    j = sorted_cold[di]
                    if j == i:
                        continue
                    
                    # Try different step sizes
                    for frac in [0.01, 0.05, 0.1, 0.2, 0.3]:
                        d = a_i * frac
                        if d <= 0 or a[i] - d < -1e-15:
                            continue
                        
                        a[i] -= d
                        a[j] += d
                        a = np.maximum(a, 0)
                        a = a / a.sum()
                        
                        new_c = fast_autocorr(a)
                        new_max = new_c.max()
                        
                        if new_max < c_max:
                            c = new_c
                            c_max = new_max
                            score = 2.0 * n * c_max
                            if score < best_score:
                                best_score = score
                                best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                        else:
                            # Revert
                            a[i] += d
                            a[j] -= d
                        if improved:
                            break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall > max_stall:
                break
            
            # Random perturbation to escape local minima
            if stall % 100 == 0:
                n_trials = 10
                for _ in range(n_trials):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a[i] > 1e-8:
                        frac = rng.uniform(0.005, 0.1)
                        d = a[i] * frac
                        a[i] -= d
                        a[j] += d
                        a = np.maximum(a, 0)
                        a = a / a.sum()
                        new_c = fast_autocorr(a)
                        if new_c.max() < c_max:
                            c = new_c
                            c_max = new_c.max()
                            stall = 0
                        else:
                            # Revert
                            a[i] += d
                            a[j] -= d
                        if stall == 0:
                            break
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
