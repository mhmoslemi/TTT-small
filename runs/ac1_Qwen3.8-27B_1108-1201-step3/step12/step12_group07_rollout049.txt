def fast_convolve(a):

    return irfft(A * A, fft_size)[:size]

def softmax_from_z(z):

        else:

            a_new = a_mult
            sc_new = sc_mult

            best_patience_score = sc_new
            patience = 0
        else:

def lse_objective_grad(z_in, T_val, n):
    """LSE objective and gradient in z-space."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700.0
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    # Gradient: d/dz_i = a_i * (g_i - sum_j a_j g_j)
    # where g_i = sum_k w_k * a_{k-i} (convolution of w with a at position i)
    # g = convolve(w, a) truncated to length n
    w_padded = w[:n]  # use first n values as approximation
    g = fast_convolve(a)  # actually we need conv of w and a
    # g_i = sum_k w_k * a_{k-i} for valid k-i
    # This is the cross-correlation, let's compute it directly
    g = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += w[k] * a[k - idx]

    S = np.dot(g, a)
    grad_z = a * (g - S)
    return obj, grad_z

                         'gtol': 1e-16, 'maxfun': iters * 200,
                         'disp': False}
            )
            z_new = res.x
            a_new = softmax_from_z(z_new)
            a_new = np.maximum(a_new, 0)
            sn = a_new.sum()
            if sn > 1e-15:
                a_new = a_new / sn
            sc = score_normalized(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            z = z_new
            if (si + 1) % 5 == 0:
                print(f"    LSE stage={si+1}/{n_stages} T={stage_T:.4f}: score={best_score:.8f}")
        except Exception:
            pass

    return best_a, best_score


def shake_and_bake(a0, time_limit, solver, n_shakes=5, shake_scale=0.05):
    """Perturb the solution and re-optimize. Returns best found."""
    n = len(a0)
    best_a = a0.copy()
    best_score = score_normalized(a0)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for shake_i in range(n_shakes):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 30:
            break

        # Generate perturbation
        scale = shake_scale * (1.0 + 0.5 * shake_i)
        noise = rng.standard_normal(n) * scale
        z = z_from_a(best_a) + noise
        a_perturbed = softmax_from_z(z)
        a_perturbed = np.maximum(a_perturbed, 0)
        sp = a_perturbed.sum()
        if sp > 1e-15:
            a_perturbed = a_perturbed / sp

        sc_pert = score_normalized(a_perturbed)
        print(f"    Shake {shake_i+1}: perturbed score={sc_pert:.8f} (best={best_score:.8f})")

        # Re-optimize from perturbed point with short budget
        reopt_time = min(120.0, remaining * 0.6)
        if reopt_time < 10:
            continue

        a_out, _ = geometric_nesterov_lp(a_perturbed, reopt_time, solver=solver, max_iters=100000)
        sc_out = score_normalized(a_out)
        if sc_out < best_score:
            best_score = sc_out
            best_a = a_out.copy()
            print(f"    Shake {shake_i+1}: NEW BEST={best_score:.8f}")

    return best_a, best_score


def adaptive_subgradient(a0, time_limit, max_steps=200000):
    """Adaptive subgradient method with Polyak-like step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = 0.5
    avg_f = best_score
    g_avg = np.zeros(n)
    alpha = 0.1  # EMA rate for gradient averaging

    # Lower bound estimate (from theory: >= 2n/(2n-1) ~ 1, use 1.4 as practical)
    f_lb = 1.4

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        f_val = 2.0 * n * max_b

        # Threshold for "active" peaks
        threshold = max_b * 0.9
        topk = np.where(b >= threshold)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])

        # Subgradient in z-space
        grad_a = np.zeros(n)
        for k in topk:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * a[k - idx]
        grad_a /= len(topk)

        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)

        # Polyak step size
        if f_val > f_lb + 1e-10:
            eta_polyak = (f_val - f_lb) / (np.linalg.norm(grad_z, ord=np.inf) ** 2 + 1e-10)
            eta = min(eta_polyak, 1.0)
        else:
            eta = 0.01

        # Update
        z = z_from_a(a) - eta * grad_z
        a_new = softmax_from_z(z)

        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()

        # Exponential moving average of gradient for smoothing
        g_avg = (1 - alpha) * g_avg + alpha * grad_z
        a = a_new

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def diverse_init(n, rng, variant):
    """Generate diverse structured initializations."""
    x = np.linspace(0, 1, n)
    if variant == 0:
        a = 1.0 + 0.5 * np.cos(2 * np.pi * x)
    elif variant == 1:
        a = np.exp(-2.0 * (x - 0.5) ** 2) * 3.0 + 0.1
    elif variant == 2:
        a = 1.0 / (1.0 + 10.0 * x * (1 - x))
    elif variant == 3:
        a = 0.5 + 0.5 * np.sin(np.pi * x) ** 2
    elif variant == 4:
        a = 1.0 + 0.3 * np.cos(4 * np.pi * x) + 0.2 * np.cos(8 * np.pi * x)
    elif variant == 5:
        # Two-scale structure
        a = 1.0 + 0.4 * np.cos(2 * np.pi * x) + 0.2 * np.cos(20 * np.pi * x)
    elif variant == 6:
        # Edge-enhanced
        a = 1.0 + 1.5 * (x * (1 - x)) ** 0.3
    else:
        a = 1.0 + 0.5 * np.random.default_rng(variant).standard_normal(n)
    a = np.maximum(a, 0.01)
    s = a.sum()
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    global _global_deadline
    rng = np.random.default_rng(seed)
    _global_deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    n_base = 1200
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = diverse_init(n_base, rng, 0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    def try_update(arr, label):
        nonlocal best_seq, best_score
        if arr is None:
            return
        arr = np.maximum(arr, 0).astype(np.float64)
        s = arr.sum()
        if s < 0.01:
            return
        arr = arr / s
        sc = evaluate_safe(arr)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = arr.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(arr)})")

    # === Phase 1: Geometric Nesterov LP (400s) ===
    print(f"\n=== Phase 1: Geometric Nesterov LP at n={n_base} (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t1 = min(400.0, remaining * 0.3)
    if t1 > 30:
        a_out, _ = geometric_nesterov_lp(best_seq, t1, solver=solver, max_iters=300000)
        try_update(a_out, "geo_nesterov_p1")
    print(f"  Phase 1 done: best={best_score:.8f}")

    # === Phase 2: LSE L-BFGS polish (250s) ===
    print(f"\n=== Phase 2: LSE L-BFGS polish (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t2 = min(250.0, remaining * 0.2)
    if t2 > 20:
        a_lse = lse_lbfgs_polish(best_seq, t2, T_start=40.0, T_end=0.01, n_stages=25)
        try_update(a_lse, "lse_polish_p2")
    print(f"  Phase 2 done: best={best_score:.8f}")

    # === Phase 3: Shake and bake (350s) ===
    print(f"\n=== Phase 3: Shake and bake (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t3 = min(350.0, remaining * 0.25)
    if t3 > 60:
        a_shake, _ = shake_and_bake(best_seq, t3, solver, n_shakes=4, shake_scale=0.03)
        try_update(a_shake, "shake_bake_p3")
    print(f"  Phase 3 done: best={best_score:.8f}")

    # === Phase 4: Second geometric Nesterov LP (250s) ===
    print(f"\n=== Phase 4: Second Geometric Nesterov LP (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t4 = min(250.0, remaining * 0.2)
    if t4 > 20:
        a_out2, _ = geometric_nesterov_lp(best_seq, t4, solver=solver, max_iters=200000)
        try_update(a_out2, "geo_nesterov_p4")
    print(f"  Phase 4 done: best={best_score:.8f}")

    # === Phase 5: Diverse starts at different lengths (200s) ===
    print(f"\n=== Phase 5: Diverse starts / length exploration (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    if remaining > 60:
        n_current = len(best_seq)
        # Try a few different lengths and diverse initializations
        candidates = [
            (n_current, "same_len"),
            (n_current + 100, "n+100"),
            (n_current - 100, "n-100"),
            (n_current + 200, "n+200"),
            (800, "n800"),
            (2000, "n2000"),
        ]
        tried = set()
        for n_try, label in candidates:
            remaining = _global_deadline - time.monotonic()
            if remaining < 30:
                break
            if n_try in tried or n_try < 100:
                continue
            tried.add(n_try)
            t5 = min(50.0, remaining * 0.4)
            if t5 < 10:
                break
            print(f"  Trying n={n_try} ({label}), budget={t5:.0f}s")
            # Interpolate or initialize
            if n_try == n_current:
                a_init = best_seq.copy()
            else:
                a_init = interpolate_sequence(best_seq, n_try)
            a_try, _ = geometric_nesterov_lp(a_init, t5, solver=solver, max_iters=50000)
            try_update(a_try, f"diverse_{label}")
    print(f"  Phase 5 done: best={best_score:.8f}")

    # === Phase 6: Final LSE polish + adaptive subgradient (150s) ===
    print(f"\n=== Phase 6: Final polish (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    if remaining > 30:
        t6a = min(80.0, remaining * 0.5)
        if t6a > 10:
            a_lse2 = lse_lbfgs_polish(best_seq, t6a, T_start=20.0, T_end=0.005, n_stages=20)
            try_update(a_lse2, "lse_final")
        remaining = _global_deadline - time.monotonic()
        if remaining > 15:
            t6b = min(70.0, remaining * 0.5)
            if t6b > 10:
                a_sg, _ = adaptive_subgradient(best_seq, t6b, max_steps=100000)
                try_update(a_sg, "subgrad_final")
    print(f"  Phase 6 done: best={best_score:.8f}")

    # === Phase 7: One more shake if time remains ===
    remaining = _global_deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 7: Final shake (score={best_score:.8f}) ===")
        a_shake2, _ = shake_and_bake(best_seq, min(80.0, remaining * 0.7), solver,
                                      n_shakes=2, shake_scale=0.02)
        try_update(a_shake2, "shake_final")
        print(f"  Phase 7 done: best={best_score:.8f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
