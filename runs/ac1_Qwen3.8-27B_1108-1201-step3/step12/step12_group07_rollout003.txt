        a = 1.0 + 0.5 * np.sin(2 * np.pi * x) + 0.3 * np.sin(4 * np.pi * x + 0.5)
    elif choice == 1:
        a = np.exp(-3.0 * (x - 0.5) ** 2) + 0.3
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * (x + 0.15)) + 0.2 * np.cos(6 * np.pi * x)
    elif choice == 3:
        a = (1.0 + 0.4 * np.sin(2 * np.pi * 3 * x + 1.0)) ** 2
    elif choice == 4:
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2) + 0.1
    elif choice == 5:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 7 * x) * np.exp(-2 * (x - 0.5)**2)
    elif choice == 6:
        a = 0.8 + 0.4 * np.cos(2 * np.pi * 2 * x + 0.3)
    elif choice == 7:
        a = (x ** 0.3) * ((1 - x) ** 0.3) + 0.1
    elif choice == 8:
        a = 1.0 + 0.25 * np.random.default_rng(choice).standard_normal(n)
    else:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 5 * x + 0.7) + 0.15 * np.cos(2 * np.pi * 11 * x)
    a = np.maximum(a, 0.01)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def lse_grad(a, T_val):
    """Compute gradient of log-sum-exp approximation of max convolution."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-20:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    obj = (2.0 * n / T_val) * log_sum_exp
    return obj, grad_z


def subgradient_nesterov_multi(a0, time_limit, max_steps=200000):
    """Subgradient Nesterov with multiple restarts at different temperatures."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a)
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    eta = 0.5
    last_improve = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve > 8000 and t_nest > 3:
            t_nest = 1.0
            eta = max(eta * 0.7, 0.05)

        T_val = max(5.0, 100.0 / (1.0 + step * 0.001))
        _, grad_z = lse_grad(a, T_val)

        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            z_extrap = z + momentum * (z - z_from_a(a_prev))
        else:
            z_extrap = z

        z_new = z_extrap - eta * grad_z
        a_new = softmax_from_z(z_new)

        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a_prev = a
            a = a_new
            z = z_new
            t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
            last_improve = step
        else:
            if sc > best_score * 1.002:
                t_nest = 1.0
                eta *= 0.92
            else:
                t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
                z = z_new
                a = a_new
            eta = max(eta * 0.9995, 0.02)

    return best_a, best_score


def hybrid_lp_gradient_nesterov(a0, time_limit, solver=None, lp_every=5, max_iters=1000000):
    """Hybrid: solve LP every `lp_every` steps, use gradient between."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a)
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    eta = 0.3
    last_improve = 0
    b_lp = a.copy()
    lp_count = 0
    restart_count = 0
    last_print = t0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if (it - last_improve) > 20000 and t_nest > 5:
            t_nest = 1.0
            eta = max(eta * 0.8, 0.05)
            restart_count += 1

        # Every lp_every steps, solve LP for best response
        if it % lp_every == 0:
            y = a
            if t_nest > 1.0:
                momentum = (t_nest - 1.0) / t_nest
                y_extrap = a + momentum * (a - a_prev)
                y_extrap = np.maximum(y_extrap, 0)
                sy = y_extrap.sum()
                if sy > 1e-15:
                    y = y_extrap / sy
            b_new, M_lp = solve_lp_best_response(y, solver)
            if b_new is not None:
                b_lp = b_new
                lp_count += 1

        # Compute gradient direction (blend LP direction with subgradient)
        T_val = max(10.0, 200.0 / (1.0 + it * 0.0005))
        _, grad_z = lse_grad(a, T_val)

        # Nesterov extrapolation
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            z_extrap = z + momentum * (z - z_from_a(a_prev))
        else:
            z_extrap = z

        # Take step: blend gradient and LP direction
        # Convert LP direction to z-gradient equivalent
        S = np.dot(a, b_lp)
        lp_grad_z = a * (2.0 * b_lp - S)

        # Blend
        blend = min(1.0, lp_count * 0.1)
        combined_grad = (1.0 - blend) * grad_z + blend * lp_grad_z

        z_new = z_extrap - eta * combined_grad
        a_new = softmax_from_z(z_new)

        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a_prev = a
            a = a_new
            z = z_new
            t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
            last_improve = it
            eta = min(eta * 1.001, 0.8)
        else:
            if sc > best_score * 1.001:
                t_nest = 1.0
                eta *= 0.93
            else:
                t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
                z = z_new
                a = a_new
            eta = max(eta * 0.999, 0.02)

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 20:
                print(f"    Hybrid iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} t={t_nest:.1f} lps={lp_count} restarts={restart_count}")
                last_print = elapsed

    return best_a, best_score


def nesterov_lp_pure(a0, time_limit, solver=None, max_iters=1000000):
    """Pure LP-based Nesterov with periodic perturbations."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    last_improve = 0
    restart_count = 0
    last_print = t0
    rng = np.random.default_rng(123)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if (it - last_improve) > 30000 and t_nest > 5:
            # Perturb to escape local minimum
            pert = 0.01 * rng.standard_normal(n)
            a_pert = a + pert
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            sc_pert = score_normalized(a_pert)
            if sc_pert < best_score * 1.01:
                a = a_pert
                a_prev = a_pert
                t_nest = 1.0
            else:
                t_nest = 1.0
            restart_count += 1

        # Extrapolation
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y = y / sy
        else:
            y = a

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha = 2.0 / (t_new + 1.0)

        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        sc_new = score_normalized(a_new)

        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            last_improve = it

        sc_cur = score_normalized(a)
        if sc_new > sc_cur * 1.0003 and t_nest > 5.0:
            t_nest = 1.0
            a_prev = a_new.copy()
            restart_count += 1
        else:
            a_prev = a.copy()
            t_nest = t_new

        a = a_new

        if (it + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 20:
                print(f"    LPNesterov iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} t={t_nest:.1f} restarts={restart_count}")
                last_print = elapsed

    return best_a, best_score


def coordinate_descent(a0, time_limit, max_passes=50):
    """Cyclic coordinate descent with adaptive steps."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(456)

    for p in range(max_passes):
        if time.monotonic() - t0 > time_limit:
            break
        if p > 0 and p % 10 == 0 and best_score < 1.503:
            break

        improved = False
        order = np.arange(n)
        if p > 5:
            rng.shuffle(order)

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[i]
            if old_val < 1e-10:
                # Try small positive value
                candidates = [old_val, 1e-4 / n, 5e-4 / n]
            else:
                # Try relative perturbations
                candidates = [old_val]
                for frac in [0.5, 0.75, 1.25, 1.5, 0.25, 2.0]:
                    candidates.append(old_val * frac)
                if p < 10:
                    candidates.append(old_val * (1.0 + 0.01 * rng.standard_normal()))

            best_c = old_val
            best_sc = score_normalized(a)

            for c in candidates:
                if c < 0:
                    c = 0.0
                a[i] = c
                s_new = a.sum()
                if s_new < 0.01:
                    a[i] = old_val
                    continue
                a_norm = a / s_new
                sc = score_normalized(a_norm)
                if sc < best_sc:
                    best_sc = sc
                    best_c = c
                a[i] = old_val

            if best_c != old_val:
                a[i] = best_c
                a = a / a.sum()
                improved = True
                if best_sc < best_score:
                    best_score = best_sc
                    best_a = a.copy()

        if not improved and p > 3:
            break

        if (p + 1) % 10 == 0:
            print(f"    CoordDescent pass={p+1}: score={best_score:.8f}")

    return best_a, best_score


def lse_lbgfs_polish(a0, time_limit, T_start=80.0, T_end=0.02, n_stages=20):
    """LSE L-BFGS with progressive temperature."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func_staged(z_val):
            obj, _ = lse_grad(z_val, stage_T)
            return obj

        def jac_staged(z_val):
            _, grad = lse_grad(z_val, stage_T)
            return grad

        iters = 30 if si < n_stages // 2 else 15
        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-22,
                         'gtol': 1e-18, 'maxfun': iters * 200,
                         'disp': False}
            )
            z_new = res.x
            a_new = softmax_from_z(z_new)
            sc = score_normalized(a_new)
            z = z_new
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            if (si + 1) % 5 == 0:
                print(f"    LSE stage={si+1}/{n_stages} T={stage_T:.3f}: score={best_score:.8f}")
        except Exception:
            pass

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def peak_targeted_refine(a0, time_limit, max_steps=30000):
    """Multiplicative weights targeting top peaks with adaptive threshold."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = 0.03
    last_improve = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve > 3000:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.90
        topk = np.where(b >= threshold)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])

        grad = np.zeros(n)
        for k in topk:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += a[k - idx]
        grad /= len(topk)

        exponent = -eta * grad
        exponent -= exponent.max()
        new_a = a * np.exp(exponent)
        sa = new_a.sum()
        if sa < 1e-15:
            eta *= 0.5
            continue
        new_a = new_a / sa

        sc = score_normalized(new_a)
        if sc < best_score:
            best_score = sc
            best_a = new_a.copy()
            a = new_a
            last_improve = step
            eta = min(eta * 1.003, 0.1)
        else:
            eta *= 0.9
            if eta < 1e-10:
                break

    return best_a, best_score
