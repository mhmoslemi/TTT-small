def project_simplex(v, target_sum=1.0):
    """Project v onto {x: x >= 0, sum(x) = target_sum}."""
    n = len(v)

    # x_i = max(v_i - tau, 0), find tau such that sum = target_sum
    # Use the sorting-based method

    cssv = np.cumsum(u)

from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(arr):
    """Safely evaluate a sequence."""

def project_simplex(v, target_sum=1.0):
    """Project v onto {x: x >= 0, sum(x) = target_sum} in O(n log n)."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv[-1] - target_sum)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    tau = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - tau, 0.0)


def cross_corr(a, w):
    """
    Compute g[i] = sum_k w[k] * a[k-i] for i=0..n-1 via FFT.
    w has length 2n-1, a has length n. Both non-negative.
    """
    n = len(a)
    N = 1
    while N < 3 * n:
        N *= 2
    W = rfft(w.astype(np.float64), N)
    A = rfft(a.astype(np.float64), N)
    C = irfft(W * np.conj(A), N)
    return C[:n]


def pnorm_obj_grad(a, p):
    """
    Objective and gradient of f_p(a) = (sum b_k^p)^(1/p) where b = a*a.
    Returns (obj, grad) where grad has length n.
    """
    n = len(a)
    b = fast_autocorr(a)
    if p == 2:
        obj = np.sqrt(np.sum(b * b))
        w = b
    else:
        bp = np.power(b, p)
        obj = np.sum(bp) ** (1.0 / p)
        w = np.power(b, p - 1)
    g = cross_corr(a, w)
    # Full gradient: 2 * (sum b^p)^(1/p - 1) * g
    if p == 2:
        scale = 2.0 / obj
    else:
        sum_bp = np.sum(np.power(b, p))
        scale = 2.0 * sum_bp ** (1.0 / p - 1.0)
    return obj, scale * g


def nesterov_pnorm(a_init, time_budget, p_start=2, p_end=16, n_stages=6, lr_init=0.02):
    """
    Nesterov-accelerated gradient on p-norm with simplex projection.
    Anneals p from p_start to p_end over n_stages.
    """
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    t_nest = 1.0
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    p_schedule = []
    if p_start == p_end:
        p_schedule = [p_start]
    else:
        p_schedule = np.geomspace(p_start, p_end, n_stages)

    lr = lr_init
    total_iters = 0

    for stage_p in p_schedule:
        if time.monotonic() - t0 > time_budget:
            break
        remaining = time_budget - (time.monotonic() - t0)
        if remaining < 5:
            break

        t_nest = 1.0
        y = a.copy()
        lr = lr_init * max(0.5, min(2.0, 4.0 / stage_p))

        # Estimate iterations we can afford
        budget_this_stage = remaining / max(1, len(p_schedule))
        start_time = time.monotonic()

        while time.monotonic() - start_time < budget_this_stage:
            if time.monotonic() - t0 > time_budget:
                break

            obj, g = pnorm_obj_grad(y, stage_p)

            # Backtracking line search
            step = lr
            f_y = obj
            a_trial = project_simplex(y - step * g)
            b_trial = fast_autocorr(a_trial)
            if stage_p == 2:
                f_trial = np.sqrt(np.sum(b_trial * b_trial))
            else:
                f_trial = np.sum(np.power(b_trial, stage_p)) ** (1.0 / stage_p)

            backtracks = 0
            while f_trial > f_y - 0.5 * step * np.dot(g, a_trial - y) + 1e-30:
                step *= 0.5
                if step < 1e-12:
                    break
                a_trial = project_simplex(y - step * g)
                b_trial = fast_autocorr(a_trial)
                if stage_p == 2:
                    f_trial = np.sqrt(np.sum(b_trial * b_trial))
                else:
                    f_trial = np.sum(np.power(b_trial, stage_p)) ** (1.0 / stage_p)
                backtracks += 1
                if backtracks > 30:
                    break

            # Nesterov update
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
            momentum = (t_nest - 1.0) / t_new
            a_new = a_trial
            y = a_new + momentum * (a_new - a)
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y / s_y
            else:
                y = a_new.copy()

            a = a_new
            t_nest = t_new
            lr = min(step * 1.25, 0.5)
            total_iters += 1

            if total_iters % 30 == 0:
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()

    # Final check
    score = 2.0 * n * fast_autocorr(a).max()
    if score < best_score:
        best_score = score
        best_a = a.copy()

    return best_a, best_score


def subgradient_fw_avg(a_init, time_budget, top_k=5, eps_frac=0.01):
    """
    Subgradient Frank-Wolfe with running average for the true max function.
    At each step: find peak lags, compute subgradient, take FW step.
    Returns the best of all iterates and the running average.
    """
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    t_iter = 1.0
    avg_a = np.zeros(n)
    n_avg = 0

    while time.monotonic() - t0 < time_budget:
        b = fast_autocorr(a)
        b_max = b.max()

        # Find peak lags
        threshold = b_max * (1.0 - eps_frac)
        peak_idx = np.where(b >= threshold)[0]
        if len(peak_idx) == 0:
            peak_idx = np.array([np.argmax(b)])
        # Also always include the global max
        gm = np.argmax(b)
        if gm not in peak_idx:
            peak_idx = np.append(peak_idx, gm)

        # Compute subgradient: g[i] = (1/|K|) * sum_{k in K} 2*a[k-i] (zero-padded)
        g = np.zeros(n)
        for k in peak_idx:
            # a[k-i] for i=0..n-1, valid when 0 <= k-i < n
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            g[lo:hi + 1] += 2.0 * a[k - np.arange(lo, hi + 1)]
        g /= len(peak_idx)

        # Frank-Wolfe: v = e_j where j = argmin g_i
        j_min = np.argmin(g)
        gamma = 2.0 / (t_iter + 2.0)
        a_new = (1.0 - gamma) * a + gamma * 0.0
        a_new[j_min] += gamma

        # Project to simplex (should already be in simplex if a is)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new

        a = a_new
        t_iter += 1
        avg_a += a
        n_avg += 1

        if t_iter % 50 == 0:
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

            avg_check = avg_a / n_avg
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()

    return best_a, best_score


def active_set_polish(a_init, time_budget):
    """
    Identify active lags (at max) and solve the KKT linear system exactly.
    At optimum: for i in support S, sum_{k in K} a[k-i] = lambda (constant).
    We solve this as a linear system with the constraint sum(a) = 1, a >= 0.
    """
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, 2.0 * n * fast_autocorr(a).max()

    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    for outer in range(20):
        if time.monotonic() - t0 > time_budget:
            break

        b = fast_autocorr(a)
        b_max = b.max()

        # Identify active lags (within tolerance of max)
        tol = b_max * 0.001 + 1e-12
        active_k = np.where(b >= b_max - tol)[0]

        # Identify support (where a > threshold)
        support_thresh = max(1e-10, a.max() * 0.01)
        support = np.where(a > support_thresh)[0]

        if len(active_k) < 2 or len(support) < 2:
            break

        # Build system: for each i in support, sum_{k in active_k} a[k-i] = lambda
        # Unknowns: a_i for i in support, and lambda
        # Plus constraint: sum_{i in support} a_i = 1

        m = len(support)
        k_active = active_k

        # Build matrix
        # Row for each i in support: sum_{k in K} a[k-i] = lambda
        # Row for sum constraint: sum a_i = 1
        A_mat = np.zeros((m + 1, m + 1))
        rhs = np.zeros(m + 1)

        for ii, i in enumerate(support):
            for k in k_active:
                # a[k-i] is a variable if k-i is in support
                if 0 <= k - i < n:
                    idx = np.searchsorted(support, k - i)
                    if idx < m and support[idx] == k - i:
                        A_mat[ii, idx] += 1.0
            A_mat[ii, m] = -1.0  # coefficient of lambda
            rhs[ii] = 0.0

        # Sum constraint
        for ii in range(m):
            A_mat[m, ii] = 1.0
        A_mat[m, m] = 0.0
        rhs[m] = 1.0

        # Solve
        try:
            sol = np.linalg.solve(A_mat, rhs)
            a_new = np.zeros(n)
            for ii, i in enumerate(support):
                a_new[i] = max(0.0, sol[ii])
            a_new[sol < 0].sum()  # check for negatives
            if np.any(sol[:m] < -1e-10):
                # Some variables went negative, shrink support
                neg_idx = np.where(sol[:m] < 0)[0]
                support = np.delete(support, neg_idx)
                a[support_thresh:]  # reset
                continue
            s_new = a_new.sum()
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
        except np.linalg.LinAlgError:
            # System is singular, try least squares
            try:
                sol, residuals, rank, sv = np.linalg.lstsq(A_mat, rhs, rcond=None)
                a_new = np.zeros(n)
                for ii, i in enumerate(support):
                    a_new[i] = max(0.0, sol[ii])
                s_new = a_new.sum()
                if s_new < 1e-10:
                    continue
                a_new = a_new / s_new
            except Exception:
                break

        score_new = 2.0 * n * fast_autocorr(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
        else:
            # No improvement, tighten tolerance and retry
            tol *= 0.5
            if tol < 1e-15:
                break

    return best_a, best_score


def spectral_shaping(a, n_iter=300, alpha=0.25):
    """Iteratively flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.05, 20.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def peak_shaving(a_init, time_budget, seed=123):
    """Coordinate-descent peak shaving for final micro-optimization."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_budget:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:6]
        improved = False

        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            src = idx[np.argmax(contrib)]

            # Find good destination
            dst_candidates = [j for j in range(n) if j != src and a[j] < a.max() * 0.3]
            if not dst_candidates:
                dst_candidates = [j for j in range(n) if j != src]
            dst_candidates_sorted = sorted(dst_candidates, key=lambda j: a[j])
            dst = dst_candidates_sorted[rng.integers(min(len(dst_candidates_sorted), 50))]

            for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0

            if stall > 200:
                break

    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

            t2b = min(rem2 * 0.20, 200.0)
            if t2b > 20:
                a_out, _ = multi_start_nesterov(best_seq, t2b, n_starts=2, p_range=(4, 32))
                try_update(a_out, f"multi_start_n{n_base}")

    # === Phase 3: Subgradient FW with averaging ===
    rem = remaining()
    print(f"\n=== Phase 3: Subgradient FW (rem={rem:.0f}s) ===")
    t3 = min(rem * 0.15, 180.0)
    if t3 > 15:
        a_out, _ = subgradient_fw_avg(best_seq, t3, top_k=5, eps_frac=0.005)
        try_update(a_out, f"subgrad_fw_n{n_base}")

    # === Phase 4: Active set KKT polishing ===
    rem = remaining()
    print(f"\n=== Phase 4: Active set KKT (rem={rem:.0f}s) ===")
    t4 = min(rem * 0.08, 80.0)
    if t4 > 10:
        a_out, _ = active_set_polish(best_seq, t4)
        try_update(a_out, f"kkt_n{n_base}")

    # === Phase 5: Peak shaving ===
    rem = remaining()
    print(f"\n=== Phase 5: Peak shaving (rem={rem:.0f}s) ===")
    t5 = min(rem * 0.12, 120.0)
    if t5 > 10:
        a_out, _ = peak_shaving(best_seq, t5, seed=777)
        try_update(a_out, f"peak1_n{n_base}")

    # === Phase 6: Try nearby lengths ===
    rem = remaining()
    print(f"\n=== Phase 6: Length exploration (rem={rem:.0f}s) ===")

    # Try n-200
    n_short = max(400, n_base - 200)
    if remaining() > 120:
        t6 = min(remaining() * 0.15, 120.0)
        if t6 > 20:
            init_short = interpolate_sequence(best_seq, n_short)
            a_s = spectral_shaping(init_short, 100, 0.2)
            a_out, _ = nesterov_pnorm(a_s, t6 * 0.8, p_start=2, p_end=8, n_stages=4, lr_init=0.02)
            try_update(a_out, f"n{n_short}")

    # Try n+200
    n_long = n_base + 200
    if remaining() > 120:
        t6b = min(remaining() * 0.15, 120.0)
        if t6b > 20:
            init_long = interpolate_sequence(best_seq, n_long)
            a_l = spectral_shaping(init_long, 100, 0.2)
            a_out, _ = nesterov_pnorm(a_l, t6b * 0.8, p_start=2, p_end=8, n_stages=4, lr_init=0.015)
            try_update(a_out, f"n{n_long}")

    # === Phase 7: Final combined polish ===
    rem = remaining()
    print(f"\n=== Phase 7: Final polish (rem={rem:.0f}s) ===")

    # Spectral + Nesterov + Peak on current best
    if remaining() > 30:
        t7 = min(remaining() * 0.25, 120.0)
        if t7 > 10:
            a_s = spectral_shaping(best_seq, 100, 0.1)
            a_out, _ = nesterov_pnorm(a_s, t7 * 0.6, p_start=4, p_end=16, n_stages=3, lr_init=0.005)
            try_update(a_out, "final_nest")

    if remaining() > 20:
        t7b = min(remaining() * 0.20, 80.0)
        if t7b > 5:
            a_out, _ = peak_shaving(best_seq, t7b, seed=999)
            try_update(a_out, "final_peak")

    # === Phase 8: One more active set + peak ===
    if remaining() > 20:
        t8 = min(remaining() * 0.15, 50.0)
        if t8 > 5:
            a_out, _ = active_set_polish(best_seq, t8)
            try_update(a_out, "final_kkt")

    if remaining() > 15:
        t8b = min(remaining() * 0.15, 40.0)
        if t8b > 5:
            a_out, _ = peak_shaving(best_seq, t8b, seed=444)
            try_update(a_out, "final_peak2")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
            best_seq = np.maximum(best_seq, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)}, time={elapsed():.0f}s)")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
