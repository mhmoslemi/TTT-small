"""
LP-based iterative optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovation: Uses exact LP subgradient (max sum(g) s.t. f*g <= M, g>=0)
via HiGHS solver, with FFT-based line search. Multi-phase: LP iterations ->
LSE polish -> multi-length exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
from scipy.optimize import linprog, minimize
import time


def fast_conv(a, b=None):
    """Linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    if b is None:
        b = a
    else:
        b = np.asarray(b, dtype=np.float64)
    n, m = len(a), len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def fast_score(a):
    """Compute 2n*max(b)/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return 2.0 * n * b.max() / (s * s)


def evaluate_safe(a):

def build_lb_matrix(f, n):
    """Build sparse constraint matrix A where A[k,j] = f[k-j] if 0<=k-j<n."""
    nrows = 2 * n - 1
    data = []
    row_idx = []
    col_idx = []
    for k in range(nrows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            js = np.arange(j_start, j_end + 1)
            data.extend(f[k - js].tolist())
            row_idx.extend([k] * len(js))
            col_idx.extend(js.tolist())
    return csr_matrix((data, (row_idx, col_idx)), shape=(nrows, n))


def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. A@g <= M*ones, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = build_lb_matrix(f, n)
    b_ub = np.ones(2 * n - 1) * M
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10.0})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def line_search_t(c0, c1, c2):
    """Find t in [0,1] minimizing max_k (1-t)^2*c0[k] + 2t(1-t)*c1[k] + t^2*c2[k]."""
    def phi(t):
        val = (1.0 - t) ** 2 * c0 + 2.0 * t * (1.0 - t) * c1 + t ** 2 * c2
        return val.max()

    a, b = 0.0, 1.0
    for _ in range(50):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        if phi(m1) < phi(m2):
            b = m2
        else:
            a = m1
    return (a + b) / 2.0


def lp_optimize(f_init, time_limit, max_iter=None, verbose=True):
    """LP-based optimization: iterate LP direction + line search."""
    n = len(f_init)
    f = np.maximum(np.asarray(f_init, dtype=np.float64), 0).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_f = f.copy()
    best_score = fast_score(f)
    t0 = time.monotonic()
    M = fast_conv(f).max()
    iteration = 0
    stall = 0
    rng = np.random.default_rng(12345)

    if max_iter is None:
        max_iter = 100000

    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break

        g = solve_lp_direction(f, M)

        if g is None or g.sum() < 1e-12:
            stall += 1
            if stall > 30:
                f = f + 0.015 * rng.standard_normal(n)
                f = np.maximum(f, 0)
                f = f / f.sum()
                M = fast_conv(f).max()
                stall = 0
            continue

        g = g / g.sum()

        c0 = fast_conv(f)
        c1 = fast_conv(f, g)
        c2 = fast_conv(g)

        t_opt = line_search_t(c0, c1, c2)
        t_opt = min(t_opt, 1.0)

        f_new = (1.0 - t_opt) * f + t_opt * g
        f_new = np.maximum(f_new, 0)
        s_new = f_new.sum()
        if s_new < 1e-15:
            f_new = f
        else:
            f_new = f_new / s_new

        M_new = fast_conv(f_new).max()
        score_new = 2.0 * n * M_new

        if score_new < best_score - 1e-14:
            f = f_new
            M = M_new
            best_score = score_new
            best_f = f.copy()
            stall = 0
            iteration += 1
        else:
            improved = False
            for t_try in [0.5 * t_opt, 0.3 * t_opt, 0.1, 0.05]:
                if t_try <= 0:
                    continue
                f_try = (1.0 - t_try) * f + t_try * g
                f_try = np.maximum(f_try, 0)
                s_try = f_try.sum()
                if s_try < 1e-15:
                    continue
                f_try = f_try / s_try
                sc_try = 2.0 * n * fast_conv(f_try).max()
                if sc_try < best_score:
                    f = f_try
                    M = fast_conv(f).max()
                    best_score = sc_try
                    best_f = f.copy()
                    stall = 0
                    improved = True
                    break
            if not improved:
                stall += 1
                if stall > 50:
                    f = f + 0.02 * rng.standard_normal(n)
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                    M = fast_conv(f).max()
                    stall = 0

        if verbose and iteration > 0 and iteration % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {iteration} ({elapsed:.0f}s): score={best_score:.8f}, M={M:.6e}, t={t_opt:.4f}")

    return best_f, best_score


def lse_obj_grad(z, T):
    """LSE objective and gradient in z-space where a=softmax(z)."""
    n = len(z)
    z_shift = z - z.max()
    e = np.exp(z_shift)
    a = e / e.sum()
    b = fast_conv(a)
    Tb = T * b
    Tb_shift = Tb - Tb.max()
    exp_Tb = np.exp(Tb_shift)
    obj = (Tb.max() + np.log(exp_Tb.sum())) / T
    w = exp_Tb / exp_Tb.sum()
    # grad_a[k] = 2 * sum_j w[j] * a[j - (k - (n-1))] ... use cross-corr
    # (f*g)[k] = sum_j f[j]*g[k-j], so grad of b[k] wrt a[m] = 2*sum_k w[k]*a[k-(k-m)]
    # Let's compute: grad_a[m] = 2 * sum_k w[k] * ( Contribution of a[m] to b[k] )
    # a[m] contributes to b[k] as a[m]*a[k-m] (if 0<=k-m<n) or a[k-m]*a[m] (if 0<=m-k<n)
    # So a[m] contributes to b[k] with coefficient a[k-m] if 0<=k-m<n, plus a[m-k] if 0<=m-k<n
    # For m = k - (k-m): the contribution is a[k-m] when the other index is in range
    # Simpler: grad_b[k] wrt a[m] = a[k-m] if 0<=k-m<n (from a[m]*a[k-m]) + a[m-k] if 0<=m-k<n
    # = a[k-m] if 0<=k-m<n else 0, PLUS a[m-k] if m>=k (i.e., a[m-k] when m-k in [0,n-1])
    # Since b is symmetric (b[k]=b[2n-2-k]), let's just compute directly:
    # grad_a[m] = 2 * sum_{k: 0<=k-m<n} w[k]*a[k-m] + 2 * sum_{k: 0<=m-k<n, k!=m} w[k]*a[m-k]
    # Actually simpler: (f*g)[k] = sum_j f[j]*g[k-j], so d(b[k])/d(a[m]) = a[k-m] if 0<=k-m<n else 0
    # (treating b as convolution, not autocorrelation)
    # But b = conv(a,a), so d(b[k])/d(a[m]) = a[k-m] (from first a) + a[m-k] (from second a) if valid
    # = a[k-m] if 0<=k-m<n, plus a[m-k] if 0<=m-k<n
    # Since a[m-k] = a[|m-k|] only if symmetric... Let's just use the cross-corr approach.

    # Cross correlation: cc[m] = sum_k w[k] * a[k-m] where 0<=k-m<n
    # This equals (w_rev * a)[m + n - 1] where w_rev is w reversed... 
    # Actually: sum_k w[k]*a[k-m] for k-m in [0,n-1] means k in [m, m+n-1]
    # = sum_{j=0}^{n-1} w[m+j] * a[j] (where w has been extended)
    # This is the cross-correlation of w and a at lag m.
    # Using FFT: if we compute conv(w_extended, a), we get this.
    
    # Simpler approach: use the fact that d(conv(a,a)[k])/d(a[m]) = a[k-m] + a[m-k] (with bounds)
    # = 2*a[k-m] if symmetric, but in general:
    # For the convolution b[k] = sum_{i} a[i]*a[k-i]:
    # db[k]/da[m] = a[k-m] (if 0<=k-m<n) + a[m-k] (if 0<=m-k<n, m!=k to avoid double counting at i=m)
    # Wait, at i=m: a[m]*a[k-m], derivative wrt a[m] gives a[k-m]
    # At i=k-m: a[k-m]*a[m], derivative wrt a[m] gives a[k-m]
    # So if m != k-m (i.e., 2m != k), we get 2*a[k-m]; if 2m==k, we get 2*a[m] (from a[m]^2)
    # Actually db[k]/da[m] = a[k-m] (from the term where first factor is a[m]) + a[k-m] (from where second factor is a[m]) = 2*a[k-m], valid when 0<=k-m<n.
    # But wait: b[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i]*a[k-i]
    # db[k]/da[m] = a[k-m] (if m is in the valid range for i, i.e., max(0,k-n+1)<=m<=min(n-1,k))
    #              + a[k-m] (if k-m is in the valid range, i.e., max(0,k-n+1)<=k-m<=min(n-1,k))
    # The second condition: 0<=k-m<=n-1 AND k-n+1<=k-m<=k, which simplifies to 0<=k-m<=n-1 (since m in [0,n-1] implies k-m in [k-n+1, k])
    # So db[k]/da[m] = 2*a[k-m] if 0<=k-m<n, else 0. (The factor of 2 accounts for both positions.)
    # Wait, that's only if m != k-m. If m = k-m (i.e., k=2m), then the term is a[m]^2 and derivative is 2*a[m].
    # But 2*a[k-m] = 2*a[m] in this case too! So the formula db[k]/da[m] = 2*a[k-m] (if 0<=k-m<n) is always correct.
    
    # So grad_a[m] = 2 * sum_k w[k] * a[k-m] (where 0<=k-m<n)
    # = 2 * sum_{j=0}^{n-1} w[m+j] * a[j]  (where m+j < 2n-1, i.e., j < 2n-1-m)
    # This is a cross-correlation! Let's compute it via FFT.
    
    # w has length 2n-1. We need (w * a)[m] = sum_j w[m+j]*a[j] for j=0..n-1 (with w indexed 0..2n-2)
    # = sum_j a[j]*w[m+j]
    # This is the cross-correlation of a and w, which equals conv(a, w_rev)[m + n - 1] where w_rev = w[::-1]
    # Or: it's the (m-th) element of the "anti-convolution"
    
    # Let's use: cc[m] = sum_{j=0}^{n-1} a[j] * w[m+j]
    # If we define w_padded of length 2n-1, and compute the "sliding dot product"
    # This is equivalent to: for each m, dot(a, w[m:m+n])
    # = cross-correlation of a and w at lag... let me think differently.
    
    # Define: extended_a = [a, 0, 0, ..., 0] of length 3n-2
    # Then conv(extended_a, w)[m + something]... this is getting complicated.
    
    # Simplest: just loop over k for the top weights (since w is concentrated on few peaks)
    # But for generality, let's use the full FFT approach:
    # cc[m] = sum_j a[j] * w[m+j] where j in [0,n-1] and m+j in [0, 2n-2]
    # = sum_j a[j] * w[m+j]
    # Let's define u = w (length 2n-1), v = a (length n)
    # cc[m] = sum_{j=0}^{n-1} v[j] * u[m+j] (with u[i]=0 for i outside [0,2n-2])
    # This is the cross-correlation <v, shift_m(u)> 
    # = conv(v, u_rev)[m + n - 1] where u_rev[i] = u[2n-2-i]
    # Let's verify: conv(v, u_rev)[k] = sum_j v[j] * u_rev[k-j] = sum_j v[j] * u[2n-2-(k-j)] = sum_j v[j] * u[2n-2-k+j]
    # We want this to equal sum_j v[j]*u[m+j], so 2n-2-k+j = m+j, i.e., k = 2n-2-m.
    # So cc[m] = conv(v, u_rev)[2n-2-m]
    
    u_rev = w[::-1].copy()
    conv_len = n + (2 * n - 1) - 1  # = 3n-2
    fft_size = 1 << (conv_len - 1).bit_length()
    V = rfft(a, fft_size)
    U = rfft(u_rev, fft_size)
    conv_result = irfft(V * U, fft_size)[:conv_len]
    # cc[m] = conv_result[2n-2-m]
    indices = 2 * n - 2 - np.arange(n)
    cc = conv_result[indices]
    grad_a = 2.0 * cc

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_optimize(a0, time_limit, T_start=50.0, T_end=0.3, n_stages=10):
    """LSE-smoothed optimization with L-BFGS and annealing."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    a_init = a_init / a_init.sum()
    z = np.log(a_init) + np.log(n)  # z = log(a) + log(n) so softmax(z) = a
    # Actually simpler: z = log(a), and softmax(z) = a / sum(a) = a (since sum=1)
    z = np.log(a_init)

    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = fast_score(a_init)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        iters = max(20, min(100, int(40 + 60 * si / max(1, n_stages))))

        def func(z_in):
            val, _ = lse_obj_grad(z_in, stage_T)
            return val

        def jac(z_in):
            _, g = lse_obj_grad(z_in, stage_T)
            return g

        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': iters, 'ftol': 1e-18,
                                    'gtol': 1e-12, 'maxfun': iters * 300,
                                    'disp': False})
            z = res.x
            z_shift = z - z.max()
            a = np.exp(z_shift)
            a = a / a.sum()
            a = np.maximum(a, 0)
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        except Exception:
            pass

    return best_a, best_score


def mirror_descent(a0, time_limit, eta0=0.05, k_max=12, seed=42):
    """Fast mirror descent with top-k subgradient."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_conv(a)
        b_max = b.max()
        b_mean = b.mean()

        top_idx = np.argsort(b)[::-1][:min(k_max, n)]
        excess = b[top_idx] - b_mean
        if excess.max() < 1e-15:
            weights = np.ones(len(top_idx)) / len(top_idx)
        else:
            excess_shift = excess - excess.max()
            weights = np.exp(excess_shift) / np.exp(excess_shift).sum()

        grad = np.zeros(n)
        for ii in range(len(top_idx)):
            m = int(top_idx[ii])
            w_m = weights[ii] * 2.0
            j_start = max(0, m - n + 1)
            j_end = min(n - 1, m)
            js = np.arange(j_start, j_end + 1)
            grad[js] += w_m * a[m - js]

        eta = eta0 / (1.0 + t / 200.0) ** 0.6
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a + 0.01 * rng.standard_normal(n)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
        a_new = a_new / s_new
        a = a_new

        if t % 10 == 0 or t < 50:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stall = 0
            else:
                stall += 1

        if stall > 300:
            a = a + 0.03 * rng.standard_normal(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            stall = 0

    return best_a, best_score


def interpolate_to_length(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def try_update(best_state, a, label):
    """Try to update best state with new sequence."""
    if a is None:
        return best_state
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return best_state
    a = a / s
    sc = fast_score(a)
    if sc < best_state[1]:
        print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
        return (a.copy(), sc)
    return best_state


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: LP-based optimization with LSE and MD fallbacks.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 40.0, 1750.0)
    rng = np.random.default_rng(seed)

    # Get parent sequence
    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = fast_score(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        best_seq = np.ones(1600) / 1600.0
        best_score = fast_score(best_seq)
        print(f"Default init: n=1600, score={best_score:.8f}")

    n = len(best_seq)
    print(f"Starting LP optimization at n={n}")

    # Phase 1: LP-based optimization (main phase)
    remaining = time_budget - (time.monotonic() - t_global)
    t_lp = min(remaining * 0.55, 950.0)
    if t_lp > 60:
        print(f"\n=== Phase 1: LP Optimization ({t_lp:.0f}s) ===")
        t0 = time.monotonic()
        a_lp, sc_lp = lp_optimize(best_seq, t_lp, verbose=True)
        best_seq, best_score = try_update((best_seq, best_score), a_lp, "LP")
        print(f"  LP done ({time.monotonic()-t0:.0f}s): score={best_score:.8f}")

    # Phase 2: LSE refinement
    remaining = time_budget - (time.monotonic() - t_global)
    t_lse = min(remaining * 0.20, 250.0)
    if t_lse > 30:
        print(f"\n=== Phase 2: LSE Refinement ({t_lse:.0f}s) ===")
        a_lse, sc_lse = lse_optimize(best_seq, t_lse, T_start=30.0, T_end=0.2, n_stages=12)
        best_seq, best_score = try_update((best_seq, best_score), a_lse, "LSE")
        print(f"  LSE done: score={best_score:.8f}")

    # Phase 3: More LP iterations on the refined solution
    remaining = time_budget - (time.monotonic() - t_global)
    t_lp2 = min(remaining * 0.20, 250.0)
    if t_lp2 > 30:
        print(f"\n=== Phase 3: LP Round 2 ({t_lp2:.0f}s) ===")
        a_lp2, sc_lp2 = lp_optimize(best_seq, t_lp2, verbose=True)
        best_seq, best_score = try_update((best_seq, best_score), a_lp2, "LP2")
        print(f"  LP2 done: score={best_score:.8f}")

    # Phase 4: Try slightly different lengths
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 120:
        for n_try in [int(n * 1.1), int(n * 0.9)]:
            if n_try <= 0 or n_try == n:
                continue
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 60:
                break
            t_sub = min(remaining * 0.4, 150.0)
            print(f"\n=== Phase 4: Try n={n_try} ({t_sub:.0f}s) ===")
            init_sub = interpolate_to_length(best_seq, n_try)
            a_sub, sc_sub = lp_optimize(init_sub, t_sub, verbose=True)
            best_seq, best_score = try_update((best_seq, best_score), a_sub, f"LP_n{n_try}")

    # Phase 5: Final LSE polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t_final = min(remaining * 0.5, 100.0)
        print(f"\n=== Phase 5: Final LSE ({t_final:.0f}s) ===")
        a_fin, _ = lse_optimize(best_seq, t_final, T_start=10.0, T_end=0.1, n_stages=8)
        best_seq, best_score = try_update((best_seq, best_score), a_fin, "FinalLSE")

    # Final evaluation
    fs = evaluate_safe(best_seq)
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")

    return [float(x) for x in best_seq]
