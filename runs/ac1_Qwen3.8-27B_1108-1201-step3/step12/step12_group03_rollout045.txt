def softmax_mirror_descent(a0, time_limit, tau_start=50.0, tau_end=500.0, verbose=False):
    """
    Mirror descent on the simplex with soft-max smoothing of the max autocorrelation.
    Uses the entropy mirror map: a_new ∝ a * exp(-lr * grad).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    t0 = time.monotonic()
    best_a = a.copy()
    best_score = score_normalized(a)

    n_iters = max(500, int(time_limit * 30))

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Anneal tau
        progress = it / max(1, n_iters)
        tau = tau_start * (tau_end / tau_start) ** progress

        # Compute autocorrelation
        b = fast_autocorr(a)
        b_max = b.max()

        # Soft-max weights (focus on peaks)
        b_shifted = b - b_max
        weights = np.exp(tau * b_shifted)
        w_sum = weights.sum()
        if w_sum < 1e-300:
            break
        weights /= w_sum

        # Compute gradient: weighted sum of 2*shift_k(a)
        grad = np.zeros(n)
        # Use top peaks for efficiency
        top_m = min(20, 2 * n - 1)
        top_indices = np.argsort(b)[::-1][:top_m]
        top_weights = weights[top_indices]
        w_norm = top_weights.sum()
        if w_norm < 1e-300:
            break
        top_weights /= w_norm

        for w, k in zip(top_weights, top_indices):
            # grad contribution: 2*w*shift_k(a)
            # shift_k(a)_i = a[k-i] if 0 <= k-i < n else 0
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                idx = np.arange(js, je + 1)
                other = k - idx
                grad[idx] += 2.0 * w * a[other]
                # For the symmetric term (when 2i != k, both contribute)
                # Actually the gradient of (a*a)_k w.r.t. a_i is:
                # a[k-i] + a[k-i] = 2*a[k-i] (counting both j=i and k-j=i)
                # But when i = k-i (i.e., 2i=k), it's just 2*a_i
                # The formula grad[idx] += 2*w*a[other] already handles this correctly
                # because when idx[i] = k - idx[i], the term a[other[i]] = a[idx[i]] = a_i
                # and we add 2*w*a_i to grad[idx[i]], which is correct.

        # Mirror descent step with adaptive learning rate
        lr = 0.01 / (1.0 + 0.1 * it / max(1, n_iters))

        # a_new(i) = a(i) * exp(-lr * grad(i)) / Z
        log_a = np.log(np.maximum(a, 1e-300))
        log_a_new = log_a - lr * grad
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new_s = a_new.sum()
        if a_new_s < 1e-300:
            break
        a_new = a_new / a_new_s

        # Also try a mixed iterate
        for frac in [0.5, 0.7, 1.0]:
            a_trial = (1 - frac) * a + frac * a_new
            a_trial = np.maximum(a_trial, 0)
            a_trial_s = a_trial.sum()
            if a_trial_s < 1e-15:
                continue
            a_trial = a_trial / a_trial_s
            sc = score_normalized(a_trial)
            if sc < best_score:
                best_score = sc
                best_a = a_trial.copy()
                a = a_trial.copy()
                break
        else:
            a = a_new

        if verbose and it % 200 == 0:
            print(f"    MD it={it}: best={best_score:.6f}, cur={score_normalized(a):.6f}")

    return best_a, best_score


def peak_shaving_cd(a0, time_limit, seed=42, verbose=False):

        top_ci = np.argsort(contrib)[::-1][:5]

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    f = np.asarray(f, dtype=np.float64)
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def make_initialization(n, init_type='gaussian', sigma=0.2):
    """Create a structured initialization of length n."""
    if init_type == 'uniform':
        return np.ones(n) / n
    elif init_type == 'gaussian':
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.5) ** 2) / (2 * sigma ** 2))
        a = np.maximum(a, 0)
        s = a.sum()
        return a / s if s > 0 else np.ones(n) / n
    elif init_type == 'triangular':
        x = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2 * x - 1)
        a = np.maximum(a, 0)
        s = a.sum()
        return a / s if s > 0 else np.ones(n) / n
    elif init_type == 'raised_cosine':
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = np.maximum(a, 0)
        s = a.sum()
        return a / s if s > 0 else np.ones(n) / n
    elif init_type == 'chirp':
        # Quadratic chirp: frequency increases linearly
        x = np.arange(n, dtype=np.float64)
        phase = 2 * np.pi * (x ** 2) / (2 * n)
        a = np.abs(np.cos(phase)) ** 2
        a = np.maximum(a, 0)
        s = a.sum()
        return a / s if s > 0 else np.ones(n) / n
    elif init_type == 'bump':
        # Sharp bump in the middle
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.5) ** 2) / (2 * 0.05 ** 2))
        a = np.maximum(a, 0)
        s = a.sum()
        return a / s if s > 0 else np.ones(n) / n
    elif init_type == 'bimodal':
        # Two bumps
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.3) ** 2) / (2 * 0.1 ** 2)) + np.exp(-((x - 0.7) ** 2) / (2 * 0.1 ** 2))
        a = np.maximum(a, 0)
        s = a.sum()
        return a / s if s > 0 else np.ones(n) / n
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization with LP Nesterov,
    soft-max polishing, and peak shaving at multiple lengths.
    """
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    print(f"=== Starting optimization with time budget {time_budget:.0f}s ===")

    # Get LP solver
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Try to get parent sequence
    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent sequence: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    # Determine base length
    if parent is not None:
        n_base = len(parent)
    else:
        n_base = 1600
        best_seq = make_initialization(n_base, 'gaussian', 0.3)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Multi-initialization warmup at base length ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: Multi-init warmup at n={n_base} (remaining={remaining:.0f}s) ===")
    init_types = ['gaussian', 'triangular', 'raised_cosine', 'chirp', 'bimodal', 'bump']
    warmup_time = min(remaining * 0.05, 60.0)
    if warmup_time > 10:
        for init_type in init_types:
            if time.monotonic() - t_global_start > time_budget - 300:
                break
            a_init = make_initialization(n_base, init_type, 0.2)
            if parent is not None:
                # Mix with parent for some inits
                a_init = 0.5 * a_init + 0.5 * interpolate_sequence(parent, n_base)
                a_init = np.maximum(a_init, 0)
                a_init = a_init / a_init.sum()
            a_out, sc_out, _ = nesterov_lp_optimize(a_init, warmup_time / len(init_types), solver, verbose=False)
            try_update(a_out, f"warmup_{init_type}")

    # === Phase 2: Main Nesterov LP at base length ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: Main Nesterov LP at n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.40, 600.0)
    if t2 > 60:
        a_out, sc_out, iters = nesterov_lp_optimize(best_seq, t2, solver, verbose=True)
        print(f"  Main NLP: {iters} iters, score={sc_out:.6f}")
        try_update(a_out, "main_nlp")

    # === Phase 3: Soft-max mirror descent polishing ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: Soft-max MD at n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.10, 120.0)
    if t3 > 15:
        a_out, sc_out = softmax_mirror_descent(best_seq, t3, tau_start=30.0, tau_end=300.0)
        print(f"  Soft-max MD: score={sc_out:.6f}")
        try_update(a_out, "softmax_md")

    # === Phase 4: Peak shaving coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: Peak shaving CD at n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.08, 100.0)
    if t4 > 10:
        a_out, sc_out = peak_shaving_cd(best_seq, t4, seed=123)
        print(f"  Peak shaving: score={sc_out:.6f}")
        try_update(a_out, "peak_shave_1")

    # === Phase 5: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: Multi-length exploration (remaining={remaining:.0f}s) ===")
    n_best = len(best_seq)
    length_candidates = [n_best - 200, n_best - 100, n_best + 100, n_best + 200, n_best + 400]
    length_candidates = [nc for nc in length_candidates if 400 <= nc <= 4000]
    # Remove duplicates and the base length
    seen = set()
    unique_lengths = []
    for nc in length_candidates:
        if nc not in seen and nc != n_best:
            seen.add(nc)
            unique_lengths.append(nc)

    for n_try in unique_lengths:
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining < 80:
            break
        print(f"\n  --- Trying n={n_try} (remaining={remaining:.0f}s) ---")
        t5 = min(remaining * 0.15, 150.0)
        if t5 < 30:
            continue
        a_init = interpolate_sequence(best_seq, n_try)
        # Also try a fresh init
        a_init2 = make_initialization(n_try, 'gaussian', 0.3)
        a_mix = 0.7 * a_init + 0.3 * a_init2
        a_mix = np.maximum(a_mix, 0)
        a_mix = a_mix / a_mix.sum()

        a_out, sc_out, iters = nesterov_lp_optimize(a_mix, t5, solver, verbose=False)
        try_update(a_out, f"nlp_n{n_try}")

        # Quick polish
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining > 30:
            t_polish = min(remaining * 0.04, 40.0)
            if t_polish > 5:
                a_ps, sc_ps = peak_shaving_cd(a_out, t_polish, seed=456)
                try_update(a_ps, f"ps_n{n_try}")

    # === Phase 6: Second Nesterov round on global best ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: Nesterov round 2 at n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t6 = min(remaining * 0.20, 250.0)
    if t6 > 30:
        a_out, sc_out, iters = nesterov_lp_optimize(best_seq, t6, solver, verbose=True)
        print(f"  NLP round 2: {iters} iters, score={sc_out:.6f}")
        try_update(a_out, "nlp_round2")

    # === Phase 7: Final polishing ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 7: Final polishing (remaining={remaining:.0f}s) ===")
    if remaining > 20:
        t7a = min(remaining * 0.3, 60.0)
        if t7a > 5:
            a_out, sc_out = softmax_mirror_descent(best_seq, t7a, tau_start=100.0, tau_end=1000.0)
            try_update(a_out, "final_md")

    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 10:
        t7b = min(remaining * 0.5, 60.0)
        if t7b > 5:
            a_out, sc_out = peak_shaving_cd(best_seq, t7b, seed=789)
            try_update(a_out, "final_ps")

    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 10:
        t7c = min(remaining * 0.5, 60.0)
        if t7c > 10:
            a_out, sc_out, iters = nesterov_lp_optimize(best_seq, t7c, solver, verbose=False)
            try_update(a_out, "final_nlp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\n=== FINAL: score={best_score:.6f}, n={len(best_seq)} ===")
    else:
        print("\nERROR: No valid sequence found!")
        best_seq = np.ones(1600) / 1600
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in best_seq]
