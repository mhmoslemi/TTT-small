def solve_lp_cutting_plane(a_fixed, solver, n_constraints=20):
    """Solve LP with cutting plane method."""
    n = len(a_fixed)
    b = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    
    # Build constraint matrix lazily
    T = build_conv_matrix(a_fixed)  # (2n-1) x n
    
    # Start with top constraints
    # First, get the current peaks
    c = fast_autocorr(a_fixed)  # This is conv(a_fixed, a_fixed)
    # But we need T@b, not T@a_fixed. So we need to estimate which constraints are active.
    # Use the dual of the previous solution, or just use a uniform b to get initial peaks.
    
    b_current = np.ones(n) / n
    c_est = T @ b_current  # (2n-1,)
    active = np.argsort(c_est)[::-1][:n_constraints]
    
    constraints = [cp.sum(b) == 1, M >= 0]
    for k in active:
        constraints.append(T[k] @ b <= M)
    
    for _ in range(20):
        prob = cp.Problem(cp.Minimize(M), constraints)
        prob.solve(solver=solver)
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        
        b_val = np.asarray(b.value)
        b_val = np.maximum(b_val, 0)
        b_val = b_val / b_val.sum()
        
        # Check for violated constraints
        c_new = T @ b_val
        max_c = c_new.max()
        M_val = float(M.value)
        
        violated = np.where(c_new > M_val * (1 + 1e-6))[0]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Cutting-plane LP Nesterov (faster than full LP, more iterations)
2. Symmetry enforcement for faster convergence
3. Diverse structured initializations
4. QP active-set refinement
5. Adaptive peak shaving

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def symmetrize(a):
    """Enforce symmetry: a[j] = a[n-1-j]."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    return a_sym


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_cutting_plane(a_fixed, T, solver, n_init_constraints=30, max_rounds=30, b_prev=None):
    """
    Solve LP: min M s.t. T@b <= M*1, sum(b)=1, b>=0
    Using cutting-plane method (start with few constraints, add violated ones).
    """
    n = len(a_fixed)
    b = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    
    # Get initial active set from current peaks or previous solution
    if b_prev is not None:
        c_est = T @ b_prev
    else:
        b_est = np.ones(n) / n
        c_est = T @ b_est
    
    active = np.argsort(c_est)[::-1][:n_init_constraints]
    
    for round_idx in range(max_rounds):
        constraints = [cp.sum(b) == 1, M >= 0]
        for k in active:
            constraints.append(T[k] @ b <= M)
        
        prob = cp.Problem(cp.Minimize(M), constraints)
        try:
            prob.solve(solver=solver)
        except Exception:
            try:
                prob.solve(solver=cp.SCS)
            except Exception:
                try:
                    prob.solve(solver=cp.ECOS)
                except Exception:
                    return None, None
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None, None
        
        b_val = np.asarray(b.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        s = b_val.sum()
        if s < 1e-15:
            return None, None
        b_val = b_val / s
        M_val = float(M.value)
        
        # Check for violated constraints
        c_new = T @ b_val
        max_violation = c_new.max() - M_val
        if max_violation < 1e-8:
            break
        
        # Add most violated constraints
        violated = np.where(c_new > M_val + 1e-10)[0]
        if len(violated) > 0:
            violated_sorted = violated[np.argsort(c_new[violated])[::-1]]
            current_active_set = set(active)
            for v in violated_sorted:
                if v not in current_active_set and len(active) < 2 * n_init_constraints:
                    active = np.append(active, v)
                    current_active_set.add(v)
    
    return b_val, M_val


def nesterov_lp_cutting(a0, time_limit, solver, n_init_constraints=30, enforce_sym=True, verbose=True):
    """
    Nesterov-accelerated LP with cutting plane inner solve.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    if enforce_sym:
        a = symmetrize(a)
        a = a / a.sum()
    
    # Build T matrix
    # T depends on y (the Nesterov iterate), so we need to rebuild it each iteration
    # But that's expensive. Instead, use the current a to build T and update when a changes significantly.
    
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    b_prev = None
    T = build_conv_matrix(y)
    iter_count = 0
    stall = 0
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Rebuild T if y has changed significantly
        # (We rebuild every iteration since T depends on y)
        T = build_conv_matrix(y)
        
        # Solve LP with cutting plane
        b_star, M_val = solve_lp_cutting_plane(y, T, solver, n_init_constraints, b_prev=b_prev)
        if b_star is None:
            continue
        
        if enforce_sym:
            b_star = symmetrize(b_star)
            b_star = b_star / b_star.sum()
        
        b_prev = b_star
        iter_count += 1
        
        # Nesterov schedule
        alpha_t = 2.0 / (t + 2.0)
        
        # Nesterov update
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_new = y_new / y_new.sum()
        
        momentum = (t - 1.0) / (t + 2.0) if t > 1 else 1.0
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        y = y_new
        a = a_new
        
        # Evaluate
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        
        # Also evaluate the running average of y's
        # (Keep a running count)
        
        if t % 200 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"      NLP-CP t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}")
        
        if stall > 5000 and t > 1000:
            if verbose:
                print(f"      Early stop at t={t}, stall={stall}")
            break
    
    return best_a, best_score, iter_count

def proj_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def projected_gradient_max(a0, time_limit, lr=0.1, verbose=True):
    """Projected gradient method on max_k (conv(a,a))_k over the simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(a)
        max_b = b.max()
        k_star = np.argmax(b)
        
        # Gradient: gc[k]/da[m] = 2*a[k-m] if 0<=k-m<n
        grad = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        for j in range(js, je + 1):
            grad[j] += 2 * a[k_star - j]
        
        # Adaptive learning rate
        # Use backtracking line search
        ft = 2.0 * n * max_b
        lr_t = lr
        for backtrack in range(20):
            a_new = proj_simplex(a - lr_t * grad)
            ft_new = 2.0 * n * fast_autocorr(a_new).max()
            if ft_new < ft - 0.5 * (lr_t * 0.5) * np.dot(grad, a - a_new):
                break
            lr_t *= 0.5
        
        a = a_new
        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if it % 500 == 0 and verbose:
            print(f"      PG it={it}: best={best_score:.6f}, cur={score:.6f}")
    
    return best_a, best_score

import numpy as np

from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def symmetrize(a):
    """Enforce symmetry: a[j] = a[n-1-j]."""
    n = len(a)
    return 0.5 * (a + a[::-1])


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:

    return cp.ECOS


def solve_lp_cutting_plane(a_fixed, T, solver, n_init_constraints=40, max_rounds=25, b_prev=None):
    """
    Solve LP: min M s.t. T@b <= M*1, sum(b)=1, b>=0
    Using cutting-plane method for speed.
    """
    n = len(a_fixed)
    b = cp.Variable(n, nonneg=True)
    M = cp.Variable()

    if b_prev is not None:
        c_est = T @ b_prev
    else:
        b_est = np.ones(n) / n
        c_est = T @ b_est

    active = list(np.argsort(c_est)[::-1][:n_init_constraints])

    b_val = None
    M_val = 0.0

    for round_idx in range(max_rounds):
        constraints = [cp.sum(b) == 1, M >= 0]
        for k in active:
            constraints.append(T[k] @ b <= M)

        prob = cp.Problem(cp.Minimize(M), constraints)
        solved = False
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solved = True
        except Exception:
            pass
        if not solved:
            for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
                try:
                    prob.solve(solver=alt)
                    if prob.status in ('optimal', 'optimal_inaccurate'):
                        solved = True
                        break
                except Exception:
                    continue
        if not solved:
            return None, None

        b_val = np.asarray(b.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        s = b_val.sum()
        if s < 1e-15:
            return None, None
        b_val = b_val / s
        M_val = float(M.value)

        # Check for violated constraints
        c_new = T @ b_new_check(b_val, T)
        max_c = c_new.max()
        if max_c <= M_val * (1 + 1e-7) + 1e-10:
            break

        violated = np.where(c_new > M_val * (1 + 1e-7))[0]
        if len(violated) == 0:
            break
        violated_sorted = violated[np.argsort(c_new[violated])[::-1]]
        current_set = set(active)
        for v in violated_sorted:
            if v not in current_set and len(active) < 3 * n_init_constraints:
                active.append(int(v))
                current_set.add(int(v))

    return b_val, M_val


def c_new_check(b_val, T):
    """Compute T @ b_val."""
    return T @ b_val

def projected_gradient_step(a0, time_limit, lr_start=0.05, verbose=False):
    """
    Projected gradient method on max_k (conv(a,a))_k over simplex.
    Uses the subgradient: grad[m] = 2*a[k*-m] at k*=argmax.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    lr = lr_start

    for it in range(500000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        max_b = b.max()
        k_star = int(np.argmax(b))

        # Gradient: grad[m] = 2*a[k*-m] for valid m
        grad = np.zeros(n)
        m_lo = max(0, k_star - n + 1)
        m_hi = min(n - 1, k_star)
        for m in range(m_lo, m_hi + 1):
            grad[m] = 2.0 * a[k_star - m]

        # Backtracking line search
        ft = 2.0 * n * max_b
        a_old = a.copy()
        lr_t = lr
        improved = False
        for bt in range(30):
            a_trial = a_old - lr_t * grad
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                lr_t *= 0.5
                continue
            a_trial = a_trial / st
            ft_new = 2.0 * n * fast_autocorr(a_trial).max()
            if ft_new < ft - 1e-14:
                a = a_trial
                lr = lr_t * 1.05
                improved = True
                break
            lr_t *= 0.5

        if not improved:
            # Try smaller fixed step
            a_trial = a_old - 0.001 * grad
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 1e-15:
                a_trial = a_trial / st
                ft_new = 2.0 * n * fast_autocorr(a_trial).max()
                if ft_new < ft:
                    a = a_trial
                    lr = 0.001

        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if it % 2000 == 0 and verbose:
            print(f"      PG it={it}: best={best_score:.6f}, lr={lr:.6f}")

    return best_a, best_score


def peak_shaving_algorithm(a0, time_limit, seed=42):
    """
    Peak shaving: iteratively move mass from positions that contribute
    to the top peak toward positions that contribute least.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        k_star = int(np.argmax(b))

        # Find contributors to the peak
        m_lo = max(0, k_star - n + 1)
        m_hi = min(n - 1, k_star)
        idx = np.arange(m_lo, m_hi + 1)
        contrib = a[idx] * a[k_star - idx]
        total = contrib.sum()
        if total < 1e-15:
            break

        # Find top contributor
        top_ci = idx[np.argmax(contrib)]
        top_other = k_star - top_ci

        # Find positions that contribute least to ALL top peaks
        top_peaks = np.argsort(b)[::-1][:min(5, 2 * n - 1)]
        harmful = np.zeros(n)
        for k2 in top_peaks:
            lo2 = max(0, k2 - n + 1)
            hi2 = min(n - 1, k2)
            for j in range(lo2, hi2 + 1):
                harmful[j] += b[k2] * 2.0 * a[k2 - j]

        # Best target: low harmful score, not the source
        harmful[top_ci] = np.inf
        if top_other != top_ci:
            harmful[top_other] = np.inf
        target = int(np.argmin(harmful))

        # Try different fractions
        improved = False
        for frac in [0.01, 0.03, 0.05, 0.08, 0.12]:
            amt = a[top_ci] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[top_ci] -= amt
            if top_other != top_ci:
                a_trial[top_other] -= amt * 0.5
            a_trial[target] += amt
            if np.any(a_trial < -1e-12):
                continue
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            score = 2.0 * n * fast_autocorr(a_trial).max()
            if score < best_score - 1e-14:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                stall = 0
                break

        if not improved:
            # Try random perturbation
            if stall % 5 == 0:
                n_pert = rng.integers(1, 4)
                sources = rng.choice(n, size=n_pert, replace=False)
                targets = rng.choice(n, size=n_pert, replace=False)
                a_trial = a.copy()
                for src, dst in zip(sources, targets):
                    frac = rng.uniform(0.005, 0.03)
                    amt = a_trial[src] * frac
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-15:
                    a_trial = a_trial / st
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            stall += 1
            if stall > 1000:
                break

    return best_a, best_score


def generate_fejer_init(n, width_frac=0.6):
    """Generate a Fejer-kernel-like initialization."""
    center = (n - 1) / 2.0
    x = np.arange(n)
    dist = np.abs(x - center) / (n * width_frac / 2.0)
    a = np.maximum(1.0 - dist, 0) ** 2
    s = a.sum()
    if s > 0:
        a = a / s
    return a


def generate_gaussian_init(n, sigma_frac=0.3):
    """Generate a Gaussian initialization."""
    center = (n - 1) / 2.0
    x = np.arange(n)
    sigma = n * sigma_frac
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    s = a.sum()
    if s > 0:
        a = a / s
    return a


def generate_poly_init(n, degree=4):
    """Generate a polynomial-based initialization (positive on [0,1], zero at endpoints)."""
    x = np.linspace(0, 1, n)
    # Use (x*(1-x))^degree as base, multiplied by a smooth modification
    base = (x * (1 - x)) ** (degree / 2.0)
    # Add slight asymmetry to break symmetry (then we'll re-symmetrize)
    a = base * (1.0 + 0.1 * np.sin(np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a = a / s
    return a


def generate_bspline_init(n, order=6):
    """Generate a B-spline-like initialization."""
    center = (n - 1) / 2.0
    x = np.arange(n)
    width = n / (order + 1)
    dist = np.abs(x - center) / width
    # B-spline of order m: (1 - |x|)^m for |x|<=1
    a = np.maximum(1.0 - dist, 0) ** order
    s = a.sum()
    if s > 0:
        a = a / s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    t_global = time.monotonic()
    time_budget = 1700.0  # Safety margin from 1800s

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Generate diverse starts
    n_base = len(best_seq) if best_seq is not None else 1600

    if parent is None:
        n_base = 1600
        best_seq = generate_fejer_init(n_base)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # Symmetrize the parent/best
    best_seq_sym = symmetrize(best_seq)
    best_seq_sym = best_seq_sym / best_seq_sym.sum()
    sym_score = evaluate_safe(best_seq_sym)
    if sym_score < best_score:
        best_score = sym_score
        best_seq = best_seq_sym.copy()
        print(f"  Symmetrized: score={best_score:.6f}")

    # Phase 1: Quick LP Nesterov on diverse starts
    print(f"\n=== Phase 1: Quick LP on diverse starts (n={n_base}) ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t1_total = min(remaining * 0.10, 120.0)
    if t1_total > 30:
        inits = {
            'parent': best_seq.copy(),
            'fejer': generate_fejer_init(n_base, 0.5),
            'fejer_wide': generate_fejer_init(n_base, 0.7),
            'gauss': generate_gaussian_init(n_base, 0.35),
            'poly': generate_poly_init(n_base, 4),
            'bspline': generate_bspline_init(n_base, 8),
        }
        t_per_init = t1_total / len(inits)
        for name, init_a in inits.items():
            if time_budget - (time.monotonic() - t_global) < 10:
                break
            init_sym = symmetrize(init_a)
            init_sym = np.maximum(init_sym, 0)
            init_sym = init_sym / init_sym.sum()
            a_out, sc_out, _ = nesterov_lp_cutting(init_sym, t_per_init, solver,
                                                     n_init_constraints=30, enforce_sym=True, verbose=False)
            try_update(a_out, f"phase1_{name}")

    # Phase 2: Long LP Nesterov with cutting plane (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Long LP Nesterov (cutting plane, n={len(best_seq)}) (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.40, 600.0)
    if t2 > 60:
        a_out, sc_out, iters = nesterov_lp_cutting(best_seq, t2, solver,
                                                    n_init_constraints=50, enforce_sym=True, verbose=True)
        print(f"  Phase 2 done: {iters} iters, score={sc_out:.6f}")
        try_update(a_out, "phase2_lp_cp")

    # Phase 3: Projected gradient (complementary method)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Projected Gradient (n={len(best_seq)}) (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 20:
        a_out, sc_out = projected_gradient_step(best_seq, t3, lr_start=0.02, verbose=False)
        print(f"  Phase 3 done: score={sc_out:.6f}")
        try_update(a_out, "phase3_pg")

    # Phase 4: Peak shaving
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Peak Shaving (n={len(best_seq)}) (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 15:
        a_out, sc_out = peak_shaving_algorithm(best_seq, t4, seed=123)
        print(f"  Phase 4 done: score={sc_out:.6f}")
        try_update(a_out, "phase4_peak")

    # Phase 5: Another LP round on the improved best
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: LP Round 2 (n={len(best_seq)}) (remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 30:
        a_out, sc_out, iters = nesterov_lp_cutting(best_seq, t5, solver,
                                                    n_init_constraints=60, enforce_sym=True, verbose=False)
        print(f"  Phase 5 done: {iters} iters, score={sc_out:.6f}")
        try_update(a_out, "phase5_lp2")

    # Phase 6: Try nearby lengths
    for n_try in [n_base + 100, n_base - 100, n_base + 200]:
        if n_try < 500:
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 100:
            break
        print(f"\n=== Phase 6: LP at n={n_try} (remaining={remaining:.0f}s) ===")
        t6 = min(remaining * 0.15, 150.0)
        if t6 < 30:
            continue
        init_try = interpolate_sequence(best_seq, n_try)
        init_try = symmetrize(init_try)
        init_try = np.maximum(init_try, 0)
        init_try = init_try / init_try.sum()
        a_out, sc_out, _ = nesterov_lp_cutting(init_try, t6, solver,
                                                n_init_constraints=40, enforce_sym=True, verbose=False)
        try_update(a_out, f"n{n_try}")

    # Phase 7: Final peak shaving + PG
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final Polish (remaining={remaining:.0f}s) ===")
    if remaining > 30:
        t7a = min(remaining * 0.4, 60.0)
        if t7a > 10:
            a_out, sc_out = peak_shaving_algorithm(best_seq, t7a, seed=789)
            try_update(a_out, "final_peak")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t7b = min(remaining * 0.4, 40.0)
            if t7b > 5:
                a_out, sc_out = projected_gradient_step(best_seq, t7b, lr_start=0.01, verbose=False)
                try_update(a_out, "final_pg")

    # Final: try both symmetric and non-symmetric version of best
    if best_seq is not None:
        bs_sym = symmetrize(best_seq)
        bs_sym = bs_sym / bs_sym.sum()
        try_update(bs_sym, "final_sym")

        # Also try a slight perturbation
        rng = np.random.default_rng(999)
        n = len(best_seq)
        a_pert = best_seq + rng.normal(0, 0.001, n)
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()
        try_update(a_pert, "final_pert")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found! Using fallback.")
        n = 1600
        fallback = generate_fejer_init(n, 0.5)
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
