    Compute a subgradient of max(c) where c = conv(a, a).
    Uses peaks above threshold_frac * max as active set.

    if len(active) > max_peaks:

    if excess.max() < 1e-15:

    b = fast_autocorr(a0)
    top_indices = np.argsort(b)[::-1][:max_cons]

    active_set = list(top_indices)

        cons = [a_var >= 0, cp.sum(a_var) == 1]
        for k in active_set:
            # c_k = sum_{m in S_k} a[m] * a[k-m]

            prob.solve(solver=cp.ECOS)
        except:
            try:
                prob.solve(solver=cp.SCS)
            except:
                return None

            return None
        
        a_new = np.maximum(a_var.value, 0)
        s = a_new.sum()
        if s < 1e-10:
            return None
        a_new = a_new / s
