    """Mirror descent with log-sum-exp smoothing."""
    n = len(a)

    # Weighted average of gradients weighted by softmax(beta * b)
    b_shifted = b - b.max()
    w = np.exp(beta * b_shifted)
    w /= w.sum()
    
    # grad[i] = 2 * sum_k w[k] * (a[k-i] + a[i-k]) (where valid)
    # Use FFT for efficiency
    fft_size = 1 << (2*n-1).bit_length()
    A_rev = rfft(a[::-1], fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    grad = 2.0 * conv[n-1:n-1+n]

    a_new = a * np.exp(-grad * beta / (n * b.max()))
    s = a_new.sum()
    if s > 0:

    beta = 5.0
    eta = 1.0
    n_iters = 0

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        b_max = b.max()
        if b_max < 1e-12:
            break

        # Compute softmax weights for smoothed objective
        beta_eff = beta / (n * b_max + 1e-10)
        b_shifted = b - b.max()
        w = np.exp(beta_eff * b_shifted)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros_like(b)
            w[np.argmax(b)] = 1.0
        else:
            w = w / w_sum

        # Compute gradient: grad[i] = 2 * sum_k w[k] * (a[k-i] + a[i-k])
        # Using FFT: convolve w with reversed a
        A_rev = rfft(a[::-1], fft_size)
        W = rfft(w, fft_size)
        conv = irfft(A_rev * W, fft_size)
        grad = 2.0 * conv[n - 1:n - 1 + n]

        # Multiplicative update: a_new[i] = a[i] * exp(-eta * grad[i]) / Z
        log_step = -eta * grad / (b_max + 1e-10)
        # Clip to avoid numerical issues
        log_step = np.clip(log_step, -50.0, 50.0)
        a_new = a * np.exp(log_step)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

            best_score = score
            best_a = a_new.copy()
            a = a_new

            beta = min(beta * 1.01, 200.0)
        elif score < best_score * 1.001:
            a = a_new
            eta = eta * 0.95
        else:
            eta = eta * 0.7
            # Keep old a, try smaller step

def spectral_smooth(a0, time_limit, verbose=True):
    """Gerchberg-Saxton variant: alternate time-domain clipping and frequency-domain flattening."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()
    n_iters = 0

    # Relaxation parameter
    gamma = 0.3
    # Power spectrum cap (fraction of max)
    cap_fraction = 0.7

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        # FFT
        A = rfft(a, fft_size)
        P = A.real ** 2 + A.imag ** 2  # |A|^2
        P_max = P.max()

        if P_max < 1e-15:
            break

        # Cap the power spectrum: interpolate towards flat
        # Target: uniform power spectrum
        P_target = P.sum() / fft_size
        P_new = gamma * P + (1.0 - gamma) * P_target
        # Apply a cap
        cap = cap_fraction * P_max
        P_new = np.minimum(P_new, cap)

        # Reconstruct with new magnitudes, old phases
        phase = A / (np.abs(A) + 1e-30)
        A_new = np.sqrt(P_new) * phase
        a_new = irfft(A_new, fft_size)[:n]

        # Post-process: clip and normalize
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new * norm / s_new

        score = fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            # Don't update a, just continue with original
            pass

        # Slowly decrease gamma (less aggressive over time)
        gamma = max(0.1, gamma * 0.999)
        cap_fraction = min(0.95, cap_fraction + 0.0001)

        n_iters += 1
        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SS-iter={n_iters} ({elapsed:.0f}s): best_M={best_score:.8f} gamma={gamma:.3f}")

    return best_a, best_score

            stall += 1
            if stall > 500:
                break

        if verbose and (int(time.monotonic() - t0) % 10 == 0):
            elapsed = time.monotonic() - t0
            print(f"    LS ({elapsed:.0f}s): best_M={best_score:.8f} stall={stall}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    choice = rng.integers(0, 6)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.2
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2.0 * 2 * np.pi * x) + 0.15 * np.cos(5.0 * 2 * np.pi * x)
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    else:
        a = np.ones(n) + 0.1 * np.sin(4.0 * np.pi * x)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """Main entry point: multi-phase optimization to minimize the sequence score."""
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # ---- Phase 1: LP Nesterov (primary global optimizer) ----
    elapsed = time.monotonic() - t_start
    remaining = time_budget - elapsed
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.30, 480.0)
    if t1 > 30:
        a_norm = best_seq.copy()
        s_a = a_norm.sum()
        if s_a > 1e-15:
            a_norm = a_norm * np.sqrt(2.0 * n_base) / s_a
        a_out, _ = nesterov_lp(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n_base}")

    # ---- Phase 2: Multiplicative Weights (mirror descent) ----
    elapsed = time.monotonic() - t_start
    remaining = time_budget - elapsed
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: MultWeights n={n_cur} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 15:
        a_norm = best_seq.copy()
        s_a = a_norm.sum()
        if s_a > 1e-15:
            a_norm = a_norm * np.sqrt(2.0 * n_cur) / s_a
        a_out, _ = multiplicative_weights(a_norm, t2, verbose=True)
        do_try_update(a_out, f"mw_n{n_cur}")

    # ---- Phase 3: Spectral Smoothing (Gerchberg-Saxton) ----
    elapsed = time.monotonic() - t_start
    remaining = time_budget - elapsed
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: SpectralSmooth n={n_cur} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 10:
        a_norm = best_seq.copy()
        s_a = a_norm.sum()
        if s_a > 1e-15:
            a_norm = a_norm * np.sqrt(2.0 * n_cur) / s_a
        a_out, _ = spectral_smooth(a_norm, t3, verbose=True)
        do_try_update(a_out, f"ss_n{n_cur}")

    # ---- Phase 4: Local Search ----
    elapsed = time.monotonic() - t_start
    remaining = time_budget - elapsed
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: LocalSearch n={n_cur} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 220.0)
    if t4 > 10:
        a_norm = best_seq.copy()
        s_a = a_norm.sum()
        if s_a > 1e-15:
            a_norm = a_norm * np.sqrt(2.0 * n_cur) / s_a
        a_out, _ = local_search(a_norm, t4, seed=123, verbose=True)
        do_try_update(a_out, f"ls_n{n_cur}")

    # ---- Phase 5: Extended length (n+300) ----
    elapsed = time.monotonic() - t_start
    remaining = time_budget - elapsed
    n_ext = n_cur + 300
    print(f"\n=== Phase 5: LP Nesterov n={n_ext} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 220.0)
    if t5 > 30:
        init_ext = interpolate_sequence(best_seq, n_ext)
        a_norm = init_ext.copy()
        s_a = a_norm.sum()
        if s_a > 1e-15:
            a_norm = a_norm * np.sqrt(2.0 * n_ext) / s_a
        a_out, _ = nesterov_lp(a_norm, t5, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n_ext}")

    # ---- Phase 6: Second pass multiplicative weights on best ----
    elapsed = time.monotonic() - t_start
    remaining = time_budget - elapsed
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: MW pass 2 n={n_cur} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 10:
        a_norm = best_seq.copy()
        s_a = a_norm.sum()
        if s_a > 1e-15:
            a_norm = a_norm * np.sqrt(2.0 * n_cur) / s_a
        a_out, _ = multiplicative_weights(a_norm, t6, beta_init=20.0, verbose=True)
        do_try_update(a_out, f"mw2_n{n_cur}")

    # ---- Phase 7: Final local search polish ----
    elapsed = time.monotonic() - t_start
    remaining = time_budget - elapsed
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: Final LS n={n_cur} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.5, 150.0)
    if t7 > 5:
        a_norm = best_seq.copy()
        s_a = a_norm.sum()
        if s_a > 1e-15:
            a_norm = a_norm * np.sqrt(2.0 * n_cur) / s_a
        a_out, _ = local_search(a_norm, t7, seed=777, verbose=False)
        do_try_update(a_out, "ls_final")

    # ---- Final verification ----
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_start:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0] * 100

    return [float(x) for x in best_seq]
