Uses the SOTA LP direction-finding with adaptive line search, combined with

import numpy as np

from scipy.optimize import linprog

import time


def fast_autocorr(a):
    """Full autocorrelation a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_score(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)


def evaluate_safe(a):
    """Call the official evaluate_sequence safely."""
    try:
        a = np.asarray(a, dtype=np.float64)
        a = np.clip(a, 0.0, 1000.0)
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:

def build_A_sparse(a):
    """Build sparse matrix A where (A @ b)[k] = sum_i a[k-i]*b[i] for valid i.
    Shape: (2n-1, n). A[k,i] = a[k-i] if 0 <= k-i < n."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    from scipy.sparse import csr_matrix
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_direction(a, M):
    """Solve LP: maximize sum(g) s.t. conv(a,g) <= M, g >= 0.
    Returns optimal g or None on failure."""
    n = len(a)
    A = build_A_sparse(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    options = {'presolve': True, 'dual_feasibility_tolerance': 1e-9,
               'primal_feasibility_tolerance': 1e-9}
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options=options)
    if not res.success:
        return None
    g = res.x
    if g is None or g.sum() < 1e-12:
        return None
    return g


def analytic_step(a, g, M, k_star, B_k, C_k):
    """Compute optimal step size analytically assuming max stays at k_star.
    A = conv(a,a)[k_star], B = conv(a,g)[k_star], C = conv(g,g)[k_star].
    Score ~ (1-t)^2*A + 2t(1-t)*B + t^2*C (with sum=1 normalization).
    Min at t = (A-B)/(A - 2B + C)."""
    Agg = M  # conv(a,a)[k_star] = M
    denom = Agg - 2.0 * B_k + C_k
    if abs(denom) < 1e-15:
        return 0.1
    t_opt = (Agg - B_k) / denom
    return np.clip(t_opt, 0.01, 0.8)

def mirror_descent_fast(a0, time_limit, eta0=0.04, decay_rate=0.7,
                        k_min=1, k_max=12, k_ramp=300, seed=42):
    """Fast mirror descent with gradient subgradient (simplified version)."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        b_max = b.max()
        b_mean = b.sum() / len(b)

        frac = min(1.0, t / float(k_ramp))
        k = max(k_min, min(k_max, int(k_min + frac * (k_max - k_min))))

        top_idx = np.argsort(b)[::-1][:k]

        excess = b[top_idx] - b_mean
        if excess.max() < 1e-15:
            weights = np.ones(k) / k
        else:
            excess_shift = excess - excess.max()
            weights = np.exp(excess_shift) / np.exp(excess_shift).sum()

        grad = np.zeros(n)
        for ii in range(k):
            m = int(top_idx[ii])
            w_m = weights[ii] * 2.0
            j_start = max(0, m - n + 1)
            j_end = min(n - 1, m)
            js = np.arange(j_start, j_end + 1)
            grad[js] += w_m * a[m - js]

        eta = eta0 / (1.0 + t / 200.0) ** decay_rate
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a + 0.02 * rng.standard_normal(n)
            a_new = np.maximum(a_new, 0)
            a_new = a_new / max(a_new.sum(), 1e-15)
        else:
            a_new = a_new / s_new
        a = a_new

        check_interval = 3 if t < 300 else 10 if t < 3000 else 30
        if t % check_interval == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stall = 0
            else:
                stall += 1

        if stall > 400:
            a = a + 0.03 * rng.standard_normal(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            stall = 0

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=123):
    """Coordinate descent peak shaving."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:min(5, n)]

        improved = False
        for k_idx in top_k:
            k_idx = int(k_idx)
            js = max(0, k_idx - n + 1)
            je = min(n - 1, k_idx)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_idx - idx]

            if contrib.sum() < 1e-15:
                continue

            src_local = int(np.argmax(contrib))
            src = int(idx[src_local])

            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k[:3]:
                k2 = int(k2)
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) < 5:
                candidates = np.where(np.arange(n) != src)[0]
            sorted_cands = candidates[np.argsort(a[candidates])[:max(3, len(candidates) // 5)]]
            if len(sorted_cands) == 0:
                continue
            dst = int(sorted_cands[rng.integers(len(sorted_cands))])

            for frac in [0.003, 0.01, 0.03, 0.06]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                sc = fast_score(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial.copy()
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break

            if improved:
                break

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                amt = a[i] * rng.uniform(0.005, 0.05)
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    sc = fast_score(a_trial)
                    if sc < best_score:
                        best_score = sc
                        best_a = a_trial.copy()
                        a = a_trial
                        stall = 0
            if stall > 200:
                break

    return best_a, best_score


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: LP optimization from parent ===
    print(f"\n=== Phase 1: LP Optimization at n={n_parent} ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_lp1 = min(remaining * 0.35, 600.0)
    if t_lp1 > 30:
        print(f"  Starting LP (budget={t_lp1:.0f}s)")
        a_out, _ = lp_optimize(best_seq, t_lp1, verbose=True, seed=1)
        try_update(a_out, f"lp_n{n_parent}")

    # === Phase 2: More LP with different t schedule ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP continued ===")
    t_lp2 = min(remaining * 0.20, 350.0)
    if t_lp2 > 20 and best_seq is not None:
        a_out, _ = lp_optimize_fast(best_seq, t_lp2, verbose=False, seed=7)
        try_update(a_out, f"lpf_n{len(best_seq)}")

    # === Phase 3: Diverse initializations + LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Diverse Inits + LP ===")
    t_div_total = min(remaining * 0.15, 250.0)
    if t_div_total > 20:
        t_per_init = t_div_total / 3
        for v in [4, 10]:  # U-shaped, random Gaussians
            init = diverse_init(n_parent, rng, variant=v)
            a_out, _ = lp_optimize_fast(init, t_per_init, verbose=False, seed=v * 13 + 1)
            try_update(a_out, f"div_v{v}_n{n_parent}")

    # === Phase 4: Longer length exploration ===
    n_long = int(round(n_parent * 1.35))
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Longer length n={n_long} ===")
    t_long_total = min(remaining * 0.22, 380.0)
    if t_long_total > 30 and best_seq is not None:
        init_long = interpolate_to_length(best_seq, n_long)
        t_lp_long = t_long_total * 0.6
        a_out, _ = lp_optimize_fast(init_long, t_lp_long, verbose=False, seed=5)
        try_update(a_out, f"lp_n{n_long}")
        # MD polish at long length
        remaining = time_budget - (time.monotonic() - t_global)
        t_md_long = min(remaining * 0.08, 100.0)
        if t_md_long > 15 and best_seq is not None:
            a_out, _ = mirror_descent_fast(best_seq, t_md_long, eta0=0.03,
                                            decay_rate=0.6, k_min=1, k_max=10,
                                            k_ramp=200, seed=3)
            try_update(a_out, f"md_n{len(best_seq)}")

    # === Phase 5: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Final Polish ===")
    if best_seq is not None:
        # Short LP
        t_final_lp = min(remaining * 0.35, 200.0)
        if t_final_lp > 15:
            a_out, _ = lp_optimize(best_seq, t_final_lp, verbose=False, seed=99)
            try_update(a_out, "final_lp")

        # Mirror descent
        remaining = time_budget - (time.monotonic() - t_global)
        t_final_md = min(remaining * 0.30, 150.0)
        if t_final_md > 10:
            a_out, _ = mirror_descent_fast(best_seq, t_final_md, eta0=0.02,
                                            decay_rate=0.75, k_min=2, k_max=15,
                                            k_ramp=150, seed=55)
            try_update(a_out, "final_md")

        # Peak shaving
        remaining = time_budget - (time.monotonic() - t_global)
        t_final_ps = min(remaining * 0.35, 120.0)
        if t_final_ps > 5:
            a_out, _ = peak_shaving(best_seq, t_final_ps, seed=444)
            try_update(a_out, "final_ps")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("ERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
