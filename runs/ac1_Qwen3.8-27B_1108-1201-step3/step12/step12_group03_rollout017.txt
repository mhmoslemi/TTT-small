def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def symmetrize(a):
    """Enforce symmetry: a_i = a_{n-1-i}."""
    n = len(a)

    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

    C = np.zeros((2*n-1, half))

def fast_autocorr(a):

def subgradient_refine(a0, time_limit, n_top=8, verbose=True):
    """
    Smoothed subgradient / mirror descent with symmetric constraint.
    Uses softmax smoothing of the max function.
    """
    n = len(a0)
    a = normalize_nonneg(symmetrize(a0))
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    # Temperature for softmax smoothing (controls sharpness)
    temp = 0.01
    
    for it in range(20000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(a)
        max_b = b.max()
        
        # Adaptive temperature: decrease over time
        temp = max(0.001, 0.01 * (0.999 ** it))
        
        # Softmax weights over lags
        shifted = b / temp - (b.max() / temp)
        weights = np.exp(shifted)
        weights = weights / weights.sum()
        
        # Weighted subgradient: sum_k w_k * grad_k
        # grad_k[i] = d/d a_i [conv(a,a)_k] = a_{k-i} + a_{i-k} (valid)
        grad = np.zeros(n)
        for k in range(len(b)):
            if weights[k] < 1e-10:
                continue
            jlo = max(0, k - n + 1)
            jhi = min(n - 1, k)
            for j in range(jlo, jhi + 1):
                other = k - j
                grad[j] += weights[k] * a[other]
                if other != j:
                    grad[other] += weights[k] * a[j]
        
        # Symmetrize gradient
        grad = symmetrize(grad)
        
        # Mirror descent step (exponential family)
        # a_new[i] = a[i] * exp(-lr * grad[i]) / Z
        progress = it / 20000.0
        lr = 0.05 * (0.001 / 0.05) ** progress
        lr = max(lr, 0.0001)
        
        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a - lr * grad
        log_a_new = symmetrize(log_a_new)
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = symmetrize(a_new)
        a_new = normalize_nonneg(a_new)
        
        # Also try a simple gradient step with projection
        a_grad = a - lr * grad
        a_grad = symmetrize(a_grad)
        a_grad = normalize_nonneg(a_grad)
        
        # Pick the better
        s1 = score_normalized(a_new)
        s2 = score_normalized(a_grad)
        
        if s1 < s2:
            a_next = a_new
            s_next = s1
        else:
            a_next = a_grad
            s_next = s2
        
        if s_next < best_score:
            best_score = s_next
            best_a = a_next.copy()
        
        a = a_next
        
        if it % 200 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"    SG it={it} ({elapsed:.0f}s): best={best_score:.7f}, cur={s_next:.7f}")
    
    return best_a, best_score

                    Q[go, gj] += 0.5
            constraints.append(cp.quad_form(a_half, Q) <= M_var)
        
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        solved = False
        for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue
        
        if not solved:
            if rd > 2:
                break
            # Widen active set and retry
            n_active = min(n_active + 10, 100)
            continue
        
        a_half_val = np.asarray(a_half.value, dtype=np.float64)
        a_half_val = np.maximum(a_half_val, 0)
        
        # Reconstruct full a
        a_full = np.zeros(n)
        for j in range(n):
            g = min(j, n - 1 - j)
            a_full[j] = a_half_val[g]
        
        a_full = a_full / a_full.sum()
        a_full = symmetrize(a_full)
        a_full = normalize_nonneg(a_full)
        
        sc = score_normalized(a_full)
        improved = False
        
        if sc < best_score:
            best_score = sc
            best_a = a_full.copy()
            a = a_full
            improved = True
        else:
            # Try mixing
            for frac in [0.3, 0.5, 0.7]:
                a_mix = (1 - frac) * a + frac * a_full
                a_mix = symmetrize(a_mix)
                a_mix = normalize_nonneg(a_mix)
                sc_mix = score_normalized(a_mix)
                if sc_mix < best_score:
                    best_score = sc_mix
                    best_a = a_mix.copy()
                    a = a_mix
                    improved = True
                    break
        
        if verbose:
            print(f"    QP rd={rd}: best={best_score:.7f}, M_qp={float(M_var.value):.8f}, improved={improved}")
        
        if not improved and rd > 3:
            break
    
    return best_a, best_score


def local_search(a0, time_limit, seed=42, verbose=True):
    """
    Local search: coordinate descent + random pairwise moves.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize_nonneg(symmetrize(a0))
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    stall = 0
    
    # Precompute which indices are "independent" (second half mirrors first)
    # For symmetric a, changing a[i] also changes a[n-1-i]
    # So we only need to consider i in 0..n//2
    
    half = n // 2 + 1
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        top_lags = np.argsort(b)[::-1][:5]
        
        improved = False
        
        # Try coordinate moves: shift mass from one position to another
        n_tries = 50
        for _ in range(n_tries):
            # Pick a source (from top contributing to peaks) and a target
            src = rng.integers(0, half)
            dst = rng.integers(0, half)
            if src == dst:
                continue
            
            # Map to full indices (use the mirror position for symmetry)
            src_full = src  # We'll adjust both src and n-1-src
            dst_full = dst
            
            for frac in [0.001, 0.003, 0.01, 0.02, 0.05]:
                amt = a[src_full] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                # Move from src (and its mirror) to dst (and its mirror)
                a_trial[src_full] -= amt
                a_trial[n - 1 - src_full] -= amt
                a_trial[dst_full] += amt
                a_trial[n - 1 - dst_full] += amt
                
                if a_trial.min() < -1e-12:
                    continue
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                a_trial = symmetrize(a_trial)
                
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-12:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            stall += 1
            # Larger random perturbation
            if stall % 20 == 0:
                a_trial = a.copy()
                n_pert = rng.integers(2, 6)
                for _ in range(n_pert):
                    s = rng.integers(0, half)
                    d = rng.integers(0, half)
                    if s == d:
                        continue
                    frac = rng.uniform(0.005, 0.03)
                    amt = a_trial[s] * frac
                    a_trial[s] -= amt
                    a_trial[n-1-s] -= amt
                    a_trial[d] += amt
                    a_trial[n-1-d] += amt
                if a_trial.min() < -1e-12:
                    continue
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                a_trial = symmetrize(a_trial)
                sc = score_normalized(a_trial)
                if sc < best_score:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    stall = 0
        
        if stall > 300:
            break
        
        if it_check := (int((time.monotonic()-t0)*10) % 50 == 0):
            pass
    
    return best_a, best_score


def coordinate_descent_fine(a0, time_limit, verbose=True):
    """
    Fine-grained coordinate descent: for each position, find optimal value by
    1D line search (quadratic in that coordinate).
    """
    n = len(a0)
    a = normalize_nonneg(symmetrize(a0))
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    half = n // 2 + 1
    
    # For a symmetric sequence, changing a[i] (i < n/2) changes both a[i] and a[n-1-i]
    # The objective as a function of a[i] (with others fixed) is quadratic
    # (because conv(a,a)_k is quadratic in a)
    # So we can do exact 1D minimization for each coordinate
    
    sweep = 0
    while time.monotonic() - t0 < time_limit:
        improved_any = False
        
        # Process coordinates in random order for better mixing
        order = np.arange(half)
        np.random.shuffle(order)
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Current value
            v = a[i]
            if v < 1e-15:
                # Try small positive values
                for trial_v in [1e-5, 1e-4, 5e-4, 1e-3]:
                    a[i] = trial_v
                    a[n-1-i] = trial_v
                    a = normalize_nonneg(a)
                    sc = score_normalized(a)
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
                        improved_any = True
                        v = trial_v
                    else:
                        a[i] = 0
                        a[n-1-i] = 0
                        break
                if v < 1e-15:
                    a[i] = 0
                    a[n-1-i] = 0
                continue
            
            # Golden section search for optimal value of a[i]
            # We keep sum(a)=1 by adjusting: if we change a[i] by dv, 
            # we need to redistribute. Simplest: change a[i] and renormalize.
            # But that changes all coordinates. Alternative: move mass from a[i] to a "sink".
            
            # Better: optimize a[i] by moving mass from position i to a fixed "background"
            # Or just do a simple perturbation search
            best_local = v
            for delta in [v * 0.1, v * 0.05, -v * 0.1, -v * 0.05, v * 0.5, -v * 0.5]:
                trial_v = v + delta
                if trial_v < 0:
                    trial_v = 0
                a[i] = trial_v
                a[n-1-i] = trial_v
                a = normalize_nonneg(a)
                sc = score_normalized(a)
                if sc < score_normalized(best_a):
                    best_local = trial_v
            
            if best_local != v:
                a[i] = best_local
                a[n-1-i] = best_local
                a = normalize_nonneg(a)
                a = symmetrize(a)
                sc = score_normalized(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    improved_any = True
        
        sweep += 1
        if not improved_any and sweep > 5:
            break
        if sweep % 10 == 0 and verbose:
            print(f"    CD sweep={sweep}: best={best_score:.7f}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Linearly interpolate a sequence to a new length, preserving symmetry."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = symmetrize(f)
    f = normalize_nonneg(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = symmetrize(f_interp)
    return normalize_nonneg(f_interp)


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: multi-phase symmetric optimization.
    """
    t_start = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    rng = np.random.default_rng(seed)
    
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    # Try to get parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 100 and np.isfinite(p).all() and p.sum() > 1e-10:
            p = symmetrize(p)
            p = normalize_nonneg(p)
            p_score = evaluate_safe(p)
            print(f"Parent (symmetrized): n={len(p)}, score={p_score:.7f}")
            best_seq = p.copy()
            best_score = p_score
    except Exception as e:
        print(f"Parent load failed: {e}")
    
    if best_seq is None:
        # Default initialization: cosine bump
        n_base = 1600
        x = np.linspace(0, np.pi, n_base)
        best_seq = np.cos(x / 2) ** 2
        best_seq = symmetrize(best_seq)
        best_seq = normalize_nonneg(best_seq)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.7f}")
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = symmetrize(np.asarray(a, dtype=np.float64))
        a = normalize_nonneg(a)
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST [{label}]: {sc:.7f} (n={len(a)}, t={elapsed:.0f}s)")
    
    n_base = len(best_seq)
    print(f"Starting length: {n_base}")
    
    # Phase 1: LP Nesterov (main workhorse)
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 1: LP Nesterov (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.40, 650.0)
    if t1 > 60:
        a_out, sc_out, iters = nesterov_lp_symmetric(best_seq, t1, solver)
        print(f"  LP Nesterov: {iters} iters, score={sc_out:.7f}")
        try_update(a_out, "lp1")
    
    # Phase 2: Subgradient refinement
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 2: Subgradient refinement (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 30:
        a_out, sc_out = subgradient_refine(best_seq, t2)
        print(f"  Subgradient: score={sc_out:.7f}")
        try_update(a_out, "sg")
    
    # Phase 3: QP active set
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 3: QP Active Set (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 20:
        a_out, sc_out = active_set_qp_refine(best_seq, t3, n_active=30)
        print(f"  QP: score={sc_out:.7f}")
        try_update(a_out, "qp")
    
    # Phase 4: LP Nesterov round 2 (exploit improvements)
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 4: LP Nesterov Round 2 (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.20, 250.0)
    if t4 > 40:
        a_out, sc_out, iters = nesterov_lp_symmetric(best_seq, t4, solver, verbose=False)
        print(f"  LP Nesterov 2: {iters} iters, score={sc_out:.7f}")
        try_update(a_out, "lp2")
    
    # Phase 5: Coordinate descent fine tuning
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 5: Coordinate Descent (remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 120.0)
    if t5 > 10:
        a_out, sc_out = coordinate_descent_fine(best_seq, t5)
        print(f"  Coord desc: score={sc_out:.7f}")
        try_update(a_out, "cd")
    
    # Phase 6: Try nearby lengths
    for n_try in [n_base - 100, n_base + 100, n_base - 200, n_base + 200]:
        if n_try < 400:
            continue
        remaining = time_budget - (time.monotonic() - t_start)
        if remaining < 80:
            break
        print(f"\n=== Phase 6: Length {n_try} (remaining={remaining:.0f}s) ===")
        t6 = min(remaining * 0.15, 100.0)
        if t6 < 30:
            continue
        init = interpolate_sequence(best_seq, n_try)
        a_out, sc_out, iters = nesterov_lp_symmetric(init, t6, solver, verbose=False)
        print(f"  LP n={n_try}: {iters} iters, score={sc_out:.7f}")
        try_update(a_out, f"lp_n{n_try}")
    
    # Phase 7: Final local search
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 7: Final Local Search (remaining={remaining:.0f}s) ===")
    t7 = min(remaining * 0.3, 80.0)
    if t7 > 5:
        a_out, sc_out = local_search(best_seq, t7, seed=123)
        print(f"  Local search: score={sc_out:.7f}")
        try_update(a_out, "local")
    
    # Phase 8: Final QP polish
    remaining = time_budget - (time.monotonic() - t_start)
    if remaining > 10:
        t8 = min(remaining * 0.5, 40.0)
        a_out, sc_out = active_set_qp_refine(best_seq, t8, n_active=20, verbose=False)
        try_update(a_out, "final_qp")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
    
    return [float(x) for x in best_seq]
