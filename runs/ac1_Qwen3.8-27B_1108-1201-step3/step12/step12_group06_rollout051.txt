Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

2. Multi-start initialization (chirp, PSWF-like, parent, random)

import numpy as np

from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Fast linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score assuming sum(a)=1: returns 2*n*max(b)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_from_z(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Inverse of softmax (log)."""
    return np.log(np.maximum(a, 1e-300))


def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def get_lp_solver():
    """Find the best available LP solver."""
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CBC]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_sparse_conv_matrix(n, active_lags, a_fixed):
    """Build convolution matrix rows for active lags only."""
    T = np.zeros((len(active_lags), n))
    for idx, k in enumerate(active_lags):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        idx_arr = np.arange(lo, hi + 1)
        T[idx, idx_arr] = a_fixed[k - idx_arr]
    return T


def solve_lp_with_constraints(a_fixed, active_lags, solver):
    """Solve LP: min M s.t. (conv rows at active_lags) @ b <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_sparse_conv_matrix(n, active_lags, a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS, cp.CBC]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

        y = y_new

        # Periodic evaluation and active set update

            if sc_a < best_score:

def lse_objective_grad(z_in, T_val, n):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    # Gradient: d/dz_i = a_i * (grad_a_i - sum_j a_j * grad_a_j)
    # where grad_a_i = d/d a_i [sum_k w[k] * b[k]]
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_lbfgs(a0, time_limit, T_start, T_min, n_stages, iters_per_stage):
    """Adaptive LogSumExp L-BFGS."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    T = T_start

    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, T, n)
            return grad

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-16, 'maxfun': iters_per_stage * 200,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)

            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()

            # Temperature scheduling
            if stage > 0:
                T = max(T_min, T * 0.7)
            if (stage + 1) % 5 == 0:
                print(f"    LSE stage={stage+1}/{n_stages} T={T:.2f}: score={sc:.8f}")

        except Exception:
            T = max(T_min, T * 0.8)

    return best_a, best_score


def peak_shave(a0, time_limit, n_steps=200000):
    """Peak-targeted subgradient with Polyak step and iterate averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()

    best_a = a.copy()
    best_score = score_normalized(a)
    lower_bound = 2.0 * n / (2.0 * n - 1.0)

    t0 = time.monotonic()
    avg_a = a.copy()
    avg_count = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b

        # Top peaks
        threshold = max_b * 0.95
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        if len(active) > 30:
            top_idx = np.argsort(b[active])[-30:]
            active = active[top_idx]

        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        S = np.dot(g, a)
        grad_z = a * (g - S)
        grad_norm = np.linalg.norm(grad_z)
        if grad_norm < 1e-15:
            break

        gap = cur_score - lower_bound
        if gap < 1e-12:
            break
        eta = gap / (2.0 * grad_norm * grad_norm)
        eta = min(eta, 0.3)

        z_new = z - eta * grad_z
        a_new = softmax_from_z(z_new)
        sc_new = score_normalized(a_new)

        if sc_new > cur_score * 1.02:
            lo_e, hi_e = 0.0, eta
            for _ in range(4):
                mid_e = (lo_e + hi_e) / 2.0
                z_mid = z - mid_e * grad_z
                a_mid = softmax_from_z(z_mid)
                if score_normalized(a_mid) < cur_score:
                    lo_e = mid_e
                else:
                    hi_e = mid_e
            eta = lo_e
            z_new = z - eta * grad_z
            a_new = softmax_from_z(z_new)
            sc_new = score_normalized(a_new)

        z = z_new
        a = a_new

        avg_a = (avg_count * avg_a + a) / (avg_count + 1)
        avg_count += 1
        if avg_count > 3000:
            avg_count = 3000
            avg_a = (avg_a * 3000 + a) / 3001

        if sc_new < best_score:
            best_score = sc_new
            best_a = a.copy()

        sc_avg = score_normalized(avg_a)
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = avg_a.copy()

        if (step + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave step={step+1} ({elapsed:.0f}s): cur={cur_score:.8f} best={best_score:.8f}")

    return best_a, best_score


def spectrally_flatten(a0, time_limit, alpha=0.3, n_iters=5000):
    """Riemann-style spectral flattening for non-negative sequences."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        A = np.fft.fft(a, fft_size)
        mag = np.abs(A)
        mag = np.maximum(mag, 1e-12)
        phase = A / mag

        # Flatten: compress dynamic range of |A|
        mag_new = mag ** (1.0 - alpha)
        A_new = phase * mag_new

        a_new = np.real(np.fft.ifft(A_new, fft_size))[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s

        # Blend
        a = 0.6 * a + 0.4 * a_new
        a = np.maximum(a, 0)
        a = a / a.sum()

        sc = score_normalized(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        # Adapt alpha: decrease as we converge
        alpha = max(0.05, alpha * 0.999)

    return best_a, best_score


def chirp_init(n, rng):
    """Generate chirp-like initialization."""
    x = np.arange(n) / n
    # Quadratic chirp wrapped in a window
    a = 1.0 + np.cos(2 * np.pi * x * x * 3)
    a = a * (1.0 - np.cos(2 * np.pi * x))  # raised cosine window
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def psfw_init(n, rng):
    """Generate prolate-spheroidal-wave-like initialization."""
    x = np.linspace(-1, 1, n)
    # Approximate PSWF using sinc-like function
    a = np.sinc(0.8 * x) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def cosine_init(n, rng):
    """Cosine-bell initialization."""
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * (x - 0.5)) ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function that minimizes the evaluation score."""
    rng = np.random.default_rng(seed)
    effective_budget = max(10.0, float(budget_s) - 30.0)
    global_deadline = time.monotonic() + effective_budget

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Total budget: {budget_s}s, effective: {effective_budget:.0f}s")

    # Get parent sequence
    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        pass

    n_base = 1200
    if best_seq is None:
        best_seq = cosine_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        n_base = n_base
        print(f"Default init: n={n_base}, score={best_score:.8f}")
    else:
        n_base = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            remain = global_deadline - time.monotonic()
            print(f"  ** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, remain={remain:.0f}s)")

    # === Phase 0: Try multiple initializations, pick the best ===
    print(f"\n=== Phase 0: Initialization (score={best_score:.8f}) ===")
    inits = {
        'parent': best_seq.copy(),
        'chirp': chirp_init(n_base, rng),
        'psfw': psfw_init(n_base, rng),
        'cosine': cosine_init(n_base, rng),
    }
    # Add random
    r = rng.random(n_base)
    r = r / r.sum()
    inits['random'] = r

    # Quick spectral flatten each for 5s to compare
    for name, init_a in inits.items():
        remain = global_deadline - time.monotonic()
        if remain < 10:
            break
        t_test = min(8.0, remain * 0.2)
        if t_test < 3:
            break
        a_test, _ = spectrally_flatten(init_a, t_test, alpha=0.5, n_iters=500)
        sc = evaluate_safe(a_test)
        print(f"    Init '{name}': quick score={sc:.8f}")
        if sc < best_score:
            best_score = sc
            best_seq = a_test.copy()
            print(f"  ** Better init '{name}': {sc:.8f}")

    # === Phase 1: Active-set Nesterov LP ===
    print(f"\n=== Phase 1: Active-Set Nesterov (score={best_score:.8f}) ===")
    remain = global_deadline - time.monotonic()
    t1 = min(0.40 * effective_budget, remain * 0.55)
    if t1 > 30:
        a_out, _ = active_set_nesterov(best_seq, t1, solver, initial_k=40, max_k=500, max_iters=500000)
        try_update(a_out, "active_nesterov_1")

        # Second pass with larger active set
        remain = global_deadline - time.monotonic()
        if remain > 30:
            t1b = min(0.15 * effective_budget, remain * 0.4)
            if t1b > 20:
                a_out2, _ = active_set_nesterov(best_seq, t1b, solver, initial_k=100, max_k=800, max_iters=300000)
                try_update(a_out2, "active_nesterov_2")

    # === Phase 2: Spectral flattening + LSE ===
    print(f"\n=== Phase 2: Spectral + LSE (score={best_score:.8f}) ===")
    remain = global_deadline - time.monotonic()
    t2_total = min(0.20 * effective_budget, remain * 0.5)
    if t2_total > 20:
        # Spectral flatten
        t2a = t2_total * 0.4
        a_sf, _ = spectrally_flatten(best_seq, t2a, alpha=0.3, n_iters=10000)
        try_update(a_sf, "spectral")

        # LSE from best
        remain = global_deadline - time.monotonic()
        if remain > 10:
            t2b = min(t2_total * 0.6, remain * 0.7)
            a_lse, _ = lse_lbfgs(best_seq, t2b, T_start=50.0, T_min=0.1, n_stages=20, iters_per_stage=40)
            try_update(a_lse, "lse_main")

    # === Phase 3: Peak shaving ===
    print(f"\n=== Phase 3: Peak Shaving (score={best_score:.8f}) ===")
    remain = global_deadline - time.monotonic()
    t3 = min(0.15 * effective_budget, remain * 0.5)
    if t3 > 15:
        a_ps, _ = peak_shave(best_seq, t3, n_steps=300000)
        try_update(a_ps, "peak_shave")

        remain = global_deadline - time.monotonic()
        if remain > 15:
            a_ps2, _ = peak_shave(best_seq, remain * 0.5, n_steps=200000)
            try_update(a_ps2, "peak_shave_2")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (score={best_score:.8f}) ===")
    remain = global_deadline - time.monotonic()
    t4_total = min(0.15 * effective_budget, remain * 0.6)
    if t4_total > 30:
        for n_try in [n_base + 300, n_base + 500, n_base + 200]:
            remain = global_deadline - time.monotonic()
            if remain < 25:
                break
            t4 = min(t4_total / 3.0, remain * 0.4)
            if t4 < 15:
                break
            print(f"    Trying n={n_try}, budget={t4:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            a_try, _ = active_set_nesterov(init_try, t4 * 0.5, solver, initial_k=40, max_k=300, max_iters=100000)
            try_update(a_try, f"len_{n_try}")

            remain = global_deadline - time.monotonic()
            if remain > 10:
                a_try2, _ = lse_lbfgs(a_try, remain * 0.4, T_start=30.0, T_min=0.2, n_stages=8, iters_per_stage=25)
                try_update(a_try2, f"len_{n_try}_lse")

    # === Phase 5: Final polish ===
    print(f"\n=== Phase 5: Final Polish (score={best_score:.8f}) ===")
    remain = global_deadline - time.monotonic()
    if remain > 20:
        # Short active-set Nesterov
        a_f1, _ = active_set_nesterov(best_seq, remain * 0.3, solver, initial_k=80, max_k=600, max_iters=100000)
        try_update(a_f1, "final_nesterov")

        remain = global_deadline - time.monotonic()
        if remain > 10:
            # Very low temperature LSE
            a_f2, _ = lse_lbfgs(best_seq, remain * 0.4, T_start=5.0, T_min=0.05, n_stages=10, iters_per_stage=30)
            try_update(a_f2, "final_lse_cold")

        remain = global_deadline - time.monotonic()
        if remain > 10:
            # Peak shave
            a_f3, _ = peak_shave(best_seq, remain * 0.5, n_steps=100000)
            try_update(a_f3, "final_shave")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"Time used: {time.monotonic() - (global_deadline - effective_budget):.0f}s")
    return [float(x) for x in best_seq]
