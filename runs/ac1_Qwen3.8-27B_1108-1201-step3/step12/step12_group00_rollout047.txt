    w = np.exp(Tc)
    w /= w.sum()
    # grad[i] = 2 * sum_k w[k] * a[k-i]
    # = 2 * cross_correlation(w, a) at lag i
    # Using FFT: cross_corr[i] = ifft(FFT(w) * conj(FFT(a_pad)))[appropriate_index]
    # Simpler: conv(w, a_rev)[n-1+i]
    a_rev = a[::-1]
    size = len(w) + n - 1  # = (2n-1) + n - 1 = 3n-2
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    Ar = rfft(a_rev, fft_size)
    conv = irfft(W * Ar, fft_size)
    grad = 2.0 * conv[n - 1 : n - 1 + n]
    return grad

def mw_best_response(a, n_iters=30, eta_scale=0.3):
    """
    Approximate max 1^T b s.t. max_k (a*b)[k] <= 1, b >= 0
    via multiplicative weights. Returns b normalized to sum=1.
    """
    n = len(a)
    a_max = a.max()
    if a_max < 1e-15:
        return np.ones(n) / n
    eta = eta_scale / a_max
    b = np.ones(n) / n
    for _ in range(n_iters):
        ab = fast_cross_conv(a, b)  # length 2n-1
        k_star = int(np.argmax(ab))
        j_start = max(0, k_star - n + 1)
        j_end = min(n - 1, k_star)
        if j_end >= j_start:
            indices = np.arange(j_start, j_end + 1)
            b[indices] *= np.exp(-eta * a[k_star - indices])
        b = np.maximum(b, 0)
        b_sum = b.sum()
        if b_sum < 1e-15:
            b = np.ones(n) / n
        else:
            b /= b_sum
    return b

def nesterov_mw(a0, time_limit, n_iters_inner=25, eta_scale=0.3):
    """Nesterov-accelerated MW best-response method."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute best response
        b = mw_best_response(y, n_iters=n_iters_inner, eta_scale=eta_scale)
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
        
        if t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Nesterov-MW t={t} ({elapsed:.1f}s): best={best_score:.8f}")
    
    return best_a, best_score

def lse_gradient_descent(a0, time_limit, T_start=20.0, T_end=0.5, n_stages=8):
    """LSE-smoothed gradient descent with momentum."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    v = np.zeros(n)  # momentum
    
    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        frac = stage / max(1, n_stages - 1)
        T = T_start * (T_end / T_start) ** frac
        lr = 1.0 / (T * n * 0.1)
        
        stage_iters = 0
        while time.monotonic() - t0 < time_limit:
            stage_iters += 1
            grad = lse_grad(a, T)
            v = 0.9 * v - lr * grad
            a_new = a + v
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s
            a = a_new
            
            if stage_iters % 20 == 0:
                score = fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
            
            # Break inner loop if time running out
            if time.monotonic() - t0 > time_limit:
                break
    
    final_score = fast_autocorr(a).max()
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    return best_a, best_score

def stochastic_local_search(a0, time_limit, seed=123):
    """Simulated annealing local search."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    current_score = best_score
    base_temp = current_score * 0.01
    n_iterations = 0
    
    while time.monotonic() - t0 < time_limit:
        n_iterations += 1
        # Choose a move type
        move_type = rng.integers(0, 4)
        
        a_trial = a.copy()
        if move_type == 0:
            # Single element adjustment
            i = rng.integers(n)
            delta = rng.uniform(-1, 1) * a.mean() * 0.5
            a_trial[i] = max(0, a_trial[i] + delta)
        elif move_type == 1:
            # Pair swap (transfer mass)
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j:
                amt = rng.uniform(0.01, 0.1) * min(a[i], a[j])
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < 0:
                    a_trial[i], a_trial[j] = a_trial[j], a_trial[i]
        elif move_type == 2:
            # Block move
            i = rng.integers(n)
            block_size = rng.integers(2, min(10, n // 4))
            j = rng.integers(n)
            if abs(i - j) > block_size:
                tmp = a_trial[i:i+block_size].copy()
                a_trial[i:i+block_size] = a_trial[j:j+block_size]
                a_trial[j:j+block_size] = tmp
        else:
            # Gaussian perturbation
            a_trial += rng.normal(0, a.mean() * 0.1, n)
            a_trial = np.maximum(a_trial, 0)
        
        if a_trial.sum() < 1e-15:
            continue
        a_trial = a_trial * norm / a_trial.sum()
        
        new_score = fast_autocorr(a_trial).max()
        
        # Acceptance criterion (simulated annealing)
        elapsed = time.monotonic() - t0
        progress = elapsed / time_limit
        temp = base_temp * (1.0 - progress) ** 2
        
        if new_score < current_score:
            a = a_trial
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        elif temp > 1e-15 and rng.random() < np.exp(-(new_score - current_score) / temp):
            a = a_trial
            current_score = new_score
        
        if n_iterations % 1000 == 0:
            print(f"    SLA iter={n_iterations} ({elapsed:.1f}s): best={best_score:.8f}")
    
    return best_a, best_score

def interpolate_seq(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    result = np.interp(new_x, old_x, f)
    result = np.maximum(result, 0.0)
    ss = result.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return result / ss

def spectral_init(n, rng):
    """Generate initial sequences using spectral methods."""
    choice = rng.integers(0, 6)
    if choice == 0:
        # Random phases, flat magnitude
        N = 1 << (2 * n - 1).bit_length()
        phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
        spectrum = np.exp(1j * phases)
        signal = irfft(spectrum, N)[:n]
        a = np.abs(signal)
    elif choice == 1:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 2:
        # Multi-frequency
        x = np.linspace(0, 2 * np.pi, n)
        n_freqs = rng.integers(3, 8)
        freqs = rng.integers(1, n // 2, n_freqs)
        amps = rng.uniform(0.1, 1.0, n_freqs)
        phases = rng.uniform(0, 2 * np.pi, n_freqs)
        a = np.ones(n) * 0.5
        for f, amp, ph in zip(freqs, amps, phases):
            a += amp * np.cos(f * x + ph)
    elif choice == 3:
        # Gaussian mixture
        n_components = rng.integers(2, 5)
        centers = rng.uniform(0.1, 0.9, n_components)
        widths = rng.uniform(0.05, 0.2, n_components)
        x = np.linspace(0, 1, n)
        a = np.zeros(n)
        for c, w in zip(centers, widths):
            a += np.exp(-((x - c) / w) ** 2)
    elif choice == 4:
        # Comb pattern
        period = rng.integers(max(2, n // 20), max(3, n // 5))
        a = (1.0 + np.cos(2 * np.pi * np.arange(n) / period)) / 2
    else:
        # Triangle wave
        x = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * x - 1.0)
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
