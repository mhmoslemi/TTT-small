def mirror_descent_best_response(a, T=5.0, n_steps=200, eta=1.0):
    """
    Approximate best response: min_b (1/T)log(sum_k exp(T*conv(a,b)_k))
    s.t. b in simplex, using mirror descent (multiplicative weights).
    Returns (b, M_approx) where M_approx = (1/T)log(sum_k exp(T*conv(a,b)_k)).
    """
    n = len(a)
    b = np.ones(n) / n  # Initialize uniformly
    
    for step in range(n_steps):
        # Compute conv(a, b)
        # conv(a,b)_k = sum_j a_j * b_{k-j}
        # This is a cross-correlation, which we can compute via FFT
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        A = rfft(a, fft_size)
        B = rfft(b, fft_size)
        c = irfft(A * B, fft_size)[:size]  # conv(a, b)
        
        # Compute LSE value
        Tc = T * c
        max_Tc = Tc.max()
        exp_Tc = np.exp(Tc - max_Tc)
        lse_val = (max_Tc + np.log(exp_Tc.sum())) / T
        w = exp_Tc / exp_Tc.sum()
        
        # Compute gradient: grad_b_j = a_j * sum_{k=j}^{j+n-1} w_k
        # (with appropriate bounds)
        # prefix sum of w
        w_pad = np.zeros(2 * n - 1 + n)  # pad to avoid boundary issues
        w_pad[:2*n-1] = w
        # Actually, S_j = sum_{k=max(0,j)}^{min(2n-2, j+n-1)} w_k
        # Let's use cumulative sum
        cumw = np.concatenate([[0], np.cumsum(w)])
        S = np.zeros(n)
        for j in range(n):
            k_lo = j  # k-j >= 0 => k >= j
            k_hi = min(2 * n - 2, j + n - 1)  # k-j <= n-1 => k <= j+n-1
            S[j] = cumw[k_hi + 1] - cumw[k_lo]
        
        # grad_b = a * S
        grad_b = a * S
        
        # Mirror descent update
        log_b = np.log(np.maximum(b, 1e-300))
        log_b_new = log_b - eta * grad_b
        log_b_new -= log_b_new.max()  # for numerical stability
        b_new = np.exp(log_b_new)
        b_new /= b_new.sum()
        b = b_new
    
    # Final evaluation
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    c = irfft(A * B, fft_size)[:size]
    Tc = T * c
    max_Tc = Tc.max()
    lse_val = (max_Tc + np.log(np.exp(Tc - max_Tc).sum())) / T
    
    return b, lse_val

    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:

def mirror_descent_br(a, b_init, T, n_steps, eta):

    Warm-started mirror descent for: min_b (1/T)log(sum_k exp(T*conv(a,b)_k)) s.t. b in simplex.
    Returns (b, lse_value).

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov zero-sum game with fast mirror-descent best response (primary)
2. p-norm smooth gradient descent with Adam optimizer (secondary)
3. Direct max-subgradient polish with Polyak step (final refinement)
4. Targeted peak-shaving local search
5. Multi-length exploration with interpolation
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full autocorrelation a*a via FFT, returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_corr_conv(x, y):
    """
    Compute conv(x_reversed, y) and return elements [len(x)-1 : 2*len(x)-1].
    This equals: result[i] = sum_j x[j] * y[j+i] for i=0..len(x)-1.
    """
    nx = len(x)
    ny = len(y)
    total_len = nx + ny - 1
    fft_size = 1 << (total_len - 1).bit_length()
    x_rev = x[::-1]
    X = rfft(x_rev, fft_size)
    Y = rfft(y, fft_size)
    conv_full = irfft(X * Y, fft_size)[:total_len]
    return conv_full[nx - 1: 2 * nx - 1]


def mirror_descent_br(a, b_init, T, n_steps, eta):
    """
    Warm-started mirror descent for min_b LSE_T(conv(a,b)) s.t. b in simplex.
    Returns (b, lse_value).
    """
    n = len(a)
    b = np.asarray(b_init, dtype=np.float64).copy()
    b = np.maximum(b, 1e-300)
    b /= b.sum()

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A_fft = rfft(a, fft_size)
    cumw = np.empty(2 * n)

    for step in range(n_steps):
        B_fft = rfft(b, fft_size)
        c = irfft(A_fft * B_fft, fft_size)[:size]

        Tc = T * c
        max_Tc = Tc.max()
        exp_Tc = np.exp(Tc - max_Tc)
        Z = exp_Tc.sum()
        w = exp_Tc / Z

        # grad_b_j = a_j * sum_{k=j}^{j+n-1} w_k
        cumw[0] = 0.0
        np.cumsum(w, out=cumw[1:])
        S = cumw[n:2 * n] - cumw[0:n]
        grad_b = a * S

        # Mirror descent update
        log_b = np.log(b) - eta * grad_b
        log_b -= log_b.max()
        b = np.exp(log_b)
        b_sum = b.sum()
        if b_sum < 1e-300:
            b = np.ones(n) / n
        else:
            b /= b_sum

    # Final LSE
    B_fft = rfft(b, fft_size)
    c = irfft(A_fft * B_fft, fft_size)[:size]
    Tc = T * c
    max_Tc = Tc.max()
    lse_val = (max_Tc + np.log(np.exp(Tc - max_Tc).sum())) / T
    return b, lse_val


def nesterov_game_fast(a0, time_limit, T=5.0, eta_inner=0.8, n_inner=40, check_interval=50):
    """
    Nesterov-accelerated zero-sum game with mirror descent best response.
    Much faster than LP-based inner solver, allows many more iterations.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    b = np.ones(n) / n
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    t0 = time.monotonic()
    stall = 0
    it_count = 0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit - 2:
            break

        # Adaptive inner steps: more early on, fewer later (warm start helps)
        if t < 200:
            n_steps = n_inner
        elif t < 2000:
            n_steps = max(10, n_inner // 2)
        else:
            n_steps = max(5, n_inner // 4)

        b, M_val = mirror_descent_br(y, b, T, n_steps, eta_inner)

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * b
        y_new = np.maximum(y_new, 0)
        ys = y_new.sum()
        if ys < 1e-15:
            y_new = b.copy()
        else:
            y_new /= ys

        if t == 1:
            a_new = b.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            a_new = y_new + beta * (y_new - y)
            a_new = np.maximum(a_new, 0)
            sa = a_new.sum()
            if sa < 1e-15:
                a_new = y_new.copy()
            else:
                a_new /= sa

        y = y_new
        a = a_new
        it_count += 1

        # Periodic score check
        if it_count % check_interval == 0:
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
                stall = 0
            else:
                stall += 1

            score_y = 2.0 * n * fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()
                stall = 0

            if it_count % (check_interval * 20) == 0:
                elapsed = time.monotonic() - t0
                print(f"      NG t={t} ({elapsed:.0f}s): best={best_score:.6f}")

        # Adaptive restart
        if stall > 300 and t > 5000:
            y = best_a.copy()
            a = best_a.copy()
            b, _ = mirror_descent_br(y, np.ones(n) / n, T, n_inner, eta_inner)
            stall = 0
            print(f"      NG restart at t={t}, best={best_score:.6f}")

    # Final check
    score_final = 2.0 * n * fast_autocorr(a).max()
    if score_final < best_score:
        best_score = score_final
        best_a = a.copy()
    score_yf = 2.0 * n * fast_autocorr(y).max()
    if score_yf < best_score:
        best_score = score_yf
        best_a = y.copy()

    return best_a, best_score


def pnorm_grad_descent(a0, time_limit, p_start=50, p_end=500, n_stages=15, lr_base=3e-3):
    """
    p-norm smooth gradient descent with Adam and Nesterov momentum.
    Minimizes (1/n)*(sum_k c_k^p)^(1/p) where c = autocorr(a).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    p_values = np.geomspace(p_start, p_end, n_stages)

    for si, p_val in enumerate(p_values):
        if time.monotonic() - t0 > time_limit - 5:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        lr = lr_base * min(1.0, 80.0 / n) * (0.85 ** si)
        m = np.zeros(n)
        v = np.zeros(n)
        t_adam = 0
        b1, b2, eps = 0.9, 0.999, 1e-8

        max_iters = max(20, int(min(remaining, 60.0) * 5))

        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit - 3:
                break

            c = fast_autocorr(a)
            # f_p = (1/n) * (sum c_k^p)^(1/p)
            # grad_a_i = (2/n) * S^(1/p-1) * sum_k c_k^(p-1) * a_{k-i}
            c_pow = np.power(np.maximum(c, 1e-30), p_val - 1)
            S = np.sum(np.power(np.maximum(c, 1e-30), p_val))
            if S < 1e-300:
                grad = np.zeros(n)
            else:
                # grad_a_i = (2/n) * S^(1/p-1) * sum_k c_k^(p-1) * a_{k-i}
                # sum_k c_k^(p-1) * a_{k-i} = cross-correlation of a with c_pow
                # = conv(a_rev, c_pow)[n-1:2n-1]
                weighted = cross_corr_conv(a, c_pow)
                grad = (2.0 / n) * np.power(S, 1.0 / p_val - 1.0) * weighted

            t_adam += 1
            m = b1 * m + (1 - b1) * grad
            v = b2 * v + (1 - b2) * grad ** 2
            m_hat = m / (1 - b1 ** t_adam)
            v_hat = v / (1 - b2 ** t_adam)
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0)
            sn = a_new.sum()
            if sn < 1e-15:
                a_new = a
            else:
                a_new /= sn
            a = a_new

            if (it + 1) % 10 == 0:
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()

        a_score = 2.0 * n * fast_autocorr(a).max()
        if a_score < best_score:
            best_score = a_score
            best_a = a.copy()

    return best_a, best_score


def subgradient_polish(a0, time_limit, n_iters=200000):
    """
    Direct subgradient on the max of autocorrelation with Polyak step.
    Uses the fact that the peak is typically near the center.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    # Track best possible (lower bound from uniform)
    lb = 2.0 * n * (1.0 / n)  # = 2.0 for uniform... not useful
    # Use current best as reference for Polyak step
    best_val = fast_autocorr(a).max()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit - 2:
            break

        c = fast_autocorr(a)
        c_max = c.max()

        # Polyak step size: alpha = (current - best) / ||grad||^2
        # Subgradient at argmax: grad_i = 2 * a_{k*-i} (reversed at k*)
        # ||grad||^2 = 4 * sum a_{k*-i}^2 (valid) <= 4 * ||a||^2
        k_star = c.argmax()
        # Compute subgradient
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        grad = np.zeros(n)
        grad[idx] = 2.0 * a[k_star - idx]

        grad_norm_sq = np.dot(grad, grad)
        if grad_norm_sq < 1e-30:
            break

        alpha = (c_max - best_val) / grad_norm_sq
        alpha = min(alpha, 0.01)  # cap step size
        if alpha < 1e-12:
            break

        a_new = a - alpha * grad
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new /= sn

        new_c_max = fast_autocorr(a_new).max()
        if new_c_max < c_max - 1e-15:
            a = a_new
            if 2.0 * n * new_c_max < best_score:
                best_score = 2.0 * n * new_c_max
                best_a = a.copy()
                best_val = new_c_max

        # Occasionally try multiple peaks
        if (it + 1) % 100 == 0:
            top_k = np.argsort(c)[::-1][:5]
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                idx2 = np.arange(js2, je2 + 1)
                grad2 = np.zeros(n)
                grad2[idx2] = 2.0 * a[k2 - idx2]
                gn2 = np.dot(grad2, grad2)
                if gn2 < 1e-30:
                    continue
                alpha2 = min((c[k2] - best_val) / gn2, 0.005)
                if alpha2 < 1e-12:
                    continue
                a_try = a - alpha2 * grad2
                a_try = np.maximum(a_try, 0)
                st = a_try.sum()
                if st < 1e-15:
                    continue
                a_try /= st
                c_try_max = fast_autocorr(a_try).max()
                if c_try_max < c_max - 1e-15:
                    a = a_try
                    c_max = c_try_max
                    if 2.0 * n * c_max < best_score:
                        best_score = 2.0 * n * c_max
                        best_a = a.copy()
                        best_val = c_max
                    break

    return best_a, best_score


def peak_shaving_local(a0, time_limit, seed=42):
    """
    Local peak-shaving: identify top peaks and move mass from peak-contributing
    positions to safe positions.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 3:
        c = fast_autocorr(a)
        c_max = c.max()
        threshold = c_max * 0.999
        top_peaks = np.where(c >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(c)[::-1][:5]
        top_peaks = top_peaks[:10]

        improved = False
        for k in top_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src_local = np.argmax(contrib)
            src = idx[src_local]

            # Find safe destinations (positions not contributing to any top peak)
            danger = np.zeros(n, dtype=bool)
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                danger[js2:je2 + 1] = True

            safe = np.where(~danger)[0]
            safe = safe[safe != src]
            if len(safe) == 0:
                safe = np.where(np.arange(n) != src)[0]
            safe_sorted = safe[np.argsort(a[safe])]
            dst_pool = safe_sorted[:max(5, len(safe_sorted) // 10)]

            for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.05, 0.08, 0.12]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                for dst in dst_pool[np.random.default_rng(seed + int(frac*1000)).integers(0, len(dst_pool), size=min(3, len(dst_pool)))]:
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    a_trial /= a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 200:
                break
            # Small random perturbation to escape
            a_trial = a.copy()
            for _ in range(3):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a_trial[i] > 1e-10:
                    amt = a_trial[i] * rng.uniform(0.001, 0.01)
                    a_trial[i] -= amt
                    a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 1e-10:
                a_trial /= st
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 4)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.6 + 0.3 * np.cos(2 * x) + 0.1 * np.cos(7 * x)
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-8 * (x - 0.3) ** 2) + 0.7 * np.exp(-8 * (x - 0.7) ** 2)
    else:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.2 * np.cos(9 * np.pi * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

    remaining = time_budget - (time.monotonic() - t_global_start)
    t3 = min(remaining * 0.12, 150.0)
    print(f"=== Phase 3: Subgradient polish ({t3:.0f}s) ===")
    if t3 > 10:
        a_out, _ = subgradient_polish(best_seq, t3)
        try_update(a_out, "subgrad_1")

    # === Phase 4: Fast Nesterov game again (from improved start) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    t4 = min(remaining * 0.20, 300.0)
    print(f"=== Phase 4: NG round 2 n={n_base} ({t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = nesterov_game_fast(best_seq, t4, T=3.0, eta_inner=1.0, n_inner=30, check_interval=50)
        try_update(a_out, f"ng2_n{n_base}")

    # === Phase 5: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 5: Multi-length ({remaining:.0f}s left) ===")
    if remaining > 100:
        lengths_to_try = []
        # Try nearby lengths
        for delta in [-200, -100, 100, 200, 300]:
            nl = n_base + delta
            if nl >= 200 and nl <= 5000:
                lengths_to_try.append(nl)

        t_per_len = min(remaining * 0.5 / max(1, len(lengths_to_try)), 80.0)
        for nl in lengths_to_try:
            if time.monotonic() - t_global_start > time_budget - 100:
                break
            if t_per_len < 15:
                break
            print(f"  Trying n={nl} ({t_per_len:.0f}s each)")
            init_nl = interpolate_sequence(best_seq, nl)
            a_out, _ = nesterov_game_fast(init_nl, t_per_len, T=5.0, eta_inner=0.8, n_inner=40, check_interval=50)
            try_update(a_out, f"ng_n{nl}")
            # Quick subgradient polish
            if time.monotonic() - t_global_start < time_budget - 60:
                a_out2, _ = subgradient_polish(a_out, min(20.0, t_per_len * 0.3))
                try_update(a_out2, f"subgrad_n{nl}")

    # === Phase 6: Peak shaving on best ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    t6 = min(remaining * 0.15, 100.0)
    print(f"=== Phase 6: Peak shaving ({t6:.0f}s) ===")
    if t6 > 10:
        a_out, _ = peak_shaving_local(best_seq, t6, seed=777)
        try_update(a_out, "peak_shave_1")

    # === Phase 7: Final subgradient polish ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 15:
        t7 = min(remaining * 0.20, 80.0)
        print(f"=== Phase 7: Final subgradient ({t7:.0f}s) ===")
        a_out, _ = subgradient_polish(best_seq, t7)
        try_update(a_out, "subgrad_final")

    # === Phase 8: Final peak shave ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 10:
        t8 = min(remaining * 0.4, 50.0)
        a_out, _ = peak_shaving_local(best_seq, t8, seed=999)
        try_update(a_out, "peak_shave_2")

    # === Phase 9: One more NG round if time allows ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 30:
        t9 = min(remaining * 0.5, 60.0)
        print(f"=== Phase 9: Final NG ({t9:.0f}s) ===")
        a_out, _ = nesterov_game_fast(best_seq, t9, T=2.0, eta_inner=1.2, n_inner=25, check_interval=30)
        try_update(a_out, "ng_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
