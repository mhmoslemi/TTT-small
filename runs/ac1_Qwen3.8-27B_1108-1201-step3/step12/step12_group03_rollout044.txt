def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def get_lp_solver():

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):

        except Exception:

    return cp.ECOS


def build_conv_matrix(a_fixed):
    """Build the (2n-1) x n convolution matrix T where (T@b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0. Returns (b, M)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

        # = sum_j [delta_{j,i} * a[k-j] + a[j] * delta_{k-j,i}]
        # = a[k-i] + a[i]  (if both 0<=i<n and 0<=k-i<n)
        # = a[k-i] (if only one is valid)
        # = 0 (if neither is valid)
        
        # Wait: sum_j delta_{k-j,i} = 1 if there exists j with k-j=i, i.e., j=k-i, and 0<=k-i<n.
        # In that case the contribution is a[k-i] * 1 = a[k-i]... no wait.
        # sum_j a[j] * delta_{k-j,i} = a[k-i] (when j=k-i is valid)
        
        # So: db[k]/da[i] = a[k-i] * I(0<=i<n, 0<=k-i<n) + a[k-i] * I(0<=i<n, 0<=k-i<n)
        
        # Hmm wait no. Let me redo:
        # sum_j delta_{j,i} * a[k-j] = a[k-i] (if 0<=i<n and 0<=k-i<n, else 0)
        # sum_j a[j] * delta_{k-j,i} = a[j*] where j* = k-i, so = a[k-i] (if 0<=k-i<n and 0<=j*<n... 
        #   j* = k-i must satisfy 0<=k-i<n for a[k-i] to be defined, and j* must be in [0,n) which is the same condition)
        # So both give the same condition and value!
        
        # Therefore: db[k]/da[i] = 2 * a[k-i] (if 0<=i<n and 0<=k-i<n, else 0)
        
        # Wait, that can't be right because when k=0: b[0] = a[0]^2, db[0]/da[0] = 2*a[0]. And 2*a[0-0] = 2*a[0]. ✓
        # When k=1: b[1] = a[0]*a[1] + a[1]*a[0] = 2*a[0]*a[1]
        # db[1]/da[0] = 2*a[1]. Formula: 2*a[1-0] = 2*a[1]. ✓
        # db[1]/da[1] = 2*a[0]. Formula: 2*a[1-1] = 2*a[0]. ✓
        # When k=2, n>=3: b[2] = a[0]*a[2] + a[1]*a[1] + a[2]*a[0]
        # db[2]/da[0] = a[2] + a[2] = 2*a[2]. Formula: 2*a[2-0] = 2*a[2]. ✓
        # db[2]/da[1] = 2*a[1] (from product rule on a[1]^2). Formula: 2*a[2-1] = 2*a[1]. ✓
        # db[2]/da[2] = a[0] + a[0] = 2*a[0]. Formula: 2*a[2-2] = 2*a[0]. ✓
        
        # Great! So the formula is: grad[i] = sum_k softmax_w[k] * 2 * a[k-i] (for valid k-i)
        
        # This can be computed via cross-correlation of softmax_w and a!
        # grad = 2 * correlate(softmax_w, a) but with appropriate length handling.
        
        # Actually, let me think about it as: grad[i] = 2 * sum_k softmax_w[k] * a[k-i] * I(0<=k-i<n)
        # = 2 * (softmax_w * a)[i] in a certain convolution sense.
        
        # Let me just do it with FFT for efficiency.
        # Define c[k] = softmax_w[k] for k=0,...,2n-2, and a_padded[j] = a[j] for j=0,...,n-1.
        # Then grad[i] = 2 * sum_k c[k] * a[k-i] where 0<=k-i<n
        # = 2 * sum_{j=0}^{n-1} c[i+j] * a[j] (reindexing with j=k-i)
        # = 2 * (cross-correlation of c and a, evaluated at lag i)
        
        # This is the cross-correlation! I can compute it with FFT.
        
        # cross_corr[m] = sum_j c[j+m] * a[j]
        # I need cross_corr[i] for i=0,...,n-1.
        
        # Using FFT: cross_corr = ifft(conj(fft(c)) * fft(a)) ... hmm, need to be careful.
        # Actually, the linear cross-correlation of c and a at lag m is:
        #   sum_j c[j+m] * a[j]
        # This can be computed as the (m)-th element of the convolution of c_reversed with a.
        # Or: ifft(fft(c, N) * conj(fft(a, N))) gives the circular cross-correlation.
        
        # For linear cross-correlation, pad to size >= len(c) + len(a) - 1 = 2n-1 + n - 1 = 3n-2.
        
        fft_size_corr = 1 << (3 * n - 1).bit_length()
        C = rfft(softmax_w, fft_size_corr)
        A = rfft(a, fft_size_corr)
        cross_corr_full = irfft(C * np.conj(A), fft_size_corr)
        # cross_corr_full[m] = sum_j c[j] * conj(a[j-m]) for circular... 
        # Actually, irfft(fft(c) * conj(fft(a))) gives the circular cross-correlation:
        # cc[m] = sum_j c[j] * a[(j-m) mod N]
        # For our purpose with zero-padding, this gives the linear cross-correlation:
        # cc[m] = sum_j c[j] * a[j-m] (with a[k]=0 outside [0,n-1])
        
        # I want: grad[i] = 2 * sum_j c[i+j] * a[j] = 2 * sum_j c[j'] * a[j'-i] where j'=i+j
        # = 2 * sum_{j'} c[j'] * a[j'-i]
        # This is cc[i] where cc[m] = sum_{j'} c[j'] * a[j'-m]
        
        # And irfft(fft(c)*conj(fft(a))) gives sum_j c[j] * a[j-m] at position m.
        # So grad[i] = 2 * cross_corr_full[i] for i = 0, ..., n-1.
        
        # Wait, let me double-check the sign convention.
        # numpy's irfft(fft(c) * conj(fft(a))) at position m gives:
        #   sum_{j} c[j] * a[(j - m) mod N]
        # With zero-padding, for m in range: sum_j c[j] * a[j-m] (where a[k]=0 if k not in [0,n-1])
        
        # I need: sum_j c[i+j] * a[j] = sum_{j'} c[j'] * a[j'-i] (with j'=i+j)
        # = (c a-reversed) at lag i... 
        # Actually this is exactly cross_corr_full[i] as computed above!
        # Because cross_corr_full[m] = sum_j c[j] * a[j-m], and setting m=i:
        # cross_corr_full[i] = sum_j c[j] * a[j-i]
        # But I want sum_j c[i+j] * a[j] = sum_{j'} c[j'] * a[j'-i] (j'=i+j)
        
        # These are the same! cross_corr_full[i] = sum_j c[j] * a[j-i] = sum_{j'} c[j'] * a[j'-i] ✓
        
        grad = 2.0 * cross_corr_full[:n]
        
        # Project gradient onto tangent space of simplex: grad - mean(grad)
        grad_tangent = grad - grad.mean()
        
        # Step size
        lr = 0.01 / (1.0 + 0.1 * it)
        
        # Update: a_new = softmax(log(a) - lr * grad_tangent) (mirror descent)
        # Or: a_new = a - lr * grad_tangent, then project
        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a - lr * grad_tangent
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new_s = a_new.sum()
        if a_new_s < 1e-30:
            break
        a_new = a_new / a_new_s
        
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        
        if it % 200 == 0:
            print(f"      SoftGrad it={it} tau={tau:.1f}: best={best_score:.6f}")

    return best_a, best_score


def spectral_flatten(a0, time_limit):
    """
    Iteratively flatten the power spectrum to reduce autocorrelation peaks.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for it in range(500):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute FFT
        A = rfft(a, fft_size)
        A_magnitude = np.abs(A)
        A_phase = np.angle(A)
        
        # Target: flat magnitude (uniform power spectrum)
        # Use power compression: |A|^alpha for alpha < 1
        alpha = 0.8 + 0.15 * (it / 500)  # Start with aggressive compression, ease off
        A_new_mag = A_magnitude ** alpha
        
        # Reconstruct
        A_new = A_new_mag * np.exp(1j * A_phase)
        a_new = irfft(A_new, fft_size)[:n]
        
        # Project: clip negatives, normalize
        a_new = np.maximum(a_new, 0)
        a_new_s = a_new.sum()
        if a_new_s < 1e-15:
            break
        a_new = a_new / a_new_s
        
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        
        # If no improvement for a while, try different alpha
        if it % 100 == 99:
            alpha = 0.7 + 0.3 * np.random.random()
    
    return best_a, best_score


def qp_active_set_refinement(a0, time_limit, n_active=25):
    """
    QP refinement: identify active constraints (peaks near max) and solve QP.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    b = fast_autocorr(a)
    max_b = b.max()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * max_b

    for round_num in range(20):
        if time.monotonic() - t0 > time_limit:
            break

        threshold = max_b * (1 - 0.01)
        active_lags = np.where(b >= threshold)[0]
        if len(active_lags) > n_active:
            active_lags = np.argsort(b[active_lags])[::-1][:n_active]

        # Build QP
        a_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()

        constraints = [cp.sum(a_var) == 1, M_var >= 0]
        for k in active_lags:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            other_idx = k - idx
            q_expr = cp.quad_form(a_var, cp.diag(np.outer(
                np.ones(len(idx)), 1.0)))  # This won't work for cross terms
            # Actually, (a*a)_k = sum_j a[j]*a[k-j] which is a quadratic form
            # Let me use a different approach: just use the bilinear form directly
            # (a*a)_k = a[idx]^T @ a[other_idx] (sum of products)
            # In cvxpy, this is: cp.sum(a_var[idx] * a_var[other_idx])
            # But this is only convex if the matrix is PSD...
            # Actually, sum_j a[idx_j] * a[other_idx_j] is not a standard quad_form.
            # Let me use: cp.sum(cp.multiply(a_var[idx], a_var[other_idx]))
            # But cvxpy may not handle this directly unless it's a recognized convex expression.
            
            # Alternative: express as quad_form with the appropriate matrix.
            # (a*a)_k = a^T W_k a where W_k has W_k[i,j] = 1/2 if i+j=k (or i=k-j)
            # Actually (a*a)_k = sum_i a[i]*a[k-i] = a^T W_k a where W_k[i,j] = 1 if j=k-i, else 0
            # But this matrix is not symmetric and not PSD in general.
            
            # For cvxpy, I'll just use the product directly.
            constraints.append(cp.sum(a_var[idx] * a_var[other_idx]) <= M_var)

        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iter=10000)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, max_iter=10000)
            except Exception:
                break

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            try:
                prob.solve(solver=cp.CLARABEL)
            except Exception:
                break
            if prob.status not in ('optimal', 'optimal_inaccurate'):
                break

        a_new = np.asarray(a_var.value, dtype=np.float64)
        a_new = np.maximum(a_new, 0)
        a_new_s = a_new.sum()
        if a_new_s < 1e-15:
            break
        a_new = a_new / a_new_s

        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            b = fast_autocorr(a)
            max_b = b.max()
        else:
            for frac in [0.3, 0.5, 0.7]:
                a_mix = (1 - frac) * a + frac * a_new
                a_mix = np.maximum(a_mix, 0)
                a_mix_s = a_mix.sum()
                if a_mix_s < 1e-15:
                    continue
                a_mix = a_mix / a_mix_s
                score_mix = 2.0 * n * fast_autocorr(a_mix).max()
                if score_mix < best_score:
                    best_score = score_mix
                    best_a = a_mix.copy()
                    a = a_mix
                    b = fast_autocorr(a)
                    max_b = b.max()
                    break
            else:
                if round_num >= 5:
                    break
                n_active = min(n_active * 2, 50)

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=123):
    """
    Peak shaving coordinate descent with line search.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        n_peaks = min(5, 2 * n - 1)
        top_peaks = np.argsort(b)[::-1][:n_peaks]

        improved = False
        for k in top_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            
            # Find the largest contributor
            ci = np.argmax(contrib)
            src = idx[ci]
            src_other = k - src
            
            # Try moving mass from src to other positions
            for frac in [0.001, 0.003, 0.005, 0.01, 0.02]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                # Find best destination: position with minimum contribution to top peaks
                dst_candidates = np.argsort(a)[:20]
                for dst in dst_candidates:
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    if src != src_other:
                        a_trial[src_other] -= amt * 0.5
                    a_trial[dst] += amt
                    if (a_trial < -1e-12).any():
                        continue
                    a_trial = np.maximum(a_trial, 0)
                    a_trial_s = a_trial.sum()
                    if a_trial_s < 1e-15:
                        continue
                    a_trial = a_trial / a_trial_s
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 5 == 0:
                # Random perturbation
                n_perturb = rng.integers(1, 4)
                sources = rng.choice(n, size=n_perturb, replace=False)
                targets = rng.choice(n, size=n_perturb, replace=False)
                a_trial = a.copy()
                for src_idx, dst_idx in zip(sources, targets):
                    frac = rng.uniform(0.003, 0.02)
                    amt = a_trial[src_idx] * frac
                    a_trial[src_idx] -= amt
                    a_trial[dst_idx] += amt
                a_trial = np.maximum(a_trial, 0)
                a_trial_s = a_trial.sum()
                if a_trial_s > 1e-15:
                    a_trial = a_trial / a_trial_s
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > max_stall:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f
