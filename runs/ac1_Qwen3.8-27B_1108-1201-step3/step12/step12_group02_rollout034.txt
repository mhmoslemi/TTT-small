"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_autocorr(a):
    """Full autocorrelation a*a using FFT. Returns array of length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_score(a):
    """Compute 2n*max(b)/(sum(a))^2 quickly."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)


def cross_corr(w, a):
    """Compute cross-correlation: result[j] = sum_i w[j+i]*a[i]."""
    n = len(a)
    a_rev = a[::-1].copy()
    conv_len = n + len(w) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a_rev, fft_size)
    conv = irfft(W * A, fft_size)
    return conv[n - 1:n - 1 + n]


def lse_obj_grad_z(z, T):
    """Objective and gradient in z-space where a = softmax(z)."""
    n = len(z)
    z_shift = z - z.max()
    e = np.exp(z_shift)
    a = e / e.sum()

    b = fast_autocorr(a)
    Tb = T * b
    Tb_shift = Tb - Tb.max()
    exp_Tb = np.exp(Tb_shift)
    obj = (Tb.max() + np.log(exp_Tb.sum())) / T

    w = exp_Tb / exp_Tb.sum()
    cc = cross_corr(w, a)
    grad_a = 2.0 * cc

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    return obj, grad_z


def interpolated_update(a, b, alpha):
    """Interpolate between a and b (both normalized), return normalized result."""
    result = (1 - alpha) * a + alpha * b
    result = np.maximum(result, 0)
    s = result.sum()
    if s < 1e-15:
        return a.copy()
    return result / s


def mirror_descent_adaptive(a0, time_limit, eta0=0.04, k_min=1, k_max=30,
                            k_ramp=500, threshold_frac=0.95, seed=42):
    """
    Mirror descent with adaptive peak selection.
    Uses ALL peaks above threshold_frac * max_b (not just fixed top-k).
    Adapters k between k_min and k_max based on peak distribution.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall_count = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        b_max = b.max()

        # Adaptive peak selection: use peaks above threshold
        threshold = threshold_frac * b_max
        peak_mask = b >= threshold
        n_peaks = peak_mask.sum()

        # Blend with top-k for stability
        frac = min(1.0, t / float(k_ramp))
        k_eff = max(k_min, min(k_max, int(k_min + frac * (k_max - k_min))))
        top_k_idx = np.argsort(b)[::-1][:k_eff]

        # Combine: use top-k always, plus additional peaks above threshold
        all_peaks = np.unique(np.concatenate([top_k_idx, np.where(peak_mask)[0]]))

        # Gaussian/softmax weights based on peak values
        peak_vals = b[all_peaks]
        peak_shifted = peak_vals - peak_vals.max()
        temp = max(0.1, (peak_vals.max() - peak_vals.min()) * 5)
        weights = np.exp(peak_shifted / temp)
        weights /= weights.sum()

        # Compute subgradient
        grad = np.zeros(n)
        for ii in range(len(all_peaks)):
            m = int(all_peaks[ii])
            w_m = weights[ii] * 2.0
            j_start = max(0, m - n + 1)
            j_end = min(n - 1, m)
            js = np.arange(j_start, j_end + 1)
            grad[js] += w_m * a[m - js]

        # Adaptive learning rate
        grad_norm = np.sqrt(np.sum(grad * a))
        eta = eta0 / (1.0 + t / 200.0) ** 0.7
        # Scale by inverse gradient magnitude for stability
        if grad_norm > 1e-15:
            eta = min(eta, 0.5 / grad_norm)

        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        a_new = a_new / s_new

        a = a_new

        # Check score periodically
        if t < 200:
            check_interval = 2
        elif t < 2000:
            check_interval = 5
        else:
            check_interval = 20

        if t % check_interval == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stall_count = 0
            else:
                stall_count += 1

        # Stall handling: perturb and restart
        if stall_count > 300:
            noise_scale = 0.02 * (1 + 0.1 * (stall_count / 100.0))
            a = a + noise_scale * rng.standard_normal(n)
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            stall_count = 0

    return best_a, best_score


def lse_refine(a0, time_limit, T_start=50.0, T_end=0.2, n_stages=10, seed=42):
    """LogSumExp L-BFGS refinement with multi-stage annealing."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s

    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = fast_score(a_init)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        iters = max(20, min(100, int(40 + 60 * si / max(1, n_stages))))

        def func_z(z_in):
            val, _ = lse_obj_grad_z(z_in, stage_T)
            return val

        def jac_z(z_in):
            _, g = lse_obj_grad_z(z_in, stage_T)
            return g

        try:
            res = minimize(fun=func_z, x0=z, jac=jac_z, method='L-BFGS-B',
                           options={'maxiter': iters, 'ftol': 1e-18,
                                    'gtol': 1e-12, 'maxfun': iters * 300,
                                    'disp': False})
            z = res.x
            z_shift = z - z.max()
            a = np.exp(z_shift)
            a = a / a.sum()
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        except Exception:
            pass

    return best_a, best_score


def peak_equalization(a0, time_limit, seed=42):
    """
    Novel peak equalization: directly solve for adjustments that
    simultaneously reduce the top autocorrelation peaks.
    Uses a "dual variable" approach inspired by KKT conditions.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    outer_stall = 0

    while time.monotonic() - t0 < time_limit:
        if outer_stall > 500:
            break

        b = fast_autocorr(a)
        b_max = b.max()

        # Find all significant peaks (above 80% of max)
        threshold = 0.80 * b_max
        peak_indices = np.where(b >= threshold)[0]

        if len(peak_indices) == 0:
            break

        # For each peak, compute its "pressure" gradient
        # grad_k[j] = 2 * a[k-j] (for valid j)
        # We want to find da such that all top peaks decrease equally
        # and sum(da) = 0, a + da >= 0

        # Build the pressure vector: weighted sum of peak gradients
        # Weights are proportional to peak excess
        excesses = b[peak_indices] - b[threshold:] if threshold > 0 else np.ones(len(peak_indices))
        # Simpler: weight by peak value
        peak_vals = b[peak_indices]
        weights = peak_vals / peak_vals.sum()

        # Total gradient pressure
        pressure = np.zeros(n)
        for ii in range(len(peak_indices)):
            m = int(peak_indices[ii])
            w = weights[ii] * 2.0
            j_start = max(0, m - n + 1)
            j_end = min(n - 1, m)
            js = np.arange(j_start, j_end + 1)
            pressure[js] += w * a[m - js]

        # Find positions to reduce (high pressure, high a) and increase (low pressure)
        # Use a "flow" approach: move mass from high-pressure to low-pressure regions
        pressure_centered = pressure - pressure.mean()

        # Propose: reduce a where pressure is high, increase where low
        delta = -pressure_centered
        delta -= delta.mean()  # Ensure sum = 0

        # Scale delta to be small
        max_delta = np.abs(delta).max()
        if max_delta < 1e-15:
            outer_stall += 1
            continue
        delta = delta / max_delta * 0.05

        # Try multiple step sizes with line search
        improved = False
        for step_frac in [0.1, 0.25, 0.5, 0.75, 1.0, 1.5, 2.0]:
            da = step_frac * delta
            a_trial = a + da
            a_trial = np.maximum(a_trial, 0)
            s_trial = a_trial.sum()
            if s_trial < 1e-15:
                continue
            a_trial = a_trial / s_trial
            sc = fast_score(a_trial)
            if sc < best_score - 1e-15:
                a = a_trial.copy()
                best_score = sc
                best_a = a.copy()
                improved = True
                outer_stall = 0
                break

        if not improved:
            outer_stall += 1
            # Random perturbation to escape
            if outer_stall % 50 == 0:
                n_perturb = max(3, n // 50)
                idx = rng.choice(n, size=n_perturb, replace=False)
                a[idx] *= (1 + 0.1 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0)
                s = a.sum()
                if s > 1e-15:
                    a = a / s
            if outer_stall % 200 == 0 and outer_stall > 0:
                # Larger restart perturbation
                a = a + 0.05 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                s = a.sum()
                if s > 1e-15:
                    a = a / s

    return best_a, best_score


def coordinate_descent_fine(a0, time_limit, seed=42):
    """
    Fine coordinate descent: for each of the top peaks, find the best
    pair (src, dst) to move mass, with exact line search.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        if stall > 300:
            break

        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:min(15, n)]
        improved = False

        for k_idx in top_k:
            k_idx = int(k_idx)
            js = max(0, k_idx - n + 1)
            je = min(n - 1, k_idx)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_idx - idx]
            total_contrib = contrib.sum()

            if total_contrib < 1e-15:
                continue

            # Source: position with highest contribution
            src_local = int(np.argmax(contrib))
            src = int(idx[src_local])

            # Target: find position that minimizes increase in max peak
            # Heuristic: positions with low a value, far from peak centers
            peak_region = np.zeros(n, dtype=bool)
            for k2 in top_k[:8]:
                k2 = int(k2)
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_region[js2:je2 + 1] = True

            candidates = np.where(~peak_region)[0]
            if len(candidates) < 10:
                candidates = np.where(np.arange(n) != src)[0]

            # Sort by a value (ascending) - move to low-value positions
            sorted_cands = candidates[np.argsort(a[candidates])[:min(20, len(candidates))]]
            if len(sorted_cands) == 0:
                continue

            dst = int(sorted_cands[rng.integers(len(sorted_cands))])

            # Try multiple transfer amounts
            for frac in [0.001, 0.003, 0.008, 0.02, 0.05, 0.1, 0.2]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                sc = fast_score(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial.copy()
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break

            if improved:
                break

        if not improved:
            stall += 1
            # Random pair move
            if rng.integers(5) == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.05)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        a_trial = a_trial / a_trial.sum()
                        sc = fast_score(a_trial)
                        if sc < best_score:
                            a = a_trial.copy()
                            best_score = sc
                            best_a = a.copy()
                            stall = 0

    return best_a, best_score


def diverse_init(n, rng, variant=None):
    """Generate diverse structured initializations with tilt emphasis."""
    if variant is None:
        variant = int(rng.integers(0, 12))

    x = np.linspace(0, 1, n)

    if variant == 0:
        # Linear tilt (high left, low right)
        a = 1 + 0.8 * (1 - 2 * x)
    elif variant == 1:
        # Stronger tilt
        a = 1 + 1.5 * (1 - 2 * x)
    elif variant == 2:
        # Sigmoid tilt
        a = 1 / (1 + np.exp(-6 * (x - 0.5)))
        a = a - a.min() + 0.1
    elif variant == 3:
        # Step function (half high, half low)
        a = np.where(x < 0.5, 1.5, 0.5)
    elif variant == 4:
        # Power-law tilt
        a = x ** 0.5 + 0.3
    elif variant == 5:
        # Inverse power-law
        a = (1 - x) ** 0.5 + 0.3
    elif variant == 6:
        # Gaussian centered off-center
        a = np.exp(-8 * (x - 0.3) ** 2) + 0.2
    elif variant == 7:
        # Two-bump asymmetric
        a = np.exp(-15 * (x - 0.2) ** 2) + 0.6 * np.exp(-15 * (x - 0.7) ** 2) + 0.1
    elif variant == 8:
        # U-shaped
        a = x ** 2 + (1 - x) ** 2 + 0.1
    elif variant == 9:
        # V-shaped (dip center)
        a = abs(x - 0.5) + 0.2
    elif variant == 10:
        # Cubic tilt
        a = 1 + 0.5 * (1 - x) ** 2 - 0.3 * x ** 2
    else:
        # Random smooth
        a = np.ones(n)
        for _ in range(4):
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.1, n * 0.4)
            amp = rng.uniform(-0.5, 1.0)
            xi = np.arange(n, dtype=np.float64)
            a += amp * np.exp(-(xi - center) ** 2 / (2 * width ** 2))
        a = np.maximum(a, 0.01)

    a = np.maximum(a, 1e-6)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
