def fast_autocorr(a):
    """Full self-convolution a*a using FFT, O(n log n)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_cross_conv(a, b):
    """Cross-convolution a*b using FFT."""
    n = max(len(a), len(b))
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def softmax_fn(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def lse_obj_and_grad(z, T):
    """LogSumExp objective and gradient in z-space."""
    a = softmax_fn(z)
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    obj_val = (max_Tb + np.log(exp_Tb.sum())) / T
    
    w = exp_Tb / exp_Tb.sum()
    
    # grad_a[i] = 2 * sum_k w[k] * a[k-i] = 2 * (w * a_rev)[i + n - 1]
    a_rev = a[::-1].copy()
    conv_len = (2 * n - 1) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    A_rev = rfft(a_rev, fft_size)
    conv = irfft(W * A_rev, fft_size)
    grad_a = 2.0 * conv[n - 1:n - 1 + n]
    
    # Transform to z-space
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj_val, grad_z

def logsumexp_optimize(a0, time_limit, T_start=100.0, T_end=0.01, n_stages=25):
    """LogSumExp with L-BFGS and multi-stage annealing."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = compute_score(a_init)
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        iters_per_stage = max(20, min(100, int(30 + 70 * (si / max(1, n_stages - 1)))))
        
        def func(z_in):
            val, _ = lse_obj_and_grad(z_in, stage_T)
            return val
        
        def jac(z_in):
            _, g = lse_obj_and_grad(z_in, stage_T)
            return g
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': iters_per_stage * 200,
                         'disp': False}
            )
            z = res.x
            a = softmax_fn(z)
            score = compute_score(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
        except Exception:
            pass
        
        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage={si} T={stage_T:.4f} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def gradient_coord_descent(a0, time_limit, seed=42):
    """
    Gradient-guided coordinate descent.
    Uses peak responsibility to guide mass redistribution.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        b_max = b.max()
        
        # Compute peak weights (soft max)
        temp = 0.005 * b_max
        w = np.exp((b - b_max) / temp)
        w = w / w.sum()
        
        # Compute "responsibility" of each position
        # resp[i] = sum_k w[k] * a[i] * a[k-i] (contrib to weighted peaks)
        a_rev = a[::-1].copy()
        conv_len = (2 * n - 1) + n - 1
        fft_size = 1 << (conv_len - 1).bit_length()
        W = rfft(w, fft_size)
        A_rev = rfft(a_rev, fft_size)
        conv = irfft(W * A_rev, fft_size)
        resp = conv[n - 1:n - 1 + n]
        
        # Source: highest resp[i] / a[i] (most dangerous per unit mass)
        # Target: lowest resp[j] (least contributing to peaks)
        src_scores = resp / np.maximum(a, 1e-15)
        tgt_scores = resp + 0.01 * a
        
        improved = False
        
        # Try top 5 sources and top 10 targets
        src_candidates = np.argsort(src_scores)[::-1][:5]
        tgt_candidates = np.argsort(tgt_scores)[:10]
        
        for src in src_candidates:
            if a[src] < 1e-12:
                continue
            for dst in tgt_candidates:
                if src == dst:
                    continue
                for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    score = compute_score(a_trial)
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Random exploration
            for _ in range(30):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.001, 0.03)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = compute_score(a_trial)
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            
            if not improved:
                stall += 1
                if stall > 200:
                    break
        
        if (time.monotonic() - t0) % 30 < 1:
            elapsed = time.monotonic() - t0
            print(f"    GCD ({elapsed:.0f}s): best={best_score:.8f}, stall={stall}")
    
    return best_a, best_score

def multiplicative_weights_best_response(a_fixed, n_iters=200, beta=50.0):
    """
    Fast best response using multiplicative weights (no LP).
    b minimizes max_j (a*b)_j approximately.
    """
    n = len(a_fixed)
    b = np.ones(n) / n
    t0 = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > 20:  # Per-call time limit
            break
        
        c = fast_cross_conv(a_fixed, b)
        c_max = c.max()
        
        # Soft peak indicator
        w = np.exp(beta * (c - c_max) / max(c_max, 1e-10))
        w = w / w.sum()
        
        # Gradient of max(c) w.r.t. b: grad_b[j] = sum_i w[i] * a_fixed[i-j]
        # = (w * a_rev)[j + n - 1]
        a_rev = a_fixed[::-1].copy()
        conv_len = len(c) + n - 1
        fft_size = 1 << (conv_len - 1).bit_length()
        W = rfft(w, fft_size)
        A_rev = rfft(a_rev, fft_size)
        conv = irfft(W * A_rev, fft_size)
        grad_b = conv[n - 1:n - 1 + n]
        
        # Multiplicative update
        mu = 0.5
        b = b * np.exp(-mu * grad_b / max(grad_b.max(), 1e-10))
        b = np.maximum(b, 1e-15)
        b = b / b.sum()
    
    return b

def nesterov_mw_optimize(a0, time_limit, max_iters=5000):
    """Nesterov with multiplicative weights (faster than LP)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    y = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_star = multiplicative_weights_best_response(y, n_iters=100, beta=50.0)
        
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        ss = y_new.sum()
        if ss < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / ss
        
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            ss = a_new.sum()
            if ss < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / ss
        
        y = y_new
        a = a_new
        
        score_a = compute_score(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        score_y = compute_score(y)
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NMW t={t} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def structured_init(n, rng):

    elif choice == 4:

        a = np.maximum(0, 1 - 2 * abs(x - 0.5)) + 0.2 * np.sin(5 * np.pi * x)
        a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def fast_autocorr(a):
    """Full self-convolution a*a using FFT, O(n log n)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_conv(a, b):
    """Cross-convolution a*b using FFT."""
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max_conv/(sum_a)^2."""
    n = len(a)
    b = fast_autocorr(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * b.max() / (s * s)

def build_conv_matrix_subsampled(a_fixed, n_constraints):
    """Build subsampled convolution matrix for LP speed."""
    n = len(a_fixed)
    size = 2 * n - 1
    if size <= n_constraints:
        idx = np.arange(size)
    else:
        idx = np.linspace(0, size - 1, n_constraints).astype(int)
    T = np.zeros((len(idx), n))
    for row, j in enumerate(idx):
        js = max(0, j - n + 1)
        je = min(n - 1, j)
        T[row, js:je + 1] = a_fixed[js:je + 1]
    return T


def solve_lp_best_response(a_fixed, solver, n_constraints):
    """LP: min M s.t. T(a_fixed)@b <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix_subsampled(a_fixed, n_constraints)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    b_val = b_val / s
    return b_val

def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def lse_obj_and_grad(z, T):
    """LogSumExp objective and gradient in z-space."""
    a = softmax_fn(z)
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    obj_val = (max_Tb + np.log(exp_Tb.sum())) / T
    w = exp_Tb / exp_Tb.sum()
    a_rev = a[::-1].copy()
    conv_len = (2 * n - 1) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    A_rev = rfft(a_rev, fft_size)
    conv = irfft(W * A_rev, fft_size)
    grad_a = 2.0 * conv[n - 1:n - 1 + n]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj_val, grad_z

            break

        if remaining < 3:
            break

        # Reheating: if stuck, increase T
        if stall_stages >= 3 and si < len(temps) - 1:
            stage_T = max(stage_T * 3.0, temps[min(si, len(temps) - 1)] * 2.0)
            stall_stages = 0
            print(f"    LSE reheating at stage {si}: T={stage_T:.4f}")

        prev_score = best_score

                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18,

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2:
    Phase 1: Nesterov MW (fast global) + Nesterov LP (refined) - 30% time
    Phase 2: LogSumExp with aggressive annealing - 20% time
    Phase 3: Gradient coordinate descent - 30% time
    Phase 4: Aggressive peak shave + multi-length + final polish - 20% time
    """
    rng = np.random.default_rng(12345)
    t_start = time.monotonic()
    time_budget = 1740.0

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    n_base = 1600

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            n_base = len(parent)
            p_score = evaluate_safe(parent)
            print(f"Parent: n={n_base}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        print("No valid parent, using default initialization")

    if best_seq is None:
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    # === Phase 1: Nesterov (MW then LP) ===
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 1: Nesterov at n={n_base} (remaining={remaining:.0f}s) ===")

    # 1a: Fast MW Nesterov
    t_mw = min(remaining * 0.15, 200.0)
    if t_mw > 20:
        a_out, _ = nesterov_mw_optimize(best_seq, t_mw, max_iters=5000)
        best_seq, best_score = try_update_state(best_seq, best_score, a_out, "nmw", t_start)

    # 1b: LP Nesterov from current best
    remaining = time_budget - (time.monotonic() - t_start)
    t_lp = min(remaining * 0.15, 200.0)
    if t_lp > 20:
        a_out, _ = nesterov_lp_optimize(best_seq, t_lp, solver, max_iters=2000)
        best_seq, best_score = try_update_state(best_seq, best_score, a_out, "nlp", t_start)

    # === Phase 2: LogSumExp ===
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 2: LogSumExp at n={n_base} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 15:
        a_out, _ = logsumexp_optimize(best_seq, t2, T_start=100.0, T_end=0.008, n_stages=25)
        best_seq, best_score = try_update_state(best_seq, best_score, a_out, "lse", t_start)

    # === Phase 3: Gradient coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 3: Gradient CD at n={n_base} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.30, 450.0)
    if t3 > 15:
        a_out, _ = gradient_coord_descent(best_seq, t3, seed=42)
        best_seq, best_score = try_update_state(best_seq, best_score, a_out, "gcd", t_start)

    # === Phase 4: Aggressive peak shave + multi-length + final polish ===
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 4: Peak shave + polish (remaining={remaining:.0f}s) ===")

    # 4a: Aggressive peak shave
    t4a = min(remaining * 0.25, 200.0)
    if t4a > 10:
        a_out, _ = aggressive_peak_shave(best_seq, t4a, seed=777)
        best_seq, best_score = try_update_state(best_seq, best_score, a_out, "aps", t_start)

    # 4b: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_start)
    if remaining > 60:
        for delta_n in [100, -100, 200]:
            remaining = time_budget - (time.monotonic() - t_start)
            if remaining < 30:
                break
            nl = n_base + delta_n
            if nl < 800 or nl > 3000:
                continue
            t_sub = min(remaining * 0.12, 80.0)
            if t_sub < 10:
                continue
            init_n = interpolate_sequence(best_seq, nl)
            a_out, _ = gradient_coord_descent(init_n, t_sub, seed=42 + delta_n)
            best_seq, best_score = try_update_state(best_seq, best_score, a_out, f"gcd_n{nl}", t_start)

    # 4c: Final LogSumExp polish
    remaining = time_budget - (time.monotonic() - t_start)
    if remaining > 20:
        t4c = min(remaining * 0.25, 80.0)
        a_out, _ = logsumexp_optimize(best_seq, t4c, T_start=20.0, T_end=0.003, n_stages=15)
        best_seq, best_score = try_update_state(best_seq, best_score, a_out, "final_lse", t_start)

    # 4d: Final coordinate descent
    remaining = time_budget - (time.monotonic() - t_start)
    if remaining > 10:
        t4d = min(remaining * 0.5, 80.0)
        a_out, _ = gradient_coord_descent(best_seq, t4d, seed=999)
        best_seq, best_score = try_update_state(best_seq, best_score, a_out, "final_gcd", t_start)

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return result
