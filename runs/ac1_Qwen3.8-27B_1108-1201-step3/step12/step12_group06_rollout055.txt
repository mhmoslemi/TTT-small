def nesterov_lp(a0, time_limit, solver=None, max_iters=1000000, jump_interval=2000, jump_noise=0.02):
    """Nesterov-accelerated LP with periodic jumps for exploration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * (it + 1) * (it + 1))) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        # Periodic jump: add noise to escape local minima
        if it > 0 and it % jump_interval == 0:
            noise = rng.normal(0, jump_noise, n)
            a_new = a_new + noise
            a_new = np.maximum(a_new, 0)
            sa = a_new.sum()
            if sa > 1e-15:
                a_new = a_new / sa
            # Reset momentum on jump
            y = a_new.copy()
            continue

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            remain = time_limit - elapsed
            print(f"    NLP it={it+1} ({elapsed:.0f}s/{time_limit:.0f}s): best={best_score:.8f} cur={sc_a:.8f} M={M_lp:.6f}")

    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    # Gradient: grad_a[j] = 2 * sum_{m: 0<=m<n} w[j+m] * a[m]
    # Use FFT: conv(w, a_rev)[n-1+j]
    a_rev = a[::-1]
    conv_w_ar = fast_convolve2(w, a_rev)
    grad_a = 2.0 * conv_w_ar[n - 1:2 * n - 1]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lsbfgs_lse(a0, time_limit, T_start=80.0, T_min=0.2, n_stages=30, iters_per_stage=50):
    """LSE L-BFGS with geometric temperature decay."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    T = T_start
    stagnant = 0
    decay = (T_min / T_start) ** (1.0 / max(1, n_stages))

    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, T, n)
            return grad

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-12, 'maxfun': iters_per_stage * 200,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)

            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()

            # Adaptive temperature
            if best_score < 1.51 and T > T_min:
                T = max(T_min, T * 0.7)
            elif stagnant >= 4:
                T = max(T_start * 0.3, T * 1.3)
                stagnant = 0
            else:
                T = max(T_min, T * decay * 1.2)

            if (stage + 1) % 5 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LSE stage={stage+1}/{n_stages} T={T:.2f}: cur={sc:.8f} best={best_score:.8f} ({elapsed:.0f}s)")

        except Exception:
            T = max(T_min, T * 0.8)

    return best_a, best_score


def peak_subgradient(a0, time_limit, n_steps=500000):
    """Peak-targeted subgradient with Polyak step and iterate averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    avg_a = a.copy()
    avg_count = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b

        threshold = max_b * 0.99
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        if len(active) > 30:
            top_idx = np.argsort(b[active])[-30:]
            active = active[top_idx]

        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        S = np.dot(g, a)
        grad_z = a * (g - S)

        grad_norm = np.linalg.norm(grad_z)
        if grad_norm < 1e-15:
            break

        lower_bound = 2.0 * n / (2.0 * n - 1.0)
        gap = cur_score - lower_bound
        if gap < 1e-10:
            break
        eta = gap / (2.0 * grad_norm * grad_norm)
        eta = min(eta, 0.3)

        z_new = z - eta * grad_z
        a_new = softmax_from_z(z_new)
        sc_new = score_normalized(a_new)

        if sc_new > cur_score * 1.02:
            lo_e, hi_e = 0.0, eta
            for _ in range(6):
                mid_e = (lo_e + hi_e) / 2.0
                z_mid = z - mid_e * grad_z
                a_mid = softmax_from_z(z_mid)
                if score_normalized(a_mid) < cur_score:
                    lo_e = mid_e
                else:
                    hi_e = mid_e
            eta = lo_e
            z_new = z - eta * grad_z
            a_new = softmax_from_z(z_new)
            sc_new = score_normalized(a_new)

        z = z_new
        a = a_new

        avg_a = (avg_count * avg_a + a) / (avg_count + 1)
        avg_count += 1
        if avg_count > 3000:
            avg_count = 3000
            avg_a = (avg_a * 3000 + a) / 3001

        if sc_new < best_score:
            best_score = sc_new
            best_a = a.copy()

        sc_avg = score_normalized(avg_a)
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = avg_a.copy()

        if (step + 1) % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PS step={step+1} ({elapsed:.0f}s): cur={cur_score:.8f} best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    choice = rng.integers(0, 10)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-3 * x ** 2)
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif choice == 4:
        a = np.ones(n) * (1.0 + 0.3 * np.cos(2 * np.pi * 5 * np.arange(n) / n))
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.2 * np.cos(3 * x) + 0.15 * np.cos(7 * x) + 0.1 * np.cos(13 * x)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = x * (1 - x) * (1 + 0.3 * np.sin(4 * np.pi * x))
    elif choice == 7:
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.3) ** 2) / 0.03) + np.exp(-((x - 0.7) ** 2) / 0.03) + 0.1
    elif choice == 8:
        x = np.linspace(0, 1, n)
        a = 1.0 / (x + 0.1) * (1 - x)
    else:
        x = np.linspace(0, 1, n)
        a = np.abs(np.sin(3 * np.pi * x)) * np.abs(np.cos(2 * np.pi * x)) + 0.1
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function. Returns best sequence found within budget."""
    rng = np.random.default_rng(seed)
    global_deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    t_start = time.monotonic()

    solver = get_lp_solver()
    print(f"LP solver: {solver}, budget: {budget_s}s")

    # Get starting sequence
    best_seq = None
    best_score = float('inf')
    n_base = 1500

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Init: n={n_base}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            remain = global_deadline - time.monotonic()
            print(f"  ** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, remain={remain:.0f}s)")

    # === Phase 1: Nesterov LP main (50% time) ===
    print(f"\n=== Phase 1: Nesterov LP n={n_base} score={best_score:.8f} ===")
    t1 = 0.50 * max(60, budget_s - 25)
    remain = global_deadline - time.monotonic()
    t1 = min(t1, remain * 0.6)
    a_out, _ = nesterov_lp(best_seq, t1, solver=solver, jump_interval=3000, jump_noise=0.015)
    try_update(a_out, "nlp_main")
    print(f"  Phase 1 done: {best_score:.8f}")

    # === Phase 2: Restart Nesterov from perturbed (15% time) ===
    print(f"\n=== Phase 2: Perturbed Nesterov ===")
    remain = global_deadline - time.monotonic()
    t2 = min(0.12 * (budget_s - 25), remain * 0.4)
    if t2 > 30:
        for attempt in range(3):
            remain = global_deadline - time.monotonic()
            if remain < 25:
                break
            noise_scale = 0.03 + 0.02 * attempt
            perturbed = best_seq + noise_scale * rng.normal(0, 1, len(best_seq))
            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()
            t2_sub = min(t2 / 3.0, remain * 0.5)
            a_p, _ = nesterov_lp(perturbed, t2_sub, solver=solver, jump_interval=2000, jump_noise=0.01)
            try_update(a_p, f"nlp_pert{attempt}")
    print(f"  Phase 2 done: {best_score:.8f}")

    # === Phase 3: LSE polishing (20% time) ===
    print(f"\n=== Phase 3: LSE Polish score={best_score:.8f} ===")
    remain = global_deadline - time.monotonic()
    t3 = min(0.18 * (budget_s - 25), remain * 0.5)
    if t3 > 30:
        a_lse1, _ = lsbfgs_lse(best_seq, t3 * 0.5, T_start=60.0, T_min=0.15, n_stages=25, iters_per_stage=40)
        try_update(a_lse1, "lse_main")

        remain = global_deadline - time.monotonic()
        if remain > 20:
            a_lse2, _ = lsbfgs_lse(best_seq, remain * 0.5, T_start=20.0, T_min=0.1, n_stages=20, iters_per_stage=35)
            try_update(a_lse2, "lse_lowT")
    print(f"  Phase 3 done: {best_score:.8f}")

    # === Phase 4: Peak subgradient (10% time) ===
    print(f"\n=== Phase 4: Peak Subgradient ===")
    remain = global_deadline - time.monotonic()
    t4 = min(0.08 * (budget_s - 25), remain * 0.4)
    if t4 > 20:
        a_sg, _ = peak_subgradient(best_seq, t4, n_steps=300000)
        try_update(a_sg, "peak_sg")

        remain = global_deadline - time.monotonic()
        if remain > 15:
            a_sg2, _ = peak_subgradient(best_seq, remain * 0.5, n_steps=200000)
            try_update(a_sg2, "peak_sg2")
    print(f"  Phase 4 done: {best_score:.8f}")

    # === Phase 5: Multi-length exploration (5% time) ===
    print(f"\n=== Phase 5: Multi-length ===")
    remain = global_deadline - time.monotonic()
    t5_total = min(0.05 * (budget_s - 25), remain * 0.6)
    if t5_total > 30:
        for n_try in [n_base + 300, n_base - 200]:
            remain = global_deadline - time.monotonic()
            if remain < 25:
                break
            t5 = min(t5_total / 2.0, remain * 0.5)
            if t5 < 15:
                break
            print(f"  Trying n={n_try}")
            init_try = interpolate_sequence(best_seq, n_try)
            a_try, _ = nesterov_lp(init_try, t5 * 0.55, solver=solver, max_iters=100000)
            try_update(a_try, f"len_{n_try}_nlp")
            remain = global_deadline - time.monotonic()
            if remain > 15 and a_try is not None:
                a_try2, _ = lsbfgs_lse(a_try, remain * 0.5, T_start=30.0, T_min=0.2, n_stages=10, iters_per_stage=25)
                try_update(a_try2, f"len_{n_try}_lse")
    print(f"  Phase 5 done: {best_score:.8f}")

    # === Phase 6: Final combined polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    remain = global_deadline - time.monotonic()
    if remain > 20:
        a_f1, _ = nesterov_lp(best_seq, remain * 0.35, solver=solver, max_iters=50000, jump_interval=5000, jump_noise=0.008)
        try_update(a_f1, "final_nlp")

        remain = global_deadline - time.monotonic()
        if remain > 15:
            a_f2, _ = lsbfgs_lse(best_seq, remain * 0.5, T_start=15.0, T_min=0.1, n_stages=15, iters_per_stage=30)
            try_update(a_f2, "final_lse")

        remain = global_deadline - time.monotonic()
        if remain > 10:
            a_f3, _ = peak_subgradient(best_seq, remain * 0.6, n_steps=100000)
            try_update(a_f3, "final_sg")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    elapsed = time.monotonic() - t_start
    print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={elapsed:.0f}s")
    return [float(x) for x in best_seq]
