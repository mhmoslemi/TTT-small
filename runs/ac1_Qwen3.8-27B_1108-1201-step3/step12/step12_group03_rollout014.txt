import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)


def score_max_only(a):
    """Compute max(conv(a,a)) assuming sum(a)=1."""
    return fast_autocorr(a).max()


def normalize(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def get_lp_solver():
    """Find a working fast LP solver."""
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            t0 = time.monotonic()
            x = cp.Variable(100, nonneg=True)
            constraints = [cp.sum(x) == 1, x <= 0.5]
            prob = cp.Problem(cp.Minimize(x[0] + 0.1 * cp.sum(x)))
            prob.solve(solver=solver_name)
            elapsed = time.monotonic() - t0
            if prob.status in ('optimal', 'optimal_inaccurate'):
                if solver_name == cp.CLARABEL:
                    return cp.CLARABEL
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a_fixed):
    """Build the (2n-1) x n convolution matrix T where (T@b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. conv(a_fixed, b)_k <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.CLARABEL, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else None
    return b_val, M_val


def nesterov_lp(a0, time_limit, solver, verbose=False):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    a = normalize(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = score_max_only(a)
    t0 = time.monotonic()

    # EMA tracking
    ema = a.copy()
    ema_alpha = 0.03
    vel = np.zeros(n)

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue

        # Nesterov step
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_new = y_new / max(y_new.sum(), 1e-15)

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            vel = momentum * vel + (1 - momentum) * (y_new - y)
            a_new = y_new + 0.5 * vel
            a_new = np.maximum(a_new, 0)
            a_new = a_new / max(a_new.sum(), 1e-15)

        y = y_new
        a = a_new

        sc = score_max_only(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        # Also check y
        sc_y = score_max_only(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        # EMA
        ema = (1 - ema_alpha) * ema + ema_alpha * a
        if t % 10 == 0:
            sc_ema = score_max_only(ema)
            if sc_ema < best_score:
                best_score = sc_ema
                best_a = ema.copy()

        if verbose and t % 200 == 0:
            print(f"    NLP t={t}: best={best_score:.8f}")

    return best_a, best_score


def mirror_descent_anneal(a0, time_limit, verbose=False):
    """Mirror descent with temperature annealing in log-space."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = score_max_only(a)
    t0 = time.monotonic()

    n_iters = 8000
    velocity = np.zeros(n)
    momentum = 0.9

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        progress = it / max(1, n_iters)
        # Temperature annealing: beta increases from 5 to 100
        beta = 5.0 + 95.0 * (progress ** 1.5)
        # Learning rate: starts at 0.08, decays
        lr = 0.08 * (0.001 / 0.08) ** (progress ** 0.8)

        b = fast_autocorr(a)
        max_b = b.max()

        # Soft-max subgradient: weight peaks by exp(beta * (b[k] - max_b))
        shifted = beta * (b - max_b)
        weights = np.exp(shifted - shifted.max())
        weights = weights / weights.sum()

        # Compute gradient: weighted average of shifts of a
        grad = np.zeros(n)
        # Use FFT-based approach for efficiency
        # grad[j] = sum_k weights[k] * a[k-j] (for valid indices)
        # This is: grad = convolve(weights, a) restricted to [0, n)
        # Actually: grad[j] = sum_{k valid} weights[k] * a[k-j] + weights[j-k]... 
        # Let me use the direct formula: grad[j] = sum_k w[k] * a[k-j] where 0<=k-j<n and 0<=k<2n-1
        # Equivalently: grad = ifft(fft(w, 2n-1) * fft(a, 2n-1)) but that's circular...
        # Let's just do it directly for the top peaks (most weight)
        top_k = min(20, len(b))
        idx_top = np.argsort(weights)[::-1][:top_k]
        for k in idx_top:
            w = weights[k]
            if w < 1e-10:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            other = k - idx
            grad[idx] += w * a[other]
            grad[other] += w * a[idx]

        # Mirror descent step with momentum
        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a - lr * (grad + momentum * velocity)
        velocity = grad + momentum * velocity
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / max(a_new.sum(), 1e-30)

        sc = score_max_only(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()

        a = a_new

        if verbose and it % 500 == 0:
            print(f"    MD it={it}, beta={beta:.1f}: best={best_score:.8f}")

    return best_a, best_score


def mc_local_search(a0, time_limit, seed=123, verbose=False):
    """Monte Carlo local search with simulated annealing targeting top peaks."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = score_max_only(a)
    t0 = time.monotonic()

    # Simulated annealing schedule
    T_start = 0.0005
    T_end = 0.000001
    n_total = int(time_limit * 8000)  # estimate ~8000 iterations/sec

    iterations = 0
    eval_count = 0

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, iterations / max(1, n_total))
        T = T_start * (T_end / T_start) ** (progress ** 0.7)

        b = fast_autocorr(a)
        max_b = b.max()

        # Top peaks
        n_peaks = min(5, len(b))
        top_peaks = np.argsort(b)[::-1][:n_peaks]

        # Choose a random strategy
        r = rng.random()
        if r < 0.5:
            # Targeted move: reduce a top peak
            k = top_peaks[rng.integers(0, n_peaks)]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            total = contrib.sum()
            if total < 1e-15:
                continue
            # Pick a contributor with probability proportional to its contribution
            probs = contrib / total
            ci = rng.choice(len(idx), p=probs)
            src = idx[ci]
            src_other = k - src

            # Find target: random position far from peak contributors
            # Score positions by their (negative) contribution to top peaks
            target_scores = np.zeros(n)
            for k2 in top_peaks[:3]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                idx2 = np.arange(js2, je2 + 1)
                # Position j contributes to peak k2 if j is in [js2, je2] or k2-j is in [0, n-1]
                # Simple: reduce score for positions that appear in peak computation
                for j in idx2:
                    target_scores[j] += a[k2 - j] if 0 <= k2 - j < n else 0
                for j in range(n):
                    if 0 <= k2 - j < n:
                        target_scores[j] += a[k2 - j]

            # Prefer positions with low target_scores (far from peaks)
            # Invert and add noise
            target_scores_inv = -target_scores + rng.normal(0, 0.1, n)
            dst = np.argmax(target_scores_inv)
        elif r < 0.8:
            # Random 2-move: swap mass between two random positions
            src = rng.integers(0, n)
            dst = rng.integers(0, n)
            if src == dst:
                continue
            src_other = src
        else:
            # Random single move
            src = rng.integers(0, n)
            dst = rng.integers(0, n)
            if src == dst:
                continue
            src_other = src

        # Amount to move
        amt = a[src] * rng.uniform(0.01, 0.08)
        if amt < 1e-12:
            continue

        a_trial = a.copy()
        a_trial[src] -= amt
        if src_other != src:
            a_trial[src_other] -= amt * 0.5
        a_trial[dst] += amt
        if (a_trial < -1e-12).any():
            continue
        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-15:
            continue
        a_trial = a_trial / s

        sc = score_max_only(a_trial)
        eval_count += 1

        # Accept/reject
        delta = sc - score_max_only(a)
        if delta < 0 or rng.random() < np.exp(-delta / T):
            a = a_trial
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        iterations += 1

        if verbose and iterations % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MC iter={iterations}: best={best_score:.8f}, cur={score_max_only(a):.8f}")

    return best_a, best_score


def peak_shaving_gradient(a0, time_limit, verbose=False):

    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()

            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            ci = np.argmax(contrib)
            src = idx[ci]
            src_other = k - src

            # Find best destination: minimize increase in max convolution
            # Try a few candidate destinations
            candidates = rng.choice(n, size=min(20, n), replace=False)

            best_dst_score = best_score

                for frac in [0.02, 0.05, 0.1]:

                    if amt < 1e-12:

                    a_trial = a.copy()
                    a_trial[src] -= amt

                    if (a_trial < -1e-12).any():

                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / max(a_trial.sum(), 1e-15)
                    sc = score_max_only(a_trial)
                    if sc < best_dst_score:
                        best_dst_score = sc
                        best_dst = (dst, frac, amt)

            if best_dst is not None:
                dst, frac, amt = best_dst
                a_trial = a.copy()
                a_trial[src] -= amt
                if src != src_other:

            sc = score_max_only(a_trial)
            if sc < best_score:
                a = a_trial
                best_score = sc
                best_a = a.copy()
            else:

def generate_structural_inits(n, rng):
    """Generate diverse structured initializations."""
    inits = []
    x = np.linspace(0, 1, n)

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine bell
    f = np.cos(np.pi * (x - 0.5)) ** 2
    inits.append(f / f.sum())

    # Alternating (asymmetric)
    for c in [0.3, 0.5, 0.7]:
        f = 1.0 + c * np.cos(2 * np.pi * x)
        f = np.maximum(f, 0)
        if f.sum() > 1e-10:
            inits.append(f / f.sum())

    # Higher frequency alternation
    for m in [2, 3, 5]:
        for c in [0.3, 0.5]:
            f = 1.0 + c * np.cos(2 * np.pi * m * x + rng.uniform(0, 2 * np.pi))
            f = np.maximum(f, 0)
            if f.sum() > 1e-10:
                inits.append(f / f.sum())

    # Geometric (asymmetric)
    for r in [0.7, 0.8, 0.85, 0.9]:
        f = r ** (np.arange(n) * 2.0 / (n - 1))
        inits.append(f / f.sum())
        f = r ** ((n - 1 - np.arange(n)) * 2.0 / (n - 1))
        inits.append(f / f.sum())

    # Sine-modulated
    for m in [1, 2, 3]:
        for phi in [0, np.pi / 4, np.pi / 2]:
            f = 1.0 + 0.4 * np.sin(2 * np.pi * m * x + phi)
            f = np.maximum(f, 0)
            if f.sum() > 1e-10:
                inits.append(f / f.sum())

    # Two-bump (asymmetric)
    for center1, center2, width in [(0.2, 0.7, 0.1), (0.1, 0.8, 0.15), (0.25, 0.75, 0.08)]:
        f = np.exp(-((x - center1) / width) ** 2) + np.exp(-((x - center2) / width) ** 2)
        inits.append(f / f.sum())

    # Tilted
    for tilt in [0.5, 1.0, 2.0]:
        f = 1.0 + tilt * (x - 0.5)
        f = np.maximum(f, 0)
        if f.sum() > 1e-10:
            inits.append(f / f.sum())

    # Random
    f = rng.random(n)
    inits.append(f / f.sum())

    # Parent (if available)
    return inits


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: multi-phase optimization combining structured initialization,
    LP-Nesterov, annealed mirror descent, Monte Carlo local search, and
    gradient-based peak shaving.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    def elapsed():
        return time.monotonic() - t_start

    def remaining():
        return time_budget - elapsed()

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}, time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = score_max_only(a_norm)
        full_sc = 2.0 * len(a_norm) * sc
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_seq = a_norm.copy()
            print(f"  *** NEW BEST [{label}]: {full_sc:.6f} (n={len(a_norm)}, t={elapsed():.0f}s)")

    # Phase 0: Get parent or default
    n_base = 1600
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = 2.0 * len(parent) * score_max_only(parent)
            print(f"Parent: n={len(parent)}, score={parent_score:.6f}")
            best_seq = parent.copy()
            best_score = parent_score
            n_base = len(parent)
    except Exception:
        pass

    if parent is None:
        x = np.linspace(0, np.pi, n_base)
        best_seq = np.cos(x / 2) ** 2
        best_seq = normalize(best_seq)
        best_score = 2.0 * n_base * score_max_only(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # Phase 1: Diverse initialization + quick LP to find best basin
    print(f"\n=== Phase 1: Diverse init + quick LP (n={n_base}) ===")
    inits = generate_structural_inits(n_base, rng)
    if parent is not None:
        inits.append(parent.copy())

    # Evaluate all inits
    init_scores = []
    for i, init in enumerate(inits):
        sc = 2.0 * n_base * score_max_only(init)
        init_scores.append((sc, i))
    init_scores.sort()
    print(f"  Top inits: {[f'{s:.5f}' for s, _ in init_scores[:5]]}")

    # Quick LP on top 3 inits
    t1_total = min(remaining() * 0.12, 180.0)
    t1_each = t1_total / 3.0
    for rank in range(3):
        if remaining() < 30:
            break
        _, idx = init_scores[rank]
        init = inits[idx]
        t1 = min(t1_each, remaining() - 20)
        if t1 < 10:
            continue
        print(f"  Quick LP on init {rank} (score={init_scores[rank][0]:.5f}), t={t1:.0f}s")
        a_out, sc_out = nesterov_lp(init, t1, solver)
        try_update(a_out, f"quick_lp_{rank}")

    # Phase 2: Extended LP-Nesterov on best
    print(f"\n=== Phase 2: Extended LP-Nesterov (n={len(best_seq)}) ===")
    t2 = min(remaining() * 0.22, 350.0)
    if t2 > 30:
        a_out, sc_out = nesterov_lp(best_seq, t2, solver, verbose=True)
        try_update(a_out, "lp_extended")

    # Phase 3: Mirror descent with annealing
    print(f"\n=== Phase 3: Mirror Descent Annealing (n={len(best_seq)}) ===")
    t3 = min(remaining() * 0.12, 200.0)
    if t3 > 20:
        a_out, sc_out = mirror_descent_anneal(best_seq, t3, verbose=True)
        try_update(a_out, "md_anneal")

    # Phase 4: Monte Carlo local search (main workhorse)
    print(f"\n=== Phase 4: MC Local Search (n={len(best_seq)}) ===")
    t4 = min(remaining() * 0.30, 500.0)
    if t4 > 20:
        a_out, sc_out = mc_local_search(best_seq, t4, seed=456, verbose=True)
        try_update(a_out, "mc_search")

    # Phase 5: Peak shaving on best
    print(f"\n=== Phase 5: Peak Shaving (n={len(best_seq)}) ===")
    t5 = min(remaining() * 0.12, 200.0)
    if t5 > 15:
        a_out, sc_out = peak_shaving_gradient(best_seq, t5, verbose=True)
        try_update(a_out, "peak_shave")

    # Phase 6: Second MC round (shorter, lower temperature)
    remaining_now = remaining()
    if remaining_now > 60:
        print(f"\n=== Phase 6: MC Local Search round 2 ===")
        t6 = min(remaining_now * 0.35, 250.0)
        if t6 > 15:
            a_out, sc_out = mc_local_search(best_seq, t6, seed=789, verbose=True)
            try_update(a_out, "mc_search_2")

    # Phase 7: Multi-length exploration
    remaining_now = remaining()
    if remaining_now > 100:
        print(f"\n=== Phase 7: Multi-length exploration ===")
        for n_try in [n_base - 100, n_base + 100, n_base + 200]:
            if n_try < 200 or remaining() < 60:
                continue
            print(f"  Trying n={n_try}")
            init_try = interpolate_to_length(best_seq, n_try)
            t7 = min(remaining() * 0.25, 120.0)
            if t7 < 20:
                continue
            # Quick LP + MC
            a_lp, _ = nesterov_lp(init_try, min(t7 * 0.5, 60.0), solver)
            sc_lp = 2.0 * n_try * score_max_only(a_lp)
            if sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.copy()
                print(f"    LP at n={n_try}: {sc_lp:.6f}")
            # MC refinement
            t_mc = min(remaining() * 0.15, 60.0)
            if t_mc > 10:
                a_mc, _ = mc_local_search(a_lp, t_mc, seed=n_try)
                sc_mc = 2.0 * n_try * score_max_only(a_mc)
                if sc_mc < best_score:
                    best_score = sc_mc
                    best_seq = a_mc.copy()
                    print(f"    MC at n={n_try}: {sc_mc:.6f}")

    # Phase 8: Final peak shaving
    remaining_now = remaining()
    if remaining_now > 30:
        print(f"\n=== Phase 8: Final refinement ===")
        t8 = min(remaining_now * 0.4, 120.0)
        if t8 > 10:
            a_out, sc_out = peak_shaving_gradient(best_seq, t8)
            try_update(a_out, "final_shave")
        remaining_now = remaining()
        if remaining_now > 20:
            t8b = min(remaining_now * 0.5, 80.0)
            if t8b > 10:
                a_out, sc_out = mc_local_search(best_seq, t8b, seed=1234)
                try_update(a_out, "final_mc")

    # Final verification
    if best_seq is not None:
        final_score = 2.0 * len(best_seq) * score_max_only(best_seq)
        print(f"\nFINAL SCORE: {final_score:.6f} (n={len(best_seq)}, t={elapsed():.0f}s)")
        result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    else:
        print("\nERROR: No valid sequence found!")
        result = [1.0]

    return result
