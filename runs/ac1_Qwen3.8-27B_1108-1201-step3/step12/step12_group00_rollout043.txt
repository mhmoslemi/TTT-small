```python
def nesterov_lp(a0, time_limit, solver_method='highs', verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    y = a.copy()
    v_prev = None
    t0 = time.monotonic()
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute current M
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        
        # LP best response
        b = lp_best_response(y, M_curr, method=solver_method)
        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y > 1e-15:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a > 1e-15:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        # Check score
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        # Restart if objective increased significantly
        if t > 10 and score_y > M_curr * 1.01:
            y = best_a.copy()
            v_prev = None
            # Reset t (optional)
        
        if verbose and t % 50 == 0:
            print(f"    LP iter={t} ({time.monotonic()-t0:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score
```

```python
    active = set(top_idx.tolist())
    
    for round in range(max_rounds):
        active_list = np.array(sorted(active))
        c = -np.ones(n)
        A_ub = T[active_list, :]
        b_ub = np.full(len(active_list), M)
        bounds = [(0, None)] * n
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status != 0:
            return None
        
        b = np.maximum(res.x, 0.0)
        
        # Check for violated constraints
        Tb = T @ b
        violated = np.where(Tb > M * (1 + 1e-6))[0]
        
        if len(violated) == 0:
            return b
        
        # Add violated constraints (limit to top violations)
        if len(violated) > 50:
            violated = violated[np.argsort(Tb[violated])[::-1][:50]]
        active.update(violated.tolist())
    
    # Return result even if not fully converged
    return b
```

```python
2. LSE (Log-Sum-Exp) smoothed gradient descent with FFT-based gradients

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

_global_best_seq = None
_global_best_score = float('inf')


def fast_autocorr(a):
    """Compute full autocorrelation (convolution of a with itself) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the global evaluate_sequence function."""
    try:
        a_list = [float(x) for x in a]
        for x in a_list:
            if x < 0 or x != x or x == float('inf'):
                return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def build_conv_matrix(a):
    """Build dense convolution matrix T where (T@b)[k] = (a*b)[k]."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, max_rounds=5, initial_frac=0.15):
    """
    Solve LP: max sum(b) s.t. T*b <= M, b >= 0
    Using cutting-plane method for efficiency.
    Returns optimal b or None on failure.
    """
    n = len(a)
    T = build_conv_matrix(a)
    conv = fast_autocorr(a)

    # Initial active set: top fraction of convolution peaks
    k_init = max(20, int(n * initial_frac))
    top_idx = np.argsort(conv)[::-1][:k_init]
    active = set(top_idx.tolist())

    b = None
    for rnd in range(max_rounds):
        active_list = np.array(sorted(active))
        c = -np.ones(n)
        A_ub = T[active_list, :]
        b_ub = np.full(len(active_list), M)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        except Exception:
            return None

        if res.status != 0:
            return None

        b = np.maximum(res.x, 0.0)

        # Check for violated constraints
        Tb = T @ b
        violated = np.where(Tb > M * (1.0 + 1e-8))[0]

        if len(violated) == 0:
            return b

        # Add most violated constraints (limit growth)
        if len(violated) > 100:
            worst = violated[np.argsort(Tb[violated])[::-1][:100]]
        else:
            worst = violated
        active.update(worst.tolist())

    return b


def normalize_to_norm(a, target_norm):
    """Normalize a to have sum = target_norm, keeping non-negativity."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * target_norm / len(a)
    return a * target_norm / s


def nesterov_lp_optimize(a0, time_limit, verbose=True):
    """
    Nesterov-accelerated LP best-response optimization.
    Minimizes max(f*f) subject to f >= 0, sum(f) = sqrt(2n).
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    t0 = time.monotonic()
    iter_count = 0
    restart_count = 0
    max_consecutive_worse = 0
    consecutive_worse = 0

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, max_rounds=4, initial_frac=0.12)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        iter_count += 1

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
            s_y = y.sum()
        y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
                s_a = s_y
            a_new = a_new * norm / s_a

        # Check scores
        score_a = fast_autocorr(a_new).max()
        score_y = fast_autocorr(y_new).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a_new.copy()
            consecutive_worse = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y_new.copy()
            consecutive_worse = 0

        # Backtracking: if momentum made things worse, restart
        if score_a > M_curr and score_y > M_curr:
            consecutive_worse += 1
        else:
            consecutive_worse = 0

        if consecutive_worse >= 5:
            # Restart from best
            y = best_a.copy()
            a = best_a.copy()
            consecutive_worse = 0
            restart_count += 1
            # Reduce effective t by resetting partway
            t = max(1, t // 2)
            continue

        y = y_new
        a = a_new

        if verbose and iter_count % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP iter={iter_count} ({elapsed:.0f}s): best={best_score:.8f}, restarts={restart_count}")

    return best_a, best_score


def lse_gradient_fft(a, T_temp):
    """
    Compute gradient of LSE surrogate f_T(a) = (1/T)log(sum exp(T*conv_k))
    using FFT. Returns gradient vector of length n.
    """
    n = len(a)
    conv = fast_autocorr(a)
    max_c = conv.max()
    # Numerical stability
    w = np.exp(T_temp * (conv - max_c))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w = w / w_sum

    # Gradient = (2/T) * (T^T @ w) where (T^T @ w)[i] = sum_j a[j]*w[j+i]
    # This is cross-correlation of a and w
    N = 1 << (3 * n).bit_length()
    A = np.fft.fft(a, N)
    W = np.fft.fft(w, N)
    R = np.fft.ifft(A * np.conj(W)).real[:n]

    return (2.0 / T_temp) * R


def lse_gradient_direct(a, T_temp):
    """Direct computation of LSE gradient (for verification or small n)."""
    n = len(a)
    conv = fast_autocorr(a)
    max_c = conv.max()
    w = np.exp(T_temp * (conv - max_c))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w = w / w_sum

    # grad[i] = (2/T) * sum_k w[k] * 2 * a[k-i] ... = (2/T) * (T^T w)[i]
    # (T^T w)[i] = sum_k T[k,i] * w[k] = sum_k a[k-i]*w[k] (where valid)
    T_mat = build_conv_matrix(a)
    return (2.0 / T_temp) * (T_mat.T @ w)


def lse_anneal(a0, time_limit, T_start=50.0, T_end=0.5, n_stages=8, verbose=True):
    """LSE smoothed gradient descent with temperature annealing."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        # Adaptive step size: start larger, decrease
        base_lr = 1.0 / (T * np.sqrt(n))
        n_steps = max(30, min(200, int(remaining * 8)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            grad = lse_gradient_fft(a, T)
            if not np.all(np.isfinite(grad)):
                break

            # Adaptive step with backtracking
            lr = base_lr / (1.0 + 0.05 * it)
            for _ in range(10):
                a_new = a - lr * grad
                a_new = np.maximum(a_new, 0.0)
                s = a_new.sum()
                if s < 1e-15:
                    break
                a_new = a_new * norm / s
                score = fast_autocorr(a_new).max()
                if score < fast_autocorr(a).max() * 1.001:
                    a = a_new
                    if score < best_score:
                        best_score = score
                        best_a = a.copy()
                    break
                lr *= 0.5

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"      LSE stage {si+1}/{n_stages} T={T:.2f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, verbose=True):
    """
    Coordinate descent: for each coordinate i, find optimal value
    that minimizes max_k conv_k given all other coordinates fixed.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximize = None  # ensure float
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    stall = 0
    n_full = 0

    while time.monotonic() - t0 < time_limit:
        if stall > 50:
            break

        improved_this_pass = False
        # Random permutation for diversification
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            # Compute base convolution (without a[i]'s contribution)
            a[i] = 0  # temporarily zero
            conv_base = fast_autocorr(a)
            a[i] = best_a[i]  # restore

            # c_k = a[k-i] + a[i-k] (the coefficient of a[i] in conv_k)
            c = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                if 0 <= k - i < n:
                    c[k] += a[k - i]
                if 0 <= i - k < n:
                    c[k] += a[i - k]

            # Now conv_k = conv_base_k + a[i] * c_k
            # We want to find a[i] >= 0 that minimizes max_k (conv_base_k + a[i]*c_k)
            # subject to the normalization (we'll renormalize after)

            # The optimal a[i] balances the peaks
            # Try several candidate values
            current_val = a[i]
            candidates = [0.0, current_val * 0.3, current_val * 0.7, current_val,
                          current_val * 1.3, current_val * 1.7]

            # Also try the "balanced" value: equalize top peaks
            top5 = np.argsort(conv_base)[::-1][:5]
            for k_star in top5:
                if c[k_star] > 1e-15:
                    # Want conv_base[k_star] + t*c[k_star] = target
                    # Estimate target as median of top peaks
                    target = np.median(conv_base[top5])
                    t_bal = (target - conv_base[k_star]) / c[k_star]
                    if t_bal > 0:
                        candidates.append(t_bal)

            best_t = current_val
            best_s = best_score

            for t_val in candidates:
                if t_val < 0:
                    continue
                a[i] = t_val
                conv_test = conv_base + t_val * c
                sc = conv_test.max()
                if sc < best_s:
                    best_s = sc
                    best_t = t_val

            a[i] = best_t
            # Renormalize
            s = a.sum()
            if s > 1e-15:
                a = a * norm / s
                sc = fast_autocorr(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    improved_this_pass = True

        if improved_this_pass:
            stall = 0
        else:
            stall += 1
            # Random perturbation to escape
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-8:
                frac = rng.uniform(0.05, 0.2)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-15:
                    a = a * norm / s
                    sc = fast_autocorr(a).max()
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
                        stall = 0

        n_full += 1
        if verbose and n_full % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      CD pass={n_full} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n)
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return a_interp / ss


def structured_init(n, rng):

    choice = rng.integers(0, 6)

    elif choice == 1:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(2.0 * x) + 0.15 * np.cos(4.0 * x) + 0.08 * np.cos(6.0 * x)
    elif choice == 2:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-5.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-5.0 * (x - 0.7) ** 2)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.sqrt(np.sin(np.pi * x))
    else:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(3.0 * x) + 0.1 * np.cos(7.0 * x)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

    t2 = min(remaining * 0.10, 120.0)
    if t2 > 10:
        n_best = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_anneal(a_norm, t2, T_start=30.0, T_end=0.5, n_stages=6, verbose=True)
        update_best(a_out, f"lse_n{n_best}")

    # === Phase 3: LP Nesterov at n+400 ===
    n2 = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: LP Nesterov n={n2} ===")
    t3 = min(remaining * 0.25, 350.0)
    if t3 > 30:
        init_long = interpolate_sequence(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t3, verbose=True)
        update_best(a_out, f"lp_n{n2}")

    # === Phase 4: LSE polish at n2 ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    n_best_now = len(best_seq)
    print(f"\n=== Phase 4: LSE polish n={n_best_now} ===")
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best_now) / best_seq.sum()
        a_out, _ = lse_anneal(a_norm, t4, T_start=10.0, T_end=0.3, n_stages=5, verbose=True)
        update_best(a_out, f"lse2_n{n_best_now}")

    # === Phase 5: Try longer n ===
    n3 = n_base + 800
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: LP Nesterov n={n3} ===")
    t5 = min(remaining * 0.20, 280.0)
    if t5 > 30:
        init_long3 = interpolate_sequence(best_seq, n3)
        a_norm = init_long3 * np.sqrt(2.0 * n3) / init_long3.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t5, verbose=True)
        update_best(a_out, f"lp_n{n3}")

    # === Phase 6: Coordinate descent on best ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    n_best_now = len(best_seq)
    print(f"\n=== Phase 6: Coordinate descent n={n_best_now} ===")
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best_now) / best_seq.sum()
        a_out, _ = coordinate_descent(a_norm, t6, verbose=True)
        update_best(a_out, f"cd_n{n_best_now}")

    # === Phase 7: Final LSE polish ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    n_best_now = len(best_seq)
    print(f"\n=== Phase 7: Final LSE n={n_best_now} ===")
    t7 = min(remaining * 0.20, 150.0)
    if t7 > 5:
        a_norm = best_seq * np.sqrt(2.0 * n_best_now) / best_seq.sum()
        a_out, _ = lse_anneal(a_norm, t7, T_start=5.0, T_end=0.1, n_stages=5, verbose=True)
        update_best(a_out, f"lse_final_n{n_best_now}")

    # === Phase 8: Final coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    n_best_now = len(best_seq)
    print(f"\n=== Phase 8: Final CD n={n_best_now} ===")
    t8 = min(remaining * 0.30, 150.0)
    if t8 > 5:
        a_norm = best_seq * np.sqrt(2.0 * n_best_now) / best_seq.sum()
        a_out, _ = coordinate_descent(a_norm, t8, verbose=True)
        update_best(a_out, f"cd_final_n{n_best_now}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        _global_best_score = best_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global_start:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
```
