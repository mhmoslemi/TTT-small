def hankel_matvec(w, a):
    """Compute Q@a where Q[i,j] = w[i+j]. Uses FFT."""
    n = len(a)
    a_rev = a[::-1]
    # (Q@a)[i] = sum_j w[i+j]*a[j] = (w * a_rev)[i + n - 1]
    fft_size = 1 << (len(w) + n - 1 - 1).bit_length()
    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)
    conv = irfft(W * AR, fft_size)
    return conv[n - 1:n - 1 + n]

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    # Algorithm from Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def compute_lse_grad(a, T):
    """Compute gradient of LSE objective w.r.t. a."""
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    
    a_rev = a[::-1].copy()
    fft_size = 1 << (n + 2 * n - 2 - 1).bit_length()
    W = rfft(w, fft_size)
    A_rev = rfft(a_rev, fft_size)
    conv = irfft(W * A_rev, fft_size)
    
    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad, w

def lse_z_to_obj_grad(z, T):
    """Objective and gradient in z-space for LSE."""
    a = softmax_fn(z)
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    obj_val = (max_Tb + np.log(exp_Tb.sum())) / T
    
    grad_a, _ = compute_lse_grad(a, T)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj_val, grad_z

def adam_lse_optimize(a0, time_limit, T_start=200.0, T_end=0.01, n_stages=25, lr=0.01):
    """Adam-optimized LSE with multi-stage annealing."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        # More iterations at lower temperatures
        iters = max(50, min(500, int(100 + 400 * (si / max(1, n_stages - 1)))))
        
        t_step = 0
        for it in range(iters):
            t_step += 1
            if time.monotonic() - t0 > time_limit:
                break
            
            obj, grad_z = lse_z_to_obj_grad(z, stage_T)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * (grad_z ** 2)
            m_hat = m / (1 - beta1 ** t_step)
            v_hat = v / (1 - beta2 ** t_step)
            z_new = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Keep z in reasonable range to avoid numerical issues
            z_new = np.clip(z_new, -30, 30)
            z = z_new
            
            # Check a-space score periodically
            if it % 20 == 0:
                a = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
        
        a = softmax_fn(z)
        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      AdamLSE stage={si}/T={stage_T:.4f} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def coordinate_descent_polish(a0, time_limit, seed=42):
    """Coordinate descent: try moving mass from position i to position j."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    # Precompute the convolution for quick updates
    b = fast_autocorr(a)
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        b_max = b.max()
        
        # Find the peak position
        k_peak = np.argmax(b)
        
        # Sources: positions contributing to the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        if len(idx) == 0:
            break
        contrib = a[idx] * a[k_peak - idx]
        
        # Sort by contribution (largest first)
        top_src = idx[np.argsort(contrib)[::-1][:5]]
        
        improved = False
        for src in top_src:
            if a[src] < 1e-10:
                continue
            
            # Find destinations that minimize increase in contribution to top peaks
            # Get top K peaks
            top_k = np.argsort(b)[::-1][:10]
            
            # Score destinations
            dest_scores = np.zeros(n)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                partner = k2 - np.arange(js2, je2 + 1)
                valid = (partner >= 0) & (partner < n)
                for j2 in range(js2, je2 + 1):
                    if valid[j2 - js2]:
                        dest_scores[j2] += a[partner[j2 - js2]]
            
            # Exclude current src and positions already contributing a lot
            dest_scores[src] = 1e10
            best_dsts = np.argsort(dest_scores)[:10]
            
            for dst in best_dsts:
                if dst == src:
                    continue
                for frac in [0.001, 0.003, 0.01, 0.03, 0.1]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random exploration
            for _ in range(3):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.001, 0.02)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= -1e-15:
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
                            break
            if stall > 200:
                break
        
        if stall % 100 == 50:
            elapsed = time.monotonic() - t0
            print(f"      CD stall={stall} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def solve_active_set_lp(a_fixed, b_current, solver, K=50):

def fast_autocorr(a):
    """Full self-convolution a*a using FFT, O(n log n)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence using the provided evaluate_sequence."""
    try:

        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where (T@b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def get_lp_solver():

    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:

            prob.solve(solver=solver_name, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_active_set(a_fixed, b_current, solver, K=100):
    """
    Solve LP with only K active constraints (top K peaks of b_current's autocorr).
    min M s.t. T[k] @ b <= M for k in active_set, sum(b)=1, b>=0
    This is much faster than the full LP with 2n-1 constraints.
    """
    n = len(a_fixed)
    # Identify active set from b_current's autocorrelation
    b_ac = fast_autocorr(b_current)
    active_peaks = np.argsort(b_ac)[::-1][:K]

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)

    constraints = [cp.sum(b_var) == 1]
    for k in active_peaks:
        constraints.append(T[k] @ b_var <= M_var)

    prob = cp.Problem(cp.Minimize(M_var), constraints)

    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        pass

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:

    b_val = b_val / s
    M_val = float(M_var.value)
    return b_val, M_val

def hankel_matvec(w, a):
    """Compute Q@a where Q[i,j] = w[i+j]. Uses FFT. w has length 2n-1."""
    n = len(a)
    a_rev = a[::-1]
    fft_size = 1 << (len(w) + n - 1 - 1).bit_length()
    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)
    conv = irfft(W * AR, fft_size)
    return conv[n - 1:n - 1 + n]


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def qp_refine(a0, time_limit, lr_init=0.005, n_inner=100, w_power=3.0):
    """
    QP-based refinement: iteratively solve min a^T Q_w a / 2 s.t. a in simplex
    where Q_w[i,j] = w[i+j]. Uses projected gradient with FFT matvec.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    lr = lr_init

    for outer in range(100):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute peak-emphasized weight distribution
        b = fast_autocorr(a)
        b_pos = np.maximum(b, 0)
        w = b_pos ** w_power
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            w = w / w_sum

        # Solve inner QP via projected gradient
        a_iter = a.copy()
        for inner in range(n_inner):
            if time.monotonic() - t0 > time_limit:
                break
            grad = hankel_matvec(w, a_iter)
            a_new = simplex_project(a_iter - lr * grad)
            diff = np.max(np.abs(a_new - a_iter))
            a_iter = a_new
            if diff < 1e-13:
                break

        # Evaluate and update
        score = 2.0 * n * fast_autocorr(a_iter).max()
        if score < best_score - 1e-14:
            a = a_iter
            best_score = score
            best_a = a.copy()
            lr = max(lr * 0.97, 1e-6)
        else:
            # Try blending
            a_blend = 0.5 * a + 0.5 * a_iter
            a_blend = simplex_project(a_blend)
            score_blend = 2.0 * n * fast_autocorr(a_blend).max()
            if score_blend < best_score - 1e-14:
                a = a_blend
                best_score = score_blend
                best_a = a.copy()
            else:
                lr = max(lr * 0.7, 1e-7)
            if lr < 2e-6:
                # Reduce w_power to be less aggressive
                w_power = max(1.5, w_power * 0.8)

        if outer % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"      QP o={outer} ({elapsed:.0f}s): best={best_score:.8f}, lr={lr:.2e}")

    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, solver, use_active=True, K=100, max_iters=50000):
    """Nesterov-accelerated LP best-response with active-set option."""
    n = len(a0)

    s = a.sum()
    if s < 1e-15:

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        if use_active:
            b_star, M_val = solve_lp_active_set(y, a, solver, K=K)
        else:
            b_star, M_val = solve_lp_full(y, solver)
        if b_star is None:
            continue

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        ss = y_new.sum()
        if ss < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / ss

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            ss = a_new.sum()
            if ss < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / ss

        y = y_new
        a = a_new

        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        avg_a += a
        n_avg += 1
        if n_avg >= 20 and n_avg % 10 == 0:
            avg_check = avg_a / n_avg
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()

        if t % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M_val:.10f}")

    return best_a, best_score


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def compute_lse_grad_and_weights(a, T):
    """Compute gradient of LSE(T * max_conv) w.r.t. a, and the weight distribution."""
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()

    a_rev = a[::-1].copy()
    fft_size = 1 << (n + 2 * n - 2 - 1).bit_length()
    W = rfft(w, fft_size)
    A_rev = rfft(a_rev, fft_size)
    conv = irfft(W * A_rev, fft_size)

    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad, w


def adam_lse_optimize(a0, time_limit, T_start=200.0, T_end=0.005, n_stages=30, lr=0.005):
    """Adam-optimized LSE with multi-stage annealing."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t_step = 0

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        # More iterations at lower temperatures where refinement matters
        frac = si / max(1, n_stages - 1)
        iters = max(50, min(800, int(80 + 700 * frac)))

        for it in range(iters):
            t_step += 1
            if time.monotonic() - t0 > time_limit:
                break

            a_curr = softmax_fn(z)
            n_curr = len(a_curr)
            b = fast_autocorr(a_curr)
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()

            # Gradient in a-space
            a_rev = a_curr[::-1].copy()
            fft_size = 1 << (n_curr + 2 * n_curr - 2 - 1).bit_length()
            W = rfft(w, fft_size)
            A_rev = rfft(a_rev, fft_size)
            conv = irfft(W * A_rev, fft_size)
            grad_a = 2.0 * conv[n_curr - 1:n_curr - 1 + n_curr]

            # Transform to z-space
            S = np.dot(grad_a, a_curr)
            grad_z = a_curr * (grad_a - S)

            # Adam update
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * (grad_z ** 2)
            m_hat = m / (1 - beta1 ** t_step)
            v_hat = v / (1 - beta2 ** t_step)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            z = np.clip(z, -40, 40)

            if it % 30 == 0:
                a_check = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a_check).max()
                if score < best_score:
                    best_score = score
                    best_a = a_check.copy()

        a_stage = softmax_fn(z)
        score_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()

        if si % 7 == 0:
            elapsed = time.monotonic() - t0
            print(f"      AdamLSE s={si} T={stage_T:.3f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, seed=42):
    """Peak-aware coordinate descent: move mass from high-contributors to low-impact positions."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        b_max = b.max()

        # Get top peaks
        top_K = min(10, 2 * n - 1)
        top_k = np.argsort(b)[::-1][:top_K]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            top_src = idx[np.argsort(contrib)[::-1][:4]]

            for src in top_src:
                if a[src] < 1e-10:
                    continue

                # Compute impact score for each destination
                # A destination j is "safe" if it contributes little to all top peaks
                impact = np.zeros(n)
                for k2 in top_k:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    for j2 in range(js2, je2 + 1):
                        partner = k2 - j2
                        if 0 <= partner < n:
                            impact[j2] += a[partner]

                impact[src] = 1e10
                best_dsts = np.argsort(impact)[:12]

                for dst in best_dsts:
                    if dst == src:
                        continue
                    for frac in [0.001, 0.003, 0.008, 0.02, 0.05, 0.1]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < -1e-14:
                            continue
                        a_trial = np.maximum(a_trial, 0)
                        ss = a_trial.sum()
                        if ss < 1e-14:
                            continue
                        a_trial = a_trial / ss
                        sc = 2.0 * n * fast_autocorr(a_trial).max()
                        if sc < best_score - 1e-14:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random kicks to escape
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.001, 0.03)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= -1e-14:
                        a_trial = np.maximum(a_trial, 0)
                        ss = a_trial.sum()
                        if ss > 1e-14:
                            a_trial = a_trial / ss
                            sc = 2.0 * n * fast_autocorr(a_trial).max()
                            if sc < best_score:
                                a = a_trial
                                best_score = sc
                                best_a = a.copy()
                                stall = 0
                                break
            if stall > 300:
                break

        if stall > 0 and stall % 200 == 100:
            elapsed = time.monotonic() - t0
            print(f"      CD stall={stall} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length using cubic-like interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    # Use linear interpolation (cubic can introduce negatives)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    Phase 5: Coordinate descent polish + final QP/LSE
    """
    rng = np.random.default_rng(777)
    t_global = time.monotonic()
    time_budget = 1730.0  # 70s safety buffer

    solver = get_lp_solver()
    print(f"LP solver: {solver}, budget={time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    n_base = 1600

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            n_base = len(parent)
            p_score = evaluate_safe(parent)
            print(f"Parent: n={n_base}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        print("No parent available, using structured init")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Init: n={n_base}, score={best_score:.8f}")

    # === Phase 1: Nesterov LP with active set ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== P1: Nesterov LP active-set, n={n_base} ({remaining:.0f}s) ===")
    t1 = min(remaining * 0.35, 500.0)
    if t1 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t1, solver, use_active=True, K=100)
        try_update(a_out, f"nlp_as_n{n_base}")

    # === Phase 2: QP refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== P2: QP refine, n={len(best_seq)} ({remaining:.0f}s) ===")
    t2 = min(remaining * 0.20, 280.0)
    if t2 > 20:
        a_out, _ = qp_refine(best_seq, t2, lr_init=0.005, n_inner=80, w_power=3.0)
        try_update(a_out, f"qp_n{len(best_seq)}")

    # === Phase 3: Adam-LSE ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== P3: Adam-LSE, n={len(best_seq)} ({remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 280.0)
    if t3 > 20:
        a_out, _ = adam_lse_optimize(best_seq, t3, T_start=300.0, T_end=0.005, n_stages=30, lr=0.004)
        try_update(a_out, f"adam_n{len(best_seq)}")

    # === Phase 4: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== P4: Multi-length ({remaining:.0f}s) ===")

    n_best = len(best_seq)
    offsets = [-300, -200, -100, 100, 200, 300, 500]
    candidate_lengths = sorted(set(n_best + off for off in offsets if 600 <= n_best + off <= 3200),
                               key=lambda x: abs(x - n_best))

    time_per = min(remaining * 0.55 / max(1, len(candidate_lengths)), 120.0)

    for nl in candidate_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 25:
            break
        tl = min(time_per, remaining * 0.25)
        if tl < 15:
            continue

        print(f"  n={nl} (budget={tl:.0f}s)")
        init = interpolate_sequence(best_seq, nl)

        # Quick NLP
        t_sub = tl * 0.55
        a_out, _ = nesterov_lp_optimize(init, t_sub, solver, use_active=True, K=80)
        try_update(a_out, f"nlp_n{nl}")

        # Quick CD
        rem2 = time_budget - (time.monotonic() - t_global)
        t_cd = max(5, min(tl * 0.3, rem2 * 0.08))
        if t_cd > 5:
            a_out2, _ = coordinate_descent(a_out, t_cd, seed=nl % 10000)
            try_update(a_out2, f"cd_n{nl}")

    # === Phase 5: Final polishing ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(best_seq)
    print(f"\n=== P5: Final polish, n={n_final} ({remaining:.0f}s) ===")

    # Coordinate descent
    t5a = min(remaining * 0.30, 200.0)
    if t5a > 10:
        a_out, _ = coordinate_descent(best_seq, t5a, seed=55555)
        try_update(a_out, "cd_polish1")

    # QP with different params
    remaining = time_budget - (time.monotonic() - t_global)
    t5b = min(remaining * 0.25, 180.0)
    if t5b > 10:
        a_out, _ = qp_refine(best_seq, t5b, lr_init=0.003, n_inner=120, w_power=5.0)
        try_update(a_out, "qp_polish1")

    # Adam-LSE with tighter params
    remaining = time_budget - (time.monotonic() - t_global)
    t5c = min(remaining * 0.25, 180.0)
    if t5c > 10:
        a_out, _ = adam_lse_optimize(best_seq, t5c, T_start=80.0, T_end=0.002, n_stages=25, lr=0.002)
        try_update(a_out, "adam_polish1")

    # Final CD
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = max(10, remaining * 0.4)
    if t6 > 5:
        a_out, _ = coordinate_descent(best_seq, t6, seed=11111)
        try_update(a_out, "cd_final")

    # Verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: {best_score:.8f} (n={len(best_seq)})")
    else:
        print("ERROR: No valid sequence!")
        return [1.0]

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return result
