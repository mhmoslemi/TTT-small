        solver = get_lp_solver()
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    p_val = np.asarray(p.value, dtype=np.float64)
    p_val = np.maximum(p_val, 0)
    s = p_val.sum()
    if s < 1e-15:
        return None, None
    p_val = p_val / s
    return p_val, float(t.value)

def solve_primal_lp(a_fixed, solver=None):
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    if solver is None:
        solver = get_lp_solver()
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def structured_init(n, rng):
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = 1.0 + 0.8 * (2.0 * x - 1.0)
    elif choice == 1:
        a = np.exp(-4.0 * (x - 0.5) ** 2) + 0.3
    elif choice == 2:
        a = 0.5 + 0.4 * np.cos(np.pi * x) + 0.1
    elif choice == 3:
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-6.0 * (x - 0.7) ** 2) + 0.1
    elif choice == 4:
        a = (x + 0.1) ** 0.7
    else:
        a = 0.7 + 0.3 * np.sin(2 * np.pi * x + 0.3)
    a = np.maximum(a, 0.01)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def multiplicative_weights_pre(a0, time_limit, max_steps=200000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = 0.3
    t_nest = 1.0
    a_prev = a.copy()
    last_improve = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve > 5000:
            break
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.90
        topk = np.where(b >= threshold)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])
        else:
            topk = topk[:min(20, len(topk))]
        grad_a = np.zeros(n)
        for k in topk:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += a[k - idx]
        grad_a /= len(topk)
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            z_cur = z_from_a(a)
            z_prev = z_from_a(a_prev)
            z_extrap = z_cur + momentum * (z_cur - z_prev)
        else:
            z_extrap = z_from_a(a)
        z_new = z_extrap - eta * grad_z
        a_new = softmax_from_z(z_new)
        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a_prev = a
            a = a_new
            t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
            last_improve = step
        else:
            if sc > best_score * 1.001:
                t_nest = 1.0
                eta *= 0.92
            else:
                t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
                a_prev = a
                a = a_new
            eta = max(eta * 0.998, 0.05)
    return best_a, best_score

def nesterov_dual_lp(a0, time_limit, solver=None, max_iters=500000, topK=5):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    last_print = t0
    restart_count = 0
    lp_fail_count = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y = y / sy
        else:
            y = a.copy()
        # Use top-K peaks to guide LP
        b_y = fast_convolve(y)
        max_b = b_y.max()
        if max_b < 1e-30:
            b_new = np.ones(n) / n
        else:
            # Weight the LP by focusing on top peaks
            # Solve dual LP for best response
            b_new, M_lp = solve_dual_lp(y, solver)
            if b_new is None:
                # Fallback: use peak-based update
                threshold = max_b * 0.90
                topk = np.where(b_y >= threshold)[0]
                if len(topk) == 0:
                    topk = np.array([int(np.argmax(b_y))])
                topk = topk[:min(topK, len(topk))]
                grad_a = np.zeros(n)
                for k in topk:
                    js = max(0, k - n + 1)
                    je = min(n - 1, k)
                    idx = np.arange(js, je + 1)
                    grad_a[idx] += y[k - idx]
                grad_a /= len(topk)
                S = np.dot(grad_a, y)
                grad_z = y * (grad_a - S)
                z_new = z_from_a(y) - 0.3 * grad_z
                b_new = softmax_from_z(z_new)
                M_lp = max_b
                lp_fail_count += 1
            else:
                lp_fail_count = 0
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha = 2.0 / (t_new + 1.0)
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa
        sc_new = score_normalized(a_new)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
        sc_cur = score_normalized(a)
        if sc_new > sc_cur * 1.0003 and t_nest > 5.0:
            t_nest = 1.0
            a_prev = a_new.copy()
            restart_count += 1
        else:
            a_prev = a.copy()
            t_nest = t_new
        a = a_new
        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 20:
                print(f"    Nesterov DualLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} t={t_nest:.1f} restarts={restart_count}")
                last_print = elapsed
    return best_a, best_score

def lse_objective_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-20:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

def lse_lbfgs(a0, time_limit, T_start=50.0, T_end=0.01, n_stages=30):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    temps = np.geomspace(T_start, T_end, n_stages)
    for si in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        stage_T = temps[si]
        stage_start = time.monotonic()
        stage_budget = min(remaining * 0.8, 30.0)
        if remaining > 100:
            stage_budget = min(15.0, remaining * 0.1)
        else:
            stage_budget = min(remaining * 0.4, 20.0)
        iters = 60 if stage_T > temps[n_stages // 3] else 40
        if stage_T < temps[n_stages * 2 // 3]:
            iters = 80
        try:
            def func_eval(z_val):
                obj, _ = lse_objective_grad(z_val, stage_T, n)
                return obj
            def jac_eval(z_val):
                _, grad = lse_objective_grad(z_val, stage_T, n)
                return grad
            res = minimize(
                fun=func_eval, x0=z, jac=jac_eval, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-25,
                         'gtol': 1e-20, 'maxfun': iters * 500,
                         'disp': False, 'maxls': 50}
            )
            z_new = res.x
            a_new = softmax_from_z(z_new)
            sc = score_normalized(a_new)
            z = z_new
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            stage_elapsed = time.monotonic() - stage_start
            if (si + 1) % 5 == 0:
                print(f"    LSE stage={si+1}/{n_stages} T={stage_T:.4f} ({stage_elapsed:.1f}s): score={best_score:.8f}")
        except Exception:
            pass
    return best_a, best_score

def peak_refine(a0, time_limit, max_steps=100000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = 0.08
    last_improve = 0
    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve > 3000:
            break
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.95
        topk = np.where(b >= threshold)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])
        topk = topk[:min(10, len(topk))]
        grad = np.zeros(n)
        for k in topk:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += a[k - idx]
        grad /= len(topk)
        exponent = -eta * grad
        exponent -= exponent.max()
        new_a = a * np.exp(exponent)
        sa = new_a.sum()
        if sa < 1e-15:
            eta *= 0.5
            continue
        new_a = new_a / sa
        sc = score_normalized(new_a)
        if sc < best_score:
            best_score = sc
            best_a = new_a.copy()
            a = new_a
            last_improve = step
            eta = min(eta * 1.003, 0.2)
        else:
            eta *= 0.88
            if eta < 1e-12:
                break
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def propose_candidate(seed=42, budget_s=1800.0):
