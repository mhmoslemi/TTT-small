import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Fast linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score assuming sum(a)=1: returns 2*n*max(b)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_from_z(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Inverse of softmax (log)."""
    return np.log(np.maximum(a, 1e-300))


def build_conv_matrix(a_fixed):
    """Build T where (T@b)[k] = (conv(a_fixed, b))[k]."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        T[i:i + n, i] = a_fixed
    return T


_lp_solver = None


def get_lp_solver():
    """Find the best available LP solver."""
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CBC, cp.GLPK]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Solve the LP: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    ones = cp.ones(2 * n - 1)
    constraints = [T @ b_var <= M_var * ones, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS, cp.CBC, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp(a0, time_limit, solver=None, max_iters=500000, restart_every=5000, restart_radius=0.05):
    """Nesterov-accelerated LP best-response with periodic perturbation restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0
    rng = np.random.default_rng(12345)
    no_improve_count = 0
    prev_best = best_score

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * (it + 1) * (it + 1))) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            no_improve_count = 0
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            no_improve_count = 0

        if best_score < prev_best - 1e-10:
            no_improve_count = 0
        else:
            no_improve_count += 1

        # Periodic restart with perturbation if stuck
        if no_improve_count > restart_every and it > 100:
            pert = rng.normal(0, restart_radius, n)
            a_pert = a + pert
            a_pert = np.maximum(a_pert, 0)
            sp = a_pert.sum()
            if sp > 1e-15:
                a_pert = a_pert / sp
            sc_pert = score_normalized(a_pert)
            if sc_pert < best_score * 1.01:
                a = a_pert
                y = a_pert.copy()
            no_improve_count = 0
            restart_radius = max(0.01, restart_radius * 0.7)

        prev_best = best_score

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 10:
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} M_lp={M_lp:.8f}")
                last_print = elapsed

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    max_b = b.max()
    shifted = T_val * b - T_val * max_b
    exp_b = np.exp(shifted)
    w = exp_b / exp_b.sum()
    log_sum_exp = T_val * max_b + np.log(exp_b.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_lbfgs(a0, time_limit, T_start=80.0, T_min=0.1, n_stages=30, iters_per_stage=30):
    """LogSumExp L-BFGS with geometric temperature schedule."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        T = T_start * (T_min / T_start) ** (stage / max(1, n_stages - 1))
        T = max(T, T_min)

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, T, n)
            return grad

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-16, 'maxfun': iters_per_stage * 200,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)

            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()

            if (stage + 1) % 5 == 0:
                print(f"    LSE stage={stage+1}/{n_stages} T={T:.3f}: score={sc:.8f} best={best_score:.8f}")

        except Exception:
            pass

    return best_a, best_score


def subgradient_optimize(a0, time_limit, n_steps=300000):
    """Peak-targeted subgradient with Polyak step and iterate averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / a.sum()
    z = z_from_a(a).copy()

    best_a = a.copy()
    best_score = score_normalized(a)
    lower_bound = 2.0 * n / (2.0 * n - 1.0)
    t0 = time.monotonic()
    avg_a = a.copy()
    avg_count = 1

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b

        # Find active peaks
        threshold = max_b * 0.95
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        if len(active) > 30:
            top_idx = np.argsort(b[active])[-30:]
            active = active[top_idx]

        # Compute subgradient
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        S = np.dot(g, a)
        grad_z = a * (g - S)

        grad_norm = np.linalg.norm(grad_z)
        if grad_norm < 1e-15:
            break

        gap = cur_score - lower_bound
        if gap < 1e-10:
            break
        eta = gap / (2.0 * grad_norm * grad_norm)
        eta = min(eta, 0.3)

        z_new = z - eta * grad_z
        a_new = softmax_from_z(z_new)
        sc_new = score_normalized(a_new)

        if sc_new > cur_score * 1.02:
            lo_e, hi_e = 0.0, eta
            for _ in range(4):
                mid_e = (lo_e + hi_e) / 2.0
                z_mid = z - mid_e * grad_z
                a_mid = softmax_from_z(z_mid)
                if score_normalized(a_mid) < cur_score:
                    lo_e = mid_e
                else:
                    hi_e = mid_e
            eta = lo_e
            z_new = z - eta * grad_z
            a_new = softmax_from_z(z_new)
            sc_new = score_normalized(a_new)

        z = z_new
        a = a_new

        avg_a = (avg_count * avg_a + a) / (avg_count + 1)
        avg_count += 1
        if avg_count > 3000:
            avg_count = 3000
            avg_a = (avg_a * 3000 + a) / 3001

        if sc_new < best_score:
            best_score = sc_new
            best_a = a.copy()

        sc_avg = score_normalized(avg_a)
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = avg_a.copy()

        if (step + 1) % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad step={step+1} ({elapsed:.0f}s): cur={cur_score:.8f} best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate structured initial sequences."""
    choice = rng.integers(0, 10)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-2 * x ** 2)
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif choice == 4:
        a = np.ones(n) * (1.0 + 0.2 * np.cos(2 * np.pi * 3 * np.arange(n) / n))
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(3 * x) + 0.15 * np.cos(7 * x)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * np.cos(np.pi * x / 3) ** 2
    elif choice == 7:
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.5) ** 2) / 0.05) + 0.3 * np.exp(-((x - 0.2) ** 2) / 0.02) + 0.3 * np.exp(-((x - 0.8) ** 2) / 0.02)
    elif choice == 8:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 5 * x) * np.exp(-2 * (x - 0.5) ** 2)
    else:
        x = np.linspace(0, 1, n)
        a = (1 - x) * np.sin(np.pi * x) + 0.5 * np.sin(2 * np.pi * x) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def nesterov_lp_multi_restart(a0, time_limit, solver=None, n_restarts=3):
    """Run Nesterov LP multiple times with different perturbations."""
    n = len(a0)
    rng = np.random.default_rng(67890)
    best_a = a0.copy()
    best_score = score_normalized(a0)
    t_total = time.monotonic()

    for r in range(n_restarts):
        remaining = time_limit - (time.monotonic() - t_total)
        if remaining < 30:
            break
        t_alloc = remaining / (n_restarts - r) * 0.9

        if r == 0:
            start = a0.copy()
        else:
            start = best_a + rng.normal(0, 0.03 / r, n)
            start = np.maximum(start, 0)
            start = start / start.sum()

        a_out, sc = nesterov_lp(start, t_alloc, solver=solver, restart_every=3000, restart_radius=0.04)
        if sc < best_score:
            best_score = sc
            best_a = a_out.copy()
            print(f"    Restart {r+1}: improved to {sc:.8f}")
        else:
            print(f"    Restart {r+1}: no improvement (got {sc:.8f}, best {best_score:.8f})")

    return best_a, best_score


def final_propose_candidate(seed=42, budget_s=1800.0):
    """Main search function."""
    rng = np.random.default_rng(seed)
    global_deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Total budget: {budget_s}s")

    # Get parent sequence
    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        best_seq = structured_init(1200, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_base = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            remain = global_deadline - time.monotonic()
            print(f"  ** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, remaining={remain:.0f}s)")

    # === Phase 1: Nesterov LP with multi-restart (50% time) ===
    print(f"\n=== Phase 1: Nesterov LP multi-restart at n={n_base} (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t1 = 0.50 * (budget_s - 20)
    t1 = min(t1, remaining * 0.7)
    a_out, _ = nesterov_lp_multi_restart(best_seq, t1, solver=solver, n_restarts=3)
    try_update(a_out, "nesterov_multi")
    print(f"  Phase 1 done: best_score={best_score:.8f}")

    # === Phase 2: LSE L-BFGS polishing (20% time) ===
    print(f"\n=== Phase 2: LSE L-BFGS (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t2 = min(0.20 * (budget_s - 20), remaining * 0.5)
    if t2 > 30:
        a_lse1, _ = lse_lbfgs(best_seq, t2 * 0.5, T_start=100.0, T_min=0.1, n_stages=25, iters_per_stage=30)
        try_update(a_lse1, "lse_pass1")

        remaining = global_deadline - time.monotonic()
        if remaining > 20:
            a_lse2, _ = lse_lbfgs(best_seq, remaining * 0.6, T_start=20.0, T_min=0.05, n_stages=15, iters_per_stage=25)
            try_update(a_lse2, "lse_pass2")

        # LSE from perturbed start
        remaining = global_deadline - time.monotonic()
        if remaining > 20:
            perturbed = best_seq + 0.04 * np.random.default_rng(999).normal(0, 1, len(best_seq))
            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()
            a_lse3, _ = lse_lbfgs(perturbed, remaining * 0.5, T_start=50.0, T_min=0.1, n_stages=12, iters_per_stage=25)
            try_update(a_lse3, "lse_perturbed")
    print(f"  Phase 2 done: best_score={best_score:.8f}")

    # === Phase 3: Subgradient optimization (15% time) ===
    print(f"\n=== Phase 3: Subgradient (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t3 = min(0.15 * (budget_s - 20), remaining * 0.5)
    if t3 > 20:
        a_sg, _ = subgradient_optimize(best_seq, t3, n_steps=500000)
        try_update(a_sg, "subgrad")

        remaining = global_deadline - time.monotonic()
        if remaining > 15:
            a_sg2, _ = subgradient_optimize(best_seq, remaining * 0.5, n_steps=300000)
            try_update(a_sg2, "subgrad2")
    print(f"  Phase 3 done: best_score={best_score:.8f}")

    # === Phase 4: Multi-length exploration (10% time) ===
    print(f"\n=== Phase 4: Multi-length (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t4_total = min(0.10 * (budget_s - 20), remaining * 0.6)
    if t4_total > 30:
        lengths_to_try = [n_base + 200, n_base + 500, n_base - 200, n_base + 100]
        for n_try in lengths_to_try:
            remaining = global_deadline - time.monotonic()
            if remaining < 25:
                break
            t4 = min(t4_total / 4.0, remaining * 0.45)
            if t4 < 15:
                break
            print(f"  Trying n={n_try}, budget={t4:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            a_try, _ = nesterov_lp(init_try, t4 * 0.6, solver=solver, restart_every=3000)
            try_update(a_try, f"len_{n_try}")

            remaining = global_deadline - time.monotonic()
            if remaining > 15:
                a_try2, _ = lse_lbfgs(a_try, remaining * 0.5, T_start=40.0, T_min=0.2, n_stages=10, iters_per_stage=20)
                try_update(a_try2, f"len_{n_try}_lse")
    print(f"  Phase 4 done: best_score={best_score:.8f}")

    # === Phase 5: Final combined polish (5% time) ===
    print(f"\n=== Phase 5: Final polish (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    if remaining > 20:
        a_f1, _ = nesterov_lp(best_seq, remaining * 0.4, solver=solver, max_iters=50000, restart_every=2000, restart_radius=0.02)
        try_update(a_f1, "final_nesterov")

        remaining = global_deadline - time.monotonic()
        if remaining > 15:
            a_f2, _ = lse_lbfgs(best_seq, remaining * 0.5, T_start=10.0, T_min=0.05, n_stages=10, iters_per_stage=20)
            try_update(a_f2, "final_lse")

        remaining = global_deadline - time.monotonic()
        if remaining > 10:
            a_f3, _ = subgradient_optimize(best_seq, remaining * 0.6, n_steps=100000)
            try_update(a_f3, "final_sg")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]


def propose_candidate(seed=42, budget_s=1800.0):
    """Entry point for the search."""
    return final_propose_candidate(seed=seed, budget_s=budget_s)
