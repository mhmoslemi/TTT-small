import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_of(a):
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def subgradient_multi_top(a, K=10, T=100):
    """Compute subgradient at top K peaks, weighted by softmax."""
    n = len(a)
    b = fast_convolve(a)
    # Get top K peaks
    top_idx = np.argpartition(b, -K)[-K:]
    top_idx = top_idx[np.argsort(b[top_idx])[::-1]]  # sort by value descending
    # Softmax weights
    top_vals = b[top_idx]
    w = np.exp(T * (top_vals - top_vals.max()))
    w /= w.sum()
    
    g = np.zeros(n)
    for idx in range(len(top_idx)):
        k = top_idx[idx]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            if k == 2 * i:
                g[i] += 2.0 * a[i] * w[idx]
            else:
                g[i] += 2.0 * a[k - i] * w[idx]
    return g, b.max()

def nesterov_subgrad(a0, time_limit, K=10, T=100, lr_base=1.0):
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    y = a.copy()
    best_a = a.copy()
    best_score = score_of(a)
    lower_bound = 1.0 / (2.0 * n - 1.0)  # theoretical lower bound for max(conv)
    t0 = time.monotonic()
    
    for it in range(10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        g, max_b = subgradient_multi_top(y, K=K, T=T)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        # Polyak step size
        cur_score = 2.0 * n * max_b
        gap = cur_score - 2.0 * n * lower_bound
        if gap < 1e-10:
            break
        eta = lr_base * gap / (2.0 * g_norm * g_norm + 1e-15)
        eta = min(eta, 2.0 / (np.sqrt(it + 1) + 1))  # cap
        
        # Nesterov step
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * (it+1) * (it+1))) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k
        
        a_old = a.copy()
        a_new = project_simplex(y - eta * g)
        y_new = a_new + momentum * (a_new - a_old)
        
        a = a_new
        y = y_new
        
        sc = score_of(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if (it + 1) % 5000 == 0:
            print(f"    NS iter={it+1}: score={best_score:.8f} cur={cur_score:.8f}")
    
    return best_a, best_score

def fast_convolve(a):
    """Fast linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score assuming sum(a)=1: returns 2*n*max(conv(a,a))."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

def project_simplex(v):
    """Project onto probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return np.array([])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_arr = np.where(cond)[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def normalize_nonneg(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(a, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def subgradient_multi_peak(a, K=10, T=80.0):
    """Compute subgradient at top K peaks with softmax weighting.
    Returns (gradient vector, max convolution value).
    """
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    
    # Get top K peaks
    if K >= len(b):
        top_idx = np.arange(len(b))
    else:
        top_idx = np.argpartition(b, -K)[-K:]
    # Sort by value descending
    top_vals = b[top_idx]
    top_idx = top_idx[np.argsort(top_vals)[::-1]]
    top_vals = top_vals[np.argsort(top_vals)[::-1]]
    
    # Softmax weights over top peaks
    shifted = top_vals - top_vals[0]
    w = np.exp(T * shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(len(top_idx)) / len(top_idx)
    else:
        w = w / w_sum
    
    # Compute gradient
    g = np.zeros(n)
    for idx in range(len(top_idx)):
        k = int(top_idx[idx])
        wi = w[idx]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            if k == 2 * i:
                g[i] += 2.0 * a[i] * wi
            else:
                g[i] += 2.0 * a[k - i] * wi
    
    return g, max_b

def build_conv_matrix(a_fixed):
    """Build convolution matrix T where (T@b)[k] = (a_fixed*b)[k]."""

    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length (preserving normalized shape)."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp

def lse_objective_and_grad(z, T, n):
    """LogSumExp objective and gradient w.r.t. z for softmax parameterization."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    
    b = fast_convolve(a)
    log_weights = T * b
    max_lw = log_weights.max()
    exp_lw = np.exp(log_weights - max_lw)
    w = exp_lw / exp_lw.sum()
    log_sum_exp = max_lw + np.log(exp_lw.sum())
    obj = (2.0 * n / T) * log_sum_exp
    
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        if k == 2 * lo and hi == lo:
            grad_a[lo] += 2.0 * w[k] * a[lo]
        else:
            for i in range(lo, hi + 1):
                if k == 2 * i:
                    grad_a[i] += 2.0 * w[k] * a[i]
                else:
                    grad_a[i] += 2.0 * w[k] * a[k - i]
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

            # Adaptive temperature
            if stagnant >= 2:
                T = max(T * 1.3, T_min)
                stagnant = 0
            else:
                if sc > best_score * 1.01:
                    stagnant += 1
                else:
                    stagnant = 0
                T = max(T * 0.7, T_min)
        except Exception:
            T = max(T * 0.8, T_min)
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function."""
    rng = np.random.default_rng(seed)
    global_deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Total budget: {budget_s}s")
    
    # Get best starting sequence
    best_seq = None
    best_score = float('inf')
    n_base = 1200
    
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent sequence: n={n_base}, score={best_score:.8f}")
    except Exception:
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            remain = global_deadline - time.monotonic()
            print(f"  ** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, remain={remain:.0f}s)")
    
    # === Phase 1: Nesterov Mirror Descent (main workhorse) - 30% ===
    print(f"\n=== Phase 1: Nesterov Mirror Descent (score={best_score:.8f}) ===")
    t1 = 0.30 * (budget_s - 30)
    remaining = global_deadline - time.monotonic()
    t1 = min(t1, remaining)
    if t1 > 10:
        a_out, _ = nesterov_mirror_descent(best_seq, t1, K=10, T=80.0, eta0=0.8)
        try_update(a_out, "nmd_main")
    
    # === Phase 2: LP Polishing - 20% ===
    print(f"\n=== Phase 2: LP Polishing (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t2 = min(0.20 * (budget_s - 30), remaining * 0.8)
    if t2 > 15:
        a_out, _ = nesterov_lp_polish(best_seq, t2, solver=solver, max_iters=30000)
        try_update(a_out, "lp_polish")
    
    # === Phase 3: Extended NMD from improved point - 20% ===
    print(f"\n=== Phase 3: Extended NMD (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t3 = min(0.20 * (budget_s - 30), remaining * 0.8)
    if t3 > 10:
        # Run with different hyperparameters
        a_out, _ = nesterov_mirror_descent(best_seq, t3, K=20, T=40.0, eta0=0.4, momentum_decay=0.8)
        try_update(a_out, "nmd_ext")
    
    # === Phase 4: LSE L-BFGS - 10% ===
    print(f"\n=== Phase 4: LSE L-BFGS (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t4 = min(0.10 * (budget_s - 30), remaining * 0.8)
    if t4 > 15:
        a_out, _ = lse_lbfgs(best_seq, t4, T_start=30.0, T_min=0.5, n_stages=12, iters_per_stage=25)
        try_update(a_out, "lse")
    
    # === Phase 5: Multi-length exploration - 10% ===
    print(f"\n=== Phase 5: Multi-length (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t5_total = min(0.10 * (budget_s - 30), remaining * 0.7)
    
    # Try a few different lengths
    lengths_to_try = []
    if n_base > 1000:
        lengths_to_try = [n_base + 400, n_base - 400]
    else:
        lengths_to_try = [n_base + 400, n_base + 800, n_base - 200]
    
    for n_try in lengths_to_try:
        if n_try < 200 or n_try > 5000:
            continue
        remaining = global_deadline - time.monotonic()
        if remaining < 20:
            break
        t5 = min(t5_total / len(lengths_to_try), remaining * 0.45)
        if t5 < 10:
            break
        
        print(f"  Trying n={n_try}, budget={t5:.0f}s")
        init_try = interpolate_sequence(best_seq, n_try)
        
        # Quick NMD
        a_try, _ = nesterov_mirror_descent(init_try, t5 * 0.7, K=10, T=60.0, eta0=0.6)
        try_update(a_try, f"nmd_n{n_try}")
        
        # Quick LP if time
        remaining = global_deadline - time.monotonic()
        if remaining > 10:
            a_try2, _ = nesterov_lp_polish(a_try, min(remaining * 0.4, 60), solver=solver, max_iters=10000)
            try_update(a_try2, f"lp_n{n_try}")
    
    # === Phase 6: Final combined polish - 10% ===
    print(f"\n=== Phase 6: Final Polish (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    
    if remaining > 10:
        # Short NMD with high K
        t6a = min(remaining * 0.3, 120)
        a_f1, _ = nesterov_mirror_descent(best_seq, t6a, K=30, T=20.0, eta0=0.3)
        try_update(a_f1, "final_nmd")
    
    remaining = global_deadline - time.monotonic()
    if remaining > 10:
        # Short LP
        t6b = min(remaining * 0.3, 120)
        a_f2, _ = nesterov_lp_polish(best_seq, t6b, solver=solver, max_iters=10000)
        try_update(a_f2, "final_lp")
    
    remaining = global_deadline - time.monotonic()
    if remaining > 10:
        # Coordinate descent polish
        t6c = min(remaining * 0.3, 60)
        a_f3, _ = coordinate_descent_polish(best_seq, t6c, n_rounds=3)
        try_update(a_f3, "coord_desc")
    
    remaining = global_deadline - time.monotonic()
    if remaining > 5:
        # One more NMD from the best
        a_f4, _ = nesterov_mirror_descent(best_seq, remaining * 0.5, K=15, T=50.0, eta0=0.5)
        try_update(a_f4, "final_nmd2")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"Time used: {time.monotonic() - (global_deadline - (budget_s - 30)):.1f}s")
    return [float(x) for x in best_seq]
