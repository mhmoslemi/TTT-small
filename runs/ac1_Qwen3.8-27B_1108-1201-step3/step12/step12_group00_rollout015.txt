"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP best-response Frank-Wolfe with Nesterov acceleration
2. Mirror descent (multiplicative weight updates) on LSE-smoothed max
3. Targeted coordinate descent peak shaving with smart source/target selection
4. Multi-length exploration with interpolation transfer
5. Adaptive temperature scheduling and diversified initializations
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver_cache = None


def _get_solver():
    global _solver_cache
    if _solver_cache is not None:
        return _solver_cache
    for s in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver_cache = s
                print(f"  Solver: {s}")
                return s
        except Exception:
            continue
    _solver_cache = cp.ECOS
    print("  Solver: ECOS (fallback)")
    return _solver_cache


def _fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def _eval(a):
    """Compute objective 2*n*max(conv(a,a))/(sum(a))^2"""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return float(2.0 * n * _fast_conv(a).max() / (s * s))


def _normalize(a, target=None):
    n = len(a)
    if target is None:
        target = np.sqrt(2.0 * n)
    s = a.sum()
    if s < 1e-15:
        return np.full(n, target / n)
    return a * target / s


def _build_T(a):
    """Conv matrix: (T @ b)[k] = (a*b)[k] for k=0..2n-2"""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def _lp_best_response(y, M, solver):
    n = len(y)
    b = cp.Variable(n, nonneg=True)
    T = _build_T(y)
    constraints = [T @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def _fw_nesterov(a0, time_budget, solver, verbose=True):
    """Frank-Wolfe + Nesterov with LP best response."""
    n = len(a0)
    target = np.sqrt(2.0 * n)
    a = _normalize(a0.copy(), target)
    best_a = a.copy()
    best_score = _fast_conv(a).max()
    y = a.copy()
    t0 = time.monotonic()

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_budget:
            break

        M = _fast_conv(y).max()
        b = _lp_best_response(y, M, solver)
        if b is None:
            continue

        g = _normalize(b, target)
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y = _normalize(np.maximum(y_new, 0.0), target)

        if t == 1:
            a = g.copy()
        else:
            mom = (t - 1.0) / (t + 2.0)
            a_new = y + mom * (y - a)
            a = _normalize(np.maximum(a_new, 0.0), target)

        sy = _fast_conv(y).max()
        if sy < best_score:
            best_score = sy
            best_a = y.copy()
        sa = _fast_conv(a).max()
        if sa < best_score:
            best_score = sa
            best_a = a.copy()

        if verbose and t % 300 == 0:
            print(f"    FW t={t} ({time.monotonic()-t0:.0f}s) best={best_score:.6f}")

    return best_a, best_score


def _mirror_lse(a0, time_budget, T_start=80.0, T_end=0.3, n_stages=12, verbose=True):
    """Mirror descent with multiplicative updates on LSE-smoothed max."""
    n = len(a0)
    target = np.sqrt(2.0 * n)
    a = _normalize(a0.copy(), target)
    best_a = a.copy()
    best_score = _fast_conv(a).max()
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_budget:
            break
        remaining = time_budget - (time.monotonic() - t0)
        if remaining < 1:
            break

        n_steps = max(30, min(400, int(remaining * 8)))
        step_base = 1.0 / (T * np.sqrt(n))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_budget:
                break

            conv = _fast_conv(a)
            max_c = conv.max()
            w = np.exp(conv / T - max_c / T)
            w /= w.sum()

            # Gradient computation
            if T < 15.0:
                top_idx = np.argsort(w)[::-1]
                n_top = min(50, len(top_idx))
                top_w_sum = w[top_idx[:n_top]].sum()
                if top_w_sum > 0.98:
                    grad = np.zeros(n)
                    for ki in top_idx[:n_top]:
                        wi = w[ki]
                        if wi < 1e-14:
                            continue
                        k = int(ki)
                        r1s = max(0, k - n + 1)
                        r1e = min(n - 1, k)
                        if r1s <= r1e:
                            idx1 = np.arange(r1s, r1e + 1)
                            grad[idx1] += wi * a[k - idx1]
                        if k < n:
                            r2s = k
                            r2e = n - 1
                            if r2s <= r2e:
                                idx2 = np.arange(r2s, r2e + 1)
                                grad[idx2] += wi * a[idx2 - k]
                else:
                    grad = _compute_grad_fft(a, w, n)
            else:
                grad = _compute_grad_fft(a, w, n)

            eta = step_base / (1.0 + 0.05 * it)
            a_new = a * np.exp(-eta * grad)
            a_new = np.maximum(a_new, 1e-30)
            a = _normalize(a_new, target)

            sc = _fast_conv(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose and si % 3 == 0:
            print(f"    MD st={si}/{n_stages} T={T:.1f} ({time.monotonic()-t0:.0f}s) best={best_score:.6f}")

    return best_a, best_score


def _compute_grad_fft(a, w, n):
    """Compute gradient via FFT: grad[i] = sum_k w[k]*(a[k-i]+a[i-k])_clipped"""
    fft_len = 1 << (3 * n - 2).bit_length()
    A_f = rfft(a, fft_len)
    W_f = rfft(w, fft_len)
    c2 = irfft(A_f * W_f, fft_len)[:n]
    A_rev_f = rfft(a[::-1].copy(), fft_len)
    conv_full = irfft(A_rev_f * W_f, fft_len)
    c1 = conv_full[n - 1:2 * n - 1]
    return c1 + c2

def _interpolate(f, new_n):
