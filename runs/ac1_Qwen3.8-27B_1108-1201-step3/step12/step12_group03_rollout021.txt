def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]

    except Exception:

def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):

        except Exception:

    return cp.ECOS


def build_conv_matrix(a_fixed):
    """Build the (2n-1) x n Toeplitz convolution matrix."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, max_iter=10000)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, max_iter=10000)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_val_var(a_var_value := b_var.value), dtype=np.float64) if hasattr(b_var, 'value') else np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, snapshot_interval=200):
    """
    Nesterov-accelerated LP best-response with snapshot restarts.
    Key improvements: slower decay, periodic restart from best snapshot.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = 1.0
    a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    snapshot_a = a.copy()
    snapshot_score = best_score
    t0 = time.monotonic()
    iter_count = 0

    for t in range(1, 500000):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue
        iter_count += 1

        # Slower Nesterov decay for longer convergence
        if t < 50:
            alpha_t = 2.0 / (t + 1.0)
        elif t < 500:
            alpha_t = 2.0 / (t + 2.0)
        else:
            alpha_t = 2.0 / (t + 2.0) * 0.8  # Slightly slower beyond 500

        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_new_s = y_new.sum()
        if y_new_s < 1e-15:
            y_new = y.copy()
            y_new_s = 1.0
        y_new = y_new / y_new_s

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            a_s = a_new.sum()
            if a_s < 1e-15:
                a_new = y_new.copy()
                a_s = 1.0
            a_new = a_new / a_s

        y = y_new
        a = a_new

        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Snapshot: periodically restart from best if stalled
        if t % snapshot_interval == 0:
            if best_score < snapshot_score * 0.9999:
                snapshot_a = best_a.copy()
                snapshot_score = best_score
            elif t > 1000 and (best_score - snapshot_score) / max(snapshot_score, 1e-10) < 1e-6:
                # Stalled: restart from snapshot with perturbation
                y = snapshot_a.copy()
                a = snapshot_a.copy()
                # Small perturbation
                pert = np.random.default_rng(t).uniform(0.99, 1.01, size=n)
                a = a * pert
                a = np.maximum(a, 0)
                a_s = a.sum()
                if a_s > 1e-15:
                    a = a / a_s
                    y = a.copy()

        if t % 500 == 0 and verbose:
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}")

        # Early stopping
        if t > 2000 and (best_score - snapshot_score) / max(snapshot_score, 1e-10) < 1e-8:
            if verbose:
                print(f"      Converged at t={t}")
            break

    return best_a, best_score, iter_count
