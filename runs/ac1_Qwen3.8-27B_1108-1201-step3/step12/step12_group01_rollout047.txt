import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_autocorr(a):
    """Full autocorrelation (convolution with self) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_topk_lse_and_grad(a, T, topk):
    """
    Compute top-k LSE objective and gradient (in a-space) efficiently.
    Objective: (1/T) * log(sum_{K in top_k} exp(T * c[K]))
    Gradient: weighted sum of peak gradients computed via FFT cross-correlation.
    """
    n = len(a)
    c = fast_autocorr(a)
    k_use = min(topk, len(c))
    top_idx = np.argsort(c)[::-1][:k_use]
    top_vals = c[top_idx]

    Tc = T * top_vals
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    lse_val = (max_Tc + np.log(exp_Tc.sum())) / T
    w = exp_Tc / exp_Tc.sum()

    # Build peak weight vector of length (2n-1)
    w_peaks = np.zeros(2 * n - 1)
    w_peaks[top_idx] = 2.0 * w

    # Gradient: g[i] = sum_K w_peaks[K] * a[K-i] for valid K-i
    # This is cross-correlation of w_peaks with a
    # g[i] = sum_K w_peaks[K] * a[K-i] = (w_peaks cross-correlate a)[i]
    # Using FFT: cross-correlation = ifft(conj(FFT(a_padded)) * FFT(w_peaks_padded))
    fft_size = 1 << (2 * n - 1 + n - 1).bit_length()
    A_pad = rfft(a, fft_size)
    W_pad = rfft(w_peaks, fft_size)
    # Cross-correlation: (w_peaks ⋆ a)[i] = sum_K w_peaks[K] * a[K-i]
    # = ifft(conj(FFT(a)) * FFT(w_peaks)) shifted appropriately
    xcorr = irfft(np.conj(A_pad) * W_pad, fft_size)
    # The cross-correlation gives sum_k w_peaks[k] * a[k-i+n-1] for i in [0, fft_size)
    # We need to extract the right portion
    # xcorr[m] = sum_k w_peaks[k] * a[k - m + (n-1)] ... let me just do it directly for safety
    # Actually let me use a simpler approach: g[i] = sum_K w_peaks[K]*a[K-i]
    # For K in [0, 2n-2] and i in [0, n-1], K-i is valid iff 0 <= K-i <= n-1
    # So K ranges from i to i+n-1
    # g[i] = sum_{K=i}^{i+n-1} w_peaks[K] * a[K-i]
    # This is a sliding window sum - can be done with FFT convolution
    # g = convolve(w_peaks, a)[n-1 : 2n-1] ... let me verify
    # conv(w_peaks, a)[m] = sum_j w_peaks[j] * a[m-j]
    # We want g[i] = sum_K w_peaks[K] * a[K-i], let K = j, m-j = K-i => m = K - i + K... 
    # Hmm, let me just do: g[i] = sum_K w_peaks[K] * a[K-i]
    # Let m = 2*(n-1) - i. Then conv(w_peaks, a)[m] = sum_j w_peaks[j]*a[m-j] = sum_j w_peaks[j]*a[2n-2-i-j]
    # Set l = j, then a[2n-2-i-l]. We want a[K-i] where K goes from i to i+n-1.
    # So 2n-2-i-l = K-i => l = 2n-2-K. As K goes from i to i+n-1, l goes from 2n-2-i to n-1-i.
    # This is getting confusing. Let me just use the direct FFT approach properly.
    
    # Simple approach: g is the convolution of w_peaks with reversed a, shifted
    # g[i] = sum_K w_peaks[K] * a[K-i]
    # Let b = reverse(a), so b[j] = a[n-1-j]
    # g[i] = sum_K w_peaks[K] * a[K-i] = sum_K w_peaks[K] * b[n-1-(K-i)] = sum_K w_peaks[K] * b[n-1-i+... ]
    
    # Actually the cleanest way: g = ifft(FFT(w_peaks) * conj(FFT(a_padded_to_2n-1)))
    # Let's just do it with a direct FFT of the right size
    size_wc = 2 * n - 1
    fft_sz = 1 << (size_wc + n - 1 - 1).bit_length()
    Wf = rfft(w_peaks, fft_sz)
    Af = rfft(a, fft_sz)
    # Product in freq domain: Wf * conj(Af) gives cross-correlation (w ⋆ a)[d] = sum_k w[k]*a[k-d]
    xcorr_full = irfft(Wf * np.conj(Af), fft_sz)
    # xcorr_full[d] = sum_k w_peaks[k] * a[k-d] for k in valid range
    # We want g[i] = sum_K w_peaks[K] * a[K-i] for i in [0, n-1]
    # This is xcorr_full[i] (with appropriate indexing)
    # xcorr_full[d] = sum_k w_peaks[k] * a[k-d] where a is zero-padded
    # For d = i (0-indexed), this gives sum_k w_peaks[k] * a[k-i] = g[i] ✓
    g = xcorr_full[:n]

    return lse_val, g, c


def adam_nesterov_topk_lse(a0, time_limit, T_start=50.0, T_end=0.3, n_stages=30, topk=30, lr_base=0.02, beta1=0.9, beta2=0.999, eps=1e-8):
    """
    ADAM + Nesterov momentum with top-k LSE objective in log-space.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    z = np.log(np.maximum(a, 1e-30))
    t0 = time.monotonic()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    # Nesterov momentum
    z_mom = np.zeros(n)
    
    temps = np.geomspace(T_start, T_end, n_stages)
    it_count = 0

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit - 3:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        # Learning rate with warmup and decay
        warmup = 50
        stage_lr = lr_base * min(1.0, (si + 1) / max(1, n_stages // 4))

        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0

        # Estimate iterations per stage
        est_time_per_iter = max(0.001, (time.monotonic() - t0) / max(1, it_count)) if it_count > 0 else 0.002
        max_it = max(20, int(remaining * 0.8 / max(est_time_per_iter, 0.001)))
        max_it = min(max_it, 5000)

        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 3:
                break

            # Compute objective and gradient
            obj, grad_a, c = compute_topk_lse_and_grad(a, stage_T, topk)

            # Convert to z-space gradient
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)

            # ADAM update
            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)

            # Nesterov: use momentum on z
            z_mom = 0.9 * z_mom + grad_z
            z = z - stage_lr * (m_hat / (np.sqrt(v_hat) + eps) + 0.1 * z_mom)
            it_count += 1

            # Periodic evaluation
            if (it + 1) % 10 == 0:
                a = np.exp(z - z.max())
                a = a / a.sum()
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()

        # End of stage
        a = np.exp(z - z.max())
        a = a / a.sum()
        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if si % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"      ADAM-LSE stage {si}/{n_stages} (T={stage_T:.1f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def subgradient_polish(a0, time_limit, lr=0.005):
    """
    Direct subgradient method on the single argmax peak.
    Minimizes conv(a,a)[argmax] via subgradient descent.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    momentum = np.zeros(n)
    beta = 0.9
    t_iter = 0

    while time.monotonic() - t0 < time_limit - 2:
        c = fast_autocorr(a)
        k_star = np.argmax(c)

        # Subgradient of conv(a,a)[k_star] w.r.t. a
        # d/da[i] of sum_j a[j]*a[k_star-j] = a[k_star-i] + a[k_star-i] (if i != k_star-i)
        # = 2*a[k_star-i] for i in valid range
        js = max(0, k_star - (n - 1))
        je = min(n - 1, k_star)
        g = np.zeros(n)
        if js <= je:
            idx = np.arange(js, je + 1)
            g[idx] = 2.0 * a[k_star - idx]

        # Center the gradient (for sum constraint)
        g_centered = g - g.mean()

        # Projected subgradient with momentum
        momentum = beta * momentum + g_centered
        lr_t = lr / (1.0 + 0.01 * t_iter)
        a = a - lr_t * momentum
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
        else:
            a = a / s
        t_iter += 1

        if t_iter % 5 == 0:
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

    return best_a, best_score


def local_mass_search(a0, time_limit, seed=42):
    """
    Local search: try moving small amounts of mass from positions
    that contribute to the top peak to positions that don't.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 2:
        c = fast_autocorr(a)
        bmax = c.max()
        k_star = np.argmax(c)

        # Positions contributing to the peak
        js = max(0, k_star - (n - 1))
        je = min(n - 1, k_star)
        contrib_idx = np.arange(js, je + 1)
        contrib = a[contrib_idx] * a[k_star - contrib_idx]
        total_contrib = contrib.sum()

        if total_contrib < 1e-15:
            break

        # Find top contributors
        top_contrib_positions = contrib_idx[np.argsort(contrib)[::-1][:10]]

        # Find "safe" destinations (low contribution to top peaks)
        # Compute danger: for each position i, how much does it contribute to top 10 peaks?
        top_peaks = np.argsort(c)[::-1][:10]
        danger = np.zeros(n)
        for kp in top_peaks:
            jsp = max(0, kp - (n - 1))
            jep = min(n - 1, kp)
            idx_p = np.arange(jsp, jep + 1)
            danger[idx_p] += c[kp] * a[idx_p] * a[kp - idx_p]

        # Safe positions: low danger, not in contrib_idx
        safe_mask = np.ones(n, dtype=bool)
        safe_mask[contrib_idx] = False
        safe_positions = np.where(safe_mask)[0]
        if len(safe_positions) == 0:
            break

        # Sort by danger (ascending) and current value (ascending)
        safe_score = danger[safe_positions] + a[safe_positions] * 10
        safe_sorted = safe_positions[np.argsort(safe_score)]
        dst_pool = safe_sorted[:max(5, len(safe_sorted) // 10)]

        improved = False
        for src in top_contrib_positions[:5]:
            if a[src] < 1e-10:
                continue
            for dst in dst_pool[:5]:
                if dst == src:
                    continue
                for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 200:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length using cubic-like smoothing."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm, left=f_norm[0], right=f_norm[-1])
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def perturbed_init(a0, rng, scale=0.05):
    """Create a perturbed version of a sequence for multi-start."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a0)) / len(a0)
    a = a / s
    noise = rng.normal(0, scale, len(a))
    a_new = a + noise * (a + 1e-6)
    a_new = np.maximum(a_new, 0)
    sn = a_new.sum()
    if sn < 1e-15:
        return a.copy()
    return a_new / sn


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, time_budget=1800.0):
    """
    Multi-phase optimization for minimum peak autocorrelation:
    1. FFT-accelerated ADAM + top-k LSE at base length
    2. Subgradient polish
    3. Local mass-redistribution search
    4. Multi-length exploration (longer sequences)
    5. Multi-start diversity
    6. Final combined polish
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    # Reserve 120s buffer for safety
    time_budget = min(float(time_budget) - 120.0, 1680.0)

    # Get parent sequence
    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    n_base = len(best_seq) if best_seq is not None else 1600

    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # === Phase 1: ADAM + top-k LSE at base length (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: ADAM-LSE n={n_base} ({remaining:.0f}s left) ===")
    t1 = min(remaining * 0.25, 400.0)
    if t1 > 30:
        a_out, _ = adam_nesterov_topk_lse(
            best_seq, t1, T_start=80.0, T_end=0.2, n_stages=30, topk=40, lr_base=0.015
        )
        try_update(a_out, f"adam_n{n_base}")

    # === Phase 2: Subgradient polish ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 2: Subgradient polish n={n_base} ({remaining:.0f}s left) ===")
    t2 = min(remaining * 0.10, 120.0)
    if t2 > 10:
        a_out, _ = subgradient_polish(best_seq, t2, lr=0.003)
        try_update(a_out, f"subgrad_n{n_base}")

    # === Phase 3: Local mass search ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 3: Local mass search n={n_base} ({remaining:.0f}s left) ===")
    t3 = min(remaining * 0.10, 120.0)
    if t3 > 10:
        a_out, _ = local_mass_search(best_seq, t3, seed=77)
        try_update(a_out, f"local_n{n_base}")

    # === Phase 4: ADAM at longer length ===
    n_long1 = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 4: ADAM-LSE n={n_long1} ({remaining:.0f}s left) ===")
    t4 = min(remaining * 0.18, 250.0)
    if t4 > 30:
        init_long1 = interpolate_sequence(best_seq, n_long1)
        a_out, _ = adam_nesterov_topk_lse(
            init_long1, t4, T_start=60.0, T_end=0.3, n_stages=25, topk=35, lr_base=0.012
        )
        try_update(a_out, f"adam_n{n_long1}")

    # === Phase 5: Subgradient polish at longer length ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 20:
        t5 = min(remaining * 0.08, 100.0)
        if t5 > 10:
            print(f"=== Phase 5: Subgradient n={n_long1} ({remaining:.0f}s left) ===")
            a_out, _ = subgradient_polish(best_seq, t5, lr=0.003)
            try_update(a_out, f"subgrad_long")

    # === Phase 6: ADAM at even longer length ===
    n_long2 = n_base + 800
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 60:
        print(f"=== Phase 6: ADAM-LSE n={n_long2} ({remaining:.0f}s left) ===")
        t6 = min(remaining * 0.15, 200.0)
        if t6 > 30:
            init_long2 = interpolate_sequence(best_seq, n_long2)
            a_out, _ = adam_nesterov_topk_lse(
                init_long2, t6, T_start=50.0, T_end=0.5, n_stages=20, topk=30, lr_base=0.01
            )
            try_update(a_out, f"adam_n{n_long2}")

    # === Phase 7: Multi-start diversity at base length ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 60:
        print(f"=== Phase 7: Multi-start ({remaining:.0f}s left) ===")
        n_starts = 3
        t7_each = min(remaining * 0.12 / n_starts, 80.0)
        for si in range(n_starts):
            if time.monotonic() - t_global_start > time_budget - 60:
                break
            if t7_each < 10:
                break
            pert = perturbed_init(best_seq, rng, scale=0.03 * (si + 1))
            a_out, _ = adam_nesterov_topk_lse(
                pert, t7_each, T_start=30.0, T_end=1.0, n_stages=12, topk=25, lr_base=0.01
            )
            try_update(a_out, f"multi_{si}")

    # === Phase 8: Final ADAM polish (low temperature) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 30:
        print(f"=== Phase 8: Final polish ({remaining:.0f}s left) ===")
        t8 = min(remaining * 0.20, 150.0)
        if t8 > 10:
            a_out, _ = adam_nesterov_topk_lse(
                best_seq, t8, T_start=10.0, T_end=0.1, n_stages=15, topk=50, lr_base=0.005
            )
            try_update(a_out, "final_adam")

    # === Phase 9: Final local search ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 20:
        t9 = min(remaining * 0.30, 100.0)
        if t9 > 10:
            print(f"=== Phase 9: Final local search ({remaining:.0f}s left) ===")
            a_out, _ = local_mass_search(best_seq, t9, seed=999)
            try_update(a_out, "final_local")

    # === Phase 10: One more subgradient pass ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 15:
        t10 = min(remaining * 0.5, 60.0)
        if t10 > 5:
            print(f"=== Phase 10: Final subgradient ({remaining:.0f}s left) ===")
            a_out, _ = subgradient_polish(best_seq, t10, lr=0.002)
            try_update(a_out, "final_subgrad")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
