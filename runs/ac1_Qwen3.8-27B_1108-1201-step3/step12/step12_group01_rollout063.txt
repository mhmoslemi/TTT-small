```python
def lse_gradient_full(a, T):
    """Compute gradient of 2n * LSE_T(conv(a,a)) w.r.t. a, assuming sum(a)=1."""
    n = len(a)
    # Step 1: conv = convolve(a, a)
    size1 = 2 * n - 1
    fft1 = 1 << (size1 - 1).bit_length()
    A = rfft(a, fft1)
    conv = irfft(A * A, fft1)[:size1]
    
    # Step 2: w = softmax(T * conv)
    Tconv = T * conv
    Tconv_shifted = Tconv - Tconv.max()
    exp_Tconv = np.exp(Tconv_shifted)
    w = exp_Tconv / exp_Tconv.sum()
    
    # Step 3: a_rev
    a_rev = a[::-1]
    
    # Step 4: W_a = convolve(w, a_rev)[n-1 : 2n-1]
    size2 = (2*n-1) + n - 1  # = 3n-1
    fft2 = 1 << (size2 - 1).bit_length()
    W = rfft(w, fft2)
    Ar = rfft(a_rev, fft2)
    Wa_full = irfft(W * Ar, fft2)
    W_a = Wa_full[n-1 : 2*n-1]
    
    # Step 5: grad_a = 2 * W_a
    grad_a = 2.0 * W_a
    
    # LSE value
    lse_val = (Tconv.max() + np.log(exp_Tconv.sum())) / T
    
    return lse_val, grad_a, conv
```
