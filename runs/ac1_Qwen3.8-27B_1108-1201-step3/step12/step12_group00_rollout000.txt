"""
Minimizes J(a) = 2n*max(conv(a,a))/(sum(a))^2 using:
1. LP-Frank-Wolfe with constraint generation and line search
2. Mirror descent (exponential tilting) with adaptive step size

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

# Global solver cache
_solver_cache = {}

def get_lp_solver():
    """Get a genesis-based LP solver for simplex method."""
    if 'lp' in _solver_cache:
        return _solver_cache['lp']
    # Try simplex-based solvers first
    for s in [cp.GUROBI, cp.CBC, cp.GLOP, cp.GLK, cp.SCIP]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Maximize(cp.sum(x)), [cp.sum(x) <= 1, x <= 0.2])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver_cache['lp'] = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    # Fallback
    for s in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Maximize(cp.sum(x)), [cp.sum(x) <= 1, x <= 0.2])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver_cache['lp'] = s
                print(f"  LP solver (fallback): {s}")
                return s
        except Exception:
            continue
    _solver_cache['lp'] = cp.SCS
    print(f"  LP solver (default): SCS")
    return cp.SCS


def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_score(a):
    """Compute the objective value (lower is better)."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    c = fast_autocorr(a)
    return float(2.0 * len(a) * c.max() / (s * s))


def build_conv_cross_matrix(y, K=None):
    """
    Build the matrix T such that (T @ b)_k = sum_i y_i * b_{k-i}
    for k = 0, ..., 2n-2.
    If K is given, only include the top-K peak indices.
    """
    n = len(y)
    size = 2 * n - 1
    if K is None:
        indices = np.arange(size)
    else:
        c = fast_autocorr(y)  # This is y*y, not y*b, but we use it to identify peaks
        # Actually we need to identify which k values are constraining
        # For now, use all or top-K based on the autocorrelation of y
        indices = np.argsort(c)[::-1][:K]
    
    T = np.zeros((len(indices), n))
    for row, k in enumerate(indices):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            T[row, j] = y[k - j]
    return T, indices


def lp_solve(y, M, K=None, solver=None):
    """
    Solve: max sum(b) s.t. b >= 0, (y*b)_k <= M for selected k.
    Returns b (length n) or None.
    """
    n = len(y)
    if solver is None:
        solver = get_lp_solver()
    
    b_var = cp.Variable(n, nonneg=True)
    
    if K is None or K >= 2 * n - 1:
        # Full LP
        T, _ = build_conv_cross_matrix(y)
    else:
        T, _ = build_conv_cross_matrix(y, K)
    
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def fw_line_search(a, b, max_iter=50):
    """
    Find best gamma in [0,1] minimizing max_k c_k((1-gamma)*a + gamma*b).
    Uses golden section search since f(gamma) is convex (max of quadratics).
    """
    n = len(a)
    
    def f(gamma):
        x = (1.0 - gamma) * a + gamma * b
        x = np.maximum(x, 0.0)
        s = x.sum()
        if s < 1e-15:
            return 0.0  # Won't happen if a and b are valid
        c = fast_autocorr(x)
        return c.max() / (s * s)  # We minimize this (proportional to J)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1, f2 = f(c1), f(c2)
    
    for _ in range(max_iter):
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = f(c1)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = f(c2)
    
    gamma = (lo + hi) / 2.0
    return gamma


def nesterov_fw_optimize(a0, time_limit, solver=None, K_full=None, K_start=30, 
                          use_line_search=True, verbose=True):
    """
    Frank-Wolfe with Nesterov acceleration and optional constraint generation.
    
    Parameters:
    - a0: initial sequence (will be normalized)
    - time_limit: seconds
    - solver: CVXPY solver
    - K_full: if set, always use this many constraints; None = all
    - K_start: initial number of constraints (grows over time)
    - use_line_search: whether to use line search for step size
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a * norm / s0
    
    if solver is None:
        solver = get_lp_solver()
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max() / (a.sum() ** 2)  # Proportional to J
    t0 = time.monotonic()
    
    y = a.copy()  # Nesterov momentum point
    avg_a = np.zeros(n)
    n_avg = 0
    n_iters = 0
    K = K_start
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Increase K over time (constraint generation -> full)
        if K_full is not None:
            K = min(K_full, K_start + t // 50)
        else:
            K = None  # Always full
        
        # Compute current max autocorrelation at y
        c_y = fast_autocorr(y)
        M = c_y.max()
        if M < 1e-15:
            break
        
        # Solve LP
        b = lp_solve(y, M, K=K, solver=solver)
        if b is None:
            # Fallback: use uniform
            b = np.ones(n) * norm / n
            n_iters += 1
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            b = np.ones(n) * norm / n
            b_sum = norm
        g = b * norm / b_sum  # Normalize to same scale
        n_iters += 1
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        # Optional line search (only every few iterations to save time)
        if use_line_search and t % 10 == 0 and time.monotonic() - t0 < time_limit - 5:
            gamma = fw_line_search(a, g, max_iter=30)
            a_ls = (1.0 - gamma) * a + gamma * g
            a_ls = np.maximum(a_ls, 0.0)
            s_ls = a_ls.sum()
            if s_ls > 1e-15:
                a_ls = a_ls * norm / s_ls
                score_ls = fast_autocorr(a_ls).max()
                if score_ls < best_score:
                    best_score = score_ls
                    best_a = a_ls.copy()
                    a = a_ls.copy()
        
        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        # Running average
        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
        
        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_score
            print(f"    FW-Nest iter={n_iters} ({elapsed:.0f}s): best_max_c={best_score:.8f}, J={J:.6f}, K={K}")
    
    return best_a, best_score


def mirror_descent_optimize(a0, time_limit, verbose=True):
    """
    Mirror descent (exponential tilting) on the simplex.
    a_i <- a_i * exp(-eta * g_i) / Z
    where g_i = 2 * a_{k*-i} (subgradient at the peak)
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Adaptive learning rate
    eta = 1.0 / np.sqrt(n)
    decay = 0.999
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_autocorr(a)
        k_star = int(np.argmax(c))
        
        # Subgradient: g_j = 2 * a_{k*-j} (with zero-padding)
        g = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                g[j] = 2.0 * a[idx]
        
        # Exponential tilting
        # a_i <- a_i * exp(-eta * g_i / max_g) / Z
        max_g = g.max()
        if max_g < 1e-15:
            break
        logits = np.log(np.maximum(a, 1e-300)) - (eta / max_g) * g
        logits -= logits.max()  # Numerical stability
        weights = np.exp(logits)
        Z = weights.sum()
        if Z < 1e-300:
            break
        a_new = weights / Z * norm
        
        # Track best
        score = fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        eta *= decay
        
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_score
            print(f"    Mirror t={t} ({elapsed:.0f}s): best_max_c={best_score:.8f}, J={J:.6f}, eta={eta:.6f}")
    
    return best_a, best_score


def coordinate_refine(a0, time_limit, verbose=True):
    """
    Coordinate-wise local refinement.
    For each coordinate, try moving mass from high-contributing positions to low-contributing ones.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a0, fast_autocorr(a0).max()
    a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    stall = 0
    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a)
        top_k = np.argsort(c)[::-1][:5]
        
        improved = False
        for k in top_k:
            # Find which pairs (i, k-i) contribute most to c_k
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            max_contrib_idx = idx[np.argmax(contrib)]
            
            # Find a position that doesn't contribute to any top peak
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True
            
            # Try moving mass from max_contrib_idx to a random non-peak position
            non_peak = np.where(~peak_mask)[0]
            if len(non_peak) == 0:
                non_peak = np.where(np.arange(n) != max_contrib_idx)[0]
            
            dst = non_peak[rng.integers(len(non_peak))]
            
            for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.05]:
                amt = a[max_contrib_idx] * frac
                if amt < 1e-20:
                    continue
                a_trial = a.copy()
                a_trial[max_contrib_idx] -= amt
                a_trial[dst] += amt
                if a_trial[max_contrib_idx] < 0:
                    continue
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score * (1.0 - 1e-10):
                    a = a_trial.copy()
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break
        
        if not improved:
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-15:
                frac = rng.uniform(0.01, 0.1)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial.copy()
                            best_score = score
                            best_a = a.copy()
                            stall = 0
                        else:
                            stall += 1
                    else:
                        stall += 1
                else:
                    stall += 1
            else:
                stall += 1
            
            if stall > 500:
                break
    
    return best_a, best_score


def interpolate_to_length(f, new_n):
    """Interpolate sequence to a new length, preserving the shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng=None):
    """Generate a structured initial sequence."""
    if rng is None:
        rng = np.random.default_rng(0)
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = np.ones(n) / n
    else:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - x) ** 0.5 + 0.3 * np.sin(4.0 * np.pi * x)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
