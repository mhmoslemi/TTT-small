    return a / s

    """Safely evaluate a sequence using the global evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        if len(a_list) < 1:
            return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def subgradient_mirror_descent(a0, time_limit, delta_frac=0.003, eta0=0.3, eta_decay=0.99995, check_interval=50):
    """
    Subgradient method with multiplicative weight (mirror descent) updates.
    Minimizes max_k (a*a)_k over the simplex.
    """
    n = len(a0)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    eta = eta0
    running_sum = np.zeros(n)
    running_count = 0
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        b_max = b.max()
        threshold = b_max * (1.0 - delta_frac)
        S = np.where(b >= threshold)[0]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in S:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[js:je+1] += 2.0 * a[k - idx]
        grad /= float(len(S))
        
        # Mirror descent (multiplicative weight) step
        log_a = np.log(np.maximum(a, 1e-300))
        log_a = log_a - eta * grad
        log_a = log_a - log_a.max()
        a = np.exp(log_a)
        s = a.sum()
        if s < 1e-300:
            a = np.ones(n) / n
        else:
            a = a / s
        
        it += 1
        eta *= eta_decay
        
        # Periodic score check
        if it % check_interval == 0:
            running_sum += a
            running_count += 1
            
            score = compute_score(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            # Check running average
            if running_count >= 20:
                avg_a = running_sum / running_count
                avg_score = compute_score(avg_a)
                if avg_score < best_score:
                    best_score = avg_score
                    best_a = avg_a.copy()
                    no_improve = 0
            
            # Reset running average periodically
            if running_count >= 200:
                running_sum = a.copy()
                running_count = 1
            
            if no_improve > 200:
                # Try a different delta
                delta_frac *= 0.5
                if delta_frac < 1e-6:
                    break
                no_improve = 0
            
            if it % (check_interval * 100) == 0:
                elapsed = time.monotonic() - t0
                print(f"    SG iter {it} ({elapsed:.0f}s): best={best_score:.6f}, eta={eta:.6f}, |S|={len(S)}")
    
    return best_a, best_score


def lse_obj_grad_z(z, T):
    """Objective and gradient for z-parameterization (a = softmax(z))."""
    a = softmax_fn(z)
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    obj_val = (max_Tb + np.log(exp_Tb.sum())) / T
    
    w = exp_Tb / exp_Tb.sum()
    
    size = 1 << (2 * n - 1).bit_length()
    A_rev = rfft(a[::-1].copy(), size)
    W = rfft(w, size)
    conv = irfft(A_rev * W, size)
    
    grad_a = 2.0 * conv[n - 1:n - 1 + n]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj_val, grad_z


def adam_lse_optimize(a0, time_limit, T_start=200.0, T_end=0.02, n_stages=25, lr=0.005):
    """
    LogSumExp-smoothed optimization with Adam optimizer.
    """
    n = len(a0)
    a_init = normalize_nonneg(a0)
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = compute_score(a_init)
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_adam = 0
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        # Adapt iterations per stage based on remaining time
        stages_left = n_stages - si
        iters = max(10, int(remaining * 100 / stages_left))
        iters = min(iters, 500)
        
        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break
            
            obj, grad = lse_obj_grad_z(z, stage_T)
            
            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)
            
            lr_t = lr / (1.0 + 0.01 * it)
            z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            
            if it % 20 == 0:
                a = softmax_fn(z)
                score = compute_score(a)
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
        
        a = softmax_fn(z)
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam-LSE stage {si}/{n_stages} (T={stage_T:.1f}, {elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a_fixed):
    """Build the (2n-1) x n convolution matrix T where (T@b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0. Returns (b, M)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)
