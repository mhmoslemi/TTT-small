def diverse_init(n, rng, method=None):
    """Generate diverse initial sequences."""
    if method is None:
        method = rng.integers(0, 10)
    
    if method == 0:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif method == 1:
        # Gaussian
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif method == 2:
        # Sine squared
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif method == 3:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** 0.5
    elif method == 4:
        # Multi-frequency
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(3 * x) + 0.2 * np.cos(7 * x)
    elif method == 5:
        # Double Gaussian
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * (x - 0.3) ** 2) + 0.7 * np.exp(-4 * (x - 0.7) ** 2)
    elif method == 6:
        # Chirp-modulated
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(10 * np.pi * x ** 2)
        a = np.maximum(a, 0)
    elif method == 7:
        # Triangle with ripple
        x = np.linspace(0, 1, n)
        tri = 1 - 2 * np.abs(x - 0.5)
        a = tri * (1 + 0.3 * np.sin(8 * np.pi * x))
        a = np.maximum(a, 0)
    elif method == 8:
        # Random smoothed
        a = rng.random(n)
        # Smooth with a small kernel
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    else:
        # Monotone decreasing with noise
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 1.2 + 0.1 * rng.random(n)
    
    a = np.maximum(a, 0)
    return normalize(a)

def perturbation(a, rng, scale=0.1):
    """Apply random perturbation to a normalized sequence."""
    n = len(a)
    # Multiplicative perturbation
    noise = 1 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0)
    a_pert = a * noise
    return normalize(a_pert)

def interpolate_to_length(a, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a = normalize(a)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a)
    return normalize(a_interp)

def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

import numpy as np

import time
from scipy.fft import rfft, irfft
from scipy.sparse import dia_matrix, csc_matrix


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(arr):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in arr]

    except Exception:
        return float('inf')


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def spectral_flatten_timed(a, time_limit, alpha=0.3):
    """Spectrally flatten with time limit."""
    n = len(a)
    a = normalize(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        ratio = target / (non_dc + 1e-30)
        ratio = np.clip(ratio, 0.05, 20.0)
        scale = np.ones(len(A))
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = normalize(a_new)
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score


def topk_subgradient(a0, time_limit, k_peaks=10, step_decay=0.999):
    """
    Subgradient descent using top-k peak averaging.
    Fast (no LP) but potentially slower convergence than LP game.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    step = 0.1

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        max_b = b.max()

        # Find top-k peaks
        if k_peaks <= 4000:
            top_idx = np.argpartition(b, -k_peaks)[-k_peaks:]
        else:
            top_idx = np.arange(len(b))

        # Compute subgradient: average of 2*shifted(a) over top peaks
        grad = np.zeros(n)
        for k in top_idx:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # grad[lo:hi+1] += 2 * a[k-hi : k-lo+1] reversed
            grad[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
        grad /= k_peaks

        # Projected subgradient step
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        # Adaptive step: reduce if score increased
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            step *= step_decay
            if step < 1e-8:
                break

        if t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"          Subgrad t={t} ({elapsed:.0f}s): best={best_score:.6f}, step={step:.6f}")

    return best_a, best_score


def build_conv_matrix_sparse(a_fixed):
    """Build sparse convolution matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    # T[k, j] = a_fixed[k-j] if 0 <= k-j <= n-1
    # Diagonal d (offset): T[k, k+d] = a_fixed[-d] for -d in [0, n-1]
    # So d ranges from -(n-1) to 0
    diags = []
    offsets = []
    for d in range(-(n-1), 1):
        val_idx = -d
        if 0 <= val_idx <= n-1:
            length = n - abs(d)
            diags.append(np.full(length, a_fixed[val_idx]))
            offsets.append(d)
    T = dia_matrix((diags, offsets), shape=(2*n-1, n))
    return T.tocsc()


def solve_lp_bresponse(a_fixed, M_bound, solver=None):
    """
    Solve LP: max sum(b) s.t. conv(a_fixed, b) <= M_bound, b >= 0.
    Returns (b_normalized, M_optimal) or (None, None).
    """
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    T = build_conv_matrix_sparse(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers = [solver] if solver else []
    for s_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        if s_name in solvers:
            continue
        solvers.append(s_name)

    for s in solvers:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                break
        except Exception:
            continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s_val = b_val.sum()
    if s_val < 1e-15:
        return None, None
    b_val = b_val / s_val
    return b_val, float(M_var.value)


def lp_game_iterate(a0, time_limit, solver=None, max_iters=100000, m_update_freq=5):
    """
    Nesterov-accelerated LP zero-sum game with dynamic M updates.
    The LP: min M' s.t. conv(y, b) <= M', sum(b)=1, b>=0
    gives both the best response b and the current game value M'.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()

    # Pre-build T for initial y (will rebuild when y changes significantly)
    T_cache = None
    y_cache = None

    lp_count = 0
    lp_fail_count = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        # Solve LP best response to y
        b_star, M_lp = solve_lp_bresponse(y, None, solver)
        if b_star is None:
            lp_fail_count += 1
            if lp_fail_count > 20:
                break
            continue
        lp_count += 1
        lp_fail_count = 0

        # Nesterov acceleration
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s

        y = y_new
        a = a_new

        # Track best
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Also check running average (smooths out oscillations)
        if t % 50 == 0:
            avg_a = y  # y is already a running average due to Nesterov
            score_avg = 2.0 * n * fast_autocorr(avg_a).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_a.copy()

        if t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"          LP-Game t={t} ({elapsed:.0f}s): best={best_score:.6f}, M_lp={M_lp:.8f}")

    return best_a, best_score

def perturbation(a, rng, scale=0.1):
    """Apply random multiplicative perturbation."""
    n = len(a)
    noise = 1 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0)
    a_pert = a * noise
    return normalize(a_pert)


def interpolate_to_length(a, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a = normalize(a)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a)
    return normalize(a_interp)

            sc_arr[1:] = np.power(ratio, 0.15)

            a_work = normalize(irfft(A_new, fft_size)[:n])

        # Subgradient for a bit
        t3_subgrad = t3 * 0.2
        if t3_subgrad > 10:
            a_work, _ = topk_subgradient(a_work, t3_subgrad, k_peaks=15)

        # LP game
        t3_lp = t3 - (time.monotonic() - t_s)
        if t3_lp > 30 and nl <= 2500:
            a_out, sc = lp_game_iterate(a_work, t3_lp, solver)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  LP n={nl}: {best_score:.6f}")
        elif t3_lp > 20:
            # Too large for LP, use subgradient
            a_out, sc = topk_subgradient(a_work, t3_lp, k_peaks=20)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  Subgrad n={nl}: {best_score:.6f}")

    # === Phase 4: Perturbation + Restart ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: Perturbation restarts (rem={remaining:.0f}s) ===")

    n_restarts = 4
    t_per_restart = remaining * 0.55 / n_restarts

    for r in range(n_restarts):
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining < 40:
            break

        # Different perturbation scale for each restart
        scale = 0.03 + 0.03 * r
        a_pert = perturbation(best_seq, rng, scale=scale)

        # Brief spectral
        n = len(a_pert)
        fft_size = 1 << (2 * n - 1).bit_length()
        a_work = a_pert.copy()
        for _ in range(50):
            A = rfft(a_work, fft_size)
            B = np.abs(A) ** 2
            non_dc = B[1:]
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))
            ratio = target / (non_dc + 1e-30)
            ratio = np.clip(ratio, 0.05, 20.0)
            sc_arr = np.ones(len(A))
            sc_arr[1:] = np.power(ratio, 0.1)
            A_new = A * sc_arr
            a_work = normalize(irfft(A_new, fft_size)[:n])

        # Optimize: subgradient if n large, LP if n moderate
        t_opt = t_per_restart - 15
        if t_opt > 20:
            if n <= 2000:
                a_out, sc = lp_game_iterate(a_work, t_opt, solver)
            else:
                a_out, sc = topk_subgradient(a_work, t_opt, k_peaks=20)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  Restart {r} (scale={scale:.3f}): {best_score:.6f}")
            else:
                print(f"  Restart {r} (scale={scale:.3f}): no improvement ({sc:.6f})")

    # === Phase 5: Final intensive polish ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: Final polish (rem={remaining:.0f}s) ===")

    if remaining > 30:
        # Try LP if length allows
        n_final = len(best_seq)
        if n_final <= 2500:
            t5 = remaining * 0.6
            a_out, sc = lp_game_iterate(best_seq, t5, solver)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  Final LP: {best_score:.6f}")
        else:
            t5 = remaining * 0.6
            a_out, sc = topk_subgradient(best_seq, t5, k_peaks=10)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  Final Subgrad: {best_score:.6f}")

        # Final spectral touch
        remaining2 = time_budget - (time.monotonic() - t_global_start)
        if remaining2 > 5:
            a_out, sc = spectral_flatten_timed(best_seq, remaining2 * 0.3, alpha=0.05)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  Final Spectral: {best_score:.6f}")

    # Final verification
    if best_seq is not None:
        best_seq = np.maximum(best_seq, 0)
        s = best_seq.sum()
        if s > 0:
            best_seq = best_seq / s
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
