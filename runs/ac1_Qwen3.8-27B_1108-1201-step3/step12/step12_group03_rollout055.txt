def fast_conv(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Algorithm from "Projection onto a simplex" by Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smooth_max_grad(a, au):
    """
    Compute gradient of smooth max: au * log(sum(exp(b_k/au)))
    where b = conv(a, a).
    
    Returns (gradient, value).
    """
    n = len(a)
    b = fast_conv(a)
    
    # Compute softmax weights
    b_scaled = b / au
    b_max = b_scaled.max()
    exp_b = np.exp(b_scaled - b_max)
    w = exp_b / exp_b.sum()
    
    value = au * (b_max + np.log(exp_b.sum()))
    
    # Gradient: grad[i] = sum_j a[j] * w[i+j]
    # = conv(a_rev, w)[n-1 : 2n-1]
    a_rev = a[::-1]
    conv_result = fast_conv_cross(a_rev, w)
    grad = conv_result[n-1:2*n-1]
    
    return grad, value


def fista_smoothed(a0, au_schedule, time_limit, verbose=True):
    """
    FISTA with smoothed max objective and exponential temperature annealing.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    x = a.copy()
    y = a.copy()
    t = 1.0
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_conv(a).max()
    t0 = time.monotonic()
    
    n_iters = len(au_schedule)
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        au = au_schedule[it]
        lr = 1.0  # Will use backtracking
        
        # Compute gradient at y
        grad_y, val_y = smooth_max_grad(y, au)
        
        # Backtracking line search
        step = 1.0
        for bt in range(20):
            x_new = project_simplex(y - step * grad_y)
            grad_x, val_x = smooth_max_grad(x_new, au)
            
            # Check: f(x_new) <= f(y) + grad_y.(x_new - y) + (1/(2*step)) * ||x_new - y||^2
            rhs = val_y + np.dot(grad_y, x_new - y) + 0.5 / step * np.sum((x_new - y)**2)
            
            if val_x <= rhs + 1e-12:
                break
            step *= 0.5
        
        # FISTA update
        t_new = 0.5 * (1 + np.sqrt(1 + 4 * t**2))
        y = x_new + (t - 1) / t_new * (x_new - x)
        y = project_simplex(y)  # Ensure y stays in simplex (optional but helps)
        x = x_new
        t = t_new
        
        # Track best
        score = 2.0 * n * fast_conv(x).max()
        if score < best_score:
            best_score = score
            best_a = x.copy()
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"        FISTA it={it}/{n_iters}, au={au:.4f}, best={best_score:.6f}, cur={score:.6f} ({elapsed:.1f}s)")
    
    return best_a, best_score


def fista_simple(a0, au, n_iters, time_limit, verbose=False):
    """
    Simple gradient descent with momentum on smoothed max.
    More robust than full FISTA for this problem.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_conv(a).max()
    t0 = time.monotonic()
    
    momentum = 0.9
    v = np.zeros(n)
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive learning rate based on au
        lr = min(0.1, n * au / 10.0)
        
        grad, val = smooth_max_grad(a, au)
        
        # Momentum update
        v = momentum * v + (1 - momentum) * grad
        a_new = a - lr * v
        a_new = project_simplex(a_new)
        
        # Check improvement
        score = 2.0 * n * fast_conv(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        # Always take the step (for exploration)
        a = a_new
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"        GD it={it}/{n_iters}, au={au:.4f}, best={best_score:.6f}, cur={score:.6f} ({elapsed:.1f}s)")
    
    return best_a, best_score


def coordinate_descent(a0, time_limit, verbose=False):
    """
    Coordinate descent: for each coordinate, find the optimal value
    that minimizes the max autocorrelation.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    for sweep in range(10000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Random order for coordinates
        order = rng.permutation(n)
        improved = False
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Try moving mass from i to another coordinate
            # Find the coordinate that most contributes to the peak
            b = fast_conv(a)
            peak_lag = np.argmax(b)
            
            # Contribution of a[i] to the peak
            # b[peak_lag] includes a[i] * a[peak_lag - i] (if valid)
            j = peak_lag - i
            if 0 <= j < n and j != i:
                # Moving mass from i to some other position
                # Try transferring to position that minimizes peak
                for frac in [0.01, 0.02, 0.05, 0.1, 0.2]:
                    amt = a[i] * frac
                    if amt < 1e-15:
                        continue
                    # Find best target
                    best_target = -1
                    best_trial_score = best_score
                    for target in rng.choice(n, size=min(20, n), replace=False):
                        if target == i:
                            continue
                        a_trial = a.copy()
                        a_trial[i] -= amt
                        a_trial[target] += amt
                        a_trial = np.maximum(a_trial, 0)
                        ss = a_trial.sum()
                        if ss < 1e-15:
                            continue
                        a_trial = a_trial / ss
                        score = 2.0 * n * fast_conv(a_trial).max()
                        if score < best_trial_score:
                            best_trial_score = score
                            best_target = target
                    
                    if best_target >= 0 and best_trial_score < best_score:
                        a[i] -= amt
                        a[best_target] += amt
                        a = np.maximum(a, 0)
                        a = a / a.sum()
                        best_score = best_trial_score
                        best_a = a.copy()
                        improved = True
                        break
        
        if not improved and sweep > 100:
            break
    
    if verbose:
        print(f"        CD final: {best_score:.6f}")
    return best_a, best_score


def qp_equalize(a0, time_limit, n_peaks=10):
    """
    QP: minimize M subject to b_k <= M for top n_peaks lags, sum(a)=1, a>=0.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_conv(a).max()
    t0 = time.monotonic()
    
    for round_num in range(20):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_conv(a)
        # Get top lags
        top_lags = np.argsort(b)[::-1][:n_peaks]
        
        # Build QP
        a_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        
        constraints = [cp.sum(a_var) == 1, M_var >= 0]
        for k in top_lags:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            other_idx = k - idx
            q_expr = cp.sum(a_var[idx] * a_var[other_idx])
            constraints.append(q_expr <= M_var)
        
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        solved = False
        for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver, max_iter=5000)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue
        
        if not solved:
            break
        
        a_new = np.asarray(a_var.value, dtype=np.float64)
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        
        score = 2.0 * n * fast_conv(a_new).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            print(f"        QP round {round_num}: {best_score:.6f}")
        else:
            # Try mixing
            for frac in [0.2, 0.4, 0.6]:
                a_mix = (1 - frac) * a + frac * a_new
                a_mix = np.maximum(a_mix, 0)
                a_mix = a_mix / a_mix.sum()
                score_mix = 2.0 * n * fast_conv(a_mix).max()
                if score_mix < best_score:
                    best_score = score_mix
                    best_a = a_mix.copy()
                    a = a_mix
                    break
            else:
                if round_num >= 5:
                    break
                n_peaks = min(n_peaks + 5, 30)
    
    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

The main issue is: can we actually beat 1.504377? The C-S bound I computed earlier was wrong (I confused b[0] = a[0]^2 with b[0] = sum a_i^2). Let me redo the analysis.

For a sequence `a` with `sum(a) = 1`:
- `b[k] = sum_j a[j] * a[k-j]` (full convolution, k = 0 to 2n-2)
- `sum_k b[k] = (sum a)^2 = 1`
- So `max_k b[k] >= 1/(2n-1)` (by averaging)
- This gives a lower bound of `2n/(2n-1) ≈ 1` for the objective.

But we also have the constraint that `a >= 0`. For non-negative `a`:
- `b[0] = a[0]^2`
- `b[2n-2] = a[n-1]^2`
- `b[n-1] = sum_j a[j] * a[n-1-j]` (the "center" of the autocorrelation)

The center lag `n-1` tends to be the largest for "peaked" sequences. For uniform, `b[n-1] = n * (1/n)^2 = 1/n`.

Can we make `b[n-1] < 1/n`? This requires `sum_j a[j] * a[n-1-j] < 1/n`.

By the rearrangement inequality, `sum_j a[j] * a[n-1-j]` is minimized when one sequence is sorted and the other is reverse-sorted. But since both are the same sequence (just reversed), this doesn't directly help.

Actually, for `sum_j a[j] * a[n-1-j]` with `sum a = 1` and `a >= 0`:
- If `a` is symmetric (`a[j] = a[n-1-j]`), this becomes `sum_j a[j]^2 >= 1/n` (by C-S).
- If `a` is NOT symmetric, we can potentially get a smaller value.

For example, if `a = [1, 0, 0, ..., 0]` (all mass at position 0), then `b[n-1] = a[0]*a[n-1] = 0`. But `b[0] = a[0]^2 = 1`, which is terrible.

The key tension is: reducing the center lag increases edge lags, and vice versa. The optimum balances these.

I think the smoothed max approach should work well here. Let me also add a "debiasing" step: after optimization, try shifting mass from the center to the edges (or vice versa) to see if the overall peak decreases.

Let me also think about the problem size. For n=1600:
- FFT size: 4096 (next power of 2 for 2n-1=3199)

import numpy as np

def fast_conv(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    """Full cross-convolution conv(a, b) using FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smooth_max_grad(a, au):
    """
    Compute gradient and value of smooth max: au * log(sum(exp(b_k/au)))
    where b = conv(a, a).
    Returns (gradient_of_length_n, value).
    """
    n = len(a)
    b = fast_conv(a)
    b_scaled = b / au
    b_max = b_scaled.max()
    exp_b = np.exp(b_scaled - b_max)
    sum_exp = exp_b.sum()
    w = exp_b / sum_exp
    value = au * (b_max + np.log(sum_exp))
    # Gradient: grad[i] = sum_j a[j] * w[i+j]
    # = conv(a[::-1], w)[n-1 : 2n-1]
    a_rev = a[::-1]
    conv_result = fast_conv_cross(a_rev, w)
    grad = conv_result[n - 1:2 * n - 1]
    return grad, value


def momentum_gd(a0, au, n_iters, lr, momentum, time_limit, verbose=False):
    """
    Momentum gradient descent on smoothed max objective.
    au: smoothing parameter (temperature)
    lr: learning rate
    momentum: momentum coefficient
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_conv(a).max()
    t0 = time.monotonic()
    v = np.zeros(n)

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        grad, _ = smooth_max_grad(a, au)

        # Adaptive LR: scale by problem size
        effective_lr = lr * n / 100.0
        effective_lr = min(effective_lr, 0.5)

        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - effective_lr * v
        a_new = project_simplex(a_new)

        score = 2.0 * n * fast_conv(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        a = a_new

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"          GD it={it}, au={au:.6f}, best={best_score:.6f}, cur={score:.6f} ({elapsed:.1f}s)")

    return best_a, best_score


def annealed_optimization(a0, time_limit, verbose=True):
    """
    Multi-stage annealing: start with high temperature (smooth), decrease to low.
    Uses momentum gradient descent at each stage.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_conv(a).max()
    t0 = time.monotonic()

    # Temperature schedule: geometric from high to low
    n_stages = 25
    au_start = 3.0 / n
    au_end = 0.02 / n
    aus = au_start * (au_end / au_start) ** (np.arange(n_stages) / (n_stages - 1))

    # Iterations per stage: more at low temperature
    base_iters = max(200, int(3000 * time_limit / 300.0))

    for stage in range(n_stages):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        au = aus[stage]
        # More iterations at lower temperatures
        iters = int(base_iters * (1.0 + stage * 0.1))
        lr = min(0.3, 2.0 * n * au)
        momentum = 0.9 + 0.05 * (stage / n_stages)

        a, sc = momentum_gd(a, au, iters, lr, momentum, remaining - 5, verbose=False)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        else:
            # Keep the better of a and best_a
            sc_a = 2.0 * n * fast_conv(a).max()
            if sc_a < best_score:
                best_score = sc_a
                best_a = a.copy()
            # But continue from a (it might be on a good trajectory)

        if verbose and stage % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"          Stage {stage}/{n_stages}, au={au:.6f}, best={best_score:.6f} ({elapsed:.1f}s)")

    return best_a, best_score


def qp_equalize(a0, time_limit, n_peaks=10):
    """
    QP: minimize M subject to b_k <= M for top n_peaks lags, sum(a)=1, a>=0.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_conv(a).max()
    t0 = time.monotonic()

    for round_num in range(30):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_conv(a)
        top_lags = np.argsort(b)[::-1][:n_peaks]

        a_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()

        constraints = [cp.sum(a_var) == 1, M_var >= 0]
        for k in top_lags:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            other_idx = k - idx
            q_expr = cp.sum(a_var[idx] * a_var[other_idx])
            constraints.append(q_expr <= M_var)

        prob = cp.Problem(cp.Minimize(M_var), constraints)
        solved = False
        for solver in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
            try:
                prob.solve(solver=solver, max_iter=10000)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue

        if not solved:
            break

        a_new = np.asarray(a_var.value, dtype=np.float64)
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss

        score = 2.0 * n * fast_conv(a_new).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            if round_num % 5 == 0:
                print(f"          QP round {round_num}: {best_score:.6f}")
        else:
            # Try mixing
            improved = False
            for frac in [0.1, 0.2, 0.3, 0.5, 0.7]:
                a_mix = (1 - frac) * a + frac * a_new
                a_mix = np.maximum(a_mix, 0)
                a_mix = a_mix / a_mix.sum()
                score_mix = 2.0 * n * fast_conv(a_mix).max()
                if score_mix < best_score - 1e-12:
                    best_score = score_mix
                    best_a = a_mix.copy()
                    a = a_mix
                    improved = True
                    break
            if not improved:
                if round_num >= 5:
                    break
                n_peaks = min(n_peaks + 5, 40)

    return best_a, best_score

def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def generate_initializations(n, seed=42):
    """Generate diverse initializations for optimization."""
    rng = np.random.default_rng(seed)
    inits = {}
    centers = np.arange(n) - (n - 1) / 2.0

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # Cosine bell
    x = np.linspace(0, np.pi, n)
    cos_bell = np.cos(x / 2.0) ** 2
    cos_bell = cos_bell / cos_bell.sum()
    inits['cosine'] = cos_bell

    # Laplace
    for sigma_factor in [3, 5, 10]:
        sigma = n / sigma_factor
        lap = np.exp(-np.abs(centers) / sigma)
        lap = lap / lap.sum()
        inits[f'laplace_{sigma_factor}'] = lap

    # U-shape (high at edges)
    u_shape = 1.0 / (1.0 + np.exp(-np.abs(centers) / (n / 4.0)))
    u_shape = u_shape / u_shape.sum()
    inits['u_shape'] = u_shape

    # Two peaks
    for dist in [0.2, 0.3]:
        p1 = np.exp(-((centers - n * dist) ** 2) / (2 * (n / 12.0) ** 2))
        p2 = np.exp(-((centers + n * dist) ** 2) / (2 * (n / 12.0) ** 2))
        two = p1 + p2
        two = two / two.sum()
        inits[f'twopc_{dist}'] = two

    # Random
    rand = np.abs(rng.standard_normal(n))
    rand = rand / rand.sum()
    inits['random'] = rand

    # Sparse random
    sparse = np.zeros(n)
    n_nonzero = max(10, n // 4)
    nonzero_idx = rng.choice(n, size=n_nonzero, replace=False)
    sparse[nonzero_idx] = np.abs(rng.standard_normal(n_nonzero))
    sparse = sparse / sparse.sum()
    inits['sparse'] = sparse

    # Gaussian with varying width
    for width in [0.15, 0.25, 0.4]:
        g = np.exp(-centers ** 2 / (2 * (n * width) ** 2))
        g = g / g.sum()
        inits[f'gauss_{width}'] = g

    return inits

        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Determine base length
    n_base = len(best_seq) if best_seq is not None else 1600

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0, np.pi, n_base)
        best_seq = np.cos(x / 2.0) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # Phase 1: Annealed optimization from parent
    print(f"\n=== Phase 1: Annealed GD from parent at n={n_base} ===")
    remaining = time_budget - (time.monotonic() - t_global_start)
    t1 = min(remaining * 0.30, 500.0)
    if t1 > 30:
        a_out, sc_out = annealed_optimization(best_seq, t1, verbose=True)
        try_update(a_out, "anneal_parent")

    # Phase 2: Try different initializations
    print(f"\n=== Phase 2: Multi-init optimization at n={n_base} ===")
    remaining = time_budget - (time.monotonic() - t_global_start)
    t2 = min(remaining * 0.30, 450.0)
    if t2 > 60:
        inits = generate_initializations(n_base, seed=seed)

        # Add parent
        inits['parent'] = best_seq.copy()

        # Score all inits
        init_scores = {}
        for name, init in inits.items():
            sc = 2.0 * n_base * fast_conv(init).max()
            init_scores[name] = sc
            if sc < best_score:
                best_score = sc
                best_seq = init.copy()

        # Sort by score
        sorted_inits = sorted(init_scores.keys(), key=lambda x: init_scores[x])

        # Allocate time: top 3 get more time
        n_to_try = min(5, len(sorted_inits))
        time_per_init = t2 / n_to_try

        for i, name in enumerate(sorted_inits[:n_to_try]):
            remaining = time_budget - (time.monotonic() - t_global_start)
            if remaining < 20:
                break
            t_alloc = min(time_per_init, remaining - 30)
            if t_alloc < 15:
                continue
            print(f"    Init '{name}' (score={init_scores[name]:.6f}), time={t_alloc:.0f}s")

            a_out, sc_out = annealed_optimization(inits[name], t_alloc, verbose=False)
            try_update(a_out, f"init_{name}")

            # Also try symmetrized version
            if name != 'parent':
                a_sym = (inits[name] + inits[name][::-1]) / 2.0
                a_sym = a_sym / a_sym.sum()
                sc_sym = 2.0 * n_base * fast_conv(a_sym).max()
                if sc_sym < best_score:
                    a_out2, sc_out2 = annealed_optimization(a_sym, t_alloc * 0.5, verbose=False)
                    try_update(a_out2, f"sym_{name}")

    # Phase 3: QP refinement
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: QP Refinement (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 20:
        a_out, sc_out = qp_equalize(best_seq, t3, n_peaks=10)
        try_update(a_out, "qp_10")

    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 30:
        t3b = min(remaining * 0.08, 80.0)
        if t3b > 10:
            a_out, sc_out = qp_equalize(best_seq, t3b, n_peaks=20)
            try_update(a_out, "qp_20")

    # Phase 4: Coordinate descent
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: Coordinate Descent (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 15:
        a_out, sc_out = coordinate_descent_peaks(best_seq, t4, seed=seed)
        try_update(a_out, "cd_1")

    # Phase 5: Another anneal from current best (with lower starting temp)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: Low-temp Anneal (remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 20:
        # Modify annealed_optimization to start from lower temp
        n = len(best_seq)
        a = best_seq.copy()
        best_low = a.copy()
        best_low_score = 2.0 * n * fast_conv(a).max()
        t_low_start = time.monotonic()

        n_stages = 15
        au_start = 0.1 / n
        au_end = 0.005 / n
        aus = au_start * (au_end / au_start) ** (np.arange(n_stages) / (n_stages - 1.0))

        for stage in range(n_stages):
            rem = t5 - (time.monotonic() - t_low_start)
            if rem < 5:
                break
            au = aus[stage]
            iters = max(300, int(2000 * t5 / 200.0))
            lr = min(0.2, 3.0 * n * au)
            momentum = 0.92
            a, sc = momentum_gd(a, au, iters, lr, momentum, rem - 5, verbose=False)
            if sc < best_low_score:
                best_low_score = sc
                best_low = a.copy()

        try_update(best_low, "lowtemp_anneal")
        a_cur = best_low

    # Phase 6: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: Multi-length Exploration (remaining={remaining:.0f}s) ===")

    # Try several nearby lengths
    length_offsets = [50, -50, 100, -100, 200, -200, 300, -300]
    tried_lengths = set()

    for offset in length_offsets:
        n_try = n_base + offset
        if n_try < 300 or n_try in tried_lengths:
            continue
        tried_lengths.add(n_try)
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining < 60:
            break

        print(f"    Trying n={n_try} (remaining={remaining:.0f}s)")
        t_alloc = min(remaining * 0.10, 120.0)
        if t_alloc < 20:
            continue

        init_try = interpolate_seq(best_seq, n_try)
        a_out, sc_out = annealed_optimization(init_try, t_alloc, verbose=False)
        try_update(a_out, f"len_{n_try}")

        # Quick QP
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining > 40:
            t_qp = min(remaining * 0.04, 40.0)
            if t_qp > 10:
                a_qp, sc_qp = qp_equalize(a_out, t_qp, n_peaks=8)
                try_update(a_qp, f"qp_len_{n_try}")

    # Phase 7: Final coordinate descent + QP
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 7: Final Refinement (remaining={remaining:.0f}s) ===")

    if remaining > 20:
        t_cd = min(remaining * 0.3, 80.0)
        if t_cd > 10:
            a_out, sc_out = coordinate_descent_peaks(best_seq, t_cd, seed=seed + 100)
            try_update(a_out, "final_cd")

    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 20:
        t_qp = min(remaining * 0.3, 60.0)
        if t_qp > 10:
            a_out, sc_out = qp_equalize(best_seq, t_qp, n_peaks=15)
            try_update(a_out, "final_qp")

    # Phase 8: One more short anneal from best
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 30:
        t_final = min(remaining * 0.4, 100.0)
        if t_final > 15:
            print(f"  Final anneal ({t_final:.0f}s)...")
            a_out, sc_out = annealed_optimization(best_seq, t_final, verbose=False)
            try_update(a_out, "final_anneal")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
