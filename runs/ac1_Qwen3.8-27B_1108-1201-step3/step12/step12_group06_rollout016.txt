def fast_convolve(a):
    """Fast linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score assuming sum(a)=1: returns 2*n*max(b)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

def softmax_from_z(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Inverse of softmax (log)."""
    return np.log(np.maximum(a, 1e-300))


def build_conv_matrix(a_fixed):
    """Build the convolution matrix T where (T@b)[k] = (a_fixed*b)[k]."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

def solve_lp_best_response(a_fixed, solver=None):
    """Solve the LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS, cp.CBC, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

    ring_buffer = np.zeros((avg_window, n))
    ring_idx = 0
    ring_count = 0
    geom_weights = np.exp(-np.arange(avg_window) * 1.0 / (avg_window * 0.5))
    geom_weights /= geom_weights.sum()

    lp_values = []

    for it in range(max_iters):

            continue

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * (it + 1) * (it + 1))) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()

            a_new = b_new.copy()

            a_new = a_new / sa

        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        # Update ring buffer
        ring_buffer[ring_idx] = a
        ring_idx = (ring_idx + 1) % avg_window
        ring_count = min(ring_count + 1, avg_window)

        # Track scores
        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        if (it + 1) % 200 == 0 and ring_count >= 100:

def lse_objective_grad(z_in, T_val, n):
    """Compute LogSumExp objective and gradient w.r.t. z using FFT for efficiency."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb / T_val if T_val > 0 else b.max()
    obj = (2.0 * n / T_val) * log_sum_exp if T_val > 0 else 2.0 * n * b.max()

    # Compute g[j] = sum_m w[m] * a[m-j] using cross-correlation (FFT)
    # g[j] = sum_m w[m] * a[m-j] = (w * a_rev)[j + n - 1] where a_rev = reversed a
    # Actually: g[j] = sum_m w[m] * a[m-j], let l = m-j, so m = l+j
    # g[j] = sum_l w[l+j] * a[l] = cross_correlation(w, a)[j]
    # This is the same as (w flipped) convolved with a, or equivalently:
    # g = ifft(conj(FFT(w)) * FFT(a)) ... let's use direct for clarity with n~1200
    # For efficiency, use the fact that g[j] = sum_{m} w[m]*a[m-j]
    # This is (w * a)[j] in the "full" sense... actually it's cross-correlation.
    # Let's use: g[j] = sum_m w[m] * a[m-j] which is (w ⊛ a)[j] (cross-correlation)
    # = ifft(conj(fft(w, size)) * fft(a, size))[j]

    size_g = 2 * n - 1
    fft_size_g = 1 << (size_g - 1).bit_length()
    W = rfft(w, fft_size_g)
    A = rfft(a, fft_size_g)
    # Cross-correlation: g[j] = sum_m w[m+j]*a[m] = ifft(conj(W)*A) won't work directly
    # Let's use: g[j] = sum_m w[m]*a[m-j]
    # Define w_pad and a_pad, then g = ifft(fft(w_pad) * conj(fft(a_pad)))... 
    # Actually simplest: g[j] = sum_m w[m]*a[m-j] = (w * a_flipped)[j + n - 1]
    # where a_flipped[k] = a[n-1-k]
    a_flipped = a[::-1]
    g_full = irfft(rfft(w, fft_size_g) * rfft(a_flipped, fft_size_g), fft_size_g)[:size_g]
    # g_full[j + n - 1] = sum_m w[m] * a_flipped[m-(j+n-1)] = sum_m w[m] * a[n-1-(m-j-n+1)]
    # = sum_m w[m] * a[j-m+n-1-n+1] = sum_m w[m] * a[j-m]
    # That's g[j] = sum_m w[m]*a[j-m]... which is the cross-correlation (w⊛a)[j]
    # But we want g[j] = sum_m w[m]*a[m-j] = (a⊛w)[j]
    # (a⊛w)[j] = sum_m a[j-m]*w[m] = sum_m a[m]*w[j-m]... 
    # Let me just compute directly for correctness since n is moderate
    g = np.zeros(n)
    for j in range(n):
        # g[j] = sum_m w[m]*a[m-j] where 0<=m-j<n i.e. j<=m<n... and 0<=m<2n-1
        lo = max(0, j)
        hi = min(size_g - 1, n - 1 + j)
        if lo <= hi:
            g[j] = np.dot(w[lo:hi+1], a[lo-j:hi+1-j])

    S = np.dot(g, a)
    grad_z = (4.0 * n / T_val) * a * (g - S)
    return obj, grad_z


def lsbfgs_lse_lowT(a0, time_limit, T_start=5.0, T_min=0.5, n_stages=30, iters_per_stage=50):
    """Very low temperature LSE L-BFGS for final polishing."""

    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    T = T_start

    for stage in range(n_stages):

            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, T, n)
            return obj

            _, grad = lse_objective_grad(z_val, T, n)
            return grad

                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-16, 'maxfun': iters_per_stage * 500,
                         'disp': False}

            sc = score_normalized(a_check)

            if sc < best_score:

                best_a = a_check.copy()

            # Decrease temperature

            if (stage + 1) % 5 == 0:
                print(f"    LSE-lowT stage={stage+1}/{n_stages} T={T:.3f}: score={sc:.10f} best={best_score:.10f}")

        except Exception:
            T = max(T_min, T * 0.9)

    return best_a, best_score


def local_perturbation_search(a0, time_limit, n_trials=50000, max_perturb=0.1):

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    t3 = min(0.15 * (budget_s - 30), remaining * 0.5)
    if t3 > 20:
        a_cont, _ = nesterov_lp_averaged(best_seq, t3, solver=solver, max_iters=1000000, avg_window=5000)
        try_update(a_cont, "nesterov_cont")
    print(f"  Phase 3 done: best_score={best_score:.8f}")

    # === Phase 4: Multi-length exploration - 10% time ===
    print(f"\n=== Phase 4: Multi-length (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t4_total = min(0.10 * (budget_s - 30), remaining * 0.6)
    if t4_total > 30:
        lengths_to_try = [n_base + 300, n_base - 200, n_base + 600, n_base + 1000]
        for n_try in lengths_to_try:
            if n_try < 100:
                continue
            remaining = global_deadline - time.monotonic()
            if remaining < 25:
                break
            t4 = min(t4_total / len(lengths_to_try), remaining * 0.4)
            if t4 < 15:
                break
            print(f"  Trying n={n_try}, budget={t4:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            a_try, _ = nesterov_lp_short(init_try, t4 * 0.65, solver=solver, max_iters=100000)
            try_update(a_try, f"len_{n_try}_nesterov")

            remaining = global_deadline - time.monotonic()
            if remaining > 15:
                a_try2, _ = lsbfgs_lse_lowT(a_try, remaining * 0.5, T_start=5.0, T_min=0.5, n_stages=10, iters_per_stage=30)
                try_update(a_try2, f"len_{n_try}_lse")
    print(f"  Phase 4 done: best_score={best_score:.8f}")

    # === Phase 5: Local perturbation search - 8% time ===
    print(f"\n=== Phase 5: Local perturbation search (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t5 = min(0.08 * (budget_s - 30), remaining * 0.6)
    if t5 > 15:
        a_local, _ = local_perturbation_search(best_seq, t5, n_trials=200000, max_perturb=0.05)
        try_update(a_local, "local_pert")
    print(f"  Phase 5 done: best_score={best_score:.8f}")

    # === Phase 6: Final combined polish - 2% time ===
    print(f"\n=== Phase 6: Final polish (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    if remaining > 20:
        # Very short Nesterov
        a_f1, _ = nesterov_lp_short(best_seq, remaining * 0.3, solver=solver, max_iters=50000)
        try_update(a_f1, "final_nesterov")

        remaining = global_deadline - time.monotonic()
        if remaining > 15:
            # Ultra-low T LSE
            a_f2, _ = lsbfgs_lse_lowT(best_seq, remaining * 0.5, T_start=2.0, T_min=0.3, n_stages=10, iters_per_stage=30)
            try_update(a_f2, "final_lse_ultra")

        remaining = global_deadline - time.monotonic()
        if remaining > 10:
            # Final local search
            a_f3, _ = local_perturbation_search(best_seq, remaining * 0.6, n_trials=100000, max_perturb=0.02)
            try_update(a_f3, "final_local")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    elapsed_total = time.monotonic() - (global_deadline - (budget_s - 30))
    print(f"Time used: {elapsed_total:.0f}s")
    return [float(x) for x in best_seq]
