def get_solver():

    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)[:2n-1]."""
    n = len(a)
    # T[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # So T[k, j] is nonzero only when j <= k < j+n, i.e., k-j in [0, n-1]
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver):
    """Find b >= 0 maximizing sum(b) subject to conv(a,b) <= M."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_conv_matrix(a)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.full(n, norm / n)
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-14:
            break
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def subgradient_of_max_autocorr(a):
    """Compute a subgradient of max_k conv(a,a)_k w.r.t. a."""
    n = len(a)

    k_star = int(np.argmax(b))

    #          + 2 * sum over valid j of a[j] where i + j = k_star (i.e., j = k_star - i)
    # Actually conv(a,a)_k = sum_i a[i] * a[k-i] (with a[j]=0 for j<0 or j>=n)
    # d/d a[i] of conv(a,a)_k = a[k-i] + a[i-k] (when both are in [0,n))

    grad = np.zeros(n)

def fast_autocorr(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_conv_matrix(a)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def rescale_to_norm(a, norm):
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), norm / len(a))
    return a * norm / s


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.full(n, norm / n)
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-14:
            break
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = rescale_to_norm(y_new, norm)

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            a_new = rescale_to_norm(a_new, norm)

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            avg_check = rescale_to_norm(avg_check, norm)
            score_avg = fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lp_linsearch_optimize(a0, time_limit, solver, verbose=True):
    """LP best-response with golden section line search along LP direction."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = rescale_to_norm(a, norm)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-14:
            break
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Golden section line search along y -> g
        # f(t) = max_autocorr(rescale_to_norm((1-t)*y + t*g, norm))
        # Search in t in [0, 1]
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        a_lo = 0.0
        a_hi = 1.0
        c = a_hi - gr * (a_hi - a_lo)
        d = a_lo + gr * (a_hi - a_lo)

        def obj_at(t):
            blend = (1.0 - t) * y + t * g
            blend = np.maximum(blend, 0.0)
            blend = rescale_to_norm(blend, norm)
            return fast_autocorr(blend).max()

        fc = obj_at(c)
        fd = obj_at(d)

        for _ in range(20):
            if fc < fd:
                a_hi = d
                d = c
                fd = fc
                c = a_hi - gr * (a_hi - a_lo)
                fc = obj_at(c)
            else:
                a_lo = c
                c = d
                fc = fd
                d = a_lo + gr * (a_hi - a_lo)
                fd = obj_at(d)

        t_best = (a_lo + a_hi) / 2.0
        y_new = (1.0 - t_best) * y + t_best * g
        y_new = np.maximum(y_new, 0.0)
        y_new = rescale_to_norm(y_new, norm)
        y = y_new

        score = fast_autocorr(y).max()
        if score < best_score:
            best_score = score
            best_a = y.copy()

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPLS iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lse_gradient(a, temperature):
    """Compute gradient of T*log(sum_k exp(b_k/T)) where b = autocorr(a)."""
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    w = np.exp((b - max_b) / temperature)
    w /= w.sum()

    # grad[i] = 2 * sum_k w[k] * a[k-i] (with a[j]=0 outside [0,n))
    # = 2 * h1[i] where h1 is cross-correlation of w and a
    # Using FFT: h1[i] = ifft(FFT(w) * conj(FFT(a)))[i] (with zero-padding)
    N = 1 << (3 * n - 2).bit_length()
    Wf = rfft(w, N)
    Af = rfft(a, N)
    h1_full = irfft(Wf * np.conj(Af), N)[:n]
    grad = 2.0 * h1_full
    return grad, w


def lse_backtrack_optimize(a0, time_limit, T_start=10.0, T_end=0.1, n_stages=12, verbose=True):
    """LSE smoothed optimization with adaptive temperature and backtracking line search."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = rescale_to_norm(a, norm)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        # Initial step size estimate
        grad, _ = lse_gradient(a, T)
        norm_grad = np.linalg.norm(grad)
        if norm_grad < 1e-15:
            continue
        step = norm / (norm_grad * 10.0)
        step = min(step, norm / n)

        n_steps = max(10, int(remaining * 3))
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            grad, _ = lse_gradient(a, T)

            # Backtracking line search
            alpha = step
            a_trial = a - alpha * grad
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = rescale_to_norm(a_trial, norm)
            score = fast_autocorr(a_trial).max()
            curr_score = fast_autocorr(a).max()

            backtracks = 0
            while score > curr_score - 1e-15 * curr_score and backtracks < 15:
                alpha *= 0.5
                a_trial = a - alpha * grad
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = rescale_to_norm(a_trial, norm)
                score = fast_autocorr(a_trial).max()
                backtracks += 1

            a = a_trial
            step = alpha * 1.1  # slightly increase step for next iteration

            if score < best_score:
                best_score = score
                best_a = a.copy()

            if verbose and (it + 1) % 50 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LSE stage={si} T={T:.2f} iter={it+1} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            peak_mask[js2:je2 + 1] = True

        candidates = np.where(~peak_mask)[0]
        if len(candidates) < 3:
            candidates = np.where(np.arange(n) != top_src)[0]

        # Pick best destination: minimize increase in max autocorr
        best_dst = None
        best_trial_score = best_score

        # Try a few candidates
        sample_size = min(20, len(candidates))
        sample_idx = rng.choice(len(candidates), size=sample_size, replace=False)
        for si in sample_idx:
            dst = candidates[si]
            for frac in [0.001, 0.005, 0.01, 0.02]:
                amt = a[top_src] * frac
                if amt < 1e-18:
                    break
                a_trial = a.copy()
                a_trial[top_src] -= amt
                a_trial[dst] += amt
                if a_trial[top_src] < 0:
                    break
                a_trial = rescale_to_norm(a_trial, norm)
                score = fast_autocorr(a_trial).max()
                if score < best_trial_score:
                    best_trial_score = score
                    best_dst = (dst, amt)

        improved = False
        if best_dst is not None:
            dst, amt = best_dst
            a[top_src] -= amt
            a[dst] += amt
            a = rescale_to_norm(a, norm)
            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
                improved = True
                stall = 0
            else:
                stall += 1
        else:
            # Random exploration
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-10:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = rescale_to_norm(a_trial, norm)
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0

        if not improved:
            stall += 1
            if stall > 200:
                break

        if verbose and (stall + 1) % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD ({elapsed:.0f}s): best={best_score:.8f}, stall={stall}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if parent is None:
        n_base = 2000
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LP Nesterov at n=2500 (interpolated from parent)
    n1 = max(n_base, 2500)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.30, 500.0)
    print(f"\n=== Phase 1: LP Nesterov n={n1} (budget={t1:.0f}s) ===")
    if t1 > 30:
        init1 = interpolate_sequence(best_seq, n1)
        norm1 = np.sqrt(2.0 * n1)
        a_norm = init1 * norm1
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n1}")

    # Phase 2: LP Line Search at current best length
    n_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 2: LP LineSearch n={n_best} (budget={t2:.0f}s) ===")
    if t2 > 30:
        norm2 = np.sqrt(2.0 * n_best)
        a_norm = best_seq * norm2
        a_out, _ = lp_linsearch_optimize(a_norm, t2, solver, verbose=True)
        do_try_update(a_out, f"lpls_n{n_best}")

    # Phase 3: LP Nesterov at longer length n=3200
    n3 = max(n_best + 500, 3200)
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.20, 350.0)
    print(f"\n=== Phase 3: LP Nesterov n={n3} (budget={t3:.0f}s) ===")
    if t3 > 30:
        init3 = interpolate_sequence(best_seq, n3)
        norm3 = np.sqrt(2.0 * n3)
        a_norm = init3 * norm3
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n3}")

    # Phase 4: LSE gradient polish at best length
    n_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 4: LSE Gradient n={n_best} (budget={t4:.0f}s) ===")
    if t4 > 20:
        norm4 = np.sqrt(2.0 * n_best)
        a_norm = best_seq * norm4
        a_out, _ = lse_backtrack_optimize(a_norm, t4, T_start=8.0, T_end=0.05,
                                           n_stages=10, verbose=True)
        do_try_update(a_out, f"lse_n{n_best}")

    # Phase 5: LSE gradient at longer length (if best is still at shorter length)
    n5 = max(n_best, 2800)
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 5: LSE Gradient n={n5} (budget={t5:.0f}s) ===")
    if t5 > 20:
        init5 = interpolate_sequence(best_seq, n5)
        norm5 = np.sqrt(2.0 * n5)
        a_norm = init5 * norm5
        a_out, _ = lse_backtrack_optimize(a_norm, t5, T_start=5.0, T_end=0.03,
                                           n_stages=8, verbose=True)
        do_try_update(a_out, f"lse_n{n5}")

    # Phase 6: Coordinate descent polish
    n_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 6: CD Polish n={n_best} (budget={t6:.0f}s) ===")
    if t6 > 10:
        norm6 = np.sqrt(2.0 * n_best)
        a_norm = best_seq * norm6
        a_out, _ = coordinate_descent_polish(a_norm, t6, seed=123, verbose=True)
        do_try_update(a_out, "cd_polish")

    # Phase 7: Final LP Nesterov at best length (if time remains)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t7 = min(remaining * 0.25, 300.0)
    print(f"\n=== Phase 7: Final LP Nesterov n={n_best} (budget={t7:.0f}s) ===")
    if t7 > 30:
        norm7 = np.sqrt(2.0 * n_best)
        a_norm = best_seq * norm7
        a_out, _ = nesterov_lp_optimize(a_norm, t7, solver, verbose=True)
        do_try_update(a_out, f"lp_final_n{n_best}")

    # Phase 8: Final LSE polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t8 = min(remaining * 0.30, 150.0)
    print(f"\n=== Phase 8: Final LSE Polish n={n_best} (budget={t8:.0f}s) ===")
    if t8 > 10:
        norm8 = np.sqrt(2.0 * n_best)
        a_norm = best_seq * norm8
        a_out, _ = lse_backtrack_optimize(a_norm, t8, T_start=3.0, T_end=0.02,
                                           n_stages=8, verbose=True)
        do_try_update(a_out, "lse_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence! Using fallback.")
        return [1.0]

    return [float(x) for x in best_seq]
