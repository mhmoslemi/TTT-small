import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_autocorr(a):
    """Full autocorrelation a*a using FFT. Returns array of length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_score(a):
    """Compute 2n*max(b)/(sum(a))^2 quickly."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)


def cross_corr_w_b(w, a):
    """
    Compute the gradient contribution: for each lag k with weight w[k],
    add w[k] * a_shifted to the gradient.
    result[j] = sum_k w[k] * a[k-j] (when 0<=k-j<n)
    """
    n = len(a)
    a_rev = a[::-1].copy()
    conv_len = n + len(w) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a_rev, fft_size)
    conv = irfft(W * A, fft_size)
    return conv[n - 1:n - 1 + n]


def lse_obj_grad_z(z, T):
    """
    Objective and gradient in z-space where a = softmax(z) / sum(softmax(z)).
    Objective: (1/T)*log(sum(exp(T*b_k)))
    """
    n = len(z)
    z_shift = z - z.max()
    e = np.exp(z_shift)
    a = e / e.sum()

    b = fast_autocorr(a)
    Tb = T * b
    Tb_shift = Tb - Tb.max()
    exp_Tb = np.exp(Tb_shift)
    obj = (Tb.max() + np.log(exp_Tb.sum())) / T

    w = exp_Tb / exp_Tb.sum()

    # grad_a = 2 * cross_corr(w, a)
    cc = cross_corr_w_b(w, a)
    grad_a = 2.0 * cc

    # grad_z = a * (grad_a - dot(grad_a, a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    return obj, grad_z


def spectrum_flatten_init(n, n_iter=200, seed=42):
    """
    Generate an initialization with relatively flat autocorrelation using
    iterative time/frequency domain projection (Gerchberg-Saxton variant).
    """
    rng = np.random.default_rng(seed)
    N = 1 << (2 * n - 1).bit_length()  # FFT size >= 2n-1
    
    # Start with random non-negative sequence
    a = rng.random(n)
    a = a / a.sum()
    
    # Target: make |A|^2 as flat as possible (excluding DC)
    for it in range(n_iter):
        # Go to frequency domain
        A = rfft(a, N)
        mag = np.abs(A)
        
        # Target magnitude: geometric mean of current magnitudes
        # (preserves the total "energy" in the spectrum)
        log_mag = np.log(mag + 1e-30)
        target_log_mag = np.full_like(log_mag, np.mean(log_mag[1:]) if len(log_mag) > 1 else log_mag[0])
        target_log_mag[0] = log_mag[0]  # Keep DC fixed (since sum(a) is fixed)
        target_mag = np.exp(target_log_mag)
        
        # Rescale: new_A = target_mag * A / |A|
        phase = A / (mag + 1e-30)
        A_new = target_mag * phase
        
        # Back to time domain
        a_new = irfft(A_new, N)[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = rng.random(n)
            s = a_new.sum()
        a_new = a_new / s
        a = a_new
    
    return a


def mirror_descent(a0, time_limit, eta0=0.05, decay_rate=0.65,
                   k_min=1, k_max=15, k_ramp=300, seed=42, verbose=False):
    """

    Uses O(n log n) per iteration via FFT.
    """

    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall_count = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        b_max = b.max()
        b_mean = b.sum() / len(b)
        b_var = np.var(b)

        # Determine current k with ramp
        frac = min(1.0, t / float(k_ramp))
        k = max(k_min, min(k_max, int(k_min + frac * (k_max - k_min))))

        # Top-k indices
        top_idx = np.argsort(b)[::-1][:k]

        # Variance-based weights: weight proportional to (b - mean)^2
        excess = b[top_idx] - b_mean
        excess_sq = excess ** 2
        if excess_sq.max() < 1e-30:
            weights = np.ones(k) / k
        else:
            # Use squared excess for sharper focus on outliers
            weights = excess_sq / (excess_sq.sum() + 1e-30)

        # Compute subgradient using FFT
        grad = np.zeros(len(b))
        grad[top_idx] = weights
        cc = cross_corr_w_b(grad, a)
        grad_a = 2.0 * cc

        # Learning rate with power decay
        eta = eta0 / (1.0 + t / 300.0) ** decay_rate

        # Multiplicative update
        a_new = a * np.exp(-eta * grad_a)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a + 0.02 * rng.standard_normal(n)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            a_new = a_new / max(s_new, 1e-15)
        else:
            a_new = a_new / s_new

        a = a_new

        # Check score periodically
        if t < 500:
            check_interval = 3
        elif t < 5000:
            check_interval = 10
        else:
            check_interval = 30

        if t % check_interval == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stall_count = 0
            else:
                stall_count += 1

        # Stall handling
        if stall_count > 400:
            noise_scale = min(0.05, 0.005 * (stall_count / 100.0))
            a = a + noise_scale * rng.standard_normal(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            stall_count = 0

    return best_a, best_score


def lse_optimize(a0, time_limit, T_start=50.0, T_end=0.3, n_stages=20, seed=42):
    """
    LogSumExp-smoothed optimization with L-BFGS and fine-grained continuation.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s

    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = fast_score(a_init)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        # More iterations at lower T (finer optimization)
        T_ratio = stage_T / T_start
        iters = max(20, min(150, int(40 + 110 * (1.0 - T_ratio))))

        def func(z_in):
            val, _ = lse_obj_grad_z(z_in, stage_T)
            return val

        def jac(z_in):
            _, g = lse_obj_grad_z(z_in, stage_T)
            return g

        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': iters, 'ftol': 1e-18,
                                    'gtol': 1e-12, 'maxfun': iters * 300,
                                    'disp': False})
            z = res.x
            z_shift = z - z.max()
            a = np.exp(z_shift)
            a = a / a.sum()
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        except Exception:
            pass

        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {si+1}/{n_stages} (T={stage_T:.2f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def multi_move_shaving(a0, time_limit, seed=123, verbose=False):
    """
    Multi-move peak shaving: simultaneously move mass from multiple sources
    to multiple destinations to reduce the top peaks.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        num_peaks = min(8, (2*n - 1) // 4)
        top_k = np.argsort(b)[::-1][:num_peaks]

        improved = False
        for k_idx in top_k[:5]:
            k_idx = int(k_idx)
            js = max(0, k_idx - n + 1)
            je = min(n - 1, k_idx)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_idx - idx]

            if contrib.sum() < 1e-15:
                continue

            # Find top-3 contributors
            src_indices = idx[np.argsort(contrib)[::-1][:3]]
            
            # Find destinations: positions far from peak areas
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k[:5]:
                k2 = int(k2)
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) < 3:
                candidates = np.where(~peak_mask | (np.arange(n) != src_indices[0]))[0]
            sorted_candidates = candidates[np.argsort(a[candidates])[:max(5, len(candidates) // 5)]]
            
            if len(sorted_candidates) == 0:
                continue
            
            dst = int(sorted_candidates[rng.integers(len(sorted_candidates))])

            for frac in [0.001, 0.003, 0.008, 0.02, 0.05]:
                a_trial = a.copy()
                for src in src_indices:
                    src = int(src)
                    amt = a_trial[src] * frac
                    if amt < 1e-15 or a_trial[src] < amt:
                        continue
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                if (a_trial < -1e-15).any():
                    continue
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = fast_score(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial.copy()
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break

            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation
            if rng.integers(2) == 0:
                nmoves = rng.integers(1, 4)
                a_trial = a.copy()
                for _ in range(nmoves):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a_trial[i] > 1e-10:
                        frac = rng.uniform(0.002, 0.03)
                        amt = a_trial[i] * frac
                        a_trial[i] -= amt
                        a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = fast_score(a_trial)
                if sc < best_score:
                    a = a_trial.copy()
                    best_score = sc
                    best_a = a.copy()
                    stall = 0

            if stall > 500:
                break

    return best_a, best_score


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def diverse_init(n, rng, variant=None):
    """Generate diverse structured initializations."""
    if variant is None:
        variant = int(rng.integers(0, 8))

    if variant == 0:
        x = np.linspace(0, 1, n)
        a = np.maximum(1 - 2 * abs(x - 0.5), 0)
    elif variant == 1:
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif variant == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-3 * x ** 2)
    elif variant == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-20 * (x - 0.25) ** 2) + np.exp(-20 * (x - 0.75) ** 2)
    elif variant == 4:
        x = np.linspace(0, 1, n)
        a = x ** 2 + (1 - x) ** 2 + 0.1
    elif variant == 5:
        x = np.linspace(0, 1, n)
        a = 1 + 0.5 * np.cos(4 * np.pi * x)
    elif variant == 6:
        x = np.linspace(-1, 1, n)
        a = np.sinc(x) ** 2
    else:
        x = np.linspace(0, 1, n)
        a = abs(x - 0.5) + 0.2

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def cross_corr_w_b(w, a):
    """
    Compute grad[j] = sum_k w[k] * a[k-j] for j = 0,...,n-1
    where w has length 2n-1 and a has length n.
    """
    n = len(a)
    a_rev = a[::-1].copy()
    conv_len = n + len(w) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a_rev, fft_size)
    conv = irfft(W * A, fft_size)
    return conv[n - 1:n - 1 + n]
