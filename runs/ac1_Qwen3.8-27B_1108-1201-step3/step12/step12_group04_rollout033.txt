import numpy as np

from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Full self-convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_fn(a):
    """Compute the objective value."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)

def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:

def normalize(a):
    """Normalize to sum=1, ensuring non-negativity."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix where T @ b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed
    return T

def get_lp_solver():

    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])

            if prob.status in ('optimal', 'optimal_inaccurate'):

def solve_lp_best_response(a_fixed, solver):
    """Solve: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.CLARABEL, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def nesterov_lp(a0, time_limit, solver, max_iters=5000):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    a = normalize(a0)
    y = a.copy()
    
    best_a = a.copy()
    best_score = score_fn(a)
    t0 = time.monotonic()
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * b_star
        y_new = normalize(y_new)
        
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = normalize(a_new)
        
        y = y_new
        a = a_new
        
        s_a = score_fn(a)
        if s_a < best_score:
            best_score = s_a
            best_a = a.copy()
        s_y = score_fn(y)
        if s_y < best_score:
            best_score = s_y
            best_a = y.copy()
        
        if t % 50 == 0:
            print(f"    NLP t={t}: best={best_score:.8f}")
    
    return best_a, best_score

def softmax_fn(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def lse_grad(a, T):
    """Gradient of LogSumExp of autocorrelation w.r.t. a."""
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    
    # grad[i] = 2 * sum_k w[k] * a[k-i] (circular correlation)
    # Using FFT: conv(w, a_rev)[n-1:n-1+n]
    a_rev = a[::-1]
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    Ar = rfft(a_rev, fft_size)
    conv = irfft(W * Ar, fft_size)
    
    # We need the cross-correlation: (w * a)[i] = sum_k w[k] * a[k-i]
    # This is conv(w, a_rev) evaluated at position i + n - 1
    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad

def logsumexp_optimize(a0, time_limit, T_start=50.0, T_end=0.01, n_stages=15):
    """LogSumExp-smoothed optimization with L-BFGS and annealing."""
    n = len(a0)
    a_init = normalize(a0)
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = score_fn(a_init)
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        def func(z_in):
            a = softmax_fn(z_in)
            b = fast_autocorr(a)
            Tb = stage_T * b
            return (Tb.max() + np.log(np.exp(Tb - Tb.max()).sum())) / stage_T
        
        def jac(z_in):
            a = softmax_fn(z_in)
            g = lse_grad(a, stage_T)
            S = np.dot(g, a)
            return a * (g - S)
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': 50, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': 5000, 'disp': False}
            )
            z = res.x
            a_new = softmax_fn(z)
            s_new = score_fn(a_new)
            if s_new < best_score:
                best_score = s_new
                best_a = a_new.copy()
        except Exception:
            pass
        
        if si % 5 == 0:
            print(f"    LSE stage={si} T={stage_T:.4f}: best={best_score:.8f}")
    
    return best_a, best_score

def coordinate_descent(a0, time_limit, seed=42):
    """Greedy coordinate descent / pairwise swap optimization."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize(a0)
    best_a = a.copy()
    best_score = score_fn(a)
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        if stall > 500:
            break
        
        b = fast_autocorr(a)
        k_max = np.argmax(b)
        
        # Find the most "productive" move
        # Try moving mass from a high-contribution position to a low-impact one
        # Compute sensitivity: how much does each a[i] contribute to b[k_max]?
        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_max - idx]
        
        # Also check sensitivity to other near-max peaks
        top_peaks = np.argsort(b)[::-1][:5]
        
        best_move_score = best_score
        best_move = None
        
        # Try top few source positions
        top_src = np.argsort(contrib)[::-1][:5]
        
        for ts in top_src:
            src = idx[ts]
            src_val = a[src]
            if src_val < 1e-10:
                continue
            
            # Find target: position that minimizes increase in max peak
            # Approximate: position with low values that doesn't contribute to top peaks
            target_scores = np.zeros(n)
            for kp in top_peaks:
                js2 = max(0, kp - n + 1)
                je2 = min(n - 1, kp)
                for j2 in range(js2, je2 + 1):
                    partner = kp - j2
                    if 0 <= partner < n:
                        target_scores[j2] += a[j2] * a[partner]
            
            # Add current value as penalty
            target_scores += 0.001 * a
            target_scores[src] = np.inf
            
            # Try a few target positions
            bottom_targets = np.argsort(target_scores)[:10]
            
            for dst in bottom_targets:
                if a[dst] < 0:  # shouldn't happen
                    continue
                for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
                    amt = src_val * frac
                    if amt < 1e-14:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = normalize(a_trial)
                    s_trial = score_fn(a_trial)
                    if s_trial < best_move_score:
                        best_move_score = s_trial
                        best_move = (src, dst, amt)
        
        if best_move is not None:
            src, dst, amt = best_move
            a[src] -= amt
            a[dst] += amt
            a = np.maximum(a, 0)
            a = normalize(a)
            best_score = best_move_score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            # Random perturbation to escape
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-8:
                amt = a[i] * rng.uniform(0.001, 0.02)
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                a_trial = normalize(a_trial)
                s_trial = score_fn(a_trial)
                if s_trial < best_score:
                    a = a_trial
                    best_score = s_trial
                    best_a = a.copy()
                    stall = 0
    
    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast Nesterov-accelerated LP best-response (workhorse, CLARABEL/ECOS)
2. LogSumExp gradient optimization with multi-stage annealing (FFT-based)
3. Smart coordinate descent / pairwise mass transfer for final polish
4. Multi-start with diverse spectral initializations
5. Averaging of Nesterov iterates for improved convergence

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_autocorr(a):
    """Full self-convolution a*a using FFT, O(n log n)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_fn(a):
    """Compute the objective 2*n*max_conv/(sum)^2."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def normalize(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where T @ b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def get_lp_solver():
    """Find a working LP solver, prefer fast ones."""
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            t0 = time.monotonic()
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                el = time.monotonic() - t0
                print(f"  LP solver {solver_name}: test time={el:.3f}s")
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """
    Solve LP: min M s.t. conv(a_fixed, b) <= M for all lags, sum(b)=1, b>=0.
    Returns (b_opt, M_val) or (None, None) on failure.
    """
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    try:
        prob.solve(solver=solver)
    except Exception:
        prob = None

    if prob is None or prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob = cp.Problem(cp.Minimize(M_var), constraints)
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue

    if prob is None or prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = float(M_var.value)
    return b_val, M_val


def nesterov_lp(a0, time_limit, solver, max_iters=100000, use_averaging=True):
    """
    Nesterov-accelerated LP best-response iteration for the symmetric game.
    Tracks best of: current iterate, momentum iterate, and running average.
    """
    n = len(a0)
    a = normalize(a0)
    y = a.copy()

    best_a = a.copy()
    best_score = score_fn(a)
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha_t) * y + alpha_t * b_star
        y_new = normalize(y_new)

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = normalize(a_new)

        y = y_new
        a = a_new

        # Track scores
        s_a = score_fn(a)
        if s_a < best_score:
            best_score = s_a
            best_a = a.copy()
        s_y = score_fn(y)
        if s_y < best_score:
            best_score = s_y
            best_a = y.copy()

        # Running average
        if use_averaging:
            avg_a += a
            n_avg += 1
            if n_avg >= 50 and n_avg % 100 == 0:
                avg_check = normalize(avg_a / n_avg)
                s_avg = score_fn(avg_check)
                if s_avg < best_score:
                    best_score = s_avg
                    best_a = avg_check.copy()

        if t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M_val:.10f}")

    return best_a, best_score


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def lse_grad_a(a, T_param):
    """
    Compute gradient of LogSumExp_T(a*a) w.r.t. a using FFT.
    Returns (grad, weights) where grad[i] = 2 * sum_k w[k] * a[k-i].
    """
    n = len(a)
    b = fast_autocorr(a)
    Tb = T_param * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()

    # Cross-correlation of w (len 2n-1) with a (len n)
    a_rev = a[::-1]
    conv_len = (2 * n - 1) + n - 1  # = 3n - 2
    fft_size = 1 << (conv_len - 1).bit_length()

    W = rfft(w, fft_size)
    Ar = rfft(a_rev, fft_size)
    conv = irfft(W * Ar, fft_size)

    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad, w


def logsumexp_optimize(a0, time_limit, T_start=50.0, T_end=0.005, n_stages=15):

    LogSumExp-smoothed optimization with L-BFGS and multi-stage temperature annealing.
    parametrized via a = softmax(z) for automatic positivity and normalization.
    """
    n = len(a0)
    a_init = normalize(a0)
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = score_fn(a_init)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func(z_in):
            a = softmax_fn(z_in)
            b = fast_autocorr(a)
            Tb = stage_T * b
            return (Tb.max() + np.log(np.exp(Tb - Tb.max()).sum())) / stage_T

        def jac(z_in):
            a = softmax_fn(z_in)
            g, _ = lse_grad_a(a, stage_T)
            S = np.dot(g, a)
            return a * (g - S)

        max_iter = max(20, min(80, int(20 + 60 * si / max(1, n_stages - 1))))
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18,
                         'gtol': 1e-15, 'maxfun': max_iter * 200,
                         'disp': False}
            )
            z = res.x
            a_new = softmax_fn(z)
            s_new = score_fn(a_new)
            if s_new < best_score:
                best_score = s_new
                best_a = a_new.copy()
        except Exception:
            pass

        if si % 5 == 0 or si == n_stages - 1:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage={si}/{n_stages-1} T={stage_T:.4f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, seed=42):
    """
    Greedy coordinate descent with smart mass transfer.
    Targets the top peaks of the autocorrelation and redistributes mass
    from high-contributing positions to low-impact positions.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize(a0)
    best_a = a.copy()
    best_score = score_fn(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        if stall > 800:
            break

        b = fast_autocorr(a)
        top_peaks = np.argsort(b)[::-1][:5]

        improved = False
        for k in top_peaks[:3]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            top_contrib = np.argsort(contrib)[::-1][:4]

            for tc in top_contrib:
                src = idx[tc]
                src_val = a[src]
                if src_val < 1e-10:
                    continue

                # Score target positions by total contribution to all top peaks
                target_penalty = np.zeros(n)
                for kp in top_peaks:
                    js2 = max(0, kp - n + 1)
                    je2 = min(n - 1, kp)
                    for j2 in range(js2, je2 + 1):
                        partner = kp - j2
                        if 0 <= partner < n:
                            target_penalty[j2] += a[j2] * a[partner]
                target_penalty[src] = np.inf
                target_penalty += 0.001 * a

                bottom_targets = np.argsort(target_penalty)[:8]

                for dst in bottom_targets:
                    if dst == src:
                        continue
                    for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                        amt = src_val * frac
                        if amt < 1e-14:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0)
                        ss = a_trial.sum()
                        if ss < 1e-15:
                            continue
                        a_trial = a_trial / ss
                        s_trial = score_fn(a_trial)
                        if s_trial < best_score - 1e-14:
                            a = a_trial
                            best_score = s_trial
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if improved:
            stall = 0
        else:
            stall += 1
            # Random perturbation to escape local optimum
            for _ in range(3):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.002, 0.03)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    s_trial = score_fn(a_trial)
                    if s_trial < best_score:
                        a = a_trial
                        best_score = s_trial
                        best_a = a.copy()
                        stall = 0
                        break

        if stall % 200 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            print(f"    CD stall={stall} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def spectral_init(n, rng):
    """Generate diverse initializations inspired by spectral design."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, 1, n)
        freq = rng.uniform(2, 15)
        a = (1.0 - np.abs(2.0 * x - 1.0)) * (0.5 + 0.5 * np.cos(2 * np.pi * freq * x))
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3) * (0.6 + 0.4 * np.cos(2 * np.pi * rng.uniform(3, 10) * x))
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2 * (0.7 + 0.3 * np.cos(2 * np.pi * rng.uniform(5, 20) * x))
    elif choice == 3:
        x = np.linspace(0, 1, n)
        n_bumps = rng.integers(2, 6)
        a = np.zeros(n)
        for _ in range(n_bumps):
            c = rng.uniform(0.1, 0.9)
            w = rng.uniform(0.05, 0.3)
            a += np.exp(-((x - c) ** 2) / (2 * w ** 2))
    elif choice == 4:
        # IFFT of random positive power spectrum
        k = n // 2 + 1
        power = rng.dirichlet(np.ones(k) * rng.uniform(0.3, 1.0))
        mag = np.sqrt(power) + 0.1
        phases = rng.uniform(0, 2 * np.pi, k)
        spectrum = mag * np.exp(1j * phases)
        a = irfft(spectrum, n)
        a = np.maximum(a, 0)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.maximum(0, 1 - 2 * abs(x - 0.5)) + 0.3 * np.sin(8 * np.pi * x)
    elif choice == 6:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(3 * x) + 0.15 * np.cos(7 * x) + 0.1 * np.cos(13 * x)
    else:
        # Smooth random via interpolation
        n_ctrl = rng.integers(5, 25)
        ctrl = rng.dirichlet(np.ones(n_ctrl) * 2)
        x_old = np.linspace(0, 1, n_ctrl)
        x_new = np.linspace(0, 1, n)
        a = np.interp(x_new, x_old, ctrl)

    return normalize(a)


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    f = normalize(f)
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f)
    return normalize(f_new)


def propose_candidate():
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Strategy: Nesterov LP (main), LogSumExp gradients (smoothing),
    coordinate descent (polish), with multi-length exploration.
    """
    t_start = time.monotonic()
    time_budget = 1740.0  # 60s safety margin

    solver = get_lp_solver()

    best_a = None
    best_score = float('inf')
    n_base = 1600

    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_a = normalize(p)
            n_base = len(best_a)
            best_score = score_fn(best_a)
            print(f"Parent loaded: n={n_base}, score={best_score:.8f}")
    except Exception:
        print("No valid parent found, using default initialization")

    if best_a is None:
        rng = np.random.default_rng(42)
        best_a = spectral_init(n_base, rng)
        best_score = score_fn(best_a)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    def try_update(a_cand, label):
        nonlocal best_a, best_score
        if a_cand is None:
            return
        a_cand = np.maximum(np.asarray(a_cand, dtype=np.float64), 0)
        s = a_cand.sum()
        if s < 1e-15:
            return
        a_cand = a_cand / s
        sc = score_fn(a_cand)
        if sc < best_score:
            best_score = sc
            best_a = a_cand.copy()
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a_cand)}, t={elapsed:.0f}s)")

    # === Phase 1: Main Nesterov LP ===
    remaining = time_budget - (time.monotonic() - t_start)
    t1 = min(remaining * 0.40, 600.0)
    print(f"\n=== Phase 1: Nesterov LP n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = nesterov_lp(best_a, t1, solver)
        try_update(a_out, "nlp1")

    # === Phase 2: LogSumExp refinement ===
    remaining = time_budget - (time.monotonic() - t_start)
    t2 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 2: LogSumExp n={len(best_a)} (budget={t2:.0f}s) ===")
    if t2 > 15:
        a_out, _ = logsumexp_optimize(best_a, t2, T_start=50.0, T_end=0.01, n_stages=15)
        try_update(a_out, "lse1")

    # === Phase 3: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_start)
    t3 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 3: Coord Descent n={len(best_a)} (budget={t3:.0f}s) ===")
    if t3 > 10:
        a_out, _ = coordinate_descent(best_a, t3, seed=123)
        try_update(a_out, "cd1")

    # === Phase 4: Second Nesterov LP pass (exploit refined point) ===
    remaining = time_budget - (time.monotonic() - t_start)
    t4 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 4: NLP refinement n={len(best_a)} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = nesterov_lp(best_a, t4, solver)
        try_update(a_out, "nlp2")

    # === Phase 5: Try nearby lengths ===
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 5: Multi-length exploration (remaining={remaining:.0f}s) ===")
    n_best = len(best_a)
    candidate_lengths = sorted(set([n_best - 200, n_best + 200, n_best + 400]),
                               key=lambda x: abs(x - n_best))
    candidate_lengths = [nl for nl in candidate_lengths if 800 <= nl <= 4000]

    time_per_len = max(20, min(remaining * 0.5 / max(1, len(candidate_lengths)), 150.0))
    for nl in candidate_lengths:
        remaining = time_budget - (time.monotonic() - t_start)
        if remaining < 30:
            break
        tl = min(time_per_len, remaining * 0.6)
        if tl < 20:
            continue
        print(f"  Trying n={nl} (time={tl:.0f}s)")
        init_nl = interpolate_seq(best_a, nl)
        # Quick Nesterov LP
        t_lp = tl * 0.7
        a_out, _ = nesterov_lp(init_nl, t_lp, solver)
        try_update(a_out, f"nlp_n{nl}")
        # Quick LSE
        remaining_sub = time_budget - (time.monotonic() - t_start)
        if remaining_sub > 10:
            t_lse = max(5, min(tl * 0.25, remaining_sub * 0.2))
            if t_lse > 5:
                a_out2, _ = logsumexp_optimize(a_out, t_lse, T_start=30.0, T_end=0.02, n_stages=8)
                try_update(a_out2, f"lse_n{nl}")

    # === Phase 6: Final polish on best ===
    remaining = time_budget - (time.monotonic() - t_start)
    n_final = len(best_a)
    print(f"\n=== Phase 6: Final polish n={n_final} (remaining={remaining:.0f}s) ===")

    t6a = min(remaining * 0.3, 120.0)
    if t6a > 5:
        a_out, _ = coordinate_descent(best_a, t6a, seed=999)
        try_update(a_out, "cd_final")

    remaining = time_budget - (time.monotonic() - t_start)
    t6b = min(remaining * 0.25, 100.0)
    if t6b > 5:
        a_out, _ = logsumexp_optimize(best_a, t6b, T_start=15.0, T_end=0.005, n_stages=10)
        try_update(a_out, "lse_final")

    remaining = time_budget - (time.monotonic() - t_start)
    t6c = min(remaining * 0.3, 80.0)
    if t6c > 10:
        a_out, _ = nesterov_lp(best_a, t6c, solver)
        try_update(a_out, "nlp_final")

    remaining = time_budget - (time.monotonic() - t_start)
    t6d = min(remaining * 0.3, 60.0)
    if t6d > 5:
        a_out, _ = coordinate_descent(best_a, t6d, seed=7777)
        try_update(a_out, "cd_final2")

    # Final verification
    if best_a is not None:
        fs = evaluate_safe(best_a)
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_a)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    # Ensure valid output
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    return result
