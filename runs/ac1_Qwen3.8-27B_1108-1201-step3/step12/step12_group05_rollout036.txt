def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(arr):

def softmax_fn(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n <= 1:
        return np.ones(n) / max(n, 1)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1)
    cond = (u - (cssv - 1.0) / rho_arr) > 0
    rho = int(np.sum(cond))
    if rho == 0:
        return np.ones(n) / n
    theta = (cssv[rho - 1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)

def compute_subgrad(a, k):
    """Compute subgradient of conv(a,a)_k w.r.t. a. O(n)."""
    n = len(a)
    g = np.zeros(n)
    if k == 0:
        g[:] = 2.0 * a
    elif k > 0:
        g[:n-k] += a[k:]       # a_{i+k}
        g[k:] += a[:n-k]       # a_{i-k}
    else:
        k_abs = -k
        g[k_abs:] += a[:n-k_abs]   # a_{i+k} = a_{i-k_abs}
        g[:n-k_abs] += a[k_abs:]   # a_{i-k} = a_{i+k_abs}
    return g

def ftrl_optimize(a0, time_limit, eta_init=0.5, eta_decay=0.9999):
    """FTRL with multiplicative weights. Fast O(n) per iteration."""
    n = len(a0)
    a = a0.copy()
    z = np.zeros(n)
    eta = eta_init
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        k_star = int(b.argmax())
        g = compute_subgrad(a, k_star)
        
        # FTRL update with adaptive eta
        z += eta * g
        # Rescale z to prevent underflow/overflow
        z -= z.mean()
        
        a = np.exp(-z)
        s = a.sum()
        if s < 1e-300:
            a = np.ones(n) / n
        else:
            a = a / s
        
        iter_count += 1
        if iter_count % 100 == 0:
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
            eta *= eta_decay
        
    score = 2.0 * n * fast_autocorr(a).max()
    if score < best_score:
        best_score = score
        best_a = a.copy()
    
    return best_a, best_score

def lse_optimize(a0, time_limit, T_start=100.0, T_end=0.02, n_stages=20, lr=0.01):
    """LSE-smoothed gradient descent with Adam."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_adam = 0
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    conv_len = 2 * n - 1
    conv_fft_size = 1 << (conv_len - 1).bit_length()
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        iters = max(10, min(200, int(remaining * 100 / (n_stages - si))))
        
        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break
            
            a = softmax_fn(z)
            A = rfft(a, fft_size)
            b = irfft(A * A, fft_size)[:size]
            
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            
            A_rev = rfft(a[::-1].copy(), conv_fft_size)
            W = rfft(w, conv_fft_size)
            conv = irfft(A_rev * W, conv_fft_size)
            grad_a = 2.0 * conv[n-1:n-1+n]
            
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)
            
            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)
            lr_t = lr / np.sqrt(1 + 0.003 * it)
            z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            
            if it % 30 == 0:
                a = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
        
        a = softmax_fn(z)
        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score

def get_lp_solver():

def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

def solve_lp_reduced(a_fixed, M_upper, active_indices, solver):
    """Solve LP with only active constraints for speed."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_full = build_conv_matrix(a_fixed)
    constraints = [T_full[active_indices] @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

    avg_a = np.zeros(n)

        active = np.argsort(b_y)[::-1][:n_constraints]

            if b_star is None:
                continue
        
        alpha_t = 2.0 / (t + 2.0)

        else:

            a_new = y_new + momentum * (y_new - y)

            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}, LP_M={M_val:.8f}")

def peak_shaving_multi(a0, time_limit, seed=123):

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:K]
        
        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            src = idx[np.argmax(contrib)]

            peak_indices = set()

def spectral_shaping(a, n_iter=500, alpha=0.3):
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    
    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.05, 20.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    
    return best_a

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FTRL (Follow-The-Regularized-Leader) multiplicative weights - fast O(n) per iter
2. LP-based Nesterov with adaptive constraint reduction and CLARABEL solver
3. LSE-smoothed Adam gradient descent with adaptive temperature
4. Aggressive multi-peak coordinate descent shaving
5. Spectral shaping for initialization
6. Multi-length exploration with interpolation
Key innovations: FTRL provides direct subgradient optimization without FFT,
adaptive LP constraint reduction speeds up the main solver, and multi-peak
shaving simultaneously attacks several autocorrelation peaks.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full autocorrelation a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(arr):
    """Safely evaluate a sequence via the global evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n <= 1:
        return np.ones(max(n, 1)) / max(n, 1)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = (u - (cssv - 1.0) / rho_arr) > 0
    rho = int(np.sum(cond))
    if rho == 0:
        return np.ones(n) / n
    theta = (cssv[rho - 1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)


def compute_subgrad(a, k):
    """
    Compute subgradient of conv(a,a)_k w.r.t. a in O(n).
    conv(a,a)_k = sum_j a_j * a_{j+k} (with a_i=0 outside [0,n-1])
    d/d a_i = a_{i+k} + a_{i-k} (with 0 for out-of-bounds)
    """
    n = len(a)
    g = np.zeros(n)
    if k == 0:
        g[:] = 2.0 * a
    elif k > 0:
        g[:n - k] += a[k:]
        g[k:] += a[:n - k]
    else:
        k_abs = -k
        g[k_abs:] += a[:n - k_abs]
        g[:n - k_abs] += a[k_abs:]
    return g


def ftrl_optimize(a0, time_limit, eta_init=1.0, eta_decay=0.9999):
    """
    FTRL with multiplicative weights. Each iteration is O(n) for subgradient
    plus O(n) FFT for score check. Much faster than LP per iteration.
    
    Strategy: accumulate subgradients, update a = exp(-z) normalized.
    Use adaptive eta that decays slowly.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    z = np.zeros(n)
    eta = eta_init
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    iter_count = 0
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    while time.monotonic() - t0 < time_limit:
        # Find peak
        A = rfft(a, fft_size)
        b = irfft(A * A, fft_size)[:size]
        k_star = int(b.argmax())

        # Compute subgradient in O(n)
        g = compute_subgrad(a, k_star)

        # FTRL update
        z += eta * g
        # Periodic rescaling to prevent overflow
        if iter_count % 1000 == 0:
            z -= z.mean()

        a = np.exp(-z)
        s = a.sum()
        if s < 1e-300:
            z -= z.max()
            a = np.exp(-z)
            s = a.sum()
            if s < 1e-300:
                a = np.ones(n) / n
            else:
                a = a / s
        else:
            a = a / s

        iter_count += 1
        if iter_count % 50 == 0:
            score = 2.0 * n * b.max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
            eta *= eta_decay

        if iter_count % 500 == 0:
            # Full score check
            A = rfft(a, fft_size)
            b = irfft(A * A, fft_size)[:size]
            score = 2.0 * n * b.max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

    # Final check
    A = rfft(a, fft_size)
    b = irfft(A * A, fft_size)[:size]
    score = 2.0 * n * b.max()
    if score < best_score:
        best_score = score
        best_a = a.copy()

    return best_a, best_score


def lse_optimize(a0, time_limit, T_start=100.0, T_end=0.02, n_stages=20, lr=0.01):
    """LSE-smoothed gradient descent with Adam optimizer and geomspace annealing."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_adam = 0
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    conv_fft_size = 1 << (2 * size - 1).bit_length()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        iters = max(10, min(200, int(remaining * 100 / (n_stages - si))))

        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break

            a = softmax_fn(z)
            A = rfft(a, fft_size)
            b = irfft(A * A, fft_size)[:size]

            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()

            A_rev = rfft(a[::-1].copy(), conv_fft_size)
            W = rfft(w, conv_fft_size)
            conv = irfft(A_rev * W, conv_fft_size)
            grad_a = 2.0 * conv[n - 1:n - 1 + n]

            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)

            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)
            lr_t = lr / np.sqrt(1 + 0.003 * it)
            z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)

            if it % 30 == 0:
                a = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()

        a = softmax_fn(z)
        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_a, best_score


def get_lp_solver():
    """Find the fastest working LP solver."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a_fixed):
    """Build convolution matrix T where (T @ b)_k = conv(a_fixed, b)_k."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_reduced(a_fixed, active_indices, solver):
    """Solve LP: min M s.t. conv(a_fixed, b)_k <= M for active k, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_full = build_conv_matrix(a_fixed)
    constraints = [T_full[active_indices] @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, max_iters=50000, n_constraints=None):
    """
    Nesterov-accelerated LP best-response with adaptive constraint set.
    Only solves LP with top n_constraints constraints for speed.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    if n_constraints is None:
        n_constraints = min(2 * n - 1, max(300, int(1.5 * n)))

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        # Find active constraints (top peaks)
        A_y = rfft(y, fft_size)
        b_y = irfft(A_y * A_y, fft_size)[:size]
        active = np.argsort(b_y)[::-1][:n_constraints]

        b_star, M_val = solve_lp_reduced(y, active, solver)
        if b_star is None:
            continue

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s

        y = y_new
        a = a_new

        # Score check
        A_a = rfft(a, fft_size)
        b_a = irfft(A_a * A_a, fft_size)[:size]
        score_a = 2.0 * n * b_a.max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        A_y = rfft(y, fft_size)
        b_y = irfft(A_y * A_y, fft_size)[:size]
        score_y = 2.0 * n * b_y.max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}, LP_M={M_val:.8f}")

    return best_a, best_score


def peak_shaving_multi(a0, time_limit, seed=123):
    """
    Aggressive multi-peak coordinate descent shaving.
    Simultaneously targets top K autocorrelation peaks.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    K = 5
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    while time.monotonic() - t0 < time_limit:
        A = rfft(a, fft_size)
        b = irfft(A * A, fft_size)[:size]
        top_k = np.argsort(b)[::-1][:K]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            src = idx[np.argmax(contrib)]

            # Collect peak-involved indices
            peak_indices = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_indices.update(range(js2, je2 + 1))

            # Find low-value candidates outside peak areas
            candidates = [j for j in range(n) if j not in peak_indices and a[j] < np.median(a)]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            candidates_sorted = sorted(candidates, key=lambda j: a[j])
            dst_pool = candidates_sorted[:max(1, len(candidates_sorted) // 4)]

            for _ in range(3):
                dst = dst_pool[rng.integers(len(dst_pool))]
                for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    A_t = rfft(a_trial, fft_size)
                    b_t = irfft(A_t * A_t, fft_size)[:size]
                    score = 2.0 * n * b_t.max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    A_t = rfft(a_trial, fft_size)
                    b_t = irfft(A_t * A_t, fft_size)[:size]
                    score = 2.0 * n * b_t.max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > 500:
                # Bigger perturbation
                mask = rng.random(n) < 0.1
                a_trial = a.copy()
                a_trial[mask] *= rng.uniform(0.5, 1.5, int(mask.sum()))
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss > 1e-15:
                    a_trial /= ss
                    A_t = rfft(a_trial, fft_size)
                    b_t = irfft(A_t * A_t, fft_size)[:size]
                    score = 2.0 * n * b_t.max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > 2000:
                break

    return best_a, best_score


def spectral_shaping(a, n_iter=500, alpha=0.3):
    """Iteratively flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.05, 20.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 1.5
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.6 + 0.3 * np.cos(3 * x) + 0.1 * np.cos(7 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x ** 2)
        a = np.maximum(a, 0)
    else:
        x = np.linspace(0, 1, n)
        tri = 1 - 2 * np.abs(x - 0.5)
        a = tri * (1 + 0.2 * np.cos(6 * np.pi * x))
        a = np.maximum(a, 0)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization combining:
    - Spectral shaping for fast initialization
    - FTRL multiplicative weights (O(n) per iter, no FFT for gradient)
    - LP Nesterov with adaptive constraint reduction (CLARABEL)
    - LSE-smoothed Adam for fine-tuning
    - Multi-peak coordinate descent shaving
    - Multi-length exploration
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Spectral shaping (fast initialization)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: Spectral shaping (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.05, 40.0)
    if t1 > 5:
        a_out = spectral_shaping(best_seq, n_iter=1000, alpha=0.3)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [spectral]: {best_score:.6f}")

    # Phase 2: FCLR - fast FTRL (O(n) per iteration)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: FTRL (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.20, 280.0)
    if t2 > 10:
        a_out, _ = ftrl_optimize(best_seq, t2, eta_init=1.5, eta_decay=0.99995)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [FTRL]: {best_score:.6f}")

    # Phase 3: LP Nesterov (main powerful optimizer)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: LP Nesterov (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.35, 500.0)
    if t3 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t3, solver)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [LP]: {best_score:.6f}")

    # Phase 4: LSE fine-tuning
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: LSE (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.18, 250.0)
    if t4 > 15:
        a_out, _ = lse_optimize(best_seq, t4, T_start=50.0, T_end=0.01, n_stages=25, lr=0.005)
        sc = evaluate_safe(a_out)

            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [LSE]: {best_score:.6f}")

    # Phase 5: Peak shaving
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: Peak shaving (remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.20, 250.0)
    if t5 > 10:
        a_out, _ = peak_shaving_multi(best_seq, t5, seed=456)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [peak_shave]: {best_score:.6f}")

    # Phase 6: Longer length exploration
    n_long = int(round(n_base * 1.3))
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: Longer n={n_long} (remaining={remaining:.0f}s) ===")
    t6 = min(remaining * 0.40, 500.0)

        t6a = t6 * 0.3
        a_out, _ = ftrl_optimize(a_spec, t6a, eta_init=1.0, eta_decay=0.99995)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [FTRL_n{n_long}]: {best_score:.6f}")

        # LP at longer length
        t6b = t6 * 0.5
        a_out, _ = nesterov_lp_optimize(best_seq, t6b, solver)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [LP_n{n_long}]: {best_score:.6f}")

        # Peak shave at longer length
        t6c = t6 * 0.2
        a_out, _ = peak_shaving_multi(best_seq, t6c, seed=789)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [peak_n{n_long}]: {best_score:.6f}")

    # Phase 7: Final LSE polish at low temperature
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 7: Final LSE polish (remaining={remaining:.0f}s) ===")
    t7 = min(remaining * 0.30, 150.0)
    if t7 > 10:
        a_out, _ = lse_optimize(best_seq, t7, T_start=5.0, T_end=0.005, n_stages=15, lr=0.002)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [final_LSE]: {best_score:.6f}")

    # Phase 8: Final peak shave
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 10:
        t8 = min(remaining * 0.60, 100.0)
        print(f"\n=== Phase 8: Final peak shave (remaining={remaining:.0f}s) ===")
        if t8 > 5:
            a_out, _ = peak_shaving_multi(best_seq, t8, seed=1011)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:

                best_seq = np.maximum(a_out, 0)
                s = best_seq.sum()
                if s > 0:
                    best_seq = best_seq / s
                print(f"  *** NEW BEST [final_peak]: {best_score:.6f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
            best_seq = np.maximum(best_seq, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            # Re-verify after normalization
            fs2 = evaluate_safe(best_seq)
            if np.isfinite(fs2) and fs2 < best_score:
                best_score = fs2
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
