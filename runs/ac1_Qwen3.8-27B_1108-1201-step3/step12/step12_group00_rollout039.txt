def build_conv_matrix(a):
    """Build the matrix T such that T @ b = conv(a, b)."""
    n = len(a)

    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response_fast(a, M, solver):

        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_conv(a).max()

    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_conv(y).max()
        if M_curr < 1e-12:
            break
        b = lp_best_response_fast(y, M_curr, solver)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_conv(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_conv(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def peak_shave(a, time_limit, seed=42):
    """Target the top autocorrelation peaks by redistributing mass."""
    rng = np.random.default_rng(seed)
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_conv(a)
        top_k = np.argsort(b)[::-1][:8]

        improved = False
        for k in top_k:
            # Find positions contributing most to this peak
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            # Find a destination with low contribution to all top peaks
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) < 10:
                candidates = np.where(np.arange(n) != src)[0]

            # Pick from low-value candidates
            cand_sorted = candidates[np.argsort(a[candidates])]
            n_try = max(3, len(cand_sorted) // 10)
            dst = cand_sorted[rng.integers(min(n_try, len(cand_sorted)))]

            for frac in [0.001, 0.003, 0.01, 0.02, 0.05, 0.1]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = normalize(a_trial, norm)
                score = fast_conv(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-12:
                frac = rng.uniform(0.02, 0.1)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = normalize(a_trial, norm)
                score = fast_conv(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                stall = 0

            if stall > 500:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def symmetric_init(n, rng):
    """Create a symmetric initial sequence with various shapes."""
    choice = rng.integers(0, 6)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        # Raised cosine
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif choice == 1:
        # Inverted parabola (dip in center)
        a = 1.0 - 0.5 * np.sin(np.pi * x) ** 2
    elif choice == 2:
        # Flat top with smooth edges
        a = np.minimum(1.0, 2.0 * np.minimum(x, 1 - x) * 3.0)
    elif choice == 3:
        # Gaussian-like
        a = np.exp(-8.0 * (x - 0.5) ** 2) + 0.5
    elif choice == 4:
        # Uniform
        a = np.ones(n)
    else:
        # Edge-enhanced
        a = 1.0 + 0.3 * (2 * x - 1) ** 2
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function. Returns the best sequence found."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if parent is None:
        n_base = 1600
        best_seq = symmetric_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # === Phase 1: Quick LP Nesterov to get a good local optimum (20% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} ===")
    t1 = min(remaining * 0.20, 300.0)
    if t1 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n_base}")

    # === Phase 2: Exponentiated gradient (main workhorse, 45% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Exponentiated Gradient n={len(best_seq)} ===")
    t2 = min(remaining * 0.45, 700.0)
    if t2 > 10:
        n_best = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = exponentiated_gradient_nesterov(
            a_norm, t2, T_start=3.0, T_end=0.005,
            eta_init=1.0 / np.sqrt(n_best),
            verbose=True, lp_solver=solver, lp_every=10000, seed=123
        )
        do_try_update(a_out, f"eg_n{n_best}")

        # Second EG run with different params
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            t2b = min(remaining * 0.15, 150.0)
            if t2b > 10:
                n_best = len(best_seq)
                a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
                a_out, _ = exponentiated_gradient_nesterov(
                    a_norm, t2b, T_start=1.0, T_end=0.001,
                    eta_init=2.0 / np.sqrt(n_best),
                    verbose=False, lp_solver=None, seed=456
                )
                do_try_update(a_out, f"eg2_n{n_best}")

    # === Phase 3: Try longer sequence (20% time) ===
    n2 = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Exponentiated Gradient n={n2} ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 20:
        init_long = interpolate_sequence(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = exponentiated_gradient_nesterov(
            a_norm, t3, T_start=3.0, T_end=0.01,
            eta_init=1.0 / np.sqrt(n2),
            verbose=True, lp_solver=solver, lp_every=20000, seed=789
        )
        do_try_update(a_out, f"eg_n{n2}")

    # === Phase 4: Even longer (15% time) ===
    n3 = n_base + 800
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Exponentiated Gradient n={n3} ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        init_long3 = interpolate_sequence(best_seq, n3)
        a_norm = init_long3 * np.sqrt(2.0 * n3) / init_long3.sum()
        a_out, _ = exponentiated_gradient_nesterov(
            a_norm, t4, T_start=5.0, T_end=0.02,
            eta_init=1.0 / np.sqrt(n3),
            verbose=True, lp_solver=None, seed=101,
        )
        do_try_update(a_out, f"eg_n{n3}")

    # === Phase 5: Peak shaving on best (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Peak Shaving n={n_best} ===")
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 5:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_shave(a_norm, t5, seed=555)
        do_try_update(a_out, "peak_shave")

    # === Phase 6: Final EG polish on best (5% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final EG Polish n={n_best} ===")
    t6 = min(remaining * 0.05, 60.0)
    if t6 > 5:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = exponentiated_gradient_nesterov(
            a_norm, t6, T_start=0.5, T_end=0.001,
            eta_init=1.0 / np.sqrt(n_best),
            verbose=False, lp_solver=None, seed=202
        )
        do_try_update(a_out, "eg_polish")

    # === Phase 7: Final peak shave ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        n_best = len(best_seq)
        t7 = min(remaining * 0.5, 80.0)
        if t7 > 5:
            print(f"\n=== Phase 7: Final Peak Shave n={n_best} ===")
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = peak_shave(a_norm, t7, seed=999)
            do_try_update(a_out, "final_shave")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
