import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:

            prob.solve(solver=s)

                _global_solver = s
                print(f"  LP solver: {s}")

        except Exception:

    _global_solver = cp.ECOS
    print(f"  LP solver (fallback): ECOS")

def fast_autocorr(a):
    """Compute linear autocorrelation via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_gradient_descent(p0, time_limit, beta_start=5.0, beta_end=200.0,
                         n_anneal=80000, verbose=True, seed=42):
    """
    LSE-smoothed gradient descent with Nesterov momentum on the simplex.
    Minimizes (1/beta)*log(sum_k exp(beta * conv(p,p)_k)) as approximation
    to min max_k conv(p,p)_k subject to p in simplex.
    """
    n = len(p0)
    p = np.maximum(np.asarray(p0, dtype=np.float64), 0.0).copy()
    s = p.sum()
    if s < 1e-15:
        p = np.ones(n) / n
    else:
        p = p / s

    best_p = p.copy()
    best_max = fast_autocorr(p).max()

    fft_size = 1 << (2 * n - 1 - 1).bit_length()

    t0 = time.monotonic()
    y = p.copy()  # Nesterov lookahead
    v = np.zeros(n)  # velocity
    total_iters = 500000

    for t in range(1, total_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        # Anneal beta
        frac = min(t / n_anneal, 1.0)
        beta = beta_start * (beta_end / beta_start) ** frac

        # Compute autocorrelation
        A = rfft(y, fft_size)
        C = irfft(A * A, fft_size)[:2 * n - 1]

        # LSE weights
        max_C = C.max()
        if max_C < 1e-30:
            break
        w = np.exp(beta * C - beta * max_C)
        Z = w.sum()
        if Z < 1e-30:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(C))] = 1.0
        else:
            w = w / Z

        # Gradient: grad_m = 2 * sum_j w_{m+j} * p_j for valid m+j
        # This is a cross-correlation of w with p
        # Use FFT: grad = 2 * ifft(fft(w) * conj(fft(p_reflected)))
        # Actually, grad_m = 2 * sum_{j=0}^{n-1} w_{m+j} * p_j (if m+j < 2n-1)
        # This is (w * p)_m where * denotes cross-correlation
        # = ifft(fft(w) * conj(fft(p))) evaluated at position m

        P_ft = rfft(p, fft_size)
        W_ft = rfft(w, fft_size)
        # Cross-correlation: ifft(conj(fft(p)) * fft(w)) gives sum_j p_j * w_{j+m}
        # We want sum_j p_j * w_{m+j} which is the same
        grad_raw = irfft(W_ft * np.conj(P_ft), fft_size)[:n]
        grad = 2.0 * grad_raw

        # Nesterov step
        lr = 1.0 / (beta * (1.0 + t * 0.001))
        v = 0.9 * v - lr * grad
        y_new = y + v
        y_new = project_simplex(y_new)

        # Check actual max
        if t % 50 == 0 or t == 1:
            A2 = rfft(y_new, fft_size)
            C2 = irfft(A2 * A2, fft_size)[:2 * n - 1]
            actual_max = C2.max()
            if actual_max < best_max:
                best_max = actual_max
                best_p = y_new.copy()

        y = y_new
        p = y_new.copy()

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE t={t} ({elapsed:.0f}s): beta={beta:.1f}, best_max={best_max:.8f}, obj={2*n*best_max:.6f}")

    return best_p, best_max


def lp_best_response(a, M, solver, fft_size=None):
    """Solve LP: max sum(b) s.t. conv(a,b)_k <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)

    # Build constraint matrix T where T[k, j] = a[k-j] if valid
    # conv(a, b)_k = sum_{j} a[k-j] * b_j
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        # b_j contributes to conv at lags j, j+1, ..., j+n-1
        # conv_k = a_{k-j} * b_j, valid when 0 <= k-j <= n-1, i.e., j <= k <= j+n-1
        lo = j
        hi = j + n
        T[lo:hi, j] = a[:n]

    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_lp(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    p = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = p.sum()
    if s < 1e-15:
        p = np.ones(n) / n
    else:
        p = p / s

    best_p = p.copy()
    best_max = fast_autocorr(p).max()

    y = p.copy()
    t0 = time.monotonic()
    n_iters = 0

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue

        b = b / b.sum()
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * b
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y

        if t == 1:
            p_new = b.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.85)
            p_new = y_new + momentum * (y_new - y)
            p_new = np.maximum(p_new, 0.0)
            s_p = p_new.sum()
            if s_p < 1e-15:
                p_new = y_new.copy()
            else:
                p_new = p_new / s_p

        y = y_new
        p = p_new

        score = fast_autocorr(p).max()
        if score < best_max:
            best_max = score
            best_p = p.copy()

        # Also check y
        score_y = fast_autocorr(y).max()
        if score_y < best_max:
            best_max = score_y
            best_p = y.copy()

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP t={n_iters} ({elapsed:.0f}s): best={best_max:.8f}, obj={2*n*best_max:.6f}")

    return best_p, best_max


def coordinate_descent(p0, time_limit, seed=42):
    """Targeted coordinate descent: move mass to reduce peak autocorrelation."""
    n = len(p0)
    rng = np.random.default_rng(seed)
    p = np.maximum(np.asarray(p0, dtype=np.float64), 0.0).copy()
    s = p.sum()
    if s < 1e-15:
        p = np.ones(n) / n
    else:
        p = p / s

    best_p = p.copy()
    best_max = fast_autocorr(p).max()
    t0 = time.monotonic()
    stall = 0

    fft_size = 1 << (2 * n - 1 - 1).bit_length()

    while time.monotonic() - t0 < time_limit:
        C = fast_autocorr(p)
        k_star = int(np.argmax(C))
        M = C[k_star]

        # Find the pair (i, j) that most reduces C_{k_star}
        # C_{k_star} = sum_i p_i * p_{k_star - i}
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        contrib = p[idx] * p[k_star - idx]

        # Source: position with highest contribution
        src_local = np.argmax(contrib)
        src = idx[src_local]

        # Destination: position that doesn't contribute to top peaks
        top_k = np.argsort(C)[::-1][:5]
        peak_mask = np.zeros(n, dtype=bool)
        for k2 in top_k:
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            peak_mask[js2:je2 + 1] = True

        candidates = np.where(~peak_mask)[0]
        if len(candidates) == 0:
            candidates = np.where(np.arange(n) != src)[0]

        # Pick candidate with lowest p value (least damage to other peaks)
        dst = candidates[np.argmin(p[candidates])]

        # Try different fractions
        improved = False
        for frac in [0.01, 0.02, 0.05, 0.1, 0.2]:
            amt = p[src] * frac
            if amt < 1e-15:
                continue
            p_trial = p.copy()
            p_trial[src] -= amt
            p_trial[dst] += amt
            if p_trial[src] < 0:
                continue
            p_trial = p_trial / p_trial.sum()
            score = fast_autocorr(p_trial).max()
            if score < best_max - 1e-12:
                p = p_trial
                best_max = score
                best_p = p.copy()
                improved = True
                stall = 0
                break

        if not improved:
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and p[i] > 1e-10:
                frac = rng.uniform(0.02, 0.1)
                amt = p[i] * frac
                p_trial = p.copy()
                p_trial[i] -= amt
                p_trial[j] += amt
                p_trial = np.maximum(p_trial, 0)
                s_t = p_trial.sum()
                if s_t > 1e-15:
                    p_trial = p_trial / s_t
                    score = fast_autocorr(p_trial).max()
                    if score < best_max:
                        p = p_trial
                        best_max = score
                        best_p = p.copy()
                        stall = 0

            stall += 1
            if stall > 200:
                break

    return best_p, best_max


def structured_init(n, rng):
    """Generate a structured initial sequence."""
    choice = rng.integers(0, 4)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        # Cosine window with oscillation
        a = (0.5 + 0.5 * np.cos(2 * np.pi * x)) * (1 + 0.3 * np.cos(4 * np.pi * x))
    elif choice == 1:
        # Double Gaussian
        a = np.exp(-8 * (x - 0.3) ** 2) + np.exp(-8 * (x - 0.7) ** 2) + 0.1
    elif choice == 2:
        # Sawtooth-like
        a = 0.5 + 0.4 * np.sin(6 * np.pi * x) + 0.1 * np.sin(12 * np.pi * x)
    else:
        # Exponential decay with oscillation
        a = np.exp(-2 * x) * (1 + 0.5 * np.cos(8 * np.pi * x))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
