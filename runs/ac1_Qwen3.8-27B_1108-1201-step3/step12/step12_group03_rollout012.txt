def fast_autocorr(a):
    """Full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def get_lp_solver():

            prob.solve(solver=solver_name)

def build_conv_matrix(a_fixed):
    """Build the (2n-1) x n convolution matrix T where (T@b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0. Returns (b, M)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def freq_domain_flatten(a0, time_limit, beta_start=5.0, beta_end=50.0, n_iters=None):
    """
    Frequency-domain power spectrum flattening.
    Iteratively modifies the power spectrum to be flatter, then projects back to non-negative.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    if n_iters is None:
        n_iters = 500

    # Zero-pad to at least 2n-1
    pad_size = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive beta
        progress = it / max(1, n_iters)
        beta = beta_start * (beta_end / beta_start) ** progress

        # Compute DFT of zero-padded sequence
        A = fft(a, pad_size)
        power = np.abs(A) ** 2

        # Target: flatten the power spectrum
        # Use a soft-max approach: increase low-power bins, decrease high-power bins
        power_shifted = power - power.mean()
        adjustment = np.tanh(beta * power_shifted / (power.std() + 1e-15))
        
        # Reduce power where it's high, increase where it's low
        # But maintain total energy
        target_power = power * (1.0 - 0.1 * adjustment)
        target_power = np.maximum(target_power, 1e-10)
        
        # Interpolate between current and target
        mix = 0.3
        new_power = (1 - mix) * power + mix * target_power
        
        # Reconstruct: new A has same phase, new magnitude
        new_magnitude = np.sqrt(new_power)
        phase = np.angle(A)
        new_A = new_magnitude * np.exp(1j * phase)
        
        # Inverse DFT
        new_a = np.real(ifft(new_A, pad_size))[:n]
        new_a = np.maximum(new_a, 0)
        new_s = new_a.sum()
        if new_s < 1e-15:
            continue
        new_a = new_a / new_s

        score = 2.0 * n * fast_autocorr(new_a).max()
        if score < best_score:
            best_score = score
            best_a = new_a.copy()
            a = new_a

    return best_a, best_score


def smooth_gradient_descent(a0, time_limit, n_iters=None):
    """
    Smooth gradient descent on exp(beta * c_k) approximation of max(c_k).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    if n_iters is None:
        n_iters = 3000

    lr = 0.5
    beta = 10.0

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        max_b = b.max()

        # Soft approximation: sum exp(beta * b_k / max_b)
        # Gradient wrt a[j]: sum_k (beta/max_b) * exp(beta*b_k/max_b) * db_k/da[j]
        weights = np.exp(beta * b / max_b)
        weights = weights / weights.sum()  # Normalize

        # Compute gradient
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            w = weights[k]
            if w < 1e-10:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            other = k - idx
            grad[idx] += 2.0 * w * a[other]
            grad[other] += 2.0 * w * a[idx]

        # Gradient step with projection to simplex
        lr_t = lr / (1.0 + 0.01 * it)
        a_new = a - lr_t * grad
        a_new = np.maximum(a_new, 0)
        a_new_s = a_new.sum()
        if a_new_s < 1e-15:
            break
        a_new = a_new / a_new_s

        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        # Adapt beta
        if it % 500 == 0:
            beta = min(beta * 1.5, 100.0)

    return best_a, best_score


def qp_refinement(a0, time_limit, n_active=30):

    max_b = b.max()
    t0 = time.monotonic()

    for round_num in range(30):
        if time.monotonic() - t0 > time_limit:
            break

        # Identify active lags (top peaks)

        if len(active_lags) > n_active:

            except Exception:

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            try:
                prob.solve(solver=cp.CLARABEL)
            except Exception:
                break
            if prob.status not in ('optimal', 'optimal_inaccurate'):
                break

        a_new = np.asarray(a_var.value, dtype=np.float64)
        a_new = np.maximum(a_new, 0)
        a_new_s = a_new.sum()
        if a_new_s < 1e-15:
            break
        a_new = a_new / a_new_s

        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score - 1e-10:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            b = fast_autocorr(a)
            max_b = b.max()
        else:
            # Try mixing
            improved = False
            for frac in [0.2, 0.4, 0.6, 0.8]:
                a_mix = (1 - frac) * a + frac * a_new
                a_mix = np.maximum(a_mix, 0)
                a_mix_s = a_mix.sum()
                if a_mix_s < 1e-15:
                    continue
                a_mix = a_mix / a_mix_s
                score_mix = 2.0 * n * fast_autocorr(a_mix).max()
                if score_mix < best_score:
                    best_score = score_mix
                    best_a = a_mix.copy()
                    a = a_mix
                    b = fast_autocorr(a)
                    max_b = b.max()
                    improved = True
                    break
            if not improved and round_num >= 5:
                break
            # Expand active set
            n_active = min(n_active + 5, 60)

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=123):
    """
    Peak shaving with contribution analysis and multi-fraction line search.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        peak_idx = np.argmax(b)
        
        # Focus on the top peaks
        n_peaks = min(5, 2 * n - 1)
        top_peaks = np.argsort(b)[::-1][:n_peaks]

        improved = False
        for k in top_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            # Find top contributors
            top_contrib = np.argsort(contrib)[::-1][:3]
            for ci in top_contrib:
                src = idx[ci]
                src_other = k - src

                # Try moving mass to a random position
                for _ in range(5):
                    dst = rng.integers(0, n)
                    if dst == src and dst == src_other:
                        continue
                    for frac in [0.005, 0.01, 0.02, 0.04]:
                        amt = a[src] * frac
                        if src != src_other:
                            amt = min(amt, a[src_other] * frac)
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        if src != src_other:
                            a_trial[src_other] -= amt * 0.5
                        a_trial[dst] += amt
                        if (a_trial < -1e-12).any():
                            continue
                        a_trial = np.maximum(a_trial, 0)
                        a_trial_s = a_trial.sum()
                        if a_trial_s < 1e-15:
                            continue
                        a_trial = a_trial / a_trial_s
                        score = 2.0 * n * fast_autocorr(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation
            if stall % 5 == 0:
                for _ in range(3):
                    src = rng.integers(0, n)
                    dst = rng.integers(0, n)
                    if src == dst:
                        continue
                    frac = rng.uniform(0.005, 0.03)
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if (a_trial < -1e-12).any():
                        continue
                    a_trial = np.maximum(a_trial, 0)
                    a_trial_s = a_trial.sum()
                    if a_trial_s < 1e-15:
                        continue
                    a_trial = a_trial / a_trial_s
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                        break
            if stall > max_stall:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_cosine(n, power=2.0):
    """Cosine-based initialization: cos(x/2)^power on [0, pi]."""
    x = np.linspace(0, np.pi, n)
    a = np.cos(x / 2.0) ** power
    a = np.maximum(a, 0)
    s = a.sum()
    return a / s


def init_exponential(n, decay=0.5):
    """Exponential decay initialization."""
    a = np.exp(-decay * np.arange(n) / n)
    s = a.sum()
    return a / s


def init_bimodal(n, width=0.3):
    """Bimodal: mass at both ends."""
    x = np.arange(n) / n
    a = np.exp(-((x - 0.1) / width) ** 2) + np.exp(-((x - 0.9) / width) ** 2)
    s = a.sum()
    return a / s


def init_skewed(n, skew=0.3):
    """Skewed cosine: shifted to create asymmetry."""
    x = np.linspace(0, np.pi, n)
    a = np.cos((x - skew * np.pi) / 2.0) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function: multi-phase optimization at multiple lengths.
    Uses LP Nesterov, frequency domain flattening, QP, and peak shaving.
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1680.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    # Get parent sequence
    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            n_parent = len(parent)
            p_score = evaluate_safe(parent)
            print(f"Parent: n={n_parent}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        n_parent = 1600

    if best_seq is None:
        n_parent = 1600
        best_seq = init_cosine(n_parent)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_parent}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            a = a * 1000.0 / max(s, 1e-15)
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Multi-start initialization ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: Multi-start init (remaining={remaining:.0f}s) ===")
    
    # Try different initializations at the parent length
    n_base = len(best_seq)
    inits = {
        'cos2': init_cosine(n_base, 2.0),
        'cos3': init_cosine(n_base, 3.0),
        'exp': init_exponential(n_base, 1.0),
        'skew': init_skewed(n_base, 0.2),
        'bimodal': init_bimodal(n_base, 0.25),
    }
    
    for name, a_init in inits.items():
        sc = evaluate_safe(a_init)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_init.copy()
            print(f"  Init {name}: {sc:.6f}")
    
    # Quick LP Nesterov on the best init (short time)
    remaining = time_budget - (time.monotonic() - t_global_start)
    t_quick = min(remaining * 0.05, 60.0)
    if t_quick > 20:
        a_out, sc_out, _ = nesterov_lp_optimize(best_seq, t_quick, solver, verbose=False)
        try_update(a_out, "quick_lp")

    # === Phase 2: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: Multi-length exploration (remaining={remaining:.0f}s) ===")
    
    # Try a few lengths around the parent
    lengths_to_try = []
    for delta in [-200, -100, 0, 100, 200, 400]:
        n_try = n_base + delta
        if 400 <= n_try <= 4000:
            lengths_to_try.append(n_try)
    
    for n_try in lengths_to_try[:4]:
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining < 100:
            break
        print(f"  Trying n={n_try}...")
        t_alloc = min(remaining * 0.15, 180.0)
        if t_alloc < 30:
            continue
        
        # Interpolate from best
        a_init = interpolate_sequence(best_seq, n_try)
        a_out, sc_out, _ = nesterov_lp_optimize(a_init, t_alloc, solver, verbose=False)
        try_update(a_out, f"lp_n{n_try}")
        
        # Quick freq domain
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining > 60:
            a_fd, sc_fd = freq_domain_flatten(a_out, min(remaining * 0.1, 40.0), n_iters=100)
            try_update(a_fd, f"freq_n{n_try}")

    # === Phase 3: Main LP Nesterov at best length ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    n_main = len(best_seq)
    print(f"\n=== Phase 3: Main LP Nesterov at n={n_main} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.40, 600.0)
    if t3 > 60:
        a_out, sc_out, iters = nesterov_lp_optimize(best_seq, t3, solver, verbose=True)
        print(f"  LP done: {iters} iters, score={sc_out:.6f}")
        try_update(a_out, "main_lp")

    # === Phase 4: Frequency domain refinement ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: Frequency domain (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 150.0)
    if t4 > 20:
        a_out, sc_out = freq_domain_flatten(best_seq, t4, beta_start=3.0, beta_end=30.0, n_iters=300)
        try_update(a_out, "freq_main")
        
        # Also try smooth gradient descent
        remaining = time_budget - (time.monotonic() - t_global_start)
        t4b = min(remaining * 0.10, 100.0)
        if t4b > 15:
            a_sg, sc_sg = smooth_gradient_descent(best_seq, t4b, n_iters=2000)
            try_update(a_sg, "smooth_grad")

    # === Phase 5: QP refinement ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: QP refinement (remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 150.0)
    if t5 > 20:
        a_out, sc_out = qp_refinement(best_seq, t5, n_active=25)
        try_update(a_out, "qp_main")

    # === Phase 6: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: Peak shaving (remaining={remaining:.0f}s) ===")
    t6 = min(remaining * 0.20, 200.0)
    if t6 > 15:
        a_out, sc_out = peak_shaving(best_seq, t6, seed=456)
        try_update(a_out, "peak1")
        
        # Second round with different seed
        remaining = time_budget - (time.monotonic() - t_global_start)
        t6b = min(remaining * 0.10, 80.0)
        if t6b > 10:
            a_out, sc_out = peak_shaving(best_seq, t6b, seed=789)
            try_update(a_out, "peak2")

    # === Phase 7: Final LP Nesterov round ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 7: Final LP round (remaining={remaining:.0f}s) ===")
    t7 = min(remaining * 0.20, 150.0)
    if t7 > 30:
        a_out, sc_out, _ = nesterov_lp_optimize(best_seq, t7, solver, verbose=False)
        try_update(a_out, "final_lp")

    # === Phase 8: Final QP + peak shave ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 8: Final QP + Peak (remaining={remaining:.0f}s) ===")
    if remaining > 30:
        t8a = min(remaining * 0.3, 45.0)
        if t8a > 10:
            a_out, sc_out = qp_refinement(best_seq, t8a, n_active=15)
            try_update(a_out, "final_qp")
    
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 15:
        t8b = min(remaining * 0.4, 40.0)
        if t8b > 5:
            a_out, sc_out = peak_shaving(best_seq, t8b, seed=999)
            try_update(a_out, "final_peak")

    # Final verification
    if best_seq is not None:
        # Ensure valid for evaluator
        best_seq = np.clip(best_seq, 0.0, 1000.0)
        s = best_seq.sum()
        if s < 0.01:
            best_seq = best_seq * 1.0 / max(s, 1e-15)
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        # Fallback: cosine at 1600
        return [float(x) for x in init_cosine(1600)]

    return [float(x) for x in best_seq]
