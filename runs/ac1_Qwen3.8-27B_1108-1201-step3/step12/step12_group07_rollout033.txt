def toeplitz_matvec(first_row, x):
    """Multiply a symmetric Toeplitz matrix (defined by first_row) with vector x using FFT."""

    fft_size = 1 << (2 * n - 1).bit_length()
    R = rfft(first_row, fft_size)
    X = rfft(x, fft_size)
    return irfft(R * X, fft_size)[:n]


def solve_qp_toeplitz(first_row, a0, max_sweeps=200):
    """
    Solve min a^T Q a s.t. a >= 0, sum(a) = 1
    where Q is symmetric Toeplitz with given first row.
    Uses active set method with CG inner iterations.
    Returns the solution or None if failed.
    """
    n = len(first_row)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    # Active set: start with all free
    active = np.ones(n, dtype=bool)

    for sweep in range(max_sweeps):
        # Solve equality-constrained QP on active set:
        # min a_F^T Q_FF a_F s.t. 1^T a_F = sum_target
        # where sum_target = 1 (since inactive vars are 0)
        # KKT: 2*Q_FF*a_F = lambda * 1
        # a_F = (lambda/2) * Q_FF^{-1} * 1
        # 1^T Q_FF^{-1} * 1 * (lambda/2) = 1
        # lambda = 2 / (1^T Q_FF^{-1} * 1)

        F = np.where(active)[0]
        nF = len(F)
        if nF == 0:
            return None

        # Build Q_FF first row (sub-Toeplitz)
        # Q[i,j] = first_row[abs(i-j)]
        # For the submatrix on indices F, we need Q_{F,F}
        # This is NOT Toeplitz in general (unless F is contiguous or arithmetic)
        # For simplicity, build the dense submatrix
        if nF <= 3000:
            Q_FF = np.zeros((nF, nF))
            for i in range(nF):
                for j in range(i, nF):
                    Q_FF[i, j] = first_row[abs(F[i] - F[j])]
                    Q_FF[j, i] = first_row[abs(F[i] - F[j])]
        else:
            # Use full matrix and extract
            Q_full = np.zeros((n, n))
            for i in range(n):
                for j in range(i, n):
                    Q_full[i, j] = first_row[abs(i - j)]
                    Q_full[j, i] = first_row[abs(i - j)]
            Q_FF = Q_full[np.ix_(F, F)]

        # Solve Q_FF * x = 1
        try:
            x = np.linalg.solve(Q_FF, np.ones(nF))
        except np.linalg.LinAlgError:
            # Add regularization
            try:
                x = np.linalg.solve(Q_FF + 1e-8 * np.eye(nF), np.ones(nF))
            except np.linalg.LinAlgError:
                return None

        denom = x.sum()
        if abs(denom) < 1e-15:
            return None

        a_F = x / denom

        # Check non-negativity
        if np.min(a_F) >= -1e-10:
            # All non-negative, this is the solution
            a_new = np.zeros(n)
            a_new[F] = np.maximum(a_F, 0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new /= s_new
            return a_new
        else:
            # Some negative - add most negative to inactive set
            min_idx = np.argmin(a_F)
            active[F[min_idx]] = False

    # If we ran out of sweeps, return best we have
    return None
