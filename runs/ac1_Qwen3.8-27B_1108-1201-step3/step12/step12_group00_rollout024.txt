def fast_autocorr(a):

def build_sparse_conv_matrix(a):
    """Build sparse matrix T where (T @ b)_k = conv(a, b)_k."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_best_response_sparse(a, M, solver, warm_start=None):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_sparse_conv_matrix(a)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    kwargs = {'verbose': False}
    if warm_start is not None:
        try:
            kwargs['warm_start'] = True
            kwargs['initial_guess'] = {'primal': {'x': warm_start}}
        except Exception:
            kwargs = {'verbose': False}
    
    try:
        prob.solve(solver=solver, **kwargs)
    except Exception:
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def nesterov_lp_optimize(a0, time_limit, solver, verbose=False, restart_interval=400):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_iters = 0
    rng = np.random.default_rng(12345)

    def run_nesterov(a_start, t_start, time_left):
        nonlocal best_a, best_score, n_iters
        y = a_start.copy()
        a_curr = y.copy()
        avg_a = np.zeros(n)
        n_avg = 0

        for t in range(t_start, t_start + 1000000):
            if time.monotonic() - t0 > time_left:
                break

            M_curr = fast_autocorr(y).max()
            if M_curr < 1e-12:
                break

            b = lp_best_response_sparse(y, M_curr, solver)
            if b is None:
                continue

            b_sum = b.sum()
            if b_sum < 1e-15:
                continue
            g = b * norm / b_sum
            n_iters += 1

            alpha = 2.0 / (t - t_start + 2.0)
            y_new = (1.0 - alpha) * y + alpha * g
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y

            if t == t_start:
                a_new = g.copy()
            else:
                momentum = (t - t_start - 1.0) / (t - t_start + 2.0)
                a_new = y_new + momentum * (y_new - y)
                a_new = np.maximum(a_new, 0.0)
                s_a = a_new.sum()
                if s_a < 1e-15:
                    a_new = y_new.copy()
                else:
                    a_new = a_new * norm / s_a

            y = y_new
            a_curr = a_new

            score_y = fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()

            score_a = fast_autocorr(a_curr).max()
            if score_a < best_score:
                best_score = score_a
                best_a = a_curr.copy()

            avg_a += a_curr
            n_avg += 1
            if n_avg % 50 == 0:
                avg_check = avg_a / n_avg
                s_avg = avg_check.sum()
                if s_avg > 1e-15:
                    avg_check = avg_check * norm / s_avg
                    score_avg = fast_autocorr(avg_check).max()
                    if score_avg < best_score:
                        best_score = score_avg
                        best_a = avg_check.copy()

        return y

    # Main loop with restarts
    t_current = 1
    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        chunk = min(remaining, 300.0)  # Max 300s per chunk
        a_start = best_a.copy()
        
        # Small random perturbation for restart diversity
        if t_current > 1:
            pert = rng.standard_normal(n) * 0.01 * best_a
            a_start = a_start + pert
            a_start = np.maximum(a_start, 0.0)
            s_p = a_start.sum()
            if s_p > 1e-15:
                a_start = a_start * norm / s_p

        a_start = run_nesterov(a_start, t_current, time.monotonic() - t0 + chunk)
        t_current += restart_interval + 1

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    NLP n={n} ({elapsed:.0f}s): iters={n_iters}, best={best_score:.8f}")

    return best_a, best_score

def spectral_peak_shave(a0, time_limit, seed=42):
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:5]

        improved = False
        for k in top_k:
            # Find the pair (i, j) that contributes most to peak k
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            src = idx[np.argmax(contrib)]
            
            # Find a destination that doesn't contribute to top peaks
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]

            # Pick candidates with lowest value
            candidates_sorted = candidates[np.argsort(a[candidates])][:max(1, len(candidates) // 4)]
            dst = candidates_sorted[rng.integers(len(candidates_sorted))]

            for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-10:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
            if stall > 200:
                break

    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
import time
import scipy.sparse as sp
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

_conv_pattern_cache = {}


def get_conv_pattern(n):
    """Pre-compute sparse pattern for the convolution constraint matrix."""
    if n in _conv_pattern_cache:
        return _conv_pattern_cache[n]
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        count = j_end - j_start + 1
        rows.extend([k] * count)
        cols.extend(range(j_start, j_end + 1))
    pattern = (np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32))
    _conv_pattern_cache[n] = pattern
    return pattern


def build_conv_sparse(a, n):
    """Build the sparse constraint matrix T where (T@b)_k = conv(a,b)_k."""
    rows, cols = get_conv_pattern(n)
    vals = a[rows - cols]
    return sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def fast_autocorr(a):
    """Compute full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate the objective function."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def lp_best_response(a, M, n):
    """Solve the best-response LP: max sum(b) s.t. conv(a,b)_k <= M, b >= 0."""
    T = build_conv_sparse(a, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30.0})
        if res.status != 0 or res.x is None:
            return None
        b_val = np.maximum(res.x, 0.0)
        s = b_val.sum()
        if s < 1e-15:
            return None
        return b_val
    except Exception:
        return None


def nesterov_lp_optimize(a0, time_limit, verbose=False, restart_interval=400):
    """Nesterov acceleration on the LP best-response game with periodic restarts."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_iters = 0
    rng = np.random.default_rng(54321)

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10.0:
            break
        chunk_time = min(remaining, 250.0)
        chunk_end = time.monotonic() + chunk_time

        # Start from best with small perturbation (except first iteration)
        if n_iters > 0:
            a_start = best_a + rng.standard_normal(n) * 0.005 * best_a
            a_start = np.maximum(a_start, 0.0)
            s_p = a_start.sum()
            if s_p > 1e-15:
                a_start = a_start * norm / s_p
        else:
            a_start = a.copy()

        # Run Nesterov for this chunk
        y = a_start.copy()
        a_curr = y.copy()
        avg_a = np.zeros(n)
        n_avg = 0
        t_local = 1

        while time.monotonic() < chunk_end:
            M_curr = fast_autocorr(y).max()
            if M_curr < 1e-12:
                break

            b = lp_best_response(y, M_curr, n)
            if b is None:
                t_local += 1
                continue

            b_sum = b.sum()
            if b_sum < 1e-15:
                t_local += 1
                continue
            g = b * norm / b_sum
            n_iters += 1

            alpha = 2.0 / (t_local + 2.0)
            y_new = (1.0 - alpha) * y + alpha * g
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y

            if t_local == 1:
                a_new = g.copy()
            else:
                momentum = (t_local - 1.0) / (t_local + 2.0)
                a_new = y_new + momentum * (y_new - y)
                a_new = np.maximum(a_new, 0.0)
                s_a = a_new.sum()
                if s_a < 1e-15:
                    a_new = y_new.copy()
                else:
                    a_new = a_new * norm / s_a

            y = y_new
            a_curr = a_new

            score_y = fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()

            score_a = fast_autocorr(a_curr).max()
            if score_a < best_score:
                best_score = score_a
                best_a = a_curr.copy()

            avg_a += a_curr
            n_avg += 1
            if n_avg % 40 == 0:
                avg_check = avg_a / n_avg
                s_avg = avg_check.sum()
                if s_avg > 1e-15:
                    avg_check = avg_check * norm / s_avg
                    score_avg = fast_autocorr(avg_check).max()
                    if score_avg < best_score:
                        best_score = score_avg
                        best_a = avg_check.copy()

            t_local += 1

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    NLP n={n} ({elapsed:.0f}s): total_iters={n_iters}, best={best_score:.8f}")

    return best_a, best_score


def lse_gradient_polish(a0, time_limit, T_val=5.0, n_steps=500):
    """Gradient descent on LogSumExp smoothing of the peak objective."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()

    for it in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-12:
            break

        w = np.exp((b - max_b) / T_val)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        # Compute gradient: grad_i = 2 * sum_d a[d] * w[i+d]
        a_rev = a[::-1]
        A_rev = rfft(a_rev, fft_size)
        W = rfft(w, fft_size)
        conv_aw = irfft(A_rev * W, fft_size)
        grad = 2.0 * conv_aw[n - 1:n - 1 + n]

        # Adaptive step size with backtracking
        step = 1.0 / (T_val * 0.5 + n * 0.1)
        for trial in range(10):
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                step *= 0.5
                continue
            a_new = a_new * norm / s_new
            score_new = fast_autocorr(a_new).max()
            if score_new < best_score or score_new < max_b * (1.0 - 1e-6):
                break
            step *= 0.5
        else:
            continue

        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score


def spectral_peak_shave(a0, time_limit, seed=42):
    """Coordinate descent targeting the top autocorrelation peaks."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:5]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            src = idx[np.argmax(contrib)]

            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]

            candidates_sorted = candidates[np.argsort(a[candidates])][:max(1, len(candidates) // 4)]
            dst = candidates_sorted[rng.integers(len(candidates_sorted))]

            for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-10:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
            if stall > 150:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving normalized shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    if t3 > 30:
        init_long = interpolate_sequence(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t3, verbose=True, restart_interval=400)
        do_try_update(a_out, f"lp_n{n2}")

    # Phase 4: Try n3
    n3 = n_base + 896
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LP Nesterov n={n3} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.18, 300.0)
    if t4 > 30:
        init_long3 = interpolate_sequence(best_seq, n3)
        a_norm = init_long3 * np.sqrt(2.0 * n3) / init_long3.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t4, verbose=True, restart_interval=400)
        do_try_update(a_out, f"lp_n{n3}")

    # Phase 5: Peak shaving on current best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Peak shaving n={n_best} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.12, 140.0)
    if t5 > 5:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = spectral_peak_shave(a_norm, t5, seed=777)
        do_try_update(a_out, "peak_shave")

    # Phase 6: Second LP round on best length (fresh start with perturbation)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: LP Nesterov round 2, n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.15, 180.0)
    if t6 > 20:
        rng2 = np.random.default_rng(seed + 100)
        a_pert = best_seq * (1.0 + 0.02 * rng2.standard_normal(n_best))
        a_pert = np.maximum(a_pert, 0.0)
        s_p = a_pert.sum()
        if s_p > 1e-15:
            a_norm = a_pert * np.sqrt(2.0 * n_best) / s_p
            a_out, _ = nesterov_lp_optimize(a_norm, t6, verbose=True, restart_interval=300)
            do_try_update(a_out, f"lp2_n{n_best}")

    # Phase 7: Final LSE + peak shave
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final polish n={n_best} (budget={remaining:.0f}s) ===")
    if remaining > 20:
        t7a = min(remaining * 0.4, 70.0)
        t7b = min(remaining * 0.4, 70.0)
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_gradient_polish(a_norm, t7a, T_val=2.0, n_steps=300)
        do_try_update(a_out, "lse_final")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            n_best2 = len(best_seq)
            a_norm = best_seq * np.sqrt(2.0 * n_best2) / best_seq.sum()
            a_out, _ = spectral_peak_shave(a_norm, min(remaining * 0.7, 60.0), seed=999)
            do_try_update(a_out, "peak_shave_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0, 1.0, 1.0]

    return [float(x) for x in best_seq]
