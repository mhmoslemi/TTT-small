import time

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_autocorr(a)
    return float(2 * n * b.max() / s**2)

def lse_gradient(a, T, topk=0):
    n = len(a)
    b = fast_autocorr(a)
    s = a.sum()
    if topk > 0 and topk < len(b):
        idx = np.argsort(b)[::-1][:topk]
        bv = b[idx]
    else:
        idx = np.arange(len(b))
        bv = b
    
    Tc = T * bv
    Tc_shift = Tc - Tc.max()
    exp_Tc = np.exp(Tc_shift)
    w = exp_Tc / exp_Tc.sum()
    f_val = (Tc.max() + np.log(exp_Tc.sum())) / T
    
    grad_b = np.zeros(n)
    for ki in range(len(idx)):
        wk = w[ki]
        if wk < 1e-15:
            continue
        k = idx[ki]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        grad_b[js:je+1] += wk * a[k - np.arange(js, je+1)]
    
    grad = 2.0 * n * (grad_b / s - 2.0 * f_val * a / (s * s))
    return f_val, grad

def adam_lse_optimize(a0, time_limit, T_start=50.0, T_end=2.0, n_stages=15, topk=15):

    if s < 1e-15:

    else:
        a = a / s

    best_score = compute_score(a)
    t0 = time.monotonic()
    
    # Use log-space parameterization
    z = np.log(np.maximum(a, 1e-30))
    
    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit - 5:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        lr = 0.01 * min(1.0, 50.0 / n)
        m = np.zeros(n)
        v = np.zeros(n)
        t_adam = 0
        b1, b2, eps = 0.8, 0.999, 1e-8
        
        max_it = max(20, int(remaining * 2))
        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 3:
                break
            a = np.exp(z - z.max())
            a = a / a.sum()

            z = z - lr * mh / (np.sqrt(vh) + eps)
            
            if (it + 1) % 5 == 0:

                a = a / a.sum()
                sc = compute_score(a)

                    best_a = a.copy()

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp gradient descent with Adam for initial convergence
2. Pairwise transfer optimization with exact 1D minimax solve (main workhorse)
3. Multi-length exploration for global improvement
4. Basin hopping with random perturbations to escape local minima
5. Final coordinated polish

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the objective locally (fast, no validation)."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_autocorr(a)
    return float(2 * n * b.max() / s ** 2)


def lse_grad(a, T, topk):
    """Compute gradient of smoothed objective in a-space."""
    n = len(a)
    b = fast_autocorr(a)
    s = a.sum()
    if topk > 0 and topk < len(b):
        idx = np.argsort(b)[::-1][:topk]
        bv = b[idx]
    else:
        idx = np.arange(len(b))
        bv = b.copy()

    Tc = T * bv
    Tc_shift = Tc - Tc.max()
    exp_Tc = np.exp(Tc_shift)
    w = exp_Tc / exp_Tc.sum()
    f_val = (Tc.max() + np.log(exp_Tc.sum())) / T

    grad_b = np.zeros(n)
    for ki in range(len(idx)):
        wk = w[ki]
        if wk < 1e-15:
            continue
        k = idx[ki]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        grad_b[js:je + 1] += wk * a[k - np.arange(js, je + 1)]

    grad = 2.0 * n * (grad_b / s - 2.0 * f_val * a / (s * s))
    return f_val, grad

def compute_peak_vec(a, peak_idx):
    """Compute vec(x) = (a[p_1-x], ..., a[p_K-x]) for all x, with 0 for OOB."""

        if x_lo <= x_hi:
            vec[x_lo:x_hi + 1, ki] = a[p - np.arange(x_lo, x_hi + 1)]
    return vec
