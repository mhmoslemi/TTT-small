import numpy as np

def fast_autocorr(a):
    """Full self-convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where (T@b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

def softmax_fn(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def compute_lse_grad(a, T):
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    a_rev = a[::-1].copy()
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    if conv_len > fft_size:
        fft_size *= 2
    W = rfft(w, fft_size)
    A_rev = rfft(a_rev, fft_size)
    conv = irfft(W * A_rev, fft_size)
    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad, w

def lse_obj_grad_z(z, T):
    a = softmax_fn(z)
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    obj_val = (max_Tb + np.log(exp_Tb.sum())) / T
    grad_a, w = compute_lse_grad(a, T)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj_val, grad_z

def get_lp_solver():
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.3])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def solve_lp_best_response(a_fixed, solver):
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = float(M_var.value)
    return b_val, M_val

def mirror_descent_optimize(a0, time_limit, n_steps=500000):
    """
    Mirror descent (multiplicative weight updates) on the simplex.
    At each step, compute subgradient of max autocorrelation, 
    update multiplicatively, project onto simplex.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Adaptive learning rate
    eta = 0.1
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(a)
        b_max = b.max()
        
        # Find all lags within 5% of max (for smoother gradient)
        threshold = b_max * 0.95
        active = np.where(b >= threshold)[0]
        
        if len(active) == 0:
            active = [int(np.argmax(b))]
        
        # Compute subgradient: sum over active lags
        # grad[i] = sum_k a[k-i] for active k (where 0 <= k-i < n)
        # This is equivalent to (w * a_rev)[i + n - 1] where w = uniform over active
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / len(active)
        
        a_rev = a[::-1]
        conv_size = n + (2 * n - 1) - 1
        fft_size = 1 << (conv_size - 1).bit_length()
        W = rfft(w, fft_size)
        AR = rfft(a_rev, fft_size)
        conv = irfft(W * AR, fft_size)
        grad = 2.0 * conv[n - 1:n - 1 + n]
        
        # Mirror descent step: a_i <- a_i * exp(-eta * grad_i)
        # Then normalize to sum = 1
        # Use log-domain for stability
        log_a = np.log(np.maximum(a, 1e-30))
        log_a -= eta * grad
        # Softmax to normalize
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-30:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / a_new_sum
        
        # Check score
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            # Adaptive: if not improving, reduce eta
            if score > best_score * 1.01:
                eta *= 0.9
            a = a_new  # Still accept to explore
        
        # Periodically print
        if step % 10000 == 0:
            elapsed = time.monotonic() - t0
            cur_score = 2.0 * n * fast_autocorr(a).max()
            print(f"      MD step={step} ({elapsed:.0f}s): best={best_score:.8f}, cur={cur_score:.8f}, eta={eta:.6f}")
    
    return best_a, best_score

def coordinated_peak_shave(a0, time_limit, seed=42):
    """
    Coordinated peak shaving: try multi-index moves to reduce peaks.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        K = min(8, 2 * n - 1)
        top_k = np.argsort(b)[::-1][:K]
        
        improved = False
        
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            # Top contributors
            top_contrib = np.argsort(contrib)[::-1][:5]
            
            for tc in top_contrib:
                src = idx[tc]
                src_val = a[src]
                if src_val < 1e-14:
                    continue
                
                # Try moving mass to random positions
                for _ in range(20):
                    dst = rng.integers(n)
                    if dst == src:
                        continue
                    for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
                        amt = src_val * frac
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_autocorr(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall > 200:
                break
            # Random perturbation
            n_trials = 10
            for _ in range(n_trials):
                i = rng.integers(n)
                if a[i] < 1e-12:
                    continue
                j = rng.integers(n)
                if i == j:
                    continue
                frac = rng.uniform(0.001, 0.03)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
                    break
    
    return best_a, best_score

def logsumexp_optimize(a0, time_limit, T_start=100.0, T_end=0.02, n_stages=20):
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        iters_per_stage = max(10, min(60, int(20 + 40 * (si / max(1, n_stages - 1)))))
        
        def func(z_in):
            val, _ = lse_obj_grad_z(z_in, stage_T)
            return val
        
        def jac(z_in):
            _, g = lse_obj_grad_z(z_in, stage_T)
            return g
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': iters_per_stage * 300,
                         'disp': False}
            )
            z = res.x
            a = softmax_fn(z)
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
        except Exception:
            pass
    
    return best_a, best_score

def coordinate_descent_polish(a0, time_limit, seed=0):
    """
    Cyclic coordinate descent with adaptive step sizes.
    For each coordinate, find the optimal value by 1D search.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    order = np.arange(n)
    
    iteration = 0
    while time.monotonic() - t0 < time_limit:
        rng.shuffle(order)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Try several fractions of a[i] to redistribute
            if a[i] < 1e-14:
                continue
            
            # Try reducing a[i] and distributing to a few targets
            for frac in [0.01, 0.05, 0.1, 0.2]:
                amt = a[i] * frac
                # Distribute to random targets
                targets = rng.choice(n, size=min(5, n), replace=False)
                for j in targets:
                    if j == i:
                        continue
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0)
                    a_trial_sum = a_trial.sum()
                    if a_trial_sum < 1e-15:
                        continue
                    a_trial = a_trial / a_trial_sum
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        break
        
        iteration += 1
        if iteration % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      CD iter={iteration} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
