def fast_autocorr(a):
    """Full linear convolution a*a using FFT. Returns array of length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_score(a):
    """Compute 2n*max(b)/(sum(a))^2 quickly."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)

def lse_grad_a(a, T):
    """
    Compute gradient of (1/T)*log(sum_k exp(T*b_k)) w.r.t. a.
    grad[j] = 2 * sum_k w[k] * a[k-j] where w = softmax(T*b).
    Using FFT: C[j] = conv(a, w_rev)[2n-2-j].
    Returns (objective_value, gradient_array_of_length_n).
    """
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    Tb_shift = Tb - Tb.max()
    exp_Tb = np.exp(Tb_shift)
    obj = (Tb.max() + np.log(exp_Tb.sum())) / T
    w = exp_Tb / exp_Tb.sum()
    # grad[j] = 2 * sum_k w[k] * a[k-j] (for valid 0<=k-j<n)
    # = 2 * C[j] where C[j] = sum_{m=0}^{n-1} w[j+m] * a[m]

    c = irfft(A * W, fft_size)
    C = c[2 * n - 2: n - 2: -1]
    return obj, 2.0 * C


def lse_obj_grad_z(z, T):
    """
    Objective and gradient in z-space where a = softmax(z) (normalized).

    n = len(z)
    z_shift = z - z.max()
    e = np.exp(z_shift)
    a = e / e.sum()
    obj, grad_a = lse_grad_a(a, T)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

    t0 = time.monotonic()
    stall_count = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        b_max = b.max()
        b_mean = b.sum() / len(b)

        frac = min(1.0, t / float(k_ramp))
        k = max(k_min, min(k_max, int(k_min + frac * (k_max - k_min))))

        top_idx = np.argsort(b)[::-1][:k]
        excess = b[top_idx] - b_mean
        if excess.max() < 1e-15:
            weights = np.ones(k) / k
        else:
            excess_shift = excess - excess.max()
            weights = np.exp(excess_shift) / np.exp(excess_shift).sum()

        # Compute subgradient
        grad = np.zeros(n)
        for ii in range(k):
            m = int(top_idx[ii])
            w_m = weights[ii] * 2.0
            j_start = max(0, m - n + 1)
            j_end = min(n - 1, m)
            js = np.arange(j_start, j_end + 1)
            grad[js] += w_m * a[m - js]

        # Base learning rate with decay
        eta = eta0 / (1.0 + t / 400.0) ** decay_rate

        # Backtracking line search (up to 5 halving steps)
        accepted = False
        for ls_step in range(6):
            eta_try = eta / (2.0 ** ls_step)
            a_new = a * np.exp(-eta_try * grad)
            a_new = np.maximum(a_new, 1e-30)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
            sc_new = fast_score(a_new)
            if sc_new <= best_score + 1e-12:
                a = a_new
                if sc_new < best_score - 1e-14:
                    best_score = sc_new
                    best_a = a.copy()
                    stall_count = 0
                else:
                    stall_count += 1
                accepted = True
                break
        if not accepted:
            # No improvement found; accept a small step anyway to avoid stalling
            a_new = a * np.exp(-eta * 0.1 * grad)
            a_new = np.maximum(a_new, 1e-30)
            a_new = a_new / a_new.sum()
            a = a_new
            stall_count += 1

        # Stall handling
        if stall_count > 300:
            noise_scale = min(0.03, 0.005 * (stall_count / 100.0))
            a = a + noise_scale * rng.standard_normal(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            stall_count = 0
            best_score = min(best_score, fast_score(a))

        if t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD t={t} ({elapsed:.0f}s): best={best_score:.6f}, bmax={b_max:.6e}, k={k}")

    return best_a, best_score


def lse_optimize_multi(a0, time_limit, T_stages=None, seed=42):
    """
    LogSumExp-smoothed optimization with L-BFGS and multi-stage temperature.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s

    if T_stages is None:
        T_stages = [20.0, 50.0, 100.0, 200.0, 400.0, 800.0, 1500.0]

    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    best_a = a_init.copy()
    best_score = fast_score(a_init)

    n_stages = len(T_stages)
    for si, stage_T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        iters = max(10, min(60, int(15 + 45 * si / max(1, n_stages - 1))))

        def func(z_in):
            val, _ = lse_obj_grad_z(z_in, stage_T)
            return val

        def jac(z_in):
            _, g = lse_obj_grad_z(z_in, stage_T)
            return g

        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': iters, 'ftol': 1e-16,
                                    'gtol': 1e-10, 'maxfun': iters * 200,
                                    'disp': False})
            z = res.x
            z_shift = z - z.max()
            a = np.exp(z_shift)
            a = a / a.sum()
            a = np.maximum(a, 0)
            a = a / a.sum()
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        except Exception:
            pass

        if si % 2 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {si+1}/{n_stages} (T={stage_T:.0f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def peak_shaving_v2(a0, time_limit, seed=123):
    """
    Improved peak shaving: uses exact gradient to find best moves.
    For each top peak, finds the coordinate that most reduces the peak
    and transfers mass to a low-contribution position.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:8]

        improved_this_iter = False
        for k_idx_val in top_k:
            k_idx = int(k_idx_val)
            # For peak at k, contributions are a[i]*a[k-i] for valid i
            js = max(0, k_idx - n + 1)
            je = min(n - 1, k_idx)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_idx - idx]
            total_contrib = contrib.sum()
            if total_contrib < 1e-15:
                continue

            # Find best source: the pair contributing most to this peak
            src_local = int(np.argmax(contrib))
            src = int(idx[src_local])
            partner = k_idx - src

            # Find best destination: position that minimizes increase in all top peaks
            # Score each candidate destination by how much it would increase peaks
            # Use a simple heuristic: prefer positions with low a value
            # and not contributing to top peaks

            # Build conflict mask: positions that contribute to any top peak
            conflict_score = np.zeros(n)
            for k2 in top_k[:5]:
                k2 = int(k2)
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                for jj in range(js2, je2 + 1):
                    other = k2 - jj
                    if 0 <= other < n:
                        conflict_score[jj] += a[js2 + (jj - js2)] * a[other] if False else a[jj] * a[other]

            # Candidates: low a, low conflict
            candidate_score = a + 0.5 * conflict_score / max(b.max(), 1e-15)
            # Exclude source and partner
            candidate_score[src] = 1e10
            if 0 <= partner < n:
                candidate_score[partner] = 1e10
            # Prefer spreading: penalize positions near other high-a positions
            dst_candidates = np.argsort(candidate_score)[:max(10, n // 20)]

            for dst in dst_candidates[:5]:
                dst = int(dst)
                for frac in [0.001, 0.003, 0.008, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    sc = fast_score(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial.copy()
                        best_score = sc
                        best_a = a.copy()
                        improved_this_iter = True
                        break
                if improved_this_iter:
                    break
            if improved_this_iter:
                break

        if not improved_this_iter:
            stall += 1
            # Random perturbation to escape
            if rng.integers(4) == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.015)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        a_trial = a_trial / a_trial.sum()
                        sc = fast_score(a_trial)
                        if sc < best_score:
                            a = a_trial.copy()
                            best_score = sc
                            best_a = a.copy()
                            stall = 0

            if stall > 300:
                break

    return best_a, best_score


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def diverse_init(n, rng, variant=None):
    """Generate diverse structured initializations."""
    if variant is None:
        variant = int(rng.integers(0, 12))

    if variant == 0:
        x = np.linspace(0, 1, n)
        a = np.maximum(1 - 2 * abs(x - 0.5), 0)
    elif variant == 1:
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif variant == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-3 * x ** 2)
    elif variant == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-20 * (x - 0.25) ** 2) + np.exp(-20 * (x - 0.75) ** 2)
    elif variant == 4:
        # U-shaped
        x = np.linspace(0, 1, n)
        a = x ** 2 + (1 - x) ** 2 + 0.05
    elif variant == 5:
        # W-shaped
        x = np.linspace(0, 1, n)
        a = 1 + 0.5 * np.cos(4 * np.pi * x)
    elif variant == 6:
        # Sinc-squared
        x = np.linspace(-1, 1, n)
        a = np.sinc(x) ** 2
    elif variant == 7:
        # Sharp U (power law edges)
        x = np.linspace(1e-6, 1 - 1e-6, n)
        a = x ** (-0.3) + (1 - x) ** (-0.3)
    elif variant == 8:
        # Extreme U
        x = np.linspace(1e-6, 1 - 1e-6, n)
        a = x ** (-0.7) + (1 - x) ** (-0.7)
    elif variant == 9:
        # V-shaped (dip in center)
        x = np.linspace(0, 1, n)
        a = abs(x - 0.5) + 0.1
    elif variant == 10:
        # Two broad bumps
        x = np.linspace(0, 1, n)
        a = np.exp(-30 * (x - 0.2) ** 2) + np.exp(-30 * (x - 0.8) ** 2) + 0.01
    else:
        # Random Gaussian bumps
        a = np.ones(n) * 0.1
        for _ in range(6):
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.05, n * 0.3)
            amp = rng.uniform(0.5, 2.0)
            x = np.arange(n, dtype=np.float64)
            a += amp * np.exp(-(x - center) ** 2 / (2 * width ** 2))
        a = np.maximum(a, 0)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def multi_scale_init(n_target, time_budget, seed=42):
    """
    Multi-scale initialization: optimize at small lengths, interpolate up.
    Returns a good initial sequence of length n_target.
    """
    rng = np.random.default_rng(seed)
    scales = [100, 200, 400, 800]
    scales = [s for s in scales if s < n_target]
    if not scales:
        return diverse_init(n_target, rng, variant=8)

    current = None
    t0 = time.monotonic()
    for scale in scales:
        remaining = time_budget - (time.monotonic() - t0)
        if remaining < 5:
            break
        t_alloc = min(remaining * 0.4, 40.0)
        if current is None:
            init = diverse_init(scale, rng, variant=8)  # Sharp U
        else:
            init = interpolate_to_length(current, scale)

        # Quick MD
        t_md = t_alloc * 0.6
        a_out, _ = mirror_descent_ls(init, t_md, eta0=0.06, decay_rate=0.7,
                                      k_min=3, k_max=10, k_ramp=100, seed=seed + scale)
        # Quick LSE
        t_lse = t_alloc * 0.4
        a_out2, _ = lse_optimize_multi(a_out, t_lse, T_stages=[30.0, 80.0, 200.0], seed=seed + scale + 1)
        current = a_out2 if fast_score(a_out2) < fast_score(a_out) else a_out

    if current is not None:
        return interpolate_to_length(current, n_target)
    return diverse_init(n_target, rng, variant=8)


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function with multi-phase optimization.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 40.0, 1750.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    # Try parent sequence
    n_target = 1600
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = fast_score(parent)
            print(f"Parent: n={len(parent)}, fast_score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
            n_target = len(parent)
    except NameError:
        pass
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = fast_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        # No parent: use multi-scale initialization
        print("No parent available. Using multi-scale initialization...")
        t_ms = min(time_budget * 0.15, 200.0)
        best_seq = multi_scale_init(n_target, t_ms, seed=seed)
        best_score = fast_score(best_seq)
        print(f"Multi-scale init: n={n_target}, score={best_score:.6f}")

    # === Phase 1: Mirror descent at parent length ===
    print(f"\n=== Phase 1: MD at n={n_target} ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_md1 = min(remaining * 0.25, 400.0)
    if t_md1 > 30:
        a_out, _ = mirror_descent_ls(best_seq, t_md1, eta0=0.035, decay_rate=0.55,
                                      k_min=5, k_max=20, k_ramp=500, seed=1)
        try_update(a_out, f"md1_n{n_target}")

    # === Phase 2: LSE continuation ===
    print(f"\n=== Phase 2: LSE at n={n_target} ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_lse1 = min(remaining * 0.30, 500.0)
    if t_lse1 > 30 and best_seq is not None:
        a_out, _ = lse_optimize_multi(best_seq, t_lse1,
                                       T_stages=[15.0, 40.0, 100.0, 250.0, 500.0, 1000.0],
                                       seed=7)
        try_update(a_out, f"lse1_n{n_target}")

    # === Phase 3: Peak shaving ===
    print(f"\n=== Phase 3: Peak Shaving at n={n_target} ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_ps1 = min(remaining * 0.12, 150.0)
    if t_ps1 > 10 and best_seq is not None:
        a_out, _ = peak_shaving_v2(best_seq, t_ps1, seed=99)
        try_update(a_out, f"ps1_n{n_target}")

    # === Phase 4: Second MD round (different params) ===
    print(f"\n=== Phase 4: MD round 2 at n={n_target} ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_md2 = min(remaining * 0.12, 150.0)
    if t_md2 > 10 and best_seq is not None:
        a_out, _ = mirror_descent_ls(best_seq, t_md2, eta0=0.02, decay_rate=0.7,
                                      k_min=8, k_max=30, k_ramp=200, seed=77)
        try_update(a_out, f"md2_n{n_target}")

    # === Phase 5: Diverse restart ===
    print(f"\n=== Phase 5: Diverse restarts ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_div = min(remaining * 0.10, 150.0)
    if t_div > 20:
        for v in [7, 8]:  # Sharp U variants
            init = diverse_init(n_target, rng, variant=v)
            # Quick MD
            t_sub = t_div / 3
            a_out, _ = mirror_descent_ls(init, t_sub, eta0=0.05, decay_rate=0.6,
                                          k_min=5, k_max=15, k_ramp=300, seed=v * 100)
            try_update(a_out, f"div_v{v}")
            # Quick LSE
            a_out2, _ = lse_optimize_multi(a_out, t_sub, T_stages=[20.0, 80.0, 300.0], seed=v * 100 + 1)
            try_update(a_out2, f"div_v{v}_lse")

    # === Phase 6: Try longer length ===
    n_long = int(round(n_target * 1.3))
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Longer length n={n_long} ===")
    t_long = min(remaining * 0.12, 150.0)
    if t_long > 20 and best_seq is not None:
        init_long = interpolate_to_length(best_seq, n_long)
        a_out, _ = mirror_descent_ls(init_long, t_long * 0.6, eta0=0.03, decay_rate=0.6,
                                      k_min=5, k_max=20, k_ramp=300, seed=2)
        try_update(a_out, f"md_n{n_long}")
        a_out2, _ = lse_optimize_multi(a_out, t_long * 0.4, T_stages=[10.0, 50.0, 200.0], seed=22)
        try_update(a_out2, f"lse_n{n_long}")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    remaining = time_budget - (time.monotonic() - t_global)
    if best_seq is not None and remaining > 20:
        # Final MD with very small steps
        t_f1 = min(remaining * 0.4, 80.0)
        if t_f1 > 10:
            a_out, _ = mirror_descent_ls(best_seq, t_f1, eta0=0.01, decay_rate=0.8,
                                          k_min=10, k_max=40, k_ramp=100, seed=555)
            try_update(a_out, "final_md")

        # Final LSE
        remaining = time_budget - (time.monotonic() - t_global)
        t_f2 = min(remaining * 0.4, 60.0)
        if t_f2 > 5:
            a_out, _ = lse_optimize_multi(best_seq, t_f2, T_stages=[50.0, 200.0, 500.0], seed=333)
            try_update(a_out, "final_lse")

        # Final peak shave
        remaining = time_budget - (time.monotonic() - t_global)
        t_f3 = min(remaining * 0.3, 40.0)
        if t_f3 > 5:
            a_out, _ = peak_shaving_v2(best_seq, t_f3, seed=4444)
            try_update(a_out, "final_ps")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = min(best_score, fs)
        print(f"\nFINAL: fast_score={fast_score(best_seq):.6f}, official_score={best_score:.6f}, n={len(best_seq)}")
    else:
        print("ERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
