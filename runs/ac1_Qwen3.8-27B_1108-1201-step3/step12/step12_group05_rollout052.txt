import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(arr):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u > (cssv - 1.0) / np.arange(1, n + 1)
    rho = np.argmax(cond)  # first True
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based LogSumExp gradient optimization (O(n log n) per iteration)
2. Spectral shaping for initialization
3. Multi-stage temperature annealing with Adam
4. LP-based Nesterov refinement for final precision
5. Peak shaving coordinate descent
6. Multi-length exploration
Key innovation: FFT cross-correlation for gradient computation enables
100x more iterations than direct methods, achieving better convergence.

import numpy as np
import cvxpy as cp

from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_value_grad(a, T):
    """
    Compute (value, gradient) of F_T(a) = (1/T) log sum_k exp(T * b_k)
    where b = convolve(a, a) has length 2n-1.
    Gradient computed via FFT cross-correlation: O(n log n).
    """
    n = len(a)
    b = fast_autocorr(a)
    
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    F_val = (max_Tb + np.log(exp_Tb.sum())) / T
    
    w = exp_Tb / exp_Tb.sum()  # length 2n-1
    
    # Gradient: grad[j] = 2 * sum_k w[k] * a[k-j] (for valid k-j in [0, n-1])
    # = 2 * h[j] where h[j] = sum_m w[j+m] * a[m]
    # h[j] = (w_rev * a)[2n-2-j] where w_rev = w[::-1]
    # (linear convolution, length 3n-2)
    
    w_rev = w[::-1]
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w_rev, fft_size)
    A = rfft(a, fft_size)
    conv = irfft(W * A, fft_size)
    
    # h[j] = conv[2n-2-j] for j=0..n-1
    # h = conv[2n-2 : n-2 : -1] = conv[n-1 : 2n-1][::-1]
    h = conv[n-1:2*n-1][::-1]
    
    grad = 2.0 * h
    return F_val, grad


def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}. O(n log n)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u > (cssv - 1.0) / np.arange(1, n + 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.argmax(cond)
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.05, 20.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def lse_adam_softmax(a0, time_limit, T_start=200.0, T_end=0.5, n_stages=20, lr=0.01):
    """
    LSE optimization with Adam in softmax (log) parameterization.
    Fast: O(n log n) per iteration via FFT gradients.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t_adam = 0
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        # Estimate iterations we can do per second
        # Each iteration is roughly O(n log n), let's estimate ~2000 iters/s for n=2000
        iters_estimate = int(remaining * 1500 / (n_stages - si))
        iters = max(30, min(500, iters_estimate))
        
        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break
            
            a = softmax_fn(z)
            obj, grad_a = lse_value_grad(a, stage_T)
            
            # Chain rule: grad_z = a * (grad_a - dot(grad_a, a))
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)
            
            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)
            
            lr_t = lr / (1 + 0.01 * it)
            z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            
            if it % 20 == 0:
                a_check = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a_check).max()
                if score < best_score:
                    best_score = score
                    best_a = a_check.copy()
        
        a = softmax_fn(z)
        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE-Adam(sigmoid) stage {si}/{n_stages} T={stage_T:.1f}: best={best_score:.8f} ({elapsed:.0f}s)")
    
    return best_a, best_score


def lse_adam_simplex(a0, time_limit, T_start=50.0, T_end=1.0, n_stages=15, lr=0.005):

    """
    n = len(a0)

    s = a.sum()

        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        iters = max(30, min(300, int(remaining * 1000 / (n_stages - si))))
        
        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break
            
            obj, grad = lse_value_grad(a, stage_T)
            
            # Project gradient onto tangent space of simplex (sum = 0)
            grad_tangent = grad - np.mean(grad)
            
            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad_tangent
            v = beta2 * v + (1 - beta2) * grad_tangent ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)
            
            lr_t = lr / (1 + 0.01 * it)
            z = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            a = proj_simplex(z)
            
            if it % 20 == 0:

        if score < best_score:

def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a_fixed):
    """Build the (2n-1) x n convolution matrix T where (T@b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0. Returns (b, M)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver):
    """Nesterov-accelerated LP best-response for the zero-sum game."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    avg_a = np.zeros(n)
    n_avg = 0
    
    t = 0
    while time.monotonic() - t0 < time_limit:
        t += 1
        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue
        
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s > 1e-15:
            y_new = y_new / s
        
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new / s
        
        y = y_new
        a = a_new
        
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        avg_a += a
        n_avg += 1
        if n_avg >= 10 and n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
        
        if t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP-Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score
