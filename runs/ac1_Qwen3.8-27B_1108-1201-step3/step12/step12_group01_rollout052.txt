                print(f"      NLP restart at t={t}, best={best_score:.6f}, lps={lp_count}")

        if t % 3000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}, lps={lp_count}")

    return best_a, best_score


def compute_topk_grad(a, T_val, topk):
    """
    Compute top-k LSE objective and gradient w.r.t. a.
    obj = (1/T)*log(sum_{k in top_k} exp(T*c_k)) where c = autocorr(a)
    """
    n = len(a)
    c = fast_autocorr(a)
    k_use = min(topk, len(c))
    top_idx = np.argsort(c)[::-1][:k_use]
    top_vals = c[top_idx]

    Tc = T_val * top_vals
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    lse_val = (max_Tc + np.log(exp_Tc.sum())) / T_val
    w = exp_Tc / exp_Tc.sum()

    # grad_a[i] = 2 * sum_K w_K * a[K-i] for valid K-i
    grad_a = np.zeros(n)
    for idx in range(k_use):
        K = top_idx[idx]
        wK = w[idx]
        js = max(0, K - (n - 1))
        je = min(n - 1, K)
        if js <= je:
            idx_range = np.arange(js, je + 1)
            grad_a[idx_range] += 2.0 * wK * a[K - idx_range]

    return lse_val, grad_a


def topk_lse_grad_z(z, T_val, topk):
    """Objective and gradient in z-space for top-k LSE."""
    a = softmax_fn(z)
    n = len(a)
    obj, grad_a = compute_topk_grad(a, T_val, topk)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def adam_topk_lse(a0, time_limit, T_start=100.0, T_end=0.2, n_stages=30, topk=30):
    """
    Adam-optimized top-k LSE with multi-stage annealing and adaptive topk.
    Uses larger topk for better global shape capture.
    """
    n = len(a0)
    a_init = normalize_a(a0)
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        # Learning rate: decrease over stages
        base_lr = 3e-3 * min(1.0, 100.0 / n)
        lr = base_lr * (0.85 ** si)

        # Increase topk as temperature decreases (focus narrows)
        current_topk = max(5, int(topk * (0.5 + 0.5 * (T_end / max(stage_T, T_end)))))

        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.8, 0.999, 1e-8

        # Adapt max iterations to remaining time
        max_it = max(20, int(min(remaining, 60.0) * 5))

        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 3:
                break

            obj, grad_z = topk_lse_grad_z(z, stage_T, current_topk)

            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            if (it + 1) % 5 == 0:
                a = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    # Reset to prevent drift
                    if (it + 1) % 30 == 0:
                        z = np.log(np.maximum(best_a, 1e-30))
                        m_z = np.zeros(n)
                        v_z = np.zeros(n)

        # Stage-end evaluation with symmetrization
        a_stage = softmax_fn(z)
        a_sym = normalize_a(symmetrize(a_stage))
        for cand in [a_stage, a_sym]:
            score_stage = 2.0 * n * fast_autocorr(cand).max()
            if score_stage < best_score:
                best_score = score_stage
                best_a = cand.copy()

        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE stage {si}/{n_stages} (T={stage_T:.1f}, topk={current_topk}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def peak_shaving_systematic(a0, time_limit, seed=42):
    """
    Systematic peak shaving with damage estimation.
    For each peak, find the worst contributing pair and try to reduce it
    by moving mass to positions with minimal damage to other peaks.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize_a(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 3:
        c = fast_autocorr(a)
        bmax = c.max()
        avg_c = c.mean()

        # Find significant peaks (above 95% of max)
        threshold = bmax * 0.95
        top_peaks = np.where(c >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(c)[::-1][:5]
        top_peaks = top_peaks[np.argsort(c[top_peaks])[::-1]][:15]

        improved = False
        for k in top_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue

            # Contribution from each pair
            contrib = a[idx] * a[k - idx]

            # Compute danger zone for all top peaks
            danger = np.zeros(n, dtype=bool)
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                danger[js2:je2 + 1] = True

            # Find the worst contributing source
            for _ in range(3):  # Try top 3 contributing sources
                src_rank = np.argsort(contrib)[::-1]
                if len(src_rank) == 0:
                    break
                src = idx[src_rank[0]]
                if a[src] < 1e-10:
                    break

                # Find safe destinations (not in danger zone, low value)
                safe = np.where(~danger)[0]
                safe = safe[safe != src]
                if len(safe) == 0:
                    safe = np.where(np.arange(n) != src)[0]

                # Score destinations by (a[dst] + estimated damage)
                # Estimated damage: how much would moving to dst increase other peaks?
                dst_scores = np.zeros(len(safe))
                for di, dst in enumerate(safe):
                    # Damage estimate: sum over peaks of a[src_part] * a[dst] contribution
                    dmg = 0.0
                    for k2 in top_peaks[:8]:
                        if 0 <= k2 - dst <= n - 1:
                            partner = k2 - dst
                            if 0 <= partner < n:
                                dmg += a[partner]
                    dst_scores[di] = a[dst] + 0.1 * dmg

                safe_sorted = safe[np.argsort(dst_scores)]
                dst_pool = safe_sorted[:max(5, len(safe_sorted) // 4)]
                if len(dst_pool) == 0:
                    dst_pool = safe[np.argsort(a[safe])][:5]
                if len(dst_pool) == 0:
                    break

                # Try multiple (dst, frac) combinations
                found = False
                for dst in dst_pool[:8]:
                    for frac in [0.002, 0.005, 0.01, 0.02, 0.04, 0.06, 0.1]:
                        amt = a[src] * frac
                        if amt < 1e-14:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < -1e-12:
                            continue
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_autocorr(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            found = True
                            break
                    if found:
                        break

                if not found:
                    # Remove this source from consideration for next iteration
                    contrib[src_rank[0]] = -1
                    if contrib.max() < 1e-14:
                        break
                if found:
                    break

            if improved:
                break

        if not improved:
            stall += 1
            # Random escape
            n_pert = rng.integers(2, 6)
            a_trial = a.copy()
            for _ in range(n_pert):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a_trial[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.03)
                    amt = a_trial[i] * frac
                    a_trial[i] -= amt
                    a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 1e-10:
                a_trial = a_trial / st
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0

            if stall > 200:
                # Try symmetrization
                a_sym = normalize_a(symmetrize(a))
                score_sym = 2.0 * n * fast_autocorr(a_sym).max()
                if score_sym < best_score:
                    a = a_sym
                    best_score = score_sym
                    best_a = a.copy()
                    stall = 0
                elif stall > 400:
                    break

    return best_a, best_score


def spectral_refinement(a0, time_limit, seed=99):
    """
    Spectral-domain refinement: iteratively adjust the FFT amplitude to flatten
    the autocorrelation, then project back to non-negative time domain.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize_a(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    # FFT size
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    # Get current spectrum
    A = rfft(a, fft_size)
    C = A * A  # |A|^2 in real FFT form

    n_iters = int(time_limit * 20)  # Rough estimate of iterations possible
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit - 2:
            break

        # Compute current autocorrelation
        c = irfft(C, fft_size)[:size]
        cmax = c.max()

        # Compute a "flatness target": we want c to be as flat as possible
        # Weight the spectrum to reduce peaks in c
        # Use softmax-like weighting on c to smooth it
        T_smooth = 50.0 + 5000.0 / (it + 1)
        c_clipped = np.maximum(c, 0)
        w = np.exp(-c_clipped * T_smooth)
        w = w / w.sum()

        # The target spectrum is proportional to 1/w (inverse of the time-domain "importance")
        # But we need to be careful: we want C_new such that IFFT(C_new) is flatter
        # Simple approach: C_new = C * (avg_c / c)^alpha, clamped
        alpha = 0.3 + 0.7 * min(1.0, it / 100)
        c_safe = np.maximum(c, cmax * 1e-3)
        C_target = C * (c.mean() / c_safe) ** alpha

        # Get new amplitude spectrum
        # C = A * A (for real FFT), so A_new = sqrt(C_target / A) * A = A * sqrt(C_target / C)
        C_safe = np.maximum(C, 1e-30)
        ratio = np.sqrt(C_target / C_safe)
        A_new = A * ratio

        # Inverse FFT
        a_new = irfft(A_new, fft_size)[:n]

        # Project to non-negative and renormalize
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        # Try both a_new and its symmetrized version
        for cand in [a_new, normalize_a(symmetrize(a_new))]:
            score = 2.0 * n * fast_autocorr(cand).max()
            if score < best_score:
                best_score = score
                best_a = cand.copy()
                # Update A to track improvement
                A = rfft(cand, fft_size)
                C = A * A

            if time.monotonic() - t0 > time_limit - 2:
                break

        if (it + 1) % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Spectral it={it+1} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 10)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 3
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 2
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.7 + 0.2 * np.cos(2 * x) + 0.1 * np.cos(5 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * (x - 0.3) ** 2) + 0.5 * np.exp(-5 * (x - 0.7) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * np.cos(np.pi * x / 3) ** 2
    elif choice == 7:
        # Sine with endpoint suppression
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2 * (1 - 0.5 * np.cos(2 * np.pi * x))
        a = np.maximum(a, 0)
    elif choice == 8:
        # Power law
        x = np.linspace(0, 1, n)
        a = x ** 0.5 * (1 - x) ** 0.3
    else:
        # Multi-scale
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) + 0.3 * np.sin(3 * np.pi * x) + 0.1 * np.sin(7 * np.pi * x)
        a = np.maximum(a, 0)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def perturbed_init(a0, rng, scale=0.1):
    """Create a perturbed version of a sequence for multi-start."""
    a = normalize_a(a0)
    noise = rng.normal(0, scale, len(a))
    a_new = a + noise * (a + 1e-6)
    a_new = np.maximum(a_new, 0)
    sn = a_new.sum()
    if sn < 1e-15:
        return a.copy()
    return a_new / sn


def propose_candidate(seed=42, time_budget=1800.0):
    """
    Multi-phase optimization combining LP Nesterov, LSE gradient,
    systematic peak shaving, spectral refinement, and multi-length search.
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(time_budget) - 90.0, 1710.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    # Get parent sequence
    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = normalize_a(a)
        if a.sum() < 0.01:
            return
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    n_base = len(best_seq) if best_seq is not None else 1600

    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # === Phase 1: LP Nesterov at base length (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} ({remaining:.0f}s left) ===")
    t1 = min(remaining * 0.35, 550.0)
    if t1 > 50:
        a_out, _ = nesterov_lp_optimize(best_seq, t1, solver)
        try_update(a_out, f"lp_n{n_base}")
        # Also try symmetrized version
        try_update(symmetrize(a_out), f"lp_sym_n{n_base}")

    # === Phase 2: LSE gradient with large topk ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 2: LSE large-topk n={n_base} ({remaining:.0f}s left) ===")
    t2 = min(remaining * 0.20, 250.0)
    if t2 > 20:
        a_out, _ = adam_topk_lse(best_seq, t2, T_start=150.0, T_end=0.1, n_stages=35, topk=40)
        try_update(a_out, f"lse_large_n{n_base}")
        try_update(symmetrize(a_out), f"lse_sym_n{n_base}")

    # === Phase 3: Systematic peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 3: Peak shaving n={n_base} ({remaining:.0f}s left) ===")
    t3 = min(remaining * 0.15, 180.0)
    if t3 > 15:
        a_out, _ = peak_shaving_systematic(best_seq, t3, seed=456)
        try_update(a_out, f"peaks_n{n_base}")

    # === Phase 4: Spectral refinement ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 4: Spectral refinement n={n_base} ({remaining:.0f}s left) ===")
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 15:
        a_out, _ = spectral_refinement(best_seq, t4, seed=77)
        try_update(a_out, f"spectral_n{n_base}")

    # === Phase 5: LP Nesterov at slightly different length ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 60:
        n_long = n_base + 200
        print(f"=== Phase 5: LP Nesterov n={n_long} ({remaining:.0f}s left) ===")
        t5 = min(remaining * 0.20, 200.0)
        if t5 > 40:
            init_long = interpolate_sequence(best_seq, n_long)
            a_out, _ = nesterov_lp_optimize(init_long, t5, solver)
            try_update(a_out, f"lp_n{n_long}")

    # === Phase 6: LSE at longer length ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 40:
        n_long2 = n_base + 400
        t6 = min(remaining *
