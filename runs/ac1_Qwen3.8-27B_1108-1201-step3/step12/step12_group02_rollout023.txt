```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adam optimizer on LSE-smoothed objective in log-space (FFT-based gradients)
2. Multi-phase temperature annealing
3. Direct subgradient with restarts
4. Multi-length exploration (1600, 2000, 2400)
5. Coordinate descent polish
Key: Adam with per-coordinate adaptive learning rates converges faster than
L-BFGS or mirror descent for this high-dimensional non-smooth problem.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

# Top-level helper functions (no closures, no lambdas)

def fast_conv(a):
    """Linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_score(a):
    """Compute 2n*max(b)/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return 2.0 * n * b.max() / (s * s)

def grad_lse(a, T):
    """
    Compute gradient of (1/T)*logsumexp(T*b) with respect to a.
    Returns array of same length as a.
    """
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_shift = Tb - Tb.max()
    exp_Tb = np.exp(Tb_shift)
    w = exp_Tb / exp_Tb.sum()  # weights, length 2n-1
    
    # g_j = 2 * sum_m a_m * w_{j+m}
    # = 2 * cross_corr(a, w)_j
    # Using the formula: cross_corr(j) = convolve(a_rev, w)[n-1+j]
    a_rev = a[::-1]
    conv_len = n + (2*n - 1) - 1  # = 3n - 2
    fft_size = 1 << (conv_len - 1).bit_length()
    A = rfft(a_rev, fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A * W, fft_size)
    g = 2.0 * conv[n-1 : n-1+n]
    return g

def grad_max(a):
    """Compute subgradient of max(b) with respect to a."""
    n = len(a)
    b = fast_conv(a)
    max_idx = np.argmax(b)
    # g_j = 2 * a_{max_idx - j} if 0 <= max_idx - j < n
    g = np.zeros(n)
    j_start = max(0, max_idx - n + 1)
    j_end = min(n-1, max_idx)
    js = np.arange(j_start, j_end + 1)
    g[js] = 2.0 * a[max_idx - js]
    return g

def softmax_grad_transform(g, a):
    """Transform gradient from a-space to z-space (where a = softmax(z))."""
    # dz/d a: grad_z = a * (grad_a - dot(grad_a, a))
    S = np.dot(g, a)
    return a * (g - S)

def adam_optimize(a_init, time_limit, T, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8, verbose=True):
    """
    Adam optimizer on LSE-smoothed objective in log-space.
    a = softmax(z), minimize (2n/T)*logsumexp(T*b(a))
    """
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 1e-30)
    a = a / a.sum()
    z = np.log(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        g_a = grad_lse(a, T)
        g_z = softmax_grad_transform(g_a, a)
        
        m = beta1 * m + (1 - beta1) * g_z
        v = beta2 * v + (1 - beta2) * g_z * g_z
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        # Numerical stability: shift
        z = z - z.max()
        a = np.exp(z)
        a = a / a.sum()
        
        if t % 20 == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                if verbose and t % 200 == 0:
                    elapsed = time.monotonic() - t0
                    print(f"      Adam(T={T:.1f}) t={t} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score

def subgrad_restart(a_init, time_limit, lr0=0.001, n_restarts=5, verbose=True):

    Direct subgradient on max(b) with random restarts.
    """
    n = len(a_init)

        while time.monotonic() - t0 < time_limit:
            t += 1
            g = grad_max(a)
            lr = lr0 / (1.0 + t / 100.0)
            a = a - lr * g
            a = np.maximum(a, 0)
            s = a.sum()
            if s < 1e-15:

                a = np.maximum(a, 0)
                a = a / a.sum()
            else:

            if t % 10 == 0:
                sc = fast_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

        if restart < n_restarts - 1:
            a = best_a + 0.02 * rng.standard_normal(n)
            a = np.maximum(a, 0)

def coord_descent(a_init, time_limit, verbose=True):
    """
    Coordinate descent: try moving mass from high-contribution positions
    to low-contribution positions to reduce the max autocorrelation.
    """
    n = len(a_init)
    rng = np.random.default_rng(456)
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_conv(a)
        max_k = int(np.argmax(b))
        
        # Find positions contributing most to b[max_k]
        js = max(0, max_k - n + 1)
        je = min(n - 1, max_k)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[max_k - idx]
        
        if contrib.sum() < 1e-15:
            break
        
        # Source: highest contributor
        src = int(idx[np.argmax(contrib)])
        
        # Destination: random position far from peak
        peak_mask = np.zeros(n, dtype=bool)
        for kk in np.argsort(b)[::-1][:5]:
            js2 = max(0, int(kk) - n + 1)
            je2 = min(n - 1, int(kk))
            peak_mask[js2:je2+1] = True
        
        candidates = np.where(~peak_mask)[0]
        if len(candidates) < 3:
            candidates = np.where(np.arange(n) != src)[0]
        
        # Try moving small amounts to random destinations
        improved = False
        for _ in range(20):
            dst = int(candidates[rng.integers(len(candidates))])
            for frac in [0.001, 0.003, 0.01, 0.03]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                sc = fast_score(a_trial)
                if sc < best_score:
                    a = a_trial.copy()
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall > 200:
                break
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.01)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    sc = fast_score(a_trial)
                    if sc < best_score:
                        a = a_trial.copy()
                        best_score = sc
                        best_a = a.copy()
                        stall = 0
        else:
            stall = 0
    
    return best_a, best_score

def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
```
