import numpy as np

from scipy.fft import rfft, irfft

_global_solver = None

def get_solver():

    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix_rows(a, k_range):
    """Build rows of the convolution matrix for specific lag indices."""
    n = len(a)
    rows = []
    for k in k_range:
        row = np.zeros(n)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        row[idx] = a[k - idx]
        rows.append(row)
    return np.array(rows)

def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def nesterov_lp(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        # Polyak-Ruppert averaging (last K/2 iterates)
        avg_a += a
        n_avg += 1
        K = n_avg

        if verbose and n_iters % 200 == 0:

def lse_gradient(a, w, T, n):
    """Compute gradient of T*log(sum_k exp(b_k/T)) where b = autocorr(a)."""
    # grad[i] = (2/T) * sum_k w[k] * (a[k-i] + a[i-k])
    # Use FFT

def lp_best_response(a, M, solver):

def proj_sum_nonneg(v, S):
    """Project v onto {x >= 0, sum(x) = S} via bisection on water level."""
    n = len(v)
    if S <= 0:
        return np.zeros(n)
    lo = v.min() - S / n - 1.0
    hi = v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5

            lo = mid

    result = np.maximum(v - lam, 0.0)
    s = result.sum()
    if s > 1e-15:

    else:

def compute_lse_gradient(a, w, T, n):
    """Compute gradient of LSE approximation. grad[i] = (2/T)*sum_k w[k]*(a[k-i]+a[i-k])."""
    N = 1 << (3 * n - 1).bit_length()
    A = rfft(a, N)
    W = rfft(w, N)
    conv = irfft(W * A, N)
    corr = irfft(W * np.conj(A), N)
    term1 = corr[:n]
    term2 = conv[:n]
    return (2.0 / T) * (term1 + term2)

def peak_shave(a0, time_limit, seed=42):
    """Spectral peak shaving: reduce mass at positions contributing to top peaks."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:8]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            src = idx[np.argmax(contrib)]

            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]

            candidates_sorted = candidates[np.argsort(a[candidates])][:max(1, len(candidates) // 5)]
            dst = candidates_sorted[rng.integers(len(candidates_sorted))]

            for frac in [0.001, 0.003, 0.008, 0.015, 0.03, 0.06]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-10:
                frac = rng.uniform(0.01, 0.08)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0

            if stall > 500:
                break

    return best_a, best_score


def coordinate_descent_refine(a0, time_limit, seed=123):
    """Local coordinate descent: try pairs of positions for mass transfer."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        k_star = int(np.argmax(b))
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_star - idx]
        src = idx[np.argmax(contrib)]

        for trial in range(50):
            j = rng.integers(n)
            if j == src:
                continue
            for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[j] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
                    break
            if stall == 0 and score < best_score:
                stall = 0
                break

        stall += 1
        if stall > 200:
            break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def structured_init(n, rng):
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-5.0 * (x - 0.3) ** 2) + 0.6 * np.exp(-5.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    else:
        a = np.ones(n)
        a = a * (1.0 + 0.2 * np.cos(2 * np.pi * 3 * np.linspace(0, 1, n)))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LP Nesterov on base length
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.28, 450.0)
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n_base}")

    # Phase 2: FISTA-LSE refinement on base length
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: FISTA-LSE n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 15:
        n_b = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_b) / best_seq.sum()
        a_out, _ = fista_lse(a_norm, t2, T_init=0.5, T_end=0.02, verbose=True)
        do_try_update(a_out, f"fista_n{n_b}")

    # Phase 3: LP Nesterov at longer length
    n2 = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LP Nesterov n={n2} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.22, 350.0)
    if t3 > 30:
        init_long = interpolate_sequence(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = nesterov_lp(a_norm, t3, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n2}")

    # Phase 4: FISTA-LSE at longer length
    remaining = time_budget - (time.monotonic() - t_global)
    n_b = len(best_seq)
    print(f"\n=== Phase 4: FISTA-LSE n={n_b} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.12, 180.0)
    if t4 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_b) / best_seq.sum()
        a_out, _ = fista_lse(a_norm, t4, T_init=0.3, T_end=0.01, verbose=False)
        do_try_update(a_out, f"fista2_n{n_b}")

    # Phase 5: LP Nesterov at even longer length
    n3 = n_base + 800
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: LP Nesterov n={n3} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.18, 250.0)
    if t5 > 30:
        init_long3 = interpolate_sequence(best_seq, n3)
        a_norm = init_long3 * np.sqrt(2.0 * n3) / init_long3.sum()
        a_out, _ = nesterov_lp(a_norm, t5, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n3}")

    # Phase 6: Peak shaving on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_b = len(best_seq)
    print(f"\n=== Phase 6: Peak shaving n={n_b} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.15, 150.0)
    if t6 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_b) / best_seq.sum()
        a_out, _ = peak_shave(a_norm, t6, seed=777)
        do_try_update(a_out, "peak_shave")

    # Phase 7: Coordinate descent refine
    remaining = time_budget - (time.monotonic() - t_global)
    n_b = len(best_seq)
    print(f"\n=== Phase 7: Coord descent n={n_b} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.10, 100.0)
    if t7 > 5:
        a_norm = best_seq * np.sqrt(2.0 * n_b) / best_seq.sum()
        a_out, _ = coordinate_descent_refine(a_norm, t7, seed=456)
        do_try_update(a_out, "coord_desc")

    # Phase 8: Final FISTA-LSE polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_b = len(best_seq)
    print(f"\n=== Phase 8: Final FISTA polish n={n_b} (budget={remaining:.0f}s) ===")
    t8 = min(remaining * 0.20, 120.0)
    if t8 > 5:
        a_norm = best_seq * np.sqrt(2.0 * n_b) / best_seq.sum()
        a_out, _ = fista_lse(a_norm, t8, T_init=0.1, T_end=0.005, verbose=False)
        do_try_update(a_out, "fista_polish")

    # Phase 9: Final peak shave
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        n_b = len(best_seq)
        t9 = min(remaining * 0.4, 60.0)
        if t9 > 5:
            a_norm = best_seq * np.sqrt(2.0 * n_b) / best_seq.sum()
            a_out, _ = peak_shave(a_norm, t9, seed=999)
            do_try_update(a_out, "peak_shave_2")

    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
