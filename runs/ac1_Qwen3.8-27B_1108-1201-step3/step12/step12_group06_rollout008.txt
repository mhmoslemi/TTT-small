import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp


def fast_convolve(a):
    """Fast linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score assuming sum(a)=1: returns 2*n*max(b)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_from_z(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Inverse of softmax (log)."""
    return np.log(np.maximum(a, 1e-300))


def build_conv_matrix(a_fixed):
    """Build the convolution matrix T where (T@b)[k] = (a_fixed*b)[k]."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def get_lp_solver():
    """Find the best available LP solver."""
    for solver_name in ['PDLP', 'ECOS', 'SCS', 'CLARABEL']:
        try:
            prob = cp.Problem(cp.Minimize(cp.sum(cp.Variable(5, nonneg=True))), [])
            prob.solve(solver=solver_name)
            return solver_name
        except Exception:
            continue
    return 'ECOS'


def solve_lp_best_response_fast(a_fixed):
    """Solve the LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0.
    Uses only the top constraints for speed."""
    n = len(a_fixed)
    size = 2 * n - 1
    
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Use FFT-based constraint generation: start with all constraints,
    # but for large n, the LP is still tractable
    T = build_conv_matrix(a_fixed)
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    for solver in ['PDLP', 'CLARABEL', 'ECOS']:
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                break
        except Exception:
            continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_game(a0, time_limit, max_iters=100000):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_lp = solve_lp_best_response_fast(y)
        if b_new is None:
            continue

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * (it + 1) * (it + 1))) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NG iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} M_lp={M_lp:.6f}")

    return best_a, best_score


def lse_obj_and_grad(z, T, n):
    """Compute LogSumExp objective and gradient w.r.t. z (log-softmax params)."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        exp_Tb = np.ones_like(Tb)
    else:
        exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp

    # Gradient: d/da [2n/T * log(sum exp(T*conv[a][k]))]
    # = 2n * sum_k w[k] * d(conv[a][k])/da
    # d(conv[a][k])/da[j] = a[k-j] (for valid j)
    grad_a = np.zeros(n)
    for k in range(len(b)):
        wk = w[k]
        if wk < 1e-18:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad_a[idx] += wk * a[k - idx]
    grad_a *= 2.0 * n

    # Transform to z-space: grad_z = a * (grad_a - sum(grad_a * a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_optimize(a0, time_limit, T_init=50.0, T_min=0.5, max_stages=30):
    """L-BFGS-B optimization of LSE objective with adaptive temperature."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    
    T = T_init
    prev_best = best_score
    
    for stage in range(max_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func(z_val):
            obj, _ = lse_obj_and_grad(z_val, T, n)
            return obj

        def jac(z_val):
            _, grad = lse_obj_and_grad(z_val, T, n)
            return grad

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': 50, 'ftol': 1e-18,
                         'gtol': 1e-14, 'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)

            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()

            # Adaptive T: based on improvement
            improvement = prev_best - best_score
            if improvement > 1e-6:
                T = max(T_min, T * 0.6)
            elif improvement > 0:
                T = max(T_min, T * 0.8)
            else:
                T = min(T_init, T * 1.3)
            
            prev_best = best_score

            if (stage + 1) % 5 == 0:
                print(f"    LSE stage={stage+1} T={T:.2f}: score={sc:.8f} best={best_score:.8f}")

        except Exception:
            T = max(T_min, T * 0.8)

    return best_a, best_score


def subgradient_polish(a0, time_limit, max_steps=1000000):

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    # Polyak: need lower bound. Use 2n/(2n-1) as theoretical bound for this problem
    # Actually the true lower bound is harder to compute. Use best_score * (1-eps) conservatively
    lb = 2.0  # Very conservative lower bound (uniform gives 2)
    
    # Actually let's use a tighter bound: for any a with sum=1, 
    # max(conv(a,a)) >= sum(a^2) >= 1/n (by Cauchy)
    # So score >= 2n * 1/n = 2. That's the same bound.
    # Better: max(conv(a,a)) >= (sum a)^2 / (number of terms) ... no that's not right.
    # Let's just use best_score * 0.99 as a practical lower bound for Polyak
    
    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b

        # Find active peaks
        threshold = max_b * 0.95
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        if len(active) > 30:
            top_idx = np.argsort(b[active])[-30:]
            active = active[top_idx]

        # Subgradient
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        # Project to tangent space
        S = np.dot(g, a)
        grad_z = a * (g - S)

        grad_norm = np.linalg.norm(grad_z)
        if grad_norm < 1e-15:
            break

        # Polyak step
        # Use a running lower bound estimate
        lb_est = best_score * 0.995  # conservative
        gap = cur_score - lb_est
        if gap < 1e-12:
            break
        eta = gap / (grad_norm * grad_norm + 1e-30)
        eta = min(eta, 1.0)

        z_new = z - eta * grad_z
        a_new = softmax_from_z(z_new)

def parabolic_seq(n, c):
    """Parabolic (raised-cosine like) sequence of length n with depth c."""
    x = np.linspace(-1, 1, n)
    a = 1.0 - c * x**2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def raised_cosine_seq(n, beta):
    """Raised cosine: a[j] = 0.5*(1+cos(pi*(2j-(n-1))/(n-1) * beta))"""
    x = np.linspace(-beta, beta, n)
    a = 0.5 * (1 + np.cos(np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def gaussian_seq(n, sigma):
    """Gaussian window."""
    x = np.linspace(-1, 1, n)
    a = np.exp(-x**2 / (2 * sigma**2))
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def trimmed_convolve(a):
    """Compute only the central portion of the convolution (where max likely is)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    full = irfft(A * A, fft_size)[:size]
    return full
