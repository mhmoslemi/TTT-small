```python
def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)

def fast_score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)

def spectral_flatten(a, alpha=0.5, n_iter=20):
    """Iteratively flatten the power spectrum to reduce autocorrelation peaks."""
    a = np.asarray(a, dtype=np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    fft_size = 1 << (2*n - 2).bit_length()
    
    for _ in range(n_iter):
        A = rfft(a, fft_size)
        P = np.abs(A)**2
        # Flatten: compress dynamic range
        P_flat = P**alpha
        # Preserve phase
        phase = A / np.where(np.abs(A) > 1e-30, np.abs(A), 1)

        if s_new < 1e-15:

        a_new = a_new / s_new
        # Check if improved

            a = a_new
        else:
            # Try with smaller alpha
            break
    
    return a
```

```python
    # g_m = sum_k w_k * a_{m+k} + sum_k w_k * a_{m-k}
    # First term: cross-correlation of w and a
    # FFT of cross-correlation: conj(W) * A
    # Second term: convolution of w and a  
    # FFT of convolution: W * A
    # Since w is symmetric (real), W is real, so conj(W) = W
    # Thus g = 2 * W * A in frequency domain... wait no.
    
    # Actually for real symmetric w, the DFT is real.
    # Cross-correlation: ifft(conj(W) * A) = ifft(W * A) (since W is real)
    # Convolution: ifft(W * A)
    # So both terms give the same result!
    # g = 2 * ifft(W * A)
    
    G = 2.0 * W * A
    g = irfft(G, size)[:n]
    return g
```

Wait, I need to be more careful. Let me define things precisely.

Let w be a sequence of length L = 2n-1, and a be a sequence of length n.

Cross-correlation: `(w ⋆ a)[m] = sum_k w[k] * a[k+m]` (sum over valid k)
Convolution: `(w * a)[m] = sum_k w[k] * a[m-k]` (sum over valid k)

In frequency domain (with zero-padding to size S >= L + n - 1):
- `FFT(w ⋆ a) = conj(FFT(w)) * FFT(a)`
- `FFT(w * a) = FFT(w) * FFT(a)`

For real symmetric w (w[k] = w[L-1-k]):
- `FFT(w)` is real (the DFT of a real even sequence is real and even)

Wait, w is not exactly "even" in the DFT sense. w is a sequence of length L=2n-1 that is palindromic: w[0]=w[L-1], w[1]=w[L-2], etc.

For a palindromic real sequence of odd length L=2n-1, its DFT satisfies: X[k] = X[-k mod S] (i.e., X is real). So yes, FFT(w) is real.

Therefore: `FFT(w ⋆ a) = FFT(w * a) = FFT(w) * FFT(a)` (since conj(FFT(w)) = FFT(w) when FFT(w) is real).

So both the cross-correlation and convolution give the same result! And the subgradient is:
`g = (w ⋆ a) + (w * a) = 2 * (w * a)`

Great, so: `g = 2 * ifft(FFT(w) * FFT(a))[:n]` (with appropriate zero-padding).

Let me verify this makes sense. For a delta function at position p (a[p]=1, rest 0):
- `b_k = a_k * a` has b_k = a[k+p] (shifted a)
- Subgradient should be `g = 2 * w convolved with delta at p = 2 * w shifted by p`
- Using FFT: `g = 2 * ifft(FFT(w) * FFT(a))` where FFT(a) = exp(-2πi p k / S), so `g[m] = 2 * w[m-p]` (approximately, with boundary effects)

That seems right.

OK so now I have an efficient way to compute the full subgradient in O(n log n). Let me use this for a Nesterov-accelerated method.

**Nesterov Accelerated Multiplicative Weights:**

Standard Nesterov for convex minimization:
- y_k = x_k + (t_k-1)/t_{k+1} * (x_k - x_{k-1})
- x_{k+1} = proj(f(y_k - eta_k * grad(f, y_k)))
- t_{k+1} = (1 + sqrt(1 + 4*t_k^2)) / 2

But for multiplicative weights on the simplex, the "projection" is the mirror map (exponential + normalization).

Let me adapt:
- y = a + momentum * (a - a_prev)  (extrapolation)
- Compute subgradient at y
- a_new = y * exp(-eta * grad) / sum
- Update momentum

Actually, Nesterov acceleration doesn't directly apply to mirror descent in the same way. Let me use a simpler momentum approach:

```
v = beta * v + (1-beta) * grad
a_new = a * exp(-eta * v) / sum
```

This is stochastic/mirror descent with momentum (similar to Adam but for multiplicative updates).

```python
import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)

def full_subgradient(a, beta):
    """Compute softmax-weighted full subgradient of max autocorrelation."""
    n = len(a)
    b = fast_autocorr(a)
    # Softmax weights
    b_shift = b - b.max()
    w = np.exp(beta * b_shift) / np.exp(beta * b_shift).sum()
    # Compute g = 2 * (w * a) via FFT
    conv_len = len(w) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    g = 2.0 * irfft(W * A, fft_size)[:n]
    return g, w, b

def accelerated_mirror_descent(a0, time_limit, beta_start=5.0, beta_end=50.0, 
                                eta0=0.1, momentum=0.9, seed=42):
    """Accelerated multiplicative weights with full softmax subgradient."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / max(s, 1e-15)
    
    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    
    v = np.zeros(n)  # momentum
    t = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        progress = min(1.0, t / 5000)
        beta = beta_start + (beta_end - beta_start) * progress
        eta = eta0 / (1 + 0.01 * t)
        
        g, w, b = full_subgradient(a, beta)
        
        # Momentum update
        v = momentum * v + (1 - momentum) * g
        
        # Multiplicative update
        a_new = a * np.exp(-eta * v)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a + 0.01 * rng.standard_normal(n)
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
        else:
            a_new = a_new / s_new
        
        a = a_new
        
        # Check score
        if t % 10 == 1:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        # Periodic restart with noise
        if t % 500 == 0:
            sc = fast_score(a)
            if sc > best_score * 1.01:
                a = best_a + 0.01 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
                v = np.zeros(n)
    
    return best_a, best_score
```

```python
import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, minimize_scalar
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)

def full_softmax_subgrad(a, beta):
    """Compute the full softmax-weighted subgradient."""
    n = len(a)
    b = fast_autocorr(a)
    b_shift = b - b.max()
    exp_b = np.exp(beta * b_shift)
    w = exp_b / exp_b.sum()
    # g = 2 * (w * a) via FFT (linear convolution)
    conv_len = len(w) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    g = 2.0 * irfft(W * A, fft_size)[:n]
    return g, b

def lse_smoothed_obj_grad(a, T):
    """LogSumExp smoothed objective and its gradient."""
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    Tb_shift = Tb - Tb.max()
    exp_Tb = np.exp(Tb_shift)
    logsumexp = Tb.max() + np.log(exp_Tb.sum())
    obj = logsumexp / T
    
    w = exp_Tb / exp_Tb.sum()
    # Gradient: (1/T) * 2 * (w * a)
    conv_len = len(w) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad = (2.0 / T) * irfft(W * A, fft_size)[:n]
    
    return obj, grad

def softmax_md(a0, time_limit, beta_start=3.0, beta_end=80.0,
               eta0=0.08, momentum_decay=0.95, seed=42, verbose=True):
    """
    Multiplicative weights with full softmax subgradient and momentum.
    Key improvements over previous:
    - Full softmax subgradient (not just top-k)
    - Momentum for faster convergence
    - Adaptive eta based on gradient norm
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / max(s, 1e-15)
    
    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    v = np.zeros(n)
    t = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        progress = min(1.0, t / 3000)
        # Anneal beta: start low (smooth), end high (sharp)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        g, b = full_softmax_subgrad(a, beta)
        g_norm = np.linalg.norm(g)
        
        # Adaptive step size: scale by 1/g_norm for stability
        eta = eta0 / (1.0 + 0.005 * t) * (1.0 / max(g_norm, 0.01))
        
        # Momentum
        v = momentum_decay * v + (1 - momentum_decay) * g
        
        # Multiplicative update
        a_new = a * np.exp(-eta * v)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()

            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()

            a_new = a_new / s_new
        
        a = a_new

                best_a = a.copy()
                stall = 0
            else:

            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            stall = 0

def lse_lbfgs(a0, time_limit, T_values=None, seed=42, verbose=True):
    """
    Multi-stage LogSumExp L-BFGS optimization.
    Works in z-space where a = softmax(z).
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    
    if T_values is None:
        T_values = [50, 30, 20, 10, 5, 3, 2, 1, 0.5, 0.3]
    
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()  # center for numerical stability
    t0 = time.monotonic()
    
    best_a = a_init.copy()
    best_score = fast_score(a_init)
    
    def func_and_grad(z_in, T):
        z_shift = z_in - z_in.max()
        a = np.exp(z_shift)
        a = a / a.sum()
        obj, grad_a = lse_smoothed_obj_grad(a, T)
        # Chain rule: grad_z = a * (grad_a - dot(grad_a, a))
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        return obj, grad_z
    
    for Ti, T in enumerate(T_values):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        # Budget per stage
        stage_time = remaining / (len(T_values) - Ti)
        iters = max(20, min(100, int(20 + 80 * Ti / len(T_values))))
        
        def func(z_in):
            val, _ = func_and_grad(z_in, T)
            return val
        
        def jac(z_in):
            _, g = func_and_grad(z_in, T)
            return g
        
        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                          options={'maxiter': iters, 'ftol': 1e-18,
                                   'gtol': 1e-12, 'maxfun': iters * 300,
                                   'disp': False})
            z = res.x
            z_shift = z - z.max()
            a = np.exp(z_shift)
            a = a / a.sum()
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        except Exception as e:
            pass
        
        if verbose and Ti % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {Ti+1}/{len(T_values)} (T={T}, {elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score

def spectral_equalize(a0, n_passes=50, power=0.7, seed=42):
    """
    Try to flatten the power spectrum iteratively.
    Uses Gerchberg-Saxton style iteration.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / max(s, 1e-15)
    
    best_a = a.copy()
    best_score = fast_score(a)
    fft_size = 1 << (2*n - 2).bit_length()
    
    for p in range(n_passes):
        A = rfft(a, fft_size)
        P = np.abs(A)**2
        # Compress dynamic range of power spectrum
        P_norm = P / P.max()
        P_flat = P_norm ** power  # power < 1 compresses
        # Reconstruct with original phase
        phase = A / np.where(np.abs(A) > 1e-30, np.abs(A), 1.0)
        a_new = irfft(np.sqrt(P_flat) * phase, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        sc = fast_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Try with the current a, just check
            break
    
    return best_a, best_score

def coordinate_descent(a0, time_limit, seed=42, verbose=True):
    """
    Fine-grained coordinate descent: move small amounts of mass
    from high-contributing positions to low-contributing ones.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / max(s, 1e-15)
    
    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        # Find the top few peaks
        top_k = np.argsort(b)[::-1][:5]
        
        improved = False
        for k_idx in top_k:
            k_idx = int(k_idx)
            # Find which pair contributes most to this peak
            js = max(0, k_idx - n + 1)
            je = min(n - 1, k_idx)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_idx - idx]
            total_contrib = contrib.sum()
            if total_contrib < 1e-15:
                continue
            
            # The position with highest contribution
            src = int(idx[np.argmax(contrib)])
            
            # Find a good destination: low a value, far from peak region
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k[:3]:
                k2 = int(k2)
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2+1] = True
            
            candidates = np.where(~peak_mask)[0]
            if len(candidates) < 10:
                candidates = np.where(np.arange(n) != src)[0]
            
            # Pick from lowest-a candidates
            cand_sorted = candidates[np.argsort(a[candidates])[:20]]
            dst = int(cand_sorted[rng.integers(len(cand_sorted))])
            
            # Try moving different fractions
            for frac in [0.001, 0.003, 0.01, 0.03, 0.1]:
                amt = min(a[src] * frac, a[src] * 0.5)
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = a_trial / a_trial.sum()
                sc = fast_score(a_trial)
                if sc < best_score:
                    a = a_trial.copy()
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > a[j]:
                frac = rng.uniform(0.001, 0.01)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = a_trial / a_trial.sum()
                sc = fast_score(a_trial)
                if sc < best_score:
                    a = a_trial.copy()
                    best_score = sc
                    best_a = a.copy()
                    stall = 0
            
            if stall > 200:
                break
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    CoordDescent ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Full softmax-weighted subgradient multiplicative weights with momentum
   (O(n log n) per step via FFT, no top-k truncation)
2. Multi-stage LogSumExp L-BFGS with refined temperature schedule
3. Spectral equalization (Gerchberg-Saxton style) as perturbation
4. Coordinate descent polishing
5. Multi-length exploration with interpolation
Key innovations over previous: full (not top-k) softmax subgradient,
momentum acceleration, spectral domain perturbation, and refined
annealing schedules.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_autocorr(a):
    """Full autocorrelation a*a using FFT. Returns array of length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_score(a):
    """Compute 2n*max(b)/(sum(a))^2 quickly."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)


def full_softmax_subgrad(a, beta):
    """Compute full softmax-weighted subgradient via FFT. O(n log n)."""
    n = len(a)
    b = fast_autocorr(a)
    b_shift = b - b.max()
    exp_b = np.exp(beta * b_shift)
    w = exp_b / exp_b.sum()
    # g = 2 * (w * a) via linear convolution
    conv_len = len(w) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    g = 2.0 * irfft(W * A, fft_size)[:n]
    return g, b


def lse_obj_grad_a(a, T):
    """LogSumExp smoothed objective and gradient w.r.t. a."""
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    Tb_shift = Tb - Tb.max()
    exp_Tb = np.exp(Tb_shift)
    logsumexp = Tb.max() + np.log(exp_Tb.sum())
    obj = logsumexp / T
    w = exp_Tb / exp_Tb.sum()
    conv_len = len(w) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad = (2.0 / T) * irfft(W * A, fft_size)[:n]
    return obj, grad

        def func(z_in):
            val, _ = _lse_func_grad(z_in, T, n)
            return val

        def jac(z_in):
            _, g = _lse_func_grad(z_in, T, n)
            return g

        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',

    return best_a, best_score


def _lse_func_grad(z_in, T, n):
    z_shift = z_in - z_in.max()
    a = np.exp(z_shift)
    a = a / a.sum()
    obj, grad_a = lse_obj_grad_a(a, T)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def time_budget_local(total, t0):
    return total - (time.monotonic() - t0)


def spectral_equalize(a0, n_passes=40, power=0.75, verbose=False):
    """Gerchberg-Saxton style spectral flattening."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / max(s, 1e-15)
    best_a = a.copy()
    best_score = fast_score(a)
    fft_size = 1 << (2 * n - 2).bit_length()

    for p in range(n_passes):
        A = rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_max = P.max()
        if P_max < 1e-30:
            break
        P_norm = P / P_max
        P_flat = P_norm ** power
        phase = A / np.where(np.abs(A) > 1e-30, np.abs(A), 1.0)
        a_new = irfft(np.sqrt(P_flat) * phase, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        sc = fast_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Try slightly different power
            P_flat2 = P_norm ** (power + 0.1)
            a_new2 = irfft(np.sqrt(P_flat2) * phase, fft_size)[:n]

            s_new2 = a_new2.sum()
            if s_new2 > 1e-15:

            else:
                break
    if verbose:

def coord_descent(a0, time_limit, seed=42, verbose=True):
    """Coordinate descent: move mass from peak contributors to low areas."""
    n = len(a0)
    rng = np.random.default_rng(seed)

    best_score = fast_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k_idx = np.argsort(b)[::-1][:8]
        improved = False
        for k_idx in top_k_idx:
            k_idx = int(k_idx)
            js = max(0, k_idx - n + 1)
            je = min(n - 1, k_idx)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_idx - idx]
            if contrib.sum() < 1e-15:
                continue
            src = int(idx[np.argmax(contrib)])
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k_idx[:4]:
                k2i = int(k2)
                js2 = max(0, k2i - n + 1)
                je2 = min(n - 1, k2i)
                peak_mask[js2:je2 + 1] = True
            candidates = np.where(~peak_mask)[0]
            if len(candidates) < 10:
                candidates = np.where(np.arange(n) != src)[0]
            cand_sorted = candidates[np.argsort(a[candidates])[:30]]
            dst = int(cand_sorted[rng.integers(len(cand_sorted))])
            for frac in [0.001, 0.003, 0.01, 0.03, 0.08, 0.15]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = a_trial / a_trial.sum()
                sc = fast_score(a_trial)
                if sc < best_score:
                    a = a_trial.copy()
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > a[j] * 2:
                frac = rng.uniform(0.002, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt

def interpolate_to_length(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: full softmax MD + LSE + spectral + coord descent + multi-length.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 45.0, 1755.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = fast_score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n = 1600
        best_seq = diverse_init(n, rng, variant=1)
        best_score = fast_score(best_seq)
        print(f"Default init: n={n}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = fast_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    n = len(best_seq)

    # === Phase 1: Full softmax MD with momentum ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Full Softmax MD with momentum (n={n}) ===")
    t1 = min(remaining * 0.22, 380.0)
    if t1 > 30:
        a_out, _ = softmax_md(best_seq, t1, beta_start=3.0, beta_end=60.0,
                              eta0=0.1, momentum=0.92, seed=1)
        try_update(a_out, "md_full_low")

    remaining = time_budget - (time.monotonic() - t_global)
    t1b = min(remaining * 0.12, 180.0)
    if t1b > 20:
        a_out, _ = softmax_md(best_seq, t1b, beta_start=15.0, beta_end=150.0,
                              eta0=0.05, momentum=0.9, seed=2)
        try_update(a_out, "md_sharp")

    # Diverse init + MD
    remaining = time_budget - (time.monotonic() - t_global)
    t1c = min(remaining * 0.08, 120.0)
    if t1c > 20:
        init_alt = diverse_init(n, rng, variant=1)
        a_out, _ = softmax_md(init_alt, t1c / 2, beta_start=5.0, beta_end=50.0,
                              eta0=0.12, momentum=0.9, seed=3)
        try_update(a_out, "md_u_shape")

    # === Phase 2: LSE L-BFGS ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LSE L-BFGS (n={n}) ===")
    t2 = min(remaining * 0.22, 330.0)
    if t2 > 30:
        T_vals = [30, 20, 12, 8, 5, 3, 2, 1, 0.5, 0.3]
        a_out, _ = lse_lbfgs(best_seq, t2, T_values=T_vals, seed=4)
        try_update(a_out, "lse_main")

    remaining = time_budget - (time.monotonic() - t_global)
    t2b = min(remaining * 0.10, 150.0)
    if t2b > 20:
        T_vals2 = [15, 8, 5, 3, 2, 1, 0.5, 0.3, 0.15]
        a_out, _ = lse_lbfgs(best_seq, t2b, T_values=T_vals2, seed=5)
        try_update(a_out, "lse_low")

    # === Phase 3: Spectral equalization ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Spectral Equalization ===")
    if remaining > 20:
        a_out, _ = spectral_equalize(best_seq, n_passes=50, power=0.7, verbose=True)
        try_update(a_out, "spectral")
        if a_out is not None and fast_score(a_out) < best_score + 0.005:
            remaining = time_budget - (time.monotonic() - t_global)
            t3b = min(remaining * 0.08, 80.0)
            if t3b > 15:
                a_out2, _ = softmax_md(a_out, t3b, beta_start=5.0, beta_end=40.0,
                                       eta0=0.08, momentum=0.92, seed=6)
                try_update(a_out2, "spectral+md")

    # === Phase 4: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Coordinate Descent ===")
    t4 = min(remaining * 0.12, 180.0)
    if t4 > 20:
        a_out, _ = coord_descent(best_seq, t4, seed=10)
        try_update(a_out, "coord")

    # === Phase 5: Try different length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = int(n * 1.3)
    print(f"\n=== Phase 5: Longer length n={n_long} ===")
    if remaining > 60:
        init_long = interpolate_to_length(best_seq, n_long)
        t5a = min(remaining * 0.20, 250.0)
        if t5a > 30:
            a_out, _ = softmax_md(init_long, t5a, beta_start=3.0, beta_end=50.0,
                                  eta0=0.06, momentum=0.9, seed=7)
            try_update(a_out, f"md_n{n_long}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            T_vals3 = [20, 10, 5, 3, 2, 1, 0.5]
            t5b = min(remaining * 0.25, 250.0)
            if t5b > 30:
                init_long2 = interpolate_to_length(best_seq, n_long)
                a_out, _ = lse_lbfgs(init_long2, t5b, T_values=T_vals3, seed=8)
                try_update(a_out, f"lse_n{n_long}")

    # === Phase 6: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Final Polish (remaining={remaining:.0f}s) ===")
    if remaining > 20 and best_seq is not None:
        # Intense MD with high beta
        t6a = min(remaining * 0.30, 80.0)
        if t6a > 10:
            a_out, _ = softmax_md(best_seq, t6a, beta_start=30.0, beta_end=200.0,
                                  eta0=0.03, momentum=0.95, seed=20)
            try_update(a_out, "final_md")

        remaining = time_budget - (time.monotonic() - t_global)
        t6b = min(remaining * 0.35, 70.0)
        if t6b > 10:
            T_vals4 = [8, 5, 3, 2, 1, 0.5, 0.3, 0.1]
            a_out, _ = lse_lbfgs(best_seq, t6b, T_values=T_vals4, seed=30)
            try_update(a_out, "final_lse")

        remaining = time_budget - (time.monotonic() - t_global)
        t6c = min(remaining * 0.35, 70.0)
        if t6c > 10:
            a_out, _ = coord_descent(best_seq, t6c, seed=50)
            try_update(a_out, "final_coord")

    # Final check
    if best_seq is not None:
        fs = fast_score(best_seq)
        print(f"\nFINAL: score={fs:.6f}, n={len(best_seq)}")
    else:
        print("ERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
```
