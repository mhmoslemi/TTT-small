from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Compute full autocorrelation (a*a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def computeeval(a):
    """Compute the objective value 2n*max(conv)/(sum)^2."""
    n = len(a)
    conv = fast_autocorr(a)
    max_c = np.max(conv)
    s = np.sum(a)
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)

def objective_normalized(a):
    """Objective with sum(a) fixed to 1: 2n * max(conv)."""
    n = len(a)
    conv = fast_autocorr(a)
    return 2.0 * n * np.max(conv)

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    # First, project onto {x >= 0, sum(x) = S}
    # This is the same as: find tau such that sum(max(v - tau, 0)) = S
    # Then result = max(v - tau, 0)
    
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv[-1] - S))[0][-1]
    thresh = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - thresh, 0.0)

def lse_fista(a0, time_limit, S=None, T_start=50.0, T_end=1.0, verbose=True):
    n = len(a0)
    if S is None:
        S = np.sqrt(2.0 * n)
    
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a * S / s
    
    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_score = np.max(fast_autocorr(a))
    
    fft_size = 1 << (3 * n - 2).bit_length()
    
    # Adaptive temperature schedule
    n_total_steps = 10000  # rough estimate
    t0 = time.monotonic()
    step_count = 0
    
    for step in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive temperature
        T = T_start * (T_end / T_start) ** (step / n_total_steps)
        T = max(T, T_end)
        
        # Compute gradient at y
        conv_y = fast_autocorr(y)
        Tconv = T * conv_y
        Tconv -= Tconv.max()
        exp_Tconv = np.exp(Tconv)
        w = exp_Tconv / exp_Tconv.sum()
        
        # Gradient via FFT
        A = rfft(y, fft_size)
        W_flip = rfft(w[::-1], fft_size)
        C = irfft(A * W_flip, fft_size)
        grad = 2.0 * C[2*n-2 : n-2 : -1]
        
        # Step size (adaptive)
        # Lipschitz constant estimate: L ~ 2 * T * max(conv_y) * 2 (rough)
        L = 2.0 * T * np.max(conv_y) * 2.0 + 1.0
        step_size = 1.0 / L
        
        # FISTA update
        z = y - step_size * grad
        x_new = project_simplex(z, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - y)
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = x_new.copy()
        else:
            y_new = y_new * S / s_y
        
        y = y_new
        t = t_new
        step_count += 1
        
        # Check convergence
        score = np.max(fast_autocorr(x_new))
        if score < best_score:
            best_score = score
            best_a = x_new.copy()
        
        if verbose and step % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-FISTA step={step} (T={T:.1f}, {elapsed:.0f}s): max_conv={best_score:.8f}")
    
    return best_a, best_score

def coordinate_descent(a, time_limit, n_passes=None, verbose=True):
    """Coordinate descent to minimize 2n*max((a*a)) subject to a>=0, sum(a)=S."""
    n = len(a)
    S = a.sum()
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0).copy()
    
    best_a = a.copy()
    best_conv = fast_autocorr(a).copy()
    best_score = best_conv.max()
    
    t0 = time.monotonic()
    max_val = a.max()
    
    pass_num = 0
    while time.monotonic() - t0 < time_limit:
        pass_num += 1
        improved = False
        max_val = a.max()
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                    break
            
            old_val = a[j]
            if old_val < 1e-15:
                # Try setting it to a small positive value
                candidates = [old_val, max_val * 0.01, max_val * 0.1]
            else:
                # Try a range of values
                candidates = [0.0, old_val * 0.5, old_val, old_val * 1.5, 2.0 * old_val, max_val]
            
            best_new_val = old_val
            best_new_score = best_score  # current best
            
            for new_val in candidates:
                if new_val < 0:
                    continue
                delta = new_val - old_val
                if abs(delta) < 1e-15:
                    continue
                
                # Update convolution
                new_conv = best_conv.copy()
                for k in range(2*n-1):
                    kj = k - j
                    if 0 <= kj < n:
                        new_conv[k] += 2 * delta * a[kj]
                    if k == 2 * j:
                        new_conv[k] += delta * delta
                
                # Check max
                new_score = new_conv.max()
                
                if new_score < best_new_score - 1e-12:
                    best_new_score = new_score
                    best_new_val = new_val
                    best_new_conv = new_conv
            
            if best_new_val != old_val:
                a[j] = best_new_val
                best_conv = best_new_conv
                best_score = best_new_score
                best_a = a.copy()
                improved = True
        
        if verbose and pass_num % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={pass_num} ({elapsed:.0f}s): score={best_score:.8f}")
        
        if not improved:
            break
    
    return best_a, best_score

    return 2.0 * n * c.max() / (s * s)


def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver):
    import cvxpy as cp
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_conv_matrix(a)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def coordinate_descent(a, time_limit, verbose=True, rng_seed=42):
    """
    Fast coordinate descent to minimize max(a*a) subject to a >= 0, sum(a) = S.
    Uses O(1) amortized update via maintaining running convolution.
    """
    n = len(a)
    S = a.sum()
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0).copy()
    c = fast_autocorr(a)
    best_a = a.copy()
    best_c = c.copy()
    best_max = c.max()

    rng = np.random.default_rng(rng_seed)
    t0 = time.monotonic()
    pass_count = 0
    stall_count = 0
    coord_order = np.arange(n)

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        if pass_count % 3 == 0:
            rng.shuffle(coord_order)

        improved = False
        for j in coord_order:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            max_a = a.max()

            # Generate candidates
            candidates = []
            if old_val > 1e-14:
                candidates.extend([0.0, old_val * 0.3, old_val * 0.7, old_val * 1.3, old_val * 1.8])
            else:
                candidates.extend([max_a * 0.05, max_a * 0.15, max_a * 0.3])
            candidates.append(max_a)

            # Smart candidate: try to equalize top peak
            k_star = int(np.argmax(c))
            kj = k_star - j
            if 0 <= kj < n:
                # Peak at k_star changes by 2*delta*a[kj] + delta^2*[k_star=2j]
                # Find second max
                c_sorted = np.sort(c)[::-1]
                second_max = c_sorted[1] if len(c_sorted) > 1 else 0.0
                if c[k_star] > second_max:
                    # Solve c[k_star] + 2*delta*a[kj] + delta^2*[k_star==2j] = second_max
                    if k_star == 2 * j:
                        # delta^2 + 2*a[j]*delta + (c[k_star] - second_max) = 0
                        disc = 4 * old_val * old_val - 4 * (c[k_star] - second_max)
                        if disc >= 0:
                            delta1 = (-2 * old_val + np.sqrt(disc)) / 2.0
                            delta2 = (-2 * old_val - np.sqrt(disc)) / 2.0
                            for d in [delta1, delta2]:
                                nv = old_val + d
                                if 0 <= nv <= 2 * max_a:
                                    candidates.append(nv)
                    else:
                        if a[kj] > 1e-15:
                            delta = (second_max - c[k_star]) / (2.0 * a[kj])
                            nv = old_val + delta
                            if 0 <= nv <= 2 * max_a:
                                candidates.append(nv)

            # Remove duplicates and out-of-range
            best_new_val = old_val
            best_new_max = c.max()
            seen = set()
            for nv in candidates:
                if nv < 0 or nv > 3 * max_a:
                    continue
                key = round(nv, 10)
                if key in seen:
                    continue
                seen.add(key)
                delta = nv - old_val
                if abs(delta) < 1e-14:
                    continue

                # Compute new max efficiently
                k_start = max(0, j)
                k_end = min(2 * n - 2, j + n - 1)

                # Affected: c[k] + 2*delta*a[k-j] for k in [k_start, k_end]
                # (plus delta^2 at k=2j)
                if k_start <= k_end:
                    idx = np.arange(k_start, k_end + 1)
                    a_vals = a[idx - j].copy()
                    # For k=2j, use old_val
                    if 2 * j >= k_start and 2 * j <= k_end:
                        a_vals[2 * j - k_start] = old_val
                    affected = c[k_start:k_end + 1] + 2.0 * delta * a_vals
                    if 2 * j >= k_start and 2 * j <= k_end:
                        affected[2 * j - k_start] += delta * delta
                    local_max = affected.max()
                else:
                    local_max = -1e18

                # Unaffected max
                unaff_max = -1e18
                if k_start > 0:
                    unaff_max = max(unaff_max, c[:k_start].max())
                if k_end < 2 * n - 2:
                    unaff_max = max(unaff_max, c[k_end + 1:].max())

                new_max = max(local_max, unaff_max)

                if new_max < best_new_max - 1e-13:
                    best_new_max = new_max
                    best_new_val = nv

            if best_new_val != old_val and abs(best_new_val - old_val) > 1e-14:
                # Apply update
                delta = best_new_val - old_val
                k_start = max(0, j)
                k_end = min(2 * n - 2, j + n - 1)
                idx = np.arange(k_start, k_end + 1)
                a_vals = a[idx - j].copy()
                if 2 * j >= k_start and 2 * j <= k_end:
                    a_vals[2 * j - k_start] = old_val
                c[k_start:k_end + 1] += 2.0 * delta * a_vals
                if 0 <= 2 * j < 2 * n - 1:
                    c[2 * j] += delta * delta
                a[j] = best_new_val
                improved = True
                stall_count = 0
                if c.max() < best_max:
                    best_max = c.max()
                    best_a = a.copy()
                    best_c = c.copy()
            else:
                stall_count += 1

        if verbose and pass_count % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={pass_count} ({elapsed:.0f}s): max_conv={best_max:.8f}")

        if stall_count > n * 3:
            # Perturb to escape
            for _ in range(min(10, n // 100)):
                i = rng.integers(n)
                j2 = rng.integers(n)
                if i != j2 and a[i] > 1e-10:
                    frac = rng.uniform(0.01, 0.05)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j2] += amt
                    # Recompute c (expensive but rare)
                    c = fast_autocorr(a)
                    if c.max() < best_max + 0.01:
                        stall_count = 0
            if time.monotonic() - t0 > time_limit * 0.9:
                break

    # Restore sum constraint
    s = a.sum()
    if s > 1e-15:
        a = a * S / s
    return best_a, best_max


def lse_fista(a0, time_limit, verbose=True):
    """LSE-smoothed FISTA on min max_k (a*a)[k] s.t. a>=0, sum(a)=S."""
    n = len(a0)
    S = a0.sum()
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a * S / a.sum()

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    t_nest = 1.0
    fft_size = 1 << (3 * n - 2).bit_length()
    t0 = time.monotonic()

    T_start = 30.0
    T_end = 0.5
    n_total = 5000

    for step in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        frac = min(1.0, step / n_total)
        T = T_start * (T_end / T_start) ** frac

        conv_y = fast_autocorr(y)
        Tconv = T * conv_y
        Tconv -= Tconv.max()
        exp_T = np.exp(Tconv)
        w = exp_T / exp_T.sum()

        # Gradient via FFT
        A = rfft(y, fft_size)
        Wflip = rfft(w[::-1], fft_size)
        C = irfft(A * Wflip, fft_size)
        grad = 2.0 * C[2 * n - 2 : n - 2 : -1]

        # Step size
        L = 4.0 * T * conv_y.max() + 1.0
        step_size = 1.0 / L

        z = y - step_size * grad
        x_new = _project_simplex(z, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        y_new = x_new + ((t_nest - 1.0) / t_new) * (x_new - y)
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = x_new.copy()
        else:
            y_new = y_new * S / s_y

        y = y_new
        t_nest = t_new

        score = fast_autocorr(x_new).max()
        if score < best_score:
            best_score = score
            best_a = x_new.copy()

        if verbose and step % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-FISTA step={step} (T={T:.2f}, {elapsed:.0f}s): max_conv={best_score:.8f}")

    return best_a, best_score


def _project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if S <= 0:
        return np.zeros(n)
    # Scale to simplex with sum=1, project, then scale back
    u = np.sort(v / S)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1)
    cond = u * rho_arr > (cssv[-1] - 1.0)
    if not cond.any():
        return np.full(n, S / n)
    rho = int(np.nonzero(cond)[0][-1])
    thresh = (cssv[rho] - 1.0) / (rho + 1.0)
    result = np.maximum(v / S - thresh, 0.0)
    return result * S


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    choice = rng.integers(0, 5)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    else:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function. Returns best sequence found."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = eval_score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_score(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LP Nesterov for coarse optimization
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget={min(remaining*0.2, 300.0):.0f}s) ===")
    t1 = min(remaining * 0.2, 300.0)
    if t1 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
        try_update(a_out, f"lp_n{n_base}")

    # Phase 2: Coordinate descent on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Coord Descent n={n_best} (budget={min(remaining*0.25, 400.0):.0f}s) ===")
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent(a_norm, t2, verbose=True, rng_seed=123)
        try_update(a_out, f"cd_n{n_best}")

    # Phase 3: Try longer length n2
    n2 = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Extended n={n2} (budget={min(remaining*0.2, 300.0):.0f}s) ===")
    t3 = min(remaining * 0.2, 300.0)
    if t3 > 20:
        init_long = interpolate_sequence(best_seq, n2)
        # Quick LP
        t3_lp = t3 * 0.4
        t3_cd = t3 * 0.6
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t3_lp, solver, verbose=False)
        try_update(a_out, f"lp_n{n2}")
        # Then coordinate descent
        if time.monotonic() - t_global < time_budget - 5:
            a_norm2 = best_seq if len(best_seq) == n2 else interpolate_sequence(best_seq, n2)
            a_norm2 = a_norm2 * np.sqrt(2.0 * n2) / a_norm2.sum()
            a_out2, _ = coordinate_descent(a_norm2, t3_cd, verbose=True, rng_seed=456)
            try_update(a_out2, f"cd_n{n2}")

    # Phase 4: Try even longer n3
    n3 = n_base + 800
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Extended n={n3} (budget={min(remaining*0.15, 250.0):.0f}s) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 20:
        init_long3 = interpolate_sequence(best_seq, n3)
        a_norm = init_long3 * np.sqrt(2.0 * n3) / init_long3.sum()
        a_out, _ = coordinate_descent(a_norm, t4, verbose=True, rng_seed=789)
        try_update(a_out, f"cd_n{n3}")

    # Phase 5: LSE-FISTA on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LSE-FISTA n={n_best} (budget={min(remaining*0.15, 200.0):.0f}s) ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_fista(a_norm, t5, verbose=True)
        try_update(a_out, f"lse_n{n_best}")

    # Phase 6: Final intensive coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final CD n={n_best} (budget={min(remaining*0.4, 500.0):.0f}s) ===")
    t6 = min(remaining * 0.4, 500.0)
    if t6 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent(a_norm, t6, verbose=True, rng_seed=1111)
        try_update(a_out, f"cd_final_n{n_best}")

    # Phase 7: Multi-restart with perturbations
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Multi-restart n={n_best} (budget={min(remaining*0.5, 300.0):.0f}s) ===")
    t7 = min(remaining * 0.5, 300.0)
    if t7 > 10:
        n_restarts = min(5, int(t7 // 40))
        t_per_restart = t7 / max(1, n_restarts)
        for r in range(n_restarts):
            if time.monotonic() - t_global > time_budget - 5:
                break
            # Perturb the best solution
            a_pert = best_seq.copy()
            s = a_pert.sum()
            a_pert = a_pert * np.sqrt(2.0 * n_best) / s
            # Add random perturbation
            noise = rng.normal(0, 0.02, n_best)
            a_pert = np.maximum(a_pert + noise, 0.0)
            sp = a_pert.sum()
            if sp < 1e-15:
                continue
            a_pert = a_pert / sp
            # Short coordinate descent
            a_out, _ = coordinate_descent(a_pert, t_per_restart, verbose=False, rng_seed=2000 + r)
            try_update(a_out, f"restart_{r}_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = eval_score(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
