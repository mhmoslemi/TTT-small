from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the objective value for a normalized (sum=1) non-negative sequence."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a = a / s
    n = len(a)
    conv = fast_autocorr(a)
    return 2.0 * n * conv.max()


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.MOSEK, cp.GLPK, cp.CBC]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver, prev_B=None):
    """
    Solve LP: min M s.t. conv(a_fixed, b)_k <= M for all k, sum(b)=1, b>=0.
    Returns (b*, M*).
    """
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, warm_start=(prev_B is not None))
        if prev_B is not None:
            try:
                b_var.value = prev_B
            except Exception:
                pass
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.MOSEK, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

                    best_score = sc
                    best_a = cand.copy()
                    improved = True

            if not improved:
                stall_count += 1
            else:
                stall_count = 0

        # Duality gap based restart
        # The duality gap is roughly M_val - min_k conv(b_star, b_star)_k
        # But we use stall count as a proxy
        if stall_count > 500 and t > 100:
            # Restart from best
            a = best_a.copy()
            y = best_a.copy()
            avg_long = np.zeros(n)
            n_long = 0
            avg_short = np.zeros(n)
            n_short = 0
            short_buf = [None] * WINDOW_SHORT
            short_idx = 0
            stall_count = 0
            prev_B = None
            if t % 2000 == 0:
                print(f"      NLP restart at t={t}, best={best_score:.8f}, M={M_val:.8f}")

        # Also try: directly use b_star (it's a valid distribution)
        if t % 20 == 0:
            sc_b = compute_score(b_star)
            if sc_b < best_score:
                best_score = sc_b
                best_a = b_star.copy()
                stall_count = 0

        if t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M_val:.8f}")

    return best_a, best_score


def peak_targeted_optimize(a0, time_limit, seed=42):
    """
    Directly target the highest peaks by moving mass from high-contribution
    positions to low-contribution positions. Uses sensitivity analysis.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    stall = 0
    MAX_STALL = 500

    while time.monotonic() - t0 < time_limit - 3:
        conv = fast_autocorr(a)
        conv_max = conv.max()

        # Find top peaks (within 1% of max)
        threshold = conv_max * 0.99
        top_peaks = np.where(conv >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(conv)[::-1][:5]
        top_peaks = top_peaks[:30]

        # Compute sensitivity: for each position j, how much does it contribute to top peaks
        sensitivity = np.zeros(n)
        for K in top_peaks:
            js = max(0, K - n + 1)
            je = min(n - 1, K)
            idx = np.arange(js, je + 1)
            # Contribution of a_j to peak K is a_{K-j} * a_j (both ways counted)
            contrib = a[K - idx]
            # Weight by how much this peak contributes to the max
            weight = conv[K] / conv_max
            sensitivity[idx] += weight * contrib

        # Find source (high sensitivity, high value) and destination (low sensitivity, low value)
        improved = False

        # Sort by sensitivity * value (total "badness")
        badness = sensitivity * a
        src_candidates = np.argsort(badness)[::-1][:20]
        dst_candidates = np.argsort(sensitivity)[:20]

        for src in src_candidates:
            if a[src] < 1e-10:
                continue
            for dst in dst_candidates:
                if dst == src:
                    continue
                # Try moving a fraction from src to dst
                for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-10:
                        continue
                    a_trial = a_trial / st
                    score = compute_score(a_trial)
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape
            n_pert = rng.integers(1, 5)
            a_trial = a.copy()
            for _ in range(n_pert):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a_trial[i] > 1e-12:
                    frac = rng.uniform(0.002, 0.03)
                    amt = a_trial[i] * frac
                    a_trial[i] -= amt
                    a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 1e-10:
                a_trial = a_trial / st
                score = compute_score(a_trial)
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
            if stall > MAX_STALL:
                break
        else:
            stall = 0

        if stall % 50 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            print(f"      Peak target stall={stall} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def spectral_flatten(a0, time_limit, seed=123):
    """
    Flatten the autocorrelation in the spectral domain.
    In frequency domain, autocorrelation is |A(w)|^2. We try to reduce
    the variance of |A(w)|^2 while keeping sum(a) = 1 and a >= 0.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    # Use iterative approach: compute FFT, identify high-|A|^2 frequencies,
    # adjust a to reduce them
    for it in range(10000):
        if time.monotonic() - t0 > time_limit - 3:
            break

        fft_size = 1 << (2 * n - 1).bit_length()
        A = rfft(a, fft_size)
        power = A.real ** 2 + A.imag ** 2

        # Find frequencies with highest power
        top_freqs = np.argsort(power)[::-1][:50]

        # For each high-power frequency, compute the "direction" in a-space
        # that would reduce it
        # d/d(a_k) |A(w)|^2 = 2 * a_k * cos(w*k) - 2 * (A_real * sin(w*k) - A_imag * cos(w*k))
        # Simplified: gradient of |A(w)|^2 w.r.t. a is 2 * (|A| * [cos, -sin] pattern)
        # Actually: d|A(w)|^2/da_k = 2 * Re(conj(A) * e^{-i*w*k})
        # = 2 * (A_r * cos(w*k) + A_i * sin(w*k))

        grad = np.zeros(n)
        for f_idx in top_freqs:
            w = 2.0 * np.pi * f_idx / fft_size
            # Gradient contribution from this frequency
            cos_w_k = np.cos(w * np.arange(n))
            sin_w_k = np.sin(w * np.arange(n))
            grad += 2.0 * (A[f_idx].real * cos_w_k + A[f_idx].imag * sin_w_k)

        # Normalize gradient
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            continue
        grad = grad / g_norm

        # Move in direction that reduces the peak
        # We want to decrease high-power frequencies, so move in -grad direction
        # But we need to maintain sum(a) = 1, so project onto the simplex
        for lr in [0.01, 0.005, 0.002, 0.001]:
            a_trial = a - lr * grad
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-10:
                continue
            a_trial = a_trial / st
            score = compute_score(a_trial)
            if score < best_score - 1e-12:
                a = a_trial
                best_score = score
                best_a = a.copy()
                break

    return best_a, best_score


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def compute_topk_grad(a, T, topk):
    """
    Compute top-k LSE objective and gradient w.r.t. a.
    obj = (1/T)*log(sum_{k in top_k} exp(T*c_k)) where c = autocorr(a)
    """
    n = len(a)
    c = fast_autocorr(a)
    k_use = min(topk, len(c))
    top_idx = np.argsort(c)[::-1][:k_use]
    top_vals = c[top_idx]

    Tc = T * top_vals
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    lse_val = (max_Tc + np.log(exp_Tc.sum())) / T
    w = exp_Tc / exp_Tc.sum()

    grad_a = np.zeros(n)
    for idx in range(k_use):
        K = top_idx[idx]
        wK = w[idx]
        js = max(0, K - (n - 1))
        je = min(n - 1, K)
        if js <= je:
            idx_range = np.arange(js, je + 1)
            grad_a[idx_range] += 2.0 * wK * a[K - idx_range]

    return lse_val, grad_a


def topk_lse_grad_z(z, T, topk):
    """Objective and gradient in z-space for top-k LSE."""
    a = softmax_fn(z)
    n = len(a)
    obj, grad_a = compute_topk_grad(a, T, topk)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def adam_topk_lse(a0, time_limit, T_start=100.0, T_end=0.3, n_stages=25, topk=15):
    """
    Adam-optimized top-k LSE with multi-stage annealing.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = compute_score(a_init)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        base_lr = 3e-3 * min(1.0, 100.0 / n)
        lr = base_lr * (0.75 ** si)

        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.8, 0.999, 1e-8

        max_it = max(20, int(min(remaining, 100.0) * 4))

        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 3:
                break

            obj, grad_z = topk_lse_grad_z(z, stage_T, topk)

            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            if (it + 1) % 10 == 0:
                a = softmax_fn(z)
                score = compute_score(a)
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    if (it + 1) % 50 == 0:
                        z = np.log(np.maximum(best_a, 1e-30))

        a_stage = softmax_fn(z)
        score_stage = compute_score(a_stage)
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()

        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE stage {si}/{n_stages} (T={stage_T:.1f}, {elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def downsample_sequence(f, new_n):
    """Downsample a normalized sequence to a new (smaller) length."""
    old_n = len(f)
    if old_n <= new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_ds = np.interp(new_x, old_x, f_norm)
    f_ds = np.maximum(f_ds, 0)
    ss = f_ds.sum()
    if ss > 1e-15:
        f_ds = f_ds / ss
    return f_ds


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 10)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 2
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.7 + 0.2 * np.cos(2 * x) + 0.1 * np.cos(5 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * (x - 0.3) ** 2) + 0.5 * np.exp(-5 * (x - 0.7) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * (0.5 + 0.5 * np.cos(2 * np.pi * x))
    elif choice == 7:
        x = np.linspace(-1, 1, n)
        a = 0.5 * (1 + np.cos(np.pi * x))
    elif choice == 8:
        # Random smooth
        x = np.linspace(0, 1, n)
        a = np.ones(n) * 0.5
        for _ in range(5):
            center = rng.uniform(0, 1)
            width = rng.uniform(0.05, 0.3)
            amp = rng.uniform(-0.3, 0.3)
            a += amp * np.exp(-((x - center) / width) ** 2)
        a = np.maximum(a, 0)
    else:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * np.exp(-2 * (x - 0.5) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def perturbed_init(a0, rng, scale=0.1):
    """Create a perturbed version of a sequence for multi-start."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a0)) / len(a0)
    a = a / s
    noise = rng.normal(0, scale, len(a))
    a_new = a + noise * (a + 1e-6)
    a_new = np.maximum(a_new, 0)
    sn = a_new.sum()
    if sn < 1e-15:
        return a.copy()
    return a_new / sn


def propose_candidate(seed=42, time_budget=1800.0):
    """
    Multi-phase optimization combining LP Nesterov, peak-targeted descent,
    spectral flattening, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(time_budget) - 90.0, 1710.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    # Get parent sequence
    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = compute_score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = compute_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    n_base = len(best_seq) if best_seq is not None else 1600

    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = compute_score(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    # === Phase 1: LP Nesterov at base length (primary workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} ({remaining:.0f}s left) ===")
    t1 = min(remaining * 0.35, 600.0)
    if t1 > 60:
        a_out, _ = nesterov_lp_optimize(best_seq, t1, solver)
        try_update(a_out, f"lp_n{n_base}")

    # === Phase 2: Peak-targeted optimization ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 2: Peak-targeted n={n_base} ({remaining:.0f}s left) ===")
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 30:
        a_out, _ = peak_targeted_optimize(best_seq, t2, seed=111)
        try_update(a_out, f"peak_n{n_base}")

    # === Phase 3: LP Nesterov again from improved start ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 3: LP Nesterov v2 n={n_base} ({remaining:.0f}s left) ===")
    t3 = min(remaining * 0.20, 350.0)
    if t3 > 40:
        a_out, _ = nesterov_lp_optimize(best_seq, t3, solver)
        try_update(a_out, f"lp2_n{n_base}")

    # === Phase 4: Top-k LSE with Adam ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 4: Top-k LSE n={n_base} ({remaining:.0f}s left) ===")
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 20:
        a_out, _ = adam_topk_lse(best_seq, t4, T_start=60.0, T_end=0.3, n_stages=20, topk=20)
        try_update(a_out, f"lse_n{n_base}")

    # === Phase 5: Peak-targeted again ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 30:
        print(f"=== Phase 5: Peak-targeted v2 ({remaining:.0f}s left) ===")
        t5 = min(remaining * 0.10, 120.0)
        if t5 > 20:
            a_out, _ = peak_targeted_optimize(best_seq, t5, seed=222)
            try_update(a_out, "peak2")

    # === Phase 6: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 60:
        print(f"=== Phase 6: Multi-length exploration ({remaining:.0f}s left) ===")
        lengths_to_try = []
        # Try slightly longer
        n_long1 = n_base + 200
        n_long2 = n_base + 400
        # Try slightly shorter (might help if parent is too long)
        n_short = max(800, n_base - 200)
        lengths_to_try = [n_long1, n_long2, n_short]

        t_each = remaining * 0.15
        for nl in lengths_to_try:
            if time.monotonic() - t_global_start > time_budget - 30:
                break
            remaining_now = time_budget - (time.monotonic() - t_global_start)
            if remaining_now < 30:
                break
            t_alloc = min(t_each, remaining_now * 0.3)
            if t_alloc < 20:
                continue
            print(f"    Trying n={nl} ({t_alloc:.0f}s)")
            if nl > n_base:
                init = interpolate_sequence(best_seq, nl)
            elif nl < n_base:
                init = downsample_sequence(best_seq, nl)
            else:
                init = best_seq.copy()
            a_out, _ = nesterov_lp_optimize(init, t_alloc, solver)
            try_update(a_out, f"lp_n{nl}")

    # === Phase 7: Spectral flattening ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 20:
        print(f"=== Phase 7: Spectral flattening ({remaining:.0f}s left) ===")
        t7 = min(remaining * 0.10, 80.0)
        if t7 > 10:
            a_out, _ = spectral_flatten(best_seq, t7, seed=333)
            try_update(a_out, "spectral")

    # === Phase 8: Final peak-targeted polish ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 15:
        print(f"=== Phase 8: Final peak polish ({remaining:.0f}s left) ===")
        t8 = min(remaining * 0.25, 100.0)
        if t8 > 10:
            a_out, _ = peak_targeted_optimize(best_seq, t8, seed=444)
            try_update(a_out, "peak_final")

    # === Phase 9: Final LP round ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 30:
        print(f"=== Phase 9: Final LP ({remaining:.0f}s left) ===")
        t9 = min(remaining * 0.5, 150.0)
        if t9 > 20:
            a_out, _ = nesterov_lp_optimize(best_seq, t9, solver)
            try_update(a_out, "lp_final")

    # === Phase 10: Ultimate peak shave ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 10:
        t10 = min(remaining * 0.6, 50.0)
        if t10 > 5:
            a_out, _ = peak_targeted_optimize(best_seq, t10, seed=555)
            try_update(a_out, "peak_ultimate")

    # Final verification
    if best_seq is not None:
        fs = compute_score(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
