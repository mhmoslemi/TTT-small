The standard Nesterov averaging uses the running average of the primal iterates. But we can also use the running average of the dual iterates to get a better primal solution.

Actually, for this specific problem, the running average of the primal iterates (the `b_t` values) is the best approach. The theory says that the average of the first `T` iterates has a gap of O(1/T) from the optimal value.

**Idea: Use a more aggressive momentum schedule.**

The standard Nesterov uses `alpha_t = 2/(t+2)`. We can try:
- `alpha_t = 2/(t + beta)` for different `beta`.
- A piecewise schedule that starts with larger steps and decreases.

**Idea: Use the "accelerated proximal gradient" method instead of Nesterov.**

The problem can be formulated as:
`min_b max_k (T(a) @ b)[k]` subject to `sum(b) = 1, b >= 0`.

This is a convex primal-concave saddle point problem. We can use the accelerated mirror descent / proximal gradient method.

The mirror for the simplex constraint is the entropy, which leads to the same best-response LP.

So Nesterov LP is already the accelerated method. The question is whether we can do better.

**Idea: Use a " trustees " approach.**

Instead of a single Nesterov sequence, run multiple Nesterov sequences with different initializations in parallel (sequentially, since we have 1 CPU). Keep the best one.

This is what I planned in Phase 1.

**Idea: Use a "warm start" for the LP.**

If we solve the LP at step `t` with initial `a_t`, the solution `b_t` is close to the solution at step `t+1` (with initial `a_{t+1}` close to `a_t`). We can use `b_t` as a warm start for the LP at step `t+1`.

However, CVXPY doesn't support warm starts for most solvers. ECOS and SCS don't have warm start options.

But we can use the solution from the previous LP as a hint for the active set, which might help the solver converge faster.

Actually, for this problem, the warm start might not help much because the LP is small enough to solve quickly.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-restart Nesterov-accelerated LP best-response
2. LogSumExp-smoothed gradient optimization (L-BFGS)
3. Targeted peak shaving via coordinate descent
4. Multi-length exploration with interpolation
"""

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

def fast_autocorr(a):
    """Full self-convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_of(a):
    """Compute the score 2*n*max_conv/sum^2 for a normalized sequence."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    b = fast_autocorr(a_norm)
    return 2.0 * n * b.max()

def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where T @ b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed
    return T

def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.3])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. T(a_fixed)@b <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def nesterov_lp(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated LP best-response with Cesàro averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    y = a.copy()  # Nesterov's "lookahead" point
    a_curr = a.copy()
    
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    
    # Cesàro average
    avg_a = np.zeros(n)
    count = 0
    
    # Exponentially weighted average (EWA)
    ewa_a = a.copy()
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP for best response to y
        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1 - alpha) * y + alpha * b_star
        y_new = np.maximum(y_new, 0)
        ss = y_new.sum()
        if ss < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / ss
        
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            ss = a_new.sum()
            if ss < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / ss
        
        y = y_new
        a_curr = a_new
        
        # Check score of current iterate
        sc = score_of(a_curr)
        if sc < best_score:
            best_score = sc
            best_a = a_curr.copy()
        
        # Check score of y
        sc_y = score_of(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        
        # Update Cesàro average
        avg_a += a_curr
        count += 1
        
        # Update EWA (decay = 0.99)
        ramda = 0.99
        ewa_a = ramda * ewa_a + (1 - ramda) * a_curr
        ewa_a = ewa_a / ewa_a.sum()
        
        # Check averages periodically
        if count >= 20 and count % 50 == 0:
            cesaro = avg_a / count
            sc_c = score_of(cesaro)
            if sc_c < best_score:
                best_score = sc_c
                best_a = cesaro.copy()
            
            sc_e = score_of(ewa_a)
            if sc_e < best_score:
                best_score = sc_e
                best_a = ewa_a.copy()
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP t={t} ({elapsed:.0f}s): best={best_score:.10f}, M={M_val:.10f}")
    
    return best_a, best_score

def lse_gradient(a, T):
    """Compute gradient of (1/T)*log(sum exp(T*(a*a))) w.r.t. a, using FFT."""
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    
    # grad[i] = 2 * sum_k w[k] * a[k-i] (valid k)
    # This is 2 * (w * a_rev)[n-1 : 2n-2] where a_rev = a[::-1]
    a_rev = a[::-1].copy()
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    A_rev = rfft(a_rev, fft_size)
    conv = irfft(W * A_rev, fft_size)
    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad

def lse_objective(a, T):
    """Compute (1/T)*log(sum exp(T*(a*a)))."""
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    return (max_Tb + np.log(np.exp(Tb - max_Tb).sum())) / T
