```python
            sc_new = score_normalized(a_new)
            
            # For LSE, we want obj to decrease
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated LogSumExp gradient (O(n log n) per step vs O(n^2) before)
2. Adam optimizer with adaptive temperature schedule for global search
3. L-BFGS with FFT gradient for local refinement at low temperatures
4. Nesterov LP best-response for global moves
5. Coordinate-descent style peak shaving for final polish
6. Multi-length exploration
Key innovation: FFT-based gradient computation enables ~10x more iterations,
allowing deeper optimization within the time budget.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Fast linear convolution a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score assuming sum(a)=1: returns 2*n*max(b)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def softmax_from_z(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Log of a for softmax parameterization."""
    return np.log(np.maximum(a, 1e-300))


def lse_obj_grad_fft(z, T, n, fft_size_g=None):
    """
    Compute LSE objective and gradient w.r.t. z using FFT for O(n log n) cost.
    F(z) = (2n/T) * log(sum_k exp(T * c[k])) where c = conv(a,a), a = softmax(z)
    Returns (objective, gradient_wrt_z).
    """
    a = softmax_from_z(z)
    c = fast_convolve(a)  # length 2n-1

    # Stable log-sum-exp
    Tc = T * c
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    sum_exp = exp_Tc.sum()
    w = exp_Tc / sum_exp  # softmax weights over peaks, length 2n-1

    log_sum_exp = max_Tc + np.log(sum_exp)
    obj = (2.0 * n / T) * log_sum_exp

    # Compute g[i] = sum_{m=0}^{n-1} w[i+m] * a[m] via FFT
    # g[i] = (w * a_rev)[i + n - 1]
    if fft_size_g is None:
        fft_size_g = 1 << (3 * n - 2 - 1).bit_length()
    a_rev = a[::-1]
    W = rfft(w, fft_size_g)
    AR = rfft(a_rev, fft_size_g)
    conv_w_ar = irfft(W * AR, fft_size_g)
    g = conv_w_ar[n - 1:2 * n - 1]  # length n

    S = np.dot(a, g)
    grad = (2.0 * n / T) * a * (g - S)

    return obj, grad


def lse_obj_grad_fft_no_grad(z, T, n):
    """Compute only the LSE objective (no gradient) - faster."""
    a = softmax_from_z(z)
    c = fast_convolve(a)
    Tc = T * c
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    sum_exp = exp_Tc.sum()
    log_sum_exp = max_Tc + np.log(sum_exp)
    obj = (2.0 * n / T) * log_sum_exp
    return obj


def adam_lse(a0, time_limit, T_start=80.0, T_end=2.0, n_stages=30, lr=0.05, verbose=True):
    """
    Adam optimizer on LSE objective with FFT gradient and geometric temperature schedule.
    """
    n = len(a0)
    z = z_from_a(a0).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8

    best_a = a0.copy()
    best_score = score_normalized(a0)

    t0 = time.monotonic()
    stage_temps = np.geomspace(T_start, T_end, n_stages)
    # Give more time to later (lower T) stages
    weights = np.zeros(n_stages)
    for i in range(n_stages):
        weights[i] = 1.0 / stage_temps[i]
    weights /= weights.sum()
    stage_ends = t0 + time_limit * np.cumsum(weights)

    fft_size_g = 1 << (3 * n - 2 - 1).bit_length()
    t_count_global = 0

    for stage in range(n_stages):
        T = stage_temps[stage]
        stage_end = stage_ends[stage]

        iters_in_stage = 0
        while time.monotonic() < stage_end:
            obj, grad = lse_obj_grad_fft(z, T, n, fft_size_g)
            if not np.isfinite(obj):
                break

            t_count_global += 1
            iters_in_stage += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** t_count_global)
            v_hat = v / (1 - beta2 ** t_count_global)

            # Adaptive learning rate decay
            lr_eff = lr / (1.0 + 0.01 * stage)
            z = z - lr_eff * m_hat / (np.sqrt(v_hat) + eps)

            # Periodic score check (every 50 iters to save time)
            if iters_in_stage % 50 == 0:
                a = softmax_from_z(z)
                sc = score_normalized(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    if verbose and (stage + 1) % 5 == 0:
                        elapsed = time.monotonic() - t0
                        print(f"    AdamLSE stage={stage+1}/{n_stages} T={T:.1f}: score={best_score:.8f} (elapsed={elapsed:.0f}s)")

    # Final check
    a = softmax_from_z(z)
    sc = score_normalized(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_a, best_score


def lbfgs_lse(a0, time_limit, T_start=10.0, T_end=0.5, n_stages=15, max_iter_per_stage=60, verbose=True):
    """
    L-BFGS on LSE objective with FFT gradient. Lower temperatures for precision.
    """
    n = len(a0)
    z = z_from_a(a0).copy()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    t0 = time.monotonic()
    stage_temps = np.geomspace(T_start, T_end, n_stages)
    stage_dur = time_limit / n_stages
    fft_size_g = 1 << (3 * n - 2 - 1).bit_length()

    for stage in range(n_stages):
        T = stage_temps[stage]
        stage_end = t0 + (stage + 1) * stage_dur

        if time.monotonic() >= stage_end:
            continue

        def func(z_val):
            obj, _ = lse_obj_grad_fft(z_val, T, n, fft_size_g)
            return obj

        def jac(z_val):
            _, grad = lse_obj_grad_fft(z_val, T, n, fft_size_g)
            return grad

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': max_iter_per_stage, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': max_iter_per_stage * 200,
                         'disp': False}
            )
            z = res.x
            a = softmax_from_z(z)
            sc = score_normalized(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            if verbose and (stage + 1) % 3 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LBFSGS-LSE stage={stage+1}/{n_stages} T={T:.2f}: score={sc:.8f} best={best_score:.8f} (elapsed={elapsed:.0f}s)")
        except Exception as e:
            if verbose:
                print(f"    LBFSGS-LSE stage {stage} error: {e}")

    return best_a, best_score


def build_conv_matrix(a_fixed):
    """Build convolution matrix T where (T@b)[k] = (a_fixed*b)[k]."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver=None):
    """Solve LP: min M s.t. (a_fixed * b)[k] <= M for all k, sum(b)=1, b>=0."""
    if solver is None:
        solver = _get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        prob = None
    if prob is None or prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS, cp.CBC, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob2 = cp.Problem(cp.Minimize(M_var), constraints)
                prob2.solve(solver=alt)
                if prob2.status in ('optimal', 'optimal_inaccurate'):
                    prob = prob2
                    break
            except Exception:
                continue
    if prob is None or prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    b_val = np.asarray(prob.variables[0].value if hasattr(prob, 'variables') else b_var.value, dtype=np.float64)
    # Actually let's be more careful
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    b_val = b_val / s
    return b_val


def _get_lp_solver():
    global _cached_solver
    if '_cached_solver' not in dir():
        pass
    try:
        return _lp_solver_cache
    except NameError:
        pass
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CBC, cp.GLPK]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                globals()['_lp_solver_cache'] = solver_name
                return solver_name
        except Exception:
            continue
    globals()['_lp_solver_cache'] = cp.ECOS
    return cp.ECOS


def nesterov_lp(a0, time_limit, max_iters=200000, verbose=True):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new = solve_lp_best_response(y)
        if b_new is None:
            continue

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        if verbose and (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 20:
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
                last_print = elapsed

    return best_a, best_score


def peak_shave(a0, time_limit, n_passes=50, verbose=True):
    """
    Coordinate-style peak shaving: identify the worst peak, then reduce its
    value by adjusting the contributing a[i] values.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        return a, score_normalized(a)
    a = a / s

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for p in range(n_passes):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        k_star = int(np.argmax(b))
        max_b = b[k_star]

        # Gradient of peak at k_star w.r.t. a
        # d(a*a)[k_star]/da[i] = a[k_star - i] for valid i
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        idx = np.arange(lo, hi + 1)
        contrib = a[k_star - idx]  # contribution of each a[i] to the peak

        # We want to reduce the peak: move mass from high-contribution positions
        # to low-contribution positions, while keeping sum=1
        # Use a "water-filling" style update
        # Reduce a[i] proportionally to contrib[i] at contributing positions
        # and increase elsewhere

        # Build full gradient vector
        grad = np.zeros(n)
        grad[idx] = contrib

        # Center the gradient (tangent to simplex)
        S = np.dot(grad, a)
        grad_tangent = grad - S  # projection onto sum=0 subspace

        # Take a small step in -gradient direction
        # Step size: choose so that the peak decreases by ~5%
        grad_norm = np.linalg.norm(grad_tangent)
        if grad_norm < 1e-15:
            break

        # Try multiple step sizes
        best_local_a = a.copy()
        best_local_score = best_score
        for frac in [0.01, 0.005, 0.001, 0.0005]:
            a_try = a - frac * grad_tangent
            a_try = np.maximum(a_try, 0)
            st = a_try.sum()
            if st < 1e-15:
                continue
            a_try = a_try / st
            sc = score_normalized(a_try)
            if sc < best_local_score:
                best_local_score = sc
                best_local_a = a_try.copy()

        a = best_local_a
        if best_local_score < best_score:
            best_score = best_local_score
            best_a = a.copy()

        if verbose and (p + 1) % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave pass={p+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate structured initial sequences."""
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-3 * x ** 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.2 * np.cos(2 * np.pi * 5 * x)
    elif choice == 3:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.2 * np.cos(3 * x) + 0.15 * np.cos(7 * x) + 0.1 * np.cos(11 * x)
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 4 * x)
    else:
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.5) ** 2) / 0.08) + 0.3 * np.exp(-((x - 0.2) ** 2) / 0.03) + 0.3 * np.exp(-((x - 0.8) ** 2) / 0.03)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Combines FFT-accelerated Adam, L-BFGS, Nesterov LP,
    and peak shaving to minimize the score.
    """
    global _lp_solver_cache
    _lp_solver_cache = None
    solver = _get_lp_solver()
    print(f"Using LP solver: {solver}")

    rng = np.random.default_rng(seed)
    global_deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)
    total_budget = max(1.0, float(budget_s) - 40.0)

    # Get parent sequence
    best_seq = None
    best_score = float('inf')
    n_base = 1200

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = score_normalized(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        best_seq = structured_init(n_base, rng)
        best_score = score_normalized(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = score_normalized(a)
        if np.isfinite(sc) and sc < best_score:
            remain = global_deadline - time.monotonic()
            print(f"  ** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, remaining={remain:.0f}s)")
            best_score = sc
            best_seq = a.copy()

    # === Phase 1: FFT-accelerated Adam (high to medium T) - 25% time ===
    print(f"\n=== Phase 1: Adam LSE FFT (score={best_score:.8f}) ===")
    t1 = 0.25 * total_budget
    a_out, _ = adam_lse(best_seq, t1, T_start=80.0, T_end=5.0, n_stages=20, lr=0.05)
    try_update(a_out, "adam_high")
    print(f"  Phase 1 done: best_score={best_score:.8f}")

    # === Phase 2: FFT-accelerated Adam (medium to low T) - 25% time ===
    print(f"\n=== Phase 2: Adam LSE FFT low (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t2 = min(0.25 * total_budget, remaining * 0.6)
    if t2 > 30:
        a_out2, _ = adam_lse(best_seq, t2, T_start=5.0, T_end=0.8, n_stages=25, lr=0.03)
        try_update(a_out2, "adam_low")
    print(f"  Phase 2 done: best_score={best_score:.8f}")

    # === Phase 3: L-BFGS LSE (low T, precision) - 20% time ===
    print(f"\n=== Phase 3: L-BFGS LSE (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t3 = min(0.20 * total_budget, remaining * 0.6)
    if t3 > 30:
        a_out3, _ = lbfgs_lse(best_seq, t3, T_start=8.0, T_end=0.3, n_stages=12, max_iter_per_stage=50)
        try_update(a_out3, "lbfgs_lse")
    print(f"  Phase 3 done: best_score={best_score:.8f}")

    # === Phase 4: Nesterov LP - 15% time ===
    print(f"\n=== Phase 4: Nesterov LP (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t4 = min(0.15 * total_budget, remaining * 0.6)
    if t4 > 30:
        a_out4, _ = nesterov_lp(best_seq, t4, max_iters=100000)
        try_update(a_out4, "nesterov_lp")

        # Follow up with L-BFGS from the LP result
        remaining = global_deadline - time.monotonic()
        if remaining > 20:
            a_out4b, _ = lbfgs_lse(best_seq, remaining * 0.5, T_start=5.0, T_end=0.3, n_stages=8, max_iter_per_stage=40)
            try_update(a_out4b, "lbfgs_from_lp")
    print(f"  Phase 4 done: best_score={best_score:.8f}")

    # === Phase 5: Peak shaving + local LSE - 10% time ===
    print(f"\n=== Phase 5: Peak Shave (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t5 = min(0.10 * total_budget, remaining * 0.6)
    if t5 > 15:
        a_out5, _ = peak_shave(best_seq, t5 * 0.6, n_passes=100)
        try_update(a_out5, "peak_shave")

        remaining = global_deadline - time.monotonic()
        if remaining > 15:
            a_out5b, _ = lbfgs_lse(best_seq, remaining * 0.7, T_start=3.0, T_end=0.2, n_stages=8, max_iter_per_stage=50)
            try_update(a_out5b, "lbfgs_after_shave")
    print(f"  Phase 5 done: best_score={best_score:.8f}")

    # === Phase 6: Multi-length exploration - 5% time ===
    print(f"\n=== Phase 6: Multi-length (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t6 = min(0.05 * total_budget, remaining * 0.7)
    if t6 > 30:
        lengths_to_try = [n_base + 200, n_base + 400, n_base - 200, n_base + 600]
        for n_try in lengths_to_try:
            remaining = global_deadline - time.monotonic()
            if remaining < 20:
                break
            t6_sub = min(t6 / len(lengths_to_try), remaining * 0.45)
            if t6_sub < 10:
                break
            print(f"  Trying n={n_try}, budget={t6_sub:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            # Quick Adam
            a_try, _ = adam_lse(init_try, t6_sub * 0.5, T_start=30.0, T_end=2.0, n_stages=8, lr=0.04, verbose=False)
            try_update(a_try, f"len_{n_try}_adam")
            # Quick L-BFGS
            remaining = global_deadline - time.monotonic()
            if remaining > 10:
                a_try2, _ = lbfgs_lse(a_try, remaining * 0.4, T_start=5.0, T_end=0.5, n_stages=5, max_iter_per_stage=30, verbose=False)
                try_update(a_try2, f"len_{n_try}_lbfgs")
    print(f"  Phase 6 done: best_score={best_score:.8f}")

    # === Final: One more Adam+LBFSGS cycle on the best ===
    remaining = global_deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Final Polish (score={best_score:.8f}) ===")
        a_fin1, _ = adam_lse(best_seq, remaining * 0.4, T_start=3.0, T_end=0.5, n_stages=8, lr=0.02, verbose=False)
        try_update(a_fin1, "final_adam")
        remaining = global_deadline - time.monotonic()
        if remaining > 10:
            a_fin2, _ = lbfgs_lse(best_seq, remaining * 0.8, T_start=2.0, T_end=0.2, n_stages=5, max_iter_per_stage=50, verbose=False)
            try_update(a_fin2, "final_lbfgs")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
