def fast_autocorr(a):
    """Full autocorrelation a*a using FFT. Returns array of length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * np.conj(A), fft_size)[:size]


def fast_score(a):
    """Compute 2n*max(b)/(sum(a))^2 quickly."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)

def autocorr_variance_grad(a):
    """
    Compute gradient of Var(c) where c = autocorr(a), normalized to sum(a)=1.
    Returns gradient with respect to a (in the cone a >= 0, sum(a)=1).
    Uses FFT for O(n log n) computation.

    grad[j] = 4/(2n-1) * sum_m e[j+m] * a[m] where e = c - c_bar

    n = len(a)
    c = fast_autocorr(a)
    c_bar = 1.0 / (2 * n - 1)
    e = c - c_bar
    
    # grad[j] = 4/(2n-1) * sum_m e[j+m] * a[m] for m=0..n-1

    # Equivalent to convolution: if a_flip[m] = a[n-1-m], then
    # (e * a_flip)[j + n - 1] = sum_m e[j+n-1-m] * a_flip[m] = sum_m e[j+n-1-m] * a[n-1-m]
    # Let p = n-1-m, m = n-1-p: = sum_p e[j+n-1-(n-1-p)] * a[p] = sum_p e[j+p] * a[p]

    grad = conv[n-1:n-1+n]

    return grad

            if s_new < 1e-15:
                a_new = a + 0.01 * rng.standard_normal(n)
                a_new = np.maximum(a_new, 0)
                s_new = a_new.sum()
            a_new = a_new / s_new
            a = a_new
            
            # Check improvement
            if t % 10 == 0 or t < 20:
                sc = fast_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    VDACE restart {restart+1}/{n_restarts} ({elapsed:.0f}s): best_score={best_score:.6f}")
    
    return best_a, best_score


def spectral_reweight(a0, time_limit, seed=42, verbose=True):
    """
    Phase-preserving spectral reweighting to flatten autocorrelation.
    Iteratively adjusts |A|^2 to reduce autocorrelation peaks while preserving phase.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / a.sum()
    
    fft_size = 1 << (2 * n - 2).bit_length()
    
    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    
    for t in range(200000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute current autocorrelation
        A = rfft(a, fft_size)
        P = np.abs(A) ** 2
        c = irfft(P, fft_size)[:2*n-1]
        c_target = 1.0 / (2 * n - 1)
        
        # Compute peak excess
        excess = c - c_target
        max_excess = excess.max()
        if max_excess < 1e-8:
            break
        
        # Build correction in frequency domain
        # We want to reduce c where it's high. In freq domain, c = IFFT(P).
        # Correction: P_new = P * (1 - eta * |FFT(excess_pos)|^2 / P) ... 
        # Simpler: multiplicative update P_new = P / (1 + eta * FFT(excess_pos))
        # But need to be careful about complex values.
        
        # Alternative: use the gradient approach in freq domain
        # d(c)/dP = circle identity (IFFT is linear), so grad_P = gradient of f(P) w.r.t. P
        # For f = max(c) approx: grad_P[k] = w[k] where w = FFT(weights_on_c)
        
        # Softmax weighting on c peaks
        T = max(1.0, max_excess * 5.0)
        w = np.exp((excess - excess.max()) / T)
        w = w / w.sum()
        
        # FFT of weights (gives correction in frequency domain)
        W = rfft(w, fft_size)
        W_real = np.real(W)
        
        # Learning rate with decay
        eta = 0.1 * (1.0 / (1.0 + t / 500.0)) ** 0.7
        
        # Update power spectrum: reduce where correction says to
        # P_new = P * (1 - eta * W_real / P.mean())
        P_correction = 1.0 - eta * W_real / max(P.mean(), 1e-15)
        P_correction = np.maximum(P_correction, 0.1)  # prevent over-correction
        P_new = P * P_correction
        
        # Reconstruct A with same phase
        phase = A / np.maximum(np.abs(A), 1e-15)
        A_new = phase * np.sqrt(np.maximum(P_new, 0))
        
        # Go back to time domain
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new
        
        # Check improvement
        if t % 20 == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SpecRew t={t} ({elapsed:.0f}s): score={fast_score(a):.6f}, best={best_score:.6f}")
    
    return best_a, best_score


def lse_refine(a0, time_limit, T_start=50.0, T_end=0.3, n_stages=10, seed=42):
    """
    LogSumExp-smoothed optimization with L-BFGS.
    Minimizes (1/T)*log(sum(exp(T*c_k))) which approximates max(c).
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = fast_score(a_init)
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        iters = max(20, min(60, int(20 + 40 * si / max(1, n_stages))))
        
        # Compute current autocorrelation for gradient
        def obj_grad(z_in):
            z_shift = z_in - z_in.max()
            e = np.exp(z_shift)
            a = e / e.sum()
            
            c = fast_autocorr(a)
            Tc = stage_T * c
            Tc_shift = Tc - Tc.max()
            exp_Tc = np.exp(Tc_shift)
            obj = (Tc.max() + np.log(exp_Tc.sum())) / stage_T
            
            # Gradient w.r.t. a
            w = exp_Tc / exp_Tc.sum()
            # grad_a[j] = 2 * sum_k w[k] * c_grad[k,j] = 2 * sum_k w[k] * a[k-j] (valid)
            # = 2 * cross_correlation(w, a)
            a_flip = a[::-1]
            conv_len = len(w) + n - 1
            fft_sz = 1 << (conv_len - 1).bit_length()
            W = rfft(w, fft_sz)
            AF = rfft(a_flip, fft_sz)
            conv = irfft(W * AF, fft_sz)[:conv_len]
            grad_a = 2.0 * conv[n-1:n-1+n]
            
            # Gradient w.r.t. z (softmax parameterization)
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)
            
            return obj, grad_z
        
        try:
            res = minimize(fun=obj_grad, x0=z, jac=True, method='L-BFGS-B',
                           options={'maxiter': iters, 'ftol': 1e-15,
                                    'gtol': 1e-9, 'maxfun': iters * 100})
            z = res.x
            z_shift = z - z.max()
            a = np.exp(z_shift)
            a = a / a.sum()
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        except Exception:
            pass
    
    return best_a, best_score


def coordinate_polish(a0, time_limit, seed=42, verbose=True):
    """
    Coordinate descent: find pairs (i,j) where moving mass from i to j
    reduces max autocorrelation. Focus on the top peaks.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / a.sum()
    
    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a)
        c_max_idx = np.argsort(c)[::-1][:8]
        
        improved = False
        for k_idx in c_max_idx[:4]:
            k_idx = int(k_idx)
            js = max(0, k_idx - n + 1)
            je = min(n - 1, k_idx)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k_idx - idx]
            if contrib.sum() < 1e-15:
                continue
            
            # Source: highest contributor
            src = int(idx[np.argmax(contrib)])
            
            # Destination: positions that don't contribute to top peaks
            peak_regions = set()
            for k2 in c_max_idx[:5]:
                k2 = int(k2)
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_regions.update(range(js2, je2 + 1))
            
            candidates = [i for i in range(n) if i not in peak_regions and i != src]
            if not candidates:
                candidates = [i for i in range(n) if i != src]
            
            # Pick from lowest a values among candidates
            candidates.sort(key=lambda i: a[i])
            top_candidates = candidates[:max(3, len(candidates) // 8)]
            
            for dst in top_candidates[:3]:
                for frac in [0.003, 0.008, 0.015, 0.03, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    sc = fast_score(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial.copy()
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if rng.integers(4) == 0 and a.sum() > 1e-10:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.005, 0.02)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] = max(0, a[i] - amt)
                    a_trial[j] += amt
                    a_trial = a_trial / a_trial.sum()
                    sc = fast_score(a_trial)
                    if sc < best_score:
                        a = a_trial.copy()
                        best_score = sc
                        best_a = a.copy()
                        stall = 0
            if stall > 300:
                break
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    CoordPolish ({elapsed:.0f}s): best={best_score:.6f}")
    return best_a, best_score


def diverse_init(n, rng, variant=None):
    """Generate diverse structured initializations."""
    if variant is None:
        variant = int(rng.integers(0, 12))
    
    x = np.linspace(0, 1, n)
    if variant == 0:
        a = np.maximum(1 - 2 * abs(x - 0.5), 0)
    elif variant == 1:
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif variant == 2:
        a = np.exp(-3 * x ** 2)
    elif variant == 3:
        a = np.exp(-20 * (x - 0.25) ** 2) + np.exp(-20 * (x - 0.75) ** 2)
    elif variant == 4:
        a = x ** 2 + (1 - x) ** 2 + 0.1
    elif variant == 5:
        a = 1 + 0.5 * np.cos(4 * np.pi * x)
    elif variant == 6:
        a = np.sinc(x - 0.5) ** 2
    elif variant == 7:
        a = x + 0.1
    elif variant == 8:
        a = 1 + 0.3 * np.cos(3 * 2 * np.pi * x) + 0.1 * np.cos(7 * 2 * np.pi * x)
    elif variant == 9:
        a = abs(x - 0.5) + 0.2
    elif variant == 10:
        # Asymmetric: heavy on left
        a = np.exp(-2 * x) + 0.3
    else:
        a = np.ones(n)
        for _ in range(4):
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.1, n * 0.3)
            amp = rng.uniform(0.2, 1.0)
            xx = np.arange(n, dtype=np.float64)
            a += amp * np.exp(-(xx - center) ** 2 / (2 * width ** 2))
        a = np.maximum(a, 0)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def combined_objective_grad(a, T_max=10.0, T_var=5.0):
    """
    Combined objective: smooth max + variance penalty.
    Returns gradient of (1/T_max)*log(sum(exp(T_max*c))) + (T_var/(2n-1))*sum(c^2)
    """
    n = len(a)
    c = fast_autocorr(a)
    
    # LSE part
    Tc = T_max * c
    Tc_shift = Tc - Tc.max()
    exp_Tc = np.exp(Tc_shift)
    w_lse = exp_Tc / exp_Tc.sum()
    
    # Variance part: weight proportional to c[k] (penalizes large values more)
    c_bar = c.mean()
    w_var = (c - c_bar) / max(np.sum((c - c_bar)**2), 1e-15) * len(c)
    w_var = np.maximum(w_var, 0)  # only penalize above-mean
    if w_var.sum() > 0:
        w_var = w_var / w_var.sum()
    else:
        w_var = np.ones(len(c)) / len(c)
    
    # Combined weights
    alpha = 0.5
    w = alpha * w_lse + (1 - alpha) * w_var
    
    # Gradient: 2 * cross_correlation(w, a)
    a_flip = a[::-1]
    conv_len = len(w) + n - 1
    fft_sz = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_sz)
    AF = rfft(a_flip, fft_sz)
    conv = irfft(W * AF, fft_sz)[:conv_len]
    grad = 2.0 * conv[n-1:n-1+n]
    
    return grad


def combined_descent(a0, time_limit, seed=42, verbose=True):
    """
    Gradient descent on combined objective (smooth max + variance).
    Uses adaptive step size and restarts.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / a.sum()
    
    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    
    T_max = 20.0
    n_restarts = 4
    iters_per_restart = 8000
    
    for restart in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break
        
        if restart > 0:
            noise_scale = 0.01 * (1 + 0.5 * restart)
            a = a + noise_scale * rng.standard_normal(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
        
        # Anneal T_max
        T_start = 20.0 / (1 + 0.3 * restart)
        T_end = 3.0 / (1 + 0.3 * restart)
        
        for t in range(iters_per_restart):
            if time.monotonic() - t0 > time_limit:
                break
            if time.monotonic() - t0 > time_limit:
                break
            
            # Anneal temperature
            prog = t / iters_per_restart
            T_cur = T_start * (T_end / T_start) ** prog
            
            # Compute gradient
            grad = combined_objective_grad(a, T_max=T_cur, T_var=2.0)
            grad = grad - np.mean(grad)
            
            # Adaptive step size
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-12:
                break
            lr = 0.005 / (1 + 0.5 * prog)
            
            # Line search (simple backtracking)
            step = lr
            for _ in range(5):
                a_trial = a - step * grad
                a_trial = np.maximum(a_trial, 0)
                s_trial = a_trial.sum()
                if s_trial < 1e-15:
                    step *= 0.5
                    continue
                a_trial = a_trial / s_trial
                
                c = fast_autocorr(a_trial)
                Tc = T_cur * c
                Tc_shift = Tc - Tc.max()
                obj_new = (Tc.max() + np.log(np.exp(Tc_shift).sum())) / T_cur
                obj_var = np.var(c)
                
                c_old = fast_autocorr(a)
                Tc_old = T_cur * c_old
                Tc_old_shift = Tc_old - Tc_old.max()
                obj_old = (Tc_old.max() + np.log(np.exp(Tc_old_shift).sum())) / T_cur + np.var(c_old)
                
                obj_new_total = (Tc.max() + np.log(np.exp(Tc_shift).sum())) / T_cur + np.var(c)
                obj_old_total = obj_old                
                if obj_new_total < obj_old_total + 1e-10:
                    break
                step *= 0.5
            
            a = a - step * grad
            a = np.maximum(a, 0)
            s_new = a.sum()
            if s_new < 1e-15:
                a = best_a + 0.01 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
            else:
                a = a / s_new
            
            if t % 15 == 0:
                sc = fast_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    CombDescent restart {restart+1}/{n_restarts} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization:
    Phase 1: Variance-driven descent to flatten autocorrelation
    Phase 2: Spectral reweighting for peak reduction
    Phase 3: Combined max+variance gradient descent
    Phase 4: LSE refinement
    Phase 5: Coordinate polish
    Phase 6: Multi-length exploration
    Phase 7: Final combined polish
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1760.0)
    rng = np.random.default_rng(seed)
    
    best_seq = None
    best_score = float('inf')
    
    # Try to use parent sequence
    n_parent = 1600
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            n_parent = len(parent)
            p_score = evaluate_safe(parent)
            print(f"Parent: n={n_parent}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = fast_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    if best_seq is None:
        best_seq = diverse_init(n_parent, rng, variant=4)
        best_score = fast_score(best_seq)
        print(f"Default init: n={n_parent}, score={best_score:.6f}")
    
    # Phase 1: Variance-Driven Descent (flatten autocorrelation)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Variance-Driven Descent at n={n_parent} (budget={remaining:.0f}s) ===")
    t_vdace = min(remaining * 0.18, 300.0)
    if t_vdace > 30:
        a_out, _ = variance_driven_descent(best_seq, t_vdace, lr_schedule='cosine',
                                            n_restarts=3, seed=1, verbose=True)
        try_update(a_out, f"vdace_n{n_parent}")
    
    # Phase 2: Spectral Reweighting
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Spectral Reweighting at n={n_parent} (budget={remaining:.0f}s) ===")
    t_spec = min(remaining * 0.12, 200.0)
    if t_spec > 20 and best_seq is not None:
        a_out, _ = spectral_reweight(best_seq, t_spec, seed=5, verbose=True)
        try_update(a_out, f"specrew_n{n_parent}")
    
    # Phase 3: Combined max+variance descent
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Combined Descent at n={n_parent} (budget={remaining:.0f}s) ===")
    t_comb = min(remaining * 0.18, 300.0)
    if t_comb > 30 and best_seq is not None:
        a_out, _ = combined_descent(best_seq, t_comb, seed=7, verbose=True)
        try_update(a_out, f"comb_n{n_parent}")
    
    # Phase 4: LSE refinement
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LSE Refinement at n={n_parent} (budget={remaining:.0f}s) ===")
    t_lse = min(remaining * 0.15, 250.0)
    if t_lse > 30 and best_seq is not None:
        a_out, _ = lse_refine(best_seq, t_lse, T_start=50.0, T_end=0.3, n_stages=10, seed=11)
        try_update(a_out, f"lse_n{n_parent}")
    
    # Phase 5: Coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Coordinate Polish at n={n_parent} (budget={remaining:.0f}s) ===")
    t_coord = min(remaining * 0.10, 150.0)
    if t_coord > 15 and best_seq is not None:
        a_out, _ = coordinate_polish(best_seq, t_coord, seed=99, verbose=True)
        try_update(a_out, f"coord_n{n_parent}")
    
    # Phase 6: Multi-length exploration
    n_long = int(round(n_parent * 1.4))
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Longer length n={n_long} (budget={remaining:.0f}s) ===")
    
    if t_global is not None:
        pass
    
    t_md_long = min(remaining * 0.12, 180.0)
    if t_md_long > 30 and best_seq is not None:
        init_long = interpolate_to_length(best_seq, n_long)
        a_out, _ = variance_driven_descent(init_long, t_md_long, lr_schedule='exp',
                                            n_restarts=2, seed=3, verbose=True)
        try_update(a_out, f"vdace_n{n_long}")
    
    # Combined descent at longer length
    remaining = time_budget - (time.monotonic() - t_global)
    t_comb_long = min(remaining * 0.10, 150.0)
    if t_comb_long > 20 and best_seq is not None:
        init_long2 = interpolate_to_length(best_seq, n_long)
        a_out, _ = combined_descent(init_long2, t_comb_long, seed=13, verbose=True)
        try_update(a_out, f"comb_n{n_long}")
    
    # Try even longer
    n_long2 = int(round(n_parent * 1.8))
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6b: Very long n={n_long2} (budget={remaining:.0f}s) ===")
    t_long2 = min(remaining * 0.08, 120.0)
    if t_long2 > 30 and best_seq is not None:
        init_l2 = interpolate_to_length(best_seq, n_long2)
        a_out, _ = variance_driven_descent(init_l2, t_long2, lr_schedule='cosine',
                                            n_restarts=2, seed=17, verbose=True)
        try_update(a_out, f"vdace_n{n_long2}")
    
    # Phase 7: Final polish on best solution
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final Polish (budget={remaining:.0f}s) ===")
    if best_seq is not None:
        # Short combined descent
        t_f1 = min(remaining * 0.20, 100.0)
        if t_f1 > 15:
            a_out, _ = combined_descent(best_seq, t_f1, seed=55, verbose=False)
            try_update(a_out, "final_comb")
        
        # Spectral reweight
        remaining = time_budget - (time.monotonic() - t_global)
        t_f2 = min(remaining * 0.15, 80.0)
        if t_f2 > 10:
            a_out, _ = spectral_reweight(best_seq, t_f2, seed=66, verbose=False)
            try_update(a_out, "final_spec")
        
        # LSE
        remaining = time_budget - (time.monotonic() - t_global)
        t_f3 = min(remaining * 0.30, 120.0)
        if t_f3 > 15:
            a_out, _ = lse_refine(best_seq, t_f3, T_start=20.0, T_end=0.1, n_stages=8, seed=33)
            try_update(a_out, "final_lse")
        
        # Coordinate polish
        remaining = time_budget - (time.monotonic() - t_global)
        t_f4 = min(remaining * 0.35, 150.0)
        if t_f4 > 10:
            a_out, _ = coordinate_polish(best_seq, t_f4, seed=444, verbose=True)
            try_update(a_out, "final_coord")
    
    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            if fs < best_score:
                best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}")
    else:
        print("ERROR: No valid sequence found!")
        return [1.0]
    
    return [float(x) for x in best_seq]
