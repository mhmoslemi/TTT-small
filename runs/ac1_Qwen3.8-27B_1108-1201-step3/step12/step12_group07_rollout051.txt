    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:

    b_val = b_val / s

def nesterov_lp(a0, time_limit, solver=None, max_iters=1000000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y = y / sy
        else:
            y = a
        
        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha = 2.0 / (t_new + 1.0)
        
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa
        
        sc_new = score_normalized(a_new)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
        
        sc_cur = score_normalized(a)
        if sc_new > sc_cur * 1.001 and t_nest > 5.0:
            t_nest = 1.0
            a_prev = a_new.copy()
        else:
            a_prev = a.copy()
            t_nest = t_new
        
        a = a_new
    
    return best_a, best_score

def active_set_descent(a0, time_limit, solver=None, max_iters=5000):
    """Novel: LP-based descent targeting only active peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    peak_frac = 0.97
    step_size = 0.05
    stall_count = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if stall_count > 200:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Find active peaks
        active_mask = b >= max_b * peak_frac
        active = np.where(active_mask)[0]
        
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        if len(active) > 30:
            top_idx = np.argsort(b[active])[-30:]
            active = active[top_idx]
        
        # Build constraint matrix for active peaks
        T_act = np.zeros((len(active), n))
        for ii, k in enumerate(active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            T_act[ii, idx] = a[k - idx]
        
        # LP: minimize M s.t. T_act @ d <= M, sum(d)=0, |d| <= step_size * a
        d_var = cp.Variable(n)
        M_var = cp.Variable()
        constraints = [
            T_act @ d_var <= M_var,
            cp.sum(d_var) == 0,
            d_var >= -step_size * a,
            d_var <= step_size * a,
        ]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            if solver is None:
                solver = get_lp_solver()
            prob.solve(solver=solver)
        except Exception:
            break
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            stall_count += 1
            continue
        
        M_val = float(M_var.value)
        
        if M_val >= -1e-8:
            # No descent direction; widen active set
            peak_frac = min(peak_frac - 0.005, 0.85)
            step_size = max(step_size * 0.95, 0.005)
            stall_count += 1
            continue
        
        d_val = np.asarray(d_var.value, dtype=np.float64)
        
        # Take step
        a_new = a + step_size * d_val
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 0.01:
            stall_count += 1
            continue
        a_new = a_new / sn
        
        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
            # Slightly reduce step for convergence
            step_size = max(step_size * 0.999, 0.003)
        else:
            step_size *= 0.8
            stall_count += 1
        
        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            if elapsed > 15:
                print(f"    ActiveSet iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} peak_frac={peak_frac:.3f} step={step_size:.4f}")
    
    return best_a, best_score

def lse_objective_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp
    
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-20:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

def lbgfs_logsumexp(a0, time_limit, T_start=100.0, T_end=0.05, n_stages=20):

    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, stage_T, n)
            return obj
        
        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, stage_T, n)
            return grad
        
        iters = 30 if si < n_stages // 2 else 15
        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-22,
                         'gtol': 1e-18, 'maxfun': iters * 200,
                         'disp': False}
            )
            z_new = res.x
            a_new = softmax_from_z(z_new)
            sc = score_normalized(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            z = z_new
        except Exception:
            pass
    
    return best_a

def log_space_perturbation(a0, time_limit, n_trials=5000, perturbation_scale=0.01):
    """Random perturbations in log-space with acceptance criterion."""
    n = len(a0)

            a = a_new

                current_scale *= 0.9
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

            active = active[top_idx]

        T_act = np.zeros((len(active), n))
        for ii, k in enumerate(active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            T_act[ii, idx] = a[k - idx]

        d_var = cp.Variable(n)
        M_var = cp.Variable()
        constraints = [
            T_act @ d_var <= M_var,
            cp.sum(d_var) == 0,
            d_var >= -step_size * a,
            d_var <= step_size * a,
        ]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            if solver is None:
                solver = get_lp_solver()
            prob.solve(solver=solver)
        except Exception:
            break

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            stall_count += 1
            continue

        M_val = float(M_var.value)

        if M_val >= -1e-8:
            peak_frac = max(peak_frac - 0.003, 0.80)
            step_size = max(step_size * 0.95, 0.003)
            stall_count += 1
            if stall_count % 50 == 0:
                print(f"    ActiveSet stall at iter={it}, peak_frac={peak_frac:.3f}, trying wider set")
            continue

        d_val = np.asarray(d_var.value, dtype=np.float64)

        a_new = a + step_size * d_val
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 0.01:
            stall_count += 1
            continue
        a_new = a_new / sn

        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
            step_size = max(step_size * 0.998, 0.002)
            if peak_frac > 0.92:
                peak_frac = max(peak_frac - 0.001, 0.88)
        else:
            step_size *= 0.75
            stall_count += 1

        if (it + 1) % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ActiveSet iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} pf={peak_frac:.3f} step={step_size:.4f} stall={stall_count}")

    return best_a, best_score


def lse_obj_and_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-20:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def _lse_func(z_val, T_val, n):
    obj, _ = lse_obj_and_grad(z_val, T_val, n)
    return obj


def _lse_jac(z_val, T_val, n):
    _, grad = lse_obj_and_grad(z_val, T_val, n)
    return grad


def lbgfs_logsumexp(a0, time_limit, T_start=100.0, T_end=0.03, n_stages=20):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        iters = 25 if si < n_stages // 2 else 12
        try:
            res = minimize(
                fun=_lse_func, x0=z, args=(stage_T, n),
                jac=_lse_jac,
                method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-22,
                         'gtol': 1e-18, 'maxfun': iters * 200,
                         'disp': False}
            )
            z_new = res.x
            a_new = softmax_from_z(z_new)
            sc = score_normalized(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            z = z_new
        except Exception:
            pass

    return best_a


def log_space_perturbation(a0, time_limit, n_trials=10000, perturbation_scale=0.005):
    """Random perturbations in log-space with acceptance criterion."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(7777)
    current_scale = perturbation_scale
    no_improve = 0

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 1000:
            break

        n_perturb = max(1, n // 40)
        idx = rng.integers(0, n, size=n_perturb)
        z = z_from_a(a)
        z_perturbed = z.copy()
        z_perturbed[idx] += rng.normal(0, current_scale, size=n_perturb)

        a_new = softmax_from_z(z_perturbed)
        sc = score_normalized(a_new)

        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            current_scale = min(current_scale * 1.005, 0.03)
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                current_scale *= 0.92

    return best_a, best_score


def peak_multiplicative_refine(a0, time_limit, max_steps=30000):
    """Multiplicative weights update targeting top peaks with adaptive threshold."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = 0.03
    threshold_frac = 0.93
    last_improve = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve > 3000:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * threshold_frac

        topk = np.where(b >= threshold)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])

        grad = np.zeros(n)
        for k in topk:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += a[k - idx]
        grad /= len(topk)

        exponent = -eta * grad
        exponent -= exponent.max()
        new_a = a * np.exp(exponent)
        sa = new_a.sum()
        if sa < 1e-15:
            eta *= 0.5
            continue
        new_a = new_a / sa

        sc = score_normalized(new_a)
        if sc < best_score:
            best_score = sc
            best_a = new_a.copy()
            a = new_a
            last_improve = step
            eta = min(eta * 1.001, 0.08)
            threshold_frac = min(threshold_frac + 0.001, 0.97)
        else:
            eta *= 0.85
            if eta < 1e-10:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def do_try_update(best_state, arr, label):
    """Update best state dict if arr is better."""
    if arr is None:
        return
    arr = np.maximum(arr, 0).astype(np.float64)
    s = arr.sum()
    if s < 0.01:
        return
    arr = arr / s
    sc = evaluate_safe(arr)
    if np.isfinite(sc) and sc < best_state['score']:
        best_state['score'] = sc
        best_state['seq'] = arr.copy()
        best_state['n'] = len(arr)
        print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(arr)})")


def propose_candidate(seed=42, budget_s=1800.0):
    global _global_deadline
    rng = np.random.default_rng(seed)
    _global_deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_state = {'score': float('inf'), 'seq': None, 'n': 0}

    n_base = 1200
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_state['seq'] = p / p.sum()
            n_base = len(best_state['seq'])
            best_state['n'] = n_base
            best_state['score'] = evaluate_safe(best_state['seq'])
            print(f"Parent: n={n_base}, score={best_state['score']:.8f}")
    except Exception:
        pass

    if best_state['seq'] is None:
        n_base = 1200
        x = np.linspace(0, 1, n_base)
        best_state['seq'] = np.exp(-4.0 * (x - 0.4)**2) + 0.4 * np.exp(-4.0 * (x - 0.7)**2) + 0.2
        best_state['seq'] = np.maximum(best_state['seq'], 0.01)
        best_state['seq'] = best_state['seq'] / best_state['seq'].sum()
        best_state['n'] = n_base
        best_state['score'] = evaluate_safe(best_state['seq'])
        print(f"Default init: n={n_base}, score={best_state['score']:.8f}")

    # === Phase 1: Nesterov LP warmup (120s) ===
    print(f"\n=== Phase 1: Nesterov LP warmup (score={best_state['score']:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t1 = min(120.0, remaining * 0.07)
    if t1 > 10:
        a_out, _ = nesterov_lp(best_state['seq'], t1, solver=solver)
        do_try_update(best_state, a_out, "nesterov_p1")
    print(f"  Phase 1 done: best_score={best_state['score']:.8f}")

    # === Phase 2: Active-set descent (350s) ===
    print(f"\n=== Phase 2: Active-set descent (score={best_state['score']:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t2 = min(350.0, remaining * 0.22)
    if t2 > 20:
        a_act, _ = active_set_descent(best_state['seq'], t2, solver=solver, init_peak_frac=0.96)
        do_try_update(best_state, a_act, "active_set_p2")
    print(f"  Phase 2 done: best_score={best_state['score']:.8f}")

    # === Phase 3: LSE L-BFGS polishing (250s) ===
    print(f"\n=== Phase 3: LSE L-BFGS (score={best_state['score']:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t3 = min(250.0, remaining * 0.17)
    if t3 > 15:
        a_lse = lbgfs_logsumexp(best_state['seq'], t3, T_start=100.0, T_end=0.02, n_stages=25)
        do_try_update(best_state, a_lse, "lse_p3")
    print(f"  Phase 3 done: best_score={best_state['score']:.8f}")

    # === Phase 4: Log-space perturbation (180s) ===
    print(f"\n=== Phase 4: Log-space perturbation (score={best_state['score']:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t4 = min(180.0, remaining * 0.13)
    if t4 > 10:
        a_pert, _ = log_space_perturbation(best_state['seq'], t4, n_trials=15000, perturbation_scale=0.004)
        do_try_update(best_state, a_pert, "perturbation_p4")
    print(f"  Phase 4 done: best_score={best_state['score']:.8f}")

    # === Phase 5: Peak multiplicative refinement (150s) ===
    print(f"\n=== Phase 5: Peak multiplicative (score={best_state['score']:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t5 = min(150.0, remaining * 0.12)
    if t5 > 10:
        a_pm, _ = peak_multiplicative_refine(best_state['seq'], t5, max_steps=20000)
        do_try_update(best_state, a_pm, "multiplicative_p5")
    print(f"  Phase 5 done: best_score={best_state['score']:.8f}")

    # === Phase 6: Nesterov LP round 2 (300s) ===
    print(f"\n=== Phase 6: Nesterov LP round 2 (score={best_state['score']:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t6 = min(300.0, remaining * 0.20)
    if t6 > 20:
        a_out2, _ = nesterov_lp(best_state['seq'], t6, solver=solver)
        do_try_update(best_state, a_out2, "nesterov_p6")
    print(f"  Phase 6 done: best_score={best_state['score']:.8f}")

    # === Phase 7: Active-set final (150s) ===
    print(f"\n=== Phase 7: Active-set final (score={best_state['score']:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t7 = min(150.0, remaining * 0.12)
    if t7 > 10:
        a_act2, _ = active_set_descent(best_state['seq'], t7, solver=solver, init_peak_frac=0.98)
        do_try_update(best_state, a_act2, "active_set_p7")
    print(f"  Phase 7 done: best_score={best_state['score']:.8f}")

    # === Phase 8: Final LSE polish (100s) ===
    print(f"\n=== Phase 8: Final LSE (score={best_state['score']:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t8 = min(100.0, remaining * 0.10)
    if t8 > 10:
        a_lse2 = lbgfs_logsumexp(best_state['seq'], t8, T_start=40.0, T_end=0.01, n_stages=15)
        do_try_update(best_state, a_lse2, "lse_final")
    print(f"  Phase 8 done: best_score={best_state['score']:.8f}")

    # === Phase 9: Length exploration ===
    print(f"\n=== Phase 9: Length exploration (score={best_state['score']:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    if remaining > 60:
        for n_try in [best_state['n'] + 150, best_state['n'] - 100, best_state['n'] + 300]:
            remaining = _global_deadline - time.monotonic()
            if remaining < 30:
                break
            if n_try < 200:
                continue
            t9 = min(50.0, remaining * 0.35)
            if t9 < 15:
                break
            print(f"  Trying n={n_try}, budget={t9:.0f}s")
            init_try = interpolate_sequence(best_state['seq'], n_try)
            a_try, _ = nesterov_lp(init_try, t9, solver=solver, max_iters=50000)
            do_try_update(best_state, a_try, f"len_{n_try}")

    # Final verification
    if best_state['seq'] is not None:
        fs = evaluate_safe(best_state['seq'])
        if np.isfinite(fs):
            best_state['score'] = fs

    print(f"\nFINAL SCORE: {best_state['score']:.8f} (n={len(best_state['seq'])})")
    return [float(x) for x in best_state['seq']]
