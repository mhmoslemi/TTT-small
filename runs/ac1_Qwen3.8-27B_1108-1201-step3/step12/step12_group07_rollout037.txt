    for solver_name in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def structured_init(n, rng):
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = 1.0 + 0.8 * (2.0 * x - 1.0)
    elif choice == 1:
        a = x + 0.1
    elif choice == 2:
        a = 2.0 * np.exp(-3.0 * x) + 0.3
    elif choice == 3:
        a = np.exp(-4.0 * (x - 0.3) ** 2) + 0.2
    elif choice == 4:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * (x + 0.1)) + 0.15 * np.cos(4 * np.pi * x)
    elif choice == 5:
        a = np.exp(-8.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-8.0 * (x - 0.7) ** 2) + 0.1
    elif choice == 6:
        a = (x + 0.05) ** 0.7
    elif choice == 7:
        a = 0.8 + 0.4 * np.sin(2 * np.pi * x + 0.5)
    elif choice == 8:
        # Two-peak asymmetric
        a = np.exp(-6*(x-0.2)**2) + 0.5*np.exp(-6*(x-0.8)**2) + 0.05
    else:
        # Quartic taper
        a = 1.0 - 0.6*(2*x-1)**4
    a = np.maximum(a, 0.01)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def spectral_init(a0, n_iters=200, T_flat=1.0):
    """Gerchberg-Saxton style spectral equalization for non-negative sequences."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    
    for it in range(n_iters):
        # FFT
        A = rfft(a, fft_size)
        # Target: flatten the magnitude
        mag = np.abs(A)
        # Keep DC, flatten the rest toward geometric mean
        if mag[0] > 1e-30:
            rest_mag = mag[1:]
            rest_mag[rest_mag < 1e-30] = 1e-30
            target_mag = np.ones_like(mag)
            target_mag[0] = mag[0]  # keep DC
            # Smooth target: blend current with flat
            target_mag[1:] = np.sqrt(mag[1:] * 1.0)  # geometric mean with 1
            new_mag = mag * 0.5 + target_mag * 0.5
            # Preserve phase
            phase = A / np.where(mag > 1e-30, mag, 1.0)
            A_new = new_mag * phase
        else:
            A_new = A
        # IFFT
        a_new = irfft(A_new, fft_size)[:n]
        # Project onto non-negative
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def multi_peak_subgrad_nesterov(a0, time_limit, n_peaks=10, max_steps=200000):
    """Nesterov mirror descent in z-space with multi-peak subgradient."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()
    a_prev = a.copy()
    z_prev = z.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    eta = 0.5
    last_improve = 0
    stall_count = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve > 5000:
            break

        b = fast_convolve(a)
        max_b = b.max()
        
        # Multi-peak: use top peaks with softmax weights
        sorted_b = np.sort(b)[::-1]
        n_use = min(n_peaks, len(b))
        threshold = sorted_b[n_use - 1] if n_use > 1 else sorted_b[0] * 0.9
        topk = np.where(b >= threshold)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])
        
        # Compute gradient in a-space
        grad_a = np.zeros(n)
        for k in topk:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * a[k - idx]
        grad_a /= len(topk)

        # Convert to z-space gradient
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)

        # Nesterov extrapolation in z-space
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            z_extrap = z + momentum * (z - z_prev)
        else:
            z_extrap = z

        z_new = z_extrap - eta * grad_z
        a_new = softmax_from_z(z_new)

        sc = score_normalized(a_new)
        if sc < best_score * (1 - 1e-10):
            best_score = sc
            best_a = a_new.copy()
            a_prev = a
            z_prev = z
            a = a_new
            z = z_new
            t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
            last_improve = step
            stall_count = 0
            eta = min(eta * 1.001, 1.0)
        else:
            stall_count += 1
            if sc > best_score * 1.001:
                t_nest = 1.0
                eta *= 0.8
            else:
                t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
                z = z_new
                a = a_new
                z_prev = z_prev
            if stall_count > 200:
                eta *= 0.5

    return best_a, best_score


def nesterov_lp_adaptive(a0, time_limit, solver=None, max_iters=1000000):
    """Nesterov LP with adaptive restarts and warm-start tracking."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    last_print = t0
    restart_count = 0
    score_history = []
    last_improve_iter = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Extrapolation
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y = y / sy
        else:
            y = a

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha = 2.0 / (t_new + 1.0)

        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        sc_new = score_normalized(a_new)

        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            last_improve_iter = it

        score_history.append(sc_new)
        # Keep only last 50 for memory
        if len(score_history) > 50:
            score_history = score_history[-50:]

        # Adaptive restart: check if we're stalling
        sc_cur = score_normalized(a)
        recent_avg = np.mean(score_history[-10:]) if len(score_history) >= 10 else sc_cur
        
        if (sc_new > sc_cur * 1.001 and t_nest > 5.0) or \
           (len(score_history) >= 20 and recent_avg > best_score * 1.0002 and it - last_improve_iter > 200):
            t_nest = 1.0
            a_prev = a_new.copy()
            restart_count += 1
        else:
            a_prev = a.copy()
            t_nest = t_new

        a = a_new

        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 10:
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} t={t_nest:.1f} restarts={restart_count}")
                last_print = elapsed

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-20:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_logsumexp(a0, time_limit, T_start=100.0, T_end=0.05, n_stages=20):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, stage_T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, stage_T, n)
            return grad

        # More iterations for early stages, fewer for later
        frac = si / max(n_stages - 1, 1)
        iters = int(50 * (1.0 - 0.5 * frac))
        iters = max(iters, 15)
        
        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-22,
                         'gtol': 1e-18, 'maxfun': iters * 300,
                         'disp': False}
            )
            z_new = res.x
            a_new = softmax_from_z(z_new)
            sc = score_normalized(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            z = z_new
            if (si + 1) % 5 == 0:
                print(f"    LSE stage={si+1}/{n_stages} T={stage_T:.3f}: score={best_score:.8f}")
        except Exception:
            pass

    return best_a


def peak_targeted_refine(a0, time_limit, max_steps=50000):
    """Multiplicative weights update targeting top peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = 0.03
    last_improve_step = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve_step > 8000:
            break

        b = fast_convolve(a)
        max_b = b.max()
        # Use a wider threshold to capture more peaks
        threshold = max_b * 0.85
        topk = np.where(b >= threshold)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])

        # Compute gradient toward peaks
        grad = np.zeros(n)
        for k in topk:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += a[k - idx]
        grad /= len(topk)

        # Multiplicative update
        exponent = -eta * grad
        exponent -= exponent.max()
        new_a = a * np.exp(exponent)
        sa = new_a.sum()
        if sa < 1e-15:
            eta *= 0.5
            continue
        new_a = new_a / sa

        sc = score_normalized(new_a)
        if sc < best_score:
            best_score = sc
            best_a = new_a.copy()
            a = new_a
            last_improve_step = step
            eta = min(eta * 1.003, 0.1)
        else:
            eta *= 0.88
            if eta < 1e-12:
                break

    return best_a, best_score


def frank_wolfe_refine(a0, time_limit, max_iters=50000):
    """Frank-Wolfe with away steps for the non-convex min-peak problem."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_improve = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if it - last_improve > 5000:
            break

        b = fast_convolve(a)
        k_star = int(np.argmax(b))

        # Subgradient
        g = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        g[idx] = 2.0 * a[k_star - idx]

        # FW step: find toward and away vertices
        # Toward: argmin g (prefer out-of-bounds indices where g=0)
        i_toward = int(np.argmin(g))
        
        # Away: argmax g among support
        support = np.where(a > 1e-10)[0]
        if len(support) > 1:
            i_away = support[np.argmax(g[support])]
            
            # Away-step FW
            # a_new = (1-t)*a + t*(e_toward - e_away) / (1 - a_away) approximately
            # Simpler: just do a coordinate swap
            amount = min(a[i_away], a[i_toward] + 0.1) * 0.1
            a_trial = a.copy()
            a_trial[i_away] -= amount
            a_trial[i_toward] += amount
            a_trial = np.maximum(a_trial, 0)
            sa = a_trial.sum()
            if sa > 1e-15:
                a_trial = a_trial / sa
                sc = score_normalized(a_trial)
                if sc < best_score:
                    best_score = sc
                    best_a = a_trial.copy()
                    a = a_trial
                    last_improve = it
                else:
                    # Regular FW step
                    t_fw = 2.0 / (it + 2)
                    a_trial2 = (1 - t_fw) * a + t_fw * np.zeros(n)
                    a_trial2[i_toward] += t_fw
                    a_trial2 = np.maximum(a_trial2, 0)
                    sa2 = a_trial2.sum()
                    if sa2 > 1e-15:
                        a_trial2 = a_trial2 / sa2
                        sc2 = score_normalized(a_trial2)
                        if sc2 < best_score:
                            best_score = sc2
                            best_a = a_trial2.copy()
                            a = a_trial2
                            last_improve = it
        else:
            # Single support: just move toward
            t_fw = 2.0 / (it + 2)
            a_trial = (1 - t_fw) * a
            a_trial[i_toward] += t_fw
            a_trial = np.maximum(a_trial, 0)
            sa = a_trial.sum()
            if sa > 1e-15:
                a_trial = a_trial / sa
                sc = score_normalized(a_trial)
                if sc < best_score:
                    best_score = sc
                    best_a = a_trial.copy()
                    a = a_trial
                    last_improve = it

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    print(f"  Phase 5 done: best_score={best_score:.8f}")

    # === Phase 6: Frank-Wolfe + LSE final polish (150s) ===
    print(f"\n=== Phase 6: Final refinement (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    if remaining > 40:
        t6a = min(60.0, remaining * 0.3)
        if t6a > 10:
            a_fw, _ = frank_wolfe_refine(best_seq, t6a, max_iters=30000)
            try_update(a_fw, "frank_wolfe")
        
        remaining = _global_deadline - time.monotonic()
        if remaining > 20:
            t6b = min(80.0, remaining * 0.5)
            if t6b > 10:
                a_lse2 = lbgfs_logsumexp(best_seq, t6b, T_start=30.0, T_end=0.01, n_stages=20)
                try_update(a_lse2, "lse_final")
        
        remaining = _global_deadline - time.monotonic()
        if remaining > 15:
            a_pt2, _ = peak_targeted_refine(best_seq, min(40.0, remaining), max_steps=20000)
            try_update(a_pt2, "peak_targeted_r2")
    print(f"  Phase 6 done: best_score={best_score:.8f}")

    # === Phase 7: Length exploration ===
    print(f"\n=== Phase 7: Length exploration (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    if remaining > 80:
        # Try a few nearby lengths
        length_tries = []
        for delta in [-200, 200, 400, 800]:
            nt = best_n + delta
            if 200 <= nt <= 5000:
                length_tries.append(nt)
        
        for n_try in length_tries[:3]:  # Try up to 3
            remaining = _global_deadline - time.monotonic()
            if remaining < 30:
                break
            t7 = min(50.0, remaining * 0.35)
            if t7 < 15:
                break
            print(f"  Trying n={n_try}, budget={t7:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            # Quick subgradient first
            a_quick, _ = multi_peak_subgrad_nesterov(init_try, min(15.0, t7 * 0.3), n_peaks=10, max_steps=50000)
            a_try, _ = nesterov_lp_adaptive(a_quick, t7 - 15.0, solver=solver, max_iters=50000)
            try_update(a_try, f"len_{n_try}")

    # === Phase 8: Final multi-peak sweep ===
    remaining = _global_deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Phase 8: Final multi-peak sweep ===")
        t8 = min(40.0, remaining * 0.8)
        a_final, _ = multi_peak_subgrad_nesterov(best_seq, t8, n_peaks=20, max_steps=100000)
        try_update(a_final, "final_sweep")
        a_pt_final, _ = peak_targeted_refine(best_seq, min(20.0, _global_deadline - time.monotonic() - 5), max_steps=10000)
        try_update(a_pt_final, "final_pt")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
