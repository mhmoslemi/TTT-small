2. Multi-stage LogSumExp L-BFGS with diagonal Hessian correction (main polish)

import numpy as np

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    try:

        if len(a_list) == 0:

        sc = evaluate_sequence(a_list)

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix_rows(a_fixed, k_indices):
    """Build rows of the convolution matrix for specific lag indices."""
    n = len(a_fixed)
    rows = []
    for k in k_indices:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        row = np.zeros(n)
        row[idx] = a_fixed[k - idx]
        rows.append(row)
    return np.array(rows)


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:

    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp(a0, time_limit, solver=None, max_iters=100000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    last_print = t0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y = y / sy
        else:
            y = a

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha = 2.0 / (t_new + 1.0)
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        sc_new = score_normalized(a_new)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()

        sc_cur = score_normalized(a)
        if sc_new > sc_cur * 1.0005 and t_nest > 5.0:
            t_nest = 1.0
            a_prev = a_new.copy()
        else:
            a_prev = a.copy()
            t_nest = t_new
        a = a_new

        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 10:
                print(f"    LP iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} t={t_nest:.1f}")
                last_print = elapsed

    return best_a, best_score


def lse_objective_grad_hess_diag(z_in, T_val, n):
    """Compute LSE objective, gradient, and diagonal Hessian approximation."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    # Gradient w.r.t. a
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-20:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    # Gradient w.r.t. z
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    # Diagonal Hessian approximation (Gauss-Newton)
    # d2 f / dz_i^2 ≈ a_i * (grad_a_i - S)^2 + a_i * grad_a_i - ...
    # Simplified: use the variance of the "exposure"
    diag_hess = a * (grad_a - S) ** 2 + a * a * (grad_a ** 2).sum() - a * (a * grad_a).sum()
    # Better approximation: for softmax, the Hessian involves the covariance
    # hess_diag ≈ a_i * (grad_a_i - S)^2 (diagonal of the "natural gradient" metric)
    diag_hess = np.maximum(a * (grad_a - S) ** 2, 1e-20)

    return obj, grad_z, diag_hess


def lbgfs_logsumexp_enhanced(a0, time_limit, T_start=200.0, T_end=0.005, n_stages=30, use_diagonal_corr=True):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        if use_diagonal_corr and stage_T < 5.0:
            # Use gradient with diagonal scaling for low temperatures
            def func_staged(z_val):
                obj, _, _ = lse_objective_grad_hess_diag(z_val, stage_T, n)
                return obj
            def jac_staged(z_val):
                _, grad, dh = lse_objective_grad_hess_diag(z_val, stage_T, n)
                if use_diagonal_corr:
                    return grad / np.sqrt(dh)
                return grad
        else:
            def func_staged(z_val):
                obj, grad, _ = lse_objective_grad_hess_diag(z_val, stage_T, n)
                return obj
            def jac_staged(z_val):
                _, grad, _ = lse_objective_grad_hess_diag(z_val, stage_T, n)
                return grad

        # More iterations for low temperature stages
        if stage_T > 10.0:
            iters = 30
        elif stage_T > 1.0:
            iters = 50
        else:
            iters = 80

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-24,
                         'gtol': 1e-20, 'maxfun': iters * 500,
                         'disp': False, 'maxls': 50}
            )
            z_new = res.x
            a_new = softmax_from_z(z_new)
            sc = score_normalized(a_new)
            z = z_new
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            if (si + 1) % 5 == 0:
                print(f"    LSE stage={si+1}/{n_stages} T={stage_T:.4f}: best={best_score:.8f}")
        except Exception as e:
            print(f"    LSE stage {si+1} error: {e}")

    return best_a, best_score


def peak_targeted_valley(a0, time_limit, max_steps=100000):
    """Peak-targeted with valley-filling and adaptive threshold."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = 0.03
    last_improve = 0
    step = 0

    while step < max_steps:
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve > 8000:
            break

        b = fast_convolve(a)
        max_b = b.max()
        min_b = b.min()

        # Adaptive threshold: start wide, tighten over time
        progress = min(step / max_steps, 1.0)
        threshold = max_b * (0.85 + 0.14 * progress)

        topk = np.where(b >= threshold)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])

        # Peak gradient
        grad_peak = np.zeros(n)
        for k in topk:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_peak[idx] += a[k - idx]
        grad_peak /= len(topk)

        # Valley gradient (elements that would raise low convolution values)
        valley_thresh = min_b + 0.1 * (max_b - min_b)
        valleys = np.where(b <= valley_thresh)[0]
        if len(valleys) > 0 and len(valleys) < n:
            grad_valley = np.zeros(n)
            for k in valleys[:min(len(valleys), 20)]:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_valley[idx] += a[k - idx]
            grad_valley /= min(len(valleys), 20)
            # Combined: reduce peak contribution, increase valley contribution
            grad = grad_peak - 0.3 * grad_valley
        else:
            grad = grad_peak

        # Normalize gradient
        grad_norm = np.abs(grad).max()
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm

        # Multiplicative update
        exponent = -eta * grad
        exponent -= exponent.max()
        new_a = a * np.exp(exponent)
        sa = new_a.sum()
        if sa < 1e-15:
            eta *= 0.5
            step += 1
            continue
        new_a = new_a / sa

        sc = score_normalized(new_a)
        if sc < best_score:
            best_score = sc
            best_a = new_a.copy()
            a = new_a
            last_improve = step
            eta = min(eta * 1.003, 0.12)
        else:
            eta *= 0.9

        step += 1

        if step % 5000 == 0:
            print(f"    PeakValley step={step} ({time.monotonic()-t0:.0f}s): best={best_score:.8f} eta={eta:.4f}")

    return best_a, best_score


def fista_smoothed(a0, time_limit, max_iters=200000):
    """FISTA with smoothed max (piecewise linear approximation)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    t_k = 1.0
    best_a = a.copy()
    best_score = score_normalized(a)
    L = 4.0 * n * n  # Upper bound on Lipschitz constant
    t0 = time.monotonic()
    step = 0

    while step < max_iters:
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(y)
        max_b = b.max()
        # Use smoothed subgradient: weight by proximity to max
        weights = np.exp(-5.0 * (max_b - b))
        weights /= weights.sum()

        grad = np.zeros(n)
        for k in range(len(b)):
            if weights[k] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * n * weights[k] * y[k - idx]

        z = y - grad / L
        a_new = project_simplex(z)

        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        y = project_simplex(y)
        a = a_new
        t_k = t_new
        step += 1

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 10)

    elif choice == 1:
        a = np.exp(-2.0 * (x - 0.5) ** 2) + 0.1
    elif choice == 2:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * x) + 0.1 * np.sin(2 * np.pi * 7 * x)
    elif choice == 3:
        a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.05)
    elif choice == 4:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * (x + 0.1)) + 0.2 * np.cos(2 * np.pi * 5 * x)
    elif choice == 5:
        a = np.exp(-5.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-5.0 * (x - 0.7) ** 2) + 0.05

        a = (1.0 + np.sin(2 * np.pi * 2 * x)) ** 0.5

        a = 1.0 + 0.4 * np.cos(2 * np.pi * x) * np.exp(-3.0 * x)

    a = np.maximum(a, 0.01)

    # === Phase 7: Final ultra-fine polish (remaining time) ===
    print(f"\n=== Phase 7: Final polish (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    if remaining > 15:
        t7 = min(remaining * 0.6, 100.0)
        if t7 > 10:
            a_final = lbgfs_logsumexp_enhanced(best_seq, t7, T_start=10.0, T_end=0.001, n_stages=20, use_diagonal_corr=True)
            try_update(a_final, "final_lse")
        remaining = _global_deadline - time.monotonic()
        if remaining > 10:
            t7b = min(remaining * 0.5, 50.0)
            if t7b > 5:
                a_pt2, _ = peak_targeted_valley(best_seq, t7b, max_steps=50000)
                try_update(a_pt2, "final_peak")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
