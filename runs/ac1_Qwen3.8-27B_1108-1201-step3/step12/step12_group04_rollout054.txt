def softmax_fn(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def lse_obj_grad(z, T):
    a = softmax_fn(z)
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    obj_val = (max_Tb + np.log(exp_Tb.sum())) / T
    
    # Gradient with respect to a
    w = exp_Tb / exp_Tb.sum()
    # grad_a[i] = 2 * sum_k w[k] * a[k-i] (valid k)
    a_rev = a[::-1].copy()
    conv_len = n + (2*n-1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    A_rev = rfft(a_rev, fft_size)
    conv = irfft(W * A_rev, fft_size)
    grad_a = 2.0 * conv[n-1:n-1+n]
    
    # Transform to z-space
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj_val, grad_z

def peak_shave(a0, time_limit, solver, K=5, max_iters=100):
    a = normalize(a0)
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:K]
        
        # For each peak, try to reduce it
        improved = False
        for k in top_k:
            # Find which indices contribute to this peak
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            
            # Find the largest contributor
            max_contrib_idx = np.argmax(contrib)
            src = idx[max_contrib_idx]
            src_val = a[src]
            if src_val < 1e-12:
                continue
            
            # Find target: position with least contribution to all top peaks
            def target_cost(j):
                total = 0.0
                for k2 in top_k:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    if js2 <= j <= je2:
                        partner = k2 - j
                        if 0 <= partner < n:
                            total += a[j] * a[partner]
                return total + 0.001 * a[j]
            
            # Try a few targets
            candidates = np.argsort([target_cost(j) for j in range(n)])
            for dst in candidates[:5]:
                if dst == src:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.05]:
                    amt = src_val * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-15:
                        continue
                    a_trial = normalize(a_trial)
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Random perturbation
            rng = np.random.default_rng(it + 42)
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:

            elapsed = time.monotonic() - t0
            print(f"  PS it={it} ({elapsed:.0f}s): score={best_score:.8f}")

    size = 2 * n - 1

            break

        # Allocate more iterations to later (colder) stages
        progress = si / max(1, n_stages - 1)
        iters_per_stage = max(8, int(10 + 50 * progress))

        def _func(z_in):
            val, _ = _lse_obj_grad(z_in, stage_T)
            return val

        def _jac(z_in):
            _, g = _lse_obj_grad(z_in, stage_T)
            return g

        try:
            res = minimize(
                fun=_func, x0=z, jac=_jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-16, 'maxfun': iters_per_stage * 400,
                         'disp': False}
            )
            z = res.x
            a = _softmax(z)
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        except Exception:
            pass

        if si % 6 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LSE stage={si}/{n_stages} T={stage_T:.4f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def peak_shave(a0, time_limit, seed=42, top_K=6, max_iters=5000):
    """
    Greedy peak shaving: iteratively find the peak in a*a, identify the
    largest contributor, and move mass to a position that minimizes
    damage to other peaks.
    """
    a = normalize(a0)
    n = len(a)
    rng = np.random.default_rng(seed)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:top_K]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            # Top 3 contributors to this peak
            top_contrib = np.argsort(contrib)[::-1][:3]
            for tc in top_contrib:
                src = int(idx[tc])
                src_val = a[src]
                if src_val < 1e-14:
                    continue

                # Score target positions by their total contribution to all top peaks
                def _target_cost(j):
                    total = 0.0
                    for k2 in top_k:
                        js2 = max(0, k2 - n + 1)
                        je2 = min(n - 1, k2)
                        if js2 <= j <= je2:
                            partner = k2 - j
                            if 0 <= partner < n:
                                total += a[j] * a[partner]
                    return total + 0.005 * a[j]

                # Find best targets (vectorized sort)
                costs = np.array([_target_cost(j) for j in range(n)])
                candidates = np.argsort(costs)[:8]

                for dst in candidates:
                    if dst == src:
                        continue
                    for frac in [0.003, 0.008, 0.015, 0.03, 0.06]:
                        amt = src_val * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < -1e-15:
                            continue
                        a_trial = normalize(a_trial)
                        sc = compute_score(a_trial)
                        if sc < best_score - 1e-15:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random exploration
            for _ in range(3):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-12:
                    frac = rng.uniform(0.001, 0.04)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= -1e-15:
                        a_trial = normalize(a_trial)
                        sc = compute_score(a_trial)
                        if sc < best_score:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            stall = 0
                            break

            if stall > 500:
                break

        if it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  PS it={it} stall={stall} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate a diverse structured initialization."""
    choice = int(rng.integers(0, 6))
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.exp(-3.0 * x ** 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(3 * x) + 0.1 * np.cos(7 * x)
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-5.0 * (x - 0.3) ** 2) + 0.6 * np.exp(-5.0 * (x - 0.7) ** 2)
    else:
        x = np.linspace(0, 1, n)
        a = np.maximum(0, 1 - 2 * np.abs(x - 0.5)) + 0.15 * np.sin(6 * np.pi * x)
        a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate():
    """
    Main search function. Multi-phase optimization:
    Phase 1: Nesterov LP at parent length (workhorse)
    Phase 2: LogSumExp gradient refinement
    Phase 3: Peak shaving
    Phase 4: Multi-length exploration
    Phase 5: Final polish
    """
    rng = np.random.default_rng(2024)
    t_global = time.monotonic()
    time_budget = 1740.0  # Leave 60s buffer

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    best_seq = None
    best_score = float('inf')
    n_base = 1600

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            n_base = len(parent)
            p_score = compute_score(parent)
            print(f"Parent: n={n_base}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        print("No valid parent, using structured init")

    def _try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = normalize(a)
        sc = compute_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        best_seq = structured_init(n_base, rng)
        best_score = compute_score(best_seq)
        print(f"Init: n={n_base}, score={best_score:.8f}")

    # === Phase 1: Nesterov LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Nesterov LP n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.35, 550.0)
    if t1 > 60:
        a_out, _ = nesterov_lp(best_seq, t1, solver)
        _try_update(a_out, f"nlp_n{n_base}")

    # === Phase 2: LogSumExp ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LogSumExp n={n_base} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 15:
        a_out, _ = lse_optimize(best_seq, t2, T_start=150.0, T_end=0.01, n_stages=30)
        _try_update(a_out, f"lse_n{n_base}")

    # === Phase 3: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Peak shaving n={n_base} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 10:
        a_out, _ = peak_shave(best_seq, t3, seed=777, top_K=6)
        _try_update(a_out, f"ps_n{n_base}")

    # === Phase 4: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Multi-length (budget={remaining:.0f}s) ===")
    candidate_lengths = []
    for offset in [-200, -100, 100, 200, 400]:
        nl = n_base + offset
        if 800 <= nl <= 3200:
            candidate_lengths.append(nl)
    candidate_lengths = sorted(set(candidate_lengths), key=lambda x: abs(x - n_base))
    time_per_len = min(remaining * 0.65 / max(1, len(candidate_lengths)), 180.0)

    for nl in candidate_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 30:
            break
        tl = min(time_per_len, remaining * 0.35)
        if tl < 15:
            continue
        print(f"  Multi-length: n={nl} (time={tl:.0f}s)")
        init_nl = interpolate_seq(best_seq, nl)

        # Quick Nesterov
        t_lp = tl * 0.6
        a_out, _ = nesterov_lp(init_nl, t_lp, solver)

        # Quick LSE
        rem2 = time_budget - (time.monotonic() - t_global)
        t_lse = max(5.0, min(tl * 0.35, rem2 * 0.1))
        if t_lse > 5 and a_out is not None:
            a_out2, _ = lse_optimize(a_out, t_lse, T_start=80.0, T_end=0.05, n_stages=12)
            _try_update(a_out2, f"lse_n{nl}")
        _try_update(a_out, f"nlp_n{nl}")

    # === Phase 5: Final refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Final refinement n={n_best} (budget={remaining:.0f}s) ===")

    # More aggressive LSE with very low temperature
    t5a = min(remaining * 0.35, 200.0)
    if t5a > 10:
        a_out, _ = lse_optimize(best_seq, t5a, T_start=30.0, T_end=0.002, n_stages=20)
        _try_update(a_out, f"lse_refine_n{n_best}")

    # Peak shaving again
    remaining = time_budget - (time.monotonic() - t_global)
    t5b = min(remaining * 0.30, 150.0)
    if t5b > 10:
        a_out, _ = peak_shave(best_seq, t5b, seed=1234, top_K=8)
        _try_update(a_out, f"ps_refine_n{n_best}")

    # One more LSE at the end
    remaining = time_budget - (time.monotonic() - t_global)
    t5c = min(remaining * 0.25, 100.0)
    if t5c > 5:
        a_out, _ = lse_optimize(best_seq, t5c, T_start=10.0, T_end=0.001, n_stages=10)
        _try_update(a_out, "final_lse")

    # Final check
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return result
