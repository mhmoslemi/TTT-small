import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def normalize_pos(a):
    """Ensure non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_lse_obj_grad(a, T, topk):
    """
    Compute top-k LSE objective and gradient w.r.t. a.
    obj = (1/T)*log(sum_{k in top_k} exp(T*c_k / (sum_a)^2))
    Actually just (1/T)*log(sum_{k in top_k} exp(T*c_k)) for normalized a.
    """
    n = len(a)
    c = fast_autocorr(a)
    k_use = min(topk, len(c))
    top_idx = np.argsort(c)[::-1][:k_use]
    top_vals = c[top_idx]

    Tc = T * top_vals
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    lse_val = (max_Tc + np.log(exp_Tc.sum())) / T
    w = exp_Tc / exp_Tc.sum()

    # grad_a[i] = 2 * sum_K w_K * a[K-i] for valid K-i
    grad_a = np.zeros(n)
    for idx in range(k_use):
        K = top_idx[idx]
        wK = w[idx]
        js = max(0, K - (n - 1))
        je = min(n - 1, K)
        if js <= je:
            idx_range = np.arange(js, je + 1)
            grad_a[idx_range] += 2.0 * wK * a[K - idx_range]

    return lse_val, grad_a


def topk_lse_grad_z(z, T, topk):
    """Objective and gradient in z-space for top-k LSE."""
    a = softmax_fn(z)
    n = len(a)
    obj, grad_a = compute_lse_obj_grad(a, T, topk)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.MOSEK, cp.GLPK]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.PDLP, cp.CLARABEL, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

    Nesterov-accelerated LP best-response with adaptive restart.
    """

    a = normalize_pos(a0)

    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    avg_long = np.zeros(n)
    n_long = 0
    avg_short = np.zeros(n)
    n_short = 0
    WINDOW_SHORT = 50
    WINDOW_LONG = 300

    short_buf = [None] * WINDOW_SHORT
    short_idx = 0
    stall_count = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)

        y_new = y_new / y_sum if y_sum > 1e-15 else b_star.copy()

        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            a_new = a_new / s_new if s_new > 1e-15 else y_new.copy()

        y = y_new
        a = a_new

        avg_long += a
        n_long += 1

        if short_buf[short_idx] is not None:
            avg_short -= short_buf[short_idx]
        short_buf[short_idx] = a.copy()
        avg_short += a
        n_short += 1
        short_idx = (short_idx + 1) % WINDOW_SHORT

        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if n_long % 50 == 0 and n_long >= 100:
            avg_check = avg_long / n_long
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
                stall_count = 0

        if n_short >= WINDOW_SHORT and n_long % 50 == 0:
            avg_s = avg_short / min(n_short, WINDOW_SHORT)
            score_s = 2.0 * n * fast_autocorr(avg_s).max()
            if score_s < best_score:
                best_score = score_s
                best_a = avg_s.copy()
                stall_count = 0

        if stall_count > 1000 and t > 500:
            a = best_a.copy()
            y = best_a.copy()
            avg_long = np.zeros(n)
            n_long = 0
            avg_short = np.zeros(n)
            n_short = 0
            short_buf = [None] * WINDOW_SHORT
            short_idx = 0
            stall_count = 0
            if t % 5000 == 0:
                print(f"      NLP restart at t={t}, best={best_score:.6f}")

        if t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}")

    return best_a, best_score


def lse_adam_optimize(a0, time_limit, T_start=150.0, T_end=0.2, n_stages=30, topk=12):
    """
    Improved Adam-optimized top-k LSE with cosine LR and adaptive topk.
    """
    n = len(a0)
    a_init = normalize_pos(a0)
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()
    best_lse = float('inf')

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit - 10:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        # Cosine learning rate schedule
        progress = si / max(1, n_stages - 1)
        base_lr = 8e-3 * min(1.0, 100.0 / n)
        lr = base_lr * (0.5 + 0.5 * np.cos(np.pi * progress))
        lr = max(lr, base_lr * 0.05)

        # Adaptive topk: more peaks at high T, fewer at low T
        cur_topk = max(4, int(topk * (0.5 + 0.5 * (1 - progress))))

        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.85, 0.999, 1e-8

        max_it = max(20, int(min(remaining, 100.0) * 4))
        stage_best = best_score

        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 5:
                break

            obj, grad_z = topk_lse_grad_z(z, stage_T, cur_topk)

            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            if (it + 1) % 10 == 0:
                a = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    stage_best = score
                    best_lse = obj
                if score < stage_best - 0.001 and (it + 1) % 50 == 0:
                    z = np.log(np.maximum(best_a, 1e-30))
                    m_z = np.zeros(n)
                    v_z = np.zeros(n)
                    t_adam = 0

        a_stage = softmax_fn(z)
        score_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()

        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE stage {si}/{n_stages} (T={stage_T:.1f}, topk={cur_topk}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def ftrl_subgradient_optimize(a0, time_limit, eta0=0.1, decay=0.9995, topk=8):
    """
    FTRL (Follow-the-Regularized-Leader) with Nesterov momentum for subgradient descent.
    Directly minimizes the smoothed max autocorrelation.
    """
    n = len(a0)
    a = normalize_pos(a0)
    z = np.log(np.maximum(a, 1e-30))
    t0 = time.monotonic()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    # Nesterov momentum
    y = a.copy()
    momentum = 0.9

    # FTRL accumulators
    G1 = np.zeros(n)  # sum of gradients
    G2 = np.zeros(n)  # sum of squared gradients
    z_prev = z.copy()

    reg_strength = 0.5

    for it in range(int(time_limit * 50)):
        if time.monotonic() - t0 > time_limit - 5:
            break

        # Nesterov step: evaluate at y
        y = normalize_pos(y)
        # Compute subgradient at y
        c = fast_autocorr(y)
        k_use = min(topk, len(c))
        top_idx = np.argsort(c)[::-1][:k_use]
        # Uniform weights on top-k for subgradient
        w = np.ones(k_use) / k_use
        grad = np.zeros(n)
        for idx in range(k_use):
            K = top_idx[idx]
            js = max(0, K - (n - 1))
            je = min(n - 1, K)
            if js <= je:
                idx_range = np.arange(js, je + 1)
                grad[idx_range] += 2.0 * w[idx] * y[K - idx_range]

        # Project gradient onto simplex tangent space
        S = np.dot(grad, y)
        grad_tangent = grad - S * y

        # FTRL update
        eta_t = eta0 / (1.0 + 0.1 * np.sqrt(it + 1))
        G1 += eta_t * grad_tangent
        G2 += eta_t ** 2 * grad_tangent ** 2

        # FTRL closed-form solution (simplified)
        z_new = -G1 / (reg_strength * np.sqrt(G2 + 1e-10))
        a_candidate = softmax_fn(z_new)

        # Nesterov momentum
        z = momentum * (z - z_prev) + z_new
        z_prev = z_new.copy()
        z_new = z.copy()

        a_new = softmax_fn(z)
        score = 2.0 * n * fast_autocorr(a_new).max()

        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        # Also check y
        score_y = 2.0 * n * c.max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Update y for next Nesterov step
        y = a_new
        z_prev = z_new

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      FTRL it={it} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def peak_shaving_v2(a0, time_limit, seed=123):
    """
    Improved peak shaving with systematic exploration of source-destination pairs.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize_pos(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 3:
        b = fast_autocorr(a)
        bmax = b.max()
        threshold = bmax * 0.998
        top_peaks = np.where(b >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(b)[::-1][:5]
        top_peaks = top_peaks[:15]

        improved = False
        for k in top_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            max_contrib = contrib.max()
            if max_contrib < 1e-15:
                continue

            # Top contributors to this peak
            src_candidates = idx[np.argsort(contrib)[::-1][:max(3, len(idx) // 10)]]

            # Compute danger zone
            danger = np.zeros(n, dtype=bool)
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                danger[js2:je2 + 1] = True

            for src in src_candidates[:5]:
                safe = np.where(~danger)[0]
                safe = safe[safe != src]
                if len(safe) == 0:
                    safe = np.where(np.arange(n) != src)[0]

                # Sort safe positions by a value (prefer low)
                safe_sorted = safe[np.argsort(a[safe])][:max(5, len(safe) // 8)]
                if len(safe_sorted) == 0:
                    continue

                for dst in safe_sorted[:8]:
                    for frac in [0.002, 0.005, 0.01, 0.02, 0.04, 0.06]:
                        amt = a[src] * frac
                        if amt < 1e-12:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < -1e-12:
                            continue
                        a_trial = normalize_pos(a_trial)
                        score = 2.0 * n * fast_autocorr(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random exploration
            n_pert = rng.integers(1, 5)
            a_trial = a.copy()
            for _ in range(n_pert):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a_trial[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.03)
                    amt = a_trial[i] * frac
                    a_trial[i] -= amt
                    a_trial[j] += amt
            a_trial = normalize_pos(a_trial)
            score = 2.0 * n * fast_autocorr(a_trial).max()
            if score < best_score:
                a = a_trial
                best_score = score
                best_a = a.copy()
                stall = 0

            if stall > 500:
                break

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, seed=456):
    """
    Systematic coordinate descent: try moving small amounts between pairs of coordinates.
    Focuses on reducing the maximum autocorrelation peak.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize_pos(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    by_value_order = np.argsort(a)[::-1]

    while time.monotonic() - t0 < time_limit - 3:
        b = fast_autocorr(a)
        bmax = b.max()
        top_peak = np.argmax(b)

        improved = False
        # Try moving from peak-contributing positions to others
        js = max(0, top_peak - n + 1)
        je = min(n - 1, top_peak)
        peak_contrib = np.arange(js, je + 1)

        # Sort by contribution
        contrib = a[peak_contrib] * a[top_peak - peak_contrib]
        src_order = peak_contrib[np.argsort(contrib)[::-1][:10]]

        for src in src_order:
            # Try dest = positions with lowest a value
            dest_candidates = by_value_order[:20]
            for dst in dest_candidates:
                if dst == src:
                    continue
                for frac in [0.001, 0.003, 0.008, 0.015, 0.03]:
                    amt = a[src] * frac
                    if amt < 1e-12:
                        continue
                    a_trial = a.copy()
                    a_trial[src] = max(0, a_trial[src] - amt)
                    a_trial[dst] += amt
                    a_trial = normalize_pos(a_trial)
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        # Update by_value_order periodically
        if (it := int(time.monotonic() - t0 * 10)) % 100 == 0:
            by_value_order = np.argsort(a)[::-1]

        if not improved:
            # Random pair swap
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                for frac in [0.002, 0.005, 0.01, 0.02]:
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = normalize_pos(a_trial)
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = normalize_pos(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 10)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 3
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 2
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.7 + 0.2 * np.cos(2 * x) + 0.1 * np.cos(5 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * (x - 0.3) ** 2) + 0.5 * np.exp(-5 * (x - 0.7) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * np.cos(np.pi * x / 3) ** 2
    elif choice == 7:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.1 * np.cos(2 * np.pi * 7 * x)
    elif choice == 8:
        x = np.linspace(0, 1, n)
        a = np.abs(np.sin(np.pi * x * 2)) ** 0.5
    else:
        x = np.linspace(-1, 1, n)
        a = 0.5 * (1 + np.cos(np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def perturbed_init(a0, rng, scale=0.1, mode='gaussian'):
    """Create a perturbed version of a sequence for multi-start."""
    a = normalize_pos(a0)
    n = len(a)
    if mode == 'gaussian':
        noise = rng.normal(0, scale, n)
        a_new = a + noise * (a + 1e-6)
    elif mode == 'peak_target':
        # Specifically perturb to reduce peaks
        c = fast_autocorr(a)
        top_k = np.argsort(c)[::-1][:5]
        peaks_weight = np.zeros(n)
        for K in top_k:
            js = max(0, K - n + 1)
            je = min(n - 1, K)
            peaks_weight[js:je+1] += c[K]
        noise = rng.normal(0, 1, n) * (1 - peaks_weight / (peaks_weight.max() + 1e-10))
        a_new = a + scale * noise * (a + 1e-6)
    else:
        noise = rng.normal(0, scale, n)
        a_new = a + scale * noise
    a_new = np.maximum(a_new, 0)
    sn = a_new.sum()
    if sn < 1e-15:
        return a.copy()
    return a_new / sn

    """

    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)

            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start

        try_update(a_out, f"lse_n{n_base}")

    # === Phase 3: FTRL Subgradient (new method) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 3: FTRL Subgradient n={n_base} ({remaining:.0f}s left) ===")

    if t3 > 15:
        a_out, _ = ftrl_subgradient_optimize(best_seq, t3, eta0=0.05, topk=10)
        try_update(a_out, f"ftrl_n{n_base}")

    # === Phase 4: Multi-start with peak-targeted perturbation ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 4: Multi-start ({remaining:.0f}s left) ===")
    t4_total = min(remaining * 0.15, 180.0)
