            mu = (t - 1.0) / (t + 2.0)
            a_new = y_new + mu * (y_new - y)
            a_new = np.maximum(a_new, 0)
            sa = a_new.sum()
            if sa < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / sa
        
        y = y_new
        a = a_new
        
        # Track best (use y-average for stability)
        sc_y = fast_conv(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        
        sc_a = fast_conv(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        
        avg += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_c = avg / n_avg
            sac = avg_c.sum()
            if sac > 1e-15:
                avg_c = avg_c * norm / sac
                sc_avg = fast_conv(avg_c).max()
                if sc_avg < best_score:
                    best_score = sc_avg
                    best_a = avg_c.copy()
        
        if verbose and it % 500 == 0:
            print(f"    NLP it={it} ({time.monotonic()-t0:.0f}s) best={best_score:.8f}")
    
    return best_a, best_score

def mirror_descent_lse(a0, time_limit, T_start=10.0, T_end=0.01, n_stages=15, verbose=True):
    """Mirror descent (multiplicative update) on LSE surrogate."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        
        stages_per_T = max(10, int(remaining * 3))
        for it in range(stages_per_T):
            if time.monotonic() - t0 > time_limit:
                break
            
            c = fast_conv(a)
            # Softmax weights
            c_max = c.max()
            w = np.exp((c - c_max) / T)
            w_sum = w.sum()
            if w_sum < 1e-300:
                w = np.zeros_like(c)
                w[c.argmax()] = 1.0
            else:
                w = w / w_sum
            
            # Gradient: grad_i = 2 * sum_k w_k * a_{k-i}
            # = 2 * correlate(w, a, mode='valid')... let me compute via FFT
            # p_i = sum_j a_j * w_{i+j}
            a_rev = a[::-1]
            fft_size = 1 << (n + 2*n - 2).bit_length()
            P = irfft(rfft(a_rev, fft_size) * rfft(w, fft_size), fft_size)
            # P[m] = sum_j a_rev[j] * w[m-j] = sum_j a[n-1-j] * w[m-j]
            # Let j' = n-1-j: P[m] = sum_{j'} a[j'] * w[m-n+1+j']
            # So p_i = P[i + n - 1]
            p = P[n-1:n-1+n]
            grad = 2.0 * p
            
            # Mirror descent: a_i <- a_i * exp(-eta * grad_i)
            # Choose eta for stability
            eta = min(1.0, 0.5 / max(grad.max(), 1e-10))
            log_a = np.log(np.maximum(a, 1e-300)) - eta * grad
            # Normalize
            log_max = log_a.max()
            a_new = np.exp(log_a - log_max)
            a_new = a_new / a_new.sum() * norm
            
            sc = fast_conv(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            
            a = a_new
        
        if verbose:
            print(f"    MD stage {si+1}/{n_stages} T={T:.4f} ({time.monotonic()-t0:.0f}s) best={best_score:.8f}")
    
    return best_a, best_score

def spectral_peak_flow(a0, time_limit, verbose=True):
    """Flow-based peak shaving using LP subproblem approximation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()
    stall = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        c = fast_conv(a)
        M = c.max()
        
        # Find top peaks
        n_peaks = min(20, 2*n-1)
        peak_idx = np.argsort(c)[::-1][:n_peaks]
        
        # Compute gradient from peaks
        grad = np.zeros(n)
        for k in peak_idx:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += a[j]
        grad *= 2.0
        
        # Flow: move mass from high-grad to low-grad positions
        # Use a "pressure" model: a_i proportional to 1/grad_i
        # But maintain non-negativity and sum constraint
        
        # Try multiple step sizes
        improved = False
        for eta in [0.01, 0.02, 0.05, 0.1, 0.2]:
            # Projected gradient step
            a_trial = a - eta * (grad - grad.mean())  # center to maintain sum
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial * norm / st
            sc = fast_conv(a_trial).max()
            if sc < best_score - 1e-12:
                a = a_trial
                best_score = sc
                best_a = a.copy()
                improved = True
                stall = 0
                break
        
        if not improved:
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                for frac in [0.001, 0.005, 0.01, 0.05]:
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        st = a_trial.sum()
                        if st > 1e-15:
                            a_trial = a_trial * norm / st
                            sc = fast_conv(a_trial).max()
                            if sc < best_score:
                                a = a_trial
                                best_score = sc
                                best_a = a.copy()
                                improved = True
                                stall = 0
                                break
            if not improved:
                stall += 1
                if stall > 500:
                    break
    
    return best_a, best_score

def interpolate(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    result = np.interp(new_x, old_x, a_norm)
    result = np.maximum(result, 0)
    rs = result.sum()
    if rs < 1e-15:
        return np.ones(new_n) / new_n
    return result / rs

def flat_top_init(n, alpha=0.5):
    """Initialize with a flat-top-like window."""
    x = np.arange(n)
    # Multi-frequency cosine window
    a = np.ones(n)
    a += alpha * np.cos(2 * np.pi * x / n)
    a += 0.3 * alpha * np.cos(4 * np.pi * x / n)
    a += 0.1 * alpha * np.cos(6 * np.pi * x / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def cosine_init(n, freq=1.0, alpha=0.3):
    """Initialize with a single cosine modulation."""
    x = np.arange(n)
    a = 1.0 + alpha * np.cos(2 * np.pi * freq * x / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
