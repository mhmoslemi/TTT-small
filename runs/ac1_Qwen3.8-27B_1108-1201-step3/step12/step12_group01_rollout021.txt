def lse_max(c, T):
    """Compute log-sum-exp approximation of max, and the softmax weights."""
    c = np.asarray(c, dtype=np.float64)
    Tc = T * c
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    w = exp_Tc / exp_Tc.sum()
    lse_val = (max_Tc + np.log(exp_Tc.sum())) / T
    return lse_val, w

    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def lse_obj_grad(a, T, topk=None):
    """
    Compute LSE-approximation of max(conv(a,a)) and its gradient.
    Returns (obj_value, gradient_wrt_a).
    """
    n = len(a)
    c = fast_autocorr(a)
    if topk is not None and topk < len(c):
        top_idx = np.argsort(c)[::-1][:topk]
        top_c = c[top_idx]
    else:
        top_idx = np.arange(len(c))
        top_c = c

    Tc = T * top_c
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    w = exp_Tc / exp_Tc.sum()
    lse_val = (max_Tc + np.log(exp_Tc.sum())) / T

    # Gradient: grad[i] = 2 * sum_{k in top} w[k] * a[k-i] (with bounds)
    # = 2 * correlate(w_full, a)[i] where w_full is the full (2n-1) weight vector
    w_full = np.zeros(2 * n - 1)
    w_full[top_idx] = w
    grad = 2.0 * np.correlate(w_full, a, mode='valid')

    return lse_val, grad


def adam_lse_optimize(a0, time_limit, T_start=50.0, T_end=0.5, n_stages=15, topk=30, lr_base=0.02):
    """
    Adam-optimized LSE in log-space with multi-stage temperature annealing.
    """
    n = len(a0)
    a_init = normalize(a0)
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = fast_autocorr(a_init).max()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit - 5:
            break

        lr = lr_base * (0.75 ** si)
        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.85, 0.999, 1e-8

        max_it = max(20, int(min(time_limit - (time.monotonic() - t0), 100.0) * 5))
        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 5:
                break

            a_cur = softmax_fn(z)
            obj, grad_a = lse_obj_grad(a_cur, stage_T, topk)
            S = np.dot(grad_a, a_cur)
            grad_z = a_cur * (grad_a - S)

            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            if (it + 1) % 10 == 0:
                a_check = softmax_fn(z)
                score = fast_autocorr(a_check).max()
                if score < best_score:
                    best_score = score
                    best_a = a_check.copy()

        a_stage = softmax_fn(z)
        score_stage = fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()

        if si % 3 == 0:
            print(f"    LSE stage {si}/{n_stages} (T={stage_T:.2f}, t={time.monotonic()-t0:.0f}s): best={best_score * 2 * n:.6f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, top_k=25, max_inner=50):
    """
    Efficient coordinate descent minimizing max(conv(a,a)).
    Uses peak-equalization to find optimal step per coordinate.
    Incrementally updates the convolution vector.
    """
    n = len(a0)
    a = normalize(a0)
    conv = fast_autocorr(a)
    best_a = a.copy()
    best_score = conv.max()
    t0 = time.monotonic()
    two_n_minus_1 = 2 * n - 1

    stall_passes = 0
    total_passes = 0

    while time.monotonic() - t0 < time_limit - 2:
        total_passes += 1
        # Refresh top peaks every few passes
        top_idx = np.argsort(conv)[::-1][:min(top_k, two_n_minus_1)]
        top_vals = conv[top_idx].copy()
        n_top = len(top_idx)

        pass_improved = False
        max_gain = 0.0

        for i in range(n):
            if a[i] < 1e-12 and i % 10 != 0:
                continue

            # Influence: for each top peak k, how much does a[i] affect conv[k]?
            # conv[k] depends on a[i] via a[k-i] (if valid)
            infl = np.zeros(n_top)
            for idx in range(n_top):
                k = top_idx[idx]
                ki = k - i
                if 0 <= ki < n:
                    infl[idx] = a[ki]

            # Generate candidate deltas
            candidates = []
            # From peak equalization pairs
            n_pairs = min(8, n_top)
            for ii in range(n_pairs):
                for jj in range(ii + 1, n_pairs):
                    d_diff = 2.0 * (infl[ii] - infl[jj])
                    if abs(d_diff) > 1e-12:
                        delta = (top_vals[jj] - top_vals[ii]) / d_diff
                        max_d = max(a[i] * 0.8, 1e-4 / n)
                        delta = max(-max_d, min(max_d, delta))
                        if abs(delta) > 1e-14:
                            candidates.append(delta)

            # Subgradient direction
            active_infl = infl[infl > 0]
            if len(active_infl) > 0:
                sg = 2.0 * active_infl.mean()
                step = min(a[i] * 0.3, 5.0 / (n * max(sg, 1e-8)))
                candidates.append(-sg * step)
                candidates.append(sg * step * 0.5)

            # Small random nudges for escape
            if total_passes % 10 == 0 and a[i] > 1e-8:
                candidates.append(a[i] * 0.01 * (1 if np.random.random() > 0.5 else -1))

            if not candidates:
                continue

            # Evaluate candidates (check only top peaks + 2i)
            best_delta = 0.0
            local_best = conv.max()

            for delta in candidates:
                if abs(delta) < 1e-14:
                    continue
                new_ai = a[i] + delta
                if new_ai < -1e-12:
                    continue
                new_ai = max(new_ai, 0.0)

                # Compute new max checking top peaks
                new_max = 0.0
                for idx in range(n_top):
                    k = top_idx[idx]
                    ki = k - i
                    if 0 <= ki < n:
                        if k == 2 * i:
                            val = conv[k] + 2.0 * delta * a[i] + delta * delta
                        else:
                            val = conv[k] + 2.0 * delta * a[ki]
                    else:
                        val = conv[k]
                    if val > new_max:
                        new_max = val

                # Check 2i if not already covered
                if 2 * i >= 0 and 2 * i < two_n_minus_1:
                    ki_2i = 2 * i - i
                    if 0 <= ki_2i < n:
                        val_2i = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
                        if val_2i > new_max:
                            new_max = val_2i

                if new_max < local_best - 1e-13:
                    local_best = new_max
                    best_delta = delta

            if best_delta != 0.0:
                old_ai = a[i]
                a[i] = max(old_ai + best_delta, 0.0)
                actual_delta = a[i] - old_ai

                if abs(actual_delta) > 1e-14:
                    # Update conv incrementally
                    js = max(0, i)
                    je = min(two_n_minus_1 - 1, i + n - 1)
                    for k in range(js, je + 1):
                        ki = k - i
                        if k == 2 * i:
                            conv[k] += 2.0 * actual_delta * old_ai + actual_delta * actual_delta
                        else:
                            conv[k] += 2.0 * actual_delta * a[ki]

                    cur_max = conv.max()
                    if cur_max < best_score - 1e-13:
                        best_score = cur_max
                        best_a = a.copy()
                        pass_improved = True
                        max_gain = max(max_gain, best_score - cur_max)

        # Periodic renormalization and conv refresh
        if total_passes % 30 == 0:
            a = normalize(a)
            conv = fast_autocorr(a)
            cur = conv.max()
            if cur < best_score:
                best_score = cur
                best_a = a.copy()

        if not pass_improved:
            stall_passes += 1
            if stall_passes > 30:
                break
        else:
            stall_passes = 0

        if total_passes % 100 == 0:
            print(f"    CD pass {total_passes} (t={time.monotonic()-t0:.0f}s): best={best_score * 2 * n:.6f}")

    return best_a, best_score


def fine_peak_shaving(a0, time_limit, seed=42):
    """
    Targeted peak shaving: identify the dominant peaks and reduce them
    by moving mass from contributing positions to safe positions.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize(a0)
    conv = fast_autocorr(a)
    best_a = a.copy()
    best_score = conv.max()
    t0 = time.monotonic()
    two_n_minus_1 = 2 * n - 1

    iteration = 0
    while time.monotonic() - t0 < time_limit - 2:
        iteration += 1
        conv = fast_autocorr(a)
        bmax = conv.max()
        best_idx = np.argmax(conv)

        # Find the top peaks (within 1% of max)
        threshold = bmax * 0.99
        top_peaks = np.where(conv >= threshold)[0]

        improved = False
        for k in top_peaks[:15]:
            # Positions that contribute to this peak: i such that i in [max(0,k-n+1), min(k,n-1)]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_range = np.arange(js, je + 1)
            if len(idx_range) == 0:
                continue

            # Contribution of each position
            contrib = a[idx_range] * a[k - idx_range]
            src = idx_range[np.argmax(contrib)]

            # Find "safe" destination: low contribution to all top peaks
            danger_score = np.zeros(n)
            for k2 in top_peaks[:15]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                for j in range(js2, je2 + 1):
                    danger_score[j] += a[k2 - j]

            # Prefer low danger and low current value
            scores = danger_score * (a + 1e-10)
            scores[src] = np.inf
            safe_candidates = np.argsort(scores)[:max(5, n // 20)]

            for dst in safe_candidates[:5]:
                if dst == src:
                    continue
                # Try moving a fraction
                for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = normalize(a_trial)
                    c_trial = fast_autocorr(a_trial)
                    if c_trial.max() < best_score - 1e-13:
                        a = a_trial
                        best_score = c_trial.max()
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved and iteration > 200:
            break

    return best_a, best_score


def structured_init(n, rng, style=None):
    """Generate structured initializations."""
    if style is None:
        style = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if style == 0:
        a = np.sin(np.pi * x)
    elif style == 1:
        a = np.sin(np.pi * x) ** 2
    elif style == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x)
    elif style == 3:
        a = np.exp(-5 * (x - 0.5) ** 2) + 0.3 * np.exp(-5 * (x - 0.2) ** 2) + 0.3 * np.exp(-5 * (x - 0.8) ** 2)
    elif style == 4:
        a = (1 + np.cos(2 * np.pi * x * 3)) / 2
    elif style == 5:
        a = np.abs(np.sin(np.pi * x * 2))
    elif style == 6:
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.6 * np.exp(-3 * (x - 0.7) ** 2)
    elif style == 7:
        a = 0.6 + 0.3 * np.cos(4 * np.pi * x) + 0.1 * np.cos(7 * np.pi * x)
    elif style == 8:
        a = np.sin(np.pi * x) * (1 + 0.3 * np.cos(6 * np.pi * x))
    else:
        a = np.maximum(np.zeros(n), 0.5 + 0.25 * np.cos(2 * np.pi * 5 * x) + 0.15 * np.cos(2 * np.pi * 2 * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)


def reflect_pattern(a, extra_pairs=0):
    """Create a symmetric extension of the sequence."""
    n = len(a)
    # Mirror: a + reverse(a)
    ext = np.concatenate([a, a[::-1]])
    return normalize(ext)


def propose_candidate(seed=42, time_budget=1800.0):
    """
    Multi-phase optimization:
    1. Quick Adam-LSE to get into good basin
    2. Extensive coordinate descent (main workhorse)
    3. Fine peak shaving
    4. Multi-length exploration
    5. Final polish
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(time_budget) - 120.0, 1680.0)
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = normalize(p)
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    n_base = len(best_seq) if best_seq is not None else 1600

    if parent is None:
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # === Phase 1: Quick Adam-LSE (get into good basin) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.15, 180.0)
    if t1 > 20:
        print(f"\n=== Phase 1: Adam-LSE n={n_base} ({t1:.0f}s) ===")
        a_out, _ = adam_lse_optimize(best_seq, t1, T_start=50.0, T_end=1.0, n_stages=12, topk=40, lr_base=0.015)
        try_update(a_out, f"adam_lse_n{n_base}")

    # === Phase 2: Coordinate Descent (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Coordinate Descent n={n_base} ({remaining:.0f}s) ===")
    t2 = min(remaining * 0.45, 600.0)
    if t2 > 30:
        a_out, _ = coordinate_descent(best_seq, t2, top_k=30)
        try_update(a_out, f"cd_n{n_base}")

    # === Phase 3: Fine peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.10, 100.0)
    if t3 > 15:
        print(f"\n=== Phase 3: Peak Shaving n={n_base} ({t3:.0f}s) ===")
        a_out, _ = fine_peak_shaving(best_seq, t3, seed=123)
        try_update(a_out, "peak_shave_1")

    # === Phase 4: Second round CD from updated best ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.20, 250.0)
    if t4 > 20:
        print(f"\n=== Phase 4: CD round 2 n={n_base} ({t4:.0f}s) ===")
        a_out, _ = coordinate_descent(best_seq, t4, top_k=40)
        try_update(a_out, "cd2_n_base")

    # === Phase 5: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        print(f"\n=== Phase 5: Multi-length ({remaining:.0f}s) ===")
        lengths_to_try = []
        # Try nearby lengths
        for dl in [-200, -100, 100, 200, 300, 400]:
            nl = n_base + dl
            if 800 <= nl <= 3000:
                lengths_to_try.append(nl)
        # Deduplicate and limit
        lengths_to_try = list(dict.fromkeys(lengths_to_try))[:4]

        t_each = remaining * 0.7 / max(len(lengths_to_try), 1)
        for nl in lengths_to_try:
            if time.monotonic() - t_global > time_budget - 60:
                break
            if t_each < 15:
                break
            print(f"  Trying n={nl} ({t_each:.0f}s each)")
            init_long = interpolate_sequence(best_seq, nl)
            # Quick CD
            a_out, _ = coordinate_descent(init_long, t_each * 0.7, top_k=25)
            try_update(a_out, f"cd_n{nl}")
            # Quick peak shave
            remaining_sub = t_each * 0.3
            if remaining_sub > 10:
                a_out, _ = fine_peak_shaving(a_out, remaining_sub, seed=nl)
                try_update(a_out, f"ps_n{nl}")

    # === Phase 6: Final CD polish on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t6 = min(remaining * 0.5, 120.0)
        print(f"\n=== Phase 6: Final CD polish ({t6:.0f}s) ===")
        a_out, _ = coordinate_descent(best_seq, t6, top_k=50)
        try_update(a_out, "final_cd")

    # === Phase 7: Final peak shave ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t7 = min(remaining * 0.6, 60.0)
        print(f"\n=== Phase 7: Final peak shave ({t7:.0f}s) ===")
        a_out, _ = fine_peak_shaving(best_seq, t7, seed=999)
        try_update(a_out, "final_ps")

    # === Phase 8: One more CD if time ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t8 = min(remaining * 0.7, 60.0)
        a_out, _ = coordinate_descent(best_seq, t8, top_k=60)
        try_update(a_out, "last_cd")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
