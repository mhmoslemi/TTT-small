    T = np.where((idx >= 0) & (idx < n), a[idx * ((idx >= 0) & (idx < n))], 0)
    # This doesn't work directly because of indexing issues
    # Let me just use the loop, it's fast enough
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T
```

from scipy.optimize import linprog

def solve_lp_scipy(a_fixed, c=None):
    """Solve LP using scipy linprog with HiGHS."""
    n = len(a_fixed)
    size = 2*n - 1
    # Variables: [b_0, ..., b_{n-1}, M]
    # Objective: minimize M -> c = [0, ..., 0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Inequality constraints: T@b - M <= 0 for each row
    # A_ub @ x <= b_ub
    T = build_T(a_fixed)
    A_ub = np.hstack([T, -np.ones((size, 1))])
    b_ub = np.zeros(size)
    
    # Equality constraint: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: b >= 0, M >= 0 (M will be >= 0 automatically)
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='HiGHS')
    if res.status != 0:
        return None, None
    b_val = res.x[:n]
    M_val = res.x[n]
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, M_val
```

Let me now think about the Nesterov/FISTA iteration more carefully:

**FISTA for the LP best-response:**

The iteration is:
1. Given y (the extrapolated point), solve the LP: b* = argmin_b max_k (y*b)_k s.t. b in simplex
2. Update: a_new = (1-alpha)*a + alpha*b*
3. Extrapolate: y_new = a_new + momentum*(a_new - a)
4. Project y_new onto simplex

The convergence rate of FISTA is O(1/k^2) for the duality gap. So after K iterations, the gap is O(1/K^2).

For our problem, the gap starts at maybe 0.5 (score - lower_bound ≈ 1.5 - 1.0 = 0.5) and we want it to be < 0.001. So we need 1/K^2 < 0.002, i.e., K > 700 iterations. That's very feasible.

But the actual convergence might be slower in practice due to:
- LP solving errors
- The non-convexity of the original problem (we're solving a convex relaxation)
- Numerical issues

In practice, the Nesterov LP converges to a good solution in a few thousand iterations, and then the improvement is marginal.

**Key improvement: Adaptive step size / restart**

Standard FISTA has a fixed step size schedule. But we can do better:
- If the objective (score) increases, restart the momentum (set momentum to 0)
- Use a backtracking line search on the step size

This is the "FISTA with restart" or "Nesterov with adaptive restart" method.

**Another improvement: Use the LP dual**

The LP dual is:
  max sum_k lambda_k
  s.t. T^T @ lambda <= mu (for some mu)
       lambda >= 0

The dual variables lambda tell us which constraints are "tight" (active). We can use this information to:
1. Identify the most important constraints
2. Focus the subgradient method on those constraints
3. Use them for a compressed sensing-like approach

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA with LP best-response (scipy HiGHS) and adaptive restarts
2. LogSumExp L-BFGS polishing with aggressive temperature schedule
3. Peak-targeted subgradient with iterate averaging
4. Multi-length exploration and final polishing
Key improvements: scipy HiGHS for faster LP, adaptive FISTA restarts,
aggressive LSE schedule, and better time management.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    """FFT-based linear convolution of a with itself."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score for sum(a)=1: returns 2*n*max(conv(a,a))."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def build_conv_matrix(a):
    """Build matrix T where (T @ b)[k] = (a convolved with b)[k]."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_highe(a_fixed):
    """
    Solve LP: min M s.t. T(a_fixed) @ b <= M, sum(b)=1, b>=0.
    Uses scipy HiGHS solver for speed.
    Returns (b_optimal, M_optimal) or (None, None) on failure.
    """
    n = len(a_fixed)
    size = 2 * n - 1
    
    # Build A_ub: [T, -1] @ [b; M] <= 0
    T = build_conv_matrix(a_fixed)
    A_ub = np.hstack([T, -np.ones((size, 1))])
    b_ub = np.zeros(size)
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Objective: min M
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Bounds
    bounds = [(0, None)] * (n + 1)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='HiGHS',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-10,
                           'primal_feasibility_tolerance': 1e-10})
    if res.status != 0:
        return None, None
    
    b_val = res.x[:n]
    M_val = float(res.x[n])
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, M_val


def simplify solve_lp_cvxpy(a_fixed, solver=None):
    """Fallback LP solver using cvxpy."""
    import cvxpy as cp
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def fista_lp(a0, time_limit, max_iters=1000000, use_highe=True):
    """
    FISTA with LP best-response and adaptive restarts.
    Minimizes max_k (a*a)_k over the simplex.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_log = t0
    
    # FISTA parameters
    k = 0
    prev_score = best_score
    restart_count = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP best response
        if use_highe:
            b_new, M_lp = solve_lp_highe(y)
        else:
            b_new, M_lp = solve_lp_cvxpy(y)
        
        if b_new is None:
            continue
        
        # FISTA update
        k += 1
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * k * k)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k
        
        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = project_simplex(a_new)
        
        y_new = a_new + momentum * (a_new - a_old)
        y_new = project_simplex(y_new)
        
        a = a_new
        y = y_new
        
        # Evaluate
        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        
        # Adaptive restart: if score went up significantly, reset momentum
        if sc_a > prev_score * 1.01 and k > 10:
            y = a.copy()
            k = 0  # Reset counter for restart
            restart_count += 1
        
        prev_score = sc_a
        
        if (it + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_log > 10:
                print(f"    FISTA iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} M_lp={M_lp:.6f} restarts={restart_count}")
                last_log = elapsed
    
    return best_a, best_score


def softmax_from_z(z):
    """Numerically stable softmax."""
    z_shift = z - z.max()
    exp_z = np.exp(z_shift)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def lse_obj_grad(z_val, T_temp, n):
    """Compute LSE objective and gradient w.r.t. z (log-softmax params)."""
    a = softmax_from_z(z_val)
    b = fast_convolve(a)
    Tb = T_temp * b
    max_Tb = min(Tb.max(), 700.0)
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    lse_val = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_temp) * lse_val
    
    # Gradient w.r.t. a
    grad_a = np.zeros(n)
    for k_pos in range(len(b)):
        if w[k_pos] < 1e-18:
            continue
        lo = max(0, k_pos - n + 1)
        hi = min(n - 1, k_pos)
        idx = np.arange(lo, hi + 1)
        grad_a[idx] += 2.0 * w[k_pos] * a[k_pos - idx]
    
    # Chain rule: grad_z = a * (grad_a - <grad_a, a>)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_lbfgs(a0, time_limit, T_init=20.0, T_min=0.5, n_stages=15, iters_per_stage=50):
    """LSE optimization with L-BFGS-B and geometric temperature schedule."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    T = T_init
    
    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        def func(z_val):
            obj, _ = lse_obj_grad(z_val, T, n)
            return obj
        
        def jac(z_val):
            _, g = lse_obj_grad(z_val, T, n)
            return g
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': iters_per_stage * 200,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            a_check = np.maximum(a_check, 0)
            a_check = a_check / a_check.sum()
            sc = score_normalized(a_check)
            
            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()
            
            # Geometric temperature decrease
            T = max(T_min, T * 0.7)
            
            if (stage + 1) % 3 == 0:
                print(f"    LSE stage={stage+1}/{n_stages} T={T:.3f}: cur={sc:.8f} best={best_score:.8f}")
        
        except Exception as e:
            T = max(T_min, T * 0.8)
    
    return best_a, best_score


def peak_subgrad(a0, time_limit, n_steps=500000):
    """Peak-targeted subgradient with Polyak step and iterate averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    z = np.log(np.maximum(a, 1e-300)).copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    lower_bound = 2.0 * n / (2.0 * n - 1.0)
    
    t0 = time.monotonic()
    avg_a = a.copy()
    avg_count = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b
        
        # Find active peaks (top 10 or 99th percentile)
        threshold = max_b * 0.95
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        if len(active) > 20:
            top_idx = np.argsort(b[active])[-20:]
            active = active[top_idx]
        
        # Subgradient: average of a shifted at each active peak
        g = np.zeros(n)
        for k_pos in active:
            lo = max(0, k_pos - n + 1)
            hi = min(n - 1, k_pos)
            idx = np.arange(lo, hi + 1)
            g[idx] += a[k_pos - idx]
        g /= len(active)
        
        # Project to tangent space
        S = np.dot(g, a)
        grad_z = a * (g - S)
        grad_norm = np.linalg.norm(grad_z)
        if grad_norm < 1e-15:
            break
        
        # Polyak step
        gap = cur_score - lower_bound
        if gap < 1e-10:
            break
        eta = gap / (grad_norm * grad_norm + 1e-15)
        eta = min(eta, 1.0)
        
        z_new = z - eta * grad_z
        a_new = softmax_from_z(z_new)
        sc_new = score_normalized(a_new)
        
        # Backtracking if step too large
        if sc_new > cur_score * 1.05:
            for _ in range(10):
                eta *= 0.5
                z_new = z - eta * grad_z
                a_new = softmax_from_z(z_new)
                if score_normalized(a_new) < cur_score:
                    break
        
        z = z_new
        a = a_new
        
        # Running average (exponential)
        beta = 0.999
        avg_a = beta * avg_a + (1 - beta) * a
        avg_count += 1
        
        if sc_new < best_score:
            best_score = sc_new
            best_a = a.copy()
        
        sc_avg = score_normalized(avg_a)
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = avg_a.copy()
        
        if (step + 1) % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakSG step={step+1} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA with LP best-response (scipy HiGHS) and adaptive momentum restarts
2. LogSumExp L-BFGS-B polishing with geometric temperature schedule
3. Peak-targeted subgradient with Polyak step and exponential averaging
4. Multi-length exploration and final combined polish
Key improvements over previous: scipy HiGHS for faster LP solving,
adaptive FISTA restarts when objective degrades, aggressive LSE schedule,
and tighter time management with safety margins.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    """FFT-based linear auto-convolution of a with itself."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score for sum(a)=1: returns 2*n*max(conv(a,a))."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def build_conv_matrix(a):
    """Build matrix T where (T @ b)[k] = (a conv b)[k] for k=0..2n-2."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_highs(a_fixed):
    """Solve LP: min M s.t. T(a_fixed)@b <= M, sum(b)=1, b>=0. Uses HiGHS."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = build_conv_matrix(a_fixed)
    A_ub = np.hstack([T, -np.ones((size, 1))])
    b_ub = np.zeros(size)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    bounds = [(0, None)] * (n + 1)
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='HiGHS',
                  options={'presolve': True})
    if res.status != 0:
        return None, None
    b_val = res.x[:n]
    M_val = float(res.x[n])
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, M_val


def solve_lp_cvxpy(a_fixed):
    """Fallback LP solver using cvxpy."""
    try:
        import cvxpy as cp
        n = len(a_fixed)
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        T = build_conv_matrix(a_fixed)
        constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        solver = None
        for s in [cp.PDLP, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=s)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solver = s
                    break
            except Exception:
                continue
        if solver is None:
            return None, None
        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        s = b_val.sum()
        if s < 1e-15:
            return None, None
        b_val = b_val / s
        return b_val, float(M_var.value)
    except Exception:
        return None, None


def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.where(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def fista_lp(a0, time_limit, max_iters=1000000, use_highe=True):
    """FISTA with LP best-response and adaptive momentum restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_log = t0
    k = 0
    prev_score = best_score
    restart_count = 0
    consecutive_worse = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if use_highe:
            b_new, M_lp = solve_lp_highs(y)
            if b_new is None and it > 100:
                b_new, M_lp = solve_lp_cvxpy(y)
        else:
            b_new, M_lp = solve_lp_cvxpy(y)
        if b_new is None:
            continue

        k += 1
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * k * k)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = project_simplex(a_new)
        y_new = a_new + momentum * (a_new - a_old)
        y_new = project_simplex(y_new)
        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            consecutive_worse = 0
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            consecutive_worse = 0

        if sc_a > prev_score * 1.005:
            consecutive_worse += 1
        else:
            consecutive_worse = max(0, consecutive_worse - 1)

        if consecutive_worse >= 5 and k > 20:
            y = a.copy()
            k = 0
            restart_count += 1
            consecutive_worse = 0

        prev_score = sc_a

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_log > 10:
                print(f"    FISTA iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} M={M_lp:.6f} restarts={restart_count}")
                last_log = elapsed

    return best_a, best_score


def softmax_from_z(z):
    """Numerically stable softmax."""
    z_shift = z - z.max()
    exp_z = np.exp(z_shift)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def lse_obj_grad(z_val, T_temp, n):
    """Compute LSE objective and gradient w.r.t. z (log-softmax parameters)."""
    a = softmax_from_z(z_val)
    b = fast_convolve(a)
    Tb = T_temp * b
    max_Tb = min(Tb.max(), 700.0)
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    lse_val = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_temp) * lse_val
    grad_a = np.zeros(n)
    for k_pos in range(len(b)):
        if w[k_pos] < 1e-18:
            continue
        lo = max(0, k_pos - n + 1)
        hi = min(n - 1, k_pos)
        idx = np.arange(lo, hi + 1)
        grad_a[idx] += 2.0 * w[k_pos] * a[k_pos - idx]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_lbfgs(a0, time_limit, T_init=20.0, T_min=0.5, n_stages=15, iters_per_stage=50):
    """LSE optimization with L-BFGS-B and geometric temperature schedule."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    T = T_init

    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func_lse(z_val):
            obj, _ = lse_obj_grad(z_val, T, n)
            return obj

        def jac_lse(z_val):
            _, g = lse_obj_grad(z_val, T, n)
            return g

        try:
            res = minimize(
                fun=func_lse, x0=z, jac=jac_lse, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': iters_per_stage * 200,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            a_check = np.maximum(a_check, 0)
            a_check = a_check / a_check.sum()
            sc = score_normalized(a_check)
            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()
            T = max(T_min, T * 0.65)
            if (stage + 1) % 5 == 0:
                print(f"    LSE stage={stage+1}/{n_stages} T={T:.3f}: cur={sc:.8f} best={best_score:.8f}")
        except Exception:
            T = max(T_min, T * 0.8)

    return best_a, best_score


def peak_subgrad(a0, time_limit, n_steps=500000):
    """Peak-targeted subgradient with Polyak step and exponential averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = np.log(np.maximum(a, 1e-300)).copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    lower_bound = 2.0 * n / (2.0 * n - 1.0)
    t0 = time.monotonic()
    avg_a = a.copy()
    beta_avg = 0.9995

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b

        threshold = max_b * 0.92
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        if len(active) > 30:
            top_idx = np.argsort(b[active])[-30:]
            active = active[top_idx]

        g = np.zeros(n)
        for k_pos in active:
            lo = max(0, k_pos - n + 1)
            hi = min(n - 1, k_pos)
            idx = np.arange(lo, hi + 1)
            g[idx] += a[k_pos - idx]
        g /= len(active)

        S = np.dot(g, a)
        grad_z = a * (g - S)
        grad_norm = np.linalg.norm(grad_z)
        if grad_norm < 1e-15:
            break

        gap = cur_score - lower_bound
        if gap < 1e-10:
            break
        eta = gap / (grad_norm * grad_norm + 1e-15)
        eta = min(eta, 1.0)

        z_new = z - eta * grad_z
        a_new = softmax_from_z(z_new)
        sc_new = score_normalized(a_new)

        if sc_new > cur_score * 1.05:
            for _ in range(8):
                eta *= 0.5
                z_new = z - eta * grad_z
                a_new = softmax_from_z(z_new)
                if score_normalized(a_new) < cur_score:
                    break

        z = z_new
        a = a_new
        avg_a = beta_avg * avg_a + (1.0 - beta_avg) * a

        if sc_new < best_score:
            best_score = sc_new
            best_a = a.copy()
        sc_avg = score_normalized(avg_a)
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = avg_a.copy()

        if (step + 1) % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakSG step={step+1} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length, preserving sum=1 normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate structured initial sequences with good spectral properties."""
    choice = rng.integers(0, 8)
    if choice == 0:

    elif choice == 1:

    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = 0.6 + 0.3 * np.cos(2 * np.pi * 4 * x) + 0.1 * np.cos(2 * np.pi * 11 * x)
    else:
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.3) ** 2) / 0.04) + 0.5 * np.exp(-((x - 0.7) ** 2) / 0.04)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


class SearchState:
    """Holds the search state to avoid closures."""

    def __init__(self):
        self.best_seq = None
        self.best_score = float('inf')
        self.deadline = 0.0

    @property
    def time_left(self):
        return max(0.0, self.deadline - time.monotonic())

    def try_update(self, a, label):
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = score_normalized(a)
        if np.isfinite(sc) and sc < self.best_score:
            self.best_score = sc
            self.best_seq = a.copy()
            print(f"  ** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, {self.time_left:.0f}s left)")


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Minimizes 2*n*max(conv(a,a))/(sum(a))^2.
    Uses FISTA-LP, LSE polishing, subgradient, and multi-length strategies.
    """
    rng = np.random.default_rng(seed)
    total_budget = max(10.0, float(budget_s))
    safety_margin = 35.0
    state = SearchState()
    state.deadline = time.monotonic() + (total_budget - safety_margin)

    n_base = 1200
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 10 and np.isfinite(p).all() and p.sum() > 1e-10:
            state.best_seq = p / p.sum()
            n_base = len(state.best_seq)
            state.best_score = score_normalized(state.best_seq)
            print(f"Parent sequence: n={n_base}, score={state.best_score:.8f}")
    except Exception:
        state.best_seq = structured_init(n_base, rng)
        state.best_score = score_normalized(state.best_seq)
        print(f"Default init: n={n_base}, score={state.best_score:.8f}")

    # === Phase 1: FISTA with LP (55% time) ===
    print(f"\n=== Phase 1: FISTA-LP (score={state.best_score:.8f}) ===")
    t1_total = min(0.55 * total_budget, state.time_left * 0.85)
    if t1_total > 30:
        cur_a = state.best_seq.copy()
        n_passes = 4
        for pass_idx in range(n_passes):
            tl = min(t1_total / (n_passes - pass_idx), state.time_left * 0.92)
            if tl < 15:
                break
            print(f"  FISTA pass {pass_idx+1}/{n_passes}, budget={tl:.0f}s")
            a_out, _ = fista_lp(cur_a, tl, max_iters=500000, use_highe=True)
            if a_out is not None:
                sc = score_normalized(a_out)
                if sc < state.best_score:
                    state.best_score = sc
                    state.best_seq = a_out.copy()
                    print(f"    Pass {pass_idx+1} improved: {sc:.8f}")
                cur_a = state.best_seq.copy()
            else:
                break
    print(f"  Phase 1 done: score={state.best_score:.8f}, {state.time_left:.0f}s left")

    # === Phase 2: LSE polishing (20% time) ===
    print(f"\n=== Phase 2: LSE polishing (score={state.best_score:.8f}) ===")
    t2_total = min(0.20 * total_budget, state.time_left * 0.75)
    if t2_total > 20:
        a_l1, _ = lse_lbfgs(state.best_seq, t2_total * 0.4, T_init=30.0, T_min=2.0,
                            n_stages=12, iters_per_stage=45)
        state.try_update(a_l1, "lse_warm")

        if state.time_left > 15:
            a_l2, _ = lse_lbfgs(state.best_seq, state.time_left * 0.4,
                                T_init=10.0, T_min=0.5, n_stages=10, iters_per_stage=40)
            state.try_update(a_l2, "lse_cool")

        if state.time_left > 15:
            pert = state.best_seq + 0.02 * rng.normal(0, 1, len(state.best_seq))
            pert = np.maximum(pert, 0)
            pert = pert / pert.sum()
            a_l3, _ = lse_lbfgs(pert, state.time_left * 0.4,
                                T_init=15.0, T_min=1.0, n_stages=10, iters_per_stage=35)
            state.try_update(a_l3, "lse_pert")

        if state.time_left > 15:
            a_l4, _ = lse_lbfgs(state.best_seq, state.time_left * 0.4,
                                T_init=4.0, T_min=0.2, n_stages=8, iters_per_stage=50)
            state.try_update(a_l4, "lse_fine")
    print(f"  Phase 2 done: score={state.best_score:.8f}, {state.time_left:.0f}s left")

    # === Phase 3: Peak subgradient (12% time) ===
    print(f"\n=== Phase 3: Peak Subgradient (score={state.best_score:.8f}) ===")
    t3_total = min(0.12 * total_budget, state.time_left * 0.7)
    if t3_total > 15:
        a_sg, _ = peak_subgrad(state.best_seq, t3_total * 0.6, n_steps=400000)
        state.try_update(a_sg, "sg_main")
        if state.time_left > 10:
            a_sg2, _ = peak_subgrad(state.best_seq, state.time_left * 0.5, n_steps=300000)
            state.try_update(a_sg2, "sg_2nd")
    print(f"  Phase 3 done: score={state.best_score:.8f}, {state.time_left:.0f}s left")

    # === Phase 4: Multi-length exploration (7% time) ===
    print(f"\n=== Phase 4: Multi-length (score={state.best_score:.8f}) ===")
    t4_total = min(0.07 * total_budget, state.time_left * 0.7)
    if t4_total > 25:
        lengths_to_try = [n_base + 300, n_base + 600, n_base - 200]
        for n_try in lengths_to_try:
            if state.time_left < 15:
                break
            t_try = min(t4_total / 3.0, state.time_left * 0.8)
            if t_try < 10:
                break
            print(f"    Trying n={n_try}")
            init_try = interpolate_seq(state.best_seq, n_try)
            a_try, _ = fista_lp(init_try, t_try * 0.65, max_iters=100000, use_highe=True)
            state.try_update(a_try, f"n{n_try}_fista")
            if state.time_left > 8 and a_try is not None:
                a_try2, _ = lse_lbfgs(a_try, state.time_left * 0.4,
                                      T_init=10.0, T_min=1.0, n_stages=6, iters_per_stage=25)
                state.try_update(a_try2, f"n{n_try}_lse")
    print(f"  Phase 4 done: score={state.best_score:.8f}, {state.time_left:.0f}s left")

    # === Phase 5: Final combined polish (6% time) ===
    print(f"\n=== Phase 5: Final Polish (score={state.best_score:.8f}) ===")
    if state.time_left > 15:
        a_f1, _ = fista_lp(state.best_seq, state.time_left * 0.3, max_iters=50000, use_highe=True)
        state.try_update(a_f1, "final_fista")
        if state.time_left > 10:
            a_f2, _ = lse_lbfgs(state.best_seq, state.time_left * 0.4,
                                T_init=3.0, T_min=0.1, n_stages=6, iters_per_stage=40)
            state.try_update(a_f2, "final_lse")
        if state.time_left > 8:
            a_f3, _ = peak_subgrad(state.best_seq, state.time_left * 0.5, n_steps=100000)
            state.try_update(a_f3, "final_sg")

    # Final verification
    if state.best_seq is not None:
        try:
            final_list = [float(max(0.0, min(1000.0, x))) for x in state.best_seq]
            fs = evaluate_sequence(final_list)
            if np.isfinite(fs):
                print(f"\nVerified via evaluate_sequence: {fs:.8f}")
                if fs < state.best_score:
                    state.best_score = fs
        except Exception:
            pass

    print(f"\nFINAL SCORE: {state.best_score:.8f} (n={len(state.best_seq)})")
    return [float(x) for x in state.best_seq]
