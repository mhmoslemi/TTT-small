def fast_autocorr(a):

def spectral_projection(a, n_iter=2000, alpha=0.4):

def lse_grad(a, T):
    """Compute gradient of (1/T)*log(sum exp(T*b_k)) w.r.t. a."""
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()

    fft_size = 1 << (2 * n - 1).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad, w


def adam_lse_optimize(a0, time_limit, T_start=100.0, T_end=1.0, n_stages=10, lr=0.005):
    """LogSumExp-smoothed optimization with Adam."""

        return e / s

    v = np.zeros(n)

        if time.monotonic() - t0 > time_limit:
            break

        if remaining < 2:
            break
        iters = max(10, min(200, int(remaining * 60 / (n_stages - si + 1))))

            a = softmax_fn(z)
            grad_a, w = lse_grad(a, stage_T)
            S = np.dot(grad_a, a)

            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2

def build_conv_matrix(a_fixed):
    """Build convolution matrix T where (T @ b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):

    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0."""

    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        import cvxpy as cp2

                    break

                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:

def subgradient_polish(a0, time_limit, n_iter_max=100000):
    """Subgradient method targeting the max convolution peak directly."""
    n = len(a0)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()

    for t in range(1, n_iter_max + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        max_b = b.max()
        # Find all peaks within tolerance of max
        tol = max_b * 0.001
        peak_mask = b >= (max_b - tol)
        peak_indices = np.where(peak_mask)[0]

        # Compute subgradient: for each peak k, grad += 2*a[reversed around k]
        # Use FFT for efficiency
        w = np.zeros(len(b))
        w[peak_indices] = 1.0 / len(peak_indices)

        fft_size = 1 << (2 * n - 1).bit_length()
        A_rev = rfft(a[::-1].copy(), fft_size)
        W = rfft(w, fft_size)
        conv = irfft(A_rev * W, fft_size)
        grad = 2.0 * conv[n - 1:n - 1 + n]

        # Step
        lr = 0.5 / np.sqrt(t)
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        sc = score_of(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

        if t % 5000 == 0:
            print(f"      Subgrad t={t}: best={best_score:.7f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):

    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: multi-phase optimization combining LP Nesterov (primary),
    spectral projection, Adam LSE, and enhanced peak shaving.
    Allocated most time to LP Nesterov as it directly optimizes worst-case.
    """
    rng = np.random.default_rng(seed)
    t_globalStart = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    solver = get_lp_solver()
    print(f"LP solver: {solver}, budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent construction
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.7f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0, 1, n_base)
        best_seq = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.15 * np.cos(7 * np.pi * x)
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.7f}")

    n_base = len(best_seq)

    # Phase 1: Quick spectral projection (30s)
    remaining = time_budget - (time.monotonic() - t_globalStart)
    print(f"\n=== Phase 1: Spectral projection n={n_base} (rem={remaining:.0f}s) ===")
    t1 = min(remaining * 0.03, 35.0)
    if t1 > 5:
        a_out = spectral_projection(best_seq, n_iter=3000, alpha=0.5)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = normalize_nonneg(a_out)
            print(f"  NEW BEST [spectral]: {best_score:.7f}")

    # Phase 2: LP Nesterov at base length (main workhorse, ~500s)
    remaining = time_budget - (time.monotonic() - t_globalStart)
    print(f"\n=== Phase 2: LP Nesterov n={n_base} (rem={remaining:.0f}s) ===")
    t2 = min(remaining * 0.35, 550.0)
    if t2 > 30:
        a_out, _, y_out = lp_nesterov_optimize(best_seq, t2, solver)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = normalize_nonneg(a_out)
            print(f"  NEW BEST [lp_n{n_base}]: {best_score:.7f}")
        # Also check y_out
        sc_y = evaluate_safe(y_out)
        if np.isfinite(sc_y) and sc_y < best_score:
            best_score = sc_y
            best_seq = normalize_nonneg(y_out)
            print(f"  NEW BEST [lp_y_n{n_base}]: {best_score:.7f}")

    # Phase 3: Reheat and more LP (200s)
    remaining = time_budget - (time.monotonic() - t_globalStart)
    print(f"\n=== Phase 3: LP Reheat n={n_base} (rem={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 30:
        # Perturb best solution
        a_pert = best_seq + 0.01 * rng.standard_normal(n_base)
        a_pert = normalize_nonneg(a_pert)
        a_out, _, y_out = lp_nesterov_optimize(a_pert, t3, solver)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = normalize_nonneg(a_out)
            print(f"  NEW BEST [lp_reheat]: {best_score:.7f}")
        sc_y = evaluate_safe(y_out)
        if np.isfinite(sc_y) and sc_y < best_score:
            best_score = sc_y
            best_seq = normalize_nonneg(y_out)
            print(f"  NEW BEST [lp_reheat_y]: {best_score:.7f}")

    # Phase 4: Interpolate to n=2400 and optimize
    n_long = 2400
    remaining = time_budget - (time.monotonic() - t_globalStart)
    print(f"\n=== Phase 4: n={n_long} spectral + LP (rem={remaining:.0f}s) ===")
    if remaining > 200:
        init_long = interpolate_sequence(best_seq, n_long)
        # Quick spectral
        a_spec = spectral_projection(init_long, n_iter=1000, alpha=0.4)
        sc = evaluate_safe(a_spec)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = normalize_nonneg(a_spec)
            print(f"  NEW BEST [spectral_n{n_long}]: {best_score:.7f}")

        # LP at longer length
        remaining2 = time_budget - (time.monotonic() - t_globalStart)
        t4 = min(remaining2 * 0.3, 350.0)
        if t4 > 30:
            a_out, _, y_out = lp_nesterov_optimize(a_spec, t4, solver)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = normalize_nonneg(a_out)
                print(f"  NEW BEST [lp_n{n_long}]: {best_score:.7f}")
            sc_y = evaluate_safe(y_out)
            if np.isfinite(sc_y) and sc_y < best_score:
                best_score = sc_y
                best_seq = normalize_nonneg(y_out)
                print(f"  NEW BEST [lp_y_n{n_long}]: {best_score:.7f}")

    # Phase 5: Peak shaving on best (150s)
    remaining = time_budget - (time.monotonic() - t_globalStart)
    print(f"\n=== Phase 5: Peak shaving (rem={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 180.0)
    if t5 > 10:
        a_out, _ = peak_shaving_enhanced(best_seq, t5, seed=456)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = normalize_nonneg(a_out)
            print(f"  NEW BEST [peak_shave]: {best_score:.7f}")

    # Phase 6: Subgradient polish (80s)
    remaining = time_budget - (time.monotonic() - t_globalStart)
    print(f"\n=== Phase 6: Subgradient polish (rem={remaining:.0f}s) ===")
    t6 = min(remaining * 0.08, 80.0)
    if t6 > 10:
        a_out, _ = subgradient_polish(best_seq, t6)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = normalize_nonneg(a_out)
            print(f"  NEW BEST [subgrad]: {best_score:.7f}")

    # Phase 7: Final LP burst (100s)
    remaining = time_budget - (time.monotonic() - t_globalStart)
    print(f"\n=== Phase 7: Final LP burst (rem={remaining:.0f}s) ===")
    t7 = min(remaining * 0.12, 120.0)
    if t7 > 20:
        a_out, _, y_out = lp_nesterov_optimize(best_seq, t7, solver)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = normalize_nonneg(a_out)
            print(f"  NEW BEST [lp_final]: {best_score:.7f}")
        sc_y = evaluate_safe(y_out)
        if np.isfinite(sc_y) and sc_y < best_score:
            best_score = sc_y
            best_seq = normalize_nonneg(y_out)
            print(f"  NEW BEST [lp_final_y]: {best_score:.7f}")

    # Phase 8: Final peak shaving (60s)
    remaining = time_budget - (time.monotonic() - t_globalStart)
    print(f"\n=== Phase 8: Final peak shaving (rem={remaining:.0f}s) ===")
    t8 = min(remaining * 0.1, 60.0)
    if t8 > 5:
        a_out, _ = peak_shaving_enhanced(best_seq, t8, seed=789)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = normalize_nonneg(a_out)
            print(f"  NEW BEST [peak_final]: {best_score:.7f}")

    # Phase 9: Adam LSE final polish (40s)
    remaining = time_budget - (time.monotonic() - t_globalStart)
    print(f"\n=== Phase 9: Adam LSE polish (rem={remaining:.0f}s) ===")
    t9 = min(remaining * 0.06, 40.0)
    if t9 > 5:
        a_out, _ = adam_lse_optimize(best_seq, t9, T_start=10.0, T_end=1.0,
                                      n_stages=5, lr=0.002)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = normalize_nonneg(a_out)
            print(f"  NEW BEST [adam_final]: {best_score:.7f}")

    # Final peak shave with remaining time
    remaining = time_budget - (time.monotonic() - t_globalStart)
    if remaining > 10:
        t10 = min(remaining * 0.5, 30.0)
        if t10 > 5:
            a_out, _ = peak_shaving_enhanced(best_seq, t10, seed=1011)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = normalize_nonneg(a_out)
                print(f"  NEW BEST [peak_very_final]: {best_score:.7f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
            best_seq = normalize_nonneg(best_seq)
        print(f"\nFINAL: score={best_score:.7f}, n={len(best_seq)}")
    else:
        print("ERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
