def get_lp_solver():
    """Find a working LP solver."""

        try:

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.3])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):

def solve_lp_best_response(a_fixed, solver):
    """
    Solve LP: min M s.t. T(a_fixed)@b <= M, sum(b)=1, b>=0.
    Returns (b, M).
    """
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix_sparse(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    try:
        prob.solve(solver=solver)
    except Exception:
        pass

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:

                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = float(M_var.value)

            if improved:
                break

        if not improved:
            stall += 1
            for _ in range(8):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.04)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= -1e-15:
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
                            break

            if stall > 500:
                break
            if stall % 200 == 0:
                elapsed = time.monotonic() - t0
                print(f"    PS stall={stall} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def lse_obj_grad_z(z, T):
    """Objective and gradient for z-parameterization (a = softmax(z))."""
    a = softmax_fn(z)
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    obj_val = (max_Tb + np.log(exp_Tb.sum())) / T

    w = exp_Tb / exp_Tb.sum()
    a_rev = a[::-1].copy()
    conv = fast_convolve(w, a_rev)
    grad_a = 2.0 * conv[n - 1:n - 1 + n]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj_val, grad_z


def logsumexp_optimize(a0, time_limit, T_start=100.0, T_end=0.01, n_stages=15):
    """LogSumExp-smoothed optimization with L-BFGS and multi-stage annealing."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-25))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        iters_per_stage = max(15, min(80, int(20 + 60 * (si / max(1, n_stages - 1)))))

        def func(z_in):
            val, _ = lse_obj_grad_z(z_in, stage_T)
            return val

        def jac(z_in):
            _, g = lse_obj_grad_z(z_in, stage_T)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 200,
                         'disp': False}
            )
            z = res.x
            a = softmax_fn(z)
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
        except Exception:
            pass

        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage={si}/T={stage_T:.4f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 2
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(3 * x) + 0.1 * np.cos(7 * x) + 0.05 * np.cos(13 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * (x - 0.3) ** 2) + 0.7 * np.exp(-4 * (x - 0.7) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = (1 - x) * (0.5 + 0.5 * np.sin(10 * np.pi * x))
    else:
        x = np.linspace(0, 1, n)
        a = np.maximum(0, 1 - 2 * abs(x - 0.5)) + 0.2 * np.sin(5 * np.pi * x)
        a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def nesterov_mw_optimize(a0, time_limit, mw_rounds=300):
    """Nesterov accelerated optimization using multiplicative weights best response."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        t = iter_count

        b_star = best_response_mw(y, n_rounds=mw_rounds)

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        ss = y_new.sum()
        if ss < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / ss

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            ss = a_new.sum()
            if ss < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / ss

        y = y_new
        a = a_new

        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        avg_a += a
        n_avg += 1
        if n_avg >= 50 and n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()

        if iter_count % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MW-NLP it={iter_count} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def propose_candidate():
    """
    Main search function. Multi-phase optimization combining:
    - Nesterov LP at parent length (primary)
    - Nesterov MW (fast) at parent length
    - Multi-length exploration
    - Local peak shaving
    - LogSumExp refinement
    - Final polish
    """
    rng = np.random.default_rng(12345)
    t_global_start = time.monotonic()
    time_budget = min(1800.0 - 30.0, 1770.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    # Load parent sequence
    best_seq = None
    best_score = float('inf')
    parent = None
    n_base = 1600

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            n_base = len(parent)
            p_score = evaluate_safe(parent)
            print(f"Parent: n={n_base}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        print("No valid parent, using default initialization")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    # === Phase 1: Nesterov LP at parent length (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: Nesterov LP at n={n_base} (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.40, 600.0)
    if t1 > 30:
        a_out, _ = nesterov_optimize(best_seq, t1, solver, use_mw=False)
        try_update(a_out, f"nlp_lp_n{n_base}")

    # === Phase 2: Nesterov MW at parent length (fast, more iterations) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: Nesterov MW at n={n_base} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 20:
        a_out, _ = nesterov_mw_optimize(best_seq, t2, mw_rounds=200)
        try_update(a_out, f"nlp_mw_n{n_base}")

    # === Phase 3: Local peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: Peak shaving at n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.10, 150.0)
    if t3 > 10:
        a_out, _ = local_peak_shave(best_seq, t3, seed=42)
        try_update(a_out, f"ps_n{len(best_seq)}")

    # === Phase 4: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: Multi-length exploration (remaining={remaining:.0f}s) ===")

    candidate_lengths = []
    for offset in [-150, 150, 300]:
        nl = n_base + offset
        if nl >= 800 and nl <= 3000:
            candidate_lengths.append(nl)
    candidate_lengths = sorted(set(candidate_lengths), key=lambda x: abs(x - n_base))

    time_per_length = min(remaining * 0.35 / max(1, len(candidate_lengths)), 200.0)

    for nl in candidate_lengths:
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining < 30:
            break
        tl = min(time_per_length, remaining * 0.3)
        if tl < 15:
            continue

        print(f"  Trying n={nl} (time={tl:.0f}s)")
        init_long = interpolate_sequence(best_seq, nl)

        # Use MW (faster) for multi-length
        t_mw = tl * 0.6
        a_out, _ = nesterov_mw_optimize(init_long, t_mw, mw_rounds=150)
        try_update(a_out, f"mw_n{nl}")

        # Quick peak shave
        rem_sub = time_budget - (time.monotonic() - t_global_start)
        t_ps = max(5, min(tl * 0.2, rem_sub * 0.1))
        if t_ps > 5:
            a_out2, _ = local_peak_shave(a_out, t_ps, seed=nl)
            try_update(a_out2, f"ps_n{nl}")

    # === Phase 5: LogSumExp refinement ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LogSumExp at n={n_best} (remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 15:
        a_out, _ = logsumexp_optimize(best_seq, t5, T_start=80.0, T_end=0.01, n_stages=15)
        try_update(a_out, f"lse_n{n_best}")

    # === Phase 6: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final polish at n={n_best} (remaining={remaining:.0f}s) ===")

    # More peak shaving
    t6a = min(remaining * 0.3, 150.0)
    if t6a > 10:
        a_out, _ = local_peak_shave(best_seq, t6a, seed=777)
        try_update(a_out, f"final_ps_n{n_best}")

    # Tight LogSumExp
    remaining = time_budget - (time.monotonic() - t_global_start)
    t6b = min(remaining * 0.25, 100.0)
    if t6b > 10:
        a_out, _ = logsumexp_optimize(best_seq, t6b, T_start=20.0, T_end=0.005, n_stages=12)
        try_update(a_out, f"final_lse_n{n_best}")

    # Another peak shave round
    remaining = time_budget - (time.monotonic() - t_global_start)
    n_best = len(best_seq)
    t6c = min(remaining * 0.25, 100.0)
    if t6c > 10:
        a_out, _ = local_peak_shave(best_seq, t6c, seed=999)
        try_update(a_out, f"final_ps2_n{n_best}")

    # One more Nesterov MW pass at the best length
    remaining = time_budget - (time.monotonic() - t_global_start)
    n_best = len(best_seq)
    t6d = min(remaining * 0.2, 80.0)
    if t6d > 10:
        a_out, _ = nesterov_mw_optimize(best_seq, t6d, mw_rounds=150)
        try_update(a_out, f"final_mw_n{n_best}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    # Ensure output is valid
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return result
