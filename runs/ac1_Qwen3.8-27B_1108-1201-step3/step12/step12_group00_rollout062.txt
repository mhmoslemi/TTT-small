    # This is (w * a)[i] where the convolution is truncated
    # Actually: sum_k w[k]*a[k-i] = sum_j w[i+j]*a[j] (with j = k-i)
    # This is the cross-correlation of w and a at lag i
    # = ifft(fft(w) * conj(fft(a))) ... hmm, let me just use the direct formula
    
    # Simpler: grad[i] = 2 * sum_k w[k] * (a[k-i] + a[i-k]) / 2 ... 
    # Actually grad[i] = sum_k w[k] * (a[k-i] + a[i-k]) where terms are 0 if index out of range
    # = (w ⊛ a)[i] + (a ⊛ w)[i] where ⊛ is cross-correlation
    # Cross-correlation: (w ⊛ a)[i] = sum_k w[k] * a[k-i]
    # = (w * a_rev) reversed... 
    
    # Let me use FFT properly:
    # (w ⊛ a)[i] = sum_k w[k] a[k-i] 
    # Let me define: c[i] = sum_k w[k] a[k-i]
    # c[i] = sum_j w[i+j] a[j] (substituting j = k-i)
    # This is the "deconvolution" or cross-correlation
    
    # Using FFT: c = ifft(fft(w, N) * conj(fft(a, N)))[:n] ... no that's not right either
    # 
    # Let me just use the fact that:
    # sum_k w[k] * b[k] where b[k] = sum_j a[j]*a[k-j]
    # d/da[i] of b[k] = a[k-i] + a[i-k] (if indices valid)
    # So grad[i] = sum_k w[k] * (a[k-i] + a[i-k])
    # 
    # Note that sum_k w[k]*a[k-i] = cross-correlation of w and a at lag i
    # And sum_k w[k]*a[i-k] = cross-correlation of a and w at lag i (same thing by symmetry of the sum)
    # Wait: sum_k w[k]*a[i-k] = sum_j w[i-j]*a[j] (let j=i-k) = cross-correlation of w and a at lag i
    # So actually grad[i] = 2 * sum_k w[k]*a[k-i] = 2 * (w ⊛ a)[i]
    # where (w ⊛ a)[i] = sum_k w[k]*a[k-i]
    #
    # (w ⊛ a)[i] = ifft(FFT(w) * conj(FFT(a))) ... let me verify
    # FFT: W[m] = sum_k w[k] exp(-2πimk/N), A[m] = sum_j a[j] exp(-2πimj/N)
    # ifft(W * conj(A))[i] = (1/N) sum_m W[m] conj(A[m]) exp(2πimi/N)
    # = (1/N) sum_m [sum_k w[k] e^{-2πimk/N}] [sum_j a[j] e^{2πimj/N}] e^{2πimi/N}
    # = (1/N) sum_k sum_j w[k] a[j] sum_m e^{2πim(j-k+i)/N}
    # = sum_k sum_j w[k] a[j] * δ_{j-k+i, 0 (mod N)}
    # = sum_k w[k] a[k-i] (mod N)
    # 
    # Yes! So (w ⊛ a)[i] = ifft(FFT(w) * conj(FFT(a)))[i] (circular, with appropriate zero-padding for linear)
    
    N = fft_size
    W_fft = rfft(w, N)  # wait, I need full FFT for this, not rfft
    # Actually for real signals, ifft(FFT(w) * conj(FFT(a))) is real, so I can use rfft/irfft
    # ifft(FFT(w) * conj(FFT(a)))[i] = ifft(rfft(w) * conj(rfft(a)))[i] for real w, a
    # But conj(rfft(a)) is not the same as rfft(a[::-1]) in general...
    # Let me just use the full approach:
    # For real w, a: ifft(FFT(w) * conj(FFT(a))) = cross-correlation
    # Using rfft: the result is real, so irfft(rfft(w) * np.conj(rfft(a))) should work
    # Wait, that's not right. Let me think again.
    # For real signals: conj(FFT(a)) = FFT(a)[::-1] (conjugate symmetry)
    # So FFT(w) * conj(FFT(a)) = FFT(w) * FFT(a)[::-1]
    # Hmm, this is getting complicated with rfft. Let me just compute it directly.
    
    # Direct computation using the convolution structure:
    # grad[i] = sum_k w[k] * (a[k-i] + a[i-k])
    # Let me define c[i] = sum_k w[k] * a[k-i] (with a[j]=0 for j outside [0,n-1])
    # c[i] = sum_{k} w[k] * a[k-i] where 0 <= k-i < n, i.e., i <= k < i+n
    # = sum_{j=0}^{n-1} w[i+j] * a[j] where 0 <= i+j < 2n-1 (since w has length 2n-1)
    # This is a "cross-correlation" that I can compute with FFT:
    # c = ifft(fft(w, N) * conj(fft(a, N)))[:n] ... but this gives circular cross-correlation
    # For linear: zero-pad to N >= (2n-1) + n - 1 = 3n-2
    
    corr = np.zeros(n)
    # Use the fact that c[i] = sum_j w[i+j] * a[j]
    # This is like a "sliding dot product" of w and a
    # FFT approach: result = ifft(fft(w, 3n) * conj(fft(a, 3n)))[:n]... 
    
    # Actually, let me just use numpy's correlate or do it with the convolution matrix approach
    # For efficiency, let me use FFT with proper zero-padding
    N_c = 1 << (3 * n - 1).bit_length()
    W_full = rfft(w, N_c)
    A_full = rfft(a, N_c)
    # cross-correlation: ifft(FFT(w) * conj(FFT(a)))
    # For real signals with rfft, conj(FFT(a)) corresponds to FFT of a flipped and...
    # Let me just use: ifft(FFT(w) * conj(FFT(a))) = correlate(w, a) in scipy notation
    # In terms of rfft: this is irfft(W_full * np.conj(A_full), N_c)[:n]
    # But wait, is this correct? For real signals:
    # FFT(w)[m] * conj(FFT(a))[m] for m = 0..N/2
    # The result has conjugate symmetry, so irfft should work
    # Let me verify: for N even, FFT result is [X[0], X[1], ..., X[N/2]]
    # conj(FFT(a))[m] = conj(A[m])
    # Product: [W[0]*conj(A[0]), W[1]*conj(A[1]), ..., W[N/2]*conj(A[N/2])]
    # This has conjugate symmetry (for real w, a), so irfft gives the correct result
    corr_full = irfft(W_full * np.conj(A_full), N_c)
    # corr_full[i] = sum_k w[k] * a[(k-i) mod N_c]
    # For i in [0, n-1] and proper zero-padding, this gives sum_k w[k] * a[k-i] (linear)
    corr = corr_full[:n]
    
    # Similarly for the other term: sum_k w[k] * a[i-k]
    # = sum_k a[k] * w[i-k] (substituting k -> i-k)
    # = same as corr (by symmetry of the sum, since both w and a are being summed over all valid indices)
    # Wait: sum_k w[k]*a[i-k] where 0 <= i-k < n means k in [i-n+1, i]
    # = sum_j a[j]*w[i-j] where j = i-k
    # This is the cross-correlation of a and w at lag i, which is corr[i] (same!)
    # Because corr[i] = sum_k w[k]*a[k-i] = sum_j w[i+j]*a[j]
    # And sum_k a[k]*w[i-k] = sum_j a[i-j]*w[j] = sum_j' a[i+j']*w[i-2j']... no this is getting confused.
    
    # Let me just be careful:
    # Term 1: T1[i] = sum_k w[k] * a[k-i] (a[k-i] = 0 if k-i out of [0,n-1])
    # Term 2: T2[i] = sum_k w[k] * a[i-k] (a[i-k] = 0 if i-k out of [0,n-1])
    # 
    # T1[i] = sum_{k=i}^{i+n-1} w[k] * a[k-i] (since k-i in [0,n-1] means k in [i, i+n-1])
    #       = sum_{j=0}^{n-1} w[i+j] * a[j]
    # 
    # T2[i] = sum_{k=0}^{i} w[k] * a[i-k] (since i-k in [0,n-1] means k in [max(0,i-n+1), i])
    #       Actually i-k in [0,n-1] means i-n+1 <= k <= i
    #       = sum_{j=0}^{n-1} w[i-j] * a[j] (letting j = i-k)
    #       = sum_{j=0}^{min(i,n-1)} w[i-j] * a[j]
    # 
    # So T1[i] = sum_j w[i+j] * a[j] and T2[i] = sum_j w[i-j] * a[j]
    # T1 + T2 = sum_j a[j] * (w[i+j] + w[i-j])
    # 
    # For T1: this is the cross-correlation I computed above: corr[i] = T1[i]
    # For T2: this is sum_j a[j]*w[i-j], which is the convolution (a * w)[i] (truncated)
    # = ifft(FFT(a) * FFT(w))[i] (with zero-padding)
    
    conv_aw = irfft(A_full * W_full, N_c)[:n]  # (a * w)[i] = sum_j a[j] * w[i-j]
    
    grad = corr + conv_aw  # grad[i] = T1[i] + T2[i]
    
    return logsumexp, grad, b, weights


def adam_lse_optimize(a0, time_limit, beta_start=5.0, beta_end=200.0, verbose=True):
    """Adam optimizer on LSE-smoothed objective."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_to_sum(a, norm_val)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5
    t_step = 0
    
    # Track for adaptive beta
    n_total_steps = int(time_limit * 10)  # estimate
    t0_time = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        t_step += 1
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        # Exponential schedule for beta
        beta = beta_start * (beta_end / beta_start) ** progress
        # Decay learning rate
        lr_t = lr / (1.0 + 0.05 * t_step)
        
        val, grad, b, weights = lse_objective_and_grad(a, beta)
        
        # Subgradient of the true max objective (for tracking)
        k_star = int(np.argmax(b))
        M_curr = b[k_star]
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t_step)
        v_hat = v / (1 - beta2 ** t_step)
        
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_to_sum(a_new, norm_val)
        
        a = a_new
        
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if verbose and t_step % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AdamLSE step={t_step} ({elapsed:.0f}s): peak={best_score:.8f}, beta={beta:.1f}")
    
    return best_a, best_score


def dual_avg_lp_optimize(a0, time_limit, solver, verbose=True):
    """LP best-response with dual averaging Nesterov."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_to_sum(a, norm_val)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_iters = 0
    
    y = a.copy()
    dual_avg = np.zeros(n)
    n_dual = 0
    force_avg = np.zeros(n)
    n_force = 0
    
    last_lp_time = 0.0
    lp_count = 0
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        lp_count += 1
        lp_time = time.monotonic() - t0 - last_lp_time
        last_lp_time = time.monotonic() - t0
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm_val / b_sum
        n_iters += 1
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize_to_sum(y_new, norm_val)
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_to_sum(a_new, norm_val)
        
        y = y_new
        a = a_new
        
        # Dual averaging: maintain running average with weights
        dual_avg += a
        n_dual += 1
        # Force-averaged: different weight schedule
        force_avg += a * (1.0 / t)
        n_force += 1.0 / t
        
        # Check best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        # Check dual average periodically
        if n_dual % 20 == 0:
            dual_avg_vec = dual_avg / n_dual
            dual_avg_vec = normalize_to_sum(dual_avg_vec, norm_val)
            score_dual = fast_autocorr(dual_avg_vec).max()
            if score_dual < best_score:
                best_score = score_dual
                best_a = dual_avg_vec.copy()
        
        if n_force > 1e-10:
            force_avg_vec = force_avg / n_force
            force_avg_vec = normalize_to_sum(force_avg_vec, norm_val)
            score_force = fast_autocorr(force_avg_vec).max()
            if score_force < best_score:
                best_score = score_force
                best_a = force_avg_vec.copy()
        
        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            avg_lp_time = elapsed / max(lp_count, 1)
            print(f"    DLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, avg_lp={avg_lp_time:.3f}s")
    
    return best_a, best_score


def spectral_coordinate_descent(a0, time_limit, seed=42, verbose=True):
    """Coordinate descent targeting top convolution peaks."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_to_sum(a, norm_val)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    n_tries = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        # Get top peaks
        top_peaks = np.argsort(b)[::-1][:15]
        
        improved = False
        for k in top_peaks:
            # Find contributing pairs for this peak
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            
            if contrib.max() < 1e-18:
                continue
            
            # Source: the pair with highest contribution
            src = idx[np.argmax(contrib)]
            
            # Destination: find position that contributes least to ALL top peaks
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True
            
            # Candidates: positions NOT in any peak's contributing range
            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]
            
            # Pick from candidates with lowest current value
            n_cands = min(len(candidates), max(10, len(candidates) // 4))
            candidates_sorted = candidates[np.argsort(a[candidates])][:n_cands]
            
            if len(candidates_sorted) == 0:
                continue
            
            dst = candidates_sorted[rng.integers(len(candidates_sorted))]
            
            # Try different transfer fractions
            for frac in [0.001, 0.003, 0.008, 0.015, 0.03, 0.05]:
                amt = a[src] * frac
                if amt < 1e-18 or a[src] - amt < 0:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = normalize_to_sum(a_trial, norm_val)
                score = fast_autocorr(a_trial).max()
                n_tries += 1
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            if stall % 50 == 0:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-15:
                    frac = rng.uniform(0.005, 0.03)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        a_trial = normalize_to_sum(a_trial, norm_val)
                        score = fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
            if stall > 500:
                break
        
        if verbose and n_tries % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDescent tries={n_tries} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, seed=0):
    rng = np.random.default_rng(seed)
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3*np.cos(2.0*x) + 0.15*np.cos(5.0*x) + 0.05*np.cos(9.0*x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0*(x-0.25)**2) + 0.6*np.exp(-4.0*(x-0.75)**2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0*x - 1.0)) ** 2
    else:
        # Multi-frequency with random phases
        x = np.linspace(0.0, 1.0, n)
        a = np.ones(n) * 0.5
        for freq in [3, 7, 13, 23]:
            phase = rng.uniform(0, 2*np.pi)
            a += 0.1 * np.cos(2*np.pi*freq*x + phase)
        a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """

    5. Final coordinate polish
    """

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad ** 2
        m_hat = m / (1 - b1 ** t_step)
        v_hat = v / (1 - b2 ** t_step)

        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_to_sum(a_new, norm_val)
        a = a_new

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and t_step % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AdamLSE step={t_step} ({elapsed:.0f}s): peak={best_score:.8f}, beta={beta:.1f}")

    return best_a, best_score


def dual_avg_lp_optimize(a0, time_limit, solver, verbose=True):
    """LP best-response with dual-averaging Nesterov acceleration."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_to_sum(a, norm_val)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_iters = 0

    y = a.copy()
    dual_avg = np.zeros(n)
    n_dual = 0
    force_avg = np.zeros(n)
    n_force = 0.0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm_val / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize_to_sum(y_new, norm_val)

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_to_sum(a_new, norm_val)

        y = y_new
        a = a_new

        dual_avg += a
        n_dual += 1
        force_avg += a / t
        n_force += 1.0 / t

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        if n_dual % 25 == 0:
            da = dual_avg / n_dual
            da = normalize_to_sum(da, norm_val)
            sc = fast_autocorr(da).max()
            if sc < best_score:
                best_score = sc
                best_a = da.copy()

        if n_force > 1e-8:
            fa = force_avg / n_force
            fa = normalize_to_sum(fa, norm_val)
            sc = fast_autocorr(fa).max()
            if sc < best_score:
                best_score = sc
                best_a = fa.copy()

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    DLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def spectral_coordinate_descent(a0, time_limit, seed=42, verbose=True):
    """Coordinate descent targeting top convolution peaks with spectral guidance."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_to_sum(a, norm_val)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    n_tries = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_peaks = np.argsort(b)[::-1][:12]

        improved = False
        for k in top_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            if contrib.max() < 1e-18:
                continue

            src = idx[np.argmax(contrib)]

            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]

            n_cands = min(len(candidates), max(5, len(candidates) // 5))
            candidates_sorted = candidates[np.argsort(a[candidates])][:n_cands]

            if len(candidates_sorted) == 0:
                continue

            dst = candidates_sorted[rng.integers(len(candidates_sorted))]

            for frac in [0.001, 0.003, 0.008, 0.02, 0.04, 0.07]:
                amt = a[src] * frac
                if amt < 1e-18 or a[src] - amt < 0:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = normalize_to_sum(a_trial, norm_val)
                score = fast_autocorr(a_trial).max()
                n_tries += 1
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

            if improved:
                break

        if not improved:
            stall += 1
            if stall % 30 == 0:
                for _ in range(3):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a[i] > 1e-12:
                        frac = rng.uniform(0.005, 0.04)
                        amt = a[i] * frac
                        a_trial = a.copy()
                        a_trial[i] -= amt
                        a_trial[j] += amt
                        if a_trial[i] >= 0:
                            a_trial = normalize_to_sum(a_trial, norm_val)
                            score = fast_autocorr(a_trial).max()
                            if score < best_score:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                stall = 0
                                break
            if stall > 400:
                break

        if verbose and n_tries % 400 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDescent tries={n_tries} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, seed=0):
    rng = np.random.default_rng(seed)
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    else:
        x = np.linspace(0.0, 1.0, n)
        a = np.ones(n) * 0.5
        for freq in [3, 7, 13, 23, 37]:
            phase = rng.uniform(0, 2 * np.pi)
            a += 0.08 * np.cos(2 * np.pi * freq * x + phase)
        a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def harmonic_descent(a0, time_limit, seed=42, verbose=True):
    """Harmonic-mean descent: smooth the convolution by redistributing mass
    from positions that feed into peaks to positions that feed into valleys."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize_to_sum(a, norm_val)

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_steps = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        b_mean = b.mean()
        b_std = b.std()
        if b_std < 1e-15:
            break

        # Identify positions that most contribute to the peak vs valley
        k_peak = int(np.argmax(b))
        k_valley = int(np.argmin(b))

        # Contribution to peak from position i: 2*a[k_peak - i] (if valid)
        # We want to move mass from high contributors to low contributors
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        peak_idx = np.arange(js, je + 1)
        peak_contrib = a[peak_idx] * a[k_peak - peak_idx]

        # Find source (high contributor to peak) and destination (low contributor to valley)
        if peak_contrib.max() > 1e-18:
            src = peak_idx[np.argmax(peak_contrib)]

            # For destination, find position with low contribution to peak AND low value
            low_contrib_mask = peak_contrib < np.median(peak_contrib)
            low_positions = peak_idx[low_contrib_mask]
            if len(low_positions) > 0:
                dst = low_positions[np.argmin(a[low_positions])]
            else:
                dst = src

            if src != dst and a[src] > 1e-14:
                for frac in [0.002, 0.005, 0.01, 0.02]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    a_trial = normalize_to_sum(a_trial, norm_val)
                    score = fast_autocorr(a_trial).max()
                    n_steps += 1
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        break

        # Also try random perturbations weighted towards low-convolution regions
        if n_steps % 10 == 0:
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = normalize_to_sum(a_trial, norm_val)
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()

        if verbose and n_steps % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    HarmonicDescent steps={n_steps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

    parent = None

            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                n_base = len(parent)
                print(f"Parent: n={n_base}, score={best_score:.6f}")
    except Exception:
