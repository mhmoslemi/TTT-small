```python
def proj_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def proj_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def subgradient_of_max_autocorr(a, topk=1):
    """Compute (scaled) subgradient of max_k (a*a)_k at a."""
    n = len(a)
    b = fast_autocorr(a)
    k_use = min(topk, len(b))
    top_idx = np.argsort(b)[::-1][:k_use]
    # Equal weight subgradient
    g = np.zeros(n)
    for idx in range(k_use):
        k = top_idx[idx]
        js = max(0, k - (n-1))
        je = min(n-1, k)
        j_range = np.arange(js, je+1)
        g[j_range] += (2.0 / k_use) * a[k - j_range]
    return g, b.max()

def nesterov_subgradient(a0, time_limit, topk=1, eta0=None, max_iters=1000000):
    """Nesterov accelerated subgradient method."""
    n = len(a0)
    a = a0.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    x_prev = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    
    if eta0 is None:
        eta0 = 0.5  # Will be tuned
    
    t0 = time.monotonic()
    stall = 0
    stall_patience = 10000
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        # Compute subgradient at y
        g, bmax = subgradient_of_max_autocorr(y, topk=topk)
        
        # Step size
        eta = eta0 / np.sqrt(t)
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = y - eta * g
        y_new = proj_simplex(y_new)
        
        if t == 1:
            x_new = y_new.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            x_new = y_new + momentum * (y_new - y)
            x_new = np.maximum(x_new, 0)
            x_new = proj_simplex(x_new)
        
        y = y_new
        a = x_new
        
        # Evaluate
        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Also check y
        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        
        if t % 100000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad t={t} ({elapsed:.0f}s): best={best_score:.6f}, eta={eta:.6f}")
        
        if stall > stall_patience:
            # Restart
            y = best_a.copy()
            x_prev = best_a.copy()
            a = best_a.copy()
            stall = 0
            eta0 = eta0 * 0.5  # Reduce step size
    
    return best_a, best_score
```

```python
"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2:
1. Nesterov LP zero-sum game (main workhorse) with adaptive restart
2. Subgradient Nesterov on the convex max-autocorrelation problem
3. Full top-k LogSumExp smooth approximation with Adam
4. Multi-scale n exploration with interpolation
5. Peak shaving with danger-zone awareness
6. Multi-start diversity for escaping local minima

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def get_lp_solver():
    """Find the fastest working LP solver."""
    best_solver = cp.ECOS
    best_time = float('inf')
    for solver_name in [cp.PDLP, cp.MOSEK, cp.ECOS, cp.SCS, cp.CLARABEL, cp.SCS]:
        try:
            x = cp.Variable(200, nonneg=True)
            M = cp.Variable()
            T_mat = np.random.rand(400, 200)
            prob = cp.Problem(cp.Minimize(M), [T_mat @ x <= M, cp.sum(x) == 1])
            t0 = time.monotonic()
            prob.solve(solver=solver_name, verbose=False)
            elapsed = time.monotonic() - t0
            if prob.status in ('optimal', 'optimal_inaccurate') and elapsed < best_time:
                best_time = elapsed
                best_solver = solver_name
        except Exception:
            continue
    return best_solver


def build_conv_matrix_vec(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b) using vectorized ops."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix_vec(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS, cp.MOSEK, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, max_iters=500000):
    """
    Nesterov-accelerated LP zero-sum game with adaptive restart and running averages.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    # Running averages
    avg_iter = np.zeros(n)
    n_avg = 0
    WINDOW = 50
    cbuf = [None] * WINDOW
    cidx = 0
    avg_cbuf = np.zeros(n)

    stall = 0
    restart_count = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit - 2:
            break

        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        ys = y_new.sum()
        if ys < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / ys

        if t == 1:
            a_new = y_new.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new

        y = y_new
        a = a_new

        # Update running averages
        avg_iter += a
        n_avg += 1
        if cbuf[cidx] is not None:
            avg_cbuf -= cbuf[cidx]
        cbuf[cidx] = a.copy()
        avg_cbuf += a
        cidx = (cidx + 1) % WINDOW

        # Evaluate candidates
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0

        if n_avg % 50 == 0 and n_avg >= 100:
            avg_check = avg_iter / n_avg
            sc = 2.0 * n * fast_autocorr(avg_check).max()
            if sc < best_score:
                best_score = sc
                best_a = avg_check.copy()
                stall = 0

        if n_avg >= WINDOW and n_avg % 50 == 0:
            avg_short = avg_cbuf / WINDOW
            sc = 2.0 * n * fast_autocorr(avg_short).max()
            if sc < best_score:
                best_score = sc
                best_a = avg_short.copy()
                stall = 0

        # Adaptive restart
        if stall > 500 and t > 100:
            a = best_a.copy()
            y = best_a.copy()
            avg_iter = np.zeros(n)
            n_avg = 0
            cbuf = [None] * WINDOW
            cidx = 0
            avg_cbuf = np.zeros(n)
            stall = 0
            restart_count += 1
            if restart_count % 5 == 0:
                elapsed = time.monotonic() - t0
                print(f"      LP restart #{restart_count} at t={t} ({elapsed:.0f}s): best={best_score:.6f}")

        if t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}, restarts={restart_count}")

    return best_a, best_score


def subgrad_nesterov(a0, time_limit, topk=3, eta0=0.3, max_iters=2000000):
    """
    Nesterov accelerated subgradient method for min max_k (a*a)_k on simplex.
    Much faster per iteration than LP Nesterov.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    eta_cur = eta0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit - 3:
            break

        # Compute subgradient at y
        b = fast_autocorr(y)
        k_use = min(topk, len(b))
        top_idx = np.argsort(b)[::-1][:k_use]
        g = np.zeros(n)
        for idx in range(k_use):
            k = top_idx[idx]
            js = max(0, k - (n - 1))
            je = min(n - 1, k)
            j_range = np.arange(js, je + 1)
            g[j_range] += (2.0 / k_use) * y[k - j_range]

        # Step size
        eta = eta_cur / np.sqrt(t)

        # Nesterov step
        alpha_t = 2.0 / (t + 2.0)
        y_new = y - eta * g
        y_new = proj_simplex(y_new)

        if t == 1:
            a_new = y_new.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            a_new = proj_simplex(a_new)

        y = y_new
        a = a_new

        # Evaluate
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0

        if t % 100000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad t={t} ({elapsed:.0f}s): best={best_score:.6f}, eta={eta:.6f}")

        # Adaptive restart
        if stall > 200000 and t > 50000:
            y = best_a.copy()
            a = best_a.copy()
            stall = 0
            eta_cur *= 0.7

    return best_a, best_score


def lse_adam_optimize(a0, time_limit, T_start=200.0, T_end=1.0, n_stages=20, topk=30):
    """
    Top-k LogSumExp smoothing with Adam optimizer.
    Uses higher T and more peaks than previous version.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit - 3:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        base_lr = 3e-3 * min(1.0, 60.0 / n)
        lr = base_lr * (0.8 ** si)
        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.85, 0.999, 1e-8

        max_it = max(20, int(min(remaining, 80.0) * 5))

        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 3:
                break

            a = softmax_fn(z)
            b = fast_autocorr(a)
            k_use = min(topk, len(b))
            top_idx = np.argsort(b)[::-1][:k_use]
            top_vals = b[top_idx]

            Tc = stage_T * top_vals
            max_Tc = Tc.max()
            exp_Tc = np.exp(Tc - max_Tc)
            lse_val = (max_Tc + np.log(exp_Tc.sum())) / stage_T
            w = exp_Tc / exp_Tc.sum()

            # Gradient w.r.t. a
            grad_a = np.zeros(n)
            for idx in range(k_use):
                K = top_idx[idx]
                wK = w[idx]
                js = max(0, K - (n - 1))
                je = min(n - 1, K)
                idx_range = np.arange(js, je + 1)
                grad_a[idx_range] += 2.0 * wK * a[K - idx_range]

            # Chain rule to z-space
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)

            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            if (it + 1) % 5 == 0:
                a_check = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a_check).max()
                if score < best_score:
                    best_score = score
                    best_a = a_check.copy()

        a_end = softmax_fn(z)
        score_end = 2.0 * n * fast_autocorr(a_end).max()
        if score_end < best_score:
            best_score = score_end
            best_a = a_end.copy()

        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {si}/{n_stages} (T={stage_T:.1f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def peak_shaving_refined(a0, time_limit, seed=42):
    """
    Refined peak shaving: identifies contributing pairs to top peaks and
    moves mass to minimize peak growth.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 3:
        b = fast_autocorr(a)
        bmax = b.max()
        threshold = bmax * 0.998
        top_peaks = np.where(b >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(b)[::-1][:10]
        top_peaks = top_peaks[:30]

        improved = False
        for k in top_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            max_c = contrib.max()
            if max_c < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            # Build danger set
            danger = np.zeros(n, dtype=bool)
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                danger[js2:je2 + 1] = True

            safe = np.where(~danger)[0]
            safe = safe[safe != src]
            if len(safe) == 0:
                safe = np.where(np.arange(n) != src)[0]
            if len(safe) == 0:
                continue

            safe_sorted = safe[np.argsort(a[safe])]
            dst_pool = safe_sorted[:max(5, len(safe_sorted) // 4)]
            if len(dst_pool) == 0:
                continue

            for attempt in range(min(5, len(dst_pool))):
                dst = dst_pool[attempt]
                for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Small random perturbation
            a_trial = a.copy()
            n_pert = rng.integers(1, 5)
            for _ in range(n_pert):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a_trial[i] > 1e-12:
                    frac = rng.uniform(0.002, 0.03)
                    amt = a_trial[i] * frac
                    a_trial[i] -= amt
                    a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 1e-10:
                a_trial = a_trial / st
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0

            if stall > 500:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n + 1)
    f_padded = np.concatenate([f_norm, [0]])
    new_x = np.linspace(0, 1, new_n + 1)
    f_interp = np.interp(new_x, old_x, f_padded)
    f_interp = f_interp[:-1]
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        return np.ones(n) / n
    return a / s


def perturbed_init(a0, rng, scale=0.08):
    """Create a perturbed version of a sequence."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a0)) / len(a0)
    a = a / s
    noise = rng.normal(0, scale, len(a))
    a_new = a + noise * (a + 1e-8)
    a_new = np.maximum(a_new, 0)
    sn = a_new.sum()
    if sn < 1e-15:
        return a.copy()
    return a_new / sn


def propose_candidate(seed=42, time_budget=1800.0):
    """
    Multi-phase hybrid optimizer:
    1. LP Nesterov (main, ~60% of time)
    2. Subgradient Nesterov (fast iterations, ~20% of time)
    3. LSE Adam (smooth refinement, ~10% of time)
    4. Multi-n exploration (~10% of time)
    5. Peak shaving polish (remaining time)
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    time_budget = min(float(time_budget) - 60.0, 1740.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    # Get parent sequence
    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    n_base = len(best_seq) if best_seq is not None else 1600

    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # === Phase 1: LP Nesterov at base length (main workhorse) ===
    elapsed = time.monotonic() - t_start
    remaining = time_budget - elapsed
    t1 = min(remaining * 0.45, 780.0)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 60:
        a_out, sc_out = nesterov_lp_optimize(best_seq, t1, solver)
        try_update(a_out, f"lp_main_n{n_base}")

    # === Phase 2: Subgradient Nesterov (fast, many iterations) ===
    elapsed = time.monotonic() - t_start
    remaining = time_budget - elapsed
    t2 = min(remaining * 0.20, 350.0)
    print(f"=== Phase 2: Subgrad Nesterov n={n_base} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, sc_out = subgrad_nesterov(best_seq, t2, topk=5, eta0=0.2)
        try_update(a_out, f"subgrad_n{n_base}")

    # === Phase 3: LSE Adam refinement ===
    elapsed = time.monotonic() - t_start
    remaining = time_budget - elapsed
    t3 = min(remaining * 0.12, 200.0)
    print(f"=== Phase 3: LSE Adam n={n_base} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, sc_out = lse_adam_optimize(best_seq, t3, T_start=300.0, T_end=2.0, n_stages=20, topk=40)
        try_update(a_out, f"lse_n{n_base}")

    # === Phase 4: Try longer n ===
    elapsed = time.monotonic() - t_start
    remaining = time_budget - elapsed
    t4 = min(remaining * 0.12, 200.0)
    print(f"=== Phase 4: LP Nesterov n={n_base + 200} (budget={t4:.0f}s) ===")
    if t4 > 40:
        n_long = n_base + 200
        init_long = interpolate_sequence(best_seq, n_long)
        a_out, sc_out = nesterov_lp_optimize(init_long, t4, solver)
        try_update(a_out, f"lp_n{n_long}")

    # === Phase 5: Multi-start with perturbation + quick LP ===
    elapsed = time.monotonic() - t_start
    remaining = time_budget - elapsed
    t5_total = min(remaining * 0.10, 150.0)
    print(f"=== Phase 5: Multi-start (budget={t5_total:.0f}s) ===")
    if t5_total > 30:
        n_starts = 3
        t5_each = t5_total / n_starts
        for si in range(n_starts):
            if time.monotonic() - t_start > time_budget - 60:
                break
            pert = perturbed_init(best_seq, rng, scale=0.04 * (si + 1))
            a_out, _ = nesterov_lp_optimize(pert, t5_each, solver)
            try_update(a_out, f"multistart_{si}")

    # === Phase 6: Peak shaving ===
    elapsed = time.monotonic() - t_start
    remaining = time_budget - elapsed
    t6 = min(remaining * 0.10, 120.0)
    print(f"=== Phase 6: Peak shaving (budget={t6:.0f}s) ===")
    if t6 > 10:
        a_out, sc_out = peak_shaving_refined(best_seq, t6, seed=777)
        try_update(a_out, "peak_shave")

    # === Phase 7: Final LSE polish (low temperature) ===
    elapsed = time.monotonic() - t_start
    remaining = time_budget - elapsed
    t7 = min(remaining * 0.20, 100.0)
    print(f"=== Phase 7: Final LSE (budget={t7:.0f}s) ===")
    if t7 > 10:
        a_out, sc_out = lse_adam_optimize(best_seq, t7, T_start=50.0, T_end=0.5, n_stages=10, topk=20)
        try_update(a_out, "final_lse")

    # === Phase 8: Final peak shave ===
    elapsed = time.monotonic() - t_start
    remaining = time_budget - elapsed
    t8 = min(remaining * 0.30, 60.0)
    print(f"=== Phase 8: Final peak shave (budget={t8:.0f}s) ===")
    if t8 > 5:
        a_out, sc_out = peak_shaving_refined(best_seq, t8, seed=999)
        try_update(a_out, "peak_shave_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
```
