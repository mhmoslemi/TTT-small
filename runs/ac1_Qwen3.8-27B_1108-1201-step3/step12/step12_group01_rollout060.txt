    n = len(a)
    v = a[i]
    s = a.sum() - v  # sum of all other elements
    
    # Compute base convolution (without a[i]'s contribution)
    conv_full = fast_autocorr(a)
    base = conv_full.copy()

    ki = k - i  # for each k

        if 0 <= j < n:
            if j == i:
                base[k] -= v * v
            else:
                base[k] -= 2 * v * a[j]
    
    # Now optimize v'
    # f(v') = 2*n * max_k(base[k] + contrib_k(v')) / (s + v')^2
    # where contrib_k(v') = 2*v'*a[k-i] if 0<=k-i<n and k-i!=i
    #                       v'^2 if k-i==i
    #                       0 otherwise

    # Linear: for k where 0<=k-i<n and k-i!=i: base[k] + 2*a[k-i]*v'
    # Quadratic: for k=2i: base[2i] + v'^2
    # Constant: for other k: base[k]

    for k in range(2*n - 1):

        if not (0 <= j < n):
            max_const = max(max_const, base[k])
    
    # Linear terms
    linear_coeffs = []  # (c, base_k) where term is base_k + c*v'
    for k in range(2*n - 1):
        j = k - i
        if 0 <= j < n and j != i:
            linear_coeffs.append((2 * a[j], base[k]))
    
    quad_base = base[2*i] if 2*i < 2*n - 1 else 0  # base[2i]

        max_val = max_const
        for c, bk in linear_coeffs:
            val = bk + c * vprime
            if val > max_val:
                max_val = val
        if 2*i < 2*n - 1:
            val = quad_base + vprime * vprime
            if val > max_val:
                max_val = val
        total = s + vprime
        if total < 1e-15:
            return float('inf')
        return 2 * n * max_val / (total * total)
    
    # Golden section search on [0, v_max]

    a_g = 0.0

    f_c = obj_at(c_g)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP Nesterov (zero-sum game) with adaptive restart and tail averaging
2. Top-k LogSumExp gradient descent with Adam
3. Coordinate descent with golden section search (new, key improvement)
4. Multi-start diversity and multi-length exploration
5. Strategic peak shaving for final polish

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def compute_topk_lse(a, T, topk):
    """
    Compute top-k LSE objective value and gradient w.r.t. a.
    obj approximates max_k conv(a,a)[k] using top-k log-sum-exp.
    """
    n = len(a)
    c = fast_autocorr(a)
    k_use = min(topk, len(c))
    top_idx = np.argsort(c)[::-1][:k_use]
    top_vals = c[top_idx]

    Tc = T * top_vals
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    lse_val = (max_Tc + np.log(exp_Tc.sum())) / T
    w = exp_Tc / exp_Tc.sum()

    grad_a = np.zeros(n)
    for idx in range(k_use):
        K = top_idx[idx]
        wK = w[idx]
        js = max(0, K - (n - 1))
        je = min(n - 1, K)
        if js <= je:
            idx_range = np.arange(js, je + 1)
            grad_a[idx_range] += 2.0 * wK * a[K - idx_range]

    return lse_val, grad_a


def topk_lse_grad_z(z, T, topk):
    """Objective and gradient in z-space for top-k LSE."""
    a = softmax_fn(z)
    n = len(a)
    obj, grad_a = compute_topk_lse(a, T, topk)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.MOSEK, cp.GLPK]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.MOSEK, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, max_iters=500000):
    """
    Nesterov-accelerated LP best-response with adaptive restart and tail averaging.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    # Tail averaging buffers
    tail_buf = [None] * 500
    tail_idx = 0
    tail_count = 0
    total_count = 0
    cum_sum = np.zeros(n)

    stall_count = 0
    restart_threshold = 600

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit - 2:
            break

        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_sum = y_new.sum()
        if y_sum < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / y_sum

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new

        y = y_new
        a = a_new

        # Update averages
        cum_sum += a
        total_count += 1

        if tail_buf[tail_idx] is not None:
            tail_count -= 1
        tail_buf[tail_idx] = a.copy()
        tail_count += 1
        tail_idx = (tail_idx + 1) % 500

        # Check candidates periodically
        if t % 20 == 0:
            score_a = 2.0 * n * fast_autocorr(a).max()
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()
                stall_count = 0

            score_y = 2.0 * n * fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()
                stall_count = 0

            # Tail average (last 200 iterates)
            if tail_count >= 200:
                tail_samples = [tail_buf[(tail_idx + i) % 500] for i in range(200)]
                tail_avg = np.mean(tail_samples, axis=0)
                score_tail = 2.0 * n * fast_autocorr(tail_avg).max()
                if score_tail < best_score:
                    best_score = score_tail
                    best_a = tail_avg.copy()
                    stall_count = 0

            # Full running average
            full_avg = cum_sum / total_count
            score_full = 2.0 * n * fast_autocorr(full_avg).max()
            if score_full < best_score:
                best_score = score_full
                best_a = full_avg.copy()
                stall_count = 0

            stall_count += 1

        # Adaptive restart
        if stall_count > restart_threshold and t > 100:
            a = best_a.copy()
            y = best_a.copy()
            cum_sum = np.zeros(n)
            total_count = 0
            tail_buf = [None] * 500
            tail_idx = 0
            tail_count = 0
            stall_count = 0
            restart_threshold = max(300, restart_threshold // 2)
            if t % 5000 == 0:
                print(f"      NLP restart at t={t}, best={best_score:.6f}")

        if t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}")

    return best_a, best_score


def adam_topk_lse(a0, time_limit, T_start=100.0, T_end=0.3, n_stages=25, topk=12):
    """Adam-optimized top-k LSE with multi-stage annealing."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit - 5:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        base_lr = 5e-3 * min(1.0, 80.0 / n)
        lr = base_lr * (0.7 ** si)

        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.8, 0.999, 1e-8

        max_it = max(30, int(min(remaining, 120.0) * 3))

        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 5:
                break

            obj, grad_z = topk_lse_grad_z(z, stage_T, topk)

            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            if (it + 1) % 8 == 0:
                a = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    if (it + 1) % 40 == 0:
                        z = np.log(np.maximum(best_a, 1e-30))

        a_stage = softmax_fn(z)
        score_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()

        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE stage {si}/{n_stages} (T={stage_T:.1f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def coord_desc_evaluate(vprime, v, s, n, const_max, lin_intercepts, lin_slopes, has_quad, quad_const):
    """Evaluate objective at a candidate value vprime for one coordinate."""
    d = vprime - v
    total = s + vprime
    if total < 1e-15:
        return float('inf')
    max_val = const_max
    if len(lin_slopes) > 0:
        lin_vals = lin_intercepts + lin_slopes * d
        max_lin = np.max(lin_vals)
        if max_lin > max_val:
            max_val = max_lin
    if has_quad:
        quad_val = quad_const + vprime * vprime
        if quad_val > max_val:
            max_val = quad_val
    return 2.0 * n * max_val / (total * total)


def coord_desc_pass(a, max_passes=20, time_limit=None, seed=42):
    """
    Coordinate descent with golden section search.
    For each coordinate i, optimize its value exactly (1D search).
    """
    n = len(a)
    rng = np.random.default_rng(seed)
    a = np.array(a, dtype=np.float64)
    conv = fast_autocorr(a).copy()
    total_sum = a.sum()
    best_score = 2.0 * n * conv.max() / (total_sum * total_sum)
    best_a = a.copy()
    t0 = time.monotonic()

    for p in range(max_passes):
        if time_limit is not None and time.monotonic() - t0 > time_limit - 3:
            break

        # Precompute prefix and suffix max of conv for O(1) const_max lookup
        conv_len = 2 * n - 1
        prefix_max = np.zeros(conv_len)
        prefix_max[0] = conv[0]
        for k in range(1, conv_len):
            prefix_max[k] = max(prefix_max[k - 1], conv[k])
        suffix_max = np.zeros(conv_len)
        suffix_max[conv_len - 1] = conv[conv_len - 1]
        for k in range(conv_len - 2, -1, -1):
            suffix_max[k] = max(suffix_max[k + 1], conv[k])

        order = rng.permutation(n)
        improved = 0

        for i in order:
            if time_limit is not None and time.monotonic() - t0 > time_limit - 3:
                break

            v = a[i]
            s = total_sum - v

            # Constant max: max of conv[k] for k not in [i, i+n-1]
            const_max = 0.0
            if i > 0:
                const_max = max(const_max, prefix_max[i - 1])
            i_end = i + n - 1
            if i_end < conv_len - 1:
                const_max = max(const_max, suffix_max[i_end + 1])

            # Linear terms: j in [0,n), j != i
            # term_j = conv[i+j] + 2*(v'-v)*a[j]
            js = np.array([j for j in range(n) if j != i])
            lin_intercepts = conv[i + js]
            lin_slopes = 2.0 * a[js]

            # Quadratic term: k = 2i
            has_quad = (2 * i < conv_len)
            quad_const = (conv[2 * i] - v * v) if has_quad else 0.0

            # Evaluate current
            f_curr = coord_desc_evaluate(v, v, s, n, const_max, lin_intercepts, lin_slopes, has_quad, quad_const)

            # Golden section search for optimal v' in [0, v_high]
            v_high = min(1.0, max(3.0 * v, 0.02))

            invphi = 2.0 / (1.0 + 5.0**0.5)
            a_g = 0.0
            b_g = v_high
            c_g = b_g - invphi * (b_g - a_g)
            d_g = a_g + invphi * (b_g - a_g)
            f_c = coord_desc_evaluate(c_g, v, s, n, const_max, lin_intercepts, lin_slopes, has_quad, quad_const)
            f_d = coord_desc_evaluate(d_g, v, s, n, const_max, lin_intercepts, lin_slopes, has_quad, quad_const)

            for _ in range(25):
                if f_c < f_d:
                    b_g = d_g
                    d_g = c_g
                    f_d = f_c
                    c_g = b_g - invphi * (b_g - a_g)
                    f_c = coord_desc_evaluate(c_g, v, s, n, const_max, lin_intercepts, lin_slopes, has_quad, quad_const)
                else:
                    a_g = c_g
                    c_g = d_g
                    f_c = f_d
                    d_g = a_g + invphi * (b_g - a_g)
                    f_d = coord_desc_evaluate(d_g, v, s, n, const_max, lin_intercepts, lin_slopes, has_quad, quad_const)

            v_opt = (a_g + b_g) / 2.0
            f_opt = coord_desc_evaluate(v_opt, v, s, n, const_max, lin_intercepts, lin_slopes, has_quad, quad_const)

            # Also check v=0
            f_zero = coord_desc_evaluate(0.0, v, s, n, const_max, lin_intercepts, lin_slopes, has_quad, quad_const)
            if f_zero < f_opt:
                v_opt = 0.0
                f_opt = f_zero

            if f_opt < f_curr - 1e-12 and abs(v_opt - v) > 1e-8:
                # Update
                dv = v_opt - v
                a[i] = v_opt
                total_sum = s + v_opt

                # Update conv
                for j in range(n):
                    k = i + j
                    if j == i:
                        conv[k] += v_opt * v_opt - v * v
                    else:
                        conv[k] += 2.0 * dv * a[j]

                current_score = 2.0 * n * conv.max() / (total_sum * total_sum)
                if current_score < best_score:
                    best_score = current_score
                    best_a = a.copy()
                    improved += 1

        if improved == 0:
            break
        if (p + 1) % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      CoordDesc pass {p+1} ({elapsed:.0f}s): best={best_score:.6f}, improved={improved}")

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=123):
    """Multi-scale peak shaving with adaptive step sizes."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 3:
        b = fast_autocorr(a)
        bmax = b.max()
        threshold = bmax * 0.995
        top_peaks = np.where(b >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(b)[::-1][:10]

        improved = False
        for k in top_peaks[:20]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            max_contrib = contrib.max()
            if max_contrib < 1e-15:
                continue

            src = idx[np.argmax(contrib)]

            danger = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:20]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                danger[js2:je2 + 1] = True

            safe = np.where(~danger)[0]
            safe = safe[safe != src]
            if len(safe) == 0:
                safe = np.where(np.arange(n) != src)[0]

            safe_sorted = safe[np.argsort(a[safe])]
            dst_pool = safe_sorted[:max(3, len(safe_sorted) // 5)]
            if len(dst_pool) == 0:
                continue
            dst = dst_pool[rng.integers(len(dst_pool))]

            for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.05, 0.08]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            n_pert = rng.integers(1, 4)
            a_trial = a.copy()
            for _ in range(n_pert):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a_trial[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.02)
                    amt = a_trial[i] * frac
                    a_trial[i] -= amt
                    a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 1e-10:
                a_trial = a_trial / st
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0

            if stall > 200:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 3
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 2
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.7 + 0.2 * np.cos(2 * x) + 0.1 * np.cos(5 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * (x - 0.3) ** 2) + 0.5 * np.exp(-5 * (x - 0.7) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * np.cos(np.pi * x / 3) ** 2
    else:
        x = np.linspace(-1, 1, n)
        a = 0.5 * (1 + np.cos(np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def perturbed_init(a0, rng, scale=0.1):
    """Create a perturbed version of a sequence for multi-start."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a0)) / len(a0)
    a = a / s
    noise = rng.normal(0, scale, len(a))
    a_new = a + noise * (a + 1e-6)
    a_new = np.maximum(a_new, 0)
    sn = a_new.sum()
    if sn < 1e-15:
        return a.copy()
    return a_new / sn


def propose_candidate(seed=42, time_budget=1800.0):
    """
    Multi-phase optimization combining LP Nesterov, LSE Adam,
    coordinate descent, and peak shaving across multiple lengths.
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(time_budget) - 120.0, 1680.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    # Get parent sequence
    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    n_base = len(best_seq) if best_seq is not None else 1600

    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # === Phase 1: LP Nesterov at base length ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} ({remaining:.0f}s left) ===")
    t1 = min(remaining * 0.30, 500.0)
    if t1 > 40:
        a_out, _ = nesterov_lp_optimize(best_seq, t1, solver)
        try_update(a_out, f"lp_n{n_base}")

    # === Phase 2: Coordinate descent at base length (NEW - key improvement) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 2: CoordDescent n={n_base} ({remaining:.0f}s left) ===")
    t2 = min(remaining * 0.18, 250.0)
    if t2 > 20:
        a_out, _ = coord_desc_pass(best_seq, max_passes=30, time_limit=t2, seed=42)
        try_update(a_out, f"coord_n{n_base}")

    # === Phase 3: LSE Adam at base length ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 3: LSE Adam n={n_base} ({remaining:.0f}s left) ===")
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 15:
        a_out, _ = adam_topk_lse(best_seq, t3, T_start=60.0, T_end=0.2, n_stages=20, topk=10)
        try_update(a_out, f"lse_n{n_base}")

    # === Phase 4: Coordinate descent after LSE ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 4: CoordDescent polish n={n_base} ({remaining:.0f}s left) ===")
    t4 = min(remaining * 0.12, 180.0)
    if t4 > 15:
        a_out, _ = coord_desc_pass(best_seq, max_passes=30, time_limit=t4, seed=789)
        try_update(a_out, f"coord2_n{n_base}")

    # === Phase 5: LP Nesterov at longer length ===
    n_long = n_base + 200
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 5: LP Nesterov n={n_long} ({remaining:.0f}s left) ===")
    t5 = min(remaining * 0.15, 250.0)
    if t5 > 40:
        init_long = interpolate_sequence(best_seq, n_long)
        a_out, _ = nesterov_lp_optimize(init_long, t5, solver)
        try_update(a_out, f"lp_n{n_long}")

    # === Phase 6: Coordinate descent at longer length ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 30:
        print(f"=== Phase 6: CoordDescent n={n_long} ({remaining:.0f}s left) ===")
        t6 = min(remaining * 0.12, 150.0)
        if t6 > 15:
            a_out, _ = coord_desc_pass(best_seq, max_passes=20, time_limit=t6, seed=321)
            try_update(a_out, f"coord_n{n_long}")

    # === Phase 7: Multi-start with coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 40:
        print(f"=== Phase 7: Multi-start ({remaining:.0f}s left) ===")
        n_starts = 2
        t7_each = min(remaining * 0.10, 80.0) / n_starts
        for si in range(n_starts):
            if time.monotonic() - t_global_start > time_budget - 60:
                break
            pert = perturbed_init(best_seq, rng, scale=0.04 * (si + 1))
            a_out, _ = coord_desc_pass(pert, max_passes=15, time_limit=t7_each, seed=100 + si)
            try_update(a_out, f"multi_start_{si}")

    # === Phase 8: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 15:
        print(f"=== Phase 8: Peak shaving ({remaining:.0f}s left) ===")
        t8 = min(remaining * 0.08, 60.0)
        if t8 > 10:
            a_out, _ = peak_shaving(best_seq, t8, seed=456)
            try_update(a_out, "peak_shave")

    # === Phase 9: Final coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 10:
        print(f"=== Phase 9: Final CoordDescent ({remaining:.0f}s left) ===")
        t9 = min(remaining * 0.30, 80.0)
        if t9 > 5:
            a_out, _ = coord_desc_pass(best_seq, max_passes=50, time_limit=t9, seed=999)
            try_update(a_out, "final_coord")

    # === Phase 10: Very fine LSE polish ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 10:
        t10 = min(remaining * 0.30, 40.0)
        if t10 > 5:
            print(f"=== Phase 10: Fine LSE polish ({remaining:.0f}s left) ===")
            a_out, _ = adam_topk_lse(best_seq, t10, T_start=3.0, T_end=0.1, n_stages=8, topk=6)
            try_update(a_out, "fine_lse")

    # === Phase 11: Final coord descent ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 5:
        t11 = min(remaining * 0.5, 30.0)
        if t11 > 3:
            a_out, _ = coord_desc_pass(best_seq, max_passes=30, time_limit=t11, seed=2024)
            try_update(a_out, "ultimate_coord")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
