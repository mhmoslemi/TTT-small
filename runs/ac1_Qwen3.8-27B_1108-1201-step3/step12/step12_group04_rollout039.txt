    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)

def build_conv_matrix(a_fixed):

    T = np.zeros((size, n))

def spectral_init(n, rng, n_iters=100):

    Tries to make |FFT(a)|^2 flat while keeping a >= 0.

def compute_lse_grad(a, T):
    """Compute gradient of (1/T)*log(sum exp(T*(a*a)_k)) w.r.t. a."""
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    
    a_rev = a[::-1].copy()
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    if conv_len > fft_size:
        fft_size *= 2
    
    W = rfft(w, fft_size)
    A_rev = rfft(a_rev, fft_size)
    conv = irfft(W * A_rev, fft_size)
    
    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad, w

def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def lse_obj_grad_z(z, T):
    """Objective and gradient for z-parameterization (a = softmax(z))."""
    a = softmax_fn(z)
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    obj_val = (max_Tb + np.log(exp_Tb.sum())) / T
    
    grad_a, w = compute_lse_grad(a, T)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj_val, grad_z

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def subgradient_descent(a0, time_limit, n_iters=5000):
    """
    Projected subgradient descent on the max-convolution objective.
    Uses the fact that the subgradient of max_b at a is 2*a_rev[k*-i] where k* is argmax.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    a_rev = a[::-1]
    step_sizes = np.geomspace(0.1, 0.001, n_iters)
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(a)
        # Find all positions where b is close to max (active set)
        b_max = b.max()
        threshold = b_max * 0.99
        active = np.where(b >= threshold)[0]
        
        # Compute subgradient: average over active peaks
        grad = np.zeros(n)
        a_rev = a[::-1]
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for i in range(js, je + 1):
                grad[i] += a_rev[k - i]
        
        grad /= max(1, len(active))
        grad *= 2.0
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        # Step size with averaging
        step = step_sizes[min(it, n_iters - 1)]
        
        # Update: a_new = a - step * grad, projected onto {a >= 0, sum(a) = 1}
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / ss
        
        # Also try the negative step (since we're minimizing, and the subgradient points "up")
        # Actually, the subgradient of max_b w.r.t. a points in the direction of increase.
        # So we should go AGAINST the gradient: a - step * grad.
        # But we also need to ensure sum stays the same, which the normalization handles.
        
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        
        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SGD it={it} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def fast_autocorr(a):
    """Full self-convolution a*a using FFT, O(n log n)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where (T@b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def get_lp_solver():

def solve_lp_best_response(a_fixed, solver):

def spectral_init(n, rng, n_iters=100):
    """
    Gerchberg-Saxton style spectral initialization.
    Alternates between spectral flattening and non-negativity projection.
    """
    a = np.ones(n) / n
    fft_size = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        A = rfft(a, fft_size)
        power = np.abs(A) ** 2

        non_dc = power[1:]
        if len(non_dc) > 0:
            target = np.mean(non_dc)
            if target > 1e-30:
                alpha = 0.3 + 0.2 * (it / max(1, n_iters - 1))
                scale = (target / np.maximum(non_dc, 1e-30)) ** alpha
                A_new = A.copy()
                A_new[1:] = A[1:] * scale
            else:
                A_new = A
        else:
            A_new = A

        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        a = a_new

    return a

def compute_lse_grad(a, T):
    """Compute gradient of (1/T)*log(sum exp(T*(a*a)_k)) w.r.t. a using FFT."""
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()

    a_rev = a[::-1].copy()
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    if conv_len > fft_size:
        fft_size *= 2

    W = rfft(w, fft_size)
    A_rev = rfft(a_rev, fft_size)
    conv = irfft(W * A_rev, fft_size)

    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad, w


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def lse_obj_grad_z(z, T):
    """Objective and gradient for z-parameterization (a = softmax(z))."""
    a = softmax_fn(z)
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    obj_val = (max_Tb + np.log(exp_Tb.sum())) / T

    grad_a, w = compute_lse_grad(a, T)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj_val, grad_z


def _lse_func_wrapper(z, T):
    val, _ = lse_obj_grad_z(z, T)
    return val


def _lse_jac_wrapper(z, T):
    _, g = lse_obj_grad_z(z, T)
    return g

def compute_subgrad(a, active_peaks):
    """Compute averaged subgradient of max-conv at given active peaks."""
    n = len(a)
    grad = np.zeros(n)
    for k in active_peaks:
        for m in range(n):
            idx = k - m
            if 0 <= idx < n:
                grad[m] += 2.0 * a[idx]
    grad /= max(1, len(active_peaks))
    return grad


def subgradient_descent(a0, time_limit, n_iters=5000):
