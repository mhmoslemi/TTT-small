import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Fast linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score assuming sum(a)=1: returns 2*n*max(b)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_from_z(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Inverse of softmax (log)."""
    return np.log(np.maximum(a, 1e-300))


def subgradient_conv(a, k):
    """Compute subgradient of conv(a,a)[k] w.r.t. a.
    grad[i] = 2*a[k-i] if 0<=k-i<n, else 0."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1)
    g[idx] = 2.0 * a[k - idx]
    return g


def active_set_subgradient(a, top_k=20):
    """Compute weighted subgradient using top peaks of convolution."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    if max_b < 1e-30:
        return np.zeros(n), b
    threshold = max_b * 0.95
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    if len(active) > top_k:
        top_idx = np.argsort(b[active])[-top_k:]
        active = active[top_idx]
    weights = b[active] / b[active].sum()
    g = np.zeros(n)
    for i, k in enumerate(active):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        g[idx] += weights[i] * 2.0 * a[k - idx]
    return g, b


def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix_reduced(a_fixed, top_k=200):
    """Build reduced convolution matrix using only top-k rows."""
    n = len(a_fixed)
    size = 2 * n - 1
    b = fast_convolve(a_fixed)
    # Get top_k indices
    if top_k < size:
        top_idx = np.argsort(b)[::-1][:top_k]
    else:
        top_idx = np.arange(size)
    T = np.zeros((len(top_idx), n))
    for row_idx, k in enumerate(top_idx):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        T[row_idx, lo:hi + 1] = a_fixed[k - (lo:hi + 1)]
    return T, top_idx


_lp_solver = None


def get_lp_solver():
    """Find the best available LP solver."""
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_reduced(a_fixed, solver=None, top_k=200):
    """Solve reduced LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T, _ = build_conv_matrix_reduced(a_fixed, top_k)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def mirror_descent(a0, time_limit, solver=None, lp_every=2000, max_iters=1000000):
    """Mirror descent with periodic LP correction."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0

    # Adaptive step size
    eta = 0.1
    eta_min = 1e-5
    momentum = 0.5
    y_z = z.copy()  # momentum buffer (in z-space)

    # Track for adaptive step
    prev_score = best_score
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute subgradient
        g, b = active_set_subgradient(a, top_k=30)

        # Project gradient to simplex tangent (for softmax parametrization)
        S = np.dot(g, a)
        grad_z = a * (g - S)

        grad_norm = np.linalg.norm(grad_z)
        if grad_norm < 1e-18:
            break

        # Momentum in z-space
        y_z = momentum * y_z + (1 - momentum) * (z - eta * grad_z)
        z = y_z

        a = softmax_from_z(z)
        cur_score = score_normalized(a)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive step size
        if no_improve > 100:
            eta = max(eta_min, eta * 0.5)
            no_improve = 0
        elif no_improve > 30:
            eta = max(eta_min, eta * 0.8)

        # Periodic LP correction
        if solver is not None and (it + 1) % lp_every == 0:
            b_lp, M_lp = solve_lp_reduced(a, solver, top_k=200)
            if b_lp is not None:
                # Blend with LP solution
                blend = min(0.3 + 0.1 * (it // lp_every), 0.9)
                a_blend = (1 - blend) * a + blend * b_lp
                a_blend = np.maximum(a_blend, 0)
                sb = a_blend.sum()
                if sb > 1e-15:
                    a_blend = a_blend / sb
                    sc_blend = score_normalized(a_blend)
                    if sc_blend < best_score:
                        best_score = sc_blend
                        best_a = a_blend.copy()
                    z = z_from_a(a_blend)
                    a = a_blend
                    y_z = z.copy()

        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 10:
                print(f"    MD iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} eta={eta:.6f}")
                last_print = elapsed

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_bfgs(a0, time_limit, T_start=80.0, T_min=0.1, n_stages=30, iters_per_stage=40):
    """Adaptive LogSumExp L-BFGS."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    T = T_start
    stagnant = 0

    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, T, n)
            return grad

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 300,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)

            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()

            if stagnant >= 3:
                T = max(T, T * 1.1)
                stagnant = 0
            else:
                T = max(T_min, T * 0.7)
                stagnant = max(0, stagnant - 1)

            if sc - best_score > best_score * 0.003:
                stagnant += 1

            if (stage + 1) % 5 == 0:
                print(f"    LSE stage={stage+1}/{n_stages} T={T:.2f}: score={sc:.8f} best={best_score:.8f}")

        except Exception:
            T = max(T_min, T * 0.8)

    return best_a, best_score


def nesterov_lp_fast(a0, time_limit, solver=None, max_iters=100000, top_k=200):
    """Nesterov-accelerated LP with reduced constraints."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_lp = solve_lp_reduced(y, solver, top_k=top_k)
        if b_new is None:
            continue

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 15:
                print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} M={M_lp:.6f}")
                last_print = elapsed

    return best_a, best_score


def structured_init(n, rng):
    """Generate structured initial sequences."""
    choice = rng.integers(0, 10)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * (x - 0.3) ** 2) + 0.5 * np.exp(-5 * (x - 0.7) ** 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * x) + 0.3
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2 * x) + 0.15 * np.cos(5 * x) + 0.1 * np.cos(9 * x)
        a = np.maximum(a, 0)
    elif choice == 5:
        # Geometric-like
        r = 0.999
        a = r ** np.arange(n)
    elif choice == 6:
        # Two bumps
        x = np.linspace(0, 1, n)
        a = np.exp(-20 * (x - 0.2) ** 2) + np.exp(-20 * (x - 0.8) ** 2) + 0.1
    elif choice == 7:
        # Smooth step
        x = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + np.exp(-10 * (x - 0.5)))
    elif choice == 8:
        # Periodic
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 7 * x) + 0.1 * np.sin(2 * np.pi * 13 * x)
        a = np.maximum(a, 0)
    else:
        x = np.linspace(0, 1, n)
        a = np.cosh(3 * (x - 0.5)) ** (-1)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def peak_shaving(a0, time_limit, n_steps=500000):
    """Aggressive peak-shaving: directly reduce the maximum convolution coefficient."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = 0.05

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        max_k = int(np.argmax(b))

        # Subgradient at the peak
        g = subgradient_conv(a, max_k)

        # Softmax gradient
        S = np.dot(g, a)
        grad_z = a * (g - S)

        grad_norm = np.linalg.norm(grad_z)
        if grad_norm < 1e-18:
            break

        # Adaptive step with bisection
        z_new = z - eta * grad_z
        a_new = softmax_from_z(z_new)
        sc_new = score_normalized(a_new)

        if sc_new > best_score:
            # Bisect
            lo, hi = 0.0, eta
            for _ in range(8):
                mid = (lo + hi) / 2.0
                z_mid = z - mid * grad_z
                a_mid = softmax_from_z(z_mid)
                sc_mid = score_normalized(a_mid)
                if sc_mid <= best_score:
                    lo = mid
                else:
                    hi = mid
            eta = lo * 0.9
            z_new = z - eta * grad_z
            a_new = softmax_from_z(z_new)
            sc_new = score_normalized(a_new)
        else:
            eta = min(eta * 1.05, 0.5)

        z = z_new
        a = a_new

        if sc_new < best_score:
            best_score = sc_new
            best_a = a.copy()

        if (step + 1) % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave step={step+1} ({elapsed:.0f}s): score={best_score:.8f} eta={eta:.6f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def hybrid_optimize(a0, time_limit, solver=None):
    """Combine mirror descent with periodic LP, then polish with LSE."""
    n = len(a0)
    t0 = time.monotonic()

    # Stage 1: Mirror descent (60% of time)
    md_time = 0.6 * time_limit
    a_md, sc_md = mirror_descent(a0, md_time, solver=solver, lp_every=3000)

    # Stage 2: Nesterov LP from MD result (20% of time)
    el = time.monotonic() - t0
    remaining = time_limit - el
    nl_time = 0.2 * time_limit
    a_nl, sc_nl = nesterov_lp_fast(a_md, nl_time, solver=solver, max_iters=50000)

    # Stage 3: LSE polishing (20% of time)
    el = time.monotonic() - t0
    remaining = time_limit - el
    lse_time = min(0.2 * time_limit, remaining * 0.7)
    if lse_time > 10:
        a_lse, sc_lse = lse_bfgs(a_nl, lse_time, T_start=50.0, T_min=0.1, n_stages=20, iters_per_stage=30)
    else:
        a_lse = a_nl
        sc_lse = sc_nl

    return a_lse


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function with adaptive multi-method optimization."""
    rng = np.random.default_rng(seed)
    global_deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Total budget: {budget_s}s, effective: {global_deadline - time.monotonic():.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to use parent sequence
    n_base = 1200
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            remain = global_deadline - time.monotonic()
            print(f"  ** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, remaining={remain:.0f}s)")

    # === Phase 1: Multi-start exploration (15% time) ===
    print(f"\n=== Phase 1: Multi-start (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    p1_budget = min(0.15 * (budget_s - 30), remaining * 0.3)
    if p1_budget > 30:
        # Try different lengths and initializations
        lengths = [800, 1000, n_base, 1500, 2000]
        per_start = p1_budget / len(lengths)
        for n_try in lengths:
            remaining = global_deadline - time.monotonic()
            if remaining < 15:
                break
            t_alloc = min(per_start, remaining * 0.45)
            if t_alloc < 10:
                continue
            print(f"  Multi-start n={n_try}, budget={t_alloc:.0f}s")
            # Use interpolation from best if available, else structured init
            if best_seq is not None and n_try != n_base:
                init_try = interpolate_sequence(best_seq, n_try)
            else:
                init_try = structured_init(n_try, rng)
            # Quick hybrid
            a_try, _ = mirror_descent(init_try, t_alloc * 0.7, solver=solver, lp_every=5000)
            try_update(a_try, f"ms_n{n_try}_md")
            # Quick LSE
            remaining = global_deadline - time.monotonic()
            if remaining > 10:
                a_try2, _ = lse_bfgs(a_try, remaining * 0.2, T_start=40.0, T_min=0.2, n_stages=8, iters_per_stage=20)
                try_update(a_try2, f"ms_n{n_try}_lse")
    print(f"  Phase 1 done: best_score={best_score:.8f}")

    # === Phase 2: Main hybrid optimization at best length (40% time) ===
    print(f"\n=== Phase 2: Main hybrid n={len(best_seq)} (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    p2_budget = min(0.40 * (budget_s - 30), remaining * 0.5)
    if p2_budget > 60:
        a_main, _ = hybrid_optimize(best_seq, p2_budget, solver=solver)
        try_update(a_main, "main_hybrid")

        # Run mirror descent again from the result with different parameters
        remaining = global_deadline - time.monotonic()
        if remaining > 30:
            a_md2, _ = mirror_descent(a_main, remaining * 0.4, solver=solver, lp_every=5000)
            try_update(a_md2, "main_md2")
    print(f"  Phase 2 done: best_score={best_score:.8f}")

    # === Phase 3: Nesterov LP focused (20% time) ===
    print(f"\n=== Phase 3: Nesterov LP (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    p3_budget = min(0.20 * (budget_s - 30), remaining * 0.5)
    if p3_budget > 60:
        a_nl, _ = nesterov_lp_fast(best_seq, p3_budget, solver=solver, max_iters=50000, top_k=300)
        try_update(a_nl, "nesterov_lp")
        # Also try with different top_k
        remaining = global_deadline - time.monotonic()
        if remaining > 30:
            a_nl2, _ = nesterov_lp_fast(best_seq, remaining * 0.4, solver=solver, max_iters=20000, top_k=100)
            try_update(a_nl2, "nesterov_lp_100")
    print(f"  Phase 3 done: best_score={best_score:.8f}")

    # === Phase 4: LSE polishing (15% time) ===
    print(f"\n=== Phase 4: LSE Polish (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    p4_budget = min(0.15 * (budget_s - 30), remaining * 0.5)
    if p4_budget > 30:
        # Multiple LSE passes with different temperatures
        a_l1, _ = lse_bfgs(best_seq, p4_budget * 0.4, T_start=100.0, T_min=0.1, n_stages=20, iters_per_stage=35)
        try_update(a_l1, "lse_high_T")

        remaining = global_deadline - time.monotonic()
        if remaining > 20:
            a_l2, _ = lse_bfgs(best_seq, remaining * 0.4, T_start=20.0, T_min=0.05, n_stages=15, iters_per_stage=30)
            try_update(a_l2, "lse_low_T")

        # LSE from perturbed point
        remaining = global_deadline - time.monotonic()
        if remaining > 15:
            pert = best_seq + 0.03 * rng.standard_normal(len(best_seq))
            pert = np.maximum(pert, 0)
            pert = pert / pert.sum()
            a_l3, _ = lse_bfgs(pert, remaining * 0.4, T_start=60.0, T_min=0.1, n_stages=12, iters_per_stage=25)
            try_update(a_l3, "lse_perturbed")
    print(f"  Phase 4 done: best_score={best_score:.8f}")

    # === Phase 5: Peak shaving (10% time) ===
    print(f"\n=== Phase 5: Peak Shaving (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    p5_budget = min(0.10 * (budget_s - 30), remaining * 0.5)
    if p5_budget > 20:
        a_ps, _ = peak_shaving(best_seq, p5_budget, n_steps=2000000)
        try_update(a_ps, "peak_shave")
        # Chain: peak shave then LSE
        remaining = global_deadline - time.monotonic()
        if remaining > 15:
            a_ps2, _ = lse_bfgs(a_ps, remaining * 0.4, T_start=15.0, T_min=0.05, n_stages=10, iters_per_stage=25)
            try_update(a_ps2, "peak_shave_lse")
    print(f"  Phase 5 done: best_score={best_score:.8f}")

    # === Phase 6: Final combined polish (remaining time) ===
    print(f"\n=== Phase 6: Final Polish (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    if remaining > 20:
        # Short combined pass
        a_f1, _ = hybrid_optimize(best_seq, remaining * 0.4, solver=solver)
        try_update(a_f1, "final_hybrid")

        remaining = global_deadline - time.monotonic()
        if remaining > 15:
            a_f2, _ = peak_shaving(best_seq, remaining * 0.4, n_steps=500000)
            try_update(a_f2, "final_peak")

        remaining = global_deadline - time.monotonic()
        if remaining > 10:
            a_f3, _ = lse_bfgs(best_seq, remaining * 0.5, T_start=10.0, T_min=0.05, n_stages=10, iters_per_stage=20)
            try_update(a_f3, "final_lse")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"Time used: {time.monotonic() - (global_deadline - (budget_s - 30)):.0f}s")
    return [float(x) for x in best_seq]
