```python
def solve_lp_best_response_active(a_fixed, solver, max_constraints=200):
    """
    Solve LP with constraint generation.
    Start with top constraints, add violated ones iteratively.
    """
    n = len(a_fixed)
    size = 2 * n - 1
    
    # Start with top 50 lags by current self-convolution
    b_self = fast_autocorr(a_fixed)
    active = np.argsort(b_self)[::-1][:max_constraints]
    
    for iteration in range(20):  # Max 20 rounds of constraint addition
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        T_sub = build_conv_matrix_rows(a_fixed, active)
        constraints = [T_sub @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        prob.solve(solver=solver)
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        
        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        b_val = b_val / b_val.sum()
        M_val = float(M_var.value)
        
        # Check all constraints
        T_full_b = build_conv_matrix(a_fixed) @ b_val
        violated = np.where(T_full_b > M_val + 1e-10)[0]
        
        if len(violated) == 0:
            break
        
        # Add violated constraints (up to some limit)
        active = np.unique(np.concatenate([active, violated[:50]]))
        active = np.argsort(T_full_b[active])[::-1][:max_constraints * 2]
    
    return b_val, M_val
```
