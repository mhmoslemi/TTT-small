"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated LSE gradient with FISTA (fast initial convergence)
2. LP Nesterov zero-sum game (global optimization)

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp


def fast_autocorr(a):
    """Full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def lse_objective_and_grad(a, T, topk=None):
    """
    Compute LSE objective and gradient for conv(a,a).
    Objective: (1/T) log(sum_k exp(T*b[k]))
    Gradient: 2 * cross_correlate(w, a) where w = softmax(T*b)
    """
    n = len(a)
    b = fast_autocorr(a)
    
    if topk is not None and topk < len(b):
        idx = np.argpartition(b, -topk)[-topk:]
        vals = b[idx]
        Tvals = T * vals
        Tvals -= Tvals.max()
        exps = np.exp(Tvals)
        s = exps.sum()
        if s < 1e-300:
            w_full = np.ones(len(b)) / len(b)
        else:
            w_full = np.zeros(len(b))
            w_full[idx] = exps / s
        lse_val = (Tvals.max() + np.log(s)) / T
    else:
        Tvals = T * b
        Tvals -= Tvals.max()
        exps = np.exp(Tvals)
        s = exps.sum()
        if s < 1e-300:
            w_full = np.ones(len(b)) / len(b)
            lse_val = b.max()
        else:
            w_full = exps / s
            lse_val = (Tvals.max() + np.log(s)) / T
    
    # Gradient: grad[m] = 2 * sum_k w[k] * a[k-m] (a[j]=0 for j not in [0,n-1])
    # = 2 * conv(w, a_flipped)[m + n - 1]
    a_flip = a[::-1]
    size2 = len(w_full) + n - 1
    fft_size2 = 1 << (size2 - 1).bit_length()
    W = rfft(w_full, fft_size2)
    AF = rfft(a_flip, fft_size2)
    conv_result = irfft(W * AF, fft_size2)[:size2]
    grad = 2.0 * conv_result[n - 1:n - 1 + n]
    
    return lse_val, grad, b


def fista_lse(a0, time_limit, T_start=200.0, T_end=1.0, n_stages=30, topk=20):
    """
    FISTA optimization of LSE objective on the simplex.
    Uses backtracking line search for step size.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit - 10:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        # Initialize FISTA
        x = a.copy()
        y = a.copy()
        k_fista = 1
        step = 1.0 / (2.0 * n * stage_T * max(1.0, x.max() * n))
        
        max_iters = max(50, int(remaining * 5))
        
        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit - 10:
                break
            
            # Compute gradient at y
            _, grad_y, _ = lse_objective_and_grad(y, stage_T, topk)
            
            # FISTA update: x_new = project(y - step * grad_y)
            x_new = project_simplex(y - step * grad_y)
            
            # Backtracking line search (every 10 iterations)
            if it % 10 == 0:
                obj_y, _, _ = lse_objective_and_grad(y, stage_T, topk)
                obj_xn, _, _ = lse_objective_and_grad(x_new, stage_T, topk)
                # Quadratic model
                diff = x_new - y
                model = obj_y + np.dot(grad_y, diff) + 0.5 / step * np.dot(diff, diff)
                if obj_xn > model + 1e-10:
                    step *= 0.5
                    x_new = project_simplex(y - step * grad_y)
                elif it > 50 and step < 10.0 / (2.0 * n * stage_T * max(1.0, x.max() * n)):
                    step *= 1.2
            
            # Nesterov momentum
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * k_fista ** 2)) / 2.0
            y = x_new + ((k_fista - 1.0) / t_new) * (x_new - x)
            y = project_simplex(y)
            x = x_new
            k_fista += 1
            
            # Check best (every 5 iterations)
            if (it + 1) % 5 == 0:
                score = 2.0 * n * fast_autocorr(x).max()
                if score < best_score:
                    best_score = score
                    best_a = x.copy()
                    a = x.copy()
        
        # Update a for next stage
        a = x.copy()
        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score


def build_conv_matrix_vec(y):
    """Build (2n-1) x n matrix T where T @ b = convolve(y, b), using stride tricks."""
    n = len(y)
    # T[k, i] = y[k-i] if 0 <= k-i <= n-1 else 0
    # Use scipy or manual construction
    rows = 2 * n - 1
    cols = n
    T = np.zeros((rows, cols))
    for i in range(cols):
        T[i:i + n, i] = y
    return T


def solve_lp_best_response(y, solver, max_time_per_solve=2.0):
    """Solve LP: max 1'b s.t. conv(y,b) <= M, sum(b)=1, b>=0. Returns (b, M)."""
    n = len(y)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix_vec(y)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    start = time.monotonic()
    try:
        prob.solve(solver=solver, max_time=max_time_per_solve)
    except Exception:
        pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = float(M_var.value)
    return b_val, M_val

def spectral_init(n, rng, n_components=8):
    """Generate initialization using sum of squared random Fourier modes."""
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.1
    for _ in range(n_components):
        freq = rng.integers(1, n // 2 + 1)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.5, 2.0)
        a += amp * np.cos(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def peak_shaving_refined(a0, time_limit, seed=42):
    """
    Refined peak shaving: identifies top contributing pairs to each peak
    and moves mass to positions that minimize impact on all peaks.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit - 5:
        b = fast_autocorr(a)
        bmax = b.max()
        # Find peaks within 0.1% of max
        threshold = bmax * 0.999
        top_peaks = np.where(b >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(b)[::-1][:5]
        
        improved = False
        for k in top_peaks[:15]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            total_contrib = contrib.sum()
            if total_contrib < 1e-15:
                continue
            
            # Find the pair with maximum contribution
            src_pos = idx[np.argmax(contrib)]
            
            # Compute danger: for each position j, how much does it contribute to top peaks?
            danger = np.zeros(n)
            for k2 in top_peaks[:15]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                for j in range(js2, je2 + 1):
                    partner = k2 - j
                    if 0 <= partner < n:
                        danger[j] += a[j] * a[partner]
            
            # Safe destinations: low danger, not source
            safe_mask = danger < 0.1 * danger.max()
            safe_mask[src_pos] = False
            safe_positions = np.where(safe_mask)[0]
            
            if len(safe_positions) == 0:
                safe_positions = np.where(np.arange(n) != src_pos)[0]
            
            # Prefer positions with low a value (to minimize disruption)
            safe_positions = safe_positions[np.argsort(a[safe_positions])]
            dst_candidates = safe_positions[:max(5, len(safe_positions) // 4)]
            
            if len(dst_candidates) == 0:
                continue
            
            dst = dst_candidates[rng.integers(len(dst_candidates))]

            for frac in [0.002, 0.005, 0.01, 0.02, 0.04, 0.07, 0.1]:
                amt = a[src_pos] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src_pos] -= amt
                a_trial[dst] += amt
                if a_trial[src_pos] < -1e-12:
                    continue
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-10:
                    continue
                a_trial = a_trial / st
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            n_pert = rng.integers(2, 6)
            a_trial = a.copy()
            for _ in range(n_pert):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a_trial[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.03)
                    amt = a_trial[i] * frac

    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: FISTA LSE n={n_base} ({remaining:.0f}s) ===")
    t1 = min(remaining * 0.20, 300.0)
    if t1 > 30:
        a_out, _ = fista_lse(best_seq, t1, T_start=300.0, T_end=0.5, n_stages=25, topk=30)
        try_update(a_out, f"fista_n{n_base}")
    
    # === Phase 2: Multi-start diversity with LSE ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 2: Multi-start ({remaining:.0f}s) ===")
    t2_total = min(remaining * 0.12, 150.0)
    if t2_total > 30:
        n_starts = 4
        t2_each = t2_total / n_starts
        inits = [
            spectral_init(n_base, rng, n_components=6),
            spectral_init(n_base, rng, n_components=12),
            structured_init(n_base, rng),
        ]
        # Also perturb the current best
        noise = rng.normal(0, 0.03, n_base)
        pert = best_seq + noise * (best_seq + 1e-6)
        pert = np.maximum(pert, 0)
        pert = pert / pert.sum()
        inits.append(pert)
        
        for si in range(min(n_starts, len(inits))):
            if time.monotonic() - t_global_start > time_budget - 60:
                break
            a_out, _ = fista_lse(inits[si], t2_each, T_start=100.0, T_end=1.0, n_stages=12, topk=20)
            try_update(a_out, f"multi_{si}_n{n_base}")
    
    # === Phase 3: LP Nesterov (main global optimizer) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 3: LP Nesterov n={n_base} ({remaining:.0f}s) ===")
    t3 = min(remaining * 0.30, 500.0)
    if t3 > 40:
        a_out, _ = nesterov_lp_optimize(best_seq, t3, solver)
        try_update(a_out, f"lp_n{n_base}")
    
    # === Phase 4: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 4: Multi-length ({remaining:.0f}s) ===")
    t4_total = min(remaining * 0.15, 200.0)
    if t4_total > 30:
        candidate_lengths = [n_base - 2, n_base - 1, n_base + 1, n_base + 2, n_base + 4]
        n_trials = min(len(candidate_lengths), max(1, int(t4_total / 25)))
        t4_each = t4_total / max(1, n_trials)
        for ci in range(n_trials):
            if time.monotonic() - t_global_start > time_budget - 60:
                break
            nl = candidate_lengths[ci]
            if nl < 100:
                continue
            init_nl = interpolate_sequence(best_seq, nl)
            a_out, _ = fista_lse(init_nl, t4_each, T_start=50.0, T_end=0.5, n_stages=10, topk=20)
            try_update(a_out, f"len_{nl}")
    
    # === Phase 5: LP Nesterov on best length found ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 60:
        n_best = len(best_seq)
        print(f"=== Phase 5: LP Nesterov n={n_best} ({remaining:.0f}s) ===")
        t5 = min(remaining * 0.20, 250.0)
        if t5 > 30:
            a_out, _ = nesterov_lp_optimize(best_seq, t5, solver)
            try_update(a_out, f"lp2_n{n_best}")
    
    # === Phase 6: Multi-stage LSE polish ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 6: LSE polish ({remaining:.0f}s) ===")
    t6 = min(remaining * 0.15, 150.0)
    if t6 > 20:
        a_out, _ = multistage_lse_polish(best_seq, t6)
        try_update(a_out, "lse_polish")
    
    # === Phase 7: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 7: Peak shaving ({remaining:.0f}s) ===")
    t7 = min(remaining * 0.12, 100.0)
    if t7 > 10:
        a_out, _ = peak_shaving_refined(best_seq, t7, seed=789)
        try_update(a_out, "peak_shave")
    
    # === Phase 8: Final LP Nesterov ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 40:
        n_best = len(best_seq)
        print(f"=== Phase 8: Final LP n={n_best} ({remaining:.0f}s) ===")
        t8 = min(remaining * 0.35, 200.0)
        if t8 > 20:
            a_out, _ = nesterov_lp_optimize(best_seq, t8, solver)
            try_update(a_out, "lp_final")
    
    # === Phase 9: Final peak shave ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 15:
        t9 = min(remaining * 0.4, 80.0)
        if t9 > 10:
            a_out, _ = peak_shaving_refined(best_seq, t9, seed=1234)
            try_update(a_out, "peak_shave_final")
    
    # === Phase 10: Ultra-fine LSE ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 15:
        t10 = min(remaining * 0.5, 60.0)
        if t10 > 5:
            a_out, _ = fista_lse(best_seq, t10, T_start=5.0, T_end=0.1, n_stages=5, topk=5)
            try_update(a_out, "ultra_lse")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
    
    return [float(x) for x in best_seq]
