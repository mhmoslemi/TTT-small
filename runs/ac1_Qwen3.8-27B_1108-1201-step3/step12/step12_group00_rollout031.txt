2. **Key observations for improvement:**
   - The objective is `2 * n * max(conv(a,a)) / (sum(a))^2`. If we normalize `sum(a) = 1`, it's `2 * n * max(conv(a,a))`.
   - The autocorrelation `conv(a,a)` has values at lags 0, 1, ..., 2n-2.
   - The peak is typically at lag 0 (the "DC" component) which equals `sum(a_i^2)`.
   - So we want to minimize `max(sum(a_i^2), max_{k>=1} sum_i a_i * a_{i+k})`.

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_peak_grad(a, k_star):
    """Compute gradient of conv(a,a)[k_star] w.r.t. a.
    grad[i] = 2*a[k_star-i] if 0<=k_star-i<n else 0."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n, k_star + 1)
    idx = np.arange(lo, hi)
    grad[idx] = 2.0 * a[k_star - idx]
    return grad


def compute_lse_grad(a, T):
    """Compute gradient of LogSumExp(T * conv(a,a))."""
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    exp_b = np.exp(T * (b - max_b))
    w = exp_b / exp_b.sum()
    # grad[i] = 2 * sum_k w[k] * a[k-i] * I(0<=k-i<n)
    # This is a cross-correlation-like computation
    grad = np.zeros(n)
    for i in range(n):
        lo = max(0, i)
        hi = min(2 * n - 1, i + n - 1)
        if lo < hi:
            k_idx = np.arange(lo, hi)
            a_idx = k_idx - i
            grad[i] = 2.0 * np.dot(w[lo:hi], a[a_idx])
    return grad


def lse_grad_fft(a, T):
    """Compute LogSumExp gradient using FFT for speed."""
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    exp_b = np.exp(T * (b - max_b))
    w = exp_b / exp_b.sum()
    # grad[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    # This is a "sliding window dot product" = cross-correlation
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # We need: grad[i] = 2 * sum_j w[i+j]*a[j]
    # This is the correlation of w and a, specifically:
    # cross_corr[i] = sum_j w[i+j]*a[j]
    # = sum_j a[j] * w[i+j]
    # Let's compute via: ifft(fft(a) * conj(fft(w))) gives sum_j a[j]*w[k-j]
    # We want sum_j a[j]*w[i+j]. Let's use:
    # rev_w[m] = w[2n-2-m], then cross_corr[i] = sum_j a[j]*w[i+j]
    # = sum_j a[j]*rev_w[2n-2-i-j] = (a * rev_w)[2n-2-i]
    rev_w = w[::-1]
    conv = irfft(A * rfft(rev_w, fft_size), fft_size)[:3 * n - 3]
    # conv[k] = sum_j a[j]*rev_w[k-j] = sum_j a[j]*w[2n-2-(k-j)]
    # We want grad[i] = 2*sum_j a[j]*w[i+j]
    # set 2n-2-k = i+j => k = 2n-2-i-j, summing over j:
    # This is getting complex, let me just use direct computation for n<=3000
    # For n up to 2400, the direct O(n^2) is about 5.7M ops, which is fine
    grad = np.zeros(n)
    for i in range(n):
        lo = i
        hi = i + n
        if lo < 2 * n - 1:
            hi = min(hi, 2 * n - 1)
            a_idx = np.arange(hi - lo)
            grad[i] = 2.0 * np.dot(w[lo:hi], a[a_idx])
    return grad


def mwu_optimize(a0, time_limit, verbose=True, lse_anneal=False):
    """Multiplicative Weight Update with adaptive step size.
    Optionally uses LogSumExp smoothing with temperature annealing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()  # Nesterov lookahead
    t0 = time.monotonic()
    t = 0
    stall_count = 0
    stall_threshold = 50000

    if lse_anneal:
        T_start = 20.0
        T_end = 1.0
    else:
        T_start = 0.0
        T_end = 0.0

    while time.monotonic() - t0 < time_limit:
        t += 1

        # Temperature for LSE smoothing
        if lse_anneal:
            progress = min(1.0, t / max(1, int(time_limit * 100)))
            T_curr = T_start * (T_end / T_start) ** progress
        else:
            T_curr = 0.0

        # Compute peak and gradient
        b_y = fast_autocorr(y)
        k_star = int(np.argmax(b_y))
        M_y = b_y[k_star]

        if T_curr > 0.5:
            # Use LSE-smoothed gradient
            max_b = b_y.max()
            exp_b = np.exp(T_curr * (b_y - max_b))
            w = exp_b / exp_b.sum()
            # grad[i] = 2 * sum_j w[i+j] * a[j]
            grad = np.zeros(n)
            for i in range(n):
                lo = i
                hi = min(i + n, 2 * n - 1)
                if lo < hi:
                    a_idx = np.arange(hi - lo)
                    grad[i] = 2.0 * np.dot(w[lo:hi], a[a_idx])
            # Scale gradient by 1/T for the LSE objective
            grad = grad / T_curr
        else:
            # Exact peak gradient
            grad = compute_peak_grad(y, k_star)

        # Adaptive step size
        grad_norm = np.max(np.abs(grad))
        if grad_norm < 1e-15:
            eta = 0.01
        else:
            eta = 1.0 / (grad_norm * np.sqrt(max(t, 1)) * 0.5 + 1e-10)
            eta = min(eta, 0.5)

        # MWU update in log space
        log_y = np.log(y)
        log_y -= eta * grad
        # Clamp for numerical stability
        log_y = np.clip(log_y, -50, 50)
        y_new = np.exp(log_y - log_y.max())
        y_new = y_new / y_new.sum()

        # Nesterov momentum
        if t <= 2:
            a_new = y_new.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / a_new.sum()

        y = y_new
        a = a_new

        # Check improvement (every few iterations for speed)
        if t % 5 == 0 or t < 100:
            score = fast_autocorr(a).max()
            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()
                stall_count = 0
            else:
                stall_count += 5

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            cur_score = fast_autocorr(a).max()
            print(f"    MWU t={t} ({elapsed:.0f}s): best={best_score:.8f}, cur={cur_score:.8f}, eta={eta:.4f}")

        # If stalled too long, try a perturbation
        if stall_count > stall_threshold and lse_anneal:
            rng = np.random.default_rng(t)
            a = a + rng.normal(0, 0.01, n) * a
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            y = a.copy()
            stall_count = 0

    return best_a, best_score


def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_conv_matrix(a)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def lse_gradient_optimize(a0, time_limit, verbose=False):
    """LogSumExp gradient descent with adaptive temperature."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    T = 10.0
    n_stages = 6
    temps = np.geomspace(T, 0.5, n_stages)

    for si, T_curr in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        # Adaptive step size
        step_base = 1.0 / (T_curr * np.sqrt(n))
        max_steps = int(min(remaining * 50, 2000))

        for it in range(max_steps):
            if time.monotonic() - t0 > time_limit:
                break
            if it % 10 == 0:
                step = step_base / (1.0 + 0.05 * it)
            else:
                step = step_base / (1.0 + 0.05 * it)

            # Compute LSE gradient
            b = fast_autocorr(a)
            max_b = b.max()
            exp_b = np.exp(T_curr * (b - max_b))
            w = exp_b / exp_b.sum()
            grad = np.zeros(n)
            for i in range(n):
                lo = i
                hi = min(i + n, 2 * n - 1)
                if lo < hi:
                    a_idx = np.arange(hi - lo)
                    grad[i] = 2.0 * np.dot(w[lo:hi], a[a_idx])
            grad = grad / T_curr

            # Gradient step with projection
            a_new = a - step * grad
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / a_new.sum()
            a = a_new

            if it % 10 == 0:
                score = fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage={si} T={T_curr:.2f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 1e-10)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    else:
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * x) * np.sin(2 * np.pi * 7 * x)
    a = np.maximum(a, 0.01)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Define optimization pipeline for a given sequence and length
    def optimize_sequence(a_init, length, time_alloc, label_prefix):
        """Run the full optimization pipeline on a sequence of given length."""
        a = interpolate_sequence(a_init, length) if len(a_init) != length else a_init.copy()
        a = np.maximum(a, 1e-10)
        a = a / a.sum()

        t_start = time.monotonic()
        elapsed_fn = lambda: time.monotonic() - t_start

        # Step 1: MWU (fast, millions of iterations)
        t_mwu = min(time_alloc * 0.4, 300.0)
        if t_mwu > 10:
            print(f"    [{label_prefix}] MWU ({t_mwu:.0f}s) n={length}")
            a_out, _ = mwu_optimize(a, t_mwu, verbose=True, lse_anneal=False)
            do_try_update(a_out, f"{label_prefix}_mwu")
            a = best_seq if best_seq is not None and len(best_seq) == length else a_out

        # Step 2: LSE gradient (smooth convergence)
        remaining = time_alloc - elapsed_fn()
        t_lse = min(remaining * 0.2, 100.0)
        if t_lse > 10:
            print(f"    [{label_prefix}] LSE ({t_lse:.0f}s) n={length}")
            a_out, _ = lse_gradient_optimize(a, t_lse, verbose=True)
            do_try_update(a_out, f"{label_prefix}_lse")
            a = best_seq if best_seq is not None and len(best_seq) == length else a_out

        # Step 3: LP best-response (refinement)
        remaining = time_alloc - elapsed_fn()
        t_lp = min(remaining * 0.25, 200.0)
        if t_lp > 10:
            print(f"    [{label_prefix}] LP ({t_lp:.0f}s) n={length}")
            a_out, _ = lp_nesterov_optimize(a, t_lp, solver, verbose=True)
            do_try_update(a_out, f"{label_prefix}_lp")
            a = best_seq if best_seq is not None and len(best_seq) == length else a_out

        # Step 4: Pair equalization (polish)
        remaining = time_alloc - elapsed_fn()
        t_pe = min(remaining * 0.3, 150.0)
        if t_pe > 5:
            print(f"    [{label_prefix}] PE ({t_pe:.0f}s) n={length}")
            a_out, _ = pair_equalization(a, t_pe, verbose=False, seed=seed)
            do_try_update(a_out, f"{label_prefix}_pe")

        return a

    # Phase 1: Optimize at base length
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.35, 550.0)
    if t1 > 30:
        optimize_sequence(best_seq, n_base, t1, "p1")

    # Phase 2: Try n=2000
    n2 = 2000
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: n={n2} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 30:
        init2 = interpolate_sequence(best_seq, n2)
        optimize_sequence(init2, n2, t2, "p2")

    # Phase 3: Try n=2400
    n3 = 2400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: n={n3} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.2, 300.0)
    if t3 > 30:
        init3 = interpolate_sequence(best_seq, n3)
        optimize_sequence(init3, n3, t3, "p3")

    # Phase 4: Final polish on the best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Final polish n={n_best} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.4, 300.0)
    if t4 > 10:
        # MWU with LSE annealing
        t4a = t4 * 0.5
        print(f"    Final MWU-LSE ({t4a:.0f}s)")
        a_out, _ = mwu_optimize(best_seq, t4a, verbose=True, lse_anneal=True)
        do_try_update(a_out, "final_mwu_lse")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t4b = min(remaining * 0.4, 150.0)
        print(f"    Final PE ({t4b:.0f}s)")
        a_out, _ = pair_equalization(best_seq, t4b, verbose=False, seed=12345)
        do_try_update(a_out, "final_pe")

    # Phase 5: Try a different initialization (random structured)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_alt = n_base
        print(f"\n=== Phase 5: Alt init n={n_alt} (budget={remaining:.0f}s) ===")
        t5 = min(remaining * 0.5, 200.0)
        if t5 > 20:
            alt_init = structured_init(n_alt, rng)
            a_out, _ = mwu_optimize(alt_init, t5 * 0.6, verbose=False, lse_anneal=False)
            do_try_update(a_out, "alt_mwu")
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 20:
                a_out2, _ = lp_nesterov_optimize(a_out, min(remaining * 0.5, 80.0), solver, verbose=False)
                do_try_update(a_out2, "alt_lp")

    # Final check
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
