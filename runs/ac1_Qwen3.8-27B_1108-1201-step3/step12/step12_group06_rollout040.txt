"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using refined multi-stage LSE,
exact active-set subgradient with Nesterov acceleration, and multi-length exploration.
Key improvements: much lower final temperatures, tighter active sets,

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def fast_convolve(a):
    """Fast linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return ifft(A * A).real[:size]


def score_normalized(a):
    """Score assuming sum(a)=1: returns 2*n*max(b)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_from_z(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Inverse of softmax (log)."""
    return np.log(np.maximum(a, 1e-300))


def lse_obj_grad(z_in, T_val, n):
    """Compute LSE objective and gradient w.r.t. log-probabilities z."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    # grad_a[i] = 2 * sum_k w[k] * a[k-i]
    # Vectorized: for each k with significant weight, add 2*w[k]*a[k-i] to grad_a[i]
    grad_a = np.zeros(n)
    # Only process k with significant weight
    thresh = 1e-12
    active_k = np.where(w > thresh)[0]
    for k in active_k:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_optimize(a0, time_limit, T_values=None, iters_per_stage=50):
    """Multi-stage LSE optimization with specified temperature schedule."""
    n = len(a0)
    if T_values is None:
        T_values = [50, 20, 10, 5, 2, 1, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
    
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    for T in T_values:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func(z_val):
            return lse_obj_grad(z_val, T, n)[0]

        def jac(z_val):
            return lse_obj_grad(z_val, T, n)[1]

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 500,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)

            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()
                print(f"      LSE T={T:.4f}: new best score={sc:.10f}")
        except Exception as e:
            print(f"      LSE T={T:.4f}: error {e}")
            continue

    return best_a, best_score


def active_set_subgradient(a0, time_limit, n_steps=1000000, active_frac=0.005):
    """Subgradient method with exact active set and Nesterov momentum."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()

    best_a = a.copy()
    best_score = score_normalized(a)
    lower_bound = 2.0 * n / (2.0 * n - 1.0)

    y = a.copy()  # Nesterov lookahead
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        # Evaluate at lookahead point
        b_y = fast_convolve(y)
        max_by = b_y.max()
        
        # Find active set at y
        thresh = max_by * (1 - active_frac)
        active = np.where(b_y >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b_y))])
        
        # Subgradient at y
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += 2.0 * y[k - idx]
        g /= len(active)
        
        # Nesterov update
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * (step + 1) * (step + 1))) / 2.0
        # Polyak step size
        cur_score = 2.0 * n * max_by
        gap = cur_score - lower_bound
        grad_norm_sq = np.dot(g, g)
        if grad_norm_sq < 1e-30:
            break
        eta = gap / (2.0 * grad_norm_sq)
        eta = min(eta, 1.0 / (step + 1))  # fallback step size
        
        # Update z
        S = np.dot(g, y)
        grad_z = y * (g - S)
        gnorm = np.linalg.norm(grad_z)
        if gnorm < 1e-15:
            break
        step_size = eta * gnorm  # scale to get reasonable z-update
        
        z_new = z - step_size * grad_z / gnorm
        a_new = softmax_from_z(z_new)
        
        # Line search: if score worsened, use bisection
        sc_new = score_normalized(a_new)
        if sc_new > cur_score * 1.05:
            # Bisection
            lo_e, hi_e = 0.0, step_size
            for _ in range(8):
                mid_e = (lo_e + hi_e) / 2.0
                z_mid = z - mid_e * grad_z / gnorm
                a_mid = softmax_from_z(z_mid)
                if score_normalized(a_mid) < cur_score:
                    lo_e = mid_e
                else:
                    hi_e = mid_e
            step_size = lo_e
            z_new = z - step_size * grad_z / gnorm
            a_new = softmax_from_z(z_new)

        # Nesterov momentum for y
        momentum = (t_k - 1.0) / t_k
        y_new = a_new + momentum * (a_new - a)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy > 1e-15:
            y_new /= sy
        else:
            y_new = a_new.copy()
        
        z = z_new
        a = a_new
        y = y_new
        
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
        
        # Also check y
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        if (step + 1) % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad step={step+1} ({elapsed:.0f}s): best={best_score:.10f} eta={eta:.2e}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-stage LSE with very low final temperatures (down to 0.0005)
2. Direct active-set subgradient with Nesterov momentum and Polyak step
3. Hankel-structured CG acceleration for Newton-like steps
4. Multi-length exploration with interpolation
5. Adaptive restarts and time management
Key improvements over previous: much lower LSE temperatures, tighter active sets,
Hankel-structured second-order steps, and better phase allocation.
"""

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def fast_convolve(a):
    """Fast linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return ifft(A * A).real[:size]


def score_normalized(a):
    """Score assuming sum(a)=1: returns 2*n*max(b)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_from_z(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Inverse of softmax (log)."""
    return np.log(np.maximum(a, 1e-300))


def lse_obj_grad(z_in, T_val, n):
    """Compute LSE objective and gradient w.r.t. log-probabilities z."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    thresh = 1e-14
    active_k = np.where(w > thresh)[0]
    for k in active_k:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_stage(a0, T, n_iters=80):
    """Single LSE stage at fixed temperature. Returns updated z and best."""
    n = len(a0)
    z = z_from_a(a0).copy()

    def func(z_val):
        return lse_obj_grad(z_val, T, n)[0]

    def jac(z_val):
        return lse_obj_grad(z_val, T, n)[1]

    try:
        res = minimize(
            fun=func, x0=z, jac=jac, method='L-BFGS-B',
            options={'maxiter': n_iters, 'ftol': 1e-18,
                     'gtol': 1e-15, 'maxfun': n_iters * 500,
                     'disp': False}
        )
        z = res.x
    except Exception:
        pass

    a_check = softmax_from_z(z)
    return z, a_check


def multi_stage_lse(a0, time_limit, T_list, iters_per_stage=80):
    """Multi-stage LSE with specified temperature schedule."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    for T in T_list:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        z, a_check = lse_stage(softmax_from_z(z), T, iters_per_stage)
        sc = score_normalized(a_check)

        if sc < best_score:
            best_score = sc
            best_a = a_check.copy()
            elapsed = time.monotonic() - t0
            print(f"      LSE T={T:.5f} ({elapsed:.0f}s): score={sc:.10f}")

    return best_a, best_score, z


def subgradient_step(a, active_indices):
    """Compute subgradient at a using given active indices."""
    n = len(a)
    g = np.zeros(n)
    for k in active_indices:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        g[idx] += 2.0 * a[k - idx]
    g /= len(active_indices)
    return g


def hankel_mvv(w, v):
    """
    Compute H @ v where H[i,j] = w[i+j] (Hankel matrix),
    using FFT. w has length 2n-1, v has length n.
    """
    n = len(v)
    # H[i,j] = w[i+j], so (Hv)[i] = sum_j w[i+j] * v[j]
    # This is cross-correlation of w and v
    # (Hv)[i] = sum_j w[i+j] * v[j]
    # Let u = v reversed. Then (Hv)[i] = sum_j w[i+j] * u[n-1-j]
    # = (w * u)[i + n - 1] where * is linear convolution... 
    # Actually let me think more carefully.
    # (Hv)[i] = sum_{j=0}^{n-1} w[i+j] * v[j]
    # Let's define this as a cross-correlation: sum_j w[i+j] v[j]
    # = sum_j w[m] v[m-i] where m = i+j, i.e., m ranges from i to i+n-1
    # This is the cross-correlation of w and v at lag i (with w as the "longer" signal)
    # Using FFT: cross_corr = ifft(conj(fft(w)) * fft(v padded))
    # But we need to be careful about the definition.
    
    # Simpler approach: embed into Toeplitz and use FFT
    # (Hv)[i] = sum_j w[i+j] v[j] for i = 0,...,n-1, j = 0,...,n-1
    # This is the "anti-diagonal" convolution.
    # Let r[j] = v[n-1-j] (reverse of v)
    # Then (Hv)[i] = sum_j w[i+j] v[j] = sum_j w[i+j] r[n-1-j]
    # Let k = n-1-j, so j = n-1-k:
    # = sum_k w[i+n-1-k] r[k] = (w * r)[i+n-1] where * is standard convolution
    # (w*r)[m] = sum_k w[m-k] r[k]
    # So (Hv)[i] = (w*r)[i+n-1]
    
    r = v[::-1]
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    W = fft(w, fft_size)
    R = fft(r, fft_size)
    conv = ifft(W * R).real
    return conv[n - 1:n - 1 + n]


def nesterov_subgradient(a0, time_limit, n_steps=500000, active_frac=0.003):
    """
    Nesterov-accelerated subgradient with Polyak step size.
    Uses exact active set from the lookahead point.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()

    best_a = a.copy()
    best_score = score_normalized(a)
    lower_bound = 2.0 * n / (2.0 * n - 1.0)

    y = a.copy()
    t0 = time.monotonic()
    best_score_t0 = best_score
    stagnant_count = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b_y = fast_convolve(y)
        max_by = b_y.max()
        cur_score = 2.0 * n * max_by

        # Find active set
        thresh = max_by * (1.0 - active_frac)
        active = np.where(b_y >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b_y))])
        if len(active) > 100:
            top_idx = np.argsort(b_y[active])[-100:]
            active = active[top_idx]

        # Subgradient at y
        g = subgradient_step(y, active)

        # Convert to z-space gradient
        S = np.dot(g, y)
        grad_z = y * (g - S)
        gnorm = np.linalg.norm(grad_z)
        if gnorm < 1e-15:
            break

        # Polyak step size
        gap = cur_score - lower_bound
        if gap < 1e-12:
            break
        eta = gap / (2.0 * gnorm * gnorm)
        eta = min(eta, 1.0 / np.sqrt(step + 1))

        # Nesterov parameter
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * (step + 1) ** 2)) / 2.0
        momentum = (t_k - 1.0) / t_k

        # Update z
        z_new = z - eta * grad_z
        a_new = softmax_from_z(z_new)

        # Line search
        sc_new = score_normalized(a_new)
        if sc_new > cur_score * 1.02:
            lo_e, hi_e = 0.0, eta
            for _ in range(10):
                mid_e = (lo_e + hi_e) / 2.0
                z_mid = z - mid_e * grad_z
                a_mid = softmax_from_z(z_mid)
                if score_normalized(a_mid) < cur_score:
                    lo_e = mid_e
                else:
                    hi_e = mid_e
            eta = lo_e * 0.9
            z_new = z - eta * grad_z
            a_new = softmax_from_z(z_new)
            sc_new = score_normalized(a_new)

        # Nesterov lookahead update
        y_new = a_new + momentum * (a_new - a)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy > 1e-15:
            y_new /= sy
        else:
            y_new = a_new.copy()

        z = z_new
        a = a_new
        y = y_new

        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            stagnant_count = 0
        else:
            stagnant_count += 1

        # Check y as well
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            stagnant_count = 0

        if stagnant_count > 200000:
            # Restart with perturbation
            pert = 0.02 * np.random.default_rng(step).normal(0, 1, n)
            a_pert = a + pert
            a_pert = np.maximum(a_pert, 0)
            a_pert /= a_pert.sum()
            z = z_from_a(a_pert)
            y = a_pert.copy()
            stagnant_count = 0

        if (step + 1) % 200000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovSG step={step+1} ({elapsed:.0f}s): best={best_score:.10f} cur={cur_score:.10f}")

    return best_a, best_score


def cg_newton_step(a, active_indices, max_cg_iters=20, tol=1e-8):
    """
    Compute a Newton-like direction using CG on the quadratic approximation
    at the active set. The Hessian is a Hankel matrix with structure.
    Returns the direction d (in a-space) such that we move a -> a + d.
    """
    n = len(a)
    # Build weights: uniform over active set
    K = len(active_indices)
    # H[i,j] = (2/K) * count of active k such that k = i+j
    # w[k] = (2/K) if k is active, 0 otherwise
    w = np.zeros(2 * n - 1)
    for k in active_indices:
        w[k] = 2.0 / K

    # Gradient: g[i] = sum_k w[k] * a[k-i] = (2/K) * sum_{k active} a[k-i]
    g = np.zeros(n)
    for k in active_indices:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        g[idx] += (2.0 / K) * a[k - idx]

    # Project gradient to simplex tangent: g_proj = g - mean(g)*a (for softmax param)
    # Actually, let's work directly in a-space with projection
    # We want to find d such that:
    #   H @ d = -g (Newton equation)
    #   sum(d) = 0 (maintain normalization)
    #   a + d >= 0 (maintain non-negativity)
    
    # CG to solve H @ d = -g (ignoring the constraint for now)
    # r = -g - H@d, start with d=0, r=-g
    d = np.zeros(n)
    r = -g.copy()
    p_vec = r.copy()
    rp = np.dot(r, p_vec)
    
    for cg_it in range(max_cg_iters):
        Hp = hankel_mvv(w, p_vec)
        pHp = np.dot(p_vec, Hp)
        if abs(pHp) < 1e-30:
            break
        alpha = rp / pHp
        d += alpha * p_vec
        r_new = r - alpha * Hp
        rp_new = np.dot(r_new, p_vec)
        if abs(rp_new) < tol * abs(rp):
            r = r_new
            break
        p_vec = r_new + (rp_new / rp) * p_vec
        r = r_new
        rp = rp_new

    # Project: ensure sum(d) = 0
    mean_d = np.dot(d, a)  # for softmax: the constraint is sum(a*d/a) = 0, i.e., dot(d, 1) adjusted
    # Actually in a-space with sum(a)=1 constraint, we need sum(d) = 0
    # But if we're in log-space, the constraint is different.
    # Let's just return d and handle projection in the caller.
    d -= np.mean(d)  # Ensure sum(d) = 0
    
    # Check non-negativity: a + d >= 0
    # If violated, scale d down
    min_val = np.min(a + d)
    if min_val < 0:
        scale = -a[np.argmin(a + d)] / (d[np.argmin(a + d)] + 1e-30)
        d *= max(0, scale * 0.9)

    return d, g


def newton_polish(a0, time_limit, max_iters=200):
    """Newton-like polish using CG on the active-set Hessian."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = score_normalized(a)
    lower_bound = 2.0 * n / (2.0 * n - 1.0)
    t0 = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b

        # Active set: indices where b[k] >= max_b * (1 - 1e-4)
        thresh = max_b * 0.9999
        active = np.where(b >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        if len(active) > 200:
            top_idx = np.argsort(b[active])[-200:]
            active = active[top_idx]

        # Newton direction
        d, g = cg_newton_step(a, active, max_cg_iters=15)
        
        if np.linalg.norm(d) < 1e-15:
            break

        # Line search
        a_new = a + d
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new /= sn
        sc_new = score_normalized(a_new)

        if sc_new < cur_score:
            a = a_new
            if sc_new < best_score:
                best_score = sc_new
                best_a = a.copy()
        else:
            # Backtracking
            step = 1.0
            for _ in range(20):
                step *= 0.5
                a_try = a + step * d
                a_try = np.maximum(a_try, 0)
                st = a_try.sum()
                if st < 1e-15:
                    break
                a_try /= st
                sc_try = score_normalized(a_try)
                if sc_try < cur_score:
                    a = a_try
                    if sc_try < best_score:
                        best_score = sc_try
                        best_a = a.copy()
                    break

        if it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Newton iter={it} ({elapsed:.0f}s): best={best_score:.10f} cur={cur_score:.10f} |active|={len(active)}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate a structured initial sequence."""
    choice = rng.integers(0, 5)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.5) ** 2) / 0.03)
    elif choice == 3:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(3 * x) + 0.1 * np.cos(7 * x)
    else:
        a = np.ones(n) * (1.0 + 0.15 * np.cos(2 * np.pi * 2 * np.arange(n) / n))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Uses multi-stage LSE, Nesterov subgradient,
    Newton polish with Hankel-structured CG, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    global_deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    # Load starting sequence
    best_seq = None
    best_score = float('inf')
    n_base = 1200

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = score_normalized(best_seq)
            print(f"Starting: n={n_base}, score={best_score:.10f}")
    except Exception:
        best_seq = structured_init(n_base, rng)
        best_score = score_normalized(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.10f}")

    # === Phase 1: Multi-stage LSE (45% time) ===
    print(f"\n=== Phase 1: Multi-stage LSE (score={best_score:.10f}) ===")
    t1 = 0.45 * (budget_s - 30)
    T_list_1 = [30, 15, 8, 4, 2, 1, 0.5, 0.3, 0.2, 0.1, 0.07, 0.05, 0.03, 0.02, 0.015, 0.01, 0.007, 0.005]
    a_out, sc_out, _ = multi_stage_lse(best_seq, t1, T_list_1, iters_per_stage=60)
    if sc_out < best_score:
        best_score = sc_out
        best_seq = a_out.copy()
        print(f"  Phase 1 improved: {best_score:.10f}")
    print(f"  Phase 1 done: best_score={best_score:.10f}")

    # === Phase 2: Nesterov Subgradient (20% time) ===
    print(f"\n=== Phase 2: Nesterov Subgradient (score={best_score:.10f}) ===")
    remaining = global_deadline - time.monotonic()
    t2 = min(0.20 * (budget_s - 30), remaining * 0.55)
    if t2 > 20:
        a_sg, sc_sg = nesterov_subgradient(best_seq, t2, n_steps=1000000, active_frac=0.002)
        if sc_sg < best_score:
            best_score = sc_sg
            best_seq = a_sg.copy()
            print(f"  Subgrad improved: {best_score:.10f}")

        # Tighter active set pass
        remaining = global_deadline - time.monotonic()
        if remaining > 20:
            a_sg2, sc_sg2 = nesterov_subgradient(best_seq, remaining * 0.5, n_steps=500000, active_frac=0.0005)
            if sc_sg2 < best_score:
                best_score = sc_sg2
                best_seq = a_sg2.copy()
                print(f"  Tight subgrad improved: {best_score:.10f}")
    print(f"  Phase 2 done: best_score={best_score:.10f}")

    # === Phase 3: Newton Polish (15% time) ===
    print(f"\n=== Phase 3: Newton Polish (score={best_score:.10f}) ===")
    remaining = global_deadline - time.monotonic()
    t3 = min(0.15 * (budget_s - 30), remaining * 0.5)
    if t3 > 15:
        a_nw, sc_nw = newton_polish(best_seq, t3, max_iters=100)
        if sc_nw < best_score:
            best_score = sc_nw
            best_seq = a_nw.copy()
            print(f"  Newton improved: {best_score:.10f}")
    print(f"  Phase 3 done: best_score={best_score:.10f}")

    # === Phase 4: Very low T LSE polish (10% time) ===
    print(f"\n=== Phase 4: Low-T LSE polish (score={best_score:.10f}) ===")
    remaining = global_deadline - time.monotonic()
    t4 = min(0.10 * (budget_s - 30), remaining * 0.5)
    if t4 > 15:
        T_list_4 = [0.02, 0.01, 0.005, 0.003, 0.002, 0.001, 0.0005]
        a_polish, sc_polish, _ = multi_stage_lse(best_seq, t4, T_list_4, iters_per_stage=50)
        if sc_polish < best_score:
            best_score = sc_polish
            best_seq = a_polish.copy()
            print(f"  Low-T LSE improved: {best_score:.10f}")
    print(f"  Phase 4 done: best_score={best_score:.10f}")

    # === Phase 5: Multi-length exploration (5% time) ===
    print(f"\n=== Phase 5: Multi-length (score={best_score:.10f}) ===")
    remaining = global_deadline - time.monotonic()
    t5 = min(0.05 * (budget_s - 30), remaining * 0.6)
    if t5 > 30:
        # Try a couple of nearby lengths
        for n_try in [n_base + 300, n_base - 200]:
            if n_try <= 200:
                continue
            remaining = global_deadline - time.monotonic()
            if remaining < 20:
                break
            t_try = min(t5 / 2.0, remaining * 0.45)
            if t_try < 10:
                break
            print(f"  Trying n={n_try}, budget={t_try:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            T_list_try = [5, 2, 1, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
            a_try, sc_try, _ = multi_stage_lse(init_try, t_try, T_list_try, iters_per_stage=30)
            if sc_try < best_score:
                best_score = sc_try
                best_seq = a_try.copy()
                print(f"  n={n_try} improved: {best_score:.10f}")
    print(f"  Phase 5 done: best_score={best_score:.10f}")

    # === Phase 6: Final combined polish (remaining time) ===
    print(f"\n=== Phase 6: Final polish (score={best_score:.10f}) ===")
    remaining = global_deadline - time.monotonic()
    if remaining > 20:
        # One more Newton pass
        a_f1, sc_f1 = newton_polish(best_seq, min(remaining * 0.3, 60), max_iters=50)
        if sc_f1 < best_score:
            best_score = sc_f1
            best_seq = a_f1.copy()
            print(f"  Final Newton improved: {best_score:.10f}")

        remaining = global_deadline - time.monotonic()
        if remaining > 15:
            # Very tight subgradient
            a_f2, sc_f2 = nesterov_subgradient(best_seq, remaining * 0.5, n_steps=300000, active_frac=0.0003)
            if sc_f2 < best_score:
                best_score = sc_f2
                best_seq = a_f2.copy()
                print(f"  Final subgrad improved: {best_score:.10f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
            print(f"\nFINAL verified score: {best_score:.10f} (n={len(best_seq)})")
        else:
            print(f"\nWARNING: evaluate_sequence returned non-finite, falling back")
            # Fallback: use the normalized sequence directly
            best_seq = np.maximum(best_seq, 0)
            best_seq /= best_seq.sum()
            fs = evaluate_safe(best_seq)
            print(f"  Fallback score: {fs:.10f}")

    return [float(x) for x in best_seq]
