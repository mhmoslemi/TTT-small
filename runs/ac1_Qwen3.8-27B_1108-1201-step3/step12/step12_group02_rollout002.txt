Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Full-softmax mirror descent with adaptive temperature annealing
   - O(n log n) per iteration via FFT
   - Smooth gradient from softmax over all convolution peaks

2. Multi-length exploration with diverse initializations
3. Local search polish (exchange moves + coordinate descent)

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

def fast_autocorr(a):
    """Full convolution a*a using FFT. Returns array of length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_score(a):
    """Compute 2n*max(b)/(sum(a))^2 quickly."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)

def cross_corr_fft(w, a):
    """
    Compute cross-correlation: result[j] = sum_k w[k] * a[k-j]
    where w has length 2n-1 and a has length n.
    result has length n.
    """
    n = len(a)
    a_rev = a[::-1]  # length n
    conv_len = len(w) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a_rev, fft_size)
    conv = irfft(W * A, fft_size)
    return conv[n-1:n-1+n]

def softmax(x, T):
    """Stable softmax with temperature T."""
    x_scaled = x / T
    x_shift = x_scaled - x_scaled.max()
    e = np.exp(x_shift)
    return e / e.sum()

def mirror_descent_smoothing(a0, time_limit, T_start=50.0, T_end=1.0,
                              eta0=0.03, decay=0.6, n_anneal=500,
                              seed=42, verbose=True):
    """
    Mirror descent (multiplicative weights) with full-softmax weighted subgradient.
    Temperature anneals from T_start to T_end over n_anneal iterations.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_autocorr(a)
        
        # Anneal temperature
        frac = min(1.0, it / float(n_anneal))
        # Use geometric annealing
        T = T_start * (T_end / T_start) ** frac
        
        # Softmax weights over all peaks
        w = softmax(b, T)
        
        # Gradient
        cc = cross_corr_fft(w, a)
        grad = 2.0 * cc
        
        # Learning rate
        eta = eta0 / (1.0 + it / 200.0) ** decay
        
        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a + 0.01 * rng.standard_normal(n)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
        a = a_new / s_new
        
        # Score check
        if it % 20 == 0 or it < 50:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stall = 0
            else:
                stall += 1

        if stall > 200:
            noise = 0.02 * rng.standard_normal(n)
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            stall = 0
            # Reset temperature to restart annealing
            # (implicitly handled by continuing to decrease T)
        
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD-smooth it={it} ({elapsed:.0f}s): best={best_score:.6f}, T={T:.2f}")

    return best_a, best_score

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

3. L-BFGS on LogSumExp for fine-grained optimization
4. Exchange-move local search for final polish

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

def fast_autocorr(a):
    """Full convolution a*a using FFT. Returns array of length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_score(a):
    """Compute 2n*max(b)/(sum(a))^2 quickly."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)

def cross_corr_fft(w, a):
    """Compute cross-correlation: result[j] = sum_k w[k] * a[k-j]."""
    n = len(a)
    a_rev = a[::-1]
    conv_len = len(w) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a_rev, fft_size)
    conv = irfft(W * A, fft_size)
    return conv[n-1:n-1+n]

def softmax_stable(x, T):
    """Stable softmax with temperature T."""
    x_scaled = np.asarray(x, dtype=np.float64) / T
    x_shift = x_scaled - x_scaled.max()
    e = np.exp(x_shift)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s

def md_full_softmax(a0, time_limit, T_start=30.0, T_end=0.5,

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:

    else:
        a = a / s

    best_score = fast_score(a)
    t0 = time.monotonic()

    it = 0

        it += 1
        b = fast_autocorr(a)

        # Geometric annealing
        frac = min(1.0, it / float(n_anneal))
        T = T_start * (T_end / T_start) ** frac

        # Full softmax weights
        w = softmax_stable(b, T)

        # Gradient: 2 * cross_corr(w, a)
        cc = cross_corr_fft(w, a)
        grad = 2.0 * cc

        # Adaptive learning rate
        eta = eta0 / (1.0 + it / 300.0) ** decay

        # Multiplicative update
        log_update = -eta * grad
        # Clip to prevent overflow/underflow
        log_update = np.clip(log_update, -10, 10)
        a_new = a * np.exp(log_update)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a + 0.01 * rng.standard_normal(n)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
        a = a_new / s_new

                best_a = a.copy()
                stall = 0
            else:
                stall += 1

            # Try a larger perturbation

            a = a + scale * rng.standard_normal(n)
            a = np.maximum(a, 0)
            s2 = a.sum()
            if s2 > 1e-15:
                a = a / s2
            stall = 0

        if verbose and it % 3000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD it={it} ({elapsed:.0f}s): best={best_score:.6f}, T={T:.3f}")

    else:

    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = fast_score(a_init)

        iters = max(10, min(50, int(20 + 30 * si / max(1, n_stages))))

        def func(z_in):
            z_in = np.asarray(z_in, dtype=np.float64)
            z_shift = z_in - z_in.max()
            e = np.exp(z_shift)
            a_local = e / e.sum()
            b = fast_autocorr(a_local)
            Tb = stage_T * b
            Tb_shift = Tb - Tb.max()
            exp_Tb = np.exp(Tb_shift)
            val = (Tb.max() + np.log(exp_Tb.sum())) / stage_T
            return val

        def jac(z_in):
            z_in = np.asarray(z_in, dtype=np.float64)
            z_shift = z_in - z_in.max()
            e = np.exp(z_shift)
            a_local = e / e.sum()
            b = fast_autocorr(a_local)
            Tb = stage_T * b
            Tb_shift = Tb - Tb.max()
            exp_Tb = np.exp(Tb_shift)
            w = exp_Tb / exp_Tb.sum()
            cc = cross_corr_fft(w, a_local)
            grad_a = 2.0 * cc
            S = np.dot(grad_a, a_local)
            grad_z = a_local * (grad_a - S)
            return grad_z

        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': iters, 'ftol': 1e-15,
                                    'gtol': 1e-8, 'maxfun': iters * 100,
                                    'disp': False})
            z = res.x
            z_shift = z - z.max()
            a = np.exp(z_shift)
            a = a / a.sum()
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        except Exception:
            pass

    return best_a, best_score


def exchange_local_search(a0, time_limit, seed=42, n_trials=50000):
    """
    Local search via random exchange moves.
    Move mass from position i to position j, accept if score improves.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, fast_score(a)
    a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    fails = 0

    # Precompute b for efficiency
    b = fast_autocorr(a)
    threshold = b.max()

    trials = 0
    while time.monotonic() - t0 < time_limit and fails < 500:
        trials += 1
        # Pick random source and destination
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j or a[i] < 1e-12:
            continue

        # Try moving a fraction of a[i] to a[j]
        fracs = [0.05, 0.1, 0.2, 0.5]
        frac = fracs[int(rng.integers(len(fracs)))]
        amt = a[i] * frac

        a_trial = a.copy()
        a_trial[i] -= amt
        a_trial[j] += amt
        if a_trial[i] < 0:
            a_trial[i] = 0
        a_trial = a_trial / a_trial.sum()

        sc = fast_score(a_trial)
        if sc < best_score - 1e-14:
            a = a_trial
            best_score = sc
            best_a = a.copy()
            threshold = fast_autocorr(a).max()
            fails = 0
        else:
            fails += 1

    return best_a, best_score
