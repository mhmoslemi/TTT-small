import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_convolution(a, b):
    """Compute full convolution using FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def logsumexp(x, beta):
    """Compute (1/beta) * log(sum(exp(beta * x))) in a numerically stable way."""
    beta_x = beta * x
    m = beta_x.max()
    if m > 500:
        # This shouldn't happen if x is bounded
        return x.max()
    e = np.exp(beta_x - m)
    s = e.sum()
    if s < 1e-300:
        return x.max()
    return (m + np.log(s)) / beta

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_smooth_gradient(a, beta):
    """
    Compute the smooth max of a*a and its gradient.
    Returns: (autocorr, smooth_value, gradient)
    """
    n = len(a)
    autocorr = fast_autocorr(a)
    
    # Softmax weights
    scaled = beta * autocorr
    m = scaled.max()
    e = np.exp(scaled - m)
    s = e.sum()
    if s < 1e-300:
        # Fallback to hard max
        k_max = autocorr.argmax()
        p = np.zeros(len(autocorr))
        p[k_max] = 1.0
    else:
        p = e / s
    
    # Gradient: grad_i = 2 * sum_k p_k * a_{k-i} (when 0 <= k-i < n)
    # = 2 * np.correlate(p, a, 'valid')[i] ... let me verify
    # np.correlate(p, a, 'valid')[i] = sum_{j=0}^{n-1} p[j+i] * a[j]
    # We want: grad_i = 2 * sum_{k: 0<=k-i<n} p_k * a_{k-i}
    # Let j = k-i: grad_i = 2 * sum_{j=0}^{n-1} p_{i+j} * a[j]
    # = 2 * np.correlate(p, a, 'valid')[i]  ✓ (since len(p) = 2n-1 >= n = len(a))
    
    grad = 2.0 * np.correlate(p, a, mode='valid')
    
    smooth_val = (m + np.log(s)) / beta if s > 1e-300 else autocorr.max()
    
    return autocorr, smooth_val, grad

def adam_smooth_optimize(a0, time_limit, beta_start=5.0, beta_end=500.0, 
                         lr=0.05, verbose=True):
    """
    Adam optimization of the logsumexp-smoothed max autocorrelation.
    Optimizes on the simplex (sum=1, a>=0).
    """
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    
    n = len(a)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    
    # Adam state
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0
    
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    t0 = time.monotonic()
    total_iters = 0
    last_print = t0
    
    # Annealing schedule: exponential
    n_anneal = max(1, int(time_limit * 20))  # anneal over roughly the first half
    
    while True:
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        
        # Anneal beta
        progress = min(1.0, elapsed / (time_limit * 0.5))
        beta = beta_start * (beta_end / beta_start) ** progress
        
        # Compute gradient
        autocorr, smooth_val, grad = compute_smooth_gradient(a, beta)
        
        # Adam step
        t_adam += 1
        m_adam = beta1 * m_adam + (1 - beta1) * grad
        v_adam = beta2 * v_adam + (1 - beta2) * (grad ** 2)
        m_hat = m_adam / (1 - beta1 ** t_adam)
        v_hat = v_adam / (1 - beta2 ** t_adam)
        
        # Learning rate schedule: decay after warmup
        if t_adam < 100:
            lr_t = lr * t_adam / 100.0
        else:
            lr_t = lr / (1 + 0.01 * (t_adam - 100))
        
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project onto simplex (sum=1, a>=0)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        a = a_new
        total_iters += 1
        
        # Evaluate
        if total_iters % 10 == 0:
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
        
        if verbose and time.monotonic() - last_print > 30:
            last_print = time.monotonic()
            cur_score = 2.0 * n * fast_autocorr(a).max()
            print(f"      Smooth: t={total_iters}, beta={beta:.1f}, cur={cur_score:.6f}, best={best_score:.6f}")
    
    return best_a, best_score, total_iters

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def build_conv_matrix(a_fixed):
    """Build the (2n-1) x n matrix T where (T@b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. (a_fixed * b)_k <= M, sum(b)=1, b>=0. Returns (b, M)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def nesterov_lp_optimize(a0, time_limit, solver, verbose=False):
    """Nesterov-accelerated LP best-response."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    iter_count = 0
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue
        iter_count += 1
        
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_new_s = y_new.sum()
        if y_new_s < 1e-15:
            y_new = y.copy()
            y_new_s = 1.0
        y_new = y_new / y_new_s
        
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            a_s = a_new.sum()
            if a_s < 1e-15:
                a_new = y_new.copy()
                a_s = 1.0
            a_new = a_new / a_s
        
        y = y_new
        a = a_new
        
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if verbose and t % 500 == 0:
            print(f"      NLP t={t}: best={best_score:.6f}")
    
    return best_a, best_score, iter_count

def qp_active_set_refinement(a0, time_limit, n_active=20):
    """QP refinement on active constraint set."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    b = fast_autocorr(a)
    max_b = b.max()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * max_b
    
    for round_num in range(30):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Identify active lags (those within 1% of max)
        threshold = max_b * (1 - 0.005)
        active_lags = np.where(b >= threshold)[0]
        if len(active_lags) > n_active:
            active_lags = np.argsort(b[active_lags])[::-1][:n_active] + active_lags
            active_lags = np.unique(active_lags[:n_active])
        
        # Build QP
        a_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        
        constraints = [cp.sum(a_var) == 1, M_var >= 0]
        for k in active_lags:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            other_idx = k - idx
            q_expr = cp.sum(a_var[idx] * a_var[other_idx])
            constraints.append(q_expr <= M_var)
        
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        solved = False
        for solver in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
            try:
                prob.solve(solver=solver, max_iter=20000)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue
        
        if not solved:
            break
        
        a_new = np.asarray(a_var.value, dtype=np.float64)
        a_new = np.maximum(a_new, 0)
        a_new_s = a_new.sum()
        if a_new_s < 1e-15:
            break
        a_new = a_new / a_new_s
        
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            b = fast_autocorr(a)
            max_b = b.max()
        else:
            # Try mixing
            improved = False
            for frac in [0.2, 0.5, 0.8]:
                a_mix = (1 - frac) * a + frac * a_new
                a_mix = np.maximum(a_mix, 0)
                a_mix_s = a_mix.sum()
                if a_mix_s < 1e-15:
                    continue
                a_mix = a_mix / a_mix_s
                score_mix = 2.0 * n * fast_autocorr(a_mix).max()
                if score_mix < best_score:
                    best_score = score_mix
                    best_a = a_mix.copy()
                    a = a_mix
                    b = fast_autocorr(a)
                    max_b = b.max()
                    improved = True
                    break
            if not improved and round_num >= 5:
                break
            # Widen active set
            n_active = min(n_active * 2, 60)
    
    return best_a, best_score

def peak_shaving(a0, time_limit, seed=42):
    """Peak shaving via coordinate descent."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        k_max = b.argmax()
        
        # Find top contributors to the peak
        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_max - idx]
        total = contrib.sum()
        if total < 1e-15:
            break
        
        # Try reducing top contributors
        found = False
        top_ci = np.argsort(contrib)[::-1][:3]
        for ci in top_ci:
            src = idx[ci]
            # Find a good target (position with low contribution to top peaks)
            top_peaks = np.argsort(b)[::-1][:5]
            for frac in [0.01, 0.02, 0.05, 0.1]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                # Try moving to a random position far from peaks
                for _ in range(5):
                    dst = rng.integers(0, n)
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    a_trial_s = a_trial.sum()
                    if a_trial_s < 1e-15:
                        continue
                    a_trial = a_trial / a_trial_s
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        found = True
                        stall = 0
                        break
                if found:
                    break
            if found:
                break
        
        if not found:
            stall += 1
            # Random perturbation
            n_pert = rng.integers(1, 4)
            for _ in range(n_pert):
                src = rng.integers(0, n)
                dst = rng.integers(0, n)
                frac = rng.uniform(0.005, 0.05)
                amt = a[src] * frac
                a[src] -= amt
                a[dst] += amt
            a = np.maximum(a, 0)
            a_s = a.sum()
            if a_s < 1e-15:
                break
            a = a / a_s
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
                stall = 0
            if stall > 200:
                break
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

Wait, I realize there's a potential issue with my smooth optimization. Let me think more carefully about the learning rate.

The objective is `f(a) = 2n * logsumexp(beta * (a*a)) / beta` (for the smooth version).

The gradient `df/da` has magnitude roughly `2 * max(a*a) * n` (very rough). For normalized a with sum=1, `max(a*a) ~ 1/n`, so the gradient magnitude is roughly `2`.

With Adam, the effective step size is `lr / (sqrt(v) + eps)` where `v` is the running average of squared gradients. If the gradient is roughly constant at magnitude 2, then `sqrt(v) ≈ 2`, and the effective step is `lr / 2`. For `lr = 0.03`, that's about 0.015 per coordinate.

Since we're on the simplex (sum=1), each coordinate is at most 1 (and typically ~1/n ≈ 0.0006). A step of 0.015 is way too large!

I need to scale the learning rate. A better approach: normalize the gradient by the typical scale of a.

Actually, let me reconsider. The issue is that Adam normalizes per-coordinate, so if coordinate i has a large gradient and coordinate j has a small gradient, Adam will take larger steps in direction j. This is actually good for our purposes.

But the absolute step size matters. If `lr = 0.03` and the gradient is ~2, the raw step is ~0.06 per coordinate. After normalization by `sqrt(v) + eps ≈ 2`, the step is ~0.03. This is indeed too large for coordinates that are ~0.0006.

Let me use a much smaller learning rate, or better yet, use a different parameterization.

**Better approach: Use the log-parameterization.**

Let `a_i = exp(z_i) / sum_j exp(z_j)` (softmax). Then we optimize over z (unconstrained).

The gradient of f with respect to z is:
`df/dz_i = df/da * da/dz_i`

`da/dz_i = a_i * (delta_{ij} - a_j)` (Jacobian of softmax)

So `df/dz_i = (df/da)_i * a_i - a_i * sum_j (df/da)_j * a_j`

This is the "natural gradient" in a sense. The advantage is that the step in z-space automatically respects the simplex constraint.

Let me implement this:

```python
def adam_smooth_optimize_param(a0, time_limit, beta_start=5.0, beta_end=200.0, lr=1.0):
    """Adam optimization using log-parameterization."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    s = a.sum()
    a = a / s
    z = np.log(a)  # log parameterization
    
    n = len(a)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / (time_limit * 0.5))
        beta = beta_start * (beta_end / beta_start) ** progress
        
        # Recompute a from z
        z_shifted = z - z.max()
        a = np.exp(z_shifted)
        a = a / a.sum()
        
        # Compute gradient of f wrt a
        autocorr, smooth_val, grad_a = compute_smooth_gradient(a, beta)
        
        # Convert to gradient wrt z
        # df/dz_i = grad_a_i * a_i - a_i * sum_j grad_a_j * a_j
        inner = np.dot(grad_a, a)
        grad_z = a * (grad_a - inner)
        
        # Adam step
        t_adam += 1
        m_adam = beta1 * m_adam + (1 - beta1) * grad_z
        v_adam = beta2 * v_adam + (1 - beta2) * (grad_z ** 2)
        m_hat = m_adam / (1 - beta1 ** t_adam)
        v_hat = v_adam / (1 - beta2 ** t_adam)
        
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Evaluate periodically
        if t_adam % 5 == 0:
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
    
    return best_a, best_score, t_adam
```

With this parameterization, the learning rate `lr` is in "z-space" which is unbounded. The typical scale of `grad_z` is:
`grad_z_i = a_i * (grad_a_i - inner)`

For `a_i ~ 1/n` and `grad_a ~ 2` (rough), `inner ~ 2/n * n = 2`... hmm, let me think more carefully.

`inner = sum_j grad_a_j * a_j ≈ (2/n) * (n * 1/n) = 2/n`

No wait: `grad_a_j ≈ 2 * max(a*a) ≈ 2/n` (for a good solution). And `a_j ≈ 1/n`. So `inner ≈ sum_j (2/n) * (1/n) = 2/n`.

And `grad_z_i = a_i * (grad_a_i - inner) ≈ (1/n) * (2/n - 2/n) ≈ 0` at the optimum.

Away from the optimum, `grad_a_i` varies between 0 and maybe `4/n`. So `grad_z_i ≈ (1/n) * (4/n) = 4/n^2`.

For n=1600: `grad_z ~ 4/2.56M ≈ 1.5e-6`.

With Adam, the effective step is `lr / sqrt(v) ≈ lr / |grad_z| ≈ lr / (1.5e-6)`. If `lr = 1.0`, the step is about 667,000 in z-space. That's way too large!

I need `lr` to be much smaller. If I want a step of about 0.1 in z-space (which corresponds to a multiplicative change of about 10% in a_i), then `lr ≈ 0.1 * |
