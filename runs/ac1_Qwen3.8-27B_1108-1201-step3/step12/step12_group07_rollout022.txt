        return float('inf')


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def structured_init(n, rng):
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = 1.0 + 0.5 * (2.0 * x - 1.0) + 0.1 * np.sin(4 * np.pi * x)
    elif choice == 1:
        a = np.exp(-3.0 * (x - 0.5)**2) + 0.3
    elif choice == 2:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x) * np.exp(-2.0 * (x - 0.5)**2)
    elif choice == 3:
        a = (1.0 - x) ** 0.7 + 0.1
    elif choice == 4:
        a = np.exp(-5.0 * (x - 0.3)**2) + 0.5 * np.exp(-5.0 * (x - 0.7)**2) + 0.1
    elif choice == 5:
        a = 1.0 + 0.3 * np.sin(3 * np.pi * x) + 0.2 * np.cos(5 * np.pi * x)
    elif choice == 6:
        a = np.maximum(0.5 * (1.0 + np.cos(2 * np.pi * (x + 0.15))), 0.01)
    elif choice == 7:
        a = 1.0 / (1.0 + np.exp(-8.0 * (x - 0.4))) + 0.1
    elif choice == 8:
        a = 0.4 + 0.6 * (x * (1.0 - x)) ** 0.3
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0.001)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def multiplicative_weight_refine(a0, time_limit, max_steps=100000):
    """Fast peak-targeted multiplicative weight descent with adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = 0.1
    last_improve = 0
    stall_count = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve > 3000:
            break

        b = fast_convolve(a)
        max_b = b.max()
        # Use top peaks with soft weighting
        threshold = max_b * 0.90
        peak_weights = np.clip((b - threshold) / max(max_b - threshold, 1e-15), 0, 2)
        
        grad = np.zeros(n)
        # Vectorized gradient computation using FFT trick
        # grad_i = sum_k peak_weights[k] * a[k-i] / sum(peak_weights)
        # This is a cross-correlation of peak_weights with a
        pw = peak_weights
        pw_sum = pw.sum()
        if pw_sum < 1e-15:
            pw_sum = 1.0
        # Use FFT for cross-correlation
        fft_size = 1 << (2 * n - 1 - 1).bit_length() if n > 1 else 2
        A_fft = rfft(a, fft_size)
        P_fft = rfft(pw / pw_sum, fft_size)
        # Cross-correlation: sum_k pw[k] * a[k-i] = (pw * reverse_a)[i]
        # Actually, let's just do it directly for clarity
        # grad[i] = sum_k pw[k] * a[k-i] (with a[j] = 0 for j out of [0,n-1])
        # This is the cross-correlation of pw and a
        # Using the fact that (pw * a)[i] in the correlation sense = sum_k pw[k] a[k-i]
        # Let's use the FFT approach: corr = ifft(conj(fft(pw)) * fft(a))
        # Or more simply: ifft(fft(pw) * conj(fft(a))) gives sum_k pw[k] a[k-i]
        # Hmm, let me just use a loop-free approach
        for k in np.where(pw > 1e-10)[0]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            grad[idx] += pw[k] * a[k - idx]
        
        # Multiplicative update
        exponent = -eta * grad
        exponent -= exponent.max()
        new_a = a * np.exp(exponent)
        sa = new_a.sum()
        if sa < 1e-15:
            eta *= 0.5
            continue
        new_a = new_a / sa

        sc = score_normalized(new_a)
        if sc < best_score:
            best_score = sc
            best_a = new_a.copy()
            a = new_a
            last_improve = step
            eta = min(eta * 1.005, 0.3)
            stall_count = 0
        else:
            eta *= 0.8
            stall_count += 1
            if eta < 1e-8:
                break

    return best_a, best_score


def lse_grad_a(a, T, n):
    """Compute gradient of LogSumExp objective w.r.t. a."""
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    
    grad_a = np.zeros(n)
    active_k = np.where(w > 1e-15)[0]
    for k in active_k:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    
    obj = (2.0 * n / T) * log_sum_exp
    return obj, grad_a, w


def adam_lse_optimize(a0, time_limit, T_start=200.0, T_end=0.01, n_stages=30, max_iter=None, seed=0):
    """Adam-based LogSumExp optimization with perturbation restarts."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    
    if max_iter is None:
        max_iter = 500000
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        # Adam optimizer state
        m = np.zeros(n)
        v = np.zeros(n)
        t_adam = 0
        lr = 0.01 / (1.0 + 0.01 * si)  # Decay learning rate per stage
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        grad_clip = 1.0

        iters_this_stage = min(max_iter // n_stages + 100, 3000)
        stage_last_improve = 0
        
        for it in range(iters_this_stage):
            if time.monotonic() - t0 > time_limit:
                break
            if it - stage_last_improve > iters_this_stage // 2 and si > 2:
                break  # No improvement for half the stage
                
            t_adam += 1
            
            # Compute gradient
            a_curr = softmax_from_z(z)
            obj, grad_a, _ = lse_grad_a(a_curr, stage_T, n)
            
            # Convert to z-space gradient
            S = np.dot(grad_a, a_curr)
            grad_z = a_curr * (grad_a - S)
            
            # Clip gradient
            gnorm = np.linalg.norm(grad_z)
            if gnorm > grad_clip:
                grad_z = grad_z * (grad_clip / gnorm)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Evaluate
            a_new = softmax_from_z(z)
            sc = score_normalized(a_new)
            
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                stage_last_improve = it
                # Slightly increase lr when improving
                lr = min(lr * 1.002, 0.05)
        
        # Print progress
        if (si + 1) % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AdamLSE stage={si+1}/{n_stages} T={stage_T:.3f}: score={best_score:.8f} ({elapsed:.0f}s)")
        
        # Perturbation restart: if we're at low T and stuck, add noise
        if si >= n_stages // 2 and si < n_stages - 2:
            if time.monotonic() - t0 < time_limit - 30:
                # Small random perturbation
                noise_scale = 0.05 * (1.0 - si / n_stages)
                z = z + noise_scale * rng.standard_normal(n)
    
    return best_a


def nesterov_lp(a0, time_limit, solver=None, max_iters=1000000, seed=0):
    """Nesterov LP with improved restart strategy."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    restart_count = 0
    last_score = best_score
    stagnation = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y = y / sy
        else:
            y = a

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha = 2.0 / (t_new + 1.0)

        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        sc_new = score_normalized(a_new)

        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            stagnation = 0
        else:
            stagnation += 1
            if sc_new > sc_new and sc_new > best_score * 1.001:
                t_nest = 1.0
                a_prev = a_new.copy()
                restart_count += 1

        sc_cur = score_normalized(a)
        if sc_new > sc_cur * 1.001 and t_nest > 5.0:
            t_nest = 1.0
            a_prev = a_new.copy()
            restart_count += 1
        else:
            a_prev = a.copy()
            t_nest = t_new

        a = a_new

        if (it + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} t={t_nest:.1f} restarts={restart_count}")

        # Stagnation-based restart with perturbation
        if stagnation > 5000 and it > 1000:
            # Add small perturbation
            perturbation = 0.02 * rng.standard_normal(n)
            a = a + perturbation
            a = np.maximum(a, 0)
            s = a.sum()
            a = a / s
            a_prev = a.copy()
            t_nest = 1.0
            stagnation = 0
            restart_count += 1

    return best_a, best_score


def low_temp_lse_polish(a0, time_limit, T_end=0.005, n_stages=15, max_iter=500000):
    """Very low temperature LSE polishing using L-BFGS."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    
    temps = np.geomspace(T_end * 10, T_end, n_stages)
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        def func(z_val):
            a_tmp = softmax_from_z(z_val)
            obj, _, _ = lse_grad_a(a_tmp, stage_T, n)
            return obj
        
        def jac(z_val):
            a_tmp = softmax_from_z(z_val)
            _, grad_a, _ = lse_grad_a(a_tmp, stage_T, n)
            S = np.dot(grad_a, a_tmp)
            return a_tmp * (grad_a - S)
        
        iters = 50 if si < n_stages // 3 else 30
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-24,
                         'gtol': 1e-20, 'maxfun': iters * 500,
                         'disp': False}
            )
            z = res.x
            a_new = softmax_from_z(z)
            sc = score_normalized(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        except Exception:
            pass
    
    return best_a


def peak_shatter(a0, time_limit, n_rounds=50):
    """Targeted peak reduction by modifying specific coordinates."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    for round_idx in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        peak_pos = int(np.argmax(b))
        peak_val = b[peak_pos]
        
        # Find which pairs contribute most to this peak
        # (a*a)[peak_pos] = sum_j a_j * a[peak_pos - j]
        js = max(0, peak_pos - n + 1)
        je = min(n - 1, peak_pos)
        idx = np.arange(js, je + 1)
        contributions = a[idx] * a[peak_pos - idx]
        
        # Find the top contributors
        top_k = min(5, len(idx))
        top_indices = idx[np.argsort(contributions)[-top_k:]]
        
        # Try reducing each top contributor
        best_trial_a = a.copy()
        best_trial_score = best_score
        
        for ti in top_indices:
            # Try reducing a[ti] by various amounts
            for frac in [0.1, 0.2, 0.3, 0.5]:
                trial = a.copy()
                reduced = trial[ti] * frac
                trial[ti] *= (1 - frac)
                # Redistribute the mass to the coordinate with lowest "pressure"
                # (i.e., the one that contributes least to the peak)
                # Find the coordinate that contributes least to the peak
                contribs_all = np.zeros(n)
                for k in range(2 * n - 1):
                    kjs = max(0, k - n + 1)
                    kje = min(n - 1, k)
                    kidx = np.arange(kjs, kje + 1)
                    if len(kidx) > 0:
                        contribs_all[kidx] += a[kidx] * a[k - kidx]
                # Redistribute to coordinate with lowest contribution
                min_coord = int(np.argmin(contribs_all))
                if min_coord == ti:
                    min_coord = (min_coord + 1) % n
                trial[min_coord] += reduced
                trial = np.maximum(trial, 0)
                ts = trial.sum()
                if ts < 1e-15:
                    continue
                trial = trial / ts
                ts_score = score_normalized(trial)
                if ts_score < best_trial_score:
                    best_trial_score = ts_score
                    best_trial_a = trial.copy()
        
        if best_trial_score < best_score:
            best_score = best_trial_score
            best_a = best_trial_a.copy()
            a = best_trial_a.copy()
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
