def fast_autocorr(a):
    """Full autocorrelation a*a (convolution) using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_score(a):
    """Compute 2n*max(b)/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase strategy:
1. Low-dimensional parameterized families (Gaussian mixtures, squared polynomials,
   Fourier series) optimized with L-BFGS for global exploration
2. Full-dimensional LogSumExp-smoothed L-BFGS optimization with multi-stage annealing
3. FFT-accelerated subgradient descent on the true max objective
4. Coordinate descent peak shaving with line search
5. Multi-length exploration via interpolation and re-optimization
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_autocorr(a):
    """Full convolution a*a using FFT. Returns array of length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_score(a):
    """Compute 2n*max(b)/(sum(a))^2 quickly."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)

def lse_smoothing_obj_grad(a, T):
    """
    Log-sum-exp smoothed objective and gradient.
    f(a) = (1/T)*log(sum_k exp(T*b_k)) where b = conv(a,a)
    Returns (objective_value, gradient_wrt_a)
    """
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    obj = (Tb_max + np.log(Z)) / T
    w = exp_Tb / Z  # weights for each lag k
    
    # Compute gradient using FFT
    # grad_a[j] = (2/T) * [conv(flip(w), a)[2n-2-j] + conv(w, a)[j]]
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    W_fft = rfft(w, fft_size)
    A_fft = rfft(a, fft_size)
    conv_wa = irfft(W_fft * A_fft, fft_size)
    w_flip = w[::-1]
    Wf_fft = rfft(w_flip, fft_size)
    conv_wfa = irfft(Wf_fft * A_fft, fft_size)
    
    # X[j] = conv_wfa[2n-2-j] for j=0,...,n-1
    X = conv_wfa[2*n-2 : n-2 : -1]
    Y = conv_wa[0:n]
    
    grad_a = (2.0 / T) * (X + Y)
    return obj, grad_a


def gaussian_mix_params_to_seq(params, n):
    """
    Convert Gaussian mixture parameters to a sequence.
    params = [w1, c1, s1, w2, c2, s2, ...] (3*J params)
    w = amplitude (log-scaled: actual_w = exp(w)), c = center (in [0,1]), s = width (log-scaled)
    """
    J = len(params) // 3
    x = np.linspace(0, 1, n)
    a = np.zeros(n)
    for j in range(J):
        w = np.exp(params[3*j])       # amplitude > 0
        c = 0.5 + 0.45 * np.tanh(params[3*j+1])  # center in [0.05, 0.95]
        s = np.exp(params[3*j+2]) * 0.1  # width
        a += w * np.exp(-(x - c)**2 / (2 * s**2))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def poly_sq_params_to_seq(params, n):
    """
    Convert squared polynomial parameters to a sequence.
    params = [c0, c1, ..., cd] (d+1 params)
    a[i] = (sum_k c_k * (i/(n-1))^k)^2
    """
    d = len(params) - 1
    x = np.linspace(0, 1, n)
    P = np.zeros(n)
    for k in range(d + 1):
        P += params[k] * x**k
    a = P**2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def fourier_params_to_seq(params, n, K):
    """
    Convert Fourier series parameters to a sequence.
    params = [A0, A1, B1, A2, B2, ..., AK, BK] (2K+1 params)
    a[i] = (A0 + sum_k Ak*cos(2*pi*k*xi) + Bk*sin(2*pi*k*xi))^2
    """
    x = np.linspace(0, 1, n)
    f = np.full(n, params[0])
    for k in range(1, K + 1):
        f += params[2*k-1] * np.cos(2*np.pi*k*x)
        f += params[2*k] * np.sin(2*np.pi*k*x)
    a = f**2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def param_optimize_gauss(n, J, rng, maxiter=200):
    """Optimize a J-bump Gaussian mixture using L-BFGS."""
    def obj_and_grad(params):
        a = gaussian_mix_params_to_seq(params, n)
        sc = fast_score(a)
        # Numerical gradient (expensive but J is small)
        eps = 1e-7
        grad = np.zeros(3 * J)
        for i in range(len(params)):
            p_plus = params.copy()
            p_plus[i] += eps
            p_minus = params.copy()
            p_minus[i] -= eps
            fp = fast_score(gaussian_mix_params_to_seq(p_plus, n))
            fm = fast_score(gaussian_mix_params_to_seq(p_minus, n))
            grad[i] = (fp - fm) / (2 * eps)
        return sc, grad
    
    best_params = None
    best_score = float('inf')
    
    for restart in range(3):
        params0 = np.zeros(3 * J)
        for j in range(J):
            params0[3*j] = rng.uniform(-1, 1)     # log amplitude
            params0[3*j+1] = rng.uniform(-1, 1)   # center param
            params0[3*j+2] = rng.uniform(-2, 0)   # log width
        
        try:
            res = minimize(obj_and_grad, params0, jac=True, method='L-BFGS-B',
                          options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-6,
                                   'maxfun': maxiter * 20, 'disp': False})
            sc = fast_score(gaussian_mix_params_to_seq(res.x, n))
            if sc < best_score:
                best_score = sc
                best_params = res.x
        except Exception:
            pass
    
    if best_params is not None:
        return gaussian_mix_params_to_seq(best_params, n), best_score
    return None, float('inf')
