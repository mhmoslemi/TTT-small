    """Compute gradient of LSE: grad[i] = 2 * sum_k p[k] * a[k-i]"""
    n = len(a)

    # We need sum_k p[k] * a[k-i] for i = 0, ..., n-1
    # This is like a cross-correlation
    # Using FFT: flip a, convolve with p
    a_flip = a[::-1]
    P = rfft(p, fft_size)
    A = rfft(a_flip, fft_size)
    result = irfft(P * A, fft_size)
    # The correlation at lag i is result[i + n - 1]
    # Let me verify: conv(p, a_flip)[m] = sum_k p[k] * a_flip[m-k] = sum_k p[k] * a[n-1-m+k]
    # We want sum_k p[k] * a[k-i]
    # So we need n-1-m+k = k-i → m = n-1+i
    grad = 2.0 * result[n-1:n-1+n]

from scipy.optimize import linprog

def solve_lp_direct(T, n):
    # Variables: b_0, ..., b_{n-1}, M (total n+1)
    # Objective: minimize M → c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: (T@b)_k - M <= 0 for each k
    # [T_row_k | -1] @ [b; M] <= 0
    A_ub = np.hstack([T.toarray(), -np.ones((T.shape[0], 1))])
    b_ub = np.zeros(T.shape[0])
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: b_i >= 0, M free (but M >= 0 implicitly)
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    return res.x[:n], res.x[-1]

def multiplicative_weights(a0, time_limit):
    """
    Multiplicative weights / mirror descent for minimizing max_k conv(a,a)_k.
    Update: a_new[i] ∝ a[i] * exp(-eta * grad[i])
    where grad[i] = sum_k p_k * 2 * a[k-i] for p_k = softmax(T * autocorr(a))
    """
    n = len(a0)
    a = np.maximum(a0, 0).copy()
    a /= a.sum()
    
    best_a = a.copy()
    best_score = 2 * n * fast_autocorr(a).max()
    
    t0 = time.monotonic()
    T = 50.0  # Temperature for softmax
    
    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_autocorr(a)
        # Softmax weights
        Tc = T * c
        Tc -= Tc.max()
        p = np.exp(Tc)
        p /= p.sum()
        
        # Gradient: grad[i] = 2 * sum_k p[k] * a[k-i]
        # = 2 * conv(p, a_flip)[n-1 + i]
        # Or compute directly
        # grad[i] = 2 * sum_k p[k] * a[k-i] where a[m]=0 for m outside [0,n-1]
        # This is: 2 * (p * a)[i] where * is cross-correlation
        # = 2 * np.correlate(p, a, 'full')[i + n - 1]... 
        # Actually, let me use FFT
        fft_size = 1 << (2 * (2*n-1) - 1).bit_length()
        P = rfft(p, fft_size)
        A = rfft(a, fft_size)
        # We want sum_k p[k] * a[k-i]
        # = (ifft(FFT(p) * FFT(a_flip)))[i + n - 1]
        A_flip = rfft(a[::-1], fft_size)
        corr = irfft(P * A_flip, fft_size)
        grad = 2.0 * corr[n-1:n-1+n]
        
        # Multiplicative weights update
        eta = 0.01  # Step size (tune this)
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 1e-15)  # Ensure positivity
        a /= a.sum()
        
        if it % 100 == 0:
            score = 2 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
    
    return best_a, best_score

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast HiGHS-based LP Nesterov (main workhorse, 50% time)
2. Multiplicative weights / mirror descent (20% time)
3. Top-k LSE with Adam (15% time)
4. Length exploration + interpolation (10% time)
5. Targeted local search for final polish (5% time)
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_autocorr(a):
    """Full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def compute_score(a):
    """Compute 2*n*max(autocorr)/(sum)^2."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-15:
        return float('inf')
    c = fast_autocorr(a)
    return 2.0 * n * c.max() / (s * s)


def solve_lp_response(a_fixed):
    """Solve LP: min max_k conv(a_fixed, b)_k s.t. sum(b)=1, b>=0.
    Returns (b_opt, M_opt)."""
    n = len(a_fixed)
    m = 2 * n - 1
    
    # Build T: T[k, j] = a_fixed[j] if 0 <= k-j < n
    T = np.zeros((m, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed[j]
    
    # LP: min M s.t. T@b - M <= 0, sum(b) = 1, b >= 0
    # Variables: [b_0, ..., b_{n-1}, M]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # Minimize M
    
    # Inequality: T@b - M <= 0
    A_ub = np.empty((m, n + 1))
    A_ub[:, :n] = T
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds
    bounds = [(0, None)] * n + [(-np.inf, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return None, None
    
    b_opt = res.x[:n]
    b_opt = np.maximum(b_opt, 0)
    s = b_opt.sum()
    if s < 1e-15:
        return None, None
    b_opt = b_opt / s
    M_opt = max(0.0, res.x[n])
    
    return b_opt, M_opt


def lp_nesterov(a0, time_limit):
    """LP Nesterov with running average and adaptive restart."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    # Running average
    avg = np.zeros(n)
    count = 0
    WINDOW = 100
    
    # Circular buffer for short average
    short_buf = [np.zeros(n) for _ in range(WINDOW)]
    short_idx = 0
    short_filled = 0
    short_avg = np.zeros(n)
    
    stall = 0
    max_stall = 500
    
    for t in range(1, 1000000):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        
        # Solve LP
        b_star, M_val = solve_lp_response(y)
        if b_star is None:
            continue
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * b_star
        y_new = np.maximum(y_new, 0)
        y_sum = y_new.sum()
        if y_sum < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / y_sum
        
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new
        
        y = y_new
        a = a_new
        
        # Long running average
        avg += a
        count += 1
        
        # Short running average
        if short_filled < WINDOW:
            short_buf[short_idx] = a
            short_avg += a
            short_filled += 1
        else:
            short_avg = short_avg - short_buf[short_idx] + a
        short_idx = (short_idx + 1) % WINDOW
        
        # Check scores periodically
        if t % 10 == 0:
            score_a = compute_score(a)
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
            
            # Check running average
            if count >= 50:
                avg_a = avg / count
                score_avg = compute_score(avg_a)
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_a.copy()
                    stall = 0
            
            if short_filled >= WINDOW:
                short_a = short_avg / WINDOW
                score_s = compute_score(short_a)
                if score_s < best_score:
                    best_score = score_s
                    best_a = short_a.copy()
                    stall = 0
        
        # Adaptive restart
        if stall > max_stall and t > 100:
            y = best_a.copy()
            a = best_a.copy()
            avg = np.zeros(n)
            count = 0
            short_avg = np.zeros(n)
            short_filled = 0
            short_idx = 0
            for i in range(WINDOW):
                short_buf[i] = np.zeros(n)
            stall = 0
            max_stall = max(200, max_stall // 2)
            if t % 2000 == 0:
                print(f"      LP Nesterov restart at t={t}, best={best_score:.6f}")
        
        if t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.6f}")
    
    return best_a, best_score


def mw_optimize(a0, time_limit, eta_init=1.0, T_init=20.0):
    """Multiplicative weights / mirror descent."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    # Precompute FFT size
    size = 2 * n - 1
    fft_size = 1 << (2 * size - 1).bit_length()
    a_flip = a[::-1]
    
    for it in range(1000000):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        
        # Compute autocorrelation
        c = fast_autocorr(a)
        
        # Softmax weights (temperature annealing)
        T_cur = max(5.0, T_init * (0.99 ** it))
        Tc = T_cur * c
        Tc -= Tc.max()
        p = np.exp(Tc)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.ones(2*n-1) / (2*n-1)
        else:
            p = p / p_sum
        
        # Gradient via FFT
        P = rfft(p, fft_size)
        A_flip = rfft(a[::-1], fft_size)
        corr = irfft(P * A_flip, fft_size)
        grad = 2.0 * corr[n-1:n-1+n]
        
        # Adaptive step size
        eta = eta_init / (1.0 + it * eta_init / (n * 0.5))
        # Or: eta = eta_init * n / (n + it)
        
        # Multiplicative weights update
        log_update = -eta * grad
        # Clip for numerical stability
        log_update = np.clip(log_update, -50, 50)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-30)
        a_sum = a.sum()
        if a_sum < 1e-15:
            a = np.ones(n) / n
        else:
            a =
