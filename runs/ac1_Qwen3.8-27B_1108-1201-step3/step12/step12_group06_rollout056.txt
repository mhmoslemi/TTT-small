def fast_convolve(a):
    """Fast linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score assuming sum(a)=1: returns 2*n*max(b)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_from_z(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def build_conv_matrix(a_fixed):
    """Build the convolution matrix T where (T@b)[k] = (a_fixed*b)[k]."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


_lp_solver = None


def get_lp_solver():
    """Find the best available LP solver."""

        return _lp_solver
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CBC]:

            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:

    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Solve the LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS, cp.CBC]:
            if alt == solver:
                continue

                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)

        if b_new is None:
            continue

        # Standard Nesterov step
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()

            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()

        y = y_new

        improved = False
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            improved = True
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            improved = True

        if improved:
            last_improve_iter = it
            stagnant_count = 0
        else:
            stagnant_count += 1

def lse_objective_grad(z_in, T_val, n):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

            return obj

            sc = score_normalized(a_check)

                best_score = sc
                best_a = a_check.copy()

            if stagnant >= 3:
                T = max(T, T * 1.3)
                stagnant = 0
            else:
                T = max(T_min, T * 0.7)

            if sc > best_score * 1.005:
                stagnant += 1
            else:
                stagnant = max(0, stagnant - 1)

            if (stage + 1) % 5 == 0:
                print(f"    LSE stage={stage+1}/{n_stages} T={T:.2f}: score={sc:.8f} best={best_score:.8f}")

        except Exception:
            T = max(T_min, T * 0.8)

    return best_a, best_score


def peak_subgradient(a0, time_limit, n_steps=200000):
    """Peak-targeted subgradient with Polyak step and iterate averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = np.log(np.maximum(a, 1e-300)).copy()

    best_a = a.copy()
    best_score = score_normalized(a)
    lower_bound = 2.0 * n / (2.0 * n - 1.0)
    t0 = time.monotonic()
    avg_a = a.copy()
    avg_count = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b

        threshold = max_b * 0.98
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        if len(active) > 50:
            top_idx = np.argsort(b[active])[-50:]
            active = active[top_idx]

        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        S = np.dot(g, a)
        grad_z = a * (g - S)

        grad_norm = np.linalg.norm(grad_z)
        if grad_norm < 1e-15:
            break

        gap = cur_score - lower_bound
        if gap < 1e-10:
            break
        eta = gap / (2.0 * grad_norm * grad_norm)
        eta = min(eta, 0.5)

        z_new = z - eta * grad_z
        a_new = softmax_from_z(z_new)
        sc_new = score_normalized(a_new)

        if sc_new > cur_score * 1.01:
            lo_e, hi_e = 0.0, eta
            for _ in range(5):
                mid_e = (lo_e + hi_e) / 2.0
                z_mid = z - mid_e * grad_z
                a_mid = softmax_from_z(z_mid)
                if score_normalized(a_mid) < cur_score:
                    lo_e = mid_e
                else:
                    hi_e = mid_e
            eta = lo_e
            z_new = z - eta * grad_z
            a_new = softmax_from_z(z_new)
            sc_new = score_normalized(a_new)

        z = z_new
        a = a_new

        avg_a = (avg_count * avg_a + a) / (avg_count + 1)
        avg_count += 1
        if avg_count > 5000:
            avg_count = 5000
            avg_a = (avg_a * 5000 + a) / 5001

        if sc_new < best_score:
            best_score = sc_new
            best_a = a.copy()

        sc_avg = score_normalized(avg_a)
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = avg_a.copy()

        if (step + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakSG step={step+1} ({elapsed:.0f}s): cur={cur_score:.8f} best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initial sequences."""
    choice = rng.integers(0, 10)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-2 * x ** 2)
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif choice == 4:
        a = np.ones(n) * (1.0 + 0.2 * np.cos(2 * np.pi * 3 * np.arange(n) / n))
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(3 * x) + 0.15 * np.cos(7 * x)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * np.cos(np.pi * x / 3) ** 2
    elif choice == 7:
        # Asymmetric: concentrated in first 2/3
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x ** 2) + 0.1
    elif choice == 8:
        # Frequency-domain design: flat spectrum approximation
        nfft = 1 << (2 * n).bit_length()
        phases = rng.uniform(0, 2 * np.pi, nfft // 2 + 1)
        F = np.zeros(nfft // 2 + 1, dtype=complex)
        F[1:nfft // 2] = np.exp(1j * phases[1:nfft // 2])
        # Make it Hermitian
        F_full = np.zeros(nfft, dtype=complex)
        F_full[:nfft // 2 + 1] = F
        if nfft % 2 == 0:
            F_full[nfft // 2 + 1:] = np.conj(F_full[nfft // 2 - 1:0:-1])
        else:
            F_full[nfft // 2 + 1:] = np.conj(F_full[nfft // 2 - 1:0:-1])
        a = np.real(irfft(F_full, nfft)[:n])
    else:
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.5) ** 2) / 0.05) + 0.3 * np.exp(-((x - 0.2) ** 2) / 0.02) + 0.3 * np.exp(-((x - 0.8) ** 2) / 0.02)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def spectral_init(n, rng):
    """Initialize using spectral flatness principle."""
    # Try to create a sequence whose autocorrelation is as flat as possible
    # Start with random phases in frequency domain
    for attempt in range(20):
        nfft = 1 << (2 * n).bit_length()
        # Random spectral phase
        freq_idx = np.arange(nfft // 2 + 1)
        # Amplitude: try different profiles
        profile = rng.choice(['flat', 'tapered', 'raised'])
        if profile == 'flat':
            mag = np.ones(len(freq_idx))
        elif profile == 'tapered':
            mag = np.exp(-0.3 * (freq_idx / nfft) ** 2)
        else:
            mag = 0.5 + 0.5 * np.cos(np.pi * freq_idx / (nfft // 2))
        
        phase = rng.uniform(0, 2 * np.pi, len(freq_idx))
        F = mag * np.exp(1j * phase)
        
        # Construct full spectrum
        F_full = np.zeros(nfft, dtype=complex)
        F_full[:nfft // 2 + 1] = F
        if nfft % 2 == 0:
            F_full[nfft // 2 + 1:] = np.conj(F_full[nfft // 2 - 1:0:-1])
        else:
            F_full[nfft // 2 + 1:] = np.conj(F_full[nfft // 2 - 1:0:-1])
        
        a = np.real(irfft(F_full, nfft)[:n])
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            a = a / s
            sc = score_normalized(a)
            if sc < 1.6:  # Good enough to use
                return a
    # Fallback
    return structured_init(n, rng)


def multi_restart_nesterov(a0, time_limit, solver, n_restarts=3, max_iters=None):
    """Run Nesterov with multiple restarts from perturbed versions."""
    if max_iters is None:
        max_iters = 300000
    n = len(a0)
    best_a = a0.copy()
    best_score = score_normalized(a0)
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    time_per_restart = time_limit / n_restarts

    for restart in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break

        if restart == 0:
            current = a0.copy()
        else:
            # Perturb the best found so far
            noise_scale = 0.05 / (restart + 1)
            noise = rng.normal(0, noise_scale, n)
            current = best_a + noise
            current = np.maximum(current, 0)
            s = current.sum()
            if s > 1e-15:
                current = current / s
            else:
                current = best_a.copy()

        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        a_out, sc_out = nesterov_lp_improved(
            current, remaining * 0.95, solver=solver,
            max_iters=max_iters // n_restarts
        )
        if sc_out < best_score:
            best_score = sc_out
            best_a = a_out.copy()
            print(f"    MultiRestart: restart={restart+1}/{n_restarts} improved to {best_score:.8f}")
        else:
            print(f"    MultiRestart: restart={restart+1}/{n_restarts} no improvement (best={best_score:.8f}, got={sc_out:.8f})")

    return best_a, best_score


def final_propose_candidate(seed=42, budget_s=1800.0):
    """Main search function with improved multi-phase approach."""
    rng = np.random.default_rng(seed)
    global_deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Total budget: {budget_s}s, effective: {global_deadline - time.monotonic():.0f}s")

    # Initialize from parent or structured
    best_seq = None
    best_score = float('inf')
    n_base = 1200

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            remain = global_deadline - time.monotonic()
            print(f"  ** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, remain={remain:.0f}s)")

    # === Phase 1: Main Nesterov with multi-restart at base length (45% time) ===
    print(f"\n=== Phase 1: Multi-restart Nesterov at n={n_base} (score={best_score:.8f}) ===")
    t1 = 0.45 * (budget_s - 25)
    a_out, _ = multi_restart_nesterov(best_seq, t1, solver, n_restarts=3)
    try_update(a_out, "nesterov_multirestart")
    print(f"  Phase 1 done: best_score={best_score:.8f}")

    # === Phase 2: LSE polish (15% time) ===
    print(f"\n=== Phase 2: LSE polish (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t2 = min(0.15 * (budget_s - 25), remaining * 0.5)
    if t2 > 30:
        a_lse1, _ = lsbfgs_lse_adaptive(best_seq, t2 * 0.6, T_start=80.0, T_min=0.15, n_stages=20, iters_per_stage=35)
        try_update(a_lse1, "lse_pass1")

        remaining = global_deadline - time.monotonic()
        if remaining > 20:
            a_lse2, _ = lsbfgs_lse_adaptive(best_seq, remaining * 0.5, T_start=20.0, T_min=0.1, n_stages=15, iters_per_stage=30)
            try_update(a_lse2, "lse_pass2")

        remaining = global_deadline - time.monotonic()
        if remaining > 20:
            # LSE from perturbed start
            perturbed = best_seq + 0.03 * rng.normal(0, 1, len(best_seq))
            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()
            a_lse3, _ = lsbfgs_lse_adaptive(perturbed, remaining * 0.5, T_start=50.0, T_min=0.2, n_stages=15, iters_per_stage=30)
            try_update(a_lse3, "lse_perturbed")
    print(f"  Phase 2 done: best_score={best_score:.8f}")

    # === Phase 3: Multi-length exploration (25% time) ===
    print(f"\n=== Phase 3: Multi-length exploration (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t3_total = min(0.25 * (budget_s - 25), remaining * 0.6)
    if t3_total > 50:
        # Try longer and shorter lengths
        lengths_to_try = sorted(set([
            n_base + 200, n_base + 400, n_base + 600,
            max(400, n_base - 200)
        ]))
        t_per_len = t3_total / len(lengths_to_try)

        for n_try in lengths_to_try:
            remaining = global_deadline - time.monotonic()
            if remaining < 30:
                break
            t5 = min(t_per_len, remaining * 0.45)
            if t5 < 20:
                break
            print(f"  Trying n={n_try}, budget={t5:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            # Nesterov at new length
            a_try, _ = nesterov_lp_improved(init_try, t5 * 0.65, solver=solver, max_iters=100000)
            try_update(a_try, f"len_{n_try}_nesterov")

            remaining = global_deadline - time.monotonic()
            if remaining > 15:
                a_try2, _ = lsbfgs_lse_adaptive(a_try, remaining * 0.5, T_start=40.0, T_min=0.3, n_stages=10, iters_per_stage=25)
                try_update(a_try2, f"len_{n_try}_lse")
    print(f"  Phase 3 done: best_score={best_score:.8f}")

    # === Phase 4: Peak subgradient (10% time) ===
    print(f"\n=== Phase 4: Peak Subgradient (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t4 = min(0.10 * (budget_s - 25), remaining * 0.5)
    if t4 > 20:
        a_sg, _ = peak_subgradient(best_seq, t4, n_steps=300000)
        try_update(a_sg, "peak_sg")
    print(f"  Phase 4 done: best_score={best_score:.8f}")

    # === Phase 5: Final combined polish (5% time) ===
    print(f"\n=== Phase 5: Final polish (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    if remaining > 20:
        a_f1, _ = nesterov_lp_improved(best_seq, remaining * 0.4, solver=solver, max_iters=50000, restart_every=200, restart_strength=0.05)
        try_update(a_f1, "final_nesterov")

        remaining = global_deadline - time.monotonic()
        if remaining > 15:
            a_f2, _ = lsbfgs_lse_adaptive(best_seq, remaining * 0.5, T_start=15.0, T_min=0.05, n_stages=10, iters_per_stage=25)
            try_update(a_f2, "final_lse_lowT")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq) if best_seq is not None else 0})")
    print(f"Time used: {time.monotonic() - (global_deadline - (budget_s - 25)):.0f}s")
    return [float(x) for x in best_seq]


def propose_candidate(seed=42, budget_s=1800.0):
    """Entry point for the search."""
    return final_propose_candidate(seed=seed, budget_s=budget_s)
