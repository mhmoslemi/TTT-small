import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time

def fast_autocorr(a):
    """Full self-convolution a*a using FFT, O(n log n)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def eval_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2 efficiently."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)

def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where (T@b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS

def solve_lp_best_response(a_fixed, solver):
    """
    Solve LP: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0.
    """
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, None
    b_val = b_val / s
    M_val = float(M_var.value)
    # Get dual variables for active constraints
    duals = None
    try:
        duals = np.asarray(T.dual_value, dtype=np.float64)
    except Exception:
        pass
    return b_val, M_val, duals

def nesterov_lp(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated LP best-response."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    
    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break
        b_star, M_val, _ = solve_lp_best_response(y, solver)
        if b_star is None:
            continue
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        ss = y_new.sum()
        if ss < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / ss
        
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            ss = a_new.sum()
            if ss < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / ss
        
        y = y_new
        a = a_new
        
        sc = eval_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        sc_y = eval_score(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP t={t} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def pair_wise_descent(a0, time_limit, top_k=5, max_rounds=10000, verbose=True):
    """
    Aggressive pair-wise coordinate descent with line search.
    For each pair (i,j), find optimal mass transfer.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    
    rng = np.random.default_rng(42)
    
    # Precompute which positions contribute to which peaks
    # For position i, it contributes to peaks k where 0 <= k-i < n, i.e., i <= k <= i+n-1
    
    round_num = 0
    while time.monotonic() - t0 < time_limit and round_num < max_rounds:
        round_num += 1
        b = fast_autocorr(a)
        # Find top peaks
        peak_indices = np.argsort(b)[::-1][:top_k]
        improved = False
        
        # For each peak, try to reduce it by moving mass
        for pk in peak_indices:
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[pk - idx]
            
            # Find the largest contributor
            max_c_idx = np.argmax(contrib)
            src = idx[max_c_idx]
            src_val = a[src]
            if src_val < 1e-10:
                continue
            
            # Find target: position that contributes least to top peaks
            # and is not src
            target_scores = []
            for j in range(n):
                if j == src or a[j] == 0:
                    continue
                # Compute contribution of j to all top peaks
                ts = 0.0
                for pk2 in peak_indices:
                    if max(0, pk2 - n + 1) <= j <= min(n - 1, pk2):
                        partner = pk2 - j
                        if 0 <= partner < n:
                            ts += a[j] * a[partner]
                target_scores.append((ts, j))
            
            if not target_scores:
                continue
            
            target_scores.sort()
            # Try top 5 targets
            for ts, dst in target_scores[:5]:
                # Line search: find optimal amount to transfer from src to dst
                # a[src] - t, a[dst] + t
                # We want to minimize max(conv)
                best_t = 0
                best_local_score = best_score
                # Try a few candidate transfer amounts
                for frac in [0.001, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
                    t_amt = src_val * frac
                    if t_amt <= 0:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= t_amt
                    a_trial[dst] += t_amt
                    sc = eval_score(a_trial)
                    if sc < best_local_score:
                        best_local_score = sc
                        best_t = t_amt
                
                if best_t > 0 and best_local_score < best_score - 1e-15:
                    a[src] -= best_t
                    a[dst] += best_t
                    best_score = best_local_score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try random moves
            for _ in range(10):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-10:
                    continue
                for frac in [0.001, 0.005, 0.01, 0.02]:
                    t_amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= t_amt
                    a_trial[j] += t_amt
                    sc = eval_score(a_trial)
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if verbose and round_num % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PWD round={round_num} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def logsumexp_optimize(a0, time_limit, T_start=100.0, T_end=0.01, n_stages=15, verbose=True):
    """LogSumExp-smoothed optimization with L-BFGS."""
    from scipy.optimize import minimize
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = eval_score(a_init)
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        iters_per_stage = max(5, min(40, int(10 + 30 * (si / max(1, n_stages - 1)))))
        
        def func(z_in):
            a = _softmax(z_in)
            b = fast_autocorr(a)
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            return (max_Tb + np.log(exp_Tb.sum())) / stage_T
        
        def jac(z_in):
            a = _softmax(z_in)
            b = fast_autocorr(a)
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            # grad_a[i] = 2 * sum_k w[k] * a[k-i] (for valid k-i)
            # This is 2 * (w * a_rev)[n-1 : 2n-2]
            a_rev = a[::-1]
            conv_len = (2*n-1) + n - 1
            fft_size = 1 << (conv_len - 1).bit_length()
            W = rfft(w, fft_size)
            A_rev = rfft(a_rev, fft_size)
            conv = irfft(W * A_rev, fft_size)
            grad_a = 2.0 * conv[n - 1:n - 1 + n]
            # Transform to z-space
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)
            return grad_z
        
        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                                    'gtol': 1e-14, 'maxfun': iters_per_stage * 200})
            z = res.x
            a = _softmax(z)
            sc = eval_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        except Exception:
            pass
        
        if verbose and si % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage={si}/T={stage_T:.3f} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def line_search_pair(a, i, j, n_trials=20):

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_autocorr(a):
    """Full self-convolution a*a using FFT, O(n log n)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where (T@b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s

def _softmax(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def _lse_func(z, T, n):
    """LogSumExp objective in z-space."""
    a = _softmax(z)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    return (max_Tb + np.log(exp_Tb.sum())) / T


def _lse_jac(z, T, n):
    """Gradient of LogSumExp objective in z-space."""
    a = _softmax(z)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    a_rev = a[::-1]
    conv_len = (2 * n - 1) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    A_rev = rfft(a_rev, fft_size)
    conv = irfft(W * A_rev, fft_size)
    grad_a = 2.0 * conv[n - 1:n - 1 + n]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return grad_z


def logsumexp_optimize(a0, time_limit, T_start=80.0, T_end=0.005, n_stages=15, verbose=True):
    """LogSumExp-smoothed optimization with L-BFGS and multi-stage annealing."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = eval_score(a_init)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        iters_per_stage = max(5, min(50, int(10 + 40 * (si / max(1, n_stages - 1)))))

        def func(z_in):
            return _lse_func(z_in, stage_T, n)

        def jac(z_in):
            return _lse_jac(z_in, stage_T, n)

        try:

    best_score = eval_score(a)
    t0 = time.monotonic()

    it = 0
    stall = 0
    n_top_peaks = 5

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_autocorr(a)
        peak_idx = np.argsort(b)[::-1][:n_top_peaks]
        max_b = b[peak_idx[0]]

        improved = False

        # For each top peak, try to reduce it
        for pk in peak_idx:
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[pk - idx]
            order = np.argsort(contrib)[::-1]

            for oi in order[:4]:
                src = idx[oi]
                src_val = a[src]
                if src_val < 1e-12:
                    continue

                # Find good destinations: positions with low contribution to top peaks
                candidates = []
                n_check = min(n, 200)
                check_idx = rng.choice(n, size=n_check, replace=False)
                for j in check_idx:
                    if j == src:
                        continue
                    # Compute contribution to top peaks
                    total_c = 0.0
                    for pk2 in peak_idx[:3]:
                        if max(0, pk2 - n + 1) <= j <= min(n - 1, pk2):
                            partner = pk2 - j
                            if 0 <= partner < n:
                                total_c += a[j] * a[partner]
                    if total_c < 0.2 * max_b:
                        candidates.append((total_c, j))

                if not candidates:
                    continue
                candidates.sort()

                for _, dst in candidates[:5]:
                    # Try several transfer amounts with exact evaluation
                    best_t = 0
                    best_sc = best_score
                    for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.05, 0.08]:
                        t_amt = src_val * frac
                        a_trial = a.copy()
                        a_trial[src] -= t_amt
                        a_trial[dst] += t_amt
                        sc = 2.0 * n * fast_autocorr(a_trial).max()
                        if sc < best_sc:
                            best_sc = sc
                            best_t = t_amt

                    if best_t > 0 and best_sc < best_score - 1e-14:
                        a[src] -= best_t
                        a[dst] += best_t
                        best_score = best_sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Simulated annealing style random moves
            temp = max(0.0005, 0.05 * np.exp(-stall / 200.0))
            for _ in range(15):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.001, 0.02)
                t_amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= t_amt
                a_trial[j] += t_amt
                sc = 2.0 * n * fast_autocorr(a_trial).max()
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    stall = 0
                    break
                elif sc < best_score * (1 + 2 * temp):
                    # Accept with probability based on "temperature"
                    if rng.random() < np.exp(-(sc - best_score) / (max_b * temp + 1e-30)):
                        a = a_trial
                        break

        if stall > 2000:
            break

        if verbose and it % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PFLS it={it} stall={stall} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def multi_peak_lp_shave(a0, time_limit, solver, verbose=True):
    """LP-based simultaneous multi-peak reduction."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()

    K = min(5, n)
    step_scales = [0.1, 0.05, 0.02, 0.01, 0.005]

    for it in range(300):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:K]
        scale = step_scales[min(it // 15, len(step_scales) - 1)]

        delta_var = cp.Variable(n)
        t_var = cp.Variable()
        a_rev = a[::-1]
        T_rev = build_conv_matrix(a_rev)

        constraints = [
            cp.sum(delta_var) == 0,
            delta_var >= -a
        ]
        for k in top_k:
            constraints.append(b[k] + 2.0 * scale * (T_rev @ delta_var)[k] <= t_var)

        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=solver)
        except Exception:
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        delta_val = np.asarray(delta_var.value, dtype=np.float64)
        a_trial = a + delta_val
        a_trial = np.maximum(a_trial, 0)
        ss = a_trial.sum()
        if ss < 1e-15:
            continue
        a_trial = a_trial / ss
        sc = eval_score(a_trial)

        if sc < best_score - 1e-14:
            a = a_trial
            best_score = sc
            best_a = a.copy()

        if verbose and it % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MPLP it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        a = np.exp(-3 * (x - 0.5) ** 2)
    elif choice == 2:
        a = (1 - x) ** 2
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.1 * np.cos(8 * np.pi * x)
        a = np.maximum(a, 0.01)
    elif choice == 4:
        a = np.exp(-4 * (x - 0.3) ** 2) + 0.6 * np.exp(-4 * (x - 0.7) ** 2)
    else:
        a = np.maximum(0, 1 - 2 * np.abs(x - 0.5)) + 0.1 * np.sin(6 * np.pi * x)
        a = np.maximum(a, 0.01)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate():
    """
    Multi-phase optimization to minimize 2*n*max(conv(a,a))/sum(a)^2.
    Combines LP best-response, LogSumExp smoothing, peak-focused local search,
    and multi-length exploration.
    """
    t_global = time.monotonic()
    time_budget = 1740.0

    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    # Load parent sequence
    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_score(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        print("No valid parent, using default initialization")

    rng_main = np.random.default_rng(42)

    if best_seq is None:
        n0 = 1600
        best_seq = structured_init(n0, rng_main)
        best_score = eval_score(best_seq)
        print(f"Default init: n={n0}, score={best_score:.8f}")

    # Track best across all phases
    results = []  # list of (score, sequence)

    def record_result(a, label):
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_score(a)
        if np.isfinite(sc) and sc < best_score:
            nonlocal_score = sc
            results.append((sc, a.copy(), label))
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Since we can't use closures with nonlocal, we'll track results differently
    # Use a mutable container
    state = {'best_seq': best_seq.copy(), 'best_score': best_score}

    def try_update(a, label):
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_score(a)
        if np.isfinite(sc) and sc < state['best_score']:
            state['best_score'] = sc
            state['best_seq'] = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    n_base = len(state['best_seq'])

    # === Phase 1: Nesterov LP at parent length ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Nesterov LP n={n_base} (rem={remaining:.0f}s) ===")
    t1 = min(remaining * 0.22, 320.0)
    if t1 > 20:
        a_out, _ = nesterov_lp(state['best_seq'], t1, solver)
        try_update(a_out, f"nlp_n{n_base}")

    # === Phase 2: LogSumExp optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LogSumExp n={n_base} (rem={remaining:.0f}s) ===")
    t2 = min(remaining * 0.18, 250.0)
    if t2 > 15:
        a_out, _ = logsumexp_optimize(state['best_seq'], t2, T_start=60.0, T_end=0.005, n_stages=14)
        try_update(a_out, f"lse_n{n_base}")

    # === Phase 3: Peak-focused local search ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Peak-focused LS n={n_base} (rem={remaining:.0f}s) ===")
    t3 = min(remaining * 0.18, 280.0)
    if t3 > 15:
        a_out, _ = peak_focused_local_search(state['best_seq'], t3)
        try_update(a_out, f"pfls_n{n_base}")

    # === Phase 4: Multi-peak LP shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Multi-peak LP n={n_base} (rem={remaining:.0f}s) ===")
    t4 = min(remaining * 0.10, 100.0)
    if t4 > 10:
        a_out, _ = multi_peak_lp_shave(state['best_seq'], t4, solver)
        try_update(a_out, f"mplp_n{n_base}")

    # === Phase 5: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Multi-length exploration (rem={remaining:.0f}s) ===")

    lengths_to_try = []
    for offset in [-200, -100, 100, 200, 400, 600]:
        nl = n_base + offset
        if 900 <= nl <= 3200:
            lengths_to_try.append(nl)
    for nl in [1200, 1400, 1800, 2200, 2500]:
        if 900 <= nl <= 3200 and nl not in lengths_to_try:
            lengths_to_try.append(nl)

    lengths_to_try = sorted(set(lengths_to_try), key=lambda x: abs(x - n_base))
    time_per_len = min(remaining * 0.55 / max(1, len(lengths_to_try)), 180.0)

    for nl in lengths_to_try:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 25:
            break
        tl = min(time_per_len, remaining * 0.25)
        if tl < 12:
            continue

        print(f"  Trying n={nl} (time={tl:.0f}s)")
        init = interpolate_sequence(state['best_seq'], nl)

        # Nesterov LP (shorter)
        t_sub1 = tl * 0.45
        a_out, _ = nesterov_lp(init, t_sub1, solver, verbose=False)

        # Quick local search
        rem2 = time_budget - (time.monotonic() - t_global)
        t_sub2 = min(tl * 0.35, rem2 * 0.1)
        if t_sub2 > 5:
            a_out2, _ = peak_focused_local_search(a_out, t_sub2, verbose=False)
            try_update(a_out2, f"pfls_n{nl}")

        try_update(a_out, f"nlp_n{nl}")

    # === Phase 6: Final polishing ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(state['best_seq'])
    print(f"\n=== Phase 6: Final polish n={n_best} (rem={remaining:.0f}s) ===")

    # Peak-focused local search (longer)
    t6a = min(remaining * 0.28, 200.0)
    if t6a > 10:
        a_out, _ = peak_focused_local_search(state['best_seq'], t6a)
        try_update(a_out, "final_pfls")

    # LogSumExp with tight params
    remaining = time_budget - (time.monotonic() - t_global)
    t6b = min(remaining * 0.22, 120.0)
    if t6b > 10:
        a_out, _ = logsumexp_optimize(state['best_seq'], t6b, T_start=15.0, T_end=0.002, n_stages=10)
        try_update(a_out, "final_lse")

    # Another round of peak-focused
    remaining = time_budget - (time.monotonic() - t_global)
    t6c = min(remaining * 0.2, 100.0)
    if t6c > 10:
        a_out, _ = peak_focused_local_search(state['best_seq'], t6c)
        try_update(a_out, "final_pfls2")

    # Multi-peak LP
    remaining = time_budget - (time.monotonic() - t_global)
    t6d = min(remaining * 0.15, 60.0)
    if t6d > 8:
        a_out, _ = multi_peak_lp_shave(state['best_seq'], t6d, solver)
        try_update(a_out, "final_mplp")

    # === Phase 7: Ultra-fine local search ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Ultra-fine polish (rem={remaining:.0f}s) ===")
    if remaining > 10:
        a_out, _ = peak_focused_local_search(state['best_seq'], remaining * 0.7, verbose=False)
        try_update(a_out, "ultra_fine")

    # Final verification
    final_seq = state['best_seq']
    final_score = state['best_score']
    fs = eval_score(final_seq)
    if np.isfinite(fs):
        final_score = fs
    print(f"\nFINAL SCORE: {final_score:.8f} (n={len(final_seq)})")

    result = [float(max(0.0, min(1000.0, x))) for x in final_seq]
    return result
