So in all cases: ∂b[k]/∂a[m] = 2*a[k-m] if 0 <= k-m < n, else 0.

Great! So the gradient of the peak b[k*] w.r.t. a is simply 2*a[k*-m] (reversed and shifted).

This means: to reduce the peak at position k*, we should decrease a at positions m where a[k*-m] is large, and increase a at positions where a[k*-m] is small.

This gives us a natural "peak reduction" direction:
- Compute g[m] = 2*a[k*-m] (the gradient of the peak)
- Move mass from high-g positions to low-g positions
- Project back to the simplex

This is essentially a subgradient step on the peak!

Now, the full gradient (for LogSumExp or max) would be a weighted average of these gradients over the active peaks.

import numpy as np

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def convolve_and_peaks(a, top_k=10):
    """Return convolution and indices of top peaks."""
    b = fast_convolve(a)
    n_peaks = min(top_k, len(b))
    idx = np.argpartition(b, -n_peaks)[-n_peaks:]
    idx = idx[np.argsort(b[idx])[::-1]]
    return b, idx


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0 or a_list[0] < 0.01:
            # Check sum
            if sum(a_list) < 0.01:
                return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


_lp_solver = None

def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CBC]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS, cp.CBC]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def peak_subgradient_v2(a0, time_limit, n_steps=300000, n_peaks=20):
    """
    Improved peak-targeted subgradient with:
    - Proper gradient computation using the formula ∂b[k]/∂a[m] = 2*a[k-m]
    - Polyak step size with gap tracking
    - Exponential moving average of iterates
    - Adaptive peak selection
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    # Exponential moving average
    avg_a = a.copy()
    avg_weight = 0.999
    
    # Track best for Polyak
    running_best = best_score
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b
        
        # Select active peaks (top n_peaks or those within 95% of max)
        threshold = max_b * 0.95
        active = np.where(b >= threshold)[0]
        if len(active) > n_peaks:
            top_idx = np.argsort(b[active])[::-1][:n_peaks]
            active = active[top_idx]
        
        # Compute gradient: for each active peak k, grad[m] += 2*a[k-m]
        g = np.zeros(n)
        for k in active:
            # ∂b[k]/∂a[m] = 2*a[k-m] if 0 <= k-m < n
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # m ranges from lo to hi, and k-m ranges from k-hi to k-lo
            m_vals = np.arange(lo, hi + 1)
            km_vals = k - m_vals
            g[m_vals] += 2.0 * a[km_vals]
        g /= len(active)
        
        # Project to tangent space of simplex
        S = np.dot(g, a)
        grad_z = a * (g - S)
        
        grad_norm = np.linalg.norm(grad_z)
        if grad_norm < 1e-15:
            break
        
        # Polyak step size
        # Lower bound: for non-negative a, by the structure, the minimum of 2n*max(b) 
        # is at least 2 (achieved by uniform). Use a tighter bound.
        lower_bound = 2.0 * n / (2.0 * n - 1.0)  # Loose lower bound
        # Actually the true lower bound is 2 (for uniform), but we can use the
        # value from the LP as a better lower bound
        gap = cur_score - lower_bound
        if gap < 1e-10:
            break
        eta = gap / (2.0 * grad_norm * grad_norm)
        eta = min(eta, 1.0)
        
        z_new = z - eta * grad_z
        a_new = softmax_from_z(z_new)
        sc_new = score_normalized(a_new)
        
        # Line search if needed
        if sc_new > cur_score * 1.005:
            lo_e, hi_e = 0.0, eta
            for _ in range(8):
                mid_e = (lo_e + hi_e) / 2.0
                z_mid = z - mid_e * grad_z
                a_mid = softmax_from_z(z_mid)
                if score_normalized(a_mid) < cur_score:
                    lo_e = mid_e
                else:
                    hi_e = mid_e
            eta = lo_e
            if eta < 1e-10:
                break
            z_new = z - eta * grad_z
            a_new = softmax_from_z(z_new)
            sc_new = score_normalized(a_new)
        
        z = z_new
        a = a_new
        
        # Update EMA
        avg_a = avg_weight * avg_a + (1 - avg_weight) * a
        
        if sc_new < best_score:
            best_score = sc_new
            best_a = a.copy()
            running_best = best_score
        
        sc_avg = score_normalized(avg_a)
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = avg_a.copy()
            running_best = best_score
        
        if (step + 1) % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakSG2 step={step+1} ({elapsed:.0f}s): cur={cur_score:.8f} best={best_score:.8f} eta={eta:.6f}")

    return best_a, best_score


def nesterov_lp_v2(a0, time_limit, solver=None, max_iters=500000, momentum_boost=1.0):
    """Nesterov-accelerated LP with adaptive momentum."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0
    stagnant_count = 0
    prev_best = best_score

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0) * momentum_boost
        alpha = min(alpha, 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            stagnant_count = 0
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            stagnant_count = 0
        
        if it % 1000 == 0:
            if best_score == prev_best:
                stagnant_count += 1
                if stagnant_count > 5:
                    # Restart with perturbation
                    rng_local = np.random.default_rng(it)
                    a = a + 0.01 * rng_local.normal(0, 1, n)
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    y = a.copy()
                    stagnant_count = 0
            prev_best = best_score

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 10:
                print(f"    NesterovLP2 iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} M_lp={M_lp:.6f}")
                last_print = elapsed

    return best_a, best_score


def lse_lsbfgs_v2(a0, time_limit, T_start=50.0, T_min=0.1, n_stages=30, iters_per_stage=30):
    """Adaptive LogSumExp L-BFGS with improved schedule."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    T = T_start
    prev_score = best_score

    def lse_obj_grad(z_val, T_val):
        a_val = softmax_from_z(z_val)
        b = fast_convolve(a_val)
        Tb = T_val * b
        max_Tb = Tb.max()
        if max_Tb > 700:
            max_Tb = 700
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        log_sum_exp = max_Tb + np.log(exp_Tb.sum())
        obj = (2.0 * n / T_val) * log_sum_exp

        # Gradient w.r.t. a
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-18:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a_val[k - idx]

        S = np.dot(grad_a, a_val)
        grad_z = a_val * (grad_a - S)
        return obj, grad_z

    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func_staged(z_val):
            obj, _ = lse_obj_grad(z_val, T)
            return obj

        def jac_staged(z_val):
            _, grad = lse_obj_grad(z_val, T)
            return grad

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-16, 'maxfun': iters_per_stage * 200,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)

            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()

            # Adaptive temperature
            improvement = prev_score - sc
            if improvement > 1e-6:
                T = max(T_min, T * 0.7)
            else:
                T = min(T_start, T * 1.3)
            prev_score = sc

            if (stage + 1) % 5 == 0:
                print(f"    LSE2 stage={stage+1}/{n_stages} T={T:.2f}: score={sc:.8f} best={best_score:.8f}")

        except Exception:
            T = max(T_min, T * 0.8)

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, n_swaps=50000):
    """
    Coordinate-wise local search: try moving mass between pairs of positions.
    Focus on moves that reduce the peak convolution value.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    for step in range(n_swaps):
        if time.monotonic() - t0 > time_limit:
            break

        # Pick two random positions
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # Try moving mass from i to j and vice versa
        # Parameterize: new_a[i] = a[i] - delta, new_a[j] = a[j] + delta
        # Find optimal delta by trying a few values
        best_delta = 0.0
        best_local_score = best_score

        # Try a range of deltas
        max_delta = min(a[i], 0.1 / n)  # Don't move too much
        for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
            delta = frac * max_delta
            a_try = a.copy()
            a_try[i] -= delta
            a_try[j] += delta
            if a_try[i] < 0 or a_try[j] < 0:
                continue
            sc = score_normalized(a_try)
            if sc < best_local_score:
                best_local_score = sc
                best_delta = delta

        # Also try reverse direction
        max_delta_r = min(a[j], 0.1 / n)
        for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
            delta = frac * max_delta_r
            a_try = a.copy()
            a_try[j] -= delta
            a_try[i] += delta
            if a_try[j] < 0 or a_try[i] < 0:
                continue
            sc = score_normalized(a_try)
            if sc < best_local_score:
                best_local_score = sc
                best_delta = -delta

        if best_delta != 0:
            a[i] -= best_delta if best_delta > 0 else -best_delta
            a[j] += best_delta if best_delta > 0 else -best_delta
            a = np.maximum(a, 0)
            s = a.sum()
            if s < 1e-15:
                break
            a = a / s

            if best_local_score < best_score:
                best_score = best_local_score
                best_a = a.copy()

        if (step + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc step={step+1} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 10)

        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-3 * x ** 2)
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)

        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(3 * x) + 0.15 * np.cos(7 * x)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * np.cos(np.pi * x / 3) ** 2
    elif choice == 7:

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def direct_peak_optimization(a0, time_limit, n_iters=100000):
    """
    Direct optimization that explicitly targets the peak positions.
    At each step, identify the most contributing terms to the peak and
    adjust those specific a[j] values.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    z = z_from_a(a)
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        k_star = int(np.argmax(b))
        cur_score = 2.0 * n * max_b
        
        # The peak is b[k_star] = sum_j a[j] * a[k_star - j]
        # The "gradient" direction to reduce this is: decrease a[j] where a[k_star-j] is large
        # and increase a[j] where a[k_star-j] is small
        
        # Compute contribution of each j to the peak
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        m_vals = np.arange(lo, hi + 1)
        contributions = a[m_vals] * a[k_star - m_vals]
        
        # Direction: move mass from high-contribution j to low-contribution j
        # (within the range that affects the peak)
        g = np.zeros(n)
        # For positions in range: gradient is proportional to a[k_star - j]
        g[m_vals] = a[k_star - m_vals]
        # For positions outside range: they don't contribute to this peak,
        # so increasing them is "free" (from this peak's perspective)
        outside = np.ones(n, dtype=bool)
        outside[m_vals] = False
        g[outside] = 0.0  # No gradient from this peak
        
        # Normalize gradient
        g = g - np.dot(g, a) / a.sum()  # Center
        grad_z = a * (g / g.max() - np.dot(g, a) / (g.max() * a.sum()))
        
        # Actually, let's use a simpler approach:
        # Move a small amount of mass from j with highest a[j]*a[k_star-j] 
        # to j with lowest (or to outside positions)
        
        # Find the top contributor
        contrib_sorted = np.argsort(contributions)[::-1]
        top_j = m_vals[contrib_sorted[0]]
        
        # Find the best target: either a low contributor or an outside position
        # For outside positions, increasing them doesn't affect this peak
        outside_positions = np.where(outside)[0]
        if len(outside_positions) > 0:
            # Pick a random outside position
            target_j = outside_positions[np.random.default_rng(it).integers(0, len(outside_positions))]
        else:
            # Pick the bottom contributor
            bottom_j = m_vals[contrib_sorted[-1]]
            target_j = bottom_j
        
        # Move a small amount
        move_amount = a[top_j] * 0.01
        a[top_j] -= move_amount
        a[target_j] += move_amount
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        
        sc = score_normalized(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if (it + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    DirectPeak it={it+1} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function."""
    rng = np.random.default_rng(seed)
    global_deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Total budget: {budget_s}s")

    best_seq = None
    best_score = float('inf')
    n_base = 1200

    # Try to use parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
