Actually, for the full convolution (not just the "valid" part), the situation is different. The full convolution of a sequence with itself has a triangular envelope if the sequence is constant. To get a flat autocorrelation, we need a sequence whose power spectrum has special properties.

This is related to the " Barker sequence" (a ±1 sequence with low autocorrelation sidelobes). For non-negative sequences, the analogous concept is a "low-peak-factor" sequence.

**Idea: Use a proxy objective**

Instead of directly minimizing max_k c_k, we could minimize a smooth proxy like:
- (1/T) * log(sum_k exp(T * c_k)) (log-sum-exp, approximates max as T→∞)
- sum_k c_k^p for large p (L_p norm, approximates max as p→∞)
- sum_k max(0, c_k - tau)^2 (hinge loss above a threshold tau)

The log-sum-exp approach is already used in the previous code (top-k LSE). The L_p norm approach might be easier to optimize with standard gradient methods.

Let me try a different approach: **mirror descent with L_p smoothing**.

The objective is: min f(a) = (sum_k c_k^p)^(1/p) subject to sum(a) = 1, a >= 0

The gradient of f with respect to a_i is:
df/da_i = (1/p) * (sum_k c_k^p)^(1/p - 1) * sum_k p * c_k^(p-1) * dc_k/da_i

where dc_k/da_i = 2 * a_{k-i} (for valid k-i).

This is a smooth function for p >= 2, and we can use gradient descent with the mirror (softmax) parameterization.

For large p (like p=10 or p=20), this approximates the max function well while being smooth enough for gradient-based optimization.

Let me implement this approach combined with the LP Nesterov method.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast LP Nesterov with warm-starting and reduced constraints
2. Multi-start diversity with diverse initializations
3. Subgradient polish on the max function
4. Multi-length exploration

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def find_solver():
    """Find the fastest working LP solver."""
    # Test solvers with a small problem and time them
    best_solver = None
    best_time = float('inf')
    
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(100, nonneg=True)
            M = cp.Variable()
            T = np.random.rand(199, 100) * 0.01
            constraints = [T @ x <= M, cp.sum(x) == 1]
            prob = cp.Problem(cp.Minimize(M), constraints)
            
            t0 = time.monotonic()
            prob.solve(solver=solver_name)
            t1 = time.monotonic()
            
            if prob.status in ('optimal', 'optimal_inaccurate'):
                elapsed = t1 - t0
                if elapsed < best_time:
                    best_time = elapsed
                    best_solver = solver_name
        except Exception:
            continue
    
    if best_solver is None:
        best_solver = cp.ECOS
    print(f"Fastest solver: {best_solver} ({best_time*1000:.1f}ms on test problem)")
    return best_solver


def solve_lp_warm(a_fixed, b_prev=None, M_prev=None, solver=cp.ECOS, reduced=False, top_k=200):
    """
    Solve LP: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0.
    Uses warm-starting if previous solution is available.
    Optionally uses reduced constraints for speed.
    """
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    if reduced:
        # Use only top_k lags
        conv_test = fast_autocorr(a_fixed)  # conv(a_fixed, a_fixed) as proxy
        top_idx = np.argsort(conv_test)[::-1][:top_k]
        T = build_conv_matrix(a_fixed)
        T_reduced = T[top_idx, :]
        constraints = [T_reduced @ b_var <= M_var, cp.sum(b_var) == 1]
    else:
        T = build_conv_matrix(a_fixed)
        constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    # Warm start
    if b_prev is not None:
        b_var.value = b_prev
    if M_prev is not None:
        M_var.value = M_prev
    
    try:
        prob.solve(solver=solver, warm_start=True)
    except Exception:
        try:
            prob.solve(solver=solver)
        except Exception:
            return None, None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        # Try alternative solvers
        for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp(a0, time_limit, solver, max_iters=200000, reduced_constraints=False):
    """
    Nesterov-accelerated LP best-response with warm-starting.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    else:
        a = a / s
    y = a.copy()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Running average (Polyak averaging)
    avg = np.zeros(n)
    n_avg = 0
    
    # Warm-start variables
    b_prev = None
    M_prev = None
    
    stall_count = 0
    stall_threshold = 2000
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP for best response to y
        b_star, M_val = solve_lp_warm(y, b_prev, M_prev, solver, 
                                        reduced=reduced_constraints)
        if b_star is None:
            continue
        
        b_prev = b_star.copy()
        M_prev = M_val
        
        # Nesterov update
        alpha_t = 2.0 / (t + 2.0)
        
        if t == 1:
            y_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            y_new = y + momentum * (y - y_prev)  # Nesterov lookahead
            # Actually, let me use the standard form:
            y_new = (1 - alpha_t) * y + alpha_t * b_star
            # Nesterov: use lookahead point
            if t > 1:
                y_lookahead = y + momentum * (y - y_prev)
                y_lookahead = np.maximum(y_lookahead, 0)
                ls = y_lookahead.sum()
                if ls > 1e-15:
                    y_lookahead = y_lookahead / ls
                # Solve LP against lookahead (already done above with y)
                # Update y
                y_new = (1 - alpha_t) * y_lookahead + alpha_t * b_star
                y_new = np.maximum(y_new, 0)
                y_new = y_new / y_new.sum()
        
        y_prev = y.copy()
        y = y_new
        
        # Update running average
        avg += y
        n_avg += 1
        
        # Check score periodically
        if t % 50 == 0:
            score_y = 2.0 * n * fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()
                stall_count = 0
            else:
                stall_count += 50
            
            if n_avg >= 100:
                avg_check = avg / n_avg
                score_avg = 2.0 * n * fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    stall_count = 0
        
        # Adaptive restart
        if stall_count > stall_threshold:
            y = best_a.copy()
            y_prev = best_a.copy()
            avg = np.zeros(n)
            n_avg = 0
            b_prev = None
            M_prev = None
            stall_count = 0
            stall_threshold = max(500, stall_threshold // 2)
            if t % 5000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NLP restart at t={t} ({elapsed:.0f}s), best={best_score:.6f}")
        
        if t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}")
    
    # Final check
    score_final = 2.0 * n * fast_autocorr(y).max()
    if score_final < best_score:
        best_score = score_final
        best_a = y.copy()
    
    if n_avg > 0:
        avg_check = avg / n_avg
        score_avg = 2.0 * n * fast_autocorr(avg_check).max()
        if score_avg < best_score:
            best_score = score_avg
            best_a = avg_check.copy()
    
    return best_a, best_score


def subgradient_polish(a0, time_limit):
    """
    Subgradient descent on the max autocorrelation with adaptive step sizes.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    step_size = 0.01
    patience = 0
    
    while time.monotonic() - t0 < time_limit - 5:
        c = fast_autocorr(a)
        cmax = c.max()
        
        # Find top peaks
        top_idx = np.argsort(c)[::-1][:20]
        
        # Subgradient of max(c_k) w.r.t. a
        # For the active set (top peaks), the subgradient is the average of their gradients
        grad = np.zeros(n)
        for k in top_idx:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * a[k - idx]
        grad = grad / len(top_idx)
        
        # Projected gradient step (stay in simplex)
        a_new = a - step_size * grad
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            continue
        a_new = a_new / sn
        
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            patience = 0
            # Increase step size slightly on success
            step_size = min(step_size * 1.05, 0.1)
        else:
            a = a  # Don't accept
            patience += 1
            if patience % 10 == 0:
                step_size *= 0.8
                if step_size < 1e-5:
                    break
    
    return best_a, best_score


def peak_shaving_focused(a0, time_limit, seed=123):
    """
    Focused peak shaving: identify the dominant peak and try to reduce it
    by redistributing mass to positions that minimally affect other peaks.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit - 5:
        b = fast_autocorr(a)
        bmax = b.max()
        
        # Find the dominant peak and nearby peaks
        top_lags = np.argsort(b)[::-1][:30]
        
        improved = False
        for k in top_lags[:15]:
            # Find which positions contribute most to this peak
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            
            # Try reducing the largest contributor
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-10:
                continue
            
            # Determine danger zone: positions that contribute to any top peak
            danger = np.zeros(n, dtype=bool)
            for k2 in top_lags[:15]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                danger[js2:je2 + 1] = True
            
            # Find safe destinations
            safe = np.where(~danger)[0]
            safe = safe[safe != src]
            if len(safe) == 0:
                safe = np.where(np.arange(n) != src)[0]
            
            # Sort by a[j] (prefer low values to minimize impact)
            safe_sorted = safe[np.argsort(a[safe])]
            dst_pool = safe_sorted[:max(5, len(safe_sorted) // 4)]
            if len(dst_pool) == 0:
                continue
            
            # Try multiple destinations and step sizes
            best_trial = None
            best_trial_score = best_score
            
            for dst in dst_pool[:10]:
                for frac in [0.002, 0.005, 0.01, 0.02, 0.04, 0.07]:
                    amt = src_val * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_trial_score:
                        best_trial_score = score
                        best_trial = a_trial
            
            if best_trial is not None:
                a = best_trial
                best_a = a.copy()
                best_score = best_trial_score
                improved = True
                stall = 0
                break
        
        if not improved:
            # Try random small perturbations
            n_pert = rng.integers(2, 6)
            a_trial = a.copy()
            for _ in range(n_pert):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a_trial[i] > 1e-8:
                    frac = rng.uniform(0.001, 0.015)
                    amt = a_trial[i] * frac
                    a_trial[i] -= amt
                    a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 1e-10:
                a_trial = a_trial / st
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_a = a.copy()
                    best_score = score
                    stall = 0
                    improved = True
            
            if not improved:
                stall += 1
                if stall > 200:
                    break
    
    return best_a, best_score


def lse_adam_polish(a0, time_limit, T_start=10.0, T_end=0.5, topk=15):
    """
    Adam-optimized top-k LogSumExp for final polish.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n)
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()
    
    lr = 0.002
    m_z = np.zeros(n)
    v_z = np.zeros(n)
    t_adam = 0
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    
    while time.monotonic() - t0 < time_limit - 5:
        a = np.exp(z - z.max())
        a = a / a.sum()
        
        c = fast_autocorr(a)
        k_use = min(topk, len(c))
        top_idx = np.argsort(c)[::-1][:k_use]
        top_vals = c[top_idx]
        
        # Adaptive temperature
        T_cur = T_start * (T_end / T_start) ** min(1.0, t_adam / 5000.0)
        
        Tc = T_cur * top_vals
        max_Tc = Tc.max()
        exp_Tc = np.exp(Tc - max_Tc)
        w = exp_Tc / exp_Tc.sum()
        
        # Gradient
        grad_a = np.zeros(n)
        for idx in range(k_use):
            K = top_idx[idx]
            wK = w[idx]
            js = max(0, K - (n - 1))
            je = min(n - 1, K)
            if js <= je:
                idx_range = np.arange(js, je + 1)
                grad_a[idx_range] += 2.0 * wK * a[K - idx_range]
        
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        
        t_adam += 1
        m_z = beta1 * m_z + (1 - beta1) * grad_z
        v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
        m_hat = m_z / (1 - beta1 ** t_adam)
        v_hat = v_z / (1 - beta2 ** t_adam)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        if t_adam % 50 == 0:
            a_check = np.exp(z - z.max())
            a_check = a_check / a_check.sum()
            score = 2.0 * n * fast_autocorr(a_check).max()
            if score < best_score:
                best_score = score
                best_a = a_check.copy()
            else:
                lr *= 0.95
    
    return best_a, best_score


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        a = np.exp(-5 * (x - 0.3) ** 2) + 0.5 * np.exp(-5 * (x - 0.7) ** 2)
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.2 * np.cos(7 * np.pi * x)
        a = np.maximum(a, 0.01)
    elif choice == 3:
        a = np.abs(np.sin(3 * np.pi * x)) ** 0.5
    elif choice == 4:
        a = (1 - x) * np.sin(np.pi * x)
        a = np.maximum(a, 0.001)
    else:
        a = np.ones(n) / n
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
