def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:

        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    sequence = [min(1000.0, x) for x in sequence]

    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def fast_autocorr(a):
    """Compute full convolution (autocorrelation) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def interp_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    return f_interp / ss

def lp_fw_optimize(a0, time_limit, solver, initial_K=500, max_K=5000, add_per_iter=50, verbose=True):
    """LP-based Frank-Wolfe with cutting planes and admissible step size."""
    n = len(a0)
    y = norm_sum1(a0)
    best_a = y.copy()
    ac_y = fast_autocorr(y)
    best_m = ac_y.max()
    t0 = time.monotonic()

    # Initialize constraint set with top-K convolution positions
    init_idx = np.argsort(ac_y)[::-1][:min(initial_K, len(ac_y))]
    constraint_set = set(int(k) for k in init_idx)

    x = y.copy()  # Nesterov iterate
    iter_count = 0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()

        if M_curr < best_m - 1e-14:
            best_m = M_curr
            best_a = y.copy()

        ac_x = fast_autocorr(x)
        if ac_x.max() < best_m - 1e-14:
            best_m = ac_x.max()
            best_a = x.copy()

        # Manage constraint set
        constraint_list = sorted(constraint_set)
        if len(constraint_list) > max_K:
            margins = np.array([M_curr - ac[k] for k in constraint_list])
            keep = np.argsort(margins)[:max_K]
            constraint_set = set(int(constraint_list[i]) for i in keep)
            constraint_list = sorted(constraint_set)

        K = len(constraint_list)
        if K == 0:
            break

        # Build constraint matrix (sparse-friendly)
        coeffs = np.zeros((K, n))
        for jj, k in enumerate(constraint_list):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start < i_end:
                idx = np.arange(i_start, i_end)
                coeffs[jj, idx] = y[k_int - idx]

        # Solve LP
        g_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(g_var) == 1, coeffs @ g_var <= M_var]
        prob = cp.Problem(cp.Minimize(M_var), constraints)

        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if len(constraint_list) > initial_K:
                constraint_set = set(int(k) for k in constraint_list[:initial_K])
                constraint_list = sorted(constraint_set)
            continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        g_val = np.maximum(np.asarray(g_var.value, dtype=np.float64), 0.0)
        g_sum = g_val.sum()
        if g_sum < 1e-15:
            continue
        g_dir = g_val / g_sum

        # Find violated constraints and add them
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        Yf = np.fft.rfft(y, fft_size)
        Gf = np.fft.rfft(g_dir, fft_size)
        conv_ys = np.fft.irfft(Yf * Gf, fft_size)[:size]

        violations = np.where(conv_ys > M_curr + 1e-7)[0]
        if len(violations) > 0:
            viol_amt = conv_ys[violations] - M_curr
            top_v = violations[np.argsort(viol_amt)[::-1][:add_per_iter]]
            for k in top_v:
                constraint_set.add(int(k))

        # Admissible step size: try several alphas
        alpha_base = 2.0 / (t + 2.0)
        best_alpha = alpha_base
        best_alpha_val = 2.0 * n * fast_autocorr((1 - alpha_base) * y + alpha_base * g_dir).max()

        # Try a few alternatives
        for trial_alpha in [alpha_base * 2.0, alpha_base * 0.5, alpha_base * 1.5]:
            trial_alpha = min(max(trial_alpha, 0.01), 1.0)
            trial = (1 - trial_alpha) * y + trial_alpha * g_dir
            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            if ts > 1e-15:
                trial = trial / ts
                tv = 2.0 * n * fast_autocorr(trial).max()
                if tv < best_alpha_val:
                    best_alpha_val = tv
                    best_alpha = trial_alpha

        # Update y with best alpha
        y_new = (1.0 - best_alpha) * y + best_alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        y = y_new

        # Nesterov update for x
        if t == 1:
            x_new = y_new.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_t * (y_new - x)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        x = x_new
        iter_count += 1

        if verbose and t % 30 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP-FW iter={t} ({elapsed:.0f}s): bestJ={J:.6f}, |S|={len(constraint_set)}, M_cur={M_curr:.6f}")

    elapsed = time.monotonic() - t0
    J = 2.0 * n * best_m
    if verbose:
        print(f"    LP-FW done ({elapsed:.0f}s): J={J:.6f}, iters={iter_count}")
    return best_a, best_m


def mirror_descend_optimize(a0, time_limit, top_k=80, beta=4.0, c0=4.0, restart_every=5000, seed=42, verbose=False):
    """Nesterov-accelerated mirror descent with soft subgradient and adaptive restarts."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    x = a.copy()
    x_prev = a.copy()
    stall_count = 0
    cur_top_k = top_k
    cur_beta = beta

    for t in range(1, 30000000):
        if time.monotonic() - t0 > time_limit:
            break

        beta_t = (t - 1.0) / (t + 1.5)
        y = x + beta_t * (x - x_prev)
        y = np.maximum(y, 1e-15)
        sy = y.sum()
        if sy < 1e-15:
            y = x.copy()
        else:
            y = y / sy

        ac = fast_autocorr(y)
        k_max = min(cur_top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 1e-30)
        w = top_vals ** cur_beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum

        # Compute soft subgradient
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue
            idx = np.arange(i_start, i_end)
            j_idx = k_int - idx
            grad[idx] += 2.0 * wi * y[j_idx]

        eta = c0 / (np.sqrt(t) + 1.0)
        log_a = np.log(y) - eta * grad
        max_log = log_a.max()
        a_new = np.exp(log_a - max_log)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        x_prev = x
        x = a_new

        M_new = fast_autocorr(x).max()
        if M_new < best_m - 1e-14:
            best_m = M_new
            best_a = x.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > restart_every:
            stall_count = 0
            # Restart with perturbation
            pert_scale = 0.03 * (1.0 + 0.5 * np.sin(np.pi * np.arange(n) / n))
            pert = rng.normal(0, 1.0, n) * pert_scale * best_a
            x_r = best_a + pert
            x_r = np.maximum(x_r, 1e-10)
            x_r = x_r / x_r.sum()
            x = x_r
            x_prev = x_r.copy()
            cur_top_k = min(cur_top_k + 30, 200)
            cur_beta = max(cur_beta - 0.8, 2.0)
            c0 = min(c0 * 1.1, 8.0)
            if verbose:
                elapsed = time.monotonic() - t0
                J = 2.0 * n * best_m
                print(f"    MD restart t={t} ({elapsed:.0f}s): J={J:.6f}, tk={cur_top_k}, b={cur_beta:.1f}")

    return best_a, best_m


def coord_polish(a0, time_limit, seed=77, verbose=True):
    """Coordinate descent with multi-scale mass transfer."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    ac0 = fast_autocorr(a)
    best_m = ac0.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 600:
        ac = fast_autocorr(a)
        n_top = min(30, len(ac))
        top_k = np.argsort(ac)[::-1][:n_top]

        improved = False

        for k in top_k[:15]:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-20:
                continue

            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Score all positions by their contribution to top peaks
            scores = np.zeros(n)
            in_peak = np.zeros(n, dtype=bool)
            for k2 in top_k[:8]:
                k2i = int(k2)
                s2 = max(0, k2i - n + 1)
                e2 = min(n - 1, k2i)
                if s2 <= e2:
                    idx2 = np.arange(s2, e2 + 1)
                    scores[idx2] += a[k2i - idx2]
                    scores[idx2] += a[idx2]
                    in_peak[s2:e2 + 1] = True

            candidates = np.where(~in_peak)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            cand_scores = scores[candidates]
            dst = candidates[np.argmin(cand_scores)]

            if dst == src:
                continue

            # Try multiple transfer fractions
            found = False
            fracs = [0.05, 0.10, 0.15, 0.25, 0.40, 0.60]
            for frac in fracs:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    found = True
                    stall = 0
                    break

            if not found and stall > 100:
                # Random exploration
                j = int(rng.integers(n))
                if j != src and a[src] > 1e-10:
                    frac = rng.uniform(0.01, 0.05)
                    amt = a[src] * frac
                    a[src] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        improved = True

        if not improved:
            stall += 1
            # Random perturbation to escape local minima
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 200 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    Coord ({elapsed:.0f}s): J={J:.6f}, stall={stall}")

    return best_a, best_m


def local_subgradient_step(a, ac, k_peak, n_top=5):
    """Compute exact subgradient at top peak positions."""
    n = len(a)
    top_idx = np.argsort(ac)[::-1][:min(n_top, len(ac))]
    grad = np.zeros(n)
    for k in top_idx:
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j_idx = k_int - idx
        for i_val, j_val in zip(idx, j_idx):
            if i_val != j_val:
                grad[i_val] += a[j_val]
            else:
                grad[i_val] += a[i_val]
    return grad

            i_end = min(n, k_int + 1)
            if i_start >= i_end:
                continue

                    else:
                        grad[i_val] += y[i_val]

        min_idx = np.argmin(grad)
        e_min = np.zeros(n)
        e_min[min_idx] = 1.0

        if y[max_idx] > 0.01:

            y_away = y - alpha_away * (np.zeros(n))

            else:

        else:
            g_dir = e_min
