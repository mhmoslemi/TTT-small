def evaluate_sequence(seq: List[float]) -> float:
    """ Evaluates the sequence for quality using full convolution normalization.
    Normalizes sequence to have sum 1, then computes J = 2*n*max(conv_full)/ (sum)^2.
    Requires sequence to be non-negative after clipping. """
    n = len(seq)
    if n < 1:
        return float('inf')
    # Convert to array and clip nonzero to zero
    arr = np.asarray(seq, dtype=np.float64)
    # Filter out negative and replace with zero (due to potential numerical issues)
    arr = np.maximum(arr, 0.0)
    # Scale sequence for normalization if needed
    sum_val = np.sum(arr)
    if sum_val < 0.01:  # Defensize below minimal sum threshold
        arr = np.ones_like(arr) / len(arr)
    else:
        arr = arr / sum_val
    # Full convolution as in self-correlation of the sequence
    conv = sp_correlate_full(arr, arr, mode='full')  # shape: len(arr)*2-1, returns full not circular
    max_conv = np.max(conv)
    return 2 * n * max_conv / (sum_val ** 2)

def project_simplex(v: np.ndarray) -> np.ndarray:
    """ Projects a vector onto the simplex (probability space). """
    if np.isscalar(v):
        v = np.array([v], dtype=float)
    n = len(v)
    y = np.zeros((n, n))
    y[0] = v
    max_val = v.max()
    s = np.count_nonzero(v) * max_val
    if s == 0.0:
        return np.zeros(n)
    # Find the minimum t such that all coordinates might exceed
    # Initialize candidate distribution
    r_sum = np.sum(v)
    # Clamping to the simplex
    unrolled = v.astype(float)
    tol = 1e-10  # small tolerance to avoid errors
    for j in np.argsort(v)[::-1]:
        # Adjust coordinates to lie within simplex
        excess = unrolled[j] - (1 - r_sum) / (n - j)
        if excess > tol:
            unrolled[j] -= excess
    # Simple normalization if sum is not too small
    np.putmask(unrolled == 0, True, [1e-3, 1e-3] * (1 / np.count_nonzero(unrolled)))
    arr = unrolled / np.sum(unrolled)
    arr = np.maximum(arr, 0.0)
    return arr
