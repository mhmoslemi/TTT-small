def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgradient_nesterov(f0, time_limit, verbose=False):
    """
    Subgradient method with Nesterov's averaging trick on the exact max-convolution.
    The objective f(a) = max_k (a*a)[k] is convex.
    Subgradient at a when max is at k*: g[j] = a[k*-j] (for valid j).
    Uses averaged subgradients over the active set for better behavior.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    # Nesterov averaging
    avg_f = f.copy()
    t0 = time.monotonic()
    it = 0
    eta = 1.0 / np.sqrt(n)  # initial step size
    
    # Adaptive parameters
    no_improve = 0
    patience = 500
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        conv = fast_convolve(f)
        M = conv.max()
        
        if M < best_obj - 1e-12:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Find active set (indices where conv is close to max)
        threshold = M - 0.01 * M  # 1% below max
        active = np.where(conv >= threshold)[0]
        
        # Compute averaged subgradient
        # For each active index k, subgradient is g_k[j] = a[k-j] for valid j
        g = np.zeros(n)
        for k in active:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je >= js:
                # g[j] += a[k-j] for j in [js, je]
                # a[k-j] where j goes from js to je
                # = a[k-je : k-js+1][::-1]
                g[js:je+1] += f[k-je:k-js+1][::-1]
        g /= len(active)
        
        # Update
        f_new = f - eta / (np.sqrt(it) + 1) * g
        f_new = project_simplex(f_new)
        
        # Nesterov averaging
        avg_f = (it / (it + 1)) * avg_f + (1 / (it + 1)) * f_new
        
        # Check averaged solution
        if it % 10 == 0:
            avg_obj = fast_convolve(avg_f).max()
            if avg_obj < best_obj - 1e-12:
                best_obj = avg_obj
                best_f = avg_f.copy()
            
            # Also check f_new directly
            new_obj = fast_convolve(f_new).max()
            if new_obj < best_obj - 1e-12:
                best_obj = new_obj
                best_f = f_new.copy()
        
        f = f_new
        
        # Adaptive step size
        if no_improve > patience:
            eta *= 0.7
            no_improve = 0
            patience = int(patience * 1.3)
            if verbose and it % 500 == 0:
                print(f"    SubGrad iter {it}: obj={2*n*best_obj:.8f}, eta={eta:.4f}")
        
        if no_improve > patience * 5:
            break

    return best_f, 2 * n * best_obj

def _smooth_gradient(f, tau):
    """Compute gradient of log-sum-exp smooth approximation using FFT."""
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()

    # Gradient: g[j] = (2/tau) * sum_l w[j+l] * f[l]
    # Use FFT: g = (2/tau) * ifft(fft(f) * fft(w_rev))[n-1:2n-1][::-1]
    w_rev = w[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)
    h = conv_result[n - 1:2 * n - 1][::-1]

    return (2.0 / tau) * h

def _solve_lp_reduced(f, M, n, active_idx):
    """Solve LP with only active constraints."""
    c = -np.ones(n)
    m = len(active_idx)
    A_ub = np.zeros((m, n))
    for i, k in enumerate(active_idx):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je >= js:
            A_ub[i, js:je+1] = f[k-je:k-js+1][::-1]
    b_ub = np.full(m, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 2})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def coordinate_descent_v2(f0, time_limit, verbose=False):
    """
    Improved coordinate descent:
    - Pairs-based optimization (adjust pairs of coordinates)
    - Focus on peak-contributing pairs
    - More aggressive redistribution
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        improved = False

        # Get top peaks
        top_k = min(8, 2 * n - 1)
        peak_indices = np.argsort(conv)[-top_k:]

        for pk in peak_indices:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            if je < js:
                continue

            idx = np.arange(js, je + 1)
            products = f[idx] * f[pk - idx]
            
            # Try adjusting the top-3 contributing pairs
            top_pairs = np.argsort(products)[-3:]
            for pair_idx in top_pairs:
                j = int(idx[pair_idx])
                k_other = pk - j
                
                if j == k_other:
                    # Diagonal: reduce f[j]
                    for factor in [0.5, 0.6, 0.7, 0.8]:
                        trial = f.copy()
                        old = f[j]
                        trial[j] = old * factor
                        diff = old - trial[j]
                        # Redistribute proportionally to non-peak elements
                        mask = _get_safe_mask(f, pk, n, top_k)
                        if mask.sum() >= 3:
                            trial[mask] += diff * f[mask] / f[mask].sum()
                            trial = np.maximum(trial, 0)
                            trial /= trial.sum()
                            obj = fast_convolve(trial).max()
                            if obj < best_obj - 1e-13:
                                f = trial
                                best_obj = obj
                                best_f = f.copy()
                                improved = True
                                break
                else:
                    # Off-diagonal: reduce product f[j]*f[k_other]
                    for scale in [0.5, 0.65, 0.8]:
                        trial = f.copy()
                        old_j = f[j]
                        trial[j] = old_j * scale
                        diff = old_j - trial[j]
                        mask = _get_safe_mask(f, pk, n, top_k)
                        mask[j] = False
                        mask[k_other] = False
                        if mask.sum() >= 3:
                            trial[mask] += diff * f[mask] / f[mask].sum()
                            trial = np.maximum(trial, 0)
                            trial /= trial.sum()
                            obj = fast_convolve(trial).max()
                            if obj < best_obj - 1e-13:
                                f = trial
                                best_obj = obj
                                best_f = f.copy()
                                improved = True
                                break

            if improved:
                break

        if not improved:
            no_improve += 1
            if no_improve > 20:
                rng = np.random.default_rng(hash((it, n)) % (2**32))
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                no_improve = 0
        else:
            no_improve = 0

        if verbose and it % 100 == 0:
            print(f"    CD2 iter {it}: obj={2*n*best_obj:.8f}")

    return best_f, 2 * n * best_obj

def _get_safe_mask(f, pk, n, top_k):
    """Get mask of elements that don't contribute to top peaks."""
    mask = np.ones(n, dtype=bool)
    conv = fast_convolve(f)
    peak_indices = np.argsort(conv)[-top_k:]
    for pk2 in peak_indices:
        pk2 = int(pk2)
        js2 = max(0, pk2 - n + 1)
        je2 = min(n - 1, pk2)
        mask[js2:je2+1] = False
    return mask

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def make_fejer(n):
    """Fejér kernel (sinc^2 shape), normalized."""
    x = np.arange(n) - (n - 1) / 2.0
    if n % 2 == 0:
        x = x + 0.5
    a = np.sinc(x / (n / 2.0)) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def make_hann(n):
    """Hann window."""
    a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
    a = np.maximum(a, 1e-10)
    a /= a.sum()
    return a

def make_hamming(n):
    """Hamming window."""
    a = 0.54 - 0.46 * np.cos(2 * np.pi * np.arange(n) / n)
    a = np.maximum(a, 1e-10)
    a /= a.sum()
    return a

def make_blackman(n):
    """Blackman window."""
    a = (0.42 - 0.5 * np.cos(2 * np.pi * np.arange(n) / n)
         + 0.08 * np.cos(4 * np.pi * np.arange(n) / n))
    a = np.maximum(a, 1e-10)
    a /= a.sum()
    return a

def make_asymmetric_gaussian(n, center_frac=0.4, sigma_frac=0.3):
    """Asymmetric Gaussian initialization."""
    x = np.arange(n) - n * center_frac
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_gradient_fft(f, tau):
    """Compute gradient of log-sum-exp smooth approximation using FFT."""
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.zeros(n)
    w /= w_sum

    w_rev = w[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)
    h = conv_result[n - 1:2 * n - 1][::-1]

    return (2.0 / tau) * h

            tau = 0.04 * fast_convolve(f).max()

            if verbose and (t + 150) % 500 == 0:
                print(f"    Adam restart: obj={2*n*best_obj:.8f}")

        if verbose and t % 2000 == 0:
            print(f"    Adam iter {t}: obj={2*n*best_obj:.8f}, tau={tau:.2e}")

    return best_f, 2 * n * best_obj


def subgradient_nesterov(f0, time_limit, verbose=False):
    """
    Subgradient method with Nesterov's averaging on exact max-convolution.
    Uses averaged subgradients over active set (near-max indices).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    avg_f = f.copy()
    t0 = time.monotonic()
    it = 0
    base_eta = 1.0 / np.sqrt(n)

    no_improve = 0
    patience = 300

    while time.monotonic() - t0 < time_limit:
        it += 1

        conv = fast_convolve(f)
        M = conv.max()

        if M < best_obj - 1e-12:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Active set: indices within 2% of max
        threshold = M - 0.02 * M
        active = np.where(conv >= threshold)[0]

        # Compute averaged subgradient
        g = np.zeros(n)
        for k in active:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je >= js:
                g[js:je + 1] += f[k - je:k - js + 1][::-1]
        g /= float(len(active))

        # Update with diminishing step
        eta = base_eta / (np.sqrt(it + 1))
        f_new = f - eta * g
        f_new = project_simplex(f_new)

        # Nesterov averaging
        avg_f = (it / (it + 1)) * avg_f + (1.0 / (it + 1)) * f_new

        # Check both current and averaged solutions periodically
        if it % 5 == 0:
            avg_obj = fast_convolve(avg_f).max()
            if avg_obj < best_obj - 1e-12:
                best_obj = avg_obj
                best_f = avg_f.copy()

        f = f_new

        # Adaptive patience
        if no_improve > patience:
            no_improve = 0
            patience = int(patience * 1.5)
            if patience > 5000:
                # Try perturbation
                rng = np.random.default_rng(it * 11 + 5)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                avg_f = f.copy()
                patience = 300

        if verbose and it % 500 == 0:
            print(f"    SubGrad iter {it}: obj={2*n*best_obj:.8f}")

        if no_improve > 10000:
            break

    return best_f, 2 * n * best_obj


def lp_fw_optimize(f0, time_limit, verbose=False):
    """
    LP-Frank-Wolfe with active set reduction.
    Solves LP: max sum(g) s.t. (f*g)[k] <= M for active k, g >= 0.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 3:
            break

        conv = fast_convolve(f)
        M_cur = conv.max()

        # Active set: top 150 most binding constraints
        top_count = min(150, 2 * n - 1)
        active_idx = np.argsort(conv)[::-1][:top_count]
        # Only include those above 40% of max
        active_idx = active_idx[conv[active_idx] >= 0.4 * M_cur]

        # Solve LP
        g = _solve_lp(f, M_cur * m_exp, n, active_idx)

        if g is None:
            m_exp *= 1.15
            if m_exp > 5.0:
                rng = np.random.default_rng(it * 13 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                M = fast_convolve(f).max()
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp *= 1.1
            continue
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 5:
                m_exp = min(m_exp * 1.03, 3.0)
            if no_improve > 30:
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                M = fast_convolve(f).max()
                m_exp = 1.0
                no_improve = 0
            continue
        else:
            no_improve = 0

        # Line search: golden section
        best_alpha = 0.0
        best_val = M_cur

        for alpha in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = np.maximum(f + alpha * (g - f), 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.04)
            hi = min(1.0, best_alpha + 0.04)
            for _ in range(12):
                mid = (lo + hi) / 2.0
                ft = np.maximum(f + mid * (g - f), 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = np.maximum(f + best_alpha * (g - f), 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 5 == 0:
                    print(f"    LP-FW iter {it}: obj={2*n*best_obj:.8f}")
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.03, 3.0)

        if no_improve > 150:
            break

    return best_f, 2 * n * best_obj


def _solve_lp(f, M, n, active_idx):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in active_idx, g >= 0."""
    c = -np.ones(n)
    m = len(active_idx)
    A_ub = np.zeros((m, n))
    for i, k in enumerate(active_idx):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je >= js:
            A_ub[i, js:je + 1] = f[k - je:k - js + 1][::-1]
    b_ub = np.full(m, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 3})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def coordinate_descent(f0, time_limit, verbose=False):
    """
    Peak-targeted coordinate descent: identifies peak-contributing pairs
    and redistributes mass to reduce peaks.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        improved = False

        top_k = min(6, 2 * n - 1)
        peak_indices = np.argsort(conv)[-top_k:]

        for pk in peak_indices:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            if je < js:
                continue

            idx = np.arange(js, je + 1)
            products = f[idx] * f[pk - idx]
            top_pairs = np.argsort(products)[-3:]

            for pair_idx in top_pairs:
                j = int(idx[pair_idx])
                k_other = pk - j

                if j == k_other:
                    for factor in [0.5, 0.65, 0.8]:
                        trial = f.copy()
                        old = f[j]
                        trial[j] = old * factor
                        diff = old * (1 - factor)
                        mask = _get_safe_mask(f, peak_indices, n, top_k)
                        if mask.sum() >= 5:
                            trial[mask] += diff * f[mask] / f[mask].sum()
                            trial = np.maximum(trial, 0)
                            trial /= trial.sum()
                            obj = fast_convolve(trial).max()
                            if obj < best_obj - 1e-13:
                                f = trial
                                best_obj = obj
                                best_f = f.copy()
                                improved = True
                                break
                else:
                    for scale in [0.5, 0.65, 0.8]:
                        trial = f.copy()
                        old_j = f[j]
                        trial[j] = old_j * scale
                        diff = old_j * (1 - scale)
                        mask = _get_safe_mask(f, peak_indices, n, top_k)
                        mask[j] = False
                        mask[k_other] = False
                        if mask.sum() >= 5:
                            trial[mask] += diff * f[mask] / f[mask].sum()
                            trial = np.maximum(trial, 0)
                            trial /= trial.sum()
                            obj = fast_convolve(trial).max()
                            if obj < best_obj - 1e-13:
                                f = trial
                                best_obj = obj
                                best_f = f.copy()
                                improved = True
                                break

            if improved:
                break

        if not improved:
            no_improve += 1
            if no_improve > 15:
                rng = np.random.default_rng(it * 17 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                no_improve = 0
        else:
            no_improve = 0

        if verbose and it % 50 == 0:
            print(f"    CD iter {it}: obj={2*n*best_obj:.8f}")

    return best_f, 2 * n * best_obj


def _get_safe_mask(f, peak_indices, n, top_k):
    """Get mask of elements not contributing to top peaks."""
    mask = np.ones(n, dtype=bool)
    for pk2 in peak_indices[:top_k]:
        pk2 = int(pk2)
        js2 = max(0, pk2 - n + 1)
        je2 = min(n - 1, pk2)
        mask[js2:je2 + 1] = False
    return mask


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_fejer(n):
    """Fejer kernel shape."""
    x = np.arange(n) - (n - 1) / 2.0
    a = np.sinc(x / (n / 2.0)) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_hann(n):
    """Hann window."""
    a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
    a = np.maximum(a, 1e-10)
    a /= a.sum()
    return a


def make_hamming(n):
    """Hamming window."""
    a = 0.54 - 0.46 * np.cos(2 * np.pi * np.arange(n) / n)
    a = np.maximum(a, 1e-10)
    a /= a.sum()
    return a


def make_blackman(n):
    """Blackman window."""
    a = (0.42 - 0.5 * np.cos(2 * np.pi * np.arange(n) / n)
         + 0.08 * np.cos(4 * np.pi * np.arange(n) / n))
    a = np.maximum(a, 1e-10)
    a /= a.sum()
    return a


def make_gaussian(n, center_frac=0.4, sigma_frac=0.25):
    """Gaussian initialization."""
    x = np.arange(n) - n * center_frac
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def symmetric_project(f, n):
    """Project to symmetric sequence and then to simplex."""
    sym = 0.5 * (f + f[::-1])
    return project_simplex(sym)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization targeting sub-1.5027 objective.
    Phase 1: Diverse initialization (30s)
    Phase 2: Adam smooth optimization (550s)
    Phase 3: Subgradient Nesterov (250s)
    Phase 4: LP-FW refinement (350s)
    Phase 5: Coordinate descent (150s)
    Phase 6: Multi-length exploration (200s)
    Phase 7: Final polishing (150s)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
        return score

    # === Phase 1: Initialization ===
    print("=== Phase 1: Initialization ===")

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_fejer(512)
        try_seq(parent, "fejer_default")

    # Diverse initializations
    for n_try in [512, 640, 768, 896, 1024]:
        if time.monotonic() > deadline - 1500:
            break
        try_seq(make_fejer(n_try), f"fejer_{n_try}")
        try_seq(make_hann(n_try), f"hann_{n_try}")
        try_seq(make_hamming(n_try), f"hamming_{n_try}")
        try_seq(make_blackman(n_try), f"blackman_{n_try}")
        try_seq(make_gaussian(n_try, 0.35, 0.22), f"gauss_{n_try}")

    # Interpolate parent to different lengths
    if parent is not None:
        for target_n in [640, 896, 1024, 1280]:
            if target_n == len(parent):
                continue
            if time.monotonic() > deadline - 1400:
                break
            interp = interpolate_sequence(parent, target_n)
            try_seq(interp, f"parent_interp_{target_n}")

    print(f"Phase 1 done. Best: {best_score:.6f} (n={len(best_seq) if best_seq is not None else 0})")

    work_seq = best_seq.copy() if best_seq is not None else parent
    work_n = len(work_seq)

    # === Phase 2: Adam optimization ===
    print(f"\n=== Phase 2: Adam (n={work_n}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem, 500)
    if t2 > 60:
        print(f"  Time: {t2:.0f}s")
        a_opt, _ = adam_optimize(work_seq, time_limit=t2 * 0.6, lr_init=0.004, verbose=True)
        try_seq(a_opt, "adam_main")

        if time.monotonic() < deadline - 600:
            rem2 = deadline - time.monotonic() - 600
            t2b = min(rem2, 200)
            a_opt2, _ = adam_optimize(best_seq, time_limit=t2b, lr_init=0.002, verbose=False)
            try_seq(a_opt2, "adam_round2")

    # === Phase 3: Subgradient Nesterov ===
    print(f"\n=== Phase 3: Subgradient (n={len(best_seq)}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 500
        if rem > 50:
            t3 = min(rem, 250)
            print(f"  Time: {t3:.0f}s")
            a_opt, _ = subgradient_nesterov(best_seq, time_limit=t3, verbose=True)
            try_seq(a_opt, "subgrad")

    # === Phase 4: LP-FW ===
    print(f"\n=== Phase 4: LP-FW (n={len(best_seq)}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 350
        if rem > 30:
            t4 = min(rem, 300)
            print(f"  Time: {t4:.0f}s")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t4, verbose=True)
            try_seq(a_opt, "lpfw")

    # === Phase 5: Coordinate Descent ===
    print(f"\n=== Phase 5: CD (n={len(best_seq)}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t5 = min(rem, 150)
            print(f"  Time: {t5:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t5, verbose=True)
            try_seq(a_opt, "cd")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-Length ===")
    if best_seq is not None:
        for target_n in [1024, 1280, 896, 640]:
            if target_n == len(best_seq):
                continue
            rem = deadline - time.monotonic() - 80
            if rem < 30:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            try_seq(f_interp, f"interp_{target_n}")

            t6 = min(rem * 0.3, 80)
            if t6 > 20:
                a_opt, _ = adam_optimize(f_interp, time_limit=t6, lr_init=0.003, verbose=False)
                try_seq(a_opt, f"adam_interp_{target_n}")

    # === Phase 7: Final polished optimization cycle ===
    print(f"\n=== Phase 7: Final Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 30:
            # Adam with very low temperature
            t7 = min(rem * 0.35, 60)
            if t7 > 10:
                a_opt, _ = adam_optimize(best_seq, time_limit=t7, lr_init=0.001, verbose=False)
                try_seq(a_opt, "final_adam")

            # LP-FW squeeze
            rem2 = deadline - time.monotonic() - 10
            if rem2 > 15:
                t7b = min(rem2, 50)
                a_opt, _ = lp_fw_optimize(best_seq, time_limit=t7b, verbose=False)
                try_seq(a_opt, "final_lpfw")

            # CD polish
            rem3 = deadline - time.monotonic() - 5
            if rem3 > 10:
                t7c = min(rem3, 25)
                a_opt, _ = coordinate_descent(best_seq, time_limit=t7c, verbose=False)
                try_seq(a_opt, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
