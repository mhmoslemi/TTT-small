def lp_direction(a, M, solver='CLARABEL'):
    """Solve LP: max sum(g) s.t. g >= 0, (a*g)_k <= M for all k."""
    n = len(a)
    
    # Build constraint matrix A such that (A @ g)[k] = (a*g)_k
    # A[k, j] = a[j] if 0 <= k-j < n, else 0
    # Equivalently: A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # Wait no: (a*g)_k = sum_j a[j] * g[k-j]
    # Let i = k-j: (a*g)_k = sum_i a[k-i] * g[i]
    # So A[k, i] = a[k-i] if 0 <= k-i < n
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            A[k, i] = a[k - i]
    
    # LP: max 1^T g s.t. A g <= M * 1, g >= 0
    # In standard form: min -1^T g s.t. A g <= M*1, g >= 0
    c = -np.ones(n)
    A_ub = A
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return -res.fun  # max sum(g)
    return None

"""
LP-based iterative optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses the "state of the art" LP direction-finding approach combined with
mirror descent for fine-tuning, diverse initializations, and symmetrization.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def l2_cross_conv(a, g):
    """Compute cross-convolution a*g via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    G = rfft(g, fft_size)
    return irfft(A * G, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (cross-convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            C[k, i] = a[k - i]
    return C


def solve_lp_direction(a, M):
    """
    Solve LP: max sum(g) s.t. g >= 0, (a*g)_k <= M for all k.
    Returns (g, optimal_value) or (None, 0) if infeasible/unbounded.
    """
    n = len(a)
    C = build_conv_matrix(a)
    c = -np.ones(n)  # minimize -sum(g)
    A_ub = C
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, -res.fun
    return None, 0

    LP-based iterative optimization.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)

    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = M
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        
        # Solve LP
        g, S = solve_lp_direction(a, M)
        if g is None or S <= 1.0 + 1e-10:
            # LP gives no improvement direction
            no_improve += 1
            if no_improve > 20:
                # Try with a slightly relaxed M

                if g is None or S <= 1.0 + 1e-10:
                    break
        else:
            no_improve = 0
        
        # Normalize g to probability vector
        p = g / S
        
        # Line search for optimal t
        best_t = 0.1
        best_trial_obj = M
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
            a_trial = (1 - t) * a + t * p
            a_trial = np.maximum(a_trial, 0)

            if trial_obj < best_trial_obj:

def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def make_bell(n, width_frac=0.6):
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def symmetrize(a):
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    ss = a_sym.sum()
    if ss > 1e-15:
        a_sym /= ss
    return a_sym
