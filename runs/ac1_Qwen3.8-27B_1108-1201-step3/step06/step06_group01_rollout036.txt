def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_weighted_subgradient(a, top_k=10, temperature=None):
    """
    Compute a weighted subgradient of max_k conv(a,a)[k] with respect to a.
    Uses top-k peaks with softmax weighting for smoother gradients.
    Returns (gradient, peak_value, peak_lag).
    """
    n = len(a)
    conv = fast_convolve(a)
    
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    peak_vals = conv[peaks]
    
    if temperature is None:
        temperature = 0.01 * max(peak_vals.max(), 1e-10)
    
    scaled = (peak_vals - peak_vals.max()) / max(temperature, 1e-15)
    w = np.exp(scaled)
    w /= w.sum()
    
    # g[j] = sum_i w_i * a[k_i - j] (where 0 <= k_i - j <= n-1)
    g = np.zeros(n)

            continue
        k = int(peaks[i])
        # a[k-j] is valid when 0 <= k-j <= n-1, i.e., max(0, k-n+1) <= j <= min(n-1, k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)

            idx = k - np.arange(j_start, j_end + 1)
            g[j_start:j_end+1] += w[i] * a[idx]
    
    peak_value = conv.max()
    peak_lag = int(np.argmax(conv))
    return g, peak_value, peak_lag


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based fictitious play (game-theoretic approach) as the primary engine
2. Nesterov-accelerated weighted subgradient for fast iterations between LP calls
3. Adaptive-temperature L-BFGS polishing with multiple restarts
4. Aggressive peak shaving and coordinate descent for final micro-optimization
5. Multi-length exploration with interpolation
Key improvements: much more time on core optimization, Nesterov momentum for
faster convergence, adaptive soft-max temperature schedule, and better
peak-shaving with diverse trial factors.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix_sparse(a):
    """Build convolution matrix T such that T @ b = conv(a, b), stored densely."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    if '_lp_solver_view' in globals():
        return _lp_solver_view
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.PDLP, cp.ECOS, cp.SCS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_view = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_view = cp.ECOS
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    """
    Given fixed a (non-negative, sum=1), find b (non-negative, sum=1)
    minimizing max_k conv(a, b)[k]. Returns (b, M).
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix_sparse(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1] + 1
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def compute_weighted_subgradient(a, top_k=10, temperature=None):
    """
    Compute weighted subgradient of max_k conv(a,a)[k] w.r.t. a.
    Uses top-k peaks with softmax weighting for smoother gradients.
    Returns (gradient, peak_value).
    """
    n = len(a)
    conv = fast_convolve(a)

    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    peak_vals = conv[peaks]

    if temperature is None:
        temperature = 0.005 * max(peak_vals.max(), 1e-10)

    scaled = (peak_vals - peak_vals.max()) / max(temperature, 1e-15)
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= ws

    g = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-15:
            continue
        k = int(peaks[i])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = k - np.arange(j_start, j_end + 1)
            g[j_start:j_end + 1] += w[i] * a[idx]

    peak_value = conv.max()
    return g, peak_value


def nesterov_subgradient_refine(a0, time_limit, top_k=10, lr_init=0.005):
    """
    Nesterov-accelerated weighted subgradient method.
    Uses momentum and Polyak-like step size for faster convergence.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    lower_bound = 2.0 * n * (1.0 / (2.0 * n - 1.0))  # averaging lower bound

    y = a.copy()  # Nesterov lookahead
    v = np.zeros(n)  # momentum
    beta = 0.9  # momentum parameter

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    lr = lr_init

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov lookahead
        y = simplex_project(2.0 * a - a - v)  # Actually standard: y = a + (t-1)/(t+2) * v
        # Use simpler momentum: y = a + beta * v
        y = simplex_project(a + beta * v)

        # Compute subgradient at y
        g, peak_val = compute_weighted_subgradient(y, top_k=top_k)

        # Polyak step size
        current_score = 2.0 * n * peak_val
        gap = max(current_score - lower_bound, 1e-10)
        g_norm_sq = np.dot(g, g)
        if g_norm_sq < 1e-20:
            break
        alpha = gap / (2.0 * n * g_norm_sq)
        alpha = min(alpha, lr)

        # Update
        v = beta * v - alpha * g
        a_new = simplex_project(a + v)

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Decay learning rate
        lr *= 0.9997

        # Restart if stuck
        if no_improve > 500:
            rng = np.random.default_rng((it * 7919 + 13) % 1000003)
            a = best_a + rng.standard_normal(n) * 0.005
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            lr = max(lr * 5.0, 0.003)
            no_improve = 0

        if it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovSG it={it} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def lbgfs_refine(a0, time_limit, T=100, maxiter=5000):
    """L-BFGS refinement in log-space with soft-max smoothing."""
    n = len(a0)

    def softmax_from_z(z):
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        return exp_z / exp_z.sum()

    def z_from_a(a_vec):
        a_safe = np.maximum(a_vec, 1e-300)
        return np.log(a_safe)

    def cross_corr(w, a_vec):
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a_vec, fft_size)
        result = irfft(np.conj(W) * A, fft_size)[:n]
        return result

    def soft_max_obj_grad(z):
        a_vec = softmax_from_z(z)
        b = fast_convolve(a_vec)
        max_b = b.max()
        exp_b = np.exp((b - max_b) / T)
        log_sum_exp = max_b / T + np.log(exp_b.sum())
        obj = (2.0 * n * T) * log_sum_exp
        w = exp_b / exp_b.sum()
        grad_a = 2.0 * n * cross_corr(w, a_vec)
        S = np.dot(grad_a, a_vec)
        grad_z = a_vec * (grad_a - a_vec * S)
        return obj, grad_z

    z0 = z_from_a(a0)
    t0 = time.monotonic()
    stopped = [False]

    def func(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        obj, _ = soft_max_obj_grad(z)
        return obj

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros_like(z)
        _, g = soft_max_obj_grad(z)
        return g

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                     'maxfun': maxiter * 50, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
    except Exception:
        a_out = a0.copy()

    return a_out

def coordinate_descent_refine(a0, time_limit, n_sweeps=50):
    """Coordinate descent: optimize one element at a time via 1D search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()

            break

            if time.monotonic() - t0 > time_limit:

            old_val = a[j]

                    a_without_j = a.copy()

                    a[j] = new_val
                    a[~(np.arange(n) == j)] = a_without_j[~(np.arange(n) == j)] * (1 - new_val) / (1 - old_val) if old_val > 0 else a_without_j[~(np.arange(n) == j)]
                    score = 2.0 * n * fast_convolve(a).max()
                    if score < best_score - 1e-14:
                        best_score = score
                        best_a = a.copy()
                    a[j] = old_val
                continue

            # Try varying a[j] while scaling others to maintain sum=1
            best_local = old_val

                a[j] = new_val

                score = 2.0 * n * fast_convolve(a).max()
                if score < best_local_score - 1e-14:
                    best_local_score = score

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def parametric_initializations(n):
    """Generate diverse parametric initializations."""
    candidates = []

    f = np.maximum(f, 0)
    candidates.append(f / f.sum())
    # Gaussian

    f = np.exp(-8 * (x - 0.5) ** 2)
    candidates.append(f / f.sum())
    # Two Gaussians
    x = np.arange(n) / n
    f = np.exp(-20 * (x - 0.3) ** 2) + np.exp(-20 * (x - 0.7) ** 2)
    candidates.append(f / f.sum())
    # Power law
    f = 1.0 / (np.arange(n) + 1) ** 0.5
    candidates.append(f / f.sum())
    # Sine wave
    x = np.arange(n) / n
    f = 0.5 + 0.5 * np.cos(2 * np.pi * x * 3)

    candidates.append(f / f.sum())
    # Triangular

    # Superposition

    # BW-like (Blackman window)

    candidates.append(f / f.sum())
    # Exponential decay
    f = np.exp(-2.0 * np.arange(n) / n)
    candidates.append(f / f.sum())
    # Symmetric exponential
    x = np.arange(n) / n
    f = np.exp(-3.0 * np.abs(x - 0.5) * 2)
    candidates.append(f / f.sum())
    return candidates


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Parametric initialization + quick selection
    2. LP-based fictitious play with interleaved subgradient
    3. Nesterov-accelerated subgradient refinement
    4. Adaptive-temperature L-BFGS polishing
    5. Peak shaving + coordinate descent
    6. Second FP round with refined start
    7. Multi-length exploration
    8. Final micro-polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    # === Phase 1: Quick initialization (30s) ===
    print("\n=== Phase 1: Initialization ===")
    t0 = time.monotonic()
    n_base = len(parent) if parent is not None else 768

    # Try parametric initializations at base length
    for i, init in enumerate(parametric_initializations(n_base)):
        if time.monotonic() - t0 > 25:
            break
        score = float(evaluate_sequence(init.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = init.copy()
            print(f"  Parametric init {i}: score={score:.8f}")

    # If parent is available, it's likely the best start
    if parent is not None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"  After init: best={best_score:.8f} (n={len(best_seq)})")

    # === Phase 2: FP with interleaved subgradient (700s) ===
    print(f"\n=== Phase 2: FP+SG (budget=700s) ===")
    t0 = time.monotonic()
    fp_budget = min(700.0, (deadline - time.monotonic()) * 0.45)
    print(f"  FP budget: {fp_budget:.0f}s, n={n_base}")

    a_fp, score_fp = fictitious_play_with_interleaved_subgrad(
        n_base, fp_budget, initial=best_seq, solver=solver
    )
    best_seq, best_score = check_and_update(a_fp, f"FP_n{n_base}", best_seq, best_score)
    elapsed = time.monotonic() - t0
    print(f"  FP done in {elapsed:.0f}s, best={best_score:.8f}")

    # === Phase 3: Nesterov subgradient (250s) ===
    print(f"\n=== Phase 3: Nesterov SG (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_alloc = min(rem * 0.5, 250.0)
            print(f"  Nesterov budget: {t_alloc:.0f}s")
            a_ns, _ = nesterov_subgradient_refine(best_seq, t_alloc, top_k=10, lr_init=0.004)
            best_seq, best_score = check_and_update(a_ns, "nesterov_sg", best_seq, best_score)

            # Second run with different top_k
            rem = deadline - time.monotonic() - 350
            if rem > 20:
                t_alloc2 = min(rem * 0.5, 100.0)
                a_ns2, _ = nesterov_subgradient_refine(best_seq, t_alloc2, top_k=20, lr_init=0.003)
                best_seq, best_score = check_and_update(a_ns2, "nesterov_sg_k20", best_seq, best_score)

    # === Phase 4: L-BFGS with adaptive temperature (250s) ===
    print(f"\n=== Phase 4: L-BFGS Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        temperatures = [200, 100, 50, 20, 10]
        for T in temperatures:
            rem = deadline - time.monotonic() - 200
            if rem < 15:
                break
            t_alloc = min(rem * 0.3, 60.0)
            print(f"  L-BFGS T={T}, budget={t_alloc:.0f}s")
            a_lb = lbgfs_refine(best_seq, t_alloc, T=T, maxiter=3000)
            best_seq, best_score = check_and_update(a_lb, f"lbgfs_T{T}", best_seq, best_score)

    # === Phase 5: Peak shaving + coordinate descent (150s) ===
    print(f"\n=== Phase 5: Local Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 15:
            t_ps = min(rem * 0.5, 80.0)
            a_ps, _ = peak_shave_local(best_seq, t_ps)
            best_seq, best_score = check_and_update(a_ps, "peak_shave", best_seq, best_score)

        rem = deadline - time.monotonic() - 60
        if rem > 10:
            t_cd = min(rem * 0.6, 50.0)
            a_cd, _ = coordinate_descent_refine(best_seq, t_cd, n_sweeps=30)
            best_seq, best_score = check_and_update(a_cd, "coord_descent", best_seq, best_score)

    # === Phase 6: Second FP round (200s) ===
    print(f"\n=== Phase 6: FP Round 2 (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 80
        if rem > 40:
            t_alloc = min(rem * 0.5, 200.0)
            print(f"  FP2 n={n_best}, budget={t_alloc:.0f}s")
            a_fp2, score_fp2 = fictitious_play_with_interleaved_subgrad(
                n_best, t_alloc, initial=best_seq, solver=solver
            )
            best_seq, best_score = check_and_update(a_fp2, "FP2", best_seq, best_score)

            # Quick L-BFGS after FP2
            rem = deadline - time.monotonic() - 40
            if rem > 15:
                a_lb2 = lbgfs_refine(best_seq, min(rem * 0.5, 40.0), T=20, maxiter=2000)
                best_seq, best_score = check_and_update(a_lb2, "lbgfs_after_fp2", best_seq, best_score)

    # === Phase 7: Multi-length exploration (100s) ===
    print(f"\n=== Phase 7: Multi-length (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        alt_lengths = []
        for offset in [-64, -32, 32, 64, 96, -96]:
            alt_n = n_best + offset
            if 256 <= alt_n <= 2048 and alt_n != n_best:
                alt_lengths.append(alt_n)

        for alt_n in alt_lengths[:4]:
            rem = deadline - time.monotonic() - 30
            if rem < 20:
                break
            t_alloc = min(rem * 0.4, 60.0)
            init_alt = interpolate_sequence(best_seq, alt_n)
            # Quick subgradient at new length
            a_alt, _ = nesterov_subgradient_refine(init_alt, t_alloc, top_k=10, lr_init=0.003)
            best_seq, best_score = check_and_update(a_alt, f"alt_n{alt_n}", best_seq, best_score)

    # === Phase 8: Final micro-polish (80s) ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_final = lbgfs_refine(best_seq, min(rem * 0.4, 30.0), T=10, maxiter=2000)
            best_seq, best_score = check_and_update(a_final, "final_lbgfs", best_seq, best_score)

        rem = deadline - time.monotonic() - 8
        if rem > 5:
            a_ps_final, _ = peak_shave_local(best_seq, min(rem * 0.7, 20.0))
            best_seq, best_score = check_and_update(a_ps_final, "final_ps", best_seq, best_score)

        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_sg_final, _ = nesterov_subgradient_refine(best_seq, min(rem * 0.7, 15.0), top_k=5, lr_init=0.002)
            best_seq, best_score = check_and_update(a_sg_final, "final_sg", best_seq, best_score)

    # Fallback
    if best_seq is None:
        if parent is not None:
            best_seq = parent
        else:
            best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
