    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)

def convolute_matrix(a):
    """Build the convolution matrix C such that C @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    C = np.zeros((size, n))
    for k in range(size):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            C[k, j] = a[j]
    return C

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

LP-based mirror descent for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses linear programming to find optimal descent directions, combined with
adaptive line search, multi-length exploration, and coordinate descent polishing.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def conv_matrix(a):
    """Build convolution matrix C where C @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    k = np.arange(size)[:, None]
    j = np.arange(n)[None, :]
    valid = (k - j) >= 0
    valid &= (k - j) < n
    C = np.where(valid, a[None, :], 0.0)
    return C


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def symmetrize(a):
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_raised_cosine(n):
    a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_blackman(n):
    i = np.arange(n)
    a = 0.42 - 0.5 * np.cos(2 * np.pi * i / (n - 1)) + 0.08 * np.cos(4 * np.pi * i / (n - 1))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_asymmetric_gaussian(n, center_frac=0.4, sigma_frac=0.25):
    x = np.arange(n) / n
    a = np.exp(-0.5 * ((x - center_frac) / sigma_frac) ** 2)
    a = a / a.sum()
    return a


def compute_objective_norm(a):
    """Compute 2*n*max(conv(a,a)) for a on the simplex (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def lp_mirror_descent(a0, time_limit, verbose=False, label=""):
    """
    LP-based mirror descent.
    At each step, solve an LP to find the best direction b,
    then interpolate towards b with adaptive step size.
    """
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        
        # Solve LP: max sum(b) s.t. C(a) @ b <= M, b >= 0
        C = conv_matrix(a)
        c_obj = -np.ones(n)
        b_ub = M * np.ones(2 * n - 1)
        
        try:
            res = linprog(c_obj, A_ub=C, b_ub=b_ub, 
                         bounds=[(0.0, None)] * n, 
                         method='highs',
                         options={'presolve': True, 'time_limit': 2.0})
        except Exception:
            # Fallback: use subgradient step
            g = a * conv  # rough subgradient
            b = project_simplex(-g)
            res = None
        
        if res is not None and res.success and res.x is not None:
            b = np.maximum(res.x, 0)
            b_sum = b.sum()
            if b_sum > 1e-10:
                b = b / b_sum
            else:
                b = a.copy()
        else:
            # Fallback: use the peak-reduction direction
            peak_idx = np.argmax(conv)
            js = max(0, peak_idx - n + 1)
            je = min(n - 1, peak_idx)
            b = a.copy()
            b[js:je+1] *= 0.8
            non_contrib = np.ones(n, dtype=bool)
            non_contrib[js:je+1] = False
            if non_contrib.sum() > 0:
                b[non_contrib] += 0.2 * a[js:je+1].sum() * a[non_contrib] / max(a[non_contrib].sum(), 1e-15)
            b = np.maximum(b, 0)
            b = b / b.sum()
        
        # Line search: find best t
        best_t = 0.0
        best_trial_obj = M
        for t in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
            a_trial = (1 - t) * a + t * b
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial = a_trial / a_trial.sum()
            trial_conv = fast_convolve(a_trial)
            trial_M = trial_conv.max()
            if trial_M < best_trial_obj:
                best_trial_obj = trial_M
                best_t = t
        
        if best_t > 0:
            a = (1 - best_t) * a + best_t * b
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            
            new_M = fast_convolve(a).max()
            new_obj = 2.0 * n * new_M
            
            if new_obj < best_obj - 1e-12:
                best_obj = new_obj
                best_a = a.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"    {label} LP iter {it}: obj={best_obj:.10f}, t={best_t:.3f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        # Restart if stuck
        if no_improve > 50:
            noise_scale = 0.001 * min(no_improve / 50, 5.0)
            rng = np.random.default_rng(it * 7919 + 13)
            a = np.maximum(best_a + rng.standard_normal(n) * noise_scale, 1e-15)
            a = a / a.sum()
            no_improve = 0
        
        # Symmetrize check every 200 iterations
        if it % 200 == 0:
            a_sym = symmetrize(a)
            sym_obj = 2.0 * n * fast_convolve(a_sym).max()
            if sym_obj < best_obj:
                best_obj = sym_obj
                best_a = a_sym.copy()
                a = a_sym.copy()
                no_improve = 0
    
    return best_a, best_obj


def coordinate_descent(a0, time_limit, verbose=False, label=""):
    """
    Fine coordinate descent: adjust one coordinate at a time.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            if old_val < 1e-12:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in [0.8, 0.9, 0.95, 1.05, 1.1, 1.25]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.3:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                trial = a.copy()
                trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                trial[mask] *= scale
                trial = np.maximum(trial, 0.0)
                ss = trial.sum()
                if ss < 1e-15:
                    continue
                trial = trial / ss
                val = 2.0 * n * fast_convolve(trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_a, best_obj


def subgradient_descent(a0, time_limit, top_k=20, temp=0.001, verbose=False, label=""):

    Smoothed subgradient descent as a diversity mechanism.
    """
    a = np.maximum(a0.astype(np.float64), 1e-15)

    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = 0.01
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        
        K = min(top_k, len(conv))
        if K < len(conv):
            peak_indices = np.argpartition(conv, -K)[-K:]
        else:
            peak_indices = np.arange(len(conv))
        
        pv = conv[peak_indices]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peak_indices))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        # Subgradient
        g = np.zeros(n)
        for idx in range(len(peak_indices)):
            k = int(peak_indices[idx])
            wk = w[idx]
            if wk < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue
            g[js:je+1] += 2.0 * wk * a[k - np.arange(js, je+1)]
        
        # Multiplicative update
        log_update = -lr * g
        log_update = np.clip(log_update, -50, 50)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        new_obj = 2.0 * n * fast_convolve(a_new).max()
        if new_obj < best_obj - 1e-12:
            best_obj = new_obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 50:
            rng = np.random.default_rng(it * 31337 + 7)
            a = np.maximum(best_a + rng.standard_normal(n) * 0.002, 1e-15)
            a = a / a.sum()
            lr = 0.005
            no_improve = 0
        
        lr *= 0.9995
        temp = min(temp * 1.002, 0.05)
        
        if it % 300 == 0:
            a_sym = symmetrize(a)
            sym_obj = 2.0 * n * fast_convolve(a_sym).max()
            if sym_obj < best_obj:
                best_obj = sym_obj
                best_a = a_sym.copy()
                a = a_sym.copy()
    
    return best_a, best_obj
