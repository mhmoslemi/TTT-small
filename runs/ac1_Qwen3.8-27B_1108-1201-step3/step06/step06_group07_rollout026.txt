"""
LP-guided optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses linear programming to find optimal improvement directions, combined with
gradient-based refinement and coordinate descent polish. Exploits the fact that
the optimal solution is necessarily asymmetric.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_A(a):
    """Build sparse matrix A where (A @ b)[k] = (a*b)[k] (linear cross-correlation)."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A.tocsr()


def solve_lp_direction(a, M):
    """Solve LP: max sum(b) s.t. (a*b)_k <= M for all k, b >= 0. Returns b* or None."""
    n = len(a)
    c = -np.ones(n)
    A = build_A(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True})
        if not result.success:
            return None
        return result.x
    except Exception:
        return None


def compute_gradient_smoothed(conv, a, T):
    """Compute gradient of T*log(sum(exp(conv_k/T))) w.r.t. a using FFT."""
    n = len(a)
    scaled = conv / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p /= p.sum()
    # grad_i = 2*T * sum_j p_{i+j} * a_j for i=0,...,n-1
    corr = np.correlate(p, a, mode='valid')
    return 2.0 * T * corr


def compute_subgradient(conv, a, top_k=20):
    """Compute a subgradient of max(conv) w.r.t. a, weighted over top peaks."""
    n = len(a)
    K = min(top_k, len(conv))
    if K < len(conv):
        peak_indices = np.argpartition(conv, -K)[-K:]
    else:
        peak_indices = np.arange(len(conv))
    
    # Use softmax weights over peaks
    pv = conv[peak_indices]
    temp = 0.001
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    # grad = 2 * sum_k w_k * a_{k-i} (valid indices)
    # = 2 * correlation of (w placed at peak indices) with a
    p = np.zeros(len(conv))
    p[peak_indices] = w
    corr = np.correlate(p, a, mode='valid')
    return 2.0 * corr


def project_simplex(v):
    """Euclidean projection onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_optimize(a0, time_limit, verbose=True):
    """Main LP-based optimization loop."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        current_obj = 2.0 * n * M
        
        # Solve LP
        b_star = solve_lp_direction(a, M)
        if b_star is None:
            if verbose:
                print(f"  LP failed at iter {it}")
            break
        
        sum_b = b_star.sum()
        if sum_b <= 1.0 + 1e-9:
            if verbose and it % 100 == 0:
                print(f"  LP optimum (sum_b={sum_b:.8f}) at iter {it}, obj={best_obj:.10f}")
            break
        
        # Normalized direction
        b_norm = b_star / sum_b
        
        # Line search
        best_t = 0.0
        best_trial_obj = current_obj
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            a_trial = (1.0 - t) * a + t * b_norm
            a_trial = np.maximum(a_trial, 0.0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial = a_trial / ss
            trial_conv = fast_convolve(a_trial)
            trial_obj = 2.0 * n * trial_conv.max()
            if trial_obj < best_trial_obj:
                best_trial_obj = trial_obj
                best_t = t
            if trial_obj < best_trial_obj * 0.999:
                break
        
        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * b_norm
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                if verbose:
                    print(f"  Stuck at iter {it}, obj={best_obj:.10f}. Restarting.")
                # Add small random perturbation
                rng = np.random.default_rng(it * 31 + 7)
                a = a + rng.standard_normal(n) * 0.001
                a = project_simplex(a)
                no_improve = 0
        
        if best_trial_obj < best_obj - 1e-12:
            best_obj = best_trial_obj
            best_a = a.copy()
            if verbose and (it % 100 == 0 or best_obj < 1.503):
                print(f"  LP iter {it}: obj={best_obj:.10f}, sum_b={sum_b:.6f}, t={best_t:.3f}")
    
    return best_a, best_obj


def gradient_optimize(a0, time_limit, T_init=0.1, T_min=0.001, lr_init=0.01,
                      verbose=True):
    """Gradient descent on the smoothed objective with annealing."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    T = T_init
    lr = lr_init
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        current_obj = 2.0 * n * M
        
        # Compute gradient of smoothed objective
        grad = compute_gradient_smoothed(conv, a, T)
        
        # Project gradient to tangent space of simplex
        grad_tangent = grad - grad.mean()
        
        # Backtracking line search
        best_lr = 0.0
        best_trial_obj = current_obj
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05]:
            if trial_lr < 1e-12:
                continue
            a_trial = a - trial_lr * grad_tangent
            a_trial = project_simplex(a_trial)
            trial_conv = fast_convolve(a_trial)
            trial_obj = 2.0 * n * trial_conv.max()
            if trial_obj < best_trial_obj:
                best_trial_obj = trial_obj
                best_lr = trial_lr
            if trial_obj < current_obj * 0.999:
                break
        
        if best_lr > 0:
            a = a - best_lr * grad_tangent
            a = project_simplex(a)
            no_improve = 0
        else:
            no_improve += 1
        
        if best_trial_obj < best_obj - 1e-12:
            best_obj = best_trial_obj
            best_a = a.copy()
            if verbose and (it % 200 == 0 or best_obj < 1.503):
                print(f"  Grad iter {it}: obj={best_obj:.10f}, T={T:.6f}, lr={best_lr:.6f}")
        
        # Anneal temperature
        T = max(T * 0.999, T_min)
        
        # Adapt learning rate
        if no_improve > 100:
            lr *= 0.5
            no_improve = 0
            if lr < 1e-10:
                break
        lr = max(lr * 0.9999, 1e-10)
        
        # Restart if very stuck
        if no_improve > 500:
            rng = np.random.default_rng(it * 7919 + 13)
            a = best_a + rng.standard_normal(n) * 0.001
            a = project_simplex(a)
            lr = lr_init * 0.5
            no_improve = 0
    
    return best_a, best_obj


def coordinate_descent(a0, time_limit, verbose=True):
    """Coordinate descent: optimize one coordinate at a time."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rounds = 0
    
    while time.monotonic() - t0 < time_limit:
        rounds += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            if old_val < 1e-12:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            # Try several multipliers
            for mult in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                trial = a.copy()
                trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                trial[mask] *= scale
                trial = np.maximum(trial, 0.0)
                ss = trial.sum()
                if ss < 1e-15:
                    continue
                trial = trial / ss
                val = fast_convolve(trial).max() * 2.0 * n
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            break
        
        if verbose and rounds % 5 == 0:
            print(f"  CD round {rounds}: obj={best_obj:.10f}")
    
    return best_a, best_obj


def make_initial(n, style="asym"):

    elif style == "gaussian":
        x = np.arange(n) - n / 2.0
        sigma = n * 0
