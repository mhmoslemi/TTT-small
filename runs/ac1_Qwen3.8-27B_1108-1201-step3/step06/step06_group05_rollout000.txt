def compute_lse_gradient(f, c, beta, top_k=None):
    """
    Compute gradient of log-sum-exp smoothed max using top-k peaks.
    Returns gradient vector of length len(f).
    """
    n = len(f)
    if top_k is not None and top_k < len(c):
        # Only use top-k peaks for a sharper gradient
        idx = np.argpartition(c, -top_k)[-top_k:]
        c_top = c[idx]
        max_c = c_top.max()
        scaled = beta * (c_top - max_c)
        exp_scaled = np.exp(scaled)
        w = exp_scaled / exp_scaled.sum()
        # Build sparse weight vector
        w_full = np.zeros(len(c))
        w_full[idx] = w
    else:
        max_c = c.max()
        scaled = beta * (c - max_c)
        exp_scaled = np.exp(scaled)
        w_full = exp_scaled / exp_scaled.sum()

    indices = np.arange(2 * n - 2, 2 * n - 2 - n, -1)

    no_improve = 0

    beta_schedule = [4, 8, 16, 32, 64, 128, 256, 512, 1024]
    top_k_schedule = [50, 40, 30, 20, 15, 10, 8, 5, 3]

    for i, (beta, top_k) in enumerate(zip(beta_schedule, top_k_schedule)):
        if time.monotonic() - t0 > time_limit * 0.9:
            break

        stage_start = time.monotonic()
        stage_time = time_limit * 0.07 * (1.0 + 0.3 * i)
        it = 0
        lr = 5.0 / (n ** 0.5)

        while time.monotonic() - stage_start < stage_time and time.monotonic() - t0 < time_limit * 0.9:
            it += 1
            c = fast_convolve(f)
            g = compute_lse_gradient(f, c, beta, top_k)
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-15:
                break

            momentum = mu * momentum - lr * g / g_norm
            f_trial = project_simplex(f + momentum)

            c_trial = fast_convolve(f_trial)
            obj_trial = c_trial.max()

            if obj_trial <= c.max() + 1e-12:
                f = f_trial
                if obj_trial < best_obj - 1e-14:
                    best_obj = obj_trial
                    best_f = f.copy()
                    no_improve = 0
                    lr = min(lr * 1.03, 2.0)
                else:
                    no_improve += 1
            else:
                no_improve += 1
                lr *= 0.8
                momentum *= 0.5

            if no_improve > 60:
                rng = np.random.default_rng(it * 13 + i * 7)
                noise_scale = 0.003 / np.sqrt(beta / 4.0 + 1)
                f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
                momentum = np.zeros(n)
                lr = 3.0 / (n ** 0.5)
                no_improve = 0

        if verbose:
            print(f"    SGD beta={beta} top_k={top_k}: best={best_obj:.10f}")

    return best_f, best_obj


def build_lp_matrix(f, n):
    """Build the constraint matrix for the LP: (f*g)_k <= M for all k."""
    # A[k, l] = f[k-l] if 0 <= k-l < n and 0 <= l < n
    # k ranges from 0 to 2n-2, l ranges from 0 to n-1
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        l_min = max(0, k - n + 1)
        l_max = min(n - 1, k)
        for l in range(l_min, l_max + 1):
            A[k, l] = f[k - l]
    return A


def build_lp_matrix_fast(f, n):
    """Build constraint matrix efficiently using toeplitz structure."""
    rows = 2 * n - 1
    cols = n
    A = np.zeros((rows, cols))
    # For each column l, A[k, l] = f[k-l] for k in [l, l+n-1]
    for l in range(cols):
        A[l:l + n, l] = f
    return A


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c_obj = -np.ones(n)
    A_ub = build_lp_matrix_fast(f, n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def golden_section_search(f, g, n, c_current, M_current, max_iters=50):
    """
    Golden-section line search for alpha in [0, 1] minimizing
    max(f*f + alpha*(g-g)*f + ...) -- actually we evaluate the full objective.
    """
    gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio
    a = 0.0
    b = 1.0
    c_pt = b - gr * (b - a)
    d_pt = a + gr * (b - a)

    def obj_at(alpha):
        ft = f + alpha * (g - f)
        ft = np.maximum(ft, 0.0)
        st = ft.sum()
        if st < 1e-15:
            return float('inf')
        ft = ft / st
        return fast_convolve(ft).max()

    fc = obj_at(c_pt)
    fd = obj_at(d_pt)

    for _ in range(max_iters):
        if fc < fd:
            b = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = b - gr * (b - a)
            fc = obj_at(c_pt)
        else:
            a = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = a + gr * (b - a)
            fd = obj_at(d_pt)

        if abs(b - a) < 1e-10:
            break

    best_alpha = (a + b) / 2.0
    return best_alpha


def lp_fw_optimize(f0, time_limit, verbose=False):
    """
    LP-based Frank-Wolfe with golden-section line search.
    Minimizes max_k (f*f)_k subject to f>=0, sum(f)=1.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    c = fast_convolve(f)
    M = c.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_factor = 1.0
    best_lp_time = 0.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        lp_start = time.monotonic()
        g_dir = solve_lp_direction(f, M * m_factor, n)
        lp_time = time.monotonic() - lp_start
        best_lp_time = max(best_lp_time, lp_time)

        if g_dir is None:
            m_factor *= 1.15
            if m_factor > 8.0:
                rng = np.random.default_rng(it * 19 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                c = fast_convolve(f)
                M = c.max()
                m_factor = 1.0
            continue

        g_sum = g_dir.sum()
        if g_sum < 1e-15:
            m_factor *= 1.08
            continue
        g_dir = g_dir / g_sum

        # Check fixed point
        diff = np.max(np.abs(g_dir - f))
        if diff < 1e-13:
            no_improve += 1
            if no_improve > 30:
                m_factor = min(m_factor * 1.05, 6.0)
            if no_improve > 100:
                rng = np.random.default_rng(it * 23 + 11)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                c = fast_convolve(f)
                M = c.max()
                m_factor = 1.0
                no_improve = 0
            if no_improve > 300:
                break
            continue
        else:
            if diff < 1e-9:
                no_improve = min(no_improve + 1, 50)
            else:
                no_improve = 0

        # Golden section line search
        best_alpha = golden_section_search(f, g_dir, n, c, M)

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g_dir - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            c = fast_convolve(f_new)
            M = c.max()
            f = f_new
            m_factor = 1.0

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 20 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.12f}, alpha={best_alpha:.4f}")
        else:
            no_improve += 1
            m_factor = min(m_factor * 1.03, 5.0)

    return best_f, best_obj


def entropic_fw_optimize(f0, time_limit, verbose=False):
    """
    Entropic Frank-Wolfe (multiplicative weights) variant.
    Uses a different geometry for potentially better directions.
    """
    f = np.maximum(f0.astype(np.float64), 1e-10)
    f = f / f.sum()
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    t0 = time.monotonic()
    lr = 0.05
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(f)
        M = c.max()

        # Compute weighted subgradient using top peaks
        k_peaks = min(20, len(c))
        peak_idx = np.argpartition(c, -k_peaks)[-k_peaks:]
        excess = np.maximum(c[peak_idx] - M * 0.95, 0)
        if excess.sum() < 1e-15:
            excess = np.ones(len(peak_idx))
        w = excess / excess.sum()

        # Subgradient: sg_j = 2 * sum_k w_k * f[k-j]
        sg = np.zeros(n)
        for i in range(len(peak_idx)):
            if w[i] < 1e-12:
                continue
            k = int(peak_idx[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += w[i] * f[k - idx]
        sg *= 2.0

        # Multiplicative weight update
        f_new = f * np.exp(-lr * sg)
        f_new = np.maximum(f_new, 1e-12)
        f_new = f_new / f_new.sum()

        c_new = fast_convolve(f_new)
        obj_new = c_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.02, 0.15)
        else:
            no_improve += 1
            lr *= 0.97

        if no_improve > 150:
            rng = np.random.default_rng(it * 31)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.002, 1e-10)
            f = f / f.sum()
            lr = 0.03
            no_improve = 0

        if lr < 1e-5:
            break

    return best_f, best_obj


def coordinate_polish(f0, time_limit, verbose=False):
    """Coordinate descent for final local refinement."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f0, fast_convolve(f0).max()
    f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-14:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue

                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial = f_trial / ss

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian-shaped initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6, power=4):
    """Bell-shaped initial sequence."""
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** power
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_raised_cosine(n):
    """Raised cosine (Hann window) initial sequence."""
    x = np.linspace(-1, 1, n)
    a = 0.5 * (1 + np.cos(np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_flat_top(n, flat_frac=0.5):
    """Flat-top window."""
    a = np.ones(n)
    # Taper the edges
    edge = int(n * (1 - flat_frac) / 2)
    if edge > 0:
        taper = np.linspace(0, 1, edge)
        a[:edge] = taper
        a[-edge:] = taper[::-1]
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_periodic_smooth(n, period=4, amplitude=0.3):
    """Periodic modulation of uniform sequence."""
    base = np.ones(n) / n
    mod = 1.0 + amplitude * np.sin(2 * np.pi * np.arange(n) / period)
    a = base * mod
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate to new length with smoothing."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    if new_n >= 5:
        kernel = np.array([0.05, 0.15, 0.6, 0.15, 0.05])
        f_smooth = np.convolve(f_interp, kernel, mode='same')
        f_smooth = np.maximum(f_smooth, 0)
    else:
        f_smooth = f_interp
    ss = f_smooth.sum()
    if ss > 1e-15:
        f_smooth /= ss
    return f_smooth


def fourier_flatten(f, n_iter=10, strength=0.3):
    """
    Flatten the auto-correlation by operating in Fourier domain.
    Reduces high-frequency peaks in the spectrum.
    """
    f = np.array(f, dtype=np.float64)
    n = len(f)
    for _ in range(n_iter):
        # Compute auto-correlation
        c = fast_convolve(f)
        # Target: make c flatter
        c_target = np.ones(2 * n - 1) * c.mean()
        # Blend: reduce peaks by adjusting f in Fourier domain
        F = np.fft.rfft(f)
        # Attenuate frequencies that contribute most to peaks
        # This is a heuristic: reduce |F|^2 at frequencies where it's large
        power = np.abs(F) ** 2
        max_power = power.max()
        if max_power > 0:
            # Compress the power spectrum
            compressed = power ** (1.0 - strength * 0.1)
            ratio = np.sqrt(compressed / (power + 1e-15))
            F_new = F * ratio
            f_new = np.fft.irfft(F_new, n)
            f_new = np.maximum(f_new, 0)
            ss = f_new.sum()
            if ss > 1e-15:
                f = f_new / ss
    return f

    if best_seq is not None:
        n_best = len(best_seq)
        restart_configs = [
            ("direct", best_seq.copy()),
            ("noise_small", None),
            ("noise_med", None),
            ("reverse_blend", None),
            ("fourier_blend", None),
        ]

        for name, f_init in restart_configs:
            rem = deadline - time.monotonic() - 30
            if rem < 30:
                break

            if f_init is None:
                if name == "noise_small":
                    f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.003)
                elif name == "noise_med":
                    f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.01)
                elif name == "reverse_blend":
                    f_init = 0.7 * best_seq + 0.3 * best_seq[::-1]
                    f_init = f_init / f_init.sum()
                elif name == "fourier_blend":
                    ff = fourier_flatten(best_seq, n_iter=3, strength=0.3)
                    f_init = 0.8 * best_seq + 0.2 * ff
                    f_init = f_init / f_init.sum()

            t_alloc = min(rem / max(1, len(restart_configs) - restart_configs.index((name, _))), 80)
            print(f"  LP-FW [{name}]: t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(f_init, time_limit=t_alloc)
            try_seq(a_opt, f"deep_{name}")

    # === Phase 7: Final coordinate polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 20:
            t_alloc = min(rem * 0.5, 40)
            a_opt, _ = coordinate_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "coord_final")

        # Final LP-FW squeeze
        rem = deadline - time.monotonic() - 5
        if rem > 15:
            t_alloc = min(rem, 30)
            a_opt2, _ = lp_fw_optimize(a_opt, time_limit=t_alloc)
            try_seq(a_opt2, "final_lpfw")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
