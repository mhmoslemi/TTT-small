"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an active-set convex QP method
with adaptive peak weighting. The key insight is that the problem is convex
(normalized to simplex), so we can use a sequence of QP subproblems that
progressively include more peak locations in the active set.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_Q_matrix(S, weights, n):
    """Build n×n matrix Q where Q[i,j] = sum of weights[k] for k in S with i+j=k."""
    Q = np.zeros((n, n))
    for idx, k in enumerate(S):
        w = weights[idx]
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            Q[i, j] += w
    return Q

def active_set_qp(a0, time_limit, max_iter=5000, K_start=10, K_max=200):
    """Active set QP method for minimizing max convolution."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Initial peak set
    b = fast_convolve(a)
    top_k = np.argsort(b)[::-1][:K_start]
    S = list(top_k)
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute weights: exponential based on peak height
        b_vals = b[S]
        b_max = b_vals.max()
        b_min = b_vals.min()
        # Weights proportional to peak height (emphasize higher peaks)
        if b_max > b_min + 1e-15:
            weights = np.exp(3.0 * (b_vals - b_max) / (b_max - b_min + 1e-15))
        else:
            weights = np.ones(len(S))
        weights = weights / weights.sum()
        
        # Build Q matrix
        Q = build_Q_matrix(S, weights, n)
        
        # Solve QP
        a_var = cp.Variable(n, nonneg=True)
        objective = cp.quad_form(a_var, cp.psd_wrap(Q))
        constraints = [cp.sum(a_var) == 1]
        prob = cp.Problem(cp.Minimize(objective), constraints)
        
        try:
            prob.solve(solver=cp.CLARABEL)  # or whatever is available
        except:
            try:
                prob.solve(solver=cp.SCS)
            except:
                try:
                    prob.solve(solver=cp.ECOS)
                except:
                    continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        
        a_new = np.asarray(a_var.value, dtype=np.float64)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        # Check convergence
        if np.linalg.norm(a_new - a) < 1e-10:
            break
        
        a = a_new
        b = fast_convolve(a)
        score = 2.0 * n * b.max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Update active set
        # Include top-K peaks + any peaks that are within 1% of max
        threshold = b.max() * 0.99
        S_new = set(np.where(b >= threshold)[0].tolist())
        top_idx = np.argsort(b)[::-1][:K_max]
        S_new = S_new | set(top_idx.tolist())
        S = sorted(S_new)
        
        # Optionally increase K
        K_current = min(K_start + it // 20, K_max)
        if len(S) < K_current:
            extra = np.argsort(b)[::-1][:K_current]
            S = sorted(set(S) | set(extra.tolist()))
    
    return best_a, best_score

3. If any a[i] < 0 for i in A, remove the most negative one from A
4. If any 2Q*a + λ > 0 for i not in A, add the most positive one to A
5. Repeat until convergence

This is the "simplex method" for QP. It can be slow in the worst case, but in practice works well.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using active-set convex QP 
progressively expanding the peak set, combined with L-BFGS log-sum-exp 
polishing and subgradient refinement. Exploits the convexity of the 
normalized problem (simplex constraint + convex objective).
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_Q_matrix(peak_indices, weights, n):
    """Build n×n matrix Q where Q[i,j] = sum_{k: i+j=k} weights[k]."""
    Q = np.zeros((n, n))
    for k, w in zip(peak_indices, weights):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            if 0 <= j < n:
                Q[i, j] += w
    return Q


def get_solver():
    """Find a working QP solver."""
    for solver_name in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.CVXOPT, cp.MOSEK, cp.GUROBI]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum_squares(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def solve_qp(a0, S, weights, solver):
    """Solve the QP: min a^T Q a s.t. a >= 0, sum(a) = 1."""
    n = len(a0)
    Q = build_Q_matrix(S, weights, n)
    a_var = cp.Variable(n, nonneg=True)
    obj = 0.5 * cp.quad_form(a_var, cp.psd_wrap(Q))
    constraints = [cp.sum(a_var) == 1]
    prob = cp.Problem(cp.Minimize(obj), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        for alt in [cp.SCS, cp.ECOS, cp.CVXOPT]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    a_val = np.asarray(a_var.value, dtype=np.float64)
    a_val = np.maximum(a_val, 0)
    s = a_val.sum()
    if s < 1e-15:
        return None
    return a_val / s


def active_set_qp_optimize(a0, time_limit, solver, K_start=10, K_max=100, 
                            weight_exp=3.0, verbose=True):
    """
    Active-set QP: iteratively solve QP minimizing weighted sum of peaks,
    growing the peak set adaptively.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    iter_count = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        b = fast_convolve(a)
        current_max = b.max()
        
        # Determine active set: peaks within threshold of max + top-K
        threshold = current_max * 0.95  # Include peaks within 95% of max
        above_thresh = np.where(b >= threshold)[0]
        
        # Adaptive K
        K = min(K_start + iter_count // 10, K_max)
        top_K = np.argsort(b)[::-1][:K]
        
        S = sorted(set(above_thresh.tolist()) | set(top_K.tolist()))
        
        # Compute weights (emphasize higher peaks)
        b_S = b[S]
        max_bS = b_S.max()
        if max_bS > 1e-15:
            # Exponential weights favoring higher peaks
            normalized = b_S / max_bS  # in [0, 1]
            weights = normalized ** weight_exp
            total_w = weights.sum()
            if total_w > 1e-15:
                weights = weights / total_w
            else:
                weights = np.ones(len(S)) / len(S)
        else:
            weights = np.ones(len(S)) / len(S)
        
        # Solve QP
        a_new = solve_qp(a, S, weights, solver)
        if a_new is None:
            break
        
        # Check improvement
        b_new = fast_convolve(a_new)
        score_new = 2.0 * n * b_new.max()
        
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Check convergence
        diff = np.linalg.norm(a_new - a)
        a = a_new
        
        if diff < 1e-10:
            if verbose:
                print(f"    AQP converged at iter {iter_count} (diff={diff:.2e})")
            break
        
        if no_improve > 30:
            # Perturb and continue
            if verbose:
                print(f"    AQP no improvement for 30 iters, perturbing...")
            rng = np.random.default_rng(iter_count * 12345)
            a = best_a + rng.standard_normal(n) * 0.001
            a = np.maximum(a, 0)
            a = a / a.sum()
            no_improve = 0
        
        if verbose and iter_count % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AQP iter={iter_count} ({elapsed:.0f}s): "
                  f"score={best_score:.6f} K={len(S)} diff={diff:.2e}")
    
    return best_a, best_score


def lbgfs_polish(a0, time_limit, T_list=None, verbose=True):
    """L-BFGS with log-sum-exp smoothing, sweeping temperature down."""
    if T_list is None:
        T_list = [100, 30, 10, 3, 1]
    
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an active-set convex QP method
with adaptive peak weighting and progressive set expansion, combined with
L-BFGS log-sum-exp polishing at multiple temperatures, FFT-accelerated
subgradient refinement, and multi-length exploration. Exploits the convexity
of the normalized problem (simplex constraint + convex max-of-quadratics objective).
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_corr_fft(w, a):
    """Compute c[m] = sum_k w[k] * a[k-m] for m in [0, n-1] using FFT."""
    n = len(a)
    size = len(w) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    B = rfft(a[::-1], fft_size)
    conv = irfft(W * B, fft_size)
    return conv[n - 1:n - 1 + n]


def build_Q_matrix(peak_indices, weights, n):
    """Build n×n symmetric matrix Q where Q[i,j] = sum_{k in peaks: i+j=k} w_k."""
    Q = np.zeros((n, n))
    for k, w in zip(peak_indices, weights):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            if 0 <= j < n:
                Q[i, j] += w
    return (Q + Q.T) / 2.0  # Ensure symmetry


def get_qp_solver():
    """Find a working QP solver."""
    global _qp_solver
    if '_qp_solver' in globals():
        return _qp_solver
    for solver_name in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.CVXOPT, cp.MOSEK, cp.GUROBI]:
        try:
            x = cp.Variable(20, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum_squares(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _qp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _qp_solver = cp.SCS
    return cp.SCS


def solve_qp_subproblem(peak_indices, weights, n, solver):
    """Solve: min a^T Q a s.t. a >= 0, sum(a) = 1, where Q[i,j] = sum_{k:i+j=k} w_k."""
    Q = build_Q_matrix(peak_indices, weights, n)
    a_var = cp.Variable(n, nonneg=True)
    obj = 0.5 * cp.quad_form(a_var, cp.psd_wrap(Q))
    constraints = [cp.sum(a_var) == 1]
    prob = cp.Problem(cp.Minimize(obj), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        for alt in [cp.SCS, cp.ECOS, cp.CVXOPT, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    a_val = np.asarray(a_var.value, dtype=np.float64)
    a_val = np.maximum(a_val, 0)
    s = a_val.sum()
    if s < 1e-15:
        return None
    return a_val / s


def active_set_qp_optimize(a0, time_limit, solver, K_start=8, K_max=80,
                           weight_power=4.0, threshold_frac=0.93, verbose=True):
    """
    Active-set QP: iteratively solve QP minimizing weighted sum of top peaks,
    growing the peak set adaptively. This is a bundle-method-like approach
    for the convex problem min max_k (a*a)[k] s.t. a in simplex.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    iter_count = 0
    no_improve = 0
    prev_S = set()

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        b = fast_convolve(a)
        current_max = b.max()

        # Determine active set
        threshold = current_max * threshold_frac
        above_thresh = set(np.where(b >= threshold)[0].tolist())

        K = min(K_start + iter_count * 2, K_max)
        top_K = set(np.argsort(b)[::-1][:K].tolist())

        S = sorted(above_thresh | top_K)

        # Compute weights emphasizing higher peaks
        b_S = b[S]
        max_bS = b_S.max()
        if max_bS > 1e-15:
            normalized = b_S / max_bS
            weights = normalized ** weight_power
            total_w = weights.sum()
            if total_w > 1e-15:
                weights = weights / total_w
            else:
                weights = np.ones(len(S)) / len(S)
        else:
            weights = np.ones(len(S)) / len(S)

        # Solve QP
        a_new = solve_qp_subproblem(S, weights, n, solver)
        if a_new is None:
            if time.monotonic() - t0 > 60:
                break
            continue

        # Evaluate
        b_new = fast_convolve(a_new)
        score_new = 2.0 * n * b_new.max()

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        diff = np.linalg.norm(a_new - a)
        a = a_new
        current_S = set(S)

        # Convergence checks
        if diff < 1e-11 and current_S == prev_S:
            if verbose:
                print(f"    AQP converged iter={iter_count} diff={diff:.2e}")
            break

        prev_S = current_S

            a = best_a + rng.standard_normal(n) * 0.002

            a = a / a.sum()
            no_improve = 0

            print(f"    AQP it={iter_count} ({elapsed:.0f}s) score={best_score:.6f} "

            a_new /= a_new.sum()
        except Exception as e:
            if verbose:
                print(f"    L-BFGS T={T} error: {e}")
            continue

        a_new = np.maximum(a_new, 0)
        a_new /= a_new.sum()
        score_new = 2.0 * n * fast_convolve(a_new).max()

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = best_a  # Don't use worse point as next starting point

        if verbose:
            print(f"    L-BFGS T={T} ({time.monotonic()-t0:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def subgradient_polish(a0, time_limit, verbose=True):
    """FFT-accelerated subgradient with softmax peak weighting and momentum."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    v = np.zeros(n)
    mom = 0.9
    lr = 0.003
    t0 = time.monotonic()
    no_improve = 0
    temp = 0.003

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        # Softmax weighting over all peaks
        Tb = temp * b
        exp_Tb = np.exp(Tb - Tb.max())
        w = exp_Tb / exp_Tb.sum()

        # Subgradient: grad_f[m] = 2 * sum_k w[k] * a[k-m]
        grad_f = 2.0 * cross_corr_fft(w, a)

        # Mirror descent (exponential update)
        v = mom * v - lr * grad_f
        a_new = np.exp(a / max(a.max(), 1e-10) * 0 + v)  # linear step then project
        # Actually use: a_new ~ exp(eta * (-grad)) projected to simplex
        # Simpler: add step and project
        a_trial = a + v
        # Project to simplex
        u = np.sort(a_trial)[::-1]
        cssv = np.cumsum(u)
        cond = u > (cssv - 1) / np.arange(1, n + 1)
        if cond.any():
            rho = np.nonzero(cond)[0][-1]
            theta = (cssv[rho] - 1.0) / (rho + 1.0)
            a_new = np.maximum(a_trial - theta, 0.0)
        else:
            a_new = np.ones(n) / n
        a_new /= a_new.sum()

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.999
        temp *= 0.9995

        if no_improve > 200:
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 100000)
            a = best_a + rng.standard_normal(n) * 0.001
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            lr = max(lr * 5, 0.001)
            no_improve = 0

        if verbose and (int((time.monotonic() - t0) * 10) % 50 == 0):
            print(f"    SubGrad ({time.monotonic()-t0:.0f}s): score={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score


def peak_shave(a0, time_limit, verbose=True):
    """Targeted peak reduction: reduce coordinates contributing most to the max peak."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        # Find current max peak (prefer interior)
        b_search = b.copy()
        b_search[0] = -np.inf
        b_search[-1] = -np.inf
        peak_k = int(np.argmax(b_search))

        if b[peak_k] <= best_score / (2.0 * n) * 1.001:
            break

        # Find top contributing pairs
        i_start = max(0, peak_k - n + 1)
        i_end = min(n - 1, peak_k)
        idx = np.arange(i_start, i_end + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_idx = np.argsort(contribs)[::-1][:5]

        improved = False
        for ci in top_idx:
            i = idx[ci]
            j = peak_k - i
            if a[i] < 1e-14:
                continue
            for factor in [0.97, 0.93, 0.88, 0.8]:
                a_trial = a.copy()
                old_val = a_trial[i]
                a_trial[i] = old_val * factor
                diff = old_val - a_trial[i]
                # Redistribute to other non-zero elements
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] += diff * a_trial[mask] / others_sum
                a_trial = np.maximum(a_trial, 0)
                a_trial /= a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_a = a.copy()
                    best_score = score
                    improved = True
                    break
            if improved:
                break

        # Try reducing both i and j
        if not improved:
            for ci in top_idx[:3]:
                i = idx[ci]
                j = peak_k - i
                if i == j or a[j] < 1e-14:
                    continue
                for f1, f2 in [(0.95, 0.95), (0.9, 0.9), (0.85, 0.9)]:
                    a_trial = a.copy()
                    diff1 = a_trial[i] * (1 - f1)
                    diff2 = a_trial[j] * (1 - f2)
                    a_trial[i] *= f1
                    a_trial[j] *= f2
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    mask[j] = False
                    others_sum = a_trial[mask].sum()
                    if others_sum < 1e-15:
                        continue
                    a_trial[mask] += (diff1 + diff2) * a_trial[mask] / others_sum
                    a_trial = np.maximum(a_trial, 0)
                    a_trial /= a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_a = a.copy()
                        best_score = score
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            if stall > 5:
                break
        else:
            stall = 0

        if verbose and (int((time.monotonic() - t0) * 5) % 10 == 0):
            print(f"    PeakShave ({time.monotonic()-t0:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate candidate and update best if improved."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate():
    """
    Multi-phase optimization:
    1. Active-set convex QP (main workhorse) at multiple lengths.
    2. L-BFGS log-sum-exp polish at decreasing temperatures.
    3. Subgradient and peak-shaving local polish.
    4. Additional QP rounds with refined starts.
    """
    deadline = time.monotonic() + 1800 - 30  # 30s safety margin

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"No parent found, using uniform n=512, score={best_score:.6f}")

    solver = get_qp_solver()
    print(f"Using QP solver: {solver}")

    # Define lengths to try
    n_parent = len(parent)
    n_values = sorted(set([
        n_parent,
        max(128, int(n_parent * 0.75)),
        min(3000, int(n_parent * 1.25)),
        min(3000, int(n_parent * 1.5)),
    ]), key=lambda x: abs(x - n_parent))

    # === Phase 1: Active Set QP ===
    print(f"\n{'='*60}")
    print(f"Phase 1: Active Set QP")
    print(f"{'='*60}")

    total_qp_time = min(700.0, (deadline - time.monotonic()) * 0.45)
    time_per_n = total_qp_time / len(n_values)

    for n_try in n_values:
        rem = deadline - time.monotonic() - 500
        if rem < 60:
            break
        t_alloc = min(time_per_n, rem * 0.85)
        if t_alloc < 30:
            continue

        print(f"\n  AQP n={n_try}, budget={t_alloc:.0f}s")
        initial = interpolate_sequence(parent, n_try)
        a_q, s_q = active_set_qp_optimize(
            initial, t_alloc, solver,
            K_start=8, K_max=80, weight_power=4.0,
            threshold_frac=0.93, verbose=True
        )
        best_seq, best_score = check_and_update(a_q, f"AQP_n{n_try}", best_seq, best_score)

    # === Phase 2: L-BFGS Polish ===
    print(f"\n{'='*60}")
    print(f"Phase 2: L-BFGS Polish (best={best_score:.6f})")
    print(f"{'='*60}")

    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_alloc = min(rem * 0.6, 350.0)
            print(f"  L-BFGS n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_lb, _ = lbgfs_polish(best_seq, t_alloc, verbose=True)
            best_seq, best_score = check_and_update(a_lb, "L-BFGS", best_seq, best_score)

    # === Phase 3: Subgradient + Peak Shaving ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Micro-polish (best={best_score:.6f})")
    print(f"{'='*60}")

    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_sg = min(rem * 0.4, 100.0)
            print(f"  Subgradient n={len(best_seq)}, budget={t_sg:.0f}s")
            a_sg, _ = subgradient_polish(best_seq, t_sg, verbose=True)
            best_seq, best_score = check_and_update(a_sg, "SubGrad", best_seq, best_score)

        rem = deadline - time.monotonic() - 80
        if rem > 15:
            t_ps = min(rem * 0.6, 80.0)
            print(f"  PeakShave n={len(best_seq)}, budget={t_ps:.0f}s")
            a_ps, _ = peak_shave(best_seq, t_ps, verbose=True)
            best_seq, best_score = check_and_update(a_ps, "PeakShave", best_seq, best_score)

    # === Phase 4: Second QP round with refined start ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Refined QP (best={best_score:.6f})")
    print(f"{'='*60}")

    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 60:
            t_alloc = min(rem * 0.6, 250.0)
            n_best = len(best_seq)
            print(f"  AQP round 2: n={n_best}, budget={t_alloc:.0f}s")
            # Use different parameters for diversity
            a_q2, _ = active_set_qp_optimize(
                best_seq, t_alloc, solver,
                K_start=15, K_max=120, weight_power=6.0,
                threshold_frac=0.90, verbose=True
            )
            best_seq, best_score = check_and_update(a_q2, "AQP_r2", best_seq, best_score)

            # Quick L-BFGS after QP
            rem = deadline - time.monotonic() - 50
            if rem > 20 and best_seq is not None:
                a_lb2, _ = lbgfs_polish(best_seq, min(rem * 0.7, 100.0), verbose=True)
                best_seq, best_score = check_and_update(a_lb2, "L-BFGS_r2", best_seq, best_score)

    # === Phase 5: Try one more length ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Alternative Length (best={best_score:.6f})")
    print(f"{'='*60}")

    if best_seq is not None:
        rem = deadline - time.monotonic() - 40
        if rem > 60:
            n_best = len(best_seq)
            # Try a longer sequence
            alt_n = min(2400, int(n_best * 1.4))
            if alt_n != n_best and alt_n >= 256:
                t_alloc = min(rem * 0.5, 150.0)
                init_alt = interpolate_sequence(best_seq, alt_n)
                print(f"  AQP alt: n={alt_n}, budget={t_alloc:.0f}s")
                a_alt, _ = active_set_qp_optimize(
                    init_alt, t_alloc, solver,
                    K_start=10, K_max=100, weight_power=4.0,
                    threshold_frac=0.92, verbose=False
                )
                best_seq, best_score = check_and_update(a_alt, f"AQP_alt_{alt_n}", best_seq, best_score)

    # === Phase 6: Final ultra-fine polish ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Final Polish (best={best_score:.6f})")
    print(f"{'='*60}")

    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            # Very brief peak shaving
            t_final = min(rem * 0.7, 30.0)
            a_fin, _ = peak_shave(best_seq, t_final, verbose=False)
            best_seq, best_score = check_and_update(a_fin, "FinalPS", best_seq, best_score)

    # Fallback
    if best_seq is None:
        best_seq = parent if parent is not None else np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence([float(x) for x in best_seq]))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
