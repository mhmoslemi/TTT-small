            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            c = fast_convolve(f_new)
            M = c.max()
            f = f_new
            m_expand = 1.0
            if M < best_obj:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 10 == 0:
                    print(f"      LP iter {it}: obj={best_obj:.12f}")
        else:
            no_improve += 1
            m_expand = min(m_expand * 1.04, 4.0)

        if no_improve > 150:
            break

    return best_f, best_obj


def simulated_annealing(f0, time_limit, verbose=False):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = compute_objective(f)
    t0 = time.monotonic()
    T = 0.01
    T_min = 0.0001
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Random perturbation
        rng = np.random.default_rng(it * 3 + 1)
        scale = T * 0.5
        f_trial = f + rng.standard_normal(n) * scale
        f_trial = project_simplex(f_trial)

        obj_trial = compute_objective(f_trial)
        obj_cur = compute_objective(f)

        # Accept if better or with probability exp(-delta/T)
        if obj_trial < obj_cur:
            f = f_trial
            no_improve = 0
        else:
            delta = obj_trial - obj_cur
            if delta < 0.1 and np.random.random() < np.exp(-delta / T):
                f = f_trial
            else:
                no_improve += 1

        if obj_trial < best_obj:
            best_obj = obj_trial
            best_f = f.copy()
            no_improve = 0
            T = max(T * 0.999, T_min)
        else:
            T = max(T * 0.9995, T_min)

        if no_improve > 500:
            if time.monotonic() - t0 > time_limit * 0.4:
                break
            f = project_simplex(best_f + np.random.default_rng(it * 17).standard_normal(n) * T)
            no_improve = 0
            T = max(T * 1.5, 0.005)

        if verbose and it % 1000 == 0:
            print(f"      SA iter {it}: T={T:.6f}, obj={compute_objective(f):.10f}, best={best_obj:.10f}")

    return best_f, best_obj


def coordinate_polish(f0, time_limit, verbose=False):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f0, compute_objective(f0)
    f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = compute_objective(f)
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        c = fast_convolve(f)

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-14:
                continue

            best_val = old_val
            best_obj_j = best_obj

            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial = f_trial / ss
                val = compute_objective(f_trial)
                if val < best_obj_j - 1e-14:
                    best_obj_j = val
                    best_val = new_val

            if best_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_val) / denom
                    f[j] = best_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                if best_obj_j < best_obj - 1e-14:
                    best_obj = best_obj_j
                    best_f = f.copy()
                    improved = True

        if not improved:
            # Try adding mass to zero coordinates
            zero_idx = np.where(f < 1e-14)[0]
            if len(zero_idx) > 0 and time.monotonic() - t0 < time_limit:
                for _ in range(min(5, len(zero_idx))):
                    j = int(zero_idx[np.random.randint(len(zero_idx))])
                    for val_try in [0.001, 0.005, 0.01]:
                        f_trial = f.copy()
                        denom = 1.0
                        f_trial[j] = val_try
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f_trial[mask] *= (1.0 - val_try)
                        f_trial = np.maximum(f_trial, 0)
                        f_trial = f_trial / f_trial.sum()
                        val = compute_objective(f_trial)
                        if val < best_obj - 1e-14:
                            f = f_trial
                            best_obj = val
                            best_f = f.copy()
                            improved = True
                            break

        if not improved:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    # Light smoothing
    if new_n > 5:
        kernel_size = min(5, new_n // 2 * 2 + 1)
        kernel = np.ones(kernel_size) / kernel_size
        f_smooth = np.convolve(f_interp, kernel, mode='same')
        f_smooth = np.maximum(f_smooth, 0)
        ss = f_smooth.sum()
        if ss > 1e-15:
            f_smooth /= ss
        return f_smooth
    return f_interp / f_interp.sum()


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization:
    1. Diverse initialization screening
    2. Subgradient descent
    3. Mirror descent
    4. LP-based Frank-Wolfe polishing
    5. Simulated annealing for exploration
    6. Multi-length exploration
    7. Coordinate descent final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = try_seq(parent, "parent")
            print(f"  Parent loaded: n={len(parent)}, score={p_score:.8f}")
    except Exception as e:
        print(f"  No parent available: {e}")

    # Phase 1: Diverse initialization
    print("=== Phase 1: Initialization Screening ===")
    t1 = time.monotonic()

    if parent is not None:
        n_parent = len(parent)
    else:
        n_parent = 768

    init_candidates = []
    init_candidates.append(("uniform", make_uniform(n_parent)))
    init_candidates.append(("gauss_025", make_gaussian(n_parent, 0.5, 0.25)))
    init_candidates.append(("gauss_035", make_gaussian(n_parent, 0.5, 0.35)))
    init_candidates.append(("gauss_050", make_gaussian(n_parent, 0.5, 0.50)))
    init_candidates.append(("gauss_070", make_gaussian(n_parent, 0.5, 0.70)))
    init_candidates.append(("bell_07", make_bell(n_parent, 0.7, 2)))
    init_candidates.append(("bell_09", make_bell(n_parent, 0.9, 2)))
    init_candidates.append(("arcoshape", make_arcoshape(n_parent)))
    init_candidates.append(("superflat", make_superflat(n_parent)))

    # Random initializations
    for i in range(10):
        a = np.abs(rng.standard_normal(n_parent))
        a = a / a.sum()
        init_candidates.append((f"rand_{i}", a))

    # Mixes with parent
    if parent is not None:
        for frac in [0.3, 0.5, 0.7]:
            a = frac * parent + (1 - frac) * make_gaussian(n_parent, 0.5, 0.4)
            a = a / a.sum()
            init_candidates.append((f"mix_{frac}", a))
            a = frac * parent + (1 - frac) * make_uniform(n_parent)
            a = a / a.sum()
            init_candidates.append((f"mixu_{frac}", a))

    best_init = None
    best_init_obj = float('inf')
    for name, a in init_candidates:
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            continue
        a = a / s
        obj = compute_objective(a)
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a
        try_seq(a, f"init_{name}")

    # Parent is also a candidate
    if parent is not None:
        obj_p = compute_objective(parent)
        if obj_p < best_init_obj:
            best_init_obj = obj_p
            best_init = parent

    print(f"  Best init obj: {best_init_obj:.8f}")
    print(f"  Phase 1 time: {time.monotonic()-t1:.1f}s")

    # Phase 2: Main optimization on best init
    print("\n=== Phase 2: Subgradient Descent ===")
    t2 = time.monotonic()
    rem = deadline - time.monotonic() - 600
    t_alloc = max(30, min(rem, 400))
    print(f"  Time allocation: {t_alloc:.0f}s, n={len(best_init)}")

    a_opt, obj = subgradient_descent(best_init, time_limit=t_alloc, verbose=True)
    try_seq(a_opt, "sgd")

    # Phase 3: Mirror descent
    print("\n=== Phase 3: Mirror Descent ===")
    t3 = time.monotonic()
    rem = deadline - time.monotonic() - 500
    if rem > 50:
        t_alloc = min(rem * 0.3, 200)
        print(f"  Time allocation: {t_alloc:.0f}s")
        a_opt2, obj2 = mirror_descent(a_opt, time_limit=t_alloc, verbose=True)
        try_seq(a_opt2, "mirror")
    print(f"  Phase 3 time: {time.monotonic()-t3:.1f}s")

    # Phase 4: Simulated annealing for exploration
    print("\n=== Phase 4: Simulated Annealing ===")
    t4 = time.monotonic()
    rem = deadline - time.monotonic() - 400
    if rem > 50:
        t_alloc = min(rem * 0.3, 200)
        print(f"  Time allocation: {t_alloc:.0f}s")
        a_opt3, obj3 = simulated_annealing(best_seq if best_seq is not None else a_opt,
                                           time_limit=t_alloc, verbose=True)
        try_seq(a_opt3, "anneal")
    print(f"  Phase 4 time: {time.monotonic()-t4:.1f}s")

    # Phase 5: Multi-length exploration
    print("\n=== Phase 5: Multi-Length Exploration ===")
    t5 = time.monotonic()
    base = best_seq if best_seq is not None else best_init

    target_lengths = []
    n_base = len(base)
    for factor in [0.5, 2, 3, 4]:
        nl = int(n_base * factor)
        if 64 <= nl <= 8192 and nl not in target_lengths:
            target_lengths.append(nl)
    # Also try some standard lengths
    for nl in [256, 512, 1024, 2048, 4096]:
        if nl not in target_lengths:
            target_lengths.append(nl)
    target_lengths = sorted(target_lengths)

    for target_n in target_lengths[:5]:
        rem = deadline - time.monotonic() - 200
        if rem < 60:
            break

        f_interp = interpolate_sequence(base, target_n)
        try_seq(f_interp, f"interp_n{target_n}")

        # Quick subgradient descent
        rem2 = deadline - time.monotonic() - 150
        if rem2 > 30:
            t_gd = min(rem2 * 0.4, 100)
            a_opt_n, _ = subgradient_descent(f_interp, time_limit=t_gd)
            try_seq(a_opt_n, f"sgd_n{target_n}")

    print(f"  Phase 5 time: {time.monotonic()-t5:.1f}s")

    # Phase 6: LP polishing on best result
    print("\n=== Phase 6: LP Polish ===")
    t6 = time.monotonic()
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 60:
            t_alloc = min(rem * 0.5, 250)
            print(f"  Time allocation: {t_alloc:.0f}s, n={len(best_seq)}")
            a_opt4, obj4 = lp_polish(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt4, "lp_polish")

            # Second LP pass with perturbation
            rem2 = deadline - time.monotonic() - 60
            if rem2 > 40:
                t_alloc2 = min(rem2 * 0.5, 100)
                n_best = len(a_opt4)
                noise = rng.standard_normal(n_best) * 0.003
                f_pert = project_simplex(a_opt4 + noise)
                a_opt5, _ = lp_polish(f_pert, time_limit=t_alloc2)
                try_seq(a_opt5, "lp_polish2")

    # Phase 7: Coordinate polish
    print("\n=== Phase 7: Coordinate Polish ===")
    t7 = time.monotonic()
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t_alloc = min(rem, 50)
            print(f"  Time allocation: {t_alloc:.0f}s")
            a_opt6, _ = coordinate_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt6, "coord")

    # Phase 8: Final squeeze - multiple short optimization runs
    print("\n=== Phase 8: Final Squeeze ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(3):
            rem = deadline - time.monotonic() - 15
            if rem < 20:
                break
            t_alloc = min(rem / (3 - p), 60)

            if p == 0:
                f_start = best_seq.copy()
            elif p == 1:
                f_start = project_simplex(best_seq + rng.standard_normal(n_best) * 0.002)
            else:
                f_start = 0.8 * best_seq + 0.2 * make_gaussian(n_best, 0.5, 0.4)
                f_start = f_start / f_start.sum()

            a_final, _ = subgradient_descent(f_start, time_limit=t_alloc)
            try_seq(a_final, f"final_{p}")

            rem2 = deadline - time.monotonic() - 10
            if rem2 > 20:
                a_final2, _ = lp_polish(a_final, time_limit=min(rem2, 40))
                try_seq(a_final2, f"final_lp_{p}")

    # Fallback
    if best_seq is None:
        best_seq = make_uniform(512)
        best_score = try_seq(best_seq, "fallback")

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"Total time: {time.monotonic() - (deadline - max(1.0, float(budget_s) - 30.0)):.1f}s")
    return [float(x) for x in best_seq]
