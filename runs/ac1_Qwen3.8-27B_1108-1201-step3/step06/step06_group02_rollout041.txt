The gradient of this is:

where p_k = exp(τ*(a*a)[k]) / sum_j exp(τ*(a*a)[j])

This is a weighted average of the gradients of all lags, weighted by their "importance" (softmax weights). For large τ, this concentrates on the peak.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-strategy approach:
- Improved LP-Frank-Wolfe with very tight M levels and FFT-accelerated line search
- NLP with smoothed max objective using FFT-accelerated analytical gradients
- Peak-targeted coordinate descent with swap operations
- Simulated annealing for escaping local minima
- Adaptive multi-phase time allocation
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def objective_normalized(a):
    """Compute 2*n*max(conv)/sum^2 for any a (auto-normalizes)."""
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    conv = fast_convolve(a_norm)
    return 2.0 * len(a_norm) * conv.max()


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    if res.success:
        return res.x
    return None


def compute_smooth_gradient(x, tau, weights=None):
    """
    Compute gradient of smoothed max of autocorrelation using FFT.
    f_smooth(x) = (1/tau) * log(sum_k exp(tau * (x*x)[k]))
    grad = sum_k p_k * grad_k(x) / sum(x)
    where p_k = exp(tau*c[k]) / sum exp(tau*c[j])
    and grad_k[i] = 2*x[k-i] (for valid indices)
    
    The key computation sum_k p_k * x[k-i] is a Hankel matrix-vector product,
    computed via FFT convolution.
    """
    n = len(x)
    conv = fast_convolve(x)
    c_shifted = conv - conv.max()
    exp_vals = np.exp(tau * c_shifted)
    p = exp_vals / exp_vals.sum()
    
    # Compute grad[i] = 2 * sum_k p[k] * x[k-i] / sum(x)
    # Using the Hankel trick: result[i] = (p * x_flipped)[i + n - 1]
    x_flipped = x[::-1]
    total_len = len(p) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    XF = np.fft.rfft(x_flipped, fft_size)
    conv_px = np.fft.irfft(P * XF, fft_size)[:total_len]
    
    # Extract the right portion: indices n-1 to 2n-2
    grad_unnorm = conv_px[n - 1:2 * n - 1]
    sx = x.sum()
    if sx < 1e-30:
        return np.zeros(n)
    return 2.0 * grad_unnorm / sx


def smooth_max_objective(x, tau):
    """Smoothed max of autocorrelation (logsumexp formulation)."""
    n = len(x)
    x_safe = np.maximum(x, 1e-30)
    sx = x_safe.sum()
    x_safe = x_safe / sx
    conv = fast_convolve(x_safe)
    c_shifted = conv - conv.max()
    return conv.max() + (1.0 / tau) * np.log(np.sum(np.exp(tau * c_shifted)))

                je = min(n - 1, pk)
                contrib = np.arange(js, je + 1)
                prods = best_f[contrib] * best_f[pk - contrib]
                src = contrib[np.argmax(prods)]
                trial = best_f.copy()
                trial[src] *= 0.8
                non_c = np.ones(n, dtype=bool)
                non_c[contrib] = False
                if non_c.sum() > 0:
                    trial[non_c] += trial[src] * 0.0 / max(trial[non_c].sum(), 1e-30)
                    diff = best_f[src] * 0.2
                    trial[non_c] += diff * trial[non_c] / max(trial[non_c].sum(), 1e-30)
                trial = np.maximum(trial, 0)
                f = trial / trial.sum()
            elif pert_type == 2:
                f = project_simplex(0.8 * best_f + 0.2 * best_f[::-1])
            else:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            
            m_idx = 0
            no_improve += 1
            if no_improve > 100:
                break
            continue
        
        g = g / g.sum()
        diff = np.max(np.abs(g - f))
        if diff < 1e-12:
            no_improve += 1
            m_idx = min(m_idx + 1, len(m_levels) - 1)
            if no_improve > 80:
                break
            continue
        no_improve = 0
        
        # Line search: coarse + fine
        best_alpha = 0.0
        best_val = current_peak
        alphas = [0.01, 0.03, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
        for alpha in alphas:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        # Fine golden section
        if best_alpha > 0:
            if best_alpha == alphas[0]:
                lo, hi = 0.0, alphas[1]
            elif best_alpha == alphas[-1]:
                lo, hi = alphas[-2], 1.0
            else:
                idx = alphas.index(best_alpha)
                lo = alphas[max(0, idx - 1)]
                hi = alphas[min(len(alphas) - 1, idx + 1)]
            
            gr = 0.61803398875
            for _ in range(15):
                a1 = hi - gr * (hi - lo)
                a2 = lo + gr * (hi - lo)
                ft1 = np.maximum(f + a1 * (g - f), 0.0)
                ft2 = np.maximum(f + a2 * (g - f), 0.0)
                ft1 /= ft1.sum()
                ft2 /= ft2.sum()
                v1 = fast_convolve(ft1).max()
                v2 = fast_convolve(ft2).max()
                if v1 < v2:
                    hi = a2
                else:
                    lo = a1
                mid = (lo + hi) / 2.0
                ftm = np.maximum(f + mid * (g - f), 0.0)
                ftm /= ftm.sum()
                vm = fast_convolve(ftm).max()
                if vm < best_val:
                    best_val = vm
                    best_alpha = mid
        
        if best_alpha < 1e-12:
            no_improve += 1
            m_idx = min(m_idx + 1, len(m_levels) - 1)
            continue
        
        f_new = np.maximum(f + best_alpha * (g - f), 0.0)
        f_new = f_new / f_new.sum()
        conv = fast_convolve(f_new)
        M = conv.max()
        f = f_new
        m_idx = 0
        
        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            if verbose and it % 10 == 0:
                print(f"    LP-FW iter {it}: obj={best_obj:.12f} (level={success_level})")
    
    return best_f, best_obj


def peak_coordinate_descent(f0, time_limit=300, verbose=True):
    """
    Peak-targeted coordinate descent with smart mass redistribution.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        if no_improve > 80:
            break
        
        conv = fast_convolve(f)
        M = conv.max()
        peak_k = int(np.argmax(conv))
        
        # Find contributing indices
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        contrib_idx = np.arange(js, je + 1)
        if len(contrib_idx) == 0:
            break
        
        products = f[contrib_idx] * f[peak_k - contrib_idx]
        max_prod_pos = np.argmax(products)
        src = contrib_idx[max_prod_pos]
        partner = peak_k - src
        
        # Try reducing src and redistributing
        best_trial = None
        best_trial_obj = M
        
        for frac in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            diff = f[src] * frac
            if diff < 1e-15:
                continue
            trial = f.copy()
            trial[src] -= diff
            # Redistribute to non-contributing positions
            non_contrib = np.ones(n, dtype=bool)
            non_contrib[contrib_idx] = False
            nc_positions = np.where(non_contrib)[0]
            if len(nc_positions) == 0:
                continue
            # Distribute proportional to current values (or uniform if all zero)
            nc_vals = trial[nc_positions]
            nc_sum = nc_vals.sum()
            if nc_sum < 1e-15:
                trial[nc_positions] += diff / len(nc_positions)
            else:
                trial[nc_positions] += diff * nc_vals / nc_sum
            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                continue
            trial = trial / ts
            obj = fast_convolve(trial).max()
            if obj < best_trial_obj - 1e-15:
                best_trial_obj = obj
                best_trial = trial
        
        # Also try swap operations
        for _ in range(3):
            rng = np.random.default_rng()
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or abs(f[i] - f[j]) < 1e-10:
                continue
            trial = f.copy()
            amount = min(f[i], f[j]) * 0.3
            if f[i] > f[j]:
                trial[i] -= amount
                trial[j] += amount
            else:
                trial[j] -= amount
                trial[i] += amount
            trial = np.maximum(trial, 0.0)
            trial = trial / trial.sum()
            obj = fast_convolve(trial).max()
            if obj < best_trial_obj - 1e-15:
                best_trial_obj = obj
                best_trial = trial
        
        if best_trial is not None:
            f = best_trial
            if best_trial_obj < best_obj - 1e-14:
                best_obj = best_trial_obj
                best_f = f.copy()
                no_improve = 0
                if verbose and best_obj < 1.51:
                    print(f"    PCD: obj={best_obj:.12f}")
        else:
            no_improve += 1
            # Try other peaks
            top_peaks = np.argsort(conv)[-8:]
            found = False
            for pk in top_peaks:
                pk = int(pk)
                if pk == peak_k:
                    continue
                js2 = max(0, pk - n + 1)
                je2 = min(n - 1, pk)
                cidx2 = np.arange(js2, je2 + 1)
                if len(cidx2) == 0:
                    continue
                prods2 = f[cidx2] * f[pk - cidx2]
                src2 = cidx2[np.argmax(prods2)]
                for frac2 in [0.1, 0.2, 0.3]:
                    diff2 = f[src2] * frac2
                    trial = f.copy()
                    trial[src2] -= diff2
                    nc2 = np.ones(n, dtype=bool)
                    nc2[cidx2] = False
                    nc2_pos = np.where(nc2)[0]
                    if len(nc2_pos) == 0:
                        continue
                    nc2_vals = trial[nc2_pos]
                    nc2_sum = nc2_vals.sum()
                    if nc2_sum < 1e-15:
                        trial[nc2_pos] += diff2 / len(nc2_pos)
                    else:
                        trial[nc2_pos] += diff2 * nc2_vals / nc2_sum
                    trial = np.maximum(trial, 0.0)
                    trial = trial / trial.sum()
                    obj2 = fast_convolve(trial).max()
                    if obj2 < best_obj - 1e-14:
                        f = trial
                        best_obj = obj2
                        best_f = f.copy()
                        no_improve = 0
                        found = True
                        break
                if found:
                    break
            if not found:
                no_improve += 1
    
    return best_f, best_obj


def simulated_annealing_optimize(f0, time_limit=200, verbose=True):
    """
    Simulated annealing with peak-aware moves.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    
    best_f = f.copy()
    best_M = fast_convolve(f).max()
    current_M = best_M
    t0 = time.monotonic()
    
    T_init = 0.001
    T_final = 0.00001
    n_steps = 3000
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        progress = step / n_steps
        T = T_init * (T_final / T_init) ** progress
        
        # Choose a random move
        rng = np.random.default_rng(step * 104729 + 3)
        move_type = rng.integers(0, 4)
        
        trial = f.copy()
        
        if move_type == 0:
            # Point mutation
            i = rng.integers(0, n)
            delta = rng.standard_normal() * 0.01
            trial[i] = max(0, trial[i] + delta)
        elif move_type == 1:
            # Swap mass between two positions
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            amount = min(trial[i], trial[j]) * rng.uniform(0.1, 0.4)
            if trial[i] > trial[j]:
                trial[i] -= amount
                trial[j] += amount
            else:
                trial[j] -= amount
                trial[i] += amount
        elif move_type == 2:
            # Local perturbation
            center = rng.integers(5, n - 5)
            width = rng.integers(2, 8)
            region = slice(max(0, center - width), min(n, center + width + 1))
            noise = rng.standard_normal(trial[region].shape) * 0.005
            trial[region] = np.maximum(trial[region] + noise, 0)
        else:
            # Blend with flipped version
            mix = rng.uniform(0.01, 0.05)
            trial = (1 - mix) * f + mix * f[::-1]
            trial = np.maximum(trial, 0)
        
        trial = np.maximum(trial, 0)
        ts = trial.sum()
        if ts < 1e-10:
            continue
        trial = trial / ts
        
        new_M = fast_convolve(trial).max()
        delta_M = new_M - current_M
        
        # Acceptance criterion
        if delta_M < 0 or rng.random() < np.exp(-delta_M / T):
            f = trial
            current_M = new_M
            if new_M < best_M - 1e-14:
                best_M = new_M
                best_f = f.copy()
                if verbose:
                    print(f"    SA step {step}: best_M={best_M:.12f}")
    
    return best_f, best_M


def make_symmetric(a):
    """Make a sequence symmetric."""
    n = len(a)
    a_sym = (a + a[::-1]) / 2.0
    a_sym = np.maximum(a_sym, 0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_cosine_mod(n, freq, amp):
    x = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a
