"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Nesterov-accelerated proximal
gradient with adaptive log-sum-exp smoothing, combined with LP-Frank-Wolfe
for initialization and multi-length exploration with jitter-based restarts.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = (u * rho) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def conv_obj(b):
    """Objective for sum-normalized b: 2*n*max(conv(b,b))."""
    n = len(b)
    return 2.0 * n * fast_convolve(b).max()


def smooth_grad(b, beta):
    """
    Compute the gradient of the log-sum-exp smoothed objective:
    F_beta(b) = 2*n/beta * log(sum_k exp(beta * conv_k(b)))
    
    Returns gradient of F_beta with respect to b.
    """
    n = len(b)
    conv = fast_convolve(b)
    # Softmax weights
    shifted = beta * (conv - conv.max())
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(conv)
        w[int(np.argmax(conv))] = 1.0
    else:
        w = w / w_sum
    
    # Gradient: dF/db_m = 2*n * sum_k w_k * d(conv_k)/db_m
    # d(conv_k)/db_m = 2*b_{k-m} (if k-m in [0,n-1])
    # So: dF/db_m = 4*n * sum_k w_k * b_{k-m}
    grad = np.zeros(n)
    # Use vectorized computation: grad[m] = 4*n * sum_j b[j] * w[m+j]
    # For efficiency, iterate over non-negligible w entries
    threshold = 1e-12
    active = np.where(w > threshold)[0]
    for k in active:
        wk = w[k]
        if wk < threshold:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        # grad[j_start:j_end+1] += 4*n * wk * b[k - j_end : k - j_start + 1][::-1]
        # b[k-j] for j in [j_start, j_end] = b[k-j_end : k-j_start+1][::-1]
        segment = b[k - j_end : k - j_start + 1][::-1]
        grad[j_start:j_end+1] += 4.0 * n * wk * segment
    
    return grad


def nesterov_proximal(b0, time_limit, beta_init=5.0, beta_final=200.0, verbose=False):
    """
    Nesterov-accelerated proximal gradient on the simplex.
    Uses log-sum-exp smoothing with gradually increasing beta.
    """
    b = project_simplex(b0.copy())
    n = len(b)
    
    best_obj = conv_obj(b)
    best_b = b.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    
    # Beta schedule: geometric from beta_init to beta_final
    n_epochs = 15
    betas = np.geomspace(beta_init, beta_final, n_epochs)
    
    for epoch_idx, beta in enumerate(betas):
        if time.monotonic() - t0 > time_limit:
            break
        
        eta = 1.0 / (2.0 * beta)  # Step size
        # Nesterov momentum
        mu = 1.0  # Initial momentum parameter
        b_curr = b.copy()
        b_prev = b.copy()
        y = b.copy()
        
        epoch_start_obj = conv_obj(b_curr)
        epoch_improved = False
        
        max_iters = 200 + 50 * epoch_idx
        
        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Compute Nesterov momentum step
            if it > 0:
                mu_new = (1.0 + np.sqrt(1.0 + 4.0 * mu * mu)) / 2.0
                y = b_curr + (mu - 1.0) / mu_new * (b_curr - b_prev)
                mu = mu_new
            else:
                y = b_curr.copy()
            
            # Compute gradient at y
            grad = smooth_grad(y, beta)
            
            # Proximal step
            b_candidate = y - eta * grad
            b_new = project_simplex(b_candidate)
            
            # Check improvement
            obj = conv_obj(b_new)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_b = b_new.copy()
                epoch_improved = True
                no_improve = 0
            else:
                no_improve += 1
            
            b_prev = b_curr
            b_curr = b_new
        
        if verbose:
            print(f"    Epoch {epoch_idx}: beta={beta:.1f}, obj={best_obj:.8f}")
        
        if no_improve > 500:
            # Reheat: add small perturbation
            rng = np.random.default_rng(epoch_idx * 137 + 42)
            b_curr = project_simplex(best_b + rng.standard_normal(n) * 0.001)
            no_improve = 0
    
    return best_b, best_obj


def lp_fw_optimize(b0, time_limit, verbose=False):
    """Improved LP-Frank-Wolfe with away-steps."""
    b = project_simplex(np.maximum(b0, 0))
    n = len(b)
    
    conv = fast_convolve(b)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_b = b.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Solve LP: max sum(g) s.t. (b*g)_k <= M, g >= 0
        # Build constraint matrix
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            k_end = min(j + n, 2 * n - 1)
            A_ub[j:k_end, j] = b[:k_end - j]
        
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M * 1.001)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, 
                         method='highs', options={'presolve': True, 'time_limit': max(1, time_limit - (time.monotonic() - t0))})
        except Exception:
            break
        
        if not res.success:
            no_improve += 5
            if no_improve > 50:
                break
            continue
        
        g = np.maximum(res.x, 0)
        g_sum = g.sum()
        if g_sum < 1e-15:
            no_improve += 10
            break
        g = g / g_sum
        
        # Check if direction is useful
        diff = np.max(np.abs(g - b))
        if diff < 1e-10:
            no_improve += 5
            if no_improve > 50:
                break
            # Try with relaxed M
            M *= 1.05
            continue
        
        # Line search for step size
        best_alpha = 0.0
        best_alpha_obj = best_obj
        
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            b_trial = (1 - alpha) * b + alpha * g
            b_trial = project_simplex(b_trial)
            obj = conv_obj(b_trial)
            if obj < best_alpha_obj:
                best_alpha_obj = obj
                best_alpha = alpha
        
        if best_alpha < 1e-8:
            no_improve += 3
            continue
        
        b = (1 - best_alpha) * b + best_alpha * g
        b = project_simplex(b)
        obj = conv_obj(b)
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_b = b.copy()
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"    LP-FW iter {it}: obj={best_obj:.8f}")
        else:
            no_improve += 1
        
        if no_improve > 80:
            break
    
    return best_b, best_obj


def make_gaussian(n, sigma_frac=0.25):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width=0.7):
    x = np.arange(n) - n / 2.0
    half_w = n * width / 2.0
    t = np.clip(np.abs(x) / half_w, 0, 1.0)
    a = (1.0 - t) ** 3
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_flat(n):
    return np.ones(n) / n


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Nesterov-accelerated proximal
gradient with adaptive log-sum-exp smoothing, combined with LP-Frank-Wolfe
for initialization, multi-length exploration, and jitter-based restarts.
Key innovation: fast proximal gradient iterations (O(n log n)) with Nesterov
momentum achieve O(1/t^2) convergence on the smoothed convex objective,
allowing many more iterations than LP-based methods in the same time.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = (u * rho) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def conv_objective(b):
    """Objective for sum-normalized b: 2*n*max(conv(b,b))."""
    n = len(b)
    return 2.0 * n * fast_convolve(b).max()


def smooth_gradient(b, beta):
    """
    Compute gradient of F_beta(b) = (2*n/beta)*log(sum_k exp(beta*conv_k(b))).
    This is the gradient of the log-sum-exp smoothed objective.
    """
    n = len(b)
    conv = fast_convolve(b)
    shifted = beta * (conv - conv.max())
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(conv)
        w[int(np.argmax(conv))] = 1.0
    else:
        w = w / w_sum

    grad = np.zeros(n)
    threshold = 1e-12
    active = np.where(w > threshold)[0]
    for k in active:
        wk = w[k]
        if wk < threshold:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        # grad[j] += 4*n * wk * b[k-j] for j in [j_start, j_end]
        # b[k-j] for j in [j_start..j_end] = b[k-j_end : k-j_start+1][::-1]
        segment = b[k - j_end : k - j_start + 1][::-1]
        grad[j_start : j_end + 1] += 4.0 * n * wk * segment
    return grad


def nesterov_proximal(b0, time_limit, beta_init=5.0, beta_final=500.0,
                       n_epochs=12, verbose=False):
    """
    Nesterov-accelerated proximal gradient on the simplex with
    geometrically increasing smoothing parameter beta.
    """
    b_curr = project_simplex(np.maximum(b0.copy(), 0.0))
    n = len(b_curr)

    best_obj = conv_objective(b_curr)
    best_b = b_curr.copy()

    t0 = time.monotonic()
    no_improve_total = 0

    betas = np.geomspace(beta_init, beta_final, n_epochs)

    for ep in range(n_epochs):
        if time.monotonic() - t0 > time_limit:
            break
        beta = betas[ep]
        eta = 1.0 / (2.0 * beta)
        max_iters = 300 + 100 * ep

        b_curr = best_b.copy() if no_improve_total > 100 * (ep + 1) else b_curr
        b_prev = b_curr.copy()
        y = b_curr.copy()
        mu = 1.0
        epoch_improved = False

        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                break

            if it > 0:
                mu_new = (1.0 + np.sqrt(1.0 + 4.0 * mu * mu)) / 2.0
                y = b_curr + (mu - 1.0) / mu_new * (b_curr - b_prev)
                y = project_simplex(y)
                mu = mu_new
            else:
                y = b_curr.copy()

            grad = smooth_gradient(y, beta)
            b_new = project_simplex(y - eta * grad)

            obj = conv_objective(b_new)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_b = b_new.copy()
                epoch_improved = True
                no_improve_total = 0
            else:
                no_improve_total += 1

            b_prev = b_curr
            b_curr = b_new

        if verbose:
            print(f"    Epoch {ep}: beta={beta:.1f}, best_obj={best_obj:.8f}")

        if no_improve_total > 300 * (ep + 1):
            # Reheat with perturbation
            rng = np.random.default_rng(ep * 271 + 13)
            b_curr = project_simplex(best_b + rng.standard_normal(n) * 0.002)
            no_improve_total = 0

    return best_b, best_obj


def subgradient_proximal(b0, time_limit, verbose=False):
    """
    Non-accelerated proximal gradient with hard max subgradient
    and diminishing step sizes. Good for final polishing.
    """
    b = project_simplex(np.maximum(b0.copy(), 0.0))
    n = len(b)
    best_obj = conv_objective(b)
    best_b = b.copy()

    t0 = time.monotonic()
    no_improve = 0
    c = 1.0  # Step size parameter

    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(b)
        max_val = conv.max()
        active = np.where(np.abs(conv - max_val) < 1e-10)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv))])

        # Average subgradient over active set
        grad = np.zeros(n)
        for k in active:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            segment = b[k - j_end : k - j_start + 1][::-1]
            grad[j_start : j_end + 1] += 2.0 * segment
        grad /= len(active)

        eta_t = c / (1.0 + it * 0.1)
        b_new = project_simplex(b - eta_t * grad)

        obj = conv_objective(b_new)
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_b = b_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        b = b_new

        if no_improve > 500:
            break

        if verbose and it % 500 == 0:
            print(f"    Subgrad PG iter {it}: obj={best_obj:.8f}")

    return best_b, best_obj


def lp_fw_step(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_end = min(j + n, 2 * n - 1)
        A_ub[j : k_end, j] = f[: k_end - j]

    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
    except Exception:
        return None
    if res.success:
        return np.maximum(res.x, 0.0)
    return None
