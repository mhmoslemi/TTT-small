"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fine coordinate descent with 
exact 1D sub-problem solving, Nesterov-accelerated subgradient descent, 
simulated annealing, and multi-scale multi-start optimization.
"""

import numpy as np
import time
from scipy.optimize import minimize_scalar

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_R(conv, a, j, n):
    """Precompute R[k] = conv[k] - terms involving a[j]."""
    R = conv.copy()
    v = a[j]
    # For k in [j, j+n-1] (clipped to [0, 2n-2]):
    k_start = j
    k_end = min(j + n - 1, 2 * n - 2)
    if k_start <= k_end:
        a_vals = a[k_start - j : k_end - j + 1]  # a[k-j] for k in range
        if 2 * j >= k_start and 2 * j <= k_end:
            # Remove terms for all k in range, then handle 2j separately
            R[k_start:k_end+1] -= 2.0 * v * a_vals
            # Add back the correct term for k=2j (should be v^2, not 2*v*a[j])
            # We subtracted 2*v*a[j] but should have subtracted v^2
            R[2*j] += 2.0 * v * a[j] - v * v
        else:
            R[k_start:k_end+1] -= 2.0 * v * a_vals
    return R

def conv_after_change(R, a, j, v, s, n):
    """Compute conv'[k] for all k given new value v at position j, using precomputed R."""
    scale = (1.0 - v) / s
    scale2 = scale * scale
    conv_new = R * scale2
    
    k_start = j
    k_end = min(j + n - 1, 2 * n - 2)
    if k_start <= k_end:
        a_vals = a[k_start - j : k_end - j + 1]
        conv_new[k_start:k_end+1] += 2.0 * v * scale * a_vals
        if 2 * j >= k_start and 2 * j <= k_end:
            conv_new[2*j] = v * v + R[2*j] * scale2
    
    return conv_new

def coord_obj(a, j, v, s, R, n):
    """Objective (max conv) for coordinate j set to v."""
    conv_new = conv_after_change(R, a, j, v, s, n)
    return conv_new.max()

def fine_coordinate_descent(a_init, time_limit, verbose=False, label=""):
    """
    Fine coordinate descent: for each coordinate, solve the 1D sub-problem
    exactly using scipy's bounded minimization.
    """
    a = np.maximum(a_init, 0.0).astype(np.float64)
    s_total = a.sum()
    if s_total < 1e-15:
        return a_init, float('inf')
    a = a / s_total
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    sweep = 0
    max_sweeps = 10000
    
    while time.monotonic() - t0 < time_limit and sweep < max_sweeps:
        sweep += 1
        sweep_improved = 0
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            v0 = a[j]
            s = 1.0 - v0
            if s < 1e-12:
                continue
            
            # Precompute R
            conv_current = fast_convolve(a)
            R = compute_R(conv_current, a, j, n)
            
            # Define the 1D objective
            def f1d(v):
                if v < -1e-10 or v > 1.0 + 1e-10:
                    return 1e10
                v = np.clip(v, 0.0, 1.0)
                return coord_obj(a, j, v, s, R, n)
            
            # Search in a window around v0
            # The optimal v is likely close to v0
            half_width = max(0.01, 2.0 * v0)
            lo = max(0.0, v0 - half_width)
            hi = min(1.0, v0 + half_width)
            if hi - lo < 1e-8:
                lo = max(0.0, v0 * 0.5)
                hi = min(1.0, v0 * 2.0 + 0.01)
            
            result = minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                                     options={'maxiter': 50, 'xatol': 1e-10})
            
            v_new = result.x
            if v_new < 0:
                v_new = 0.0
            if v_new > 1.0:
                v_new = 1.0
            
            if abs(v_new - v0) > 1e-12:
                # Apply the change
                a[j] = v_new
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= (1.0 - v_new) / s
                a = np.maximum(a, 0.0)
                
                new_conv = fast_convolve(a)
                new_obj = new_conv.max()
                
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    sweep_improved += 1
        
        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} CD sweep {sweep}: obj={best_obj:.10f}, improved={sweep_improved}, time={elapsed:.0f}s")
        
        if sweep_improved == 0:
            break
    
    return best_a, best_obj

    """Golden section search for 1D minimization."""

    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)

            fd = f(d)
    return (lo + hi) / 2.0


def fine_coordinate_descent(a_init, time_limit, verbose=False, label=""):
    """
    Fine coordinate descent with exact 1D sub-problem solving.
    For each coordinate j, minimizes max_k conv'(a_j=k) where conv' is computed
    in O(n) using precomputed residual R[k].
    """
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    s_total = a.sum()
    if s_total < 1e-15:
        return a_init.copy(), float('inf')
    a = a / s_total
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    sweep = 0
    no_improve_sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        sweep_improved = 0
        conv = fast_convolve(a)

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            v0 = a[j]
            s = 1.0 - v0
            if s < 1e-12:
                continue

            # Precompute R[k] = conv[k] - terms involving a[j]
            R = conv.copy()
            k_start = j
            k_end = min(j + n - 1, 2 * n - 2)
            if k_start <= k_end:
                idx = np.arange(k_start, k_end + 1)
                a_vals = a[idx - j]
                R[idx] -= 2.0 * v0 * a_vals
                if 2 * j >= k_start and 2 * j <= k_end:
                    R[2 * j] += v0 * v0

            # Define 1D objective (vectorized)
            idx_arr = idx
            a_vals_arr = a_vals
            R_arr = R
            two_j = 2 * j
            k_s = k_start
            k_e = k_end
            s_val = s
            n_val = n

            def f1d(v):
                if v < -1e-12 or v > 1.0 + 1e-12:
                    return 1e10
                v = np.clip(v, 0.0, 1.0)
                sc = (1.0 - v) / s_val
                conv_new = R_arr * (sc * sc)
                if k_s <= k_e:
                    conv_new[k_s:k_e + 1] += 2.0 * v * sc * a_vals_arr
                    if two_j >= k_s and two_j <= k_e:
                        conv_new[two_j] = v * v + R_arr[two_j] * (sc * sc)
                return conv_new.max()

            # Search window around v0
            margin = max(0.02, 3.0 * v0 + 0.005)
            lo = max(0.0, v0 - margin)
            hi = min(1.0, v0 + margin)
            if hi - lo < 1e-10:
                lo = max(0.0, v0 * 0.5)
                hi = min(1.0, v0 + 0.05)

            # Coarse grid search to find good region
            grid = np.linspace(lo, hi, 20)
            vals = np.array([f1d(v) for v in grid])
            best_grid_idx = np.argmin(vals)

            # Local refinement around best grid point
            local_lo = grid[max(0, best_grid_idx - 1)]
            local_hi = grid[min(len(grid) - 1, best_grid_idx + 1)]
            v_opt = golden_section_1d(f1d, local_lo, local_hi, max_iter=50, tol=1e-12)

            if abs(v_opt - v0) > 1e-13:
                a[j] = v_opt
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= (1.0 - v_opt) / s
                a = np.maximum(a, 0.0)
                ss = a.sum()
                if ss > 1e-15:
                    a /= ss

                new_obj = fast_convolve(a).max()
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    sweep_improved += 1

        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} CD sweep {sweep}: obj={best_obj:.10f}, "
                  f"improved_coords={sweep_improved}, t={elapsed:.0f}s")

        if sweep_improved == 0:
            no_improve_sweeps += 1
            if no_improve_sweeps >= 3:
                break
        else:
            no_improve_sweeps = 0

    return best_a, best_obj


def nesterov_subgradient(a_init, time_limit, top_k=20, temp_init=0.001,
                         lr_init=0.01, verbose=False, label=""):
    """Nesterov-accelerated subgradient method on the simplex."""
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    # Nesterov momentum
    y = a.copy()
    momentum = 0.0
    best_obj = fast_convolve(a).max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(y)
        M = conv.max()

        K = min(top_k, len(conv))
        if K < len(conv):
            peak_indices = np.argpartition(conv, -K)[-K:]
        else:
            peak_indices = np.arange(len(conv))

        pv = conv[peak_indices]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peak_indices))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        g = compute_subgradient(y, peak_indices, w)
        g_centered = g - g.mean()

        # Nesterov update
        new_momentum = (1.0 + momentum) / (1.0 + it + momentum)
        y_trial = y - lr * (g_centered + new_momentum * (y - a))
        y = project_simplex(y_trial)
        a = a + (y - a) * (1.0 + momentum) / (1.0 + it + momentum)
        a = project_simplex(a)
        momentum = (1.0 + momentum) / 2.0 if it % 100 == 0 else momentum

        conv_a = fast_convolve(a)
        new_obj = conv_a.max()

        if new_obj < best_obj - 1e-13:
            best_obj = new_obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    {label} NS iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        temp = min(temp * 1.003, 0.03)
        lr = max(lr * 0.9998, 1e-8)

        if no_improve > 300:
            rng = np.random.default_rng(it * 6151 + 7)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            y = a.copy()
            momentum = 0.0
            no_improve = 0
            lr = lr_init * 0.5

    return best_a, best_obj


def simulated_annealing(a_init, time_limit, n_iter_per_step=50,
                        initial_temp=0.005, verbose=False, label=""):
    """Simulated annealing on the simplex with coordinate-level moves."""
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    curr_obj = best_obj

    t0 = time.monotonic()
    temp = initial_temp
    total_iters = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        batch_start = time.monotonic()
        
        for _ in range(n_iter_per_step):
            if time.monotonic() - t0 > time_limit:
                break
            total_iters += 1

            # Random move: pick 2 coordinates, transfer mass
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue

            # Transfer fraction of min(a[i], a[j])
            transfer = min(a[i], a[j]) * rng.uniform(0.01, 0.3)
            if transfer < 1e-12:
                continue

            a_trial = a.copy()
            a_trial[i] -= transfer
            a_trial[j] += transfer
            a_trial = np.maximum(a_trial, 0.0)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial /= ss

            trial_obj = fast_convolve(a_trial).max()
            delta = trial_obj - curr_obj

            if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
                a = a_trial
                curr_obj = trial_obj
                if trial_obj < best_obj - 1e-14:
                    best_obj = trial_obj
                    best_a = a.copy()

        # Cool
        temp *= 0.95
        batch_time = time.monotonic() - batch_start

        if verbose and total_iters % 5000 == 0:
            print(f"    {label} SA iter {total_iters}: best={best_obj:.10f}, "
                  f"curr={curr_obj:.10f}, T={temp:.6f}, t={time.monotonic()-t0:.0f}s")

        if temp < 1e-8:
            break

    return best_a, best_obj


def mirror_descent(a_init, time_limit, top_k=25, temp_init=0.0005,
                   lr_init=0.01, verbose=False, label=""):
    """Multiplicative weight mirror descent with backtracking."""
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()

        K = min(top_k, len(conv))
        if K < len(conv):
            peak_indices = np.argpartition(conv, -K)[-K:]
        else:
            peak_indices = np.arange(len(conv))

        pv = conv[peak_indices]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peak_indices))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        g = compute_subgradient(a, peak_indices, w)

        # Backtracking
        best_lr = lr * 0.1
        best_trial = M * 1.01
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            if trial_lr < 1e-12:
                continue
            log_u = np.clip(-trial_lr * g, -50, 50)
            a_t = np.maximum(a * np.exp(log_u), 1e-15)
            a_t /= a_t.sum()
            t_obj = fast_convolve(a_t).max()
            if t_obj <= best_trial:
                best_lr = trial_lr
                best_trial = t_obj
                if t_obj < M * 0.998:
                    break

        log_u = np.clip(-best_lr * g, -50, 50)
        a = np.maximum(a * np.exp(log_u), 1e-15)
        a /= a.sum()

        new_obj = fast_convolve(a).max()
        if new_obj < best_obj - 1e-13:
            best_obj = new_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        temp = min(temp * 1.002, 0.04)
        if no_improve > 80:
            lr *= 0.7
            no_improve = 0
        lr = max(lr * 0.9999, 1e-9)

        if no_improve > 400:
            rng = np.random.default_rng(it * 9973 + 11)
            a = np.maximum(best_a + rng.standard_normal(n) * 0.002, 1e-15)
            a /= a.sum()
            lr = lr_init * 0.3
            no_improve = 0

        # Periodic symmetrization
        if it % 500 == 0:
            a_sym = 0.5 * (a + a[::-1])
            a_sym = np.maximum(a_sym, 1e-15)
            a_sym /= a_sym.sum()
            sym_obj = fast_convolve(a_sym).max()
            if sym_obj < best_obj:
                best_obj = sym_obj
                best_a = a_sym.copy()
                a = a_sym.copy()

    return best_a, best_obj


def symmetrize(a):
    """Make sequence palindromic."""
    a_sym = 0.5 * (a + a[::-1].copy())
    a_sym = np.maximum(a_sym, 0.0)
    ss = a_sym.sum()
    if ss > 1e-15:
        a_sym /= ss
    return a_sym


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, maintain sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian-shaped initialization."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    """Bell-shaped with flat top."""
    x = np.arange(n) - n / 2.0
    half_w = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_w, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_asymmetric(n, decay=0.4):
    """Exponentially decaying (asymmetric)."""
    a = np.exp(-np.arange(n) / (n * decay))
    a /= a.sum()
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization: fine coordinate descent, Nesterov subgradient,
    mirror descent, simulated annealing, multi-n exploration, and final polish.
    """

    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        a = a / s
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_gaussian(768, 0.3)
        p_score = try_seq(parent, "default_init")

    # === PHASE 1: Multi-n exploration (250s) ===
    print("\n=== Phase 1: Multi-n Exploration ===")
    target_ns = [512, 640, 768, 896, 1024, 1280]
    if parent is not None:
        pass  # already included
    
    phase1_time = 250
    t_per_n = phase1_time / (len(target_ns) + 2)

    for n_try in target_ns:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        t_alloc = min(t_per_n, remaining - 10)
        if t_alloc < 5:
            break

        # Multiple initializations for this n
        inits = [
            (interpolate_sequence(parent, n_try), f"parent_n{n_try}"),
            (make_gaussian(n_try, 0.25), f"gauss_tight_n{n_try}"),
            (make_gaussian(n_try, 0.35), f"gauss_wide_n{n_try}"),
            (make_asymmetric(n_try, 0.35), f"asym_n{n_try}"),
        ]

        best_for_n = None
        best_for_n_obj = float('inf')

        for a_init, lab in inits:
            if time.monotonic() > deadline - 60:
                break
            t_sub = min(t_alloc / 4, remaining - 20)
            if t_sub < 2:
                continue
            a_opt, obj = fine_coordinate_descent(a_init, time_limit=t_sub,
                                                  verbose=False, label=lab)
            try_seq(a_opt, lab)
            if obj < best_for_n_obj:
                best_for_n_obj = obj
                best_for_n = a_opt

        if best_for_n is not None:
            try_seq(best_for_n, f"best_n{n_try}")

    # === PHASE 2: Intensive Coordinate Descent on best (500s) ===
    print(f"\n=== Phase 2: Intensive CD (best_n={len(best_seq)}) ===")

    t_cd = min(500, remaining * 0.45)

    # === PHASE 4: Mirror Descent (200s) ===
    print(f"\n=== Phase 4: Mirror Descent ===")

    print(f"  Time: {t_md:.0f}s")

    # === PHASE 8: Multi-restart with noise ===
    print(f"\n=== Phase 8: Multi-restart ===")
    remaining = deadline - time.monotonic()
    if remaining > 100:
        n_restarts = 3
        t_per_r = min(remaining * 0.25 / n_restarts, 80)
        for r in range(n_restarts):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            if r == 0:
                f_init = best_seq + rng.standard_normal(len(best_seq)) * 0.003
            elif r == 1:
                f_init = symmetrize(best_seq) + rng.standard_normal(len(best_seq)) * 0.002
            else:
                f_init = 0.95 * best_seq + 0.05 * (np.ones(len(best_seq)) / len(best_seq))
                f_init += rng.standard_normal(len(best_seq)) * 0.001

            f_init = np.maximum(f_init, 0.0)
            f_init /= f_init.sum()

            t_alloc = min(t_per_r, remaining - 20)
            if t_alloc < 10:
                break
            a_opt, obj = fine_coordinate_descent(f_init, time_limit=t_alloc,
                                                 verbose=False, label=f"restart_{r}")
            try_seq(a_opt, f"restart_{r}")

    # === PHASE 9: Final ultra-fine CD ===
    print(f"\n=== Phase 9: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        t_final = min(remaining - 10, 120)
        a_opt, obj = fine_coordinate_descent(best_seq, time_limit=t_final,
                                             verbose=True, label="polish")
        try_seq(a_opt, "polish")

        # Try symmetrized final
        a_s = symmetrize(best_seq)
        try_seq(a_s, "final_sym")

    # === PHASE 10: Try different n by interpolation of final best ===
    print(f"\n=== Phase 10: Interpolation to other n ===")
    remaining = deadline - time.monotonic()
    if remaining > 60:
        for n_try in [len(best_seq) - 64, len(best_seq) + 64]:
            if n_try < 128 or remaining < 30:

                                                 verbose=False, label=f"interp_n{n_try}")

            remaining = deadline - time.monotonic()

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
