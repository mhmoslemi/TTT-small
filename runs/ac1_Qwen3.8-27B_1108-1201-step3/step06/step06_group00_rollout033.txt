import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def eval_quick(a):
    """Quick local evaluation."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S * S))

If we think of this as a "sliding dot product" of a with a shifted version of itself, then in the plateau region, we're computing the correlation of a with itself at various lags.

For the correlation to be flat in the plateau region, we need the autocorrelation of a to be flat (excluding lag 0). This is the "low autocorrelation" property!

So the problem reduces to: find a non-negative sequence of length n with sum 1, such that its autocorrelation (excluding lag 0) is as flat as possible, AND the edge effects don't create larger peaks.

The "ideal" autocorrelation would be 0 at all non-zero lags (like a Barker sequence), but that's impossible for non-negative real sequences (since all terms are positive, the autocorrelation at any non-zero lag is positive).

So the best we can do is make the autocorrelation as small and as flat as possible at non-zero lags.

For a non-negative sequence, the autocorrelation at lag d is:
R(d) = sum_{j=d}^{n-1} a[j]*a[j-d]

This is always positive. To minimize the maximum of R(d) over d=1,...,n-1, we want the sequence to be "spread out" in a way that reduces pairwise products at all lags.

One approach: use a sequence that is as "random" as possible (white noise). For i.i.d. positive random variables with mean 1/n, the autocorrelation at lag d would be approximately (n-d)/n^2 (the expected value), with some variance. The maximum would be approximately (n-d)/n^2 + several standard deviations.

For a uniform-like sequence, the autocorrelation at lag d is exactly (n-d)/n^2 (deterministic). The maximum is at d=0 (which is n/n^2 = 1/n) or at d=1 (which is (n-1)/n^2 ≈ 1/n).

So for the uniform case, the autocorrelation decreases linearly from 1/n to 0. The "peak" is at lag 0 (or lag 1).

To make it flatter, we want the autocorrelation to be more constant (not decreasing). But since R(d) = sum_{j=d}^{n-1} a[j]*a[j-d] and the number of terms decreases with d, we'd need the individual products to increase with d on average. This means a[j]*a[j-d] should be larger for large d, which implies a should be larger at positions that are far apart.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-length subgradient descent with adaptive temperature
2. Exact 1D coordinate descent with bisection
3. L-BFGS soft-max refinement
4. Genetic diversification

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def get_subgradient(a, b, w):
    """Compute subgradient of max(conv) at point a with weights w on peaks."""
    n = len(a)
    g = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        g[idx] += w[k] * a[k - idx]
    return g


def subgradient_opt(a0, time_limit, lr_init=0.003, momentum=0.92,
                    temp_init=0.0005, temp_growth=1.01,
                    n_top=None, seed=42, batch_size=50):
    """Subgradient with Nesterov momentum and temperature-annealed peaks."""
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.astype(np.float64)
    a = a / a.sum()
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    cur_lr = lr_init
    cur_temp = temp_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        for _ in range(batch_size):
            if time.monotonic() - t0 > time_limit:
                break
            b = fast_convolve(a)
            
            # Temperature-weighted peak selection
            scaled = b / cur_temp
            scaled -= scaled.max()
            w = np.exp(scaled)
            ws = w.sum()
            if ws < 1e-30:
                w = np.zeros(len(b))
                w[b.argmax()] = 1.0
            else:
                w /= ws
            
            if n_top is not None and n_top < len(b):
                top_idx = np.argpartition(b, -n_top)[-n_top:]
                w_sel = w[top_idx]
                wsel_sum = w_sel.sum()
                if wsel_sum > 1e-30:
                    w_sel /= wsel_sum
                else:
                    w_sel = np.zeros_like(w_sel)
                    w_sel[0] = 1.0
                w_full = np.zeros(len(b))
                w_full[top_idx] = w_sel
                w = w_full
            
            g = get_subgradient(a, b, w)
            v = momentum * v - cur_lr * g
            a_new = project_simplex(a + v)
            
            b_new = fast_convolve(a_new)
            obj_new = b_new.max()
            
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            a = a_new
        
        if no_improve > 2000:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.7, lr_init * 0.1)
            no_improve = 0
        elif no_improve > 1000:
            cur_lr *= 0.95
        
        cur_temp *= temp_growth
        cur_temp = min(cur_temp, 0.3)
    
    return best_a, best_obj


def coordinate_descent(a0, time_limit, top_m=50):
    """
    Coordinate descent: for each coordinate, find the best value
    by golden section search on the 1D objective.
    """
    t0 = time.monotonic()
    n = len(a0)
    a = a0.astype(np.float64) / a0.sum()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    def obj_1d(j, t):
        """Objective when a[j] = t and others scaled to maintain sum=1."""
        if t < 0 or t > 0.9999:
            return float('inf')
        a_try = a.copy()
        a_try[j] = t
        remaining = 1.0 - t
        if remaining <= 0:
            return float('inf')
        old_sum_others = 1.0 - a[j]
        if old_sum_others < 1e-10:
            return float('inf')
        scale = remaining / old_sum_others
        mask = np.ones(n, dtype=bool)
        mask[j] = False
        a_try[mask] *= scale
        b = fast_convolve(a_try)
        return b.max()
    
    # Focus on top_m coordinates (largest values) and their neighbors
    improved = True
    sweep_count = 0
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        sweep_count += 1
        # Sort by value, optimize top ones first
        order = np.argsort(a)[::-1]
        for j in order:
            if time.monotonic() - t0 > time_limit:

def softmax_obj_grad(z, T, n):
    """Soft-max smoothed objective in log-space."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    w = exp_Tb / exp_Tb.sum()
    
    # Gradient: d(obj)/d(a) = 2n * cross_corr(w, a) / (sum(a))^2
    # Since sum(a) = 1: d(obj)/d(a) = 2n * cross_corr(w, a)
    g = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        g[idx] += w[k] * a[k - idx]
    g *= 2.0 * n
    
    # Chain rule: d(obj)/d(z) = a * (g - a * (g . a))
    S = np.dot(g, a)
    grad_z = a * (g - a * S)
    return obj, grad_z

    t0 = time.monotonic()

        if time.monotonic() - t0 > time_limit:

        obj, _ = softmax_obj_grad(z, T, n)
        return obj

            return np.zeros_like(z)
        _, g = softmax_obj_grad(z, T, n)

            return a_out
    except Exception:
        pass
    return a0

2. L-BFGS refinement with soft-max smoothing in log-space

4. Multi-length exploration via interpolation and re-optimization

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, b, w):
    """Compute weighted subgradient of max(conv) at point a."""
    n = len(a)
    g = np.zeros(n)
    active = np.where(w > 1e-14)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        g[idx] += w[k] * a[k - idx]
    return g


def subgradient_opt(a0, time_limit, lr_init=0.003, momentum=0.92,
                    temp_init=0.0005, temp_growth=1.008,
                    n_top=None, seed=42, batch=40, kick_size=0.003):
    """
    Subgradient with Nesterov momentum and temperature-annealed peak weighting.
    Optimizes on the simplex (sum=1, a>=0).
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.astype(np.float64)
    S = a.sum()
    a = a / S
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    cur_lr = lr_init
    cur_temp = temp_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit:
                break
            b = fast_convolve(a)

            # Temperature-weighted peak selection
            scaled = b / cur_temp
            scaled -= scaled.max()
            w = np.exp(scaled)
            ws = w.sum()
            if ws < 1e-30:
                w = np.zeros(len(b))
                w[b.argmax()] = 1.0
            else:
                w /= ws

            if n_top is not None and n_top < len(b):
                top_idx = np.argpartition(b, -n_top)[-n_top:]
                w_sel = w[top_idx].copy()
                wsel_sum = w_sel.sum()
                if wsel_sum > 1e-30:
                    w_sel /= wsel_sum
                else:
                    w_sel = np.zeros_like(w_sel)
                    w_sel[top_idx[0]] = 1.0
                w_full = np.zeros(len(b))
                w_full[top_idx] = w_sel
                w = w_full

            g = compute_subgradient(a, b, w)
            v = momentum * v - cur_lr * g
            a_new = project_simplex(a + v)

            b_new = fast_convolve(a_new)
            obj_new = b_new.max()

            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
            a = a_new

        # Adaptive adjustments
        if no_improve > 3000:
            a = project_simplex(best_a + rng.standard_normal(n) * kick_size)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.6, lr_init * 0.05)
            no_improve = 0

            cur_lr *= 0.92

            cur_lr *= 0.97
