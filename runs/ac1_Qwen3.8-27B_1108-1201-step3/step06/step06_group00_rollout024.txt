            if 0 <= i < n:

from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize

    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)

        # Get indices of top_k values
        top_idx = np.argpartition(conv, -top_k)[-top_k:]
        A_sub = A[top_idx, :]
        b_sub = np.full(top_k, M)
    else:
        A_sub = A
        b_sub = np.full(2*n-1, M)
    
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_sub, b_ub=b_sub, bounds=bounds, method='highs',
                 options={'presolve': True})
    
    if res.success:
        return np.maximum(res.x, 0), True
    return None, False


def lp_optimize(a0, time_limit, seed=42, verbose=True, top_k_frac=0.1):
    """
    LP-based alternating optimization with line search.
    """
    t_start = time.monotonic()
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
        return a, 1.0/n
    a /= S
    
    best_a = a.copy()
    best_M = fast_convolve(a).max()
    best_obj = 2.0 * n * best_M
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)
    
    top_k = max(20, int(n * top_k_frac))
    
    if verbose:
        print(f"  LP opt: n={n}, initial M={best_M:.10f}, obj={best_obj:.8f}")
    
    while time.monotonic() - t_start < time_limit:
        it += 1
        elapsed = time.monotonic() - t_start
        remaining = time_limit - elapsed
        if remaining < 3.0:
            break
        
        M_cur = fast_convolve(a).max()
        
        # Solve LP
        g, ok = solve_lp(a, M_cur, top_k=top_k)
        if not ok or g is None:
            no_improve += 5
            if no_improve > 100:
                break
            # Perturb
            a = a + rng.standard_normal(n) * 0.005
            a = np.maximum(a, 0)
            a /= a.sum()
            continue
        
        S_g = g.sum()
        if S_g < 1e-10:
            no_improve += 10
            continue
        g = g / S_g
        
        # Line search
        found = False
        for t in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1, 0.05]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-15:
                continue
            a_try /= s
            M_try = fast_convolve(a_try).max()
            if M_try < M_cur - 1e-15:
                a = a_try
                found = True
                break
        
        if not found:
            # Try a very small step
            a_try = a + 0.01 * (g - a)
            a_try = np.maximum(a_try, 0.0)
            a_try /= a_try.sum()
            M_try = fast_convolve(a_try).max()
            if M_try < best_M - 1e-15:
                a = a_try
                found = True
            else:
                # No improvement
                no_improve += 1
                if no_improve > 50:
                    # Perturb
                    a = best_a + rng.standard_normal(n) * 0.01
                    a = np.maximum(a, 0)
                    a /= a.sum()
                    no_improve = 0
                continue
        
        # Update best
        M_new = fast_convolve(a).max()
        if M_new < best_M - 1e-15:
            best_M = M_new
            best_a = a.copy()
            best_obj = 2.0 * n * best_M
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    it={it}: M={best_M:.10f}, obj={best_obj:.8f}, t={elapsed:.0f}s")
        else:
            no_improve += 1
    
    return best_a, best_M

def peak_reduction(a, n_iter=1000):
    """Reduce the peak by redistributing mass from peak-contributing elements."""
    n = len(a)
    a = a.copy()
    
    for _ in range(n_iter):
        b = fast_convolve(a)
        k_star = b.argmax()
        M = b[k_star]
        
        # Find contributing elements
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        contributors = np.arange(lo, hi + 1)
        weights = a[contributors] * a[k_star - contributors]
        
        # Move mass from the heaviest contributor to the lightest non-contributor
        max_idx = contributors[np.argmax(weights)]
        # Find an element not in contributors with smallest value
        non_contrib = np.setdiff1d(np.arange(n), contributors)
        if len(non_contrib) == 0:
            break
        min_idx = non_contrib[np.argmin(a[non_contrib])]
        
        # Transfer a small amount
        delta = 0.01 * a[max_idx]
        a[max_idx] -= delta
        a[min_idx] += delta
        a = np.maximum(a, 0)
        a /= a.sum()
    
    return a
