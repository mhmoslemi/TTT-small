import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def geometric_sequence(n, beta, skew=0.0):
    """
    Generate a geometric-like sequence: a[i] = exp(-beta*i/n + skew*(i/n - 0.5))
    normalized to sum 1. beta > 0 gives front-loaded decay.
    skew adds asymmetry in the decay rate.
    """
    x = np.arange(n) / n
    a = np.exp(-beta * x + skew * (x - 0.5) * x)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    return a


def parametric_sequence(n, alpha, beta, gamma):
    """
    a[i] = exp(-alpha*(i/n) - beta*(i/n)^2 + gamma*(i/n)^3) / Z
    3-parameter family for flexible shapes.
    """
    x = np.arange(n) / n
    a = np.exp(-alpha * x - beta * x * x + gamma * x * x * x)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    return a


def compute_subgradient(f, c, tau=10.0):
    """
    Compute smoothed subgradient of max(c) using softmax weights.
    Correct formula: dc[k]/da[j] = 2*a[k-j] for valid j.
    """
    n = len(f)
    len_c = len(c)
    k_top = max(10, int(len_c * 0.15))
    peaks = np.argpartition(c, -k_top)[-k_top:]
    pv = c[peaks]
    scaled = tau * pv
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        idx = np.arange(js, je + 1)
        # CORRECT subgradient: 2 * a[k - j]
        contrib = 2.0 * f[k - idx]
        sg[js:je + 1] += w[i] * contrib
    return sg


def subgradient_optimize(f0, time_limit, verbose=False,
                         lr_init=0.01, tau_init=5.0, tau_rate=1.005,
                         lr_decay=0.9995, restart_every=150):
    """
    Proximal subgradient descent on probability simplex with correct gradient.
    """
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    tau = tau_init
    lr = lr_init
    no_improve = 0
    it = 0
    best_tau_at_best = tau

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        sg = compute_subgradient(f, conv, tau=tau)

        # Proximal step: project f - lr * sg onto simplex
        f_new = project_simplex(f - lr * sg)

        # Evaluate
        obj_new = fast_convolve(f_new).max()

        if obj_new > M * 1.02:
            # Try smaller step
            lr *= 0.7
            if lr < 1e-8:
                break
            continue

        f = f_new

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
            best_tau_at_best = tau
            if verbose and it % 100 == 0:
                print(f"        SubGrad iter {it}: obj={best_obj:.10f}, tau={tau:.1f}, lr={lr:.6f}")
        else:
            no_improve += 1

        tau *= tau_rate
        lr = max(lr * lr_decay, 1e-8)

        if no_improve > restart_every:
            rng = np.random.default_rng(it * 7 + 13)
            noise_scale = 0.002 + 0.003 * (no_improve / restart_every)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            lr = lr_init * 0.5
            tau = best_tau_at_best
            no_improve = 0
            if no_improve > restart_every * 3:
                break

        if lr < 1e-9:
            break

    return best_f, best_obj

def coordinate_descent(f0, time_limit, verbose=False):
    """
    Cyclic coordinate descent: optimize each coordinate in turn
    while keeping sum = 1.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    converged = False

    while not converged and time.monotonic() - t0 < time_limit:
        converged = True
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj
            if f[j] < 1e-10:
                continue

            old_val = f[j]
            other_sum = 1.0 - old_val
            if other_sum < 1e-10:
                continue

            # Try a range of values for f[j], rescaling others
            best_j_obj = best_obj
            best_j_val = old_val

            # Geometric spacing of trial values
            ratios = [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 0.3]
            for ratio in ratios:
                new_val = old_val * ratio
                if new_val < 0 or new_val >= 0.95:
                    continue
                scale = (1.0 - new_val) / other_sum
                f_trial = f * scale
                f_trial[j] = new_val
                f_trial = np.maximum(f_trial, 0)
                ts = f_trial.sum()
                if ts < 1e-10:
                    continue
                f_trial = f_trial / ts
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-12:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / other_sum
                f = f * scale
                f[j] = best_j_val
                f = np.maximum(f, 0)
                f = f / f.sum()
                if best_j_obj < best_obj - 1e-12:
                    best_obj = best_j_obj
                    best_f = f.copy()
                converged = False

    return best_f, best_obj

            blended, blend_obj = blend_optimize(best_seq, a_geom, time_limit=5)
            sc = try_seq(blended, f"blend_b{beta}")
            if sc < best_score + 0.01 and rem > 50:
                t_alloc = min(rem * 0.2, 80)
                a_opt, obj = subgradient_optimize(blended, time_limit=t_alloc,
                                                  lr_init=0.003, tau_init=5.0)
                try_seq(a_opt, f"subgrad_blend_b{beta}")

    # === Phase 6: Random restarts ===
    print("\n=== Phase 6: Random Restarts ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for r in range(5):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t_alloc = min(rem / max(1, 5 - r), 100)
            if t_alloc < 15:
                t_alloc = 15

            if r == 0:
                noise = rng.standard_normal(n_best) * 0.003
                f_init = project_simplex(best_seq + noise)
            elif r == 1:
                # Blend with reversed
                f_init = 0.6 * best_seq + 0.4 * best_seq[::-1]
                f_init = np.maximum(f_init, 0)
                f_init = f_init / f_init.sum()
            elif r == 2:
                # Different geometric
                a_g = geometric_sequence(n_best, rng.uniform(1.5, 4.0))
                f_init = 0.5 * best_seq + 0.5 * a_g
                f_init = f_init / f_init.sum()
            elif r == 3:
                # Smaller perturbation
                noise = rng.standard_normal(n_best) * 0.001
                f_init = project_simplex(best_seq + noise)
            else:
                # Multiply by a window function
                x = np.arange(n_best) / n_best
                window = 1.0 + 0.3 * np.sin(2 * np.pi * x * rng.uniform(1, 3))
                f_init = best_seq * np.maximum(window, 0.1)
                f_init = f_init / f_init.sum()

            print(f"  Restart {r}: t={t_alloc:.0f}s")
            a_opt, obj = subgradient_optimize(f_init, time_limit=t_alloc,
                                              lr_init=0.005, tau_init=3.0,
                                              tau_rate=1.004, restart_every=100)
            try_seq(a_opt, f"restart_{r}")

    # === Phase 7: Final coordinate descent polish ===
    print("\n=== Phase 7: Final Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 20:
            t_alloc = min(rem * 0.4, 80)
            print(f"  Final CoordDescent: t={t_alloc:.0f}s")
            a_opt, obj = coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_coord")

    # === Phase 8: One more subgradient with different params ===
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 20:
            t_alloc = min(rem * 0.5, 60)
            print(f"  Final SubGrad (low lr): t={t_alloc:.0f}s")
            a_opt, obj = subgradient_optimize(best_seq, time_limit=t_alloc,
                                              lr_init=0.002, tau_init=20.0,
                                              tau_rate=1.001, lr_decay=0.9999,
                                              restart_every=80)
            try_seq(a_opt, "final_subgrad")

    # Fallback
    if best_seq is None:
        best_seq = geometric_sequence(768, 2.5)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = geometric_sequence(768, 2.5)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
