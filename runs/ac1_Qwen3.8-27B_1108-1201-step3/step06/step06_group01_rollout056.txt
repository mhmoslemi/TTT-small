        c = fast_convolve(a)
        c_max = c.max()
        score = 2.0 * n * c_max
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()

        # Adaptive parameters
        alpha = max(0.01, 0.5 * np.exp(-it * 0.05))
        threshold = 0.7 * c_max

        # Identify peaks that need reduction
        peak_mask = c > threshold
        n_peaks = peak_mask.sum()
        if n_peaks == 0:
            break

        # Compute spectrum
        A = rfft(a, fft_size)
        P = np.abs(A)**2

        # Compute "excess" auto-correlation
        excess = np.maximum(c - threshold, 0)
        E = rfft(excess, fft_size)

        # Modify spectrum: reduce power at frequencies that contribute to peaks
        # The idea: if the auto-correlation has excess at certain lags,
        # reduce the spectral components that create those lags
        # P_new = P / (1 + beta * |DFT(excess)|^2 / P) — simplified approach
        beta = alpha * max(1.0, c_max) * 0.1
        P_excess = np.abs(E)**2
        denom = 1.0 + beta * P_excess / (P + 1e-20)
        P_new = P / denom

        # Update amplitude, keep phase
        A_new_mag = np.sqrt(P_new)
        phase = np.angle(A)
        A_new = A_new_mag * np.exp(1j * phase)

        # Recover time domain
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        # Damped update
        damping = min(1.0, 2.0 / (1 + it))
        a = (1 - damping) * a + damping * a_new
        a = np.maximum(a, 0)
        a = a / a.sum()

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Spectral iter={it} ({elapsed:.0f}s): score={best_score:.6f} peaks={n_peaks}")

        if it > 5000:
            break

    return best_a, best_score


def lbgfs_refine_adaptive(a0, time_limit, T_init=200, T_min=5, maxiter=5000):
    """
    L-BFGS with adaptive temperature annealing.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    z0 = np.log(np.maximum(a, 1e-30))
    t0 = time.monotonic()
    best_score = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    stopped = [False]

    def _func(z):
        if time.monotonic() - t0 > time_limit or stopped[0]:
            stopped[0] = True
            return 1e15
        # Softmax from log-domain
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a_local = exp_z / exp_z.sum()
        # Compute smoothed objective
        T_current = max(T_min, T_init * np.exp(-0.5 * (time.monotonic() - t0) / max(1, time_limit * 0.5)))
        b = fast_convolve(a_local)
        max_b = b.max()
        scaled = T_current * (b - max_b)
        exp_scaled = np.exp(scaled)
        log_sum_exp = max_b + np.log(exp_scaled.sum() / T_current)
        obj = 2.0 * n * log_sum_exp
        # Track best
        score_exact = 2.0 * n * max_b
        if score_exact < best_score - 1e-13:
            best_score_local[0] = score_exact
            best_a_local[0] = a_local.copy()
        return obj

    def _jac(z):
        if time.monotonic() - t0 > time_limit or stopped[0]:
            return np.zeros_like(z)
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a_local = exp_z / exp_z.sum()
        T_current = max(T_min, T_init * np.exp(-0.5 * (time.monotonic() - t0) / max(1, time_limit * 0.5)))
        b = fast_convolve(a_local)
        max_b = b.max()
        scaled = T_current * (b - max_b)
        exp_scaled = np.exp(scaled)
        w = exp_scaled / exp_scaled.sum()
        # Gradient w.r.t. a
        # d/d a[i] sum_k w[k]*conv(a,a)[k]
        grad_a = np.zeros(n)
        # Find significant peaks
        top_k = max(10, int(0.05 * len(b)))
        peak_idx = np.where(w > w.max() * 1e-4)[0]
        if len(peak_idx) > top_k:
            top_w = np.sort(w)[::-1][top_k]
            peak_idx = peak_idx[w[peak_idx] >= top_w]
        for k in peak_idx:
            wk = w[k]
            if wk < 1e-8:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            grad_a[idx] += 2.0 * wk * a_local[k - idx]
            mirror = k - idx
            mask = mirror != idx
            if mask.any():
                grad_a[mirror[mask]] += 2.0 * wk * a_local[idx[mask]]
        # Convert to gradient w.r.t. z: grad_z = a*(grad_a - a*(grad_a a))
        S = np.dot(grad_a, a_local)
        grad_z = a_local * (grad_a - a_local * S)
        return 2.0 * n * grad_z

    best_score_local = [best_score]
    best_a_local = [best_a.copy()]

    try:
        res = minimize(
            fun=_func, x0=z0, jac=_jac, method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-9,
                     'maxfun': maxiter * 50, 'disp': False}
        )
        z_out = res.x
        z_shifted = z_out - z_out.max()
        a_out = np.exp(z_shifted)
        a_out = a_out / a_out.sum()
    except Exception:
        a_out = a

    # Update best from local tracking
    if best_score_local[0] < best_score:
        best_score = best_score_local[0]
        best_a = best_a_local[0]

    score_out = 2.0 * n * fast_convolve(a_out).max()
    if score_out < best_score:
        best_score = score_out
        best_a = a_out

    return best_a, best_score


def coordinate_descent(a0, time_limit, n_sweeps=100, verbose=True):
    """
    Coordinate descent: for each coordinate i, find the optimal value
    given all others fixed. The objective as a function of a single
    coordinate is piecewise quadratic.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break

        # Smart ordering: process coordinates near the peak first
        c = fast_convolve(a)
        peak_k = int(np.argmax(c))
        # Order: start from peak region, expand outward
        center = peak_k // 2 if peak_k < n else n - 1
        order = np.argsort(np.abs(np.arange(n) - center))

        improved_this_sweep = False
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            # Try modifying a[i]
            # The contribution of a[i] to conv(a,a)[k] is:
            #   2 * a[i] * a[k-i] (for k != 2i)
            #   a[i]^2 (for k = 2i)
            # So changing a[i] by delta changes conv[k] by:
            #   2*delta*a[k-i] + delta^2 (for k=2i)
            #   2*delta*a[k-i] (for k != 2i)
            # We want to find delta that minimizes the max of the resulting conv
            # while maintaining sum(a) = 1 (redistribute to others)

            # Simpler approach: try a few candidate values
            current = a[i]
            if current < 1e-15:
                # Try adding a small amount
                for frac in [0.001, 0.0005, 0.002]:
                    a_trial = a.copy()
                    delta = current + frac / n
                    a_trial[i] = delta
                    # Redistribute
                    others = a_trial.copy()
                    others[i] = 0
                    s_others = others.sum()
                    needed = 1.0 - delta
                    if s_others > 1e-10 and needed > 0:
                        others = others * needed / s_others
                        a_trial = others
                        a_trial[i] = delta
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved_this_sweep = True
                        break
            else:
                # Try reducing
                for frac in [0.95, 0.9, 0.8, 1.05, 1.1]:
                    new_val = current * frac
                    if new_val < 0:
                        continue
                    diff = current - new_val
                    if abs(diff) < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[i] = new_val
                    # Redistribute diff to others proportionally
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    others = a_trial[mask]
                    s_others = others.sum()
                    if s_others > 1e-10:
                        a_trial[mask] = others + diff * others / s_others
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved_this_sweep = True
                        break

        if verbose and (sweep + 1) % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc sweep={sweep+1} ({elapsed:.0f}s): score={best_score:.6f}")

        if not improved_this_sweep:
            break

    return best_a, best_score


def peak_shave_local(a0, time_limit, n_steps=3000):
    """Aggressive local peak shaving with multi-peak awareness."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        b_mid = b.copy()
        b_mid[0] = -np.inf
        b_mid[-1] = -np.inf
        peak_k = int(np.argmax(b_mid))
        if b[peak_k] <= best_score / (2.0 * n) + 1e-15:
            # Try second peak
            b_mid[peak_k] = -np.inf
            peak_k2 = int(np.argmax(b_mid))
            if b[peak_k2] <= best_score / (2.0 * n) + 1e-15:
                break
            peak_k = peak_k2

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_contrib = np.argsort(contribs)[::-1][:5]

        improved = False
        for ci in top_contrib:
            i = idx[ci]
            j = peak_k - i
            old_val = a[i]
            if old_val < 1e-14:
                continue
            for factor in [0.97, 0.95, 0.92, 0.88, 0.85]:
                new_val = old_val * factor
                diff = old_val - new_val
                if diff < 1e-15:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a[mask]
                if others.sum() < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                a_trial[mask] = others + diff * others / others.sum()
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try two-pass: reduce both i and j
            for ci in top_contrib:
                i = idx[ci]
                j = peak_k - i
                if i == j or a[j] < 1e-14 or i < 0 or j < 0 or i >= n or j >= n:
                    continue
                for f1 in [0.95, 0.92]:
                    for f2 in [0.95, 0.92]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-13:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if not improved:
                break

        if step % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave step={step} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate and return updated best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def frank_wolfe_refine(a0, time_limit, n_iters=10000, verbose=True):
    """
    Frank-Wolfe (conditional gradient) method with multi-peak awareness.
    Each step: compute subgradient, find best simplex vertex, move towards it.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute auto-correlation and find active peaks
        c = fast_convolve(a)
        c_max = c.max()
        score = 2.0 * n * c_max
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()

        # Softmax weighting over peaks
        temp = 0.01 * c_max
        scaled = (c - c_max) / temp
        # Only consider peaks above threshold
        threshold = c_max * 0.95
        active = c > threshold
        w = np.zeros_like(c)
        w[active] = np.exp(scaled[active])
        ws = w.sum()
        if ws < 1e-15:
            w[int(np.argmax(c))] = 1.0
        else:
            w /= ws

        # Get top peaks for gradient
        peak_indices = np.where(active)[0]
        if len(peak_indices) > 50:
            top_w = np.sort(w)[::-1][50]
            peak_indices = peak_indices[w[peak_indices] >= top_w]
        peak_weights = w[peak_indices]

        # Compute subgradient
        grad = compute_subgradient_fast(a, peak_indices, peak_weights)

        # Frank-Wolfe: find vertex i* = argmin grad[i]
        i_star = int(np.argmin(grad))
        e_star = np.zeros(n)
        e_star[i_star] = 1.0

        # Adaptive step size
        if it > 0:
            step = 2.0 / (it + 2)
        else:
            step = 0.5

        # Update
        a_new = (1 - step) * a + step * e_star
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = a_new

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW iter={it} ({elapsed:.0f}s): score={best_score:.6f} step={step:.4f}")

    return best_a, best_score


def propose_candidate():
    """
    Multi-phase optimization:
    1. Fictitious play (alternating LP) at parent length and nearby.
    2. Spectral refinement.
    3. Frank-Wolfe with multi-peak subgradient.
    4. Adaptive-temperature L-BFGS.
    5. Coordinate descent.
    6. Peak shaving.
    7. Final micro-polish.
    """
    deadline = time.monotonic() + 1785.0

    best_seq = None
    best_score = float('inf')

    solver = _lp_solver_name()
    print(f"Using LP solver: {solver}")

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        # Fallback: uniform sequence
        n_init = 768
        parent = np.ones(n_init) / n_init
        # Add structure
        t = np.linspace(0, 2 * np.pi, n_init)
        parent = parent * (1 + 0.1 * np.cos(t) + 0.05 * np.cos(3 * t))
        parent = np.maximum(parent, 0)
        parent = parent / parent.sum()
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Fallback initial: n={len(parent)}, score={best_score:.6f}")

    n_parent = len(parent)

    # === Phase 1: Fictitious Play ===
    print(f"\n=== Phase 1: Fictitious Play (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 600
    if rem > 60:
        t_fp = min(rem * 0.7, 700.0)
        print(f"  FP n={n_parent}, budget={t_fp:.0f}s")
        a_fp, score_fp, iters = fictitious_play_optimize(
            n_parent, t_fp, initial=parent, solver=solver, verbose=True
        )
        best_seq, best_score = check_and_update(a_fp, f"FP_n{n_parent}", best_seq, best_score)

    # === Phase 2: Spectral Refinement ===
    print(f"\n=== Phase 2: Spectral Refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_spec = min(rem * 0.5, 150.0)
            print(f"  Spectral n={len(best_seq)}, budget={t_spec:.0f}s")
            a_spec, _ = spectral_refine(best_seq, t_spec, verbose=True)
            best_seq, best_score = check_and_update(a_spec, "spectral", best_seq, best_score)

    # === Phase 3: Frank-Wolfe ===
    print(f"\n=== Phase 3: Frank-Wolfe (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_fw = min(rem * 0.5, 200.0)
            print(f"  FW n={len(best_seq)}, budget={t_fw:.0f}s")
            a_fw, _ = frank_wolfe_refine(best_seq, t_fw, verbose=True)
            best_seq, best_score = check_and_update(a_fw, "frank_wolfe", best_seq, best_score)

    # === Phase 4: Adaptive L-BFGS ===
    print(f"\n=== Phase 4: Adaptive L-BFGS (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            for T_val in [100, 30, 10]:
                rem = deadline - time.monotonic() - 100
                if rem < 15:
                    break
                t_lb = min(rem * 0.5, 100.0)
                print(f"  L-BFGS T={T_val}, n={len(best_seq)}, budget={t_lb:.0f}s")
                a_lb, _ = lbgfs_refine_adaptive(best_seq, t_lb, T_init=T_val, T_min=max(3, T_val // 10))
                best_seq, best_score = check_and_update(a_lb, f"lbgfs_T{T_val}", best_seq, best_score)

    # === Phase 5: Coordinate Descent ===
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_cd = min(rem * 0.7, 120.0)
            print(f"  CoordDesc n={len(best_seq)}, budget={t_cd:.0f}s")
            a_cd, _ = coordinate_descent(best_seq, t_cd, verbose=True)
            best_seq, best_score = check_and_update(a_cd, "coord_desc", best_seq, best_score)

    # === Phase 6: Peak Shaving ===
    print(f"\n=== Phase 6: Peak Shaving (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 50
        if rem > 15:
            t_ps = min(rem * 0.8, 80.0)
            a_ps, _ = peak_shave_local(best_seq, t_ps)
            best_seq, best_score = check_and_update(a_ps, "peak_shave", best_seq, best_score)

    # === Phase 7: Second FP round with refined start ===
    print(f"\n=== Phase 7: FP Round 2 (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 40
        if rem > 30:
            t_fp2 = min(rem * 0.7, 150.0)
            print(f"  FP2 n={len(best_seq)}, budget={t_fp2:.0f}s")
            a_fp2, score_fp2, iters2 = fictitious_play_optimize(
                len(best_seq), t_fp2, initial=best_seq, solver=solver, verbose=False
            )
            best_seq, best_score = check_and_update(a_fp2, "FP2", best_seq, best_score)

            # Quick spectral + L-BFGS on result
            if best_seq is not None:
                rem = deadline - time.monotonic() - 20
                if rem > 15:
                    a_s2, _ = spectral_refine(best_seq, min(rem * 0.4, 30.0), verbose=False)
                    best_seq, best_score = check_and_update(a_s2, "spectral2", best_seq, best_score)
                    rem = deadline - time.monotonic() - 10
                    if rem > 10:
                        a_l2, _ = lbgfs_refine_adaptive(best_seq, min(rem * 0.7, 25.0), T_init=20, T_min=5)
                        best_seq, best_score = check_and_update(a_l2, "lbgfs2", best_seq, best_score)

    # === Phase 8: Try nearby lengths ===
    print(f"\n=== Phase 8: Nearby Lengths (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        alt_lengths = [n_best - 64, n_best + 64, n_best + 128]
        alt_lengths = [x for x in alt_lengths if 128 <= x <= 4096]

        for alt_n in alt_lengths[:2]:
            rem = deadline - time.monotonic() - 30
            if rem < 30:
                break
            t_alt = min(rem * 0.5, 100.0)
            init_alt = interpolate_sequence(best_seq, alt_n)
            # Quick FP
            a_alt, score_alt, _ = fictitious_play_optimize(
                alt_n, t_alt, initial=init_alt, solver=solver, verbose=False
            )
            best_seq, best_score = check_and_update(a_alt, f"alt_n{alt_n}", best_seq, best_score)

    # === Phase 9: Final micro-polish ===
    print(f"\n=== Phase 9: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            a_final, _ = peak_shave_local(best_seq, min(rem * 0.8, 20.0))
            best_seq, best_score = check_and_update(a_final, "final_ps", best_seq, best_score)

        rem = deadline - time.monotonic() - 3
        if rem > 5:
            a_final2, _ = coordinate_descent(best_seq, min(rem * 0.8, 15.0), n_sweeps=20, verbose=False)
            best_seq, best_score = check_and_update(a_final2, "final_cd", best_seq, best_score)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
