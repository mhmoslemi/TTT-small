import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver, preferring PDLP."""
    global _lp_solver
    if '_lp_solver' in globals():
        return _lp_solver
    for solver_name in [cp.PDLP, cp.GUROBI, cp.MOSEK, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b (non-negative, sum=1) minimizing max_k conv(a, b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.ECOS, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def fictitious_play(n, time_limit, initial=None, solver=None, verbose=True):
    """Run fictitious play for the auto-correlation minimization game."""
    if solver is None:
        solver = get_lp_solver()
    t0 = monotonic()
    if initial is None:
        a = np.ones(n) / n
        t = np.linspace(0, 2 * np.pi, n)
        a = a + 0.01 * np.cos(t)
        a = np.maximum(a, 0)
        a = a / a.sum()
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    b = a.copy()
    a_avg = a.copy().astype(np.float64)
    b_avg = b.copy().astype(np.float64)
    best_score = float('inf')
    best_a = a.copy()
    iters = 0
    M_a = None
    M_b = None

    while monotonic() - t0 < time_limit:
        iters += 1
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
        a_new, M_a = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new

        a_avg = (iters * a_avg + a) / (iters + 1)
        b_avg = (iters * b_avg + b) / (iters + 1)

        candidates = [a, b, 0.5*(a+b), a_avg, 0.5*(a_avg+b_avg)]
        for cand in candidates:
            cand = np.maximum(cand, 0)
            s = cand.sum()
            if s < 1e-10:
                continue
            cand_norm = cand / s
            max_conv = fast_convolve(cand_norm).max()
            score = 2.0 * n * max_conv

                best_score = score
                best_a = cand_norm.copy()

        if verbose and iters % 20 == 0:
            elapsed = monotonic() - t0
            ma_str = f"{M_a:.6f}" if M_a is not None else "N/A"
            mb_str = f"{M_b:.6f}" if M_b is not None else "N/A"
            print(f"  FP n={n} iter={iters} ({elapsed:.0f}s): M_a={ma_str} M_b={mb_str} best={best_score:.6f}")

def damped_lp_refine(a0, time_limit, solver=None, max_iters=3000):
    """Damped alternating LP refinement."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()

    for it in range(max_iters):
        if monotonic() - t0 > time_limit:
            break
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            continue
        alpha = min(1.0, 3.0 / (it + 1))
        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        if (it + 1) % 200 == 0:
            elapsed = monotonic() - t0
            print(f"    DampedLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def lbgfs_smooth(a0, time_limit, T=100, maxiter=5000):
    """L-BFGS refinement in log-space with soft-max smoothing."""
    n = len(a0)
    z0 = np.log(np.maximum(a0, 1e-300))
    z0 = z0 - z0.mean()  # Center for numerical stability
    t0 = monotonic()
    stopped = [False]

    def objective(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15, np.zeros(n)
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        # Soft-max of b with temperature T
        max_b = b.max()
        scaled_b = (b - max_b) / T
        exp_b = np.exp(scaled_b)
        lse = max_b + T * np.log(exp_b.sum())
        obj = 2.0 * n * lse / T  # Actually this should be 2n * (1/T) * log(sum(exp(b/T)))
        # Wait, let me redo this. We want to approximate max(b) by (1/T)*log(sum(exp(b_i/T)))
        # So obj = 2n * (1/T) * log(sum(exp(b_i/T)))
        # = 2n * (max_b/T + log(sum(exp((b_i - max_b)/T))))
        
        # Gradient: d obj / d a[j] for each j
        # d/d a[j] of max_k (a*a)[k] ≈ 2 * sum_k w_k * a[k-j] (cross-correlation)
        w = exp_b / exp_b.sum()
        # cross_corr(w, a)[j] = sum_k w[k] * a[k-j] (with appropriate bounds)
        # Actually: gradient of (a*a)[k] wrt a[j] = 2*a[k-j] (if valid)
        # So gradient of sum_k w_k * (a*a)[k] wrt a[j] = 2 * sum_k w_k * a[k-j]
        # This is 2 * (w * a)[j] in the correlation sense
        # cross_corr(w, a)[j] = sum_k w[k] * a[k-j + (n-1)] ... hmm let me use direct computation
        
        # Actually, let's use: grad[k] = 2 * sum_i w[i] * a[i+k] (for valid indices)
        # This is 2 * (a correlated with w)
        # Using FFT: correlation of w and a
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        # Correlation: ifft(conj(W) * A)
        corr = irfft(np.conj(W) * A, fft_size)[:n]
        grad_a = 2.0 * n * corr  # This is d/d a of (2n * soft_max)
        
        # Transform from a-space to z-space
        # a = softmax(z), so d a / d z = a * (I - a*a^T) (Jacobian of softmax)
        # grad_z = a * (grad_a - a * sum(grad_a * a))
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - a * S)
        
        return 2.0 * n * (max_b + T * np.log(exp_b.sum())) / T, grad_z

    def func(z):
        val, _ = objective(z)
        return val

    def jac(z):
        _, g = objective(z)
        return g

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                     'maxfun': maxiter * 20, 'disp': False}
        )
        z_final = res.x
        z_shifted = z_final - z_final.max()
        a_out = np.exp(z_shifted)
        a_out = a_out / a_out.sum()
    except Exception:
        a_out = a0

    return a_out

def subgradient_optimize(a0, time_limit, lr_init=0.005, momentum=0.9, top_k=30):
    """Subgradient method with momentum and adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    v = np.zeros(n)
    lr = lr_init
    t0 = monotonic()
    no_improve = 0
    step_count = 0

    while monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        
        # Temperature for weighting
        temp = 0.003 * max(pv.max(), 1e-10)
        scaled = (pv - pv.max()) / temp
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

        v = momentum * v - lr * sg

        # Project onto simplex
        a_candidate = a + v
        u = np.sort(a_candidate)[::-1]
        cssv = np.cumsum(u)
        cond = u * np.arange(1, n + 1) > (cssv - 1)
        if cond.any():
            rho = np.nonzero(cond)[0][-1]
            theta = (cssv[rho] - 1.0) / (rho + 1.0)
            a_new = np.maximum(a_candidate - theta, 0.0)
        else:
            a_new = np.ones(n) / n

        a_new = a_new / a_new.sum()
        max_conv = fast_convolve(a_new).max()
        score = 2.0 * n * max_conv
        step_count += 1

        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.999
        if no_improve > 200:
            a = best_a + np.random.standard_normal(n) * 0.002
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            lr = max(lr * 2, 0.003)
            no_improve = 0

        if step_count % 1000 == 0:
            elapsed = monotonic() - t0
            print(f"    SubGrad {elapsed:.0f}s: score={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate and return updated best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from time import monotonic


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver, preferring PDLP for speed."""
    global _lp_solver
    if '_lp_solver' in globals():
        return _lp_solver
    for solver_name in [cp.PDLP, cp.GUROBI, cp.MOSEK, cp.SCS, cp.ECOS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b (non-negative, sum=1) minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.PDLP, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def solve_lp_best_response_symmetric(a_fixed, solver=None):
    """LP best response with palindromic constraint on b."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    half = (n + 1) // 2
    b_half = cp.Variable(half, nonneg=True)
    M_var = cp.Variable()

    # Build full b from half
    if n % 2 == 0:
        b_full = cp.hstack([b_half, cp.flip(b_half)])
    else:
        b_full = cp.hstack([b_half, cp.flip(b_half[1:])])

    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_full <= M_var,
        cp.sum(b_full) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.PDLP, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_half_val = np.asarray(b_half.value, dtype=np.float64)
    b_half_val = np.maximum(b_half_val, 0)
    if n % 2 == 0:
        b_val = np.concatenate([b_half_val, b_half_val[::-1]])
    else:
        b_val = np.concatenate([b_half_val, b_half_val[1:][::-1]])
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def fictitious_play(n, time_limit, initial=None, solver=None, verbose=True, use_symmetry=False):
    """Run fictitious play for auto-correlation minimization."""
    if solver is None:
        solver = get_lp_solver()
    t0 = monotonic()
    if initial is None:
        a = np.ones(n) / n
        t = np.linspace(0, 2 * np.pi, n)
        a = a + 0.005 * np.cos(3 * t)
        a = np.maximum(a, 0)
        a = a / a.sum()
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    b = a.copy()
    a_avg = a.copy().astype(np.float64)
    b_avg = b.copy().astype(np.float64)
    best_score = float('inf')
    best_a = a.copy()
    iters = 0
    M_a = None
    M_b = None
    stagnation = 0

    while monotonic() - t0 < time_limit:
        iters += 1

        # Solve LP best response for b given a
        if use_symmetry:
            b_new, M_b = solve_lp_best_response_symmetric(a, solver)
        else:
            b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new

        # Solve LP best response for a given b
        if use_symmetry:
            a_new, M_a = solve_lp_best_response_symmetric(b, solver)
        else:
            a_new, M_a = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new

        # Update running averages with decreasing weight on new samples
        if iters <= 100:
            a_avg = (iters * a_avg + a) / (iters + 1)
            b_avg = (iters * b_avg + b) / (iters + 1)
        else:
            a_avg = (iters * a_avg + a) / (iters + 1)
            b_avg = (iters * b_avg + b) / (iters + 1)

        # Evaluate candidates
        candidates = [a, b, 0.5 * (a + b), a_avg, 0.5 * (a_avg + b_avg)]
        for cand in candidates:
            cand = np.maximum(cand, 0)
            s = cand.sum()
            if s < 1e-10:
                continue
            cand_norm = cand / s
            max_conv = fast_convolve(cand_norm).max()
            score = 2.0 * n * max_conv
            if score < best_score - 1e-12:
                best_score = score
                best_a = cand_norm.copy()
                stagnation = 0
            else:
                stagnation += 1

        # If stagnating, add small perturbation to escape
        if stagnation > 200:
            noise = np.random.standard_normal(n) * 0.001
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            b = b + np.random.standard_normal(n) * 0.001
            b = np.maximum(b, 0)
            b = b / b.sum()
            stagnation = 0

        if verbose and iters % 25 == 0:
            elapsed = monotonic() - t0
            ma_str = f"{M_a:.6f}" if M_a is not None else "N/A"
            mb_str = f"{M_b:.6f}" if M_b is not None else "N/A"
            print(f"  FP n={n} iter={iters} ({elapsed:.0f}s): M_a={ma_str} M_b={mb_str} best={best_score:.6f}")

    return best_a, best_score, iters


def damped_lp_refine(a0, time_limit, solver=None, max_iters=5000):
    """Damped alternating LP refinement with adaptive step size."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()

    for it in range(max_iters):
        if monotonic() - t0 > time_limit:
            break
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            continue
        alpha = min(1.0, 2.0 / (it + 1))
        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        if (it + 1) % 300 == 0:
            elapsed = monotonic() - t0
            print(f"    DampedLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def lbgfs_smooth(a0, time_limit, T=100, maxiter=5000):
    """L-BFGS refinement in log-space with soft-max smoothing."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-30)
    z0 = np.log(a_safe)
    z0 = z0 - z0.mean()
    t0 = monotonic()
    stopped = [False]

    def func(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        max_b = b.max()
        scaled_b = (b - max_b) / T
        exp_b = np.exp(scaled_b)
        lse = max_b + T * np.log(exp_b.sum())
        return 2.0 * n * lse / T

    def jac(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros(n)
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        max_b = b.max()
        scaled_b = (b - max_b) / T
        exp_b = np.exp(scaled_b)
        w = exp_b / exp_b.sum()

        # Gradient of sum_k w[k]*(a*a)[k] wrt a = 2 * sum_k w[k] * a[k-m]
        # Use direct computation
        grad_a = np.zeros(n)
        for k in range(len(w)):
            if w[k] < 1e-10:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]

        grad_a *= 2.0 * n
        # Transform to z-space
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - a * S)
        return grad_z

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                     'maxfun': maxiter * 15, 'disp': False}
        )
        if not stopped[0]:
            z_final = res.x
            z_shifted = z_final - z_final.max()
            a_out = np.exp(z_shifted)
            a_out = a_out / a_out.sum()
        else:
            a_out = a0
    except Exception:
        a_out = a0

    return a_out


def subgradient_optimize(a0, time_limit, lr_init=0.004, momentum=0.9, top_k=40):
    """Subgradient method with momentum and adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    v = np.zeros(n)
    lr = lr_init
    t0 = monotonic()
    no_improve = 0
    step_count = 0

    while monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        temp = 0.002 * max(pv.max(), 1e-10)
        scaled = (pv - pv.max()) / temp
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

        v = momentum * v - lr * sg
        a_candidate = a + v

        # Project onto simplex
        u = np.sort(a_candidate)[::-1]
        cssv = np.cumsum(u)
        cond = u * np.arange(1, n + 1) > (cssv - 1)
        if cond.any():
            rho = np.nonzero(cond)[0][-1]
            theta = (cssv[rho] - 1.0) / (rho + 1.0)
            a_new = np.maximum(a_candidate - theta, 0.0)
        else:
            a_new = np.ones(n) / n

        a_new = a_new / a_new.sum()
        max_conv = fast_convolve(a_new).max()
        score = 2.0 * n * max_conv
        step_count += 1

        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.999
        if no_improve > 300:
            a = best_a + np.random.standard_normal(n) * 0.002
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            lr = max(lr * 2, 0.003)
            no_improve = 0

        if step_count % 2000 == 0:
            elapsed = monotonic() - t0
            print(f"    SubGrad {elapsed:.0f}s: score={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score


def peak_shave(a0, time_limit, n_steps=10000):
    """Aggressive local peak shaving with multiple strategies."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    failed = 0

    for step in range(n_steps):
        if monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))

        if b[peak_k] <= best_score / (2.0 * n) + 1e-15:
            break

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        if contribs.sum() < 1e-20:
            break

        top_contrib = np.argsort(contribs)[::-1][:5]
        improved = False

        # Strategy 1: Reduce single contributor
        for ci in top_contrib:
            i = int(idx[ci])
            old_val = a[i]
            if old_val < 1e-14:
                continue
            for factor in [0.95, 0.9, 0.8, 0.7]:
                new_val = old_val * factor
                diff = old_val - new_val
                if diff < 1e-15:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                a_trial[mask] = others + diff * others / others_sum
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        # Strategy 2: Reduce pair
        if not improved:
            for ci in top_contrib:
                i = int(idx[ci])
                j = int(peak_k - i)
                if i == j or a[j] < 1e-14:
                    continue
                for f1 in [0.9, 0.85]:
                    for f2 in [0.9, 0.85]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-13:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        # Strategy 3: Small random perturbation
        if not improved:
            for attempt in range(5):
                a_trial = a + np.random.standard_normal(n) * 0.0005
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            failed += 1
            if failed > 50:
                break
        else:
            failed = 0

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving non-negativity."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate and return updated best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining fast LP fictitious play,
    L-BFGS smoothing, subgradient descent, and peak shaving.
    Explores multiple sequence lengths with adaptive time allocation.
    """
    np.random.seed(seed)
    deadline = monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score
    else:
        best_seq = np.ones(768) / 768.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Initial: n=768, score={best_score:.6f}")

    n_parent = len(best_seq)

    # === Phase 1: Fictitious Play at multiple lengths ===
    print(f"\n=== Phase 1: Fictitious Play ===")
    n_values = sorted(set([n_parent, 1024, 2048]), key=lambda x: abs(x - n_parent))
    n_values = n_values[:3]

    rem = deadline - monotonic() - 500
    total_fp_time = max(60.0, min(700.0, rem))
    # Allocate more time to parent length
    weights = [0.5, 0.3, 0.2][:len(n_values)]

    for i, n_try in enumerate(n_values):
        rem = deadline - monotonic() - 400
        if rem < 40:
            break
        t_alloc = min(total_fp_time * weights[i], rem * 0.8)
        if t_alloc < 20:
            continue

        print(f"  FP n={n_try}, budget={t_alloc:.0f}s")
        initial = None
        if len(best_seq) == n_try:
            initial = best_seq
        else:
            initial = interpolate_sequence(best_seq, n_try)

        a_fp, score_fp, iters = fictitious_play(n_try, t_alloc, initial=initial, solver=solver, verbose=True)
        best_seq, best_score = check_and_update(a_fp, f"FP_n{n_try}_it{iters}", best_seq, best_score)

    # === Phase 2: Damped LP on best result ===
    print(f"\n=== Phase 2: Damped LP (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 300
        if rem > 30:
            t_alloc = min(rem * 0.5, 250.0)
            print(f"  DampedLP n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_dl, _ = damped_lp_refine(best_seq, t_alloc, solver=solver)
            best_seq, best_score = check_and_update(a_dl, "damped_lp", best_seq, best_score)

    # === Phase 3: L-BFGS with multiple temperatures ===
    print(f"\n=== Phase 3: L-BFGS Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        for T in [30, 80, 200]:
            rem = deadline - monotonic() - 200
            if rem < 15:
                break
            t_alloc = min(rem * 0.35, 70.0)
            print(f"  L-BFGS T={T}, n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_lb = lbgfs_smooth(best_seq, t_alloc, T=T)
            best_seq, best_score = check_and_update(a_lb, f"lbgfs_T{T}", best_seq, best_score)

    # === Phase 4: Subgradient optimization ===
    print(f"\n=== Phase 4: Subgradient (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 150
        if rem > 20:
            t_sg = min(rem * 0.5, 100.0)
            n_best = len(best_seq)
            top_k = max(20, n_best // 6)
            a_sg, _ = subgradient_optimize(best_seq, t_sg, lr_init=0.003, momentum=0.9, top_k=top_k)
            best_seq, best_score = check_and_update(a_sg, "subgrad", best_seq, best_score)

    # === Phase 5: Peak shaving ===
    print(f"\n=== Phase 5: Peak Shaving (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 100
        if rem > 10:
            t_ps = min(rem * 0.6, 80.0)
            a_ps, _ = peak_shave(best_seq, t_ps)
            best_seq, best_score = check_and_update(a_ps, "peak_shave", best_seq, best_score)

    # === Phase 6: Second FP round with refined start ===
    print(f"\n=== Phase 6: FP Round 2 (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - monotonic() - 80
        if rem > 40:
            t_alloc = min(rem * 0.5, 150.0)
            print(f"  FP round 2: n={n_best}, budget={t_alloc:.0f}s")
            a_fp2, _, iters2 = fictitious_play(n_best, t_alloc, initial=best_seq, solver=solver, verbose=False)
            best_seq, best_score = check_and_update(a_fp2, f"FP2_n{n_best}", best_seq, best_score)

            # Quick damped LP on new best
            if best_seq is not None:
                rem = deadline - monotonic() - 40
                if rem > 15:
                    a_dl2, _ = damped_lp_refine(best_seq, min(rem * 0.5, 60.0), solver=solver)
                    best_seq, best_score = check_and_update(a_dl2, "damped_lp_2", best_seq, best_score)

    # === Phase 7: Try alternative lengths ===
    print(f"\n=== Phase 7: Length Exploration (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        alt_lengths = []
        for factor in [0.8, 1.25, 1.5]:
            alt_n = int(n_best * factor)
            if 256 <= alt_n <= 4096 and alt_n != n_best and alt_n not in alt_lengths:
                alt_lengths.append(alt_n)

        for alt_n in alt_lengths[:2]:
            rem = deadline - monotonic() - 50
            if rem < 30:
                break
            t_alloc = min(rem * 0.5, 100.0)
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, _, _ = fictitious_play(alt_n, t_alloc, initial=init_alt, solver=solver, verbose=False)
            best_seq, best_score = check_and_update(a_alt, f"FP_alt_n{alt_n}", best_seq, best_score)

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 10
        if rem > 15:
            t_final = min(rem * 0.5, 40.0)
            a_final = lbgfs_smooth(best_seq, t_final, T=100, maxiter=3000)
            best_seq, best_score = check_and_update(a_final, "final_lbgfs", best_seq, best_score)

        rem = deadline - monotonic() - 5
        if rem > 8:
            t_ps = min(rem * 0.6, 15.0)
            a_ps_final, _ = peak_shave(best_seq, t_ps)
            best_seq, best_score = check_and_update(a_ps_final, "final_ps", best_seq, best_score)

    # Final verification
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
