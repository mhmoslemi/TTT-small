def simplex_project(v, n):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        rho_val = np.nonzero(cond)[0][-1]
        theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

                current_a = a_new
        except Exception as e:

def aggressive_peak_shave(a0, time_limit, n_max_iters=10000):
    """
    Aggressive peak shaving: at each step, identify the peak in auto-correlation,
    and find the best single-element perturbation that reduces it.
    Uses binary search for the optimal perturbation amount.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        # Find the peak (exclude k=0 and k=2n-2 as they are always a[0]^2 and a[n-1]^2)
        b_search = b.copy()
        b_search[0] = -np.inf
        b_search[-1] = -np.inf
        peak_k = int(np.argmax(b_search))
        peak_val = b[peak_k]
        
        if peak_val <= best_score / (2.0 * n) * (1 + 1e-10):
            # All peaks are at the optimal level
            break
        
        # The peak is at b[peak_k] = sum_i a[i]*a[peak_k-i]
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        
        # Try reducing each coefficient that contributes to the peak
        best_trial = None
        best_trial_score = best_score
        
        # Focus on the top contributors
        contribs = a[idx] * a[peak_k - idx]
        top_m = min(10, len(idx))
        top_idx = idx[np.argsort(contribs)[::-1][:top_m]]
        
        for i in top_idx:
            if a[i] < 1e-12:
                continue
            # Binary search for the reduction factor
            lo, hi = 0.0, a[i]
            for _ in range(8):
                mid = (lo + hi) / 2
                a_trial = a.copy()
                diff = a[i] - mid
                a_trial[i] = mid
                # Redistribute the difference to other elements (proportionally)
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a[mask].sum()
                if others_sum < 1e-15:
                    break
                a_trial[mask] = a[mask] + diff * a[mask] / others_sum
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_trial_score:
                    best_trial_score = score
                    best_trial = a_trial.copy()
                    lo = mid
                else:
                    hi = mid
        
        if best_trial is not None and best_trial_score < best_score - 1e-13:
            a = best_trial
            best_score = best_trial_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                # Perturb to escape
                rng = np.random.default_rng(int(time.monotonic() * 1e9) % 2**32)
                a = best_a + rng.standard_normal(n) * 0.002
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
    
    return best_a, best_score

def coordinate_descent_peaks(a0, time_limit):

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):

        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive-temperature log-sum-exp
smoothing with L-BFGS in log-space, aggressive peak shaving via binary search,
pair-swap local search, and multi-length exploration. Key improvements over
LP-based fictitious play: differentiable surrogate objectives enable gradient
descent, adaptive temperature annealing handles the non-smooth max, and
pair-swap moves provide a simple escape mechanism from local minima.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_smooth_max_grad(b, T):
    """Log-sum-exp approximation of max(b) and softmax weights."""
    b_max = b.max()
    if T < 1e-10:
        w = np.zeros(len(b))
        w[np.argmax(b)] = 1.0
        return b_max, w
    scaled = T * (b - b_max)
    exp_scaled = np.exp(scaled)
    lse = b_max + np.log(exp_scaled.sum()) / T
    w = exp_scaled / exp_scaled.sum()
    return lse, w


def convolve_gradient(w, a, n):
    """
    Compute gradient of LSE(conv(a,a)) w.r.t. a given weights w over conv output.
    grad_a[j] = 2 * sum_m a[m] * w[j+m] (cross-correlation of a and w).
    """
    size = 3 * n - 2
    fft_size = 1 << (size - 1).bit_length()
    af = a[::-1]
    W = rfft(w, fft_size)
    Af = rfft(af, fft_size)
    conv_result = irfft(Af * W, fft_size)[:size]
    c = conv_result[n - 1:2 * n - 1]
    return 2.0 * c


def simplex_project(v, n):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        idx = np.nonzero(cond)[0][-1]
        theta = (cssv[idx] - 1.0) / (idx + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def compute_score(a):
    """Compute the objective 2n*max_conv for sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def log_space_lbgfs(a0, time_limit, T_schedule=None, maxiter=5000, verbose=False):
    """
    L-BFGS optimization in log-space with adaptive temperature annealing.
    Returns (best_a, best_score).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    s = a.sum()
    a = a / s
    t0 = time.monotonic()

    if T_schedule is None:
        T_schedule = [3000, 1500, 700, 300, 100, 50, 20, 10]

    current_a = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for Ti, T in enumerate(T_schedule):
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break

        z = np.log(np.maximum(current_a, 1e-300))
        stopped = [False]

        def func(z, T=T):
            if stopped[0]:
                return 1e15
            a_s = np.exp(z - z.max())
            a_s = a_s / a_s.sum()
            b = fast_convolve(a_s)
            obj, _ = lse_smooth_max_grad(b, T)
            return 2.0 * n * obj

        def jac(z, T=T):
            if stopped[0]:
                return np.zeros_like(z)
            a_s = np.exp(z - z.max())
            a_s = a_s / a_s.sum()
            b = fast_convolve(a_s)
            _, w = lse_smooth_max_grad(b, T)
            grad_a = convolve_gradient(w, a_s, n)
            S = np.dot(grad_a, a_s)
            return a_s * (grad_a - S * a_s)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': max(50, maxiter // len(T_schedule)),
                         'gtol': 1e-12, 'ftol': 1e-15,
                         'maxfun': max(100, 2 * maxiter // len(T_schedule)),
                         'disp': False}
            )
            if not stopped[0]:
                a_new = np.exp(res.x - res.x.max())
                a_new = a_new / a_new.sum()
                a_new = np.maximum(a_new, 0)
                score = compute_score(a_new)
                if score < best_score:
                    best_score = score
                    best_a = a_new.copy()
                current_a = a_new
                if verbose:
                    print(f"      T={T}: L-BFGS score={score:.6f}")
        except Exception as e:
            if verbose:
                print(f"      L-BFGS T={T} error: {e}")

    return best_a, best_score


def aggressive_peak_shave(a0, time_limit, n_max_iters=5000, verbose=False):
    """
    Aggressive peak shaving: identify peaks and use binary search to find
    the optimal single-element reduction that minimizes the new max.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    no_improve = 0
    rng = np.random.default_rng(12345)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        b_search = b.copy()
        b_search[0] = -np.inf
        b_search[-1] = -np.inf
        peak_k = int(np.argmax(b_search))
        peak_val = b[peak_k]
        target = best_score / (2.0 * n)

        if peak_val <= target * (1 + 1e-9):
            break

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        # Focus on top contributors
        top_m = min(8, len(idx))
        top_local = np.argsort(contribs)[::-1][:top_m]
        top_idx = idx[top_local]

        best_trial = None
        best_trial_score = best_score

        for i in top_idx:
            if a[i] < 1e-12:
                continue
            # Binary search for optimal reduction
            lo, hi = 0.0, a[i]
            for _ in range(10):
                mid = (lo + hi) / 2.0
                a_trial = a.copy()
                diff = a[i] - mid
                a_trial[i] = mid
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a[mask].sum()
                if others_sum < 1e-15:
                    break
                a_trial[mask] = a[mask] + diff * a[mask] / others_sum
                a_trial = a_trial / a_trial.sum()
                score = compute_score(a_trial)
                if score < best_trial_score:
                    best_trial_score = score
                    best_trial = a_trial.copy()
                    lo = mid
                else:
                    hi = mid

        if best_trial is not None and best_trial_score < best_score - 1e-12:
            a = best_trial
            best_score = best_trial_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                a = best_a + rng.standard_normal(n) * 0.003
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0

        if verbose and (it + 1) % 500 == 0:
            print(f"      PeakShave iter={it+1}: score={best_score:.6f}")

    return best_a, best_score


def pair_swap_search(a0, time_limit, verbose=False):
    """
    Pair-swap local search: transfer small amounts between pairs of elements
    to reduce the peak auto-correlation. Uses random sampling of pairs.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Sample a random pair
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        a_i = a[i]
        a_j = a[j]
        if a_i < 1e-12 and a_j < 1e-12:
            continue

        # Try transferring from i to j (and vice versa)
        improved = False
        for src, dst in [(i, j), (j, i)]:
            a_src = a[src]
            if a_src < 1e-12:
                continue
            for frac in [0.02, 0.05, 0.1, 0.2, 0.3]:
                amt = a_src * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = a_trial / a_trial.sum()
                score = compute_score(a_trial)
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            no_improve += 1
            if no_improve > 200:
                # Perturb from best
                a = best_a + rng.standard_normal(n) * 0.002
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
        else:
            no_improve = 0

        if verbose and it % 1000 == 0:
            print(f"      PairSwap iter={it}: score={best_score:.6f}")

    return best_a, best_score


def targeted_peak_reduction(a0, time_limit, verbose=False):
    """
    Directly target the peak: for the dominant peak, find which pair of
    (i,j) with i+j=peak_k to adjust, and optimize the transfer amount.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        b_search = b.copy()
        b_search[0] = -np.inf
        b_search[-1] = -np.inf
        peak_k = int(np.argmax(b_search))
        peak_val = b[peak_k]
        target = best_score / (2.0 * n)

        if peak_val <= target * (1 + 1e-8):
            break

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)

        # Find the pair (i, peak_k-i) that contributes most to the peak
        contribs = a[idx] * a[peak_k - idx]
        top_m = min(6, len(idx))
        top_local = np.argsort(contribs)[::-1][:top_m]

        improved = False
        for li in top_local:
            i = idx[li]
            j = peak_k - i
            if i == j:
                # Single element at the peak center: reduce it and redistribute
                if a[i] < 1e-12:
                    continue
                for frac in [0.05, 0.1, 0.2, 0.3]:
                    a_trial = a.copy()
                    amt = a[i] * frac
                    a_trial[i] -= amt
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    others = a[mask]
                    osum = others.sum()
                    if osum < 1e-15:
                        continue
                    a_trial[mask] = others + amt * others / osum
                    a_trial = a_trial / a_trial.sum()
                    score = compute_score(a_trial)
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
            else:
                # Transfer between i and j to reduce their product
                # a[i]*a[j] is maximized when a[i]=a[j], minimized when one is 0
                # But we want to minimize the TOTAL peak, not just this term
                a_i, a_j = a[i], a[j]
                if a_i + a_j < 1e-12:
                    continue
                # Try making them more unequal
                for t_frac in [0.1, 0.2, 0.3, 0.5]:
                    # Move t_frac of the difference (making them more unequal)
                    diff = a_i - a_j
                    if abs(diff) < 1e-14:
                        amt = (a_i + a_j) * t_frac * 0.5
                    else:
                        amt = abs(diff) * t_frac + min(a_i, a_j) * t_frac
                    amt = min(amt, min(a_i, a_j) * 0.9)
                    if amt < 1e-15:
                        continue
                    if a_i > a_j:
                        a_trial = a.copy()
                        a_trial[i] -= amt
                        a_trial[j] += amt
                    else:
                        a_trial = a.copy()
                        a_trial[j] -= amt
                        a_trial[i] += amt
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    score = compute_score(a_trial)
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break

            if improved:
                break

        if not improved:
            no_improve += 1
            if no_improve > 50:
                a = best_a + rng.standard_normal(n) * 0.002
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
        else:
            no_improve = 0

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
