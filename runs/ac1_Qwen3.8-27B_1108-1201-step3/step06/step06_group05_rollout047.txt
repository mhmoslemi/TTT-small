def softmax_gd_optimize(f0, time_limit, verbose=False):

    f = np.maximum(f0.astype(np.float64), 1e-10)
    f = f / f.sum()
    n = len(f)

    theta = np.log(f)  # f = softmax(theta) (approximately, since sum(exp(theta)) != 1)
    # Actually, let's just use theta = log(f) and normalize
    # softmax(theta) = exp(theta) / sum(exp(theta))
    # We want softmax(theta) ≈ f, so theta = log(f) works (up to a constant)
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    lr = 0.1
    beta_schedule = [4, 8, 16, 32, 64, 128, 256]
    
    for beta in beta_schedule:
        stage_time = time_limit / len(beta_schedule)
        stage_start = time.monotonic()
        it = 0

            it += 1
            f = np.exp(theta - theta.max())
            f = f / f.sum()
            
            c = fast_convolve(f)
            g_f = compute_smoothed_gradient(f, c, beta)
            
            # Chain rule: d/d theta = (diag(f) - f f^T) @ g_f
            # This is the Jacobian of softmax times the gradient in f-space
            g_theta = f * (g_f - np.dot(f, g_f))
            
            theta = theta - lr * g_theta
            # Numerical stability
            theta = theta - theta.mean()  # center to prevent overflow
            
            f_new = np.exp(theta - theta.max())
            f_new = f_new / f_new.sum()
            
            c_new = fast_convolve(f_new)
            obj_new = c_new.max()
            
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_f = f_new.copy()
                no_improve = 0
            else:
                no_improve += 1

                lr *= 0.7
                if lr < 1e-6:
                    break

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_gradient(f, c, beta):
    n = len(f)
    scaled = beta * c
    scaled -= scaled.max()
    exp_scaled = np.exp(scaled)
    w = exp_scaled / exp_scaled.sum()
    w_rev = w[::-1].copy()
    conv_len = n + len(w_rev) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv = np.fft.irfft(F * W, fft_size)[:conv_len]
    indices = np.arange(2 * n - 2, 2 * n - 2 - n, -1)
    g = 2.0 * conv[indices]
    return g


def compute_pnorm_gradient(f, c, p):
    n = len(f)
    max_c = np.max(c)
    if max_c < 1e-15:
        return np.zeros(n)
    c_norm = c / max_c
    w = c_norm ** (p - 1)
    w = w / w.sum()
    w_rev = w[::-1].copy()
    conv_len = n + len(w_rev) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv = np.fft.irfft(F * W, fft_size)[:conv_len]
    indices = np.arange(2 * n - 2, 2 * n - 2 - n, -1)
    g = 2.0 * conv[indices]
    return g


def softmax_gd_optimize(f0, time_limit, verbose=False):
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)
    theta = np.log(f + 1e-30)
    theta -= theta.mean()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    beta1 = 0.75
    beta2 = 0.95
    eps = 1e-10
    stages = [(4, 0.12), (8, 0.12), (16, 0.12), (32, 0.15), (64, 0.18), (128, 0.18), (256, 0.13)]
    cum_frac = 0.0
    for beta, dur_frac in stages:
        stage_end = t0 + time_limit * (cum_frac + dur_frac)
        cum_frac += dur_frac
        lr = 2.0 / n
        it = 0
        no_improve = 0
        while time.monotonic() < min(stage_end, t0 + time_limit * 0.98):
            it += 1
            f = np.exp(theta)
            f = f / f.sum()
            c = fast_convolve(f)
            g_f = compute_smoothed_gradient(f, c, beta)
            g_theta = f * (g_f - np.dot(f, g_f))
            m_adam = beta1 * m_adam + (1.0 - beta1) * g_theta
            v_adam = beta2 * v_adam + (1.0 - beta2) * g_theta ** 2
            m_hat = m_adam / (1.0 - beta1 ** it)
            v_hat = v_adam / (1.0 - beta2 ** it)
            theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)
            theta -= theta.mean()
            if np.max(np.abs(theta)) > 50:
                theta = np.clip(theta, -50, 50)
            f_new = np.exp(theta)
            f_new = f_new / f_new.sum()
            c_new = fast_convolve(f_new)
            obj_new = c_new.max()
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_f = f_new.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 40:
                rng = np.random.default_rng(it * 11 + int(beta))
                noise = 0.002 / np.sqrt(beta / 4.0)
                f_pert = np.maximum(best_f + rng.standard_normal(n) * noise, 1e-12)
                f_pert = f_pert / f_pert.sum()
                theta = np.log(f_pert + 1e-30)
                theta -= theta.mean()
                m_adam = np.zeros(n)
                v_adam = np.zeros(n)
                lr = 2.0 / n
                no_improve = 0
            if verbose and it % 500 == 0:
                print(f"      softmaxGD beta={beta} it={it} obj={c_new.max():.10f} best={best_obj:.10f}")
        if verbose:
            print(f"    softmaxGD beta={beta} done: best={best_obj:.10f}")
    return best_f, best_obj


def mirror_descent_optimize(f0, time_limit, verbose=False):
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    eta = 1.0 / (2.0 * np.sqrt(n) * np.log(n + 1))
    it = 0
    no_improve = 0
    top_k = min(20, 2 * n - 1)
    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(f)
        M = c.max()
        k_idx = np.argpartition(c, -top_k)[-top_k:]
        weights = c[k_idx] - np.min(c)
        weights = np.maximum(weights, 0.0)
        if weights.sum() < 1e-15:
            weights = np.ones(len(k_idx))
        weights = weights / weights.sum()
        sg = np.zeros(n)
        for i in range(len(k_idx)):
            if weights[i] < 1e-12:
                continue
            k = int(k_idx[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += weights[i] * 2.0 * f[k - idx]
        f_new = f * np.exp(-eta * sg)
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()
        c_new = fast_convolve(f_new)
        obj_new = c_new.max()
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            eta = min(eta * 1.01, 1.0)
        else:
            f = f_new
            no_improve += 1
            eta *= 0.97
        if no_improve > 100:
            rng = np.random.default_rng(it * 17 + 3)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.001, 1e-12)
            f = f / f.sum()
            eta = 1.0 / (2.0 * np.sqrt(n) * np.log(n + 1))
            no_improve = 0
        if verbose and it % 100 == 0:
            print(f"      mirror it={it} obj={c_new.max():.10f} best={best_obj:.10f}")
    return best_f, best_obj


def solve_lp_active(f, M, n, active_idx=None):
    c = fast_convolve(f)
    if active_idx is None or len(active_idx) < 5:
        top_k = min(80, 2 * n - 1)
        active_idx = np.argpartition(c, -top_k)[-top_k:]
    A_ub = np.zeros((len(active_idx), n))
    for i, k in enumerate(active_idx):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        A_ub[i, idx] = f[k - idx]
    c_obj = -np.ones(n)
    b_ub = np.full(len(active_idx), M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x, active_idx
    return None, active_idx

def kkt_check(f, n, tol=1e-8):
    c = fast_convolve(f)
    M = c.max()
    active = np.where(np.abs(c - M) < M * tol)[0]
    if len(active) == 0:
        return False, 0.0
    sg = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * f[k - idx]
    sg /= len(active)
    positive = f > 1e-10
    if not positive.any():
        return False, 0.0
    vals = sg[positive]
    spread = vals.max() - vals.min()
    return spread < M * tol, spread


def coordinate_polish(f0, time_limit, verbose=False):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f0, fast_convolve(f0).max()
    f = f / s
    n = len(f)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-14:
                continue
            best_j_obj = best_obj
            best_j_val = old_val
            for mult in [0.7, 0.8, 0.85, 0.9, 1.1, 1.15, 1.2, 1.4]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.4:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial = f_trial / ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        if not improved:
            break
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    kernel = np.array([0.05, 0.15, 0.6, 0.15, 0.05])
    f_smooth = np.convolve(f_interp, kernel, mode='same')
    f_smooth = np.maximum(f_smooth, 0)
    ss = f_smooth.sum()
    if ss > 1e-15:
        f_smooth /= ss
    return f_smooth

            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_raised_cosine(768)
        try_seq(parent, "default_parent")

    n_parent = len(parent)
    print("\n=== Phase 1: Initialization & Screening ===")
    t1 = time.monotonic()
    candidates = []
    candidates.append(("rc", make_raised_cosine(n_parent)))
    candidates.append(("gauss", make_gaussian(n_parent, 0.3)))
    candidates.append(("gauss_w", make_gaussian(n_parent, 0.35)))
    candidates.append(("bell3", make_bell_power(n_parent, 0.8, 3)))
    candidates.append(("bell5", make_bell_power(n_parent, 0.7, 5)))
    candidates.append(("flattop", make_flat_top(n_parent, 0.5)))
    candidates.append(("flattop_w", make_flat_top(n_parent, 0.6)))
    best_init = None
    best_init_obj = float('inf')
    for name, a in candidates:
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            continue
        a = a / s
        c = fast_convolve(a)
        obj = 2 * len(a) * c.max()
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a
        try_seq(a, f"init_{name}")
    for trial in range(4):
        noise = rng.standard_normal(n_parent) * 0.008
        a_pert = np.maximum(parent + noise, 0.0)
        a_pert = a_pert / a_pert.sum()
        c = fast_convolve(a_pert)
        obj = 2 * n_parent * c.max()
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a_pert
        try_seq(a_pert, f"init_pert{trial}")
    if not np.allclose(best_init, parent) and fast_convolve(parent).max() < best_init_obj / (2 * n_parent):
        best_init = parent
    print(f"  Best init obj={best_init_obj:.8f}")
    print(f"  Phase 1 time: {time.monotonic()-t1:.1f}s")

    print("\n=== Phase 2: Softmax GD (main workhorse) ===")
    t2 = time.monotonic()
    rem = deadline - time.monotonic() - 700
    t_alloc = max(60, min(rem, 400))
    print(f"  Allocation: {t_alloc:.0f}s, n={len(best_init)}")
    a_opt, obj = softmax_gd_optimize(best_init, time_limit=t_alloc, verbose=True)
    try_seq(a_opt, "softmax_gd")
    if not np.allclose(best_init, parent):
        rem2 = deadline - time.monotonic() - 600
        if rem2 > 80:
            t_alloc2 = min(rem2 * 0.4, 150)
            a_opt2, _ = softmax_gd_optimize(parent, time_limit=t_alloc2)
            try_seq(a_opt2, "softmax_gd_parent")
    print(f"  Phase 2 time: {time.monotonic()-t2:.1f}s")

    print("\n=== Phase 3: Mirror Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 500
        if rem > 60:
            t_alloc = min(rem * 0.3, 150)
            print(f"  Allocation: {t_alloc:.0f}s")
            a_opt, _ = mirror_descent_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "mirror")
        print(f"  Phase 3 time: {time.monotonic()-t2:.1f}s")

    print("\n=== Phase 4: LP-FW High Precision ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 80:
            t_alloc = min(rem * 0.6, 350)
            print(f"  Allocation: {t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "lpfw_main")
            converged, spread = kkt_check(a_opt, len(a_opt))
            print(f"  KKT check: converged={converged}, spread={spread:.2e}")
    print(f"  Phase 4 done")

    print("\n=== Phase 5: Multi-Length Exploration ===")
    base = best_seq if best_seq is not None else best_init
    for target_n in [1024, 640, 896, 1152]:
        rem = deadline - time.monotonic() - 120
        if rem < 100:
            break
        f_interp = interpolate_sequence(base, target_n)
        try_seq(f_interp, f"interp_n{target_n}")
        t_gd = min(rem * 0.25, 80)
        if t_gd > 30:
            a_opt, _ = softmax_gd_optimize(f_interp, time_limit=t_gd)
            try_seq(a_opt, f"gd_n{target_n}")
        rem2 = deadline - time.monotonic() - 80
        if rem2 > 50:
            t_lpfw = min(rem2 * 0.5, 100)
            a_opt, _ = lp_fw_optimize(a_opt, time_limit=t_lpfw)
            try_seq(a_opt, f"lpfw_n{target_n}")
    print("  Phase 5 done")

    print("\n=== Phase 6: Deep LP-FW with variants ===")
    if best_seq is not None:
        n_best = len(best_seq)
        variants = [best_seq.copy()]
        noise1 = rng.standard_normal(n_best) * 0.001
        variants.append(project_simplex(best_seq + noise1))
        rc = make_raised_cosine(n_best)
        variants.append(0.85 * best_seq + 0.15 * rc)
        variants[-1] = variants[-1] / variants[-1].sum()
        for vi, f_init in enumerate(variants):
            rem = deadline - time.monotonic() - 80
            if rem < 40:
                break
            t_alloc = min(rem * 0.4, 100)
            print(f"  Variant {vi}: t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(f_init, time_limit=t_alloc, verbose=(vi == 0))
            try_seq(a_opt, f"deep_lpfw_{vi}")
    print("  Phase 6 done")

    print("\n=== Phase 7: Final Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 30:
            t_alloc = min(rem * 0.5, 60)
            a_opt, _ = coordinate_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "coord_final")
        rem = deadline - time.monotonic() - 10
        if rem > 20:
            t_alloc = min(rem, 40)
            a_opt, _ = lp_fw_optimize(a_opt, time_limit=t_alloc)
            try_seq(a_opt, "final_lpfw")
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
