import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

    """Construct the matrix A where (A*b)_k = (f*b)_k for linear convolution."""
    n = len(f)
    k = np.arange(2 * n - 1)[:, None]
    i = np.arange(n)[None, :]
    idx = k - i
    mask = (idx >= 0) & (idx < n)
    A = np.where(mask, f[np.clip(idx, 0, n-1)], 0.0)
    return A

def lp_direction(f, M):

    n = len(f)
    A = construct_convolution_matrix(f)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), 
                  bounds=[(0, None)]*n, method='highs',
                  options={'presolve': True})
    if res.success:
        return res.x
    return np.ones(n) / n

def lp_optimize(f0, time_limit, step_init=0.5, verbose=False, label=""):

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(len(f0)) / len(f0)
        s = 1.0
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    step = step_init
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        if b_sum < 1e-10:

        g = b / b_sum

        best_trial = None
        best_trial_obj = M
        for t in [step, step*0.5, step*0.25, step*0.1, 0.01]:
            f_trial = (1 - t) * f + t * g
            f_trial = np.maximum(f_trial, 0.0)

                best_f = f.copy()

                if verbose and it % 100 == 0:
                    print(f"  {label} LP iter {it}: obj={best_obj:.10f}")
            else:

        else:

            # Try smaller step
            for t in [0.005, 0.001, 0.0001]:
                f_trial = (1 - t) * f + t * g
                f_trial = np.maximum(f_trial, 0.0)
                s = f_trial.sum()
                if s < 1e-10:

                f_trial = f_trial / s
                obj = fast_convolve(f_trial).max()
                if obj < best_obj - 1e-15:
                    f = f_trial
                    best_obj = obj
                    best_f = f.copy()
                    no_improve = 0

                f = np.maximum(f, 0.0)

                f = f_sym.copy()

    A = construct_A(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return np.ones(n) / n


def lp_optimize(f0, time_limit, step_init=0.5, verbose=False, label=""):
    """LP-based optimization with backtracking line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    step = step_init
    lp_time_total = 0.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Solve LP
        lp_start = time.monotonic()
        b = lp_direction(f, M)
        lp_time_total += time.monotonic() - lp_start

        b_sum = b.sum()
        if b_sum < 1e-10:
            b = np.ones(n) / n
            b_sum = 1.0
        g = b / b_sum

        # Line search
        improved = False
        candidates = [step, step * 0.5, step * 0.25, step * 0.1, 0.01, 0.001]
        for t in candidates:
            if t <= 0:
                continue
            f_trial = (1.0 - t) * f + t * g
            f_trial = np.maximum(f_trial, 0.0)
            ss = f_trial.sum()
            if ss < 1e-10:
                continue
            f_trial = f_trial / ss
            obj = fast_convolve(f_trial).max()
            if obj < M - 1e-15:
                f = f_trial
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_f = f.copy()
                no_improve = 0
                improved = True
                if verbose and it % 50 == 0:
                    print(f"    {label} LP iter {it}: obj={best_obj:.10f}, step={t:.4f}")
                break

        if not improved:
            no_improve += 1
            if no_improve > 30:
                # Perturb and restart
                rng = np.random.default_rng(it * 7919 + 13)
                noise_scale = min(0.001 + 0.003 * (no_improve / 200.0), 0.02)
                f = best_f + rng.standard_normal(n) * noise_scale
                f = np.maximum(f, 0.0)
                f = f / f.sum()
                no_improve = 0
                step = step_init * 0.3

        # Adapt step
        if no_improve > 10 and no_improve % 10 == 0:
            step *= 0.8
        elif no_improve == 0 and step < step_init:
            step = min(step * 1.2, step_init)

        # Symmetrize periodically
        if it % 100 == 0:
            f_sym = 0.5 * (f + f[::-1])
            f_sym = np.maximum(f_sym, 0.0)
            f_sym = f_sym / f_sym.sum()
            sym_obj = fast_convolve(f_sym).max()
            if sym_obj < best_obj:
                best_obj = sym_obj
                best_f = f_sym.copy()
                f = f_sym.copy()

        # If LP is too slow, stop
        if it > 10 and lp_time_total / it > 2.0:
            if verbose:
                print(f"    {label} LP too slow ({lp_time_total/it:.2f}s/iter), stopping")
            break

    return best_f, best_obj


def spectral_flatten(f0, time_limit, alpha_init=0.1, verbose=False, label=""):
    """Frequency-domain spectral flattening: iteratively compress |FFT(f)| to flatten autocorrelation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    alpha = alpha_init
    N = 1 << (2 * n - 1).bit_length()

    while time.monotonic() - t0 < time_limit:
        it += 1
        F = np.fft.fft(f, N)
        absF = np.abs(F)
        absF_sq = absF ** 2
        mean_pow = absF_sq.mean()

        # Compress the magnitude spectrum
        # Variant: F_new = F * (mean_pow / (absF_sq + eps))^0.5 (water-filling-like)
        eps = 1e-10
        compression = np.sqrt(mean_pow / (absF_sq + eps)**0.5)
        compression = np.clip(compression, 0.5, 2.0)
        F_new = F * (1.0 - alpha) + alpha * F * compression
        # Alternative: simply scale down peaks
        F_new = F * np.sqrt(mean_pow / (absF_sq + eps))

        f_trial = np.real(np.fft.ifft(F_new, N)[:n])
        f_trial = np.maximum(f_trial, 0.0)
        ss = f_trial.sum()
        if ss < 1e-10:
            f_trial = f.copy()
        else:
            f_trial = f_trial / ss

        obj = fast_convolve(f_trial).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_trial.copy()
            f = f_trial
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    {label} SF iter {it}: obj={best_obj:.10f}, alpha={alpha:.4f}")
        else:
            no_improve += 1
            # Blend with previous
            f_blend = 0.5 * f + 0.5 * f_trial
            f_blend = np.maximum(f_blend, 0.0)
            f_blend = f_blend / f_blend.sum()
            blend_obj = fast_convolve(f_blend).max()
            if blend_obj < best_obj - 1e-14:
                best_obj = blend_obj
                best_f = f_blend.copy()
                f = f_blend
                no_improve = 0

        # Adapt alpha
        if no_improve > 20:
            alpha *= 0.8
            if alpha < 0.001:
                # Try different strategy
                alpha = 0.3
                rng = np.random.default_rng(it * 31 + 7)
                f = best_f + rng.standard_normal(n) * 0.001
                f = np.maximum(f, 0.0)
                f = f / f.sum()
                no_improve = 0
        elif no_improve == 0:
            alpha = min(alpha * 1.05, 0.5)

        if it % 200 == 0:
            f_sym = 0.5 * (f + f[::-1])
            f_sym = np.maximum(f_sym, 0.0)
            f_sym = f_sym / f_sym.sum()
            sym_obj = fast_convolve(f_sym).max()
            if sym_obj < best_obj:
                best_obj = sym_obj
                best_f = f_sym.copy()
                f = f_sym.copy()

    return best_f, best_obj


def subgradient_optimize(f0, time_limit, top_k=20, temp=0.01, lr_init=0.01,
                         verbose=False, label=""):
    """Subgradient method with softmax-weighted peaks and symmetrization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    lr = lr_init

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Get top peaks with softmax weights
        K = min(top_k, len(conv))
        if K < len(conv):
            peak_idx = np.argpartition(conv, -K)[-K:]
        else:
            peak_idx = np.arange(len(conv))

        pv = conv[peak_idx]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(K)
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute subgradient: grad_i = 2 * sum_k w_k * f_{k-i}
        grad = np.zeros(n)
        for idx in range(K):
            k = int(peak_idx[idx])
            w_k = w[idx]
            if w_k < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue
            grad[js:je + 1] += 2.0 * w_k * f[k - np.arange(js, je + 1)]

        # Center the gradient (project to tangent space of simplex)
        grad_c = grad - grad.mean()

        # Backtracking line search
        improved = False
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            if trial_lr < 1e-10:
                continue
            f_trial = f - trial_lr * grad_c
            f_trial = np.maximum(f_trial, 1e-15)
            f_trial = f_trial / f_trial.sum()
            obj = fast_convolve(f_trial).max()
            if obj < M - 1e-15:
                f = f_trial
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_f = f.copy()
                no_improve = 0
                improved = True
                if verbose and it % 200 == 0:
                    print(f"    {label} SG iter {it}: obj={best_obj:.10f}")
                break

        if not improved:
            no_improve += 1
            if no_improve > 100:
                rng = np.random.default_rng(it * 104729 + 3)
                f = best_f + rng.standard_normal(n) * 0.002
                f = np.maximum(f, 1e-15)
                f = f / f.sum()
                lr = lr_init * 0.3
                no_improve = 0

        # Adapt
        if no_improve > 50:
            lr *= 0.5
        lr = max(lr * 0.9999, 1e-8)
        temp = min(temp * 1.002, 0.1)

        if it % 500 == 0:
            f_sym = 0.5 * (f + f[::-1])
            f_sym = np.maximum(f_sym, 1e-15)
            f_sym = f_sym / f_sym.sum()
            sym_obj = fast_convolve(f_sym).max()
            if sym_obj < best_obj:
                best_obj = sym_obj
                best_f = f_sym.copy()
                f = f_sym.copy()

        if lr < 1e-9:
            break

    return best_f, best_obj


def fine_tune(f0, time_limit, verbose=False, label=""):
    """Fine coordinate descent: try small multiplicative adjustments per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    rounds = 0

    while time.monotonic() - t0 < time_limit:
        rounds += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in [0.8, 0.9, 0.95, 1.05, 1.1, 1.25]:
                new_val = old_val * mult
                if new_val < 0:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                trial = f.copy()
                trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                trial[mask] *= scale
                trial = np.maximum(trial, 0.0)
                ss = trial.sum()
                if ss < 1e-15:
                    continue
                trial = trial / ss
                val = fast_convolve(trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0.0)
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def symmetrize(a):
    """Symmetrize a sequence."""
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    ss = a_sym.sum()
    if ss > 1e-15:
        a_sym = a_sym / ss
    return a_sym


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian-like initial sequence normalized to sum=1."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a


def make_bell(n, width_frac=0.6):
    """Bell-shaped with flat top."""
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a


def make_inverse_center(n):
    """High at edges, low in center - might give flatter convolution."""
    x = np.arange(n) - n / 2.0
    a = 1.0 + 0.5 * np.cos(2 * np.pi * x / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, maintaining sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization: LP-based direction finding, spectral flattening,
    subgradient method, and coordinate descent across multiple sequence lengths
    with diverse initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, lbl=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{lbl}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    n_base = 768
    if parent is not None:
        n_base = len(parent)

    # === Phase 1: Quick initialization evaluation ===
    print("=== Phase 1: Initialization ===")
    t0 = time.monotonic()

    init_list = []
    if parent is not None:
        init_list.append((parent.copy(), "parent", n_base))
        init_list.append((symmetrize(parent), "parent_sym", n_base))

    for n_try in [n_base, 1024, 1536]:
        init_list.append((make_gaussian(n_try, 0.25), f"gauss_{n_try}", n_try))
        init_list.append((make_gaussian(n_try, 0.35), f"gaussw_{n_try}", n_try))
        init_list.append((make_bell(n_try, 0.5), f"bell_{n_try}", n_try))
        init_list.append((make_inverse_center(n_try), f"invc_{n_try}", n_try))

    best_init = None
    best_init_obj = float('inf')
    for a_init, lbl, n_init in init_list:
        s = a_init.sum()
        if s < 1e-10:
            continue
        a_norm = a_init / s
        try_seq(a_norm, f"init_{lbl}")
        c = fast_convolve(a_norm)
        obj = 2 * len(a_norm) * c.max()
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a_norm.copy()

    if best_init is None:
        best_init = make_gaussian(n_base, 0.3)

    print(f"  Best init: obj={best_init_obj:.6f}, n={len(best_init)}")
    t1 = time.monotonic()
    print(f"  Phase 1 time: {t1-t0:.1f}s")

    # === Phase 2: LP-based optimization (main) ===
    print("\n=== Phase 2: LP-based Optimization ===")
    remaining = deadline - time.monotonic()
    t_lp = remaining * 0.45
    print(f"  Time allocated: {t_lp:.0f}s")

    # Run LP on best init
    a_opt, obj = lp_optimize(best_init, time_limit=t_lp * 0.6, step_init=0.5,
                             verbose=True, label="LP_main")
    try_seq(a_opt, "LP_main")

    # Run LP on parent symmetrized
    if parent is not None and time.monotonic() < deadline - 60:
        remaining = deadline - time.monotonic()
        t_lp2 = min(remaining * 0.4, 300)
        a_sym_p = symmetrize(parent)
        a_opt, obj = lp_optimize(a_sym_p, time_limit=t_lp2, step_init=0.4,
                                 verbose=True, label="LP_par")
        try_seq(a_opt, "LP_par")

    # === Phase 3: Spectral flattening ===
    print("\n=== Phase 3: Spectral Flattening ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_sf = remaining * 0.20
        print(f"  Time allocated: {t_sf:.0f}s")
        a_opt, obj = spectral_flatten(best_seq, time_limit=t_sf, alpha_init=0.15,
                                      verbose=True, label="SF_main")
        try_seq(a_opt, "SF_main")

        # Try on symmetrized version
        if time.monotonic() < deadline - 30:
            a_s = symmetrize(best_seq)
            a_opt, obj = spectral_flatten(a_s, time_limit=min(remaining * 0.1, 60),
                                          alpha_init=0.1, verbose=True, label="SF_sym")
            try_seq(a_opt, "SF_sym")

    # === Phase 4: LP again on current best ===
    print("\n=== Phase 4: LP Refinement ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_lp3 = remaining * 0.15
        print(f"  Time allocated: {t_lp3:.0f}s")
        a_opt, obj = lp_optimize(best_seq, time_limit=t_lp3, step_init=0.2,
                                 verbose=True, label="LP_refine")
        try_seq(a_opt, "LP_refine")

    # === Phase 5: Subgradient fine-tuning ===
    print("\n=== Phase 5: Subgradient Fine-tuning ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_sg = remaining * 0.10
        print(f"  Time allocated: {t_sg:.0f}s")
        a_opt, obj = subgradient_optimize(best_seq, time_limit=t_sg, top_k=15,
                                          temp=0.005, lr_init=0.005,
                                          verbose=True, label="SG_fine")
        try_seq(a_opt, "SG_fine")

    # === Phase 6: Final LP + coordinate descent ===
    print("\n=== Phase 6: Final Polish ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_final = remaining * 0.20
        # Final LP
        t_lp_f = t_final * 0.5
        a_opt, obj = lp_optimize(best_seq, time_limit=t_lp_f, step_init=0.1,
                                 verbose=True, label="LP_final")
        try_seq(a_opt, "LP_final")
        # Coordinate descent
        if time.monotonic() < deadline - 10:
            t_cd = min(deadline - time.monotonic() - 10, t_final * 0.5)
            if t_cd > 5:
                a_opt, obj = fine_tune(best_seq, time_limit=t_cd,
                                       verbose=True, label="CD_final")
                try_seq(a_opt, "CD_final")

        # Final symmetrization check
        a_s = symmetrize(best_seq)
        sc = try_seq(a_s, "final_sym")
        # Also try without symmetrization
        sc2 = try_seq(best_seq, "final_asym")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
