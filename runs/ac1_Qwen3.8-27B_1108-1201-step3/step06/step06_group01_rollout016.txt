def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_fast(a):
    """Compute the score without calling evaluate_sequence."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    max_b = fast_convolve(a_norm).max()
    return float(2 * len(a_norm) * max_b)


def simplex_project(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    # = a_m * (grad_a_m - <grad_a, a>)
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    
    return obj, grad_z


def lbgfs_optimize(z0, T, n, time_budget, maxiter=5000):
    """L-BFGS optimization in log-space with soft-max smoothing."""
    t0 = time.monotonic()
    stopped = [False]
    
    def func(z):
        if time.monotonic() - t0 > time_budget:
            stopped[0] = True
            return 1e15
        obj, _ = soft_max_objective(z, T, n)
        return obj
    
    def jac(z):
        if time.monotonic() - t0 > time_budget:
            stopped[0] = True
            return np.zeros_like(z)
        _, g = soft_max_objective(z, T, n)
        return g
    
    try:
        res = minimize(
            fun=func, x0=z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 20, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
    except Exception:
        a_out = softmax_from_z(z0)
    
    return a_out


def peak_shave(a0, time_budget, n_trials_per_peak=10):
    """
    Peak-targeted local search: identify the peak lag, then try to reduce it
    by moving mass between paired elements.
    """
    n = len(a0)
    a = np.maximum(a0, 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = evaluate_fast(a)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_budget:
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        
        # Get the pairs that contribute to the peak
        j_start = max(0, peak_k - n + 1)
        j_end = min(n - 1, peak_k)
        js = np.arange(j_start, j_end + 1)
        paired = peak_k - js
        contribs = a[js] * a[paired]
        
        # Try reducing the largest contributors
        order = np.argsort(contribs)[::-1]
        improved = False
        
        for idx in order[:n_trials_per_peak]:
            i = js[idx]
            j = paired[idx]
            if a[i] < 1e-12 or a[j] < 1e-12:
                continue
            
            # Try moving mass from the larger element to the smaller
            # (to reduce the product while keeping sum constant)
            if a[i] > a[j]:
                src, dst = i, j
            else:
                src, dst = j, i
            
            for frac in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3]:
                delta = a[src] * frac
                if a[dst] + delta > 1.0:
                    continue
                a_trial = a.copy()
                a_trial[src] -= delta
                a_trial[dst] += delta
                score = evaluate_fast(a_trial)
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try moving mass to a far element (not paired with peak)
            for _ in range(20):
                i = np.random.randint(n)
                j = np.random.randint(n)
                if i == j or a[i] < 1e-12:
                    continue
                delta = a[i] * 0.05
                a_trial = a.copy()
                a_trial[i] -= delta
                a_trial[j] += delta
                score = evaluate_fast(a_trial)
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            break
    
    return best_a, best_score


def subgradient_descent(a0, time_budget, lr_init=0.01, momentum=0.9, patience=200):
    """Subgradient method with momentum for the non-smooth max auto-correlation."""
    n = len(a0)
    a = np.maximum(a0, 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = evaluate_fast(a)
    v = np.zeros(n)
    lr = lr_init
    no_improve = 0
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    step = 0
    while time.monotonic() - t0 < time_budget:
        b = fast_convolve(a)
        # Find top-k peaks
        k = min(20, 2 * n - 1)
        peaks = np.argpartition(b, -k)[-k:]
        pv = b[peaks]
        # Soft weighting
        temp = 0.01 * max(pv.max(), 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        w /= w.sum()
        
        # Compute subgradient
        # d b_k / d a_j = 2 * a[k-j] (for valid indices)
        sg = np.zeros(n)
        for ii in range(len(peaks)):
            if w[ii] < 1e-12:
                continue
            kk = int(peaks[ii])
            # sg[j] += 2 * w[ii] * a[kk - j] for valid j
            js = np.arange(max(0, kk - n + 1), min(n - 1, kk) + 1)
            # a[kk - j] for j in js
            paired = kk - js
            sg[js] += 2.0 * w[ii] * a[paired]
        
        v = momentum * v - lr * sg
        a_new = simplex_project(a + v)
        a_new = a_new / a_new.sum()
        
        score = evaluate_fast(a_new)
        step += 1
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            a = a  # keep current
        
        lr *= 0.999
        
        if no_improve > patience:
            # Restart with perturbation
            a = best_a + rng.standard_normal(n) * 0.01
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            lr = max(lr * 5, 0.005)
            no_improve = 0
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def generate_initializations(n, rng):
    """Generate diverse initial conditions."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Power-law ramps (various exponents)
    t = np.arange(n, dtype=np.float64)
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = (t + 1) ** p
        inits.append(a / a.sum())
        a2 = (n - t) ** p
        inits.append(a2 / a2.sum())
    
    # Exponential ramps
    for alpha in [0.5, 1.0, 2.0]:
        a = np.exp(alpha * t / n)
        inits.append(a / a.sum())
        a2 = np.exp(-alpha * t / n)
        inits.append(a2 / a2.sum())
    
    # Periodic
    for freq in [1, 2, 3, 5]:
        a = 1 + 0.5 * np.cos(2 * np.pi * freq * t / n)
        a = np.maximum(a, 0)
        inits.append(a / a.sum())
    
    # Random
    a = rng.random(n)
    inits.append(a / a.sum())
    
    # Piecewise: half uniform
    a = np.zeros(n)
    a[n//2:] = 1.0
    inits.append(a / a.sum())
    
    # V-shape
    center = n / 2
    a = np.abs(t - center) + 0.1
    inits.append(a / a.sum())
    
    # Inverse V (peak at center)
    a = n - np.abs(t - center) + 0.1
    inits.append(a / a.sum())
    
    return inits
