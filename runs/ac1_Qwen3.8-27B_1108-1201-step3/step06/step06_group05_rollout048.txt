def fast_convolve(a):

    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_topk_gradient(f, c, k):
    """Gradient focusing on top-k peaks."""
    n = len(f)
    peak_idx = np.argpartition(c, -k)[-k:]
    w = np.zeros(len(c))
    w[peak_idx] = 1.0 / k
    
    w_rev = w[::-1].copy()
    conv_len = n + len(w_rev) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv = np.fft.irfft(F * W, fft_size)[:conv_len]
    indices = np.arange(2 * n - 2, 2 * n - 2 - n, -1)
    g = 2.0 * conv[indices]
    return g

def compute_weighted_gradient(f, c, k, power):
    """Gradient with power-weighted top-k peaks."""
    n = len(f)
    c_max = c.max()
    c_min = c.min()
    if c_max - c_min < 1e-15:
        w = np.ones(len(c)) / len(c)
    else:
        # Weight by normalized peak height raised to power
        c_norm = (c - c_min) / (c_max - c_min)
        w = c_norm ** power
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.ones(len(c)) / len(c)
        else:
            w = w / w_sum
        # Focus on top k

        w_masked = np.zeros(len(c))

        w_sum2 = w_masked.sum()
        if w_sum2 < 1e-15:
            w = np.ones(len(c)) / len(c)
        else:

def ripple_lp(f, M, K=None):
    """
    Solve the ripple LP:
    min t
    s.t. (f*f)_k + 2*(f*r)_k <= t for all k (or top-K k)
         f[j] + r[j] >= 0 for all j
         sum(r) = 0
    
    Returns the optimal ripple r.
    """
    n = len(f)
    c = fast_convolve(f)
    
    # Select active peaks
    if K is None or K >= len(c):
        peak_idx = np.arange(len(c))
    else:
        peak_idx = np.argpartition(c, -K)[-K:]
    
    c_active = c[peak_idx]
    m = len(peak_idx)
    
    # The constraint (f*r)_k = sum_j f[j] * r[k-j]
    # Build the matrix B where B[i, j] = f[j] if j is in [max(0, k_i - n + 1), min(n-1, k_i)]
    # i.e., B[i, j] = f[k_i - j] if 0 <= k_i - j < n, else 0
    
    # For each peak k_i, (f*r)_{k_i} = sum_j f[k_i - j] * r[j] for valid j
    # = sum_j A[i,j] * r[j] where A[i,j] = f[k_i-j] if 0 <= k_i-j < n
    
    # Build A (m x n)
    A = np.zeros((m, n))
    for i, k in enumerate(peak_idx):
        # r[j] contributes to (f*r)_k when f[k-j] is defined, i.e., 0 <= k-j < n
        # j ranges from max(0, k-n+1) to min(n-1, k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A[i, j] = f[k - j]
    
    n_vars = n + 1  # r[0..n-1] and t
    # Objective: minimize t -> c_obj = [0, 0, ..., 0, 1]
    c_obj = np.zeros(n_vars)
    c_obj[-1] = 1.0
    
    # Inequality constraints:
    # A @ r - t <= -c_active  (i.e., t >= c_active + A @ r)
    # Rewritten: A @ r - t <= -c_active... no wait.
    # We want: c_active[i] + (A @ r)[i] <= t for all i
    # i.e., (A @ r)[i] - t <= -c_active[i]
    
    # r[r] >= -f[j], i.e., -r[j] <= f[j]
    
    # Equality: sum(r) = 0
    
    # Let x = [r_0, ..., r_{n-1}, t]
    # Inequalities:
    # For i = 0..m-1: A[i,:] @ r - t <= -c_active[i]
    #   -> A[i,:] @ r + [0,...,0,-1] @ t <= -c_active[i]
    # For j = 0..n-1: -r[j] <= f[j]
    #   -> -e_j^T @ x <= f[j]
    
    A_ub = np.zeros((m + n, n_vars))
    b_ub = np.zeros(m + n)
    
    for i in range(m):
        A_ub[i, :n] = A[i, :]
        A_ub[i, n] = -1.0
        b_ub[i] = -c_active[i]
    
    for j in range(n):
        A_ub[m + j, j] = -1.0
        b_ub[m + j] = f[j]
    
    # Equality: sum(r) = 0
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: r[j] >= -f[j] (already in inequalities), t >= 0
    bounds = [(None, None)] * n + [(0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

def project_simplex(v):
    n = len(v)
    if n == 1:

    cssv = np.cumsum(u)
