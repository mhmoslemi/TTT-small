```
def spectral_shaping(a, n_iters, alpha):
    N = next_power_of_2(2*len(a)-1)
    for _ in range(n_iters):
        A = rfft(a, N)
        P = A.real**2 + A.imag**2  # |A|^2
        # Flatten: P_new = P^alpha / sum(P^alpha) * sum(P)
        # This reduces dynamic range
        P_new = P**alpha
        P_new = P_new / P_new.sum() * P.sum()
        # Reconstruct with new magnitude, same phase
        A_new = A / np.sqrt(P + 1e-30) * np.sqrt(P_new)
        a_new = irfft(A_new, N)[:len(a)]
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        a = a_new
    return a
```

```python
import numpy as np

from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT, zero-padded to next power of 2."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def eval_local(a):
    """Compute 2n*max(conv)/(sum)^2 locally."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S * S))


def subgrad_at_peak(a, k):
    """Compute subgradient of (a*a)[k] with respect to a."""
    n = len(a)
    g = np.zeros(n)
    # (a*a)[k] = sum_i a[i]*a[k-i]
    # d/d a[j] = a[k-j] + a[j-k] (with bounds)
    for j in range(n):
        val = 0.0
        ik = k - j
        if 0 <= ik < n:
            val += a[ik]
        jk = j - k
        if 0 <= jk < n:
            val += a[jk]
        g[j] = val
    return g


def subgrad_fft(a, k):
    """Fast subgradient computation using FFT for a single lag k."""
    n = len(a)
    # (a*a)[k] = sum_i a[i]*a[k-i]
    # d/d a[j] (a*a)[k] = a[k-j]*I[0<=k-j<n] + a[j-k]*I[0<=j-k<n]
    g = np.zeros(n)
    # First term: a[k-j] for j = max(0,k-n+1) ... min(k, n-1)
    lo1 = max(0, k - n + 1)
    hi1 = min(n - 1, k)
    if lo1 <= hi1:
        indices = np.arange(lo1, hi1 + 1)
        g[indices] += a[k - indices]
    # Second term: a[j-k] for j = max(k, 0) ... min(n-1, k+n-1)
    lo2 = max(0, k)
    hi2 = min(n - 1, k + n - 1)
    if lo2 <= hi2:
        indices = np.arange(lo2, hi2 + 1)
        g[indices] += a[indices - k]
    return g


def subgrad_weighted(a, b, weights):
    """Weighted combination of subgradients over active peaks."""
    n = len(a)
    g = np.zeros(n)
    for k in range(len(b)):
        if weights[k] < 1e-12:
            continue
        g += weights[k] * subgrad_fft(a, k)
    return g


def spectral_shaping(a, n_iters=20, alpha=0.7, floor_frac=0.05):
    """
    Iteratively flatten the power spectrum to reduce auto-convolution peak.
    Works in frequency domain: compress dynamic range of |FFT(a)|^2.
    """
    n = len(a)
    fft_size = 1 << (2 * n - 2).bit_length()
    a_work = a.copy()
    
    for it in range(n_iters):
        A = rfft(a_work, fft_size)
        P = np.abs(A) ** 2
        P_sum = P.sum()
        if P_sum < 1e-30:
            break
        
        # Add a floor to flatten: P_new = (P + floor) / (1 + floor/P_avg)
        # This compresses the dynamic range
        P_floor = floor_frac * P_sum / len(P)
        P_new = P + P_floor
        # Renormalize to preserve total energy
        P_new = P_new * P_sum / P_new.sum()
        
        # Apply alpha compression
        P_new = P_new ** alpha
        P_new = P_new * P_sum / P_new.sum()
        
        # Reconstruct: keep phase, change magnitude
        phase = A / (np.abs(A) + 1e-30)
        A_new = phase * np.sqrt(P_new)
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_work = a_new / s
    
    return a_work


def subgradient_optimize(a0, time_limit, init_lr=None, n_active=None,
                         use_polyak=True, lower_bound=None, seed=42):
    """
    Subgradient method for minimizing max(conv(a,a)) on the simplex.
    Uses gap-aware weighting and optional Polyak step size.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    
    # Initial evaluation
    b = fast_convolve(a)
    cur_obj = b.max()
    best_obj = cur_obj
    best_a = a.copy()
    
    if lower_bound is None:
        lower_bound = 1.0 / (2 * n - 1)  # theoretical minimum of max_b when sum=1
    
    if init_lr is None:
        init_lr = 1.0 / (n * 0.1)
    
    lr = init_lr
    no_improve = 0
    momentum_vec = np.zeros(n)
    momentum_alpha = 0.85
    
    # Gap-aware weighting parameters
    gap_scale = 5.0  # how sharply to weight near-max peaks
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        b_max = b.max()
        cur_obj = b_max
        
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Compute gap-aware weights
        gaps = b_max - b
        # Weight: exp(-gap / gap_scale) -- peaks close to max get high weight
        max_gap = gaps.max()
        if max_gap < 1e-15:
            # All peaks equal weight
            w = np.ones(len(b)) / len(b)
        else:
            w = np.exp(-gaps / (gap_scale * max_gap * 0.01 + 1e-15))
            w = np.exp(-gaps / max(0.01 * max_gap, 1e-10))
            w_sum = w.sum()
            if w_sum < 1e-30:
                w = np.zeros(len(b))
                w[b.argmax()] = 1.0
            else:
                w /= w_sum
        
        # Restrict to top peaks for efficiency
        if n_active is not None and n_active < len(b):
            top_idx = np.argpartition(b, -n_active)[-n_active:]
            w_sel = w[top_idx]
            w_sel /= w_sel.sum()
            w_tmp = np.zeros(len(w))
            w_tmp[top_idx] = w_sel
            w = w_tmp
        
        # Compute subgradient
        g = subgrad_weighted(a, b, w)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-30:
            g = rng.standard_normal(n)
            g_norm = np.linalg.norm(g)
        g = g / g_norm  # normalize
        
        # Polyak step size
        if use_polyak:
            obj_scaled = 2.0 * n * cur_obj  # objective with sum=1
            lb_scaled = 2.0 * n * lower_bound
            gap = obj_scaled - lb_scaled
            if gap < 1e-10:
                lr = 1e-6
            else:
                lr = min(gap / (2.0 * n * g_norm ** 2) * 0.5, init_lr * 5.0)
                lr = max(lr, 1e-8)
        
        # Nesterov-like momentum
        momentum_vec = momentum_alpha * momentum_vec - lr * g
        a_new = project_simplex(a + momentum_vec)
        
        # Check improvement
        b_new = fast_convolve(a_new)
        obj_new = b_new.max()
        
        if obj_new < cur_obj:
            a = a_new
            # Slightly increase lr if making progress
            lr *= 1.01
        else:
            a = a_new  # still move (to explore)
            lr *= 0.95
        
        # Adaptive gap_scale: decrease over time (sharper focus)
        gap_scale *= 0.999
        if gap_scale < 0.5:
            gap_scale = 0.5
        
        # Periodic restart from best
        if no_improve > 500:
            noise = rng.standard_normal(n) * 0.005
            a = project_simplex(best_a + noise)
            momentum_vec = np.zeros(n)
            lr = init_lr
            no_improve = 0
        elif no_improve > 200:
            lr *= 0.8
    
    return best_a, best_obj


def spectral_equalize(a, n_passes=10, strength=0.3):
    """
    Gentle spectral equalization: reduce peaks in the power spectrum
    without destroying the time-domain structure too much.
    """
    n = len(a)
    fft_size = 1 << (2 * n - 2).bit_length()
    a_work = a.copy()
    
    for _ in range(n_passes):
        A = rfft(a_work, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()
        
        # Damp high-power bins, boost low-power bins (multiplicative)
        ratio = P / (P_mean + 1e-30)
        # Compress: P_new = P * (1/strength * (1 - strength) + strength/ratio^strength)
        # Simpler: P_new = P * ratio^(-strength) = P^(1-strength) * P_mean^strength
        P_new = P * (ratio ** (-strength))
        P_new = P_new * P.sum() / P_new.sum()  # preserve total energy
        
        # Reconstruct
        phase = A / (np.abs(A) + 1e-30)
        A_new = phase * np.sqrt(P_new)
        a_new = irfft(A_new, fft_size)[:n]
        
        # Blend with original to avoid too much disruption
        a_work = (1 - 0.5) * a_work + 0.5 * np.maximum(a_new, 0)
        a_work = np.maximum(a_work, 0)
        a_work = a_work / a_work.sum()
    
    return a_work


def coordinate_polish(a0, time_limit, n_passes=100):
    """
    Coordinate descent: for each element, find the best scaling factor
    that minimizes the max auto-convolution, keeping sum = 1.
    """
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    
    # Focus on coordinates that are part of the auto-convolution peak
    b = fast_convolve(a)
    peak_lags = np.where(b > 0.95 * b.max())[0]
    
    multipliers = np.array([0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0])
    
    for _ in range(n_passes):
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-14:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < -1e-10 or new_val > 0.99:
                    continue
                if new_val < 0:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0 or scale > 10:
                    continue
                
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-10:
                    continue
                a_trial /= ss
                
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = np.maximum(a, 0)
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_a, best_obj


def lbgfs_refine(a0, T, time_limit, maxiter=2000):
    """
    L-BFGS refinement using soft-max smooth approximation of the max.
    Works in log-space to handle non-negativity.
    """
    t0 = time.monotonic()
    n = len(a0)
    
    def z_to_a(z):
        z_shifted = z - z.max()
        e = np.exp(z_shifted)
        return e / e.sum()
    
    def obj_grad(z):
        a = z_to_a(z)
        b = fast_convolve(a)
        Tb = T * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        log_sum_exp = max_Tb + np.log(exp_Tb.sum())
        obj_val = 2.0 * n * log_sum_exp / T
        
        w = exp_Tb / exp_Tb.sum()
        # Gradient w.r.t. a
        grad_a = np.zeros(n)
        for k in range(len(w)):
            if w[k] < 1e-12:
                continue
            grad_a += w[k] * subgrad_fft(a, k)
        
        # Gradient w.r.t. z: grad_z = a * (grad_a - a * dot(grad_a, a))
        da_dz_diag = a
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S * a)
        
        return obj_val, grad_z
    
    z0 = np.log(np.maximum(a0, 1e-300))
    best_obj = obj_grad(z0)[0]
    best_z = z0.copy()
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return obj_grad(z)[0]
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return obj_grad(z)[1]
    
    try:
        res = minimize(
            fun=func, x0=z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-8,
                     'maxfun': maxiter * 10, 'disp': False}
        )
        if np.isfinite(res.fun) and res.fun < best_obj:
            full_obj = eval_local(z_to_a(res.x))
            cur_full = eval_local(a0)
            if full_obj < cur_full:
                best_z = res.x.copy()
                best_obj = full_obj
    except Exception:
        pass
    
    return z_to_a(best_z), best_obj


def interpolate_to_length(f, new_n):
    """Interpolate sequence to new length, preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_structural_init(n, kind="flat"):
    """Generate structured initializations."""
    x = np.linspace(0, 1, n)
    if kind == "flat":
        a = np.ones(n)
    elif kind == "gauss_03":
        a = np.exp(-0.5 * ((x - 0.5) / 0.3) ** 2)
    elif kind == "gauss_05":
        a = np.exp(-0.5 * ((x - 0.5) / 0.5) ** 2)
    elif kind == "cosine":
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif kind == "tukey":
        a = np.ones(n)
        taper = n // 4
        if taper > 0:
            t = np.linspace(0, 1, taper)
            a[:taper] = 0.5 * (1 + np.cos(np.pi * (2 * t - 1)))
            a[-taper:] = a[:taper][::-1]
    elif kind == "bathtub":
        # Low at edges, high in middle
        a = 1.0 - 0.5 * np.cos(2 * np.pi * x)
    elif kind == "dip":
        # High at edges, low in middle
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

            try_seq(a_sg2, "sg_from_best")
    
    # === Phase 4: L-BFGS refinement ===
    print(f"\n=== Phase 4: L-BFGS (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for T in [30, 80, 150]:
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            t_alloc = min(rem * 0.1, 60.0)
            if t_alloc < 5:
                continue
            print(f"  L-BFGS T={T}, n={n_best}, t={t_alloc:.0f}s")
            a_lb, _ = lbgfs_refine(best_seq, T=T, time_limit=t_alloc, maxiter=2000)
            try_seq(a_lb, f"lbgfs_T{T}")
    
    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length exploration ===")
    if best_seq is not None:
        n_base = len(best_seq)
        target_lengths = []
        for factor in [0.5, 0.75, 1.0, 1.25, 1.5, 2.0]:
            tl = int(n_base * factor)
            tl = max(256, min(4096, tl))
            if tl != n_base:
                target_lengths.append(tl)
        
        # Also try specific good lengths
        for tl in [512, 768, 1024, 1536, 2048]:
            if tl not in target_lengths and tl != n_base:
                target_lengths.append(tl)
        
        target_lengths = sorted(set(target_lengths), key=lambda x: abs(x - n_base))
        
        for target_n in target_lengths[:5]:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            
            f_interp = interpolate_to_length(best_seq, target_n)
            t_alloc = min(rem * 0.12, 80.0)
            if t_alloc < 5:
                continue
            print(f"  n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = subgradient_optimize(
                f_interp, time_limit=t_alloc,
                init_lr=0.008, n_active=max(20, target_n // 15),
                use_polyak=True, seed=seed + target_n
            )
            try_seq(a_opt, f"len_{target_n}")
            
            # Quick spectral shaping on this length
            rem = deadline - time.monotonic()
            if rem > 10:
                a_s = spectral_shaping(best_seq if len(best_seq) == target_n else f_interp,
                                       n_iters=20, alpha=0.7, floor_frac=0.05)
                try_seq(a_s, f"spec_{target_n}")
    
    # === Phase 6: Diverse structural initializations ===
    print(f"\n=== Phase 6: Diverse initializations ===")
    if best_seq is not None:
        n_base = len(best_seq)
    else:
        n_base = n_parent
    
    structures = ["gauss_03", "gauss_05", "cosine", "tukey", "bathtub", "dip"]
    for kind in structures:
        rem = deadline - time.monotonic()
        if rem < 20:
            break
        init_a = make_structural_init(n_base, kind)
        t_alloc = min(rem * 0.08, 40.0)
        if t_alloc < 3:
            continue
        print(f"  Structure: {kind}, t={t_alloc:.0f}s")
        a_opt, _ = subgradient_optimize(
            init_a, time_limit=t_alloc,
            init_lr=0.02, n_active=max(15, n_base // 20),
            use_polyak=True, seed=seed + hash(kind) % 10000
        )
        try_seq(a_opt, f"struct_{kind}")
    
    # === Phase 7: Deep subgradient on best ===
    print(f"\n=== Phase 7: Deep subgradient (score={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        if rem > 30:
            t_alloc = min(rem * 0.2, 200.0)
            print(f"  Deep SG, n={n_best}, t={t_alloc:.0f}s")
            a_deep, _ = subgradient_optimize(
                best_seq, time_limit=t_alloc,
                init_lr=0.002, n_active=max(50, n_best // 8),
                use_polyak=True, seed=seed + 77777
            )
            try_seq(a_deep, "deep_sg")
    
    # === Phase 8: Spectral equalization polish ===
    print(f"\n=== Phase 8: Spectral equalization polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 10:
            for strength in [0.2, 0.4, 0.6]:
                rem = deadline - time.monotonic()
                if rem < 5:
                    break
                a_eq = spectral_equalize(best_seq, n_passes=15, strength=strength)
                try_seq(a_eq, f"eq_s{strength}")
                
                # Follow up with short subgradient
                rem = deadline - time.monotonic()
                if rem > 15 and best_score < float('inf'):
                    t_alloc = min(rem * 0.3, 30.0)
                    a_f, _ = subgradient_optimize(
                        best_seq, time_limit=t_alloc,
                        init_lr=0.003, n_active=max(30, len(best_seq) // 10),
                        use_polyak=True, seed=seed + int(strength * 1000)
                    )
                    try_seq(a_f, f"after_eq_s{strength}")
    
    # === Phase 9: Coordinate descent ===
    print(f"\n=== Phase 9: Coordinate descent (score={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 15:
            t_cd = min(rem * 0.15, 60.0)
            print(f"  CD polish, t={t_cd:.0f}s")
            a_cd, _ = coordinate_polish(best_seq, time_limit=t_cd)
            try_seq(a_cd, "cd_polish")
    
    # === Phase 10: Final perturbation + re-optimize ===
    print(f"\n=== Phase 10: Perturbations (score={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for attempt in range(8):
            rem = deadline - time.monotonic()
            if rem < 8:
                break
            # Vary perturbation size
            noise_scale = 0.003 * (1 + attempt * 0.5)
            noise = rng.standard_normal(n_best) * noise_scale
            a_pert = project_simplex(best_seq + noise)
            
            t_alloc = min(rem * 0.4, 25.0)
            a_p, _ = subgradient_optimize(
                a_pert, time_limit=t_alloc,
                init_lr=0.005, n_active=max(20, n_best // 12),
                use_polyak=True, seed=seed + attempt * 31337
            )
            try_seq(a_p, f"perturb_{attempt}")
    
    # === Phase 11: Final L-BFGS ===
    print(f"\n=== Phase 11: Final L-BFGS (score={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 15:
            for T in [50, 120]:
                rem = deadline - time.monotonic()
                if rem < 8:
                    break
                t_alloc = min(rem * 0.4, 30.0)
                a_fin, _ = lbgfs_refine(best_seq, T=T, time_limit=t_alloc, maxiter=1500)
                try_seq(a_fin, f"final_T{T}")
    
    # === Phase 12: Ultimate polish ===
    print(f"\n=== Phase 12: Ultimate polish (score={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        # Very gentle spectral equalization
        if rem > 5:
            a_ultra = spectral_equalize(best_seq, n_passes=5, strength=0.1)
            try_seq(a_ultra, "ultra_eq")
        
        # Final micro subgradient
        rem = deadline - time.monotonic()
        if rem > 10:
            t_micro = min(rem * 0.5, 20.0)
            a_micro, _ = subgradient_optimize(
                best_seq, time_limit=t_micro,
                init_lr=0.0005, n_active=max(40, len(best_seq) // 6),
                use_polyak=True, seed=seed + 999999
            )
            try_seq(a_micro, "micro_sg")
        
        # Final CD
        rem = deadline - time.monotonic()
        if rem > 8:
            a_cd2, _ = coordinate_polish(best_seq, time_limit=min(rem * 0.5, 15.0))
            try_seq(a_cd2, "final_cd")
    
    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*60}")
    
    return [float(x) for x in best_seq]
```
