"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using LP-based Frank-Wolfe iterations,
mirror descent, and multi-length search. The LP step solves for the maximal-mass
direction g that keeps the cross-convolution f*g bounded by the current peak M.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.optimize import minimize as scipy_minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_objective(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def symmetrize(a):
    """Make sequence palindromic."""
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym


def build_lp_matrix(f, n):
    """Build the constraint matrix A for the LP.
    A[k, j] = f[j] if max(0, k-n+1) <= j <= min(n-1, k), else 0.
    Shape: (2n-1) x n
    """
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A[k, js:je + 1] = f[js:je + 1]
    return A

    Returns the normalized direction p = g / sum(g), or None if infeasible.
    """
    n = len(f)
    A = build_lp_matrix(f, n)
    c = -np.ones(n)  # minimize -sum(g)
    b_ub = np.full(2 * n - 1, M)
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    
    if not res.success:
        return None
    
    g = res.x
    s = g.sum()
    if s < 1e-10:
        return None
    p = g / s
    return p


def lp_frank_wolfe_step(f, time_limit, step_candidates=None, verbose=False, label=""):
    """
    LP-based Frank-Wolfe iterations.

    3. Blend: f_new = (1-t)*f + t*p for optimal t

    if step_candidates is None:
        step_candidates = [0.1, 0.2, 0.3, 0.5, 0.7]

    best_f = f.copy()

    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        # Solve LP
        p = lp_frank_wolfe(f, M)
        if p is None:

            continue

        best_t = step_candidates[0]
        best_trial_obj = M
        for t in step_candidates:
            f_trial = (1 - t) * f + t * p
            f_trial = np.maximum(f_trial, 0.0)
            f_trial = f_trial / f_trial.sum()
            trial_conv = fast_convolve(f_trial)
            trial_obj = trial_conv.max()

                best_trial_obj = trial_obj
                best_t = t
        
        if best_trial_obj < best_obj - 1e-14:
            f = (1 - best_t) * f + best_t * p

            best_f = f.copy()
            no_improve = 0

def mirror_descent(a0, time_limit, top_k=30, temp_init=0.0005, lr_init=0.01,
                   lr_decay=0.9999, verbose=False, label=""):

            k = int(peak_indices[idx])

                continue

        else:
            no_improve += 1
        
        temp = min(temp * 1.002, 0.05)
        if no_improve > 100:
            lr *= 0.5
            no_improve = 0
        lr = max(lr * lr_decay, 1e-8)
        
        if no_improve > 500:
            rng = np.random.default_rng(it * 7919 + 13)

                best_a = a_sym.copy()

        if lr < 1e-9:

    return best_a, best_obj

def coordinate_polish(a0, time_limit, verbose=False, label=""):
    """Coordinate descent: adjust one coordinate at a time."""
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in [0.8, 0.9, 1.1, 1.25]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                trial = a.copy()
                trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                trial[mask] *= scale
                trial = np.maximum(trial, 0.0)
                ss = trial.sum()
                if ss < 1e-15:
                    continue
                trial = trial / ss
                val = fast_convolve(trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_a, best_obj


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_tent(n):
    """Triangular (tent) shape."""
    mid = n // 2
    if n % 2 == 0:
        x = np.minimum(np.arange(n), n - 1 - np.arange(n))
    else:
        x = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = x + 1.0
    a = a / a.sum()
    return a


def make_parabola(n):
    """Parabolic shape."""
    x = np.arange(n) - (n - 1) / 2.0
    a = 1.0 - (x / ((n - 1) / 2.0)) ** 2
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

The (f*f) term is bounded by M (by definition). The (f*p) term is bounded by M (by the LP constraint). The (p*p) term needs to be checked.

If t is small, the t^2 term is negligible, and the max of f_new*f_new is approximately (1-t)^2 * M + 2t(1-t) * M = (1-t)*M*(1-t+2t) = (1+t)(1-t)*M ≈ M(1-t^2) < M.

So for small t, the new peak is less than M. This guarantees progress!

The step size t should be chosen to balance: larger t gives faster progress but might violate the bound (due to the t^2 term). A safe choice is t = 0.1 or so.

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using LP-based Frank-Wolfe iterations,
smoothed mirror descent, standard mirror descent, and coordinate polish.
The LP step finds the maximal-mass direction g keeping cross-convolution f*g
bounded by the current peak, providing principled descent directions.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def symmetrize(a):
    """Make sequence palindromic."""
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym


def build_lp_matrix(f, n):
    """Build constraint matrix for LP: A[k,j] = f[j] if j in [k-n+1, k]."""
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A[k, js:je + 1] = f[js:je + 1]
    return A


def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M, g >= 0. Returns normalized g."""
    n = len(f)
    A = build_lp_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'disp': False})
    if not res.success:
        return None
    g = res.x
    s = g.sum()
    if s < 1e-10:
        return None
    return g / s

def compute_subgradient(a, peak_indices, weights):
    """Compute weighted subgradient of max_k (a*a)_k."""
    n = len(a)
    g = np.zeros(n)
    for idx in range(len(peak_indices)):
        k = int(peak_indices[idx])
        w_k = weights[idx]
        if w_k < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        g[js:je + 1] += 2.0 * w_k * a[k - np.arange(js, je + 1)]
    return g
