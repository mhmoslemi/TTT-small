    """Stable softmax of t*b."""
    tb = t * b
    tb = tb - tb.max()
    e = np.exp(tb)

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    n_opt = n // 2 if use_sym else n

    # Get initial state
    b = fast_convolve(a)
    best_obj = b.max()
    best_a = a.copy()
    t = t_start
    eta = eta_init
    momentum = momentum_init
    no_improve = 0
    it = 0
    t0 = time.monotonic()

    total_iters_target = 0
    t_anneal = max(0.999, (t_end / t_start) ** (1.0 / max(1, int(time_limit / 0.05))))

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.5:
            break

        # Compute convolution
        b = fast_convolve(a)
        current_obj = b.max()

        # Compute softmax weights
        p = softmax_stable(t, b)

        # Compute gradient via FFT
        grad = compute_gradient_fft(a, p)

        if use_sym:
            # Symmetrize gradient
            grad_sym = grad[:n_opt] + grad[n_opt:]
            # Work with half-vector
            a_half = a[:n_opt]
            grad_use = grad_sym
            n_use = n_opt
        else:
            a_half = a
            grad_use = grad
            n_use = n

        # Nesterov momentum
        if momentum > 0:
            grad_use = momentum * getattr(smoothed_mirror_descent, '_v', np.zeros(n_use)) + grad_use
            smoothed_mirror_descent._v = grad_use.copy()
            momentum = (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum)) / 2.0
            momentum = min(momentum, 2.0)

        # Mirror descent update with backtracking
        best_eta = 0
        best_new_obj = current_obj
        a_trial = a_half.copy()

        for backtrack in range(8):
            eta_try = eta * (0.5 ** backtrack)
            if eta_try < 1e-12:
                break

            if use_sym:
                a_new_half = a_half * np.exp(-eta_try * grad_use)
                a_new = np.concatenate([a_new_half[:], a_new_half[::-1]])
                a_new = np.maximum(a_new, 1e-15)
                a_new = a_new / a_new.sum()
            else:
                a_new = a_half * np.exp(-eta_try * grad_use)
                a_new = np.maximum(a_new, 1e-15)
                a_new = a_new / a_new.sum()

            b_new = fast_convolve(a_new)
            obj_new = b_new.max()

            if obj_new < best_obj + 1e-14:
                best_new_obj = obj_new
                best_eta = eta_try
                a_trial = a_new
                break

        if best_eta > 0:
            a = a_trial
            eta = min(eta * 1.1, 2.0)
            b = fast_convolve(a)
            current_obj = b.max()

            if current_obj < best_obj - 1e-12:
                best_obj = current_obj
                best_a = a.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"    SMD iter {it}: obj={best_obj:.8f}, t={t:.1f}, eta={eta:.4f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            eta *= 0.8

        # Anneal temperature
        t *= t_anneal
        t = min(t, 500.0)

        # Periodic restart
        if no_improve > 100:
            noise_amp = 0.002 * (1 + no_improve / 200.0)
            if use_sym:
                half = best_a[:n_opt] + rng.standard_normal(n_opt) * noise_amp
                half = np.maximum(half, 1e-10)
                a = np.concatenate([half, half[::-1]])
            else:
                a = best_a + rng.standard_normal(n) * noise_amp
                a = np.maximum(a, 1e-10)
            a = a / a.sum()
            eta = eta_init * 0.5
            momentum = 0.0
            smoothed_mirror_descent._v = np.zeros(n_use)
            no_improve = 0

        if no_improve > 500:
            break

    return best_a, best_obj


def peak_targeted_cd(a0, time_limit, verbose=False, seed=0):
    """
    Peak-targeted coordinate descent: identify top peaks and perform
    targeted mass redistribution to reduce them.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    b = fast_convolve(a)
    best_obj = b.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if no_improve > 80:
            break

        b = fast_convolve(a)
        M = b.max()
        # Get top peaks
        k_actual = min(10, len(b))
        peak_indices = np.argpartition(b, -k_actual)[-k_actual:]
        peak_indices = peak_indices[np.argsort(b[peak_indices])[::-1]]

        found_improvement = False
        for peak_k in peak_indices:
            peak_k = int(peak_k)
            # Contributing indices
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            contrib_idx = np.arange(js, je + 1)
            if len(contrib_idx) == 0:
                continue

            products = a[contrib_idx] * a[peak_k - contrib_idx]
            total = products.sum()
            if total < 1e-15:
                continue

            # Try reducing the top contributor
            for top_frac in [0.05, 0.1, 0.2, 0.3]:
                order = np.argsort(products)[::-1]
                for oi in order[:3]:
                    src = contrib_idx[oi]
                    partner = peak_k - src
                    diff = a[src] * top_frac
                    if diff < 1e-12:
                        continue

                    trial = a.copy()
                    trial[src] -= diff

                    # Redistribute to non-contributing coords
                    mask = np.ones(n, dtype=bool)
                    mask[contrib_idx] = False
                    if not mask.any():
                        mask[:] = True
                        mask[src] = False

                    nc_vals = trial[mask]
                    nc_sum = nc_vals.sum()
                    if nc_sum < 1e-15:
                        # Redistribute to all non-src coords
                        mask2 = np.ones(n, dtype=bool)
                        mask2[src] = False
                        nc_vals2 = trial[mask2]
                        nc_sum2 = nc_vals2.sum()
                        if nc_sum2 < 1e-15:
                            continue
                        trial[mask2] += diff * nc_vals2 / nc_sum2
                    else:
                        trial[mask] += diff * nc_vals / nc_sum

                    trial = np.maximum(trial, 0.0)
                    ts = trial.sum()
                    if ts < 1e-15:
                        continue
                    trial = trial / ts

                    obj = fast_convolve(trial).max()
                    if obj < best_obj - 1e-12:
                        a = trial
                        best_obj = obj
                        best_a = a.copy()
                        no_improve = 0
                        found_improvement = True
                        break
                if found_improvement:
                    break
            if found_improvement:
                break

        if not found_improvement:
            no_improve += 1
            # Try a random perturbation
            if no_improve % 20 == 0:
                amp = 0.001 + 0.002 * (no_improve / 80.0)
                a = best_a + rng.standard_normal(n) * amp
                a = project_simplex(a)

    return best_a, best_obj


def multiplicative_descent(a0, time_limit, verbose=False, seed=0,
                           lr_init=0.02, decay=0.9998, top_k=20):
    """Mirror descent with top-k peak subgradient."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 1e-12)
    a = a / a.sum()
    n = len(a)

    b = fast_convolve(a)
    best_obj = b.max()
    best_a = a.copy()
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        M = b.max()

        # Top-k peaks
        k_act = min(top_k, len(b))
        peaks = np.argpartition(b, -k_act)[-k_act:]
        pv = b[peaks]
        w = pv / pv.sum()

        # Subgradient
        sg = np.zeros(n)
        for i in range(k_act):
            if w[i] < 1e-10:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

        # Multiplicative update
        a = a * np.exp(-lr * sg)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        co = fast_convolve(a).max()
        if co < best_obj - 1e-14:
            best_obj = co
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    MD iter {it}: obj={best_obj:.8f}, lr={lr:.6f}")
        else:
            no_improve += 1

        lr *= decay

        if no_improve > 300:
            a = best_a + rng.standard_normal(n) * 0.001
            a = np.maximum(a, 1e-12)
            a = a / a.sum()
            lr = lr_init * 0.3
            no_improve = 0

        if lr < 1e-10:
            break

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian-like initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_cosine_mod(n, freq, amp):
    """Cosine-modulated uniform sequence."""
    x = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_lorentzian(n, gamma):
    """Lorentzian-shaped sequence."""
    x = np.arange(n) - n / 2.0
    a = 1.0 / (1.0 + (x / gamma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell_flat(n, width=0.7):
    """Flat-top bell."""
    x = np.arange(n) - n / 2.0
    hw = n * width / 2.0
    t = np.clip(np.abs(x) / hw, 0, 1.0)
    a = (1.0 - t) ** 3
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization combining smoothed mirror descent,
    peak-targeted coordinate descent, and multiplicative descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # === Phase 1: Diverse initializations ===
    print("\n=== Phase 1: Initializations ===")
    t_phase = time.monotonic()

    n_parent = len(parent) if parent is not None else 768
    n_values = [n_parent, 640, 896, 1024, 512, 1280]
    n_values = sorted(set(n_values))

    init_candidates = []
    for n_try in n_values:
        for sigma in [0.2, 0.25, 0.3, 0.35, 0.4]:
            init_candidates.append((f'g{sigma}_n{n_try}', make_gaussian(n_try, sigma)))
        for freq in [1, 2, 3]:
            for amp in [0.1, 0.2, 0.3]:
                init_candidates.append((f'c{freq}_{amp}_n{n_try}', make_cosine_mod(n_try, freq, amp)))
        for gamma in [0.2, 0.3, 0.4, 0.5]:
            init_candidates.append((f'L{gamma}_n{n_try}', make_lorentzian(n_try, n_try * gamma)))
        for w in [0.5, 0.7, 0.9]:
            init_candidates.append((f'B{w}_n{n_try}', make_bell_flat(n_try, w)))
        init_candidates.append((f'U_n{n_try}', np.ones(n_try) / n_try))

    if parent is not None:
        init_candidates.append(('parent', parent))

    # Quick evaluation
    scored_inits = []
    for label, a_init in init_candidates:
        sc = objective_normalized(a_init)
        scored_inits.append((sc, a_init, label))
    scored_inits.sort(key=lambda x: x[0])

    # Try top inits with evaluate_sequence
    for i in range(min(10, len(scored_inits))):
        sc, a_init, label = scored_inits[i]
        if sc < best_score / (2 * len(a_init) * (1.0 / (2 * len(a_init)))):
            try_seq(a_init, f"init_{label}")
        if time.monotonic() > deadline - 1500:
            break

    best_init = scored_inits[0][1].copy()
    best_init_score = scored_inits[0][0]
    print(f"  Best init: {scored_inits[0][2]}, obj={best_init_score:.6f}")
    print(f"  Phase 1 time: {time.monotonic()-t_phase:.1f}s")

    # === Phase 2: Smoothed Mirror Descent (main workhorse) ===
    print("\n=== Phase 2: Smoothed Mirror Descent ===")
    rem = deadline - time.monotonic() - 200
    if rem > 30:
        # Try top 3 initializations
        for trial in range(min(3, len(scored_inits))):
            rem = deadline - time.monotonic() - 150
            if rem < 20:
                break
            sc, a_init, label = scored_inits[trial]
            t_alloc = min(rem * 0.7, 300)
            if t_alloc < 10:
                break
            print(f"  SMD [{label}] n={len(a_init)}: alloc={t_alloc:.0f}s")
            a_opt, obj = smoothed_mirror_descent(a_init, time_limit=t_alloc,
                                                 t_start=0.5, t_end=150.0,
                                                 eta_init=0.3, momentum_init=0.0,
                                                 verbose=True, seed=seed + trial)
            try_seq(a_opt, f"smd_{label}")

    # Also try with symmetry
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            n_best = len(best_seq)
            if n_best % 2 == 0:
                t_alloc = min(rem * 0.5, 150)
                print(f"  SMD sym: n={n_best}, alloc={t_alloc:.0f}s")
                a_opt, obj = smoothed_mirror_descent(best_seq, time_limit=t_alloc,
                                                     t_start=1.0, t_end=100.0,
                                                     eta_init=0.2, momentum_init=0.1,
                                                     verbose=True, seed=seed + 10,
                                                     symmetric=True)
                try_seq(a_opt, "smd_sym")

    # === Phase 3: Multiplicative Descent ===
    print("\n=== Phase 3: Multiplicative Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_alloc = min(rem * 0.6, 120)
            print(f"  MD: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, obj = multiplicative_descent(best_seq, time_limit=t_alloc,
                                                verbose=True, seed=seed + 20,
                                                lr_init=0.03, top_k=30)
            try_seq(a_opt, "md")

    # === Phase 4: Peak-Targeted Coordinate Descent ===
    print("\n=== Phase 4: Peak-Targeted CD ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_alloc = min(rem * 0.7, 100)
            print(f"  PCD: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, obj = peak_targeted_cd(best_seq, time_limit=t_alloc,
                                          verbose=True, seed=seed + 30)
            try_seq(a_opt, "pcd")

    # === Phase 5: Multi-length exploration ===
    print("\n=== Phase 5: Multi-Length Exploration ===")
    if best_seq is not None:
        base_n = len(best_seq)
        target_ns = [512, 640, 768, 896, 1024, 1280]
        for target_n in target_ns:
            if target_n == base_n:
                continue
            rem = deadline - time.monotonic() - 80
            if rem < 15:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")
            if sc < best_score + 0.01:
                t_alloc = min(rem * 0.3, 60)
                if t_alloc > 10:
                    print(f"  SMD n={target_n}: alloc={t_alloc:.0f}s")
                    a_opt, obj = smoothed_mirror_descent(f_interp, time_limit=t_alloc,
                                                         t_start=1.0, t_end=100.0,
                                                         eta_init=0.2, seed=seed + target_n)
                    try_seq(a_opt, f"smd_n{target_n}")

    # === Phase 6: Fine-tuning rounds ===
    print("\n=== Phase 6: Fine-Tuning ===")
    if best_seq is not None:
        # Multiple short passes with different parameters
        configs = [
            ('smd_fine1', dict(t_start=50, t_end=300, eta_init=0.05, momentum_init=0.0)),
            ('smd_fine2', dict(t_start=10, t_end=200, eta_init=0.1, momentum_init=0.05)),
            ('md_fine', None),
            ('pcd_fine', None),
        ]
        for label, cfg in configs:
            rem = deadline - time.monotonic() - 30
            if rem < 10:
                break
            t_alloc = min(rem * 0.4, 40)
            if t_alloc < 5:
                break
            if cfg is not None:
                print(f"  {label}: alloc={t_alloc:.0f}s")
                a_opt, obj = smoothed_mirror_descent(best_seq, time_limit=t_alloc,
                                                     seed=seed + 100, **cfg)
            elif label == 'md_fine':
                print(f"  {label}: alloc={t_alloc:.0f}s")
                a_opt, obj = multiplicative_descent(best_seq, time_limit=t_alloc,
                                                    seed=seed + 200, lr_init=0.01, top_k=50)
            elif label == 'pcd_fine':
                print(f"  {label}: alloc={t_alloc:.0f}s")
                a_opt, obj = peak_targeted_cd(best_seq, time_limit=t_alloc, seed=seed + 300)
            else:
                continue
            try_seq(a_opt, label)

    # === Phase 7: Final micro-steps ===
    print("\n=== Phase 7: Final Micro-Steps ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            t_alloc = min(rem, 20)
            print(f"  Final SMD: alloc={t_alloc:.0f}s")
            a_opt, obj = smoothed_mirror_descent(best_seq, time_limit=t_alloc,
                                                 t_start=100, t_end=500,
                                                 eta_init=0.02, momentum_init=0.0,
                                                 seed=seed + 999)
            try_seq(a_opt, "final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
