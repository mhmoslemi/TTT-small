import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if not len(rho):
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_gradient(f, conv, tau, top_frac=0.25):
    """Compute smoothed gradient of log-sum-exp of top convolution peaks."""
    n = len(f)
    len_c = len(conv)
    k_top = max(10, int(len_c * top_frac))
    peaks = np.argpartition(conv, -k_top)[-k_top:]
    pv = conv[peaks]
    scaled = tau * pv
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        idx = np.arange(js, je + 1)
        contrib = f[k - idx] + f[idx]
        # Fix diagonal: if k == 2*j, the term is 2*f[j] not f[j]+f[j]... actually it is f[k-j]+f[j] = f[j]+f[j] = 2*f[j] which is correct
        sg[js:je + 1] += w[i] * contrib
    return sg / max(tau * conv[peaks].max(), 1e-15) * tau

def projected_gradient_momentum(f0, time_limit, verbose=False,
                                 tau_init=50.0, tau_min=5.0,
                                 lr_init=0.005, momentum_init=0.9,
                                 decay_tau=0.9995, decay_lr=0.9999,
                                 decay_mom=1.0):
    """
    Projected gradient descent with momentum on the smoothed max-conv objective.
    Uses Euclidean projection onto simplex. Much better conditioned than
    multiplicative updates for this problem.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    tau = tau_init
    lr = lr_init
    momentum = momentum_init
    vel = np.zeros(n)
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Compute smoothed gradient
        sg = compute_smoothed_gradient(f, conv, tau, top_frac=0.25)

        # Momentum update
        vel = momentum * vel + lr * sg
        y = f - vel
        f_new = project_simplex(y)

        # Line search (backtracking)
        obj_new = fast_convolve(f_new).max()
        if obj_new > M * 1.01:
            for c in [0.5, 0.25, 0.1, 0.05, 0.01]:
                f_trial = f - c * vel
                f_trial = project_simplex(f_trial)
                obj_trial = fast_convolve(f_trial).max()
                if obj_trial < M:
                    f_new = f_trial
                    obj_new = obj_trial
                    break
            else:
                lr *= 0.5
                vel *= 0.5
                if lr < 1e-9:
                    break
                continue

        # Update
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"      PGM iter {it}: obj={best_obj:.10f}, tau={tau:.1f}, lr={lr:.6f}")
        else:
            no_improve += 1

        f = f_new

        # Adaptive parameters
        tau = max(tau * decay_tau, tau_min)
        lr *= decay_lr
        momentum = min(momentum * decay_mom, 0.95)

        # Restart
        if no_improve > 300:
            rng = np.random.default_rng(it * 13 + 7)
            noise = rng.standard_normal(n) * 0.002
            f = project_simplex(best_f + noise)
            vel = np.zeros(n)
            lr = lr_init * 0.5
            no_improve = 0

def fw_away_steps(f0, time_limit, verbose=False, max_away=20):
    """
    Frank-Wolfe with away steps. Uses the structure that the linear
    minimization oracle over the simplex picks a vertex.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:

    else:
        f = f / s
    n = len(f)

    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    vertex_set = []  # Track vertices for away steps
    tau = 20.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Compute gradient using smoothed max
        k_top = max(5, int(len(conv) * 0.2))
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        pv = conv[peaks]

        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        g = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue
            idx = np.arange(js, je + 1)
            g[js:je + 1] += w[i] * (f[k - idx] + f[idx])

        # FW toward step: minimize <g, s> over simplex → s = e_{argmin g}
        j_toward = int(np.argmin(g))

        # Away step: maximize <g, s> over vertex set
        j_away = -1
        beta = 0.0
        if vertex_set:
            g_away = max(g[v] for v in vertex_set)
            if g_away > max(g[j_toward], 0):
                j_away = max(vertex_set, key=lambda v: g[v])
                # Optimal beta for away step
                gamma = (f[j_toward] * (g[j_toward] - g[j_away])) / (
                    f[j_toward] * (g[j_toward] - g[j_away]) +
                    f[j_away] * (g[j_away] - g[j_toward]))
                beta = max(0.0, min(1.0, gamma))

        # Standard FW line search
        best_alpha = 0.0
        best_val = M
        direction = np.zeros(n)
        direction[j_toward] = 1.0
        if j_away >= 0 and beta > 0.01:
            direction[j_away] = -beta

                continue
            f_trial = f_trial / st
            val = fast_convolve(f_trial).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine line search
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.03)
            hi = min(1.0, best_alpha + 0.03)
            for _ in range(5):
                mid = (lo + hi) / 2.0
                f_trial = f + mid * direction
                f_trial = np.maximum(f_trial, 0.0)
                st = f_trial.sum()
                if st < 1e-15:
                    break
                f_trial = f_trial / st
                val = fast_convolve(f_trial).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha < 1e-6:
            no_improve += 1
            tau = max(tau * 0.95, 2.0)
            if no_improve > 80:
                # Perturb and restart
                rng = np.random.default_rng(it * 3 + 11)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                vertex_set = []
                no_improve = 0
            continue
        else:
            no_improve = 0

        f_new = f + best_alpha * direction
        f_new = np.maximum(f_new, 0.0)

        f = f_new

        # Update vertex set

def coordinate_descent_pairs(f0, time_limit, verbose=False):

        improved = False
        rng = np.random.default_rng(full_pass * 1000 + 42)
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj
            if f[j] < 1e-14:
                continue

            for target in [j + 1, j - 1, j + 2, j - 2]:
                if target < 0 or target >= n:
                    continue
                if f[target] < 1e-14 and target != j:
                    continue

                    if f_trial[j] < 0 or f_trial[target] < 0:

                    val = fast_convolve(f_trial).max()
                    if val < best_obj - 1e-14:
                        f = f_trial
                        best_obj = val
                        best_f = f_trial.copy()

                else:

                break

        if not improved:
            # Try single coordinate optimization
            single_improved = False
            for j in order:
                if time.monotonic() - t0 > time_limit:
                    return best_f, best_obj
                if f[j] < 1e-12:
                    continue

                best_j_obj = best_obj
                best_j_val = f[j]
                others_sum = 1.0 - f[j]
                if others_sum < 1e-15:

def lp_direction_reduced(f, M, n, top_k=50):
    """
    Solve reduced LP for FW direction using only top-k near-active constraints.
    Much faster than full LP.
    """
    conv = fast_convolve(f)
    # Get top-k constraint indices
    k_indices = np.argpartition(conv, -top_k)[-top_k:]
    k_indices = sorted(k_indices.tolist())

    # Build reduced constraint matrix
    A_rows = []
    b_rows = []
    for k in k_indices:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        row = np.zeros(n)
        idx = np.arange(js, je + 1)
        row[js:je + 1] = f[k - idx] + f[idx]
        A_rows.append(row)
        b_rows.append(M)

    if not A_rows:
        return None

    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    c_obj = np.ones(n)  # maximize sum(g), so minimize -sum(g)

    try:
        res = linprog(-c_obj, A_ub=A_ub, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': 5})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using cubic-like interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f, kind='cubic') if old_n >= 4 else np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
