In log-space (a_i = exp(z_i) / sum_j exp(z_j)):
f(z) = 2n * soft_max(conv(softmax(z), softmax(z); T)) / (sum(softmax(z)))^2
     = 2n * soft_max(conv(softmax(z), softmax(z); T))  [since sum(softmax(z)) = 1]

So we just need to minimize 2n * soft_max(conv(a,a); T) where a = softmax(z).

The gradient with respect to z:
∂f/∂z_i = 2n * ∂soft_max/∂a_i * ∂a_i/∂z_i + 2n * sum_j ∂soft_max/∂a_j * ∂a_j/∂z_i

Since a = softmax(z): ∂a_i/∂z_i = a_i(1-a_i), ∂a_j/∂z_i = -a_j*a_i for j≠i

So: ∂f/∂z_i = 2n * a_i * (g_i - sum_j g_j * a_j)

where g_i = ∂soft_max/∂a_i is the subgradient component.

This is the standard "natural gradient" or "Fisher information matrix" scaling for softmax.

OK let me now write the code. I'll focus on making it efficient and effective.

Key implementation details:
1. FFT size: for n=1152, 3n-2 = 3454, next power of 2 = 4096. That's manageable.
2. For the soft-max, I need to be careful about numerical stability (subtract max before exp).
3. L-BFGS-B should work well in log-space since the variables are unbounded.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- FFT-accelerated L-BFGS with soft-max smoothing in log-space

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_local(a):
    """Compute objective 2n*max(conv)/(sum)^2 locally."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S * S))


def softmax(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def log_from_a(a):
    """Convert positive vector to log-space."""
    return np.log(np.maximum(a, 1e-300))


def compute_gradient_fft(w, a):
    """
    Compute g[i] = sum_j w[j+i] * a[j] for i=0..n-1
    using FFT cross-correlation.
    w: length 2n-1 (weights over convolution positions)
    a: length n
    Returns: g of length n
    """
    n = len(a)
    m = len(w)  # should be 2n-1
    # Need FFT size >= m + n - 1
    L = 1 << (m + n - 2).bit_length()
    W = rfft(w, L)
    A = rfft(a, L)
    # Cross-correlation: ifft(conj(W) * A)
    c = irfft(np.conj(W) * A, L)
    return c[:n]


def soft_max_obj_grad(z, T):
    """
    Compute soft-max smoothed objective and gradient in log-space.
    Objective: 2n * (1/T) * log(sum(exp(T*b_k)))
    where b = conv(softmax(z), softmax(z))
    """
    n = len(z)
    a = softmax(z)
    b = fast_convolve(a)
    
    # Soft-max computation (numerically stable)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    log_sum_exp = Tb_max / T + np.log(exp_Tb.sum()) / T
    # Actually: soft_max = (1/T) * log(sum(exp(T*b))) = (1/T) * (Tb_max + log(sum(exp(Tb - Tb_max))))
    # = b_max + (1/T) * log(sum(exp(T*b - T*b_max)))
    
    # Weights
    w = exp_Tb / exp_Tb.sum()
    
    # Gradient of soft_max w.r.t. a: cross-correlation of w and a
    g_a = compute_gradient_fft(w, a)
    
    # Objective value
    obj = 2.0 * n * (Tb_max + np.log(exp_Tb.sum())) / T
    
    # Gradient w.r.t. z (natural gradient for softmax)
    # d(obj)/dz_i = 2n * a_i * (g_a_i - sum_j g_a_j * a_j)
    S = np.dot(g_a, a)
    grad_z = 2.0 * n * a * (g_a - S)
    
    return obj, grad_z


def lbgfs_optimize(a0, time_limit, T_schedule=None, maxiter=10000):
    """
    L-BFGS optimization in log-space with soft-max smoothing.
    a0: initial point (non-negative, will be normalized)
    time_limit: time in seconds
    T_schedule: list of temperatures to try (decreasing)
    """
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 0)
    S = a.sum()

        return a0.copy(), float('inf')
    a = a / S
    
    if T_schedule is None:
        T_schedule = [200, 100, 50, 30, 20, 15, 10]
    
    z = log_from_a(a)
    best_obj = eval_local(a)
    best_a = a.copy()
    
    per_T_time = time_limit / len(T_schedule)
    
    for T in T_schedule:
        t_rem = (time_limit + t0) - time.monotonic()
        if t_rem < 5:
            break
        t_alloc = min(per_T_time, t_rem)
        
        z_current = log_from_a(best_a)
        
        def func(z):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            obj, _ = soft_max_obj_grad(z, T)
            return obj if np.isfinite(obj) else 1e15
        
        def jac(z):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            _, g = soft_max_obj_grad(z, T)
            return g
        
        try:
            res = minimize(
                fun=func, x0=z_current, jac=jac, method='L-BFGS-B',
                options={
                    'maxiter': 5000,
                    'ftol': 1e-15,
                    'gtol': 1e-12,
                    'maxfun': 20000,
                    'disp': False
                }
            )
            a_trial = softmax(res.x)
            trial_obj = eval_local(a_trial)
            if trial_obj < best_obj:
                best_obj = trial_obj
                best_a = a_trial
            print(f"    T={T}: L-BFGS done, obj={trial_obj:.6f}, best={best_obj:.6f}")
        except Exception as e:
            print(f"    T={T}: L-BFGS error: {e}")
    
    return best_a, best_obj


def active_set_subgradient(a0, time_limit, K=20, lr_init=0.001, momentum=0.9):
    """
    Subgradient method focusing on top K peaks, with FFT-accelerated gradient.
    """
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    lr = lr_init
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        # Start a batch
        b = fast_convolve(a)
        b_max = b.max()
        
        # Get top K peaks
        if K < len(b):
            top_idx = np.argpartition(b, -K)[-K:]
        else:
            top_idx = np.arange(len(b))
        
        # Softmax weights over top peaks
        b_top = b[top_idx]
        T_peak = max(1e-6, (b_top.max() - b_top.min()) * 0.5)
        w_top = np.exp(b_top / T_peak - b_top.max() / T_peak)
        w_top /= w_top.sum()
        
        # Build full weight vector
        w = np.zeros(len(b))
        w[top_idx] = w_top
        
        # Compute gradient via FFT
        g = compute_gradient_fft(w, a)
        
        # Nesterov step with simplex projection
        v = momentum * v - lr * g
        a_new = a + v
        # Project onto {x >= 0, sum(x) = 1}
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-10:
            a_new /= s
        else:
            a_new = best_a.copy()
        
        # Evaluate
        b_new = fast_convolve(a_new)
        obj_new = b_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Revert if no improvement for a while
            if no_improve > 200:
                a = best_a + np.random.default_rng(it).standard_normal(n) * 0.001
                a = np.maximum(a, 0)
                a = a / a.sum()
                v = np.zeros(n)
                lr *= 0.8
                no_improve = 0
        
        a = a_new
        it += 1
        
        # Adaptive learning rate
        if it % 500 == 0:
            lr *= 0.95
    
    return best_a, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length preserving the sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
