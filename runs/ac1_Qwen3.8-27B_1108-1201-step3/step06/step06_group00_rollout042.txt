```python
"""
Search for low-peak squared step functions.

The score is 2*n*max(a*a)/(sum a)^2, where a*a is the full convolution.
The implementation combines:
- fast log-domain subgradient / heavy-ball optimization with exact line search,
- smoothed L-BFGS in log-space with scale-aware peak smoothing,
- exact cyclic coordinate descent with one-coordinate affine rescaling,
- an optional linear-programmed direction from the current operating point,
- interpolation to nearby lengths and perturbation restarts.
"""

import time
import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


# -----------------------------------------------------------------------------
# Global state used only by the top-level scipy callback functions.
# -----------------------------------------------------------------------------

_LBF_N = 0
_LBF_NFT = 0
_LBF_NCORR = 0
_LBF_PIVOT = 0
_LBF_BETA = 0.0
_LBF_DEADLINE = 0.0
_LBF_BEST_SCORE = float("inf")
_LBF_BEST_A = None


# -----------------------------------------------------------------------------
# Basic utilities
# -----------------------------------------------------------------------------

def next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def normalize_to_A(A):
    A = np.asarray(A, dtype=np.float64).copy()
    n = len(A)
    if n == 0:
        return A
    A = np.nan_to_num(A, nan=0.0, posinf=0.0, neginf=0.0)
    A = np.maximum(A, 0.0)
    S = float(A.sum())
    if S <= 0.0:
        return np.ones(n, dtype=np.float64)
    return A * (float(n) / S)


def local_score_A(A):
    A = normalize_to_A(A)
    n = len(A)
    if n == 0:
        return float("inf")
    S = float(A.sum())
    if S <= 0.0:
        return float("inf")
    b = convolve_fft(A)
    return float(2.0 * n * b.max() / (S * S))


def convolve_fft(a, N=None):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    if N is None or N < size:
        N = next_pow2(size)
    padded = np.zeros(N, dtype=np.float64)
    padded[:n] = a
    F = rfft(padded, N)
    out = irfft(F * F, N)[:size]
    return out


def gradient_fft(A, W, factor=2.0, N=None):
    """
    If b_k = sum_j A_j W_{k-j} style correlation is needed, this computes
    factor * sum_k W_k A_{k-i} for i=0..n-1.
    """
    A = np.asarray(A, dtype=np.float64)
    W = np.asarray(W, dtype=np.float64)
    n = len(A)
    L = 2 * n - 1
    if N is None or N < L + n - 1:
        N = next_pow2(L + n - 1)
    Ap = np.zeros(N, dtype=np.float64)
    Ap[:n] = A
    Wp = np.zeros(N, dtype=np.float64)
    Wp[:L] = W
    Af = rfft(Ap, N)
    Wf = rfft(Wp, N)
    corr = irfft(np.conj(Wf) * Af, N)[:n]
    return factor * corr


def direct_gradient_top(A, idx, weights, factor=2.0):
    """
    Compute sum_k weights[k] * db_k/dA using only selected peaks.
    """
    n = len(A)
    G = np.zeros(n, dtype=np.float64)
    for k, w in zip(idx, weights):
        if w <= 0.0:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi < lo:
            continue
        j_start = k - hi
        j_end = k - lo + 1
        G[lo:hi + 1] += factor * w * A[j_start:j_end][::-1]
    return G


def select_active_weights(b, top_k, beta_scale=10.0):
    L = len(b)
    K = int(max(1, min(top_k, L)))
    if K >= L:
        idx = np.arange(L)
    else:
        idx = np.argpartition(b, -K)[-K:]
    vals = b[idx]
    bmax = float(vals.max())
    temp = bmax / float(max(1.0, beta_scale))
    if temp <= 1e-15:
        temp = 1e-15
    w = np.exp((vals - bmax) / temp)
    s = float(w.sum())
    if s <= 0.0:
        w = np.zeros(K, dtype=np.float64)
        w[int(np.argmax(vals))] = 1.0
        return idx, w
    return idx, w / s


def interpolate_p(p, new_n):
    p = np.asarray(p, dtype=np.float64)
    old_n = len(p)
    p = np.nan_to_num(p, nan=0.0, posinf=0.0, neginf=0.0)
    p = np.maximum(p, 0.0)
    S = float(p.sum())
    if S <= 0.0:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    p = p / S
    if old_n == new_n:
        return p.copy()
    if old_n == 1:
        return np.ones(new_n, dtype=np.float64) / float(new_n)
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    q = np.interp(x_new, x_old, p)
    q = np.maximum(q, 0.0)
    qS = float(q.sum())
    if qS <= 0.0:
        q = np.ones(new_n, dtype=np.float64)
    else:
        q = q / qS
    return q


# -----------------------------------------------------------------------------
# Log-domain subgradient optimizer
# -----------------------------------------------------------------------------

def log_gradient_optimize(A0, time_limit=60.0, use_energy=False, top_k=64,
                          beta_scale=10.0, eta0=0.015, momentum=0.55,
                          seed=0, restart_interval=1200, verbose=False):
    A = normalize_to_A(A0)
    n = len(A)
    if n <= 1:
        return A

    end = time.monotonic() + max(0.0, float(time_limit))
    Nconv = next_pow2(2 * n - 1)
    Ncorr = next_pow2(3 * n) if use_energy else 0

    rng = np.random.default_rng(int(seed))
    best_A = A.copy()
    b = convolve_fft(A, Nconv)
    S = float(A.sum())
    best_score = float(2.0 * n * b.max() / (S * S))
    cur_score = best_score

    vel = np.zeros(n, dtype=np.float64)
    base_t = float(max(1e-6, eta0))
    no_improve = 0

    if verbose:
        print(f"    log-gradient start n={n} score={best_score:.8f} "
              f"energy={use_energy} top_k={top_k} beta={beta_scale}")

    while time.monotonic() < end:
        b = convolve_fft(A, Nconv)
        S = float(A.sum())
        cur_score = float(2.0 * n * b.max() / (S * S))

        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_A = A.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > restart_interval:
            noise = rng.standard_normal(n) * 0.012
            A = best_A * (1.0 + noise)
            A = normalize_to_A(A)
            vel = np.zeros(n, dtype=np.float64)
            base_t = float(max(eta0, 1e-5))
            no_improve = 0

        if use_energy:
            W = b / max(float(b.sum()), 1e-300)
            L = gradient_fft(A, W, factor=4.0, N=Ncorr)
        else:
            idx, w = select_active_weights(b, top_k, beta_scale)
            L = direct_gradient_top(A, idx, w, factor=2.0)

        S = float(A.sum())
        if S <= 0.0:
            break
        mean_L = float(np.dot(A, L) / S)
        grad_z = A * (L - mean_L)
        grad_z -= float(np.mean(grad_z))

        if not np.all(np.isfinite(grad_z)) or float(np.max(np.abs(grad_z))) <= 1e-16:
            no_improve += 1
            if no_improve > 10:
                break
            continue

        direction = momentum * vel - grad_z
        direction -= float(np.mean(direction))
        direction = np.nan_to_num(direction, nan=0.0, posinf=0.0, neginf=0.0)

        accepted = False
        for fac in (1.0, 0.5, 0.25):
            t = base_t * fac
            zs = t * direction
            if np.max(np.abs(zs)) > 25.0:
                zs = np.clip(zs, -25.0, 25.0)
            trial = A * np.exp(zs)
            St = float(trial.sum())
            if St <= 0.0:
                continue
            trial *= float(n) / St
            bt = convolve_fft(trial, Nconv)
            score_t = float(2.0 * n * bt.max() / (St * St))
            if score_t < cur_score - 1e-13:
                A = trial
                cur_score = score_t
                vel = zs.copy()
                accepted = True
                if fac == 1.0:
                    base_t = min(base_t * 1.06, 0.35)
                else:
                    base_t = max(base_t * 0.75, 1e-6)
                break

        if not accepted:
            base_t *= 0.5
            vel = np.zeros(n, dtype=np.float64)
            if base_t < 1e-8:
                A = best_A * (1.0 + rng.standard_normal(n) * 0.008)
                A = normalize_to_A(A)
                vel = np.zeros(n, dtype=np.float64)
                base_t = float(max(eta0, 1e-5))
                no_improve = 0

    if cur_score < best_score:
        best_score = cur_score
        best_A = A.copy()

    if verbose:
        print(f"    log-gradient done  n={n} score={best_score:.8f}")
    return best_A


# -----------------------------------------------------------------------------
# Smoothed L-BFGS optimizer in log-space
# -----------------------------------------------------------------------------

def _lbf_compute(x):
    x = np.asarray(x, dtype=np.float64)
    n = _LBF_N
    pivot = _LBF_PIVOT
    beta = _LBF_BETA
    N = _LBF_NCORR

    z = x.copy()
    z[pivot] = 0.0
    z = np.nan_to_num(z, nan=0.0, posinf=50.0, neginf=-50.0)
    z = z - float(np.max(z))
    ez = np.exp(z)
    s = float(ez.sum())
    if s <= 0.0:
        p = np.ones(n, dtype=np.float64) / float(n)
    else:
        p = ez / s
    A = float(n) * p
    b = convolve_fft(A, N)
    S = float(A.sum())
    bmax = float(b.max())
    exact = float(2.0 * n * bmax / (S * S))

    m = bmax
    xb = beta * b - beta * m
    ex = np.exp(xb)
    se = float(ex.sum())
    if se <= 0.0:
        W = np.zeros(len(b), dtype=np.float64)
        W[int(np.argmax(b))] = 1.0
        f = m
    else:
        W = ex / se
        f = float((m + np.log(se)) / beta)
    return A, b, exact, W, f


def _lbf_func(x):
    if time.monotonic() > _LBF_DEADLINE:
        return 1e100
    try:
        A, b, exact, W, f = _lbf_compute(x)
        global _LBF_BEST_SCORE, _LBF_BEST_A
        if np.isfinite(exact) and exact < _LBF_BEST_SCORE:
            _LBF_BEST_SCORE = exact
            _LBF_BEST_A = A.copy()
        return f
    except Exception:
        return 1e100


def _lbf_jac(x):
    if time.monotonic() > _LBF_DEADLINE:
        return np.zeros_like(x, dtype=np.float64)
    try:
        A, b, exact, W, f = _lbf_compute(x)
        L = gradient_fft(A, W, factor=2.0, N=_LBF_NCORR)
        S = float(A.sum())
        mean_L = float(np.dot(A, L) / S)
        g = A * (L - mean_L)
        g[_LBF_PIVOT] = 0.0
        return np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
    except Exception:
        return np.zeros_like(x, dtype=np.float64)


def lbgfs_smooth(A0, focus=10.0, time_limit=30.0, maxiter=250, pivot=None,
                 verbose=False):
    A = normalize_to_A(A0)
    n = len(A)
    if n <= 1:
        return A

    A = A + 1e-12
    A = normalize_to_A(A)
    b = convolve_fft(A)
    bmax = float(max(b.max(), 1e-12))
    beta = float(max(1e-12, focus / bmax))

    Ncorr = next_pow2(max(3 * n, 2 * n - 1))
    if pivot is None:
        pivot = int(n // 2)

    z0 = np.log(A / float(n))
    z0 = z0 - float(z0[pivot])
    z0[pivot] = 0.0
    z0 = np.clip(z0, -50.0, 50.0)

    bounds = [(-50.0, 50.0)] * n
    bounds[pivot] = (0.0, 0.0)

    global _LBF_N, _LBF_NFT, _LBF_NCORR, _LBF_PIVOT, _LBF_BETA
    global _LBF_DEADLINE, _LBF_BEST_SCORE, _LBF_BEST_A

    _LBF_N = n
    _LBF_NFT = 2 * n - 1
    _LBF_NCORR = Ncorr
    _LBF_PIVOT = int(pivot)
    _LBF_BETA = beta
    _LBF_DEADLINE = time.monotonic() + max(0.0, float(time_limit))
    _LBF_BEST_SCORE = local_score_A(A)
    _LBF_BEST_A = A.copy()

    if verbose:
        print(f"    L-BFGS smooth focus={focus:.3f} beta={beta:.6e} n={n} "
              f"t={time_limit:.0f}s")

    try:
        res = minimize(fun=_lbf_func,
                       x0=z0,
                       jac=_lbf_jac,
                       method="L-BFGS-B",
                       bounds=bounds,
                       options={
                           "maxiter": int(max(50, maxiter)),
                           "ftol": 1e-13,
                           "gtol": 1e-8,
                           "maxls": 50,
                           "maxcor": 20,
                           "disp": False,
                       })
    except Exception:
        res = None

    best_A = _LBF_BEST_A if _LBF_BEST_A is not None else A.copy()
    best_sc = float(local_score_A(best_A))

    if res is not None and hasattr(res, "x") and time.monotonic() < _LBF_DEADLINE:
        try:
            A2, b2, ex2, W2, f2 = _lbf_compute(res.x)
            if np.isfinite(ex2) and ex2 < best_sc:
                best_A = A2
                best_sc = ex2
        except Exception:
            pass

    if verbose:
        print(f"    L-BFGS smooth done n={n} score={best_sc:.8f}")
    return best_A


# -----------------------------------------------------------------------------
# Exact cyclic coordinate descent
# -----------------------------------------------------------------------------

def coord_eval(x, c0, cross, d, diag_idx):
    if d <= 1e-15:
        return 1.0
    if x <= 0.0:
        return float((c0 / (d * d)).max())
    if x >= 1.0:
        return 1.0
    s = (1.0 - x) / d
    vb = c0 * (s * s) + cross * (x * s)
    vb[diag_idx] += x * x
    return float(vb.max())


def optimize_coordinate(p, c, i, n, iters=20, idx_arr=None):
    pi = float(p[i])
    d = 1.0 - pi
    L = 2 * n - 1
    if d <= 1e-14:
        return p, c, float(2.0 * n * c.max())

    if idx_arr is None:
        idx_arr = np.arange(n)
    k_arr = i + idx_arr
    diag_idx = 2 * i

    c0 = c.copy()
    vals = (2.0 * pi) * p
    vals[i] = 0.0
    c0[k_arr] -= vals
    c0[diag_idx] -= pi * pi

    cross = np.zeros(L, dtype=np.float64)
    vals2 = 2.0 * p
    vals2[i] = 0.0
    cross[k_arr] = vals2

    best_x = pi
    best_val = float(c.max())

    if 0.0 < pi < 1.0:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        a = 0.0
        b = 1.0
        c1 = b - gr * (b - a)
        d1 = a + gr * (b - a)
        fc = coord_eval(c1, c0, cross, d, diag_idx)
        fd = coord_eval(d1, c0, cross, d, diag_idx)
        if fc < best_val:
            best_val = fc
            best_x = c1
        if fd < best_val:
            best_val = fd
            best_x = d1
        for _ in range(max(0, int(iters))):
            if fc < fd:
                b = d1
                d1 = c1
                fd = fc
                c1 = b - gr * (b - a)
                fc = coord_eval(c1, c0, cross, d, diag_idx)
            else:
                a = c1
                c1 = d1
                fc = fd
                d1 = a + gr * (b - a)
                fd = coord_eval(d1, c0, cross, d, diag_idx)
            if c1 < best_val:
                best_val = c1
                best_x = c1
            if d1 < best_val:
                best_val = d1
                best_x = d1

    for xx in (0.0, 1.0, float(pi)):
        if xx < 0.0 or xx > 1.0:
            continue
        vx = coord_eval(xx, c0, cross, d, diag_idx)
        if vx < best_val - 1e-15:
            best_val = vx
            best_x = float(xx)

    if best_x <= 0.0:
        best_x = 0.0
    elif best_x >= 1.0:
        best_x = 1.0

    if abs(best_x - pi) < 1e-13 and best_val >= best_val - 1e-14:
        return p, c, float(c.max())

    s = (1.0 - best_x) / d
    p_new = p * s
    p_new[i] = best_x
    p_new = np.maximum(p_new, 0.0)
    pS = float(p_new.sum())
    if pS <= 0.0:
        p_new = np.zeros(n, dtype=np.float64)
        p_new[i] = 1.0
    else:
        p_new *= 1.0 / pS

    c_new = c0 * (s * s) + cross * (best_x * s)
    c_new[diag_idx] += best_x * best_x

    return p_new, c_new, float(2.0 * n * c_new.max())


def coordinate_polish(A0, time_limit=120.0, seed=0, iters=None):
    A = normalize_to_A(A0)
    n = len(A)
    if n <= 1:
        return A

    if iters is None:
        if n <= 1500:
            iters = 20
        elif n <= 3000:
            iters = 14
        else:
            iters = 10

    p = A / float(n)
    Nconv = next_pow2(2 * n - 1)
    c = convolve_fft(p, Nconv)
    cur_score = float(2.0 * n * c.max())

    best_p = p.copy()
    best_score = cur_score

    end = time.monotonic() + max(0.0, float(time_limit))
    rng = np.random.default_rng(int(seed))
    order = list(range(n))
    rng.shuffle(order)
    idx_arr = np.arange(n)

    no_improve_sweeps = 0

    while time.monotonic() < end:
        sweep_score_start = best_score
        for i in order:
            if time.monotonic() >= end:
                break
            p_new, c_new, sc = optimize_coordinate(p, c, i, n, iters=iters,
                                                   idx_arr=idx_arr)
            if sc <= cur_score + 1e-12:
                p = p_new
                c = c_new
                cur_score = sc
            if sc < best_score - 1e-12:
                best_score = sc
                best_p = p.copy()

        # correct accumulated drift
        c = convolve_fft(p, Nconv)
        cur_score = float(2.0 * n * c.max())
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_p = p.copy()

        if best_score < sweep_score_start - 1e-12:
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1
            if no_improve_sweeps >= 3:
                break
        rng.shuffle(order)

    return float(n) * best_p

# -----------------------------------------------------------------------------

def lp_direction_polish(A0, time_limit=60.0, cap_factor=50.0, verbose=False):
    A = normalize_to_A(A0)
    n = len(A)
    if n > 1400 or n <= 1:
        return A

    b = convolve_fft(A)
    M = float(b.max())
    if M <= 0.0:
        return A

    U = float(cap_factor * max(1.0, float(A.max())))
    rows = 2 * n - 1
    A_ub = np.zeros((rows, n), dtype=np.float64)
    idx = np.arange(n)
    for i in range(n):
        A_ub[i + idx, idx] = float(A[i])

    c_obj = -np.ones(n, dtype=np.float64)
    b_ub = np.full(rows, M, dtype=np.float64)
    bounds = [(0.0, U)] * n

    G = None
    try:
        res = linprog(c_obj,
                      A_ub=A_ub,
                      b_ub=b_ub,
                      bounds=bounds,
                      method="highs",
                      options={"time_limit": max(1.0, float(time_limit) - 5.0)})
        if res is not None and res.success and res.x is not None:
            G = np.asarray(res.x, dtype=np.float64)
    except Exception:
        G = None

    if G is None or not np.all(np.isfinite(G)):
        return A
    G = np.maximum(G, 0.0)
    GS = float(G.sum())
    if GS <= 0.0:
        return A
    H = n * G / GS

    best_A = A.copy()
    best_score = local_score_A(A)
    for t in (0.10, 0.25, 0.40, 0.50, 0.60, 0.75, 0.90, 1.0):
        mix = (1.0 - t) * A + t * H
        mix = normalize_to_A(mix)
        sc = local_score_A(mix)
        if sc < best_score - 1e-12:
            best_score = sc
            best_A = mix

    return best_A

def consider_candidate(A, best_p, best_score, tag=""):
    A = np.asarray(A, dtype=np.float64)
    A = np.nan_to_num(A, nan=0.0, posinf=0.0, neginf=0.0)
    A = np.maximum(A, 0.0)
    S = float(A.sum())
    if S <= 0.0:
        return best_p, best_score
    p = A / S
    sc = local_score_A(p)
    if np.isfinite(sc) and sc < best_score - 1e-12:
        if tag:
            print(f"  NEW BEST [{tag}] n={len(p)} score={sc:.8f}")
        return p, sc
    return best_p, best_score

def propose_candidate(seed=42, budget_s=1800.0):
    """Search for a nonnegative sequence minimizing the convolution-peak score."""
    end = time.monotonic() + max(1.0, float(budget_s) - 40.0)
    rng = np.random.default_rng(int(seed))

    # Load parent if available.
    parent = None
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        if h.ndim == 1 and h.size > 0:
            h = np.nan_to_num(h, nan=0.0, posinf=0.0, neginf=0.0)
            h = np.maximum(h, 0.0)
            if h.sum() > 0.0:
                parent = h / float(h.sum())
    except Exception:
        parent = None

    if parent is None:
        parent = np.ones(768, dtype=np.float64) / 768.0

    best_p = parent.copy()
    best_score = local_score_A(best_p)
    n_parent = len(parent)
    print(f"Parent initial: n={n_parent} score={best_score:.8f}")

    # Quick interpolation scan to see if another length is immediately better.
    for factor in (0.6, 0.75, 0.9, 1.1, 1.25, 1.5, 2.0):
        nn = int(round(n_parent * factor))
        if nn < 64 or nn > 8192:
            continue
        q = interpolate_p(parent, nn)
        best_p, best_score = consider_candidate(nn * q, best_p, best_score,
                                                tag=f"init_len{nn}")

    A = len(best_p) * best_p
    n_primary = len(best_p)

    # ---------------------------- Phase 1: primary -------------------------
    rem = end - time.monotonic()
    if rem > 60:
        t = min(rem - 20.0, 150.0)
        if t >= 10:
            print(f"Phase 1a: coordinate polish first, t={t:.0f}s, n={n_primary}")
            A2 = coordinate_polish(A, time_limit=t, seed=seed + 1)
            best_p, best_score = consider_candidate(A2, best_p, best_score,
                                                    tag="cd_first")
            A = len(best_p) * best_p

    rem = end - time.monotonic()
    if rem > 60:
        t = min(rem - 20.0, 70.0)
        if t >= 10:
            print(f"Phase 1b: energy grad, t={t:.0f}s, n={len(A)}")
            A2 = log_gradient_optimize(A, time_limit=t, use_energy=True,
                                       eta0=0.012, momentum=0.5, seed=seed + 7)
            best_p, best_score = consider_candidate(A2, best_p, best_score,
                                                    tag="energy")
            A = len(best_p) * best_p

    rem = end - time.monotonic()
    if rem > 60:
        t = min(rem - 20.0, 70.0)
        if t >= 10:
            print(f"Phase 1c: broad peak grad, t={t:.0f}s, n={len(A)}")
            A2 = log_gradient_optimize(A, time_limit=t, use_energy=False,
                                       top_k=256, beta_scale=2.0,
                                       eta0=0.018, momentum=0.55, seed=seed + 11)
            best_p, best_score = consider_candidate(A2, best_p, best_score,
                                                    tag="broad_grad")
            A = len(best_p) * best_p

    rem = end - time.monotonic()
    if rem > 50:
        t = min(rem - 20.0, 70.0)
        if t >= 10:
            print(f"Phase 1d: L-BFGS flat, t={t:.0f}s, n={len(A)}")
            A2 = lbgfs_smooth(A, focus=1.0, time_limit=t, maxiter=180)
            best_p, best_score = consider_candidate(A2, best_p, best_score,
                                                    tag="lbfgs_flat")
            A = len(best_p) * best_p

    rem = end - time.monotonic()
    if rem > 50:
        t = min(rem - 20.0, 70.0)
        if t >= 10:
            print(f"Phase 1e: high peak grad, t={t:.0f}s, n={len(A)}")
            A2 = log_gradient_optimize(A, time_limit=t, use_energy=False,
                                       top_k=32, beta_scale=25.0,
                                       eta0=0.010, momentum=0.72, seed=seed + 13)
            best_p, best_score = consider_candidate(A2, best_p, best_score,
                                                    tag="high_grad")
            A = len(best_p) * best_p

    rem = end - time.monotonic()
    if rem > 60:
        t = min(rem - 30.0, 110.0)
        if t >= 10:
            print(f"Phase 1f: L-BFGS peak, t={t:.0f}s, n={len(A)}")
            A2 = lbgfs_smooth(A, focus=20.0, time_limit=t, maxiter=300)
            best_p, best_score = consider_candidate(A2, best_p, best_score,
                                                    tag="lbfgs_peak")
            A = len(best_p) * best_p

    rem = end - time.monotonic()
    if rem > 90:
        t = min(rem - 30.0, 240.0)
        if t >= 15:
            print(f"Phase 1g: coordinate polish, t={t:.0f}s, n={len(A)}")
            A2 = coordinate_polish(A, time_limit=t, seed=seed + 17)
            best_p, best_score = consider_candidate(A2, best_p, best_score,
                                                    tag="cd_main")
            A = len(best_p) * best_p

    # ---------------------------- Phase 2: second polish -------------------
    rem = end - time.monotonic()
    if rem > 60:
        t = min(rem - 20.0, 50.0)
        if t >= 10:
            print(f"Phase 2a: ultra-high peak grad, t={t:.0f}s, n={len(A)}")
            A2 = log_gradient_optimize(A, time_limit=t, use_energy=False,
                                       top_k=12, beta_scale=80.0,
                                       eta0=0.004, momentum=0.82, seed=seed + 23)
            best_p, best_score = consider_candidate(A2, best_p, best_score,
                                                    tag="ultra_grad")
            A = len(best_p) * best_p

    rem = end - time.monotonic()
    if rem > 50:
        t = min(rem - 20.0, 70.0)
        if t >= 10:
            print(f"Phase 2b: L-BFGS ultra-peak, t={t:.0f}s, n={len(A)}")
            A2 = lbgfs_smooth(A, focus=80.0, time_limit=t, maxiter=350)
            best_p, best_score = consider_candidate(A2, best_p, best_score,
                                                    tag="lbfgs_ultra")
            A = len(best_p) * best_p

    rem = end - time.monotonic()
    if rem > 80:
        t = min(rem - 30.0, 150.0)
        if t >= 15:
            print(f"Phase 2c: final coordinate polish, t={t:.0f}s, n={len(A)}")
            A2 = coordinate_polish(A, time_limit=t, seed=seed + 29)
            best_p, best_score = consider_candidate(A2, best_p, best_score,
                                                    tag="cd_second")
            A = len(best_p) * best_p

    # Optional LP direction on the primary length.
    rem = end - time.monotonic()
    if rem > 120 and len(A) <= 1400:
        t = min(rem - 50.0, 90.0)
        if t >= 30:
            print(f"Phase 2d: LP direction polish, t={t:.0f}s, n={len(A)}")
            A2 = lp_direction_polish(A, time_limit=t)
            best_p, best_score = consider_candidate(A2, best_p, best_score,
                                                    tag="lp_dir")
            A = len(best_p) * best_p

    # ---------------------------- Phase 3: length sweep --------------------
    for factor in (0.75, 1.25, 1.5, 2.0):
        rem = end - time.monotonic()
        if rem < 140:
            break
        nn = int(round(len(best_p) * factor))
        if nn < 128 or nn > 65536 or abs(nn - len(best_p)) < 32:
            continue
        q = interpolate_p(best_p, nn)
        best_p2, best_score2 = consider_candidate(nn * q, best_p, best_score,
                                                  tag=f"interp_len{nn}")
        if best_p2 is not best_p:
            best_p, best_score = best_p2, best_score2

        rem = end - time.monotonic()
        if rem < 110:
            continue
        Aexp = nn * best_p

        t = min(rem - 80.0, 55.0)
        if t >= 15:
            Aexp2 = log_gradient_optimize(Aexp, time_limit=t, use_energy=False,
                                          top_k=64, beta_scale=10.0,
                                          eta0=0.012, momentum=0.6,
                                          seed=seed + 31 * factor)
            best_p, best_score = consider_candidate(Aexp2, best_p, best_score,
                                                    tag=f"exp_len{nn}_grad")
            Aexp = len(best_p) * best_p

        rem = end - time.monotonic()
        if rem < 90:
            continue
        t = min(rem - 50.0, 55.0)
        if t >= 15:
            Aexp2 = lbgfs_smooth(Aexp, focus=20.0, time_limit=t, maxiter=250)
            best_p, best_score = consider_candidate(Aexp2, best_p, best_score,
                                                    tag=f"exp_len{nn}_lbfgs")
            Aexp = len(best_p) * best_p

        rem = end - time.monotonic()
        if rem < 80:
            continue
        t = min(rem - 30.0, 70.0)
        if t >= 15:
            Aexp2 = coordinate_polish(Aexp, time_limit=t, seed=seed + 37 * int(factor * 10))
            best_p, best_score = consider_candidate(Aexp2, best_p, best_score,
                                                    tag=f"exp_len{nn}_cd")

    # Another quick no-optimization length scan using the improved profile.
    for factor in (0.8, 1.0, 1.1, 1.2, 1.4, 1.6):
        nn = int(round(len(best_p) * factor))
        if nn < 64 or nn > 8192:
            continue
        q = interpolate_p(best_p, nn)
        best_p, best_score = consider_candidate(nn * q, best_p, best_score,
                                                tag=f"lscan_{nn}")

    # ---------------------------- Phase 4: final micro ---------------------
    rem = end - time.monotonic()
    if rem > 40:
        t = min(rem - 15.0, 45.0)
        if t >= 8:
            A = len(best_p) * best_p
            A2 = coordinate_polish(A, time_limit=t, seed=seed + 41)
            best_p, best_score = consider_candidate(A2, best_p, best_score,
                                                    tag="micro_cd")
            A = len(best_p) * best_p

    rem = end - time.monotonic()
    if rem > 25:
        t = min(rem - 8.0, 25.0)
        if t >= 6:
            A = len(best_p) * best_p
            A2 = log_gradient_optimize(A, time_limit=t, use_energy=False,
                                       top_k=8, beta_scale=150.0,
                                       eta0=0.0018, momentum=0.9,
                                       seed=seed + 43)
            best_p, best_score = consider_candidate(A2, best_p, best_score,
                                                    tag="micro_grad")
            A = len(best_p) * best_p

    # A few perturbation restarts if there is still comfortable time.
    rem = end - time.monotonic()
    for attempt in range(5):
        if rem < 35:
            break
        t_sg = 12.0
        t_cd = min(12.0, max(5.0, rem - 20.0))
        if t_cd < 5:
            break
        A = len(best_p) * best_p
        A = A * (1.0 + rng.standard_normal(len(A)) * 0.004)
        A = normalize_to_A(A)

        A2 = log_gradient_optimize(A, time_limit=t_sg, use_energy=False,
                                   top_k=24, beta_scale=40.0, eta0=0.005,
                                   momentum=0.75, seed=seed + 50 + attempt)
        best_p, best_score = consider_candidate(A2, best_p, best_score,
                                                tag=f"pert_{attempt}_sg")
        A = len(best_p) * best_p

        A2 = coordinate_polish(A, time_limit=t_cd, seed=seed + 60 + attempt)
        best_p, best_score = consider_candidate(A2, best_p, best_score,
                                                tag=f"pert_{attempt}_cd")

    # Final cleanup and verification.
    best_p = np.asarray(best_p, dtype=np.float64)
    best_p = np.nan_to_num(best_p, nan=0.0, posinf=0.0, neginf=0.0)
    best_p = np.maximum(best_p, 0.0)
    S = float(best_p.sum())
    if S <= 0.0:
        best_p = parent.copy()
        S = float(best_p.sum())
    if S <= 0.0:
        best_p = np.ones(512, dtype=np.float64) / 512.0
        S = 1.0
    else:
        best_p = best_p / S

    # Limit values just in case; evaluator also clips.
    best_p = np.minimum(best_p, 1000.0)

    try:
        final_score = float(evaluate_sequence([float(x) for x in best_p]))
    except Exception:
        final_score = float("inf")

    if not np.isfinite(final_score):
        # Fallback to parent if evaluator rejects something.
        p2 = parent.copy()
        S2 = float(p2.sum())
        if S2 > 0:
            p2 /= S2
            p2 = np.nan_to_num(p2, nan=0.0, posinf=0.0, neginf=0.0)
            p2 = np.maximum(p2, 0.0)
            p2 = np.minimum(p2, 1000.0)
            try:
                final_score = float(evaluate_sequence([float(x) for x in p2]))
            except Exception:
                final_score = float("inf")
            if np.isfinite(final_score):
                best_p = p2
        else:
            best_p = np.ones(512, dtype=np.float64) / 512.0
            final_score = float("inf")

    print("=" * 60)
    print(f"FINAL SCORE: {final_score:.8f}  n={len(best_p)}")
    print("=" * 60)

    return [float(x) for x in best_p]
```
