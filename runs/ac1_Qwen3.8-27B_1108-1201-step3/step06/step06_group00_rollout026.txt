Yes, that's correct.

For n=1152, this matrix is 2303 x 1152 ≈ 2.6 million floats ≈ 21 MB. That's fine for memory.

The LP will have:
- 1152 variables (b_0, ..., b_{n-1})
- 2303 inequality constraints (conv(a,b) <= M)
- Non-negativity constraints (b >= 0)
- Objective: maximize sum(b)

This is a very standard LP formulation. With HiGHS (the default in scipy), this should solve in milliseconds to a few seconds.

Let me now think about the overall algorithm more carefully:

```
Initialize a (from parent or other)
Normalize: a = a / sum(a)
b = conv(a, a)
M = max(b)

For each iteration:

    if res.success:

    # Line search for step size
```

This should work. Let me think about potential issues:

1. The LP might be infeasible if the active constraints are too tight. In that case, we should relax (add more tolerance or include more constraints).

2. The direction b_dir might be very similar to a, leading to no improvement. In that case, we should try a different approach.

3. Numerical precision: for large n, the LP might have numerical issues.

Let me also add a fallback: if the LP fails, use a subgradient step instead.

Now, let me think about the line search more carefully. The objective as a function of t is:
f(t) = 2n * max(conv((1-t)*a + t*b_dir, (1-t)*a + t*b_dir))

= 2n * max((1-t)^2 * conv(a,a) + 2t(1-t) * conv(a,b_dir) + t^2 * conv(b_dir, b_dir))

This is the max of 2n-1 quadratic functions of t, so it's convex in t. We can minimize it using ternary search or golden section search.

Actually, since it's convex, we can also use the fact that the minimum is where the derivative changes sign. But for simplicity, let's just do a grid search followed by golden section.

```python
"""
LP-based optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses iterative linear programming to find improving directions,
combined with line search and coordinate descent polishing.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(n, a):
    """Build the dense convolution constraint matrix A where A[k,j] = a[k-j] if valid."""
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi+1] = a[k-hi:k-lo+1]
    return A

def solve_lp_direction(n, a, M, active_indices):
    """Solve the LP to find the best improving direction.
    Maximize sum(b) subject to conv(a, b)[active] <= M, b >= 0.
    """
    A = build_conv_matrix(n, a)
    A_active = A[active_indices, :]
    b_ub = np.full(len(active_indices), M)
    c = -np.ones(n)  # Minimize -sum(b) = maximize sum(b)
    
    res = linprog(c, A_ub=A_active, b_ub=b_ub, 
                  bounds=[(0, None)]*n, method='highs')
    
    if res.success and res.x is not None:
        return res.x
    return None

def line_search(a, b_dir, n, max_iter=50):
    """Golden section search for optimal step size t in [0, 1]."""
    def obj(t):
        a_t = (1-t)*a + t*b_dir
        a_t = np.maximum(a_t, 0)
        s = a_t.sum()
        if s < 1e-15:
            return float('inf')
        a_t = a_t / s
        b_t = fast_convolve(a_t)
        return 2*n*b_t.max()
    
    # First, check if t=1 is better
    obj_0 = obj(0.0)
    obj_1 = obj(1.0)
    
    if obj_1 < obj_0:
        lo, hi = 0.0, 1.0
    else:
        # Try some intermediate values
        best_t = 0.0
        best_obj = obj_0
        for t_test in [0.1, 0.2, 0.3, 0.5, 0.7]:
            o = obj(t_test)
            if o < best_obj:
                best_obj = o
                best_t = t_test
        if best_t == 0.0:
            return 0.0, obj_0
        # Narrow the range
        if best_t < 0.5:
            lo, hi = 0.0, min(1.0, best_t * 2)
        else:
            lo, hi = max(0.0, best_t * 0.5), 1.0
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ≈ 0.618
    a_gs = lo + (1 - gr) * (hi - lo)
    b_gs = lo + gr * (hi - lo)
    fa = obj(a_gs)
    fb = obj(b_gs)
    
    for _ in range(max_iter):
        if fa < fb:
            hi = b_gs
            b_gs = a_gs
            fb = fa
            a_gs = lo + (1 - gr) * (hi - lo)
            fa = obj(a_gs)
        else:
            lo = a_gs
            a_gs = b_gs
            fa = fb
            b_gs = lo + gr * (hi - lo)
            fb = obj(b_gs)
    
    t_opt = (lo + hi) / 2
    return t_opt, obj(t_opt)

def lp_iterate(a, n, tol=1e-4, max_iter=200, time_limit=60.0):
    """Iterative LP-based optimization."""
    t0 = time.monotonic()
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    best_a = a.copy()
    b = fast_convolve(a)
    M = b.max()
    best_M = M
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Identify active constraints (within tolerance of max)
        threshold = M - tol * M
        active = np.where(b >= threshold)[0]
        
        # If too many active constraints, use only the top ones
        if len(active) > 100:
            top_idx = np.argsort(b[active])[-100:]
            active = active[top_idx]
        
        # Solve LP
        b_dir = solve_lp_direction(n, a, M, active)
        
        if b_dir is None:
            # LP failed, try with relaxed constraints
            threshold = M - 2 * tol * M
            active = np.where(b >= threshold)[0]
            b_dir = solve_lp_direction(n, a, M * 1.01, active)
        
        if b_dir is None:
            break
        
        b_dir = b_dir / b_dir.sum()
        
        # Line search
        t_opt, obj_new = line_search(a, b_dir, n)
        
        if t_opt > 0 and obj_new < best_M * 2 * n - 1e-10:  # Actually obj_new already includes 2n
            # Wait, obj_new = 2*n*b_trial.max(), and we want to compare with 2*n*best_M
            # So: obj_new < 2*n*best_M  =>  b_trial.max() < best_M
            a = (1 - t_opt) * a + t_opt * b_dir
            a = np.maximum(a, 0)
            a = a / a.sum()
            b = fast_convolve(a)
            M = b.max()
            if M < best_M:
                best_M = M
                best_a = a.copy()
        else:
            # No improvement, try a smaller tolerance or break
            tol *= 0.5
            if tol < 1e-8:
                break
    
    return best_a, best_M

def coordinate_polish(a0, time_limit, n_multipliers=7):
    """Coordinate descent with multiple multipliers per element."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_convolve(a).max()
    
    multipliers = np.array([0.5, 0.7, 0.85, 1.0, 1.18, 1.4, 2.0])
    
    improved = True
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_M
            old_val = a[j]
            if old_val < 1e-14:
                continue
            best_j_M = best_M
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 1.0:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0 or scale > 1e6:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                val = fast_convolve(a_trial).max()
                if val < best_j_M - 1e-15:
                    best_j_M = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = np.maximum(a, 0)
                a = a / a.sum()
                if best_j_M < best_M - 1e-15:
                    best_M = best_j_M
                    best_a = a.copy()
                    improved = True
    return best_a, best_M

def subgradient_optimize(a0, time_limit=60.0, lr_init=0.01, momentum=0.9,
                         temp_init=0.001, temp_growth=1.02, seed=42):
    """Subgradient method with momentum (as backup/polish)."""
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 0)
    a = a / a.sum()
    v = np.zeros(n)
    best_a = a.copy()
    best_M = fast_convolve(a).max()
    cur_lr = lr_init
    cur_temp = temp_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        b_max = b.max()
        
        # Temperature-weighted peaks
        scaled = b / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(b))
            w[b.argmax()] = 1.0
        else:
            w /= w_sum
        
        # Subgradient
        g = np.zeros(n)
        nonzero_k = np.where(w > 1e-12)[0]
        for k in nonzero_k:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += w[k] * a[k - idx]
        
        # Nesterov step
        v = momentum * v - cur_lr * g
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a
        else:
            a_new = a_new / s
        
        b_new = fast_convolve(a_new)
        M_new = b_new.max()
        
        if M_new < best_M - 1e-15:
            best_M = M_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        if no_improve > 2000:
            a = np.maximum(best_a + rng.standard_normal(n) * 0.003, 0)
            a = a / a.sum()
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.7, lr_init * 0.1)
            no_improve = 0
        elif no_improve > 1000:
            cur_lr *= 0.95
        
        cur_temp = min(cur_temp * temp_growth, 0.5)
    
    return best_a, best_M
```

Wait, I need to reconsider the LP approach. The issue is that `build_conv_matrix` is called at every LP solve, and it's O(n^2). For n=1152, that's about 1.3M operations per call, which should be fine (a few milliseconds).

But the LP itself might be slow. Let me think about the size: 1152 variables, up to 2303 constraints. With HiGHS, this should be solvable in under a second typically.

Let me also reconsider the line search. The golden section search calls `fast_convolve` at each evaluation, which is O(n log n). With 50 iterations of golden section, that's 50 * O(n log n) per line search. For n=1152, each conv is about 1152 * 11 ≈ 12K operations, so 50 * 12K = 600K operations per line search. That's fine.

Actually, I realize there might be an issue with the line search. The objective function f(t) = 2n * max(conv((1-t)*a + t*b, (1-t)*a + t*b)) is not necessarily convex in t! It's the max of quadratic functions, and the max of convex functions is convex. Wait, each conv coefficient as a function of t is a quadratic (convex in t since the coefficient of t^2 is positive). So the max of convex functions is convex. Good, so golden section search should work.

Actually wait, let me double-check. conv((1-t)*a + t*b, (1-t)*a + t*b)[k]
= (1-t)^2 * conv(a,a)[k] + 2t(1-t) * cross(a,b)[k] + t^2 * conv(b,b)[k]

where cross(a,b)[k] = sum_i a[i]*b[k-i] (the cross-correlation).

The coefficient of t^2 is conv(b,b)[k] >= 0 (since b >= 0). So each coefficient is a convex quadratic in t. The max of convex functions is convex. Great, so the golden section search is valid.

But wait, there's a subtlety: after the update, we clip to non-negative and renormalize. This breaks the convexity. So the line search is only approximately valid. In practice, for small t, the clipping and renormalization don't change much, so it should be fine.

Let me also think about whether the LP direction is always improving. The LP finds b that maximizes sum(b) subject to conv(a,b) <= M (at active constraints). """
LP-based iterative optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations:
- Iterative LP direction finding (linearizes one factor of convolution)
- Convex line search along LP directions
- Active constraint management for efficient LPs
- Multi-phase strategy: LP -> subgradient -> coordinate descent -> alternation
- Sparse matrix support and safeguards for zero elements

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(n, a):
    """Build dense convolution constraint matrix A[k,j] = a[k-j] if valid."""
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[k, lo:hi + 1] = a[k - hi:k - lo + 1]
    return A


def solve_lp_direction(n, a, M, active_indices, upper_bound=1.0):
    """Solve LP: max sum(b) s.t. conv(a,b)[active] <= M, 0 <= b <= upper_bound."""
    A_full = build_conv_matrix(n, a)
    A_act = A_full[active_indices, :]
    b_ub = np.full(len(active_indices), M)
    c = -np.ones(n)
    bounds = [(0.0, upper_bound)] * n

    try:
        res = linprog(c, A_ub=A_act, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def line_search_convex(a, b_dir, n, max_evals=20):
    """Efficient line search for convex objective in t."""
    evals = [0]

    def obj(t):
        evals[0] += 1
        a_t = (1 - t) * a + t * b_dir
        a_t = np.maximum(a_t, 0)
        s = a_t.sum()
        if s < 1e-15:
            return float('inf')
        a_t = a_t / s
        return fast_convolve(a_t).max()

    obj0 = obj(0.0)
    obj1 = obj(1.0)

    if obj1 < obj0 * (1 - 1e-10):
        lo, hi = 0.0, 1.0
    else:
        # Minimum is near t=0, do a quick search
        candidates = [0.05, 0.1, 0.2, 0.4]
        best_t = 0.0
        best_o = obj0
        for tc in candidates:
            if evals[0] >= max_evals:
                break
            o = obj(tc)
            if o < best_o:
                best_o = o
                best_t = tc
        if best_t == 0.0:
            return 0.0, obj0
        # Narrow around best_t
        lo = max(0.0, best_t * 0.5)
        hi = min(1.0, best_t * 1.5)

    # Bisection on convex function
    gr = (np.sqrt(5) - 1) / 2
    a_gs = lo + (1 - gr) * (hi - lo)
    b_gs = lo + gr * (hi - lo)
    fa = obj(a_gs)
    fb = obj(b_gs)

    for _ in range(10):
        if evals[0] >= max_evals:
            break
        if fa < fb:
            hi = b_gs
            b_gs = a_gs
            fb = fa
            a_gs = lo + (1 - gr) * (hi - lo)
            fa = obj(a_gs)
        else:
            lo = a_gs
            a_gs = b_gs
            fa = fb
            b_gs = lo + gr * (hi - lo)
            fb = obj(b_gs)

    t_opt = (lo + hi) / 2
    return t_opt, obj(t_opt)


def lp_iterate(a0, n, tol_frac=1e-4, max_iter=300, time_limit=120.0,
               max_active=80, verbose=False):
    """Iterative LP-based optimization with active set management."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 0)
    a = a / a.sum()

    best_a = a.copy()
    b = fast_convolve(a)
    M = b.max()
    best_M = M
    stall_count = 0

    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break

        # Identify active constraints
        threshold = M * (1 - tol_frac)
        active = np.where(b >= threshold)[0]

        if len(active) == 0:
            active = np.array([b.argmax()])

        if len(active) > max_active:
            top_vals = b[active]
            top_idx = np.argsort(top_vals)[-max_active:]
            active = active[top_idx]

        # Solve LP
        b_dir = solve_lp_direction(n, a, M, active)

        if b_dir is None:
            # Relax constraints
            b_dir = solve_lp_direction(n, a, M * 1.001, np.arange(2 * n - 1))
            if b_dir is None:
                break

        b_sum = b_dir.sum()
        if b_sum < 1e-15:
            break
        b_dir = b_dir / b_sum

        # Check if direction is too similar to current (stagnation)
        cos_sim = np.dot(a, b_dir) / (np.linalg.norm(a) * np.linalg.norm(b_dir) + 1e-15)
        if cos_sim > 0.9999:
            stall_count += 1
            tol_frac *= 0.5
            if tol_frac < 1e-8:
                break
            continue
        stall_count = 0

        # Line search
        t_opt, M_new = line_search_convex(a, b_dir, n, max_evals=15)

        if t_opt > 1e-6 and M_new < best_M * (1 - 1e-12):
            a = (1 - t_opt) * a + t_opt * b_dir
            a = np.maximum(a, 0)
            a = a / a.sum()
            b = fast_convolve(a)
            M = b.max()
            if M < best_M:
                best_M = M
                best_a = a.copy()
                if verbose and it % 20 == 0:
                    print(f"    LP iter {it}: M={M:.8f}, t={t_opt:.4f}, active={len(active)}")
        else:
            # No improvement, tighten tolerance
            tol_frac *= 0.7
            if tol_frac < 1e-9:
                break

    return best_a, best_M


def soft_max_grad_step(a0, time_limit=60.0, T=50.0, lr=0.01, momentum=0.9, seed=42):
    """Soft-max smoothed gradient descent with momentum."""
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 0)
    a = a / a.sum()
    v = np.zeros(n)
    best_a = a.copy()
    best_M = fast_convolve(a).max()
    cur_T = T
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)

        # Softmax weights
        Tb = T * b
        Tb -= Tb.max()
        w = np.exp(Tb)
        w = w / w.sum()

        # Gradient: g[i] = sum_k w[k] * a[k-i] for valid range
        g = np.zeros(n)
        nz = np.where(w > 1e-14)[0]
        for k in nz:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += w[k] * a[k - idx]

        # Momentum step
        v = momentum * v - lr * g
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        M_new = fast_convolve(a_new).max()

        if M_new < best_M - 1e-14:
            best_M = M_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Adaptive adjustments
        if no_improve > 3000:
            a = np.maximum(best_a + rng.standard_normal(n) * 0.002, 0)
            a = a / a.sum()
            v = np.zeros(n)
            lr = max(lr * 0.8, 0.001)
            no_improve = 0
        elif no_improve > 1500:
            lr *= 0.95

        cur_T = min(cur_T * 1.005, 200.0)

    return best_a, best_M


def coordinate_polish(a0, time_limit=60.0):
    """Coordinate descent with multiple multipliers."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_convolve(a).max()

    multipliers = np.array([0.4, 0.6, 0.8, 0.9, 1.1, 1.25, 1.5, 2.0])

    improved = True
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_M
            old_val = a[j]
            if old_val < 1e-14:
                continue
            best_j_M = best_M
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 1.0:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0 or scale > 1e4:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                val = fast_convolve(a_trial).max()
                if val < best_j_M - 1e-14:
                    best_j_M = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = np.maximum(a, 0)
                a = a / a.sum()
                if best_j_M < best_M - 1e-14:
                    best_M = best_j_M
                    best_a = a.copy()
                    improved = True

    return best_a, best_M


def make_power_init(n, p=1.0):
    """Power-law shaped initialization."""
    x = np.arange(n, dtype=np.float64) + 1.0
    a = x ** p
    return a / a.sum()


def make_symmetric_v_init(n, p=1.0):
    """V-shaped (symmetric, minimum at center) initialization."""
    x = np.linspace(0, 1, n)
    a = np.abs(x - 0.5) ** p + 0.01
    return a / a.sum()


def make_bimodal_init(n, sigma=0.15):

    return a / a.sum()
