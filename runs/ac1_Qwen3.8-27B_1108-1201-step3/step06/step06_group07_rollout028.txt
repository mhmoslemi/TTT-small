    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using improved mirror descent with:
- Chirp-modulated initializations (closer to optimal flat-autocorrelation structure)
- Larger top-k subgradient for more accurate descent direction
- No forced symmetrization (symmetric seqs have lower bound 2, we're below 2)
- Multi-n exploration
- Pairwise mass-transfer local search
- Adaptive temperature and step size

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u * np.arange(1, n + 1) > (cssv - 1.0))
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, peak_indices, weights):
    """Compute weighted subgradient of max_k (a*a)_k."""
    n = len(a)
    g = np.zeros(n)
    for idx in range(len(peak_indices)):
        k = int(peak_indices[idx])
        w_k = weights[idx]
        if w_k < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        g[js:je + 1] += 2.0 * w_k * a[k - np.arange(js, je + 1)]
    return g


def mirror_descent_v2(a0, time_limit, top_k=150, temp_init=0.0005, lr_init=0.01,
                      lr_decay=0.99995, verbose=False, label=""):
    """
    Improved multiplicative weight mirror descent on the simplex.
    Uses larger top-k, adaptive temperature, and no symmetrization.
    """
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()

        # Find top peaks
        K = min(top_k, len(conv))
        if K < len(conv):
            peak_indices = np.argpartition(conv, -K)[-K:]
        else:
            peak_indices = np.arange(len(conv))

        # Softmax weights over peaks
        pv = conv[peak_indices]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peak_indices))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute subgradient
        g = compute_subgradient(a, peak_indices, w)

        # Backtracking line search
        best_lr = lr * 0.05
        best_trial_obj = M * 1.01
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05]:
            if trial_lr < 1e-10:
                continue
            log_update = -trial_lr * g
            log_update = np.clip(log_update, -40, 40)
            a_trial = a * np.exp(log_update)
            a_trial = np.maximum(a_trial, 1e-15)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial = a_trial / ss
            trial_conv = fast_convolve(a_trial)
            trial_obj = trial_conv.max()
            if trial_obj <= best_trial_obj:
                best_lr = trial_lr
                best_trial_obj = trial_obj
                if trial_obj < M * 0.998:
                    break

        # Apply update
        log_update = -best_lr * g
        log_update = np.clip(log_update, -40, 40)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        conv = fast_convolve(a)
        new_obj = conv.max()

        if new_obj < best_obj - 1e-14:
            best_obj = new_obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    {label} iter {it}: obj={best_obj:.10f}, lr={best_lr:.6f}, temp={temp:.6f}")
        else:
            no_improve += 1

        # Adapt temperature: gradually increase to include more peaks
        temp = min(temp * 1.003, 0.08)

        # Adapt learning rate
        lr = max(lr * lr_decay, 1e-9)
        if no_improve > 200:
            lr *= 0.3
            no_improve = 100

        # Restart if stuck
        if no_improve > 800:
            rng = np.random.default_rng(it * 7919 + 13)
            # "Push away" from peak-contributing positions
            k_max = int(np.argmax(conv))
            js = max(0, k_max - n + 1)
            je = min(n - 1, k_max)
            push = a.copy()
            push[js:je+1] *= 0.7
            push = np.maximum(push + rng.standard_normal(n) * 0.001, 1e-15)
            push = push / push.sum()
            a = push
            lr = lr_init * 0.2
            no_improve = 0
            temp = temp_init * 2

        if lr < 1e-10:
            break

    return best_a, best_obj


def additive_flow(a0, time_limit, top_k=150, temp_init=0.001, dt_init=0.005,
                  verbose=False, label=""):
    """
    Additive (rectilinear) flow: da = -dt * (g - mean(g)), projected to simplex.
    Different geometry than multiplicative, can find different minima.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    dt = dt_init
    temp = temp_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()

        K = min(top_k, len(conv))
        if K < len(conv):
            peak_indices = np.argpartition(conv, -K)[-K:]
        else:
            peak_indices = np.arange(len(conv))

        pv = conv[peak_indices]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peak_indices))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        g = compute_subgradient(a, peak_indices, w)
        g_centered = g - g.mean()

        # Backtracking
        best_dt = dt * 0.1
        best_trial_obj = M * 1.01
        for trial_dt in [dt, dt * 0.5, dt * 0.25, dt * 0.1]:
            if trial_dt < 1e-12:
                continue
            a_trial = a - trial_dt * g_centered
            a_trial = project_simplex(a_trial)
            trial_obj = fast_convolve(a_trial).max()
            if trial_obj <= best_trial_obj:
                best_dt = trial_dt
                best_trial_obj = trial_obj
                if trial_obj < M * 0.999:
                    break

        a = a - best_dt * g_centered
        a = project_simplex(a)

        conv = fast_convolve(a)
        new_obj = conv.max()

        if new_obj < best_obj - 1e-14:
            best_obj = new_obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    {label} iter {it}: obj={best_obj:.10f}, dt={best_dt:.6f}")
        else:
            no_improve += 1

        temp = min(temp * 1.003, 0.06)
        dt = max(dt * 0.9999, 1e-10)

        if no_improve > 300:
            rng = np.random.default_rng(it * 104729 + 3)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            dt = dt_init * 0.3
            no_improve = 0
            temp = temp_init

        if dt < 1e-11:
            break

    return best_a, best_obj


def coordinate_descent(a0, time_limit, verbose=False, label=""):
    """Coordinate descent: adjust one coordinate at a time."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        # Randomize coordinate order
        perm = np.random.permutation(n)

        for j in perm:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            if old_val < 1e-13:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in [0.6, 0.8, 0.95, 1.05, 1.2, 1.5, 2.0]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.8:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                trial = a.copy()
                trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                trial[mask] *= scale
                trial = np.maximum(trial, 0.0)
                ss = trial.sum()
                if ss < 1e-15:
                    continue
                trial = trial / ss
                val = fast_convolve(trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-12:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            break

    return best_a, best_obj


def pairwise_local_search(a0, time_limit, n_iter=5000, verbose=False, label=""):
    """
    Pairwise mass transfer: move mass from position i to position j.
    Effective at escaping local minima of gradient methods.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    # Precompute which positions contribute to the current peak
    conv = fast_convolve(a)
    peak_k = int(np.argmax(conv))

    it = 0
    while time.monotonic() - t0 < time_limit and it < n_iter:
        it += 1
        # Pick a random pair (i, j) where i has mass and j is "far" from peak contributors
        i = rng.integers(0, n)
        if a[i] < 1e-8:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # Try moving a fraction of a[i] to a[j]
        for frac in [0.1, 0.3, 0.5]:
            transfer = a[i] * frac
            if transfer < 1e-12:
                continue
            trial = a.copy()
            trial[i] -= transfer
            trial[j] += transfer
            trial = np.maximum(trial, 0.0)
            ss = trial.sum()
            if ss < 1e-10:
                continue
            trial = trial / ss
            obj = fast_convolve(trial).max()
            if obj < best_obj - 1e-14:
                a = trial
                best_obj = obj
                best_a = a.copy()
                if verbose and it % 500 == 0:
                    print(f"    {label} PL iter {it}: obj={best_obj:.10f}")
                break

    return best_a, best_obj


def make_gaussian(n, sigma_frac=0.3, center_frac=0.5):
    """Gaussian initialization."""
    x = np.arange(n) - n * center_frac
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_chirp_gaussian(n, sigma_frac=0.35, alpha=1.0, epsilon=0.4, center_frac=0.5):
    """
    Chirp-modulated Gaussian: has more uniform autocorrelation structure.
    The chirp component helps flatten the autocorrelation peak.
    """
    x = np.arange(n) - n * center_frac
    sigma = n * sigma_frac
    gauss = np.exp(-0.5 * (x / sigma) ** 2)
    # Quadratic phase (chirp)
    phase = 2.0 * np.pi * alpha * (np.arange(n) - n * center_frac) ** 2 / (n * n)
    mod = 1.0 + epsilon * np.cos(phase)
    a = gauss * np.maximum(mod, 0.0)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_triangle(n, width_frac=0.8, center_frac=0.5):
    """Triangle-shaped initialization."""
    x = np.abs(np.arange(n) / n - center_frac)
    a = np.maximum(1.0 - x / (width_frac / 2.0), 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_asymmetric_ramp(n, decay=0.5, center_frac=0.5):
    """Asymmetric ramp (helps break symmetry)."""
    x = np.arange(n) / n - center_frac
    a = np.exp(-n * decay * np.abs(x)) * (1.0 + 0.5 * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_two_bumps(n, c1=0.3, c2=0.7, sigma=0.12):
    """Two-bump initialization (asymmetric)."""
    x = np.arange(n) / n
    a = np.exp(-0.5 * ((x - c1) / sigma) ** 2) + 0.8 * np.exp(-0.5 * ((x - c2) / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_flat_top(n, width_frac=0.6, round_frac=0.1):
    """Flat-top with rounded edges."""
    x = np.abs(np.arange(n) / n - 0.5)
    half_w = width_frac / 2.0
    r = round_frac
    a = np.where(x < half_w - r, 1.0,
                 np.where(x < half_w, 0.5 * (1 + np.cos(np.pi * (x - (half_w - r)) / r)), 0.0))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
        return score

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # === Phase 1: Generate and evaluate diverse initializations ===
    print("=== Phase 1: Initializations ===")
    init_candidates = []

    n_values = [512, 640, 768, 896, 1024]
    if parent is not None:
        n_parent = len(parent)
        if n_parent not in n_values:
            n_values.insert(0, n_parent)

    for n in n_values:
        # Chirp-like (most promising for flat autocorrelation)
        for alpha in [0.5, 1.0, 2.0]:
            for eps in [0.3, 0.5, 0.7]:
                for sf in [0.25, 0.35, 0.45]:
                    a = make_chirp_gaussian(n, sf, alpha, eps)
                    init_candidates.append((a, f"chirp_a{alpha}_e{eps}_s{sf}_n{n}"))

        # Gaussian
        for sf in [0.2, 0.3, 0.4, 0.5]:
            for cf in [0.35, 0.5, 0.65]:
                a = make_gaussian(n, sf, cf)
                init_candidates.append((a, f"gauss_s{sf}_c{cf}_n{n}"))

        # Asymmetric ramp
        for dec in [0.3, 0.5, 0.8]:
            a = make_asymmetric_ramp(n, dec)
            init_candidates.append((a, f"ramp_d{dec}_n{n}"))

        # Two bumps
        a = make_two_bumps(n)
        init_candidates.append((a, f"2bump_n{n}"))

        # Flat top
        a = make_flat_top(n)
        init_candidates.append((a, f"flat_n{n}"))

        # Triangle
        a = make_triangle(n, 0.7)
        init_candidates.append((a, f"tri_n{n}"))

    if parent is not None:
        init_candidates.append((parent, "parent"))
        # Asymmetric perturbation of parent
        noise = rng.standard_normal(len(parent)) * 0.05
        p_perturbed = np.maximum(parent + noise, 0)
        p_perturbed = p_perturbed / p_perturbed.sum()
        init_candidates.append((p_perturbed, "parent_pert"))
        # Shifted parent
        p_shift = np.roll(parent, n_parent // 10)
        p_shift = np.maximum(p_shift, 0)
        p_shift = p_shift / p_shift.sum()
        init_candidates.append((p_shift, "parent_shift"))

    # Evaluate all initializations quickly
    best_init = None
    best_init_obj = float('inf')
    for a_init, label in init_candidates:
        s = a_init.sum()
        if s < 1e-10:
            continue
        a_norm = a_init / s
        conv = fast_convolve(a_norm)
        c_obj = 2 * len(a_norm) * conv.max()
        if c_obj < best_init_obj:
            best_init_obj = c_obj
            best_init = a_norm
            best_init_label = label

    print(f"  Best init: {best_init_label} with obj={best_init_obj:.6f} (n={len(best_init)})")
    try_seq(best_init, "best_init")

    # Also try the second and third best inits
    # (quick evaluation, no optimization)
    scored_inits = []
    for a_init, label in init_candidates:
        s = a_init.sum()
        if s < 1e-10:
            continue
        a_norm = a_init / s
        conv = fast_convolve(a_norm)
        c_obj = 2 * len(a_norm) * conv.max()
        scored_inits.append((c_obj, a_norm, label))
    scored_inits.sort(key=lambda x: x[0])

    # === Phase 2: Main optimization on top initializations ===
    print("\n=== Phase 2: Main Optimization ===")

    # Run mirror descent on the best init
    remaining = deadline - time.monotonic()
    t_md = remaining * 0.30
    print(f"  MD on best init: {t_md:.0f}s")
    a_opt, obj = mirror_descent_v2(best_init, time_limit=t_md, top_k=150,
                                   temp_init=0.0004, lr_init=0.01,
                                   verbose=True, label="MD_best")
    try_seq(a_opt, "MD_best")

    # Run additive flow on the result
    remaining = deadline - time.monotonic()
    t_af = remaining * 0.15
    print(f"  Additive flow: {t_af:.0f}s")
    if best_seq is not None:
        a_opt, obj = additive_flow(best_seq, time_limit=t_af, top_k=150,
                                   temp_init=0.001, dt_init=0.003,
                                   verbose=True, label="AF")
        try_seq(a_opt, "AF")

    # Run mirror descent on second-best init (if different n or very different)
    remaining = deadline - time.monotonic()
    if remaining > 60:
        # Find a diverse second-best init
        second_init = None
        for obj2, a2, label2 in scored_inits[1:]:
            if len(a2) != len(best_init) or obj2 - best_init_obj > 0.1:
                second_init = a2
                break
        if second_init is None and len(scored_inits) > 1:
            second_init = scored_inits[1][1]

        if second_init is not None:
            t_md2 = min(remaining * 0.20, 200)
            print(f"  MD on diverse init: {t_md2:.0f}s")
            a_opt, obj = mirror_descent_v2(second_init, time_limit=t_md2, top_k=150,
                                           temp_init=0.0006, lr_init=0.008,
                                           verbose=True, label="MD_div")
            try_seq(a_opt, "MD_div")

    # === Phase 3: Pairwise local search ===
    print("\n=== Phase 3: Pairwise Local Search ===")
    remaining = deadline - time.monotonic()
    t_pl = min(remaining * 0.15, 120)
    print(f"  Time: {t_pl:.0f}s")
    if best_seq is not None and t_pl > 10:
        a_opt, obj = pairwise_local_search(best_seq, time_limit=t_pl,
                                           n_iter=10000, verbose=True, label="PL")
        try_seq(a_opt, "PL")

    # === Phase 4: Coordinate descent polish ===
    print("\n=== Phase 4: Coordinate Descent ===")
    remaining = deadline - time.monotonic()
    t_cd = min(remaining * 0.15, 120)
    print(f"  Time: {t_cd:.0f}s")
    if best_seq is not None and t_cd > 10:
        a_opt, obj = coordinate_descent(best_seq, time_limit=t_cd,
                                        verbose=True, label="CD")
        try_seq(a_opt, "CD")

    # === Phase 5: More mirror descent with different params ===
    print("\n=== Phase 5: MD Fine-tune ===")
    remaining = deadline - time.monotonic()
    t_md3 = min(remaining * 0.20, 200)
    if best_seq is not None and t_md3 > 30:
        # Use smaller top-k for finer peak targeting
        a_opt, obj = mirror_descent_v2(best_seq, time_limit=t_md3 * 0.6, top_k=50,
                                       temp_init=0.0001, lr_init=0.005,
                                       verbose=True, label="MD_fine")
        try_seq(a_opt, "MD_fine")

        # Larger top-k for broader optimization
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_opt, obj = mirror_descent_v2(best_seq, time_limit=min(remaining * 0.5, 100),
                                           top_k=300, temp_init=0.001, lr_init=0.004,
                                           verbose=True, label="MD_broad")
            try_seq(a_opt, "MD_broad")

    # === Phase 6: Additive flow on best ===
    print("\n=== Phase 6: Additive Flow 2 ===")
    remaining = deadline - time.monotonic()
    t_af2 = min(remaining * 0.15, 100)
    if best_seq is not None and t_af2 > 20:
        a_opt, obj = additive_flow(best_seq, time_limit=t_af2, top_k=200,
                                   temp_init=0.0005, dt_init=0.002,
                                   verbose=True, label="AF2")
        try_seq(a_opt, "AF2")

    # === Phase 7: Final local search + CD ===
    print("\n=== Phase 7: Final Polish ===")
    remaining = deadline - time.monotonic()
    if best_seq is not None and remaining > 20:
        t_final = min(remaining - 15, 80)
        # Quick pairwise search
        t_pl2 = t_final * 0.4
        a_opt, obj = pairwise_local_search(best_seq, time_limit=t_pl2,
                                           n_iter=5000, verbose=True, label="PL_final")
        try_seq(a_opt, "PL_final")

        # Very quick CD
        remaining = deadline - time.monotonic()
        if remaining > 10:
            t_cd2 = min(remaining - 5, 40)
            a_opt, obj = coordinate_descent(best_seq, time_limit=t_cd2,
                                            verbose=True, label="CD_final")
            try_seq(a_opt, "CD_final")

        # Try a few random perturbations of the best
        remaining = deadline - time.monotonic()
        if remaining > 5:
            for attempt in range(5):
                if time.monotonic() > deadline - 3:
                    break
                noise = rng.standard_normal(len(best_seq)) * 0.005
                perturbed = np.maximum(best_seq + noise, 0)
                perturbed = perturbed / perturbed.sum()
                try_seq(perturbed, f"perturbed_{attempt}")

    # Fallback
    if best_seq is None:
        best_seq = make_gaussian(768, 0.35)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
