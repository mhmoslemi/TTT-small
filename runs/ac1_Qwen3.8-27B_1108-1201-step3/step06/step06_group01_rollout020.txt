from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_subgradient(a, conv_val, top_k=20, temperature=None):
    """
    Compute a subgradient of f(a) = max_k conv(a,a)[k].
    Uses softmax weighting over the top-k peaks.
    Returns (gradient, active_lags, weights).
    """
    n = len(a)
    k_actual = min(top_k, len(conv_val))

    # Exclude the very first and last (they are just a[0]^2 and a[-1]^2)
    # Actually, let's include all for correctness
    idx = np.argpartition(conv_val, -k_actual)[-k_actual:]
    
    # Softmax weighting
    if temperature is None:
        # Adaptive temperature: proportional to the spread of top peaks
        peak_vals = conv_val[idx]
        max_val = peak_vals.max()
        min_top = peak_vals.min()
        spread = max_val - min_top
        if spread < 1e-15:
            temperature = max_val * 0.01
        else:
            temperature = spread * 0.3
    temperature = max(temperature, 1e-15)

    w = np.exp(scaled)

def exponentiated_gradient_step(a, g, eta):
    """One step of mirror descent with negative entropy."""
    # To maintain sum(a) = 1, subtract a weighted mean
    # a_new = a * exp(-eta * (g - <g, a>)) normalized
    shift = np.dot(g, a)
    log_ratio = -eta * (g - shift)
    # For numerical stability
    log_ratio = log_ratio - log_ratio.max()
    ratio = np.exp(log_ratio)
    a_new = a * ratio
    s = a_new.sum()
    if s < 1e-300:
        return a.copy()
    a_new = a_new / s
    return a_new

        conv_z = fast_convolve(z)
        g, _, _ = compute_subgradient(z, conv_z, top_k=top_k)
        
        # FISTA update: a_new = prox_{eta * f}(z - eta * g)
        # prox is the mirror descent step (for negative entropy)
        a_prev = a.copy()
        a = exponentiated_gradient_step(z, g, eta)
        
        # FISTA momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        z = a + ((t_k - 1.0) / t_new) * (a - a_prev)
        # Project z onto simplex (approximately)
        z = np.maximum(z, 0)
        z_sum = z.sum()
        if z_sum < 1e-15:
            z = a.copy()
        else:
            z = z / z_sum
        t_k = t_new
        
        # Evaluate
        f_val = fast_convolve(a).max()
        if f_val < best_f - 1e-15:
            best_f = f_val
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        # Adaptive step size
        if stagnant > 100:
            eta *= 0.95
            stagnant = 0
        elif stagnant > 10:
            eta = min(eta * 1.02, 1.0 / np.sqrt(n))
        
        # Random restart if completely stuck
        if stagnant > max_stagnant:
            rng = np.random.default_rng(hash((iters, n)) % 2**31)
            a = best_a + rng.standard_normal(n) * 0.01
            a = np.maximum(a, 0)
            a = a / a.sum()
            z = a.copy()
            t_k = 1.0
            eta = 1.0 / (2.0 * np.sqrt(n))
            stagnant = 0
            max_stagnant = int(max_stagnant * 0.8)
        
        if verbose and iters % 200 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_f
            print(f"    FISTA iter={iters} ({elapsed:.0f}s): f={best_f:.8f} score={score:.6f} eta={eta:.6f}")
    
    return best_a, 2.0 * n * best_f


def exponentiated_gradient_optimize(a0, time_limit, top_k=30, verbose=True):
    """
    Exponentiated gradient (mirror descent) with backtracking.
    More robust than FISTA but potentially slower.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_f = fast_convolve(a).max()
    eta = 1.0 / np.sqrt(n)
    t0 = time.monotonic()
    iters = 0
    stagnant = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        
        conv_a = fast_convolve(a)
        f_val = conv_a.max()
        
        if f_val < best_f - 1e-15:
            best_f = f_val
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        # Compute subgradient
        g, _, _ = compute_subgradient(a, conv_a, top_k=top_k)
        
        # Mirror descent step
        a_new = exponentiated_gradient_step(a, g, eta)
        
        # Check improvement
        f_new = fast_convolve(a_new).max()
        if f_new < f_val:
            a = a_new
        else:
            # Backtrack
            eta *= 0.8
            a_new = exponentiated_gradient_step(a, g, eta)
            f_new = fast_convolve(a_new).max()
            if f_new < f_val:
                a = a_new
            else:
                a = a_new  # Accept anyway (subgradient method doesn't require monotone decrease)
        
        # Periodic restart
        if stagnant > 300:
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 2**31)
            a = best_a + rng.standard_normal(n) * 0.005
            a = np.maximum(a, 0)
            a = a / a.sum()
            eta = 1.0 / np.sqrt(n)
            stagnant = 0
        
        if verbose and iters % 500 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_f
            print(f"    EG iter={iters} ({elapsed:.0f}s): f={best_f:.8f} score={score:.6f} eta={eta:.6f}")
    
    return best_a, 2.0 * n * best_f


def peak_equalize(a0, time_limit, top_k=50, verbose=True):
    """
    Peak equalization: iteratively try to make the top peaks equal.
    Uses a "water-filling" approach: identify the highest peak, reduce it,
    and distribute the reduction to other positions.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_f = fast_convolve(a).max()
    t0 = time.monotonic()
    iters = 0
    stagnant = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv_a = fast_convolve(a)
        f_val = conv_a.max()
        
        if f_val < best_f - 1e-15:
            best_f = f_val
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        # Find the top peaks
        k_actual = min(top_k, len(conv_a))
        peaks = np.argpartition(conv_a, -k_actual)[-k_actual:]
        peak_vals = conv_a[peaks]
        
        # Target: reduce the max peak to the average of top peaks
        target = peak_vals.mean()
        if f_val <= target + 1e-15:
            if stagnant > 50:
                break
            continue
        
        # For the highest peak, compute the gradient and take a small step
        highest_k = int(np.argmax(conv_a))
        # Gradient of a^T Q_k a at a is 2 * Q_k a
        js = max(0, highest_k - n + 1)
        je = min(n - 1, highest_k)
        grad = np.zeros(n)
        segment = a[highest_k - je : highest_k - js + 1][::-1]
        grad[js:je+1] = 2.0 * segment
        
        # Step in negative gradient direction (projected)
        step = 0.01 / np.linalg.norm(grad)
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        f_new = fast_convolve(a_new).max()
        if f_new < f_val:
            a = a_new
        else:
            # Try smaller step
            step *= 0.1
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            f_new = fast_convolve(a_new).max()
            if f_new < f_val:
                a = a_new
            else:
                # Accept anyway for exploration
                a = a_new
        
        if stagnant > 500:
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 2**31)
            a = best_a + rng.standard_normal(n) * 0.003
            a = np.maximum(a, 0)
            a = a / a.sum()
            stagnant = 0
            if iters > 10000:
                break
        
        if verbose and iters % 500 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_f
            print(f"    PEAK iter={iters} ({elapsed:.0f}s): f={best_f:.8f} score={score:.6f}")
    
    return best_a, 2.0 * n * best_f


def lbgfs_polish(a0, time_limit, top_k=50, verbose=True):
    """
    L-BFGS polishing using the exact (nonsmooth) subgradient.
    Uses scipy's L-BFGS-B with a smoothed objective.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Use log-parameterization to maintain non-negativity
    # a = softmax(z)
    z = np.log(np.maximum(a, 1e-15))
    
    def obj_and_grad(z_val, T=100):
        """Softmax-smoothed objective and gradient."""
        a_val = np.exp(z_val - z_val.max())
        a_val = a_val / a_val.sum()
        conv_val = fast_convolve(a_val)
        
        # Soft-max approximation of max
        scaled = conv_val / T
        scaled = scaled - scaled.max()
        exp_scaled = np.exp(scaled)
        log_sum_exp = T * (scaled.max() + np.log(exp_scaled.sum()))
        obj = 2.0 * n * log_sum_exp
        
        # Gradient via chain rule
        w = exp_scaled / exp_scaled.sum()
        # d(obj)/d(a_j) = 2n * sum_k w_k * d(conv_k)/d(a_j)
        # d(conv_k)/d(a_j) = a[k-j] + a[j-k] (by symmetry, and accounting for both terms)
        # Actually for auto-correlation: d(conv_k)/d(a_j) = a[k-j] (accounting for valid range)
        # More precisely: conv_k = sum_i a[i]*a[k-i], so d(conv_k)/d(a_j) = a[k-j] + a[j-k]
        # But since conv_k = conv_{2n-2-k} and we're dealing with auto-correlation...
        # Let's just use: d(conv_k)/d(a_j) = a[k-j] if 0 ≤ k-j ≤ n-1, else 0
        # (This is correct because conv_k = sum_i a[i]*a[k-i], partial wrt a_j gives a[k-j])
        
        grad_a = np.zeros(n)
        k_actual = min(top_k, len(conv_val))
        top_peaks = np.argpartition(conv_val, -k_actual)[-k_actual:]
        for k in top_peaks:
            k = int(k)
            wk = w[k]
            if wk < 1e-13:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            segment = a_val[k - je : k - js + 1][::-1]
            grad_a[js:je+1] += 2.0 * n * wk * segment
        
        # Chain rule through softmax
        # d(softmax(z)_i)/d(z_j) = a_i * (delta_ij - a_j)
        grad_z = a_val * (grad_a - np.dot(grad_a, a_val))
        
        return obj, grad_z
    
    # Run L-BFGS
    stopped = [False]
    
    def func(z_val):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        o, _ = obj_and_grad(z_val)
        return o
    
    def jac(z_val):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z_val)
        _, g = obj_and_grad(z_val)
        return g
    
    try:
        res = minimize(
            fun=func, x0=z, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': 5000, 'ftol': 1e-14, 'gtol': 1e-12,
                     'maxfun': 15000, 'disp': False}
        )
        a_out = np.exp(res.x - res.x.max())
        a_out = a_out / a_out.sum()
        a_out = np.maximum(a_out, 0)
        score = 2.0 * n * fast_convolve(a_out).max()
        if score < best_score:
            best_score = score
            best_a = a_out
    except Exception:
        pass
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate and return updated best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score
