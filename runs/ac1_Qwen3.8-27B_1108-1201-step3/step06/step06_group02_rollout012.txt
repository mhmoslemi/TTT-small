"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Fast projected subgradient descent with log-sum-exp smoothing
- Peak-targeted local search with pairwise mass transfer
- Multi-length exploration with interpolation
- Perturbation-based restarts for escaping local minima
- Final coordinate descent polish

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_gradient(a, weights, N_cache=None):
    """Compute smoothed gradient: grad_j = 2 * sum_k weights[k] * a[k-j]."""
    n = len(a)
    if N_cache is None:
        N_cache = 1 << (3 * n - 1).bit_length()
    W = np.fft.rfft(weights, N_cache)
    A = np.fft.rfft(a, N_cache)
    grad = 2.0 * np.fft.irfft(W * np.conj(A), N_cache)[:n]
    return grad


def compute_weights(conv, T, top_k=None):
    """Compute softmax weights over convolution values."""
    if top_k is not None and top_k < len(conv):
        idx = np.argpartition(conv, -top_k)[-top_k:]
        weights = np.zeros(len(conv))
        vals = conv[idx]
        scaled = vals * T
        scaled -= scaled.max()
        exp_vals = np.exp(scaled)
        weights[idx] = exp_vals / exp_vals.sum()
    else:
        scaled = conv * T
        scaled -= scaled.max()
        exp_vals = np.exp(scaled)
        weights = exp_vals / exp_vals.sum()
    return weights


def subgradient_descent(a0, time_limit, T_init=30.0, T_end=200.0,

    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    N_fft = 1 << (3 * n - 1).bit_length()
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    
    lr = lr_init
    no_improve = 0
    it = 0
    total_iters = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        total_iters += 1
        
        # Compute weights and gradient
        T = T_init + (T_end - T_init) * min(1.0, it / max(1, int(time_limit / 0.05)))
        weights = compute_weights(conv, T, top_k)
        grad = compute_smoothed_gradient(a, weights, N_fft)
        
        # Step
        a_new = a - lr * grad
        a_new = project_simplex(a_new)
        
        # Evaluate
        new_conv = fast_convolve(a_new)
        new_obj = new_conv.max()
        
        if new_obj < best_obj - 1e-15:
            best_obj = new_obj
            best_a = a_new.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    SGD iter {it}: obj={best_obj:.8f}, T={T:.1f}, lr={lr:.6f}")
        else:
            a_new2 = a - lr * 0.5 * grad
            a_new2 = project_simplex(a_new2)
            new_conv2 = fast_convolve(a_new2)
            new_obj2 = new_conv2.max()
            if new_obj2 < best_obj - 1e-15:
                best_obj = new_obj2
                best_a = a_new2.copy()
                a = a_new2
                lr *= 0.7
                no_improve = 0
            else:
                a = a_new  # Still move (might help escape flat regions)
                lr *= 0.85
                no_improve += 1
        
        conv = fast_convolve(a)
        
        if no_improve > 50:

            rng = np.random.default_rng(it * 7 + 3)
            noise = rng.standard_normal(n) * 0.005
            a = project_simplex(best_a + noise)
            conv = fast_convolve(a)
            lr = lr_init * 0.8
            no_improve = 0
            if verbose:
                print(f"    SGD restart at iter {it}, resetting lr")
    
    return best_a, best_obj


def peak_shave(a0, time_limit, verbose=False):

    a = np.maximum(a0.astype(np.float64), 0.0)

            total = products.sum()
            if total < 1e-15:
                continue
            
            # Find the pair with maximum product

            src = int(contrib[max_idx_in])
            partner = int(pk - src)

            non_contrib = np.ones(n, dtype=bool)
            non_contrib[contrib] = False
            if non_contrib.sum() < 1:

                non_contrib[src] = False
            
            if non_contrib.sum() == 0:
                no_improve += 1
                continue
            
            # Try different transfer fractions
            best_trial_obj = M
            best_trial = None

                delta = a[src] * frac

                trial[src] -= delta

                nc_sum = nc_vals.sum()

                    # Uniform distribution
                    trial[non_contrib] += delta / non_contrib.sum()
                else:

                trial = np.maximum(trial, 0.0)

                trial = trial / ts

                    improved = True

def coordinate_descent(a0, time_limit, verbose=False):
    """Pairwise swap coordinate descent."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if no_improve > 200:
            break
        
        improved = False
        # Try random pairs of coordinates
        rng = np.random.default_rng(it + 1)
        n_attempts = 50
        
        for _ in range(n_attempts):
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            
            # Try transferring fraction of a[i] to a[j]
            for frac in [0.1, 0.3, 0.5]:
                delta = a[i] * frac
                if delta < 1e-15:
                    continue
                trial = a.copy()
                trial[i] -= delta
                trial[j] += delta
                trial = np.maximum(trial, 0.0)
                ts = trial.sum()
                if ts < 1e-15:
                    continue
                trial = trial / ts
                obj = fast_convolve(trial).max()
                if obj < best_obj - 1e-14:
                    a = trial
                    best_obj = obj
                    best_a = a.copy()
                    no_improve = 0
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            no_improve += 1
    
    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian-like initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_cosine_mod(n, freq, amp):
    """Cosine-modulated uniform sequence."""
    x = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_multi_cosine(n, n_freqs=5, max_amp=0.3, seed=0):
    """Multi-frequency cosine-modulated sequence."""
    rng = np.random.default_rng(seed)
    a = np.ones(n)
    for m in range(1, n_freqs + 1):
        amp = rng.uniform(0.05, max_amp)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * m * np.arange(n) / n + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_tapered(n, taper_power=2):
    """Tapered (raised cosine) sequence."""
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x) ** taper_power
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def objective_normalized(a):
    """Compute 2*n*max(conv)/sum^2 for normalized a (sum=1)."""
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max()

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_gradient(a, weights, N_fft):
    """Compute smoothed gradient: grad_j = 2 * sum_k weights[k] * a[k-j]."""
    n = len(a)
    W = np.fft.rfft(weights, N_fft)
    A = np.fft.rfft(a, N_fft)
    grad = 2.0 * np.fft.irfft(W * np.conj(A), N_fft)[:n]
    return grad


def compute_weights(conv, T, top_k):
    """Compute softmax weights over convolution values."""
    if top_k is not None and top_k < len(conv) and top_k > 0:
        idx = np.argpartition(conv, -top_k)[-top_k:]
        weights = np.zeros(len(conv))
        vals = conv[idx]
        scaled = vals * T
        scaled -= scaled.max()
        exp_vals = np.exp(scaled)
        weights[idx] = exp_vals / exp_vals.sum()
    else:
        scaled = conv * T
        scaled -= scaled.max()
        exp_vals = np.exp(scaled)
        weights = exp_vals / exp_vals.sum()
    return weights


def multiplicative_descent(a0, time_limit, T_init=50.0, T_end=500.0,
                           lr_init=0.01, top_k=20, verbose=False, seed=0):
    """Mirror descent (multiplicative weights) with log-sum-exp smoothing."""
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    N_fft = 1 << (3 * n - 1).bit_length()

    conv = fast_convolve(a)
    best_obj = conv.max()

    lr = lr_init

        grad = compute_smoothed_gradient(a, weights, N_fft)

        # Multiplicative update

        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        new_obj = new_conv.max()

        if new_obj < best_obj - 1e-15:
            best_obj = new_obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            lr *= 1.01  # Slightly increase lr when improving

                a = a_half

    beta = 0.9

    while time.monotonic() - t0 < time_limit:

        weights = compute_weights(conv, T, top_k)
        grad = compute_smoothed_gradient(a, weights, N_fft)

        a_new = a - lr * v

        new_conv = fast_convolve(a_new)

def peak_shave(a0, time_limit, verbose=False, seed=0):
    """Targeted peak reduction via mass transfer from peak contributors."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit and no_improve < 200:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()

        # Get top peaks
        n_peaks = min(5, 2 * n - 1)
        peak_indices = np.argsort(conv)[-n_peaks:][::-1]

        improved = False
        for pk in peak_indices:
            pk = int(pk)

            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            contrib = np.arange(js, je + 1)
            if len(contrib) == 0:
                continue

            products = a[contrib] * a[pk - contrib]
            total = products.sum()
            if total < 1e-15:
                continue

            max_idx_in = np.argmax(products)
            src = int(contrib[max_idx_in])
            partner = int(pk - src)

            # Non-contributing positions
            non_contrib = np.ones(n, dtype=bool)
            non_contrib[contrib] = False
            if non_contrib.sum() < 1:
                non_contrib[:] = True
                non_contrib[src] = False

            if non_contrib.sum() == 0:
                no_improve += 1
                break

            best_trial_obj = M
            best_trial = None

            for frac in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
                delta = a[src] * frac
                if delta < 1e-15:
                    continue
                trial = a.copy()
                trial[src] -= delta
                nc_vals = trial[non_contrib]
                nc_sum = nc_vals.sum()
                if nc_sum < 1e-15:
                    trial[non_contrib] += delta / non_contrib.sum()
                else:
                    trial[non_contrib] += delta * nc_vals / nc_sum
                trial = np.maximum(trial, 0.0)
                ts = trial.sum()
                if ts < 1e-15:
                    continue
                trial = trial / ts

                obj = fast_convolve(trial).max()
                if obj < best_trial_obj - 1e-14:
                    best_trial_obj = obj
                    best_trial = trial

            if best_trial is not None:
                a = best_trial
                if best_trial_obj < best_obj - 1e-14:
                    best_obj = best_trial_obj
                    best_a = a.copy()
                    no_improve = 0
                    improved = True
                    if verbose and it % 50 == 0:
                        print(f"    PeakShave iter {it}: obj={best_obj:.8f}")
                break

        if not improved:
            no_improve += 1
            if no_improve % 30 == 0:
                a = project_simplex(best_a + rng.standard_normal(n) * 0.003)

    return best_a, best_obj


def coordinate_descent(a0, time_limit, verbose=False, seed=0):

    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian-like initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_tapered(n, power=2):
    """Raised cosine tapered sequence."""
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x) ** power
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_multi_cosine(n, n_freqs=5, max_amp=0.3, seed=0):
    """Multi-frequency cosine-modulated sequence."""
    rng = np.random.default_rng(seed)
    a = np.ones(n)
    for m in range(1, n_freqs + 1):
        amp = rng.uniform(0.05, max_amp)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * m * np.arange(n) / n + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def objective_normalized(a):
    """Compute 2*n*max(conv)/sum^2 for normalized a (sum=1)."""
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max()

                best_score = sc
                best_seq = a_opt.copy()
                print(f"  NEW BEST [Shave]: {sc:.6f}")

    # === Phase 4: Second Multiplicative Descent (sharper) ===
    print("\n=== Phase 4: Sharper MD ===")
    if best_seq is not None:
        t_alloc = min(300, deadline - time.monotonic())
        if t_alloc > 30:
            print(f"  n={len(best_seq)}, time={t_alloc:.0f}s")
            a_opt, obj = multiplicative_descent(best_seq, time_limit=t_alloc,
                                                T_init=200, T_end=2000,
                                                lr_init=0.003, top_k=5,
                                                verbose=True, seed=seed + 3)
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  NEW BEST [MD2]: {sc:.6f}")

    # === Phase 5: Multi-length exploration ===
    print("\n=== Phase 5: Multi-length Exploration ===")
    if best_seq is not None:
        base_n = len(best_seq)
        target_lengths = [1024, 896, 640, 512, 1280]
        for target_n in target_lengths:
            if target_n == base_n:
                continue
            rem = deadline - time.monotonic()
            if rem < 80:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            sc = float(evaluate_sequence(f_interp.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = f_interp.copy()
                print(f"  NEW BEST [interp_n{target_n}]: {sc:.6f}")

            # Optimize if promising
            if sc < best_score + 0.15:
                t_alloc = min(150, rem - 30)
                if t_alloc > 20:
                    print(f"  Optimizing n={target_n}, time={t_alloc:.0f}s")
                    a_opt, obj = multiplicative_descent(f_interp, time_limit=t_alloc,
                                                        T_init=50, T_end=300,
                                                        lr_init=0.008, top_k=15,
                                                        verbose=False, seed=seed + target_n)
                    sc2 = float(evaluate_sequence(a_opt.tolist()))
                    if np.isfinite(sc2) and sc2 < best_score:
                        best_score = sc2
                        best_seq = a_opt.copy()
                        print(f"  NEW BEST [opt_n{target_n}]: {sc2:.6f}")

    # === Phase 6: Perturbation-based exploration ===
    print("\n=== Phase 6: Perturbation Exploration ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rng = np.random.default_rng(seed + 100)

        perturbations = []
        # Shifts
        for shift in [1, 2, 3, 5, 10]:
            perturbations.append((f'shift{shift}', np.roll(best_seq, shift)))
        # Flip
        perturbations.append(('flip', best_seq[::-1]))
        # Blend with uniform
        for mix in [0.05, 0.1, 0.15, 0.2, 0.3]:
            perturbations.append((f'uni{mix}', (1 - mix) * best_seq + mix * (np.ones(n_best) / n_best)))
        # Noise
        for amp in [0.005, 0.01, 0.02, 0.03]:
            noise = rng.standard_normal(n_best) * amp
            perturbations.append((f'noise{amp}', np.maximum(best_seq + noise, 0)))
        # Taper blend
        taper = make_tapered(n_best)
        for mix in [0.1, 0.2, 0.3, 0.5]:
            perturbations.append((f'taper{mix}', (1 - mix) * best_seq + mix * taper))

        for label, f_pert in perturbations:
            rem = deadline - time.monotonic()
            if rem < 50:
                break
            f_pert = project_simplex(f_pert)
            t_alloc = min(80, rem - 20)
            if t_alloc < 15:
                break
            print(f"  Perturbation [{label}]: time={t_alloc:.0f}s")
            a_opt, obj = multiplicative_descent(f_pert, time_limit=t_alloc,
                                                T_init=50, T_end=300,
                                                lr_init=0.008, top_k=15,
                                                verbose=False, seed=seed + hash(label) % 10000)
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  NEW BEST [pert_{label}]: {sc:.6f}")

    # === Phase 7: Coordinate Descent Polish ===
    print("\n=== Phase 7: Coordinate Descent ===")
    if best_seq is not None:
        t_alloc = min(200, deadline - time.monotonic())
        if t_alloc > 20:
            print(f"  n={len(best_seq)}, time={t_alloc:.0f}s")
            a_opt, obj = coordinate_descent(best_seq, time_limit=t_alloc,
                                            verbose=True, seed=seed + 200)
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  NEW BEST [CD]: {sc:.6f}")

    # === Phase 8: Final MD polish ===
    print("\n=== Phase 8: Final Polish ===")
    if best_seq is not None:
        t_alloc = min(150, deadline - time.monotonic())
        if t_alloc > 15:
            print(f"  n={len(best_seq)}, time={t_alloc:.0f}s")
            a_opt, obj = multiplicative_descent(best_seq, time_limit=t_alloc,
                                                T_init=500, T_end=5000,
                                                lr_init=0.002, top_k=3,
                                                verbose=True, seed=seed + 300)
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  NEW BEST [FinalMD]: {sc:.6f}")

    # === Phase 9: Final peak shave ===
    if best_seq is not None:
        t_alloc = min(60, deadline - time.monotonic())
        if t_alloc > 10:
            print(f"\n=== Phase 9: Final Peak Shave ===")
            a_opt, obj = peak_shave(best_seq, time_limit=t_alloc,
                                    verbose=True, seed=seed + 400)
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  NEW BEST [FinalShave]: {sc:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
