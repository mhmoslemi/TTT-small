Let me parameterize differently: `eta = alpha / ||grad||_inf` where `alpha` is the desired "step size" in the updated direction. With `alpha = 0.1`, each component changes by at most a factor of `exp(0.1) ≈ 1.1`.

**Active Set Method:**
1. Compute `conv(f,f)`.
2. Identify the "active" positions where `conv_k` is close to the max.
3. For each active position `k`, the contribution is `sum_{i+j=k} f_i f_j`.
4. To reduce the max, we want to reduce the contributions at all active positions simultaneously.
5. This is a small optimization problem: adjust `f` to minimize `max_k conv_k` locally.

This is essentially what the LP-FW does, but I can make it more targeted.

**Another idea: "Flattening" approach**
1. Compute `conv(f,f)`.
2. The "excess" at each position is `conv_k - target` where `target` is the current max or a lower target.
3. For positions with positive excess, we need to reduce the convolution.
4. The gradient of the excess at position `k` w.r.t. `f_i` is `2 * f[k-i]`.
5. Combine the gradients from all excess positions (weighted by the excess).
6. Take a step in the negative combined gradient direction.

This is a weighted subgradient method, which is essentially what the mirror descent does.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
(1) Mirror descent (multiplicative weights) with adaptive step,
(2) LP-based Frank-Wolfe with adaptive threshold tightening,
(3) Targeted 2-variable pair optimization at peak positions,

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_objective(f):
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    return 2.0 * n * M, conv, M


def mirror_descent_optimize(f0, time_limit, verbose=False):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0
    alpha = 0.1  # step size parameter
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        # Find top peaks
        n_peaks = min(5, 2 * n - 1)
        peak_indices = np.argsort(conv)[-n_peaks:]
        
        # Weighted subgradient: weight each peak by how much it exceeds the mean
        mean_conv = 1.0 / (2 * n - 1)  # average convolution entry when sum(f)=1
        weights = np.zeros(2 * n - 1)
        for pk in peak_indices:
            weights[pk] = max(0, conv[pk] - mean_conv)
        if weights.sum() < 1e-15:
            weights[peak_indices[0]] = 1.0
        weights /= weights.sum()
        
        # Compute combined subgradient
        grad = np.zeros(n)
        for pk in peak_indices:
            w = weights[pk]
            js = max(0, int(pk) - n + 1)
            je = min(n - 1, int(pk))
            idx = np.arange(js, je + 1)
            grad[idx] += w * 2.0 * f[int(pk) - idx]
        
        # Mirror descent update
        grad_max = np.abs(grad).max()
        if grad_max < 1e-15:
            break
        eta = alpha / grad_max
        f_new = f * np.exp(-eta * grad)
        f_new = np.maximum(f_new, 0)
        ss = f_new.sum()
        if ss < 1e-15:
            break
        f_new = f_new / ss
        
        # Check improvement
        conv_new = fast_convolve(f_new)
        obj_new = 2.0 * n * conv_new.max()
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    MD iter {it}: obj={best_obj:.8f}")
        else:
            no_improve += 1
            # Still update f (might help escape)
            f = f_new
        
        # Adaptive alpha
        if no_improve > 50:
            alpha *= 0.9
        elif no_improve == 0:
            alpha = min(alpha * 1.05, 0.5)
        
        if no_improve > 200:
            # Restart with perturbation
            rng = np.random.default_rng(it * 7 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            no_improve = 0
            alpha = 0.15
            if verbose:
                print(f"    MD restart at iter {it}")
        
        if no_improve > 500:
            break

    return best_f, best_obj


def lp_fw_optimize(f0, time_limit, verbose=False):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0
    m_factor = 1.0  # relaxation factor for M
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 100 == 0 and verbose:
            print(f"    LP-FW iter {it}: obj={best_obj:.8f}, m_factor={m_factor:.3f}")
        
        # Solve LP: max sum(g) s.t. (f*g)_k <= M*m_factor, g >= 0
        g = _solve_lp(f, M * m_factor, n)
        if g is None or g.sum() < 1e-15:
            m_factor = min(m_factor * 1.15, 5.0)
            if m_factor > 5.0:
                rng = np.random.default_rng(it * 13 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                m_factor = 1.0
            continue
        
        g = g / g.sum()
        
        # Check if g is close to f
        if np.max(np.abs(g - f)) < 1e-10:
            m_factor = min(m_factor * 1.1, 3.0)
            continue
        
        # Line search: try several alphas
        best_alpha = 0.0
        best_val = M
        
        # Coarse search
        for a in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = np.maximum(f + a * (g - f), 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = a
        
        # Fine search (bisection around best_alpha)
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(12):
                mid = (lo + hi) / 2.0
                ft = np.maximum(f + mid * (g - f), 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-10:
            f_new = np.maximum(f + best_alpha * (g - f), 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new
            m_factor = max(1.0, m_factor * 0.95)  # Tighten threshold
            
            if M < best_obj / (2.0 * n) - 1e-14:
                best_obj = 2.0 * n * M
                best_f = f.copy()
        else:
            m_factor = min(m_factor * 1.1, 4.0)

    return best_f, best_obj


def _solve_lp(f, M, n):
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 5})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def pair_optimize(f0, time_limit, verbose=False):
    """Targeted 2-variable optimization at peak positions."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_obj = 2.0 * n * fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        improved = False

        # Get top peaks
        n_peaks = min(8, 2 * n - 1)
        peak_indices = np.argsort(conv)[-n_peaks:]

        for pk in peak_indices:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            if je < js:
                continue

            idx = np.arange(js, je + 1)
            products = f[idx] * f[pk - idx]
            j_max = idx[np.argmax(products)]
            j_other = pk - j_max

            if j_max == j_other:
                # Diagonal: reduce f[j_max], redistribute to others
                old_val = f[j_max]
                if old_val < 1e-10:
                    continue
                for factor in [0.3, 0.5, 0.7, 0.85]:
                    trial = f.copy()
                    trial[j_max] = old_val * factor
                    diff = old_val - trial[j_max]
                    # Redistribute to non-contributing elements
                    mask = np.ones(n, dtype=bool)
                    for pk2 in peak_indices[:3]:
                        pk2 = int(pk2)
                        js2 = max(0, pk2 - n + 1)
                        je2 = min(n - 1, pk2)
                        mask[js2:je2 + 1] = False
                    mask[j_max] = False
                    if mask.sum() > 0 and f[mask].sum() > 1e-15:
                        trial[mask] += diff * f[mask] / f[mask].sum()
                    trial = np.maximum(trial, 0)
                    trial /= trial.sum()
                    obj = fast_convolve(trial).max()
                    if obj < M - 1e-14:
                        f = trial
                        M = obj
                        if 2.0 * n * M < best_obj:
                            best_obj = 2.0 * n * M
                            best_f = f.copy()
                        improved = True
                        break
            else:
                # Off-diagonal: adjust the pair
                for scale in [0.5, 0.7, 0.85]:
                    trial = f.copy()
                    old_jm = trial[j_max]
                    trial[j_max] = old_jm * scale
                    diff = old_jm - trial[j_max]
                    # Add to j_other or distribute
                    for add_frac in [0.0, 0.3, 0.5, 1.0]:
                        trial2 = trial.copy()
                        trial2[j_other] += diff * add_frac
                        remaining = diff * (1 - add_frac)
                        mask = np.ones(n, dtype=bool)
                        mask[j_max] = False
                        mask[j_other] = False
                        if mask.sum() > 0 and f[mask].sum() > 1e-15:
                            trial2[mask] += remaining * f[mask] / f[mask].sum()
                        trial2 = np.maximum(trial2, 0)
                        trial2 /= trial2.sum()
                        obj = fast_convolve(trial2).max()
                        if obj < M - 1e-14:
                            f = trial2
                            M = obj
                            if 2.0 * n * M < best_obj:
                                best_obj = 2.0 * n * M
                                best_f = f.copy()
                            improved = True
                            break
                    if improved:
                        break

        if not improved:
            no_improve += 1
            if no_improve > 20:
                rng = np.random.default_rng(it * 17 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                no_improve = 0
        else:
            no_improve = 0

        if no_improve > 100:
            break

    return best_f, best_obj


def adam_smooth_optimize(f0, time_limit, tau_init=None, lr_init=0.003, verbose=False):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    if tau_init is None:
        tau = 0.08 * conv.max()
    else:
        tau = tau_init

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    lr = lr_init
    t = 0

    best_obj = conv.max()
    best_f = f.copy()
    no_improve = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)
        scaled = conv / tau
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        # h[j] = sum_{l} w[j+l] * f[l] (cross-convolution)
        w_rev = w[::-1]
        total_len = n + len(w) - 1
        fft_size = 1 << (total_len - 1).bit_length()
        A = np.fft.rfft(f, fft_size)
        W = np.fft.rfft(w_rev, fft_size)
        conv_result = np.fft.irfft(A * W, fft_size)
        h = conv_result[n - 1:2 * n - 1][::-1]

        grad = (2.0 / tau) * h

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new
        tau *= 0.9997
        if tau < 1e-8:
            tau = 1e-8

        if no_improve > 100:
            lr *= 0.85
        if no_improve > 400:
            rng = np.random.default_rng(t * 13 + 7)
            noise = 0.002 + 0.003 * min(no_improve / 1000.0, 3.0)
            f = project_simplex(best_f + rng.standard_normal(n) * noise)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = lr_init * 0.7
            tau = 0.05 * fast_convolve(f).max()

        if verbose and t % 500 == 0:
            print(f"    Adam iter {t}: obj={2*n*best_obj:.8f}, tau={tau:.2e}")

    return best_f, 2 * n * best_obj


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, center_frac, sigma_frac):
    x = np.arange(n) - n * center_frac
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac):
    x = np.arange(n) - n / 2.0
    half = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_two_peak(n, pos1_frac, pos2_frac, sigma_frac, weight1=0.6):
    a = weight1 * np.exp(-0.5 * ((np.arange(n) - n * pos1_frac) / (n * sigma_frac)) ** 2)
    a += (1 - weight1) * np.exp(-0.5 * ((np.arange(n) - n * pos2_frac) / (n * sigma_frac)) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_gaussian(n, center_frac, sigma_frac):
    x = np.arange(n) - n * center_frac
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a
