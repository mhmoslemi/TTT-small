import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_objective(f):
    """Compute the normalized max auto-convolution (sum f = 1 assumed)."""
    conv = fast_convolve(f)
    return conv.max()


def compute_subgradient_conv(f, conv, top_k=20, temperature=0.01):
    """Compute weighted subgradient of log-sum-exp(conv)."""
    n = len(f)
    len_c = len(conv)
    k_top = min(top_k, len_c)
    peaks = np.argpartition(conv, -k_top)[-k_top:]
    pv = conv[peaks]
    scaled = pv / max(temperature, 1e-10)
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        idx = np.arange(js, je + 1)
        contrib = f[k - idx] + f[idx]
        # Handle the self-pairing term: when k - m == m, i.e., m == k/2
        # In that case the derivative is 2*f[m] not f[m]+f[m]... actually it IS 2*f[m]
        # because d/da[m] of a[m]*a[k-m] when k-m=m gives 2*a[m]
        sg[js:je + 1] += w[i] * contrib
    return sg


def parametric_shape(n, param):
    """Generate a parameterized asymmetric shape.
    param = [alpha, beta, gamma, delta]
    a[j] = exp(-alpha*(j/n)^beta) * (1 + gamma*sin(pi*j/n)) * (1 + delta*(j/n))
    """
    alpha, beta, gamma, delta = param
    x = np.arange(n) / max(n - 1, 1)
    a = np.exp(-alpha * x ** beta) * (1.0 + gamma * np.sin(np.pi * x)) * (1.0 + delta * x)
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def parametric_obj(param, n):
    """Objective for parameterized shape."""
    a = parametric_shape(n, param)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def optimize_parametric(n, seed=42, time_limit=30):

            k_top = 5
        elif ratio > 1.5:
            k_top = 15
        else:
            k_top = max(30, len(conv) // 10)

        sg = compute_subgradient_conv(f, conv, top_k=k_top, temperature=1.0 / tau)

        # Momentum step
        velocity = momentum_decay * velocity - lr * sg
        f_new = f * np.exp(velocity)
        f_new = np.maximum(f_new, 1e-14)
        s = f_new.sum()
        if s < 1e-14:
            f_new = f.copy()
        else:
            f_new = f_new / s

        # Line search
        obj_new = fast_convolve(f_new).max()
        if obj_new > M * 1.02:
            # Try smaller steps
            found = False
            for damp in [0.5, 0.3, 0.1, 0.05]:
                f_trial = f * (1.0 - damp) + f_new * damp
                f_trial = np.maximum(f_trial, 1e-14)
                f_trial = f_trial / f_trial.sum()
                obj_trial = fast_convolve(f_trial).max()
                if obj_trial < M:
                    f_new = f_trial
                    obj_new = obj_trial
                    found = True
                    break
            if not found:
                lr *= 0.5
                velocity *= 0.5
                continue

        f = f_new

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
            best_tau = tau
            if verbose and it % 200 == 0:
                print(f"        SMDv2 iter {it}: obj={best_obj:.10f}, tau={tau:.1f}, lr={lr:.6f}")
        else:
            no_improve += 1

        tau *= tau_rate
        lr *= lr_decay

        if no_improve > restart_every:
            rng = np.random.default_rng(it * 13 + 7)
            noise_scale = 0.002 + 0.003 * min(no_improve / restart_every, 2.0)
            f = np.maximum(best_f + rng.standard_normal(n) * noise_scale, 1e-14)
            f = f / f.sum()
            lr = lr_init * 0.4
            tau = best_tau
            velocity = np.zeros(n)
            no_improve = 0

        if lr < 1e-8:
            break

    return best_f, best_obj


def peak_shaving(f0, time_limit, verbose=False):
    """
    Direct peak shaving: identify the worst peak position k*, then for each
    pair (j, k*-j) contributing to it, try reducing a[j] and a[k*-j].
    """
    f = np.maximum(f0.astype(np.float64), 1e-14)
    f = f / f.sum()
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    lr = 0.005
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        k_star = int(np.argmax(conv))

        # Find all pairs contributing to peak k_star
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)

        # Subtract gradient contribution
        grad = np.zeros(n)
        idx = np.arange(js, je + 1)
        grad[js:je + 1] = f[k_star - idx] + f[idx]

        # Also handle secondary peaks
        top_peaks = np.argsort(conv)[-5:]
        for k2 in top_peaks:
            k2 = int(k2)
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            if je2 >= js2:
                idx2 = np.arange(js2, je2 + 1)
                grad[js2:je2 + 1] += 0.3 * (f[k2 - idx2] + f[idx2])

        # Multiplicative update
        f_new = f * np.exp(-lr * grad)
        f_new = np.maximum(f_new, 1e-14)
        f_new = f_new / f_new.sum()

        obj_new = fast_convolve(f_new).max()
        if obj_new < M - 1e-15:
            f = f_new
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"        PeakShave iter {it}: obj={best_obj:.10f}")
            lr = min(lr * 1.01, 0.03)
        else:
            no_improve += 1
            lr *= 0.8

        if no_improve > 80:
            rng = np.random.default_rng(it * 23 + 11)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.003, 1e-14)
            f = f / f.sum()
            lr = 0.005
            no_improve = 0

        if lr < 1e-9:
            break

    return best_f, best_obj


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c_obj = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'timelimit': 30})
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False, lp_m_expand=1.01):
    """LP-based Frank-Wolfe with careful line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp = min(m_exp * 1.15, 10.0)
            if m_exp >= 10.0:
                rng = np.random.default_rng(it * 19 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp = min(m_exp * 1.1, 5.0)
            continue
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-11:
            no_improve += 1
            if no_improve > 80:
                m_exp = min(m_exp * 1.1, 5.0)
            if no_improve > 200:
                rng = np.random.default_rng(it * 41 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue
        else:
            no_improve = 0

        # Line search (coarse then fine)
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            # Fine search
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 10 == 0:
                    print(f"        LP-FW iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 5.0)

        if no_improve > 200:
            break

    return best_f, best_obj


def coordinate_descent_v2(f0, time_limit, verbose=False):
    """
    Improved coordinate descent: for each coordinate, do a golden-section
    search to find the optimal value while keeping others scaled proportionally.
    """
    f = np.maximum(f0.astype(np.float64), 1e-14)
    f = f / f.sum()
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    sweeps = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        sweeps += 1
        rng = np.random.default_rng(sweeps * 7 + 1)
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj

            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            # Log-spaced perturbations around current value
            ratios = [0.3, 0.5, 0.7, 0.8, 0.9, 1.1, 1.2, 1.5, 2.0, 3.0, 0.1]
            for r in ratios:
                new_val = old_val * r
                if new_val < 0 or new_val > 0.8:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 1e-14)
                ss = f_trial.sum()
                if ss < 1e-14:
                    continue
                f_trial = f_trial / ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val and best_j_obj < best_obj - 1e-15:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 1e-14)
                    f = f / f.sum()
                best_obj = best_j_obj
                best_f = f.copy()
                improved = True

        if not improved:
            # Try a random subset with larger perturbations
            if sweeps < 50:
                rng2 = np.random.default_rng(sweeps * 13 + 5)
                subset = rng2.choice(n, size=min(50, n), replace=False)
                sub_improved = False
                for j in subset:
                    old_val = f[j]
                    for new_val in [old_val * 0.1, old_val * 0.5, old_val * 2.0, old_val * 5.0, 0.0]:
                        if new_val < 0 or new_val > 0.9:
                            continue
                        denom = 1.0 - old_val
                        if denom < 1e-10:
                            continue
                        scale = (1.0 - new_val) / denom
                        f_trial = f.copy()
                        f_trial[j] = new_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        f_trial[mask] *= scale
                        f_trial = np.maximum(f_trial, 1e-14)
                        ss = f_trial.sum()
                        if ss < 1e-14:
                            continue
                        f_trial = f_trial / ss
                        val = fast_convolve(f_trial).max()
                        if val < best_obj - 1e-14:
                            best_obj = val
                            f = f_trial
                            best_f = f.copy()
                            sub_improved = True
                            break
                if not sub_improved:
                    break
            else:
                break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def diverse_init(n, variant, seed=42):
    """Generate diverse asymmetric initial sequences."""
    rng = np.random.default_rng(seed + variant * 1000)
    x = np.arange(n) / max(n - 1, 1)

    if variant == 0:
        a = np.exp(-2.0 * x)
    elif variant == 1:
        a = np.exp(-3.0 * x ** 0.5)
    elif variant == 2:
        a = (1 - x) ** 3
    elif variant == 3:
        a = np.exp(-2.0 * x) * (1 + 0.5 * np.sin(2 * np.pi * x))
        a = np.maximum(a, 0)
    elif variant == 4:
        # Two-scale: fast decay then slow decay
        a = np.where(x < 0.3, np.exp(-5 * x), np.exp(-5 * 0.3) * np.exp(-0.5 * (x - 0.3)))
    elif variant == 5:
        # Power law
        a = (1 - x + 0.1) ** (-0.3) * np.exp(-1.0 * x)
    elif variant == 6:
        # Gaussian-like but skewed
        a = np.exp(-8 * (x - 0.3) ** 2) + 0.3 * np.exp(-8 * (x - 0.7) ** 2)
    elif variant == 7:
        # Ramp with exponential
        a = x * np.exp(-3.0 * x)
    elif variant == 8:
        # Pure random with structure
        a = np.ones(n)
        a[:n // 4] *= 1.3
        a[n // 4: n // 2] *= 1.1
        a[n // 2: 3 * n // 4] *= 0.8
        a[3 * n // 4:] *= 0.6
        a += rng.standard_normal(n) * 0.05
        a = np.maximum(a, 0.01)
    elif variant == 9:
        # Cosine window, asymmetric
        a = np.cos(np.pi * x / 2) ** 4 + 0.05
    elif variant == 10:
        # Logarithmic
        a = 1.0 / (1.0 + x * 5.0)
    elif variant == 11:
        # Sigmoid-like transition
        a = 1.0 / (1.0 + np.exp(10 * (x - 0.4)))
    else:
        a = np.exp(-1.5 * x)

    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    Phase 1: Parameterized shape search (fast)
    Phase 2: SMD v2 with momentum on best inits
    Phase 3: Peak shaving
    Phase 4: LP-Frank-Wolfe
    Phase 5: Multi-length exploration
    Phase 6: Coordinate descent polish
    Phase 7: Final squeeze
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # === Phase 1: Parameterized shape search ===
    print("=== Phase 1: Parameterized Shape Search ===")
    t_phase1_start = time.monotonic()
    test_lengths = [640, 768, 896, 1024]

    param_results = []
    for n_try in test_lengths:
        if time.monotonic() - t_phase1_start > 60:
            break
        a_param, obj_param = optimize_parametric(n_try, seed=seed, time_limit=15)
        param_results.append((obj_param, a_param, f"param_n{n_try}"))
        sc = try_seq(a_param, f"param_n{n_try}")

    # Also try diverse inits
    for n_try in test_lengths:
        if time.monotonic() - t_phase1_start > 50:
            break
        for v in range(12):
            a_init = diverse_init(n_try, v, seed=seed)
            sc = try_seq(a_init, f"div_{v}_n{n_try}")
            param_results.append((2 * n_try * fast_convolve(a_init).max(), a_init, f"div_{v}_n{n_try}"))

    # Sort by objective and keep top candidates
    param_results.sort(key=lambda x: x[0])
    top_candidates = param_results[:6]
    print(f"  Top init objects: {[f'{l}:{o:.4f}' for o, _, l in top_candidates[:4]]}")

    # === Phase 2: SMD v2 with momentum ===
    print("\n=== Phase 2: SMD v2 (Momentum) ===")
    rem = deadline - time.monotonic() - 900
    if rem > 60:
        t_alloc_each = min(rem / 3, 200)
        for idx in range(min(3, len(top_candidates))):
            rem_now = deadline - time.monotonic() - 600
            if rem_now < 30:
                break
            _, a_init, label = top_candidates[idx]
            t_alloc = min(rem_now * 0.9, t_alloc_each)
            if t_alloc < 20:
                t_alloc = 20
            print(f"  SMDv2 on {label} (n={len(a_init)}): t={t_alloc:.0f}s")
            a_opt, obj = smoothed_mirror_descent_v2(a_init, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, f"smdv2_{label}")

    # === Phase 3: Peak Shaving ===
    print("\n=== Phase 3: Peak Shaving ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 500
        if rem > 30:
            t_alloc = min(rem * 0.6, 150)
            if t_alloc < 20:
                t_alloc = 20
            print(f"  PeakShave on best (n={len(best_seq)}): t={t_alloc:.0f}s")
            a_opt, obj = peak_shaving(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "peakshave")

    # === Phase 4: LP-Frank-Wolfe ===
    print("\n=== Phase 4: LP-Frank-Wolfe ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 60:
            t_alloc = min(rem * 0.7, 200)
            if t_alloc < 30:
                t_alloc = 30
            print(f"  LP-FW on best (n={len(best_seq)}): t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "lpfw_main")

    # === Phase 5: Multi-length exploration ===
    print("\n=== Phase 5: Multi-Length Exploration ===")
    if best_seq is not None:
        base = best_seq
        base_n = len(base)
        target_ns = [512, 640, 768, 896, 1024, 1200]
        target_ns = [tn for tn in target_ns if tn != base_n]

        for target_n in target_ns:
            rem = deadline - time.monotonic() - 200
            if rem < 40:
                break
            f_interp = interpolate_sequence(base, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")
            if sc < best_score + 0.02:
                # Run SMD on this length
                t_alloc = min(rem * 0.3, 80)
                if t_alloc < 15:
                    t_alloc = 15
                print(f"  SMDv2 n={target_n}: t={t_alloc:.0f}s")
                a_opt, obj = smoothed_mirror_descent_v2(f_interp, time_limit=t_alloc)
                try_seq(a_opt, f"smdv2_n{target_n}")

    # === Phase 6: More SMD with different params ===
    print("\n=== Phase 6: Additional SMD Runs ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for r in range(3):
            rem = deadline - time.monotonic() - 100
            if rem < 25:
                break
            t_alloc = min(rem * 0.5, 100)
            if t_alloc < 15:
                t_alloc = 15

            if r == 0:
                # Small noise
                noise = rng.standard_normal(n_best) * 0.003
                f_init = np.maximum(best_seq + noise, 1e-14)
                f_init = f_init / f_init.sum()
                tau_i = 20.0
                lr_i = 0.005
            elif r == 1:
                # Blend with parametric
                a_param = optimize_parametric(n_best, seed=seed + 100, time_limit=5)[0]
                f_init = 0.6 * best_seq + 0.4 * a_param
                f_init = f_init / f_init.sum()
                tau_i = 10.0
                lr_i = 0.008
            else:
                # Reverse blend
                f_init = 0.5 * best_seq + 0.5 * best_seq[::-1]
                f_init = f_init / f_init.sum()
                tau_i = 15.0
                lr_i = 0.006

            print(f"  SMDv2 restart {r}: t={t_alloc:.0f}s, tau={tau_i}, lr={lr_i}")
            a_opt, obj = smoothed_mirror_descent_v2(f_init, time_limit=t_alloc,
                                                     tau_init=tau_i, lr_init=lr_i)
            try_seq(a_opt, f"restart_{r}")

    # === Phase 7: Coordinate Descent Polish ===
    print("\n=== Phase 7: Coordinate Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_alloc = min(rem * 0.7, 100)
            if t_alloc < 15:
                t_alloc = 15
            print(f"  CD (n={len(best_seq)}): t={t_alloc:.0f}s")
            a_opt, obj = coordinate_descent_v2(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "cd_polish")

    # === Phase 8: Final LP-FW squeeze ===
    print("\n=== Phase 8: Final Squeeze ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 30:
            t_alloc = min(rem * 0.8, 60)
            if t_alloc < 15:
                t_alloc = 15
            print(f"  Final LP-FW: t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_lpfw")

        # One more CD
        rem = deadline - time.monotonic() - 15
        if rem > 15:
            t_alloc = min(rem * 0.8, 30)
            if t_alloc < 10:
                t_alloc = 10
            print(f"  Final CD: t={t_alloc:.0f}s")
            a_opt, obj = coordinate_descent_v2(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
