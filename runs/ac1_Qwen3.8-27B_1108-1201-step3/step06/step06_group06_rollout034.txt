Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
(1) Mirror descent with true max-convolution subgradient (multiplicative weights),
(2) Exact coordinate descent on the non-smooth objective,
(3) LP-based Frank-Wolfe with adaptive margin,
(4) Multi-length exploration with interpolation.
Key innovation: Direct subgradient of the true max (not smoothed), combined
with multiplicative weight updates on the simplex for fast convergence.
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgradient_of_max_conv(a):
    """Compute subgradient of max_k conv(a,a)[k] with respect to a."""
    conv = fast_convolve(a)
    k_max = np.argmax(conv)
    n = len(a)
    # grad[i] = 2 * a[k_max - i] for valid indices (0 <= k_max - i < n)
    # i.e., i in range max(0, k_max-n+1) to min(n-1, k_max)
    grad = np.zeros(n)
    i_lo = max(0, k_max - n + 1)
    i_hi = min(n - 1, k_max)
    for i in range(i_lo, i_hi + 1):
        grad[i] = 2.0 * a[k_max - i]
    return grad, conv, k_max

def mirror_descent_step(a, grad, eta):
    """Multiplicative weights update on the simplex."""
    log_a = np.where(a > 0, np.log(a), -100)
    new_log = log_a - eta * grad
    new_log -= new_log.max()
    a_new = np.exp(new_log)
    a_new /= a_new.sum()
    return a_new

def md_optimize(a0, time_limit, eta_init=0.1, verbose=False):
    """Mirror descent with true max-convolution subgradient."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    n = len(a)
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    eta = eta_init
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    # Momentum for the subgradient direction
    mom = np.zeros(n)
    beta = 0.7
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        grad, conv, k_max = subgradient_of_max_conv(a)
        
        # Add momentum
        mom = beta * mom + (1 - beta) * grad
        
        # Mirror descent step
        a_new = mirror_descent_step(a, mom, eta)
        
        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Adaptive eta
        if no_improve > 50:
            eta *= 0.9
            mom = 0  # Reset momentum
            no_improve = 0
        if no_improve > 200:
            # Restart from best with perturbation
            rng = np.random.default_rng(it * 13 + 7)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
            eta = eta_init * 0.5
            mom = np.zeros(n)
            no_improve = 0
        
        if verbose and it % 1000 == 0:
            print(f"    MD iter {it}: obj={2*n*best_obj:.8f}, eta={eta:.4f}")
    
    return best_a, 2 * n * best_obj

def lse_gradient(f, tau):

def nesterov_optimize(f0, time_limit, tau_init=None, lr_init=0.003, verbose=False):

    y = f.copy()  # lookahead point
    momentum = 0.9

    no_improve = 0

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

            break

        c = -np.ones(n)

def spectral_refine(f0, time_limit, verbose=False):
    """Spectral domain refinement: iteratively reduce auto-conv peaks."""
    a = np.maximum(f0.astype(np.float64), 0.0)
    s = a.sum()
    n = len(a)
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    fft_size = 1 << (2 * n - 2).bit_length()
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    eta = 0.05
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)

        b = fast_convolve(a)
        M = b.max()
        if M < best_obj - 1e-14:
            best_obj = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Top peaks
        top_k = min(5, len(b))
        peak_indices = np.argsort(b)[-top_k:]

        # Combined gradient in spectral domain
        N = 2 * n - 1
        freqs = np.arange(len(mag))
        grad_spec = np.zeros(len(mag))
        for pk in peak_indices:
            pk = int(pk)
            weight = b[pk] / b[peak_indices].sum()
            grad_spec += weight * np.cos(2 * np.pi * freqs * pk / N)

        # Multiplicative update
        mag_new = mag * (1.0 - eta * np.maximum(grad_spec, 0))
        mag_new[0] = mag[0]  # Preserve DC
        mag_new = np.maximum(mag_new, 1e-10)

        # Reconstruct
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            a = best_a.copy()
        else:
            a = a_new / sn

        # Decay eta
        eta *= 0.998
        if no_improve > 100:
            eta = 0.03
            rng = np.random.default_rng(it * 7 + 3)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
            no_improve = 0

        if verbose and it % 200 == 0:
            print(f"    Spectral iter {it}: obj={2 * n * best_obj:.8f}, eta={eta:.4f}")

    return best_a, 2 * n * best_obj


def shape_gaussian(n, center, sigma):
    x = np.arange(n) / n - center
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def shape_bell(n, width):
    x = np.arange(n) / n - 0.5
    t = np.clip(np.abs(x) / (width / 2), 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def shape_exp(n, decay):
    a = np.exp(-np.arange(n) / (n * decay))
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def shape_two_bump(n, center1, center2, sigma):
    x = np.arange(n) / n
    a = np.exp(-0.5 * ((x - center1) / sigma) ** 2) + np.exp(-0.5 * ((x - center2) / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def shape_sine_power(n, power, phase):
    a = np.sin(np.pi * (np.arange(n) / n + phase)) ** power
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def shape_power_law(n, alpha, offset):
    a = (1.0 + offset + np.arange(n) / n) ** (-alpha)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

        if rem > 10:
            a_opt, _ = lp_fw_active(best_seq, time_limit=min(rem, 30),
                                    max_constraints=40, verbose=False)
            try_seq(a_opt, "final_lpfw")

    # Fallback
    if best_seq is None:
        best_seq = shape_gaussian(768, 0.35, 0.25)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
