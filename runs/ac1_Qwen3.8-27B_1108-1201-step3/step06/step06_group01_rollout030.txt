            y_new = (t * a + beta * y - (beta-1) * v) / (t + beta)
            # Simplified: y = a + (t/(t+beta)) * (a - a_prev)

        # Track best

        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        v = a_new - a  # velocity for Nesterov
        
        if t % 1000 == 0:
            print(f"    Nesterov t={t} T={T:.4f} score={best_score:.6f}")

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using coordinate descent with 
analytic 1D line search, Nesterov-accelerated smoothed gradient method,
and multi-start diverse initializations. Different from the previous LP-based
fictitious play approach, this uses direct coordinate-wise optimization with
exact line search and momentum-based gradient descent on the smoothed objective.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_normalized(a):
    """Compute the score for a non-negative sequence."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a_n = a / s
    return 2.0 * len(a_n) * fast_convolve(a_n).max()

def smooth_max_weights(x, T):
    """Compute softmax weights for smoothed max."""
    x_shifted = (x - x.max()) / T
    w = np.exp(x_shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
    else:
        w /= w_sum
    return w

def auto_corr_grad(a, w):
    """
    Compute gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a.
    a has length n, w has length 2n-1.
    grad[i] = 2 * sum_{m=0}^{n-1} a[m] * w[i+m]
    """
    n = len(a)
    a_rev = a[::-1]
    N = 1 << (3 * n - 2).bit_length()
    W = rfft(w, N)
    AR = rfft(a_rev, N)
    full_conv = irfft(W * AR, N)[:3*n-2]
    cross = full_conv[n-1 : 2*n-1]
    return 2.0 * cross

def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    # First project onto {x >= 0, sum(x) = 1}, then scale
    v = np.maximum(v, 0)
    s = v.sum()
    if s < 1e-30:
        return np.ones(n) * target_sum / n
    return v / s * target_sum

def momentum_smoothed_descent(a0, budget=60.0, T0=5.0, T_min=0.5, 
                                lr_init=0.02, momentum=0.9, n_max_steps=5000):
    """Momentum-based descent on smoothed auto-correlation max."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s  # normalize to sum=1
    
    v = np.zeros(n)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for step in range(n_max_steps):
        if time.monotonic() - t0 > budget:
            break
        
        T = max(T_min, T0 * np.exp(-0.005 * step))
        conv_a = fast_convolve(a)
        w = smooth_max_weights(conv_a, T)
        grad = auto_corr_grad(a, w)
        
        # Normalize gradient for stable updates
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        grad_norm = grad / g_norm
        
        # Momentum update
        v = momentum * v - lr_init * np.exp(-0.002 * step) * grad_norm
        a_new = a + v
        
        # Project onto simplex
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s_new
        
        # Evaluate
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_score

    a = a / s
    
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    conv_a = fast_convolve(a)
    
    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > budget:
            break
        sweep_improved = False
        
        for i in range(n):
            if time.monotonic() - t0 > budget:
                break
            
            x_old = a[i]
            if x_old < 1e-15:
                # Try small positive values
                for x_try in [1e-4, 1e-3, 0.001, 0.002]:
                    a[i] = x_try
                    sc = 2.0 * n * fast_convolve(a).max()
                    if sc < best_score:

                        sweep_improved = True
                    else:
                        a[i] = 0
                        break

            # Try various multiplicities

            best_x = x_old
            best_sc = 2.0 * n * conv_a.max()  # current score

                sc = 2.0 * n * fast_convolve(a).max()
                if sc < best_sc:
                    best_sc = sc
                    best_x = x_old * f

            if best_x != x_old:
                conv_a = fast_convolve(a)
                sweep_improved = True
            if best_sc < best_score:
                best_score = best_sc
                sweep_improved = True
        
        # If no improvement in a full sweep, try finer steps
        if not sweep_improved:
            # Finer search
            for i in range(n):
                if time.monotonic() - t0 > budget:
                    break
                x_old = a[i]
                if x_old < 1e-10:
                    continue
                found = False
                for f in np.linspace(0.85, 1.15, 13):
                    a[i] = x_old * f
                    sc = 2.0 * n * fast_convolve(a).max()
                    if sc < best_score - 1e-15:
                        best_score = sc
                        found = True
                    else:
                        a[i] = x_old
                        break
                if not found:
                    a[i] = x_old
            break
    
    return a, best_score
