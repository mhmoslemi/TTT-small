```python
import numpy as np
from scipy.optimize import linprog, minimize
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def objective_normalized(a):
    """Compute 2*n*max(conv)/sum^2 for normalized a (sum=1)."""
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max()
```

One more thought: the "lazy" LP-FW idea. At each iteration, we check if the current sequence $f$ already satisfies $(f*f)[k] \le M$ for all $k$ (trivially true at the current peak). The LP finds a direction $g$ that maximizes $\sum g_j$ subject to $(f*g)[k] \le M$. If $g$ is very close to $f$ (i.e., the direction doesn't change much), we can skip the LP and just do a small perturbation.

The "greedy" direction: find the set of $j$ such that $f_j = 0$ or $j$ is "safe" (doesn't contribute to any lag near the peak). Then set $g_j = 1$ for safe $j$ and $g_j = 0$ otherwise.

Actually, the "safe" set depends on the threshold. Let me define:

- "Hot" lags: $K = \{k : (f*f)[k] > (1-\epsilon) M\}$ for some small $\epsilon$.
- For each hot lag $k$, the "contributing" positions are $[k-n+1, k] \cap [0, n-1]$.
- "Safe" positions: those not in any contributing set.

The greedy direction is $g_j = 1$ if $j$ is safe, $g_j = 0$ otherwise.

"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def objective_normalized(a):
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max()

def softmax_obj_and_grad(a, T):
    """Compute soft-max objective and its gradient."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = conv / T
    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    log_sum = np.log(exp_s.sum())
    obj = T * (log_sum + scaled.max())  # = T * log(sum(exp(conv/T)))
    w = exp_s / exp_s.sum()
    # Gradient
    a_rev = a[::-1]
    fft_size = 1 << (3 * n - 1).bit_length()
    A_rev = np.fft.rfft(a_rev, fft_size)
    W = np.fft.rfft(w, fft_size)
    corr = np.fft.irfft(A_rev * W, fft_size)
    c = corr[n-1:2*n-1]
    grad = 2.0 * c
    return obj, grad

def lp_fw_optimize(f0, time_limit, verbose=False, max_m_levels=5):
    """LP-Frank-Wolfe with escalation and line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1.0:
            break
        
        n_constraints = 2 * n - 1
        # Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0
        c = -np.ones(n)
        A_ub = np.zeros((n_constraints, n))
        for j in range(n):
            A_ub[j:j+n, j] = f
        b_ub = np.full(n_constraints, M)
        bounds = [(0, None)] * n
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        
        if not res.success or res.x.sum() < 1e-12:
            # Try with relaxed M
            M_relaxed = M * 1.1
            b_ub2 = np.full(n_constraints, M_relaxed)
            res = linprog(c, A_ub=A_ub, b_ub=b_ub2, bounds=bounds, method='highs')
            if not res.success or res.x.sum() < 1e-12:
                no_improve += 1
                if no_improve > 30:
                    break
                # Perturb
                rng = np.random.default_rng(it)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                conv = fast_convolve(f)
                M = conv.max()
                continue
        
        g = res.x / res.x.sum()
        
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            continue
        no_improve = 0
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = np.maximum(f + alpha * (g - f), 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha < 1e-10:
            no_improve += 1
            continue
        
        f_new = np.maximum(f + best_alpha * (g - f), 0.0)
        f_new = f_new / f_new.sum()
        conv = fast_convolve(f_new)
        M = conv.max()
        f = f_new
        
        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            if verbose and it % 10 == 0:
                print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
    
    return best_f, best_obj

def softmax_lbfgs(f0, time_limit, verbose=False):
    """Optimize soft-max objective with L-BFGS, decreasing T."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:

    M = conv.max()
    best_obj_true = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    T = M * 0.1  # Start with T = 10% of the peak
    
    while time.monotonic() - t0 < time_limit:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5.0:
            break
        
        # Define objective and gradient for L-BFGS
        def obj_func(x):
            x = np.maximum(x, 0)
            sx = x.sum()
            if sx < 1e-15:
                return 1e10, np.zeros_like(x)
            x = x / sx
            return softmax_obj_and_grad(x, T)
        
        # Use scipy minimize with L-BFGS-B (handles bounds)
        from scipy.optimize import minimize
        result = minimize(
            lambda x: softmax_obj_and_grad(x, T)[0],
            f.copy(),
            jac=lambda x: softmax_obj_and_grad(x, T)[1],
            method='L-BFGS-B',
            bounds=[(0, None)] * n,
            options={'maxiter': 50, 'ftol': 1e-12, 'gtol': 1e-8}
        )
        
        f_new = np.maximum(result.x, 0)
        f_new = f_new / f_new.sum()
        
        # Check true objective
        true_obj = fast_convolve(f_new).max()
        if true_obj < best_obj_true - 1e-14:
            best_obj_true = true_obj
            best_f = f_new.copy()
            if verbose:
                print(f"    SoftMax T={T:.6f}: true_obj={best_obj_true:.10f}")
        
        f = f_new
        T *= 0.7  # Decrease T
        
        if T < M * 0.001:
            T = M * 0.001
    
    return best_f, best_obj_true

def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
