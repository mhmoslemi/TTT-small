import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def cross_correlate_fft(w, f):
    """Compute force[i] = sum_k w[k] * f[k-i] (valid range) via FFT."""
    n = len(f)
    w_rev = w[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    cross = np.fft.irfft(F * W, fft_size)
    return cross[n - 1:2 * n - 1][::-1]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def md_optimize(f0, time_limit, verbose=False):
    """Accelerated mirror descent for minimizing max(f*f) on the simplex."""
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    eta = 0.1  # Step size
    tau_w = 0.03  # Temperature for subgradient weights
    t_damp = 0.4  # Damping factor
    
    f_prev = f.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Extrapolation (Nesterov acceleration)
        momentum = (it - 1) / (it + 2)
        y = f + momentum * (f - f_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()
        
        # Compute convolution at y
        conv_y = fast_convolve(y)
        M = conv_y.max()
        
        if M < best_obj - 1e-12:
            best_obj = M
            best_f = y.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    MD iter {it}: obj={2*n*best_obj:.8f}, eta={eta:.4f}")
        else:
            no_improve += 1
        
        # Subgradient weights
        scaled = conv_y / tau_w
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        
        # Force computation
        force = cross_correlate_fft(w, y)
        
        # Mirror descent update
        log_new = -eta * force
        log_new -= log_new.max()  # for numerical stability
        new_f = np.exp(log_new)
        new_f = np.maximum(new_f, 1e-30)
        new_f = new_f / new_f.sum()
        
        # Damped update
        f_prev = f.copy()
        f = (1 - t_damp) * f + t_damp * new_f
        f = np.maximum(f, 1e-15)
        f = f / f.sum()
        
        # Adaptive parameters
        if no_improve > 100:
            eta *= 0.8
            t_damp = min(0.8, t_damp * 1.05)
            if eta < 0.0005:
                # Restart from best

                f_prev = f.copy()
                eta = 0.05
                t_damp = 0.3
                no_improve = 0
        
        # Slowly cool the temperature
        tau_w *= 0.9999
    
    return best_f, 2 * n * best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
(1) Diversified initialization with spectral shaping,
(2) Accelerated mirror descent (multiplicative weights) as main workhorse,
(3) LP-Frank-Wolfe refinement for local improvement,
(4) Spectral flattening for global restructuring,
(5) Multi-length interpolation and final polish.
Key innovation: combining multiplicative mirror descent with adaptive
subgradient focusing and LP-based direction finding for robust convergence.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_correlate_fft(w, f):
    """Compute force[i] = sum_k w[k] * f[k-i] via FFT (valid range only)."""
    n = len(f)
    w_rev = w[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    cross = np.fft.irfft(F * W, fft_size)
    return cross[n - 1:2 * n - 1][::-1]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_gaussian(n, center=0.4, sigma=0.25):
    x = np.arange(n) - n * center
    a = np.exp(-0.5 * (x / (n * sigma)) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width=0.6):
    x = np.arange(n) - n / 2.0
    half = n * width / 2.0
    t = np.clip(np.abs(x) / half, 0, 1.0)
    a = (1.0 - t) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_tapered(n, power=2.0):
    x = np.arange(n) / (n - 1)
    a = (4.0 * x * (1.0 - x)) ** power
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def mirror_descent(f0, time_limit, verbose=False):
    """Accelerated mirror descent with adaptive step size and subgradient focusing."""
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    eta = 0.08
    tau_w = 0.025
    damp = 0.35

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        conv = fast_convolve(f)
        M = conv.max()

        if M < best_obj - 1e-12:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    MD iter {it}: obj={2 * n * best_obj:.8f}, eta={eta:.4f}")
        else:
            no_improve += 1

        # Adaptive subgradient focusing: concentrate on top peaks
        scaled = conv / tau_w
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        # Compute force via cross-correlation
        force = cross_correlate_fft(w, f)

        # Mirror descent multiplicative update
        log_update = -eta * force
        log_update -= log_update.max()
        new_f = np.exp(log_update)
        new_f = np.maximum(new_f, 1e-30)
        s = new_f.sum()
        if s < 1e-20:
            new_f = np.ones(n) / n
        else:
            new_f = new_f / s

        # Damped combination
        f = (1.0 - damp) * f + damp * new_f
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        # Adaptive parameters
        if no_improve > 80:
            eta *= 0.75
            if eta < 0.002:
                if verbose:
                    print(f"    MD restart at iter {it}")
                rng = np.random.default_rng(it * 13 + 7)
                f = np.maximum(best_f + rng.standard_normal(n) * 0.003, 1e-15)
                f = f / f.sum()
                eta = 0.06
                tau_w = 0.02
                no_improve = 0
        elif no_improve > 30:
            tau_w *= 0.9

        # Slow cooling of focusing temperature
        if it % 100 == 0:
            tau_w = max(0.003, tau_w * 0.995)

    return best_f, 2 * n * best_obj


def spectral_flatten(f0, time_limit, verbose=False):
    """Spectral flattening: iteratively flatten the power spectrum."""
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)
    fft_size = 1 << (2 * n - 2).bit_length()

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    alpha = 0.4
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        F = np.fft.rfft(f, fft_size)
        mag = np.abs(F)
        mag = np.maximum(mag, 1e-15)
        phase = np.angle(F)

        # Flatten magnitude: reduce peaks in spectrum
        new_mag = mag ** (1.0 - alpha)
        new_F = new_mag * np.exp(1j * phase)
        f_new = np.fft.irfft(new_F, fft_size)[:n]
        f_new = np.maximum(f_new, 1e-15)
        s = f_new.sum()
        if s < 1e-15:
            break
        f_new = f_new / s

        conv = fast_convolve(f_new)
        obj = conv.max()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    Spectral iter {it}: obj={2 * n * best_obj:.8f}")
        else:
            no_improve += 1

        # Damped update
        f = 0.6 * f + 0.4 * f_new
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        if no_improve > 100:
            alpha *= 0.85
            if alpha < 0.05:
                break
        if it % 100 == 0:
            alpha = max(0.05, alpha * 0.97)

    return best_f, 2 * n * best_obj


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive relaxation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
        s = 1.0
    f = f / s

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        g = _solve_lp_dir(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.15
            if m_exp > 8.0:
                rng = np.random.default_rng(it * 17 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-12:
            m_exp *= 1.1
            continue
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 50:
                m_exp = min(m_exp * 1.15, 6.0)
            if no_improve > 150:
                break
            continue
        else:
            no_improve = 0

        # Line search (bisection)
        best_alpha = 0.0
        best_val = M

        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = np.maximum(f + alpha * (g - f), 0.0)
            st = ft.sum()
            if st < 1e-12:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.08)
            hi = min(1.0, best_alpha + 0.08)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = np.maximum(f + mid * (g - f), 0.0)
                st = ft.sum()
                if st < 1e-12:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-8:
            f_new = np.maximum(f + best_alpha * (g - f), 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new
            m_exp = 1.0

            if M < best_obj - 1e-12:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 10 == 0:
                    print(f"    LP-FW iter {it}: obj={2 * n * best_obj:.8f}")
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)

        if no_improve > 200:
            break

    return best_f, 2 * n * best_obj


def _solve_lp_dir(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    B = np.zeros((2 * n - 1, n))
    for j in range(n):
        B[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=B, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 30, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def gradient_cd(f0, time_limit, verbose=False):
    """Subgradient method with simplex projection and momentum."""
    f = np.maximum(f0.astype(np.float64), 1e-10)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    lr = 0.02 / n
    momentum_arr = np.zeros(n)
    beta = 0.9

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        conv = fast_convolve(f)
        M = conv.max()

        if M < best_obj - 1e-12:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    GCD iter {it}: obj={2 * n * best_obj:.8f}")
        else:
            no_improve += 1

        # Subgradient: weighted combination of top peaks
        top_k = min(5, 2 * n - 1)
        peak_idx = np.argsort(conv)[-top_k:]
        peak_vals = conv[peak_idx]
        weights = peak_vals / peak_vals.sum()

        grad = np.zeros(n)
        for k, w in zip(peak_idx, weights):
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * w * f[k - idx]

        # Momentum update
        momentum_arr = beta * momentum_arr + (1 - beta) * grad
        f_new = f - lr * momentum_arr
        f_new = project_simplex(f_new)

        f = f_new

        # Adaptive lr
        if no_improve > 100:
            lr *= 0.7
            if lr < 1e-8:
                rng = np.random.default_rng(it * 7 + 11)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                lr = 0.01 / n
                momentum_arr = np.zeros(n)
                no_improve = 0

    return best_f, 2 * n * best_obj


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
