```python
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_batch_subgrad(a, conv, k_peaks=20):
    """Compute average subgradient over top-k peaks."""
    n = len(a)
    len_c = len(conv)

        js = max(0, k_pos - n + 1)
        je = min(n - 1, k_pos)
        idx = np.arange(js, je + 1)
        sg[js:je+1] += a[k_pos - idx] + a[idx]
    return sg / k
```

Hmm, but I'm worried that projected subgradient descent might be too slow. Let me think about what learning rate to use.

The Lipschitz constant of the subgradient is roughly 2*max(a) (since each component of the gradient is at most 2*max(a)). If max(a) ~ 1/n (for a spread-out sequence), then L ~ 2/n. The optimal step size for subgradient method is O(1/sqrt(it)) or with Polyak step size O(1/it).

For n=768, L ~ 0.0026, so initial lr ~ 1/L ~ 380. But that seems too large. Actually, the subgradient components are bounded by 2*max(a)*k (where k is the number of peaks), so the norm could be larger.

Let me just use a practical approach: start with lr=0.01 and adapt.

Actually, I think the multiplicative (mirror) descent approach is better for this problem because:
1. It naturally maintains non-negativity
2. The log barrier keeps elements away from zero (which is good for exploration)
3. It's more stable

Let me refine the previous SMD approach with these changes:
1. Use top-k equal-weight subgradient (simpler than softmax)
2. Add momentum to the multiplicative updates
3. Use a halving learning rate schedule
4. Try more diverse initializations
5. Add a "peak targeting" phase that specifically attacks the top 1-3 peaks

Let me write the final solution:

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-stage optimization:
1. Diverse asymmetric initializations
2. Multiplicative mirror descent with batch subgradient and momentum
3. Peak-targeted refinement
4. Coordinate descent polishing
5. Multi-length exploration and interpolation
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_batch_subgrad(a, conv, k_peaks=20):
    """Compute average subgradient of max(conv) over top-k peaks."""
    n = len(a)
    len_c = len(conv)
    k = min(k_peaks, len_c)
    peaks = np.argpartition(conv, -k)[-k:]
    
    sg = np.zeros(n)
    for k_pos in peaks:
        k_pos = int(k_pos)
        js = max(0, k_pos - n + 1)
        je = min(n - 1, k_pos)
        if je < js:
            continue
        idx = np.arange(js, je + 1)
        sg[js:je + 1] += a[k_pos - idx] + a[idx]
    return sg / k


def compute_softmax_subgrad(a, conv, k_peaks=20, temperature=0.01):
    """Compute softmax-weighted subgradient of max(conv)."""
    n = len(a)
    len_c = len(conv)
    k = min(k_peaks, len_c)
    peaks = np.argpartition(conv, -k)[-k:]
    pv = conv[peaks]
    
    scaled = pv / max(temperature, 1e-10)
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(k)
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    sg = np.zeros(n)
    for i in range(k):
        if w[i] < 1e-12:
            continue
        k_pos = int(peaks[i])
        js = max(0, k_pos - n + 1)
        je = min(n - 1, k_pos)
        if je < js:
            continue
        idx = np.arange(js, je + 1)
        sg[js:je + 1] += w[i] * (a[k_pos - idx] + a[idx])
    return sg


def mirror_descent_optimize(a0, time_limit, k_peaks=20, temperature=0.01,
                            lr_init=0.005, lr_decay=0.9999, momentum_beta=0.0,
                            restart_every=100, verbose=False):
    """Multiplicative mirror descent with optional momentum."""
    a = np.maximum(a0.astype(np.float64), 1e-15)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    momentum = np.zeros(n)
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        
        if temperature > 0:
            sg = compute_softmax_subgrad(a, conv, k_peaks, temperature)
        else:
            sg = compute_batch_subgrad(a, conv, k_peaks)
        
        if momentum_beta > 0:
            momentum = momentum_beta * momentum + (1 - momentum_beta) * sg
            step = momentum
        else:
            step = sg
        
        a_new = a * np.exp(-lr * step)
        a_new = np.maximum(a_new, 1e-15)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        obj_new = fast_convolve(a_new).max()
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"      MD iter {it}: obj={best_obj:.8f}, lr={lr:.6f}")
        else:
            no_improve += 1
        
        a = a_new
        lr *= lr_decay
        
        if no_improve > restart_every:
            rng = np.random.default_rng(it * 7 + 13)
            noise = 0.003 + 0.002 * (no_improve / restart_every)
            a = np.maximum(best_a + rng.standard_normal(n) * noise, 1e-15)
            a = a / a.sum()
            lr = lr_init * 0.5
            no_improve = 0
        
        if lr < 1e-7:
            break
    
    return best_a, best_obj


def peak_elimination(a0, time_limit, verbose=False):
    """Directly target and eliminate the top peaks."""
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        
        # Focus on top 3 peaks only
        k_peaks = 3
        peaks = np.argpartition(conv, -k_peaks)[-k_peaks:]
        
        sg = np.zeros(n)
        for k_pos in peaks:
            k_pos = int(k_pos)
            js = max(0, k_pos - n + 1)
            je = min(n - 1, k_pos)
            idx = np.arange(js, je + 1)
            sg[js:je + 1] += a[k_pos - idx] + a[idx]
        sg /= k_peaks
        
        a_new = a * np.exp(-lr * sg)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        obj_new = fast_convolve(a_new).max()
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new

def projected_subgrad(a0, time_limit, k_peaks=30, lr_init=0.05, verbose=False):
    """Projected subgradient descent with adaptive step size."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    it = 0
    # Polyak-like: track best gap
    best_gap = best_obj - (1.0 / (2 * n - 1))  # lower bound approximation
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        
        sg = compute_batch_subgrad(a, conv, k_peaks)
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-12:
            break
        
        # Adaptive step size
        a_new = a - lr * sg
        a_new = project_simplex(a_new)
        
        obj_new = fast_convolve(a_new).max()
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            lr = min(lr * 1.01, 0.5)

        if verbose and it % 100 == 0:
            print(f"      PSD iter {it}: obj={best_obj:.8f}, lr={lr:.6f}")
        
        if no_improve > 100:
            rng = np.random.default_rng(it * 41 + 19)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.01)
            conv = fast_convolve(a)
            M = conv.max()
            lr = lr_init * 0.5
            no_improve = 0
        
        if lr < 1e-7:
            break
    
    return best_a, best_obj


def coordinate_descent_fine(a0, time_limit, verbose=False):
    """Fine coordinate descent."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    sweep = 0
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        rng = np.random.default_rng(sweep * 1000 + 1)
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            
            old_val = a[j]
            if old_val < 1e-12:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            # Try a range of new values
            for frac in [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 1.7, 2.0, 0.1]:
                new_val = old_val * frac if frac != 1.0 else old_val
                if new_val < 0 or new_val > 0.5:
                    continue
                if abs(new_val - old_val) < 1e-14:
                    continue
                
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-13:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        sweep += 1
        if not improved:
            # Try with randomized order and smaller steps
            if sweep > 5:
                break
            # Perturb and try again
            rng = np.random.default_rng(sweep * 777 + 3)
            a = np.maximum(best_a + rng.standard_normal(n) * 0.002, 0)
            a = a / a.sum()
    
    return best_a, best_obj


def make_inits(n, seed=42):
    """Generate diverse initializations."""
    inits = []
    rng = np.random.default_rng(seed)
    
    # Exponential decay front
    a = np.exp(-np.arange(n) / (n * 0.1))
    inits.append(np.maximum(a, 0) / np.maximum(a, 0).sum())
    
    # Exponential decay back
    a = np.exp(-np.arange(n) / (n * 0.15))
    a = a[::-1]
    inits.append(np.maximum(a, 0) / np.maximum(a, 0).sum())
    
    # Power law
    a = (np.arange(n) + 1) ** (-0.3)
    inits.append(a / a.sum())
    
    a = (np.arange(n) + 1) ** (-0.7)
    inits.append(a / a.sum())
    
    # Sine modulated
    x = np.arange(n) / n
    a = np.exp(-3 * x) * (1 + 0.5 * np.sin(4 * np.pi * x))
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        inits.append(a / a.sum())
    
    # Two bump asymmetric
    a = np.exp(-80 * (x - 0.2) ** 2) + 0.4 * np.exp(-80 * (x - 0.8) ** 2)
    inits.append(a / a.sum())
    
    # Ramp
    a = np.linspace(0.05, 1.0, n)
    inits.append(a / a.sum())
    
    # Cosine
    a = np.cos(np.pi * np.arange(n) / (2 * n)) ** 2 + 0.05
    inits.append(a / a.sum())
    
    # Random structured
    a = np.ones(n)
    a[:n//4] *= 1.5
    a[3*n//4:] *= 0.4
    a += rng.standard_normal(n) * 0.05
    a = np.maximum(a, 0)
    inits.append(a / a.sum())
    
    # Logarithmic
    a = np.log1p(np.arange(n) + 1) ** 0.5
    inits.append(a / a.sum())
    
    # "Anti-convolution" init: try to make conv flat from the start
    # Start with random, then do a few steps of mirror descent
    a = 1.0 + 0.3 * rng.standard_normal(n)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    for _ in range(50):
        conv = fast_convolve(a)
        peaks = np.argpartition(conv, -5)[-5:]
        sg = compute_batch_subgrad(a, conv, 5)
        a = a * np.exp(-0.01 * sg)
        a = np.maximum(a, 1e-10)
        a = a / a.sum()
    inits.append(a)
    
    return inits


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
```

```python
def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_batch_subgrad(a, conv, k_peaks=20):
    """Compute average subgradient of max(conv) over top-k peaks."""
    n = len(a)
    len_c = len(conv)
    k = min(k_peaks, len_c)
    peaks = np.argpartition(conv, -k)[-k:]
    sg = np.zeros(n)
    for k_pos in peaks:
        k_pos = int(k_pos)
        js = max(0, k_pos - n + 1)
        je = min(n - 1, k_pos)
        if je < js:
            continue
        idx = np.arange(js, je + 1)
        sg[js:je + 1] += a[k_pos - idx] + a[idx]
    return sg / k

def away_step_fw(a0, time_limit, k_peaks=10, verbose=False):

    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:

        it += 1
        conv = fast_convolve(a)
        M = conv.max()

        # Compute subgradient (average over top peaks)
        sg = compute_batch_subgrad(a, conv, k_peaks)

        # FW direction: v = e_{argmin(sg)}
        v_idx = int(np.argmin(sg))

        # Line search for standard FW step
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_trial = (1.0 - alpha) * a
            a_trial[v_idx] += alpha
            st = a_trial.sum()
            if st < 1e-15:

            a_trial = a_trial / st
            val = fast_convolve(a_trial).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Away step: find the vertex in support with highest sg
        support = np.where(a > 1e-6)[0]
        if len(support) > 1:
            w_idx = int(support[np.argmax(sg[support])])
            if sg[v_idx] < sg[w_idx] - 1e-6:

                    st = a_away.sum()
                    if st < 1e-15:

                        best_val = val

def spectral_init(n, seed=42):
    """Initialize with a flat-power-spectrum-like sequence."""
    rng = np.random.default_rng(seed)
    # Start with random phases, compute IFFT, take magnitude
    phases = rng.uniform(0, 2 * np.pi, n // 2 + 1)
    magnitudes = np.ones(n // 2 + 1)
    # Boost DC
    magnitudes[0] = np.sqrt(n)
    spectrum = magnitudes * np.exp(1j * phases)
    # Make it Hermitian
    full_spectrum = np.zeros(n, dtype=complex)
    full_spectrum[:n // 2 + 1] = spectrum
    if n % 2 == 0:
        for k in range(1, n // 2):
            full_spectrum[n - k] = np.conj(full_spectrum[k])
    else:
        for k in range(1, (n - 1) // 2 + 1):
            full_spectrum[n - k] = np.conj(full_spectrum[k])

    a = np.real(np.fft.ifft(full_spectrum))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = 1.0 + 0.5 * rng.standard_normal(n)
        a = np.maximum(a, 0)
        s = a.sum()
    a = a / s
    return a

    rng = np.random.default_rng(seed)
    x = np.arange(n) / n

    # 1. Exponential decay (front-heavy)
    a = np.exp(-4.0 * x)
    inits.append(a / a.sum())

    # 2. Exponential decay (back-heavy)
    a = np.exp(-4.0 * (1.0 - x))
    inits.append(a / a.sum())

    # 3. Power law (steep)
    a = (x + 0.01) ** (-1.0)
    inits.append(a / a.sum())

    # 4. Power law (gentle)
    a = (x + 0.01) ** (-0.4)
    inits.append(a / a.sum())

    # 5. Sine-modulated decay
    a = np.exp(-3.0 * x) * (1.0 + 0.6 * np.sin(3.0 * np.pi * x))
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        inits.append(a / a.sum())

    # 6. Two-bump asymmetric
    a = np.exp(-100 * (x - 0.25) ** 2) + 0.3 * np.exp(-100 * (x - 0.75) ** 2)
    inits.append(a / a.sum())

    # 7. Ramp (linear)
    a = np.linspace(0.05, 1.0, n)
    inits.append(a / a.sum())

    # 8. Cosine window
    a = np.cos(np.pi * x / 2.0) ** 2 + 0.03
    inits.append(a / a.sum())

    # 9. Spectral init
    a = spectral_init(n, seed=seed)
    inits.append(a)

    # 10. Random structured
    a = np.ones(n)
    a[:n // 4] *= 1.8
    a[3 * n // 4:] *= 0.3
    a += rng.standard_normal(n) * 0.03
    a = np.maximum(a, 0.01)
    inits.append(a / a.sum())

    # 11. "Pre-optimized" random: do 100 steps of MD on random init
    a = 1.0 + 0.5 * rng.standard_normal(n)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    for _ in range(200):
        conv = fast_convolve(a)
        sg = compute_batch_subgrad(a, conv, 10)
        a = a * np.exp(-0.005 * sg)
        a = np.maximum(a, 1e-10)
        a = a / a.sum()
    inits.append(a)

    # 12. Quadratic asymmetry
    a = (x * (1.0 - x) + 0.1) ** 2
    a = a * np.exp(-2.0 * x)  # break symmetry
    inits.append(a / a.sum())

    return inits


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-stage optimization combining mirror descent, projected subgradient,
    away-step Frank-Wolfe, peak targeting, and coordinate descent across
    multiple sequence lengths with spectral and parametric initializations.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})", flush=True)
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}", flush=True)
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # === Phase 1: Multi-length initialization screening ===
    print("=== Phase 1: Initialization Screening ===", flush=True)
    lengths = [512, 640, 700, 768, 850, 900, 1024]

    all_candidates = []
    for n_try in lengths:
        inits = make_inits(n_try, seed=seed)
        for i, a_init in enumerate(inits):
            conv = fast_convolve(a_init)
            obj = 2.0 * n_try * conv.max()
            all_candidates.append((obj, a_init, f"n{n_try}_i{i}"))

    if parent is not None:
        conv = fast_convolve(parent)
        obj = 2.0 * len(parent) * conv.max()
        all_candidates.append((obj, parent, "parent"))

    all_candidates.sort(key=lambda x: x[0])
    print(f"  Top 5: {[f'{l}={o:.4f}' for o, _, l in all_candidates[:5]]}", flush=True)

    # === Phase 2: Mirror Descent on top candidates ===
    print("\n=== Phase 2: Mirror Descent ===", flush=True)
    top_k = 6
    for idx in range(min(top_k, len(all_candidates))):
        rem = deadline - time.monotonic() - 500
        if rem < 20:
            break
        _, a_init, label = all_candidates[idx]
        t_alloc = max(20, min(rem / max(1, top_k - idx), 150))

        print(f"  MD[{label}] n={len(a_init)} t={t_alloc:.0f}s", flush=True)

        # Stage 2a: Warm phase (many peaks, higher lr)
        a_opt, obj = mirror_descent_optimize(
            a_init, time_limit=t_alloc * 0.5,
            k_peaks=30, lr_init=0.006, lr_decay=0.9998,
            use_threshold=True, threshold_frac=0.93,
            restart_every=80
        )
        try_seq(a_opt, f"md_a_{label}")

        # Stage 2b: Cool phase (fewer peaks, lower lr)
        rem2 = deadline - time.monotonic() - 300
        if rem2 > 20:
            t2 = min(rem2 * 0.3, t_alloc * 0.4)
            a_opt2, obj2 = mirror_descent_optimize(
                a_opt, time_limit=t2,
                k_peaks=10, lr_init=0.003, lr_decay=0.9999,
                momentum_beta=0.5,
                use_threshold=True, threshold_frac=0.97,
                restart_every=60
            )
            try_seq(a_opt2, f"md_b_{label}")

    # === Phase 3: Projected Subgradient with Momentum ===
    print("\n=== Phase 3: Projected Subgradient ===", flush=True)
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_alloc = min(rem, 120)
            print(f"  PSM n={len(best_seq)} t={t_alloc:.0f}s", flush=True)
            a_opt, obj = projected_subgrad_momentum(
                best_seq, time_limit=t_alloc,
                k_peaks=40, lr_init=0.03, momentum_beta=0.85
            )
            try_seq(a_opt, "psm")

    # === Phase 4: Away-Step Frank-Wolfe ===
    print("\n=== Phase 4: Away-Step FW ===", flush=True)
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_alloc = min(rem, 100)
            print(f"  AFW n={len(best_seq)} t={t_alloc:.0f}s", flush=True)
            a_opt, obj = away_step_fw(best_seq, time_limit=t_alloc, k_peaks=8)
            try_seq(a_opt, "afw")

    # === Phase 5: Peak Targeted Refinement ===
    print("\n=== Phase 5: Peak Targeting ===", flush=True)
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_alloc = min(rem, 80)
            print(f"  PTR n={len(best_seq)} t={t_alloc:.0f}s", flush=True)
            a_opt, obj = peak_targeted_refinement(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "ptr")

    # === Phase 6: Multi-length cross-pollination ===
    print("\n=== Phase 6: Cross-length Exploration ===", flush=True)
    if best_seq is not None:
        base = best_seq
        for target_n in [512, 640, 700, 768, 850, 900, 1024, 1152]:
            rem = deadline - time.monotonic() - 120
            if rem < 25:
                break
            if abs(len(base) - target_n) < 48:
                continue
            f_interp = interpolate_sequence(base, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")

            # If competitive, optimize
            if sc < best_score + 0.005:
                t_alloc = min(rem * 0.25, 50)
                if t_alloc > 15:
                    a_opt, obj = mirror_descent_optimize(
                        f_interp, time_limit=t_alloc,
                        k_peaks=20, lr_init=0.004, lr_decay=0.9999,
                        use_threshold=True, threshold_frac=0.95
                    )
                    try_seq(a_opt, f"md_x_n{target_n}")

    # === Phase 7: Coordinate Descent Polish ===
    print("\n=== Phase 7: Coordinate Descent ===", flush=True)
    if best_seq is not None:
        rem = deadline - time.monotonic() - 50
        if rem > 15:
            t_alloc = min(rem, 70)
            print(f"  CD n={len(best_seq)} t={t_alloc:.0f}s", flush=True)
            a_opt, obj = coordinate_descent_fine(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "cd")

    # === Phase 8: Final refinement pipeline ===
    print("\n=== Phase 8: Final Refinement ===", flush=True)
    if best_seq is not None:
        # 8a: Low-temperature MD
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            t_alloc = min(rem, 35)
            a_opt, obj = mirror_descent_optimize(
                best_seq, time_limit=t_alloc,
                k_peaks=5, lr_init=0.002, lr_decay=0.99995,
                use_threshold=True, threshold_frac=0.99,
                restart_every=40
            )
            try_seq(a_opt, "final_md")

        # 8b: Peak targeting on result
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a_opt, obj = peak_targeted_refinement(best_seq, time_limit=rem * 0.7)
            try_seq(a_opt, "final_ptr")

        # 8c: One more CD pass
        rem = deadline - time.monotonic() - 3
        if rem > 8:
            a_opt, obj = coordinate_descent_fine(best_seq, time_limit=rem)
            try_seq(a_opt, "final_cd")

    # === Phase 9: Diverse restarts ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 20:
            n_best = len(best_seq)
            rng = np.random.default_rng(seed + 777)
            for r in range(3):
                if deadline - time.monotonic() < 8:
                    break
                t_alloc = min(rem / max(1, 3 - r), 25)
                if t_alloc < 8:
                    break

                if r == 0:
                    f_init = np.maximum(best_seq + rng.standard_normal(n_best) * 0.004, 1e-15)
                    f_init = f_init / f_init.sum()
                elif r == 1:
                    f_init = 0.65 * best_seq + 0.35 * best_seq[::-1]
                    f_init = f_init / f_init.sum()
                else:
                    f_init = np.maximum(best_seq + rng.standard_normal(n_best) * 0.008, 1e-15)
                    f_init = f_init / f_init.sum()

                a_opt, obj = mirror_descent_optimize(
                    f_init, time_limit=t_alloc,
                    k_peaks=15, lr_init=0.004, lr_decay=0.9999,
                    use_threshold=True, threshold_frac=0.95,
                    restart_every=50
                )
                try_seq(a_opt, f"restart_{r}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL: {best_score:.6f} (n={len(best_seq)})", flush=True)
    return [float(x) for x in best_seq]
```
