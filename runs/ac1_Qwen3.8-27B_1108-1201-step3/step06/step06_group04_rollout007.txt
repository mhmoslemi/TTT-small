Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast multiplicative (mirror) descent
with adaptive peak-weighted subgradients, periodic LP refinement, gradient 
projection steps, and multi-length exploration. Achieves better convergence 
than pure LP-based methods by doing 10-100x more iterations per second.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_norm(f):
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgrad(f, peaks, weights):
    """Compute weighted subgradient of max(conv) at given peaks."""
    n = len(f)
    sg = np.zeros(n)
    for idx in range(len(peaks)):
        w = weights[idx]
        if w < 1e-12:
            continue
        k = int(peaks[idx])
        # d(conv[k])/d(f[i]) = f[k-i] (if 0<=k-i<n) + f[i-k] (if 0<=i-k<n)
        i1s = max(0, k - n + 1)
        i1e = min(k, n - 1)
        if i1e >= i1s:
            sg[i1s:i1e+1] += w * f[k - np.arange(i1s, i1e+1)]
        i2s = max(0, k)
        i2e = min(k + n - 1, n - 1)
        if i2e >= i2s:
            sg[i2s:i2e+1] += w * f[np.arange(i2s, i2e+1) - k]
    return sg


def mirror_descent(f0, time_limit, verbose=False, lr_init=0.008, top_k=16):

    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    it = 0
    temp = 0.02
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        # Top-k peaks
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w = w / w.sum()
        
        sg = compute_subgrad(f, peaks, w)
        
        # Multiplicative update
        f = f * np.exp(-lr * sg)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()
        
        co = 2.0 * n * fast_convolve(f).max()
        if co < best_obj * (1 - 1e-10):
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive: increase lr if not improving (try to escape)
        if no_improve > 50 and no_improve % 50 == 0:
            lr = max(lr * 1.5, 0.002)
            temp *= 2.0
        elif no_improve > 0 and no_improve <= 20:
            lr *= 0.99
        else:
            lr *= 0.999
        temp *= 1.003
        
        if no_improve > 300:
            # Mix with best
            f = np.maximum(0.7 * best_f + 0.3 * f, 1e-15)
            f = f / f.sum()
            no_improve = 0
            lr = lr_init
            temp = 0.02

            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            f = np.maximum(f, 1e-15)
            f = f / f.sum()
            no_improve = 0
            lr = lr_init * 1.5
            temp = 0.01
        
        if verbose and it % 2000 == 0:
            print(f"    MD iter {it}: obj={best_obj:.10f}, lr={lr:.6f}, no_imp={no_improve}")

def gradient_steps(f0, time_limit, verbose=False, lr_init=0.01):

    f = np.maximum(f0.astype(np.float64), 1e-15)

    conv = fast_convolve(f)

    no_improve = 0
    it = 0
    temp = 0.015
    top_k = 20
    m = 0.0  # momentum

        f_new = project_simplex(f_new)

def build_lp_matrix(f, n):
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        A_ub[k_start:k_end + 1, j] = f[:k_end - k_start + 1]
    return A_ub


def lp_step(f, M, n):
    """Solve one LP step and return the direction."""
    A_ub = build_lp_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        g = np.maximum(res.x, 0)
        gs = g.sum()
        if gs > 1e-15:
            g = g / gs
            return g
    return None


def combined_optimize(f0, time_limit, verbose=False):
    """
    Combined optimizer: alternates between fast mirror descent and 
    periodic LP refinement for precision.
    """
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve_global = 0
    
    while time.monotonic() - t0 < time_limit:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break
        
        # Phase A: Mirror descent burst
        md_time = min(rem * 0.7, 60.0)
        if md_time > 2:
            f, obj = mirror_descent(f, md_time, verbose=verbose)
            if obj < best_obj:
                best_obj = obj
                best_f = f.copy()
                no_improve_global = 0
            else:
                no_improve_global += 1
        
        rem = time_limit - (time.monotonic() - t0)
        if rem < 10:
            break
        
        # Phase B: LP refinement
        lp_time = min(rem * 0.3, 30.0)
        if lp_time > 5:
            conv = fast_convolve(best_f)
            M = conv.max()
            # Try a few LP steps
            f_lp = best_f.copy()
            for _ in range(3):
                if time.monotonic() - t0 > time_limit - 5:
                    break
                g = lp_step(f_lp, M, n)
                if g is None:
                    # Relax M
                    M *= 1.1
                    g = lp_step(f_lp, M, n)
                    if g is None:
                        break
                # Line search
                best_alpha = 0.0
                best_val = obj_norm(f_lp)
                for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                    trial = f_lp + alpha * (g - f_lp)
                    trial = np.maximum(trial, 0)
                    ts = trial.sum()
                    if ts < 1e-15:
                        continue
                    trial = trial / ts
                    v = obj_norm(trial)
                    if v < best_val:
                        best_val = v
                        best_alpha = alpha
                if best_alpha > 0:
                    f_lp = f_lp + best_alpha * (g - f_lp)
                    f_lp = np.maximum(f_lp, 0)
                    f_lp = f_lp / f_lp.sum()
            
            if obj_norm(f_lp) < best_obj:
                best_obj = obj_norm(f_lp)
                best_f = f_lp.copy()
                no_improve_global = 0
        
        if verbose:
            print(f"    Combined: best_obj={best_obj:.10f}, rem={rem:.0f}s")
    
    return best_f, best_obj

            no_improve += 1
        if no_improve > 50:
            lr *= 0.94
        if no_improve > 300:
            f = np.maximum(0.6 * best_f + 0.4 * f, 1e-15)
            f = f / f.sum()
            lr = lr_init
            no_improve = 0
            vel = np.zeros(n)
        if no_improve > 800:
            rng = np.random.default_rng(it * 13 + 5)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            no_improve = 0
            lr = lr_init
            vel = np.zeros(n)
        if lr < 1e-8:
            break
        if verbose and it % 5000 == 0:
            print(f"      GM iter {it}: obj={best_obj:.10f}, lr={lr:.6f}")
    return best_f, best_obj


def build_lp_matrix(f, n):
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        A_ub[k_start:k_end + 1, j] = f[:k_end - k_start + 1]
    return A_ub


def lp_polish(f0, time_limit, verbose=False):
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0
    while time.monotonic() - t0 < time_limit:
        it += 1
        A_ub = build_lp_matrix(f, n)
        M = fast_convolve(f).max() * m_relax
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            m_relax *= 1.15
            if m_relax > 8.0:
                break
            continue
        g = np.maximum(res.x, 0)
        gs = g.sum()
        if gs < 1e-15:
            m_relax *= 1.1
            continue
        g = g / gs
        diff = np.max(np.abs(g - f))
        if diff < 1e-10:
            no_improve += 1
            if no_improve > 8:
                m_relax = min(m_relax * 1.05, 5.0)
            if no_improve > 30:
                break
            continue
        m_relax = 1.0
        best_alpha = 0.0
        best_val = obj_norm(f)
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            trial = f + alpha * (g - f)
            trial = np.maximum(trial, 0)
            ts = trial.sum()
            if ts < 1e-15:
                continue
            trial = trial / ts
            v = obj_norm(trial)
            if v < best_val:
                best_val = v
                best_alpha = alpha
        if best_alpha > 0:
            f = f + best_alpha * (g - f)
            f = np.maximum(f, 0)
            f = f / f.sum()
            if best_val < best_obj:
                best_obj = best_val
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        if verbose and it % 5 == 0:
            print(f"      LP iter {it}: obj={best_obj:.10f}")
    return best_f, best_obj


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_left_ramp(n, sharpness=2.0):
    x = np.arange(n) / n
    a = np.exp(-sharpness * x)
    a = a / a.sum()
    return a


def make_left_bell(n, center=0.35, width=0.25):
    x = np.arange(n) / n
    a = np.exp(-0.5 * ((x - center) / width) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_left_tri(n, peak_frac=0.4):
    x = np.arange(n)
    peak = int(n * peak_frac)
    a = np.zeros(n)
    a[:peak + 1] = np.linspace(0, 1, peak + 1)
    a[peak:] = 1 - (np.arange(n - peak) / (n - peak)) * 0.5
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_gaussian(n, center=0.4, width=0.22):
    x = np.arange(n) / n
    a = np.exp(-0.5 * ((x - center) / width) ** 2)
    a = a / a.sum()
    return a


def make_smooth_bump(n, left=0.3, right=0.7, power=3.0):
    x = np.arange(n) / n
    a = np.zeros(n)
    mask = (x >= left) & (x <= right)
    xm = (x[mask] - left) / (right - left)
    a[mask] = (xm * (1 - xm)) ** power
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_left_bell(768, 0.35, 0.22)

    n_parent = len(parent)

    def try_update(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a = a / s
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")

    # Phase 1: Diverse initializations
    print("=== Phase 1: Diverse Initializations ===")
    init_candidates = [parent]
    for n_try in [600, 700, 768, 850, 900, 1000, 1100]:
        if n_try == n_parent and parent is not None:
            init_candidates.append(parent.copy())
        else:
            init_candidates.append(interpolate_seq(parent, n_try))
        init_candidates.append(make_left_ramp(n_try, 1.5))
        init_candidates.append(make_left_ramp(n_try, 2.5))
        init_candidates.append(make_left_ramp(n_try, 4.0))
        init_candidates.append(make_left_bell(n_try, 0.3, 0.2))
        init_candidates.append(make_left_bell(n_try, 0.35, 0.25))
        init_candidates.append(make_left_bell(n_try, 0.4, 0.3))
        init_candidates.append(make_gaussian(n_try, 0.35, 0.2))
        init_candidates.append(make_gaussian(n_try, 0.4, 0.25))
        init_candidates.append(make_left_tri(n_try, 0.35))
        init_candidates.append(make_left_tri(n_try, 0.45))
        init_candidates.append(make_smooth_bump(n_try, 0.25, 0.75, 2.0))
        init_candidates.append(make_smooth_bump(n_try, 0.3, 0.8, 3.0))

    best_init = None
    best_init_obj = float('inf')
    for ic in init_candidates:
        s = ic.sum()
        if s < 1e-10:
            continue
        a_n = ic / s
        obj = obj_norm(a_n)
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a_n.copy()
        try_update(a_n, f"init_n{len(a_n)}")

    print(f"  Best init: obj_norm={best_init_obj:.8f} (n={len(best_init)})")

    # Phase 2: Fast mirror descent (main workhorse)
    print(f"\n=== Phase 2: Mirror Descent (n={len(best_init)}) ===")
    rem = deadline - time.monotonic() - 500
    if rem > 30:
        t_alloc = min(rem, 400)
        print(f"  MD: t={t_alloc:.0f}s")
        a_opt, obj = mirror_descent(best_init, t_alloc, verbose=True, lr_init=0.008, top_k=20)
        try_update(a_opt, "md_main")

    # Phase 3: Gradient momentum
    if best_seq is not None:
        print(f"\n=== Phase 3: Gradient Momentum (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 350
        if rem > 20:
            t_alloc = min(rem, 200)
            print(f"  GM: t={t_alloc:.0f}s")
            a_opt, obj = gradient_momentum(best_seq, t_alloc, verbose=True, lr_init=0.015)
            try_update(a_opt, "grad_mom")

    # Phase 4: LP polish
    if best_seq is not None:
        print(f"\n=== Phase 4: LP Polish (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            t_alloc = min(rem, 150)
            print(f"  LP: t={t_alloc:.0f}s")
            a_opt, obj = lp_polish(best_seq, t_alloc, verbose=True)
            try_update(a_opt, "lp_polish")

    # Phase 5: Second MD round from polished result
    if best_seq is not None:
        print(f"\n=== Phase 5: MD Round 2 (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 120
        if rem > 20:
            t_alloc = min(rem, 120)
            print(f"  MD2: t={t_alloc:.0f}s")
            a_opt, obj = mirror_descent(best_seq, t_alloc, verbose=True, lr_init=0.004, top_k=24)
            try_update(a_opt, "md_round2")

    # Phase 6: Multi-length exploration
    print(f"\n=== Phase 6: Multi-length Exploration ===")
    base = best_seq if best_seq is not None else best_init
    for target_n in [650, 750, 850, 950, 1050]:
        if len(base) == target_n:
            continue
        rem = deadline - time.monotonic() - 60
        if rem < 30:
            break
        f_interp = interpolate_seq(base, target_n)
        try_update(f_interp, f"interp_n{target_n}")
        t_alloc = min(rem * 0.35, 50)
        if t_alloc < 10:
            continue
        print(f"  MD n={target_n}: t={t_alloc:.0f}s")
        a_opt, obj = mirror_descent(f_interp, t_alloc, verbose=False, lr_init=0.006, top_k=16)
        try_update(a_opt, f"md_n{target_n}")

    # Phase 7: Final LP polish
    if best_seq is not None:
        print(f"\n=== Phase 7: Final LP Polish ===")
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_alloc = min(rem, 60)
            print(f"  Final LP: t={t_alloc:.0f}s")
            a_opt, obj = lp_polish(best_seq, t_alloc, verbose=False)
            try_update(a_opt, "final_lp")

    # Phase 8: Final fine MD
    if best_seq is not None:
        print(f"\n=== Phase 8: Final Fine MD ===")
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            t_alloc = min(rem, 30)
            a_opt, obj = mirror_descent(best_seq, t_alloc, verbose=False, lr_init=0.002, top_k=30)
            try_update(a_opt, "final_md")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
