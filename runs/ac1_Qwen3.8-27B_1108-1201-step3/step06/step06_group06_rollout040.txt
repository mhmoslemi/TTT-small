def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_gradient(f, tau):
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    # h[j] = sum_m f[m] * w[m+j] via cross-correlation FFT
    total = 3 * n + 2
    fft_size = 1 << (total - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w, fft_size)
    # cross_corr[j] = sum_m f[m] * w[m+j] = ifft(FFT(f) * conj(FFT(w)))[j]
    corr = np.fft.irfft(F * np.conj(np.fft.fft(w, fft_size))[:fft_size // 2 + 1], fft_size)
    # Simpler: use full FFT
    F_full = np.fft.fft(f, fft_size)
    W_full = np.fft.fft(w, fft_size)
    corr = np.fft.ifft(F_full * np.conj(W_full)).real[:n]
    grad = (2.0 / tau) * corr
    return grad, conv


def nesterov_smooth_optimize(f0, time_limit, tau_init=None, lr_init=0.005, verbose=False):

def make_power_law(n, alpha, direction='end'):
    """Create a power-law sequence weighted toward one end."""
    if direction == 'end':
        a = np.arange(1, n + 1) ** alpha
    elif direction == 'start':
        a = np.arange(n, 0, -1) ** alpha
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_gaussian(n, center_frac=0.5, sigma_frac=0.25):
    x = np.arange(n) - n * center_frac
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_exponential(n, rate=2.0, direction='end'):
    x = np.arange(n, dtype=np.float64)
    if direction == 'end':
        a = np.exp(rate * x / n)
    else:
        a = np.exp(-rate * x / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
