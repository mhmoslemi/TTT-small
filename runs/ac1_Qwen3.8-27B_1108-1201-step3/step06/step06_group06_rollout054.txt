import numpy as np

import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def project_simplex_sym(v, n):
    """Project onto simplex restricted to symmetric sequences of length n."""
    half_n = (n + 1) // 2
    # Collapse to half: sum pairs
    v_half = np.zeros(half_n)
    for i in range(half_n):
        j = n - 1 - i
        if i == j:
            v_half[i] = v[i]
        else:
            v_half[i] = (v[i] + v[j]) / 2.0
    # Project v_half onto simplex with weights
    # A symmetric sequence a has sum = 2*sum(a[0:half_n]) - a[mid] if n odd
    # Let's just project and expand
    v_proj = project_simplex(v)
    # Symmetrize
    v_sym = (v_proj + v_proj[::-1]) / 2.0
    v_sym = np.maximum(v_sym, 0)
    s = v_sym.sum()
    if s > 1e-15:
        v_sym /= s
    return v_sym


def compute_smooth_grad(f, tau):
    """
    Compute gradient of F(a) = log(sum_k exp((a*a)[k]/tau)).
    Returns (gradient vector, convolution).
    """
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(len(conv)) / len(conv)
    else:
        w /= w_sum

    # h[j] = sum_l w[j+l] * f[l], computed via cross-correlation
    w_rev = w[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)
    h = conv_result[n - 1:2 * n - 1][::-1]

    grad = (2.0 / tau) * h
    return grad, conv


def adam_nesterov_optimize(f0, time_limit, tau_init=None, lr_init=0.003, verbose=False, use_nesterov=True):
    """
    Optimize smooth surrogate with Nesterov momentum and adaptive temperature.
    """

    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    M0 = conv.max()
    if tau_init is None:
        tau = 0.06 * M0
    else:
        tau = tau_init

    m = np.zeros(n)  # momentum
    v = np.zeros(n)  # RMS
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    t = 0
    y = f.copy()  # Nesterov lookahead

    best_obj = M0
    best_f = f.copy()
    no_improve = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        t += 1

        # Nesterov step
        if use_nesterov:
            grad, conv = compute_smooth_grad(y, tau)
            y_new = 0.5 * (f + y) - 0.5 * lr * grad / (np.sqrt(v + eps) + 1e-10)
            y = project_simplex(y_new)
        else:
            grad, conv = compute_smooth_grad(f, tau)
            y = f.copy()

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new

        # Adaptive temperature
        tau *= 0.9998
        if tau < 1e-8:
            tau = 1e-8

        # Adaptive learning rate
        if no_improve > 80:
            lr *= 0.88
        if no_improve > 350:
            rng = np.random.default_rng(t * 13 + 7)
            noise_scale = 0.001 + 0.004 * min(no_improve / 1000.0, 2.0)
            # Multiplicative perturbation
            pert = np.exp(rng.standard_normal(n) * noise_scale * 0.5)
            f = project_simplex(best_f * pert)
            m = np.zeros(n)
            v = np.zeros(n)
            y = f.copy()
            no_improve = 0
            lr = lr_init * 0.6
            tau = 0.04 * fast_convolve(f).max()

            print(f"    Adam iter {t}: obj={obj_score:.8f}, tau={tau:.2e}, lr={lr:.5f}")

def build_lp_matrix_block(f, k_start, k_end):
    """Build a block of the LP constraint matrix for rows k_start to k_end-1."""
    n = len(f)
    block = np.zeros((k_end - k_start, n))
    for row_idx, k in enumerate(range(k_start, k_end)):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_hi >= j_lo:
            block[row_idx, j_lo:j_hi + 1] = f[j_lo:j_hi + 1]
    return block


def solve_lp_direction(f, M, n, use_sparse=True):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns optimal g or None.
    """
    c = -np.ones(n)
    n_rows = 2 * n - 1

    # Build constraint matrix
    A_ub = np.zeros((n_rows, n))
    for k in range(n_rows):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_hi >= j_lo:
            A_ub[k, j_lo:j_hi + 1] = f[j_lo:j_hi + 1]

    b_ub = np.full(n_rows, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def solve_lp_direction_block(f, M, n):
    """Solve LP with the full constraint matrix. Optimized for repeated calls."""
    c = -np.ones(n)
    n_rows = 2 * n - 1

    # Build the banded constraint matrix
    A_ub = np.zeros((n_rows, n))
    for k in range(n_rows):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_hi >= j_lo:
            A_ub[k, j_lo:j_hi + 1] = f[j_lo:j_hi + 1]

    b_ub = np.full(n_rows, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 15})

    except Exception:
        pass
    return None


def lp_fw_optimize(f0, time_limit, verbose=False, use_symmetry=False, tight_factor=0.98):
    """
    LP-based Frank-Wolfe with adaptive constraint tightening.
    At each step: solve LP with M * tight_factor to force improvement.
    If infeasible, relax gradually.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0  # relaxation factor for M
    consecutive_relax = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0

        # Solve LP with tight constraint
        M_tight = M * tight_factor * m_relax
        g = solve_lp_direction_block(f, M_tight, n)

        if g is None:
            # Infeasible - relax
            m_relax *= 1.05
            consecutive_relax += 1
            if consecutive_relax > 10:
                # Give up on this direction, perturb
                m_relax = 1.0
                consecutive_relax = 0
                no_improve += 5
                if no_improve > 30:
                    rng = np.random.default_rng(it * 31 + 7)
                    pert = np.exp(rng.standard_normal(n) * 0.003)
                    f = project_simplex(best_f * pert)
                    conv = fast_convolve(f)
                    M = conv.max()
                    no_improve = 0
            continue

        consecutive_relax = max(0, consecutive_relax - 1)
        g_sum = g.sum()
        if g_sum < 1e-15:
            m_relax *= 1.1
            continue

        g = g / g_sum

            gs = g.sum()
            if gs > 1e-15:
                g /= gs

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            m_relax = min(m_relax * 1.02, 3.0)
            if no_improve > 40:
                rng = np.random.default_rng(it * 31 + 7)

                m_relax = 1.0
                no_improve = 0
            continue
        else:

        # Aggressive line search

        best_val = M

            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine bisection
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.08)
            hi = min(1.0, best_alpha + 0.08)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M_new = conv.max()
            f = f_new
            m_relax = 1.0

            if M_new < best_obj - 1e-14:
                best_obj = M_new
                best_f = f.copy()
                M = M_new
                if verbose and it % 10 == 0:
                    print(f"    LP-FW iter {it}: obj={2*n*best_obj:.8f} (t={elapsed:.0f}s)")

def peak_targeted_optimize(f0, time_limit, verbose=False):
    """
    Peak-targeted optimization: identify top peaks and use local
    quadratic approximation to find descent directions that reduce them.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        improved = False

        # Get top peaks
        n_peaks = min(8, 2 * n - 1)
        peak_indices = np.argsort(conv)[-n_peaks:]

        for pk in peak_indices:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            if je < js:
                continue

            idx = np.arange(js, je + 1)
            products = f[idx] * f[pk - idx]
            top_products = np.argsort(products)[-3:]

            for tp in top_products:
                j_max = idx[tp]
                j_other = pk - j_max

                if j_max == j_other:
                    # Diagonal: reduce f[j_max]^2
                    old_val = f[j_max]
                    if old_val < 1e-12:
                        continue
                    for factor in [0.4, 0.6, 0.75, 0.9]:
                        trial = f.copy()
                        trial[j_max] = old_val * factor
                        diff = old_val - trial[j_max]
                        # Redistribute proportionally to non-peak elements
                        mask = np.ones(n, dtype=bool)
                        for pk2 in peak_indices[:5]:
                            js2 = max(0, int(pk2) - n + 1)
                            je2 = min(n - 1, int(pk2))
                            mask[js2:je2 + 1] = False
                        mask[j_max] = False
                        if mask.sum() > 0 and f[mask].sum() > 1e-15:
                            trial[mask] += diff * f[mask] / f[mask].sum()
                        trial = np.maximum(trial, 0)
                        ss = trial.sum()
                        if ss > 1e-15:
                            trial /= ss
                        obj = fast_convolve(trial).max()
                        if obj < best_obj - 1e-14:
                            f = trial
                            best_obj = obj
                            best_f = f.copy()
                            improved = True
                            break
                else:
                    # Off-diagonal: reduce f[j_max]*f[j_other]
                    old_jm = f[j_max]
                    old_jo = f[j_other]
                    for sjm in [0.5, 0.7, 0.85]:
                        trial = f.copy()
                        trial[j_max] = old_jm * sjm
                        diff = old_jm - trial[j_max]
                        # Redistribute to elements not contributing to peaks
                        mask = np.ones(n, dtype=bool)
                        mask[j_max] = False
                        mask[j_other] = False
                        for pk2 in peak_indices[:5]:
                            js2 = max(0, int(pk2) - n + 1)
                            je2 = min(n - 1, int(pk2))
                            mask[js2:je2 + 1] = False
                        if mask.sum() > 0 and f[mask].sum() > 1e-15:
                            trial[mask] += diff * f[mask] / f[mask].sum()
                        trial = np.maximum(trial, 0)
                        ss = trial.sum()
                        if ss > 1e-15:
                            trial /= ss
                        obj = fast_convolve(trial).max()
                        if obj < best_obj - 1e-14:
                            f = trial
                            best_obj = obj
                            best_f = f.copy()
                            improved = True
                            break

        if not improved:
            no_improve += 1
            if no_improve > 12:
                # Larger perturbation
                rng = np.random.default_rng(it * 7 + int(time.monotonic() * 100) % 1000)
                pert = np.exp(rng.standard_normal(n) * 0.005)
                f = project_simplex(best_f * pert)
                no_improve = 0
                # Alternating: also try a structured perturbation
                if it % 3 == 0:
                    shift = rng.integers(1, max(2, n // 10))
                    f_shifted = np.zeros(n)
                    f_shifted[shift:] = f[:-shift]
                    f_shifted[:shift] = f[-shift:] * 0.9
                    f = project_simplex(0.7 * best_f + 0.3 * f_shifted)
        else:
            no_improve = 0

        if verbose and it % 20 == 0:
            print(f"    Peak-CD iter {it}: obj={2*n*best_obj:.8f}")

    return best_f, 2 * n * best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian_init(n, center=0.5, sigma=0.25):
    """Gaussian initialization."""
    x = np.arange(n) / n - center
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell_init(n, width=0.6):
    """Flat-topped bell."""
    x = np.arange(n) / n - 0.5
    half = width / 2.0
    t = np.clip(np.abs(x) / half, 0, 1.0)
    a = 1.0 - t ** 6
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_boundary_aware_init(n):
    """
    Initialization that accounts for boundary structure.
    The convolution (a*a)[0] = a[0]^2, so we want a[0] small.
    Similarly for a[n-1]. The "inner" values can be larger.
    """
    x = np.arange(n) / (n - 1)
    # Taper at boundaries: a[j] ~ min(j, n-1-j, 1) * bell
    taper = np.minimum(np.minimum(x, 1 - x) * n, 1.0)
    taper = np.clip(taper, 0.1, 1.0)
    # Inner bell
    center = n / 2.0
    sigma = n * 0.2
    bell = np.exp(-0.5 * (np.arange(n) - center) ** 2 / sigma ** 2)
    a = taper * bell
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def multiplicative_descent(f0, time_limit, verbose=False, lr=0.3):
    """
    Multiplicative weight / mirror descent approach.
    Update: a_j <- a_j * exp(-lr * grad_j) where grad_j from smooth approximation.
    This naturally maintains non-negativity.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    tau = 0.04 * M
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        grad, conv = compute_smooth_grad(f, tau)
        M = conv.max()

        # Multiplicative update
        # Scale gradient to reasonable range
        grad_scale = np.abs(grad).max()
        if grad_scale > 1e-15:
            grad_norm = grad / grad_scale
        else:
            grad_norm = grad

        f_new = f * np.exp(-lr * grad_norm)
        f_new = np.maximum(f_new, 0)
        ss = f_new.sum()
        if ss > 1e-15:
            f_new /= ss
        else:
            f_new = f.copy()

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Try smaller step
            for factor in [0.5, 0.2]:
                f_try = f * np.exp(-lr * factor * grad_norm)
                f_try = np.maximum(f_try, 0)
                ss_try = f_try.sum()
                if ss_try > 1e-15:
                    f_try /= ss_try
                    obj_try = fast_convolve(f_try).max()
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_f = f_try.copy()
                        f_new = f_try
                        no_improve = 0
                        break

        f = f_new

        # Adaptive temperature
        tau *= 0.9995
        if tau < 1e-9:
            tau = 1e-9

        # Adaptive lr
        if no_improve > 50:
            lr *= 0.9
            if lr < 0.01:
                break

        if no_improve > 200:
            rng = np.random.default_rng(it * 11 + 3)
            pert = np.exp(rng.standard_normal(n) * 0.003)
            f = project_simplex(best_f * pert)
            no_improve = 0
            lr = min(lr * 2, 0.3)
            tau = 0.03 * fast_convolve(f).max()

        if verbose and it % 100 == 0:
            print(f"    MulDescent iter {it}: obj={2*n*best_obj:.8f}")

    return best_f, 2 * n * best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization for minimum max-convolution ratio.
    
    Phases:
    1. Initialization & evaluation (15s)
    2. Adam Nesterov warmup (200s)
    3. LP-Frank-Wolfe main (500s)
    4. Peak-targeted CD (150s)
    5. Multiplicative descent (100s)
    6. Multi-length exploration (300s)
    7. Final LP-FW polish (remaining time)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
        return score

    # === Phase 1: Initialization ===
    print("=== Phase 1: Initialization ===")

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_bell_init(768, 0.6)
        try_seq(parent, "default_init")

    # Diverse initializations
    inits = []
    for n_try in [512, 768, 1024]:
        inits.append((n_try, make_gaussian_init(n_try, 0.5, 0.22), "gauss"))
        inits.append((n_try, make_bell_init(n_try, 0.55), "bell"))
        inits.append((n_try, make_boundary_aware_init(n_try), "boundary"))
        # Exponential
        a_exp = np.exp(-np.arange(n_try) / (n_try * 0.35))
        a_exp /= a_exp.sum()
        inits.append((n_try, a_exp, "exp"))
        # Symmetric Gaussian (narrower)
        inits.append((n_try, make_gaussian_init(n_try, 0.5, 0.15), "narrow_gauss"))

    t_init_limit = time.monotonic() + 12
    for i, (n_try, a_init, name) in enumerate(inits):
        if time.monotonic() > t_init_limit:
            break
        try_seq(a_init, f"init_{name}_{n_try}")

    # Pick best starting point
    work_seq = best_seq if best_seq is not None else parent
    work_n = len(work_seq)
    print(f"  Best init: n={work_n}, score={best_score:.6f}")

    # === Phase 2: Adam Nesterov Warmup ===
    print(f"\n=== Phase 2: Adam Nesterov (n={work_n}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem, 200)
    if t2 > 30:
        print(f"  Time allocated: {t2:.0f}s")
        # First Adam run
        a_opt, obj = adam_nesterov_optimize(work_seq, time_limit=t2 * 0.55,
                                            lr_init=0.004, use_nesterov=True, verbose=True)
        try_seq(a_opt, "adam_nested")

        # Second Adam run from result with different params
        if time.monotonic() < deadline - 900:
            rem2 = deadline - time.monotonic() - 900
            t2b = min(rem2, 60)
            if t2b > 20:
                a_opt2, obj2 = adam_nesterov_optimize(best_seq, time_limit=t2b,
                                                      tau_init=0.03 * fast_convolve(best_seq).max(),
                                                      lr_init=0.002, use_nesterov=False, verbose=False)
                try_seq(a_opt2, "adam_fine")

    # === Phase 3: LP-Frank-Wolfe Main ===
    print(f"\n=== Phase 3: LP-FW Main ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 500
        if rem > 60:
            t3 = min(rem, 500)
            print(f"  Time allocated: {t3:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t3 * 0.7,
                                        verbose=True, use_symmetry=False, tight_factor=0.97)
            try_seq(a_opt, "lpfw_main")

            # Second LP-FW with symmetry
            if time.monotonic() < deadline - 400:
                rem2 = deadline - time.monotonic() - 400
                t3b = min(rem2, 130)
                if t3b > 30:
                    # Symmetrize best
                    sym_best = (best_seq + best_seq[::-1]) / 2.0
                    sym_best = np.maximum(sym_best, 0)
                    sym_best /= sym_best.sum()
                    a_opt2, obj2 = lp_fw_optimize(sym_best, time_limit=t3b,
                                                  verbose=True, use_symmetry=True, tight_factor=0.95)
                    try_seq(a_opt2, "lpfw_sym")

    # === Phase 4: Peak-Targeted CD ===
    print(f"\n=== Phase 4: Peak-Targeted CD ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 40:
            t4 = min(rem, 150)
            print(f"  Time allocated: {t4:.0f}s")
            a_opt, obj = peak_targeted_optimize(best_seq, time_limit=t4, verbose=True)
            try_seq(a_opt, "peak_cd")

    # === Phase 5: Multiplicative Descent ===
    print(f"\n=== Phase 5: Multiplicative Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 30:
            t5 = min(rem, 100)
            print(f"  Time allocated: {t5:.0f}s")
            a_opt, obj = multiplicative_descent(best_seq, time_limit=t5, verbose=True)
            try_seq(a_opt, "mul descent")

    # === Phase 6: Multi-Length Exploration ===
    print(f"\n=== Phase 6: Multi-Length Exploration ===")
    if best_seq is not None:
        base = best_seq
        target_lengths = [int(base.size * 1.33), int(base.size * 0.75), 1024, 1536]
        # Remove duplicates and out-of-range
        target_lengths = [t for t in target_lengths if 256 <= t <= 3000]
        target_lengths = list(dict.fromkeys(sorted(target_lengths)))  # deduplicate, keep order

        for target_n in target_lengths[:4]:
            rem = deadline - time.monotonic() - 80
            if rem < 40:
                break
            if target_n == len(base):
                continue
            f_interp = interpolate_sequence(base, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")

            # Quick optimization on interpolated
            t6 = min(rem * 0.35, 80)
            if t6 > 15:
                a_opt, obj = adam_nesterov_optimize(f_interp, time_limit=t6 * 0.6,
                                                    lr_init=0.003, use_nesterov=True, verbose=False)
                try_seq(a_opt, f"adam_n{target_n}")
                # Quick LP-FW
                rem6 = deadline - time.monotonic() - 60
                if rem6 > 20:
                    t6b = min(rem6 * 0.4, 30)
                    if t6b > 10:
                        a_opt2, obj2 = lp_fw_optimize(a_opt, time_limit=t6b, verbose=False)
                        try_seq(a_opt2, f"lpfw_n{target_n}")

    # === Phase 7: Final Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 20:
            t7 = min(rem, 80)
            # LP-FW final
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t7 * 0.6,
                                        verbose=False, tight_factor=0.96)
            try_seq(a_opt, "final_lpfw")

            rem2 = deadline - time.monotonic() - 10
            if rem2 > 15:
                # Adam final
                a_opt2, obj2 = adam_nesterov_optimize(best_seq, time_limit=min(rem2, 30),
                                                      tau_init=0.015 * fast_convolve(best_seq).max(),
                                                      lr_init=0.001, use_nesterov=True, verbose=False)
                try_seq(a_opt2, "final_adam")

    # === Phase 8: Random restart attempts ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 20:
            n_restarts = min(5, int(rem / 10))
            for r in range(n_restarts):
                if time.monotonic() > deadline:
                    break
                perturbation_sizes = [0.005, 0.01, 0.02, 0.015, 0.008]
                ps = perturbation_sizes[r % len(perturbation_sizes)]
                pert = np.exp(rng.standard_normal(len(best_seq)) * ps)
                f_pert = project_simplex(best_seq * pert)
                sc = try_seq(f_pert, f"restart_{r}")
                # Very quick LP-FW
                rem_r = deadline - time.monotonic() - 3
                if rem_r > 8 and sc < best_score + 0.01:
                    a_opt, obj = lp_fw_optimize(f_pert, time_limit=min(rem_r, 15), verbose=False)
                    try_seq(a_opt, f"restart_lpfw_{r}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
