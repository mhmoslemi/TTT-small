def lse_objective(c, beta):
    """Log-sum-exp approximation to max(c)."""
    m = c.max()
    if m < 1e-30:
        return 0.0, np.full(len(c), 1.0 / len(c))
    scaled = beta * (c - m)
    exp_s = np.exp(scaled)
    w = exp_s / exp_s.sum()
    return m + np.log(exp_s.sum()) / beta, w

def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_hann(n):
    x = np.linspace(-1, 1, n)
    a = 0.5 * (1 + np.cos(np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, power=4):
    x = np.linspace(-1, 1, n)
    a = (1 - np.abs(x)) ** power
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_to_length(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    s = f_interp.sum()
    if s > 1e-15:
        f_interp /= s
    return f_interp

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Min-max LP based Frank-Wolfe (primary, finds direction minimizing peak of f*g)
2. LSE-smoothed gradient descent (backup for diversity)

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_constraint_matrix(f, n):
    """Build (2n-1) x n matrix where A[k,j] = f[k-j] if 0<=k-j<n else 0."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_minmax_lp(f, n):
    """
    Solve: min t s.t. (f*g)_k <= t for all k, sum(g)=1, g>=0.
    Returns (g_opt, t_opt) or (None, None).
    """
    A = build_constraint_matrix(f, n)
    # Variables: [g_0, ..., g_{n-1}, t]
    # Objective: minimize t -> c = [0,...,0,1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    # Inequality: A @ g <= t -> A @ g - t <= 0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    # Bounds: g_j >= 0, t free (but t >= 0 implicitly)
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None


def solve_fw_lp(f, M, n):
    """
    Solve: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0.
    Returns g or None.
    """
    A = build_constraint_matrix(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lse_weights(c, beta):
    """Compute softmax(beta*c) weights for LSE approximation."""
    m = c.max()
    if m < 1e-30:
        return np.full(len(c), 1.0 / len(c))
    scaled = beta * (c - m)
    exp_s = np.exp(scaled)
    s = exp_s.sum()
    if s < 1e-300:
        return np.full(len(c), 1.0 / len(c))
    return exp_s / s


def lse_gradient(f, w):
    """
    Gradient of LSE_beta(conv(f,f)) w.r.t. f.
    grad_j = sum_k w_k * f_{k-j} + w_{2j} * f_j
    """
    n = len(f)
    # Cross-correlation part: sum_k w_k * f_{k-j}
    f_flipped = f[::-1]
    conv_len = len(w) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)[:conv_len]
    indices = np.arange(n - 1, 2 * n - 1)
    grad = conv[indices].copy()
    # Special term: w_{2j} * f_j
    grad += w[2 * np.arange(n)] * f
    return grad

        g = None
        if use_minmax:
            g, t_pred = solve_minmax_lp(f, n)
            if g is None:
                use_minmax = False
            if g is not None:
                gs = g.sum()
                if gs < 1e-15:
                    g = None
                else:
                    g = g / gs
        if g is None:
            # Fallback to standard FW LP
            M_mult = 1.0 + 0.01 * min(1.0, no_improve / 50.0)
            g = solve_fw_lp(f, M * M_mult, n)
            if g is None:
                no_improve += 1
                if no_improve > 50:
                    rng = np.random.default_rng(it * 13 + 7)
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                    c = fast_convolve(f)
                    M = c.max()
                    no_improve = 0
                continue
            gs = g.sum()
            if gs < 1e-15:
                no_improve += 1
                continue
            g = g / gs

        # Check convergence
        diff = np.max(np.abs(g - f))
        if diff < 1e-11:
            no_improve += 1
            if no_improve > 30:
                rng = np.random.default_rng(it * 19 + 3)
                scale = 0.001 * (1 + no_improve / 30.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                c = fast_convolve(f)
                M = c.max()
                no_improve = 0
            continue
        else:
            no_improve = 0

        # Line search
        best_t = 0.0
        best_val = M

        # Quick checks at common step sizes
        for t_candidate in [0.3, 0.5, 0.2, 0.4, 0.6, 0.15, 0.7, 1.0, 0.1]:
            ft = (1.0 - t_candidate) * f + t_candidate * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_t = t_candidate

        # Bisection refinement around best_t
        if best_t > 0:
            lo = max(0.0, best_t - 0.04)
            hi = min(1.0, best_t + 0.04)
            for _ in range(12):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_t = mid
                    lo = mid
                else:
                    hi = mid

        if best_t > 1e-10:
            f_new = (1.0 - best_t) * f + best_t * g
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            M = fast_convolve(f_new).max()
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 100 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.12f}, t={best_t:.4f}, time={time.monotonic()-t0:.1f}s")
        else:
            no_improve += 1

        if no_improve > 500:
            break

    return best_f, best_obj


def smoothed_gd_main(f0, time_limit, verbose=False):
    """LSE-smoothed gradient descent with momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    c = fast_convolve(f)
    best_obj = c.max()

    t0 = time.monotonic()
    momentum = np.zeros(n)
    mu = 0.85
    lr = 0.02

    betas = [4, 8, 16, 32, 64, 128, 256, 512]
    for beta in betas:
        if time.monotonic() - t0 > time_limit:
            break
        stage_start = time.monotonic()
        stage_time = min(100, (time_limit - (time.monotonic() - t0)) * 0.15)
        it = 0
        no_improve = 0

        while time.monotonic() - stage_start < stage_time and time.monotonic() - t0 < time_limit * 0.92:
            it += 1
            c = fast_convolve(f)
            w = lse_weights(c, beta)
            g = lse_gradient(f, w)

            g_norm = np.linalg.norm(g)
            if g_norm < 1e-14:
                break

            lr_use = lr / (g_norm + 1e-10)
            momentum = mu * momentum - lr_use * g
            f_trial = f + momentum
            f_trial = project_simplex(f_trial)

            c_trial = fast_convolve(f_trial)
            obj_trial = c_trial.max()

            if obj_trial <= c.max() + 1e-12:
                f = f_trial
                if obj_trial < best_obj - 1e-13:
                    best_obj = obj_trial
                    best_f = f.copy()
                    no_improve = 0
                    lr = min(lr * 1.01, 0.08)
                else:
                    no_improve += 1
            else:
                no_improve += 1
                lr *= 0.92
                momentum *= 0.5

            if no_improve > 60:
                rng = np.random.default_rng(it * 7 + int(beta))
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                momentum = np.zeros(n)
                lr = 0.02
                no_improve = 0

        if verbose:
            print(f"  LSE-GD beta={beta}: best={best_obj:.10f}, time={time.monotonic()-t0:.1f}s")

    return best_f, best_obj


def coord_descent_fast(f0, time_limit, verbose=False):
    """
    Coordinate descent with FFT-based incremental updates.
    When changing a_j by delta, the convolution changes by:
    c_new[k] = c[k] + 2*delta*f[k-j] - delta^2 * [k==2j]
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f0.astype(np.float64), fast_convolve(f0).max()
    f = f / s
    n = len(f)

    c = fast_convolve(f)
    M = c.max()
    best_f = f.copy()
    best_obj = M
    t0 = time.monotonic()

    # Precompute the shift ranges for each j
    while time.monotonic() - t0 < time_limit:
        improved = False
        # Shuffle coordinate order for diversity
        order = np.random.default_rng(int(time.monotonic() * 1000) % 10000).permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-14:
                continue

            best_new_val = old_val
            best_new_obj = M

            # Try several multipliers
            for mult in [0.6, 0.8, 0.9, 1.1, 1.2, 1.4]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.4:
                    continue
                delta = new_val - old_val

                # Compute new convolution peak using incremental update
                # c_new[k] = c[k] + 2*delta*f[k-j] - delta^2*[k==2j]
                # We only need the max, so compute incrementally
                # The change is: 2*delta * f shifted by j, minus delta^2 at 2j
                # The new values in the affected range:
                js = j  # k starts at j (k-j >= 0)
                je = min(2 * n - 2, j + n - 1)  # k-j <= n-1

                if js > je:
                    continue

                affected = c[js:je + 1] + 2.0 * delta * f[:je - js + 1]
                peak_affected = affected.max()

                # Also check k=2j for the delta^2 term
                if 2 * j <= 2 * n - 2:
                    val_2j = c[2 * j] + 2.0 * delta * f[j] - delta * delta
                    peak_affected = max(peak_affected, val_2j)

                # Check unaffected part (should be unchanged, but verify max)
                # The unaffected max is max(c) excluding the affected range
                if js > 0:
                    left_max = c[:js].max()
                else:
                    left_max = -np.inf
                if je < 2 * n - 2:
                    right_max = c[je + 1:].max()
                else:
                    right_max = -np.inf
                unaffected_max = max(left_max, right_max)
                if 2 * j < js or 2 * j > je:
                    unaffected_max = max(unaffected_max, c[2 * j])

                new_peak = max(peak_affected, unaffected_max)

                if new_peak < best_new_obj - 1e-14:
                    best_new_obj = new_peak
                    best_new_val = new_val

            if best_new_val != old_val:
                f[j] = best_new_val
                f = f / f.sum()
                c = fast_convolve(f)  # Full recompute for accuracy
                M = c.max()
                if M < best_obj - 1e-14:
                    best_obj = M
                    best_f = f.copy()
                    improved = True

        if not improved:
            # Try a different order
            c = fast_convolve(f)
            M = c.max()
            if M >= best_obj - 1e-13:
                break

    return best_f, best_obj


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian-shaped initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a
