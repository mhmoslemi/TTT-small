def fast_convolve(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def make_theoretical_init(n):
    """a[k] proportional to C(2k,k)/4^k, normalized."""
    k = np.arange(n)
    # C(2k,k) / 4^k = prod_{i=1}^k (k+i)/(4*i)
    # Compute iteratively to avoid overflow
    c = np.ones(n)
    for i in range(1, n):

        # Actually: C(2i,i)/4^i = C(2i-2,i-1)/4^(i-1) * (2i)(2i-1)/(i^2 * 4)
        # = c[i-1] * (2i*(2i-1))/(4*i*i) = c[i-1] * (2i-1)/(2*i)
    # Let me redo: c[i] = C(2i, i) / 4^i
    # c[0] = 1
    # c[i] = c[i-1] * (2i)(2i-1) / (i*i * 4) = c[i-1] * (2i-1)/(2i)

    if s < 1e-15:

    return c / s

def conv_grad_at_k(a, k):
    """Compute d(conv(a,a)[k])/d(a) as a vector."""
    n = len(a)
    grad = np.zeros(n)
    # For each m: if 0 <= k-m < n:
    #   grad[m] += 2 * a[k-m] if 2m != k
    #   grad[m] = 2 * a[m] if 2m == k
    m_start = max(0, k - n + 1)
    m_end = min(n - 1, k)
    if m_start > m_end:
        return grad
    ms = np.arange(m_start, m_end + 1)
    others = k - ms
    # For all m in range, grad[m] += 2 * a[k-m] (if m != k-m)
    # For m where 2m == k, grad[m] = 2 * a[m] (which is the same as 2*a[k-m] since k-m=m)
    # So actually: grad[m] = 2 * a[k-m] for all valid m
    # (When 2m=k, a[k-m] = a[m], so it's 2*a[m], which is correct.)
    grad[ms] = 2.0 * a[others]
    return grad

Wait, I realize there's an issue. Let me re-derive.

conv[k] = sum_j a[j] * a[k-j] (over valid j)

d(conv[k])/d(a[m]) = ?

The terms in the sum that involve a[m] are:
- j = m: term is a[m]*a[k-m], partial derivative = a[k-m] (valid if 0<=k-m<n)
- j = k-m: term is a[k-m]*a[m], partial derivative = a[k-m] (valid if 0<=k-m<n, which is the same condition)

If m != k-m (i.e., 2m != k): these are two different terms, so total derivative = a[k-m] + a[k-m] = 2*a[k-m]
If m == k-m (i.e., 2m = k): these are the same term a[m]^2, derivative = 2*a[m] = 2*a[k-m] (since k-m=m)

So in all cases: d(conv[k])/d(a[m]) = 2*a[k-m] (if 0<=k-m<n and 0<=m<n)

This is much simpler than I thought! The gradient is just 2*a reversed and shifted.

So conv_grad_at_k(a, k)[m] = 2 * a[k-m] for m in [max(0, k-n+1), min(n-1, k)], and 0 otherwise.

This means the gradient at peak position k is basically a "reversed and shifted" copy of a, scaled by 2.

This is great because it means computing the gradient is O(n) and very fast.

Now, for the "sum of top K" subgradient:
subgrad = sum_{k in topK} conv_grad_at_k(a, k)

This is O(n*K) which is fine.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Theoretical kernel initialization (truncated (1-x)^{-1/2})
2. Adam optimization with sum-of-top-K subgradient
3. Active-set fine-tuning with exact max subgradient
4. Multi-start diversification
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_theoretical_init(n):
    """a[k] proportional to C(2k,k)/4^k, normalized to sum=1."""
    c = np.ones(n)
    for i in range(1, n):
        c[i] = c[i - 1] * (2 * i - 1) / (2 * i)
    s = c.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return c / s


def conv_grad_at_k(a, k):
    """Gradient of conv(a,a)[k] w.r.t. a. Result is 2*a[k-m] for valid m."""
    n = len(a)
    grad = np.zeros(n)
    m_start = max(0, k - n + 1)
    m_end = min(n - 1, k)
    if m_start > m_end:
        return grad
    ms = np.arange(m_start, m_end + 1)
    grad[ms] = 2.0 * a[k - ms]
    return grad


def topk_subgradient(a, K):
    """Subgradient of sum of top K convolution values."""
    conv = fast_convolve(a)
    topk_idx = np.argsort(conv)[-K:]
    grad = np.zeros(len(a))
    for k in topk_idx:
        k = int(k)
        grad += conv_grad_at_k(a, k)
    return grad, conv


def adam_topk_optimize(f0, time_limit, K=5, lr_init=0.01, verbose=False):
    """
    Adam optimizer minimizing sum of top K convolution values.
    f is normalized (sum=1, non-negative).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    t = 0

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    no_improve = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        t += 1
        grad, conv = topk_subgradient(f, K)
        M = conv.max()

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new

        # Adaptive: reduce lr if stuck
        if no_improve > 50:
            lr *= 0.9
        if no_improve > 200:
            # Restart from best with perturbation
            rng = np.random.default_rng(t * 7 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = lr_init * 0.5

        if verbose and t % 200 == 0:
            print(f"    Adam-TopK iter {t}: score={2*n*best_obj:.8f}, lr={lr:.2e}")

    return best_f, 2 * n * best_obj


def active_set_fine_tune(f0, time_limit, verbose=False):
    """
    Fine-tuning using exact max subgradient with adaptive step size.
    Focus on reducing the single peak (or top 1-3 peaks).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    step_size = 0.01
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        # Find the top 1-3 peaks
        top_k = min(3, len(conv))
        peak_idx = np.argsort(conv)[-top_k:]
        
        # Use subgradient of sum of top peaks
        grad = np.zeros(n)
        for pk in peak_idx:
            pk = int(pk)
            grad += conv_grad_at_k(f, pk)
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Projected gradient step
        # direction = -grad, projected to sum=0 (to maintain sum=1)
        direction = -grad
        direction -= direction.mean()  # Make sum zero
        
        # Try different step sizes
        best_alpha = 0.0
        best_val = M
        for alpha in [step_size, step_size*0.5, step_size*2, step_size*0.1]:
            f_trial = f + alpha * direction
            f_trial = np.maximum(f_trial, 0)
            st = f_trial.sum()
            if st < 1e-15:
                continue
            f_trial = f_trial / st
            val = fast_convolve(f_trial).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 0:
            f_new = f + best_alpha * direction
            f_new = np.maximum(f_new, 0)
            f_new = f_new / f_new.sum()
            f = f_new
            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            step_size *= 0.5
        
        if no_improve > 0 and no_improve % 10 == 0:
            step_size *= 0.8
        
        if verbose and it % 100 == 0:
            print(f"    ActiveSet iter {it}: score={2*n*best_obj:.8f}, step={step_size:.2e}")
        
        if no_improve > 500:
            # Try a perturbation
            rng = np.random.default_rng(it * 13)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.0005)
            no_improve = 0
            step_size = 0.01

    return best_f, 2 * n * best_obj


def coordinate_descent_fine(f0, time_limit, verbose=False):
    """
    Coordinate descent: adjust pairs of elements to reduce the peak.
    For the peak at position k, the most contributing pair is (j, k-j)
    where a[j]*a[k-j] is maximized. We transfer mass from this pair
    to less-contributing pairs.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        peak_k = int(np.argmax(conv))
        
        # Find the most contributing pair for the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        products = f[idx] * f[peak_k - idx]
        j_star = idx[np.argmax(products)]
        j_other = peak_k - j_star
        
        improved = False
        if j_star != j_other and f[j_star] > 1e-10 and f[j_other] > 1e-10:
            # Transfer mass from (j_star, j_other) to other positions
            # Reduce both by a small amount
            for frac in [0.01, 0.005, 0.001, 0.05]:
                delta = min(f[j_star], f[j_other]) * frac
                trial = f.copy()
                trial[j_star] -= delta
                trial[j_other] -= delta
                # Redistribute to positions that don't contribute to the peak
                # (or contribute less)
                contrib_mask = np.ones(n, dtype=bool)
                contrib_mask[js:je+1] = False  # Don't add to peak-contributing positions
                if contrib_mask.sum() > 0 and trial[contrib_mask].sum() > 0:
                    # Add proportionally to existing values
                    weights = trial[contrib_mask]
                    weights = weights / weights.sum()
                    trial[contrib_mask] += 2 * delta * weights
                trial = np.maximum(trial, 0)
                trial = trial / trial.sum()
                obj = fast_convolve(trial).max()
                if obj < best_obj - 1e-12:
                    f = trial
                    best_obj = obj
                    best_f = f.copy()
                    improved = True
                    break
        elif j_star == j_other and f[j_star] > 1e-10:
            # Diagonal: reduce a[j_star]^2
            for frac in [0.01, 0.005, 0.001]:
                delta = f[j_star] * frac
                trial = f.copy()
                trial[j_star] -= delta
                # Redistribute
                mask = np.ones(n, dtype=bool)
                mask[js:je+1] = False
                if mask.sum() > 0 and trial[mask].sum() > 0:
                    weights = trial[mask]
                    weights = weights / weights.sum()
                    trial[mask] += delta * weights
                trial = np.maximum(trial, 0)
                trial = trial / trial.sum()
                obj = fast_convolve(trial).max()
                if obj < best_obj - 1e-12:
                    f = trial
                    best_obj = obj
                    best_f = f.copy()
                    improved = True
                    break
        
        if not improved:
            no_improve += 1
            if no_improve > 30:
                # Try a random pair transfer
                rng = np.random.default_rng(it * 31 + 7)
                i1 = rng.integers(n)
                i2 = rng.integers(n)
                if i1 != i2 and f[i1] > 1e-8 and f[i2] > 1e-8:
                    for frac in [0.005, 0.001]:
                        delta = min(f[i1], f[i2]) * frac
                        trial = f.copy()
                        trial[i1] -= delta
                        trial[i2] += delta  # Transfer from i1 to i2
                        trial = np.maximum(trial, 0)
                        trial = trial / trial.sum()
                        obj = fast_convolve(trial).max()
                        if obj < best_obj - 1e-12:
                            f = trial
                            best_obj = obj
                            best_f = f.copy()
                            improved = True
                            break
                no_improve = 0
        
        if verbose and it % 200 == 0:
            print(f"    CD iter {it}: score={2*n*best_obj:.8f}")

    return best_f, 2 * n * best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
