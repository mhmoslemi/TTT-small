import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_smooth_grad(a, conv, T):
    """Compute gradient of T*log(sum(exp(conv/T))) with respect to a."""
    n = len(a)
    c_max = conv.max()
    scaled = (conv - c_max) / T
    scaled = np.clip(scaled, -100, 100)
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    # g_j = 2 * sum_{k=j}^{j+n-1} w[k] * a[k-j] = 2 * dot(a, w[j:j+n])
    # Use np.correlate for efficiency
    g = 2.0 * np.correlate(w, a, mode='valid')
    return g


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def symmetrize(a):
    """Symmetrize a sequence (make it palindromic), keep sum=1."""
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    ss = a_sym.sum()
    if ss > 1e-15:
        a_sym /= ss
    return a_sym

def smooth_adam(a0, time_limit, T_init=0.05, T_min=0.001, lr_init=0.005,
                beta1=0.9, beta2=0.999, eps=1e-8, verbose=False, label=""):

    Smooth Adam optimization with temperature annealing.
    Minimizes max(conv) subject to a>=0, sum(a)=1.

    ss = a.sum()
    if ss < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / ss
    n = len(a)

    best_obj = conv.max()
    best_a = a.copy()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    # Temperature schedule
    T = T_init
    conv_scale = conv.max()

    t0 = time.monotonic()
    no_improve = 0
    lr = lr_init

    while time.monotonic() - t0 < time_limit:

        conv = fast_convolve(a)
        M = conv.max()

        v = beta2 * v + (1 - beta2) * g ** 2

        a_trial = np.maximum(a_trial, 0.0)

            best_obj = new_obj

        else:

            no_improve = 0

            T = T_init * conv_scale * 0.5

                m = np.zeros(n)
                v = np.zeros(n)

    while time.monotonic() - t0 < time_limit:
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            if old_val < 1e-12:
                continue

            # Try a range of multipliers
            best_j_obj = best_obj
            best_j_val = old_val

            for mult in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                trial = a.copy()
                trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                trial[mask] *= scale
                trial = np.maximum(trial, 0.0)
                ss = trial.sum()
                if ss < 1e-15:
                    continue
                trial = trial / ss
                val = fast_convolve(trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            break

    return best_a, best_obj


def two_phase_optimize(a0, time_limit, verbose=False, label=""):
    """
    Two-phase optimization: first smooth Adam (coarse), then mirror descent (fine).
    """
    n = len(a0)
    t0 = time.monotonic()

    # Phase 1: Smooth Adam (60% of time)
    t1 = time_limit * 0.6
    a1, obj1 = smooth_adam(a0, time_limit=t1, T_init=0.05, T_min=0.002,
                           lr_init=0.008, verbose=verbose, label=f"{label}_SA")

    # Phase 2: Mirror descent polish (40% of time)
    t_remaining = time_limit - (time.monotonic() - t0)
    a2, obj2 = mirror_descent_v2(a1, time_limit=max(t_remaining, 1.0),
                                 top_k=50, temp_init=0.001, lr_init=0.005,
                                 verbose=verbose, label=f"{label}_MD")

    if obj2 < obj1:
        return a2, obj2
    return a1, obj1


def perturb_and_reoptimize(a0, time_limit, n_perturbations=5, verbose=False, label=""):
    """
    Apply various perturbations and re-optimize from each.
    """
    n = len(a0)
    rng = np.random.default_rng(12345)
    best_a = a0.copy()
    best_obj = fast_convolve(a0).max()
    t0 = time.monotonic()

    perturbations = []

    # Symmetrize
    perturbations.append(("sym", symmetrize(a0)))

    # Small noise
    noise = rng.standard_normal(n) * 0.003
    a_noisy = np.maximum(a0 + noise, 0)
    a_noisy /= a_noisy.sum()
    perturbations.append(("noise", a_noisy))

    # Blend with uniform
    a_blend = 0.85 * a0 + 0.15 * (np.ones(n) / n)
    a_blend /= a_blend.sum()
    perturbations.append(("blend_uniform", a_blend))

    # Blend with Gaussian
    a_gauss = make_gaussian(n, 0.2)
    a_blend2 = 0.8 * a0 + 0.2 * a_gauss
    a_blend2 /= a_blend2.sum()
    perturbations.append(("blend_gauss", a_blend2))

    # Edge reduction (reduce values at edges, boost center)
    a_edge = a0.copy()
    quarter = n // 4
    edge_sum = a_edge[:quarter].sum() + a_edge[-quarter:].sum()
    a_edge[:quarter] *= 0.7
    a_edge[-quarter:] *= 0.7
    freed = a0.sum() - a_edge.sum()
    a_edge[quarter:3*quarter] += freed
    a_edge = np.maximum(a_edge, 0)
    a_edge /= a_edge.sum()
    perturbations.append(("edge_reduce", a_edge))

    t_per = (time_limit - 5) / max(len(perturbations), 1)

    for name, a_init in perturbations:
        if time.monotonic() - t0 > time_limit - 5:
            break
        t_alloc = min(t_per, time_limit - (time.monotonic() - t0) - 5)
        if t_alloc < 5:
            break

        a_opt, obj = smooth_adam(a_init, time_limit=t_alloc, T_init=0.03,
                                 T_min=0.001, lr_init=0.005,
                                 verbose=False, label=f"{label}_{name}")
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_opt
            if verbose:
                print(f"    {label} perturb {name}: obj={best_obj:.10f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization for minimizing 2*n*max(conv)/(sum)^2.
    Uses smooth Adam with temperature annealing as primary method,
    combined with mirror descent, coordinate descent, and perturbations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    n_base = 768
    if parent is not None:
        n_base = len(parent)

    # === Phase 1: Quick initialization and evaluation ===
    print("=== Phase 1: Initializations ===")
    init_candidates = []

    if parent is not None:
        init_candidates.append((parent, "parent"))
        init_candidates.append((symmetrize(parent), "parent_sym"))

    # Generate diverse initializations at the base length
    for n_try in [n_base]:
        init_candidates.append((make_gaussian(n_try, 0.25), f"gauss25_n{n_try}"))
        init_candidates.append((make_gaussian(n_try, 0.35), f"gauss35_n{n_try}"))
        init_candidates.append((make_super_gaussian(n_try, 4.0), f"super4_n{n_try}"))
        init_candidates.append((make_super_gaussian(n_try, 6.0), f"super6_n{n_try}"))
        init_candidates.append((make_bell(n_try, 0.7), f"bell70_n{n_try}"))
        init_candidates.append((make_optimal_shape(n_try), f"optshape_n{n_try}"))

    # Quick eval
    best_init = None
    best_init_obj = float('inf')
    for a_init, lbl in init_candidates:
        s = a_init.sum()
        if s < 1e-10:
            continue
        a_norm = a_init / s
        conv = fast_convolve(a_norm)
        c_obj = 2 * len(a_norm) * conv.max()
        if c_obj < best_init_obj:
            best_init_obj = c_obj
            best_init = a_norm

    # Evaluate the best init
    try_seq(best_init, "best_init")
    if parent is not None:
        try_seq(parent, "parent")

    print(f"  Best init continuous obj: {best_init_obj:.6f}")

    # === Phase 2: Main Smooth Adam (largest allocation) ===
    print(f"\n=== Phase 2: Smooth Adam (main) ===")
    remaining = deadline - time.monotonic()
    t_main = remaining * 0.35
    print(f"  Time: {t_main:.0f}s")

    a_opt, obj = smooth_adam(best_init, time_limit=t_main, T_init=0.05,
                             T_min=0.001, lr_init=0.008,
                             verbose=True, label="SA_main")
    try_seq(a_opt, "SA_main")

    # === Phase 3: Mirror Descent V2 ===
    print(f"\n=== Phase 3: Mirror Descent V2 ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_md = remaining * 0.20
        print(f"  Time: {t_md:.0f}s")
        a_opt, obj = mirror_descent_v2(best_seq, time_limit=t_md,
                                       top_k=50, temp_init=0.0005, lr_init=0.008,
                                       verbose=True, label="MD2")
        try_seq(a_opt, "MD2")

    # === Phase 4: Two-phase from different init ===
    print(f"\n=== Phase 4: Two-phase (Gaussian init) ===")
    remaining = deadline - time.monotonic()
    t_2p = remaining * 0.15
    print(f"  Time: {t_2p:.0f}s")
    a_gauss = make_gaussian(n_base, 0.3)
    a_opt, obj = two_phase_optimize(a_gauss, time_limit=t_2p,
                                    verbose=True, label="2ph_gauss")
    try_seq(a_opt, "2ph_gauss")

    # === Phase 5: Perturb and re-optimize ===
    print(f"\n=== Phase 5: Perturbations ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_pert = remaining * 0.15
        print(f"  Time: {t_pert:.0f}s")
        a_opt, obj = perturb_and_reoptimize(best_seq, time_limit=t_pert,
                                            verbose=True, label="perturb")
        try_seq(a_opt, "perturb")

    # === Phase 6: Coordinate Descent Polish ===
    print(f"\n=== Phase 6: Coordinate Descent ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_cd = remaining * 0.10
        print(f"  Time: {t_cd:.0f}s")
        a_opt, obj = coordinate_descent(best_seq, time_limit=t_cd,
                                        verbose=True, label="CD")
        try_seq(a_opt, "CD")

    # === Phase 7: Final Smooth Adam with low temperature (fine-tune) ===
    print(f"\n=== Phase 7: Fine SA ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_fine = remaining * 0.08
        print(f"  Time: {t_fine:.0f}s")
        a_opt, obj = smooth_adam(best_seq, time_limit=t_fine, T_init=0.01,
                                 T_min=0.0005, lr_init=0.003,
                                 verbose=True, label="SA_fine")
        try_seq(a_opt, "SA_fine")

    # === Phase 8: Try different length ===
    print(f"\n=== Phase 8: Alternative lengths ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            # Try n_base + 128
            n_alt = n_base + 128
            # Interpolate parent to new length
            old_x = np.linspace(0, 1, len(best_seq))
            new_x = np.linspace(0, 1, n_alt)
            a_interp = np.interp(new_x, old_x, best_seq)
            a_interp = np.maximum(a_interp, 0)
            a_interp /= a_interp.sum()

            t_alt = min(remaining * 0.3, 120)
            a_opt, obj = smooth_adam(a_interp, time_limit=t_alt, T_init=0.03,
                                     T_min=0.001, lr_init=0.005,
                                     verbose=True, label="alt_len")
            try_seq(a_opt, f"alt_n{n_alt}")

    # === Phase 9: Final symmetrization and polish ===
    print(f"\n=== Phase 9: Final Polish ===")
    if best_seq is not None:
        a_s = symmetrize(best_seq)
        try_seq(a_s, "final_sym")

        remaining = deadline - time.monotonic()
        if remaining > 15:
            t_pol = min(remaining - 5, 30)
            a_opt, obj = mirror_descent_v2(best_seq, time_limit=t_pol,
                                           top_k=30, temp_init=0.0001, lr_init=0.003,
                                           verbose=False, label="polish")
            try_seq(a_opt, "polish")

    # Fallback
    if best_seq is None:
        best_seq = make_gaussian(n_base, 0.3)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
