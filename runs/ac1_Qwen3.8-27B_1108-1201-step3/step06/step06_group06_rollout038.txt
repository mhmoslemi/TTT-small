"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) diverse initialization, (2) Adam with log-sum-exp smooth surrogate,
(3) Frank-Wolfe with argmin-vertex oracle, (4) multi-length exploration,
(5) final polish. Key innovation: extremely efficient FW step (no LP needed,

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_gradient(f, tau):

    # = sum_{k=j}^{min(2n-2, j+n-1)} w_k * f_{k-j}
    
    # Using FFT: multiply FFT(w) * FFT(f_reversed) and take appropriate slice
    # Let's define: h = ifft(fft(w, N) * fft(f[::-1], N))[:N]
    # Then h[k] = sum_j w[j] * f_rev[k-j] = sum_j w[j] * f[n-1-(k-j)] = sum_j w[j] * f[n-1-k+j]

def adam_smooth(f0, time_limit, tau_init=None, lr_init=0.005, verbose=False):
    """Adam optimization of the smooth log-sum-exp surrogate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    if tau_init is None:
        tau = 0.05 * conv.max()
    else:
        tau = tau_init

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    lr = lr_init
    t = 0

    best_obj = conv.max()
    best_f = f.copy()
    no_improve = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        t += 1
        grad, conv = smooth_gradient(f, tau)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new

        # Adaptive temperature
        tau *= 0.9995
        if tau < 5e-8:
            tau = 5e-8

        # Adaptive learning rate
        if no_improve > 200:
            lr *= 0.7
        if no_improve > 500:
            break

        if verbose and t % 200 == 0:
            print(f"    Adam iter {t}: obj={2*n*best_obj:.8f}, tau={tau:.2e}, lr={lr:.5f}")

    return best_f, 2 * n * best_obj


def frank_wolfe(f0, time_limit, verbose=False):
    """
    Frank-Wolfe for min max_k a^T T_k a s.t. a in simplex.
    Vertex oracle: e_{argmin (subgradient)_j}
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        # Find active peaks (within 1% of max)
        threshold = 0.99 * M
        active = np.where(conv >= threshold)[0]
        
        # Compute averaged subgradient over active peaks
        # Subgradient for peak k: 2 * (T_k * f) where (T_k * f)_j = sum_l f_l * 1[k-l in [0,n-1]] * ... 
        # Actually (T_k * f)_j = f_{k-j} if 0 <= k-j < n, else 0
        # So (T_k * f) is just a shifted and truncated version of f.
        # For averaged subgradient: grad_j = 2 * sum_{k in active} (1/|active|) * f_{k-j} (if valid)
        
        # More efficiently: grad = 2 * (1/|active|) * sum_{k in active} shifted_f_k
        # where shifted_f_k[j] = f[k-j] if 0 <= k-j < n
        
        # This is equivalent to: for each j, grad[j]/2 = (1/|active|) * sum_{k in active, 0<=k-j<n} f[k-j]
        # = (1/|active|) * sum_{l: 0<=l<n, k-j=l for some k in active} f[l]
        # = (1/|active|) * sum_{l in S_j} f[l] where S_j = {l : l+j in active, 0<=l<n}
        
        # Even more efficiently using convolution:
        # Let indicator = 1[k in active] for k in 0..2n-2
        # Then grad[j]/2 = sum_{l=0}^{n-1} f[l] * indicator[j+l]
        # = (f * indicator_rev)[j + (2n-2)] ... hmm
        # Actually: sum_l f[l] * indicator[j+l] 
        # Let's define: for each j, this is the dot product of f with indicator[j:j+n]
        # We can compute this via FFT.
        
        indicator = np.zeros(2*n-1)
        indicator[active] = 1.0
        indicator /= len(active)

        # = (f * indicator)[reversed] ... 
        # Let's use: h[j] = sum_l f[l] * indicator[j+l]
        # This is the cross-correlation of f and indicator.
        # Using FFT: h = ifft(conj(FFT(f)) * FFT(indicator)) ... 
        # Actually: h[j] = sum_l f[l] * indicator[j+l]
        # = sum_l f[l] * conj(indicator[l-j]) (if indicator is real)
        # = cross_correlation(f, indicator)[j]
        # = ifft(FFT(f) * conj(FFT(indicator)))[j]
        # Hmm, let me just verify the formula.
        
        # cross_correlation(x, y)[j] = sum_l x[l] * conj(y[l-j])
        # We want h[j] = sum_l f[l] * indicator[j+l]
        # Let m = j+l, l = m-j: h[j] = sum_m f[m-j] * indicator[m] (for valid m-j)
        # = sum_m indicator[m] * f[m-j]
        # This is the cross-correlation of indicator and f: (indicator ⊛ f)[j]
        # = ifft(conj(FFT(indicator)) * FFT(f))[j]
        
        fft_size = 1 << (3*n).bit_length()  # need enough room
        F = np.fft.rfft(f, fft_size)
        I = np.fft.rfft(indicator, fft_size)
        h = np.fft.irfft(np.conj(I) * F, fft_size)[:n]
        
        grad = 2.0 * h
        
        # Find vertex: argmin of grad
        j_min = np.argmin(grad)
        
        # Check if we're already at the vertex
        if f[j_min] > 1 - 1e-10:
            no_improve += 1
            if no_improve > 50:
                break
            continue
        
        # Line search for step size
        # FW step size: t_k = 2/(k+2) or greedy
        best_alpha = 2.0 / (it + 2.0)
        best_val = M
        
        # Try a few step sizes
        for alpha in [0.1, 0.3, 0.5, 2.0/(it+2), 0.7, 0.9, 1.0]:
            if alpha > 1.0:

            f_trial = f + alpha * (np.eye(n)[j_min] - f)
            f_trial = np.maximum(f_trial, 0)
            st = f_trial.sum()
            if st < 1e-15:
                continue
            f_trial = f_trial / st
            val = fast_convolve(f_trial).max()
            if val < best_val:

                best_alpha = alpha
        
        if best_alpha < 1e-10:
            no_improve += 1
            if no_improve > 100:
                break
            continue
        
        f = f + best_alpha * (np.eye(n)[j_min] - f)
        f = np.maximum(f, 0)
        f = f / f.sum()
        
        iter_count += 1
        if best_val < best_obj - 1e-14:
            best_obj = best_val
            best_f = f.copy()
            no_improve = 0
            if verbose and iter_count % 50 == 0:
                print(f"    FW iter {iter_count}: obj={2*n*best_obj:.8f}")

def fw_smooth_hybrid(f0, time_limit, verbose=False):
    """
    Hybrid: alternate between smooth gradient steps and FW steps.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    tau = 0.08 * conv.max()
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Smooth gradient step
        grad, conv = smooth_gradient(f, tau)
        lr = max(0.001, 0.01 * 0.9995**it)
        f_new = f - lr * grad
        f_new = project_simplex(f_new)
        
        conv_new = fast_convolve(f_new)
        if conv_new.max() < best_obj:
            best_obj = conv_new.max()
            best_f = f_new.copy()
        f = f_new
        
        # Decay temperature
        tau *= 0.999
        if tau < 1e-7:
            tau = 1e-7
            
        # Every 50 iterations, do a FW step
        if it % 50 == 0:
            conv = fast_convolve(f)
            M = conv.max()

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gauss(n, center=0.5, sigma=0.25):
    x = np.arange(n) / n - center
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a /= a.sum()
    return a

def make_bell(n, width=0.6):

def make_rand(n, seed=42):
    rng = np.random.default_rng(seed)
    a = np.abs(rng.standard_normal(n))

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines: (1) diverse initialization, (2) Adam with log-sum-exp smooth surrogate
using FFT-based gradients, (3) Frank-Wolfe with argmin-vertex oracle and
averaged subgradients for active peaks, (4) multi-length exploration,
(5) coordinate-descent micro-refinement. Key innovations: efficient subgradient
computation via FFT, adaptive multi-length search, and hybrid smooth/FW switching.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_gradient(f, tau):
    """
    Compute gradient of log(sum_k exp(conv_k/tau)) w.r.t. f.
    Returns (gradient vector, convolution).
    Uses FFT for O(n log n) computation.
    """
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    w = w / w_sum

    # h[j] = sum_l f[l] * w[j+l] for l=0..n-1 (w has length 2n-1)
    # Use g = f[::-1], then h[j] = (g*w)[j+n-1]
    g = f[::-1]
    total_len = n + (2 * n - 1) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    G = np.fft.rfft(g, fft_size)
    W = np.fft.rfft(w, fft_size)
    conv_gw = np.fft.irfft(G * W, fft_size)
    h = conv_gw[n - 1:2 * n - 1]

    grad = (2.0 / tau) * h
    return grad, conv


def adam_smooth(f0, time_limit, tau_init=None, lr_init=0.005, verbose=False):
    """Adam optimization of the smooth log-sum-exp surrogate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    if tau_init is None:
        tau = 0.06 * conv.max()
    else:
        tau = tau_init

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    lr = lr_init
    t = 0

    best_obj = conv.max()
    best_f = f.copy()
    no_improve = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        t += 1
        grad, conv = smooth_gradient(f, tau)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new

        tau *= 0.9994
        if tau < 3e-8:
            tau = 3e-8

        if no_improve > 300:
            lr *= 0.7
        if no_improve > 800:
            break

        if verbose and t % 200 == 0:
            print(f"    Adam iter {t}: obj={2 * n * best_obj:.8f}, tau={tau:.2e}, lr={lr:.5f}")

    return best_f, 2 * n * best_obj


def compute_subgradient_averaged(f, conv, peak_fraction=0.02):
    """
    Compute averaged subgradient of max_k (f*f)_k over peaks within
    peak_fraction of the max. Returns the gradient vector.
    """
    n = len(f)
    M = conv.max()
    threshold = M * (1.0 - peak_fraction)
    active = np.where(conv >= threshold)[0]

    # For each active peak k, the subgradient contribution to coordinate j is:
    #   2 * f[k-j] if 0 <= k-j < n, else 0
    # Averaged over active peaks:
    #   grad[j] = (2/|active|) * sum_{k in active} f[k-j] * 1[0 <= k-j < n]
    #           = (2/|active|) * sum_{l=0}^{n-1} f[l] * 1[j+l in active]

    # indicator[i] = 1 if i in active, else 0
    indicator = np.zeros(2 * n - 1)
    indicator[active] = 1.0
    indicator /= len(active)

    # h[j] = sum_l f[l] * indicator[j+l]
    # = (f_rev * indicator)[j + n - 1] where f_rev = f[::-1]
    f_rev = f[::-1]
    total_len = n + (2 * n - 1) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    F = np.fft.rfft(f_rev, fft_size)
    I = np.fft.rfft(indicator, fft_size)
    conv_fi = np.fft.irfft(F * I, fft_size)
    h = conv_fi[n - 1:2 * n - 1]

    return 2.0 * h


def frank_wolfe(f0, time_limit, verbose=False, peak_fraction=0.02):
    """
    Frank-Wolfe for min max_k (a*a)_k s.t. a in simplex.
    Uses averaged subgradient over active peaks and argmin vertex oracle.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Compute averaged subgradient
        grad = compute_subgradient_averaged(f, conv, peak_fraction)

        # Vertex oracle: argmin of subgradient
        j_min = np.argmin(grad)

        # Check if already at vertex
        if f[j_min] > 1 - 1e-12:
            no_improve += 1
            if no_improve > 50:
                break
            continue

        # Line search: try multiple step sizes
        best_alpha = 2.0 / (it + 2.0)
        best_val = M
        e_j = np.zeros(n)
        e_j[j_min] = 1.0

        alphas = [2.0 / (it + 2), 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        for alpha in alphas:
            f_trial = f + alpha * (e_j - f)
            f_trial = np.maximum(f_trial, 0)
            st = f_trial.sum()
            if st < 1e-15:
                continue
            f_trial = f_trial / st
            val = fast_convolve(f_trial).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha < 1e-12:
            no_improve += 1
            if no_improve > 80:
                break
            continue

        f = f + best_alpha * (e_j - f)
        f = np.maximum(f, 0)
        f = f / f.sum()

        if best_val < best_obj - 1e-14:
            best_obj = best_val
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 30 == 0:
                print(f"    FW iter {it}: obj={2 * n * best_obj:.8f}")

    return best_f, 2 * n * best_obj


def fw_pairwise(f0, time_limit, verbose=False):
    """
    Pairwise Frank-Wolfe: at each step, move from the worst coordinate
    (largest subgradient) to the best coordinate (smallest subgradient).
    This often converges faster than standard FW for our problem.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        grad = compute_subgradient_averaged(f, conv, 0.02)

        # Find best (min grad) and worst (max grad among positive entries)
        j_best = np.argmin(grad)
        # For "away" step, find the index in support with largest grad
        support = np.where(f > 1e-10)[0]
        if len(support) > 0:
            j_away = support[np.argmax(grad[support])]
        else:
            j_away = j_best

        # Try both standard FW (move to j_best) and away-step (move away from j_away toward j_best)
        best_f_new = f.copy()
        best_val = M

        # Standard FW
        e_j = np.zeros(n)
        e_j[j_best] = 1.0
        for alpha in [0.1, 0.3, 0.5, 2.0 / (it + 2), 0.8, 1.0]:
            f_t = f + alpha * (e_j - f)
            f_t = np.maximum(f_t, 0)
            f_t /= f_t.sum()
            val = fast_convolve(f_t).max()
            if val < best_val:
                best_val = val
                best_f_new = f_t.copy()

        # Away FW: move from j_away to j_best
        if j_away != j_best and f[j_away] > 1e-10:
            # f_new = f - gamma * f[j_away] * e_{j_away} + gamma * f[j_away] * e_{j_best}
            # = f with f[j_away] decreased by gamma*f[j_away] and f[j_best] increased by same
            for gamma in [0.2, 0.5, 0.8, 1.0]:
                f_t = f.copy()
                amt = gamma * f[j_away]
                f_t[j_away] -= amt
                f_t[j_best] += amt
                f_t = np.maximum(f_t, 0)
                f_t /= f_t.sum()
                val = fast_convolve(f_t).max()
                if val < best_val:
                    best_val = val
                    best_f_new = f_t.copy()

        if best_val < M - 1e-14:
            f = best_f_new
            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 30 == 0:
                    print(f"    FW-PW iter {it}: obj={2 * n * best_obj:.8f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            if no_improve > 60:
                # Restart with perturbation
                rng_local = np.random.default_rng(it * 7 + 3)
                noise = rng_local.standard_normal(n) * 0.002
                f = project_simplex(best_f + noise)
                conv = fast_convolve(f)
                M = conv.max()
                no_improve = 0

    return best_f, 2 * n * best_obj


def coordinate_descent_refine(f0, time_limit, verbose=False):
    """
    Coordinate descent: iteratively adjust individual coordinates to reduce max conv.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        improved = False

        # Find the peak
        k_star = np.argmax(conv)

        # Find the coordinate pair contributing most to the peak
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        if je < js:
            no_improve += 1
            break

        idx = np.arange(js, je + 1)
        products = f[idx] * f[k_star - idx]
        j_max = idx[np.argmax(products)]
        j_other = k_star - j_max

        if j_max == j_other:
            # Diagonal: reduce f[j_max]
            if f[j_max] < 1e-8:
                no_improve += 1
                break
            for frac in [0.7, 0.85, 0.9]:
                new_val = f[j_max] * frac
                diff = f[j_max] - new_val
                trial = f.copy()
                trial[j_max] = new_val
                # Redistribute to random non-contributing coordinates
                contrib = set()
                for k in range(2 * n - 1):
                    if conv[k] > 0.8 * M:
                        ls = max(0, k - n + 1)
                        le = min(n - 1, k)
                        for l in range(ls, le + 1):
                            contrib.add(l)
                non_contrib = np.array([i for i in range(n) if i not in contrib])
                if len(non_contrib) > 0:
                    trial[non_contrib] += diff * f[non_contrib] / max(f[non_contrib].sum(), 1e-15)
                trial = np.maximum(trial, 0)
                trial /= trial.sum()
                val = fast_convolve(trial).max()
                if val < M - 1e-12:
                    f = trial
                    M = val
                    improved = True
                    break
        else:
            # Off-diagonal: reduce product by scaling one down
            for scale in [0.5, 0.7, 0.85]:
                new_val = f[j_max] * scale
                diff = f[j_max] - new_val
                trial = f.copy()
                trial[j_max] = new_val
                # Redistribute to j_other (if it doesn't worsen) or to random
                if f[j_other] > 0:
                    # Add to a "safe" location
                    safe_idx = np.random.default_rng(int(it * 13)).integers(0, n)
                    while safe_idx in (j_max, j_other) and n > 2:
                        safe_idx = np.random.default_rng(int(it * 13 + safe_idx)).integers(0, n)
                    trial[safe_idx] += diff
                else:
                    trial[(j_max + n // 2) % n] += diff
                trial = np.maximum(trial, 0)
                trial /= trial.sum()
                val = fast_convolve(trial).max()
                if val < M - 1e-12:
                    f = trial
                    M = val
                    improved = True
                    break

        if improved:
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    CD iter {it}: obj={2 * n * best_obj:.8f}")
        else:
            no_improve += 1
            if no_improve > 30:
                break

    return best_f, 2 * n * best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gauss(n, center=0.5, sigma=0.25):
    x = np.arange(n) / n - center
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a /= a.sum()
    return a


def make_bell(n, width=0.6):
    x = np.arange(n) / n - 0.5
    t = np.clip(np.abs(x) / (width / 2), 0, 1.0)
    a = np.maximum(1 - t ** 4, 1e-10)
    a /= a.sum()
    return a


def make_rand_smooth(n, seed=42):
    rng = np.random.default_rng(seed)
    a = np.abs(rng.standard_normal(n)) + 0.5
    # Smooth it
    kernel = np.ones(5) / 5
    a = np.convolve(a, kernel, mode='same')
    a /= a.sum()
    return a


def make_ramp(n, decay=0.4):
    a = np.exp(-np.arange(n) / (n * decay))
    a /= a.sum()
    return a


def make_flat_top(n):
    """Raised cosine window."""
    a = 0.5 * (1 + np.cos(2 * np.pi * (np.arange(n) / n - 0.5)))
    a = np.maximum(a, 1e-10)
    a /= a.sum()
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: Diverse initialization (50s)
    Phase 2: Adam smooth optimization (500s)
    Phase 3: Pairwise Frank-Wolfe (400s)
    Phase 4: Multi-length exploration (350s)
    Phase 5: Coordinate descent + final polish (300s)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return score

    # === Phase 0: Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # === Phase 1: Diverse initialization ===
    print("=== Phase 1: Initialization ===")
    candidates = []
    for n_try in [512, 640, 768, 1024]:
        candidates.append((make_gauss(n_try, 0.35, 0.20), n_try))
        candidates.append((make_gauss(n_try, 0.5, 0.22), n_try))
        candidates.append((make_bell(n_try, 0.5), n_try))
        candidates.append((make_bell(n_try, 0.7), n_try))
        candidates.append((make_flat_top(n_try), n_try))
        candidates.append((make_rand_smooth(n_try, seed + n_try), n_try))
        candidates.append((make_ramp(n_try, 0.35), n_try))
        candidates.append((np.ones(n_try) / n_try, n_try))

    for i, (a_init, n_init) in enumerate(candidates):
        if time.monotonic() > deadline - 1400:
            break
        try_seq(a_init, f"init_{i}_n{n_init}")

    if best_seq is None:
        best_seq = make_gauss(768, 0.4, 0.22)
        best_score = try_seq(best_seq, "fallback")

    print(f"  Best init: n={len(best_seq)}, score={best_score:.8f}")

    # === Phase 2: Adam smooth optimization ===
    print(f"\n=== Phase 2: Adam Smooth ===")
    rem = deadline - time.monotonic()
    t2 = min(rem, 500)
    if t2 > 60:
        print(f"  Time: {t2:.0f}s, n={len(best_seq)}")
        a_opt, obj = adam_smooth(best_seq, time_limit=t2 * 0.7, lr_init=0.005, verbose=True)
        try_seq(a_opt, "adam_main")

        if time.monotonic() < deadline - 400:
            rem2 = deadline - time.monotonic() - 400
            t2b = min(rem2, 180)
            tau_fine = 0.02 * fast_convolve(best_seq).max()
            a_opt2, obj2 = adam_smooth(best_seq, time_limit=t2b, tau_init=tau_fine, lr_init=0.002, verbose=False)
            try_seq(a_opt2, "adam_fine")

    # === Phase 3: Pairwise Frank-Wolfe ===
    print(f"\n=== Phase 3: Pairwise FW ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 350
        if rem > 60:
            t3 = min(rem, 400)
            print(f"  Time: {t3:.0f}s")
            a_opt, obj = fw_pairwise(best_seq, time_limit=t3, verbose=True)
            try_seq(a_opt, "fw_pw")

            # Standard FW as well
            if time.monotonic() < deadline - 250:
                rem2 = deadline - time.monotonic() - 250
                t3b = min(rem2, 150)
                a_opt2, obj2 = frank_wolfe(best_seq, time_limit=t3b, verbose=False)
                try_seq(a_opt2, "fw_std")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-Length ===")
    if best_seq is not None:
        base = best_seq
        target_lengths = [1024, 1280, 1536, 640, 512, 2048]
        # Sort by distance from current length
        target_lengths.sort(key=lambda x: abs(x - len(base)))

        for target_n in target_lengths:
            rem = deadline - time.monotonic() - 120
            if rem < 40:
                break
            if target_n == len(base):
                continue
            if target_n > 2500:
                continue

            f_interp = interpolate_sequence(base, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")

            rem2 = deadline - time.monotonic() - 100
            if rem2 > 30:
                t4 = min(rem2 * 0.35, 100)
                a_opt, obj = adam_smooth(f_interp, time_limit=t4, lr_init=0.004)
                try_seq(a_opt, f"adam_n{target_n}")

                rem3 = deadline - time.monotonic() - 80
                if rem3 > 20 and best_seq is not None:
                    t4b = min(rem3 * 0.3, 60)
                    work = best_seq if len(best_seq) == target_n else f_interp
                    a_opt2, obj2 = fw_pairwise(work, time_limit=t4b)
                    try_seq(a_opt2, f"fw_n{target_n}")

    # === Phase 5: Coordinate descent + perturbed restarts ===
    print(f"\n=== Phase 5: CD + Perturbations ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 30:
            t5 = min(rem, 120)
            a_opt, obj = coordinate_descent_refine(best_seq, time_limit=t5, verbose=False)
            try_seq(a_opt, "cd")

        # Perturbed FW restarts
        base = best_seq
        for attempt in range(4):
            rem = deadline - time.monotonic() - 40
            if rem < 25:
                break
            noise_scale = 0.002 + 0.002 * attempt
            perturbed = project_simplex(base + rng.standard_normal(len(base)) * noise_scale)
            sc = try_seq(perturbed, f"pert_{attempt}")

            rem2 = deadline - time.monotonic() - 30
            if rem2 > 15:
                t5b = min(rem2 * 0.4, 50)
                a_opt, obj = fw_pairwise(perturbed, time_limit=t5b)
                try_seq(a_opt, f"fw_pert_{attempt}")

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 15:
            t6 = min(rem, 60)
            tau_final = 0.01 * fast_convolve(best_seq).max()
            a_opt, obj = adam_smooth(best_seq, time_limit=t6 * 0.5,
                                     tau_init=tau_final, lr_init=0.001)
            try_seq(a_opt, "final_adam")

            rem2 = deadline - time.monotonic() - 10
            if rem2 > 10:
                t7 = min(rem2, 30)
                a_opt2, obj2 = frank_wolfe(best_seq, time_limit=t7)
                try_seq(a_opt2, "final_fw")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
