import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def nesterov_mirror_descent(f0, time_limit, verbose=False,
                            lr_init=0.01, momentum=0.9,
                            restart_every=150):
    """
    Nesterov-accelerated mirror descent for minimizing max(conv)
    on the probability simplex with non-negative constraint.
    """
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    # Nesterov momentum
    y = f.copy()
    t = 1.0
    t0_time = time.monotonic()
    lr = lr_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0_time < time_limit:
        it += 1

        conv_y = fast_convolve(y)
        M = conv_y.max()

        # Compute subgradient at y
        # Use top-k peaks with softmax weighting
        len_c = len(conv_y)
        k_top = max(10, int(len_c * 0.15))
        peaks = np.argpartition(conv_y, -k_top)[-k_top:]
        pv = conv_y[peaks]
        
        # Temperature-adaptive softmax
        temp = max((pv.max() - pv.min()) * 0.5, 1e-10)
        scaled = (pv - pv.max()) / temp
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute weighted subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue
            idx = np.arange(js, je + 1)
            contrib = y[k - idx] + y[idx]
            sg[js:je + 1] += w[i] * contrib

        # Mirror descent step (multiplicative)
        f_new = y * np.exp(-lr * sg)
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()

        # Nesterov update
        t_new = (1 + np.sqrt(1 + 4 * t * t)) / 2
        beta = (t - 1) / t_new
        y_new = (1 + beta) * f_new - beta * f
        y_new = np.maximum(y_new, 1e-15)
        y_new = y_new / y_new.sum()

        # Evaluate
        obj_new = fast_convolve(f_new).max()

        if obj_new < M:
            f = f_new
            y = y_new
            t = t_new
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"      NMD iter {it}: obj={best_obj:.10f}, lr={lr:.6f}")
            # Slightly increase lr if stable
            lr = min(lr * 1.01, 0.05)
        else:
            # Rollback and reduce lr
            f = f  # keep old f
            y = f  # reset momentum
            t = 1.0
            lr *= 0.7
            no_improve += 1

        if no_improve > restart_every:
            rng = np.random.default_rng(it * 7 + 13)
            noise = rng.standard_normal(n) * 0.003
            f = np.maximum(best_f + noise, 1e-15)
            f = f / f.sum()
            y = f.copy()
            t = 1.0
            lr = lr_init
            no_improve = 0

        if lr < 1e-7:
            break

    return best_f, best_obj


def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c_obj = -np.ones(n)
    # Build constraint matrix: (2n-1) x n, sparse
    # (f*g)[k] = sum_j f[j] * g[k-j] for valid k-j
    # Row k: A[k, j] = f[j] if 0 <= k-j < n, else 0
    # i.e., A[k, j] = f[j] if max(0, k-n+1) <= j <= min(n-1, k)
    
    # For efficiency, build the sparse matrix
    from scipy.sparse import csr_matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def make_ramp_init(n, power=1.0, seed=42):
    """Generate a ramp-shaped initialization."""
    x = np.arange(n)
    a = (x + 1) ** power
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_exp_init(n, rate=5.0, seed=42):
    """Generate an exponential initialization."""
    x = np.arange(n)
    a = np.exp(-rate * x / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_parabola_init(n, seed=42):
    """Generate a parabolic initialization."""
    x = np.arange(n) / n
    a = 1.0 - (2*x - 0.3)**2  # Off-center parabola
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_two_bump_init(n, pos1=0.25, pos2=0.7, width=0.15, seed=42):
    """Generate a two-bump initialization."""
    x = np.arange(n) / n
    a = np.exp(-((x - pos1)/width)**2) + 0.5 * np.exp(-((x - pos2)/width)**2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_random_structured(n, seed=42):
    """Generate a random but structured initialization."""
    rng = np.random.default_rng(seed)
    # Start with a smooth base
    a = np.ones(n)
    # Apply random smooth perturbations

    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
