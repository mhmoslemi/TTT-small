```
    u=np.sort(v)[::-1]
```

```
def bernstein_density(n,p,D):
    x=(np.arange(n)+0.5)/n
    w=np.exp(p-np.max(p)); w/=w.sum()
    f=np.zeros(n)
    om=1.0-x
    for k in range(D+1):
       if w[k] == 0: continue
       f += w[k] * math.comb(D,k) * (x**k) * (om**(D-k))
    f=np.maximum(f,0); s=f.sum(); if s<1e-15: return np.ones(n)/n
    return f/s
```

```python
"""
Search for nonnegative step-function heights minimizing
    2*n*max(np.convolve(a,a))/(sum(a))^2.

The program uses:
  - parent-sequence interpolation to longer lengths,
  - low-dimensional Bernstein / power densities as diverse starts,
  - dense smooth-max multiplicative mirror descent,
  - dense smooth-max Euclidean projected-gradient polishing,
  - adaptive temperature, step size, restarts, and multi-length refinement.

No filesystem/network access is used.  The function propose_candidate()
is the entry point.
"""

import numpy as np
import time
import math

try:
    from scipy import optimize as scopt
    HAS_SCIPY = True
except Exception:
    HAS_SCIPY = False


# ----------------------------------------------------------------------
# Basic helpers
# ----------------------------------------------------------------------

def _next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def fast_convolve(a):
    """Linear convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([0.0])
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return np.ascontiguousarray(conv)


def score_array(a):
    """Objective value for an arbitrary nonnegative sequence (scale invariant)."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return float("inf")
    a[~np.isfinite(a)] = 0.0
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 1e-10:
        return float("inf")
    a = a / s
    conv = fast_convolve(a)
    m = float(np.max(conv))
    if m < 0.0 or not np.isfinite(m):
        return float("inf")
    return float(2.0 * a.size * m)


def score_normalized(a):
    """Faster objective when a is already approximately normalized."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return float("inf")
    s = float(a.sum())
    if s < 1e-10:
        return float("inf")
    if abs(s - 1.0) > 1e-6:
        a = a / s
    conv = fast_convolve(a)
    m = float(np.max(conv))
    if m < 0.0:
        m = 0.0
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * a.size * m)


def normalize_nonnegative(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a[~np.isfinite(a)] = 0.0
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 1e-10:
        return a
    return a / s


def add_candidate(a, best_a, best_score, cand_list, label=""):
    """
    Consider a candidate sequence.  Returns updated (best_a, best_score, cand_list).
    cand_list stores up to 8 promising candidates as (score, array, label).
    """
    if a is None:
        return best_a, best_score, cand_list
    try:
        arr = np.asarray(a, dtype=np.float64).ravel()
    except Exception:
        return best_a, best_score, cand_list
    if arr.size == 0:
        return best_a, best_score, cand_list

    arr = np.nan_to_num(arr, nan=0.0, posinf=1000.0, neginf=0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s < 1e-10:
        return best_a, best_score, cand_list

    an = arr / s
    sc = score_array(an)
    if not np.isfinite(sc):
        return best_a, best_score, cand_list

    if sc < best_score:
        print(f"  new best [{label}]: {sc:.9f}  n={arr.size}")
        best_a = an.copy()
        best_score = float(sc)
        cand_list.insert(0, (float(sc), best_a.copy(), label))
    elif best_score < float("inf") and sc < best_score * 1.02:
        cand_list.append((float(sc), an.copy(), label))

    if len(cand_list) > 8:
        worst_idx = 0
        worst_val = -float("inf")
        for i in range(len(cand_list)):
            if cand_list[i][0] > worst_val:
                worst_val = cand_list[i][0]
                worst_idx = i
        cand_list.pop(worst_idx)

    return best_a, best_score, cand_list

def interpolate_sequence_midpoint(a, new_n):
    a = normalize_nonnegative(a)
    n = a.size
    if n == 0:
        return np.ones(int(new_n)) / max(1, int(new_n))
    if n == int(new_n):
        return a
    if n == 1:
        return np.full(int(new_n), float(a[0]))

    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(int(new_n)) + 0.5) / int(new_n)
    b = np.interp(x_new, x_old, a)
    b = np.maximum(b, 0.0)
    s = float(b.sum())
    if s < 1e-10:
        return np.full(int(new_n), 1.0 / int(new_n))
    return b / s


def interpolate_sequence_endpoint(a, new_n):
    a = normalize_nonnegative(a)
    n = a.size
    if n == 0:
        return np.ones(int(new_n)) / max(1, int(new_n))
    if n == int(new_n):
        return a
    if n == 1 or int(new_n) == 1:
        return np.full(int(new_n), float(a.sum()))

    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, int(new_n))
    b = np.interp(x_new, x_old, a)
    b = np.maximum(b, 0.0)
    s = float(b.sum())
    if s < 1e-10:
        return np.full(int(new_n), 1.0 / int(new_n))
    return b / s

def make_power_density(n, alpha=0.0, beta=0.0, lam=0.0):
    n = int(n)
    x = (np.arange(n) + 0.5) / n
    f = np.ones(n, dtype=np.float64)
    if alpha != 0.0:
        f = f * np.power(np.maximum(x, 1e-300), float(alpha))
    if beta != 0.0:
        f = f * np.power(np.maximum(1.0 - x, 1e-300), float(beta))
    if lam != 0.0:
        f = f * np.exp(np.clip(float(lam) * x, -100.0, 100.0))
    f = np.maximum(f, 0.0)
    s = float(f.sum())
    if s < 1e-10:
        return np.ones(n) / n
    return f / s

def softmax_weights(c, temp):
    c = np.asarray(c, dtype=np.float64).ravel()
    n = c.size
    if n == 0:
        return np.array([])
    if temp <= 1e-300:
        w = np.zeros(n)
        w[int(np.argmax(c))] = 1.0
        return w
    x = (c - float(np.max(c))) / temp
    x = np.maximum(x, -1000.0)
    e = np.exp(x)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros(n)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def active_uniform_weights(c, rel_delta=0.05):
    c = np.asarray(c, dtype=np.float64).ravel()
    n = c.size
    if n == 0:
        return np.array([])
    m = float(np.max(c))
    if m <= 0.0 or not np.isfinite(m):
        w = np.zeros(n)
        w[int(np.argmax(c))] = 1.0
        return w
    threshold = m * (1.0 - float(rel_delta))
    mask = c >= threshold
    if not mask.any():
        w = np.zeros(n)
        w[int(np.argmax(c))] = 1.0
        return w
    w = np.zeros(n)
    w[mask] = 1.0 / float(np.count_nonzero(mask))
    return w


def dense_subgradient(w, a):
    """
    g_i = 2 * sum_k w_k a_{k-i}
    computed by FFT cross-correlation.
    """
    n = a.size
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a, size)
    corr = np.fft.irfft(np.conj(fw) * fa, size)
    g = 2.0 * corr[:n]
    return np.ascontiguousarray(g)


def project_simplex(v):
    """Project onto {x >= 0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return v
    if n == 1:
        out = np.array([max(0.0, float(v[0]))])
        if out[0] > 0:
            return out
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if s < 1e-15:
        return np.ones(n) / n
    return out / s

def local_md_dense(a0, time_limit, eta=0.08, rel_temp=0.05, verbose=False, label=""):
    """
    Multiplicative mirror descent in log-coordinates using dense smooth-max weights.
    """
    a = np.asarray(a0, dtype=np.float64).ravel()
    a = np.maximum(a, 1e-15)
    s = float(a.sum())
    if s > 0:
        a = a / s
    else:
        a = np.ones_like(a) / max(1, a.size)

    n = a.size
    y = np.log(a)
    best_a = a.copy()
    best_score = score_normalized(a)

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    if verbose:
        print(f"  {label} md_dense start: score={best_score:.9f}, n={n}, eta={eta:.4f}, tempRel={rel_temp:.4f}")

    while time.monotonic() - t0 < max(0.0, float(time_limit)):
        it += 1
        conv = fast_convolve(a)
        m = float(np.max(conv))
        if m < 0.0 or not np.isfinite(m):
            break
        cur = 2.0 * n * m

        if cur < best_score - 1e-14:
            best_score = cur
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 5000:
            break

        temp = m * rel_temp
        if temp < 1e-16:
            temp = 1e-16
        w = softmax_weights(conv, temp)
        if rel_temp > 0.004 and (it % 4 == 0):
            aw = active_uniform_weights(conv, 0.05)
            w = 0.65 * w + 0.35 * aw
            w = w / max(1e-300, float(w.sum()))

        g = dense_subgradient(w, a)
        g = g * float(n)
        g = g - float(np.mean(g))
        gnorm = float(np.max(np.abs(g)))
        if not np.isfinite(gnorm) or gnorm < 1e-14:
            break
        g = g / gnorm

        best_trial_score = cur
        best_trial_y = y
        best_trial_a = a
        improved = False

        for fac in (1.0, 0.5, 0.25, 0.1):
            te = eta * fac
            if te < 1e-9:
                continue
            yt = y - te * g
            yt = yt - float(np.max(yt))
            at = np.exp(yt)
            sd = float(at.sum())
            if sd < 1e-15:
                continue
            at = at / sd
            ts = score_normalized(at)
            if ts < best_trial_score - 1e-14:
                best_trial_score = ts
                best_trial_y = yt
                best_trial_a = at
                improved = True
                if ts < cur * 0.999:
                    break

        if improved:
            y = best_trial_y
            a = best_trial_a
            if best_trial_score < best_score - 1e-13:
                best_score = best_trial_score
                best_a = a.copy()
                no_improve = 0
                eta = min(eta * 1.05, 0.5)
                rel_temp = max(rel_temp * 0.99, 0.001)
            else:
                no_improve += 1
                eta = max(eta * 0.9, 1e-4)
        else:
            no_improve += 1
            eta = max(eta * 0.7, 1e-5)
            if no_improve > 40:
                rel_temp = min(rel_temp * 1.2, 0.25)
            if no_improve > 180:
                base = best_a if best_a is not None else a
                base = np.maximum(base, 1e-15)
                base = base / base.sum()
                rng = np.random.default_rng(1234567 + it)
                sigma_y = 0.05 + 0.15 * min((no_improve - 180) / 400.0, 1.0)
                yn = np.log(base) + rng.standard_normal(n) * sigma_y
                yn = yn - float(np.max(yn))
                a = np.exp(yn)
                a = a / a.sum()
                y = yn
                eta = 0.08
                rel_temp = 0.05
                no_improve = 0

        if verbose and it % 200 == 0:
            print(f"    {label} md_dense it={it:6d} cur={cur:.9f} best={best_score:.9f} eta={eta:.4f} tempRel={rel_temp:.5f}")

    return best_a, float(best_score)

                no_improve = 0
                dt = min(dt * 1.05, 0.1)
                rel_temp = max(rel_temp * 0.99, 0.001)
            else:
                no_improve += 1
                dt = max(dt * 0.9, 1e-5)
        else:
            no_improve += 1
            dt = max(dt * 0.7, 1e-6)
            if no_improve > 40:
                rel_temp = min(rel_temp * 1.2, 0.25)
            if no_improve > 180:
                base = best_a if best_a is not None else a
                base = np.maximum(base, 1e-15)
                base = base / base.sum()
                rng = np.random.default_rng(7654321 + it)
                yn = np.log(base) + rng.standard_normal(n) * 0.1
                yn = yn - float(np.max(yn))
                a = np.exp(yn)
                a = a / a.sum()
                dt = 0.02
                rel_temp = 0.05
                no_improve = 0

        if verbose and it % 200 == 0:
            print(f"    {label} add_dense it={it:6d} cur={cur:.9f} best={best_score:.9f} dt={dt:.5f} tempRel={rel_temp:.5f}")

    return best_a, float(best_score)

# Optional low-dimensional Bernstein search

_HAS_LOW = False
_low_basis = None
_low_n = 0
_low_D = 0
_low_deadline = 0.0


def initialize_lowdim(n, D):
    global _low_basis, _low_n, _low_D
    n = int(n)
    D = int(D)
    x = (np.arange(n) + 0.5) / n
    om = 1.0 - x
    basis = np.empty((D + 1, n), dtype=np.float64)
    for k in range(D + 1):
        ck = float(math.comb(D, k))
        if k == 0:
            xk = np.ones(n)
        else:
            xk = x ** k
        if D - k == 0:
            omk = np.ones(n)
        else:
            omk = om ** (D - k)
        basis[k] = ck * xk * omk
    basis *= (D + 1)
    _low_basis = basis
    _low_n = n
    _low_D = D
    return True


def lowdim_objective(p):
    global _low_basis
    if _low_basis is None:
        return float("inf")
    p = np.asarray(p, dtype=np.float64)
    w = np.exp(p - float(np.max(p)))
    w /= w.sum()
    f = _low_basis.T @ w
    f = np.maximum(f, 0.0)
    return score_array(f)


def lowdim_callback(xk):
    global _low_deadline
    return time.monotonic() > _low_deadline


def bernstein_density(n, p, D):
    n = int(n)
    D = int(D)
    p = np.asarray(p, dtype=np.float64)
    w = np.exp(p - float(np.max(p)))
    w /= w.sum()
    x = (np.arange(n) + 0.5) / n
    om = 1.0 - x
    f = np.zeros(n, dtype=np.float64)
    for k in range(D + 1):
        if w[k] <= 0.0:
            continue
        ck = float(math.comb(D, k))
        if k == 0:
            xk = 1.0
        else:
            xk = x ** k
        if D - k == 0:
            omk = 1.0
        else:
            omk = om ** (D - k)
        f += w[k] * ck * xk * omk
    f = np.maximum(f, 0.0)
    s = float(f.sum())
    if s < 1e-12:
        return np.ones(n) / n
    return f / s


def run_lowdim_search(time_limit, seed=1):
    global _low_deadline
    results = []
    if not HAS_SCIPY or time_limit < 5.0:
        return results

    D = 6
    n = 256
    initialize_lowdim(n, D)
    _low_deadline = time.monotonic() + float(time_limit)
    rng = np.random.default_rng(seed)
    npar = D + 1

    for s in range(8):
        p0 = rng.standard_normal(npar) * 0.5
        if s == 0:
            p0 = np.linspace(-1.0, 2.0, npar)
        elif s == 1:
            p0 = np.linspace(2.0, -1.0, npar)
        elif s == 2:
            p0 = np.linspace(-2.0, 0.5, npar)
        elif s == 3:
            p0 = np.linspace(0.5, 2.0, npar)
        try:
            res = scopt.minimize(
                lowdim_objective,
                p0,
                method="Powell",
                options={"maxiter": 50, "maxfev": 200, "ftol": 1e-8, "gtol": 1e-5},
                callback=lowdim_callback,
            )
            if res is not None and np.isfinite(res.fun):
                results.append((float(res.fun), np.asarray(res.x, dtype=np.float64).copy()))
        except Exception as e:
            print("  lowdim error:", e)
            break

        if time.monotonic() > _low_deadline:
            break

    return results

def eval_final_list(lst):
    try:
        return float(evaluate_sequence(lst))
    except Exception:
        return float("inf")

def propose_candidate(seed=42, budget_s=1800):
    budget_s = max(30.0, float(budget_s))
    start = time.monotonic()
    deadline = start + max(1.0, budget_s - 30.0)
    rng = np.random.default_rng(seed)

    best_a = None
    best_score = float("inf")
    cand_list = []

    # --- Parent sequence, if available ---
    parent = None
    try:
        parr = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parr.size > 0:
            parr[~np.isfinite(parr)] = 0.0
            parr = np.clip(parr, 0.0, 1000.0)
            if parr.sum() > 1e-10:
                parent = normalize_nonnegative(parr)
                best_a, best_score, cand_list = add_candidate(parent, best_a, best_score, cand_list, "parent")
    except Exception:
        parent = None

    # --- Static parametric candidates ---
    n_power_short = [1024, 2048, 3072]
    alphas_coarse = np.linspace(0.05, 2.0, 41)

    for n in n_power_short:
        for alpha in alphas_coarse:
            if time.monotonic() > deadline - 900.0:
                break
            a = make_power_density(n, alpha=alpha)
            best_a, best_score, cand_list = add_candidate(a, best_a, best_score, cand_list, f"power_a{alpha:.2f}_n{n}")

    # A few beta/lam families to create more diverse starts.
    for n in [1536, 2048, 3072]:
        for alpha in [0.35, 0.5, 0.65, 0.85, 1.05, 1.25]:
            for beta in [0.0, 0.15, 0.35, 0.60]:
                if time.monotonic() > deadline - 800.0:
                    break
                a = make_power_density(n, alpha=alpha, beta=beta)
                best_a, best_score, cand_list = add_candidate(a, best_a, best_score, cand_list,
                                                              f"pow_ab_{alpha:.2f}_{beta:.2f}_n{n}")

        for lam in [-0.8, -0.4, -0.2, 0.2, 0.5, 0.9]:
            if time.monotonic() > deadline - 700.0:
                break
            a = make_power_density(n, alpha=0.0, beta=0.0, lam=lam)
            best_a, best_score, cand_list = add_candidate(a, best_a, best_score, cand_list, f"exp_lam{lam:.2f}_n{n}")

    # Fallback if nothing worked.
    if best_a is None:
        best_a = np.ones(768) / 768.0
        best_score = score_array(best_a)
        cand_list = [(best_score, best_a.copy(), "uniform")]

    # --- Interpolate current best to several lengths (often improves discretization error) ---
    target_ns = [1024, 1536, 2048, 3072, 4096]
    for tn in target_ns:
        if tn <= best_a.size or time.monotonic() > deadline - 760.0:
            continue
        a1 = interpolate_sequence_midpoint(best_a, tn)
        best_a, best_score, cand_list = add_candidate(a1, best_a, best_score, cand_list, f"interp_mid_n{tn}")
        a2 = interpolate_sequence_endpoint(best_a, tn)
        best_a, best_score, cand_list = add_candidate(a2, best_a, best_score, cand_list, f"interp_ep_n{tn}")

    # --- Optional low-dimensional Bernstein search ---
    if time.monotonic() < deadline - 650.0:
        low_budget = min(70.0, max(10.0, (deadline - time.monotonic()) * 0.05))
        print(f"=== Low-dim Bernstein search (budget {low_budget:.0f}s) ===")
        try:
            low_res = run_lowdim_search(low_budget, seed=int(rng.integers(0, 1_000_000)))
            for sc, p in low_res[:6]:
                for tn in [1024, 2048, 3072, 4096]:
                    if time.monotonic() > deadline - 600.0:
                        break
                    f = bernstein_density(tn, p, 6)
                    best_a, best_score, cand_list = add_candidate(f, best_a, best_score, cand_list, f"bern_n{tn}")
        except Exception as e:
            print("  lowdim blocked:", e)

    # --- Choose length for primary local optimization ---
    local_a = best_a.copy()
    local_n = best_a.size
    if local_n < 1024:
        local_n = 2048
        local_a = interpolate_sequence_midpoint(best_a, local_n)
    elif local_n < 2048:
        local_n = 2048
        local_a = interpolate_sequence_midpoint(best_a, local_n)
    elif local_n > 4096:
        local_n = 4096
        local_a = interpolate_sequence_midpoint(best_a, local_n)

    best_a, best_score, cand_list = add_candidate(local_a, best_a, best_score, cand_list, "local_start")

    # --- Main mirror-descent polishing ---
    rem = deadline - time.monotonic()
    if rem > 120.0:
        t_main = rem * 0.50
        print(f"=== Main MD dense (budget {t_main:.0f}s, n={local_n}) ===")
        try:
            oa, osn = local_md_dense(local_a, t_main, eta=0.08, rel_temp=0.05, verbose=True, label="main")
            best_a, best_score, cand_list = add_candidate(oa, best_a, best_score, cand_list, "main_md")
        except Exception as e:
            print("  main md error:", e)

    # --- Additive polishing on current global best ---
    rem = deadline - time.monotonic()
    if rem > 90.0 and best_a is not None:
        t_add = rem * 0.22
        print(f"=== Additive dense (budget {t_add:.0f}s, n={best_a.size}) ===")
        try:
            oa, osn = local_add_dense(best_a, t_add, dt=0.02, rel_temp=0.03, verbose=True, label="add")
            best_a, best_score, cand_list = add_candidate(oa, best_a, best_score, cand_list, "additive")
        except Exception as e:
            print("  additive error:", e)

    # --- If we are still < 4096, try a bigger-length refinement ---
    rem = deadline - time.monotonic()
    if rem > 120.0 and best_a is not None and best_a.size < 4096:
        t_big = rem * 0.28
        big_n = 4096
        a_big = interpolate_sequence_midpoint(best_a, big_n)
        best_a, best_score, cand_list = add_candidate(a_big, best_a, best_score, cand_list, f"big_start_n{big_n}")
        print(f"=== Big-n MD dense (budget {t_big:.0f}s, n={big_n}) ===")
        try:
            oa, osn = local_md_dense(a_big, t_big, eta=0.06, rel_temp=0.035, verbose=True, label="big")
            best_a, best_score, cand_list = add_candidate(oa, best_a, best_score, cand_list, "big_md")
        except Exception as e:
            print("  big md error:", e)

    # --- Refine a promising secondary candidate if available ---
    rem = deadline - time.monotonic()
    if rem > 80.0 and len(cand_list) > 1:
        # pick the second best by stored score, avoiding the current best length if possible
        second = None
        for item in cand_list:
            sc, arr, lab = item
            if abs(int(arr.size) - int(best_a.size)) > 0:
                second = item
                break
        if second is None:
            second = cand_list[1]
        t_sec = rem * 0.30
        print(f"=== Secondary candidate MD (budget {t_sec:.0f}s, n={second[1].size}) ===")
        try:
            oa, osn = local_md_dense(second[1], t_sec, eta=0.06, rel_temp=0.04, verbose=True, label="secondary")
            best_a, best_score, cand_list = add_candidate(oa, best_a, best_score, cand_list, "secondary")
        except Exception as e:
            print("  secondary error:", e)

    # --- Tiny final polish from global best ---
    rem = deadline - time.monotonic()
    if rem > 30.0 and best_a is not None:
        t_fin = min(90.0, rem - 5.0)
        print(f"=== Final micro-polish (budget {t_fin:.0f}s, n={best_a.size}) ===")
        try:
            oa, osn = local_md_dense(best_a, t_fin, eta=0.03, rel_temp=0.012, verbose=False, label="final")
            best_a, best_score, cand_list = add_candidate(oa, best_a, best_score, cand_list, "final_polish")
        except Exception as e:
            print("  final polish error:", e)

    # --- Make a final list and verify with evaluate_sequence when available ---
    final_list = [float(x) for x in best_a]
    final_score = eval_final_list(final_list)
    if not np.isfinite(final_score):
        # fallback to a simple valid sequence if validation failed for some reason
        final_list = [1.0 / 768.0] * 768
        final_score = eval_final_list(final_list)

    # Compare a few extra candidates using the exact evaluator.
    seen_lens = set()
    for item in cand_list[:6]:
        sc, arr, lab = item
        ll = int(arr.size)
        if ll in seen_lens:
            continue
        seen_lens.add(ll)
        lst = [float(x) for x in arr]
        sc2 = eval_final_list(lst)
        if np.isfinite(sc2) and sc2 < final_score:
            final_score = sc2
            final_list = lst

    print(f"FINAL candidate score (our FFT): {best_score:.9f}, n={len(final_list)}")
    print(f"FINAL candidate score (evaluate_sequence): {final_score:.9f}")

    return final_list
```
