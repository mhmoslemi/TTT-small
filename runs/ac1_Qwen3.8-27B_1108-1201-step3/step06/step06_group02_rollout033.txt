import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_smooth_grad(a, s):
    """
    Compute gradient of (1/s)*log(sum exp(s*conv_k)) w.r.t. a.
    Returns (gradient, peak_value).
    """
    n = len(a)
    conv = fast_convolve(a)
    max_c = conv.max()
    log_w = s * (conv - max_c)
    exp_w = np.exp(log_w)
    p = exp_w / exp_w.sum()
    # Cross-correlation: grad_j = sum_k p[k] * a[k-j] for valid k-j
    N = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, N)
    P = np.fft.rfft(p, N)
    corr = np.fft.irfft(A * P.conj(), N)
    grad = corr[:n]
    return grad, max_c

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_line_search(f, g, f_conv):
    """
    Grid-based line search for alpha in [0,1] minimizing
    max_k conv((1-alpha)f + alpha*g, (1-alpha)f + alpha*g)
    """
    n = len(f)
    g_conv = fast_convolve(g)
    # Cross-convolution f*g
    fft_size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    G = np.fft.rfft(g, fft_size)
    fg_conv = np.fft.irfft(F * G, fft_size)[:2 * n - 1]

    best_alpha = 0.0
    best_val = f_conv.max()

    # Coarse grid
    alphas = np.linspace(0.0, 1.0, 50)
    vals = np.zeros(len(alphas))
    for i, alpha in enumerate(alphas):
        val = (1 - alpha) ** 2 * f_conv + 2 * alpha * (1 - alpha) * fg_conv + alpha ** 2 * g_conv
        vals[i] = val.max()
    best_idx = np.argmin(vals)
    best_alpha = alphas[best_idx]
    best_val = vals[best_idx]

    # Refine with golden section around best
    if 0 < best_idx < len(alphas) - 1:
        lo = alphas[best_idx - 1]
        hi = alphas[best_idx + 1]
        gr = 0.618
        for _ in range(15):
            a1 = hi - gr * (hi - lo)
            a2 = lo + gr * (hi - lo)
            v1 = ((1 - a1) ** 2 * f_conv + 2 * a1 * (1 - a1) * fg_conv + a1 ** 2 * g_conv).max()
            v2 = ((1 - a2) ** 2 * f_conv + 2 * a2 * (1 - a2) * fg_conv + a2 ** 2 * g_conv).max()
            if v1 < v2:
                hi = a2
            else:
                lo = a1
            mid = (lo + hi) / 2.0
            vm = ((1 - mid) ** 2 * f_conv + 2 * mid * (1 - mid) * fg_conv + mid ** 2 * g_conv).max()
            if vm < best_val:
                best_val = vm
                best_alpha = mid

    return best_alpha, best_val


def lp_fw_optimize(f0, time_limit, verbose=False):
    """
    LP-Frank-Wolfe with grid line search and adaptive M levels.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    f_conv = fast_convolve(f)
    M = f_conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_levels = [1.0, 1.02, 1.05, 1.1, 1.2, 1.4, 1.7, 2.2, 3.0]
    m_idx = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2.0:
            break

        f_conv = fast_convolve(f)
        M = f_conv.max()
        g = None

        for level_try in range(m_idx, len(m_levels)):
            M_try = M * m_levels[level_try]
            g = solve_lp_direction(f, M_try, n)
            if g is not None and g.sum() > 1e-12:
                m_idx = level_try
                break
            m_idx = level_try + 1

        if g is None or g.sum() < 1e-12:
            # Perturbation to escape
            rng = np.random.default_rng(it * 13 + 7)
            noise_type = it % 4
            if noise_type == 0:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            elif noise_type == 1:
                shift = rng.integers(1, max(2, n // 10))
                f = project_simplex(np.roll(best_f, int(shift)))
            elif noise_type == 2:
                f = project_simplex(0.8 * best_f + 0.2 * best_f[::-1])
            else:
                # Blend with a cosine window
                x = np.arange(n) / n
                cos_win = 1.0 + 0.1 * np.cos(2 * np.pi * x)
                f = project_simplex(best_f * cos_win)
            no_improve += 1
            if no_improve > 60:
                break
            continue

        g = g / g.sum()
        diff = np.max(np.abs(g - f))
        if diff < 1e-10:
            no_improve += 1
            m_idx = min(m_idx + 1, len(m_levels) - 1)
            if no_improve > 50:
                break
            continue
        no_improve = 0

        # Line search
        best_alpha, best_val = lp_fw_line_search(f, g, f_conv)

        if best_alpha < 1e-10:
            no_improve += 1
            m_idx = min(m_idx + 1, len(m_levels) - 1)
            continue

        f_new = f + best_alpha * (g - f)
        f_new = np.maximum(f_new, 0.0)
        sn = f_new.sum()
        if sn < 1e-15:
            continue
        f_new = f_new / sn
        f_new_conv = fast_convolve(f_new)
        M_new = f_new_conv.max()
        f = f_new
        m_idx = 0

        if M_new < best_obj - 1e-14:
            best_obj = M_new
            best_f = f.copy()
            if verbose and it % 20 == 0:
                print(f"    LP-FW iter {it}: obj={best_obj:.10f}")

    return best_f, best_obj


def multiplicative_descent(f0, time_limit, verbose=False):
    """Mirror descent with adaptive temperature over peaks."""
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    lr = 0.004
    no_improve = 0
    temperature = 0.001
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        k_actual = min(30, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        f = f * np.exp(-lr * sg)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        co = fast_convolve(f).max()
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"    MD iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        lr *= 0.9997
        temperature *= 1.005

        if no_improve > 150:
            rng = np.random.default_rng(it * 37 + 3)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.002, 1e-15)
            f = f / f.sum()
            lr = 0.003
            no_improve = 0

        if lr < 1e-8:
            break

    return best_f, best_obj


def make_gaussian(n, sigma_frac=0.25):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.5):
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_cosine_mod(n, freq, amp):
    x = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_multi_cosine(n, n_freqs=3, max_amp=0.2, seed=0):
    rng = np.random.default_rng(seed)
    a = np.ones(n)
    for m in range(1, n_freqs + 1):
        amp = rng.uniform(0.05, max_amp)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * m * np.arange(n) / n + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_periodic_block(n, block_len, seed=0):
    rng = np.random.default_rng(seed)
    block = rng.uniform(0.3, 1.0, size=block_len)
    n_blocks = n // block_len
    a = np.tile(block, n_blocks)
    if len(a) < n:
        a = np.concatenate([a, np.zeros(n - len(a))])
    a = a[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def objective_normalized(a):
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max()


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization combining:
    1. Improved LP-Frank-Wolfe with grid line search
    2. Log-sum-exp smoothed Nesterov projected gradient
    3. Multiplicative descent
    4. Multi-length exploration
    5. Diverse initializations and final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_gaussian(768)
        p_score = try_seq(parent, "default_parent")

    print(f"\n=== Phase 1: Diverse Initializations ===")
    t0 = time.monotonic()

    init_candidates = []
    n_values = [512, 640, 768, 896, 1024]

    for n_try in n_values:
        init_candidates.append(('gauss', n_try, make_gaussian(n_try, 0.25)))
        init_candidates.append(('bell', n_try, make_bell(n_try, 0.5)))
        init_candidates.append(('cos1', n_try, make_cosine_mod(n_try, 1, 0.2)))
        init_candidates.append(('cos2', n_try, make_cosine_mod(n_try, 2, 0.15)))
        init_candidates.append(('multi1', n_try, make_multi_cosine(n_try, 3, 0.2, seed=n_try)))
        init_candidates.append(('multi2', n_try, make_multi_cosine(n_try, 5, 0.15, seed=n_try + 100)))
        init_candidates.append(('block', n_try, make_periodic_block(n_try, 16, seed=n_try)))
        init_candidates.append(('flat', n_try, np.ones(n_try) / n_try))

    if parent is not None:
        init_candidates.append(('parent', len(parent), parent))

    # Score all initializations
    scored_inits = []
    for label, n_try, a_init in init_candidates:
        sc = objective_normalized(a_init)
        scored_inits.append((sc, a_init, label, n_try))
        if sc < 1.6:
            try_seq(a_init, f"init_{label}_{n_try}")
    scored_inits.sort(key=lambda x: x[0])

    print(f"  Best init: {scored_inits[0][2]}, n={scored_inits[0][3]}, obj={scored_inits[0][0]:.6f}")
    print(f"  Phase 1 time: {time.monotonic()-t0:.1f}s")

    # Phase 2: LP-FW on parent (most promising)
    print(f"\n=== Phase 2: LP-FW on Parent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 600
        if rem > 60:
            t_alloc = min(rem, 500)
            print(f"  LP-FW: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "lpfw_parent")

    # Phase 3: LSE smooth optimization
    print(f"\n=== Phase 3: LSE Smooth Optimization ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_alloc = min(rem, 250)
            print(f"  LSE: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, obj = lse_optimize(best_seq, time_limit=t_alloc, s_init=20.0, s_end=2000.0, verbose=True)
            try_seq(a_opt, "lse")

    # Phase 4: LP-FW again (alternating)
    print(f"\n=== Phase 4: LP-FW Round 2 ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 40:
            t_alloc = min(rem, 250)
            print(f"  LP-FW R2: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "lpfw_r2")

    # Phase 5: Multiplicative descent
    print(f"\n=== Phase 5: Multiplicative Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_alloc = min(rem, 150)
            print(f"  MD: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, obj = multiplicative_descent(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "md")

    # Phase 6: Multi-length exploration
    print(f"\n=== Phase 6: Multi-Length Exploration ===")
    if best_seq is not None:
        base_n = len(best_seq)
        for target_n in [1024, 896, 640, 512, 640, 896]:
            if target_n == base_n:
                continue
            rem = deadline - time.monotonic() - 100
            if rem < 20:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")
            if sc < best_score + 0.02:
                t_alloc = min(rem * 0.3, 60)
                if t_alloc > 10:
                    print(f"  LP-FW n={target_n}: alloc={t_alloc:.0f}s")
                    a_opt, obj = lp_fw_optimize(f_interp, time_limit=t_alloc, verbose=False)
                    try_seq(a_opt, f"lpfw_n{target_n}")

    # Phase 7: LP-FW from best initialization (different from parent)
    print(f"\n=== Phase 7: LP-FW from Alt Init ===")
    for trial_idx in range(1, min(4, len(scored_inits))):
        rem = deadline - time.monotonic() - 80
        if rem < 30:
            break
        sc_init, a_init, label, n_init = scored_inits[trial_idx]
        if sc_init > best_score * 1.1:
            continue
        t_alloc = min(rem * 0.4, 80)
        if t_alloc < 10:
            break
        print(f"  LP-FW [{label}] n={n_init}: alloc={t_alloc:.0f}s (init_obj={sc_init:.6f})")
        a_opt, obj = lp_fw_optimize(a_init, time_limit=t_alloc, verbose=False)
        try_seq(a_opt, f"lpfw_alt_{label}")

    # Phase 8: Final polish - alternating LSE and LP-FW
    print(f"\n=== Phase 8: Final Polish ===")
    if best_seq is not None:
        # LSE polish
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            t_alloc = min(rem * 0.4, 40)
            print(f"  LSE polish: alloc={t_alloc:.0f}s")
            a_opt, obj = lse_optimize(best_seq, time_limit=t_alloc, s_init=100.0, s_end=5000.0, verbose=False)
            try_seq(a_opt, "lse_polish")

        # LP-FW polish
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            t_alloc = min(rem, 30)
            print(f"  LP-FW polish: alloc={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, "lpfw_polish")

        # MD polish
        rem = deadline - time.monotonic() - 5
        if rem > 8:
            t_alloc = min(rem, 15)
            print(f"  MD polish: alloc={t_alloc:.0f}s")
            a_opt, obj = multiplicative_descent(best_seq, time_limit=t_alloc, verbose=False)
            try_seq(a_opt, "md_polish")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
