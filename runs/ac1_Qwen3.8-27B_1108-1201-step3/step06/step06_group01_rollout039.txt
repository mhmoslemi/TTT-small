import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from time import monotonic

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a = a / s
    return 2.0 * n * fast_convolve(a).max()

def softmax(z):
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()

def obj_and_grad_theta(theta, T, n):
    a = softmax(theta)
    b = fast_convolve(a)  # length 2n-1
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    lse = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * lse
    w = exp_Tb / exp_Tb.sum()
    # grad_a[j] = 4n * sum_{m=0}^{n-1} w[j+m] * a[m] = 4n * correlate(w, a, 'valid')[j]
    grad_a = 4.0 * n * np.correlate(w.astype(np.float64), a, mode='valid')
    S = np.dot(grad_a, a)
    grad_theta = a * (grad_a - S)
    return obj, grad_theta

def lbgfs_anneal(a0, time_budget, T_start=5000, T_end=50, maxiter=5000):
    n = len(a0)
    a_safe = np.maximum(a0, 1e-30)
    a_safe = a_safe / a_safe.sum()
    theta0 = np.log(a_safe)
    t0 = monotonic()
    
    best_theta = theta0.copy()
    best_score = compute_score(a0)
    
    # Adaptive temperature: decrease T over iterations
    n_iter = maxiter
    T_schedule = T_start * (T_end / T_start) ** (np.arange(n_iter) / n_iter)
    
    # We'll use a wrapper that changes T over time
    state = {'iter': 0, 'T': T_start}
    
    def func(theta):
        if monotonic() - t0 > time_budget:
            return 1e15
        T = T_start * (T_end / T_start) ** (state['iter'] / n_iter)
        state['iter'] += 1
        obj, _ = obj_and_grad_theta(theta, T, n)
        return obj
    
    def jac(theta):
        T = T_start * (T_end / T_start) ** (state['iter'] / n_iter)
        _, g = obj_and_grad_theta(theta, T, n)
        return g
    
    try:
        res = minimize(
            fun=func, x0=theta0, jac=jac, method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10,
                     'maxfun': maxiter * 20, 'disp': False}
        )
        a_out = softmax(res.x)
        score = compute_score(a_out)
        if score < best_score:
            best_score = score
            best_theta = res.x.copy()
    except Exception as e:
        print(f"  L-BFGS error: {e}")
    
    return best_theta, best_score

Now, for the coordinate descent, let me think of a more effective approach.

**Improved Coordinate Descent:**

The key insight: if the peak is at lag k*, and the active set is {j : 0<=j<n, 0<=k*-j<n}, then reducing any a_j in the active set reduces b_{k*}. But we need to add the mass somewhere that doesn't increase other peaks.

A good heuristic: add the mass to the coordinate with the smallest "sensitivity" to other peaks. The sensitivity of coordinate j to lag k is `a_{k-j}` (if 0<=k-j<n). So we want to add to a coordinate j where `max_k a_{k-j}` is small (i.e., j is "far" from all high coordinates).

Actually, a simpler heuristic: add to the coordinate with the smallest current value. This tends to work well because:
1. It makes the distribution more uniform (reducing b_0)
2. It's unlikely to create a new peak (since we're adding to a small value)

Actually, I realize there's a much better approach for the coordinate descent: **use the subgradient information directly**.

At the current point a, the subgradient of the objective (max of b_k) is:
`g = 4n * sum_{k in active peaks} w_k * a_shifted_k`

where `a_shifted_k[j] = a_{k-j}` (the "shifted" version of a).

If the peak is unique at k*, then `g = 4n * a_shifted_{k*}`.

The direction of steepest descent is `-g` projected onto the tangent space of the simplex. The tangent space is {v : sum v = 0, v >= -a (for staying in the simplex)}.

The projected gradient direction is:
`d = -(g - mean(g))` (projection onto sum=0 hyperplane)

Then we take a step: `a_new = a + alpha * d`, projected onto the simplex.

This is essentially the gradient descent method, but we can make it more effective by:
1. Using Nesterov momentum
2. Using a backtracking line search
3. Projecting onto the simplex at each step

    n = len(v)

        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

        try:
            res = minimize(
                fun=func, x0=theta0, jac=jac, method='L-BFGS-B',
                options={'maxiter': 5000, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxfun': 100000, 'disp': False}
            )
            a_new = softmax_fn(res.x)
            score = compute_score(a_new)
            if verbose:
                print(f"    L-BFGS T={T:5d}: score={score:.6f} (iters={res.nit})")
            
            if score < compute_score(a_current):
                a_current = a_new
        except Exception as e:
            if verbose:
                print(f"    L-BFGS T={T} error: {e}")
        
        # Check time
        if monotonic() - t0 > time_budget * 0.8:
            break
    
    return a_current


def projected_gradient_descent(a0, time_budget, verbose=True):
    """PGD with multi-peak subgradient and Nesterov momentum."""
    n = len(a0)
    a = a0.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    v = np.zeros(n)
    lr = 0.005
    momentum = 0.85
    t0 = monotonic()
    step = 0
    no_improve = 0
    
    while monotonic() - t0 < time_budget:
        step += 1
        b = fast_convolve(a)
        
        # Multi-peak subgradient
        top_k = min(20, len(b))
        top_peaks = np.argpartition(b, -top_k)[-top_k:]
        peak_vals = b[top_peaks]
        range_pv = peak_vals.max() - peak_vals.min()
        temp = max(0.005 * range_pv, 1e-10)
        w_peaks = np.exp((peak_vals - peak_vals.max()) / temp)
        w_peaks = w_peaks / w_peaks.sum()
        
        # Compute subgradient using correlation
        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_peaks):
            w_full[int(k)] = w_peaks[i]
        
        g = 4.0 * n * np.correlate(w_full, a, mode='valid')
        
        # Project onto tangent space
        g_proj = g - np.mean(g)
        
        # Nesterov momentum
        v = momentum * v - lr * g_proj
        a_new = a + v
        a_new = project_simplex(a_new)
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                rng = np.random.default_rng(step * 7919)
                a = best_a + rng.standard_normal(n) * 0.002
                a = np.maximum(a, 0)
                a = a / a.sum()
                v = np.zeros(n)
                lr = 0.003
                no_improve = 0
            elif
            no_improve > 50:
                lr *= 0.98
        
        lr *= 0.9998
        
        if verbose and step % 2000 == 0:
            print(f"    PGD step={step}: score={best_score:.6f} lr={lr:.6f}")
    
    return best_a, best_score


def coord_targeted(a0, time_budget, verbose=True):
    """Targeted coordinate descent: move mass from peak contributors to safe targets."""
    n = len(a0)
    a = a0.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = monotonic()
    step = 0
    no_improve = 0
    rng = np.random.default_rng(12345)
    
    while monotonic() - t0 < time_budget:
        step += 1
        b = fast_convolve(a)
        k_star = int(np.argmax(b))
        
        # Active set for this lag
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        active = np.arange(js, je + 1)
        non_active = np.setdiff1d(np.arange(n), active)
        
        # Find worst contributor
        if k_star > 0:
            contribs = np.array([a[j] * a[k_star - j] for j in active])
            j_worst = active[np.argmax(contribs)]
        else:
            j_worst = int(np.argmax(a))
        
        # Choose target: prefer non-active, low-value
        if len(non_active) > 0:
            targets = non_active[np.argsort(a[non_active])[:5]]
        else:
            targets = active[np.argsort(a[active])[:5]]
        targets = [int(t) for t in targets if t != j_worst]
        
        improved = False
        for tgt in targets[:3]:
            for factor in [0.97, 0.94, 0.9, 0.85, 0.8]:
                a_new = a.copy()
                diff = a[j_worst] * (1.0 - factor)
                if diff < 1e-15:
                    break
                a_new[j_worst] = a[j_worst] * factor
                a_new[tgt] += diff
                a_new = a_new / a_new.sum()
                score = 2.0 * n * fast_convolve(a_new).max()
                if score < best_score - 1e-14:
                    a = a_new
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try moving to a random coordinate
            tgt = int(rng.integers(n))
            if tgt != j_worst and a[j_worst] > 1e-10:
                for factor in [0.95, 0.9, 0.85]:
                    a_new = a.copy()
                    diff = a[j_worst] * (1.0 - factor)
                    a_new[j_worst] = a[j_worst] * factor
                    a_new[tgt] += diff
                    a_new = a_new / a_new.sum()
                    score = 2.0 * n * fast_convolve(a_new).max()
                    if score < best_score - 1e-14:
                        a = a_new
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
        
        if not improved:
            no_improve += 1
            if no_improve > 100:
                a = best_a + rng.standard_normal(n) * 0.003
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
        else:
            no_improve = 0
        
        if verbose and step % 5000 == 0:
            print(f"    CD step={step}: score={best_score:.6f}")
    
    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f_norm)
    f_new = np.maximum(f_new, 0)
    ss = f_new.sum()
    if ss > 1e-15:
        f_new = f_new / ss
    return f_new


def make_initializations(parent, n):
    """Generate diverse initializations of length n."""
    inits = []
    
    # 1. Interpolated parent
    if parent is not None:
        inits.append(interpolate_seq(parent, n))
    
    # 2. Uniform
    inits.append(np.ones(n) / n)
    
    # 3. Cosine modulation
    t = np.arange(n)
    a = 1.0 + 0.3 * np.cos(2 * np.pi * t / n)
    a = np.maximum(a, 0)
    inits.append(a / a.sum())
    
    # 4. Higher frequency cosine
    a = 1.0 + 0.2 * np.cos(4 * np.pi * t / n)
    a = np.maximum(a, 0)
    inits.append(a / a.sum())
    
    # 5. Mildly random
    rng = np.random.default_rng(42)
    a = 1.0 + 0.1 * rng.standard_normal(n)
    a = np.maximum(a, 0.01)
    inits.append(a / a.sum())
    
    # 6. Triangular
    a = 1.0 - 0.5 * np.abs(2 * t / n - 1)
    a = np.maximum(a, 0)
    inits.append(a / a.sum())
    
    # 7. Two-band
    a = 1.0 + 0.15 * np.cos(2 * np.pi * t / n + np.pi/3) + 0.1 * np.cos(4 * np.pi * t / n)
    a = np.maximum(a, 0)
    inits.append(a / a.sum())
    
    return inits
