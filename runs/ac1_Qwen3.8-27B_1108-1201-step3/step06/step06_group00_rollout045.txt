"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using ADMM (Alternating Direction
Method of Multipliers) with a quadratic penalty on the convolution peak,
combined with simplex-projected gradient for the a-update, adaptive penalty
parameter, multiple restarts with diverse initializations, and final polish
via coordinate descent. This fundamentally differs from the prior subgradient
approach by using a consensus-form ADMM that separates the convolution
computation from the simplex constraint.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def conv_gradient(a, e):
    """
    Compute gradient of (1/2)||conv(a) - d||^2 w.r.t. a, given e = conv(a) - d.
    Returns grad where grad[i] = 2 * sum_j e[j] * a[j-i] (a[x]=0 for x outside [0,n-1]).
    """
    n = len(a)
    # grad[i] = 2 * (e * a_rev)[n-1+i]
    a_rev = a[::-1]
    conv_size = len(e) + n - 1
    fft_size = 1 << (conv_size - 1).bit_length()
    E = rfft(e, fft_size)
    AR = rfft(a_rev, fft_size)
    conv = irfft(E * AR, fft_size)[:conv_size]
    return 2.0 * conv[n-1:n-1+n]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def eval_local(a):
    """Compute objective 2n*max(conv)/(sum)^2 locally (assuming sum=1)."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def admm_solve(a0, time_limit, rho_init=10.0, inner_iters=50, sigma=1.5,
               rho_growth=1.05, max_rho=1e6, seed=42):
    """
    ADMM solver for min max_j b_j(a) subject to sum(a)=1, a>=0.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    n = len(a0)
    
    # Initialize
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    b = fast_convolve(a)
    z = b.copy()
    u = np.zeros(2 * n - 1)
    rho = float(rho_init)
    
    best_a = a.copy()
    best_peak = b.max()
    no_improve = 0
    
    outer_iter = 0
    while time.monotonic() - t_start < time_limit:
        outer_iter += 1
        
        # === a-update: minimize (rho/2)||conv(a) - (z-u)||^2 + (0.01/2)||a||^2 s.t. sum(a)=1, a>=0
        d = z - u
        # Use projected gradient with backtracking
        step = 1.0 / (rho * 4.0 * n + 0.01)  # rough Lipschitz bound
        step = min(step, 0.5)
        
        for _ in range(inner_iters):
            if time.monotonic() - t_start > time_limit:
                break
            b_a = fast_convolve(a)
            e = b_a - d
            grad = rho * conv_gradient(a, e) + 0.01 * a
            a_new = project_simplex(a - step * grad)
            if not np.allclose(a_new, a, atol=1e-14):
                a = a_new
            else:
                break
        
        # === z-update and t-update
        b_a = fast_convolve(a)
        t_val = max(0.0, (b_a + u).max())
        z = np.minimum(b_a + u, t_val)
        
        # === u-update (with over-relaxation)
        u = u + sigma * (b_a - z)
        
        # Track best
        peak = z.max()
        if peak < best_peak - 1e-12:
            best_peak = peak
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive rho
        residual_primal = np.linalg.norm(b_a - z)
        residual_dual = rho * np.linalg.norm(u - (u + sigma * (b_a - z)))
        # Actually dual residual is rho * norm(z_new - z_old) = rho * norm(z - z_prev)
        # Let me just use primal residual
        if residual_primal < 1e-6:
            rho = min(rho * rho_growth, max_rho)
        elif residual_primal > 1e-2 and rho > rho_init:
            rho = max(rho / rho_growth, rho_init)
            # Undo u update and recompute
            u = u - sigma * (b_a - z)
            z = np.minimum(b_a + u, t_val)
    
    return best_a, best_peak
