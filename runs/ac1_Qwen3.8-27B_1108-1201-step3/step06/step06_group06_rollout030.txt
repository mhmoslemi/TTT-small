import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_obj_and_grad(f, tau):
    """Compute log-sum-exp objective and its gradient."""
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    # Gradient: 2/tau * sum_k w[k] * a[k-j]
    # h[j] = sum_k w[k] * a[k-j] = (w_rev * a)[j] where w_rev[k] = w[-k]
    # Using FFT: h = irfft(rfft(a) * rfft(w_padded_and_flipped))
    w_rev = w[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)
    h = conv_result[n - 1:2 * n - 1][::-1]
    
    obj = tau * (np.log(w.sum()) + scaled.max())  # = tau * logsumexp(conv/tau)
    grad = (2.0 / tau) * h
    return obj, grad, conv, w

def fw_optimize_nonsmooth(f0, time_limit, verbose=False):
    """Frank-Wolfe on the non-smooth objective max_k conv_k(a)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
        s = 1.0
    else:
        f = f / s
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        gamma = 2.0 / (it + 2.0)  # FW step size
        
        # Find active peak
        conv = fast_convolve(f)
        M = conv.max()
        peak_idx = int(np.argmax(conv))
        
        # Subgradient: grad[j] = sum_{k active} a[k-j]
        # For single active peak at peak_idx:
        # grad[j] = a[peak_idx - j] for valid j
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        indices = np.arange(js, je + 1)
        grad_vals = f[peak_idx - indices]  # a[peak_idx - j] for j in indices
        
        # FW vertex: e_j where j minimizes grad[j]
        # Unconstrained positions have grad = 0, so we prefer those
        # But if all positions in the "active range" have positive grad,
        # the minimum is at the position with smallest a[peak_idx - j]
        min_pos_in_range = indices[np.argmin(grad_vals)]
        
        # Check if any position outside the range has grad=0 (always true if n > range)
        # If peak_idx < n-1 or peak_idx >= n-1, there are positions with grad=0
        # Actually for full conv, positions j where peak_idx - j < 0 or peak_idx - j >= n have grad 0
        # So if 0 <= peak_idx <= 2n-2, the range of j with non-zero grad is [max(0,peak_idx-n+1), min(n-1, peak_idx)]
        
        if js > 0 or je < n - 1:
            # There exist positions with grad=0, prefer those
            # Among those, pick one that's far from current support
            outside = np.setdiff1d(np.arange(n), indices)
            # Pick the one with smallest f value (to spread mass)
            v_idx = outside[np.argmin(f[outside])]
        else:
            v_idx = min_pos_in_range
        
        v = np.zeros(n)
        v[v_idx] = 1.0
        
        # FW update
        f_new = (1 - gamma) * f + gamma * v
        f_new = np.maximum(f_new, 0)
        f_new /= f_new.sum()
        
        conv_new = fast_convolve(f_new)
        M_new = conv_new.max()
        
        if M_new < best_obj - 1e-14:
            best_obj = M_new
            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    FW iter {it}: obj={2*n*best_obj:.8f}")
        else:
            no_improve += 1
        
        f = f_new
        
        if no_improve > 200:
            rng = np.random.default_rng(it * 13)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            no_improve = 0
            
        if no_improve > 1000:
            break
    
    return best_f, 2.0 * n * best_obj


def peak_aware_perturb(f, strength=0.01):
    """Perturb by reducing peak-contributing elements and increasing others."""
    n = len(f)
    conv = fast_convolve(f)
    peak_idx = int(np.argmax(conv))
    
    # Find top contributing pair
    js = max(0, peak_idx - n + 1)
    je = min(n - 1, peak_idx)
    indices = np.arange(js, je + 1)
    products = f[indices] * f[peak_idx - indices]
    j_main = indices[np.argmax(products)]
    
    f_new = f.copy()
    # Reduce the main contributor
    f_new[j_main] *= (1 - strength)
    diff = f[j_main] - f_new[j_main]
    
    # Redistribute to elements that don't contribute to the peak
    mask = np.ones(n, dtype=bool)
    for j in indices:
        mask[j] = False
        mask[peak_idx - j] = False
    
    if mask.sum() > 0 and f[mask].sum() > 1e-15:
        f_new[mask] += diff * f[mask] / f[mask].sum()
    
    f_new = np.maximum(f_new, 0)
    f_new /= f_new.sum()
    return f_new


def cosine_window(n):
    """Cosine-tapered window."""
    x = np.arange(n)
    a = np.cos(np.pi * (x + 0.5) / (2 * n))
    a = np.maximum(a, 0)
    a /= a.sum()
    return a


def raised_cosine(n, oversample=0.3):
    """Raised cosine with adjustable width."""
    x = np.arange(n) / (n - 1)
    t = (x - 0.5) / (0.5 - oversample)
    t = np.clip(t, -1, 1)
    a = 0.5 * (1 + np.cos(np.pi * t))
    a = np.maximum(a, 0)
    a /= a.sum()
    return a


def spectral_init(n, mode='b enn'):
    """Spectral-based initialization."""
    if mode == 'benr':
        # Benford-like: decreasing geometric
        a = np.geomspace(1, 0.1, n)
    elif mode == 'bimodal':
        # Two peaks
        x = np.arange(n)
        a = np.exp(-0.5 * ((x - n*0.25) / (n*0.15))**2) + np.exp(-0.5 * ((x - n*0.75) / (n*0.15))**2)
    elif mode == 'ring':
        # High at edges, low in middle
        x = np.arange(n) / (n - 1)
        a = x * (1 - x) * 4  # parabola
        a = a ** 2  # sharper
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

            # Check descent condition (simplified)

            if obj_z + np.dot(grad_z, dx) + (1.0 / (2.0 * step)) * np.dot(dx, dx) + 0.01 >= 0:
                pass
            # Simpler: just check if objective at trial is reasonable
            obj_trial, _, conv_trial, _ = smooth_obj_grad(x_trial, tau)
            if obj_trial <= obj_z + 0.1 * step * grad_norm_sq:
                break
            step *= 0.5

        x = x_trial
        step = min(step * 1.5, 1.0 / (4.0 * conv_z.max() / tau**2 + 1.0) * 2)

        # FISTA momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        z_new = x + ((t_k - 1.0) / t_new) * (x - z)
        z_new = project_simplex(z_new)
        z = z_new
        t_k = t_new

        conv_x = fast_convolve(x)
        M = conv_x.max()
        if M < best_obj - 1e-15:
            best_obj = M
            best_f = x.copy()
            no_improve = 0
        else:
            no_improve += 1

        tau *= tau_decay
        if tau < 1e-9:
            tau = 1e-9

        if no_improve > 100 and no_improve % 100 == 0:
            rng = np.random.default_rng(it * 13 + 7)
            noise = 0.003 * min(1.0 + no_improve / 300.0, 3.0)
            f = project_simplex(best_f + rng.standard_normal(n) * noise)
            x = f.copy()
            z = f.copy()
            t_k = 1.0
            step = 1.0
            no_improve = 0
            tau = 0.05 * fast_convolve(f).max()
            if verbose:
                print(f"    FISTA restart iter {it}: obj={2*n*best_obj:.8f}, tau={tau:.2e}")

        if verbose and it % 500 == 0:
            print(f"    FISTA iter {it}: obj={2*n*best_obj:.8f}, tau={tau:.2e}")

    return best_f, best_obj, 2.0 * n * best_obj


def adam_optimize(f0, time_limit, tau_init=None, lr_init=0.004, verbose=False):
    """Adam optimizer on log-sum-exp surrogate with adaptive temperature."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    if tau_init is None:
        tau = 0.06 * conv.max()
    else:
        tau = tau_init

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    lr = lr_init
    t = 0

    best_obj = conv.max()
    best_f = f.copy()
    no_improve = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        t += 1
        obj, grad, conv, w = smooth_obj_grad(f, tau)
        M = conv.max()

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        M_new = conv_new.max()

        if M_new < best_obj - 1e-14:
            best_obj = M_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new

        # Adaptive temperature based on convolution flatness
        sorted_conv = np.sort(conv_new)
        gap = sorted_conv[-1] - sorted_conv[-min(10, len(sorted_conv)):]
        avg_gap = gap.mean() if len(gap) > 0 else conv_new.max()
        target_tau = 0.3 * avg_gap
        tau = 0.95 * tau + 0.05 * target_tau
        tau = max(tau, 1e-9)

        if no_improve > 150:
            lr *= 0.8
        if no_improve > 500:
            rng = np.random.default_rng(t * 17 + 3)
            noise = 0.002 + 0.004 * min(no_improve / 1000.0, 2.0)
            f = project_simplex(best_f + rng.standard_normal(n) * noise)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = lr_init * 0.8
            tau = 0.05 * fast_convolve(f).max()
            if verbose:
                print(f"    Adam restart iter {t}: obj={2*n*best_obj:.8f}")

        if verbose and t % 500 == 0:
            print(f"    Adam iter {t}: obj={2*n*best_obj:.8f}, tau={tau:.2e}, lr={lr:.5f}")

    return best_f, best_obj, 2.0 * n * best_obj


def fw_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe on non-smooth max-convolution objective."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        gamma = 2.0 / (it + 2.0)

        conv = fast_convolve(f)
        M = conv.max()
        peak_idx = int(np.argmax(conv))

        # Subgradient direction
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        indices = np.arange(js, je + 1)
        grad_vals = f[peak_idx - indices]

        # Find FW vertex
        if js > 0 or je < n - 1:
            outside = np.setdiff1d(np.arange(n), indices)
            # Pick position with smallest f value among outside
            v_idx = int(outside[np.argmin(f[outside])])
        else:
            v_idx = int(indices[np.argmin(grad_vals)])

        v = np.zeros(n)
        v[v_idx] = 1.0

        f_new = (1.0 - gamma) * f + gamma * v
        f_new = np.maximum(f_new, 0.0)
        ss = f_new.sum()
        if ss > 1e-15:
            f_new /= ss

        conv_new = fast_convolve(f_new)
        M_new = conv_new.max()

        if M_new < best_obj - 1e-14:
            best_obj = M_new
            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    FW iter {it}: obj={2*n*best_obj:.8f}")
        else:
            no_improve += 1

        f = f_new

        if no_improve > 500:
            rng = np.random.default_rng(it * 19 + 11)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            no_improve = 0

        if no_improve > 3000:
            break

    return best_f, best_obj, 2.0 * n * best_obj


def peak_aware_perturb(f, strength=0.01, n_peaks=3):
    """Reduce peak-contributing elements, redistribute to non-contributing."""
    n = len(f)
    conv = fast_convolve(f)
    peak_indices = np.argsort(conv)[-n_peaks:]

    f_new = f.copy()
    for pk in peak_indices:
        pk = int(pk)
        js = max(0, pk - n + 1)
        je = min(n - 1, pk)
        indices = np.arange(js, je + 1)
        products = f_new[indices] * f_new[pk - indices]
        j_main = int(indices[np.argmax(products)])

        old_val = f_new[j_main]
        f_new[j_main] *= (1.0 - strength)
        diff = old_val - f_new[j_main]

        # Redistribute to non-contributing positions
        mask = np.ones(n, dtype=bool)
        for pk2 in peak_indices:
            pk2 = int(pk2)
            js2 = max(0, pk2 - n + 1)
            je2 = min(n - 1, pk2)
            mask[js2:je2 + 1] = False

        if mask.sum() > 0 and f_new[mask].sum() > 1e-15:
            f_new[mask] += diff * f_new[mask] / f_new[mask].sum()

    f_new = np.maximum(f_new, 0.0)
    ss = f_new.sum()
    if ss > 1e-15:
        f_new /= ss
    return f_new


def cosine_window(n):
    x = np.arange(n)
    a = np.cos(np.pi * (x + 0.5) / (2.0 * n))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def raised_cosine(n, width=0.3):
    x = np.arange(n) / (n - 1)
    t = (x - 0.5) / (0.5 - width)
    t = np.clip(t, -1.0, 1.0)
    a = 0.5 * (1.0 + np.cos(np.pi * t))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def asymmetric_gaussian(n, center=0.4, sigma=0.25):
    x = np.arange(n) - n * center
    a = np.exp(-0.5 * (x / (n * sigma)) ** 2)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def bimodal_init(n, pos1=0.3, pos2=0.7, sigma=0.12):
    x = np.arange(n)
    a = (np.exp(-0.5 * ((x - n * pos1) / (n * sigma)) ** 2) +
         np.exp(-0.5 * ((x - n * pos2) / (n * sigma)) ** 2))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def ring_init(n, power=2.0):
    x = np.arange(n) / (n - 1)
    a = (x * (1.0 - x) * 4.0) ** power
    a = np.maximum(a, 0.0)
    ss = a.sum()
    if ss > 1e-15:
        a /= ss
    return a


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: Diverse initialization
    Phase 2: FISTA smooth optimization (main)
    Phase 3: Adam refinement with flatness-adaptive temperature
    Phase 4: Peak-aware perturbation + re-optimization
    Phase 5: Frank-Wolfe non-smooth polish
    Phase 6: Multi-length exploration
    Phase 7: Final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
        return score

    # === Phase 1: Diverse Initialization ===
    print("=== Phase 1: Diverse Initialization ===")

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # Generate diverse initializations at multiple lengths
    inits = []
    for n_try in [512, 768, 1024]:
        inits.append((f"cos_{n_try}", cosine_window(n_try)))
        inits.append((f"rc_{n_try}", raised_cosine(n_try, 0.25)))
        inits.append((f"rc2_{n_try}", raised_cosine(n_try, 0.35)))
        inits.append((f"gauss_{n_try}", asymmetric_gaussian(n_try, 0.35, 0.22)))
        inits.append((f"gauss2_{n_try}", asymmetric_gaussian(n_try, 0.5, 0.18)))
        inits.append((f"bimodal_{n_try}", bimodal_init(n_try)))
        inits.append((f"ring_{n_try}", ring_init(n_try, 1.5)))
        inits.append((f"ring2_{n_try}", ring_init(n_try, 2.5)))
        # Exponential decay
        a_exp = np.exp(-np.arange(n_try) / (n_try * 0.4))
        a_exp /= a_exp.sum()
        inits.append((f"exp_{n_try}", a_exp))
        # Symmetric exponential
        x = np.arange(n_try) - n_try // 2
        a_sexp = np.exp(-0.5 * (x / (n_try * 0.3)) ** 2)
        a_sexp /= a_sexp.sum()
        inits.append((f"sexc_{n_try}", a_sexp))
        # Random with smoothing
        a_rnd = np.abs(rng.standard_normal(n_try))
        # Smooth it
        kernel = np.ones(7) / 7.0
        a_rnd = np.convolve(a_rnd, kernel, mode='same')
        a_rnd = np.maximum(a_rnd, 0)
        a_rnd /= a_rnd.sum()
        inits.append((f"rand_{n_try}", a_rnd))

    for label, a_init in inits:
        if time.monotonic() > deadline - 1400:
            break
        try_seq(a_init, f"init_{label}")

    if best_seq is None:
        best_seq = parent if parent is not None else cosine_window(768)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"  Best after init: score={best_score:.6f}, n={len(best_seq)}")

    # === Phase 2: FISTA Main Optimization ===
    print(f"\n=== Phase 2: FISTA Optimization (n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.30, 500)
    if t2 > 60:
        print(f"  Time: {t2:.0f}s")
        a_opt, obj, score = fista_optimize(best_seq, time_limit=t2, verbose=True)
        try_seq(a_opt, "fista_main")

    # === Phase 3: Adam Refinement ===
    print(f"\n=== Phase 3: Adam Refinement ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 60:
            t3 = min(rem, 400)
            print(f"  Time: {t3:.0f}s")
            a_opt, obj, score = adam_optimize(best_seq, time_limit=t3,
                                               tau_init=0.04 * fast_convolve(best_seq).max(),
                                               lr_init=0.003, verbose=True)
            try_seq(a_opt, "adam_refine")

    # === Phase 4: Peak-Aware Perturbation ===
    print(f"\n=== Phase 4: Peak-Aware Perturbation ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        for strength in [0.005, 0.01, 0.015, 0.025]:
            if rem < 40:
                break
            f_pert = peak_aware_perturb(best_seq, strength)
            sc = try_seq(f_pert, f"perturb_{strength}")
            # Quick refinement from perturbed point
            t4 = min(rem * 0.2, 60)
            if t4 > 20:
                a_opt, obj, score = adam_optimize(f_pert, time_limit=t4,
                                                   tau_init=0.03 * fast_convolve(f_pert).max(),
                                                   lr_init=0.002, verbose=False)
                try_seq(a_opt, f"perturb_opt_{strength}")
            rem = deadline - time.monotonic() - 200

    # === Phase 5: Frank-Wolfe ===
    print(f"\n=== Phase 5: Frank-Wolfe ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t5 = min(rem, 250)
            print(f"  Time: {t5:.0f}s")
            a_opt, obj, score = fw_optimize(best_seq, time_limit=t5, verbose=True)
            try_seq(a_opt, "fw")

    # === Phase 6: Multi-Length Exploration ===
    print(f"\n=== Phase 6: Multi-Length Exploration ===")
    if best_seq is not None:
        base = best_seq
        for target_n in [1024, 1280, 1536, 2048]:
            rem = deadline - time.monotonic() - 80
            if rem < 40:
                break
            f_interp = interpolate_sequence(base, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")
            # Quick FISTA on interpolated
            t6 = min(rem * 0.25, 60)
            if t6 > 15:
                a_opt, obj, score = fista_optimize(f_interp, time_limit=t6, verbose=False)
                try_seq(a_opt, f"interp_fista_n{target_n}")

    # === Phase 7: Final Polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 20:
            t7 = min(rem * 0.5, 60)
            a_opt, obj, score = adam_optimize(
                best_seq, time_limit=t7,
                tau_init=0.02 * fast_convolve(best_seq).max(),
                lr_init=0.001, verbose=False)
            try_seq(a_opt, "polish_adam")

        rem = deadline - time.monotonic() - 10
        if rem > 10:
            t8 = min(rem, 30)
            a_opt, obj, score = fw_optimize(best_seq, time_limit=t8, verbose=False)
            try_seq(a_opt, "polish_fw")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
