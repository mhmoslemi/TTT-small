"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
(1) Fast log-sum-exp Adam with Nesterov-style momentum and adaptive temperature,
(2) Direct max-subgradient with Nesterov acceleration (O(n) per iteration),

(4) Multi-length exploration with interpolation,
(5) Final coordinate descent polish.
Key innovation: exploits O(n) subgradient structure (flipped/shifted copy of a),

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_at_peak(a, k, n):
    """Compute subgradient of (a*a)[k] w.r.t. a: g[i] = 2*a[k-i] (0 if out of bounds)."""
    g = np.zeros(n)
    for j in range(max(0, k - n + 1), min(n, k + 1)):
        if 0 <= k - j < n:
            g[j] = 2.0 * a[k - j]
    return g


def subgradient_at_peak_fast(a, k, n):
    """O(n) subgradient computation using slice operations."""
    g = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    if je < js:
        return g
    idx = np.arange(js, je + 1)
    other = k - idx
    g[idx] = 2.0 * a[other]
    return g


def compute_smooth_grad_fft(f, tau):
    """
    Compute gradient of log(sum_k exp(conv[k]/tau)) via FFT.
    Returns (gradient, convolution, logsumexp_value).
    """
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / tau
    scaled_max = scaled.max()
    w = np.exp(scaled - scaled_max)
    w_sum = w.sum()
    w /= w_sum

    # h[j] = sum_{k} w[k] * a[k-j] = (w_reversed * a)[j] via FFT
    # Actually: h[j] = sum_k w[k] * f[k-j], which is correlation of w and f
    # = convolve(f, w_flipped)[n-1 : 2n-1]
    w_flipped = w[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_flipped, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)
    h = conv_result[n - 1: 2 * n - 1]

    grad = (1.0 / tau) * h
    lse = scaled_max + np.log(w_sum)
    return grad, conv, lse


def adam_smooth_optimize(f0, time_limit, tau_init=None, lr_init=0.003, tau_decay=0.9995,
                         restart_threshold=150, verbose=False):
    """
    Adam optimization of log-sum-exp smoothed max-convolution.
    With adaptive temperature decay and periodic restarts.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    M = conv.max()
    if tau_init is None:
        tau = 0.06 * M
    else:
        tau = tau_init

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    t = 0

    best_obj = M
    best_f = f.copy()
    no_improve = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        t += 1
        grad, conv, lse = compute_smooth_grad_fft(f, tau)
        M = conv.max()

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new

        # Adaptive temperature
        tau *= tau_decay
        if tau < 1e-6:
            tau = 1e-6

        # Adaptive learning rate
        if no_improve == restart_threshold:
            lr *= 0.7
        elif no_improve == restart_threshold + 50:
            lr *= 0.7
        elif no_improve > restart_threshold + 100:
            # Restart with perturbation
            rng = np.random.default_rng(t * 131 + 17)
            noise = 0.001 + 0.002 * min(no_improve / 500.0, 2.0)
            f = project_simplex(best_f + rng.standard_normal(n) * noise)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = lr_init * 0.6
            tau = 0.04 * fast_convolve(f).max()
            if verbose:
                print(f"      Adam restart at t={t}: score={2*n*best_obj:.8f}")

        if verbose and t % 1000 == 0:
            print(f"      Adam t={t}: score={2*n*best_obj:.8f}, tau={tau:.2e}, lr={lr:.5f}")

    return best_f, 2 * n * best_obj


def nesterov_subgradient_optimize(f0, time_limit, verbose=False):
    """
    Nesterov-accelerated subgradient method on the non-smooth max-convolution.
    Each iteration is O(n) since the subgradient is a flipped/shifted copy of a.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    f_prev = f.copy()
    y = f.copy()  # Nesterov extrapolation point

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    t = 0
    lr_base = 0.02

    # Track recent peaks for averaging subgradient
    peak_history = []
    peak_window = 20

    while time.monotonic() - t0 < time_limit:
        t += 1

        # Nesterov extrapolation
        beta_t = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta_prev = (1.0 + np.sqrt(1.0 + 4.0 * (t - 1) ** 2)) / 2.0
        momentum = (1 - beta_prev) / (1 - beta_t)

        y = f + momentum * (f - f_prev)
        y = np.maximum(y, 0)
        ys = y.sum()
        if ys > 1e-15:
            y = y / ys

        # Compute subgradient at y
        conv = fast_convolve(y)
        M = conv.max()

        # Use averaged subgradient over recent peaks for stability
        peak_idx = np.argmax(conv)
        peak_history.append(peak_idx)
        if len(peak_history) > peak_window:
            peak_history.pop(0)

        # Average subgradient over top peaks
        top_k = min(5, 2 * n - 1)
        top_peaks = np.argsort(conv)[-top_k:]
        peak_vals = conv[top_peaks]
        weights = peak_vals / peak_vals.sum()

        g_avg = np.zeros(n)
        for pk, w in zip(top_peaks, weights):
            g_avg += w * subgradient_at_peak_fast(y, int(pk), n)

        # Learning rate decay
        lr_t = lr_base / np.sqrt(t + 1)

        f_prev = f.copy()
        f_new = y - lr_t * g_avg
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()

        f = f_new

        if verbose and t % 5000 == 0:
            print(f"      Nesterov t={t}: score={2*n*best_obj:.8f}")

    return best_f, 2 * n * best_obj


def mirror_descent_optimize(f0, time_limit, verbose=False):
    """
    Mirror descent (exponential/multiplicative updates) on the smoothed objective.
    Uses log-sum-exp smoothing with adaptive temperature.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    M = conv.max()
    tau = 0.05 * M

    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    t = 0
    lr = 0.01

    while time.monotonic() - t0 < time_limit:
        t += 1
        grad, conv, lse = compute_smooth_grad_fft(f, tau)
        M = conv.max()

        # Mirror descent update: f[i] <- f[i] * exp(-lr * grad[i]), then normalize
        log_f = np.log(f + 1e-300) - lr * grad
        log_f -= log_f.max()  # numerical stability
        f_new = np.exp(log_f)
        f_new_sum = f_new.sum()
        if f_new_sum < 1e-30:
            f_new = f.copy()
        else:
            f_new /= f_new_sum

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()

        f = f_new
        tau *= 0.9995
        if tau < 1e-6:
            tau = 1e-6

        if t % 100 == 0:
            lr *= 0.95

        if verbose and t % 2000 == 0:
            print(f"      Mirror t={t}: score={2*n*best_obj:.8f}")

    return best_f, 2 * n * best_obj


def multi_peak_subgradient(f0, time_limit, n_top=10, verbose=False):
    """
    Subgradient method using weighted average of top-k peak subgradients.
    With adaptive step size (Polyak-style).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    t = 0
    lr = 0.05

    # Store best f encountered for Polyak average
    f_bar = f.copy()
    n_bar = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)
        M = conv.max()

        if M < best_obj - 1e-15:
            best_obj = M
            best_f = f.copy()

        # Top-k peaks with softmax weighting
        top_k = min(n_top, 2 * n - 1)
        top_indices = np.argsort(conv)[-top_k:]
        top_vals = conv[top_indices]
        # Softmax weights (sharper for more focused correction)
        shifted = top_vals - top_vals.max()
        w = np.exp(shifted / (0.01 * M + 1e-30))
        w /= w.sum()

        # Average subgradient
        g = np.zeros(n)
        for pk, wi in zip(top_indices, w):
            g += wi * subgradient_at_peak_fast(f, int(pk), n)

        # Update
        f_new = f - lr * g
        f_new = project_simplex(f_new)

        # Update Polyak average
        f_bar = (n_bar * f_bar + f_new) / (n_bar + 1)
        n_bar += 1

        f = f_new

        # Adaptive lr
        if t % 200 == 0:
            lr *= 0.97
            if lr < 0.001:
                lr = 0.001

        if verbose and t % 5000 == 0:
            print(f"      MultiPeak t={t}: score={2*n*best_obj:.8f}")

    # Try Polyak average
    conv_bar = fast_convolve(f_bar)
    obj_bar = conv_bar.max()
    n_bar_norm = f_bar.sum()
    if n_bar_norm > 1e-15:
        f_bar = f_bar / n_bar_norm
        conv_bar = fast_convolve(f_bar)
        obj_bar = conv_bar.max()
        if obj_bar < best_obj:
            best_obj = obj_bar
            best_f = f_bar

    return best_f, 2 * n * best_obj


def coordinate_descent_polish(f0, time_limit, verbose=False):
    """
    Coordinate descent: modify pairs of coordinates to reduce the peak.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()
        improved = False

        # Get the peak and top contributors
        peak_idx = int(np.argmax(conv))

        # Find which pairs contribute to this peak
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        idx = np.arange(js, je + 1)
        other = peak_idx - idx
        products = f[idx] * f[other]

        # Try reducing the largest contributor
        j_max = idx[np.argmax(products)]
        j_other = int(peak_idx - j_max)

        if j_max == j_other:
            # Diagonal: reduce f[j_max]
            for factor in [0.5, 0.7, 0.85]:
                trial = f.copy()
                old = trial[j_max]
                trial[j_max] *= factor
                diff = old - trial[j_max]
                # Redistribute to elements not contributing to top peaks
                top_peaks = np.argsort(conv)[-3:]
                mask = np.ones(n, dtype=bool)
                for pk in top_peaks:
                    js2 = max(0, int(pk) - n + 1)
                    je2 = min(n - 1, int(pk))
                    mask[js2:je2 + 1] = False
                if mask.sum() > 0 and f[mask].sum() > 1e-15:
                    trial[mask] += diff * f[mask] / f[mask].sum()
                trial = np.maximum(trial, 0)
                ts = trial.sum()
                if ts > 1e-15:
                    trial /= ts
                    obj = fast_convolve(trial).max()
                    if obj < best_obj - 1e-15:
                        f = trial
                        best_obj = obj
                        best_f = f.copy()
                        improved = True
                        break
        else:
            # Off-diagonal: reduce the product
            for scale1 in [0.5, 0.7, 0.85]:
                trial = f.copy()
                old = trial[j_max]
                trial[j_max] *= scale1
                diff = old - trial[j_max]
                mask = np.ones(n, dtype=bool)
                mask[j_max] = False
                mask[j_other] = False
                top_peaks = np.argsort(conv)[-3:]
                for pk in top_peaks:
                    js2 = max(0, int(pk) - n + 1)
                    je2 = min(n - 1, int(pk))
                    mask[js2:je2 + 1] = False
                if mask.sum() > 0 and f[mask].sum() > 1e-15:
                    trial[mask] += diff * f[mask] / f[mask].sum()
                trial = np.maximum(trial, 0)
                ts = trial.sum()
                if ts > 1e-15:
                    trial /= ts
                    obj = fast_convolve(trial).max()
                    if obj < best_obj - 1e-15:
                        f = trial
                        best_obj = obj
                        best_f = f.copy()
                        improved = True
                        break

        if not improved:
            no_improve += 1
            if no_improve > 20:
                rng = np.random.default_rng(int(time.monotonic() * 10000) % (2**31))
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                no_improve = 0
        else:
            no_improve = 0

    return best_f, 2 * n * best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_parabolic(n, curvature=0.5):
    """Parabolic (concave) initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 + curvature * (2*x - 1)**2
    a = np.maximum(a, 0.01)
    a /= a.sum()
    return a


def make_cosine_modulated(n, freq=1.0, amplitude=0.3):
    """Cosine-modulated initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 + amplitude * np.cos(2 * np.pi * freq * x)
    a = np.maximum(a, 0.01)
    a /= a.sum()
    return a


def make_two_harmonic(n, f1=1.0, f2=3.0, a1=0.2, a2=0.1):
    """Two-harmonic modulation."""
    x = np.linspace(0, 1, n)
    a = 1.0 + a1 * np.cos(2 * np.pi * f1 * x) + a2 * np.cos(2 * np.pi * f2 * x)
    a = np.maximum(a, 0.01)
    a /= a.sum()
    return a

    n_parent = len(parent) if parent is not None else 768

    # === Phase 1: Diverse initialization ===
    print("\n=== Phase 1: Initialization ===")
    if parent is None:
        parent = make_cosine_modulated(768, 1.0, 0.3)
        try_seq(parent, "default_init")

    # Try various parametric initializations
    inits = []
    for n_try in [n_parent, 512, 1024]:
        inits.append((make_parabolic(n_try, 0.3), f"parabolic_n{n_try}"))
        inits.append((make_parabolic(n_try, 0.7), f"parabolic7_n{n_try}"))
        inits.append((make_cosine_modulated(n_try, 1.0, 0.2), f"cos1_n{n_try}"))
        inits.append((make_cosine_modulated(n_try, 2.0, 0.15), f"cos2_n{n_try}"))
        inits.append((make_two_harmonic(n_try, 1.0, 3.0, 0.2, 0.1), f"2harm_n{n_try}"))

    for a_init, label in inits:
        if time.monotonic() > deadline - 1500:
            break
        try_seq(a_init, label)

    work_seq = best_seq if best_seq is not None else parent
    work_n = len(work_seq)
    print(f"  Starting optimization: n={work_n}, score={best_score:.6f}")

    # === Phase 2: Adam smooth optimization (primary) ===
    print(f"\n=== Phase 2: Adam Smooth (n={work_n}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem, 700)
    if t2 > 100:
        print(f"  Time: {t2:.0f}s")
        a_opt, obj = adam_smooth_optimize(work_seq, time_limit=t2 * 0.55,
                                          lr_init=0.004, tau_decay=0.9994,
                                          verbose=True)
        try_seq(a_opt, "adam_main")

        # Second pass with different params
        rem2 = deadline - time.monotonic() - 500
        if rem2 > 80:
            t2b = min(rem2, 300)
            a_opt2, obj2 = adam_smooth_optimize(best_seq, time_limit=t2b,
                                                tau_init=0.03 * fast_convolve(best_seq).max(),
                                                lr_init=0.002, tau_decay=0.9997,
                                                verbose=True)
            try_seq(a_opt2, "adam_fine")

    # === Phase 3: Nesterov subgradient ===
    print(f"\n=== Phase 3: Nesterov Subgradient ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 50:
            t3 = min(rem, 250)
            print(f"  Time: {t3:.0f}s")
            a_opt, obj = nesterov_subgradient_optimize(best_seq, time_limit=t3, verbose=True)
            try_seq(a_opt, "nesterov")

    # === Phase 4: Mirror descent ===
    print(f"\n=== Phase 4: Mirror Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 350
        if rem > 50:
            t4 = min(rem, 200)
            print(f"  Time: {t4:.0f}s")
            a_opt, obj = mirror_descent_optimize(best_seq, time_limit=t4, verbose=True)
            try_seq(a_opt, "mirror")

    # === Phase 5: Multi-peak subgradient ===
    print(f"\n=== Phase 5: Multi-Peak Subgradient ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 50:
            t5 = min(rem, 200)
            print(f"  Time: {t5:.0f}s")
            a_opt, obj = multi_peak_subgradient(best_seq, time_limit=t5, n_top=8, verbose=True)
            try_seq(a_opt, "multipk")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-Length Exploration ===")
    if best_seq is not None:
        base = best_seq
        for target_n in [1024, 1280, 640, 512, 1536]:
            rem = deadline - time.monotonic() - 80
            if rem < 40:
                break
            f_interp = interpolate_sequence(base, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")

            # Quick Adam on interpolated
            t6 = min(rem * 0.3, 80)
            if t6 > 15:
                a_opt, obj = adam_smooth_optimize(f_interp, time_limit=t6,
                                                  lr_init=0.003, tau_decay=0.9995)
                try_seq(a_opt, f"adam_n{target_n}")

    # === Phase 7: Second round of Adam from best ===
    print(f"\n=== Phase 7: Adam Round 2 ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 30:
            t7 = min(rem, 150)
            # Perturb and re-optimize
            n_b = len(best_seq)
            perturbed = project_simplex(best_seq + rng.standard_normal(n_b) * 0.003)
            a_opt, obj = adam_smooth_optimize(perturbed, time_limit=t7 * 0.6,
                                              lr_init=0.003, tau_decay=0.9996, verbose=True)
            try_seq(a_opt, "adam_r2_pert")

    # === Phase 8: Coordinate descent polish ===
    print(f"\n=== Phase 8: CD Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t8 = min(rem, 100)
            a_opt, obj = coordinate_descent_polish(best_seq, time_limit=t8, verbose=True)
            try_seq(a_opt, "cd_polish")

    # === Phase 9: Final Adam squeeze ===
    print(f"\n=== Phase 9: Final Squeeze ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            t9 = min(rem, 50)
            conv_b = fast_convolve(best_seq)
            a_opt, obj = adam_smooth_optimize(best_seq, time_limit=t9,
                                              tau_init=0.015 * conv_b.max(),
                                              lr_init=0.001, tau_decay=0.9999)
            try_seq(a_opt, "final_squeeze")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
