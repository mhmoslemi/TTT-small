```python
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Adam optimization in log-space
with soft-max gradient, periodic LP-Frank-Wolfe refinement, and multi-length
exploration. Key innovations: FFT-accelerated soft-gradient, adaptive temperature
scheduling, and hybrid switching between gradient and LP-based methods.

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def obj_normalized(f):
    """Objective for sum-normalized sequence: 2*n*max(conv)."""
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)

def build_lp_matrix(f, n):
    """Build constraint matrix for LP."""
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        length = k_end - k_start + 1
        A_ub[k_start:k_end + 1, j] = f[:length]
    return A_ub

def solve_lp_direction(f, M, n):
    """Solve LP direction."""
    A_ub = build_lp_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def soft_gradient(f, beta):
    """Compute gradient of 2n * soft_max(conv(f,f)) using softmax weights."""
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv * beta
    m = scaled.max()
    w = np.exp(scaled - m)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[m.argmax()] = 1.0
    else:
        w /= w_sum

        js = max(0, k - n + 1)
        je = min(n - 1, k)

    grad *= 4.0 * n  # 2 (from chain rule) * 2n (objective coeff)
    return grad, conv, w

    v_hat = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = lr_init
    beta = beta_init

    t0 = time.monotonic()
    no_improve = 0
    it = 0

        it += 1
        t += 1
        
        # Compute gradient
        grad_f, conv, w = soft_gradient(f, beta)

        # dF/dtheta[j] = f[j] * (dF/df[j] - sum_i dF/df[i] * f[i])
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using reduced-constraint LP Frank-Wolfe
with momentum, adaptive constraint selection, FFT-accelerated evaluation,
coordinate descent polishing, and multi-length exploration.
Key improvements: sparse LP (top-k binding constraints), directional momentum,
polynomial line search, and aggressive local refinement.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_normalized(f):
    """Objective for sum-normalized sequence: 2*n*max(conv)."""
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_reduced_lp(f, M, n, top_k=60):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for top-k binding constraints, g >= 0."""
    conv = fast_convolve(f)
    # Find the top-k indices where conv is largest (most binding)
    if top_k >= len(conv):
        top_k = len(conv)
    threshold_idx = np.argpartition(conv, -top_k)[-top_k:]
    threshold_idx = threshold_idx[np.argsort(conv[threshold_idx])[::-1]]

    k = len(threshold_idx)
    A_ub = np.zeros((k, n))
    b_ub = np.full(k, M)

    for row_i, ck in enumerate(threshold_idx):
        ck = int(ck)
        js = max(0, ck - n + 1)
        je = min(n - 1, ck)
        # A_ub[row_i, j] = f[ck - j] for j in [js, je]
        # f[ck-je] ... f[ck-js]
        A_ub[row_i, js:je + 1] = f[ck - je:ck - js + 1][::-1]

    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 5})
    if res.success:
        return np.maximum(res.x, 0)
    return None


def polynomial_line_search(f, g, f_obj, n, max_evals=8):
    """
    Find alpha in [0,1] minimizing obj(f + alpha*(g-f)) projected to simplex.
    Uses coarse grid + quadratic interpolation.
    """
    # Coarse grid
    alphas = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.85, 1.0]
    vals = []
    for a in alphas:
        ft = f + a * (g - f)
        ft = np.maximum(ft, 0.0)
        s = ft.sum()
        if s < 1e-15:
            vals.append(f_obj)
        else:
            ft = ft / s
            vals.append(obj_normalized(ft))
    vals = np.array(vals)
    best_i = int(np.argmin(vals))
    best_alpha = alphas[best_i]
    best_val = vals[best_i]

    # Refine with quadratic interpolation using 3 best points
    sorted_idx = np.argsort(vals)[:5]
    if len(sorted_idx) >= 3:
        i1, i2, i3 = sorted_idx[0], sorted_idx[1], sorted_idx[2]
        x1, x2, x3 = alphas[i1], alphas[i2], alphas[i3]
        y1, y2, y3 = vals[i1], vals[i2], vals[i3]
        # Quadratic fit
        denom = (x1 - x2) * (x1 - x3) * (x2 - x3)
        if abs(denom) > 1e-12:
            a_q = (x3 * (y2 - y1) + x2 * (y1 - y3) + x1 * (y3 - y2)) / denom
            b_q = (x3*x3*(y1-y2) + x2*x2*(y3-y1) + x1*x1*(y2-y3)) / denom
            if a_q > 1e-8:
                x_star = -b_q / (2 * a_q)
                if 0 < x_star < 1:
                    ft = f + x_star * (g - f)
                    ft = np.maximum(ft, 0.0)
                    s = ft.sum()
                    if s > 1e-15:
                        ft = ft / s
                        v_star = obj_normalized(ft)
                        if v_star < best_val:
                            best_alpha = x_star
                            best_val = v_star
                    evals_left = max_evals - 1
                    if v_star < best_val:
                        # Fine search around x_star
                        for da in [0.02, -0.02, 0.01, -0.01]:
                            if evals_left <= 0:
                                break
                            evals_left -= 1
                            ax = x_star + da
                            if 0 <= ax <= 1:
                                ft2 = f + ax * (g - f)
                                ft2 = np.maximum(ft2, 0.0)
                                s2 = ft2.sum()
                                if s2 > 1e-15:
                                    ft2 = ft2 / s2
                                    v2 = obj_normalized(ft2)
                                    if v2 < best_val:
                                        best_val = v2
                                        best_alpha = ax
    return best_alpha, best_val


def lp_fw_optimize(f0, time_limit, verbose=False, n_constraints=60, momentum_beta=0.5):
    """
    LP Frank-Wolfe with reduced constraints, momentum, and polynomial line search.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    momentum_dir = None
    n_cons = n_constraints

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        it += 1

        # Solve reduced LP
        g = solve_reduced_lp(f, M * m_exp, n, top_k=n_cons)
        if g is None:
            m_exp *= 1.2
            if m_exp > 8.0:
                # Perturb to escape
                rng = np.random.default_rng(it * 13 + 5)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
                momentum_dir = None
            continue

        g = np.maximum(g, 0)
        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp *= 1.1
            continue
        g = g / g_sum

        # Apply momentum
        if momentum_dir is not None:
            g = (1 - momentum_beta) * g + momentum_beta * momentum_dir
            g = np.maximum(g, 0)
            gs = g.sum()
            if gs > 1e-15:
                g = g / gs
        momentum_dir = g.copy()

        diff = np.max(np.abs(g - f))
        if diff < 1e-10:
            no_improve += 2
            if no_improve > 30:
                m_exp = min(m_exp * 1.1, 8.0)
            if no_improve > 100:
                rng = np.random.default_rng(it * 29 + 11)
                scale = 0.004 * min(m_exp, 3.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
                momentum_dir = None
            continue
        elif diff > 1e-4:
            no_improve = 0
        elif diff > 1e-6:
            no_improve += 1

        # Line search
        alpha, alpha_val = polynomial_line_search(f, g, M, n, max_evals=8)
        if alpha < 1e-8:
            no_improve += 1
            continue

        f_new = f + alpha * (g - f)
        f_new = np.maximum(f_new, 0.0)
        sn = f_new.sum()
        if sn < 1e-15:
            no_improve += 1
            continue
        f_new = f_new / sn

        conv_new = fast_convolve(f_new)
        M_new = conv_new.max()
        obj_new = 2.0 * n * M_new

        f = f_new
        M = M_new
        m_exp = 1.0

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt constraint count
        if no_improve > 20:
            n_cons = min(120, n_cons + 10)
        elif no_improve == 0:
            n_cons = max(30, n_cons - 5)

        if verbose and it % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW iter {it}: obj={best_obj:.10f}, M={M:.6f}, "
                  f"n_cons={n_cons}, m_exp={m_exp:.2f}, t={elapsed:.1f}s")

        if no_improve > 200:
            break

    return best_f, best_obj


def coordinate_descent_polish(f0, time_limit, verbose=False):
    """
    Aggressive coordinate descent targeting peak convolution values.
    Tries reducing top contributors to each peak and redistributing.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = obj_normalized(f)
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit and no_improve < 80:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        improved = False

        # Target top peaks
        top_k = min(8, len(conv))
        peak_indices = np.argsort(conv)[-top_k:]

        for k in peak_indices:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue

            idx = np.arange(js, je + 1)
            products = f[idx] * f[k - idx]
            total = products.sum()
            if total < 1e-15:
                continue

            # Sort by contribution
            order = np.argsort(products)[::-1]
            n_try = min(5, len(order))

            for oi in range(n_try):
                j_target = int(idx[order[oi]])
                old_val = f[j_target]
                if old_val < 1e-12:
                    continue

                # Try several reduction factors
                for mult in [0.6, 0.7, 0.8, 0.9, 1.1, 1.3]:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.4:
                        continue

                    trial = f.copy()
                    trial[j_target] = new_val
                    d = old_val - new_val

                    # Redistribute to non-contributing positions
                    if abs(d) > 1e-15:
                        non_peak = np.ones(n, dtype=bool)
                        non_peak[js:je + 1] = False
                        cnt = non_peak.sum()
                        if cnt > 0:
                            w = f[non_peak].copy()
                            wsum = w.sum()
                            if wsum > 1e-15:
                                trial[non_peak] += d * w / wsum
                            else:
                                trial[non_peak] += d / cnt

                    trial = np.maximum(trial, 0)
                    tsum = trial.sum()
                    if tsum < 1e-15:
                        continue
                    trial = trial / tsum
                    obj = obj_normalized(trial)
                    if obj < best_obj - 1e-13:
                        best_obj = obj
                        best_f = trial.copy()
                        f = trial
                        improved = True
                        # Recompute products after update
                        idx = np.arange(js, je + 1)
                        products = f[idx] * f[k - idx]

        if not improved:
            no_improve += 1
            if no_improve > 15:
                # Aggressive: scale down all top peak contributors
                k_peak = int(np.argmax(fast_convolve(f)))
                js = max(0, k_peak - n + 1)
                je = min(n - 1, k_peak)
                trial = f.copy()
                trial[js:je + 1] *= 0.9
                d = f.sum() - trial.sum()
                non_contrib = np.ones(n, dtype=bool)
                non_contrib[js:je + 1] = False
                nc_sum = f[non_contrib].sum()
                if nc_sum > 1e-15:
                    trial[non_contrib] += d * f[non_contrib] / nc_sum
                trial = np.maximum(trial, 0)
                trial = trial / trial.sum()
                obj = obj_normalized(trial)
                if obj < best_obj:
                    f = trial
                    best_obj = obj
                    best_f = f.copy()
                    no_improve = 0
                else:
                    break
        else:
            no_improve = 0

        if verbose and it % 10 == 0:
            print(f"    CD iter {it}: obj={best_obj:.10f}, no_improve={no_improve}")

    return best_f, best_obj


def multiplicative_descent(f0, time_limit, verbose=False):
    """
    Multiplicative weight descent targeting top peaks with adaptive temperature.
    """
    f = np.maximum(f0, 1e-15).astype(np.float64)
    f = f / f.sum()
    n = len(f)

    best_obj = obj_normalized(f)
    best_f = f.copy()
    t0 = time.monotonic()
    lr = 0.006
    temperature = 0.001
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)

        # Select top peaks
        k_actual = min(20, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]

        # Soft weights
        scaled = pv / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Gradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-10:
                continue
            kk = int(peaks[i])
            js = max(0, kk - n + 1)
            je = min(n - 1, kk)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[kk - idx]

        # Multiplicative update
        f = f * np.exp(-lr * sg)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        co = obj_normalized(f)
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt
        lr *= 0.9995
        temperature *= 1.01

        if no_improve > 100:
            rng = np.random.default_rng(it * 37 + 7)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.003, 1e-15)
            f = f / f.sum()
            lr = max(0.004, lr)
            no_improve = 0

        if verbose and it % 200 == 0:
            print(f"    MD iter {it}: obj={best_obj:.10f}, lr={lr:.5f}, temp={temperature:.4f}")

        if lr < 1e-7:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, sigma_frac=0.25):
    """Gaussian initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def try_sequence(a, best_seq, best_score, label=""):
    """Evaluate a candidate sequence, return updated best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining reduced-LP Frank-Wolfe, coordinate descent,
    multiplicative descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_gaussian(768, 0.25)
        best_seq, best_score = try_sequence(parent, best_seq, best_score, "default")

    n_parent = len(parent)

    # === Phase 1: Quick multi-length initializations (5% time) ===
    print("=== Phase 1: Initializations ===")
    t1_start = time.monotonic()
    t1_budget = 0.05 * (budget_s - 30)
    rem_after_p1 = t1_start + t1_budget

    n_tries = [n_parent]
    for dn in [-64, -32, 32, 64, 128]:
        nt = n_parent + dn
        if 128 <= nt <= 2048:
            n_tries.append(nt)

    best_init = None
    best_init_obj = float('inf')

    for nt in n_tries:
        if time.monotonic() > rem_after_p1:
            break
        f_interp = interpolate_sequence(parent, nt)
        obj = obj_normalized(f_interp)
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = f_interp.copy()
        best_seq, best_score = try_sequence(f_interp, best_seq, best_score, f"init_n{nt}")

    # Also try gaussians
    for sigma in [0.2, 0.25, 0.3, 0.35, 0.4]:
        fg = make_gaussian(n_parent, sigma)
        best_seq, best_score = try_sequence(fg, best_seq, best_score, f"gauss_{sigma}")

    if best_init is None:
        best_init = parent.copy()

    print(f"  Best init: obj={best_init_obj:.6f} (n={len(best_init)})")

    # === Phase 2: Main LP-FW (45% time) ===
    print(f"\n=== Phase 2: Main LP-FW (n={len(best_init)}) ===")
    t2_budget = 0.45 * (budget_s - 30)
    t2_alloc = min(t2_budget, max(10, deadline - time.monotonic() - int(0.4 * (budget_s - 30))))
    if t2_alloc > 30:
        print(f"  Allocating {t2_alloc:.0f}s")
        a_opt, obj = lp_fw_optimize(best_init, time_limit=t2_alloc, verbose=True,
                                    n_constraints=60, momentum_beta=0.3)
        best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "lpfw_main")
    else:
        print("  Not enough time, skipping main LP-FW")

    # === Phase 3: Coordinate descent polish (15% time) ===
    if best_seq is not None:
        print(f"\n=== Phase 3: CD Polish (n={len(best_seq)}) ===")
        t3_budget = 0.15 * (budget_s - 30)
        t3_alloc = min(t3_budget, max(5, deadline - time.monotonic() - int(0.25 * (budget_s - 30))))
        if t3_alloc > 15:
            print(f"  Allocating {t3_alloc:.0f}s")
            a_opt, obj = coordinate_descent_polish(best_seq, time_limit=t3_alloc, verbose=True)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "cd_polish")

    # === Phase 4: Multiplicative descent (15% time) ===
    if best_seq is not None:
        print(f"\n=== Phase 4: Multiplicative Descent (n={len(best_seq)}) ===")
        t4_budget = 0.15 * (budget_s - 30)
        t4_alloc = min(t4_budget, max(5, deadline - time.monotonic() - int(0.15 * (budget_s - 30))))
        if t4_alloc > 15:
            print(f"  Allocating {t4_alloc:.0f}s")
            a_opt, obj = multiplicative_descent(best_seq, time_limit=t4_alloc, verbose=True)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "md")

    # === Phase 5: Second LP-FW with different params (10% time) ===
    if best_seq is not None:
        print(f"\n=== Phase 5: LP-FW Round 2 (n={len(best_seq)}) ===")
        t5_budget = 0.10 * (budget_s - 30)
        t5_alloc = min(t5_budget, max(5, deadline - time.monotonic() - int(0.1 * (budget_s - 30))))
        if t5_alloc > 20:
            print(f"  Allocating {t5_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t5_alloc, verbose=True,
                                        n_constraints=100, momentum_beta=0.5)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "lpfw_r2")

    # === Phase 6: Multi-length exploration (10% time) ===
    base = best_seq if best_seq is not None else best_init
    print(f"\n=== Phase 6: Multi-length Exploration ===")
    for target_n in [n_parent - 32, n_parent + 32, n_parent + 64]:
        if target_n < 128 or target_n > 2048:
            continue
        if len(base) == target_n:
            continue
        rem = deadline - time.monotonic() - int(0.05 * (budget_s - 30))
        if rem < 30:
            break
        t_alloc = min(rem * 0.4, 60)
        if t_alloc < 15:
            continue
        f_interp = interpolate_sequence(base, target_n)
        best_seq, best_score = try_sequence(f_interp, best_seq, best_score, f"interp_n{target_n}")
        print(f"  LP-FW n={target_n}: t={t_alloc:.0f}s")
        a_opt, obj = lp_fw_optimize(f_interp, time_limit=t_alloc, verbose=False,
                                    n_constraints=50, momentum_beta=0.3)
        best_seq, best_score = try_sequence(a_opt, best_seq, best_score, f"lpfw_n{target_n}")

    # === Phase 7: Final CD + LP-FW polish (5% time) ===
    if best_seq is not None:
        print(f"\n=== Phase 7: Final Polish ===")
        rem = deadline - time.monotonic() - 5
        if rem > 20:
            t_cd = min(rem * 0.4, 30)
            t_lpfw = min(rem * 0.5, 40)
            t_left = rem - t_cd - t_lpfw - 5

            if t_cd > 10:
                print(f"  Final CD: t={t_cd:.0f}s")
                a_opt, obj = coordinate_descent_polish(best_seq, time_limit=t_cd)
                best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "fin_cd")

            rem2 = deadline - time.monotonic() - 5
            if rem2 > 10 and best_seq is not None:
                t_lpfw2 = min(t_lpfw, rem2 - 5)
                if t_lpfw2 > 10:
                    print(f"  Final LP-FW: t={t_lpfw2:.0f}s")
                    a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_lpfw2,
                                                n_constraints=80, momentum_beta=0.4)
                    best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "fin_lpfw")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
