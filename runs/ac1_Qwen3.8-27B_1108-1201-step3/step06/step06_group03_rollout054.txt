Actually, c_k = sum_j a_j a_{k-j} = a^T P_k a where P_k is the permutation matrix that reverses indices (P_k[j, l] = 1 if j + l = k).

The gradient of a^T P a with respect to a is (P + P^T) a. Since P_k is symmetric (P_k[j,l] = 1 iff j+l=k, which is symmetric), the gradient is 2*P_k*a.

(P_k*a)[j] = a[k-j]. So the gradient is 2*a[k-j]. Confirmed.

So the bug is real. The fix should improve optimization quality.

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)

def compute_corrected_subgradient(f, c, top_frac=0.15, temperature=0.01):
    """Compute weighted subgradient of max(c) using CORRECT derivative: 2*a[k-j]."""
    n = len(f)
    len_c = len(c)
    k_top = max(5, int(len_c * top_frac))
    peaks = np.argpartition(c, -k_top)[-k_top:]
    pv = c[peaks]
    scaled = pv / max(temperature, 1e-10)
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:

        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)

def smoothed_mirror_descent(f0, time_limit, verbose=False,
                            tau_init=5.0, lr_init=0.01, tau_decay=1.003,
                            lr_decay=0.9998, restart_every=200):

    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()

    lr = lr_init
    no_improve = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        conv = fast_convolve(f)
        M = conv.max()

            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)

                continue
            idx = np.arange(js, je + 1)
            sg[js:je + 1] += w[i] * 2.0 * f[k - idx]

        f_new = f * np.exp(-lr * sg)
        f_new = np.maximum(f_new, 1e-15)
        s = f_new.sum()
        if s < 1e-15:
            f_new = f.copy()
        else:
            f_new = f_new / s

        obj_new = fast_convolve(f_new).max()

                f_trial = f * (1.0 - damp) + f_new * damp
                f_trial = np.maximum(f_trial, 1e-15)
                f_trial = f_trial / f_trial.sum()
                obj_trial = fast_convolve(f_trial).max()
                if obj_trial < M:
                    f_new = f_trial
                    obj_new = obj_trial
                    break
            else:
                lr *= 0.5
                continue

        f = f_new

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        tau *= tau_decay
        lr *= lr_decay

        if no_improve > restart_every:
            rng = np.random.default_rng(iter_count * 7 + 13)
            noise_scale = 0.003
            f = np.maximum(best_f + rng.standard_normal(n) * noise_scale, 1e-15)
            f = f / f.sum()
            lr = lr_init * 0.3
            tau = best_tau_for_restart
            no_improve = 0

        if lr < 1e-7:
            break

    return best_f, best_obj

    """Compute log-sum-exp smoothed objective and its gradient (corrected)."""
    conv = fast_convolve(f)
    tc = t * conv
    tc_max = tc.max()
    exp_tc = np.exp(tc - tc_max)
    lse_val = tc_max + np.log(exp_tc.sum())
    
    # Objective: 2*n*lse/t
    obj = 2.0 * n * lse_val / t
    
    # Gradient of LSE with respect to conv: softmax(t*conv)
    weights = exp_tc / exp_tc.sum()
    
    # Gradient with respect to f (using CORRECT derivative: 2*f[k-j])
    grad_f = np.zeros(n)
    # Only iterate over significant weights for efficiency
    k_top = max(20, int(len(conv) * 0.3))
    top_idx = np.argpartition(weights, -k_top)[-k_top:]

        if w < 1e-15:
            continue
        js = max(0, int(k) - n + 1)
        je = min(n - 1, int(k))
        idx = np.arange(js, je + 1)
        grad_f[js:je+1] += w * 2.0 * f[int(k) - idx]
    
    grad_obj = 2.0 * n * grad_f / t
    return obj, grad_obj

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:

    return f_interp

    """Smoothed mirror descent with CORRECTED gradient (2*a[k-j])."""
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    tau = tau_init
    lr = lr_init
    no_improve = 0
    iter_count = 0
    best_tau = tau

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        conv = fast_convolve(f)
        M = conv.max()

        len_c = len(conv)
        k_top = max(10, int(len_c * top_frac))
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        pv = conv[peaks]
        scaled = tau * pv
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # CORRECTED gradient: 2*f[k - idx]
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue
            idx = np.arange(js, je + 1)
            sg[js:je + 1] += w[i] * 2.0 * f[k - idx]

        f_new = f * np.exp(-lr * sg)
        f_new = np.maximum(f_new, 1e-15)
        s = f_new.sum()
        if s < 1e-15:
            f_new = f.copy()
        else:
            f_new = f_new / s

        obj_new = fast_convolve(f_new).max()
        if obj_new > M * 1.05:
            improved = False
            for damp in [0.5, 0.25, 0.1, 0.05, 0.02]:
                f_trial = f * (1.0 - damp) + f_new * damp
                f_trial = np.maximum(f_trial, 1e-15)
                f_trial = f_trial / f_trial.sum()
                obj_trial = fast_convolve(f_trial).max()
                if obj_trial < M:
                    f_new = f_trial
                    obj_new = obj_trial
                    improved = True
                    break
            if not improved:
                lr *= 0.5
                if lr < 1e-8:
                    break
                continue

        f = f_new

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
            best_tau = tau
            if verbose and iter_count % 200 == 0:
                print(f"      SMD iter {iter_count}: obj={best_obj:.10f}, tau={tau:.1f}")
        else:
            no_improve += 1

        tau *= tau_decay
        lr *= lr_decay

        if no_improve > restart_every:
            seed = iter_count * 7 + 13
            rng = np.random.default_rng(seed)
            noise_scale = 0.002 + 0.003 * min(1.0, no_improve / (restart_every * 2))
            f = np.maximum(best_f + rng.standard_normal(n) * noise_scale, 1e-15)
            f = f / f.sum()
            lr = lr_init * 0.5
            tau = best_tau
            no_improve = 0

    return best_f, best_obj


def _softmax_lse_func_and_grad(x, n, temp, k_top_frac):
    """Compute objective and gradient for softmax-parameterized LSE objective."""
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    a = exp_x / exp_x.sum()

    conv = fast_convolve(a)
    tc = temp * conv
    tc_max = tc.max()
    exp_tc = np.exp(tc - tc_max)
    lse_val = tc_max + np.log(exp_tc.sum())

    obj = 2.0 * n * lse_val / temp

    weights = exp_tc / exp_tc.sum()

    grad_a = np.zeros(n)
    len_conv = len(conv)
    k_top = max(30, int(len_conv * k_top_frac))
    top_idx = np.argpartition(weights, -k_top)[-k_top:]
    for k in top_idx:
        w = weights[k]
        if w < 1e-15:
            continue
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[js:je + 1] += w * 2.0 * a[k - idx]

    # Chain rule for softmax: dx_obj_i = (2/temp) * a_i * (grad_a_i - dot(grad_a, a))
    dx_obj = (2.0 / temp) * a * (grad_a - np.dot(grad_a, a))
    return obj, dx_obj


def lbfqsb_optimize(f0, time_limit, verbose=False):
    """L-BFGS-B with softmax parameterization and multi-temperature LSE."""
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    t0 = time.monotonic()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    # Warm start: x = log(f) (approximately, since softmax(x) ~ f when f is the target)
    x0 = np.log(np.maximum(f, 1e-30))
    x0 -= x0.mean()  # center it

    temperatures = [20.0, 50.0, 150.0, 500.0, 1500.0]
    k_top_fracs = [0.4, 0.3, 0.2, 0.15, 0.1]

    for idx_temp in range(len(temperatures)):
        if time.monotonic() - t0 > time_limit - 15:
            break

        temp = temperatures[idx_temp]
        kfrac = k_top_fracs[idx_temp]

        def func_wrapper(x):
            obj, grad = _softmax_lse_func_and_grad(x, n, temp, kfrac)
            return obj, grad

        # Use current best as warm start
        x_current = np.log(np.maximum(best_f, 1e-30))
        x_current -= x_current.mean()

        max_iter = 300
        result = minimize(func_wrapper, x_current, method='L-BFGS-B',
                          jac=True, options={'maxiter': max_iter, 'ftol': 1e-16,
                                             'gtol': 1e-12, 'maxls': 60})

        if result.success or result.fun < 2 * n * best_obj:
            x_opt = result.x
            a_new = np.exp(x_opt - x_opt.max())
            a_new = a_new / a_new.sum()
            obj_new = fast_convolve(a_new).max()

            if obj_new < best_obj:
                best_obj = obj_new
                best_f = a_new.copy()
                if verbose:
                    print(f"      L-BFGS-B temp={temp}: obj={best_obj:.10f}")

    return best_f, best_obj


def peak_targeted_descent(f0, time_limit, verbose=False):
    """Directly target worst peaks with corrected gradient."""
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = 0.02
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        k_best = int(np.argmax(conv))

        # Corrected gradient for worst peak + top-5
        js = max(0, k_best - n + 1)
        je = min(n - 1, k_best)
        idx = np.arange(js, je + 1)
        grad = 2.0 * f[k_best - idx].copy()
        full_grad = np.zeros(n)
        full_grad[js:je + 1] = grad

        # Add top-5 peaks with decreasing weight
        top5 = np.argsort(conv)[-5:]
        for rank, k2 in enumerate(top5):
            if rank == 0:
                continue
            weight = 0.3 / (rank ** 0.5)
            k2 = int(k2)
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            idx2 = np.arange(js2, je2 + 1)
            full_grad[js2:je2 + 1] += weight * 2.0 * f[k2 - idx2]

        f_new = f * np.exp(-lr * full_grad)
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()

        obj_new = fast_convolve(f_new).max()
        if obj_new < M * (1 - 1e-10):
            f = f_new
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"      PTD iter {it}: obj={best_obj:.10f}")
            lr = min(lr * 1.03, 0.08)
        else:
            lr *= 0.6
            no_improve += 1

        if no_improve > 80:
            seed = it * 41 + 19
            rng = np.random.default_rng(seed)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.003, 1e-15)
            f = f / f.sum()
            lr = 0.01
            no_improve = 0

        if lr < 1e-9:
            break

    return best_f, best_obj


def coordinate_descent_fine(f0, time_limit, verbose=False):
    """Fine coordinate descent with 1D golden section search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    pass_count = 0

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        improved = False
        seed = pass_count * 1000 + 7
        rng = np.random.default_rng(seed)
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj

            old_val = f[j]
            if old_val < 1e-14:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            # Try a range of new values
            for frac in [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 1.7, 2.0, 0.1]:
                new_val = old_val * frac
                if new_val < 0 or new_val > 0.5:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial = f_trial / ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

        if verbose and pass_count % 5 == 0:
            print(f"      CD pass {pass_count}: obj={best_obj:.10f}")

    return best_f, best_obj


def random_restart_search(best_f, time_limit, n_restarts=3):
    """Try random perturbations around the best solution."""
    n = len(best_f)
    t0 = time.monotonic()
    best_obj = fast_convolve(best_f).max()
    best = best_f.copy()

    for r in range(n_restarts):
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2:
            break
        t_alloc = rem / (n_restarts - r)

        rts = sms = r * 1000 + 3
        rng = np.random.default_rng(rts)
        # Different perturbation strategies
        if r == 0:
            noise = rng.standard_normal(n) * 0.003
            f_init = np.maximum(best + noise, 1e-15)
        elif r == 1:
            # Blend with reversed
            f_init = 0.8 * best + 0.2 * best[::-1]
        elif r == 2:
            # Larger perturbation
            noise = rng.standard_normal(n) * 0.008
            f_init = np.maximum(best + noise, 1e-15)
        else:
            # Localized perturbation
            f_init = best.copy()
            center = rng.integers(n // 4, 3 * n // 4)
            width = n // 10
            sl = slice(max(0, center - width), min(n, center + width))
            f_init[sl] *= np.exp(rng.standard_normal(width * 2) * 0.02)
            f_init = np.maximum(f_init, 1e-15)

        f_init = f_init / f_init.sum()
        f_opt, obj = smd_optimize(f_init, time_limit=t_alloc, tau_init=3.0,
                                  lr_init=0.008, restart_every=80)
        if obj < best_obj:
            best_obj = obj
            best = f_opt

    return best, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    Phase 1: Diverse asymmetric initializations
    Phase 2: SMD coarse optimization (corrected gradient)
    Phase 3: L-BFGS-B fine optimization (softmax param)
    Phase 4: Peak-targeted descent
    Phase 5: Multi-length exploration
    Phase 6: Random restarts
    Phase 7: Coordinate descent polish
    Phase 8: Final L-BFGS-B squeeze
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # === Phase 1: Diverse initializations ===
    print("=== Phase 1: Diverse Initializations ===")
    init_candidates = []
    test_lengths = [512, 640, 768, 896, 1024]

    for n_try in test_lengths:
        for v in range(13):
            a_init = make_asymmetric_init(n_try, variant=v, seed=seed)
            init_candidates.append((a_init, f"asym_{v}_n{n_try}"))
            try_seq(a_init, f"init_{v}_n{n_try}")

    if parent is not None:
        init_candidates.append((parent, "parent"))

    # Score by fast objective
    scored_inits = []
    for a_init, label in init_candidates:
        s = a_init.sum()
        if s < 1e-10:
            continue
        a_norm = a_init / s
        conv = fast_convolve(a_norm)
        obj = 2 * len(a_norm) * conv.max()
        scored_inits.append((obj, a_norm, label, len(a_norm)))

    scored_inits.sort(key=lambda x: x[0])
    top_inits = scored_inits[:8]
    print(f"  Top 5 inits: {[f'{l}:{o:.4f}(n={nn})' for o, _, l, nn in top_inits[:5]]}")

    # === Phase 2: SMD coarse optimization ===
    print("\n=== Phase 2: SMD Coarse Optimization ===")
    # Take top 4 with diverse lengths
    seen_lengths = set()
    selected = []
    for item in top_inits:
        if item[3] not in seen_lengths:
            selected.append(item)
            seen_lengths.add(item[3])
        if len(selected) >= 4:
            break
    # Fill if needed
    for item in top_inits:
        if item not in selected:
            selected.append(item)
        if len(selected) >= 4:
            break

    rem = deadline - time.monotonic() - 800
    for idx, (obj_init, a_init, label, n_len) in enumerate(selected):
        if rem < 20:
            break
        t_alloc = min(rem / max(1, 4 - idx), 250)
        if t_alloc < 10:
            t_alloc = 10
        print(f"  SMD on {label} (n={n_len}): t={t_alloc:.0f}s")
        a_opt, obj = smd_optimize(a_init, time_limit=t_alloc, verbose=True,
                                  tau_init=5.0, lr_init=0.01)
        try_seq(a_opt, f"smd_{label}")
        rem = deadline - time.monotonic() - 800

    # === Phase 3: L-BFGS-B fine optimization ===
    print("\n=== Phase 3: L-BFGS-B Fine Optimization ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 500
        if rem > 30:
            t_alloc = min(rem, 300)
            print(f"  L-BFGS-B on best (n={len(best_seq)}): t={t_alloc:.0f}s")
            a_opt, obj = lbfqsb_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "lbfqsb_main")

    # === Phase 4: Peak-targeted descent ===
    print("\n=== Phase 4: Peak-Targeted Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 350
        if rem > 20:
            t_alloc = min(rem, 150)
            print(f"  PTD on best (n={len(best_seq)}): t={t_alloc:.0f}s")
            a_opt, obj = peak_targeted_descent(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "ptd")

    # === Phase 5: Multi-length exploration ===
    print("\n=== Phase 5: Multi-Length Exploration ===")
    if best_seq is not None:
        base = best_seq
        target_lengths = [512, 640, 896, 1024, 1152, 1280]
        for target_n in target_lengths:
            rem = deadline - time.monotonic() - 250
            if rem < 30:
                break
            if len(base) == target_n:
                continue
            f_interp = interpolate_sequence(base, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")
            if sc < best_score + 0.005:
                t_alloc = min(rem * 0.3, 60)
                if t_alloc < 10:
                    t_alloc = 10
                print(f"  SMD n={target_n}: t={t_alloc:.0f}s")
                a_opt, obj = smd_optimize(f_interp, time_limit=t_alloc,
                                          tau_init=3.0, lr_init=0.008)
                try_seq(a_opt, f"smd_n{target_n}")
                if time.monotonic() > deadline - 200:
                    break

    # === Phase 6: Random restarts ===
    print("\n=== Phase 6: Random Restarts ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            print(f"  Restarts: t={rem:.0f}s")
            a_opt, obj = random_restart_search(best_seq, time_limit=min(rem, 120),
                                               n_restarts=3)
            try_seq(a_opt, "restarts")

    # === Phase 7: Coordinate descent polish ===
    print("\n=== Phase 7: Coordinate Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 15:
            t_alloc = min(rem, 80)
            print(f"  CD: t={t_alloc:.0f}s")
            a_opt, obj = coordinate_descent_fine(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "cd")

    # === Phase 8: Final L-BFGS-B + SMD squeeze ===
    print("\n=== Phase 8: Final Squeeze ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 20:
            t_alloc1 = min(rem * 0.5, 60)
            t_alloc2 = rem - t_alloc1 - 5
            if t_alloc1 > 10:
                print(f"  Final L-BFGS-B: t={t_alloc1:.0f}s")
                a_opt, obj = lbfqsb_optimize(best_seq, time_limit=t_alloc1, verbose=True)
                try_seq(a_opt, "final_lbfqsb")
            if t_alloc2 > 10:
                print(f"  Final SMD: t={t_alloc2:.0f}s")
                a_opt, obj = smd_optimize(best_seq, time_limit=t_alloc2,
                                          tau_init=50.0, lr_init=0.003,
                                          restart_every=50, top_frac=0.1)
                try_seq(a_opt, "final_smd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
