def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using improved LP-Frank-Wolfe with
direct min-max LP formulation, asymmetric initialization, enhanced line search,
multi-length exploration, and coordinate descent polish.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_normalized(f):
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()


def solve_lp_direct(f, n):
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            A_ub[k, i] = f[k - i]
        A_ub[k, n] = -1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [1.0]
    b_ub = np.zeros(2 * n - 1)
    bounds = [(0, None)] * n + [(-np.inf, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[n]
    return None, None

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-Frank-Wolfe with direct
min-max LP formulation. Key insight: asymmetric sequences can achieve scores
below 2 by making conv[n-1] = sum a[i]*a[n-1-i] < 1/n. The LP directly
minimizes max_k (f*g)[k] over the simplex, providing a globally optimal
direction for each FW step. Includes multi-length exploration, adaptive
restarts, and coordinate descent polish.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_convolve(f, g):
    n = len(f)
    m = len(g)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    G = np.fft.rfft(g, fft_size)
    return np.fft.irfft(F * G, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_normalized(f):
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()


def build_conv_matrix(f, n):
    T = np.zeros((2 * n - 1, n))
    for i in range(n):
        T[i:i + n, i] = f
    return T
