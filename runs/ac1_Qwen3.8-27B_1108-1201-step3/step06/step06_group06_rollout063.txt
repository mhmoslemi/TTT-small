import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fast_convolve(a):

    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]

    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using KKT-based active-set optimization:
- Solves the KKT linear system for the weighted max of active quadratic forms

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_obj(a):
    """Return (objective_value, convolution) for normalized a."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv

    Q = sum_{i,j} m[i,j] a[i] a[j]
    
    Where m[i,j] = (1/|K*|) * sum_{k in K*} 1[i+j=k]
    
    Since i+j = j+i, m[i,j] = m[j,i], so it IS symmetric.
    
    The gradient: dQ/da[p] = sum_j m[p,j] a[j] + sum_i m[i,p] a[i] = 2 * sum_j m[p,j] a[j]
    (using symmetry)
    
    Actually: dQ/da[p] = sum_j m[p,j] a[j] + sum_i a[i] m[i,p] = 2 * (M a)[p]
    where M = m (the matrix).
    
    So the gradient is 2*Ma.
    The KKT conditions for min Q s.t. 1^T a = 1, a >= 0:
    2Ma + λ*1 - μ = 0
    For active set (support) S: 2M[S,S] a_S + λ*1 = 0
    1^T a_S = 1
    """
    m = len(support)
    if m == 0:
        return None
    if m == 1:
        i = support[0]
        # Q = m[i,i] * a[i]^2 = m[i,i] (since a[i] = 1)
        # This is already determined: a[i] = 1
        return np.array([1.0]), 0.0
    
    # Build the system: [2*M_SS  1; 1^T  0] [a_S; λ] = [0; 1]
    # M_eff[i,j] = (1/|K*|) * (number of k in K* with support[i]+support[j] = k)
    
    active_set = set(int(k) for k in active_set)
    k_star = len(active_set)
    
    # Build sparse representation
    # For the (m+1) x (m+1) system:
    # Row i (0-indexed, i < m): 2 * sum_j M[i,j] * a[j] + λ = 0
    # Row m: sum_i a[i] = 1
    
    # System matrix
    A_sys = np.zeros((m + 1, m + 1))
    b_sys = np.zeros(m + 1)
    b_sys[m] = 1.0
    
    # Fill in the quadratic form matrix
    # M[i,j] = (1/k_star) * count of k in K* with support[i] + support[j] = k
    sup = np.array(support)
    sums = sup[:, None] + sup[None, :]  # m x m matrix of sums
    # For each (i,j) pair, count how many k in K* equal support[i]+support[j]
    # Since K* is a set, there's at most 1 k for each sum value
    M_ss = np.zeros((m, m))
    for i in range(m):
        for j in range(m):
            s_val = sup[i] + sup[j]
            if s_val in active_set:
                M_ss[i, j] = 1.0 / k_star

    A_sys[:m, m] = 1.0  # λ coefficient
    A_sys[m, :m] = 1.0  # sum constraint
    A_sys[m, m] = 0.0
    
    try:
        sol = np.linalg.solve(A_sys, b_sys)
    except np.linalg.LinAlgError:
        try:
            sol = np.linalg.lstsq(A_sys, b_sys, rcond=None)[0]
        except:
            return None
    
    a_S = sol[:m]
    lam = sol[m]
    
    # Check feasibility
    if np.min(a_S) < -1e-10:
        return None
    
    return a_S, lam

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using KKT active-set optimization:
- Exploits the quadratic structure to solve exact optimality conditions
- Iteratively levels active peaks via sparse linear systems
- Combines with subgradient Frank-Wolfe, multi-length exploration, and perturbation restarts
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Return (2n*max(conv), conv) for normalized a."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv


def solve_kkt_system(support, active_set, weights=None):
    """
    Solve KKT: [2*N  -1][a_S] = [0]
               [1^T   0][lam]   [1]
    where N[i,j] = sum_k w_k * 1[support[i]+support[j] == k]
    """
    m = len(support)
    if m < 1:
        return None
    if m == 1:
        return np.array([1.0]), 0.0

    active_set = list(active_set)
    k_star = len(active_set)
    if k_star == 0:
        return None

    if weights is None:
        weights_np = np.ones(k_star) / k_star
    else:
        weights_np = np.asarray(weights, dtype=np.float64)
        weights_np = np.maximum(weights_np, 0)
        wsum = weights_np.sum()
        if wsum < 1e-15:
            weights_np = np.ones(k_star) / k_star
        else:
            weights_np /= wsum

    active_dict = {}
    for idx, k in enumerate(active_set):
        active_dict[int(k)] = weights_np[idx]

    sup = np.array(support, dtype=np.int64)
    sums_mat = sup[:, None] + sup[None, :]

    N_mat = np.zeros((m, m), dtype=np.float64)
    for i in range(m):
        for j in range(m):
            s_val = sums_mat[i, j]
            if s_val in active_dict:
                N_mat[i, j] = active_dict[s_val]

    A_sys = np.zeros((m + 1, m + 1), dtype=np.float64)
    b_sys = np.zeros(m + 1, dtype=np.float64)
    b_sys[m] = 1.0

    A_sys[:m, :m] = 2.0 * N_mat
    A_sys[:m, m] = -1.0
    A_sys[m, :m] = 1.0
    A_sys[m, m] = 0.0

    try:
        sol = np.linalg.solve(A_sys, b_sys)
    except np.linalg.LinAlgError:
        try:
            sol = np.linalg.lstsq(A_sys, b_sys, rcond=None)[0]
        except Exception:
            return None

    a_S = sol[:m]
    lam = sol[m]

    if np.any(a_S < -1e-8):
        return None

    a_S = np.maximum(a_S, 0.0)
    ss = a_S.sum()
    if ss < 1e-15:
        return None
    a_S /= ss
    return a_S, lam


def kkt_leveling_optimize(a0, time_limit, verbose=False, max_active=12, max_support=300):
    """
    KKT-based active-set leveling for min max_k conv(a,a)[k].
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    n = len(a)
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_val, _ = compute_obj(a)
    t0 = time.monotonic()
    outer_it = 0
    stall = 0

    while time.monotonic() - t0 < time_limit and outer_it < 10000:
        outer_it += 1
        conv = fast_convolve(a)
        M = conv.max()

        # Active set: peaks within 2% of max
        tol = 0.02 * M
        active_mask = conv >= (M - tol)
        active = list(np.where(active_mask)[0])
        if not active:
            active = [int(np.argmax(conv))]

        # Cap active set size
        if len(active) > max_active:
            active_idx = np.argsort(conv)[-max_active:]
            active = list(active_idx)

        # Support: top elements
        if n > max_support:
            top_idx = np.argsort(a)[-max_support:]
            support = list(top_idx)
        else:
            support = list(np.where(a > 1e-10)[0])

        if len(support) < 2:
            break

        # Inner leveling loop
        a_cand = a.copy()
        converged = False
        for inner in range(30):
            a_S, lam = solve_kkt_system(support, active)
            if a_S is None:
                # Remove smallest from support
                min_idx = min(support, key=lambda i: a[i])
                support.remove(min_idx)
                if len(support) < 2:
                    break
                continue

            a_cand = np.zeros(n)
            for i, s_idx in enumerate(support):
                a_cand[s_idx] = a_S[i]
            a_cand = np.maximum(a_cand, 0)
            ss = a_cand.sum()
            if ss < 1e-15:
                break
            a_cand /= ss

            # Check leveling
            conv_cand = fast_convolve(a_cand)
            M_cand = conv_cand.max()
            # Find actual peaks of candidate
            tol_c = 0.005 * M_cand
            cand_peaks = list(np.where(conv_cand >= (M_cand - tol_c))[0])
            if len(cand_peaks) > max_active:
                cand_peaks = list(np.argsort(conv_cand)[-max_active:])

            if set(cand_peaks) == set(int(x) for x in active):
                converged = True
                break
            else:
                active = cand_peaks
                # Update support if elements became zero
                new_support = [i for i in support if a_cand[i] > 1e-10]
                if len(new_support) < 2:
                    new_support = list(np.where(a_cand > 1e-10)[0])
                if len(new_support) >= 2:
                    support = new_support

        # Evaluate candidate
        val_cand, _ = compute_obj(a_cand)

        if val_cand < best_val - 1e-13:
            best_val = val_cand
            best_a = a_cand.copy()
            a = a_cand
            stall = 0
            if verbose and outer_it % 10 == 0:
                print(f"    KKT iter {outer_it}: obj={best_val:.8f}, |active|={len(active)}")
        else:
            stall += 1

        # Perturbation to escape local optima
        if not converged or stall > 5:
            rng = np.random.default_rng(outer_it * 1000033 + 7)
            noise = 0.001 + 0.003 * min(stall / 20.0, 1.0)
            a = project_simplex(a_cand + rng.standard_normal(n) * noise)
            stall = max(0, stall - 2)

    return best_a, best_val


def subgrad_fw(a0, time_limit, verbose=False, top_k=5, max_iter=200000):
    """Subgradient Frank-Wolfe with exact line search on the true objective."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    n = len(a)
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_val, _ = compute_obj(a)
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit and it < max_iter:
        it += 1
        _, conv = compute_obj(a)
        M = conv.max()

        # Active set
        active = np.argsort(conv)[-top_k:]

        # Subgradient computation
        g = np.zeros(n)
        for k in active:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                g[j] += 2.0 * a[k - j]
        g /= len(active)

        # FW direction: move mass to argmin(g)
        j_min = int(np.argmin(g))

        # Line search
        best_t = 0.0
        best_f = M + 1e-15

        for t in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.8, 1.0]:
            trial = (1.0 - t) * a
            trial[j_min] += t
            trial = np.maximum(trial, 0.0)
            st = trial.sum()
            if st < 1e-15:
                continue
            trial /= st
            val, _ = compute_obj(trial)
            if val < best_f:
                best_f = val
                best_t = t

        # Fine search
        if best_t > 0.005:
            lo = max(0.0, best_t - 0.04)
            hi = min(1.0, best_t + 0.04)
            for _ in range(12):
                mid = (lo + hi) / 2.0
                trial = (1.0 - mid) * a
                trial[j_min] += mid
                trial = np.maximum(trial, 0.0)
                st = trial.sum()
                if st < 1e-15:
                    break
                trial /= st
                val, _ = compute_obj(trial)
                if val < best_f:
                    best_f = val
                    best_t = mid
                    lo = mid
                else:
                    hi = mid

        if best_t > 1e-8 and best_f < M - 1e-13:
            a = (1.0 - best_t) * a
            a[j_min] += best_t
            a = np.maximum(a, 0.0)
            a /= a.sum()
            if best_f < best_val - 1e-13:
                best_val = best_f
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 200:
            rng = np.random.default_rng(it * 7919 + 13)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            no_improve = 0
            top_k = min(top_k + 2, 20)

        if verbose and it % 5000 == 0:
            print(f"    FW iter {it}: obj={best_val:.8f}, no_improve={no_improve}")

    return best_a, best_val


def coordinate_descent(a0, time_limit, verbose=False):
    """
    Targeted coordinate descent: for each peak, try reducing the
    largest-contributing coefficient and redistributing to low-impact positions.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    n = len(a)
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_val, _ = compute_obj(a)
    t0 = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - t0 < time_limit and it < 50000:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        peak_idx = int(np.argmax(conv))

        # Find the pair contributing most to the peak
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        products = np.array([a[j] * a[peak_idx - j] for j in range(js, je + 1)])
        j_best = js + int(np.argmax(products))
        j_other = peak_idx - j_best

        # Try reducing a[j_best] and redistributing
        improved = False
        for factor in [0.7, 0.8, 0.9, 0.5, 0.95]:
            trial = a.copy()
            old_val = trial[j_best]
            new_val = old_val * factor
            diff = old_val - new_val
            trial[j_best] = new_val

            # Redistribute to positions that minimize impact on peaks
            # Score each position by how much adding mass would increase the max
            if diff > 1e-15:
                # Use the convolution to find "safe" positions
                # A position p is "safe" if (f * delta_p)[k] is small for all k near peak
                # Approximate: positions far from peak - p
                # Simple heuristic: redistribute to positions with smallest a
                bottom_quarter = np.argsort(a)[:n // 4]
                targets = [p for p in bottom_quarter if p != j_best and p != j_other]
                if targets:
                    t_arr = np.array(targets)
                    t_vals = a[t_arr]
                    if t_vals.sum() > 1e-15:
                        weights = t_vals / t_vals.sum()
                    else:
                        weights = np.ones(len(targets)) / len(targets)
                    trial[t_arr] += diff * weights

                    trial = np.maximum(trial, 0.0)
                    trial /= trial.sum()
                    val, _ = compute_obj(trial)
                    if val < best_val - 1e-13:
                        best_val = val
                        best_a = trial.copy()
                        a = trial
                        improved = True
                        break

        if not improved:
            stall += 1
            if stall > 30:
                # Try a random pair swap
                rng = np.random.default_rng(it * 31 + 3)
                i1 = rng.integers(0, n)
                i2 = rng.integers(0, n)
                if i1 != i2:
                    amt = a[i1] * 0.1
                    trial = a.copy()
                    trial[i1] -= amt
                    trial[i2] += amt
                    trial = np.maximum(trial, 0.0)
                    trial /= trial.sum()
                    val, _ = compute_obj(trial)
                    if val < best_val - 1e-13:
                        best_val = val
                        best_a = trial.copy()
                        a = trial
                        stall = 0

    return best_a, best_val


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_diverse_init(n, seed=42):
    """Create diverse initial sequences."""
    rng = np.random.default_rng(seed)
    inits = []

    # Gaussian
    x = np.arange(n) - n * 0.4
    sigma = n * 0.25
    g = np.exp(-0.5 * (x / sigma) ** 2)
    g /= g.sum()
    inits.append(g)

    # Uniform
    inits.append(np.ones(n) / n)

    # Exponential decay
    e = np.exp(-np.arange(n) / (n * 0.3))
    e /= e.sum()
    inits.append(e)

    # Bimodal
    x = np.arange(n)
    b = np.exp(-0.5 * ((x - n * 0.25) / (n * 0.1)) ** 2) + np.exp(-0.5 * ((x - n * 0.75) / (n * 0.1)) ** 2)
    b /= b.sum()
    inits.append(b)

    # Random
    r = rng.dirichlet(np.ones(n) * 0.5)
    inits.append(r)

    # Flat-top
    half = n * 0.3
    ft = np.ones(n)
    ft[:n//2] = 1.0 - 0.3 * (np.arange(n//2) / half) ** 2
    ft[n//2:] = 1.0 - 0.3 * (np.arange(n - n//2, 0, -1) / half) ** 2
    ft = np.maximum(ft, 0)
    ft /= ft.sum()
    inits.append(ft)

    return inits
