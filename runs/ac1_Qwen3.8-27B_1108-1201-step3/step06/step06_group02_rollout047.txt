"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def objective_normalized(a):
    """Compute 2*n*max(conv)/sum^2 for normalized a (sum=1)."""
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max()


def compute_subgradient(a, conv, top_k, temperature, focal_beta):
    """Compute weighted subgradient of max(conv) with focal peak weighting."""
    n = len(a)
    conv_size = len(conv)
    k_select = min(top_k, conv_size)
    peak_indices = np.argpartition(conv, -k_select)[-k_select:]
    peak_vals = conv[peak_indices]
    
    # Focal weighting: w_k = (conv[k]/max_conv)^focal_beta * softmax
    max_pv = peak_vals.max()
    if max_pv < 1e-30:
        return np.zeros(n)
    focal_w = (peak_vals / max_pv) ** focal_beta
    scaled = focal_w / temperature
    scaled -= scaled.max()
    weights = np.exp(scaled)
    ws = weights.sum()
    if ws < 1e-30:
        weights = np.zeros(len(peak_indices))
        weights[np.argmax(peak_vals)] = 1.0
    else:
        weights /= ws
    
    sg = np.zeros(n)
    for i in range(len(peak_indices)):
        w = weights[i]
        if w < 1e-15:
            continue
        k = int(peak_indices[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        partners = k - idx
        sg[idx] += 2.0 * w * a[partners]
    return sg


def adam_multiplicative_descent(f0, time_limit, lr_init=0.003, beta1=0.9, beta2=0.999,
                                eps=1e-8, top_k=30, temp_init=0.005,
                                temp_decay=1.002, focal_beta=2.0, restart_every=150,
                                no_improve_limit=300):
    """Adam-enhanced multiplicative subgradient descent."""
    f = np.maximum(f0.astype(np.float64), 1e-10)
    f = f / f.sum()
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    temperature = temp_init
    no_improve = 0
    t0 = time.monotonic()
    lr = lr_init
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1.0:
            break
        
        conv = fast_convolve(f)
        M = conv.max()
        
        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > no_improve_limit:
            break
        
        # Restart with perturbation if stuck
        if no_improve > restart_every:
            rng = np.random.default_rng(t * 7919 + 13)
            amp = 0.002 * np.sqrt(no_improve / restart_every)
            f = np.maximum(best_f + rng.standard_normal(n) * amp, 1e-10)
            f = f / f.sum()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            no_improve = 0
            lr = lr_init * 0.5
            temperature = temp_init * 0.5
            continue
        
        # Compute subgradient
        sg = compute_subgradient(f, conv, top_k, temperature, focal_beta)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Multiplicative update in log domain
        update = lr * m_hat / (np.sqrt(v_hat) + eps)
        f = f * np.exp(-np.clip(update, -5, 5))
        f = np.maximum(f, 1e-12)
        f = f / f.sum()
        
        # Schedule
        temperature *= temp_decay
        lr *= 0.9998
        
    return best_f, best_obj


def spectral_equalization(f, time_limit, n_steps=200, alpha_init=0.1):
    """Periodic spectral equalization to flatten the power spectrum."""
    f = np.maximum(f.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    alpha = alpha_init
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 50:
            break
        
        # Compute spectrum
        A = np.fft.rfft(f, fft_size)
        P = np.abs(A) ** 2
        phase = np.angle(A)
        
        # Compute target: compress dynamic range
        P_mean = P.mean()
        P_max = P.max()
        P_min = P.min()
        
        # Compress: P_new = P^(1/(1+alpha))
        # This reduces peaks and raises valleys
        P_compressed = P ** (1.0 / (1.0 + alpha))
        
        # Reconstruct
        A_new = np.sqrt(P_compressed) * np.exp(1j * phase)
        f_new = np.fft.irfft(A_new, fft_size)[:n]
        
        # Project to non-negative
        f_new = np.maximum(f_new, 0.0)
        sn = f_new.sum()
        if sn < 1e-15:
            no_improve += 1
            alpha *= 0.9
            continue
        f_new = f_new / sn
        
        # Evaluate
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            alpha = max(0.01, alpha * 0.995)
        else:
            no_improve += 1
            # Try with original f blended
            f_blend = 0.7 * f + 0.3 * f_new
            f_blend = np.maximum(f_blend, 0)
            f_blend = f_blend / f_blend.sum()
            conv_blend = fast_convolve(f_blend).max()
            if conv_blend < best_obj - 1e-14:

                best_f = f_blend.copy()
                f = f_blend

                    alpha = 0.1

                    rng = np.random.default_rng(step * 31 + 7)
                    f = np.maximum(best_f + rng.standard_normal(n) * 0.001, 1e-10)
                    f = f / f.sum()
    
    return best_f, best_obj


def peak_coordinate_descent(f0, time_limit):
    """Coordinate descent targeting the worst peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        if no_improve > 80:
            break
        
        conv = fast_convolve(f)
        M = conv.max()
        peak_k = int(np.argmax(conv))
        
        # Find contributing indices
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        contrib_idx = np.arange(js, je + 1)
        if len(contrib_idx) == 0:
            break
        
        products = f[contrib_idx] * f[peak_k - contrib_idx]
        
        # Try reducing the most contributing coordinate
        max_prod_pos = np.argmax(products)
        source = contrib_idx[max_prod_pos]
        
        # Try multiple reduction fractions
        found = False
        for frac in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            diff = f[source] * frac
            if diff < 1e-15:
                continue
            trial = f.copy()
            trial[source] -= diff
            # Distribute to non-contributing coordinates
            non_contrib = np.ones(n, dtype=bool)
            non_contrib[contrib_idx] = False
            if non_contrib.sum() < 1:
                non_contrib[:] = True
                non_contrib[source] = False
            nc_vals = trial[non_contrib]
            nc_sum = nc_vals.sum()
            if nc_sum < 1e-15:
                continue
            trial[non_contrib] += diff * nc_vals / nc_sum
            trial = np.maximum(trial, 0.0)
            trial = trial / trial.sum()
            obj = fast_convolve(trial).max()
            if obj < best_obj - 1e-14:
                f = trial
                best_obj = obj
                best_f = f.copy()
                found = True
                break
        
        if not found:
            # Try swap: move mass from source to a specific target
            non_contrib = np.ones(n, dtype=bool)
            non_contrib[contrib_idx] = False
            if non_contrib.sum() >= 1:
                # Pick the coordinate with lowest current contribution to any top peak
                top_peaks = np.argsort(conv)[-3:]
                contrib_all = np.zeros(n)
                for pk in top_peaks:
                    js2 = max(0, pk - n + 1)
                    je2 = min(n - 1, pk)
                    idx2 = np.arange(js2, je2 + 1)
                    contrib_all[idx2] += f[pk - idx2]
                targets = np.where(non_contrib)[0]
                if len(targets) > 0:
                    target = targets[np.argmin(contrib_all[targets])]
                    for frac in [0.1, 0.2, 0.3]:
                        diff = f[source] * frac
                        trial = f.copy()
                        trial[source] -= diff
                        trial[target] += diff
                        trial = np.maximum(trial, 0)
                        trial = trial / trial.sum()
                        obj = fast_convolve(trial).max()
                        if obj < best_obj - 1e-14:
                            f = trial
                            best_obj = obj
                            best_f = f.copy()
                            found = True
                            break
            if not found:
                no_improve += 1
        else:
            no_improve = 0
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian-like initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    """Bell-shaped initial sequence with flat top."""
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_cosine_mod(n, freq, amp):
    """Cosine-modulated uniform sequence."""
    x = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_multi_cosine(n, n_freqs=3, max_amp=0.3, seed=0):
    """Multi-frequency cosine-modulated sequence."""
    rng = np.random.default_rng(seed)
    a = np.ones(n)
    for m in range(1, n_freqs + 1):
        amp = rng.uniform(0.05, max_amp)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * m * np.arange(n) / n + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_legendre_based(n, p=None):
    """Create sequence inspired by Legendre quadratic residues."""
    if p is None:
        # Find a prime near n
        p = n
        while True:
            if is_prime(p):
                break
            p += 1
    # Legendre symbol sequence
    # a[j] = 1 if j^2 is a quadratic residue mod p (and j != 0), 0 otherwise
    # Simpler: use the quadratic character to weight positions
    a = np.zeros(n)
    for j in range(1, p):
        # Check if j is a quadratic residue mod p
        if pow(j, (p - 1) // 2, p) == 1:
            # Map to position in [0, n)
            pos = j % n
            a[pos] += 1.0 / p
    # Also set a[0] based on convention
    a[0] = 2.0 / p  # Slightly boost the center
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def is_prime(n):
    """Simple primality test."""
    if n < 2:
        return False
    if n < 4:
        return True
    if n % 2 == 0 or n % 3 == 0:
        return False
    i = 5
    while i * i <= n:
        if n % i == 0 or n % (i + 2) == 0:
            return False
        i += 6
    return True

    scored_inits.sort(key=lambda x: x[0])
    print(f"  Best init: obj={scored_inits[0][0]:.6f} [{scored_inits[0][2]}] n={len(scored_inits[0][1])}")
    print(f"  Phase 1 time: {time.monotonic()-t0:.1f}s")
    
    # === Phase 2: Adam Multiplicative Descent on top initializations ===
    print("\n=== Phase 2: Adam Multiplicative Descent ===")
    
    # Try top 4 initializations
    for trial_idx in range(min(4, len(scored_inits))):
        rem = deadline - time.monotonic() - 200
        if rem < 60:
            break
        sc_init, a_init, label = scored_inits[trial_idx]
        t_alloc = min(rem * 0.5, 300)
        print(f"  AdamMD [{label}] n={len(a_init)}: alloc={t_alloc:.0f}s (init_obj={sc_init:.6f})")
        a_opt, obj = adam_multiplicative_descent(
            a_init, time_limit=t_alloc, lr_init=0.003, beta1=0.9, beta2=0.999,
            top_k=30, temp_init=0.005, temp_decay=1.002, focal_beta=2.0,
            restart_every=120, no_improve_limit=250
        )
        try_seq(a_opt, f"adammd_{label}")
    
    # === Phase 3: Mixed Descent (MD + Spectral) ===
    print("\n=== Phase 3: Mixed Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 60:
            t_alloc = min(rem * 0.6, 300)
            print(f"  Mixed: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, obj = mixed_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "mixed_main")
    
    # Also try mixed descent from a different initialization
    if len(scored_inits) > 1:
        rem = deadline - time.monotonic() - 150
        if rem > 60:
            sc2, a2, label2 = scored_inits[1]
            t_alloc = min(rem * 0.5, 200)
            print(f"  Mixed [{label2}]: n={len(a2)}, alloc={t_alloc:.0f}s")
            a_opt, obj = mixed_descent(a2, time_limit=t_alloc)
            try_seq(a_opt, f"mixed_{label2}")
    
    # === Phase 4: Peak Coordinate Descent ===
    print("\n=== Phase 4: Peak Coordinate Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_alloc = min(rem * 0.5, 120)
            print(f"  PCD: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, obj = peak_coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "pcd_main")
    
    # === Phase 5: Spectral Equalization Standalone ===
    print("\n=== Phase 5: Spectral Equalization ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            t_alloc = min(rem * 0.5, 100)
            print(f"  SpecEq: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, obj = spectral_equalization(best_seq, time_limit=t_alloc, n_steps=200, alpha_init=0.1)
            try_seq(a_opt, "speceq_main")
    
    # === Phase 6: Multi-length exploration ===
    print("\n=== Phase 6: Multi-Length Exploration ===")
    if best_seq is not None:
        base_n = len(best_seq)
        target_lengths = [512, 640, 768, 896, 1024, 1152, 1280]
        for target_n in target_lengths:
            if target_n == base_n:
                continue
            rem = deadline - time.monotonic() - 80
            if rem < 15:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")
            
            if sc < best_score + 0.02:
                t_alloc = min(rem * 0.3, 80)
                if t_alloc > 15:
                    print(f"  AdamMD n={target_n}: alloc={t_alloc:.0f}s")
                    a_opt, obj = adam_multiplicative_descent(
                        f_interp, time_limit=t_alloc, lr_init=0.002,
                        top_k=25, temp_init=0.004, focal_beta=2.0,
                        restart_every=100, no_improve_limit=200
                    )
                    try_seq(a_opt, f"adammd_interp_n{target_n}")
    
    # === Phase 7: Refined fine-tuning with very small steps ===
    print("\n=== Phase 7: Fine Tuning ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t_alloc = min(rem * 0.6, 60)
            print(f"  Fine AdamMD: alloc={t_alloc:.0f}s")
            a_opt, obj = adam_multiplicative_descent(
                best_seq, time_limit=t_alloc, lr_init=0.001, beta1=0.95, beta2=0.9995,
                top_k=15, temp_init=0.002, temp_decay=1.001, focal_beta=3.0,
                restart_every=80, no_improve_limit=150
            )
            try_seq(a_opt, "fine_adam")
        
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            t_alloc = min(rem * 0.7, 30)
            print(f"  Fine PCD: alloc={t_alloc:.0f}s")
            a_opt, obj = peak_coordinate_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "fine_pcd")
    
    # === Phase 8: Try different focal beta and top_k ===
    print("\n=== Phase 8: Hyperparameter Sweep ===")
    if best_seq is not None:
        for focal_b in [1.5, 4.0, 6.0]:
            rem = deadline - time.monotonic() - 20
            if rem < 15:
                break
            t_alloc = min(rem * 0.4, 40)
            if t_alloc < 10:
                break
            print(f"  AdamMD foiluncal={focal_b}: alloc={t_alloc:.0f}s")
            a_opt, obj = adam_multiplicative_descent(
                best_seq, time_limit=t_alloc, lr_init=0.0015,
                top_k=20, temp_init=0.003, temp_decay=1.002,
                focal_beta=focal_b, restart_every=80, no_improve_limit=120
            )
            try_seq(a_opt, f"sweep_fb{focal_b}")
    
    # === Phase 9: Final squeeze ===
    print("\n=== Phase 9: Final Squeeze ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 8:
            t_alloc = min(rem, 10)
            print(f"  Final polish: alloc={t_alloc:.0f}s")
            a_opt, obj = adam_multiplicative_descent(
                best_seq, time_limit=t_alloc, lr_init=0.0005,
                top_k=10, temp_init=0.001, temp_decay=1.001, focal_beta=4.0,
                restart_every=50, no_improve_limit=80
            )
            try_seq(a_opt, "final")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
