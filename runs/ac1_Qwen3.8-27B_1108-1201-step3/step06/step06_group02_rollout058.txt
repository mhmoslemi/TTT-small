```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using mirror descent, 
pairwise coordinate moves, and spectral flattening.
"""

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def objective(a):
    """Compute the objective for normalized a (sum=1)."""
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max()

def subgradient_peak(a, k):
    """Compute subgradient of r[k] with respect to a."""
    n = len(a)
    sg = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        sg[j] = 2.0 * a[k - j]
    return sg
```

def soft_subgradient(a, conv, temperature, n_peaks=10):
    n = len(a)
    # Get top peaks
    k_actual = min(n_peaks, len(conv))
    peak_indices = np.argpartition(conv, -k_actual)[-k_actual:]
    peak_vals = conv[peak_indices]
    
    # Soft weights
    scaled = peak_vals / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    # Combine subgradients
    sg = np.zeros(n)
    for i in range(len(peak_indices)):
        if w[i] < 1e-12:
            continue
        k = int(peak_indices[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += w[i] * 2.0 * a[k - idx]
    return sg

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-strategy approach:
- Mirror descent with soft subgradients (main workhorse)
- Pairwise coordinate descent for peak reduction
- Spectral domain flattening
- Multi-start with diverse initializations and adaptive length exploration

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def objective_normalized(a):
    """Compute 2*n*max(conv)/sum^2 for normalized a (sum=1)."""
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max()


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def soft_subgradient(a, conv, temperature, n_peaks=20):
    """Compute soft subgradient of max(r_k) weighted over top peaks."""
    n = len(a)
    k_actual = min(n_peaks, len(conv))
    peak_indices = np.argpartition(conv, -(k_actual - 1))[-(k_actual - 1):] if k_actual > 1 else np.array([np.argmax(conv)])
    peak_vals = conv[peak_indices]
    scaled = peak_vals / max(temperature, 1e-15)
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        sg = np.zeros(n)
        k = int(np.argmax(conv))
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] = 2.0 * a[k - idx]
        return sg
    w = w / ws
    sg = np.zeros(n)
    for i in range(len(peak_indices)):
        if w[i] < 1e-12:
            continue
        k = int(peak_indices[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += w[i] * 2.0 * a[k - idx]
    return sg

    a = np.maximum(a0.astype(np.float64), 0.0)

    best_a = a.copy()

    temperature = temp_init_frac * conv.max()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:

            break

        sg = soft_subgradient(a, conv, temperature, n_peaks=20)

        log_update -= log_update.max()  # numerical stability
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-18)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:

            if verbose and it % 100 == 0:
                print(f"    MD iter {it}: obj={best_obj:.10f}, lr={lr:.6f}")
        else:

            log_half = -0.5 * lr * sg
            log_half -= log_half.max()
            a_half = a * np.exp(log_half)
            a_half = np.maximum(a_half, 1e-18)
            a_half = a_half / a_half.sum()
            M_half = fast_convolve(a_half).max()

def spectral_flatten(a0, time_limit, verbose=False, alpha=0.5):
    """Spectral domain flattening: compress dynamic range of |FFT|^2."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_obj = objective_normalized(a)
    best_a = a.copy()
    t0 = time.monotonic()
    cur_alpha = alpha
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > 50:
            break

        it += 1
        N = 1 << (2 * n).bit_length()
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)

        # Flatten: compress dynamic range
        mag_min = mag[mag > 1e-15].min() if (mag > 1e-15).any() else 1e-15
        mag_max = mag.max()
        if mag_max < 1e-15:
            break
        # Normalize and apply power compression
        mag_norm = mag / mag_max
        mag_flat = mag_norm ** cur_alpha * mag_max

        # Reconstruct
        A_new = mag_flat * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            no_improve += 1
            break
        a_new = a_new / s_new

        obj_new = objective_normalized(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            if verbose and it % 10 == 0:
                print(f"    Spectral iter {it}: obj={best_obj:.10f}, alpha={cur_alpha:.3f}")
        elif obj_new < objective_normalized(a) - 1e-12:
            a = a_new
            no_improve += 1
        else:
            no_improve += 1

        # Adapt alpha
        cur_alpha = max(0.1, min(0.9, cur_alpha * (0.97 if no_improve > 10 else 1.02)))

    return best_a, best_obj


def multi_peak_descent(a0, time_limit, verbose=False):

    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()

    while time.monotonic() - t0 < time_limit:
        if no_improve > 150:
            break

        peak_indices = np.argpartition(conv, -(top_k - 1))[-(top_k - 1):] if top_k > 1 else np.array([np.argmax(conv)])
        peak_vals = conv[peak_indices]
        temp = 0.01 * M
        w = peak_vals / temp
        w -= w.max()
        w = np.exp(w)
        w /= w.sum()

        sg = np.zeros(n)

            sg[idx] += w[i] * 2.0 * a[k - idx]

            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            if verbose and it % 50 == 0:

            a_half = np.maximum(a_half, 1e-18)
            a_half = a_half / a_half.sum()

def make_gaussian(n, sigma_frac=0.3):

    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    a /= a.sum()
    return a


def make_sine(n):
    """Sine-squared shape (zero at edges)."""
    x = np.arange(n)
    a = np.sin(np.pi * (x + 0.5) / n) ** 2
    a = np.maximum(a, 1e-10)
    a /= a.sum()

def make_raised_cosine(n, beta=0.3):
    """Raised cosine window."""
    x = np.arange(n) - n / 2.0 + 0.5
    t = 2.0 * np.abs(x) / n
    a = np.ones(n)
    mask = t > (1 - beta)
    a[mask] = 0.5 * (1 + np.cos(np.pi * (t[mask] - (1 - beta)) / beta))
    a = np.maximum(a, 0)
    a /= a.sum()
    return a


def make_cosine_mod(n, freq, amp):

    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n)

def make_multi_cosine(n, n_freqs=4, max_amp=0.25, seed=0):

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

            parent = p / p.sum()
            p_score = try_seq(parent, "parent")
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = make_gaussian(768, 0.3)
        try_seq(parent, "default")

    # === Phase 1: Diverse initializations ===
    print("\n=== Phase 1: Diverse Initializations ===")
    t0 = time.monotonic()

    n_base = len(parent)
    init_candidates = []

    for n_try in [n_base, 512, 640, 896, 1024]:
        init_candidates.append(('gauss', make_gaussian(n_try, 0.3)))
        init_candidates.append(('sine', make_sine(n_try)))
        init_candidates.append(('raised', make_raised_cosine(n_try, 0.3)))
        init_candidates.append(('cos1', make_cosine_mod(n_try, 1, 0.2)))
        init_candidates.append(('cos2', make_cosine_mod(n_try, 2, 0.15)))
        init_candidates.append(('multi', make_multi_cosine(n_try, 4, 0.2, seed=n_try)))
        init_candidates.append(('shape', make_optimal_shape(n_try, 1.5)))
        init_candidates.append(('shape2', make_optimal_shape(n_try, 2.0)))
        init_candidates.append(('shape3', make_optimal_shape(n_try, 3.0)))
        init_candidates.append(('uniform', np.ones(n_try) / n_try))

    if parent is not None:
        init_candidates.append(('parent', parent))

    # Score all and pick top
    scored = []
    for label, a in init_candidates:
        sc = objective_normalized(a)
        scored.append((sc, a, label))
        if sc < 2.0:
            try_seq(a, f"init_{label}_n{len(a)}")
    scored.sort(key=lambda x: x[0])

    print(f"  Best init: obj={scored[0][0]:.6f} ({scored[0][2]}, n={len(scored[0][1])})")
    print(f"  Phase 1 time: {time.monotonic() - t0:.1f}s")

    # === Phase 2: Mirror Descent on best candidates ===
    print("\n=== Phase 2: Mirror Descent ===")
    rem = deadline - time.monotonic() - 400
    for trial in range(min(4, len(scored))):
        rem = deadline - time.monotonic() - 350
        if rem < 60:
            break
        sc, a_init, label = scored[trial]
        t_alloc = min(rem * 0.45, 180)
        if t_alloc < 30:
            break
        print(f"  MD [{label}] n={len(a_init)}: alloc={t_alloc:.0f}s (init_obj={sc:.6f})")
        a_opt, obj = mirror_descent(a_init, time_limit=t_alloc, verbose=True, lr_init=0.08, temp_init_frac=0.02)
        try_seq(a_opt, f"md_{label}")

    # === Phase 3: Multi-peak Descent ===
    print("\n=== Phase 3: Multi-Peak Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 40:
            t_alloc = min(rem * 0.6, 150)
            print(f"  MPD: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, obj = multi_peak_descent(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "mpd")

    # === Phase 4: Pairwise Descent ===
    print("\n=== Phase 4: Pairwise Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_alloc = min(rem, 120)
            print(f"  PD: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, obj = pairwise_descent(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "pd")

    # === Phase 5: Spectral Flattening ===
    print("\n=== Phase 5: Spectral Flattening ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_alloc = min(rem, 80)
            print(f"  Spectral: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, obj = spectral_flatten(best_seq, time_limit=t_alloc, alpha=0.5, verbose=True)
            try_seq(a_opt, "spectral")
            # Try different alpha
            a_opt2, obj2 = spectral_flatten(best_seq, time_limit=min(t_alloc * 0.5, 40), alpha=0.3, verbose=False)
            try_seq(a_opt2, "spectral_a03")

    # === Phase 6: Multi-length exploration ===
    print("\n=== Phase 6: Multi-Length Exploration ===")
    if best_seq is not None:
        base_n = len(best_seq)
        for target_n in [1024, 896, 768, 640, 512]:
            if target_n == base_n:
                continue
            rem = deadline - time.monotonic() - 60
            if rem < 30:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")
            if sc < best_score + 0.1:
                t_alloc = min(rem * 0.3, 50)
                if t_alloc > 15:
                    a_opt, obj = mirror_descent(f_interp, time_limit=t_alloc, verbose=False)
                    try_seq(a_opt, f"md_interp_n{target_n}")

    # === Phase 7: Refined optimization from best ===
    print("\n=== Phase 7: Refined Optimization ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 40
        if rem > 20:
            t_alloc = min(rem * 0.5, 60)
            print(f"  MD polish: alloc={t_alloc:.0f}s")
            a_opt, obj = mirror_descent(best_seq, time_limit=t_alloc, lr_init=0.03, temp_init_frac=0.01)
            try_seq(a_opt, "md_polish")

        rem = deadline - time.monotonic() - 15
        if rem > 10:
            t_alloc = min(rem * 0.6, 30)
            print(f"  PD polish: alloc={t_alloc:.0f}s")
            a_opt, obj = pairwise_descent(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "pd_polish")

    # === Phase 8: Final perturbations and re-optimization ===
    print("\n=== Phase 8: Final Perturbations ===")
    if best_seq is not None and len(best_seq) > 0:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 5
        if rem > 15:
            perturbations = []
            for shift in [1, 2, 3]:
                perturbations.append(('shift', np.roll(best_seq, shift)))
            perturbations.append(('flip', best_seq[::-1]))
            for mix in [0.05, 0.1, 0.15]:
                perturbations.append((f'mix{mix}', (1 - mix) * best_seq + mix * (np.ones(n_best) / n_best)))
            for amp in [0.002, 0.005, 0.01]:
                noise = np.random.default_rng(int(amp * 1000) + 999).standard_normal(n_best) * amp
                perturbations.append((f'noise{amp}', np.maximum(best_seq + noise, 0)))

            for label, f_pert in perturbations:
                rem = deadline - time.monotonic() - 3
                if rem < 8:
                    break
                f_pert = f_pert / f_pert.sum()
                sc = objective_normalized(f_pert)
                if sc < objective_normalized(best_seq) + 0.01:
                    t_alloc = min(rem * 0.4, 20)
                    if t_alloc > 5:
                        a_opt, obj = mirror_descent(f_pert, time_limit=t_alloc, lr_init=0.05)
                        try_seq(a_opt, f"pert_{label}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence([float(x) for x in best_seq]))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
