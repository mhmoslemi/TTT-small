Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using pure gradient-based optimization

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def soft_max_and_gradient(a, T, n):
    """Compute soft-max smoothed objective and gradient."""
    b = fast_convolve(a)
    # For efficiency, only consider a subset of peaks
    # Use top-K + uniform background
    k = min(100, len(b))
    top_idx = np.argpartition(b, -k)[-k:]
    
    # Soft-max over all entries (approximate with top-K)
    b_shifted = b - b.max()
    exp_b = np.exp(b_shifted / T)
    log_sum_exp = b.max() + T * np.log(exp_b.sum())
    
    # Weight for gradient
    w = exp_b / exp_b.sum()
    
    # Compute gradient
    grad = np.zeros(n)
    # Only compute for significant weights
    significant = w > 1e-12
    for idx in np.where(significant)[0]:
        kk = int(idx)
        weight = w[idx]
        js = max(0, kk - n + 1)
        je = min(n - 1, kk)
        i_range = np.arange(js, je + 1)
        grad[i_range] += 2.0 * weight * a[kk - i_range]
    
    obj = 2.0 * n * log_sum_exp
    return obj, grad, w


def adam_step(params, grads, state, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer step. Returns updated params and state."""
    m = state['m']
    v = state['v']
    t = state['t'] + 1
    state['t'] = t
    
    m = beta1 * m + (1 - beta1) * grads
    v = beta2 * v + (1 - beta2) * grads ** 2
    
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    
    update = m_hat / (np.sqrt(v_hat) + eps)
    
    state['m'] = m
    state['v'] = v
    
    return params - lr * update


def project_simplex(v):
    """Project v onto the probability simplex (non-negative, sums to 1)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        out = np.maximum(v - theta, 0.0)
    else:
        out = np.ones(n) / n
    return out


def gradient_descent_optimize(a0, time_limit, T_init=1.0, T_final=0.01, 
                               lr_init=0.01, n_restarts=3, seed=42):
    """
    Pure gradient descent with soft-max, Adam optimizer, and annealing.
    Works in probability simplex (sum=1, non-negative).
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    t0 = time.monotonic()
    total_steps = 0
    
    for restart in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Initialize optimizer state
        state = {'m': np.zeros(n), 'v': np.zeros(n), 't': 0}
        
        # For restarts > 0, perturb
        if restart > 0:
            noise_scale = 0.05 / (restart + 1)
            a = best_a + rng.standard_normal(n) * noise_scale
            a = np.maximum(a, 0)
            a = a / a.sum()
            state = {'m': np.zeros(n), 'v': np.zeros(n), 't': 0}
        
        # Annealing: T goes from T_init to T_final
        # Use a schedule based on elapsed time
        n_steps_target = 5000  # target steps per restart
        restart_start = time.monotonic()
        
        for step in range(n_steps_target):
            if time.monotonic() - t0 > time_limit:
                break
            if step % 200 == 0 and (time.monotonic() - restart_start) > time_limit * 0.8:
                break
            
            # Annealing schedule (cosine)
            progress = min(1.0, step / n_steps_target)
            T = T_final + (T_init - T_final) * 0.5 * (1 + np.cos(np.pi * progress))
            lr = lr_init * (1.0 - 0.5 * progress)
            
            # Compute objective and gradient
            obj, grad, w = soft_max_and_gradient(a, T, n)
            
            # Adam step
            a_new = adam_step(a, grad, state, lr)
            
            # Project to simplex
            a_new = project_simplex(a_new)
            
            # Evaluate true objective
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = best_a.copy()
                continue
            a_new = a_new / s
            
            max_conv = fast_convolve(a_new).max()
            score = 2.0 * n * max_conv
            total_steps += 1
            
            if score < best_score - 1e-14:
                best_score = score
                best_a = a_new.copy()
            
            a = a_new
    
    return best_a, best_score, total_steps

def fast_convolve(a):
