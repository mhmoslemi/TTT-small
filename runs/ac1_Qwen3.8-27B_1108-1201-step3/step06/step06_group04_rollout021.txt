```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using exact coordinate descent with
multi-start 1D optimization, endpoint suppression, multi-length exploration,
and stochastic restarts. Replaces the LP-Frank-Wolfe approach with a direct
coordinate-wise optimization that solves each 1D subproblem exactly via
golden section search on the convex (in practice) 1D objective.
"""
import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_conv_coefficients(a, j):
    """For coordinate j, compute C, L, Q such that conv[k] = C[k] + L[k]*x + Q[k]*x^2.
    Returns C, L, Q arrays of length 2n-1, and S0 = sum(a) - a[j]."""
    n = len(a)
    x = a[j]
    S0 = a.sum() - x
    conv = fast_convolve(a)
    
    # Q[k] = 1 if k == 2j, else 0
    Q = np.zeros(2 * n - 1)
    if 2 * j < 2 * n - 1:
        Q[2 * j] = 1.0
    
    # L[k] = 2*a[k-j] if 0 <= k-j < n and k != 2j, else 0
    L = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        kj = k - j
        if 0 <= kj < n and k != 2 * j:
            L[k] = 2.0 * a[kj]
    
    C = conv - L * x - Q * x * x
    return C, L, Q, S0

def minimize_1d(C, L, Q, S0, n, x_current):
    """Minimize 2*n*max(C + L*x + Q*x^2)/(S0+x)^2 over x >= 0."""
    def f(x):
        vals = C + L * x + Q * x * x
        m = vals.max()
        s = S0 + x
        if s < 1e-15:
            return np.inf
        return 2.0 * n * m / (s * s)
    
    # Multi-start: evaluate at several points
    candidates = [0.0, x_current, max(0, x_current * 0.5), x_current * 2.0,
                  S0 / n, 2.0 * S0 / n]
    candidates = [max(0.0, c) for c in candidates]
    
    best_x = x_current
    best_f = f(x_current)
    
    for cx in candidates:
        fx = f(cx)
        if fx < best_f:
            best_f = fx
            best_x = cx
    
    # Golden section search around best_x
    span = max(0.5 * best_x, S0 / (2 * n), 0.01)
    lo = max(0.0, best_x - span)
    hi = min(4.0 * best_x + S0 / n + 1.0, best_x + span + 1.0)
    
    # Expand hi if needed
    while f(hi) < best_f and hi < 10.0 * S0:
        hi *= 2
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = max(0.0, best_x - span), min(hi, max(best_x + span, 2.0))
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    c_val = f(c_pt)
    d_val = f(d_pt)
    
    for _ in range(50):
        if c_val < d_val:
            hi = d_pt
            d_pt = c_pt
            d_val = c_val
            c_pt = hi - gr * (hi - lo)
            c_val = f(c_pt)
        else:
            lo = c_pt
            c_pt = d_pt
            c_val = d_val
            d_pt = lo + gr * (hi - lo)
            d_val = f(d_pt)
        if abs(hi - lo) < 1e-14:
            break
        mid = (lo + hi) / 2.0
        mid_val = f(mid)
        if mid_val < best_f:
            best_f = mid_val
            best_x = mid
    
    return max(0.0, best_x)

def coordinate_descent(a, max_passes=1000, time_limit=None, t0=None, verbose=False):
    """Cyclic coordinate descent for the max-convolution objective."""
    a = a.copy().astype(np.float64)
    n = len(a)
    
    if time_limit is not None and t0 is None:
        t0 = time.monotonic()
    
    best_obj = 2.0 * n * fast_convolve(a).max() / (a.sum() ** 2)
    best_a = a.copy()
    no_improve = 0
    
    for p in range(max_passes):
        if time_limit is not None and time.monotonic() - t0 > time_limit:
            break
        
        improved = False
        for j in range(n):
            C, L, Q, S0 = compute_conv_coefficients(a, j)
            x_new = minimize_1d(C, L, Q, S0, n, a[j])
            if abs(x_new - a[j]) > 1e-12:
                a[j] = x_new
                improved = True
        
        obj = 2.0 * n * fast_convolve(a).max() / (a.sum() ** 2)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if verbose and p % 50 == 0:
            print(f"  CD pass {p}: obj={best_obj:.10f}, no_improve={no_improve}")
        
        if no_improve > 20:
            break
    
    return best_a, best_obj
```

def update_conv_after_change(conv, a, j, x_old, x_new, n):
    """Update conv after changing a[j] from x_old to x_new."""
    delta = x_new - x_old
    if abs(delta) < 1e-15:
        return conv
    k_lo = j  # k-j >= 0 → k >= j
    k_hi = min(2 * n - 2, j + n - 1)  # k-j <= n-1 → k <= j+n-1
    if k_lo > k_hi:
        return conv
    kj_range = np.arange(k_lo - j, k_hi - j + 1)  # = 0, 1, ..., (k_hi - j)

        candidates.append(make_flat_spectrum(n_try, rng, num_trials=15))
        # Sine
        candidates.append(make_sine_seq(n_try, 1.0))
        candidates.append(make_sine_seq(n_try, 2.0))
        # Gaussian
        candidates.append(make_gaussian_seq(n_try, 0.2))
        candidates.append(make_gaussian_seq(n_try, 0.35))
        # Cosine
        candidates.append(make_cosine_seq(n_try, 2.0))
        candidates.append(make_cosine_seq(n_try, 4.0))
        # Random
        r = rng.random(n_try)
        r /= r.sum()
        candidates.append(r)

        best_init = None
        best_init_obj = np.inf
        for c in candidates:
            c = np.maximum(c, 0.0)
            s = c.sum()
            if s < 1e-10:
                continue
            c = c / s
            o = obj_val(c)
            if o < best_init_obj:
                best_init_obj = o
                best_init = c.copy()
            best_seq, best_score = try_candidate(c, best_seq, best_score, f"init_n{n_try}")

        inits[n_try] = (best_init, best_init_obj)
        print(f"  n={n_try}: best init obj={best_init_obj:.8f}")

    # === Phase 2: Coordinate descent on best initializations ===
    print("\n=== Phase 2: Coordinate Descent ===")
    # Sort lengths by init quality
    sorted_lengths = sorted(len(inits.keys()), key=lambda x: inits[x][1])

    # Allocate time: more time to better initializations
    total_cd_time = max(300.0, budget_s - 120.0)
    # Give 50% to best, 25% to second, 25% to rest
    time_allocs = []
    for i, n_try in enumerate(sorted_lengths):
        if i == 0:
            time_allocs.append(total_cd_time * 0.40)
        elif i == 1:
            time_allocs.append(total_cd_time * 0.25)
        elif i == 2:
            time_allocs.append(total_cd_time * 0.15)
        else:
            time_allocs.append(total_cd_time * 0.075)

    for idx, n_try in enumerate(sorted_lengths):
        rem = deadline - time.monotonic()
        t_alloc = min(time_allocs[idx], rem - 30.0)
        if t_alloc < 10:
            break
        init = inits[n_try][0]
        print(f"  CD n={n_try}: t={t_alloc:.0f}s (init_obj={inits[n_try][1]:.8f})")
        a_opt, obj = coordinate_descent(init, time_limit=t_alloc, t0=t0, verbose=True)
        best_seq, best_score = try_candidate(a_opt, best_seq, best_score, f"cd_n{n_try}")

    # === Phase 3: Stochastic restarts on best ===
    print(f"\n=== Phase 3: Stochastic Restarts (n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    if rem > 60 and best_seq is not None:
        n_best = len(best_seq)
        num_restarts = min(5, int(rem // 60))
        for r in range(num_restarts):
            rem2 = deadline - time.monotonic()
            t_alloc = min(rem2 - 20.0, 60.0)
            if t_alloc < 10:
                break
            # Perturb
            scale = 0.005 * (r + 1)
            noisy = best_seq + rng.standard_normal(n_best) * scale
            noisy = np.maximum(noisy, 0.0)
            noisy /= noisy.sum()
            a_opt, obj = coordinate_descent(noisy, time_limit=t_alloc, t0=t0)
            best_seq, best_score = try_candidate(a_opt, best_seq, best_score, f"restart_{r}")

    # === Phase 4: Try symmetric variant ===
    if best_seq is not None:
        print(f"\n=== Phase 4: Symmetric Variant ===")
        n_best = len(best_seq)
        sym = 0.5 * (best_seq + best_seq[::-1])
        sym = np.maximum(sym, 0.0)
        sym /= sym.sum()
        best_seq, best_score = try_candidate(sym, best_seq, best_score, "symmetric")

        rem = deadline - time.monotonic()
        if rem > 20:
            a_opt, obj = coordinate_descent(sym, time_limit=min(rem - 10, 40), t0=t0)
            best_seq, best_score = try_candidate(a_opt, best_seq, best_score, "sym_cd")

    # === Phase 5: Final polish ===
    if best_seq is not None:
        print(f"\n=== Phase 5: Final Polish ===")
        rem = deadline - time.monotonic()
        if rem > 15:
            a_opt, obj = coordinate_descent(best_seq, time_limit=min(rem - 5, 50), t0=t0)
            best_seq, best_score = try_candidate(a_opt, best_seq, best_score, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(640) / 640
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
