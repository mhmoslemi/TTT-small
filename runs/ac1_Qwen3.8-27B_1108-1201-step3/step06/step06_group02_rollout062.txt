def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    # Sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_max_gradient(a, conv, T):
    """
    Compute the gradient of the smoothed max of conv with respect to a.
    Uses FFT for efficiency.
    
    Returns: grad (same shape as a)
    """
    n = len(a)
    # Softmax weights
    conv_shifted = conv - conv.max()
    scaled = T * conv_shifted
    scaled -= scaled.max()  # numerical stability (already shifted by max, so max is 0)
    # Actually conv_shifted max is 0, so scaled max is 0
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:

        w = np.ones(len(conv)) / len(conv)
    else:
        w /= w_sum

    W = np.fft.rfft(w_rev, fft_size)
    h = np.fft.irfft(A * W, fft_size)[:conv_size]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated smoothed gradient
descent with log-sum-exp annealing, combined with peak-targeted coordinate
descent and multi-length exploration. Key innovations:
- O(n log n) gradient computation via FFT (replacing expensive LP solves)
- Smoothed max (log-sum-exp) with adaptive temperature annealing
- Multiplicative (mirror descent) updates on the simplex
- Peak-targeted coordinate descent with swap operations for final polish

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max_gradient(a, conv, T):
    """
    Compute gradient of (1/T)*log(sum exp(T*conv)) w.r.t. a using FFT.
    Returns array of same shape as a.
    """
    n = len(a)
    shifted = conv - conv.max()
    w = np.exp(T * shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(len(conv)) / len(conv)
    else:
        w = w / w_sum

    w_rev = w[::-1]
    conv_len = 3 * n - 2
    fft_size = 1 << (conv_len - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    h = np.fft.irfft(A * W, fft_size)[:conv_len]
    grad = 2.0 * h[n - 1:2 * n - 1][::-1]
    return grad


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian-like initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_cosine_mod(n, freq, amp):
    """Cosine-modulated uniform sequence."""
    x = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_multi_cosine(n, n_freqs=3, max_amp=0.3, seed=0):
    """Multi-frequency cosine-modulated sequence."""
    rng = np.random.default_rng(seed)
    a = np.ones(n)
    for m in range(1, n_freqs + 1):
        amp = rng.uniform(0.05, max_amp)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * m * np.arange(n) / n + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_linear_ramp(n, alpha=1.0):
    """Linearly increasing sequence (power law)."""
    a = np.arange(1, n + 1) ** alpha
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bimodal(n, gap_frac=0.3):
    """Two Gaussian bumps."""
    x = np.arange(n)
    c1 = n * (0.5 - gap_frac / 2)
    c2 = n * (0.5 + gap_frac / 2)
    sigma = n * 0.15
    a = np.exp(-0.5 * ((x - c1) / sigma) ** 2) + np.exp(-0.5 * ((x - c2) / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

        else:
            no_improve += 1

        grad = smooth_max_gradient(a, conv, T)

        else:
            grad_clipped = grad

        update = np.exp(-eta * grad_clipped)

        if a_new_sum < 1e-15:

        else:
            a_new = a_new / a_new_sum

        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

            a = a_new

def multiplicative_descent(a0, time_limit, lr_init=0.003, decay=0.9997,
                           top_k=20, verbose=False):
    """Mirror descent focusing on top peaks."""
    a = np.maximum(a0.astype(np.float64), 1e-12)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 1.0:
            break

        conv = fast_convolve(a)
        M = conv.max()

        # Focus on top peaks
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        # Weight by how close to max
        w = np.exp(0.5 * (pv - M))
        w = w / w.sum()

        # Subgradient from top peaks
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

        # Multiplicative update
        a = a * np.exp(-lr * sg)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        co = fast_convolve(a).max()
        if co < best_obj - 1e-14:
            best_obj = co
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"    MD iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        lr *= decay

        if no_improve > 300:
            rng = np.random.default_rng(it * 31 + 7)
            a = np.maximum(best_a + rng.standard_normal(n) * 0.002, 1e-15)
            a = a / a.sum()
            lr = lr_init * 0.5
            no_improve = 0

        if lr < 1e-10:
            break

    return best_a, best_obj

                best_seq, time_limit=t_alloc,
                lr_init=0.001, decay=0.9998, top_k=50)
            try_seq(a_opt, "md_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    total_time = time.monotonic() - (deadline - max(1.0, float(budget_s) - 15.0))
    print(f"Total time: {total_time:.1f}s")
    return [float(x) for x in best_seq]
