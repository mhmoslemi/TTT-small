import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """T @ b = conv(a, b), T is (2n-1) x n"""
    n = len(a)
    m = len(a)
    T = np.zeros((2*n-1, m))
    for j in range(m):
        T[j:j+n, j] = a
    return T


def build_conv_matrix_transpose(a):
    """T^T, shape (n, 2n-1). Column k of T^T has a[valid range] placed appropriately."""
    n = len(a)
    Tt = np.zeros((n, 2*n-1))
    for k in range(2*n-1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je+1)
        Tt[idx, k] = a[k - idx]
    return Tt


def get_lp_solver():
    global _lp_solver
    if '_lp_solver' in globals():
        return _lp_solver
    for solver_name in [cp.MINOS, cp.CBC, cp.GLPK, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response_primal(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b (non-negative, sum=1) minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.GLPK, cp.CBC]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, None
    b_val = b_val / s
    return b_val, float(M_var.value), None


def solve_lp_best_response_dual(a_fixed, solver=None):
    """
    Given fixed a (non-negative, sum=1), solve the dual LP to find
    the weights y over lags and the best response b.
    Dual: max sum(y) subject to T(a)^T @ y <= 1, y >= 0
    where T(a)^T has columns that are the "fingerprint" of each lag.
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    # Dual: max 1^T y subject to T^T y <= 1, y >= 0
    # T^T is n x (2n-1), y is (2n-1) x 1
    y_var = cp.Variable(2*n-1, nonneg=True)
    Tt = build_conv_matrix_transpose(a_fixed)
    constraints = [
        Tt @ y_var <= 1
    ]
    prob = cp.Problem(cp.Maximize(cp.sum(y_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.GLPK, cp.CBC]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    y_val = np.asarray(y_var.value, dtype=np.float64)
    y_val = np.maximum(y_val, 0)
    return y_val, float(prob.value)


def fictitious_play(a_init, time_limit, solver=None, verbose=True):
    """Run fictitious play. Returns (best_a, best_score, a_avg)."""
    if solver is None:
        solver = get_lp_solver()
    t0 = time.monotonic()
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0)
    a = a / a.sum()
    b = a.copy()
    a_avg = a.copy().astype(np.float64)
    b_avg = b.copy().astype(np.float64)
    best_score = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        # Best response for b given a
        b_new, M_b, _ = solve_lp_best_response_primal(a, solver)
        if b_new is not None:
            b = b_new
        
        # Best response for a given b
        a_new, M_a, _ = solve_lp_best_response_primal(b, solver)
        if a_new is not None:
            a = a_new
        
        # Update running averages
        a_avg = (iters * a_avg + a) / (iters + 1)
        b_avg = (iters * b_avg + b) / (iters + 1)
        
        # Evaluate candidates
        for cand in [a, b, 0.5*(a+b), a_avg, 0.5*(a_avg+b_avg)]:
            cand = np.maximum(cand, 0)
            s = cand.sum()
            if s < 1e-10:
                continue
            cand_norm = cand / s
            score = 2.0 * n * fast_convolve(cand_norm).max()
            if score < best_score:
                best_score = score
                best_a = cand_norm.copy()
        
        if verbose and iters % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"  FP iter={iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score, a_avg


def soft_max_objective(z, n, T_temp, w_lags=None):
    """Compute soft-max smoothed objective and gradient in log-space."""
    # z -> a via softmax
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    
    # Compute auto-correlation
    c = fast_convolve(a)
    
    if w_lags is not None:
        # Weighted soft-max over selected lags
        Tc = T_temp * c * w_lags
        Tc_max = Tc.max()
        exp_Tc = np.exp(Tc - Tc_max)
        log_sum_exp = Tc_max + np.log(exp_Tc.sum())
        w = exp_Tc / exp_Tc.sum()
        obj = (2.0 * n / T_temp) * log_sum_exp
    else:
        Tc = T_temp * c
        Tc_max = Tc.max()
        exp_Tc = np.exp(Tc - Tc_max)
        log_sum_exp = Tc_max + np.log(exp_Tc.sum())
        w = exp_Tc / exp_Tc.sum()
        obj = (2.0 * n / T_temp) * log_sum_exp
    
    # Gradient: grad_a = 2 * cross_corr(w, a)
    # cross_corr(w, a)[i] = sum_l w[l] * a[l-i]
    size = n + (2*n-1) - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad_a = irfft(np.conj(W) * A, fft_size)[:n] * 2.0
    
    # Gradient in z-space
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return obj, grad_z


def lbgfs_polish(a0, time_limit, solver=None, verbose=True):
    """L-BFGS polishing with adaptive temperature."""
    n = len(a0)
    t0 = time.monotonic()
    
    # Start with moderate temperature, increase over time
    temperatures = [50, 100, 200, 500, 1000, 2000, 5000]
    
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    
    z = np.log(np.maximum(a0, 1e-300))
    
    for T in temperatures:
        if time.monotonic() - t0 > time_limit:
            break
        
        stopped = [False]
        
        def func(z_inner):
            if time.monotonic() - t0 > time_limit:
                stopped[0] = True
                return 1e15
            obj, _ = soft_max_objective(z_inner, n, T)
            return obj
        
        def jac(z_inner):
            if time.monotonic() - t0 > time_limit:
                stopped[0] = True
                return np.zeros_like(z_inner)
            _, g = soft_max_objective(z_inner, n, T)
            return g
        
        try:
            maxiter = min(500, max(50, int((time_limit - (time.monotonic() - t0)) * 10)))
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxfun': maxiter * 50, 'disp': False}
            )
            a_out = np.exp(res.x - res.x.max())
            a_out = a_out / a_out.sum()
            a_out = np.maximum(a_out, 0)
            
            score = 2.0 * n * fast_convolve(a_out).max()
            if score < best_score:
                best_score = score
                best_a = a_out.copy()
                z = res.x.copy()
            
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    L-BFGS T={T} ({elapsed:.0f}s): score={best_score:.8f}")
        except Exception as e:
            if verbose:
                print(f"    L-BFGS T={T} failed: {e}")
    
    return best_a, best_score


def subgradient_polish(a0, time_limit, top_k=30, verbose=True):
    """Subgradient method with momentum for fine-tuning."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    v = np.zeros(n)
    mom = 0.9
    lr = 0.005
    t0 = time.monotonic()
    no_improve = 0
    step_count = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        k_actual = min(top_k, len(c))
        peaks = np.argpartition(c, -k_actual)[-k_actual:]
        pv = c[peaks]
        
        # Temperature for weighting peaks
        temp = 0.003 * max(pv.max(), 1e-10)
        scaled = (pv - pv.max()) / temp
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute subgradient
        grad = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * w[i] * a[k - idx]

        v = mom * v - lr * grad

        # Project onto simplex
        a_candidate = a + v
        u = np.sort(a_candidate)[::-1]
        cssv = np.cumsum(u)
        cond = u * np.arange(1, n + 1) > (cssv - 1)
        if cond.any():
            rho = np.nonzero(cond)[0][-1]
            theta = (cssv[rho] - 1.0) / (rho + 1.0)
            a_new = np.maximum(a_candidate - theta, 0.0)
        else:
            a_new = np.ones(n) / n

        a_new = a_new / a_new.sum()
        score = 2.0 * n * fast_convolve(a_new).max()
        step_count += 1

        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.999
        if no_improve > 200:
            # Restart with perturbation
            rng = np.random.default_rng(step_count)
            a = best_a + rng.standard_normal(n) * 0.001
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            lr = 0.003
            no_improve = 0

        if verbose and step_count % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad {elapsed:.0f}s: score={best_score:.8f} lr={lr:.6f}")

    return best_a, best_score


def peak_shave(a0, time_limit, verbose=True):
    """Local peak shaving - reduce contributions to the peak lag."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    step = 0

    while time.monotonic() - t0 < time_limit:
        step += 1
        c = fast_convolve(a)
        peak_k = int(np.argmax(c))
        
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        if contribs.max() < 1e-20:
            break
        
        improved = False
        # Find top contributors and try to reduce them
        top_n = min(5, len(contribs))
        top_idx = np.argsort(contribs)[::-1][:top_n]
        
        for ci in top_idx:
            i = idx[ci]
            j = peak_k - i
            old_val = a[i]
            if old_val < 1e-14:
                continue
            for factor in [0.98, 0.95, 0.9, 0.85]:
                new_val = old_val * factor
                diff = old_val - new_val
                if diff < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a[mask]
                other_sum = others.sum()
                if other_sum < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                a_trial[mask] = others + diff * others / other_sum
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try paired reduction
            for ci in top_idx:
                i = idx[ci]
                j = peak_k - i
                if i == j or a[j] < 1e-14:
                    continue
                for f1 in [0.95, 0.9, 0.85]:
                    for f2 in [0.95, 0.9, 0.85]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        s = a_trial.sum()
                        if s < 1e-15:
                            continue
                        a_trial = a_trial / s
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-15:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if not improved:
                break
    
    if verbose:
        print(f"    PeakShave: {step} steps, score={best_score:.8f}")
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):

        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score
