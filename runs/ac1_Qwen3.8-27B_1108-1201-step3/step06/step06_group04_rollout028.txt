"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach combining:
1. Gradient descent with softmax-smoothed max objective (differentiable proxy)
2. LP-based Frank-Wolfe direction for local refinement
3. Parametric family optimization for global exploration
4. Multi-length, multi-start strategy to avoid local minima
Key innovation: uses the analytical gradient of the smoothed self-convolution peak,
exploiting the structure d(conv(a,a)[k])/d(a[i]) = 2*a[k-i] for valid indices.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_normalized(f):
    """Objective for sum-normalized sequence: 2*n*max(conv)."""
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()


def softmax_grad_objective(a, T):
    """
    Compute the smoothed objective and its gradient.
    Returns (smoothed_obj, gradient) where gradient is w.r.t. a.
    """
    n = len(a)
    conv = fast_convolve(a)
    # Softmax approximation of max
    c_scaled = conv / T
    c_max = c_scaled.max()
    exp_vals = np.exp(c_scaled - c_max)
    log_sum_exp = c_max + np.log(exp_vals.sum())
    smoothed_max = T * log_sum_exp
    obj = 2.0 * n * smoothed_max
    # Softmax weights
    p = exp_vals / exp_vals.sum()
    # Gradient: grad[i] = 2 * sum_j p[j+i] * a[j] for valid indices
    grad = np.zeros(n)
    for i in range(n):
        valid = min(n, len(p) - i)
        if valid > 0:
            grad[i] = 2.0 * np.dot(a[:valid], p[i:i + valid])
    return obj, grad


def gradient_descent_smoothed(f0, time_limit, T_init=None, verbose=False):

    Gradient descent with softmax-smoothed max objective.
    Uses adaptive temperature and learning rate.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)

    M0 = conv0.max()

    else:
        T = T_init

    best_obj = 2.0 * n * M0
    best_f = f.copy()

    t0 = timeomonotonic()
    lr = 0.1
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        obj, grad = softmax_grad_objective(f, T)

        # Project gradient to tangent space (zero sum)
        grad_mean = grad.mean()
        grad_tangent = grad - grad_mean

        # Gradient step with projection onto simplex
        f_new = f - lr * grad_tangent
        f_new = project_simplex(f_new)

        # Evaluate actual objective
        conv_new = fast_convolve(f_new)
        actual_obj = 2.0 * n * conv_new.max()

        if actual_obj < best_obj - 1e-14:
            best_obj = actual_obj
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            # Slowly decrease learning rate
            lr *= 0.995
            # Slowly decrease temperature
            T *= 0.998
        else:
            no_improve += 1
            # Backtrack: try smaller step
            for backtrack in range(5):
                f_try = f - lr * (0.5 ** backtrack) * grad_tangent
                f_try = project_simplex(f_try)
                conv_try = fast_convolve(f_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_f = f_try.copy()
                    f = f_try
                    no_improve = 0
                    break
            else:

                T *= 1.05  # Increase temperature to smooth out
                if no_improve > 50:

                    T = 0.05 * conv.max()
                    lr = 0.1
                    no_improve = 0

        if T < 1e-10:

        if lr < 1e-10:
            break

def build_lp_matrix(f, n):
    """Build the constraint matrix for the LP: (f*g)[k] for all k."""
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        length = k_end - k_start + 1
        A_ub[k_start:k_end + 1, j] = f[:length]
    return A_ub


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    A_ub = build_lp_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': 30})
    if res.success:
        return res.x
    return None


def lp_refine(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe refinement."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.2
            if m_exp > 5.0:
                no_improve += 5
                m_exp = 1.0
            continue

        g = np.maximum(g, 0)
        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp *= 1.1
            continue
        g = g / g_sum

        diff = np.max(np.abs(g - f))
        if diff < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        # Line search
        best_alpha = 0
        best_val = best_obj
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            f_try = f + alpha * (g - f)
            f_try = np.maximum(f_try, 0)
            f_try = f_try / f_try.sum()
            val = 2.0 * n * fast_convolve(f_try).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            f = f + best_alpha * (g - f)
            f = np.maximum(f, 0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            if 2.0 * n * M < best_obj:
                best_obj = 2.0 * n * M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 5.0)

        if verbose and it % 10 == 0:
            print(f"    LP iter {it}: obj={best_obj:.10f}, m_exp={m_exp:.2f}")

        if no_improve > 60:
            break

    return best_f, best_obj


def parametric_optimize(n, time_limit, verbose=False):
    """
    Optimize over parametric families:
    a[j] = exp(-alpha * x^gamma) * exp(beta * x) where x = j/(n-1)
    """
    def param_obj(params):
        alpha, gamma, beta = params
        if alpha < 0 or gamma < 0 or abs(beta) > 10:
            return 1e10
        x = np.linspace(0, 1, n)
        a = np.exp(-alpha * np.power(x, gamma)) * np.exp(beta * x)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            return 1e10
        a = a / s
        return obj_normalized(a)

    best_obj = float('inf')
    best_params = None
    best_a = None

    # Try multiple initial points
    inits = [
        [1.0, 2.0, 0.0],
        [0.5, 1.0, 0.5],
        [0.5, 1.0, -0.5],
        [1.0, 3.0, 0.0],
        [0.3, 1.0, 1.0],
        [2.0, 1.5, 0.0],
        [0.8, 2.5, -0.3],
        [1.5, 1.0, 0.3],
    ]

    t0 = time.monotonic()
    per_init = max(1.0, time_limit / len(inits))

    for init in inits:
        if time.monotonic() - t0 > time_limit:
            break
        res = minimize(param_obj, init, method='Nelder-Mead',
                       options={'maxiter': 500, 'xatol': 1e-8, 'fatol': 1e-10,
                                'adaptive': True})
        if res.fun < best_obj:
            best_obj = res.fun
            best_params = res.x
            alpha, gamma, beta = best_params
            x = np.linspace(0, 1, n)
            best_a = np.exp(-alpha * np.power(x, gamma)) * np.exp(beta * x)
            best_a = np.maximum(best_a, 0)
            best_a = best_a / best_a.sum()
            if verbose:
                print(f"      Param: alpha={alpha:.3f}, gamma={gamma:.3f}, "
                      f"beta={beta:.3f}, obj={best_obj:.8f}")

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, sigma_frac=0.25, center=0.5):
    """Gaussian initial sequence."""
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - center) / sigma_frac) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_exponential(n, alpha=2.0):
    """Exponential decay sequence."""
    x = np.linspace(0, 1, n)
    a = np.exp(-alpha * x)
    a = a / a.sum()
    return a


def make_power_law(n, gamma=2.0):
    """Power law decay sequence."""
    x = np.linspace(0.01, 1, n)
    a = np.power(x, -gamma)
    a = a / a.sum()

def make_sinc_like(n, width=0.3):

    while time.monotonic() - t0 < time_limit:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break

        if phase == 0:
            # Gradient descent phase
            t_gd = min(rem, rem * gd_time_frac)
            if t_gd < 3:
                phase = 1
                continue
            f_new, obj_new = gradient_descent_smoothed(f, time_limit=t_gd, verbose=False)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_f = f_new.copy()
                no_improve_total = 0
            else:
                no_improve_total += 1
            f = f_new
            phase = 1
        else:
            # LP refinement phase
            t_lp = min(rem, rem * (1 - gd_time_frac))
            if t_lp < 3:
                phase = 0
                continue
            f_new, obj_new = lp_refine(f, time_limit=t_lp, verbose=False)
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_f = f_new.copy()
                no_improve_total = 0
            else:
                no_improve_total += 1
            f = f_new
            phase = 0

        if verbose:
            print(f"    Combined: phase={phase}, obj={best_obj:.10f}, "
                  f"no_improve={no_improve_total}")

        if no_improve_total > 10:
            # Try perturbation
            rng = np.random.default_rng(hash(str(time.monotonic())) % (2**32))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            no_improve_total = 0

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization combining gradient descent on smoothed objective,
    LP-based Frank-Wolfe, parametric family optimization, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_gaussian(768, 0.25)
        best_seq, best_score = try_sequence(parent, best_seq, best_score, "default")

    n_parent = len(parent)

    # === Phase 1: Quick parametric search at parent length ===
    print("=== Phase 1: Parametric Search ===")
    t1 = min(120, (deadline - time.monotonic()) * 0.1)
    if t1 > 10:
        print(f"  Parametric (n={n_parent}): t={t1:.0f}s")
        a_param, obj_param = parametric_optimize(n_parent, time_limit=t1, verbose=True)
        if a_param is not None:
            best_seq, best_score = try_sequence(a_param, best_seq, best_score,
                                                 f"param_n{n_parent}")

    # === Phase 2: Gradient descent from best ===
    base = best_seq if best_seq is not None else parent
    print(f"\n=== Phase 2: Gradient Descent (n={len(base)}) ===")
    rem = deadline - time.monotonic() - 600
    if rem > 30:
        t2 = min(rem, 250)
        print(f"  GD: t={t2:.0f}s")
        a_opt, obj = gradient_descent_smoothed(base, time_limit=t2, verbose=True)
        best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "gd_main")

    # === Phase 3: LP refinement ===
    if best_seq is not None:
        print(f"\n=== Phase 3: LP Refinement (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 400
        if rem > 20:
            t3 = min(rem, 150)
            print(f"  LP: t={t3:.0f}s")
            a_opt, obj = lp_refine(best_seq, time_limit=t3, verbose=True)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "lp_main")

    # === Phase 4: Multi-length exploration ===
    base = best_seq if best_seq is not None else parent
    print(f"\n=== Phase 4: Multi-length Exploration ===")
    target_lengths = [640, 700, 768, 832, 900]
    for tl in target_lengths:
        if len(base) == tl:
            continue
        rem = deadline - time.monotonic() - 200
        if rem < 40:
            break
        f_interp = interpolate_sequence(base, tl)
        best_seq, best_score = try_sequence(f_interp, best_seq, best_score,
                                             f"interp_n{tl}")
        # Combined optimize at this length
        t_alloc = min(rem * 0.3, 80)
        if t_alloc < 15:
            continue
        print(f"  Combined n={tl}: t={t_alloc:.0f}s")
        a_opt, obj = combined_optimize(f_interp, time_limit=t_alloc, verbose=False)
        best_seq, best_score = try_sequence(a_opt, best_seq, best_score,
                                             f"combined_n{tl}")

    # === Phase 5: Diverse initializations + combined optimize ===
    base = best_seq if best_seq is not None else parent
    n_base = len(base)
    print(f"\n=== Phase 5: Diverse Init + Optimize ===")
    diverse_inits = [
        ("gauss_02", make_gaussian(n_base, 0.2)),
        ("gauss_03", make_gaussian(n_base, 0.3)),
        ("exp_2", make_exponential(n_base, 2.0)),
        ("exp_1", make_exponential(n_base, 1.0)),
        ("powlaw", make_power_law(n_base, 2.0)),
        ("sinc", make_sinc_like(n_base, 0.3)),
        ("tilted_gauss", None),  # Will compute
    ]
    # Create tilted gaussian
    x = np.linspace(0, 1, n_base)
    tilted = np.exp(-0.5 * ((x - 0.4) / 0.25) ** 2) * np.exp(1.5 * x)
    tilted = np.maximum(tilted, 0)
    tilted = tilted / tilted.sum()
    diverse_inits[6] = ("tilted_gauss", tilted)

    for name, f_init in diverse_inits:
        rem = deadline - time.monotonic() - 100
        if rem < 30:
            break
        best_seq, best_score = try_sequence(f_init, best_seq, best_score,
                                             f"div_{name}")
        t_alloc = min(rem * 0.25, 60)
        if t_alloc < 10:
            continue
        print(f"  {name}: t={t_alloc:.0f}s")
        a_opt, obj = combined_optimize(f_init, time_limit=t_alloc, verbose=False)
        best_seq, best_score = try_sequence(a_opt, best_seq, best_score,
                                             f"opt_{name}")

    # === Phase 6: Deep refinement on best ===
    if best_seq is not None:
        print(f"\n=== Phase 6: Deep Refinement (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 30
        if rem > 30:
            t6 = min(rem, 200)
            # Split between GD and LP
            t6a = t6 * 0.5
            t6b = t6 * 0.5
            print(f"  Deep GD: t={t6a:.0f}s")
            a_opt, obj = gradient_descent_smoothed(best_seq, time_limit=t6a, verbose=True)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "deep_gd")

            if best_seq is not None:
                rem2 = deadline - time.monotonic() - 10
                t6b_actual = min(t6b, rem2)
                if t6b_actual > 5:
                    print(f"  Deep LP: t={t6b_actual:.0f}s")
                    a_opt, obj = lp_refine(best_seq, time_limit=t6b_actual, verbose=True)
                    best_seq, best_score = try_sequence(a_opt, best_seq, best_score,
                                                         "deep_lp")

    # === Phase 7: Final micro-optimization ===
    if best_seq is not None:
        print(f"\n=== Phase 7: Final Polish ===")
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            # Very short GD with low temperature
            t7 = min(rem, 50)
            a_opt, obj = gradient_descent_smoothed(best_seq, time_limit=t7,
                                                   T_init=1e-5, verbose=False)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "polish_gd")

            rem2 = deadline - time.monotonic() - 2
            if rem2 > 5 and best_seq is not None:
                a_opt, obj = lp_refine(best_seq, time_limit=min(rem2, 25), verbose=False)
                best_seq, best_score = try_sequence(a_opt, best_seq, best_score,
                                                     "polish_lp")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
