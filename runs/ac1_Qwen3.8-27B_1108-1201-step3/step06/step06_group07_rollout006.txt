```python
"""
LP-based optimization with mirror descent refinement for minimizing
2*n*max(conv(a,a))/sum(a)^2.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_lp_matrix(a):
    """Build the constraint matrix A where A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    # Use sparse matrix for efficiency
    rows = []
    cols = []
    vals = []
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        for k in range(k_start, k_end + 1):
            idx = k - j
            if 0 <= idx < n:
                rows.append(k)
                cols.append(j)
                vals.append(a[idx])
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

def lp_optimize(a, M, time_limit=None):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
```

```python
            a_sym = np.maximum(a_sym, 1e-15)
            a_sym = a_sym / a_sym.sum()
            sym_obj = fast_convolve(a_sym).max()
            if sym_obj < best_obj:
                best_obj = sym_obj
                best_a = a_sym.copy()
                a = a_sym.copy()

        if lr < 1e-10:
            break

    return best_a, best_obj


def additive_descent(a0, time_limit, top_k=30, temp_init=0.001, dt_init=0.003,
                     verbose=False, label=""):
    """Additive (rectilinear) flow on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / max(a.sum(), 1e-15)
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    dt = dt_init
    temp = temp_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()

        K = min(top_k, len(conv))
        if K < len(conv):
            peak_indices = np.argpartition(conv, -K)[-K:]
        else:
            peak_indices = np.arange(len(conv))

        pv = conv[peak_indices]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peak_indices))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        g = np.zeros(n)
        for idx in range(len(peak_indices)):
            k = int(peak_indices[idx])
            w_k = w[idx]
            if w_k < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue
            g[js:je + 1] += 2.0 * w_k * a[k - np.arange(js, je + 1)]

        g_centered = g - g.mean()

        best_dt = dt * 0.1
        best_trial = M * 1.01
        for trial_dt in [dt, dt * 0.5, dt * 0.25, dt * 0.1]:
            if trial_dt < 1e-12:
                continue
            a_trial = a - trial_dt * g_centered
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                a_trial = a.copy()
            else:
                a_trial = a_trial / s
            trial_obj = fast_convolve(a_trial).max()
            if trial_obj <= best_trial:
                best_dt = trial_dt
                best_trial = trial_obj
                if trial_obj < M * 0.998:
                    break

        a = a - best_dt * g_centered
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
        else:
            a = a / s

        conv = fast_convolve(a)
        new_obj = conv.max()

        if new_obj < best_obj - 1e-13:
            best_obj = new_obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    {label} AD iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        temp = min(temp * 1.003, 0.05)
        dt = max(dt * 0.9999, 1e-10)

        if no_improve > 150:
            rng = np.random.default_rng(it * 104729 + 3)
            a = np.maximum(best_a + rng.standard_normal(n) * 0.002, 0)
            a = a / max(a.sum(), 1e-15)
            dt = dt_init * 0.3
            no_improve = 0

        if it % 500 == 0:
            a_sym = 0.5 * (a + a[::-1])
            a_sym = np.maximum(a_sym, 0)
            s = a_sym.sum()
            if s > 1e-10:
                a_sym = a_sym / s
                sym_obj = fast_convolve(a_sym).max()
                if sym_obj < best_obj:
                    best_obj = sym_obj
                    best_a = a_sym.copy()
                    a = a_sym.copy()

    return best_a, best_obj


def coordinate_descent(a0, time_limit, verbose=False, label=""):
    """Fine coordinate descent with binary search on each coordinate."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / max(a.sum(), 1e-15)
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    rounds = 0

    while time.monotonic() - t0 < time_limit:
        rounds += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in [0.7, 0.85, 0.93, 1.07, 1.15, 1.3]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                trial = a.copy()
                trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                trial[mask] *= scale
                trial = np.maximum(trial, 0.0)
                ss = trial.sum()
                if ss < 1e-15:
                    continue
                trial = trial / ss
                val = fast_convolve(trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0.0)
                    a = a / max(a.sum(), 1e-15)
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            break

    return best_a, best_obj


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_triangular(n):
    a = np.minimum(np.arange(1, n + 1), n - np.arange(1, n + 1) + 1)
    a = a.astype(np.float64)
    a /= a.sum()
    return a


def make_bell(n, width_frac=0.6):
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bimodal(n, c1=0.3, c2=0.7, sigma=0.08):
    x = np.arange(n) / n
    a = np.exp(-0.5 * ((x - c1) / sigma) ** 2) + np.exp(-0.5 * ((x - c2) / sigma) ** 2)
    a = a / a.sum()
    return a


def make_sinc_like(n, width=6):
    x = np.arange(n) - n / 2.0
    a = np.sinc(x * width / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_flat_top(n, flat_frac=0.4, roll=0.6):
    x = np.arange(n) / n
    a = np.where(np.abs(x - 0.5) < flat_frac / 2, 1.0,
                 1.0 - (np.abs(x - 0.5) - flat_frac / 2) / max(roll * 0.5, 1e-10) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization: LP-guided search, mirror descent, additive
    descent, coordinate descent, across diverse initializations and lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # === Phase 1: LP optimization from parent ===
    print("\n=== Phase 1: LP from parent ===")
    if parent is not None:
        remaining = deadline - time.monotonic()
        t_alloc = min(remaining * 0.25, 400)
        a_p = parent / parent.sum()
        n_p = len(a_p)
        a_norm = a_p * np.sqrt(2 * n_p)
        a_opt, obj = lp_optimize(a_norm, time_limit=t_alloc, verbose=True, label="LP_parent")
        a_out = a_opt / np.sqrt(2 * n_p)
        try_seq(a_out, "LP_parent")

    # === Phase 2: LP with different lengths ===
    print("\n=== Phase 2: LP with different n ===")
    n_candidates = [640, 704, 768, 832, 896, 1024]
    if parent is not None:
        n_candidates = [len(parent) - 32, len(parent), len(parent) + 32,
                        len(parent) + 64, len(parent) + 128]
        n_candidates = [n for n in n_candidates if 32 <= n <= 2048]

    remaining = deadline - time.monotonic()
    t_per_n = min(remaining * 0.20 / len(n_candidates), 80)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        t_alloc = min(t_per_n, deadline - time.monotonic() - 30)
        if t_alloc < 15:
            break
        a_init = make_gaussian(n_try, 0.28) * np.sqrt(2 * n_try)
        a_opt, obj = lp_optimize(a_init, time_limit=t_alloc, verbose=False,
                                 label=f"LP_n{n_try}")
        a_out = a_opt / np.sqrt(2 * n_try)
        try_seq(a_out, f"LP_n{n_try}")

    # === Phase 3: Mirror descent refinement ===
    print("\n=== Phase 3: Mirror Descent ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_alloc = min(remaining * 0.20, 350)
        a_norm = best_seq / best_seq.sum()
        a_opt, obj = mirror_descent(a_norm, time_limit=t_alloc, top_k=30,
                                    temp_init=0.0003, lr_init=0.01,
                                    verbose=True, label="MD_refine")
        try_seq(a_opt, "MD_refine")

        # Symmetrized version
        a_sym = 0.5 * (a_opt + a_opt[::-1])
        a_sym = np.maximum(a_sym, 1e-15)
        a_sym = a_sym / a_sym.sum()
        try_seq(a_sym, "MD_sym")

        if time.monotonic() < deadline - 120:
            t_alloc2 = min(120, deadline - time.monotonic() - 30)
            a_opt2, obj2 = mirror_descent(a_sym, time_limit=t_alloc2, top_k=15,
                                          temp_init=0.0001, lr_init=0.005,
                                          verbose=False, label="MD_fine")
            try_seq(a_opt2, "MD_fine")

    # === Phase 4: Additive descent ===
    print("\n=== Phase 4: Additive Descent ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_alloc = min(remaining * 0.12, 200)
        if t_alloc > 20:
            a_norm = best_seq / best_seq.sum()
            a_opt, obj = additive_descent(a_norm, time_limit=t_alloc, top_k=30,
                                          temp_init=0.001, dt_init=0.003,
                                          verbose=True, label="AD")
            try_seq(a_opt, "AD")

    # === Phase 5: Diverse initialization + LP ===
    print("\n=== Phase 5: Diverse init + LP ===")
    remaining = deadline - time.monotonic()
    n_base = len(best_seq) if best_seq is not None else 768
    inits = [
        (make_gaussian(n_base, 0.22), f"gauss_n{n_base}_narrow"),
        (make_gaussian(n_base, 0.35), f"gauss_n{n_base}_wide"),
        (make_bell(n_base, 0.5), f"bell_n{n_base}"),
        (make_bell(n_base, 0.7), f"bell_w_n{n_base}"),
        (make_triangular(n_base), f"tri_n{n_base}"),
        (make_bimodal(n_base), f"bimodal_n{n_base}"),
        (make_sinc_like(n_base, 4), f"sinc_n{n_base}"),
        (make_flat_top(n_base, 0.3, 0.5), f"flat_n{n_base}"),
        (make_flat_top(n_base, 0.5, 0.3), f"flat_w_n{n_base}"),
    ]

    t_per_init = min(remaining * 0.15 / len(inits), 45)

    for a_init, label in inits:
        if time.monotonic() > deadline - 60:
            break
        t_alloc = min(t_per_init, deadline - time.monotonic() - 30)
        if t_alloc < 15:
            break
        a_scaled = a_init * np.sqrt(2 * n_base)
        a_opt, obj = lp_optimize(a_scaled, time_limit=t_alloc, verbose=False,
                                 label=f"LP_{label}")
        a_out = a_opt / np.sqrt(2 * n_base)
        try_seq(a_out, f"LP_{label}")

    # === Phase 6: Coordinate descent polish ===
    print("\n=== Phase 6: Coordinate Descent ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_alloc = min(remaining * 0.10, 150)
        if t_alloc > 15:
            a_norm = best_seq / best_seq.sum()
            a_opt, obj = coordinate_descent(a_norm, time_limit=t_alloc,
                                            verbose=True, label="CD")
            try_seq(a_opt, "CD")
            a_sym = 0.5 * (a_opt + a_opt[::-1])
            a_sym = np.maximum(a_sym, 0)
            a_sym = a_sym / max(a_sym.sum(), 1e-15)
            try_seq(a_sym, "CD_sym")

    # === Phase 7: Final LP polish on best ===
    print("\n=== Phase 7: Final LP Polish ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_alloc = min(remaining * 0.10, 120)
        if t_alloc > 15:
            n_b = len(best_seq)
            a_scaled = (best_seq / best_seq.sum()) * np.sqrt(2 * n_b)
            a_opt, obj = lp_optimize(a_scaled, time_limit=t_alloc, verbose=True,
                                     label="LP_final")
            a_out = a_opt / np.sqrt(2 * n_b)
            try_seq(a_out, "LP_final")
            a_sym = 0.5 * (a_out + a_out[::-1])
            a_sym = np.maximum(a_sym, 0)
            a_sym = a_sym / max(a_sym.sum(), 1e-15)
            try_seq(a_sym, "LP_final_sym")

    # === Phase 8: Multi-restart mirror descent ===
    print("\n=== Phase 8: Multi-restart MD ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        n_restarts = 3
        t_per = min(remaining * 0.10 / n_restarts, 60)
        if t_per > 15:
            for r in range(n_restarts):
                if time.monotonic() > deadline - 30:
                    break
                if r == 0:
                    f_init = best_seq / best_seq.sum()
                elif r == 1:
                    noise = rng.standard_normal(len(best_seq)) * 0.003
                    f_init = np.maximum(best_seq / best_seq.sum() + noise, 1e-15)
                    f_init = f_init / f_init.sum()
                else:
                    f_init = 0.85 * best_seq / best_seq.sum() + 0.15 * (np.ones(len(best_seq)) / len(best_seq))
                    f_init = f_init / f_init.sum()

                t_alloc = min(t_per, deadline - time.monotonic() - 30)
                if t_alloc < 10:
                    break
                a_opt, obj = mirror_descent(f_init, time_limit=t_alloc, top_k=20,
                                            temp_init=0.0002, lr_init=0.006,
                                            verbose=False, label=f"step_{r}")
                try_seq(a_opt, f"restart_{r}")

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
            print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.8f}")

    return [float(x) for x in best_seq]
```
