- Peak-targeted coordinate descent with swap operations

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """FFT-based linear auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex (sum=1, v>=0)."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_normalized(a):
    """Compute 2*n*max(conv) for normalized a (sum=1)."""
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max()


def make_sloped(n, direction=1):
    """Linearly sloped sequence (anti-palindromic structure)."""
    x = np.arange(n, dtype=np.float64)
    if direction == 1:
        a = 1.0 + 0.8 * (2.0 * x / (n - 1) - 1.0)
    else:
        a = 1.0 - 0.8 * (2.0 * x / (n - 1) - 1.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s
    return a


def make_anti_palindromic(n, eps):
    """Sequence with anti-palindromic asymmetry."""
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = 1.0 + eps * (x - center) / center
    a = np.maximum(a, 0.001)
    s = a.sum()
    a /= s
    return a


def make_exponential(n, alpha, side=0):
    """Exponential decay from one side."""
    x = np.arange(n, dtype=np.float64)
    if side == 0:
        a = np.exp(-alpha * x / n)
    else:
        a = np.exp(-alpha * (n - 1 - x) / n)
    s = a.sum()
    if s < 1e-15:

    else:

def make_gaussian(n, center_frac, sigma_frac):

    x = np.arange(n, dtype=np.float64)

    sigma = sigma_frac * n

def make_multi_freq(n, n_freqs, seed):

def subgradient_preopt(a0, time_limit, lr_init=0.02, momentum=0.9):
    """Fast multiplicative subgradient descent with momentum for pre-optimization."""
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    best_obj = obj_normalized(a)
    best_a = a.copy()
    t0 = time.monotonic()
    lr = lr_init
    velocity = np.zeros(n)
    no_improve = 0
    it = 0
    k_top = min(20, 2 * n - 1)

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()

        # Get top-k peaks for subgradient
        top_k = np.argpartition(conv, -k_top)[-k_top:]
        top_k_vals = conv[top_k]
        # Weight peaks by exponential of normalized value
        max_v = top_k_vals.max()
        if max_v < 1e-15:
            break
        weights = np.exp((top_k_vals - max_v) * 5.0)
        wsum = weights.sum()
        weights /= wsum

        # Compute subgradient
        sg = np.zeros(n)
        for idx in range(len(top_k)):
            if weights[idx] < 1e-12:
                continue
            k = int(top_k[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                sg[j] += 2.0 * weights[idx] * a[k - j]

        # Momentum update
        velocity = momentum * velocity + (1 - momentum) * sg
        # Multiplicative update
        a = a * np.exp(-lr * velocity)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive learning rate
        if no_improve > 50:
            lr *= 0.8
            if lr < 1e-6:
                # Restart with perturbation
                rng = np.random.default_rng(it)
                a = np.maximum(best_a + rng.standard_normal(n) * 0.005, 1e-15)
                a = a / a.sum()
                lr = lr_init * 0.5
                no_improve = 0

        # Decay learning rate slowly
        lr *= 0.9999

    return best_a, best_obj


def solve_lp_reduced(a, M, K):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for top-K peaks, g >= 0."""
    n = len(a)
    conv = fast_convolve(a)
    # Get top K peak indices
    if K < len(conv):
        top_K = set(np.argpartition(conv, -K)[-K:].tolist())
    else:
        top_K = set(range(len(conv)))

    # Build constraint matrix (only for top K peaks)
    constraints = []
    for k in sorted(top_K):
        row = np.zeros(n)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            row[j] = a[k - j]
        constraints.append(row)

    if not constraints:
        return np.ones(n) / n

    A_ub = np.array(constraints)
    b_ub = np.full(len(constraints), M)
    c = -np.ones(n)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={"presolve": True, "time_limit": 10})
    if res.success and res.x is not None:
        return res.x
    return None


def lp_fw_optimize(a0, time_limit, K=50, verbose=False):
    """LP Frank-Wolfe with reduced constraints and multi-level M."""
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = M
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_levels = [1.0, 1.02, 1.05, 1.1, 1.2, 1.5, 2.0, 3.0]
    m_idx = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 3.0:
            break

        conv = fast_convolve(a)
        M_cur = conv.max()

        # Try LP with escalating M levels
        g = None
        for level_try in range(m_idx, len(m_levels)):
            M_try = M_cur * m_levels[level_try]
            g = solve_lp_reduced(a, M_try, K)
            if g is not None and g.sum() > 1e-12:
                m_idx = level_try
                break
            m_idx = level_try + 1

        if g is None or g.sum() < 1e-12:
            # Perturb and restart
            rng = np.random.default_rng(it * 17 + 3)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            m_idx = 0
            no_improve += 1
            if no_improve > 40:
                break
            continue

        g = g / g.sum()

        # Check if direction is meaningful
        diff = np.max(np.abs(g - a))
        if diff < 1e-10:
            no_improve += 1
            m_idx = min(m_idx + 1, len(m_levels) - 1)
            if no_improve > 30:
                break
            continue
        no_improve = 0

        # Line search: coarse grid
        best_alpha = 0.0
        best_val = M_cur
        alphas = [0.01, 0.03, 0.06, 0.1, 0.15, 0.25, 0.4, 0.6, 0.8, 1.0]
        for alpha in alphas:
            a_t = np.maximum(a + alpha * (g - a), 0.0)
            st = a_t.sum()
            if st < 1e-15:
                continue
            a_t = a_t / st
            val = fast_convolve(a_t).max()
            if val < best_val - 1e-15:
                best_val = val
                best_alpha = alpha

        # Fine line search around best
        if best_alpha > 0:
            if best_alpha == alphas[0]:
                lo, hi = 0.0, alphas[1]
            elif best_alpha == alphas[-1]:
                lo, hi = alphas[-2], 1.0
            else:
                idx = alphas.index(best_alpha)
                lo = alphas[idx - 1]
                hi = alphas[idx + 1]

            for _ in range(15):
                a1 = lo + 0.382 * (hi - lo)
                a2 = lo + 0.618 * (hi - lo)
                ft1 = np.maximum(a + a1 * (g - a), 0.0)
                ft2 = np.maximum(a + a2 * (g - a), 0.0)
                ft1 = ft1 / ft1.sum()
                ft2 = ft2 / ft2.sum()
                v1 = fast_convolve(ft1).max()
                v2 = fast_convolve(ft2).max()
                if v1 < v2:
                    hi = a2
                else:
                    lo = a1
                mid = (lo + hi) / 2.0
                ftm = np.maximum(a + mid * (g - a), 0.0)
                ftm = ftm / ftm.sum()
                vm = fast_convolve(ftm).max()
                if vm < best_val:
                    best_val = vm
                    best_alpha = mid

        if best_alpha < 1e-10:
            no_improve += 1
            m_idx = min(m_idx + 1, len(m_levels) - 1)
            continue

        a = np.maximum(a + best_alpha * (g - a), 0.0)
        a = a / a.sum()
        conv = fast_convolve(a)
        M = conv.max()
        m_idx = 0

        if M < best_obj - 1e-14:
            best_obj = M
            best_a = a.copy()
            if verbose and it % 10 == 0:
                print(f"    LP-FW iter {it}: obj={best_obj:.10f}")

    return best_a, best_obj


def peak_coordinate_descent(a0, time_limit, verbose=False):
    """Peak-targeted coordinate descent with mass redistribution and swaps."""
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > 80:
            # Try perturbation
            rng = np.random.default_rng(int(time.monotonic() * 1000))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            no_improve = 0
            if not np.isfinite(obj_normalized(a)):
                break

        conv = fast_convolve(a)
        M = conv.max()
        peak_k = int(np.argmax(conv))

        # Find contributing indices
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        contrib = np.arange(js, je + 1)
        if len(contrib) == 0:
            break

        products = a[contrib] * a[peak_k - contrib]
        total = products.sum()
        if total < 1e-15:
            break

        # Target the most contributing pair
        max_idx_local = np.argmax(products)
        src = contrib[max_idx_local]
        partner = peak_k - src

        # Strategy 1: Reduce source, distribute to non-contributing
        non_contrib = np.ones(n, dtype=bool)
        non_contrib[contrib] = False
        if non_contrib.sum() < 1:
            non_contrib[:] = True
            non_contrib[src] = False

        best_trial = None
        best_trial_obj = M

        for frac in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            diff = a[src] * frac
            if diff < 1e-15:
                continue
            trial = a.copy()
            trial[src] -= diff
            nc_vals = trial[non_contrib]
            nc_sum = nc_vals.sum()
            if nc_sum < 1e-15:
                continue
            trial[non_contrib] += diff * nc_vals / nc_sum
            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                continue
            trial = trial / ts
            obj = fast_convolve(trial).max()
            if obj < best_trial_obj - 1e-15:
                best_trial_obj = obj
                best_trial = trial

        # Strategy 2: Swap between two non-adjacent coordinates
        if best_trial is None or best_trial_obj >= M - 1e-15:
            # Find a coordinate with high value and a far-away coordinate with low value
            if n > 10:
                # Pick source from top quarter, target from bottom quarter
                sorted_idx = np.argsort(a)
                target_pool = sorted_idx[:n // 4]
                source_pool = sorted_idx[-n // 4:]

                for _ in range(5):
                    si = source_pool[np.random.randint(len(source_pool))]
                    ti = target_pool[np.random.randint(len(target_pool))]
                    if si == ti:
                        continue
                    for frac in [0.1, 0.3, 0.5]:
                        diff = min(a[si] * frac, a[ti] * frac * 3)
                        if diff < 1e-15:
                            continue
                        trial = a.copy()
                        trial[si] -= diff
                        trial[ti] += diff
                        trial = np.maximum(trial, 0.0)
                        trial = trial / trial.sum()
                        obj = fast_convolve(trial).max()
                        if obj < best_trial_obj - 1e-15:
                            best_trial_obj = obj
                            best_trial = trial

        if best_trial is not None and best_trial_obj < M - 1e-15:
            a = best_trial
            if best_trial_obj < best_obj - 1e-14:
                best_obj = best_trial_obj
                best_a = a.copy()
                no_improve = 0
                if verbose and no_improve == 0 and (int(time.monotonic() * 100) % 7 == 0):
                    print(f"    PCD: peak_k={peak_k}, obj={best_obj:.10f}")
        else:
            no_improve += 1

        # Also try targeting second peak
        if no_improve % 10 == 5:
            conv2 = fast_convolve(a)
            # Mask the top peak and find second
            conv2_masked = conv2.copy()
            conv2_masked[peak_k] = -np.inf
            peak_k2 = int(np.argmax(conv2_masked))
            if peak_k2 != peak_k:
                js2 = max(0, peak_k2 - n + 1)
                je2 = min(n - 1, peak_k2)
                contrib2 = np.arange(js2, je2 + 1)
                if len(contrib2) > 0:
                    products2 = a[contrib2] * a[peak_k2 - contrib2]
                    src2 = contrib2[np.argmax(products2)]
                    nc2 = np.ones(n, dtype=bool)
                    nc2[contrib2] = False
                    if nc2.sum() > 0:
                        for frac in [0.1, 0.2, 0.3]:
                            diff2 = a[src2] * frac
                            trial = a.copy()
                            trial[src2] -= diff2
                            nc2_vals = trial[nc2]
                            nc2_sum = nc2_vals.sum()
                            if nc2_sum > 1e-15:
                                trial[nc2] += diff2 * nc2_vals / nc2_sum
                                trial = np.maximum(trial, 0.0)
                                trial = trial / trial.sum()
                                obj2 = fast_convolve(trial).max()
                                if obj2 < best_obj - 1e-14:
                                    a = trial
                                    best_obj = obj2
                                    best_a = a.copy()
                                    no_improve = 0
                                    break

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Linear interpolation of sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def two_phase_optimize(a0, time_limit, verbose=False):
    """Two-phase: fast subgradient pre-opt then LP-FW polish."""
    # Phase A: Subgradient pre-optimization (fast)
    t0 = time.monotonic()
    pre_time = min(60, time_limit * 0.3)
    a_pre, obj_pre = subgradient_preopt(a0, time_limit=pre_time)
    t_pre = time.monotonic() - t0
    if verbose:
        print(f"  Pre-opt: {t_pre:.1f}s, obj={obj_pre:.10f}")

    # Phase B: LP-FW (slower but more precise)
    rem = time_limit - (time.monotonic() - t0) - 5
    if rem > 10:
        a_final, obj_final = lp_fw_optimize(a_pre, time_limit=rem, K=50, verbose=verbose)
        if verbose:
            print(f"  LP-FW: obj={obj_final:.10f}")
    else:
        a_final, obj_final = a_pre, obj_pre

    return a_final, obj_final


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization for minimum peak autocorrelation sequences.
    Combines fast subgradient pre-optimization, LP-Frank-Wolfe,
    peak-targeted coordinate descent, and multi-length exploration.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Try to load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_sloped(768, direction=1)
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default parent: n={len(parent)}, score={p_score:.6f}")
        best_score = p_score
        best_seq = parent.copy()

    def try_and_update(a, label):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # === Phase 1: Diverse initializations ===
    print("\n=== Phase 1: Diverse Initializations ===")
    t0 = time.monotonic()
    n_vals = [512, 640, 768, 896, 1024]

    inits = []
    for n_try in n_vals:
        inits.append(('sloped_up', n_try, make_sloped(n_try, 1)))
        inits.append(('sloped_dn', n_try, make_sloped(n_try, -1)))
        inits.append(('anti_pal_03', n_try, make_anti_palindromic(n_try, 0.3)))
        inits.append(('anti_pal_05', n_try, make_anti_palindromic(n_try, 0.5)))
        inits.append(('exp_1', n_try, make_exponential(n_try, 1.0, 0)))
        inits.append(('exp_2', n_try, make_exponential(n_try, 2.0, 0)))
        inits.append(('exp_3', n_try, make_exponential(n_try, 3.0, 1)))
        inits.append(('gauss_center', n_try, make_gaussian(n_try, 0.5, 0.35)))
        inits.append(('gauss_off', n_try, make_gaussian(n_try, 0.3, 0.3)))
        inits.append(('multifreq', n_try, make_multi_freq(n_try, 4, seed=n_try)))
        inits.append(('random', n_try, make_random_pos(n_try, seed=n_try + 7)))
        inits.append(('flat', n_try, np.ones(n_try) / n_try))

    # Add parent
    if parent is not None:
        inits.append(('parent', len(parent), parent))

    # Score all initializations
    scored = []
    for label, n_try, a_init in inits:
        sc = obj_normalized(a_init)
        scored.append((sc, a_init, label))
        if sc < 2.0:
            try_and_update(a_init, f"init_{label}_{n_try}")

    scored.sort(key=lambda x: x[0])
    print(f"  Best init: {scored[0][2]}, obj={scored[0][0]:.6f}")
    print(f"  Phase 1 time: {time.monotonic()-t0:.1f}s")

    # === Phase 2: Two-phase optimization on top initializations ===
    print("\n=== Phase 2: Two-Phase Optimization ===")
    top_k_inits = min(5, len(scored))
    for trial_idx in range(top_k_inits):
        rem = deadline - time.monotonic() - 200
        if rem < 60:
            break
        sc_init, a_init, label = scored[trial_idx]
        # Allocate time: more time for better initializations
        t_alloc = min(rem * 0.45, 180)
        if t_alloc < 30:
            break
        print(f"  TwoPhase [{label}] n={len(a_init)}: alloc={t_alloc:.0f}s (init={sc_init:.6f})")
        a_opt, obj_opt = two_phase_optimize(a_init, time_limit=t_alloc, verbose=True)
        try_and_update(a_opt, f"2phase_{label}")

    # === Phase 3: LP-FW directly on current best ===
    print("\n=== Phase 3: LP-FW on Best ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_alloc = min(rem, 200)
            print(f"  LP-FW: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, obj_opt = lp_fw_optimize(best_seq, time_limit=t_alloc, K=50, verbose=True)
            try_and_update(a_opt, "lpfw_best")

    # === Phase 4: Peak coordinate descent ===
    print("\n=== Phase 4: Peak Coordinate Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_alloc = min(rem, 120)
            print(f"  PCD: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, obj_opt = peak_coordinate_descent(best_seq, time_limit=t_alloc, verbose=True)
            try_and_update(a_opt, "pcd")

    # === Phase 5: Multi-length exploration ===
    print("\n=== Phase 5: Multi-Length Exploration ===")
    if best_seq is not None:
        base_n = len(best_seq)
        target_ns = [512, 640, 768, 896, 1024, 1280]
        for target_n in target_ns:
            if target_n == base_n:
                continue
            rem = deadline - time.monotonic() - 80
            if rem < 15:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            sc = try_and_update(f_interp, f"interp_n{target_n}")
            if sc < best_score + 0.03:
                t_alloc = min(rem * 0.35, 80)
                if t_alloc > 10:
                    print(f"  Optimize n={target_n}: alloc={t_alloc:.0f}s")
                    a_opt, obj_opt = two_phase_optimize(f_interp, time_limit=t_alloc)
                    try_and_update(a_opt, f"opt_interp_n{target_n}")

    # === Phase 6: Perturbation-based exploration ===
    print("\n=== Phase 6: Perturbation Exploration ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rng = np.random.default_rng(seed)

        perturbations = []
        # Small Gaussian noise
        for amp in [0.001, 0.003, 0.005, 0.01, 0.02]:
            noise = rng.standard_normal(n_best) * amp
            perturbations.append((f'noise{amp}', np.maximum(best_seq + noise, 0)))

        # Shift
        for shift in [1, 2, 3, 5]:
            perturbations.append((f'shift{shift}', np.roll(best_seq, shift)))

        # Blend with different templates
        for template in [make_sloped(n_best, 1), make_sloped(n_best, -1),
                         make_exponential(n_best, 2.0, 0)]:
            for mix in [0.1, 0.2, 0.3]:
                blend = (1 - mix) * best_seq + mix * template
                perturbations.append((f'blend{mix}', blend))

        for label, f_pert in perturbations:
            rem = deadline - time.monotonic() - 40
            if rem < 10:
                break
            f_pert = project_simplex(f_pert)
            # Quick evaluation
            sc_quick = obj_normalized(f_pert)
            if sc_quick < best_score / (2.0 * len(f_pert)) + 0.05:
                t_alloc = min(rem * 0.25, 40)
                if t_alloc > 5:
                    a_opt, obj_opt = two_phase_optimize(f_pert, time_limit=t_alloc)
                    try_and_update(a_opt, f"pert_{label}")

    # === Phase 7: Final polish ===
    print("\n=== Phase 7: Final Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 15:
            # Quick LP-FW squeeze
            t_alloc = min(rem * 0.5, 30)
            print(f"  Final LP-FW: alloc={t_alloc:.0f}s")
            a_opt, obj_opt = lp_fw_optimize(best_seq, time_limit=t_alloc, K=30)
            try_and_update(a_opt, "final_lp")

        rem = deadline - time.monotonic() - 5
        if rem > 10:
            # Quick PCD polish
            t_alloc = min(rem, 15)
            print(f"  Final PCD: alloc={t_alloc:.0f}s")
            a_opt, obj_opt = peak_coordinate_descent(best_seq, time_limit=t_alloc)
            try_and_update(a_opt, "final_pcd")

    # === Fallback ===
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    total_time = time.monotonic() - (deadline - max(1.0, float(budget_s) - 30.0))
    print(f"Total time: {total_time:.1f}s")

    return [float(x) for x in best_seq]
