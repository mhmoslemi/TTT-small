"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

2. Power-smooth subgradient optimization with Adam

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def power_smooth_grad(f, p, tau):
    """
    Compute gradient of (sum_k (conv[k]/tau + eps)^p)^(1/p).
    Returns (gradient, objective_value).
    """
    n = len(f)
    conv = fast_convolve(f)
    base = conv / tau + 1e-15
    obj = (np.sum(base ** p)) ** (1.0 / p) * tau  # Scale back
    
    # Weight vector
    w = (base ** (p - 1)) / tau  # d(conv[k]) factor
    
    # grad[m] = scale * (2 * CC[m] - diag_correction[m])
    # where CC[m] = sum_j w[j+m] * f[j] = (w * f_flip)[n-1+m]
    a_flip = f[::-1]
    total_len = len(w) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    AF = np.fft.rfft(a_flip, fft_size)
    conv_result = np.fft.irfft(W * AF, fft_size)
    
    CC = conv_result[n - 1 : 2 * n - 1]  # CC[m] for m = 0, ..., n-1
    
    # Diagonal correction: w[2m] * f[m] for 2m <= 2n-2
    grad = 2.0 * CC
    # For m where 2m < len(w) = 2n-1, i.e., m <= n-1:
    two_m = 2 * np.arange(n)
    valid = two_m < len(w)
    grad[valid] -= w[two_m[valid]] * f[valid]
    
    # Outer scale factor
    scale = (np.sum(base ** p)) ** (1.0 / p - 1.0)
    grad *= scale
    
    return grad, obj

def adam_power_optimize(f0, time_limit, p=4, tau=None, lr_init=0.005, verbose=False):
    """
    Adam optimization of the power-smooth objective.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    
    conv = fast_convolve(f)
    if tau is None:
        tau = 0.05 * conv.max()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    lr = lr_init
    t = 0
    
    best_obj = conv.max()
    best_f = f.copy()
    no_improve = 0
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        grad, smooth_obj = power_smooth_grad(f, p, tau)
        
        # Also track the actual max-conv objective
        conv = fast_convolve(f)
        M = conv.max()
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)
        
        conv_new = fast_convolve(f_new)
        M_new = conv_new.max()
        
        if M_new < best_obj - 1e-15:
            best_obj = M_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        f = f_new
        
        # Adaptive: decay tau (sharpen the approximation)
        if t % 100 == 0:
            tau *= 0.95
            tau = max(tau, 1e-8 * conv_new.max())
        
        # Adaptive learning rate
        if no_improve > 50:
            lr *= 0.8
        if no_improve > 200:
            # Restart with perturbation
            rng = np.random.default_rng(t * 7 + 3)
            noise = 0.005 + 0.01 * min(no_improve / 500.0, 2.0)
            f = project_simplex(best_f + rng.standard_normal(n) * noise)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = lr_init
            conv_r = fast_convolve(f)
            tau = 0.05 * conv_r.max()
            
        if verbose and t % 200 == 0:
            print(f"    Adam iter {t}: 2n*max={2*n*best_obj:.8f}, tau={tau:.2e}, lr={lr:.5e}")
    
    return best_f, 2 * n * best_obj

def peak_cd(f0, time_limit, verbose=False):
    """
    Peak-targeted coordinate descent.
    For each peak, find the coordinate that contributes most and reduce it,
    redistributing to "safe" coordinates.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        improved = False
        
        # Get the top peaks
        n_peaks = min(10, 2 * n - 1)
        peak_idx = np.argsort(conv)[-n_peaks:]
        
        for pk in sorted(peak_idx, reverse=True):
            pk = int(pk)
            # Find the coordinate that contributes most to this peak
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            products = f[idx] * f[pk - idx]
            j_max = idx[np.argmax(products)]
            
            # Try reducing f[j_max] by various amounts
            old_val = f[j_max]
            if old_val < 1e-8:
                continue
            
            for frac in [0.3, 0.5, 0.7]:
                new_val = old_val * (1 - frac)
                diff = old_val - new_val
                
                trial = f.copy()
                trial[j_max] = new_val
                
                # Redistribute to coordinates that DON'T contribute to any peak
                # (i.e., coordinates where the "opposite" position has low f value)
                safe = np.zeros(n)
                for i in range(n):
                    if i == j_max:
                        continue
                    # Check if i contributes to any peak
                    contributes = False
                    for pk2 in peak_idx[:5]:
                        opposite = int(pk2) - i
                        if 0 <= opposite < n and f[opposite] > 0.01 * f.max():
                            contributes = True
                            break
                    if not contributes:
                        safe[i] = f[i]
                
                if safe.sum() > 1e-10:
                    trial += diff * safe / safe.sum()
                    trial = np.maximum(trial, 0)
                    trial /= trial.sum()
                    
                    obj = fast_convolve(trial).max()
                    if obj < best_obj - 1e-14:
                        f = trial
                        best_obj = obj
                        best_f = f.copy()
                        improved = True
                        break
        
        if not improved:
            no_improve += 1
            if no_improve > 20:
                # Perturb
                rng = np.random.default_rng(it * 13)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                no_improve = 0
                if time.monotonic() - t0 > time_limit * 0.8:
                    break
        else:
            no_improve = 0
        
        if verbose and it % 50 == 0:
            print(f"    Peak CD iter {it}: 2n*max={2*n*best_obj:.8f}")
    
    return best_f, 2 * n * best_obj

def fine_grad_descent(f0, time_limit, verbose=False):
    """
    Fine gradient descent using the actual subgradient of max(conv).
    The subgradient of max_k conv[k] at a is: sum_k alpha_k * d(conv[k])/da
    where alpha_k are the barycentric coordinates of the max (uniform over argmax).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0
    lr = 0.01
    decay = 0.999
    smoothing = 100  # number of peaks to "soften"
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        # Compute subgradient: weighted by proximity to max
        # Use a soft version: weights proportional to exp((conv - M) / sigma)
        sigma = 0.1 * M + 1e-15
        w = np.exp((conv - M) / sigma)
        w /= w.sum()
        
        # Gradient of sum_k w[k] * conv[k]
        # = sum_k w[k] * d(conv[k])/da
        # d(conv[k])/da[m] = 2*a[k-m] - a[m]*delta[k,2m]
        a_flip = f[::-1]
        total_len = len(w) + n - 1
        fft_size = 1 << (total_len - 1).bit_length()
        W = np.fft.rfft(w, fft_size)
        AF = np.fft.rfft(a_flip, fft_size)
        conv_result = np.fft.irfft(W * AF, fft_size)
        
        CC = conv_result[n - 1 : 2 * n - 1]
        grad = 2.0 * CC
        two_m = 2 * np.arange(n)
        valid = two_m < len(w)
        grad[valid] -= w[two_m[valid]] * f[valid]
        
        # Step
        f_new = f - lr * grad
        f_new = project_simplex(f_new)
        
        conv_new = fast_convolve(f_new)
        M_new = conv_new.max()
        
        if M_new < best_obj - 1e-15:
            best_obj = M_new
            best_f = f_new.copy()
        
        f = f_new
        lr *= decay
        
        if verbose and it % 100 == 0:
            print(f"    FineGD iter {it}: 2n*max={2*n*best_obj:.8f}, lr={lr:.5e}")
    
    return best_f, 2 * n * best_obj

def make_blackman(n):

def make_flat_topped(n, width=0.6):

def make_comb(n, teeth, width=1):
    """Comb structure: teeth blocks of given width."""
    a = np.zeros(n)
    spacing = n // teeth
    for i in range(teeth):
        start = i * spacing
        end = min(start + width, n)
        a[start:end] = 1.0
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def make_ridged(n, base_sigma=0.15, ridge_count=3):
    """Multi-ridge structure."""
    a = np.zeros(n)
    x = np.arange(n) / n
    for i in range(ridge_count):
        center = (i + 0.5) / ridge_count
        a += np.exp(-0.5 * ((x - center) / base_sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    result = np.interp(new_x, old_x, f)
    result = np.maximum(result, 0)
    s = result.sum()
    if s > 1e-15:
        result /= s
    return result

def two_coordinate_swap(f, i, j, alpha):
    """
    Move mass from coordinate i to j: f[i] -= alpha*f[i], f[j] += alpha*f[i].
    Returns the new sequence (already on simplex).
    """
    trial = f.copy()
    delta = alpha * f[i]
    trial[i] -= delta
    trial[j] += delta
    trial = np.maximum(trial, 0)
    s = trial.sum()
    if s > 1e-15:
        trial /= s
    return trial

def coordinate_descent(f0
