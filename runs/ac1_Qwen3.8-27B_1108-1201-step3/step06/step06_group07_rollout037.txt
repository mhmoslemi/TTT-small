def spectral_cap_update(a, cap_factor=0.98, newton_iters=1, target_integral=1.0):
    a = np.maximum(a,0); s=a.sum(); if s==0 return None; a=a/s
    n=len(a); size=2*n-1; L=1<<(size-1).bit_length()
    ap=np.zeros(L); ap[:n]=a
    A=np.fft.fft(ap)
    C=A*A
    c_full=np.fft.ifft(C).real
    c=c_full[:size]
    # clean small negative
    c=np.maximum(c,0)
    current_max=c.max()
    target_cap=current_max*cap_factor
    avg=target_integral/size
    if target_cap < avg: target_cap=avg*1.001
    c_target=c.copy()
    mask=c>target_cap
    if mask.sum()==0: return a, current_max
    c_target[mask]=target_cap
    deficit = target_integral - c_target.sum()
    if deficit > 1e-12:
       slack = np.maximum(target_cap - c_target,0)
       if slack.sum() > 1e-15:
          c_target += deficit*slack/slack.sum()
       else:
          c_target += deficit/size
    elif deficit < -1e-12:
       # target_cap above some? remove from non-mask maybe
       over = np.maximum(c_target,0)
       c_target -= (-deficit)*over/over.sum()
    c_target=np.maximum(c_target,0)
    Ct=np.fft.fft(c_target, L)
    # Newton
    A_cur=A
    for _ in range(newton_iters):
       # where abs(A_cur) small
       denom=np.where(np.abs(A_cur)>1e-300, A_cur, 1e-300+0j)
       A_next=0.5*(A_cur + Ct/denom)
       A_cur=A_next
    a_new=np.fft.ifft(A_cur).real[:n]
    a_new=np.maximum(a_new,0)
    ss=a_new.sum()
    if ss < 1e-10: return None
    a_new/=ss
    # after normalization, conv max? return
    conv=fast_convolve(a_new)
    return a_new, float(conv.max())

"""
Hybrid search for low-peak non-negative step-function coefficients.

Strategy:
- Keep a global archive of good sequences of different lengths.
- Use a fast "spectral waterfilling" step:
  * pad to a zero-free circular length,
  * compute the circular/linear convolution c = a*a,
  * cap c at a target peak level,
  * redistribute the removed mass into slack lags,
  * move a toward the spectral square root of the capped target,
  * project back to non-negative support.
- Use log-domain Adam on a softmax parameterization to polish smooth
  log-sum-exp approximations of the max-convolution objective.
- Use small linear-programming direction moves (minimize max cross-convolution
  with sum equal to one) when scipy/HiGHS is available.
- Use L-BFGS-B polish on smoothed surrogates.
- Blend diverse archive members to reduce peaks by convex averaging.

No closures or lambdas are used. evaluate_sequence is assumed to be available
in the global namespace.
"""

import time
import numpy as np

try:
    from scipy.optimize import linprog, minimize

    HAVE_SCIPY = True
except Exception:
    HAVE_SCIPY = False


_FFT_CACHE = {}
_GRAD_CACHE = {}
_LP_CACHE = {}


def _next_pow2(x):
    if x <= 1:
        return 1
    return 1 << (int(x) - 1).bit_length()


def get_fft_size(n):
    val = _FFT_CACHE.get(n)
    if val is None:
        val = _next_pow2(2 * n - 1)
        _FFT_CACHE[n] = val
    return val


def get_grad_size(n):
    val = _GRAD_CACHE.get(n)
    if val is None:
        val = _next_pow2(3 * n - 2)
        _GRAD_CACHE[n] = val
    return val


def get_lp_pattern(n):
    val = _LP_CACHE.get(n)
    if val is None:
        rows = (np.arange(n, dtype=np.int64)[:, None] +
                np.arange(n, dtype=np.int64)[None, :]).ravel()
        cols = np.repeat(np.arange(n, dtype=np.int64), n)
        val = (rows, cols)
        _LP_CACHE[n] = val
    return val


def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    L = get_fft_size(n)
    A = np.fft.rfft(a, L)
    return np.fft.irfft(A * A, L)[:size]


def normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.array([1.0], dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 1e-300:
        return np.ones(a.shape[0], dtype=np.float64) / a.shape[0]
    return a / s


def local_score(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float("inf")
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 1e-300:
        return float("inf")
    c = fast_convolve(a)
    if c.size == 0:
        return float("inf")
    return float(2.0 * a.size * c.max() / (s * s))


def symmetrize(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return a
    a = np.where(np.isfinite(a), a, 0.0)
    a = 0.5 * (a + a[::-1])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 1e-300:
        return np.ones(a.shape[0], dtype=np.float64) / a.shape[0]
    return a / s


def interpolate_sequence(f, new_n):
    f = np.asarray(f, dtype=np.float64)
    old_n = f.size
    if new_n <= 0:
        return np.array([1.0], dtype=np.float64)
    if old_n == 0:
        return np.ones(new_n, dtype=np.float64) / new_n
    if old_n == 1:
        return np.ones(new_n, dtype=np.float64) / new_n
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    out = np.interp(new_x, old_x, f)
    return normalize_nonneg(out)


def make_gaussian(n, sigma_frac=0.30):
    if n <= 0:
        return np.array([1.0])
    x = np.arange(n, dtype=np.float64) - 0.5 * n
    sigma = max(1.0, n * sigma_frac)
    a = np.exp(-0.5 * (x / sigma) ** 2)
    return normalize_nonneg(a)


def make_bell(n, width_frac=0.60):
    if n <= 0:
        return np.array([1.0])
    x = np.arange(n, dtype=np.float64) - 0.5 * n
    half = max(1.0, 0.5 * n * width_frac)
    t = np.clip(np.abs(x) / half, 0.0, 1.0)
    a = 1.0 - t ** 4
    return normalize_nonneg(a)


def make_bimodal(n, c1=0.28, c2=0.72, sigma=0.08):
    if n <= 0:
        return np.array([1.0])
    x = np.arange(n, dtype=np.float64) / float(n)
    a = (np.exp(-0.5 * ((x - c1) / sigma) ** 2) +
         np.exp(-0.5 * ((x - c2) / sigma) ** 2))
    return normalize_nonneg(a)


def make_split(n):
    if n <= 1:
        return np.ones(max(1, n), dtype=np.float64)
    third = max(1, n // 3)
    a = np.zeros(n, dtype=np.float64)
    a[:third] = 1.0
    a[-third:] = 1.0
    return normalize_nonneg(a)


def make_modulated(n, freq=1.0, amp=0.25):
    if n <= 0:
        return np.array([1.0])
    x = np.linspace(0.0, 1.0, n)
    a = (1.0
         + amp * np.cos(2.0 * np.pi * freq * x)
         + 0.5 * amp * np.cos(4.0 * np.pi * freq * x))
    a = np.maximum(a, 0.0)
    return normalize_nonneg(a)


def make_random_dirichlet(n, seed):
    if n <= 0:
        return np.array([1.0])
    rng = np.random.default_rng(seed)
    a = rng.dirichlet(np.ones(n, dtype=np.float64))
    return normalize_nonneg(a)


def smooth_sequence(a, sigma_frac=0.03):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n <= 3:
        return normalize_nonneg(a)
    sigma = float(n) * sigma_frac
    radius = int(max(1, 4 * sigma))
    x = np.arange(-radius, radius + 1, dtype=np.float64)
    k = np.exp(-0.5 * (x / sigma) ** 2)
    k = k / k.sum()
    out = np.convolve(a, k, mode="same")
    return normalize_nonneg(out)


def update_archive(archive, a, label):
    an = normalize_nonneg(a)
    sc = local_score(an)
    if not np.isfinite(sc):
        return archive

    replaced = False
    for i in range(len(archive)):
        if archive[i][1] == label:
            if sc < archive[i][0]:
                archive[i] = (sc, label, an.copy())
            replaced = True
            break

    if not replaced:
        archive.append((sc, label, an.copy()))

    if len(archive) > 30:
        scores = [e[0] for e in archive]
        order = np.argsort(scores)
        new_archive = []
        for idx in order[:30]:
            new_archive.append(archive[int(idx)])
        archive = new_archive
    return archive


def select_entries(entries, k, unique_lengths=False):
    if not entries or k <= 0:
        return []
    scores = [e[0] for e in entries]
    order = np.argsort(scores)
    selected = []

    if unique_lengths:
        used_lengths = set()
        for idx in order:
            e = entries[int(idx)]
            n = len(e[2])
            if n in used_lengths:
                continue
            used_lengths.add(n)
            selected.append(e)
            if len(selected) >= k:
                break

    if len(selected) < k:
        used_labels = set()
        for e in selected:
            used_labels.add(e[1])
        for idx in order:
            e = entries[int(idx)]
            if e[1] in used_labels:
                continue
            selected.append(e)
            used_labels.add(e[1])
            if len(selected) >= k:
                break
    return selected


def softmax_u(u):
    u = np.asarray(u, dtype=np.float64)
    if u.size == 0:
        return np.array([1.0], dtype=np.float64)
    m = 0.0
    if u.size > 0:
        umax = float(u.max())
        if umax > 0.0:
            m = umax
    exp_u = np.exp(u - m)
    last = np.exp(-m)
    D = float(exp_u.sum() + last)
    if D <= 1e-300:
        out = np.ones(u.size + 1, dtype=np.float64)
        out[-1] = 1.0
        return out
    out = np.empty(u.size + 1, dtype=np.float64)
    out[:-1] = exp_u / D
    out[-1] = last / D
    return out


def init_u(a):
    a = normalize_nonng(a) if False else normalize_nonneg(a)
    n = a.size
    if n == 1:
        return np.array([], dtype=np.float64)
    # Prefer a representation with a reasonably large last coordinate.
    if a[-1] < a[0]:
        a = a[::-1]
    eps = 1e-15
    af = np.maximum(a, eps)
    af = af / af.sum()
    u = np.log(af[:-1] / af[-1])
    return np.clip(u, -40.0, 40.0)


def compute_weights_from_conv(conv, tau):
    conv = np.asarray(conv, dtype=np.float64)
    if conv.size == 0:
        return conv
    m = float(conv.max())
    if not np.isfinite(m):
        w = np.zeros_like(conv)
        w[0] = 1.0
        return w
    if tau <= 1e-14:
        w = np.zeros_like(conv)
        idx = int(np.argmax(conv))
        w[idx] = 1.0
        return w
    e = np.exp((conv - m) / tau)
    s = float(e.sum())
    if s <= 1e-300:
        w = np.zeros_like(conv)
        idx = int(np.argmax(conv))
        w[idx] = 1.0
    else:
        w = e / s
    return w


def smooth_conv_value(conv, tau):
    conv = np.asarray(conv, dtype=np.float64)
    if conv.size == 0:
        return 0.0
    m = float(conv.max())
    if tau <= 1e-14:
        return m
    e = np.exp((conv - m) / tau)
    s = float(e.sum())
    if s <= 1e-300:
        return m
    return float(tau * (m + np.log(s)))


def grad_a_from_weights(w, a):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if n <= 0:
        return np.array([], dtype=np.float64)
    L = get_grad_size(n)
    ar = a[::-1]
    FW = np.fft.rfft(w, L)
    FA = np.fft.rfft(ar, L)
    g = np.fft.irfft(FW * FA, L)[n - 1:n - 1 + n]
    return 2.0 * g


def gradient_u_from_data(u, a, conv, tau):
    n = a.size
    if n <= 1:
        return np.zeros(0, dtype=np.float64)
    w = compute_weights_from_conv(conv, tau)
    g = grad_a_from_weights(w, a)
    S = float(np.dot(a[:-1], g[:-1]))
    return a[:-1] * (g[:-1] - S)


def objective_u(u, tau):
    u = np.asarray(u, dtype=np.float64)
    a = softmax_u(u)
    conv = fast_convolve(a)
    return smooth_conv_value(conv, tau)


def gradient_u(u, tau):
    u = np.asarray(u, dtype=np.float64)
    a = softmax_u(u)
    conv = fast_convolve(a)
    return gradient_u_from_data(u, a, conv, tau)


def spectral_cap_update_from_cap(a, cap, newton_iters=1):
    an = normalize_nonneg(a)
    n = an.size
    if n <= 1:
        return None, local_score(an)
    size = 2 * n - 1
    L = _next_pow2(size)

    ap = np.zeros(L, dtype=np.float64)
    ap[:n] = an
    A = np.fft.fft(ap)
    C = A * A
    c = np.fft.ifft(C).real[:size]
    c = np.maximum(c, 0.0)

    curr_max = float(c.max())
    score = float(2.0 * n * curr_max)
    if not np.isfinite(score):
        return None, float("inf")
    if cap >= curr_max * 0.9999:
        return None, score

    avg = 1.0 / float(size)
    if cap < avg * 1.005:
        cap = avg * 1.005

    ct = c.copy()
    mask = ct > cap
    if not mask.any():
        return None, score
    ct[mask] = cap

    deficit = 1.0 - float(ct.sum())
    if deficit > 1e-12:
        slack = np.maximum(cap - ct, 0.0)
        slack_sum = float(slack.sum())
        if slack_sum > 1e-15:
            ct += deficit * (slack / slack_sum)
        else:
            ct += deficit / float(size)
    elif deficit < -1e-12:
        over = ct.copy()
        over_sum = float(over.sum())
        if over_sum > 1e-15:
            ct += deficit * (over / over_sum)
    ct = np.maximum(ct, 0.0)
    ct_sum = float(ct.sum())
    if ct_sum <= 1e-300:
        return None, score
    ct = ct / ct_sum

    Ct = np.fft.fft(ct, L)

    A_cur = A.astype(np.complex128).copy()
    A_cur[0] = 1.0 + 0.0j

    for _ in range(int(max(1, newton_iters))):
        mask = np.abs(A_cur) > 1e-30
        quot = np.zeros_like(A_cur, dtype=np.complex128)
        np.divide(Ct, A_cur, out=quot, where=mask)
        A_cur = np.where(mask, 0.5 * (A_cur + quot), A_cur)

        partner = (-np.arange(L)) % L
        A_cur = 0.5 * (A_cur + np.conjugate(A_cur[partner]))
        A_cur[0] = 1.0 + 0.0j

    a_new = np.fft.ifft(A_cur).real[:n]
    a_new = np.maximum(a_new, 0.0)
    ss = float(a_new.sum())
    if ss <= 1e-100:
        return None, score
    a_new = a_new / ss
    sc = local_score(a_new)
    if not np.isfinite(sc):
        return None, score
    return a_new, sc


def spectral_improve(a, time_limit, verbose=False):
    a = normalize_nonneg(a)
    if a.size == 0 or time_limit <= 0:
        return a

    n = a.size
    end = time.monotonic() + float(time_limit)
    best_a = a.copy()
    best_score = local_score(a)

    factors = [0.9995, 0.999, 0.995, 0.99, 0.985, 0.98, 0.97, 0.96, 0.94, 0.90]
    no_improve = 0
    it = 0

    while time.monotonic() < end:
        it += 1
        rng = np.random.default_rng(1000003 + 31 * it)

        cur_score = local_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0

        base_cap = best_score / (2.0 * float(n))
        improved = False

        for fac in factors:
            cap = base_cap * fac
            try:
                new_a, new_sc = spectral_cap_update_from_cap(a, cap, newton_iters=1)
            except Exception:
                new_a, new_sc = None, float("inf")
            if new_a is None:
                continue
            if new_sc < best_score - 1e-14:
                a = new_a
                best_score = new_sc
                best_a = a.copy()
                improved = True
                break
            if new_sc < cur_score - 1e-12:
                a = new_a
                improved = True
                break

            s_sym = symmetrize(new_a)
            sc_sym = local_score(s_sym)
            if np.isfinite(sc_sym) and sc_sym < best_score - 1e-14:
                a = s_sym
                best_score = sc_sym
                best_a = a.copy()
                improved = True
                break

        if not improved:
            cap = base_cap * 0.96
            try:
                new_a, new_sc = spectral_cap_update_from_cap(a, cap, newton_iters=2)
            except Exception:
                new_a, new_sc = None, float("inf")
            if new_a is not None and new_sc < best_score - 1e-14:
                a = new_a
                best_score = new_sc
                best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 6:
                noise = 0.0005 + 0.002 * float(min(no_improve, 50)) / 50.0
                a = normalize_nonneg(a + rng.standard_normal(n) * noise)
            elif no_improve > 60:
                noise = 0.001
                a = normalize_nonneg(best_a + rng.standard_normal(n) * noise)
                no_improve = 0

        if it % 25 == 0:
            a_s = symmetrize(a)
            sc_s = local_score(a_s)
            if np.isfinite(sc_s) and sc_s < best_score - 1e-14:
                best_score = sc_s
                best_a = a_s.copy()
                a = a_s.copy()

        if verbose and it % 200 == 0:
            print("    spectral it=%d score=%.10f" % (it, best_score))

    return best_a


def adam_u(a, time_limit, verbose=False):
    a = normalize_nonneg(a)
    if a.size == 0 or time_limit <= 0:
        return a

    n = a.size
    rep = a
    if rep[-1] < rep[0]:
        rep = rep[::-1]
    u = init_u(rep)

    end = time.monotonic() + float(time_limit)
    t0 = end - float(time_limit)

    best_a = a.copy()
    best_score = local_score(a)

    m = np.zeros(n - 1, dtype=np.float64)
    v = np.zeros(n - 1, dtype=np.float64)
    lr = 0.025
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8
    it = 0
    no_improve = 0

    K0 = 35.0
    K1 = 4000.0

    while time.monotonic() < end:
        it += 1
        a_cur = softmax_u(u)
        conv = fast_convolve(a_cur)
        score = float(2.0 * n * conv.max())

        if score < best_score - 1e-14:
            best_score = score
            best_a = a_cur.copy()
            no_improve = 0

        prog = 0.0
        if time_limit > 0:
            prog = min(1.0, (time.monotonic() - t0) / float(time_limit))
        K = K0 + (K1 - K0) * prog
        tau = max(1e-14, float(conv.max()) / K)

        g = gradient_u_from_data(u, a_cur, conv, tau)
        if not np.all(np.isfinite(g)):
            break

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        bias1 = 1.0 - beta1 ** it
        bias2 = 1.0 - beta2 ** it
        step = (m / bias1) / (np.sqrt(v / bias2) + eps)

        accepted = False
        for mult in (1.0, 0.5, 0.25, 0.10):
            trial_u = np.clip(u - mult * lr * step, -40.0, 40.0)
            ta = softmax_u(trial_u)
            tconv = fast_convolve(ta)
            tsc = float(2.0 * n * tconv.max())
            if np.isfinite(tsc) and tsc < score - 1e-15:
                u = trial_u
                a_cur = ta
                score = tsc
                accepted = True
                break

        if accepted:
            if score < best_score - 1e-14:
                best_score = score
                best_a = a_cur.copy()
            no_improve = 0
            if no_improve == 0 and it % 500 == 0:
                lr = min(lr * 1.02, 0.10)
        else:
            lr = max(lr * 0.75, 1e-5)
            no_improve += 1
            if no_improve > 80 or lr <= 1e-5:
                bbest = best_a
                brep = bbest if bbest[-1] >= bbest[0] else bbest[::-1]
                base_u = init_u(brep)
                noise = 0.02 + 0.04 * min(1.0, float(no_improve) / 500.0)
                rng = np.random.default_rng(97 + 101 * it)
                u = np.clip(base_u + rng.standard_normal(n - 1) * noise, -40.0, 40.0)
                m = np.zeros(n - 1, dtype=np.float64)
                v = np.zeros(n - 1, dtype=np.float64)
                lr = 0.015
                no_improve = 0

        if it % 250 == 0:
            a_s = symmetrize(a_cur)
            sc_s = local_score(a_s)
            if np.isfinite(sc_s) and sc_s < best_score - 1e-14:
                best_score = sc_s
                best_a = a_s.copy()
                brep = best_a if best_a[-1] >= best_a[0] else best_a[::-1]
                u = init_u(brep)
                m = np.zeros(n - 1, dtype=np.float64)
                v = np.zeros(n - 1, dtype=np.float64)
                no_improve = 0

        if verbose and it % 400 == 0:
            print("    adam it=%d score=%.10f lr=%.6f" % (it, best_score, lr))

    return best_a


def lp_direction(a):
    if not HAVE_SCIPY:
        return None
    an = normalize_nonneg(a)
    n = an.size
    if n > 900:
        return None

    conv = fast_convolve(an)
    M = float(conv.max())
    if M <= 0:
        return None
    M *= 1.0005

    R = 2 * n - 1
    rows_conv, cols_conv = get_lp_pattern(n)
    data_conv = np.tile(an, n)

    rows_t = np.arange(R, dtype=np.int64)
    cols_t = np.full(R, n, dtype=np.int64)
    data_t = -np.ones(R, dtype=np.float64)

    rows_all = np.concatenate((rows_conv, rows_t))
    cols_all = np.concatenate((cols_conv, cols_t))
    data_all = np.concatenate((data_conv, data_t))

    try:
        A = sparse.csr_matrix((data_all, (rows_all, cols_all)), shape=(R, n + 1))
    except Exception:
        return None

    b_ub = np.zeros(R, dtype=np.float64)
    c = np.zeros(n + 1, dtype=np.float64)
    c[-1] = 1.0

    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0], dtype=np.float64)

    bnds = []
    for _ in range(n):
        bnds.append((0.0, 1.0))
    bnds.append((0.0, None))

    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bnds=bnds,
            method="highs",
            options={"presolve": True}
        )
    except Exception:
        return None

    if not res.success:
        return None

    b = np.asarray(res.x[:n], dtype=np.float64)
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s <= 1e-12:
        return None
    b = b / s
    t = float(res.x[-1])
    if not np.isfinite(t):
        return None
    return b


def lp_improve(a, time_limit, verbose=False):
    a = normalize_nonneg(a)
    if not HAVE_SCIPY or a.size == 0 or time_limit <= 0:
        return a

    end = time.monotonic() + float(time_limit)
    best_a = a.copy()
    best_score = local_score(a)
    it = 0

    while time.monotonic() < end:
        it += 1
        if it > 12:
            break
        g = lp_direction(a)
        if g is None:
            break

        cur_score = local_score(a)
        improved = False
        for t in (1.0, 0.7, 0.5, 0.3, 0.2, 0.1):
            trial = normalize_nonneg((1.0 - t) * a + t * g)
            sc = local_score(trial)
            if np.isfinite(sc) and sc < cur_score - 1e-13:
                a = trial
                improved = True
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = trial.copy()
                break

        if not improved:
            gs = symmetrize(g)
            for t in (1.0, 0.5, 0.2):
                trial = normalize_nonneg((1.0 - t) * a + t * gs)
                sc = local_score(trial)
                if np.isfinite(sc) and sc < cur_score - 1e-13:
                    a = trial
                    improved = True
                    if sc < best_score - 1e-14:
                        best_score = sc
                        best_a = trial.copy()
                    break

        if improved and verbose:
            print("    lp it=%d score=%.10f" % (it, best_score))

        if not improved:
            break

    return best_a


def lbfgs_polish(a, time_budget, verbose=False):
    a = normalize_nonneg(a)
    if not HAVE_SCIPY or a.size == 0 or time_budget <= 0:
        return a

    n = a.size
    if n <= 1:
        return a

    rep = a if a[-1] >= a[0] else a[::-1]
    u = init_u(rep)

    end = time.monotonic() + float(time_budget)
    best_a = a.copy()
    best_score = local_score(a)

    cur_tau = best_score / (2.0 * n * 50.0)
    cur_tau = max(1e-12, cur_tau)

    t_est0 = time.monotonic()
    try:
        objective_u(u, cur_tau)
        gradient_u(u, cur_tau)
        est = max((time.monotonic() - t_est0) / 2.0, 1e-5)
    except Exception:
        est = 0.01

    factors = (10.0, 40.0, 150.0, 600.0, 2000.0)
    for factor in factors:
        if time.monotonic() >= end - 5.0:
            break
        remaining = end - time.monotonic()
        tau = best_score / (2.0 * n * float(factor))
        tau = max(1e-12, min(1e-3, tau))
        maxfun = int(min(1200, max(80, remaining / (25.0 * est))))
        if maxfun < 40:
            continue
        try:
            bounds = []
            for _ in range(n - 1):
                bounds.append((-40.0, 40.0))
            res = minimize(
                objective_u,
                u,
                jac=gradient_u,
                args=(tau,),
                method="L-BFGS-B",
                bounds=bounds,
                options={
                    "maxiter": max(20, maxfun // 4),
                    "ftol": 1e-15,
                    "gtol": 1e-8,
                    "maxfun": maxfun,
                    "disp": False,
                    "iprint": -1
                }
            )
            u_new = np.clip(np.asarray(res.x, dtype=np.float64), -40.0, 40.0)
            a_new = softmax_u(u_new)
            sc = local_score(a_new)
            if np.isfinite(sc) and sc < best_score - 1e-15:
                best_score = sc
                best_a = a_new.copy()
                u = u_new

            a_s = symmetrize(a_new)
            sc_s = local_score(a_s)
            if np.isfinite(sc_s) and sc_s < best_score - 1e-15:
                best_score = sc_s
                best_a = a_s.copy()
                brep = best_a if best_a[-1] >= best_a[0] else best_a[::-1]
                u = init_u(brep)

            if verbose:
                print("    lbfgs factor=%.0f score=%.10f" % (factor, best_score))
        except Exception as ee:
            if verbose:
                print("    lbfgs error: %s" % str(ee))

    return best_a


def blend_archive(archive, time_limit, verbose=False):
    if not archive or len(archive) < 2 or time_limit <= 0:
        return archive

    end = time.monotonic() + float(time_limit)

    while time.monotonic() < end:
        top = select_entries(archive, 12, unique_lengths=False)

        for i in range(len(top)):
            a_s = symmetrize(top[i][2])
            archive = update_archive(archive, a_s, top[i][1] + "_sym")

        top = select_entries(archive, 12, unique_lengths=False)
        best_score = min([e[0] for e in archive])
        improved = False
        L = len(top)

        for i in range(L):
            if time.monotonic() >= end:
                break
            a1 = top[i][2]
            for j in range(i + 1, L):
                if time.monotonic() >= end:
                    break
                a2 = top[j][2]
                if len(a1) != len(a2):
                    continue
                for t in (0.5, 0.3, 0.7, 0.2, 0.8, 0.1, 0.9):
                    trial = normalize_nonneg((1.0 - t) * a1 + t * a2)
                    sc = local_score(trial)
                    if np.isfinite(sc) and sc < best_score - 1e-13:
                        lab = "blend_" + top[i][1] + "_" + top[j][1] + "_t%s" % str(t)
                        archive = update_archive(archive, trial, lab)
                        best_score = sc
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if verbose and improved:
            print("    blend improved to %.10f" % best_score)
        if not improved:
            break

    return archive


def improve_sequence(a, time_limit, mode="deep", verbose=False):
    a = normalize_nonneg(a)
    if a.size == 0 or time_limit <= 0:
        return a

    t0 = time.monotonic()
    deadline = t0 + float(time_limit)

    if mode == "deep":
        spec_frac = 0.32
    else:
        spec_frac = 0.42
    t_spec = float(time_limit) * spec_frac
    if t_spec > 5.0:
        try:
            a = spectral_improve(a, t_spec, verbose=verbose)
        except Exception as ee:
            if verbose:
                print("    spectral phase error: %s" % str(ee))

    a_s = symmetrize(a)
    if local_score(a_s) < local_score(a):
        a = a_s

    rem = deadline - time.monotonic()
    if rem > 5.0:
        t_adam = rem * 0.60 if mode == "quick" else rem * 0.55
        if t_adam > 5.0:
            try:
                a = adam_u(a, t_adam, verbose=verbose)
            except Exception as ee:
                if verbose:
                    print("    adam phase error: %s" % str(ee))
        a_s = symmetrize(a)
        if local_score(a_s) < local_score(a):
            a = a_s

    if mode == "deep":
        rem = deadline - time.monotonic()
        t_lp = rem * 0.12
        if t_lp > 20.0 and a.size <= 900 and HAVE_SCIPY:
            try:
                a = lp_improve(a, t_lp, verbose=verbose)
            except Exception as ee:
                if verbose:
                    print("    lp phase error: %s" % str(ee))
        a_s = symmetrize(a)
        if local_score(a_s) < local_score(a):
            a = a_s

        rem = deadline - time.monotonic()
        t_lbf = rem * 0.40
        if t_lbf > 30.0:
            try:
                a = lbfgs_polish(a, t_lbf, verbose=verbose)
            except Exception as ee:
                if verbose:
                    print("    lbfgs phase error: %s" % str(ee))
        a_s = symmetrize(a)
        if local_score(a_s) < local_score(a):
            a = a_s

    rem = deadline - time.monotonic()
    if rem > 8.0:
        try:
            a = spectral_improve(a, rem * 0.85, verbose=False)
        except Exception:
            pass
        a_s = symmetrize(a)
        if local_score(a_s) < local_score(a):
            a = a_s

    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    entries = []

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and p.sum() > 1e-300:
            parent = normalize_nonneg(p)
    except Exception:
        parent = None

    if parent is None:
        parent = make_gaussian(768, 0.30)

    entries = update_archive(entries, parent, "parent")
    entries = update_archive(entries, symmetrize(parent), "parent_sym")
    entries = update_archive(entries, smooth_sequence(parent, 0.02), "parent_sm2")
    entries = update_archive(entries, smooth_sequence(parent, 0.05), "parent_sm5")

    base_lengths = [640, 768, 896, 1024, 1152, 1280]
    for n in base_lengths:
        if n <= 0:
            continue
        a = interpolate_sequence(parent, n)
        entries = update_archive(entries, a, "interp_%d" % n)
        entries = update_archive(entries, symmetrize(a), "interp_sym_%d" % n)

    for n in (768, 1024):
        entries = update_archive(entries, make_gaussian(n, 0.25), "gauss_%d" % n)
        entries = update_archive(entries, make_bell(n, 0.50), "bell_%d" % n)
        entries = update_archive(entries, make_bimodal(n), "bimodal_%d" % n)
        entries = update_archive(entries, make_split(n), "split_%d" % n)
        entries = update_archive(entries, make_modulated(n, 1.0, 0.22), "mod1_%d" % n)
        entries = update_archive(entries, make_random_dirichlet(n, 1000 + n), "rand_%d" % n)

    if entries:
        init_best = min([e[0] for e in entries])
        print("Initial best local score: %.10f" % init_best)

    quick_starts = select_entries(entries, k=4, unique_lengths=True)
    for i in range(len(quick_starts)):
        remaining = deadline - time.monotonic()
        if remaining < 45.0:
            break
        e = quick_starts[i]
        t_quick = min(60.0, remaining * 0.25)
        if t_quick < 18.0:
            t_quick = max(10.0, remaining * 0.5)
        print("Quick preoptimize %s for %.1fs" % (e[1], t_quick))
        try:
            a_opt = improve_sequence(e[2], t_quick, mode="quick", verbose=False)
        except Exception as ex:
            a_opt = e[2]
            print("Quick optimize error: %s" % str(ex))
        entries = update_archive(entries, a_opt, e[1] + "_quick")

    deep_starts = select_entries(entries, k=2, unique_lengths=True)
    remaining_main = deadline - time.monotonic() - 220.0
    time_per = max(40.0, remaining_main / max(1, len(deep_starts)))

    for i in range(len(deep_starts)):
        remaining = deadline - time.monotonic() - 220.0
        if remaining < 80.0:
            break
        e = deep_starts[i]
        t_deep = min(time_per, remaining)
        if t_deep < 60.0:
            t_deep = max(30.0, remaining * 0.5)
        print("Deep optimize %s for %.1fs" % (e[1], t_deep))
        try:
            a_opt = improve_sequence(e[2], t_deep, mode="deep", verbose=True)
        except Exception as ex:
            a_opt = e[2]
            print("Deep optimize error: %s" % str(ex))
        entries = update_archive(entries, a_opt, e[1] + "_deep%d" % (i + 1))

    top1 = select_entries(entries, 1)
    if top1:
        remaining = deadline - time.monotonic()
        if remaining > 60.0:
            a_best = top1[0][2]
            t_lbf = min(220.0, remaining * 0.35)
            print("Final L-BFGS polish for %.1fs" % t_lbf)
            try:
                a_lb = lbfgs_polish(a_best, t_lbf, verbose=True)
                entries = update_archive(entries, a_lb, "final_lbfgs")
            except Exception as ex:
                print("Final L-BFGS error: %s" % str(ex))

            remaining = deadline - time.monotonic()
            if remaining > 30.0:
                a_best2 = select_entries(entries, 1)[0][2]
                t_spec = min(150.0, remaining * 0.40)
                print("Final spectral polish for %.1fs" % t_spec)
                try:
                    a_sp = spectral_improve(a_best2, t_spec, verbose=False)
                    entries = update_archive(entries, a_sp, "final_spectral")
                except Exception as ex:
                    print("Final spectral error: %s" % str(ex))

    remaining = deadline - time.monotonic()
    if remaining > 30.0:
        t_blend = min(150.0, remaining * 0.60)
        print("Archive blending for %.1fs" % t_blend)
        entries = blend_archive(entries, t_blend, verbose=True)

    if not entries:
        fallback = make_gaussian(768, 0.30)
        print("Fallback used. local=%.10f" % local_score(fallback))
        return [float(x) for x in fallback]

    top_eval = select_entries(entries, k=6, unique_lengths=False)
    final_a = top1[0][2] if top1 else entries[0][2]
    final_local = local_score(final_a)

    best_actual = float("inf")
    best_actual_seq = final_a
    for e in top_eval:
        seq = [float(x) for x in e[2]]
        try:
            sc = float(evaluate_sequence(seq))
        except Exception:
            sc = local_score(e[2])
        if np.isfinite(sc) and sc < best_actual:
            best_actual = sc
            best_actual_seq = e[2]

    print("Best local score: %.10f" % final_local)
    print("Best actual score: %.10f (n=%d)" % (best_actual, len(best_actual_seq)))

    return [float(x) for x in best_actual_seq]
