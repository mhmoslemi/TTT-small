def fast_convolve(a):

def cross_corr(a, w):

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho_idx = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(a, temp=0.01, n_top=50):
    """Compute subgradient of 2n*max(conv(a,a)) using temperature-weighted peaks."""
    n = len(a)
    b = fast_convolve(a)
    b_max = b.max()
    
    # Temperature-weighted active set
    scaled = b / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    
    if n_top is not None and n_top < len(b):
        # Keep only top n_top
        top_idx = np.argpartition(b, -n_top)[-n_top:]
        w_sel = w[top_idx]
        w_sel_sum = w_sel.sum()
        if w_sel_sum > 1e-30:
            w_full = np.zeros(len(b))
            w_full[top_idx] = w_sel / w_sel_sum
            w = w_full
        else:
            w = np.zeros(len(b))
            w[b.argmax()] = 1.0
    else:
        w_sum = w.sum()
        if w_sum > 1e-30:
            w /= w_sum
        else:
            w = np.zeros(len(b))
            w[b.argmax()] = 1.0
    
    # Subgradient: g[i] = 2n * sum_k w[k] * a[k-i] * I(valid)
    g = cross_corr(a, w)
    # The factor 2 is already handled (cross_corr gives sum w[k]*a[k-i])
    # We need 2n * g
    g = 2.0 * n * g
    return g, b_max

def mirror_descent_log(a0, time_limit, eta_init=0.1, temp_init=0.01,

    best_a /= best_a.sum()
    best_obj = fast_convolve(best_a).max()
    
    cur_eta = eta_init
    cur_temp = temp_init

    it = 0

            a = np.exp(z - z.max())
            a /= a.sum()

                no_improve = 0
            else:
                no_improve += 1
            
            # Mirror descent update in log-space
            # a_i <- a_i * exp(-eta * g_i) / Z
            # z_i <- z_i - eta * g_i (then renormalize)
            z -= cur_eta * g
            z -= z.max()  # prevent overflow
            # Clip to prevent underflow
            z = np.clip(z, -50, 50)
            
            # Adaptive step size
            cur_eta = eta_init / (1 + 0.001 * it)

            z -= z.max()
            z += rng.standard_normal(n) * 0.001
            no_improve = 0
            cur_temp = 0.001

            cur_eta *= 0.9
    
    return best_a, best_obj


def lbgfs_refine(a0, T, time_limit, maxiter=5000):
    """L-BFGS in log-space with soft-max smoothed objective."""
    t0 = time.monotonic()
    n = len(a0)
    a_init = a0.copy().astype(np.float64)
    a_init = np.maximum(a_init, 1e-15)
    a_init /= a_init.sum()
    z0 = np.log(a_init)
    
    best_obj = float('inf')
    best_z = z0.copy()
    
    def soft_obj_grad(z):
        a = np.exp(z - z.max())
        a /= a.sum()
        b = fast_convolve(a)
        Tb = T * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        log_sum_exp = max_Tb + np.log(exp_Tb.sum())
        obj = (2.0 * n / T) * log_sum_exp
        w = exp_Tb / exp_Tb.sum()
        # grad w.r.t. a: 2n * (w cross-correlated with a)
        grad_a = 2.0 * n * cross_corr(a, w)
        # grad w.r.t. z: a * (grad_a - a * sum(grad_a * a))
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - a * S)
        return obj, grad_z
    
    try:
        b0 = fast_convolve(a_init).max()
        best_obj = b0
        
        def func(z):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            obj, _ = soft_obj_grad(z)
            return obj
        
        def jac(z):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            _, g = soft_obj_grad(z)
            return g
        
        res = minimize(
            fun=func, x0=z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 30, 'disp': False}
        )
        final_obj, _ = soft_obj_grad(res.x)
        if np.isfinite(final_obj) and final_obj < best_obj * 1.001:
            a_res = np.exp(res.x - res.x.max())
            a_res /= a_res.sum()
            true_obj = fast_convolve(a_res).max()
            if true_obj < best_obj:
                best_obj = true_obj
                best_z = res.x
    except Exception:
        pass
    
    a_out = np.exp(best_z - best_z.max())
    a_out /= a_out.sum()
    return a_out, fast_convolve(a_out).max()

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian_init(n, sigma_frac=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma_frac) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_cosine_init(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_tukey_init(n, r=0.5):
    x = np.linspace(0, 1, n)
    a = np.ones(n)
    taper = max(1, int(r * n // 2))
    if taper > 0 and taper < n // 2:
        t = np.linspace(0, 1, taper)
        vals = 0.5 * (1 + np.cos(np.pi * (2 * t / r - 1)))
        a[:taper] = vals
        a[-taper:] = vals[::-1]
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_u_shape_init(n, alpha=2.0):

    x = np.linspace(0, 1, n)
    a = (x - 0.5) ** (2 * alpha) + 0.1
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_flat_top_init(n, frac=0.6):

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Frank-Wolfe with away-step,
log-space mirror descent with adaptive step size, L-BFGS polishing,
multi-resolution warm-starting, and diverse restarts.

Key innovations over prior approach:

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_corr(a, w):
    """Compute g[i] = sum_k w[k] * a[k-i] for valid k-i in range [0,n-1]."""
    n = len(a)
    m = len(w)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    return irfft(np.conj(W) * A, fft_size)[:n]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho_idx = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1)
    return np.maximum(v - theta, 0.0)


def eval_obj(a):
    """Compute objective 2n*max(conv)/(sum)^2 locally."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S * S))


def compute_subgradient(a, temp=0.01, n_top=50):
    """Compute subgradient of 2n*max(conv(a,a)) using temperature-weighted peaks."""
    n = len(a)
    b = fast_convolve(a)
    b_max = b.max()
    
    scaled = b / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    
    if n_top is not None and n_top < len(b):
        top_idx = np.argpartition(b, -n_top)[-n_top:]
        w_sel = w[top_idx]
        w_sel_sum = w_sel.sum()
        if w_sel_sum > 1e-30:
            w_full = np.zeros(len(b))
            w_full[top_idx] = w_sel / w_sel_sum
            w = w_full
        else:
            w = np.zeros(len(b))
            w[b.argmax()] = 1.0
    else:
        w_sum = w.sum()
        if w_sum > 1e-30:
            w /= w_sum
        else:
            w = np.zeros(len(b))
            w[b.argmax()] = 1.0
    
    g = cross_corr(a, w)
    g = 2.0 * n * g
    return g, b_max, b


def fw_away_optimize(a0, time_limit, lr_init=0.5, temp_init=0.001,
                     temp_growth=1.02, n_top=50, seed=42):
    """Frank-Wolfe with away-step for minimizing 2n*max(conv)/(sum)^2 on simplex."""
    rng = np.random.default_rng(seed)

    a = a0.copy().astype(np.float64)

            if time.monotonic() - t0 > time_limit:
                break

                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            j_in = np.argmin(g)
            support = np.where(a > 1e-10)[0]
            
            if len(support) > 0:
                j_out = support[np.argmax(g[support])]
                if g[j_out] - g[j_in] > 1e-10:
                    t_max = a[j_out]
                    t = min(cur_lr * 2.0 / (it_count + 1), t_max)
                    if t > 1e-15:

                        a[j_out] -= t

                        if ss > 1e-15:

            ss = a.sum()

                a = best_a.copy()

def lbgfs_refine(a0, T, time_limit, maxiter=5000):

    t0 = time.monotonic()
    n = len(a0)
    a_init = a0.copy().astype(np.float64)
    a_init = np.maximum(a_init, 1e-15)

    z0 = np.log(a_init)
    
    best_obj = float('inf')
    best_z = z0.copy()

    def soft_obj_grad(z):
        a = np.exp(z - z.max())
        a /= a.sum()
        b = fast_convolve(a)
        Tb = T * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        log_sum_exp = max_Tb + np.log(exp_Tb.sum())
        obj = (2.0 * n / T) * log_sum_exp
        w = exp_Tb / exp_Tb.sum()
        grad_a = 2.0 * n * cross_corr(a, w)
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - a * S)
        return obj, grad_z
    
    try:
        b0 = fast_convolve(a_init).max()
        best_obj = b0
        
        def func(z):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            obj, _ = soft_obj_grad(z)
            return obj
        
        def jac(z):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            _, g = soft_obj_grad(z)
            return g
        
        res = minimize(
            fun=func, x0=z0, jac=jac, method='L-BFGS-B',

def coordinate_polish(a0, time_limit):
    """Coordinate descent: try scaling pairs of elements."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    multipliers = np.array([0.5, 0.7, 0.85, 1.15, 1.3, 1.5])
    
    improved = True
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-12:
                continue
            best_j_obj = best_obj + 1e-6
            best_j_val = old_val
            best_k_val = a[k] if (k := n - 1 - j) != j else a[(j + 1) % n]
            
            k = n - 1 - j
            if k == j:
                k = (j + 1) % n
            
            for mult in multipliers:
                new_j = old_val * mult
                if new_j < 0 or new_j > 0.99:
                    continue
                diff = old_val - new_j
                new_k = a[k] + diff
                if new_k < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_j
                a_trial[k] = new_k
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                val = fast_convolve(a_trial).max()

                    best_j_obj = val
                    best_j_val = new_j
                    best_k_val = new_k
            
            if best_j_val != old_val:

                a[k] = a[k] + (old_val - best_j_val)

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian_init(n, sigma_frac=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma_frac) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_cosine_init(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_tukey_init(n, r=0.5):
    x = np.linspace(0, 1, n)
    a = np.ones(n)
    taper = max(1, int(r * n // 2))
    if taper > 0 and taper < n // 2:
        t = np.linspace(0, 1, taper)
        vals = 0.5 * (1 + np.cos(np.pi * (2 * t / r - 1)))
        a[:taper] = vals
        a[-taper:] = vals[::-1]
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_u_shape_init(n, alpha=2.0):
    x = np.linspace(0, 1, n)
    a = (x - 0.5) ** (2 * alpha) + 0.1
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_flat_top_init(n, frac=0.6):
    a = np.ones(n)
    taper = max(1, int((1 - frac) * n / 2))
    if taper > 0 and taper < n // 2:
        t = np.linspace(0, 1, taper)
        a[:taper] = 0.5 * (1 + np.cos(np.pi * t))
        a[-taper:] = a[:taper][::-1]
    a = np.maximum(a, 1e-10)
    return a / a.sum()
