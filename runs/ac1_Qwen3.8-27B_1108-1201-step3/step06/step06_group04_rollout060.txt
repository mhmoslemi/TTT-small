import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_gradient(a, M):
    """Compute subgradient of max(conv) at point a, given peak index M."""
    n = len(a)
    g = np.zeros(n)
    # g[i] = 2*a[M-i] if 0 <= M-i < n
    i_start = max(0, M - n + 1)
    i_end = min(n - 1, M)
    idx = np.arange(i_start, i_end + 1)
    g[idx] = 2.0 * a[M - idx]
    return g

def obj_normalized(a):
    """Objective for sum-normalized sequence: 2*n*max(conv)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def projected_gradient_descent(a0, time_limit, verbose=False):
    """Projected gradient with momentum and backtracking."""
    a = a0.copy()
    n = len(a)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = obj_normalized(a)
    
    t0 = time.monotonic()
    momentum = np.zeros(n)
    beta = 0.9  # momentum coefficient
    lr = 0.05  # initial learning rate
    no_improve = 0
    max_no_improve = 100
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        conv = fast_convolve(a)
        M = int(np.argmax(conv))
        g = compute_gradient(a, M)
        
        # Heavy ball momentum
        momentum = beta * momentum - lr * g
        a_new = a + momentum
        
        # Project onto simplex
        a_new = project_simplex(a_new)
        
        # Check improvement
        obj_new = obj_normalized(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            a = a_new  # still update, but track no improvement
            no_improve += 1
            
        # Adapt learning rate
        if no_improve > 30:
            lr *= 0.9
            if no_improve > 60:
                beta *= 0.95
            if no_improve > max_no_improve:
                # Restart with perturbation
                rng = np.random.default_rng(it)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
                momentum = np.zeros(n)
                lr = 0.05
                beta = 0.9
                no_improve = 0
        
        if verbose and it % 100 == 0:
            print(f"    PGD iter {it}: best_obj={best_obj:.10f}, lr={lr:.6f}")
    
    return best_a, best_obj

"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
- Analytical gradient structure (gradient is a shifted copy of a)

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, M):
    """
    Compute subgradient of max_k conv(a,a)[k] at point a.
    Uses the analytical structure: grad[i] = 2*a[M-i] for valid i.
    """
    n = len(a)
    g = np.zeros(n)
    i_start = max(0, M - n + 1)
    i_end = min(n - 1, M)
    if i_end >= i_start:
        idx = np.arange(i_start, i_end + 1)
        g[idx] = 2.0 * a[M - idx]
    return g


def obj_normalized(a):
    """Objective for sum-normalized sequence: 2*n*max(conv)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_lp_matrix(f, n):
    """Build constraint matrix for LP: (f*g)[k] for all k, with band structure."""
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        length = k_end - k_start + 1
        A_ub[k_start:k_end + 1, j] = f[:length]
    return A_ub


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    A_ub = build_lp_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0)
    return None


def make_hann(n):
    """Hann window normalized to sum=1."""
    a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / (n - 1)))
    a = np.maximum(a, 0)
    s = a.sum()
    return a / s if s > 1e-15 else np.ones(n) / n


def make_kaiser(n, beta):
    """Kaiser window normalized to sum=1."""
    from scipy.special import i0
    r = np.arange(n) - (n - 1) / 2.0
    r = r / ((n - 1) / 2.0)
    a = i0(beta * np.sqrt(np.maximum(1 - r**2, 0)))
    a = np.maximum(a, 0)
    s = a.sum()
    return a / s if s > 1e-15 else np.ones(n) / n

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def spsa_optimize(a0, time_limit, verbose=False):
    """
    SPSA (Simultaneous Perturbation Stochastic Approximation) with
    simplex projection for escaping local minima.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    best_obj = obj_normalized(a)
    best_a = a.copy()

    t0 = time.monotonic()
    c = 0.01  # perturbation size
    a_sp = 0.05  # step size
    gamma = 0.6  # decay rate
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        rng = np.random.default_rng(it * 101 + 17)
        delta = rng.integers(-1, 2, size=n).astype(np.float64)

        # Perturb in both directions
        a_plus = project_simplex(a + c * delta)
        a_minus = project_simplex(a - c * delta)

        obj_plus = obj_normalized(a_plus)
        obj_minus = obj_normalized(a_minus)

        # Gradient estimate
        denom = 2.0 * c
        if abs(denom) < 1e-15:
            break
        grad_est = (obj_plus - obj_minus) / denom * delta

        # Update
        a_new = a - a_sp * grad_est
        a_new = project_simplex(a_new)

        obj_new = obj_normalized(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        a_sp = 0.05 / (1 + 0.01 * it) ** gamma
        c = 0.01 / (1 + 0.005 * it) ** 0.6

        if no_improve > 50:
            # Restart near best
            rng2 = np.random.default_rng(it * 23 + 5)
            a = project_simplex(best_a + rng2.standard_normal(n) * 0.005)
            a_sp = 0.05
            c = 0.01
            no_improve = 0

        if verbose and it % 50 == 0:
            print(f"    SPSA iter {it}: best={best_obj:.10f}")

    return best_a, best_obj

        init_candidates.append(make_kaiser(n_try, 5.0))
        init_candidates.append(make_kaiser(n_try, 8.0))
        init_candidates.append(make_kaiser(n_try, 12.0))
        init_candidates.append(make_flat_top(n_try))
        init_candidates.append(make_raised_cosine(n_try, 0.4))
        init_candidates.append(make_raised_cosine(n_try, 0.6))
        init_candidates.append(make_raised_cosine(n_try, 0.8))
        init_candidates.append(make_peaked(n_try, 1.5))
        init_candidates.append(make_peaked(n_try, 3.0))
        init_candidates.append(np.ones(n_try) / n_try)

    best_init = None
    best_init_obj = float('inf')
    for ic in init_candidates:
        s = ic.sum()
        if s < 1e-10:
            continue
        a_norm = ic / s
        obj = obj_normalized(a_norm)
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a_norm.copy()
        best_seq, best_score = try_sequence(a_norm, best_seq, best_score,
                                             f"init_n{len(a_norm)}")

    print(f"  Best init: obj={best_init_obj:.6f} (n={len(best_init)})")

    # === Phase 2: Main LP-FW ===
    print(f"\n=== Phase 2: LP-FW (n={len(best_init)}) ===")
    rem = deadline - time.monotonic() - 600
    if rem > 60:
        t_alloc = min(rem, 350)
        print(f"  LP-FW: t={t_alloc:.0f}s")
        a_opt, obj = lp_fw_optimize(best_init, time_limit=t_alloc, verbose=True)
        best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "lpfw_main")

    # === Phase 3: Momentum PGD ===
    base = best_seq if best_seq is not None else best_init
    print(f"\n=== Phase 3: Momentum PGD (n={len(base)}) ===")
    rem = deadline - time.monotonic() - 400
    if rem > 30:
        t_alloc = min(rem, 200)
        print(f"  PGD: t={t_alloc:.0f}s")
        a_opt, obj = momentum_pgd(base, time_limit=t_alloc, verbose=True)
        best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "pgd_main")

    # === Phase 4: Structured Coordinate Descent ===
    if best_seq is not None:
        print(f"\n=== Phase 4: Structured CD (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_alloc = min(rem, 150)
            print(f"  CD: t={t_alloc:.0f}s")
            a_opt, obj = structured_cd(best_seq, time_limit=t_alloc, verbose=True)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "cd_main")

    # === Phase 5: Multi-length exploration ===
    base = best_seq if best_seq is not None else best_init
    print(f"\n=== Phase 5: Multi-length LP-FW ===")
    for target_n in [720, 840, 896, 1024]:
        if len(base) == target_n:
            continue
        rem = deadline - time.monotonic() - 200
        if rem < 50:
            break
        f_interp = interpolate_sequence(base, target_n)
        best_seq, best_score = try_sequence(f_interp, best_seq, best_score,
                                             f"interp_n{target_n}")
        t_alloc = min(rem * 0.35, 70)
        if t_alloc < 20:
            continue
        print(f"  LP-FW n={target_n}: t={t_alloc:.0f}s")
        a_opt, obj = lp_fw_optimize(f_interp, time_limit=t_alloc, verbose=False)
        best_seq, best_score = try_sequence(a_opt, best_seq, best_score,
                                             f"lpfw_n{target_n}")

    # === Phase 6: SPSA for escape ===
    if best_seq is not None:
        print(f"\n=== Phase 6: SPSA (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_alloc = min(rem, 120)
            print(f"  SPSA: t={t_alloc:.0f}s")
            a_opt, obj = spsa_optimize(best_seq, time_limit=t_alloc, verbose=True)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "spsa")

    # === Phase 7: Deep LP-FW on best ===
    if best_seq is not None:
        print(f"\n=== Phase 7: Deep LP-FW (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 100
        if rem > 40:
            t_alloc = min(rem, 150)
            print(f"  Deep LP-FW: t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "deep_lpfw")

    # === Phase 8: PGD polish ===
    if best_seq is not None:
        print(f"\n=== Phase 8: PGD Polish (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 50
        if rem > 20:
            t_alloc = min(rem, 80)
            print(f"  PGD polish: t={t_alloc:.0f}s")
            a_opt, obj = momentum_pgd(best_seq, time_limit=t_alloc, verbose=False)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "pgd_polish")

    # === Phase 9: Final CD ===
    if best_seq is not None:
        print(f"\n=== Phase 9: Final CD (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_alloc = min(rem, 40)
            print(f"  Final CD: t={t_alloc:.0f}s")
            a_opt, obj = structured_cd(best_seq, time_limit=t_alloc, verbose=False)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "fin_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
