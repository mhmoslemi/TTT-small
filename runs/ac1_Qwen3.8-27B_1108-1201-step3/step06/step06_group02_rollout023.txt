def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def objective(a):
    """Compute 2*n*max(conv)/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

def multi_peak_subgradient(a, n_peaks=50, temperature=0.01):
    """Compute weighted subgradient from top peaks."""
    n = len(a)
    conv = fast_convolve(a)
    # Select top peaks
    k = min(n_peaks, len(conv))
    top_idx = np.argpartition(conv, -k)[-k:]
    # Weight by softmax
    pv = conv[top_idx]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    # Compute subgradient
    sg = np.zeros(n)
    for i in range(k):
        if w[i] < 1e-12:

        kk = int(top_idx[i])

                    sg[j] += 2.0 * w[i] * a[kk - j]

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def objective_val(a):
    """Compute 2*n*max(conv)/sum^2 for a sequence."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def compute_subgradient(a, peak_indices, weights):

def multi_peak_subgradient_optimize(a0, time_limit, lr_init=0.3, decay=0.9997,

                                     temp_start=0.005, temp_decay=1.002,

    best_a = a.copy()

    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.5:
            break

        conv = fast_convolve(a)
        current_max = conv.max()

        # Adaptive number of peaks
        progress = min(1.0, it / max(1, int(time_limit / 0.5)))
        n_peaks = int(n_peaks_start + (n_peaks_end - n_peaks_start) * progress)
        n_peaks = min(n_peaks, len(conv))

        # Select top peaks with softmax weighting

        pv = conv[top_idx]

            sg = sg / sg_max

        # Multiplicative update
        a = a * np.exp(-lr * sg)
        a = np.maximum(a, 1e-15)
        s = a.sum()

            break

def simulated_annealing(a0, time_limit, verbose=False):
    """
    Simulated annealing: random perturbations with Boltzmann acceptance.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    best_obj = objective_val(a)
    best_a = a.copy()
    current_obj = best_obj
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    T_start = 0.001
    T_end = 1e-6
    total_iters = int(time_limit / 0.01)  # estimate
    cooling = (T_end / T_start) ** (1.0 / max(1, total_iters))

    T = T_start
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Perturbation: modify 1-5 random coordinates
        n_mod = rng.integers(1, 6)
        trial = a.copy()
        mod_idx = rng.choice(n, size=n_mod, replace=False)
        for i in mod_idx:
            factor = rng.uniform(0.5, 1.5)
            trial[i] *= factor
        trial = np.maximum(trial, 0.0)
        ts = trial.sum()
        if ts < 1e-15:
            T *= cooling
            continue
        trial = trial / ts
        trial_obj = objective_val(trial)

        # Acceptance
        delta = trial_obj - current_obj
        if delta < 0 or rng.random() < np.exp(-delta / T):
            a = trial
            current_obj = trial_obj
            if current_obj < best_obj - 1e-12:
                best_obj = current_obj
                best_a = a.copy()

        T *= cooling

        if verbose and it % 5000 == 0:
            print(f"      SA iter {it}: best={best_obj:.8f}, T={T:.8f}")

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, sigma_frac=0.35):
    """Gaussian-like initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_super_gaussian(n, power=4, width_frac=0.7):

    return a


def make_cosine_envelope(n, freq=1, amp=0.3):

def make_triangular(n):

    half = plateau_frac / 2.0
    edge = 0.1
    a = np.zeros(n)
    for i in range(n):
        if half - edge <= x[i] <= half + edge:
            a[i] = 1.0
        elif half - edge - edge <= x[i] < half - edge:
            t = (x[i] - (half - 2 * edge)) / edge
            a[i] = 0.5 * (1 - np.cos(np.pi * t))
        elif half + edge < x[i] <= half + edge + edge:
            t = (x[i] - (half + edge)) / edge
            a[i] = 0.5 * (1 + np.cos(np.pi * t))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization for minimizing the convolution peak objective.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # --- Load parent ---
    parent = None
    parent_n = 0
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_n = len(parent)
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={parent_n}, score={p_score:.8f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_gaussian(768, 0.35)
        parent_n = len(parent)
        try_seq(parent, "default_parent")

    print(f"\n=== Phase 1: Diverse Initializations ===")
    t0 = time.monotonic()

    # Generate diverse initializations at multiple lengths
    n_values = [512, 640, 768, 832, 896, 960, 1024]
    init_candidates = []

    for n_try in n_values:
        init_candidates.append((f'gauss_{n_try}', n_try, make_gaussian(n_try, 0.35)))
        init_candidates.append((f'sgauss_{n_try}', n_try, make_super_gaussian(n_try, 4, 0.7)))
        init_candidates.append((f'cos_{n_try}', n_try, make_cosine_envelope(n_try, 1, 0.25)))
        init_candidates.append((f'tri_{n_try}', n_try, make_triangular(n_try)))
        init_candidates.append((f'plat_{n_try}', n_try, make_plateau(n_try, 0.5)))
        init_candidates.append((f'flat_{n_try}', n_try, np.ones(n_try) / n_try))

    # Add parent
    init_candidates.append((f'parent_{parent_n}', parent_n, parent))

    # Evaluate all
    scored_inits = []
    for label, n_try, a_init in init_candidates:
        sc = objective_val(a_init)
        scored_inits.append((sc, a_init, label))
        if sc < best_score * 1.05:
            # Only call evaluate_sequence for promising ones (cheaper check first)
            try_seq(a_init, f"init_{label}")

    scored_inits.sort(key=lambda x: x[0])
    print(f"  Top 5 inits:")
    for i, (sc, ai, lab) in enumerate(scored_inits[:5]):
        print(f"    {lab}: obj={sc:.6f}")
    print(f"  Phase 1 time: {time.monotonic()-t0:.1f}s")

    # --- Phase 2: Main optimization on best initializations ---
    print(f"\n=== Phase 2: Subgradient Optimization ===")
    # Allocate time to top 3 initializations
    top_init_count = min(4, len(scored_inits))
    t_per_init = (deadline - time.monotonic() - 400) / max(1, top_init_count)

    for trial_idx in range(top_init_count):
        rem = deadline - time.monotonic() - 350
        if rem < 30:
            break
        sc_init, a_init, label = scored_inits[trial_idx]
        t_alloc = min(t_per_init, rem * 0.6)
        if t_alloc < 15:
            t_alloc = 15
        print(f"  Subgradient MD [{label}] n={len(a_init)}: t={t_alloc:.0f}s (init_obj={sc_init:.6f})")
        a_opt, obj = multi_peak_subgradient_optimize(
            a_init, time_limit=t_alloc,
            lr_init=0.4, decay=0.9998,
            n_peaks_start=30, n_peaks_end=150,
            temp_start=0.003, temp_decay=1.003,
            verbose=True
        )
        try_seq(a_opt, f"md_{label}")

    # Also try with reversed parent (symmetry exploitation)
    if parent is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            parent_rev = parent[::-1]
            avg_sym = 0.5 * parent + 0.5 * parent_rev
            avg_sym = np.maximum(avg_sym, 0)
            avg_sym = avg_sym / avg_sym.sum()
            t_alloc = min(rem * 0.3, 80)
            print(f"  Subgradient MD [symmetric_parent]: t={t_alloc:.0f}s")
            a_opt, obj = multi_peak_subgradient_optimize(
                avg_sym, time_limit=t_alloc,
                lr_init=0.4, decay=0.9998,
                n_peaks_start=30, n_peaks_end=150,
                temp_start=0.003, temp_decay=1.003,
                verbose=True
            )
            try_seq(a_opt, "md_sym_parent")

    # --- Phase 3: Coordinate Descent Fine-Tuning ---
    print(f"\n=== Phase 3: Coordinate Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 20:
            t_alloc = min(rem * 0.5, 120)
            print(f"  CD: n={len(best_seq)}, t={t_alloc:.0f}s")
            a_opt, obj = coordinate_descent(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "cd")

    # --- Phase 4: Targeted Peak Reduction ---
    print(f"\n=== Phase 4: Targeted Peak Reduction ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            t_alloc = min(rem * 0.5, 100)
            print(f"  TPR: n={len(best_seq)}, t={t_alloc:.0f}s")
            a_opt, obj = targeted_peak_reduction(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "tpr")

    # --- Phase 5: Simulated Annealing ---
    print(f"\n=== Phase 5: Simulated Annealing ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_alloc = min(rem * 0.4, 80)
            print(f"  SA: n={len(best_seq)}, t={t_alloc:.0f}s")
            a_opt, obj = simulated_annealing(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "sa")

    # --- Phase 6: Multi-length exploration ---
    print(f"\n=== Phase 6: Multi-Length Exploration ===")
    if best_seq is not None:
        base_n = len(best_seq)
        target_ns = [640, 768, 832, 896, 960, 1024]
        for target_n in target_ns:
            if target_n == base_n:
                continue
            rem = deadline - time.monotonic() - 100
            if rem < 30:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            sc = objective_val(f_interp)
            if sc < best_score * 1.02:
                t_alloc = min(rem * 0.3, 60)
                if t_alloc > 15:
                    print(f"  MD interp n={target_n}: t={t_alloc:.0f}s (obj={sc:.6f})")
                    a_opt, obj = multi_peak_subgradient_optimize(
                        f_interp, time_limit=t_alloc,
                        lr_init=0.3, decay=0.9997,
                        n_peaks_start=40, n_peaks_end=120,
                        temp_start=0.004, temp_decay=1.003,
                        verbose=False
                    )
                    try_seq(a_opt, f"md_interp_{target_n}")
                    # Quick CD polish
                    rem2 = deadline - time.monotonic() - 80
                    if rem2 > 10:
                        a_opt2, obj2 = coordinate_descent(a_opt, time_limit=min(rem2*0.3, 30))
                        try_seq(a_opt2, f"cd_interp_{target_n}")

    # --- Phase 7: Iterate optimization-refinement cycle ---
    print(f"\n=== Phase 7: Iterative Refinement ===")
    if best_seq is not None:
        for cycle in range(3):
            rem = deadline - time.monotonic() - 50
            if rem < 30:
                break
            # Subgradient round
            t_md = min(rem * 0.3, 60)
            if t_md > 10:
                a_opt, obj = multi_peak_subgradient_optimize(
                    best_seq, time_limit=t_md,
                    lr_init=0.2, decay=0.9997,
                    n_peaks_start=50, n_peaks_end=200,
                    temp_start=0.002, temp_decay=1.004,
                    verbose=False
                )
                try_seq(a_opt, f"refine_md_{cycle}")

            # CD round
            rem = deadline - time.monotonic() - 30
            if rem > 15:
                t_cd = min(rem * 0.3, 40)
                if t_cd > 8:
                    a_opt, obj = coordinate_descent(best_seq, time_limit=t_cd)
                    try_seq(a_opt, f"refine_cd_{cycle}")

            # TPR round
            rem = deadline - time.monotonic() - 20
            if rem > 15:
                t_tpr = min(rem * 0.3, 30)
                if t_tpr > 8:
                    a_opt, obj = targeted_peak_reduction(best_seq, time_limit=t_tpr)
                    try_seq(a_opt, f"refine_tpr_{cycle}")

    # --- Phase 8: Final polish ---
    print(f"\n=== Phase 8: Final Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            a_opt, obj = coordinate_descent(best_seq, time_limit=rem)
            try_seq(a_opt, "final_cd")

        rem = deadline - time.monotonic() - 2
        if rem > 5:
            a_opt, obj = targeted_peak_reduction(best_seq, time_limit=rem)
            try_seq(a_opt, "final_tpr")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"Total time used: {time.monotonic() - (deadline - max(1.0, float(budget_s) - 10.0)):.1f}s / {budget_s}s")
    return [float(x) for x in best_seq]
