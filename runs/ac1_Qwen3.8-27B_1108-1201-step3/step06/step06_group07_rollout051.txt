def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def fast_cross_convolve(f, g):
    """Linear cross-convolution of f and g via FFT."""
    nf = len(f)
    ng = len(g)
    size = nf + ng - 1
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    G = np.fft.rfft(g, fft_size)
    return np.fft.irfft(F * G, fft_size)[:size]

def build_conv_matrix(f):
    """Build the convolution matrix C such that C @ g = convolve(f, g)."""
    n = len(f)
    C = lil_matrix((2*n-1, n))
    for k in range(2*n-1):
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        for m in range(m_start, m_end + 1):
            C[k, m] = f[k - m]
    return C.tocsr()

        g = cp.Variable(n)
        constraints = [g >= 0]

    if g.value is None:

import numpy as np
import time
import cvxpy as cp
from scipy.sparse import coo_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(f):
    """Build sparse convolution matrix C such that C @ g = np.convolve(f, g)."""
    n = len(f)
    rows_list = []
    cols_list = []
    data_list = []
    for k in range(2 * n - 1):
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        m_vals = np.arange(m_start, m_end + 1)
        rows_list.extend([k] * len(m_vals))
        cols_list.extend(m_vals.tolist())
        data_list.extend(f[k - m_vals].tolist())
    C = coo_matrix((data_list, (rows_list, cols_list)), shape=(2 * n - 1, n))
    return C.tocsr()


def solve_lp_direction(f, M, symmetric=False):
    """Solve LP: max sum(g) s.t. g >= 0, convolve(f,g) <= M. Returns g normalized to sum=1."""
    n = len(f)
    g = cp.Variable(n)
    constraints = [g >= 0]
    if symmetric:
        for i in range(n // 2):
            constraints.append(g[i] == g[n - 1 - i])
    C = build_conv_matrix(f)
    constraints.append(C @ g <= M * np.ones(2 * n - 1))
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    solvers_to_try = [cp.GLOP, cp.CBC, cp.SCIP, cp.ECOS, cp.CLARABEL]
    for solver in solvers_to_try:
        try:
            prob.solve(solver=solver)
            if g.value is not None:
                g_val = np.array(g.value).flatten()
                g_val = np.maximum(g_val, 0.0)
                s = g_val.sum()
                if s > 1e-10:
                    return g_val / s
                return None
        except Exception:
            continue
    return None

    n = len(a)

    best_obj = conv.max()
    best_a = a.copy()

    lr = lr_init
    temp = temp_init
    no_improve = 0
    it = 0

        it += 1
        conv = fast_convolve(a)
        M = conv.max()

        K = min(top_k, len(conv))
        if K < len(conv):
            peak_indices = np.argpartition(conv, -K)[-K:]
        else:
            peak_indices = np.arange(len(conv))

        pv = conv[peak_indices]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peak_indices))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        g = np.zeros(n)
        for idx in range(len(peak_indices)):
            k = int(peak_indices[idx])
            w_k = w[idx]
            if w_k < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue
            g[js:je + 1] += 2.0 * w_k * a[k - np.arange(js, je + 1)]

        best_lr = lr * 0.1
        best_trial_obj = M * 1.01
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            if trial_lr < 1e-10:
                continue
            log_update = -trial_lr * g
            log_update = np.clip(log_update, -50, 50)
            a_trial = a * np.exp(log_update)
            a_trial = np.maximum(a_trial, 1e-15)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial = a_trial / ss
            trial_obj = fast_convolve(a_trial).max()
            if trial_obj <= best_trial_obj:
                best_lr = trial_lr
                best_trial_obj = trial_obj
                if trial_obj < M * 0.999:
                    break

        log_update = -best_lr * g
        log_update = np.clip(log_update, -50, 50)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        conv = fast_convolve(a)
        new_obj = conv.max()

        if new_obj < best_obj - 1e-13:
            best_obj = new_obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    {label} MD iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        temp = min(temp * 1.002, 0.05)
        if no_improve > 100:
            lr *= 0.5
            no_improve = 0
        lr = max(lr * 0.9999, 1e-8)

        if no_improve > 500:
            rng = np.random.default_rng(it * 7919 + 13)
            noise_scale = 0.001 + 0.002 * min(no_improve / 5000, 1.0)
            a = np.maximum(best_a + rng.standard_normal(n) * noise_scale, 1e-15)
            a = a / a.sum()
            lr = lr_init * 0.3
            no_improve = 0

        if it % 500 == 0:
            a_sym = 0.5 * (a + a[::-1])
            a_sym = np.maximum(a_sym, 1e-15)
            a_sym = a_sym / a_sym.sum()
            sym_obj = fast_convolve(a_sym).max()
            if sym_obj < best_obj:
                best_obj = sym_obj
                best_a = a_sym.copy()
                a = a_sym.copy()

        if lr < 1e-9:
            break

    return best_a, best_obj


def coordinate_descent(f0, time_limit, verbose=False, label=""):
    """Fine coordinate descent for polishing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-10:
        return f0, float('inf')
    f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    rounds = 0

    while time.monotonic() - t0 < time_limit:
        rounds += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in [0.8, 0.9, 0.97, 1.03, 1.1, 1.25]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                trial = f.copy()
                trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                trial[mask] *= scale
                trial = np.maximum(trial, 0.0)
                ss = trial.sum()
                if ss < 1e-15:
                    continue
                trial = trial / ss
                val = fast_convolve(trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0.0)
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            if verbose:
                print(f"    {label} CD converged in {rounds} rounds, obj={best_obj:.10f}")
            break

    return best_f, best_obj


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian-like initial sequence normalized to sum=1."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def symmetrize(a):
    """Symmetrize a sequence (make it palindromic)."""
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    ss = a_sym.sum()
    if ss > 1e-15:
        a_sym /= ss
    return a_sym


def interpolate_to_length(a, new_n):
    """Interpolate a sequence to a new length, maintaining sum=1."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a)
    a_interp = np.maximum(a_interp, 0)
    ss = a_interp.sum()
    if ss > 1e-15:
        a_interp /= ss
    return a_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: LP-based direction finding as the main engine,
    combined with mirror descent, symmetric LP, coordinate descent polishing,
    and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, lbl=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{lbl}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is not None:
        n_base = len(parent)
    else:
        n_base = 768

    # === Phase 1: LP Optimization (main workhorse) ===
    print(f"\n=== Phase 1: LP Optimization (n={n_base}) ===")
    remaining = deadline - time.monotonic()
    t_lp = remaining * 0.45
    print(f"  Time allocated: {t_lp:.0f}s")

    f_init = parent.copy() if parent is not None else make_gaussian(n_base, 0.3)
    f_opt, obj = lp_optimize(f_init, time_limit=t_lp, verbose=True, label="LP_main")
    try_seq(f_opt, "LP_main")

    # === Phase 2: Mirror Descent refinement ===
    print(f"\n=== Phase 2: Mirror Descent ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_md = remaining * 0.18
        print(f"  Time allocated: {t_md:.0f}s")
        f_opt, obj = mirror_descent(best_seq, time_limit=t_md, top_k=50,
                                    temp_init=0.0002, lr_init=0.008,
                                    verbose=True, label="MD")
        try_seq(f_opt, "MD")

    # === Phase 3: Symmetric LP ===
    print(f"\n=== Phase 3: Symmetric LP ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_lp2 = remaining * 0.18
        print(f"  Time allocated: {t_lp2:.0f}s")
        f_sym = symmetrize(best_seq)
        f_opt, obj = lp_optimize(f_sym, time_limit=t_lp2, verbose=True,
                                 label="LP_sym", use_sym=True)
        try_seq(f_opt, "LP_sym")

    # === Phase 4: Coordinate Descent Polish ===
    print(f"\n=== Phase 4: CD Polish ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_cd = remaining * 0.12
        print(f"  Time allocated: {t_cd:.0f}s")
        f_opt, obj = coordinate_descent(best_seq, time_limit=t_cd,
                                        verbose=True, label="CD1")
        try_seq(f_opt, "CD1")

    # === Phase 5: LP with relaxed M (explore different region) ===
    print(f"\n=== Phase 5: LP Relaxed ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_lp3 = remaining * 0.15
        print(f"  Time allocated: {t_lp3:.0f}s")
        f_opt, obj = lp_optimize(best_seq, time_limit=t_lp3, verbose=True,
                                 label="LP_relax", m_relax=1.02)
        try_seq(f_opt, "LP_relax")

    # === Phase 6: Try slightly different lengths ===
    print(f"\n=== Phase 6: Different Lengths ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        for n_try in [n_base + 64, n_base - 64, n_base + 128]:
            if remaining < 90:
                break
            if n_try < 128:
                continue
            f_interp = interpolate_to_length(best_seq, n_try)
            t_alloc = min(remaining * 0.2, 180)
            print(f"  Trying n={n_try}, time={t_alloc:.0f}s")
            f_opt, obj = lp_optimize(f_interp, time_limit=t_alloc,
                                     verbose=True, label=f"LP_n{n_try}")
            try_seq(f_opt, f"LP_n{n_try}")
            remaining = deadline - time.monotonic()

    # === Phase 7: Final coordinate descent ===
    print(f"\n=== Phase 7: Final CD ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            t_final = min(remaining - 10, 150)
            print(f"  Final CD time: {t_final:.0f}s")
            f_opt, obj = coordinate_descent(best_seq, time_limit=t_final,
                                            verbose=True, label="FinalCD")
            try_seq(f_opt, "FinalCD")

            # Try symmetrized final
            f_s = symmetrize(best_seq)
            try_seq(f_s, "final_sym")
            if best_seq is not None:
                f_opt, obj = coordinate_descent(symmetrize(best_seq),
                                                time_limit=min(remaining - 20, 60),
                                                verbose=False, label="FinalCD_sym")
                try_seq(f_opt, "FinalCD_sym")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
