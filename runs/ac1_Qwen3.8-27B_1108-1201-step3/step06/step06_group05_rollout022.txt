def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)

    Smoothed approximation to max(c) using top-k average.
    If beta=0, returns exact top-k average (non-smooth).
    If beta>0, uses softmax-weighted top-k.

    if k >= len(c):
        return c.max()
    # Sort and take top k
    sorted_c = np.sort(c)[::-1]
    top_k = sorted_c[:k]
    if beta == 0:
        return top_k.mean()
    else:
        # Softmax over top k
        w = np.exp(beta * (top_k - top_k[0]))
        w /= w.sum()
        return float(np.dot(w, top_k))

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def pnorm_gradient(f, c, p):
    """
    Compute gradient of ||c||_p where c = conv(f,f).
    For p > 1: grad = (1/||c||_p^(p-1)) * c^(p-1) convolved back to f-space.
    For p = inf limit: gradient at the argmax.
    """
    n = len(f)
    max_c = np.max(c)
    if max_c < 1e-15:
        return np.zeros(n)
    
    if p >= 64:
        # Use top-1 (hard max) subgradient
        idx = np.argmax(c)
        w = np.zeros(len(c))
        w[idx] = 1.0
    else:
        # c_norm^(p-1) * (1/||c||_p^(p-1))
        c_pos = np.maximum(c, 0)
        pnorm = (np.sum(c_pos ** p)) ** (1.0 / p)
        if pnorm < 1e-15:
            return np.zeros(n)
        w = c_pos ** (p - 1) / (pnorm ** (p - 1))
    
    # Build weight sequence of length 2n-1
    w_seq = np.zeros(2 * n - 1)
    w_seq[:len(w)] = w
    
    # g[j0] = 2 * sum_k w_seq[k] * f[k-j0] for valid k-j0
    # Computed via: g[j0] = 2 * (w_seq * f_ref)[n-1-j0]
    f_ref = f[::-1].copy()
    conv_len = (2 * n - 1) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = np.fft.rfft(w_seq, fft_size)
    F = np.fft.rfft(f_ref, fft_size)
    conv = np.fft.irfft(W * F, fft_size)[:conv_len]
    
    indices = np.arange(n - 1, -1, -1)
    g = 2.0 * conv[indices]
    return g


def pnorm_value(c, p):
    """Compute ||c||_p."""
    if p >= 64:
        return np.max(c)
    c_pos = np.maximum(c, 0)
    return (np.sum(c_pos ** p)) ** (1.0 / p)


def gradient_descent_pnorm(f0, time_limit, p_schedule=None, verbose=False):
    """
    Adaptive p-norm gradient descent with Nesterov momentum.
    p increases over time (geometric schedule) to sharpen the max approximation.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    c = fast_convolve(f)
    best_obj = c.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    
    if p_schedule is None:
        p_schedule = [4, 8, 16, 32, 64, 128, 256, 512]
    
    # Nesterov momentum
    v = np.zeros(n)
    beta_mom = 0.9
    
    for p in p_schedule:
        if time.monotonic() - t0 > time_limit * 0.88:
            break
        
        stage_start = time.monotonic()
        remaining = time_limit * 0.88 - (time.monotonic() - t0)
        stage_time = min(remaining, time_limit * 0.1)
        
        it = 0
        no_improve = 0
        # Initial learning rate
        c_curr = fast_convolve(f)
        g = pnorm_gradient(f, c_curr, p)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        lr = 0.5 / (g_norm + 1e-10)
        lr = min(lr, 0.1 / np.sqrt(n))
        
        while time.monotonic() - stage_start < stage_time:
            it += 1
            
            # Nesterov lookahead
            y = f + beta_mom * v
            c_y = fast_convolve(y)
            g_y = pnorm_gradient(y, c_y, p)
            g_norm = np.linalg.norm(g_y)
            
            if g_norm < 1e-14:
                break
            
            # Adaptive step size
            lr = min(lr, 1.0 / (g_norm * 2.0 + 1e-10))
            lr = min(lr, 0.5 / np.sqrt(n))
            
            # Momentum update
            v = beta_mom * v - lr * g_y
            f_trial = project_simplex(f + v)
            
            # Evaluate
            c_trial = fast_convolve(f_trial)
            obj_trial = c_trial.max()
            
            # Backtracking if needed
            if obj_trial > best_obj + 1e-8:
                lr *= 0.5
                v = beta_mom * v - lr * g_y
                f_trial = project_simplex(f + v)
                c_trial = fast_convolve(f_trial)
                obj_trial = c_trial.max()
            
            if obj_trial <= pnorm_value(c_curr, min(p, 2)) + 1e-10:
                f = f_trial
                c_curr = c_trial
                if obj_trial < best_obj - 1e-13:
                    best_obj = obj_trial
                    best_f = f.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
                lr *= 0.7
                # Revert momentum
                v = np.zeros(n)
            
            if no_improve > 60:
                # Perturb and restart momentum
                rng = np.random.default_rng(it * 7 + int(p) * 3)
                noise_scale = 0.003 * np.sqrt(4.0 / p)
                f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
                v = np.zeros(n)
                lr = 0.3 / (g_norm + 1e-10)
                no_improve = 0
                c_curr = fast_convolve(f)
            
            if verbose and it % 200 == 0:
                print(f"    GD p={p} it={it}: obj={c_curr.max():.10f}, best={best_obj:.10f}")
        
        if verbose:
            print(f"  GD p={p} done: best={best_obj:.10f}, t={time.monotonic()-t0:.1f}s")
    
    return best_f, best_obj


def solve_lp_direction_reduced(f, M, n, K=None, warm_start=None):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Uses only top K most violated constraints for speed.
    """
    if K is None:
        K = min(2 * n - 1, max(50, n // 2))
    
    # Compute current convolution to find most violated constraints
    # At the current point f, the constraint (f*g)[k] <= M is "most violated"
    # when it's hardest to satisfy. We approximate by looking at where f*f is largest.
    c = fast_convolve(f)
    
    # Select top K constraint indices (where convolution peak is)
    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
    else:
        top_idx = np.arange(len(c))
    
    # Build reduced LP
    A_ub = np.zeros((len(top_idx), n))
    for ii, k in enumerate(top_idx):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A_ub[ii, js:je + 1] = f[js:je + 1][::-1]
    
    b_ub = np.full(len(top_idx), M)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    
    options = {'presolve': True, 'dual_feasibility_tolerance': 1e-9,
               'primal_feasibility_tolerance': 1e-9}
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options=options)
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, K=None, verbose=False):
    """
    LP-based Frank-Wolfe with constraint reduction.
    Minimizes max_k (f*f)_k subject to f>=0, sum(f)=1.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    if K is None:
        K = min(2 * n - 1, max(64, n))
    
    c = fast_convolve(f)
    M = c.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        g = solve_lp_direction_reduced(f, M * m_exp, n, K=K)
        if g is None:
            m_exp *= 1.15
            if m_exp > 8.0:
                rng = np.random.default_rng(it * 19 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                c = fast_convolve(f)
                M = c.max()
                m_exp = 1.0
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp *= 1.1
            continue
        g = g / g_sum
        
        # Check fixed point
        diff = np.max(np.abs(g - f))
        if diff < 1e-11:
            no_improve += 1
            if no_improve > 30:
                m_exp = min(m_exp * 1.05, 5.0)
            if no_improve > 100:
                rng = np.random.default_rng(it * 23 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                c = fast_convolve(f)
                M = c.max()
                m_exp = 1.0
                no_improve = 0
            continue
        else:
            if diff < 1e-8:
                no_improve += 1
            else:
                no_improve = 0
        
        # Armijo line search
        best_alpha = 0.0
        best_val = M
        alpha = 1.0
        
        for _ in range(30):
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                break
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
            if val > M * (1.0 - 1e-4 * alpha):
                alpha *= 0.5
            else:
                alpha *= 0.75
        
        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            M = fast_convolve(f_new).max()
            f = f_new
            m_exp = 1.0
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 10 == 0:
                    print(f"    LP-FW it={it}: obj={best_obj:.12f}")
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.04, 6.0)
        
        if no_improve > 150:
            break
    
    return best_f, best_obj


def spectral_flatten(f, iterations=20):
    """
    Attempt to flatten the power spectrum of f to reduce peak autocorrelation.
    Uses iterative amplitude modification in the DFT domain.
    """
    n = len(f)
    f_work = f.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    for _ in range(iterations):
        # Compute DFT
        A = np.fft.rfft(f_work)
        # Target: flat magnitude (except DC)
        magnitude = np.abs(A)
        # Set target magnitude to geometric mean of non-DC magnitudes
        if len(magnitude) > 1:
            non_dc = magnitude[1:]
            target_mag = np.sqrt(np.mean(non_dc ** 2))
        else:
            target_mag = magnitude[0] if len(magnitude) > 0 else 1.0
        
        # Blend towards flat
        new_mag = 0.7 * magnitude + 0.3 * target_mag
        new_mag[0] = magnitude[0]  # Keep DC
        
        # Reconstruct with new magnitude, same phase
        phase = np.angle(A)
        A_new = new_mag * np.exp(1j * phase)
        f_new = np.fft.irfft(A_new, n * 2)[:n]
        f_new = np.maximum(f_new, 0)
        ss = f_new.sum()
        if ss < 1e-15:
            break
        f_new = f_new / ss
        
        obj = fast_convolve(f_new).max()
        if obj < best_obj:
            best_obj = obj
            best_f = f_new.copy()
            f_work = f_new
        else:
            f_work = f_new  # Continue from here
    
    return best_f, best_obj


def coordinate_polish(f0, time_limit, verbose=False):
    """
    Fine coordinate descent with binary search for optimal value.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f0, fast_convolve(f0).max()
    f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        c = fast_convolve(f)
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-12:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            # Try multiplier values
            for mult in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                f_trial[mask] *= (1.0 - new_val) / denom
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial = f_trial / ss
                
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if abs(best_j_val - old_val) > 1e-15:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_f, best_obj


def make_gaussian(n, sigma_frac=0.3):
    """Create a Gaussian initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_raised_cosine(n):
    """Raised cosine (Hann window) shape."""
    x = np.linspace(-1, 1, n)
    a = 0.5 * (1 + np.cos(np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6, power=4):
    """Bell-shaped initial sequence."""
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** power
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_uniform(n):
    return np.ones(n) / n


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def evaluate_local(a):
    """Local evaluation without calling evaluate_sequence (for speed)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_convolve(a)
    return float(2 * len(a) * c.max() / (s ** 2))

        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:

            parent = p / p.sum()
