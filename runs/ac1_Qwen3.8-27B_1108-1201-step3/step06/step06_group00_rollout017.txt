from scipy.sparse import lil_matrix

def solve_lp_direction(f, M, n):
    """

    """
    # A is (2n-1) x n, A[k,j] = f[k-j] if 0 <= k-j < n

            if 0 <= idx < n:

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    return res.x

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def eval_local(a):
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S * S))

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based directional optimization
with constraint generation, subgradient descent, and multi-scale refinement.

The key innovation is using a linear program at each step to find the optimal
direction that reduces the peak of the auto-convolution while maintaining
non-negativity, following the approach from the state-of-the-art LP iteration.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
from scipy.sparse import csr_array, eye as sp_eye, vstack as sp_vstack

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_local(a):
    """Compute objective 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S * S))


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix_indices(n):
    """Precompute (row, col) indices for the convolution matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)


def solve_lp_direction(f, M, row_idx, col_idx, time_limit=10.0):
    """
    Solve: maximize sum(g) subject to (f*g)[k] <= M for all k, g >= 0.
    Uses precomputed sparsity pattern.
    """
    n = len(f)
    # Build sparse A matrix
    vals = f[np.array(row_idx) - ... ]  # Need to be careful here
    # Actually A[k,j] = f[k-j], and we stored (k,j) pairs
    # So vals[i] = f[row_idx[i] - col_idx[i]]
    vals = f[row_idx - col_idx]
    A = csr_array((vals, (row_idx, col_idx)), shape=(2*n-1, n))
    
    # Add non-negativity constraints: -g <= 0
    I_neg = -sp_eye(n, format='csr')
    A_full = sp_vstack([A, I_neg], format='csr')
    b_full = np.concatenate([M * np.ones(2*n-1), np.zeros(n)])
    c = -np.ones(n)
    
    res = linprog(c, A_ub=A_full, b_ub=b_full, method='highs',
                  options={'time_limit': time_limit, 'presolve': True})
    if res.success:
        return res.x, res.fun
    return None, None

def subgradient_optimize(a0, time_limit, lr_init=0.005, momentum=0.9,
                         temp_init=0.001, temp_growth=1.02,
                         kick_interval=2000, kick_size=0.003,
                         n_top=None, seed=42):
    """Projected subgradient with Nesterov momentum and temperature-weighted peaks."""
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    S = a.sum()
    a = a / S
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    cur_lr = lr_init
    cur_temp = temp_init
    no_improve = 0
    it_count = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        b_max = b.max()
        
        # Temperature-weighted peaks
        scaled = b / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(b))
            w[b.argmax()] = 1.0
        else:
            w /= w_sum
        
        if n_top is not None and n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w_sel = w[top_idx]
            w_sel_sum = w_sel.sum()
            if w_sel_sum > 1e-30:
                w_sel /= w_sel_sum
                w_full = np.zeros(len(b))
                w_full[top_idx] = w_sel
                w = w_full
            else:
                w = np.zeros(len(b))
                w[b.argmax()] = 1.0
        
        # Subgradient: g[i] = sum_k w[k] * a[k-i] (valid range)
        g = np.zeros(n)
        active_k = np.where(w > 1e-12)[0]
        for k in active_k:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += w[k] * a[k - idx]
        
        # Nesterov step
        v = momentum * v - cur_lr * g
        a_new = project_simplex(a + v)
        
        b_new = fast_convolve(a_new)
        obj_new = b_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        it_count += 1
        
        if no_improve > kick_interval:
            a = project_simplex(best_a + rng.standard_normal(n) * kick_size)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.7, lr_init * 0.1)
            no_improve = 0
        elif no_improve > kick_interval // 2:
            cur_lr *= 0.95
        
        cur_temp *= temp_growth
        if cur_temp > 0.5:
            cur_temp = 0.5
    
    return best_a, best_obj


def coordinate_polish(a0, time_limit, multipliers=None):
    """Coordinate descent: try scaling each element by various multipliers."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    if multipliers is None:
        multipliers = np.array([0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 1.8, 2.5])
    
    improved_global = True
    while improved_global and time.monotonic() - t0 < time_limit:
        improved_global = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-14:
                continue
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved_global = True
    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

Wait, I have a bug in the LP solve. The normalization in the paper uses sum(a) = sqrt(2n), but I need to be careful about how I handle this.

Let me reconsider. The objective is:
f(a) = 2n * max(conv(a,a)) / (sum(a))^2

This is scale-invariant. So we can fix sum(a) = 1 and minimize 2n * max(conv(a,a)).

In the LP approach:
- Fix a (with sum = 1)
- M = max(conv(a,a))

def cross_convolve(f, g):
    """Compute (f*g)[k] for k = 0..2n-2 where f and g have the same length n."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]
