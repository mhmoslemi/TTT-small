def evaluate_sequence(sequence):
    """Evaluates a sequence of coefficients with the given metric.
    Returns np.inf if the input is invalid."""
    n = len(sequence)
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:
        return np.inf
    for x in sequence:
        if not isinstance(x, float) or isinstance(x, bool) or np.isnan(x) or np.isinf(x):
            return np.inf
    total_sum = np.sum(sequence)
    if total_sum < 0.01:
        return np.inf
    conv = np.convolve(sequence, sequence, mode='full')
    max_val = np.max(conv)
    return float(2 * n * max_val / (total_sum ** 2))

def project_simplex(v):
    """Project the probability vector onto the simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u * np.arange(1, n+1)) > (cssv - 1)
    if ~cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
