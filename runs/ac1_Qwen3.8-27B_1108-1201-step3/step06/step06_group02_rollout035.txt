import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def objective_normalized(a):
    """Compute 2*n*max(conv)/sum^2 for normalized a (sum=1)."""
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max()


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M, n, warm_start=None):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns (g, y_dual) where y_dual is the dual solution.
    """
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    options = {'presolve': True}
    if warm_start is not None:
        options['presolve'] = False
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options=options)
    if res.success:
        return res.x, res.ineqlin.marginals if hasattr(res.ineqlin, 'marginals') else None
    return None, None


def curvature_step_size(f, g, k_star):
    """
    Compute curvature-aware step size.
    f: current point (normalized)
    g: LP direction (normalized)
    k_star: active peak index
    Returns suggested alpha in [0.01, 1.0].
    """
    d = g - f
    # Second derivative of conv[k_star] along d
    d_conv_k = 0.0
    js = max(0, k_star - len(f) + 1)
    je = min(len(f) - 1, k_star)
    for j in range(js, je + 1):
        d_conv_k += 2.0 * d[j] * d[k_star - j]
    # Actually use FFT for efficiency
    conv_d = fast_convolve(d)
    second_deriv = 2.0 * conv_d[k_star]
    
    # First derivative (directional derivative of max conv)
    conv_f = fast_convolve(f)
    # Subgradient at f
    js = max(0, k_star - len(f) + 1)
    je = min(len(f) - 1, k_star)
    first_deriv = 0.0
    for j in range(js, je + 1):
        first_deriv += 2.0 * f[j] * d[k_star - j]
    
    if second_deriv > 1e-10:
        # Convex: use quadratic step
        alpha = -first_deriv / second_deriv
        alpha = min(max(alpha, 0.01), 1.0)
    elif second_deriv < -1e-10:
        # Concave: can take full step
        alpha = 1.0
    else:
        alpha = 0.3  # Default
    return alpha

        # Compute via FFT
        # g[i] = 2 * sum_{k=0}^{2n-2} w[k] * f[k-i] (where f[j]=0 outside [0,n-1])
        # This is the cross-correlation of w and f
        N = 1 << (2 * n - 1).bit_length()
        W = np.fft.rfft(w, N)
        F = np.fft.rfft(f, N)
        # cross-correlation: ifft(W * conj(F))
        cross = np.fft.irfft(W * np.conj(F), N)[:n]
        g = 2.0 * cross
        
        # Multiplicative update (mirror descent)
        f = f * np.exp(-lr * g)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()
        
        co = fast_convolve(f).max()
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 20 == 0:
                print(f"    PNorm iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
        
        # Adaptive learning rate
        if no_improve > 30:
            lr *= 0.8
            if no_improve > 100:
                # Perturbation
                rng = np.random.default_rng(it * 43 + 11)
                f = np.maximum(best_f + rng.standard_normal(n) * 0.001, 1e-15)
                f /= f.sum()
                lr = 0.01
                no_improve = 0
        if no_improve % 50 == 0 and no_improve > 0:
            lr *= 0.9
        
        if lr < 1e-10:
            break

    return best_f, best_obj


def peak_targeted_descent(f0, time_limit, verbose=False):
    """
    Peak-targeted coordinate descent: identifies top peaks and performs
    targeted mass redistribution to reduce them.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > 80:
            break

        conv = fast_convolve(f)
        M = conv.max()
        peak_k = int(np.argmax(conv))

        # Find contributing indices for the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        contrib_idx = np.arange(js, je + 1)
        if len(contrib_idx) == 0:
            break

        products = f[contrib_idx] * f[peak_k - contrib_idx]
        total_contrib = products.sum()
        if total_contrib < 1e-15:
            break

        max_prod_idx = contrib_idx[np.argmax(products)]
        partner_idx = peak_k - max_prod_idx

        if f[max_prod_idx] >= f[partner_idx]:
            source = max_prod_idx
            partner = partner_idx
        else:
            source = partner_idx
            partner = max_prod_idx

        non_contrib = np.ones(n, dtype=bool)
        non_contrib[contrib_idx] = False
        if non_contrib.sum() < 2:
            non_contrib[:] = True
            non_contrib[source] = False

        best_trial = None
        best_trial_obj = M

        for frac in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            diff = f[source] * frac
            if diff < 1e-15:
                continue
            trial = f.copy()
            trial[source] -= diff
            nc_vals = trial[non_contrib]
            nc_sum = nc_vals.sum()
            if nc_sum < 1e-15:
                continue
            trial[non_contrib] += diff * nc_vals / nc_sum
            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                continue
            trial = trial / ts
            obj = fast_convolve(trial).max()
            if obj < best_trial_obj - 1e-15:
                best_trial_obj = obj
                best_trial = trial

        if best_trial is not None:
            f = best_trial
            if best_trial_obj < best_obj - 1e-14:
                best_obj = best_trial_obj
                best_f = f.copy()
                no_improve = 0
                if verbose and no_improve == 0:
                    print(f"    PTD: peak_k={peak_k}, obj={best_obj:.10f}")
        else:
            no_improve += 1
            # Try other peaks
            top_peaks = np.argsort(conv)[-3:]
            found = False
            for pk in top_peaks:
                pk = int(pk)
                if pk == peak_k:
                    continue
                js2 = max(0, pk - n + 1)
                je2 = min(n - 1, pk)
                cidx2 = np.arange(js2, je2 + 1)
                if len(cidx2) == 0:
                    continue
                prods2 = f[cidx2] * f[pk - cidx2]
                src2 = cidx2[np.argmax(prods2)]
                for frac2 in [0.1, 0.2, 0.3]:
                    diff2 = f[src2] * frac2
                    trial = f.copy()
                    trial[src2] -= diff2
                    nc2 = np.ones(n, dtype=bool)
                    nc2[cidx2] = False
                    nc2_vals = trial[nc2]
                    nc2_sum = nc2_vals.sum()
                    if nc2_sum < 1e-15:
                        continue
                    trial[nc2] += diff2 * nc2_vals / nc2_sum
                    trial = np.maximum(trial, 0.0)
                    trial /= trial.sum()
                    obj2 = fast_convolve(trial).max()
                    if obj2 < best_obj - 1e-14:
                        f = trial
                        best_obj = obj2
                        best_f = f.copy()
                        no_improve = 0
                        found = True
                        break
                if found:
                    break
            if not found:
                no_improve += 1

    return best_f, best_obj


def coordinate_descent_polish(f0, time_limit, verbose=False):
    """
    Coordinate descent: try adjusting each element individually.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    f /= f.sum()
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit and no_improve < 200:
        # Pick a random coordinate to adjust
        i = rng.integers(0, n)
        if f[i] < 1e-10:
            no_improve += 1
            continue
        
        # Try moving mass from i to j
        j = rng.integers(0, n)
        if i == j:
            continue
        
        best_delta = 0.0
        best_obj_trial = best_obj
        
        for frac in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
            delta = f[i] * frac
            trial = f.copy()
            trial[i] -= delta
            trial[j] += delta
            if trial[i] < 0:
                continue
            trial /= trial.sum()
            obj = fast_convolve(trial).max()
            if obj < best_obj_trial - 1e-15:
                best_obj_trial = obj
                best_delta = delta
        
        if best_delta > 0:
            f[i] -= best_delta
            f[j] += best_delta
            f = np.maximum(f, 0.0)
            f /= f.sum()
            if best_obj_trial < best_obj:
                best_obj = best_obj_trial
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

    return best_f, best_obj


def make_init_gaussian(n, sigma_frac=0.25):
    """Gaussian-like initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_init_cosine(n, freq=1, amp=0.2):
    """Cosine-modulated uniform sequence."""
    x = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_init_multi_cosine(n, n_freqs=4, max_amp=0.25, seed=0):
    """Multi-frequency cosine-modulated sequence."""
    rng = np.random.default_rng(seed)
    a = np.ones(n)
    for m in range(1, n_freqs + 1):
        amp = rng.uniform(0.05, max_amp)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * m * np.arange(n) / n + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_init_flat_spectrum(n, seed=0):
    """
    Initialize with approximately flat power spectrum.
    Use random phases in frequency domain, then project to non-negative.
    """
    rng = np.random.default_rng(seed)
    # Generate random phases
    phases = rng.uniform(0, 2 * np.pi, size=n)
    # Build complex spectrum with unit magnitude and random phases
    # Use only real frequencies for real signal
    spectrum = np.exp(1j * phases)
    # Make it conjugate symmetric for real signal
    if n % 2 == 0:
        spectrum[n//2 + 1:] = np.conj(spectrum[n//2 - 1:0:-1])
        spectrum[n//2] = 1.0
    else:
        spectrum[(n+1)//2:] = np.conj(spectrum[(n-1)//2:0:-1])
    
    # Inverse FFT to get time domain
    a = np.real(np.fft.ifft(spectrum)) * n  # Scale to sum ~ n
    # Make non-negative: use soft thresholding
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        # Fallback
        a = np.maximum(a + 0.5, 0)
        s = a.sum()
    a /= s
    return a


def make_init_tukey(n, alpha=0.3):
    """Tukey (tapered cosine) window."""
    n_half = n // 2
    a = np.ones(n)
    for i in range(n_half):
        if i < int(alpha * n_half):
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * n_half) - 1)))
            a[n - 1 - i] = a[i]
        elif i > n_half - int(alpha * n_half):
            # Already set by symmetry
            pass
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
