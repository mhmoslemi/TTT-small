from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive-temperature gradient
descent with momentum, L-BFGS soft-max polishing, alternating subgradient
minimization, and peak-targeted local search. Exploits the structure of the
auto-convolution for efficient gradient computation via FFT-based cross-correlation.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Project vector v onto the probability simplex {x : x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(a, T):
    """
    Compute the gradient of the soft-max smoothed objective.
    f_smooth(a) = (1/T) * log(sum_k exp(T * conv(a,a)[k]))
    Returns gradient of f_smooth with respect to a (before normalization).
    """
    n = len(a)
    c = fast_convolve(a)
    # Softmax weights
    c_max = c.max()
    exp_c = np.exp(T * (c - c_max))
    w = exp_c / exp_c.sum()
    # Gradient: g[i] = 2 * sum_k w[k] * a[k-i]
    # = 2 * conv(w, a_rev)[n-1+i] for i = 0, ..., n-1
    a_rev = a[::-1]
    conv_w = cross_convolve(w, a_rev)
    g = 2.0 * conv_w[n-1:n-1+n]
    return g, w, c


def soft_max_obj(a, T):
    """Compute soft-max smoothed objective value."""
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    log_sum_exp = c_max + np.log(np.exp(T * (c - c_max)).sum()) / T
    return 2.0 * n * log_sum_exp


def score_from_conv(c, n, sum_a):
    """Compute score from convolution."""
    return 2.0 * n * c.max() / (sum_a ** 2)

                                momentum=0.9, lr_init=0.01, verbose=True):

    Adaptive-temperature gradient descent with momentum on the simplex.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    v = np.zeros(n)  # momentum
    t0 = time.monotonic()
    n_total = max(1, int((time_limit - 1) / 0.01))  # rough estimate
    iters = 0
    lr = lr_init
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        # Adaptive temperature
        progress = min(1.0, iters / max(1, int(time_limit * 50)))
        T = T_init * (T_final / T_init) ** progress
        
        # Compute gradient
        g, w, c = compute_gradient(a, T)
        
        # Account for normalization constraint
        # We're optimizing over the simplex, so we need to subtract the mean
        # (projection handles this, but for the gradient direction, centering helps)
        g_centered = g - np.mean(g)
        
        # Momentum update
        v = momentum * v - lr * g_centered
        
        # Candidate
        a_candidate = a + v
        a_candidate = project_simplex(a_candidate)
        
        # Evaluate
        max_conv = fast_convolve(a_candidate).max()
        score = 2.0 * n * max_conv
        
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_candidate.copy()
            a = a_candidate
        
        # Adaptive learning rate
        if iters % 50 == 0:
            lr *= 0.98
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    GD iter={iters} ({elapsed:.0f}s): T={T:.1f} score={best_score:.6f}")
        
        # Early stopping check
        if iters % 200 == 0:
            elapsed = time.monotonic() - t0
            if elapsed > time_limit * 0.9:
                break
    
    return best_a, best_score


def lbgfs_optimize(a0, time_limit, T=50, maxiter=5000):
    """
    L-BFGS optimization of the soft-max smoothed objective on the simplex.
    Uses log-space parameterization for automatic simplex constraint handling.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    
    # Log-space parameterization
    a_safe = np.maximum(a, 1e-10)
    z0 = np.log(a_safe / a_safe.sum())
    
    t0 = time.monotonic()
    stopped = [False]
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        # softmax
        z_shifted = z - z.max()
        a_tmp = np.exp(z_shifted)
        a_tmp = a_tmp / a_tmp.sum()
        return soft_max_obj(a_tmp, T)
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros_like(z)
        z_shifted = z - z.max()
        a_tmp = np.exp(z_shifted)
        a_tmp = a_tmp / a_tmp.sum()
        # Gradient in a-space
        g_a, w, c = compute_gradient(a_tmp, T)
        # Convert to z-space: grad_z = a * (grad_a - a * sum(a * grad_a))
        S = np.dot(a_tmp, g_a)
        grad_z = a_tmp * (g_a - a_tmp * S)
        return grad_z
    
    try:
        res = minimize(
            fun=func,
            x0=z0,
            jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                     'maxfun': maxiter * 50, 'disp': False}
        )
        z_out = res.x
        z_shifted = z_out - z_out.max()
        a_out = np.exp(z_shifted)
        a_out = a_out / a_out.sum()
    except Exception:
        a_out = a
    
    return a_out


def alternating_subgradient(a0, b0, time_limit, n_iter_max=50000, lr=0.01, verbose=True):
    """
    Alternating subgradient method for the bilinear problem.
    Alternates between optimizing a (given b) and b (given a).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    b = np.maximum(np.asarray(b0, dtype=np.float64), 0.0)
    s = b.sum()
    b = b / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    t0 = time.monotonic()
    iters = 0
    lr_cur = lr
    
    while time.monotonic() - t0 < time_limit and iters < n_iter_max:
        iters += 1
        # Update b given a: minimize max_k conv(a, b)[k]
        # Subgradient of max_k conv(a,b)[k] w.r.t. b is: a reversed/shifted
        ab_conv = cross_convolve(a, b)
        k_peak = int(np.argmax(ab_conv))
        # Subgradient w.r.t. b: the row of T corresponding to k_peak
        # (T @ b)[k_peak] = sum_i a[i] * b[k_peak - i]
        # grad_b[j] = a[k_peak - j] (with a[m] = 0 for m outside [0, n-1])
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        grad_b = np.zeros(n)
        grad_b[idx] = a[k_peak - idx]
        # Center and step
        grad_b = grad_b - np.mean(grad_b)
        b = project_simplex(b - lr_cur * grad_b)
        
        # Update a given b: similar
        ba_conv = cross_convolve(b, a)
        k_peak_a = int(np.argmax(ba_conv))
        js = max(0, k_peak_a - n + 1)
        je = min(n - 1, k_peak_a)
        idx = np.arange(js, je + 1)
        grad_a = np.zeros(n)
        grad_a[idx] = b[k_peak_a - idx]
        grad_a = grad_a - np.mean(grad_a)
        a = project_simplex(a - lr_cur * grad_a)
        
        # Evaluate
        max_conv = fast_convolve(a).max()
        score = 2.0 * n * max_conv
        if score < best_score - 1e-15:
            best_score = score
            best_a = a.copy()
        
        # Also evaluate the cross to get a different candidate
        abc = 0.5 * (a + b)
        abc = abc / abc.sum()
        max_conv_ab = fast_convolve(abc).max()
        score_ab = 2.0 * n * max_conv_ab
        if score_ab < best_score - 1e-15:
            best_score = score_ab
            best_a = abc.copy()
        
        # Decay learning rate
        if iters % 1000 == 0:
            lr_cur *= 0.5
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    AltSG iter={iters} ({elapsed:.0f}s): score={best_score:.6f} lr={lr_cur:.6f}")
    
    return best_a, best_score


def peak_shave(a0, time_limit, n_steps=10000):
    """
    Targeted peak shaving: identify the peak of the auto-convolution and
    try to reduce it by adjusting specific elements.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(a)
        peak_k = int(np.argmax(c))
        
        # Find contributing elements
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Get top contributors
        top_k = min(10, len(idx))
        top_contrib = np.argsort(contribs)[::-1][:top_k]
        
        improved = False
        for ci in top_contrib:
            i = idx[ci]
            j = peak_k - i
            if i == j or a[i] < 1e-12:
                continue
            
            # Try reducing a[i] and redistributing
            for factor in [0.95, 0.9, 0.85, 0.8]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-15:
                    continue
                
                # Redistribute to other elements (not i, not j)
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                mask[j] = False
                others = a[mask]
                if others.sum() < 1e-15:
                    continue
                
                a_trial = a.copy()
                a_trial[i] = new_val
                a_trial[mask] = others * (1.0 + diff / others.sum())
                a_trial = a_trial / a_trial.sum()
                
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try reducing both a[i] and a[j]
            for ci in top_contrib:
                i = idx[ci]
                j = peak_k - i
                if i == j:
                    continue
                for f1 in [0.95, 0.9, 0.8]:
                    for f2 in [0.95, 0.9, 0.8]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        s_trial = a_trial.sum()
                        if s_trial < 1e-15:
                            continue
                        a_trial = a_trial / s_trial
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            
            if not improved:
                # Random perturbation to escape local minimum
                rng = np.random.default_rng(step * 7919 + 13)
                a_trial = a + rng.standard_normal(n) * 0.002
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                else:
                    # Accept with small probability (simulated annealing)
                    if rng.random() < 0.01:
                        a = a_trial
                    else:
                        break  # Stuck, break out
    
    return best_a, best_score
