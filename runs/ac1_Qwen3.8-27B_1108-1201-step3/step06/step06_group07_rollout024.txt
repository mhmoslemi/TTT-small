Dual: min d^T y s.t. A^T y >= c, y >= 0
= M * sum_k y_k
s.t. sum_k A[k,j] * y_k >= 1 for all j
= M * sum_k y_k
s.t. sum_{k: 0<=k-j<n} a[j] * y_k >= 1 for all j
= M * sum_k y_k
s.t. a[j] * sum_{k=j}^{j+n-1} y_k >= 1 for all j (since A[k,j] = a[j] when 0 <= k-j < n, i.e., j <= k <= j+n-1)

So: sum_{k=j}^{j+n-1} y_k >= 1/a[j] for all j (assuming a[j] > 0).

And y_k >= 0.

This is a nice structure! The dual is asking for a non-negative sequence y (of length 2n-1) such that the sliding window sum of y over any window of length n starting at position j is at least 1/a[j], and we want to minimize the total sum of y.

This gives us an interpretation: y is like a "weight" on the convolution positions, and we want the total weight to be small while ensuring that each position j gets enough "coverage" from the y weights in its window.

By strong duality, the optimal value of the primal (max sum b_j) equals the optimal value of the dual (min M * sum y_k).

So if the optimal dual value is M * Y* where Y* = sum y_k*, then the optimal primal value is also M * Y*. The rescaled sequence b' = b / (M * Y*) will have sum 1, and we can check if its peak convolution is less than M.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based flow (novel), improved
mirror descent, aggressive coordinate descent, and diverse multi-start search.
Key innovation: uses linear programming to find "maximally spread" directions
compatible with the current peak constraint, providing a fundamentally different
optimization geometry than subgradient methods.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(a, peak_indices, weights):
    """Compute weighted subgradient of max_k (a*a)_k."""
    n = len(a)
    g = np.zeros(n)
    for idx in range(len(peak_indices)):
        k = int(peak_indices[idx])
        w_k = weights[idx]
        if w_k < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        g[js:je + 1] += 2.0 * w_k * a[k - np.arange(js, je + 1)]
    return g

def solve_lp_for_flow(a, M):
    """
    Solve: max sum(b) s.t. conv(a,b)_k <= M for all k, b >= 0.
    Returns optimal b and optimal value.
    """
    n = len(a)
    num_constraints = 2 * n - 1
    
    # Build constraint matrix A: A[k,j] = a[j] if 0 <= k-j < n, else 0
    # Use sparse format
    rows = []
    cols = []
    data = []
    for j in range(n):
        if a[j] < 1e-15:
            continue
        for k in range(j, j + n):
            if k < num_constraints:
                rows.append(k)
                cols.append(j)
                data.append(a[j])
    
    A = csr_matrix((data, (rows, cols)), shape=(num_constraints, n))
    
    # Objective: minimize -sum(b)
    c = -np.ones(n)
    b_ub = np.full(num_constraints, M)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        b_opt = np.maximum(result.x, 0.0)
        return b_opt, float(-result.fun)
    else:
        # Fallback: return a itself (which is always feasible)
        return a.copy(), float(a.sum())

                best_a = a_sym.copy()
                a = a_sym.copy()
        
        if lr < 1e-9:
            break
    
    return best_a, best_obj

def fine_coordinate_descent(a0, time_limit, verbose=False, label=""):
    """Fine coordinate descent with multiple trial values per coordinate."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            if old_val < 1e-12:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.8:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                trial = a.copy()
                trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                trial[mask] *= scale
                trial = np.maximum(trial, 0.0)
                ss = trial.sum()
                if ss < 1e-15:
                    continue
                trial = trial / ss
                val = fast_convolve(trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            # Try reverse order
            improved2 = False
            for j in range(n - 1, -1, -1):
                if time.monotonic() - t0 > time_limit:
                    break
                old_val = a[j]
                if old_val < 1e-12:
                    continue
                best_j_obj = best_obj
                best_j_val = old_val
                for mult in [0.6, 0.8, 0.9, 1.1, 1.25, 1.5]:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.8:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    trial = a.copy()
                    trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    trial[mask] *= scale
                    trial = np.maximum(trial, 0.0)
                    ss = trial.sum()
                    if ss < 1e-15:
                        continue
                    trial = trial / ss
                    val = fast_convolve(trial).max()
                    if val < best_j_obj - 1e-14:
                        best_j_obj = val
                        best_j_val = new_val
                if best_j_val != old_val:
                    denom = 1.0 - old_val
                    if denom > 1e-10:
                        scale = (1.0 - best_j_val) / denom
                        a[j] = best_j_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        a[mask] *= scale
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
                    if best_j_obj < best_obj - 1e-14:
                        best_obj = best_j_obj
                        best_a = a.copy()
                        improved2 = True
            if not improved2:
                break
    
    return best_a, best_obj

def paired_coordinate_descent(a0, time_limit, verbose=False, label=""):
    """Adjust pairs of coordinates simultaneously for faster convergence."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        # Find the peak position
        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        
        if je <= js:
            break
        
        # Try adjusting pairs within the peak-contributing range
        idx = np.arange(js, je + 1)
        products = a[idx] * a[k_peak - idx]
        
        # Find top contributors
        top_contrib = idx[np.argsort(products)[-min(6, len(idx)):]]
        
        for j in top_contrib:
            if time.monotonic() - t0 > time_limit:
                break
            j = int(j)
            old_val = a[j]
            if old_val < 1e-12:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in [0.3, 0.5, 0.7, 0.85, 0.95, 1.05, 1.2, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.8:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                trial = a.copy()
                trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                trial[mask] *= scale
                trial = np.maximum(trial, 0.0)
                val = fast_convolve(trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            # Fall back to standard coordinate descent
            for j in range(n):
                if time.monotonic() - t0 > time_limit:
                    break
                old_val = a[j]
                if old_val < 1e-12:
                    continue
                best_j_obj = best_obj
                best_j_val = old_val
                for mult in [0.7, 0.9, 1.1, 1.3]:
                    new_val = old_val * mult
                    if new_val < 0 or new_val > 0.8:
                        continue
                    denom = 1.0 - old_val
                    if denom < 1e-10:
                        continue
                    scale = (1.0 - new_val) / denom
                    trial = a.copy()
                    trial[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    trial[mask] *= scale
                    trial = np.maximum(trial, 0.0)
                    val = fast_convolve(trial).max()
                    if val < best_j_obj - 1e-14:
                        best_j_obj = val
                        best_j_val = new_val
                if best_j_val != old_val:
                    denom = 1.0 - old_val
                    if denom > 1e-10:
                        scale = (1.0 - best_j_val) / denom
                        a[j] = best_j_val
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        a[mask] *= scale
                        a = np.maximum(a, 0.0)
                        a = a / a.sum()
                    if best_j_obj < best_obj - 1e-14:
                        best_obj = best_j_obj
                        best_a = a.copy()
                        improved = True
            if not improved:
                break
    
    return best_a, best_obj

def make_gaussian(n, sigma_frac=0.3):
    """Create a Gaussian-like initial sequence normalized to sum=1."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def make_arcsine(n):
    """Arcsine distribution on [0,1] discretized."""
    x = (np.arange(n) + 0.5) / n
    x = np.clip(x, 1e-6, 1 - 1e-6)
    a = 1.0 / (np.pi * np.sqrt(x * (1.0 - x)))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def make_bell(n, width_frac=0.6):
    """Bell-shaped with flat top."""
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def make_bimodal(n, center1_frac=0.25, center2_frac=0.75, sigma=0.08):
    """Bimodal Gaussian."""
    x = np.arange(n) / n
    a = np.exp(-0.5 * ((x - center1_frac) / sigma) ** 2) + \
        np.exp(-0.5 * ((x - center2_frac) / sigma) ** 2)
    a = a / a.sum()
    return a

def make_sidelobe_free(n):
    """Try to create a sequence with low sidelobes using iterative method."""
    a = np.ones(n) / n
    for _ in range(50):
        conv = fast_convolve(a)
        M = conv.max()
        k = int(np.argmax(conv))
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        # Reduce elements contributing to peak
        idx = np.arange(js, je + 1)
        a[idx] *= 0.95
        a = np.maximum(a, 0)
        a = a / a.sum()
    return a

def symmetrize(a):
    """Symmetrize a sequence (make it palindromic)."""
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    ss = a_sym.sum()
    if ss > 1e-15:
        a_sym /= ss
    return a_sym

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, maintaining sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

            return float('inf')
