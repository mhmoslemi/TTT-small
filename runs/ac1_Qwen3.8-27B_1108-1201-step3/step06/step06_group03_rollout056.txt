def fast_convolve(a):

def project_nonnegative_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u * rho > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)

If we have a continuous function f(x) on [0,1] and sample it at n points, then:
- sum(a) ≈ n * integral(f) dx (if we normalize, sum=1 means integral = 1/n)
- b[k] ≈ (1/n^2) * sum_j a[j]*a[k-j] ≈ (1/n) * (f*f)(k/n) (approximately)
- max_b ≈ (1/n) * max(f*f)
- score ≈ 2*n*(1/n)*max(f*f) = 2*max(f*f)

So the score is approximately independent of n (for a continuous shape)! This means the length doesn't matter much for very large n, and we should focus on finding the right shape.

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_conv_max(a):
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    a_n = a / s
    return fast_convolve(a_n).max()


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    return f_new


def nesterov_smd(f0, time_limit, lr_init=0.015, momentum_init=0.9,
                  tau_init=8.0, tau_decay=1.004, tau_max=300.0,
                  peak_frac=0.2, restart_every=120, verbose=False):
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    y = f.copy()
    v = np.zeros(n)
    t0 = time.monotonic()
    tau = tau_init
    lr = lr_init
    mu = momentum_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        len_c = len(conv)
        k_top = max(8, int(len_c * peak_frac))
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        pv = conv[peaks]
        scaled = tau * pv
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        grad = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue
            idx = np.arange(js, je + 1)
            grad[js:je + 1] += w[i] * 2.0 * f[k - idx]

        # Nesterov step
        v = momentum_init * v + grad
        y = f - lr * v
        y = np.maximum(y, 1e-15)
        y = y / y.sum()

        # Evaluate
        obj_y = fast_convolve(y).max()
        if obj_y < M * 1.001:
            f = y
        else:
            # Backtrack: try intermediate points
            found = False
            for alpha in [0.5, 0.25, 0.1, 0.05]:
                f_trial = f * (1.0 - alpha) + y * alpha
                f_trial = np.maximum(f_trial, 1e-15)
                f_trial = f_trial / f_trial.sum()
                obj_trial = fast_convolve(f_trial).max()
                if obj_trial < M:
                    f = f_trial
                    found = True
                    break
            if not found:
                lr *= 0.5
                v = v * 0.5
                if lr < 1e-9:
                    break
                continue

        if obj_y < best_obj - 1e-14:
            best_obj = obj_y
            best_f = f.copy()
            no_improve = 0
            lr = min(lr * 1.005, 0.04)
            if verbose and it % 25 == 0:
                print(f"        NSMD iter {it}: 2n*max={best_obj * n:.6f}, tau={tau:.1f}, lr={lr:.5f}")
        else:
            no_improve += 1

        tau = min(tau * tau_decay, tau_max)
        if no_improve > 30 and no_improve % 30 == 0:
            mu = min(mu * 1.01, 0.97)

        if no_improve > restart_every:
            rng = np.random.default_rng(it * 13 + 7)
            scale = 0.002 + 0.004 * min(float(no_improve) / restart_every, 1.0)
            f = np.maximum(best_f + rng.standard_normal(n) * scale, 1e-15)
            f = f / f.sum()
            v = np.zeros(n)
            lr = lr_init * 0.5
            tau = tau_init
            no_improve = 0
            mu = momentum_init

    return best_f, best_obj


def solve_lp_dir(f, M, n):
    c_obj = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def coordinate_polish(f0, time_limit, verbose=False):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    epoch = 0

    while time.monotonic() - t0 < time_limit:
        epoch += 1
        improved = False
        rng = np.random.default_rng(epoch * 1000 + 42)
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj

            old_val = f[j]
            if old_val < 1e-10:
                # Try setting a small value
                for new_val in [0.001, 0.005, 0.01]:
                    f_trial = f.copy()
                    f_trial[j] = new_val
                    f_trial = np.maximum(f_trial, 0)
                    st = f_trial.sum()
                    if st < 1e-15:
                        continue
                    f_trial = f_trial / st
                    val = fast_convolve(f_trial).max()
                    if val < best_obj - 1e-14:
                        f = f_trial
                        best_obj = val
                        best_f = f.copy()
                        improved = True
                        break
                continue

            best_j_val = old_val
            best_j_obj = best_obj
            denom = 1.0 - old_val
            if denom < 1e-10:
                continue

            for frac in [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0, 3.0]:
                new_val = old_val * frac
                if new_val < 0 or new_val > 0.5:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= denom / (1.0 - new_val) if (1.0 - new_val) > 1e-10 else 1.0
                f_trial = np.maximum(f_trial, 0)
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial = f_trial / st
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale = denom / (1.0 - best_j_val) if (1.0 - best_j_val) > 1e-10 else 1.0
                f[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] *= scale
                f = np.maximum(f, 0)
                f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

    return best_f, best_obj


def peak_directed_descent(f0, time_limit, verbose=False):
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = 0.008
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Focus on top 5 peaks
        top_k = 5
        peaks = np.argsort(conv)[-top_k:]

        grad = np.zeros(n)
        for k in peaks:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue
            idx = np.arange(js, je + 1)
            grad[js:je + 1] += 2.0 * f[k - idx]

        grad /= top_k

        f_new = f * np.exp(-lr * grad)
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()

        obj_new = fast_convolve(f_new).max()
        if obj_new < M:
            f = f_new
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.01, 0.03)
                if verbose and it % 30 == 0:
                    print(f"        PDD iter {it}: 2n*max={best_obj * n:.6f}")
        else:
            lr *= 0.7
            no_improve += 1

        if no_improve > 80:
            rng = np.random.default_rng(it * 23 + 11)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.003, 1e-15)
            f = f / f.sum()
            lr = 0.008
            no_improve = 0

        if lr < 1e-9:
            break

    return best_f, best_obj

                f_init = 0.7 * best_seq + 0.3 * asym
                f_init = f_init / f_init.sum()
            elif r == 2:
                f_init = 0.5 * best_seq + 0.5 * best_seq[::-1]
                f_init = f_init / f_init.sum()
            elif r == 3:
                # Structural perturbation: shift mass
                shift = int(rng.integers(n_best // 20, n_best // 8))
                f_init = np.roll(best_seq, shift)
                f_init = np.maximum(f_init, 0)
                f_init = f_init / f_init.sum()
            else:
                noise = rng.standard_normal(n_best) * 0.008
                f_init = np.maximum(best_seq + noise, 1e-15)
                f_init = f_init / f_init.sum()

            print("  Restart {} (n={}): t={:.0f}s".format(r, n_best, t_alloc))
            a_opt, _ = nesterov_smd(f_init, time_limit=t_alloc,
                                    lr_init=0.01, tau_init=5.0, tau_decay=1.006,
                                    restart_every=80)
            try_seq(a_opt, "restart_{}".format(r))

    # Phase 7: Coordinate polish
    print("\n=== Phase 7: Coordinate Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 50
        if rem > 20:
            t_alloc = min(rem, 80)
            print("  CD (n={}): t={:.0f}s".format(len(best_seq), t_alloc))
            a_opt, _ = coordinate_polish(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, "cd")

    # Phase 8: Final LP-FW squeeze
    print("\n=== Phase 8: Final Squeeze ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            t_alloc = min(rem, 40)
            print("  Final LP-FW: t={:.0f}s".format(t_alloc))
            a_opt, _ = lp_fw(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_lpfw")

    # Phase 9: Final NSMD with high temperature
    if best_seq is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            t_alloc = min(rem, 20)
            print("  Final NSMD (high tau): t={:.0f}s".format(t_alloc))
            a_opt, _ = nesterov_smd(best_seq, time_limit=t_alloc,
                                    lr_init=0.003, tau_init=50.0,
                                    tau_decay=1.002, tau_max=500.0,
                                    restart_every=40)
            try_seq(a_opt, "final_nsmd")

    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print("\nFINAL SCORE: {:.6f} (n={})".format(best_score, len(best_seq)))
    return [float(x) for x in best_seq]
