def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def subgradient_refine(a0, time_limit, lr_init=0.008, top_k=50):
    """Subgradient method with momentum for fine-tuning."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    v = np.zeros(n)
    mom = 0.92
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    step_count = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        # Temperature for softmax weighting of peaks
        temp = 0.005 * max(pv.max(), 1e-10)
        scaled = (pv - pv.max()) / temp
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

        v = mom * v - lr * sg

        # Project onto simplex
        a_candidate = a + v
        # Simplex projection
        u = np.sort(a_candidate)[::-1]
        cssv = np.cumsum(u)
        cond = u * np.arange(1, n + 1) > (cssv - 1)
        if cond.any():
            rho = np.nonzero(cond)[0][-1]
            theta = (cssv[rho] - 1.0) / (rho + 1.0)
            a_new = np.maximum(a_candidate - theta, 0.0)
        else:
            a_new = np.ones(n) / n

        a_new = a_new / a_new.sum()
        max_conv = fast_convolve(a_new).max()
        score = 2.0 * n * max_conv
        step_count += 1

        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9995
        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 100000)
            a = best_a + rng.standard_normal(n) * 0.003
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            lr = max(lr * 3, 0.002)
            no_improve = 0

        if step_count % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad {elapsed:.0f}s: score={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score


def peak_shave_local(a0, time_limit, n_steps=2000):

    best_a = a.copy()

    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        # Find the peak (excluding trivial endpoints)
        b_mid = b.copy()
        b_mid[0] = -np.inf
        b_mid[-1] = -np.inf
        peak_k = int(np.argmax(b_mid))
        if b[peak_k] <= best_score / (2.0 * n) + 1e-15:
            break

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_contrib = np.argsort(contribs)[::-1][:3]

        improved = False
        for ci in top_contrib:
            i = idx[ci]
            j = peak_k - i
            old_val = a[i]
            if old_val < 1e-14:
                continue
            for factor in [0.95, 0.9, 0.8, 0.7]:
                new_val = old_val * factor
                diff = old_val - new_val
                if diff < 1e-15:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a[mask]
                if others.sum() < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                a_trial[mask] = others + diff * others / others.sum()
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try reducing both i and j
            for ci in top_contrib:
                i = idx[ci]
                j = peak_k - i
                if i == j or a[j] < 1e-14:
                    continue
                for f1 in [0.9, 0.8]:
                    for f2 in [0.9, 0.8]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-13:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if not improved:
                break

def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def cross_corr(w, a):
    """Cross-correlation for gradient computation."""
    n = len(a)
    m = len(w)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    return irfft(np.conj(W) * A, fft_size)[:n]


def soft_max_obj_grad(z, T, n):
    """Soft-max smoothed objective and gradient in log-space."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    w = exp_Tb / exp_Tb.sum()
    grad_a = 2.0 * n * cross_corr(w, a)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    return obj, grad_z


def lbgfs_refine(a0, time_limit, T=200, maxiter=3000):
    """L-BFGS refinement in log-space with soft-max smoothing."""
    from scipy.optimize import minimize
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    stopped = [False]

    def func(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        obj, _ = soft_max_obj_grad(z, T, n)
        return obj

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros_like(z)
        _, g = soft_max_obj_grad(z, T, n)
        return g

    try:
        res = minimize(
            fun=func,
            x0=z0,
            jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 30, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
    except Exception:
        a_out = a0

    return a_out


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving non-negativity and sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
