    Ar = rfft(a_rev, fft_size)
    conv_result = irfft(Ar * W, fft_size)
    # H[l] = conv_result[n-1+l] for l = 0, ..., n-1
    H = conv_result[n - 1:n - 1 + n]
    return H


def smooth_obj_and_grad(z, T, n):
    """
    Compute smoothed objective and gradient in log-space.
    f(z) = 2n * (1/T) * log(sum_k exp(T * b[k]))
    where b = conv(softmax(z), softmax(z)).
    """
    a = softmax_fn(z)
    b = fast_convolve(a)

    # Log-sum-exp
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    lse = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * lse

    # Softmax weights
    w = exp_Tb / exp_Tb.sum()

    # Gradient computation
    # H[l] = 2 * sum_k w[k] * a[k-l] (approximately)
    H = 2.0 * cross_corr_for_grad(w, a)

    # S = 2 * sum_k w[k] * b[k] (approximately)
    S = 2.0 * np.dot(w, b)

    # grad_z[l] = 2n * a[l] * (H[l] - S)
    grad = 2.0 * n * a * (H - S)

    return obj, grad


def annealed_lbgfs(a0, time_limit, T_start=2000, T_end=20, n_T_steps=8, verbose=True):
    """
    Annealed L-BFGS in log-space. Start with large T (smooth), decrease T
    to approach the true max objective.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = compute_score_internal(a)
    z = log_from_a(a)
    t0 = time.monotonic()

    # Create T schedule (geometric)
    T_schedule = np.geomspace(T_start, T_end, n_T_steps)

    rem_time = time_limit - (time.monotonic() - t0)
    time_per_T = rem_time / n_T_steps

    for Ti_idx, T in enumerate(T_schedule):
        T = float(T)
        elapsed = time.monotonic() - t0
        if elapsed > time_limit * 0.85:
            break
        t_alloc = min(time_per_T * (1 + 0.5 * Ti_idx), time_limit - elapsed - 10)
        if t_alloc < 5:
            break

        stopped = [False]

        def func(z):
            if time.monotonic() - t0 > time_limit * 0.95:
                stopped[0] = True
                return 1e15
            obj, _ = smooth_obj_and_grad(z, T, n)
            return obj

        def jac(z):
            if time.monotonic() - t0 > time_limit * 0.95:
                stopped[0] = True
                return np.zeros_like(z)
            _, g = smooth_obj_and_grad(z, T, n)
            return g

        try:
            max_iter = max(100, int(t_alloc * 50))
            res = minimize(
                fun=func,
                x0=z,
                jac=jac,
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-12,
                    'maxfun': max_iter * 50,
                    'maxls': 50,
                    'disp': False
                }
            )
            z = res.x
        except Exception:
            pass

        a_new = softmax_fn(z)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
            score = compute_score_internal(a_new)
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
            if verbose:
                print(f"    AnnealL-BFGS T={T:.0f}: score={best_score:.8f} (elapsed={time.monotonic()-t0:.0f}s)")

    return best_a, best_score


def adam_log_optimize(a0, time_limit, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8,
                      T=100, verbose=True):
    """Adam optimizer in log-space with soft-max smoothing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    z = log_from_a(a)
    best_a = a.copy()
    best_score = compute_score_internal(a)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        obj, grad = smooth_obj_and_grad(z, T, n)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = softmax_fn(z)
        score = compute_score_internal(a_new)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Decay learning rate
        lr *= 0.9999

        # Restart if stuck
        if no_improve > 2000:
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 1000000)
            z = log_from_a(best_a) + rng.standard_normal(n) * 0.01
            m = np.zeros(n)
            v = np.zeros(n)
            lr = 0.005
            no_improve = 0

        if t % 5000 == 0:
            if verbose:
                print(f"    AdamLog t={t}: score={best_score:.8f} lr={lr:.6f} (elapsed={time.monotonic()-t0:.0f}s)")

    return best_a, best_score


def spectral_shape_pass(a0, n_iter=50, gamma=0.3, verbose=True):
    """
    Spectral shaping: iteratively flatten the auto-correlation by modifying
    the power spectrum and finding a non-negative spectral factor.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = compute_score_internal(a)
    t0 = time.monotonic()
    L_fft = 1 << (2 * n - 1).__ class __.bit_length if False else 1 << (2 * n - 1 - 1).bit_length()

    for it in range(n_iter):
        if time.monotonic() - t0 > 60:
            break
        # Compute current auto-correlation
        b = fast_convolve(a)
        b_max = b.max()
        b_mean = b.mean()

        # Form target: compress the peaks
        b_target = b.copy()
        # Reduce peaks, boost valleys
        b_target = (1 - gamma) * b + gamma * b_mean * np.ones_like(b)

        # Ensure non-negative
        b_target = np.maximum(b_target, 0)

        # Get target spectrum
        A = rfft(a, L_fft)
        S = np.abs(A) ** 2
        B = rfft(b_target, L_fft)
        # New target spectrum (should be close to |A|^2 but with b flattened)
        # We want |A_new|^2 such that IDFT(|A_new|^2) = b_target
        # So A_new^2 = DFT(b_target)
        # But DFT(b_target) might not be non-negative
        S_new = rfft(b_target, L_fft)
        S_new = np.maximum(S_new.real, 0)  # Ensure non-negative (real part of |A|^2 should be real)

        # Actually, for real a, |A(k)|^2 = |A(L-k)|^2 and is real and non-negative.
        # DFT of auto-correlation IS |A|^2, so it should be non-negative.
        # Let's just use it directly.
        S_target = rfft(b_target, L_fft)
        S_target = np.maximum(S_target, 0)

        # Current amplitude and phase
        amp = np.sqrt(np.maximum(S, 0))
        phase = np.angle(A)

        # Target amplitude
        amp_target = np.sqrt(S_target)

        # Blend: move toward target amplitude, keep phase
        A_new = amp_target * np.exp(1j * phase)

        # Time domain
        a_new = irfft(A_new, L_fft)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        score = compute_score_internal(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller gamma
            break

        if verbose and it % 10 == 0:
            print(f"    SpectralShape iter={it}: score={best_score:.8f}")

    return best_a, best_score


def profile_flatten(a0, time_limit, n_steps=5000, verbose=True):
    """
    Profile-flattening coordinate descent:
    At each step, find the peak, identify the top contributors,
    and redistribute mass to flatten the profile.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = compute_score_internal(a)
    t0 = time.monotonic()
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 200:
            break

        b = fast_convolve(a)
        # Find peak (exclude endpoints which are just a[0]^2, a[n-1]^2)
        b_search = b.copy()
        b_search[0] = -np.inf
        b_search[-1] = -np.inf
        peak_k = int(np.argmax(b_search))

        # Get contributors
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        total_contrib = contribs.sum()
        if total_contrib < 1e-15:
            break

        # Find top contributor
        top_i = idx[np.argmax(contribs)]
        top_j = peak_k - top_i

        # Try to move mass from top_i to a "quiet" location
        # Find a location that minimizes the increase in other peaks
        # Heuristic: find index with smallest a value that is far from the peak
        candidates = np.arange(n)
        # Exclude indices near the peak that would create new peaks
        dist = np.abs(candidates - top_i)
        # Prefer far indices with small values
        score_vals = a[candidates] / (1.0 + 0.1 * dist)
        quiet_idx = candidates[np.argmin(score_vals)]

        if quiet_idx == top_i or a[quiet_idx] > a[top_i]:
            no_improve += 1
            continue

        # Move a small fraction of mass
        for frac in [0.01, 0.005, 0.002]:
            delta = a[top_i] * frac
            a_trial = a.copy()
            a_trial[top_i] -= delta
            a_trial[quiet_idx] += delta
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            score = compute_score_internal(a_trial)
            if score < best_score - 1e-14:
                best_score = score
                best_a = a_trial.copy()
                a = a_trial
                no_improve = 0
                break

        if verbose and step % 500 == 0:
            print(f"    ProfileFlat step={step}: score={best_score:.8f} (elapsed={time.monotonic()-t0:.0f}s)")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Load parent / initialize
    2. Annealed L-BFGS with analytic gradients
    3. Adam optimization in log-space
    4. Spectral shaping
    5. Profile-flattening coordinate descent
    6. Multi-length exploration
    7. Final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score
    else:
        # Initialize with a cosine-shaped distribution (better than uniform)
        n_init = 768
        t = np.linspace(0, np.pi, n_init)
        a_init = 0.5 * (1 + np.cos(t))
        a_init = np.maximum(a_init, 0)
        a_init = a_init / a_init.sum()
        best_seq = a_init
        best_score = float(evaluate_sequence(a_init.tolist()))
        print(f"Initialized cosine: n={n_init}, score={best_score:.8f}")

    # === Phase 1: Annealed L-BFGS (main workhorse) ===
    print(f"\n=== Phase 1: Annealed L-BFGS (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_alloc = min(rem * 0.35, 600.0)
        print(f"  Budget: {t_alloc:.0f}s, n={len(best_seq)}")
        a_lb, score_lb = annealed_lbgfs(
            best_seq, t_alloc,
            T_start=5000, T_end=10, n_T_steps=6,
            verbose=True
        )
        best_seq, best_score = check_and_update(a_lb, "anneal_lbgfs", best_seq, best_score)

    # === Phase 2: Adam optimization ===
    print(f"\n=== Phase 2: Adam Log-Space (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_alloc = min(rem * 0.2, 300.0)
        print(f"  Budget: {t_alloc:.0f}s")
        # Try different T values
        for T_val in [50, 200]:
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            t_a = min(rem * 0.4, t_alloc / 2)
            a_ad, score_ad = adam_log_optimize(
                best_seq, t_a, lr=0.005, T=T_val, verbose=True
            )
            best_seq, best_score = check_and_update(a_ad, f"adam_T{T_val}", best_seq, best_score)

    # === Phase 3: Spectral shaping ===
    print(f"\n=== Phase 3: Spectral Shaping (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        for gamma in [0.2, 0.4, 0.6]:
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            a_sp, score_sp = spectral_shape_pass(best_seq, n_iter=30, gamma=gamma, verbose=True)
            best_seq, best_score = check_and_update(a_sp, f"spectral_g{gamma}", best_seq, best_score)

    # === Phase 4: Profile flattening ===
    print(f"\n=== Phase 4: Profile Flattening (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_alloc = min(rem * 0.2, 120.0)
        a_pf, score_pf = profile_flatten(best_seq, t_alloc, n_steps=10000, verbose=True)
        best_seq, best_score = check_and_update(a_pf, "profile_flat", best_seq, best_score)

    # === Phase 5: Second L-BFGS round with tighter T ===
    print(f"\n=== Phase 5: Fine L-BFGS (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_alloc = min(rem * 0.25, 300.0)
        print(f"  Budget: {t_alloc:.0f}s, n={len(best_seq)}")
        a_lb2, score_lb2 = annealed_lbgfs(
            best_seq, t_alloc,
            T_start=500, T_end=5, n_T_steps=4,
            verbose=True
        )
        best_seq, best_score = check_and_update(a_lb2, "fine_lbgfs", best_seq, best_score)

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-Length (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_best = len(best_seq)
        alt_lengths = []
        for factor in [1.25, 1.5, 0.75, 2.0]:
            alt_n = int(n_best * factor)
            if 256 <= alt_n <= 4096 and alt_n != n_best:
                alt_lengths.append(alt_n)
        alt_lengths = sorted(set(alt_lengths))[:3]

        for alt_n in alt_lengths:
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            t_alloc = min(rem * 0.3, 200.0)
            print(f"  Trying n={alt_n}, budget={t_alloc:.0f}s")
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, score_alt = annealed_lbgfs(
                init_alt, t_alloc,
                T_start=2000, T_end=20, n_T_steps=4,
                verbose=False
            )
            best_seq, best_score = check_and_update(a_alt, f"multilength_n{alt_n}", best_seq, best_score)

            # Quick Adam on new best
            if best_score < 1.51:
                rem = deadline - time.monotonic()
                if rem > 20:
                    a_ad2, _ = adam_log_optimize(best_seq, min(rem * 0.3, 60.0),
                                                  lr=0.003, T=50, verbose=False)
                    best_seq, best_score = check_and_update(a_ad2, f"adam_ml_n{alt_n}",
                                                            best_seq, best_score)

    # === Phase 7: Final micro-polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t_alloc = min(rem * 0.5, 60.0)
        a_fin, _ = annealed_lbgfs(best_seq, t_alloc,
                                   T_start=200, T_end=3, n_T_steps=3,
                                   verbose=True)
        best_seq, best_score = check_and_update(a_fin, "final_lbgfs", best_seq, best_score)

    rem = deadline - time.monotonic()
    if rem > 10:
        t_pf = min(rem * 0.6, 30.0)
        a_pf2, _ = profile_flatten(best_seq, t_pf, n_steps=5000, verbose=False)
        best_seq, best_score = check_and_update(a_pf2, "final_pf", best_seq, best_score)

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.8f}")

    return [float(x) for x in best_seq]
