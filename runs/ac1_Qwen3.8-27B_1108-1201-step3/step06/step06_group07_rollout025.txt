"""
LP-based optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.

import numpy as np

from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Linear cross-convolution a * b via FFT."""
    n, m = len(a), len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, peak_indices, weights):
    """Compute weighted subgradient of max_k (a*a)_k."""
    n = len(a)
    g = np.zeros(n)
    for idx in range(len(peak_indices)):
        k = int(peak_indices[idx])
        w_k = weights[idx]
        if w_k < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        g[js:je + 1] += 2.0 * w_k * a[k - np.arange(js, je + 1)]
    return g


def solve_lp_direction(a, M, verbose=False):
    """
    Solve LP: maximize sum(g) subject to cross_convolve(a, g)_k <= M for all k, g >= 0.
    Returns the optimal g (normalized to sum=1) or None if infeasible.
    """
    n = len(a)
    num_constraints = 2 * n - 1
    
    # Build constraint matrix: A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # This is a Toeplitz-like matrix
    # For efficiency, build it as a sparse-like structure
    # Actually for scipy linprog, we need dense or csc/csr
    from scipy.sparse import csr_matrix
    
    # Build the matrix
    rows = []
    cols = []
    vals = []
    for k in range(num_constraints):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            i = k - j
            if 0 <= i < n:
                rows.append(k)
                cols.append(j)
                vals.append(a[i])
    
    A = csr_matrix((vals, (rows, cols)), shape=(num_constraints, n))
    
    # LP: maximize sum(g) = -minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: A @ g <= M
    b = np.full(num_constraints, M)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if result.success:
            g = result.x
            s = g.sum()
            if s > 1e-15:
                return g / s, s
            else:
                return None, 0
        else:
            return None, 0
    except Exception:
        return None, 0

def mirror_descent(a0, time_limit, top_k=30, temp_init=0.0005, lr_init=0.01,

def coordinate_descent(a0, time_limit, verbose=False, label=""):
    """Coordinate descent with binary search on each coordinate."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            if old_val < 1e-14:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.8:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                trial = a.copy()
                trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                trial[mask] *= scale
                trial = np.maximum(trial, 0.0)
                ss = trial.sum()
                if ss < 1e-15:
                    continue
                trial = trial / ss
                val = fast_convolve(trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_a, best_obj


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def symmetrize(a):
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    ss = a_sym.sum()
    if ss > 1e-15:

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

LP-based optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.

import numpy as np

from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, peak_indices, weights):
    """Compute weighted subgradient of max_k (a*a)_k."""
    n = len(a)
    g = np.zeros(n)
    for idx in range(len(peak_indices)):
        k = int(peak_indices[idx])
        w_k = weights[idx]
        if w_k < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        g[js:je + 1] += 2.0 * w_k * a[k - np.arange(js, je + 1)]
    return g


def build_lp_matrix(a):
    """Build the constraint matrix for the LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    num_constraints = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(j, min(j + n, num_constraints)):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(num_constraints, n))
    return A


def solve_lp_direction(a, M):
    """
    Solve LP: maximize sum(g) subject to cross_conv(a, g)_k <= M, g >= 0.
    Returns normalized g or None.
    """
    n = len(a)
    num_constraints = 2 * n - 1
    
    A = build_lp_matrix(a)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b = np.full(num_constraints, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if result.success and result.fx < -1e-15:
            g = np.maximum(result.x, 0)
            s = g.sum()
            if s > 1e-15:
                return g / s
        return None
    except Exception:
        return None

Primary method: LP-based direction finding (solves max sum(g) s.t. cross_conv(a,g)<=M).
Secondary: mirror descent with adaptive temperature and backtracking.
Polish: coordinate descent + symmetrization.

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, peak_indices, weights):
    """Weighted subgradient of max_k (a*a)_k."""
    n = len(a)
    g = np.zeros(n)
    for idx in range(len(peak_indices)):
        k = int(peak_indices[idx])
        w_k = weights[idx]
        if w_k < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        g[js:je + 1] += 2.0 * w_k * a[k - np.arange(js, je + 1)]
    return g


def build_lp_matrix(a):
    """Build sparse constraint matrix for cross-convolution LP."""
    n = len(a)
    num_constraints = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(j, min(j + n, num_constraints)):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(num_constraints, n))


def solve_lp_direction(a, M):
    """
    Solve: max sum(g) s.t. cross_conv(a,g)_k <= M for all k, g >= 0.
    Returns normalized g (sum=1) or None if infeasible/failed.
    """
    n = len(a)
    num_constraints = 2 * n - 1
    A = build_lp_matrix(a)
    c = -np.ones(n)
    b = np.full(num_constraints, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if result.success and result.x is not None:
            g = np.maximum(result.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return None

    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    """Flat-topped bell shape."""
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bimodal(n, c1=0.3, c2=0.7, sigma=0.08):
    """Two-lobe Gaussian."""
    x = np.arange(n) / n
    a = np.exp(-0.5 * ((x - c1) / sigma) ** 2) + np.exp(-0.5 * ((x - c2) / sigma) ** 2)
    a = a / a.sum()
    return a


def symmetrize(a):
    """Make sequence palindromic."""
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    ss = a_sym.sum()
    if ss > 1e-15:
        a_sym /= ss
    return a_sym


def interpolate_sequence(f, new_n):
    """Resample sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-method optimization for step function minimization.
    Phase 1: Diverse initialization.
    Phase 2: LP-based direction finding (primary, 40% time).
    Phase 3: Mirror descent (20% time).
    Phase 4: Second LP pass from symmetrized best (15% time).
    Phase 5: Coordinate descent polish (10% time).
    Phase 6: Multi-restart + different lengths (remaining time).
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    n_base = len(parent) if parent is not None else 768

    # === Phase 1: Initialization ===
    print("=== Phase 1: Initialization ===")
    init_candidates = []
    init_candidates.append((make_gaussian(n_base, 0.22), "g22"))
    init_candidates.append((make_gaussian(n_base, 0.30), "g30"))
    init_candidates.append((make_gaussian(n_base, 0.38), "g38"))
    init_candidates.append((make_bell(n_base, 0.5), "bell50"))
    init_candidates.append((make_bell(n_base, 0.65), "bell65"))
    init_candidates.append((make_bimodal(n_base), "bimodal"))
    if parent is not None:
        init_candidates.append((parent, "parent"))
        init_candidates.append((symmetrize(parent), "parent_sym"))

    best_init = None
    best_init_obj = float('inf')
    for a_init, label in init_candidates:
        s = a_init.sum()
        if s < 1e-10:
            continue
        obj = try_seq(a_init / s, f"init_{label}")
        conv_obj = 2 * len(a_init) * fast_convolve(a_init / s).max()
        if conv_obj < best_init_obj:
            best_init_obj = conv_obj
            best_init = a_init / s

    if best_init is None:
        best_init = make_gaussian(n_base, 0.3)

    print(f"  Best initialization obj: {best_init_obj:.6f}")

    # === Phase 2: LP-based optimization (primary) ===
    print(f"\n=== Phase 2: LP Optimization ===")
    remaining = deadline - time.monotonic()
    t_lp = remaining * 0.40
    print(f"  Allocated: {t_lp:.0f}s")
    a_opt, obj = lp_optimize(best_init, time_limit=t_lp, t_init=0.3,
                             verbose=True, label="LP1")
    try_seq(a_opt, "LP1")

    # === Phase 3: Mirror Descent ===
    print(f"\n=== Phase 3: Mirror Descent ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_md = remaining * 0.20
        print(f"  Allocated: {t_md:.0f}s")
        a_opt, obj = mirror_descent(best_seq, time_limit=t_md, top_k=30,
                                    temp_init=0.0003, lr_init=0.012,
                                    verbose=True, label="MD1")
        try_seq(a_opt, "MD1")

    # === Phase 4: Second LP pass (from symmetrized) ===
    print(f"\n=== Phase 4: LP Pass 2 ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_lp2 = remaining * 0.18
        print(f"  Allocated: {t_lp2:.0f}s")
        a_sym = symmetrize(best_seq)
        a_opt, obj = lp_optimize(a_sym, time_limit=t_lp2, t_init=0.2,
                                 verbose=True, label="LP2")
        try_seq(a_opt, "LP2")

    # === Phase 5: Coordinate Descent ===
    print(f"\n=== Phase 5: Coordinate Descent ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_cd = remaining * 0.12
        print(f"  Allocated: {t_cd:.0f}s")
        a_opt, obj = coordinate_descent(best_seq, time_limit=t_cd,
                                        verbose=True, label="CD1")
        try_seq(a_opt, "CD1")

    # === Phase 6: Multi-restart and length exploration ===
    print(f"\n=== Phase 6: Multi-Restart / Lengths ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            # Try a few noise restarts
            n_restarts = 3
            t_per = min(remaining * 0.12 / n_restarts, 90)
            for r in range(n_restarts):
                remaining = deadline - time.monotonic()
                if remaining < 40:
                    break
                if r == 0:
                    f_init = best_seq.copy()
                elif r == 1:
                    noise = rng.standard_normal(len(best_seq)) * 0.003
                    f_init = np.maximum(best_seq + noise, 1e-15)
                    f_init = f_init / f_init.sum()
                else:
                    f_init = 0.85 * best_seq + 0.15 * (np.ones(len(best_seq)) / len(best_seq))
                    f_init = f_init / f_init.sum()

                t_alloc = min(t_per, remaining - 30)
                if t_alloc < 15:
                    break
                a_opt, obj = mirror_descent(f_init, time_limit=t_alloc, top_k=25,
                                            temp_init=0.0004, lr_init=0.008,
                                            verbose=False, label=f"R{r}")
                try_seq(a_opt, f"restart{r}")

            # Try different lengths
            for n_try in [600, 900, 1024]:
                remaining = deadline - time.monotonic()
                if remaining < 40:
                    break
                a_interp = interpolate_sequence(best_seq, n_try)
                t_sub = min(remaining * 0.15, 100)
                remaining = deadline - time.monotonic()
                if remaining < 20:
                    break
                a_opt, obj = mirror_descent(a_interp, time_limit=t_sub, top_k=25,
                                            temp_init=0.0005, lr_init=0.008,
                                            verbose=False, label=f"n{n_try}")
                try_seq(a_opt, f"len{n_try}")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            # Fine MD
            t_polish = min(remaining - 10, 80)
            a_opt, obj = mirror_descent(best_seq, time_limit=t_polish, top_k=8,
                                        temp_init=0.00005, lr_init=0.003,
                                        verbose=True, label="polish")
            try_seq(a_opt, "polish")

            # Symmetrize
            a_s = symmetrize(best_seq)
            try_seq(a_s, "sym_final")

            # Quick CD
            if time.monotonic() < deadline - 20:
                t_cd2 = min(deadline - time.monotonic() - 10, 50)
                a_opt, obj = coordinate_descent(best_seq, time_limit=t_cd2,
                                                verbose=False, label="CD2")
                try_seq(a_opt, "CD2")
                # Symmetrize again after CD
                try_seq(symmetrize(a_opt), "CD2_sym")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
