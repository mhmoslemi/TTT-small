import numpy as np
from scipy.fft import rfft, irfft

import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_conv(a, b):
    """Linear cross-convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_objective(a):
    """Compute 2n*max(conv)/(sum)^2 for a given sequence."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S * S))


def compute_grad_active(a, active_peaks):
    """
    Compute the subgradient at active peaks.
    grad_j = 2 * sum_k w_k * a[k-j] (with zero padding)
    Uses uniform weights over active peaks.
    """
    n = len(a)
    g = np.zeros(n)
    # a_ext[m] = a[m] if 0<=m<n else 0
    # (grad b_k)_j = 2 * a_ext[k-j]
    # So g_j = 2 * sum_k (1/|S|) * a_ext[k-j]
    inv_len = 2.0 / len(active_peaks)
    for k in active_peaks:
        # a_ext[k-j] for j=0..n-1 means a[k-j] when 0<=k-j<n
        # k-j in [0, n-1] means j in [k-n+1, k]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            # j ranges from lo to hi, and a[k-j] means indices k-hi to k-lo
            idx = k - np.arange(lo, hi + 1)
            g[lo:hi+1] += inv_len * a[idx]
    return g

def logsumexp_obj(a, T):
    """Log-sum-exp smoothed objective (sum=1 normalization assumed)."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    m = Tb.max()
    return m / T + np.log(np.exp(Tb - m).sum()) / T


def logsumexp_grad(a, T):
    """Gradient of log-sum-exp smoothed objective w.r.t. a (on simplex)."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    m = Tb.max()
    w = np.exp(Tb - m)
    w /= w.sum()
    # grad_a = 2 * sum_k w[k] * a_ext[k-j]
    g = compute_grad_active(a, np.arange(len(b)))
    # Actually we need weighted version
    g = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-12:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = k - np.arange(lo, hi + 1)
        g[lo:hi+1] += 2.0 * w[k] * a[idx]
    # Project to tangent space for simplex constraint
    g_t = g - g.mean()
    return g_t

                idx = k - np.arange(lo, hi + 1)
                g_a[lo:hi+1] += 2.0 * w[k] * a_vec[idx]
            # Chain rule: grad_z = a * (g_a - a * dot(g_a, a))
            S = np.dot(g_a, a_vec)
            grad_z = a_vec * (g_a - a_vec * S)
            return grad_z
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': 5000, 'ftol': 1e-14, 'gtol': 1e-8,
                         'maxls': 50, 'disp': False}
            )
            if time.monotonic() - t0 < time_limit:
                z_shifted = res.x - res.x.max()
                a_new = np.exp(z_shifted)
                a_new = a_new / a_new.sum()
                obj_new = fast_convolve(a_new).max()
                if obj_new < best_raw_obj:
                    best_raw_obj = obj_new
                    best_a = a_new.copy()
                    z = res.x - res.x.max() + np.log(a0.max())  # rescale
        except Exception:
            pass
    
    return best_a, best_raw_obj


def coordinate_descent(a0, time_limit):
    """
    Coordinate descent: for each element, find optimal value
    that minimizes the max convolution.
    """
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best = a.copy()
    best_obj = fast_convolve(a).max()
    
    # Precompute: b without element j's contribution
    # b_k = sum_i a_i a_{k-i}
    # If we change a_j to new_val and scale others:
    #   a_i' = a_i * (1-new_val)/(1-a_j) for i != j
    #   a_j' = new_val
    # This is complex. Instead, use a simpler approach:
    # Try adding/removing from a_j and uniformly from others.
    
    multipliers = [0.1, 0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 3.0]
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best, best_obj
            old = a[j]
            if old < 1e-15:
                continue
            rem = 1.0 - old
            if rem < 1e-10:
                continue
            
            for mult in multipliers:
                new_j = old * mult
                if new_j < 0 or new_j >= 1.0:
                    continue
                scale = (1.0 - new_j) / rem
                if scale < 0 or scale > 10:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_j
                mask = np.arange(n) != j
                a_trial[mask] = a[mask] * scale
                a_trial = a_trial / a_trial.sum()
                b_trial = fast_convolve(a_trial)
                obj_trial = b_trial.max()
                if obj_trial < best_obj - 1e-14:
                    a = a_trial
                    best_obj = obj_trial
                    best = a.copy()
                    improved = True
                    break
        
        if not improved:
            break
    
    return best, best_obj


def peak_equalization(a0, time_limit):
    """
    Try to equalize the top peaks by adjusting the sequence.
    Uses a simple iterative approach: identify the single maximum peak,
    and move mass to reduce it.
    """
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best = a.copy()
    best_obj = fast_convolve(a).max()
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        b_max = b.max()
        max_k = b.argmax()
        
        # The peak at max_k is b[max_k] = sum_i a_i * a[max_k - i]
        # To reduce it, we should decrease the "large" a_i values that
        # contribute most to this sum.
        # Contribution of a_j to b[max_k]: 2 * a[max_k - j] (if valid)
        # So elements j where a[max_k-j] is large contribute most.
        
        # Find elements to decrease: those j where a[max_k-j] is large
        # and elements to increase: those j where a[max_k-j] is small
        contrib = np.zeros(n)
        lo = max(0, max_k - n + 1)
        hi = min(n - 1, max_k)
        for j in range(lo, hi + 1):
            contrib[j] = 2.0 * a[max_k - j]
        
        # Move mass from high-contribution to low-contribution elements
        # (subject to non-negativity)
        contrib_centered = contrib - contrib.mean()
        
        # Take a small step
        alpha = 0.001
        a_new = a - alpha * contrib_centered
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        b_new = fast_convolve(a_new)
        if b_new.max() < best_obj - 1e-15:
            best_obj = b_new.max()
            best = a_new.copy()
            a = a_new
        else:
            # Try different alpha
            alpha = 0.0001
            a_new = a - alpha * contrib_centered
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-10:
                break
            a_new = a_new / s
            b_new = fast_convolve(a_new)
            if b_new.max() < best_obj - 1e-15:
                best_obj = b_new.max()
                best = a_new.copy()
                a = a_new
            else:
                break
    
    return best, best_obj


def make_powerlaw_init(n, p, direction='increasing'):
    """Power-law initialization: a_i = (i+1)^p or (n-i)^p."""
    i = np.arange(n)
    if direction == 'increasing':
        a = (i + 1.0) ** p
    elif direction == 'decreasing':
        a = (n - i) ** p
    else:  # 'centered'
        center = n / 2.0
        a = np.abs(i + 0.5 - center) ** (-p) if p > 0 else np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_sinusoidal_init(n, freq=1.0, phase=0.0):
    """Sinusoidal modulation of a base ramp."""
    i = np.arange(n)
    base = (i + 1.0) / n  # linear ramp
    mod = 0.5 + 0.5 * np.sin(2 * np.pi * freq * i / n + phase)
    a = base * mod
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_product_ramp_init(n):
    """Product of two ramps: a_i = min(i+1, n-i) / norm."""
    i = np.arange(n)
    a = np.minimum(i + 1.0, n - i)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_curved_ramp_init(n, curvature=2.0):
    """Curved ramp: a_i = ((i+1)/n)^curvature."""
    i = np.arange(n)
    a = ((i + 1.0) / n) ** curvature
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining active-set subgradient,
    L-BFGS smoothing, coordinate descent, and peak equalization.
    Explores structured initializations and multiple lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
        return score
    
    t_start = time.monotonic()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        n_default = 1024
        parent = make_powerlaw_init(n_default, 1.0)
        parent_score = float(evaluate_sequence(parent.tolist()))
        print(f"No parent, using power-law n={n_default}, score={parent_score:.6f}")
        best_seq = parent.copy()
        best_score = parent_score
    
    n_parent = len(parent)
    
    # === Phase 1: Active-set subgradient on parent ===
    print(f"\n=== Phase 1: Active-set subgradient (n={n_parent}) ===")
    rem = deadline - time.monotonic() - 600
    if rem > 60:
        t_alloc = min(rem, 300.0)
        print(f"  Allocating {t_alloc:.0f}s")
        a1, _ = active_set_subgrad(
            parent, time_limit=t_alloc,
            lr_init=0.005, n_top=max(30, n_parent // 20), seed=seed
        )
        try_seq(a1, "sg_parent")
    
    # === Phase 2: L-BFGS smoothing on current best ===
    print(f"\n=== Phase 2: L-BFGS smoothing (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for T in [30, 80, 200]:
            rem = deadline - time.monotonic() - 300
            if rem < 20:
                break
            t_alloc = min(rem * 0.25, 60.0)
            if t_alloc < 10:
                continue
            print(f"  L-BFGS T={T}, t={t_alloc:.0f}s")
            a2, _ = lbgfs_smoothed(
                best_seq, T_list=[T], time_limit=t_alloc
            )
            try_seq(a2, f"lbgfs_T{T}")
    
    # === Phase 3: Structured initializations ===
    print(f"\n=== Phase 3: Structured initializations ===")
    n_base = len(best_seq) if best_seq is not None else n_parent
    inits = {}
    
    # Power-law variations
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        inits[f"pl_inc_{p}"] = make_powerlaw_init(n_base, p, 'increasing')
        inits[f"pl_dec_{p}"] = make_powerlaw_init(n_base, p, 'decreasing')
    
    # Curved ramps
    for c in [0.5, 1.0, 2.0, 3.0, 5.0]:
        inits[f"curve_{c}"] = make_curved_ramp_init(n_base, c)
    
    # Product ramp (triangular-ish)
    inits["prod_ramp"] = make_product_ramp_init(n_base)
    
    # Sinusoidal modulations
    inits["sin_1"] = make_sinusoidal_init(n_base, 1.0, 0)
    inits["sin_2"] = make_sinusoidal_init(n_base, 2.0, 0.5)
    inits["sin_05"] = make_sinusoidal_init(n_base, 0.5, 0)
    
    # Perturbed best
    inits["perturb_best"] = project_simplex(best_seq + rng.standard_normal(n_base) * 0.02)
    
    # Quick optimize each
    for name, init_a in list(inits.items()):
        rem = deadline - time.monotonic() - 300
        if rem < 15:
            break
        t_alloc = min(rem * 0.12, 25.0)
        if t_alloc < 5:
            continue
        
        # Quick subgradient
        a_opt, _ = active_set_subgrad(
            init_a, time_limit=t_alloc,
            lr_init=0.01, n_top=max(20, n_base // 25),
            seed=seed + hash(name) % 100000
        )
        try_seq(a_opt, f"init_{name}")
    
    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length exploration (best={best_score:.6f}) ===")
    if best_seq is not None:
        target_lengths = []
        for factor in [0.5, 0.75, 1.0, 1.25, 1.5, 2.0]:
            tl = int(n_parent * factor)
            tl = max(256, min(4096, tl))
            if tl not in target_lengths and tl != n_parent:
                target_lengths.append(tl)
        target_lengths.sort(key=lambda x: abs(x - n_parent))
        
        for target_n in target_lengths[:5]:
            rem = deadline - time.monotonic() - 200
            if rem < 30:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            # Also try a power-law init at this length
            pl_init = make_powerlaw_init(target_n, 1.0)
            
            for init_a, label in [(f_interp, "interp"), (pl_init, "powerlaw")]:
                rem = deadline - time.monotonic() - 150
                if rem < 20:
                    continue
                t_alloc = min(rem * 0.3, 50.0)
                if t_alloc < 10:
                    continue
                print(f"  n={target_n}, {label}, t={t_alloc:.0f}s")
                a_opt, _ = active_set_subgrad(
                    init_a, time_limit=t_alloc,
                    lr_init=0.005, n_top=max(20, target_n // 25),
                    seed=seed + target_n * 7 + (0 if label == "interp" else 1)
                )
                try_seq(a_opt, f"len{target_n}_{label}")
                
                # Quick L-BFGS polish
                if best_seq is not None and len(best_seq) == target_n:
                    rem = deadline - time.monotonic() - 120
                    if rem > 15:
                        t_lb = min(rem * 0.3, 30.0)
                        a_lb, _ = lbgfs_smoothed(best_seq, T_list=[50], time_limit=t_lb)
                        try_seq(a_lb, f"lbgfs_n{target_n}")
    
    # === Phase 5: Deep optimization on best ===
    print(f"\n=== Phase 5: Deep optimization (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_alloc = min(rem, 120.0)
            print(f"  Deep subgradient n={n_best}, t={t_alloc:.0f}s")
            a5, _ = active_set_subgrad(
                best_seq, time_limit=t_alloc,
                lr_init=0.002, n_top=max(30, n_best // 10),
                seed=seed + 99999
            )
            try_seq(a5, "deep_sg")
    
    # === Phase 6: Peak equalization ===
    print(f"\n=== Phase 6: Peak equalization (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_alloc = min(rem, 40.0)
            print(f"  Peak eq, t={t_alloc:.0f}s")
            a6, _ = peak_equalization(best_seq, time_limit=t_alloc)
            try_seq(a6, "peak_eq")
    
    # === Phase 7: L-BFGS final polish ===
    print(f"\n=== Phase 7: Final L-BFGS (best={best_score:.6f}) ===")
    if best_seq is not None:
        for T in [20, 50, 100, 200]:
            rem = deadline - time.monotonic() - 60
            if rem < 15:
                break
            t_alloc = min(rem * 0.3, 40.0)
            if t_alloc < 8:
                continue
            print(f"  Final L-BFGS T={T}, t={t_alloc:.0f}s")
            a7, _ = lbgfs_smoothed(best_seq, T_list=[T], time_limit=t_alloc)
            try_seq(a7, f"final_T{T}")
    
    # === Phase 8: Coordinate descent polish ===
    print(f"\n=== Phase 8: Coordinate descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            t_cd = min(rem, 45.0)
            print(f"  CD, t={t_cd:.0f}s")
            a8, _ = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a8, "cd")
    
    # === Phase 9: One more round of subgradient + polish ===
    print(f"\n=== Phase 9: Final push (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            t_alloc = min(rem * 0.4, 30.0)
            a9, _ = active_set_subgrad(
                best_seq, time_limit=t_alloc,
                lr_init=0.001, n_top=max(20, len(best_seq) // 10),
                seed=seed + 777777
            )
            try_seq(a9, "push_sg")
            
            rem = deadline - time.monotonic() - 10
            if rem > 10:
                t_pe = min(rem * 0.4, 20.0)
                a10, _ = peak_equalization(best_seq, time_limit=t_pe)
                try_seq(a10, "push_pe")
    
    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")
    
    return [float(x) for x in best_seq]
