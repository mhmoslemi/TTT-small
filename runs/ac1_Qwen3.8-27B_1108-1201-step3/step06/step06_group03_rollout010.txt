Uses the FW LP (min t s.t. (f*g)[k] <= t for all k, sum(g)=1, g>=0) to find descent

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def make_conv_matrix(f):
    """Build the (2n-1) x n cross-convolution matrix for f * g."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def fw_lp_solve(f):
    """Solve FW LP: min t s.t. (f*g)[k] <= t, sum(g)=1, g>=0. Returns (g, t_opt)."""
    n = len(f)
    A_conv = make_conv_matrix(f)
    A_ub = np.hstack([A_conv, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    c = np.zeros(n + 1)
    c[n] = 1.0
    bounds = [(0, None)] * (n + 1)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

        return res.x[:n].copy(), res.x[n]
    return None, None

def subgradient_step(f, M, top_k=5, lr=0.02):
    """One subgradient step targeting the top peaks."""
    n = len(f)
    conv = fast_convolve(f)
    peaks = np.argsort(conv)[-top_k:]
    peak_vals = conv[peaks]
    peak_vals = peak_vals - peak_vals.min() + 1e-10
    weights = peak_vals / peak_vals.sum()

    grad = np.zeros(n)
    for i, k in enumerate(peaks):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[js:je + 1] += weights[i] * (f[k - idx] + f[idx])

    f_new = f * np.exp(-lr * grad)
    f_new = np.maximum(f_new, 1e-15)
    f_new = f_new / f_new.sum()
    return f_new

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        if t_alloc > 30:
            print(f"  FW on parent n={len(parent)}, t={t_alloc:.0f}s")
            a_opt, obj = run_fw_optimize(parent, time_limit=t_alloc, label="parent")
            try_seq(a_opt, "fw_parent")
    else:
        # No parent, use best init at n=768
        best_768 = None
        for o, a, l, n in init_pool:
            if n == 768:
                best_768 = a
                break
        if best_768 is not None:
            t_alloc = min(deadline - time.monotonic(), 200)
            if t_alloc > 30:
                a_opt, obj = run_fw_optimize(best_768, time_limit=t_alloc, label="768")
                try_seq(a_opt, "fw_768")

    # Phase 3: LP-FW on best longer initializations
    print("\n=== Phase 3: LP-FW on Longer Sequences ===")
    longer_inits = []
    for o, a, l, n in init_pool:
        if n >= 1024:
            longer_inits.append((o, a, l, n))

    # Pick diverse longer inits (different lengths, best per length)
    seen_lengths = set()
    selected = []
    for o, a, l, n in longer_inits:
        if n not in seen_lengths:
            selected.append((o, a, l, n))
            seen_lengths.add(n)
        if len(selected) >= 4:
            break

    # Also try interpolating the best 768 result to longer lengths
    if best_seq is not None and len(best_seq) < 1024:
        for target_n in [1024, 1280]:
            if target_n not in seen_lengths:
                f_interp = interpolate_sequence(best_seq, target_n)
                conv = fast_convolve(f_interp)
                obj_est = 2 * target_n * conv.max()
                selected.append((obj_est, f_interp, f"interp_n{target_n}", target_n))
                seen_lengths.add(target_n)

    selected.sort(key=lambda x: x[0])

    for idx, (o, a, l, n) in enumerate(selected[:3]):
        rem = deadline - time.monotonic() - 200
        if rem < 60:
            break
        t_alloc = min(rem / max(1, 3 - idx), 450)
        if t_alloc < 60:
            t_alloc = 60
        print(f"  FW on {l} (n={n}), t={t_alloc:.0f}s, init_obj={o:.4f}")
        a_opt, obj = run_fw_optimize(a, time_limit=t_alloc, label=l)
        try_seq(a_opt, f"fw_{l}")

    # Phase 4: Subgradient polish on overall best
    print("\n=== Phase 4: Subgradient Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            t_alloc = min(rem, 120)
            print(f"  SGP on best (n={len(best_seq)}), t={t_alloc:.0f}s")
            a_opt, obj = quick_subgradient_polish(best_seq, time_limit=t_alloc, label="best")
            try_seq(a_opt, "sg_polish")

    # Phase 5: Further FW on the best
    print("\n=== Phase 5: Further FW on Best ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 60:
            t_alloc = min(rem, 300)
            print(f"  FW polish on best (n={len(best_seq)}), t={t_alloc:.0f}s")
            a_opt, obj = run_fw_optimize(best_seq, time_limit=t_alloc, label="polish")
            try_seq(a_opt, "fw_polish")

    # Phase 6: Try a few random restarts from best
    print("\n=== Phase 6: Random Restarts ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for r in range(3):
            rem = deadline - time.monotonic() - 40
            if rem < 30:
                break
            t_alloc = min(rem / max(1, 3 - r), 100)
            if t_alloc < 20:
                t_alloc = 20

            if r == 0:
                noise = rng.standard_normal(n_best) * 0.004
                f_init = np.maximum(best_seq + noise, 1e-10)
                f_init = f_init / f_init.sum()
            elif r == 1:
                # Blend with a fresh asymmetric init
                asym = make_asymmetric_init(n_best, variant=rng.integers(0, 10), seed=seed + 100 + r)
                f_init = 0.8 * best_seq + 0.2 * asym
                f_init = f_init / f_init.sum()
            else:
                # Slightly larger perturbation
                noise = rng.standard_normal(n_best) * 0.008
                f_init = np.maximum(best_seq + noise, 1e-10)
                f_init = f_init / f_init.sum()

            print(f"  Restart {r}: t={t_alloc:.0f}s")
            a_opt, obj = run_fw_optimize(f_init, time_limit=t_alloc, label=f"restart{r}")
            try_seq(a_opt, f"fw_restart{r}")

    # Phase 7: Final subgradient squeeze
    print("\n=== Phase 7: Final Squeeze ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            t_alloc = min(rem, 60)
            a_opt, obj = quick_subgradient_polish(best_seq, time_limit=t_alloc, label="final")
            try_seq(a_opt, "final_sgp")

    # Fallback
    if best_seq is None:
        best_seq = make_asymmetric_init(768, variant=0, seed=seed)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback used: score={best_score:.6f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        # Should not happen, but just in case
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
