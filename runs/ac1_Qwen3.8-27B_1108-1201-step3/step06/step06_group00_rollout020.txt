    size = 2 * n - 1

    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]

    rho_vals = np.arange(1, n + 1)

    rho = np.nonzero(cond)[0][-1]

    return np.maximum(v - theta, 0.0)

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Lp-approximation with normalized projected gradient descent
2. Basin hopping (simulated annealing) with mass-transfer moves
3. Subgradient refinement with temperature-weighted peak selection
4. Coordinate descent polish
Explores multiple sequence lengths for optimal trade-off.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import math

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)]).astype(np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n, dtype=np.float64) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0).astype(np.float64)

def cross_corr_fft(x, y):
    """Cross-correlation: result[j] = sum_m x[j+m] * y[m] for j = 0,...,len(y)-1."""
    ny = len(y)
    y_rev = y[::-1].copy()
    size = ny + len(x) - 1
    fft_size = 1 << (size - 1).bit_length()
    Y = rfft(y_rev, fft_size)
    X = rfft(x, fft_size)
    return irfft(Y * X, fft_size)[ny-1:ny-1+ny]

def lpp_gradient(a, p):
    """Gradient of sum((a*a)_k^p) with respect to a."""
    n = len(a)
    c = fast_convolve(a)
    c = np.maximum(c, 1e-300)
    if p <= 32:
        c_pm1 = c ** (p - 1)
    else:
        log_c = np.log(c)
        c_pm1 = np.exp((p - 1) * log_c)
    return p * cross_corr_fft(c_pm1, a)

def true_obj(a):
    """Compute the true objective: 2*n*max(conv(a,a))/(sum(a))^2. Assumes sum(a)=1."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def lpp_normalized_grad_descent(a0, p_schedule, time_limit, check_every=200):
    """
    Normalized projected gradient descent for Lp approximation.
    For each p in schedule, iteratively moves in the negative gradient direction
    with fixed step length, projecting onto the simplex after each step.
    """
    t0 = time.monotonic()
    n = len(a0)
    a = project_simplex(a0.astype(np.float64).copy())
    best_a = a.copy()
    best_maxc = fast_convolve(a).max()
    step = 1.0 / math.sqrt(n)
    
    for p in p_schedule:
        if time.monotonic() - t0 > time_limit - 2:
            break
        for it in range(50000):
            if time.monotonic() - t0 > time_limit - 2:
                break
            grad = lpp_gradient(a, p)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            a_new = project_simplex(a - step * grad / gnorm)
            a = a_new
            if it % check_every == 0:
                maxc = fast_convolve(a).max()
                if maxc < best_maxc:
                    best_maxc = maxc
                    best_a = a.copy()
        # End of p level - update best
        maxc = fast_convolve(a).max()
        if maxc < best_maxc:
            best_maxc = maxc
            best_a = a.copy()
    
    return best_a, best_maxc

def basin_hopping(a0, time_limit, T0=0.01, Tend=1e-7, n_transfers=1):
    """
    Basin hopping / simulated annealing with mass-transfer moves.
    Transfers small amounts of mass between pairs of elements.
    """
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_maxc = fast_convolve(a).max()
    cur_maxc = best_maxc
    rng = np.random.default_rng(42)
    total_duration = time_limit
    step = 0.005
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Compute current temperature
        frac = (time.monotonic() - t0) / total_duration
        T = T0 * (Tend / T0) ** frac
        
        # Generate move: transfer mass between 1 or 2 random pairs
        if n_transfers == 1:
            i, j = rng.choice(n, size=2, replace=False)
            delta = step * rng.uniform(0.5, 1.0) * rng.choice([-1, 1])
            if a[i] - delta < 0 or a[j] + delta > 0.999:
                continue
            trial = a.copy()
            trial[i] -= delta
            trial[j] += delta
        else:
            trial = a.copy()
            for _ in range(n_transfers):
                i, j = rng.choice(n, size=2, replace=False)
                delta = step * rng.uniform(0.3, 1.0) * rng.choice([-1, 1])
                if trial[i] - delta >= 0:
                    trial[i] -= delta
                    trial[j] += delta
        
        # Evaluate
        maxc_new = fast_convolve(trial).max()
        delta_obj = maxc_new - cur_maxc
        
        # Metropolis acceptance
        if delta_obj < 0 or (T > 1e-15 and rng.random() < math.exp(-delta_obj / T)):
            a = trial
            cur_maxc = maxc_new
            if cur_maxc < best_maxc:
                best_maxc = cur_maxc
                best_a = a.copy()
        else:
            # Try a different step size occasionally
            if it % 10000 == 0:
                step *= 0.9
                if step < 0.0001:
                    step = 0.001
    
    return best_a, best_maxc
