Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using aggressive LP-Frank-Wolfe
with adaptive target, diverse asymmetric initializations, multi-length
exploration, and coordinate descent polish.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def make_init(n, variant, seed=42):

    rng = np.random.default_rng(seed + variant * 137)

        a = np.exp(-x / (n * 0.1))
    elif variant == 1:

def smoothed_mirror_descent(f0, time_limit, verbose=False,
                            tau=10.0, lr_init=0.01, top_frac=0.1):
    """Smoothed mirror descent for quick progress."""
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        # Top-k softmax weighting
        len_c = len(conv)
        k_top = max(10, int(len_c * top_frac))
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        pv = conv[peaks]
        scaled = tau * pv
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue
            idx = np.arange(js, je + 1)
            contrib = f[k - idx] + f[idx]
            sg[js:je + 1] += w[i] * contrib
        
        f_new = f * np.exp(-lr * sg)
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()
        
        obj_new = fast_convolve(f_new).max()
        if obj_new < M:
            f = f_new
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"    SMD iter {it}: obj={best_obj:.10f}")
            lr = min(lr * 1.01, 0.05)
        else:
            lr *= 0.7
            no_improve += 1

            f = np.maximum(best_f + rng.standard_normal(n) * 0.003, 1e-15)
            f = f / f.sum()
            lr = lr_init
            no_improve = 0
    
    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with exact 1D optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    sweep = 0
    
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        improved = False
        rng = np.random.default_rng(sweep * 1000 + 42)
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            old = f[j]
            if old < 1e-14:
                continue
            
            best_val = best_obj
            best_new = old
            
            # Try several values
            for frac in [0.3, 0.5, 0.7, 0.8, 1.2, 1.5, 2.0, 0.1]:
                new_val = old * frac
                if new_val < 0:
                    continue
                # Renormalize: keep sum = 1
                # f[j] -> new_val, others scale by (1-new_val)/(1-old)
                if abs(1.0 - old) < 1e-10:
                    continue
                scale = (1.0 - new_val) / (1.0 - old)
                if scale < 0:
                    continue
                ft = f.copy()
                ft[j] = new_val
                ft[:j] *= scale
                ft[j+1:] *= scale
                ft = np.maximum(ft, 0)
                ss = ft.sum()
                if ss < 1e-15:
                    continue
                ft = ft / ss
                val = fast_convolve(ft).max()
                if val < best_val - 1e-15:
                    best_val = val
                    best_new = new_val
            
            if best_new != old:
                if abs(1.0 - old) > 1e-10:
                    scale = (1.0 - best_new) / (1.0 - old)
                    f[j] = best_new
                    f[:j] *= scale
                    f[j+1:] *= scale
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                if best_val < best_obj - 1e-15:
                    best_obj = best_val
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_f, best_obj


def interpolate_seq(f, new_n):
    """Interpolate to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    fi = np.interp(new_x, old_x, f)
    fi = np.maximum(fi, 0)
    ss = fi.sum()
    if ss > 1e-15:
        fi /= ss
    return fi

def fast_convolve(a):

    n = len(a)

                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M_new = conv.max()
            f = f_new
            M = M_new
            m_relax = max(1.0, m_relax * 0.97)

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 100 == 0:
                    elapsed = time.monotonic() - t0
                    print(f"    LP-FW iter {it} ({elapsed:.0f}s): obj={best_obj:.10f}")
        else:
            m_relax = min(m_relax * 1.03, max_m_relax)

    return best_f, best_obj


def smoothed_mirror_descent(f0, time_limit, verbose=False,
                            tau=10.0, lr_init=0.01, top_frac=0.1):
    """Smoothed mirror descent for quick initial progress."""
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        len_c = len(conv)
        k_top = max(10, int(len_c * top_frac))
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        pv = conv[peaks]
        scaled = tau * pv
        scaled = scaled - scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue
            idx = np.arange(js, je + 1)
            contrib = f[k - idx] + f[idx]
            sg[js:je + 1] += w[i] * contrib

        f_new = f * np.exp(-lr * sg)
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()

        obj_new = fast_convolve(f_new).max()
        if obj_new < M:
            f = f_new
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 200 == 0:
                    print(f"    SMD iter {it}: obj={best_obj:.10f}")
            lr = min(lr * 1.01, 0.05)
        else:
            lr *= 0.7
            no_improve += 1

        if no_improve > 200:
            rng = np.random.default_rng(it * 7 + 13)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.003, 1e-15)
            f = f / f.sum()
            lr = lr_init
            no_improve = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with multiple trial values per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    sweep = 0

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        improved = False
        rng = np.random.default_rng(sweep * 1000 + 42)
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            old = f[j]
            if old < 1e-14:
                continue

            best_val = best_obj
            best_new = old

            for frac in [0.3, 0.5, 0.7, 0.8, 1.2, 1.5, 2.0, 0.1]:
                new_val = old * frac
                if new_val < 0:
                    continue
                if abs(1.0 - old) < 1e-10:
                    continue
                scale = (1.0 - new_val) / (1.0 - old)
                if scale < 0:
                    continue
                ft = f.copy()
                ft[j] = new_val
                ft[:j] = ft[:j] * scale
                ft[j + 1:] = ft[j + 1:] * scale
                ft = np.maximum(ft, 0)
                ss = ft.sum()
                if ss < 1e-15:
                    continue
                ft = ft / ss
                val = fast_convolve(ft).max()
                if val < best_val - 1e-15:
                    best_val = val
                    best_new = new_val

            if best_new != old:
                if abs(1.0 - old) > 1e-10:
                    scale = (1.0 - best_new) / (1.0 - old)
                    f[j] = best_new
                    f[:j] = f[:j] * scale
                    f[j + 1:] = f[j + 1:] * scale
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                if best_val < best_obj - 1e-15:
                    best_obj = best_val
                    best_f = f.copy()
                    improved = True

        if not improved:
            break
        if verbose and sweep % 5 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.10f}")

    return best_f, best_obj


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    fi = np.interp(new_x, old_x, f)
    fi = np.maximum(fi, 0)
    ss = fi.sum()
    if ss > 1e-15:
        fi = fi / ss
    return fi


def score_seq(a):
    """Score a sequence using evaluate_sequence."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    try:
        return float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    1. Diverse asymmetric initialization
    2. SMD warmup for quick progress
    3. Aggressive LP-FW (main phase)
    4. Second LP-FW pass
    5. Multi-length exploration
    6. Coordinate descent polish
    7. Perturbation + re-optimize cycles
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def update_best(a, label):
        nonlocal best_seq, best_score
        sc = score_seq(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.asarray(a, dtype=np.float64).copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")
        return sc

    # Try parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            pn = p / p.sum()
            ps = score_seq(pn)
            if np.isfinite(ps) and ps < best_score:
                best_score = ps
                best_seq = pn.copy()
            print(f"Parent: n={len(pn)}, score={ps:.6f}")
    except Exception:
        pass

    # === Phase 1: Diverse initialization (20s) ===
    print("=== Phase 1: Initialization ===")
    t_phase1_end = time.monotonic() + 20.0
    test_lengths = [512, 640, 768, 896]

    candidates = []
    for n_try in test_lengths:
        for v in range(16):
            a = make_init(n_try, v, seed=seed)
            candidates.append(a)

    # Quick score with FFT
    scored = []
    for a in candidates:
        cons = fast_convolve(a)
        obj = 2 * len(a) * cons.max()
        scored.append((obj, a))

    scored.sort()
    top5 = scored[:5]
    print(f"  Top 5: {[f'{o:.6f}(n={len(a)})' for o, a in top5]}")

    for i, (obj, a) in enumerate(top5[:3]):
        update_best(a, f"init_{i}")

    # === Phase 2: SMD Warmup (60s) ===
    print("=== Phase 2: SMD Warmup ===")
    start_f = top5[0][1]
    rem = deadline - time.monotonic() - 1300
    t2 = max(20, min(rem, 60))
    print(f"  t={t2:.0f}s")
    a_opt, _ = smoothed_mirror_descent(start_f, time_limit=t2, verbose=True)
    update_best(a_opt, "smd_warm")

    # === Phase 3: Aggressive LP-FW (main, ~1000s) ===
    print("=== Phase 3: LP-FW Main ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 500
        t3 = max(120, min(rem, 1100))
        print(f"  t={t3:.0f}s, n={len(best_seq)}")
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t3, verbose=True)
        update_best(a_opt, "lpfw_main")

    # === Phase 4: Second LP-FW pass ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 350
        if rem > 80:
            print("=== Phase 4: LP-FW Pass 2 ===")
            t4 = min(rem, 250)
            print(f"  t={t4:.0f}s")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t4, verbose=True)
            update_best(a_opt, "lpfw_2")

    # === Phase 5: Multi-length exploration ===
    print("=== Phase 5: Multi-Length ===")
    if best_seq is not None:
        base_n = len(best_seq)
        for target_n in [640, 800, 896, 1024, 1280]:
            rem = deadline - time.monotonic() - 200
            if rem < 50:
                break
            if target_n == base_n:
                continue
            fi = interpolate_seq(best_seq, target_n)
            sc = score_seq(fi)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = fi.copy()
                print(f"  NEW BEST [interp_{target_n}]: {sc:.8f}")
            if sc < best_score + 0.05:
                t5 = min(rem * 0.25, 100)
                if t5 < 20:
                    t5 = 20
                print(f"  LP-FW n={target_n}: t={t5:.0f}s")
                a_opt, _ = lp_fw_optimize(fi, time_limit=t5)
                update_best(a_opt, f"lpfw_{target_n}")

    # === Phase 6: Coordinate descent polish ===
    print("=== Phase 6: Coordinate Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t6 = min(rem, 100)
            print(f"  t={t6:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t6, verbose=True)
            update_best(a_opt, "cd")

    # === Phase 7: Perturbation + short LP-FW cycles ===
    print("=== Phase 7: Perturbation Cycles ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for cyc in range(3):
            rem = deadline - time.monotonic() - 30
            if rem < 30:
                break
            t7 = min(rem / max(1, 3 - cyc), 60)
            if t7 < 20:
                t7 = 20

            noise_scale = 0.002 + 0.001 * cyc
            noise = rng.standard_normal(n_best) * noise_scale
            f_init = np.maximum(best_seq + noise, 0.0)
            f_init = f_init / f_init.sum()

            print(f"  Cycle {cyc}: t={t7:.0f}s, scale={noise_scale:.4f}")
            a_opt, _ = lp_fw_optimize(f_init, time_limit=t7, verbose=False)
            update_best(a_opt, f"perturb_{cyc}")

    # === Phase 8: Final squeeze ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 15:
            print("=== Phase 8: Final Squeeze ===")
            t8 = min(rem, 40)
            print(f"  t={t8:.0f}s")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t8, verbose=True)
            update_best(a_opt, "final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = score_seq(best_seq)

    # Final verification
    fs = score_seq(best_seq)
    if np.isfinite(fs):
        best_score = fs
        best_seq = np.asarray(best_seq, dtype=np.float64)
    else:
        best_seq = np.ones(512) / 512
        best_score = 2.0

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
