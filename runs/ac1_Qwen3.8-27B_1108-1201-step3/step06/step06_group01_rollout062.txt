    """Update a using mirror descent given b's strategy."""
    # Compute convolution
    conv = np.convolve(a, b)  # length 2n-1
    max_val = conv.max()
    
    # Find active peaks (within epsilon)
    eps = 1e-10
    active = np.where(conv >= max_val - eps)[0]
    
    # Compute gradient
    grad = np.zeros(n)
    for k in active:
        # conv[k] = sum_i a[i]*b[k-i]
        # d conv[k] / d a[i] = b[k-i]
        for i in range(n):

    a_new = np.maximum(a_new, 0)

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = max(len(a), len(b))
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def get_lp_solver():
    global _lp_solver
    if '_lp_solver' in globals():
        return _lp_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    """Find b minimizing max_k conv(a_fixed, b)[k] subject to b>=0, sum(b)=1."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build constraint: conv(a, b) <= M
    # conv(a,b)[k] = sum_i a[i]*b[k-i]
    # We need: for all k, sum_i a[i]*b[max(0, k-i) ... ] <= M
    # This is a set of 2n-1 linear constraints
    
    # Build the (2n-1) x n matrix T such that T @ b = conv(a, b)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed
    
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(max(M_var.value, 0))

    Mirror descent (multiplicative weights) update for player A.
    Given B's strategy b, compute subgradient and update a.
    """
    # Compute cross-convolution a * b
    conv = cross_convolve(a, b)
    max_val = conv.max()
    
    # Find peaks (within small epsilon)
    eps = 1e-8 * max_val
    active = np.where(conv >= max_val - eps)[0]
    
    # Compute subgradient: grad[i] = sum_{k in active} b[k-i] for valid k-i
    grad = np.zeros(n)
    for k in active:
        # contribution to grad[i] is b[k-i] if 0 <= k-i < n
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min <= i_max:
            # b[k-i] for i in [i_min, i_max]
            indices = np.arange(i_min, i_max + 1)
            b_indices = k - indices  # = k - i
            # Clip to valid range
            valid = (b_indices >= 0) & (b_indices < n)
            grad[indices[valid]] += b[b_indices[valid]]
    
    # Multiplicative weights update
    # a_new[i] = a[i] * exp(-eta * grad[i]) / Z
    log_a = np.log(np.maximum(a, 1e-300))
    log_a_new = log_a - eta * grad
    # Normalize
    log_a_new -= log_a_new.max()
    a_new = np.exp(log_a_new)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    a_new = a_new / s
    return a_new


def mirror_descent_fp(n, time_limit, initial=None, solver=None, verbose=True):
    """
    Accelerated fictitious play using multiplicative weights (mirror descent).
    """
    if solver is None:
        solver = get_lp_solver()
    t0 = time.monotonic()
    
    if initial is None:
        a = np.ones(n) / n
        # Small perturbation
        rng = np.random.default_rng(42)
        a = a + 0.005 * rng.standard_normal(n)
        a = np.maximum(a, 0)
        a = a / a.sum()
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        a = a / s
    
    b = a.copy()
    
    # Running averages
    a_avg = a.copy().astype(np.float64)
    b_avg = b.copy().astype(np.float64)
    
    best_score = float('inf')
    best_a = a.copy()
    
    # Learning rate schedule
    eta_init = 2.0 / np.sqrt(n)
    
    iters = 0
    while time.monotonic() - t0 < time_limit:
        iters += 1
        eta = eta_init / np.sqrt(iters + 1)
        
        # Get best responses via LP
        b_br, M_b = solve_lp_best_response(a, solver)
        if b_br is not None:
            b = b_br
        
        a_br, M_a = solve_lp_best_response(b, solver)
        if a_br is not None:
            a = a_br
        
        # Multiplicative weights update (this is the key difference from standard FP)
        # Actually, the best response IS the multiplicative weights update in the limit
        # So let's just use the best response directly with damping
        
        # Damped update
        alpha = 2.0 / (iters + 2)
        a = (1 - alpha) * a + alpha * a_br if a_br is not None else a
        b = (1 - alpha) * b + alpha * b_br if b_br is not None else b
        a = np.maximum(a, 0)
        a = a / a.sum()
        b = np.maximum(b, 0)
        b = b / b.sum()
        
        # Running average
        a_avg = (iters * a_avg + a) / (iters + 1)
        b_avg = (iters * b_avg + b) / (iters + 1)
        
        # Evaluate
        if iters % 5 == 0:
            for cand in [a, b, a_avg, b_avg, 0.5*(a+b), 0.5*(a_avg+b_avg)]:
                cand = np.maximum(cand, 0)
                s = cand.sum()
                if s < 1e-10:
                    continue
                cand = cand / s
                max_conv = fast_convolve(cand).max()
                score = 2.0 * n * max_conv
                if score < best_score:
                    best_score = score
                    best_a = cand.copy()
        
        if verbose and iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  MD-FP n={n} iter={iters} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score, iters


def subgradient_projected_gradient(a0, time_limit, n_peaks=20, verbose=True):
    """
    Projected gradient descent using exact subgradient of max auto-correlation.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    t0 = time.monotonic()
    lr = 0.01
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Compute auto-correlation
        conv = fast_convolve(a)
        max_val = conv.max()
        
        # Find active peaks
        eps = max(1e-8, 1e-4 * max_val)
        active = np.where(conv >= max_val - eps)[0]
        if len(active) > n_peaks:
            # Keep only the top n_peaks
            vals = conv[active]
            top_idx = np.argsort(vals)[::-1][:n_peaks]
            active = active[top_idx]
        
        # Compute subgradient (average over active peaks)
        grad = np.zeros(n)
        for k in active:
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            indices = np.arange(i_min, i_max + 1)
            a_indices = k - indices
            grad[indices] += a[a_indices]
        # Factor of 2 for auto-correlation (symmetry)
        grad *= 2.0 / len(active)
        
        # Project onto tangent space of simplex (subtract mean)
        grad -= grad.mean()
        
        # Gradient step with backtracking
        best_lr = lr
        best_a_trial = a.copy()
        best_obj = best_score
        
        for shrink in range(20):
            trial_lr = lr * (0.5 ** shrink)
            a_trial = a - trial_lr * grad
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                break
            a_trial = a_trial / s
            obj = 2.0 * n * fast_convolve(a_trial).max()
            if obj < best_score - 1e-14:
                best_lr = trial_lr
                best_a_trial = a_trial
                best_obj = obj
                break
            elif obj < best_obj:
                best_lr = trial_lr
                best_a_trial = a_trial
                best_obj = obj
            else:
                break
        
        if best_obj < best_score - 1e-14:
            a = best_a_trial
            best_score = best_obj
            best_a = a.copy()
            no_improve = 0
            lr = min(lr * 1.1, 0.1)
        else:
            no_improve += 1
            lr *= 0.7
            if no_improve > 100:
                # Random restart perturbation
                rng = np.random.default_rng(it)
                a = best_a + 0.001 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
                lr = 0.01
                no_improve = 0
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  SubGradPG iter={it} ({elapsed:.0f}s): score={best_score:.6f} lr={lr:.6f}")
    
    return best_a, best_score


def local_search(a0, time_limit, verbose=True):
    """
    Local search: coordinate descent + pairwise operations.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        improved = False
        
        # Coordinate descent: try adjusting individual elements
        # Focus on elements that contribute to peaks
        conv = fast_convolve(a)
        max_val = conv.max()
        eps = 1e-6 * max_val
        active = np.where(conv >= max_val - eps)[0]
        
        # Find elements that contribute to active peaks
        important = set()
        for k in active:
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            for i in range(i_min, i_max + 1):
                if a[i] > 0.05 * a.mean():  # Only consider significant elements
                    important.add(i)
        
        # Try adjusting important elements
        candidates = list(important)[:min(len(important), 50)]
        if not candidates:
            candidates = list(range(n))
        
        # Shuffle for diversity
        rng.shuffle(candidates)
        
        for i in candidates:
            if time.monotonic() - t0 > time_limit:
                break
            for factor in [0.9, 0.8, 0.7, 1.1, 1.2, 1.3]:
                a_trial = a.copy()
                old_val = a_trial[i]
                a_trial[i] = old_val * factor
                if a_trial[i] < 0:
                    continue
                # Redistribute
                diff = a_trial[i] - old_val
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                if others.sum() < 1e-15:
                    continue
                a_trial[mask] = others - diff * others / others.sum()
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial = a_trial / s
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            # Try pairwise swap
            i1 = rng.integers(n)
            i2 = rng.integers(n)
            if i1 != i2:
                for delta in [0.001, 0.005, 0.01, 0.02, -0.001, -0.005, -0.01, -0.02]:
                    a_trial = a.copy()
                    a_trial[i1] += delta
                    a_trial[i2] -= delta
                    if a_trial[i1] < 0 or a_trial[i2] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
        
        if not improved and it % 100 == 0:
            # Periodic perturbation
            a = best_a + 0.002 * rng.standard_normal(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LocalSearch iter={it} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score


def dual_guided_refinement(a0, time_limit, solver=None, verbose=True):
    """
    Use the LP dual to identify which lags are most constraining,
    then do targeted optimization.
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Solve LP to get best response and dual variables
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j+n, j] = a
        
        constraints = [
            T @ b_var <= M_var,
            cp.sum(b_var) == 1
        ]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except:
            break
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        
        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        b_val = b_val / b_val.sum()
        
        # Damped update
        alpha = 1.0 / (0.5 * it + 1)
        a_new = (1 - alpha) * a + alpha * b_val
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            a = a_new  # Still update but don't record as best
            alpha *= 0.5  # Dampen more if no improvement
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  DualRefine iter={it} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    a_norm = a / s

        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def get_lp_solver():
    """Find a working LP solver."""
    if '_lp_solver' in globals():
        return _lp_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return cp.ECOS


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver=None, max_compact=50):
    """
    Find b (non-negative, sum=1) minimizing max_k conv(a_fixed, b)[k].
    Uses compact LP if possible (only active constraints).
    Returns (b, M).
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = float(max(M_var.value, 0.0))
    return b_val, M_val

def subgradient_refine(a0, time_limit, n_peaks=40, momentum=0.9, verbose=True):
    """
    Projected gradient with exact subgradient from all active peaks,
    momentum, and backtracking line search.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    t0 = time.monotonic()
    v = np.zeros(n)
    lr = 0.05
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        max_val = conv.max()

        # Find active peaks (within 0.5% of max)
        eps = max(1e-10, 0.005 * max_val)
        active = np.where(conv >= max_val - eps)[0]
        if len(active) > n_peaks:
            vals = conv[active]
            top_idx = np.argsort(vals)[::-1][:n_peaks]
            active = active[top_idx]

        # Compute subgradient: grad[i] = 2 * sum_{k in active} a[k-i] / |active|
        grad = np.zeros(n)
        for k in active:
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            indices = np.arange(i_min, i_max + 1)
            a_idx = k - indices
            grad[indices] += a[a_idx]
        grad *= 2.0 / len(active)

        # Project to tangent space of simplex
        grad -= grad.mean()

        # Momentum update
        v = momentum * v - lr * grad

        # Projected gradient step with backtracking
        improved = False
        for shrink in range(15):
            trial_step = v * (0.5 ** shrink)
            a_trial = a + trial_step
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                break
            a_trial = a_trial / s
            obj = 2.0 * n * fast_convolve(a_trial).max()
            if obj < best_score - 1e-14:
                a = a_trial
                best_score = obj
                best_a = a.copy()
                no_improve = 0
                improved = True
                break
            if obj <= best_score + 1e-12:
                a = a_trial
                no_improve += 1
                break

        if not improved:
            no_improve += 1
            lr *= 0.8

        if no_improve > 200:
            # Perturbation to escape
            rng_local = np.random.default_rng(it * 7919)
            a = best_a + 0.002 * rng_local.standard_normal(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            lr = max(lr * 2, 0.02)
            no_improve = 0

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  SubGrad it={it} ({elapsed:.0f}s): score={best_score:.6f} lr={lr:.5f}")

    return best_a, best_score

        b = np.maximum(b, 0)
        b = b / b.sum()

        a_avg = (iters * a_avg + a) / (iters + 1)
        b_avg = (iters * b_avg + b) / (iters + 1)

        # Convergence detection
        if value is not None and prev_value is not None:
            if abs(value - prev_value) < 1e-7:
                converged_iters += 1
            else:
                converged_iters = 0
        prev_value = value

        # Evaluate
        if iters % 5 == 0:
            for cand in [a_avg, b_avg, 0.5 * (a_avg + b_avg), a, b]:
                cand = np.maximum(cand, 0)
                s = cand.sum()
                if s < 1e-10:
                    continue
                cand = cand / s
                score = 2.0 * n * fast_convolve(cand).max()
                if score < best_score:
                    best_score = score
                    best_a = cand.copy()

        if verbose and iters % 100 == 0:
            elapsed = time.monotonic() - t0
            val_s = f"{value:.7f}" if value is not None else "N/A"
            print(f"  DGF n={n} it={iters} ({elapsed:.0f}s): V={val_s} best={best_score:.6f} conv={converged_iters}")

        # Early stopping if well converged
        if converged_iters > 500 and iters > 500:
            break

    # Final candidates
    for cand in [a, b, a_avg, b_avg, 0.5 * (a_avg + b_avg)]:
        cand = np.maximum(cand, 0)
        s = cand.sum()
        if s < 1e-10:
            continue
        cand = cand / s
        score = 2.0 * n * fast_convolve(cand).max()
        if score < best_score:
            best_score = score
            best_a = cand.copy()

    return best_a, best_score, iters


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate candidate and update best."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def softmax_smoothed_grad(z, T, n):
    """Compute soft-max smoothed objective and gradient in log-space."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    lse = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * lse
    w = exp_Tb / exp_Tb.sum()

    # Gradient w.r.t. a: 2n * cross_corr(w, a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    cross = irfft(np.conj(W) * A, fft_size)[:n]
    grad_a = 2.0 * n * cross

    # Gradient w.r.t. z
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    return obj, grad_z


def lbgfs_refine(a0, time_limit, T=100, maxiter=5000, verbose=True):
    """L-BFGS refinement in log-space with soft-max smoothing."""
    from scipy.optimize import minimize
    n = len(a0)
    a_safe = np.maximum(a0, 1e-10)
    a_safe = a_safe / a_safe.sum()
    z0 = np.log(a_safe)
    t0 = time.monotonic()
    stopped = [False]

    def func(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        obj, _ = softmax_smoothed_grad(z, T, n)
        return obj

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros_like(z)
        _, g = softmax_smoothed_grad(z, T, n)
        return g

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                     'maxfun': maxiter * 50, 'disp': False}
        )
        a_out = np.exp(res.x - res.x.max())
        a_out = a_out / a_out.sum()
    except Exception:
        a_out = a0.copy()

    if verbose:
        score_out = 2.0 * n * fast_convolve(a_out).max()
        print(f"  L-BFGS done: score={score_out:.6f}")

    return a_out


def propose_candidate():

    1. Fictitious play with LP best responses (convergence to Nash eq)
    2. Dual-guided FP (tighter convergence detection)

    4. L-BFGS with soft-max smoothing
    5. Local search (coordinate descent + pairwise)
    6. Multi-length exploration
    7. Final micro-polish
    """

    best_score = float('inf')
