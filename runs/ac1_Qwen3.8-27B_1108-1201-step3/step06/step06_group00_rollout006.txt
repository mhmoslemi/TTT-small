"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based peak suppression
iterative method combined with gradient refinement.

Key idea: At each iteration, solve a linear program to find a sequence g
that maximizes its sum subject to the cross-correlation with the current
sequence being bounded by the current peak. Blending the improved sequence
with the current one reduces the peak autocorrelation.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_peak_suppression(f, M, solver=None):
    """
    Solve: max sum(g) s.t. g >= 0, conv(f,g)[k] <= M for all k.
    Returns g or None if infeasible.
    """
    import cvxpy as cp
    
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    constraints = []
    
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # conv(f,g)[k] = sum_{i=lo}^{hi} f[i] * g[k-i]
        idx = k - np.arange(lo, hi + 1)
        constraints.append(f[lo:hi+1] @ g[idx] <= M)
    
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        # Try alternate solver
        try:
            prob.solve(solver=cp.CLARABEL, verbose=False)
        except Exception:
            return None
    
    if g.value is None or prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    g_val = np.maximum(g.value, 0)
    return g_val

def lp_iteration(f, n_iters=50, t_blend=0.3, solver=None, time_limit=None, t0=None):
    """
    Iterative LP-based peak suppression.
    """
    if t0 is None:
        t0 = time.monotonic()
    
    f = f.copy().astype(np.float64)
    s = f.sum()
    f = f / s  # normalize to sum=1
    
    best_f = f.copy()
    best_M = fast_convolve(f).max()
    
    for it in range(n_iters):
        if time_limit is not None and time.monotonic() - t0 > time_limit:
            break
        
        M = fast_convolve(f).max()
        if M < best_M:
            best_M = M
            best_f = f.copy()
        
        # Solve LP
        g = solve_lp_peak_suppression(f, M, solver=solver)
        if g is None:
            break
        
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-10:
            # No improvement possible
            break
        
        # Rescale g to sum=1
        g = g / sum_g
        
        # Check that the new sequence doesn't have worse peak
        M_g = fast_convolve(g).max()
        
        # Blend
        f_new = (1 - t_blend) * f + t_blend * g
        f_new = np.maximum(f_new, 0)
        f_new = f_new / f_new.sum()
        
        M_new = fast_convolve(f_new).max()
        
        if M_new >= M - 1e-12:
            # Blending didn't help, try smaller t
            t_blend = max(t_blend * 0.5, 0.01)
            f_new = (1 - t_blend) * f + t_blend * g
            f_new = np.maximum(f_new, 0)
            f_new = f_new / f_new.sum()
            M_new = fast_convolve(f_new).max()
            if M_new >= M - 1e-12:
                break
        
        f = f_new
        # Reset blend factor
        t_blend = min(t_blend * 1.2, 0.5)
    
    return best_f, best_M

def subgradient_optimize(a0, time_limit, lr_init=0.01, momentum=0.9,
                         temp_init=0.001, temp_growth=1.02,
                         kick_interval=500, kick_size=0.003, seed=42):
    """Subgradient optimization on the simplex."""
    from scipy.fft import rfft, irfft
    
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    cur_lr = lr_init
    cur_temp = temp_init
    no_improve = 0
    it_count = 0
    
    while time.monotonic() - t0 < time_limit:
        # Batch of iterations
        batch_size = 20
        for _ in range(batch_size):
            if time.monotonic() - t0 >= time_limit:
                break
            it_count += 1
            b = fast_convolve(a)
            b_max = b.max()
            
            # Temperature-weighted peak selection
            scaled = b / cur_temp
            scaled -= scaled.max()
            w = np.exp(scaled)
            w_sum = w.sum()
            if w_sum < 1e-30:
                w = np.zeros(len(b))
                w[b.argmax()] = 1.0
            else:
                w /= w_sum
            
            # Subgradient computation
            g = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-10:
                    continue
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                idx = np.arange(lo, hi + 1)
                g[idx] += w[k] * a[k - idx]
            
            # Nesterov step
            v = momentum * v - cur_lr * g
            a_new = project_simplex(a + v)
            
            b_new = fast_convolve(a_new)
            obj_new = b_new.max()
            
            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            a = a_new
        
        # Adaptive adjustments
        if no_improve > kick_interval:
            a = project_simplex(best_a + rng.standard_normal(n) * kick_size)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.7, lr_init * 0.1)
            no_improve = 0
        elif no_improve > kick_interval // 2:
            cur_lr *= 0.95
        
        cur_temp *= temp_growth
        if cur_temp > 0.5:
            cur_temp = 0.5
    
    return best_a, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))

def lbgfs_refine(a0, T, time_limit, maxiter=5000):
    """L-BFGS refinement with soft-max objective."""
    from scipy.optimize import minimize
    t0 = time.monotonic()
    n = len(a0)
    z0 = z_from_a(a0)
    
    def softmax_obj(z):
        a = softmax_from_z(z)
        b = fast_convolve(a)
        Tb = T * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        log_sum_exp = max_Tb + np.log(exp_Tb.sum())
        return (2.0 * n / T) * log_sum_exp
    
    def softmax_grad(z):
        a = softmax_from_z(z)
        b = fast_convolve(a)
        Tb = T * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        
        # grad_a = 2n * cross_corr(w, a)
        # cross_corr(w, a)[i] = sum_k w[k] * a[k-i] for valid k, i
        # Using FFT: ifft(conj(FFT(w)) * FFT(a))
        size = n + len(w) - 1
        fft_size = 1 << (size - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        grad_a = irfft(np.conj(W) * A, fft_size)[:n]
        grad_a *= 2.0 * n
        
        # Jacobian of softmax: J = diag(a) - outer(a, a)
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - a * S)
        return grad_z
    
    best_obj = softmax_obj(z0)
    best_z = z0.copy()
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return softmax_obj(z)
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return softmax_grad(z)
    
    try:
        res = minimize(
            fun=func, x0=z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 20, 'disp': False}
        )
        final_obj = func(res.x)
        if np.isfinite(final_obj) and final_obj < best_obj:
            best_z = res.x
            best_obj = final_obj
    except Exception:
        pass
    
    return softmax_from_z(best_z), best_obj

def coordinate_polish(a0, time_limit, n_trials_per_elem=8):
    """Coordinate descent on individual elements."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    multipliers = np.array([0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0])
    
    improved = True
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-14:
                continue
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 0.999:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                remaining = 1.0 - new_val
                if remaining < 1e-10:

                total_old = a.sum() - old_val
                if total_old < 1e-10:
                    continue
                a_trial[mask] *= remaining / total_old
                a_trial = a_trial / a_trial.sum()
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based peak suppression
iterative method combined with subgradient refinement, smooth L-BFGS polish,
and coordinate descent. The LP formulation finds the maximum-mass sequence
g whose cross-correlation with the current sequence is bounded, enabling
progressive peak reduction through careful blending.

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_peak(f, M, prefer_sparse=False):
    """
    Solve: max sum(g) s.t. g >= 0, conv(f,g)[k] <= M for all k.
    Returns g array or None if failed.
    """
    import cvxpy as cp

    n = len(f)
    g = cp.Variable(n, nonneg=True)
    constraints = []

    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = (k - np.arange(lo, hi + 1)).astype(int)
        constraints.append(f[lo:hi + 1] @ g[idx] <= M)

    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)

    solvers = [cp.CLARABEL, cp.ECOS, cp.SCS, cp.GLPK]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                return np.maximum(g.value.astype(np.float64), 0)
        except Exception:
            continue

    return None


def lp_iteration(f, time_limit, t0=None, adaptive_blend=True, verbose=False):

    Returns (best_f, best_M) where f is normalized to sum=1.

    f = f.copy().astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        return f, np.inf
    f = f / s

    best_f = f.copy()
    best_M = fast_convolve(f).max()
    t_blend = 0.3
    stagnation = 0

    for it in range(2000):
        if time.monotonic() - t0 > time_limit:
            break

        M = fast_convolve(f).max()
        if M < best_M - 1e-14:
            best_M = M
            best_f = f.copy()
            stagnation = 0

        g = solve_lp_peak(f, M)
        if g is None:
            break

        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-8:
            stagnation += 5
            if stagnation > 10:
                break
            continue

        g = g / sum_g

        if adaptive_blend:
            trial_ts = [t_blend, t_blend * 0.5, t_blend * 0.75, 0.1, 0.05]
            best_t = None
            best_M_trial = M
            for t in trial_ts:

                f_trial = np.maximum(f_trial, 0)

                if ss < 1e-10:

            f = (1 - best_t) * f + best_t * g
            f = np.maximum(f, 0)
            f = f / f.sum()
            t_blend = min(best_t * 1.1, 0.5)

def subgradient_optimize(a0, time_limit, lr_init=0.01, momentum=0.9,
                         temp_init=0.001, temp_growth=1.02,
                         kick_interval=300, kick_size=0.003,
                         n_top=None, seed=42):
    """Subgradient optimization on the simplex with Nesterov momentum."""
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    cur_lr = lr_init
    cur_temp = temp_init
    no_improve = 0
    it_count = 0

    while time.monotonic() - t0 < time_limit:
        batch_size = 30
        for _ in range(batch_size):
            if time.monotonic() - t0 >= time_limit:
                break
            it_count += 1
            b = fast_convolve(a)

            scaled = b / cur_temp
            scaled -= scaled.max()
            w = np.exp(scaled)
            w_sum = w.sum()
            if w_sum < 1e-30:
                w = np.zeros(len(b))
                w[b.argmax()] = 1.0
            else:
                w /= w_sum

            if n_top is not None and n_top < len(b):
                top_idx = np.argpartition(b, -n_top)[-n_top:]
                w_sel = w[top_idx]
                ws = w_sel.sum()
                if ws > 1e-30:
                    w_sel /= ws
                w_full = np.zeros(len(b))
                w_full[top_idx] = w_sel
                w = w_full

            g = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-10:
                    continue
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                idx = np.arange(lo, hi + 1)
                g[idx] += w[k] * a[k - idx]

            v = momentum * v - cur_lr * g
            a_new = project_simplex(a + v)

            b_new = fast_convolve(a_new)
            obj_new = b_new.max()

            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1

            a = a_new

        if no_improve > kick_interval:
            a = project_simplex(best_a + rng.standard_normal(n) * kick_size)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.7, lr_init * 0.05)
            no_improve = 0
        elif no_improve > kick_interval // 2:
            cur_lr *= 0.95

        cur_temp *= temp_growth
        if cur_temp > 0.3:
            cur_temp = 0.3

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def lbgfs_refine(a0, T, time_limit, maxiter=5000):
    """L-BFGS in log-space with soft-max smoothed objective."""
    from scipy.optimize import minimize
    t0 = time.monotonic()
    n = len(a0)
    z0 = z_from_a(a0)

    def _obj(z):
        a = softmax_from_z(z)
        b = fast_convolve(a)
        Tb = T * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        log_sum_exp = max_Tb + np.log(exp_Tb.sum())
        return (2.0 * n / T) * log_sum_exp

    def _grad(z):
        a = softmax_from_z(z)
        b = fast_convolve(a)
        Tb = T * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()

        size = n + len(w) - 1
        fft_size = 1 << (size - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        grad_a = irfft(np.conj(W) * A, fft_size)[:n]
        grad_a *= 2.0 * n

        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - a * S)
        return grad_z

    best_obj = _obj(z0)
    best_z = z0.copy()

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return _obj(z)

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return _grad(z)

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 20, 'disp': False}
        )
        final_obj = func(res.x)
        if np.isfinite(final_obj) and final_obj < best_obj:
            best_z = res.x
            best_obj = final_obj
    except Exception:
        pass

    return softmax_from_z(best_z), best_obj


def coordinate_polish(a0, time_limit):
    """Coordinate descent: try scaling each element by various multipliers."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    multipliers = np.array([0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 1.8, 2.5])

    improved = True
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-14:
                continue
            best_j_obj = best_obj
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale <= 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = a / a.sum()
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

    return best_a, best_obj

        for k in peak_positions:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                # Element i contributes a[i]*a[k-i] to peak k
                pressures[i] += a[k - i]
                if k - i != i:
                    pressures[k - i] += a[i]

        # Reduce elements with high pressure, increase low-pressure elements
        total_pressure = pressures.sum() + 1e-15
        target = a * (1.0 - lr) + (1.0 / n) * lr * pressures / total_pressure * n
        target = np.maximum(target, 0)
        ts = target.sum()
        if ts < 1e-10:
            break
        target = target / ts

        # Add small noise
        if step % 100 == 0:
            target = target + rng.standard_normal(n) * 0.001 * (1.0 - step / 10000)
            target = np.maximum(target, 0)
            target = target / target.sum()

        a = target
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
        elif best_obj - obj > 0.01 * best_obj:
            a = best_a.copy()
            break

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses LP-based peak suppression as the main engine, combined with
    subgradient descent, L-BFGS smoothing, and coordinate descent polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        print("No parent, using uniform n=768")
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    n_parent = len(parent)

    # === Phase 1: LP-based iteration on parent ===
    print(f"\n=== Phase 1: LP iteration (n={n_parent}) ===")
    rem = deadline - time.monotonic() - 600
    if rem > 30:
        t_alloc = min(rem, 500.0)
        print(f"  Allocating {t_alloc:.0f}s")
        a_lp, M_lp = lp_iteration(parent, time_limit=t_alloc, verbose=True)
        try_seq(a_lp, f"lp_n{n_parent}")

    # === Phase 2: Subgradient refinement ===
    print(f"\n=== Phase 2: Subgradient (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_alloc = min(rem, 200.0)
            print(f"  t={t_alloc:.0f}s")
            a_sg, _ = subgradient_optimize(
                best_seq, time_limit=t_alloc,
                lr_init=0.005, momentum=0.92,
                temp_init=0.0005, temp_growth=1.02,
                kick_interval=500, kick_size=0.003,
                n_top=max(20, len(best_seq) // 10), seed=seed
            )
            try_seq(a_sg, f"sg_n{len(a_sg)}")

    # === Phase 3: LP iteration on best (second round) ===
    print(f"\n=== Phase 3: LP round 2 (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_alloc = min(rem, 300.0)
            print(f"  t={t_alloc:.0f}s")
            a_lp2, _ = lp_iteration(best_seq, time_limit=t_alloc)
            try_seq(a_lp2, f"lp2_n{len(a_lp2)}")

    # === Phase 4: L-BFGS refinement ===
    print(f"\n=== Phase 4: L-BFGS (best={best_score:.6f}) ===")
    if best_seq is not None:
        for T in [30, 80, 150]:
            rem = deadline - time.monotonic() - 200
            if rem < 15:
                break
            t_alloc = min(rem * 0.3, 60.0)
            if t_alloc < 10:
                continue
            print(f"  L-BFGS T={T}, t={t_alloc:.0f}s")
            a_lb, _ = lbgfs_refine(best_seq, T=T, time_limit=t_alloc, maxiter=3000)
            try_seq(a_lb, f"lbgfs_T{T}")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_base = len(best_seq)
        target_lengths = []
        for factor in [0.75, 1.25, 1.5, 2.0]:
            tl = int(n_base * factor)
            tl = max(256, min(4096, tl))
            if tl != n_base and tl not in target_lengths:
                target_lengths.append(tl)

        target_lengths.sort(key=lambda x: abs(x - n_base))
        for target_n in target_lengths[:3]:
            rem = deadline - time.monotonic() - 150
            if rem < 30:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            # Quick LP on new length
            t_alloc = min(rem * 0.4, 80.0)
            if t_alloc < 10:
                continue
            print(f"  n={target_n}, LP t={t_alloc:.0f}s")
            a_opt, _ = lp_iteration(f_interp, time_limit=t_alloc)
            try_seq(a_opt, f"lp_multin_n{target_n}")

            if best_seq is not None and len(best_seq) == target_n:
                rem = deadline - time.monotonic() - 100
                if rem > 20:
                    t_sg = min(rem * 0.4, 40.0)
                    a_sg2, _ = subgradient_optimize(
                        best_seq, time_limit=t_sg,
                        lr_init=0.003, momentum=0.9,
                        temp_init=0.001, temp_growth=1.03,
                        kick_interval=400, kick_size=0.003,
                        n_top=max(15, target_n // 10), seed=seed + target_n
                    )
                    try_seq(a_sg2, f"sg_multin_n{target_n}")

    # === Phase 6: Peak-targeted descent ===
    print(f"\n=== Phase 6: Peak-targeted (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_alloc = min(rem, 80.0)
            print(f"  t={t_alloc:.0f}s")
            a_pt, _ = peak_targeted_descent(
                best_seq, time_limit=t_alloc,
                n_peaks=10, lr=0.002, seed=seed + 12345
            )
            try_seq(a_pt, "peak_targeted")

            # Second round with different params
            rem = deadline - time.monotonic() - 60
            if rem > 15:
                t_alloc = min(rem, 40.0)
                a_pt2, _ = peak_targeted_descent(
                    best_seq, time_limit=t_alloc,
                    n_peaks=20, lr=0.001, seed=seed + 67890
                )
                try_seq(a_pt2, "peak_targeted2")

    # === Phase 7: Final L-BFGS polish ===
    print(f"\n=== Phase 7: Final L-BFGS (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 50
        if rem > 15:
            for T in [50, 100]:
                rem = deadline - time.monotonic() - 30
                if rem < 10:
                    break
                t_alloc = min(rem * 0.5, 30.0)
                a_fin, _ = lbgfs_refine(best_seq, T=T, time_limit=t_alloc, maxiter=5000)
                try_seq(a_fin, f"final_T{T}")

    # === Phase 8: Coordinate descent ===
    print(f"\n=== Phase 8: Coordinate descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_cd = min(rem, 45.0)
            print(f"  t={t_cd:.0f}s")
            a_cd, _ = coordinate_polish(best_seq, time_limit=t_cd)
            try_seq(a_cd, "cd")

    # === Phase 9: Final subgradient micro-opt ===
    print(f"\n=== Phase 9: Micro-opt (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            t_alloc = min(rem, 20.0)
            a_micro, _ = subgradient_optimize(
                best_seq, time_limit=t_alloc,
                lr_init=0.001, momentum=0.95,
                temp_init=0.0002, temp_growth=1.005,
                kick_interval=1000, kick_size=0.001,
                n_top=max(30, len(best_seq) // 5),
                seed=seed + 99999
            )
            try_seq(a_micro, "micro")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'=' * 50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'=' * 50}")

    return [float(x) for x in best_seq]
