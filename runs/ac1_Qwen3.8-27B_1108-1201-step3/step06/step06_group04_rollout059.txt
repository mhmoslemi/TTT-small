import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_normalized(f):
    """Objective for sum-normalized non-negative sequence: 2*n*max(conv)."""
    n = len(f)
    if n == 0:
        return np.inf
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()


def compute_subgradient(a, k_indices, n):
    """
    Compute weighted subgradient of max_k conv[k] at point a.
    For each active peak k: d conv[k]/d a[i] = 2*a[k-i] if 0<=k-i<n else 0.
    """
    grad = np.zeros(n)
    for k in k_indices:
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_end < i_start:
            continue
        indices = np.arange(i_start, i_end + 1)
        grad[indices] += 2.0 * a[k - indices]
    return grad


def compute_weighted_subgradient(a, conv, n, top_k=3, temperature=0.01):
    """
    Compute a smoothed subgradient using softmax-weighted top peaks.
    Returns (gradient, effective_peak_count).
    """
    k_actual = min(top_k, len(conv))
    # Get top k actual peak values and indices
    peak_vals = np.sort(conv)[-k_actual:][::-1]
    peak_indices = np.argsort(conv)[-k_actual:][::-1]

    # Weight by softmax of (peak_val - max) / temperature
    shifted = (peak_vals - peak_vals[0]) / temperature
    weights = np.exp(shifted)
    weights /= weights.sum()

    grad = np.zeros(n)
    for w, k in zip(weights, peak_indices):
        if w < 1e-10:
            continue
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_end < i_start:
            continue
        indices = np.arange(i_start, i_end + 1)
        grad[indices] += w * 2.0 * a[k - indices]

    return grad, len(peak_indices)


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def make_cosine(n):

def make_bimodal(n, gap=0.3):

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        cur_max = conv.max()
        cur_obj = 2.0 * n * cur_max

        # Check improvement
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Periodically re-seed from best if stuck
        if no_improve > 300:
            f = best_f.copy()
            rng = np.random.default_rng(it * 17 + 3)
            f = f + rng.standard_normal(n) * 0.003
            f = np.maximum(f, 1e-15)
            f = f / f.sum()
            no_improve = 0
            eta = max(eta, 0.01)
            continue

        if no_improve > 150:
            # Increase temperature to explore more peaks
            temperature = min(temperature * 1.02, 5.0)
            eta = max(eta * 1.01, 0.005)

        # Compute weighted subgradient
        grad, _ = compute_weighted_subgradient(f, conv, n, top_k=top_k,
                                               temperature=temperature)

        # Normalize gradient for stability
        grad_max = np.max(np.abs(grad))
        if grad_max > 1e-15:
            grad = grad / grad_max

        # Momentum update
        if it < momentum_ramp:
            momentum = momentum_target * it / momentum_ramp
        grad_buf = momentum * grad_buf + (1 - momentum) * grad
        effective_grad = grad_buf

        # Multiplicative update (mirror descent)
        log_update = -eta * effective_grad
        # Clip to avoid overflow
        log_update = np.clip(log_update, -5.0, 5.0)
        f_new = f * np.exp(log_update)
        f_new = np.maximum(f_new, 1e-15)
        s = f_new.sum()
        if s < 1e-15:
            f_new = f.copy()
        else:
            f_new = f_new / s
        f = f_new

        # Decay step size
        eta *= eta_decay
        if eta < eta_min:
            eta = eta_min
        temperature *= temp_growth

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter {it}: obj={best_obj:.10f}, eta={eta:.6f}, "
                  f"temp={temperature:.3f}, t={elapsed:.0f}s")

        if no_improve > 2000:
            break

    return best_f, best_obj


def projected_cd(f0, time_limit, verbose=False):
    """
    Coordinate descent on the simplex, targeting peak reduction.
    Tries moving mass from peak-contributing coordinates to others.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 200:
        conv = fast_convolve(f)
        cur_max = conv.max()
        cur_obj = 2.0 * n * cur_max
        improved = False

        # Find top peaks
        k_actual = min(8, 2 * n - 1)
        peak_indices = np.argpartition(conv, -k_actual)[-k_actual:]
        peak_indices = peak_indices[np.argsort(conv[peak_indices])[::-1]]

        for k in peak_indices[:5]:
            k = int(k)
            if conv[k] < cur_max * 0.95:
                break
            # Find contributing pairs
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_end < i_start:
                continue
            indices = np.arange(i_start, i_end + 1)
            products = f[indices] * f[k - indices]
            total = products.sum()
            if total < 1e-15:
                continue

            # Try reducing top contributors
            order = np.argsort(products)[::-1]
            for oi in range(min(5, len(order))):
                j_idx = int(indices[order[oi]])
                partner = k - j_idx
                old_val = f[j_idx]
                if old_val < 1e-12:
                    continue

                # Try different reduction factors
                for mult in [0.3, 0.5, 0.7, 0.85]:
                    new_val = old_val * mult
                    delta = old_val - new_val
                    if delta < 1e-12:
                        continue

                    trial = f.copy()
                    trial[j_idx] = new_val

                    # Redistribute delta to coordinates NOT contributing to this peak
                    # and NOT contributing to other top peaks
                    safe = np.ones(n, dtype=bool)
                    for kp in peak_indices[:3]:
                        kp = int(kp)
                        s2 = max(0, kp - n + 1)
                        e2 = min(n - 1, kp)
                        if e2 >= s2:
                            safe[s2:e2 + 1] = False
                    safe[j_idx] = False

                    if safe.sum() < 3:
                        # Fallback: redistribute to all non-peak coords
                        safe = np.ones(n, dtype=bool)
                        safe[j_idx] = False

                    nsafe = safe.sum()
                    if nsafe < 1:
                        continue
                    # Proportional redistribution
                    w = f[safe]
                    wsum = w.sum()
                    if wsum < 1e-15:
                        trial[safe] += delta / nsafe
                    else:
                        trial[safe] += delta * (w / wsum)

                    trial = np.maximum(trial, 0)
                    ts = trial.sum()
                    if ts < 1e-15:
                        continue
                    trial = trial / ts

                    obj = 2.0 * n * fast_convolve(trial).max()
                    if obj < best_obj - 1e-14:
                        best_obj = obj
                        best_f = trial.copy()
                        f = trial
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            no_improve += 1
            # Aggressive perturbation
            if no_improve > 30:
                conv_cur = fast_convolve(f)
                k_peak = int(np.argmax(conv_cur))
                i_start = max(0, k_peak - n + 1)
                i_end = min(n - 1, k_peak)
                trial = f.copy()
                trial[i_start:i_end + 1] *= 0.9
                delta = f.sum() - trial.sum()
                non_peak = np.ones(n, dtype=bool)
                non_peak[i_start:i_end + 1] = False
                if non_peak.sum() > 0:
                    nw = f[non_peak].sum()
                    if nw > 1e-15:
                        trial[non_peak] += delta * f[non_peak] / nw
                trial = np.maximum(trial, 0)
                trial = trial / trial.sum()
                obj = 2.0 * n * fast_convolve(trial).max()
                if obj < best_obj * 1.01:  # Accept slightly worse to escape
                    f = trial
                    if obj < best_obj:
                        best_obj = obj
                        best_f = trial.copy()
                        no_improve = 0
            if no_improve > 150:
                # Full restart from best with perturbation
                rng = np.random.default_rng(int(time.monotonic() * 1000))
                f = best_f + rng.standard_normal(n) * 0.005
                f = np.maximum(f, 1e-10)
                f = f / f.sum()
                no_improve = 0
        else:
            no_improve = 0

        if verbose and it % 100 == 0:
            print(f"    CD: obj={best_obj:.10f}, no_improve={no_improve}")

    return best_f, best_obj


def accelerated_subgradient(f0, time_limit, verbose=False):
    """
    Nesterov-accelerated projected subgradient method on the simplex.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    y = f.copy()  # Nesterov point
    f_prev = f.copy()
    beta = 1.0  # momentum parameter (1/(1+t))

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    eta0 = 0.05

    while time.monotonic() - t0 < time_limit and no_improve < 500:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        # Compute subgradient at y
        conv_y = fast_convolve(y)
        cur_max = conv_y.max()
        k_peak = int(np.argmax(conv_y))

        # Also check a few nearby peaks for smoothing
        top_k = min(3, 2 * n - 1)
        peaks = np.argpartition(conv_y, -top_k)[-top_k:]
        peaks = peaks[np.argsort(conv_y[peaks])[::-1]]
        # Weight: exponential decay from max
        pk_vals = conv_y[peaks]
        w = np.exp(-(pk_vals[0] - pk_vals) / max(cur_max * 0.01, 1e-10))
        w /= w.sum()

        grad = np.zeros(n)
        for wi, kk in zip(w, peaks):
            kk = int(kk)
            i_start = max(0, kk - n + 1)
            i_end = min(n - 1, kk)
            if i_end >= i_start:
                indices = np.arange(i_start, i_end + 1)
                grad[indices] += wi * 2.0 * y[kk - indices]

        # Step size: eta_t = eta0 / (1 + t*0.1)
        eta_t = eta0 / (1.0 + 0.05 * it)

        # Subgradient step
        f_new_raw = y - eta_t * grad

        # Project onto simplex
        f_new = project_simplex(f_new_raw)
        f_new = np.maximum(f_new, 0)
        ss = f_new.sum()
        if ss > 1e-15:
            f_new = f_new / ss
        else:
            f_new = f.copy()

        # Nesterov update
        beta_new = 2.0 / (1.0 + np.sqrt(1.0 + 4.0 * beta * beta))
        y_new = (f_new + (beta * beta / beta_new) * (f_new - f_prev)) / (1.0 + beta * beta)
        y_new = project_simplex(y_new)
        y_new = np.maximum(y_new, 0)
        ys = y_new.sum()
        if ys > 1e-15:
            y_new = y_new / ys

        y = y_new
        f_prev = f_new
        f = f_new
        beta = beta_new

        # Check
        conv_f = fast_convolve(f)
        obj_f = 2.0 * n * conv_f.max()
        if obj_f < best_obj - 1e-13:
            best_obj = obj_f
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and it % 200 == 0:
            print(f"    ASG iter {it}: obj={best_obj:.10f}, t={time.monotonic()-t0:.0f}s")

        if no_improve > 200:
            # Reset momentum
            y = best_f.copy()
            f_prev = best_f.copy()
            f = best_f.copy()
            beta = 1.0
            no_improve = 0

    return best_f, best_obj


def fine_polish(f0, time_limit, verbose=False):
    """
    Very fine-grained polishing: try small perturbations of individual
    coordinates and pairs, accepting only strict improvements.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f, obj_normalized(f)
    f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()
    t0 = time.monotonic()

    # Identifying critical coordinates for the dominant peak
    k_peak = int(np.argmax(conv))
    i_start = max(0, k_peak - n + 1)
    i_end = min(n - 1, k_peak)

    # Iterate: for each coordinate, try small multiplicative changes
    max_rounds = 50
    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        improved_round = False
        conv = fast_convolve(f)
        cur_max = conv.max()

        # Process coordinates in order of contribution to peak
        if i_end >= i_start:
            indices = np.arange(i_start, i_end + 1)
            contribs = f[indices] * f[k_peak - indices]
            order = np.argsort(contribs)[::-1]
        else:
            order = np.arange(n)

        for oi in order:
            if time.monotonic() - t0 > time_limit:
                break
            j = int(indices[oi]) if i_end >= i_start else int(oi)
            old_val = f[j]
            if old_val < 1e-14:
                continue

            # Try a range of multipliers
            for mult in [0.8, 0.9, 0.95, 1.05, 1.1, 1.25]:
                new_val = old_val * mult
                delta = old_val - new_val
                if abs(delta) < 1e-14:
                    continue

                trial = f.copy()
                trial[j] = new_val

                # Redistribute to a "safe" set of coordinates
                safe = np.ones(n, dtype=bool)
                # Mark all peak-contributing coords as unsafe
                for kp in np.where(conv >= cur_max * 0.98)[0][:5]:
                    kp = int(kp)
                    s2 = max(0, kp - n + 1)
                    e2 = min(n - 1, kp)
                    safe[s2:e2 + 1] = False
                safe[j] = False

                nsafe = safe.sum()
                if nsafe < 2:
                    safe = np.ones(n, dtype=bool)
                    safe[j] = False

                if delta > 0:
                    # Redistributing mass: add proportionally
                    w = trial[safe]
                    wsum = w.sum()
                    if wsum > 1e-15:
                        trial[safe] += delta * w / wsum
                    else:
                        trial[safe] += delta / nsafe
                else:
                    # Removing mass: remove proportionally from safe coords
                    w = trial[safe]
                    wsum = w.sum()
                    if wsum > abs(delta) + 1e-15:
                        trial[safe] -= abs(delta) * w / wsum
                        trial[safe] = np.maximum(trial[safe], 0)
                    else:
                        continue

                trial = np.maximum(trial, 0)
                ts = trial.sum()
                if ts < 1e-15:
                    continue
                trial = trial / ts

                obj = 2.0 * n * fast_convolve(trial).max()
                if obj < best_obj - 1e-15:
                    best_obj = obj
                    best_f = trial.copy()
                    f = trial
                    improved_round = True
                    break

        if not improved_round:
            break

        if verbose:
            print(f"    Polish round {rnd}: obj={best_obj:.10f}")

    return best_f, best_obj


def evaluate_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-10:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization combining accelerated mirror descent,
    Nesterov subgradient, projected coordinate descent, and fine polishing.
    Explores multiple sequence lengths and diverse initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Try to load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if np.isfinite(p_score) and p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # === Phase 1: Initialization and screening ===
    print("=== Phase 1: Initialization ===")
    init_candidates = []
    lengths_to_try = [512, 640, 768, 800, 1024, 1280]

    if parent is not None:
        for nl in lengths_to_try:
            init_candidates.append(interpolate_sequence(parent, nl))
        init_candidates.append(parent.copy())

    for nl in lengths_to_try:
        init_candidates.append(make_gaussian(nl, 0.25))
        init_candidates.append(make_gaussian(nl, 0.35))
        init_candidates.append(make_gaussian(nl, 0.4))
        init_candidates.append(make_uniform(nl))
        init_candidates.append(make_exponential(nl, 2.0))
        init_candidates.append(make_exponential(nl, 4.0))
        init_candidates.append(make_cosine(nl))
        init_candidates.append(make_platform(nl, 0.5))
        init_candidates.append(make_platform(nl, 0.7))
        init_candidates.append(make_bimodal(nl, 0.4))

    # Screen initializations
    best_init = None
    best_init_obj = float('inf')
    for ic in init_candidates:
        if len(ic) < 10:
            continue
        s = ic.sum()
        if s < 1e-10:
            continue
        a_norm = ic / s
        obj = obj_normalized(a_norm)
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a_norm.copy()
        best_seq, best_score = evaluate_and_update(a_norm, best_seq, best_score,
                                                    f"init_n{len(a_norm)}")

    print(f"  Best init: obj={best_init_obj:.6f} (n={len(best_init)})")

    # === Phase 2: Mirror Descent on best init ===
    print("\n=== Phase 2: Mirror Descent ===")
    rem = deadline - time.monotonic() - 600
    if rem > 60:
        t_alloc = min(rem, 300)
        print(f"  MD: n={len(best_init)}, t={t_alloc:.0f}s")
        a_opt, obj = mirror_descent_optimize(best_init, time_limit=t_alloc, verbose=True)
        best_seq, best_score = evaluate_and_update(a_opt, best_seq, best_score, "md_main")

    # === Phase 3: Accelerated Subgradient ===
    if best_seq is not None:
        print(f"\n=== Phase 3: Accelerated Subgradient (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 400
        if rem > 40:
            t_alloc = min(rem, 200)
            print(f"  ASG: t={t_alloc:.0f}s")
            a_opt, obj = accelerated_subgradient(best_seq, time_limit=t_alloc, verbose=True)
            best_seq, best_score = evaluate_and_update(a_opt, best_seq, best_score, "asg")

    # === Phase 4: Coordinate Descent ===
    if best_seq is not None:
        print(f"\n=== Phase 4: Coordinate Descent ===")
        rem = deadline - time.monotonic() - 250
        if rem > 30:
            t_alloc = min(rem, 150)
            print(f"  CD: t={t_alloc:.0f}s")
            a_opt, obj = projected_cd(best_seq, time_limit=t_alloc, verbose=False)
            best_seq, best_score = evaluate_and_update(a_opt, best_seq, best_score, "cd")

    # === Phase 5: Multi-length refinement ===
    print(f"\n=== Phase 5: Multi-length MD ===")
    base = best_seq if best_seq is not None else best_init
    for target_n in [640, 800, 900, 1024]:
        if len(base) == target_n:
            continue
        rem = deadline - time.monotonic() - 150
        if rem < 50:
            break
        f_interp = interpolate_sequence(base, target_n)
        best_seq, best_score = evaluate_and_update(f_interp, best_seq, best_score,
                                                    f"interp_n{target_n}")
        t_alloc = min(rem * 0.35, 100)
        if t_alloc < 20:
            continue
        print(f"  MD n={target_n}: t={t_alloc:.0f}s")
        a_opt, obj = mirror_descent_optimize(f_interp, time_limit=t_alloc, verbose=False)
        best_seq, best_score = evaluate_and_update(a_opt, best_seq, best_score,
                                                    f"md_n{target_n}")

    # === Phase 6: Deep mirror descent on best ===
    if best_seq is not None:
        print(f"\n=== Phase 6: Deep MD (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 100
        if rem > 40:
            t_alloc = min(rem, 180)
            print(f"  Deep MD: t={t_alloc:.0f}s")
            a_opt, obj = mirror_descent_optimize(best_seq, time_limit=t_alloc, verbose=True)
            best_seq, best_score = evaluate_and_update(a_opt, best_seq, best_score, "deep_md")

    # === Phase 7: Perturbed variants ===
    if best_seq is not None:
        print(f"\n=== Phase 7: Perturbed MD ===")
        n_best = len(best_seq)
        for p in range(4):
            rem = deadline - time.monotonic() - 60
            if rem < 30:
                break
            t_alloc = min(rem / (4 - p), 80)

            if p == 0:
                noise = rng.standard_normal(n_best) * 0.002
                f_init = project_simplex(best_seq + noise)
            elif p == 1:
                f_init = 0.9 * best_seq + 0.1 * (np.ones(n_best) / n_best)
                f_init = f_init / f_init.sum()
            elif p == 2:
                # Reverse blend
                f_init = 0.5 * best_seq + 0.5 * best_seq[::-1]
                f_init = f_init / f_init.sum()
            else:
                # Random restart with structure preservation
                f_init = best_seq.copy()
                mask = rng.random(n_best) < 0.3
                f_init[mask] *= rng.uniform(0.5, 1.5, mask.sum())
                f_init = np.maximum(f_init, 1e-10)
                f_init = f_init / f_init.sum()

            print(f"  Perturb {p}: t={t_alloc:.0f}s")
            a_opt, obj = mirror_descent_optimize(f_init, time_limit=t_alloc, verbose=False)
            best_seq, best_score = evaluate_and_update(a_opt, best_seq, best_score,
                                                        f"perturb_{p}")

    # === Phase 8: Fine polishing ===
    if best_seq is not None:
        print(f"\n=== Phase 8: Fine Polish ===")
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t_alloc = min(rem, 120)
            print(f"  Polish: t={t_alloc:.0f}s")
            a_opt, obj = fine_polish(best_seq, time_limit=t_alloc, verbose=True)
            best_seq, best_score = evaluate_and_update(a_opt, best_seq, best_score, "polish")

    # === Phase 9: Final CD + MD ===
    if best_seq is not None:
        print(f"\n=== Phase 9: Final refinement ===")
        rem = deadline - time.monotonic() - 15
        if rem > 20:
            t1 = rem * 0.5
            t2 = rem * 0.5

            if t1 > 10:
                a_opt, obj = projected_cd(best_seq, time_limit=t1)
                best_seq, best_score = evaluate_and_update(a_opt, best_seq, best_score, "fin_cd")

            rem2 = deadline - time.monotonic() - 10
            if rem2 > 10 and best_seq is not None:
                a_opt, obj = mirror_descent_optimize(best_seq, time_limit=rem2)
                best_seq, best_score = evaluate_and_update(a_opt, best_seq, best_score, "fin_md")

            rem3 = deadline - time.monotonic() - 5
            if rem3 > 10 and best_seq is not None:
                a_opt, obj = fine_polish(best_seq, time_limit=rem3)
                best_msg, best_sco = evaluate_and_update(a_opt, best_seq, best_score, "fin_polish")

    # Fallback
    if best_seq is None:
        best_seq = make_gaussian(768, 0.3)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
