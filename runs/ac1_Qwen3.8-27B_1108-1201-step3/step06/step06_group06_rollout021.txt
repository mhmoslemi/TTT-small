        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        f = f_new
        tau = max(tau * 0.9995, 1e-10)
        if no_improve > 200:
            lr *= 0.8
        if no_improve > 600:
            rng = np.random.default_rng(t * 13 + 7)
            noise = 0.002 + 0.002 * min(no_improve / 1000.0, 2.0)
            f = project_simplex(best_f + rng.standard_normal(n) * noise)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = lr_init * 0.6
            tau = 0.05 * fast_convolve(f).max()
    return best_f, best_f, 2 * n * best_obj

def away_fw_optimize(f0, time_limit, verbose=False):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    tau = 0.05 * M
    # Active set
    S = set(np.nonzero(f > 1e-10)[0].tolist())
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        if len(S) == 0:
            S = set(range(n))
        grad, conv = smooth_grad(f, tau)
        M = conv.max()
        # Find towards and away vertices
        S_list = list(S)
        g_S = grad[S_list]
        s_idx = S_list[np.argmin(g_S)]  # towards
        a_idx = S_list[np.argmax(g_S)]  # away
        # Try towards step
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = f.copy()
            ft[s_idx] += alpha * (1 - ft[s_idx])
            ft /= ft.sum()
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        # Try away step
        if f[a_idx] > 1e-8:
            for beta in [0.1, 0.2, 0.3, 0.5]:
                ft = f.copy()
                ft[a_idx] *= (1 - beta)
                ft /= ft.sum()
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = -beta  # negative means away
        if best_val < M - 1e-14:
            if best_alpha > 0:
                f[s_idx] += best_alpha * (1 - f[s_idx])
                S.add(s_idx)
            else:
                f[a_idx] *= (1 + best_alpha)  # best_alpha is negative
                if f[a_idx] < 1e-10:
                    S.discard(a_idx)
            f /= f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"    AwayFW iter {it}: obj={2*n*best_obj:.8f}")
        else:
            no_improve += 1
            if no_improve > 50:
                tau = max(tau * 0.95, 1e-10)
            if no_improve > 200:
                rng = np.random.default_rng(it * 17 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                S = set(np.nonzero(f > 1e-10)[0].tolist())
                no_improve = 0
                tau = 0.05 * fast_convolve(f).max()
    return best_f, 2 * n * best_obj

def mirror_descent_optimize(f0, time_limit, verbose=False):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    tau = 0.05 * M
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        grad, conv = smooth_grad(f, tau)
        M = conv.max()
        # Mirror descent step: f <- f * exp(-eta * grad) / sum(...)
        eta = 1.0 / (np.max(np.abs(grad)) + 1e-10)
        log_f = np.log(f + 1e-30) - eta * grad
        log_f -= log_f.max()
        f_new = np.exp(log_f)
        f_new = f_new / f_new.sum()
        f_new = np.maximum(f_new, 0)
        f_new = f_new / f_new.sum()
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 100 == 0:
                print(f"    Mirror iter {it}: obj={2*n*best_obj:.8f}")
        else:
            no_improve += 1
            # Backtrack
            for backtrack in [0.5, 0.2, 0.1]:
                ft = f + backtrack * (f_new - f)
                ft = project_simplex(ft)
                val = fast_convolve(ft).max()
                if val < M:
                    f = ft
                    break
        tau = max(tau * 0.999, 1e-10)
        if no_improve > 200:
            rng = np.random.default_rng(it * 23 + 11)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            no_improve = 0
    return best_f, 2 * n * best_obj

def peak_cd(f0, time_limit, verbose=False):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()
        improved = False
        # Find peak
        pk = int(np.argmax(conv))
        # Find contributing pairs
        js = max(0, pk - n + 1)
        je = min(n - 1, pk)
        idx = np.arange(js, je + 1)
        products = f[idx] * f[pk - idx]
        # Top contributors
        top_contrib = idx[np.argsort(products)][-3:]
        for j_max in top_contrib:
            j_other = pk - j_max
            old_val = f[j_max]
            if old_val < 1e-12:
                continue
            # Try reducing j_max and redistributing
            for factor in [0.7, 0.8, 0.9, 0.95]:
                new_val = old_val * factor
                diff = old_val - new_val
                trial = f.copy()
                trial[j_max] = new_val
                # Redistribute to non-contributing elements
                mask = np.ones(n, dtype=bool)
                # Exclude elements contributing to top 3 peaks
                top_peaks = np.argsort(conv)[-3:]
                for pk2 in top_peaks:
                    js2 = max(0, int(pk2) - n + 1)
                    je2 = min(n - 1, int(pk2))
                    mask[js2:je2 + 1] = False
                if mask.sum() > 0:
                    trial[mask] += diff * f[mask] / f[mask].sum()
                trial = np.maximum(trial, 0)
                trial /= trial.sum()
                obj = fast_convolve(trial).max()
                if obj < best_obj - 1e-14:
                    f = trial
                    best_obj = obj
                    best_f = f.copy()
                    improved = True
                    break
            if improved:
                break
        if not improved:
            no_improve += 1
            if no_improve > 20:
                rng = np.random.default_rng(int(time.monotonic() * 10000))
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                no_improve = 0
        else:
            no_improve = 0
        if verbose and it % 1000 == 0:
            print(f"    PeakCD iter {it}: obj={2*n*best_obj:.8f}")
        it += 1
    return best_f, 2 * n * best_obj

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def make_fejer(n):
    """Fejér-kernel-like initialization."""
    x = np.arange(n) - (n - 1) / 2.0
    a = (1.0 - np.abs(x) / ((n - 1) / 2.0 + 0.5)) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def make_w_shape(n):
    """W-shaped initialization (two peaks)."""
    x = np.arange(n)
    center1 = n * 0.3
    center2 = n * 0.7
    sigma = n * 0.15
    a = np.exp(-0.5 * ((x - center1) / sigma) ** 2) + np.exp(-0.5 * ((x - center2) / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def make_asymmetric_gaussian(n, center_frac=0.4, sigma_frac=0.3):
    x = np.arange(n) - n * center_frac
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a
