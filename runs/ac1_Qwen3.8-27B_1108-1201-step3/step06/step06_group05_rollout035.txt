"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an improved multi-phase approach:
1. Diverse initialization with parametric shapes and parent
2. LP-based Frank-Wolfe with cutting-plane acceleration (main workhorse)
3. Spectral equalization for diversification

6. Final coordinate descent polish

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_cutting_plane(a, M, active_set, max_constraints=100):
    """
    Solve LP: max sum(g) s.t. (a*g)[k] <= M for k in active_set, g >= 0.
    Returns (g, new_active_set) where new_active_set includes violated constraints.
    """
    n = len(a)
    k_list = list(active_set)
    m = len(k_list)
    
    if m == 0:
        # No constraints: return uniform
        return np.ones(n) / n, active_set
    
    # Build constraint matrix for active constraints only
    # A[k_list[i], j] = a[k_list[i] - j] if 0 <= k_list[i] - j < n
    A = np.zeros((m, n))
    for idx, k in enumerate(k_list):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A[idx, j] = a[k - j]
    
    c = -np.ones(n)
    b_ub = np.full(m, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': 30, 'mip_rel_gap': 0.0})
    
    if not res.success or res.x is None:
        return None, active_set
    
    g = res.x
    
    # Check which constraints are (near) violated for the new point
    # We'll check all constraints, not just active ones
    ag = np.convolve(a, g)
    violated = np.where(ag > M * 0.999)[0]
    
    # Update active set: keep top constraints by violation
    if len(violated) > max_constraints:
        # Keep the most violated ones
        viol_vals = ag[violated]
        top_idx = violated[np.argsort(viol_vals)[::-1][:max_constraints]]
        new_active = set(top_idx.tolist())
    else:
        new_active = set(violated.tolist())
    
    # Also always keep the current peaks
    current_peaks = np.where(ag >= M * 0.95)[0]
    new_active.update(current_peaks.tolist())
    
    if len(new_active) > max_constraints * 2:
        # Trim: keep most violated
        all_ag = ag[list(new_active)]
        top = list(new_active)
        top.sort(key=lambda k: ag[k], reverse=True)
        new_active = set(top[:max_constraints])
    
    return g, new_active

    """
    Diversify by modifying the spectral content while maintaining non-negativity.
    Tries to flatten the convolution in the frequency domain.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    
    # Try different spectral modifications
    for trial in range(20):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Random spectral perturbation
        A = np.fft.rfft(a)
        
        # Perturb the magnitude with a smooth filter
        mags = np.abs(A)
        phases = np.angle(A)
        
        # Apply a random smooth filter to the magnitude
        freqs = np.arange(len(mags))
        sigma_f = rng.uniform(0.2, 0.8) * len(mags)
        center = rng.uniform(0.1, 0.9) * len(mags)
        filter_curve = 1.0 + rng.uniform(0.1, 0.5) * np.exp(-0.5 * ((freqs - center) / sigma_f) ** 2)
        
        mags_new = mags * filter_curve
        A_new = mags_new * np.exp(1j * phases)
        a_new = np.fft.irfft(A_new, n)
        
        # Project to simplex
        a_new = project_simplex(a_new)
        
        obj = fast_convolve(a_new).max()
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"    Spectral trial {trial}: obj={obj:.10f}")
    
    return best_a, best_obj


def greedy_peak_reduction(a0, time_limit, verbose=False):
    """
    Greedy approach: identify peaks, find responsible coordinates,
    and redistribute mass to reduce peaks.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        M = b.max()
        
        # Find top peaks
        n_peaks = min(10, 2 * n - 1)
        peak_indices = np.argsort(b)[::-1][:n_peaks]
        
        # For each peak, find the most contributing coordinate
        improved = False
        for k in peak_indices:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Find which a[j] contribute to this peak
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            
            # Contribution of a[j] to b[k] is a[j] * a[k-j]
            # Reducing a[j] by delta reduces b[k] by delta * a[k-j] (approximately)
            # but also affects other b values
            
            # Try reducing the largest contributor
            contributions = np.array([a[j] * a[k - j] for j in range(js, je + 1)])
            j_rel = js + np.argmax(contributions)
            
            if a[j_rel] < 1e-10:
                continue
            
            # Try different reduction amounts
            best_red_obj = best_obj
            best_red_frac = 0.0
            
            for frac in [0.05, 0.1, 0.15, 0.2, 0.3]:
                delta = a[j_rel] * frac
                if delta < 1e-12:
                    continue
                
                # Reduce a[j_rel] and distribute to least-contributing positions
                a_trial = a.copy()
                a_trial[j_rel] -= delta
                
                # Distribute to positions that don't contribute to top peaks
                peak_js = set()
                for pk in peak_indices:
                    for j in range(max(0, pk - n + 1), min(n - 1, pk) + 1):
                        peak_js.add(j)
                
                safe_js = [j for j in range(n) if j not in peak_js and j != j_rel]
                if not safe_js:
                    safe_js = [j for j in range(n) if j != j_rel]
                
                if not safe_js:
                    continue
                
                # Distribute proportionally to current values (or uniformly)
                weights = np.array([a[j] + 1e-10 for j in safe_js])
                weights = weights / weights.sum()
                
                for idx, j in enumerate(safe_js):
                    a_trial[j] += delta * weights[idx]
                
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                
                obj = fast_convolve(a_trial).max()
                if obj < best_red_obj - 1e-12:
                    best_red_obj = obj
                    best_red_frac = frac
                    best_a_trial = a_trial.copy()
            
            if best_red_frac > 0:
                a = best_a_trial
                if best_red_obj < best_obj - 1e-12:
                    best_obj = best_red_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            # Try small random perturbations
            rng = np.random.default_rng(it * 41 + 7)
            for _ in range(5):
                noise = rng.standard_normal(n) * 0.001
                a_trial = project_simplex(a + noise)
                obj = fast_convolve(a_trial).max()
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a_trial.copy()
                    a = a_trial
                    improved = True
            if not improved:
                break
    
    return best_a, best_obj


def coordinate_polish(a0, time_limit, verbose=False):
    """Coordinate descent: adjust pairs of coordinates to reduce max convolution."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, fast_convolve(a0).max()
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Precompute which pairs of indices affect which convolution values
    while time.monotonic() - t0 < time_limit:
        improved = False
        b = fast_convolve(a)
        
        # Focus on coordinates with significant mass
        significant = np.where(a > 1e-6)[0]
        
        for j in significant:
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            if old_val < 1e-8:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            best_j_rest = None
            
            for mult in [0.7, 0.8, 0.9, 0.95, 1.05, 1.1, 1.2, 1.3]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-13:
                    best_j_obj = val
                    best_j_val = new_val
                    best_j_rest = a_trial.copy()
            
            if best_j_val != old_val and best_j_rest is not None:
                a = best_j_rest
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_a, best_obj


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_super_sine(n, power=2):
    """a[i] = sin(pi*(i+0.5)/n)^power, normalized."""
    x = np.pi * (np.arange(n) + 0.5) / n
    a = np.sin(x) ** power
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_flattened(n, flat_frac=0.5, smooth=0.3):
    """Flattened top: constant in middle, smooth transition to zero at edges."""
    x = np.arange(n) / (n - 1.0) * 2.0 - 1.0  # [-1, 1]
    edge = flat_frac / 2.0
    if np.abs(x) <= edge:
        a = 1.0
    else:
        t = (np.abs(x) - edge) / (1.0 - edge)
        t = np.clip(t, 0, 1)
        a = (1 - t) ** (1.0 / smooth) if smooth > 0 else (1 - t)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    kernel = np.array([0.1, 0.2, 0.4, 0.2, 0.1])
    f_smooth = np.convolve(f_interp, kernel, mode='same')
    f_smooth = np.maximum(f_smooth, 0)
    ss = f_smooth.sum()
    if ss > 1e-15:
        f_smooth /= ss
    return f_smooth


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization with LP-FW cutting plane, spectral diversification,
    greedy peak reduction, and coordinate polish.

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)

        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:

            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
