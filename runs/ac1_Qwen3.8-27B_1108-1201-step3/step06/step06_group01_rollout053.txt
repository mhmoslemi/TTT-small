    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_two(a, b):
    """Linear convolution of two sequences via FFT."""
    n1 = len(a)
    n2 = len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def compute_grad(a, w):
    """
    Compute gradient of 2n * sum_k w[k] * b[k] where b = conv(a,a).
    w is a probability vector (sum=1) of length 2n-1.
    Returns gradient of length n.
    """
    n = len(a)
    a_flipped = a[::-1].copy()
    # Term 2: conv(w, a)[:n]
    conv_wa = convolve_two(w, a)
    term2 = conv_wa[:n]
    # Term 1: conv(w, a_flipped)[n-1 : 2n-1]
    conv_waf = convolve_two(w, a_flipped)
    term1 = conv_waf[n-1 : 2*n-1]
    return 2.0 * n * (term1 + term2)


def soft_max_obj_grad(a, T):
    """
    Compute soft-max smoothed objective and gradient.
    Objective: 2n * logsumexp(T * conv(a,a))
    """
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = 2.0 * n * log_sum_exp
    w = exp_Tb / exp_Tb.sum()
    grad = compute_grad(a, w)
    return obj, grad


def softmax(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def mirror_descent(a0, time_limit, T_init=50.0, T_final=5.0, 
                   eta_init=0.01, momentum=0.9):
    """
    Mirror descent (multiplicative weights) on the simplex with adaptive temperature.
    """
    n = len(a0)
    a = np.maximum(a0.copy(), 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    v = np.zeros(n)  # momentum buffer (in log-space)
    t0 = time.monotonic()
    no_improve = 0
    step = 0
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        # Adaptive temperature
        progress = min(1.0, step / (time_limit / 0.05))  # rough progress estimate
        T = T_init * (T_final / T_init) ** progress
        
        b = fast_convolve(a)
        Tb = T * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        
        # Compute gradient
        g = compute_grad(a, w)
        
        # Normalize gradient for stability
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-30:
            break
        g = g / g_norm
        
        # Momentum in log-space
        v = momentum * v - eta_init * g
        lr = eta_init / (1.0 + 0.1 * step)
        v = momentum * v - lr * g
        
        # Multiplicative update (mirror step)
        # a_i <- a_i * exp(v_i) / Z
        z = np.log(a) + v
        a_new = softmax(z)
        
        # Evaluate
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        if no_improve > 200:
            # Restart with perturbation
            a = best_a + np.random.standard_normal(n) * 0.002
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            v = np.zeros(n)
            no_improve = 0
        
        if step % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD step={step} ({elapsed:.0f}s): T={T:.1f} score={best_score:.6f}")
    
    return best_a, best_score


def lbgfs_logspace(a0, time_limit, T_levels=None, maxiter_per_level=2000):
    """
    L-BFGS in log-space with multi-level temperature annealing.
    """
    n = len(a0)
    if T_levels is None:
        T_levels = [200, 100, 50, 20, 10, 5]
    
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    
    for T in T_levels:
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 5:
            break
        
        t_level = remaining * 0.8
        obj_calls = 0
        
        def func(z_vec):
            nonlocal obj_calls
            obj_calls += 1
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a_vec = softmax(z_vec)
            b = fast_convolve(a_vec)
            Tb = T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            obj = 2.0 * n * log_sum_exp
            return obj
        
        def jac(z_vec):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_vec)
            a_vec = softmax(z_vec)
            b = fast_convolve(a_vec)
            Tb = T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            # Gradient w.r.t. a
            grad_a = compute_grad(a_vec, w)
            # Gradient w.r.t. z: grad_z = a * (grad_a - a * dot(grad_a, a))
            inner = np.dot(grad_a, a_vec)
            grad_z = a_vec * (grad_a - a_vec * inner)
            return grad_z
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': maxiter_per_level, 'ftol': 1e-15,
                         'gtol': 1e-12, 'maxfun': maxiter_per_level * 20,
                         'disp': False}
            )
            a_new = softmax(res.x)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new / s
                score = 2.0 * n * fast_convolve(a_new).max()
                if score < best_score - 1e-14:
                    best_score = score
                    best_a = a_new.copy()
                    z = res.x
        except Exception as e:
            print(f"    L-BFGS T={T} error: {e}")
            continue
        
        elapsed = time.monotonic() - t0
        print(f"    L-BFGS T={T} ({elapsed:.0f}s): obj_calls={obj_calls} score={best_score:.6f}")
    
    return best_a, best_score


def peak_redistribute(a0, time_limit):
    """
    Peak redistribution: identify the peak in auto-correlation,
    and redistribute mass from peak-contributing positions to reduce it.
    """
    n = len(a0)
    a = np.maximum(a0.copy(), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    iterations = 0
    
    while time.monotonic() - t0 < time_limit:
        iterations += 1
        b = fast_convolve(a)
        # Find the peak (can be at any position, not just center)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]
        
        if peak_val <= best_score / (2.0 * n) + 1e-15:
            break
        
        # Find the positions that contribute to this peak
        # b[peak_k] = sum_i a[i] * a[peak_k - i] (for valid i)
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Find the top contributors
        top = np.argsort(contribs)[::-1]
        
        improved = False
        for ci in top[:5]:
            i = idx[ci]
            j = peak_k - i
            if a[i] < 1e-12:
                continue
            
            # Try reducing a[i] and distributing to other positions
            # that don't contribute to this peak
            old_val = a[i]
            for factor in [0.7, 0.5, 0.3, 0.1]:
                new_val = old_val * factor
                diff = old_val - new_val
                if diff < 1e-14:
                    continue
                
                a_trial = a.copy()
                a_trial[i] = new_val
                
                # Distribute diff to positions that don't contribute to peak_k
                # A position p contributes to peak_k if 0 <= peak_k - p < n
                contrib_mask = np.zeros(n, dtype=bool)
                pk_start = max(0, peak_k - n + 1)
                pk_end = min(n - 1, peak_k)
                contrib_mask[pk_start:pk_end+1] = True
                contrib_mask[i] = False  # we already reduced i
                
                if not contrib_mask.any():
                    # All positions contribute; distribute to the least contributing
                    a_trial[~np.arange(n) == i] += diff * a[~np.arange(n) == i] / (s - old_val)
                else:
                    mask = contrib_mask
                    others_sum = a_trial[mask].sum()
                    if others_sum > 1e-15:
                        a_trial[mask] += diff * a_trial[mask] / others_sum
                    else:
                        # Distribute uniformly
                        a_trial[mask] += diff / mask.sum()
                
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            
            if improved:
                break
        
        if not improved:
            # Try reducing both i and j simultaneously
            for ci in top[:3]:
                i = idx[ci]
                j = peak_k - i
                if i == j or a[j] < 1e-12:
                    continue
                for f1 in [0.5, 0.3]:
                    for f2 in [0.5, 0.3]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j]
