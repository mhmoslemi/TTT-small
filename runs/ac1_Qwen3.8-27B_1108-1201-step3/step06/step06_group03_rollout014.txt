"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase optimization:
Adam with log-sum-exp smoothing, LP-Frank-Wolfe with adaptive relaxation,

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient_topk(f, conv, k, temperature=1.0):
    """Compute subgradient using top-k peaks with softmax weighting."""
    n = len(f)
    len_c = len(conv)
    k = max(1, min(k, len_c))
    peaks = np.argpartition(conv, -k)[-k:]
    pv = conv[peaks]
    scaled = pv * temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(k)
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    # Full-length weight array
    w_full = np.zeros(len_c)
    w_full[peaks] = w
    
    # grad[j] = 2 * sum_m f[m] * w_full[j+m] for j=0,...,n-1
    # Use FFT for efficiency
    fft_size = 1 << (len_c + n - 1).bit_length()
    W = np.fft.rfft(w_full, fft_size)
    F = np.fft.rfft(f, fft_size)
    # Cross-correlation: grad[j] = 2 * sum_m f[m] * w_full[j+m]
    # = 2 * ifft(conj(fft(w_full)) * fft(f))[j + n - 1] ... 
    # Actually, let me use a simpler approach.
    # sum_m f[m] * w_full[m+j] is the cross-correlation of w_full and f at lag j.
    # Using FFT: correlate(a, b) = ifft(conj(fft(a)) * fft(b))
    # Wait, I need to be careful with the definition.
    # Let's use: grad[j] = 2 * (w_full * f)[j] where * denotes cross-correlation
    # Cross-correlation: (w_full ⋆ f)[j] = sum_m w_full[m] * f[m-j] ... no
    # Let me just define it as: C[j] = sum_m f[m] * w_full[m+j]
    # This is the cross-correlation of w_full with f, i.e., C = correlate(w_full, f)
    # In numpy: np.correlate(w_full, f, 'full')
    # But for efficiency with FFT:
    # C[j] = sum_m f[m] * w_full[m+j]
    # Let's flip f: f_flip[m] = f[n-1-m]
    # Then C[j] = sum_m f[n-1-m] * w_full[m+j] ... this is getting complicated.
    # For n=768, np.correlate is fast enough (O(n * len_c) ≈ 768 * 1535 ≈ 1.2M ops).
    
    corr = np.correlate(w_full, f, mode='full')
    # np.correlate(w_full, f, 'full')[i] = sum_n w_full[n + i - (len(f)-1)] * f[n]
    # We want C[j] = sum_m f[m] * w_full[m+j]
    # Let i - (len(f)-1) = j, so i = j + len(f) - 1 = j + n - 1
    # np.correlate(w_full, f, 'full')[j + n - 1] = sum_n w_full[n + j] * f[n] = C[j] ✓
    
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad

def adam_lse(f0, time_limit, lr=0.005, tau_start=3.0, tau_end=80.0,
             topk_frac=0.15, restart_every=150, verbose=False, seed=42):
    """Adam optimization with log-sum-exp smoothing."""
    rng = np.random.default_rng(seed)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    t = 0
    t0 = time.monotonic()
    no_improve = 0
    
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        frac = min(elapsed / time_limit, 1.0)
        tau = tau_start * (tau_end / tau_start) ** frac
        
        conv = fast_convolve(f)
        M = conv.max()
        
        # Compute subgradient
        k = max(5, int(len(conv) * topk_frac))
        grad = compute_subgradient_topk(f, conv, k, temperature=tau)
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            grad = grad / gnorm
        
        # Adam update
        m_vec = beta1 * m_vec + (1 - beta1) * grad
        v_vec = beta2 * v_vec + (1 - beta2) * (grad ** 2)
        m_hat = m_vec / (1 - beta1 ** t)
        v_hat = v_vec / (1 - beta2 ** t)
        
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f - step)
        
        obj_new = fast_convolve(f_new).max()
        
        if obj_new < M - 1e-15:
            f = f_new
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_f = f.copy()
                no_improve = 0
                if verbose and t % 200 == 0:
                    print(f"      AdamLSE iter {t}: obj={best_obj:.10f}, tau={tau:.1f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            # Halve learning rate
            lr = max(lr * 0.7, 1e-7)
        
        if no_improve > restart_every:
            noise = rng.standard_normal(n) * (0.002 + 0.002 * min(no_improve / restart_every, 2.0))
            f = project_simplex(best_f + noise)
            m_vec = np.zeros(n)
            v_vec = np.zeros(n)
            t = 0
            lr = max(0.005, 1e-6)
            no_improve = 0
    
    return best_f, best_obj

def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    c_obj = -np.ones(n)
    # Build constraint matrix: (f*g)[k] = sum_j f[j]*g[k-j] <= M
    # For each k, the constraint is: sum_{j: 0<=k-j<n} f[j]*g[k-j] <= M
    # Let i = k-j, so j = k-i: sum_{i: 0<=k-i<n} f[k-i]*g[i] <= M
    # For a fixed j (column), the entries are f[j] at row (j+i) for i=0,...,n-1
    # So A[k, j] = f[j] if 0 <= k-j < n, else 0
    # Equivalently: A[k, j] = f[j] if j <= k <= j+n-1
    
    # Build sparse matrix
    A_rows = []
    A_cols = []
    A_vals = []
    for j in range(n):
        rows = np.arange(j, j + n)
        A_rows.extend(rows.tolist())
        A_cols.extend([j] * n)
        A_vals.extend([f[j]] * n)
    
    from scipy.sparse import csr_matrix
    A_mat = csr_matrix((A_vals, (A_rows, A_cols)), shape=(2*n-1, n))
    b_mat = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_mat, b_ub=b_mat, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def lp_fw(f0, time_limit, verbose=False, seed=42):
    """LP-based Frank-Wolfe with adaptive relaxation."""
    rng = np.random.default_rng(seed)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    relax = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        g = lp_direction(f, M * relax, n)
        if g is None:
            relax = min(relax * 1.1, 5.0)
            if relax > 3.0:
                # Restart
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                relax = 1.0
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            relax = min(relax * 1.05, 5.0)
            continue
        g = g / g_sum
        
        # Check if g is close to f (FW point)
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            relax = min(relax * 1.02, 10.0)
            if no_improve > 100:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                conv = fast_convolve(f)
                M = conv.max()
                relax = 1.0
                no_improve = 0
            continue
        else:
            no_improve = 0
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new
            relax = 1.0
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                if verbose and it % 10 == 0:
                    print(f"      LPFW iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            relax = min(relax * 1.02, 5.0)
        
        if no_improve > 200:
            break
    
    return best_f, best_obj

def coordinate_descent(f0, time_limit, verbose=False, seed=42):
    """Aggressive coordinate descent."""
    rng = np.random.default_rng(seed)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    sweep = 0
    
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        improved = False
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj
            
            old_val = f[j]
            if old_val < 1e-14:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            # Try a range of perturbations
            base_deltas = [-0.4, -0.3, -0.2, -0.1, -0.05, 0.05, 0.1, 0.2, 0.3, 0.5, -0.5]
            deltas = [old_val * d for d in base_deltas]
            # Also try small absolute perturbations
            deltas.extend([0.001, -0.001, 0.005, -0.005])
            
            for delta in deltas:
                new_val = old_val + delta
                if new_val < 0 or new_val > 0.8:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial = f_trial / ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            # Try a random restart with small perturbation
            if sweep > 3:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                best_obj_cur = fast_convolve(f).max()
                if best_obj_cur < best_obj:
                    best_obj = best_obj_cur
                    best_f = f.copy()
                sweep = 0
    
    return best_f, best_obj

def make_exp_init(n, rate, seed=42):
    """Exponential decay initialization."""
    x = np.arange(n)
    a = np.exp(-rate * x / (n - 1))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses Adam with log-sum-exp, LP-Frank-Wolfe, coordinate descent,
and simulated annealing. All on the probability simplex (sum=1, a>=0).
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_topk(f, conv, k, temperature):
    n = len(f)
    len_c = len(conv)
    k = max(1, min(k, len_c))
    peaks = np.argpartition(conv, -k)[-k:]
    pv = conv[peaks]
    scaled = pv * temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(k)
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    w_full = np.zeros(len_c)
    w_full[peaks] = w
    
    corr = np.correlate(w_full, f, mode='full')
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


def adam_lse(f0, time_limit, lr, tau_start, tau_end, topk_frac, restart_every, verbose, seed):

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()

        f = f / s

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    m_vec = np.zeros(n)
    v_vec = np.zeros(n)

    no_improve = 0

    beta2 = 0.999

    while time.monotonic() - t0 < time_limit:

        elapsed = time.monotonic() - t0
        frac = min(elapsed / time_limit, 1.0)
        tau = tau_start * (tau_end / tau_start) ** frac

        M = conv.max()
        
        k = max(5, int(len(conv) * topk_frac))

        if gnorm > 1e-15:

        v_vec = beta2 * v_vec + (1 - beta2) * (grad ** 2)
        m_hat = m_vec / (1 - beta1 ** t)
        v_hat = v_vec / (1 - beta2 ** t)

            v_vec = np.zeros(n)
