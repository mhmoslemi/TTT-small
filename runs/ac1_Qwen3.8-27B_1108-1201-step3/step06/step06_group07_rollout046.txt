"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using LP-based mirror descent
with adaptive constraint tightening, diverse initializations, and coordinate
descent polishing. The LP approach is more principled than gradient methods:
it finds the steepest descent direction in the simplex by solving a linear
program where the cross-convolution with the current sequence is bounded.

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build the (2n-1) x n convolution matrix A where A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A[k, j] = a[k - j]
    return A

def lp_solve(a, M, max_sum=None):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)  # minimize -sum(g)
    A = build_conv_matrix(a)
    b = np.full(2*n-1, M)
    
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                           'primal_feasibility_tolerance': 1e-9})
    if not res.success:
        return None
    return res.x

def coordinate_descent(a0, time_limit, verbose=False):
    """Fine coordinate descent: adjust one coordinate at a time."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()

    while improved_any and time.monotonic() - t0 < time_limit:
        improved_any = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            if old_val < 1e-12:
                continue
            
            best_j_obj = best_M
            best_j_val = old_val
            
            # Try several adjustments
            for mult in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                trial = a.copy()
                trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                trial[mask] *= scale
                trial = np.maximum(trial, 0.0)
                ss = trial.sum()
                if ss < 1e-15:
                    continue
                trial = trial / ss
                val = fast_convolve(trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                if best_j_obj < best_M - 1e-14:
                    best_M = best_j_obj
                    best_a = a.copy()
                    improved_any = True

def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_asymmetric(n, center_frac=0.4, sigma_frac=0.25):
    """Asymmetric Gaussian centered away from middle."""
    x = np.arange(n)
    center = n * center_frac
    sigma = n * sigma_frac
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def symmetrize(a):
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    ss = a_sym.sum()
    if ss > 1e-15:
        a_sym /= ss
    return a_sym

            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

        n_base = len(parent)

    print("=== Phase 1: Initial evaluations ===")
    init_candidates = []

        init_candidates.append((make_gaussian(n_try, 0.25), f"gauss_{n_try}"))

        init_candidates.append((make_bell(n_try, 0.5), f"bell_{n_try}"))
    
    best_init = None
    best_init_score = float('inf')
    for a_init, label in init_candidates:
        s = a_init.sum()
        if s < 1e-10:
            continue
        a_norm = a_init / s
        sc = try_seq(a_norm, f"init_{label}")
        if sc < best_init_score:
            best_init_score = sc
            best_init = a_norm.copy()
    
    if best_init is None:
        best_init = make_gaussian(n_base, 0.3)

    # === Phase 2: LP-based mirror descent ===

            # Start from symmetrized version

    # === Phase 4: Coordinate descent ===
    if best_seq is not None:

            a_opt, obj = coordinate_descent(best_seq, time_limit=t_cd, verbose=True)

    if best_seq is not None:

            n_restarts = 3

                remaining = deadline - time.monotonic()

                    f_init = np.maximum(best_seq + noise, 1e-10)
                    f_init = f_init / f_init.sum()
                else:
