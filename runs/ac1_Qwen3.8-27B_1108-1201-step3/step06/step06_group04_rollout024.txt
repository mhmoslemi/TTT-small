`f(a + t*d) = (a + t*d)^T C_{k*} (a + t*d)`
`= a^T C_{k*} a + 2t * a^T C_{k*} d + t^2 * d^T C_{k*} d`
`= M + 2t * (C_{k*} a)^T d + t^2 * d^T C_{k*} d`
`= M + 2t * (grad/2)^T d + t^2 * d^T C_{k*} d`
`= M + t * grad^T d + t^2 * d^T C_{k*} d`

Now, `grad^T d = grad^T (e_j - a) = grad[j] - grad^T a`
`grad^T a = (2*C_{k*}*a)^T * a = 2*a^T C_{k*} a = 2*M`

So `grad^T d = grad[j] - 2*M`

And `d^T C_{k*} d = (e_j - a)^T C_{k*} (e_j - a) = (C_{k*})[j,j] - 2*(C_{k*}*a)[j] + a^T C_{k*} a`
`= (C_{k*})[j,j] - grad[j] + M`

where `(C_{k*})[j,j] = 1` if `2j = k*`, else 0 (from our earlier analysis of C_k).

So the quadratic approximation is:
`f(a + t*d) ≈ M + t*(grad[j] - 2*M) + t^2 * ((C_{k*})[j,j] - grad[j] + M)`

Let `A = grad[j] - 2*M` (linear coefficient, should be negative since grad[j] is the minimum)
Let `B = (C_{k*})[j,j] - grad[j] + M` (quadratic coefficient)

Minimize `M + A*t + B*t^2` over `t ∈ [0, 1]`:
- If B > 0: `t* = -A/(2B)` (clipped to [0,1])
- If B ≤ 0: `t* = 1` (quadratic is decreasing or linear)

This gives us an O(1) line search (no FFT needed)! We just need to compute `grad[j]` and `(C_{k*})[j,j]`, both of which are O(1).

This is a huge speedup! Now each FW iteration is just:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast Frank-Wolfe with O(1) quadratic step-size (exploiting gradient structure of auto-convolution)
2. CVXPY-based QCQP refinement with active constraint management
3. Multiplicative mirror descent for final polish
4. Multi-length exploration with diverse initializations

Key insight: The gradient of conv(a,a)[k] w.r.t. a[i] is 2*a[k-i] (for valid indices),

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, active_ks, conv_vals, M):
    """Compute weighted subgradient from active constraints."""
    n = len(a)
    grad = np.zeros(n)
    total_w = 0.0
    for k in active_ks:
        w = (conv_vals[k] / M) ** 4 if M > 0 else 1.0
        total_w += w
        # grad[i] += w * 2 * a[k-i] for valid i
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_hi >= i_lo:
            grad[i_lo:i_hi + 1] += 2.0 * w * a[k - i_hi:k - i_lo + 1][::-1]
    if total_w > 0:
        grad /= total_w
    return grad


def fast_fw_optimize(a0, time_limit, verbose=False, n_top=30):
    """
    Fast Frank-Wolfe using the gradient structure of auto-convolution.
    Each iteration is O(n log n) dominated by FFT.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    momentum = 0.0
    y = a.copy()  # Nesterov momentum variable

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        # Nesterov extrapolation
        m_t = 0.5 / (np.sqrt(it) + 1.0)
        y = a + m_t * (y - a)
        y = np.maximum(y, 0)
        y = y / y.sum()

        conv = fast_convolve(y)
        M = conv.max()
        obj = 2.0 * n * M

        # Find active set (top peaks)
        top_k = min(n_top, len(conv))
        active_ks = np.argpartition(conv, -top_k)[-top_k:]
        active_ks = [int(k) for k in active_ks if conv[k] > 0.5 * M]

        if not active_ks:
            active_ks = [int(np.argmax(conv))]

        # Compute subgradient
        grad = compute_subgradient(y, active_ks, conv, M)

        # Find FW direction (minimize linear approximation over simplex)
        j_star = int(np.argmin(grad))

        # Quadratic step size
        # f(y + t*(e_j - y)) ≈ M + t*(grad[j] - 2*M) + t^2*(C[j,j] - grad[j] + M)
        k_star = int(active_ks[0])  # use primary active constraint
        A_coeff = grad[j_star] - 2.0 * M
        C_jj = 1.0 if (2 * j_star == k_star and k_star % 2 == 0) else 0.0
        B_coeff = C_jj - grad[j_star] + M

        if B_coeff > 1e-10:
            t_opt = max(0.0, min(1.0, -A_coeff / (2.0 * B_coeff)))
        else:
            t_opt = 1.0 if A_coeff < 0 else 0.0

        # Clip step to avoid being too aggressive
        t_opt = min(t_opt, 2.0 / (it + 2.0) * 3.0)  # cap standard FW rate

        # Update
        a_new = (1.0 - t_opt) * y + t_opt * np.eye(n)[j_star, :]
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new / sn

        # Evaluate
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Periodic verification and perturbation
        if no_improve > 500:
            if no_improve % 500 == 0:
                # Try a small perturbation
                rng = np.random.default_rng(it + 7)
                a_try = project_simplex(a + rng.standard_normal(n) * 0.001)
                obj_try = 2.0 * n * fast_convolve(a_try).max()
                if obj_try < best_obj:
                    a = a_try
                    best_obj = obj_try
                    best_a = a_try.copy()
                    no_improve = 0
            if no_improve > 5000:
                break

        if verbose and it % 5000 == 0:
            print(f"    FW iter {it}: obj={best_obj:.10f}, t={elapsed:.1f}s")

    return best_a, best_obj


def cvxpy_refine(a0, time_limit, K_constraints=50, verbose=False):
    """
    Solve a reduced QCQP using CVXPY. Only include top-K convolution constraints.
    """
    try:
        import cvxpy as cp
    except ImportError:
        return a0, 2.0 * len(a0) * fast_convolve(a0).max()

    a_init = np.maximum(a0.astype(np.float64), 0.0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(len(a0)) / len(a0)
    a_init = a_init / s
    n = len(a_init)

    best_a = a_init.copy()
    best_obj = 2.0 * n * fast_convolve(a_init).max()

    t0 = time.monotonic()

    # Get initial active constraints
    conv_init = fast_convolve(a_init)
    top_k = np.argpartition(conv_init, -K_constraints)[-K_constraints:]
    top_k = sorted(set(int(k) for k in top_k))

    while time.monotonic() - t0 < time_limit:
        # Build CVXPY problem with current active constraints
        try:
            a_var = cp.Variable(n)
            t_var = cp.Variable()

            constraints = [cp.sum(a_var) == 1, a_var >= 0]

            for k in top_k:
                j_lo = max(0, k - n + 1)
                j_hi = min(n - 1, k)
                if j_hi < j_lo:
                    continue
                j = np.arange(j_lo, j_hi + 1)
                terms = a_var[j] * a_var[k - j]
                constraints.append(cp.sum(terms) <= t_var)

            objective = cp.Minimize(2.0 * n * t_var)
            prob = cp.Problem(objective, constraints)

            # Try solving
            try:
                prob.solve(solver=cp.SCS, max_iters=10000, eps=1e-6)
            except Exception:
                try:
                    prob.solve(solver=cp.CLARABEL)
                except Exception:
                    try:
                        prob.solve(solver=cp.ECOS, max_iters=10000)
                    except Exception:
                        break

            if prob.status in ['optimal', 'optimal_inaccurate'] and a_var.value is not None:
                a_new = np.maximum(np.asarray(a_var.value).flatten(), 0.0)
                s_new = a_new.sum()
                if s_new < 1e-12:
                    break
                a_new = a_new / s_new
                obj_new = 2.0 * n * fast_convolve(a_new).max()

                if obj_new < best_obj - 1e-10:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    if verbose:
                        print(f"    CVXPY: obj={best_obj:.10f} (n={n}, {len(top_k)} constraints)")

                # Update active constraints
                conv_new = fast_convolve(a_new)
                new_top = np.argpartition(conv_new, -K_constraints)[-K_constraints:]
                new_top = set(int(k) for k in new_top)
                old_top = set(top_k)
                # Add new violated/near-active constraints
                top_k = sorted(old_top | new_top)
                top_k = top_k[:K_constraints * 2]  # cap the number
            else:
                break
        except Exception as e:
            if verbose:
                print(f"    CVXPY error: {e}")
            break

    return best_a, best_obj


def multiplicative_descent(a0, time_limit, verbose=False, lr_init=0.003):
    """Mirror descent with multiplicative weight updates targeting top peaks."""
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    temperature = 0.005
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()

        # Get top peaks
        k_actual = min(20, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        # Compute weighted gradient
        pv = conv[peaks]
        scaled = pv / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            kk = int(peaks[i])
            i_lo = max(0, kk - n + 1)
            i_hi = min(n - 1, kk)
            idx = np.arange(i_lo, i_hi + 1)
            sg[idx] += 2.0 * w[i] * a[kk - idx]

        a = a * np.exp(-lr * sg)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        obj = 2.0 * n * fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9995
        temperature *= 1.01

        if no_improve > 200:
            rng = np.random.default_rng(it * 43 + 11)
            a = np.maximum(best_a + rng.standard_normal(n) * 0.002, 1e-15)
            a = a / a.sum()
            lr = max(lr_init * 0.5, 1e-6)
            no_improve = 0

        if lr < 1e-9:
            break

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_asymmetric_smooth(n, alpha=1.5, beta=0.5):
    """Create an asymmetric smooth initial sequence."""
    x = np.linspace(0, 1, n)
    a = np.exp(-alpha * x) * (1.0 + beta * np.sin(np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_gaussian(n, sigma_frac=0.3, center=0.5):
    """Gaussian initial sequence."""
    x = np.linspace(0, 1, n)
    sigma = sigma_frac
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_ramped(n, power=2.0):
    """Ramped (monotone) initial sequence."""
    x = np.linspace(0, 1, n)
    a = x ** power
    a = a / a.sum()
    return a

def peak_shaving_cd(a0, time_limit, verbose=False):

        conv = fast_convolve(a)
        M = conv.max()
        k_peak = int(np.argmax(conv))
        improved = False

        # Identify contributing indices
        j_lo = max(0, k_peak - n + 1)
        j_hi = min(n - 1, k_peak)
        idx = np.arange(j_lo, j_hi + 1)
        products = a[idx] * a[k_peak - idx]
        total = products.sum()

        if total < 1e-15:
            no_improve += 1
            break

        # Try reducing top contributors and redistributing
        order = np.argsort(products)[::-1]
        for oi in range(min(6, len(order))):

            if old_val < 1e-12:

                if new_val < 0:
                    continue

                trial = a.copy()
                trial[j_max] = new_val
                d = old_val - new_val

                if abs(d) > 1e-15:
                    # Redistribute to non-contributing positions
                    mask = np.ones(n, dtype=bool)
                    mask[j_lo:j_hi + 1] = False
                    if mask.sum() > 0:
                        w = a[mask]
                        wsum = w.sum()
                        if wsum > 1e-15:

                trial = np.maximum(trial, 0)
                tsum = trial.sum()
                if tsum < 1e-15:
                    continue
                trial = trial / tsum

                obj = 2.0 * n * fast_convolve(trial).max()

                    best_obj = obj
                    best_a = trial.copy()
                    a = trial
                    improved = True

            # Aggressive: scale down all peak contributors

            d = a.sum() - trial.sum()
            mask = np.ones(n, dtype=bool)
            mask[j_lo:j_hi + 1] = False
            if mask.sum() > 0:
                w = a[mask]
                wsum = w.sum()
                if wsum > 1e-15:
                    trial[mask] += d * w / wsum
            trial = np.maximum(trial, 0)
            trial = trial / trial.sum()
            obj = 2.0 * n * fast_convolve(trial).max()
            if obj < best_obj:
                a = trial
                best_obj = obj
                best_a = trial.copy()
                no_improve = 0
            else:

            no_improve = 0

        rem = deadline - time.monotonic() - 350
        if rem > 30:
            t_alloc = min(rem, 180)
            print(f"  CVXPY: t={t_alloc:.0f}s")
            a_opt, obj = cvxpy_refine(best_seq, time_limit=t_alloc, K_constraints=40, verbose=True)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "cvxpy")

    # === Phase 4: Peak Shaving ===
    if best_seq is not None:
        print(f"\n=== Phase 4: Peak CD (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            t_alloc = min(rem, 80)
            print(f"  Peak CD: t={t_alloc:.0f}s")
            a_opt, obj = peak_shaving_cd(best_seq, time_limit=t_alloc, verbose=True)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "peak_cd")

    # === Phase 5: Multiplicative Descent ===
    if best_seq is not None:
        print(f"\n=== Phase 5: Multiplicative Descent (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 120
        if rem > 20:
            t_alloc = min(rem, 80)
            print(f"  MD: t={t_alloc:.0f}s")
            a_opt, obj = multiplicative_descent(best_seq, time_limit=t_alloc, verbose=True)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "md")

    # === Phase 6: Second round of FW on best ===
    if best_seq is not None:
        print(f"\n=== Phase 6: Second FW Pass (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 100
        if rem > 40:
            t_alloc = min(rem, 100)
            print(f"  FW pass 2: t={t_alloc:.0f}s")
            a_opt, obj = fast_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "fw_pass2")

    # === Phase 7: Multi-length exploration ===
    base = best_seq if best_seq is not None else best_init
    print(f"\n=== Phase 7: Multi-length Exploration ===")
    for target_n in [700, 800, 850, 900]:
        if len(base) == target_n:
            continue
        rem = deadline - time.monotonic() - 60
        if rem < 40:
            break
        f_interp = interpolate_sequence(base, target_n)
        best_seq, best_score = try_sequence(f_interp, best_seq, best_score, f"interp_n{target_n}")
        t_alloc = min(rem * 0.3, 40)
        if t_alloc < 15:
            continue
        print(f"  FW n={target_n}: t={t_alloc:.0f}s")
        a_opt, obj = fast_fw_optimize(f_interp, time_limit=t_alloc, verbose=False)
        best_seq, best_score = try_sequence(a_opt, best_seq, best_score, f"fw_n{target_n}")

    # === Phase 8: Final Polish ===
    if best_seq is not None:
        print(f"\n=== Phase 8: Final Polish (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 10
        if rem > 20:
            # Quick peak shaving
            t1 = min(rem * 0.4, 30)
            if t1 > 5:
                a_opt, obj = peak_shaving_cd(best_seq, time_limit=t1)
                best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "fin_peak")

            rem2 = deadline - time.monotonic() - 5
            if rem2 > 10 and best_seq is not None:
                t2 = min(rem2 * 0.5, 20)
                if t2 > 5:
                    a_opt, obj = fast_fw_optimize(best_seq, time_limit=t2, n_top=50)
                    best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "fin_fw")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
