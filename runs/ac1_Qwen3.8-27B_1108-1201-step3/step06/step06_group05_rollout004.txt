Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Active-set LP-based Frank-Wolfe (sparse constraints for speed)
2. Multi-n exploration (768 to 2048)
3. Spectral-guided initialization

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def make_hann(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def make_flat_top(n, width_frac=0.8):

    s = a.sum()
    if s > 1e-15:

    return a

    M = c.max()
    best_f = f.copy()

            active_idx = np.argsort(c)[::-1][:max_active]
            M = c[active_idx].max()

            # (f * g)_k = sum_j f_j * g_{k-j} for j in [max(0, k-n+1), min(n-1, k)]
            js = max(0, k - n + 1)

            A_ub[i, js:je+1] = f[js:je+1]
            b_ub[i] = M

            if no_improve > 30:
                c = fast_convolve(f)
                M = c.max() * 1.05

            continue

def gradient_descent(f0, time_limit, beta_start=4, beta_end=256, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()

    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    c = fast_convolve(f)
    best_obj = c.max()
    
    t0 = time.monotonic()
    momentum = np.zeros(n)
    mu = 0.8
    lr = 0.1
    
    betas = [beta_start]
    b = beta_start
    while b < beta_end:
        b = min(b * 2, beta_end)
        betas.append(b)
    
    for beta in betas:
        stage_start = time.monotonic()
        stage_time = (time_limit / len(betas)) * 0.8
        no_improve = 0
        
        while time.monotonic() - stage_start < stage_time and time.monotonic() - t0 < time_limit * 0.9:
            c = fast_convolve(f)
            # Compute gradient of LSE approximation to max
            scaled = beta * c
            scaled -= scaled.max()
            exp_scaled = np.exp(scaled)
            w = exp_scaled / exp_scaled.sum()
            
            # Gradient: g_j = 2 * sum_i f_i * w_{j+i}

            fft_size = 1 << (n + len(w_rev) - 2).bit_length()
            F = np.fft.rfft(f, fft_size)
            W = np.fft.rfft(w_rev, fft_size)
            conv = np.fft.irfft(F * W, fft_size)[:n + len(w_rev) - 1]
            indices = np.arange(2 * n - 2, 2 * n - 2 - n, -1)
            g = 2.0 * conv[indices]
            
            # Normalize and step
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-15:
                break
            g = g / g_norm
            
            momentum = mu * momentum - lr * g
            f_trial = f + momentum
            f_trial = project_simplex(f_trial)
            
            c_trial = fast_convolve(f_trial)
            obj_trial = c_trial.max()
            
            if obj_trial < best_obj:
                best_obj = obj_trial
                best_f = f_trial.copy()
            if obj_trial <= c.max() + 1e-8:
                f = f_trial
                no_improve = 0
            else:
                no_improve += 1
                lr *= 0.9
                f = best_f.copy()
            
            if no_improve > 50:
                f = project_simplex(best_f + np.random.default_rng(int(time.monotonic()*1e6)).standard_normal(n) * 0.001)
                momentum = np.zeros(n)
                no_improve = 0
                lr = 0.1 / (n ** 0.5)
        
        lr = 0.1 / (beta * 0.2 * n ** 0.5)
    
    return best_f, best_obj

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_gaussian(n, sigma_frac=0.35):
    """Gaussian-shaped initialization."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_hann(n):
    """Hann window initialization."""
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s > 1e-15:

    return a


def make_flat_top(n, width_frac=0.85):

            a[i] = 1.0
        else:
            t = (ax - w) / (1.0 - w + 1e-10)
            a[i] = 0.5 * (1.0 + np.cos(np.pi * t))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.7, power=4):

    s = a.sum()
    if s > 1e-15:

def gradient_descent_smoothed(f0, time_limit, verbose=False):

    mu = 0.85

    for beta in betas:
        if time.monotonic() - t0 > total_time:
            break

        stage_start = time.monotonic()
        lr = 0.5 / (np.sqrt(n) * 2.0)
        no_improve = 0
        it = 0

        while time.monotonic() - stage_start < per_beta and time.monotonic() - t0 < total_time:
            it += 1
            c = fast_convolve(f)
            scaled = beta * c
            scaled -= scaled.max()
            exp_scaled = np.exp(scaled)
            w = exp_scaled / exp_scaled.sum()

            w_rev = w[::-1]
            conv_len = n + len(w_rev) - 1

            F = np.fft.rfft(f, fft_size)
            W = np.fft.rfft(w_rev, fft_size)
            conv = np.fft.irfft(F * W, fft_size)[:conv_len]
            indices = np.arange(2 * n - 2, 2 * n - 2 - n, -1)
            g = 2.0 * conv[indices]

            g_norm = np.linalg.norm(g)
            if g_norm < 1e-15:

            g = g / g_norm

            momentum = mu * momentum - lr * g
            f_trial = f + momentum

def spectral_flatten(f0, time_limit, n_peaks=20, verbose=False):
    """
    Spectral-domain peak reduction: iteratively adjusts the sequence

    """

    n = len(f)

    no_improve = 0

        M = c.max()

        # Find top peaks
        k = min(n_peaks, len(c))
        peak_idx = np.argpartition(c, -k)[-k:]
        peak_vals = c[peak_idx]
        excess = np.maximum(peak_vals - np.median(c), 0)
        w = excess / (excess.sum() + 1e-15)

        # Subgradient from peaks
        sg = np.zeros(n)
        for i in range(k):
            if w[i] < 1e-12:
                continue
            ki = int(peak_idx[i])
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            idx = np.arange(js, je + 1)
            sg[idx] += w[i] * f[ki - idx]

        # Multiplicative update
        f = f * np.exp(-lr * 2.0 * sg)
        f = np.maximum(f, 1e-12)
        f = f / f.sum()

        c_new = fast_convolve(f)
        obj_new = c_new.max()

        if obj_new < best_obj - 1e-13:

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    s = f_interp.sum()
    if s > 1e-15:
        f_interp /= s
    return f_interp

            ("flat_blend", lambda: (0.5 * best_seq + 0.5 * make_flat_top(n_best, 0.8)) / (0.5 * best_seq + 0.5 * make_flat_top(n_best, 0.8)).sum()),
            ("gauss_blend", lambda: (0.7 * best_seq + 0.3 * make_gaussian(n_best, 0.4)) / (0.7 * best_seq + 0.3 * make_gaussian(n_best, 0.4)).sum()),
            ("reverse", lambda: best_seq[::-1].copy() / best_seq[::-1].sum()),
        ]

        for si, (sname, sfn) in enumerate(strategies):
            rem = deadline - time.monotonic()
            if rem < 25:
                break
            t_alloc = min(rem / max(1, len(strategies) - si), 120)

            f_init = sfn()
            a_opt, obj = active_set_lp_fw(f_init, n_best, time_limit=t_alloc, max_active=50, verbose=(si == 0))
            try_seq(a_opt, f"deep_{sname}")

    # === Phase 3: Try nearby n values ===
    print("\n=== Phase 3: Nearby n Exploration ===")
    if best_seq is not None:
        n_best = len(best_seq)
        nearby_ns = [n_best + 128, n_best - 128, n_best + 256, n_best + 512]
        seen = set()

        for n_try in nearby_ns:
            if n_try < 256 or n_try > 4096 or n_try in seen:
                continue
            seen.add(n_try)

            rem = deadline - time.monotonic()
            if rem < 35:
                break

            f_interp = interpolate_sequence(best_seq, n_try)
            try_seq(f_interp, f"interp_n{n_try}")

            t_alloc = min(rem * 0.4, 100)
            if t_alloc > 15:
                a_opt, obj = active_set_lp_fw(f_interp, n_try, time_limit=t_alloc, max_active=50)
                try_seq(a_opt, f"lpfw_n{n_try}")

    # === Phase 4: Final polish ===
    print("\n=== Phase 4: Final Polish ===")
    if best_seq is not None:
        n_final = len(best_seq)
        rem = deadline - time.monotonic()
        if rem > 15:
            t_final = min(rem, 60)
            a_opt, obj = active_set_lp_fw(best_seq, n_final, time_limit=t_final, max_active=60, verbose=True)
            try_seq(a_opt, "final_polish")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
