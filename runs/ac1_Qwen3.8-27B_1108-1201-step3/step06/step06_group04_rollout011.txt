def solve_flat_system(n, w, S):
    """
    Solve L(a) = 2*lambda * 1 for j in S, sum(a_S) = 1.
    L(a)[j] = sum_k w[k] * (a[k-j] + a[j-k]) (with valid indices)
    Returns a (full length n) and lambda.
    """
    S_arr = np.array(S)
    m = len(S_arr)
    # Build (m+1) x (m+1) system
    # Unknowns: a[S_arr] (m vars) + lambda (1 var) = m+1 vars
    # Equations:
    #   For each j in S: L(a)[j] - 2*lambda = 0  (m equations)
    #   sum(a[S_arr]) = 1  (1 equation)
    
    # L(a)[j] = sum_k w[k] * (a[k-j] + a[j-k])
    # For j in S: sum_k w[k] * (a[k-j] + a[j-k]) - 2*lambda = 0
    
    # If k-j is in S: contributes w[k] * a[k-j]
    # If j-k is in S: contributes w[k] * a[j-k]
    # If k-j not in S: a[k-j] = 0, no contribution
    # If j-k not in S: a[j-k] = 0, no contribution
    
    # Build matrix
    # Let idx map: position in S_arr -> index in the unknown vector
    # Unknown vector: [a[S[0]], a[S[1]], ..., a[S[m-1]], lambda]
    
    A_mat = np.zeros((m+1, m+1))
    b_vec = np.zeros(m+1)
    
    # S index map
    S_set = set(S_arr.tolist())
    S_idx = {v: i for i, v in enumerate(S_arr)}
    
    for i, j in enumerate(S_arr):
        # Equation i: L(a)[j] - 2*lambda = 0
        for k in range(2*n-1):
            wk = w[k]
            if wk == 0:
                continue
            # Term 1: a[k-j] if 0 <= k-j < n
            kj = k - j
            if 0 <= kj < n and kj in S_set:
                A_mat[i, S_idx[kj]] += wk
            # Term 2: a[j-k] if 0 <= j-k < n
            jk = j - k
            if 0 <= jk < n and jk in S_set:
                A_mat[i, S_idx[jk]] += wk
        # Lambda term
        A_mat[i, m] = -2.0
    
    # Last equation: sum(a[S]) = 1
    A_mat[m, :m] = 1.0
    b_vec[m] = 1.0
    
    # Solve
    try:
        sol = np.linalg.solve(A_mat, b_vec)
    except np.linalg.LinAlgError:
        sol = np.linalg.lstsq(A_mat, b_vec, rcond=None)[0]
    
    a = np.zeros(n)
    for i, j in enumerate(S_arr):
        a[j] = sol[i]
    lam = sol[m]
    
    return a, lam

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an active-set linear solver
combined with soft-max gradient descent and multi-length exploration.
The key new idea is solving the KKT conditions directly via linear algebra,
which gives much more precise local optima than iterative first-order methods.

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def obj_normalized(a):
    """Objective for sum-normalized sequence."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def active_set_solve(a_init, n, tol_active=1e-4, max_iter=50):
    """
    Solve the KKT system for the active set method.
    At optimum: L(a)[j] = 2*lambda for j in support, L(a)[j] >= 2*lambda for j not in support.
    Where L(a)[j] = sum_k w[k] * (a[k-j] + a[j-k]) with valid indices.
    """
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = obj_normalized(a)
    
    for outer in range(max_iter):
        # Compute convolution and active set
        conv = fast_convolve(a)
        M = conv.max()
        
        # Active set: lags within tol of max
        active_mask = conv >= M * (1 - tol_active)
        A_set = np.where(active_mask)[0]
        
        if len(A_set) == 0:
            A_set = np.array([np.argmax(conv)])
        
        # Weights
        w = np.zeros(2*n-1)
        w[A_set] = 1.0 / len(A_set)
        
        # Support: indices where a > threshold
        support = np.where(a > 1e-10)[0]
        
        # Solve the linear system
        a_new = solve_linear_system(n, w, support)
        
        if a_new is None:
            # System singular or infeasible, try removing some support
            # Remove the smallest elements
            min_idx = np.argmin(a[support])
            support = np.delete(support, min_idx)
            a_new = solve_linear_system(n, w, support)
            if a_new is None:

            support = np.setdiff1d(support, neg_indices)
            a_new2 = solve_linear_system(n, w, support)
            if a_new2 is not None:
                a_new = a_new2

                break

        a_new = a_new / a_new.sum()

            a = a_new
            break
        
        # Check if active set changed
        conv_new = fast_convolve(a_new)
        M_new = conv_new.max()

            a = a_new
            # One more iteration to verify
            break
        else:
            a = a_new
    
    obj_new = obj_normalized(a)
    if obj_new < best_obj:
        best_obj = obj_new
        best_a = a.copy()
    
    return best_a, best_obj

def solve_linear_system(n, w, support):
    """
    Solve: sum_k w[k]*(a[k-j]+a[j-k]) = 2*lambda for j in support
           sum_{j in support} a[j] = 1
    """
    m = len(support)
    if m == 0:
        return None
    
    support_set = set(support.tolist())
    support_idx = {v: i for i, v in enumerate(support)}
    
    # Build (m+1) x (m+1) system
    # Unknowns: a[support[0]], ..., a[support[m-1]], lambda
    # Equations: for each j in support: L(a)[j] - 2*lambda = 0
    #            sum(a[support]) = 1
    
    A_mat = np.zeros((m+1, m+1))
    b_vec = np.zeros(m+1)
    
    for i, j in enumerate(support):
        for k in range(2*n-1):
            wk = w[k]
            if wk == 0:
                continue
            # a[k-j] contribution
            kj = k - j
            if 0 <= kj < n and kj in support_set:
                A_mat[i, support_idx[kj]] += wk
            # a[j-k] contribution
            jk = j - k
            if 0 <= jk < n and jk in support_set:
                A_mat[i, support_idx[jk]] += wk
        A_mat[i, m] = -2.0
    
    A_mat[m, :m] = 1.0
    b_vec[m] = 1.0
    
    try:
        sol = np.linalg.solve(A_mat, b_vec)
    except np.linalg.LinAlgError:
        sol = np.linalg.lstsq(A_mat, b_vec, rcond=None)[0]
    
    a = np.zeros(n)
    for i, j in enumerate(support):
        a[j] = sol[i]
    
    # Check: all support values should be non-negative (for the support to be valid)
    # We'll check this in the caller
    
    return a

def soft_max_gradient_descent(a_init, n, time_limit, beta=20, lr=0.01, verbose=False):
    """
    Gradient descent on the soft-max smoothed objective.
    """
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = obj_normalized(a)
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    # Adam state
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        
        # Softmax weights
        shifted = beta * conv - beta * M
        p = np.exp(shifted)
        p_sp = p.sum()
        if p_sp < 1e-30:
            p = np.zeros(2*n-1)
            p[np.argmax(conv)] = 1.0
        else:
            p /= p_sp
        
        # Compute gradient
        # grad[j] = beta * sum_k p[k] * (a[k-j] + a[j-k]) (valid)
        # Use the FFT approach
        pr = p[::-1]  # reverse p
        fft_len = 1 << (3*n-2-1).bit_length()
        A_fft = np.fft.rfft(a, fft_len)
        PR_fft = np.fft.rfft(pr, fft_len)
        c = np.fft.irfft(A_fft * PR_fft, fft_len)[:3*n-2]
        
        term1 = c[2*(n-1) - np.arange(n)]
        grad = beta * (term1 + term1[2*(n-1) - np.arange(n)])
        
        # Normalize gradient (subtract mean to keep sum=1)
        grad = grad - grad.mean()
        
        # Adam update
        m_v = beta1 * m_v + (1 - beta1) * grad
        v_v = beta2 * v_v + (1 - beta2) * grad**2
        m_hat = m_v / (1 - beta1**it)
        v_hat = v_v / (1 - beta2**it)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        obj_new = obj_normalized(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        if no_improve > 100:
            # Reset with perturbation
            rng = np.random.default_rng(it)
            a = np.maximum(best_a + rng.standard_normal(n) * 0.001, 0)
            a = a / a.sum()
            no_improve = 0
            m_v = np.zeros(n)
            v_v = np.zeros(n)
        
        if verbose and it % 100 == 0:
            print(f"    SGD iter {it}: obj={best_obj:.10f}, beta={beta:.1f}")
    
    return best_a, best_obj

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
