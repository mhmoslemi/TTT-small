"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using sparse constraint-generation
LP-Frank-Wolfe with momentum, peak-targeted refinement, and multi-scale
optimization. Key improvements over previous approach: constraint generation
(much faster LP solves), adaptive momentum, and focused peak shaving.
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def convolve_pair(f, g):
    """Cross-convolution of f and g via FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    G = np.fft.rfft(g, fft_size)
    return np.fft.irfft(F * G, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)

def obj_normalized(f):
    """Objective for sum-normalized sequence: 2*n*max(conv)."""
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()

def make_gaussian(n, sigma_frac=0.25):
    """Gaussian initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def make_bell(n, width=0.7):
    """Bell-shaped initial sequence."""
    x = np.arange(n) - n / 2.0
    half_w = n * width / 2.0
    t = np.clip(np.abs(x) / half_w, 0, 1.0)
    a = (1.0 - t) ** 3
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def make_asymmetric(n, decay=3.0):
    """Asymmetric exponential decay."""
    x = np.arange(n)
    a = np.exp(-decay * x / n)
    a = a / a.sum()
    return a

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def evaluate_seq(f):
    """Safely evaluate a sequence."""
    try:
        f_list = [float(x) for x in f]
        return float(evaluate_sequence(f_list))
    except:
        return float('inf')

def sparse_lp_direction(f, M, n, init_constraints=None, max_constraints=None):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for active constraints, g >= 0.
    Uses constraint generation: start with a few constraints, add violated ones.
    """
    if max_constraints is None:
        max_constraints = min(2 * n - 1, 200)
    
    if init_constraints is None:
        # Initialize with top constraints based on f*f
        conv = fast_convolve(f)
        init_constraints = np.argsort(conv)[-min(30, 2*n-1)]:]
        init_constraints = np.sort(init_constraints)
    
    active = list(init_constraints)
    A_ub_rows = []
    b_ub_vals = []
    
    for k in active:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        row = np.zeros(n)
        for j in range(js, je + 1):
            row[j] = f[k - j]
        A_ub_rows.append(row)
        b_ub_vals.append(M)
    
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    g = None
    for cg_iter in range(20):  # Max constraint generation iterations
        A_ub = np.array(A_ub_rows)
        b_ub = np.array(b_ub_vals)
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            return None
        g = res.x
        
        # Check all constraints for violation
        fg = convolve_pair(f, g)
        violations = np.where(fg > M + 1e-10)[0]
        
        if len(violations) == 0:
            break
        
        # Add the most violated constraint
        worst = violations[np.argmax(fg[violations])]
        if worst not in active:
            k = int(worst)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            row = np.zeros(n)
            for j in range(js, je + 1):
                row[j] = f[k - j]
            A_ub_rows.append(row)
            b_ub_vals.append(M)
            active.append(worst)
        
        if len(active) >= max_constraints:
            break
    
    if g is not None:
        g = np.maximum(g, 0)
        if g.sum() > 1e-15:
            g = g / g.sum()
    return g

def lp_fw_sparse(f0, time_limit, verbose=False, momentum_beta=0.7):
    """
    LP-Frank-Wolfe with sparse constraint generation and momentum.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    momentum_dir = np.zeros(n)
    active_constraints = None
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP direction
        g = sparse_lp_direction(f, M * m_exp, n, init_constraints=active_constraints)
        if g is None:
            m_exp *= 1.1
            if m_exp > 5.0:
                # Perturbation
                rng = np.random.default_rng(it * 31 + 7)
                scale = 0.003 * min(m_exp, 3.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                active_constraints = None
                no_improve = 0
            continue
        
        # Update momentum
        new_dir = g - f
        momentum_dir = momentum_beta * momentum_dir + (1 - momentum_beta) * new_dir
        
        # Line search
        # Try a few step sizes
        best_alpha = 0.0
        best_alpha_val = M
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            f_trial = f + alpha * momentum_dir
            f_trial = np.maximum(f_trial, 0.0)
            st = f_trial.sum()
            if st < 1e-15:
                continue
            f_trial = f_trial / st
            cvt = fast_convolve(f_trial)
            if cvt.max() < best_alpha_val:
                best_alpha_val = cvt.max()
                best_alpha = alpha
        
        if best_alpha < 1e-8:
            # Try with raw direction too
            for alpha in [0.05, 0.1, 0.2, 0.5]:
                f_trial = f + alpha * (g - f)
                f_trial = np.maximum(f_trial, 0.0)
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial = f_trial / st
                cvt = fast_convolve(f_trial)
                if cvt.max() < best_alpha_val:
                    best_alpha_val = cvt.max()
                    best_alpha = alpha
        
        if best_alpha < 1e-8:
            no_improve += 1
            if no_improve > 10:
                m_exp = min(m_exp * 1.05, 4.0)
            if no_improve > 50:
                # Perturbation
                rng = np.random.default_rng(it * 17 + 3)
                scale = 0.002
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                active_constraints = None
                no_improve = 0
                momentum_dir = np.zeros(n)
            continue
        
        # Update
        f = f + best_alpha * momentum_dir
        f = np.maximum(f, 0.0)
        f = f / f.sum()
        conv = fast_convolve(f)
        M = conv.max()
        
        # Update active constraints for next iteration
        active_constraints = np.argsort(conv)[-min(50, 2*n-1):]
        active_constraints = np.sort(active_constraints)
        
        m_exp = 1.0
        current_obj = 2.0 * n * M
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if verbose and it % 10 == 0:
            print(f"    LP-FW-Sparse iter {it}: obj={best_obj:.10f}, alpha={best_alpha:.3f}")
        
        if no_improve > 100:
            break
    
    return best_f, best_obj

def peak_shaving(f0, time_limit, verbose=False):
    """
    Peak-targeted coordinate descent: directly reduces top convolution peaks.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and no_improve < 30:
        conv = fast_convolve(f)
        M = conv.max()
        improved = False
        
        # Target top peaks
        top_k = min(8, len(conv))
        peak_indices = np.argsort(conv)[-top_k:]
        
        for k in peak_indices:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue
            
            idx = np.arange(js, je + 1)
            products = f[idx] * f[k - idx]
            total = products.sum()
            if total < 1e-15:
                continue
            
            # Try reducing top contributors
            order = np.argsort(products)[::-1]
            for oi in range(min(5, len(order))):
                j_max = int(idx[order[oi]])
                old_val = f[j_max]
                if old_val < 1e-12:
                    continue
                
                for mult in [0.3, 0.5, 0.7, 0.8, 1.2, 1.5]:
                    new_val = old_val * mult
                    if new_val < 0:
                        continue
                    
                    trial = f.copy()
                    trial[j_max] = new_val
                    d = old_val - new_val
                    if abs(d) > 1e-15:
                        # Redistribute to non-peak area
                        non_peak = np.ones(n, dtype=bool)
                        # Don't redistribute to elements that contribute to this peak
                        for j in range(js, je + 1):
                            if j != j_max:
                                non_peak[j] = False
                        cnt = non_peak.sum()
                        if cnt > 0:
                            w = trial[non_peak].copy()
                            wsum = w.sum()
                            if wsum > 1e-15:
                                trial[non_peak] += d * w / wsum
                    
                    trial = np.maximum(trial, 0)
                    tsum = trial.sum()
                    if tsum < 1e-15:
                        continue
                    trial = trial / tsum
                    obj = 2.0 * n * fast_convolve(trial).max()
                    if obj < best_obj - 1e-12:
                        best_obj = obj
                        best_f = trial.copy()
                        f = trial
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            no_improve += 1
            # Try global scale adjustment of peak region
            k = int(np.argmax(conv))
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            
            for scale in [0.85, 0.9, 0.95, 1.05, 1.1]:
                trial = f.copy()
                trial[js:je+1] *= scale
                d = (1.0 - scale) * f[js:je+1].sum()
                non_contrib = np.ones(n, dtype=bool)
                non_contrib[js:je+1] = False
                if non_contrib.sum() > 0:
                    nc_w = trial[non_contrib].copy()
                    nw = nc_w.sum()
                    if nw > 1e-15:
                        trial[non_contrib] += d * nc_w / nw
                trial = np.maximum(trial, 0)
                trial = trial / trial.sum()
                obj = 2.0 * n * fast_convolve(trial).max()
                if obj < best_obj:
                    f = trial
                    best_obj = obj
                    best_f = f.copy()
                    no_improve = 0

def subgradient_descent(f0, time_limit, verbose=False):
    """
    Subgradient method for min max_k (f*f)_k s.t. sum(f)=1, f>=0.
    The subgradient at the maximizing index k* is: sg[j] = 2*f[k*-j] (for valid j)
    """
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)
    
    best_obj = 2.0 * n * fast_convolve(f).max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    lr = 0.01
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        # Subgradient: use top few peaks with weighting
        k_top = min(10, len(conv))
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        pv = conv[peaks]
        # Weights: softmax with temperature
        temp = M * 0.1
        w = np.exp((pv - pv.max()) / temp)
        w /= w.sum()
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-10:
                continue
            kk = int(peaks[i])
            js = max(0, kk - n + 1)
            je = min(n - 1, kk)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[kk - idx]
        
        # Projected subgradient step
        f_new = f - lr * sg
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()
        
        co = 2.0 * n * fast_convolve(f_new).max()
        if co < best_obj - 1e-12:
            best_obj = co
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive learning rate
        if no_improve % 20 == 0 and no_improve > 0:
            lr *= 0.95
        
        if no_improve > 200:
            # Restart with perturbation
            rng = np.random.default_rng(it * 41 + 13)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.003, 1e-15)
            f = f / f.sum()
            lr = 0.01
            no_improve = 0
        
        if lr < 1e-10:
            break
    
    return best_f, best_obj

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def convolve_pair(f, g):
    """Cross-convolution of f and g via FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    G = np.fft.rfft(g, fft_size)
    return np.fft.irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def make_gaussian(n, sigma_frac=0.25):
    """Gaussian initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width=0.7):
    """Bell-shaped initial sequence."""
    x = np.arange(n) - n / 2.0
    half_w = n * width / 2.0
    t = np.clip(np.abs(x) / half_w, 0, 1.0)
    a = (1.0 - t) ** 3
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_asymmetric(n, decay=3.0):
    """Asymmetric exponential decay initial sequence."""
    x = np.arange(n)
    a = np.exp(-decay * x / n)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def build_row(f, k, n):
    """Build constraint row for index k: (f*g)[k] = sum_j f[k-j]*g[j]."""
    row = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        row[j] = f[k - j]
    return row


def sparse_lp_direction(f, M, n, active_set=None, max_iter=15):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for active constraints, g >= 0.
    Uses constraint generation: start with active set, add violated ones.
    """
    if active_set is None or len(active_set) == 0:
        conv = fast_convolve(f)
        k_init = min(40, 2 * n - 1)
        active_set = np.sort(np.argsort(conv)[-k_init:])

    active_set = list(active_set)
    rows = []
    for k in active_set:
        rows.append(build_row(f, int(k), n))

    c = -np.ones(n)
    bounds = [(0, None)] * n
    g = None

    for _ in range(max_iter):
        A_ub = np.array(rows)
        b_ub = np.full(len(rows), M)
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            return None, active_set
        g = res.x

        # Check all constraints
        fg = convolve_pair(f, g)
        tol = 1e-9
        violations = np.where(fg > M + tol)[0]

        if len(violations) == 0:
            break

        # Add most violated
        worst = int(violations[np.argmax(fg[violations])])
        if worst not in active_set:
            active_set.append(worst)
            rows.append(build_row(f, worst, n))

        # Also add a few more if close
        close = np.where(fg > M * 0.99)[0]
        for cv in close[:3]:
            cv = int(cv)
            if cv not in active_set:
                active_set.append(cv)
                rows.append(build_row(f, cv, n))
                break

    if g is not None:
        g = np.maximum(g, 0)
        gs = g.sum()
        if gs > 1e-15:
            g = g / gs
        else:
            return None, active_set

    return g, active_set

                m_exp = min(m_exp * 1.05, 4.0)
            if no_improve > 40:
                rng = np.random.default_rng(it * 17 + 3)
                scale = 0.002
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                active_set = None
                momentum_dir = np.zeros(n)
                no_improve = 0
            continue

        # Apply update
        f = f + best_alpha * momentum_dir
        f = np.maximum(f, 0.0)
        f = f / f.sum()
        conv = fast_convolve(f)
        M = conv.max()

        # Update active set
        k_keep = min(60, 2 * n - 1)
        active_set = np.sort(np.argsort(conv)[-k_keep:])

        m_exp = 1.0
        current_obj = 2.0 * n * M
        if current_obj < best_obj - 1e-11:
            best_obj = current_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and it % 15 == 0:
            print(f"    LP-FW iter {it}: obj={best_obj:.10f}, alpha={best_alpha:.3f}, m_exp={m_exp:.2f}")

        if no_improve > 80:
            break

    return best_f, best_obj


def peak_shaving(f0, time_limit, verbose=False):
    """
    Peak-targeted coordinate descent: directly reduces top convolution peaks
    by shrinking contributing elements and redistributing mass.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 25:
        conv = fast_convolve(f)
        M = conv.max()
        improved = False

        top_k = min(10, len(conv))
        peak_indices = np.argsort(conv)[-top_k:]

        for k in peak_indices:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue

            idx = np.arange(js, je + 1)
            products = f[idx] * f[k - idx]
            total = products.sum()
            if total < 1e-15:
                continue

            order = np.argsort(products)[::-1]
            for oi in range(min(6, len(order))):
                j_max = int(idx[order[oi]])
                old_val = f[j_max]
                if old_val < 1e-12:
                    continue

                for mult in [0.4, 0.55, 0.7, 0.8, 0.9, 1.1, 1.3]:
                    new_val = old_val * mult
                    if new_val < 0:
                        continue

                    trial = f.copy()
                    trial[j_max] = new_val
                    d = old_val - new_val
                    if abs(d) > 1e-15:
                        non_peak = np.ones(n, dtype=bool)
                        non_peak[js:je + 1] = False
                        non_peak[j_max] = True  # allow redistributing back
                        cnt = non_peak.sum()
                        if cnt > 0:
                            w = trial[non_peak].copy()
                            wsum = w.sum()
                            if wsum > 1e-15:
                                trial[non_peak] += d * w / wsum

                    trial = np.maximum(trial, 0)
                    tsum = trial.sum()
                    if tsum < 1e-15:
                        continue
                    trial = trial / tsum
                    obj = 2.0 * n * fast_convolve(trial).max()
                    if obj < best_obj - 1e-11:
                        best_obj = obj
                        best_f = trial.copy()
                        f = trial
                        improved = True
                        break

        if not improved:
            no_improve += 1
            # Global peak region scaling
            k = int(np.argmax(fast_convolve(f)))
            js = max(0, k - n + 1)
            je = min(n - 1, k)

            done = False
            for scale in [0.82, 0.88, 0.94, 1.06, 1.12]:
                trial = f.copy()
                trial[js:je + 1] *= scale
                d = (1.0 - scale) * f[js:je + 1].sum()
                non_contrib = np.ones(n, dtype=bool)
                non_contrib[js:je + 1] = False
                if non_contrib.sum() > 0:
                    nc_w = trial[non_contrib].copy()
                    nw = nc_w.sum()
                    if nw > 1e-15:
                        trial[non_contrib] += d * nc_w / nw
                trial = np.maximum(trial, 0)
                tsum = trial.sum()
                if tsum < 1e-15:
                    continue
                trial = trial / tsum
                obj = 2.0 * n * fast_convolve(trial).max()
                if obj < best_obj:
                    f = trial
                    best_obj = obj
                    best_f = f.copy()
                    no_improve = 0
                    done = True
                    break
            if done:
                no_improve = 0

        if improved:
            no_improve = 0

        if verbose:
            print(f"    Peak CD: obj={best_obj:.10f}, no_improve={no_improve}")

    return best_f, best_obj


def pairwise_swap(f0, time_limit, verbose=False):
    """
    Local search: try transferring small amounts of mass between pairs of
    elements to reduce the max convolution value.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 20:
        conv = fast_convolve(f)
        M = conv.max()
        improved = False

        # Identify peak region
        k_peak = int(np.argmax(conv))
        # Elements contributing to the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)

        # Try moving mass from peak contributors to non-peak elements
        contributors = np.arange(js, je + 1)
        non_contrib = np.setdiff1d(np.arange(n), contributors)

        if len(non_contrib) == 0:
            no_improve += 1
            break

        # Sort contributors by value (move from largest)
        for i in np.argsort(f[contributors])[::-1]:
            if time.monotonic() - t0 > time_limit:
                break
            src = int(contributors[i])
            src_val = f[src]
            if src_val < 1e-6:
                continue

            # Try moving fractions to best non-contributor
            for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                amount = src_val * frac
                if amount < 1e-8:
                    continue

                # Choose target: element with smallest value in non_peak
                # (spreads mass to reduce concentration)
                for j in np.argsort(f[non_contrib])[:3]:
                    dst = int(non_contrib[j])
                    trial = f.copy()
                    trial[src] -= amount
                    trial[dst] += amount
                    trial = np.maximum(trial, 0)
                    tsum = trial.sum()
                    if tsum < 1e-15:
                        continue
                    trial = trial / tsum
                    obj = 2.0 * n * fast_convolve(trial).max()
                    if obj < best_obj - 1e-11:
                        best_obj = obj
                        best_f = trial.copy()
                        f = trial
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            # Try random perturbations
            rng = np.random.default_rng(int(time.monotonic() * 1000) % (2**31))
            for _ in range(20):
                src = rng.integers(n)
                dst = rng.integers(n)
                if src == dst:
                    continue
                amount = f[src] * 0.15
                if amount < 1e-8:
                    continue
                trial = f.copy()
                trial[src] -= amount
                trial[dst] += amount
                trial = np.maximum(trial, 0)
                tsum = trial.sum()
                if tsum < 1e-15:
                    continue
                trial = trial / tsum
                obj = 2.0 * n * fast_convolve(trial).max()
                if obj < best_obj - 1e-11:
                    best_obj = obj
                    best_f = trial.copy()
                    f = trial
                    improved = True
                    break

        if verbose and it % 5 == 0:
            print(f"    Pairwise: obj={best_obj:.10f}")

    return best_f, best_obj


def subgradient_refine(f0, time_limit, verbose=False):
    """
    Subgradient method with adaptive step size for final polishing.
    """
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)

    best_obj = 2.0 * n * fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = 0.005
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Use top peaks for subgradient
        k_top = min(15, len(conv))
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        pv = conv[peaks]
        temp = max(M * 0.05, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            kk = int(peaks[i])
            js = max(0, kk - n + 1)
            je = min(n - 1, kk)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[kk - idx]

        # Projected subgradient step
        f_new = f - lr * sg
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()

        co = 2.0 * n * fast_convolve(f_new).max()
        if co < best_obj - 1e-12:
            best_obj = co
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 30:
            lr *= 0.9
            if no_improve > 150:
                rng = np.random.default_rng(it * 41 + 13)
                f = np.maximum(best_f + rng.standard_normal(n) * 0.002, 1e-15)
                f = f / f.sum()
                lr = 0.005
                no_improve = 0

        if lr < 1e-9:
            break

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization for minimizing 2*n*max(conv)/(sum)^2.
    Uses sparse constraint-generation LP-FW as main engine, supplemented by
    peak shaving, pairwise swap, subgradient methods, and multi-length search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = safe_eval(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent
    except Exception:
        pass

    if parent is None:
        parent = make_gaussian(768, 0.25)
        sc = safe_eval(parent)
        if sc < best_score:
            best_score = sc
            best_seq = parent.copy()

    n_parent = len(parent)

    # === Phase 1: Quick diverse initialization ===
    print("=== Phase 1: Diverse Init ===")
    init_candidates = [parent / parent.sum()]
    for n_try in [640, 704, 768, 832, 896]:
        if n_try == n_parent:
            init_candidates.append(parent / parent.sum())
        else:
            init_candidates.append(interpolate_sequence(parent / parent.sum(), n_try))
        init_candidates.append(make_gaussian(n_try, 0.20))
        init_candidates.append(make_gaussian(n_try, 0.25))
        init_candidates.append(make_gaussian(n_try, 0.30))
        init_candidates.append(make_bell(n_try, 0.55))
        init_candidates.append(make_bell(n_try, 0.75))
        init_candidates.append(make_asymmetric(n_try, 2.5))

    for ic in init_candidates:
        sc = safe_eval(ic)
        if sc < best_score:
            best_score = sc
            best_seq = ic.copy()
            print(f"  Better init: {sc:.6f} (n={len(ic)})")

    print(f"  Best after init: {best_score:.6f}")

    # === Phase 2: Main sparse LP-FW ===
    n_main = len(best_seq)
    print(f"\n=== Phase 2: Sparse LP-FW (n={n_main}) ===")
    rem = deadline - time.monotonic() - 700
    if rem > 80:
        t_alloc = min(rem, 450)
        print(f"  Alloc: {t_alloc:.0f}s")
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
        sc = safe_eval(a_opt)
        if sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f}")

    # === Phase 3: Peak shaving ===
    if best_seq is not None:
        print(f"\n=== Phase 3: Peak Shaving (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 500
        if rem > 40:
            t_alloc = min(rem, 150)
            print(f"  Alloc: {t_alloc:.0f}s")
            a_opt, _ = peak_shaving(best_seq, time_limit=t_alloc, verbose=True)
            sc = safe_eval(a_opt)
            if sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  NEW BEST: {sc:.6f}")

    # === Phase 4: Pairwise swap ===
    if best_seq is not None:
        print(f"\n=== Phase 4: Pairwise Swap (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 350
        if rem > 30:
            t_alloc = min(rem, 100)
            print(f"  Alloc: {t_alloc:.0f}s")
            a_opt, _ = pairwise_swap(best_seq, time_limit=t_alloc, verbose=True)
            sc = safe_eval(a_opt)
            if sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  NEW BEST: {sc:.6f}")

    # === Phase 5: Subgradient polish ===
    if best_seq is not None:
        print(f"\n=== Phase 5: Subgradient (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 250
        if rem > 30:
            t_alloc = min(rem, 80)
            print(f"  Alloc: {t_alloc:.0f}s")
            a_opt, _ = subgradient_refine(best_seq, time_limit=t_alloc, verbose=True)
            sc = safe_eval(a_opt)
            if sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  NEW BEST: {sc:.6f}")

    # === Phase 6: Multi-length exploration ===
    base = best_seq
    print(f"\n=== Phase 6: Multi-length ===")
    for target_n in [700, 768, 832, 896, 1024]:
        if base is None:
            break
        rem = deadline - time.monotonic() - 200
        if rem < 60:
            break
        if len(base) == target_n:
            continue
        f_interp = interpolate_sequence(base, target_n)
        sc = safe_eval(f_interp)
        if sc < best_score:
            best_score = sc
            best_seq = f_interp.copy()
            print(f"  Interp n={target_n}: {sc:.6f}")
        t_alloc = min(rem * 0.3, 50)
        if t_alloc < 20:
            continue
        print(f"  LP-FW n={target_n}: t={t_alloc:.0f}s")
        a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_alloc, verbose=False)
        sc = safe_eval(a_opt)
        if sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {sc:.6f} (n={target_n})")
            base = best_seq

    # === Phase 7: Deep LP-FW on current best ===
    if best_seq is not None:
        print(f"\n=== Phase 7: Deep LP-FW (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 120
        if rem > 50:
            t_alloc = min(rem, 150)
            print(f"  Alloc: {t_alloc:.0f}s")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            sc = safe_eval(a_opt)
            if sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  NEW BEST: {sc:.6f}")

    # === Phase 8: Perturbed restarts ===
    if best_seq is not None:
        print(f"\n=== Phase 8: Perturbed Restarts ===")
        n_best = len(best_seq)
        for p in range(4):
            rem = deadline - time.monotonic() - 60
            if rem < 30:
                break
            t_alloc = min(rem / (4 - p), 50)

            if p == 0:
                noise = rng.standard_normal(n_best) * 0.001
                f_init = project_simplex(best_seq + noise)
            elif p == 1:
                f_init = 0.96 * best_seq + 0.04 * (np.ones(n_best) / n_best)
                f_init = f_init / f_init.sum()
            elif p == 2:
                # Slight smoothing via convolution
                kernel = np.array([0.1, 0.25, 0.3, 0.25, 0.1])
                f_init = np.convolve(best_seq, kernel, mode='same')
                f_init = np.maximum(f_init, 0)
                f_init = f_init / f_init.sum()
            else:
                # Anti-smoothing: sharpen slightly
                kernel = np.array([-0.1, 0.3, 0.8, 0.3, -0.1])
                f_init = np.convolve(best_seq, kernel, mode='same')
                f_init = np.maximum(f_init, 0)
                f_init = f_init / f_init.sum()

            print(f"  Variant {p}: t={t_alloc:.0f}s")
            a_opt, _ = lp_fw_optimize(f_init, time_limit=t_alloc, verbose=False)
            sc = safe_eval(a_opt)
            if sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  NEW BEST: {sc:.6f}")

    # === Phase 9: Final polish ===
    if best_seq is not None:
        print(f"\n=== Phase 9: Final Polish ===")
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            t1 = rem * 0.4
            t2 = rem * 0.3
            t3 = rem * 0.3

            if t1 > 5:
                a_opt, _ = peak_shaving(best_seq, time_limit=t1)
                sc = safe_eval(a_opt)
                if sc < best_score:
                    best_score = sc
                    best_seq = a_opt.copy()

            rem2 = deadline - time.monotonic() - 8
            if rem2 > 5 and best_seq is not None:
                t2a = min(t2, rem2 - 5)
                if t2a > 3:
                    a_opt, _ = pairwise_swap(best_seq, time_limit=t2a)
                    sc = safe_eval(a_opt)
                    if sc < best_score:
                        best_score = sc
                        best_seq = a_opt.copy()

            rem3 = deadline - time.monotonic() - 3
            if rem3 > 3 and best_seq is not None:
                a_opt, _ = lp_fw_optimize(best_seq, time_limit = min(t3, rem3))
                sc = safe_eval(a_opt)
                if sc < best_score:
                    best_score = sc
                    best_seq = a_opt.copy()

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = safe_eval(best_seq)

    # Final verification
    fs = safe_eval(best_seq)
    if np.isfinite(fs):
        best_score = min(best_score, fs)

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
