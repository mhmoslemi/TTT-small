def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n <= 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)

    return 2.0 * n * conv.max()

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n <= 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_normalized(f):
    """Objective for sum-normalized sequence: 2*n*max(conv)."""
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()


def smooth_obj_and_grad(f, tau):
    """
    Compute smoothed objective and gradient using log-sum-exp.
    f is assumed to be on the simplex (sum=1, f>=0).
    Returns (smoothed_obj, gradient_vector).
    """
    n = len(f)
    conv = fast_convolve(f)
    
    # Log-sum-exp for numerical stability
    c_max = conv.max()
    shifted = conv - c_max
    log_sum_exp = c_max + np.log(np.exp(tau * shifted).sum() / (2*n-1))
    
    # Softmax weights
    p = np.exp(tau * shifted)
    p = p / p.sum()
    
    # Gradient: grad[j] = sum_k p[k] * a[k-j] for valid k-j
    # This is a cross-correlation computation
    grad = np.zeros(n)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        if k_end > k_start:
            grad[j] = np.dot(p[k_start:k_end], f[:k_end - k_start])
    
    return 2.0 * n * log_sum_exp, grad

def build_lp_matrix(f, n):
    """Build constraint matrix for LP: rows are convolution indices, cols are g components."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        length = k_end - k_start + 1
        A[k_start:k_end + 1, j] = f[:length]
    return A


def solve_lp_mindir(f, n):
    """
    Solve LP: min M s.t. (f*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M) or (None, None) if infeasible.
    """
    # Variables: [g_0, ..., g_{n-1}, M]
    # Objective: minimize M -> c = [0,...,0, 1]
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize M
    
    # Constraints: (f*g)[k] - M <= 0 for all k
    A_fg = np.zeros((2 * n - 1, n + 1))
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        length = k_end - k_start + 1
        A_fg[k_start:k_end + 1, j] = f[:length]
    A_fg[:, n] = -1.0  # -M
    
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: g >= 0, M unbounded below (but will be >= 0 naturally)
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_fg, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        g = np.maximum(res.x[:n], 0)
        M = res.x[n]
        gs = g.sum()
        if gs > 1e-15:
            g = g / gs
        return g, M
    return None, None

def multiplicative_descent(f0, time_limit, verbose=False):
    """Mirror descent with multiplicative updates targeting peaks."""
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)
    
    best_obj = obj_normalized(f)
    best_f = f.copy()
    
    t0 = time.monotonic()
    lr = 0.02
    no_improve = 0
    it = 0
    temp = 0.01
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        
        # Weight peaks
        k_actual = min(30, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / (temp + 1e-15)
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        w = w / ws
        
        # Compute action
        action = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            action[idx] += 2.0 * w[i] * f[k - idx]
        
        # Multiplicative update
        f = f * np.exp(-lr * action)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()
        
        co = obj_normalized(f)
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        lr *= 0.9995
        temp *= 1.005
        
        if no_improve > 100:
            rng = np.random.default_rng(it * 31)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.003, 1e-15)
            f = f / f.sum()
            lr = 0.02
            no_improve = 0
        
        if lr < 1e-7:
            break
        
        if verbose and it % 50 == 0:
            print(f"      MD it={it}: obj={best_obj:.10f}, lr={lr:.6f}")
    
    return best_f, best_obj


def peak_shaving(f0, time_limit, verbose=False):
    """Aggressive peak-targeted coordinate descent."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_obj = obj_normalized(f)
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and no_improve < 60:
        conv = fast_convolve(f)
        M = conv.max()
        improved = False
        
        # Top peaks
        top_k = min(15, len(conv))
        peak_indices = np.argsort(conv)[-top_k:]
        
        for k in peak_indices:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue
            
            idx = np.arange(js, je + 1)
            products = f[idx] * f[k - idx]
            total = products.sum()
            if total < 1e-15:
                continue
            
            # Try reducing top contributors
            order = np.argsort(products)[::-1]
            for oi in range(min(5, len(order))):
                j_max = int(idx[order[oi]])
                old_val = f[j_max]
                if old_val < 1e-12:
                    continue
                
                for mult in [0.4, 0.6, 0.75, 0.9, 1.2, 1.5]:
                    new_val = old_val * mult
                    if new_val < 0:
                        continue
                    
                    trial = f.copy()
                    trial[j_max] = new_val
                    d = old_val - new_val
                    if abs(d) > 1e-15:
                        # Redistribute to non-contributors
                        mask = np.ones(n, dtype=bool)
                        mask[js:je + 1] = False
                        mask[j_max] = False  # Already handled
                        cnt = mask.sum()
                        if cnt > 0:
                            w = f[mask]
                            wsum = w.sum()
                            if wsum > 1e-15:
                                trial[mask] += d * w / wsum
                    
                    trial = np.maximum(trial, 0)
                    tsum = trial.sum()
                    if tsum < 1e-15:
                        continue
                    trial = trial / tsum
                    obj = obj_normalized(trial)
                    if obj < best_obj - 1e-13:
                        best_obj = obj
                        best_f = trial.copy()
                        f = trial
                        improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 10:
                # Aggressive: reduce all peak contributors
                k = int(np.argmax(fast_convolve(f)))
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                trial = f.copy()
                trial[js:je + 1] *= 0.9
                d = f.sum() - trial.sum()
                mask = np.ones(n, dtype=bool)
                mask[js:je + 1] = False
                if mask.sum() > 0:
                    w = f[mask]
                    nw = w.sum()
                    if nw > 1e-15:
                        trial[mask] += d * w / nw
                trial = np.maximum(trial, 0)
                trial = trial / trial.sum()
                obj = obj_normalized(trial)
                if obj < best_obj:
                    f = trial
                    best_obj = obj
                    best_f = f.copy()
                    no_improve = 0
                elif no_improve > 30:
                    break
        else:
            no_improve = 0
        
        if verbose and no_improve == 0:
            print(f"      Peak CD: obj={best_obj:.10f}")
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, sigma_frac=0.25):
    """Gaussian initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_super_gaussian(n, width=0.3, power=4):
    """Super-Gaussian (flatter top) initial sequence."""
    x = np.arange(n) - n / 2.0
    hw = n * width / 2
    t = np.abs(x) / hw
    a = np.where(t < 1, (1 - t**power)/(1-0), 0)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_asymmetric(n, decay=3.0):
    """Asymmetric exponential decay."""
    x = np.arange(n)
    a = np.exp(-decay * x / n)
    a = a / a.sum()
    return a


def make_bell(n, width=0.7):
    """Cubic bell-shaped initial sequence."""
    x = np.arange(n) - n / 2.0
    half_w = n * width / 2.0
    t = np.clip(np.abs(x) / half_w, 0, 1.0)
    a = (1.0 - t) ** 3
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_lorentzian(n, gamma_frac=0.2):
    """Lorentzian (Cauchy) shaped sequence."""
    x = np.arange(n) - n / 2.0
    gamma = n * gamma_frac
    a = 1.0 / (1.0 + (x / gamma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def try_sequence(a, best_seq, best_score, label=""):
    """Evaluate a candidate sequence, return updated best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization combining smoothed gradient descent,
    LP-based Frank-Wolfe, multiplicative descent, and peak shaving.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass
    
    if parent is None:
        parent = make_gaussian(768, 0.25)
        best_seq, best_score = try_sequence(parent, best_seq, best_score, "default")
        print(f"Default: n={len(parent)}, score={best_score:.6f}")
    
    n_parent = len(parent)
    
    # === Phase 1: Diverse initialization (30s) ===
    print("=== Phase 1: Diverse Initializations ===")
    t1_start = time.monotonic()
    t1_limit = 30.0
    
    init_candidates = [parent]
    
    for n_try in [384, 512, 640, 768, 896, 1024]:
        if time.monotonic() - t1_start > t1_limit:
            break
        if n_try == n_parent:
            init_candidates.append(parent.copy())
        else:
            init_candidates.append(interpolate_sequence(parent, n_try))
        init_candidates.append(make_gaussian(n_try, 0.2))
        init_candidates.append(make_gaussian(n_try, 0.25))
        init_candidates.append(make_gaussian(n_try, 0.3))
        init_candidates.append(make_gaussian(n_try, 0.35))
        init_candidates.append(make_super_gaussian(n_try, 0.25, 4))
        init_candidates.append(make_super_gaussian(n_try, 0.3, 6))
        init_candidates.append(make_asymmetric(n_try, 2.0))
        init_candidates.append(make_asymmetric(n_try, 4.0))
        init_candidates.append(make_bell(n_try, 0.5))
        init_candidates.append(make_bell(n_try, 0.7))
        init_candidates.append(make_bell(n_try, 0.9))
        init_candidates.append(make_lorentzian(n_try, 0.15))
        init_candidates.append(make_lorentzian(n_try, 0.25))
    
    best_init = None
    best_init_obj = float('inf')
    for ic in init_candidates:
        s = ic.sum()
        if s < 1e-10:
            continue
        a_norm = ic / s
        obj = obj_normalized(a_norm)
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a_norm.copy()
        best_seq, best_score = try_sequence(a_norm, best_seq, best_score,
                                             f"init_n{len(a_norm)}")
    
    print(f"  Best init: obj={best_init_obj:.6f} (n={len(best_init)})")
    
    # === Phase 2: Nesterov Smooth Descent (400s) ===
    print(f"\n=== Phase 2: Nesterov Smooth Descent (n={len(best_init)}) ===")
    rem = deadline - time.monotonic() - 600
    if rem > 60:
        t_alloc = min(rem, 400)
        print(f"  Time: {t_alloc:.0f}s")
        a_opt, obj = nesterov_smooth_descent(best_init, time_limit=t_alloc, verbose=True)
        best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "nsgd")
    
    # === Phase 3: LP-Frank-Wolfe (400s) ===
    if best_seq is not None:
        print(f"\n=== Phase 3: LP-FW (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 500
        if rem > 60:
            t_alloc = min(rem, 400)
            print(f"  Time: {t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "lpfw1")
    
    # === Phase 4: Peak Shaving (200s) ===
    if best_seq is not None:
        print(f"\n=== Phase 4: Peak Shaving (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 400

            t_alloc = min(rem, 200)
            print(f"  Time: {t_alloc:.0f}s")
            a_opt, obj = peak_shaving(best_seq, time_limit=t_alloc, verbose=True)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "peak1")
    
    # === Phase 5: Multiplicative Descent (200s) ===
    if best_seq is not None:
        print(f"\n=== Phase 5: Mult Descent (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_alloc = min(rem, 200)
            print(f"  Time: {t_alloc:.0f}s")
            a_opt, obj = multiplicative_descent(best_seq, time_limit=t_alloc, verbose=True)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "md1")
    
    # === Phase 6: LP-FW again on best (200s) ===
    if best_seq is not None:
        print(f"\n=== Phase 6: LP-FW refinement (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 250
        if rem > 50:
            t_alloc = min(rem, 200)
            print(f"  Time: {t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "lpfw2")
    
    # === Phase 7: Multi-length exploration (150s) ===
    base = best_seq if best_seq is not None else best_init
    print(f"\n=== Phase 7: Multi-length exploration ===")

            break
        f_interp = interpolate_sequence(base, target_n)

                                             f"interp_n{target_n}")
        t_alloc = min(rem * 0.4, 50)
        if t_alloc < 20:

        print(f"  LP-FW n={target_n}: t={t_alloc:.0f}s")
        a_opt, obj = lp_fw_optimize(f_interp, time_limit=t_alloc, verbose=False)
        best_seq, best_score = try_sequence(a_opt, best_seq, best_score,
                                             f"lpfw_n{target_n}")
    
    # === Phase 8: Final polish (100s) ===
    if best_seq is not None:
        print(f"\n=== Phase 8: Final Polish ===")
        rem = deadline - time.monotonic() - 20
        if rem > 20:
            t1 = min(rem * 0.4, 50)
            t2 = min(rem * 0.3, 40)
            t3 = min(rem * 0.3, 40)

                best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "fin_peak")
            
            rem2 = deadline - time.monotonic() - 15

                best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "fin_md")
            
            rem3 = deadline - time.monotonic() - 10
            if rem3 > 5 and best_seq is not None and t3 > 5:
                print(f"  LP-FW: t={t3:.0f}s")
                a_opt, obj = lp_fw_optimize(best_seq, time_limit=t3)
                best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "fin_lpfw")
    
    # === Phase 9: Perturbed restarts ===
    if best_seq is not None:
        print(f"\n=== Phase 9: Perturbed LP-FW ===")
        n_best = len(best_seq)
        for p in range(3):
            rem = deadline - time.monotonic() - 15
            if rem < 30:
                break
            t_alloc = min(rem / (3 - p), 60)
            
            if p == 0:
                noise = rng.standard_normal(n_best) * 0.003
                f_init = project_simplex(best_seq + noise)
            elif p == 1:
                f_init = 0.9 * best_seq + 0.1 * (np.ones(n_best) / n_best)
                f_init = f_init / f_init.sum()
            else:
                # Flip and blend
                f_init = 0.5 * best_seq + 0.5 * best_seq[::-1]
                f_init = f_init / f_init.sum()
            
            print(f"  Variant {p}: t={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(f_init, time_limit=t_alloc, verbose=False)
            best_seq, best_score = try_sequence(a_opt, best_seq, best_score,
                                                 f"perturb_{p}")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
