def peak_reduction_cd(f0, time_limit):
    """Direct peak reduction via coordinate descent on peak-contributing pairs."""
    n = len(f0)

        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)

        improved = False
        for ti in top_idx:

            j2 = int(k_star - j1)
            if j1 == j2:
                continue

                trial[j2] -= reduce_amt
                # Redistribute to positions far from the peak

                contrib_mask = (np.arange(n) >= js) & (np.arange(n) <= je)

                    total_non = a[non_contrib].sum()
                    if total_non > 1e-15:

                    best_obj = obj

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_initialization(n, shape='gaussian', **kwargs):

    else:

    a = np.maximum(a, 0)
    s = a.sum()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Diverse initialization at multiple lengths
2. Adam with log-sum-exp smoothed surrogate (fast convergence)
3. Momentum subgradient method on actual max with softmax-averaged subgradients
   (key: O(n) subgradient via identity grad_k[j]=2*a[k-j], enabling >1M iterations)
4. Direct peak-reduction coordinate descent
5. Multi-length exploration with interpolation and re-optimization
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_subgradient(a, temp):
    """
    Compute softmax-weighted subgradient of max(conv(a,a)).
    Uses identity: subgrad_k[j] = 2*a[k-j] for valid indices.
    Returns (subgradient vector of length n, convolution, max value).
    """
    n = len(a)
    c = fast_convolve(a)
    max_c = c.max()
    scaled = (c - max_c) / (temp * max_c + 1e-15)
    w = np.exp(scaled)
    w /= w.sum()
    w_rev = w[::-1]
    total_len = 3 * n - 2
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_r = np.fft.irfft(A * W, fft_size)[:total_len]
    idx = 2 * n - 2 - np.arange(n)
    h = 2.0 * conv_r[idx]
    return h, c, max_c


def compute_logsumexp_grad(f, tau):
    """Compute gradient of log(sum exp(conv/tau)) smoothed surrogate."""
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    w_rev = w[::-1]
    total_len = 3 * n - 2
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_r = np.fft.irfft(A * W, fft_size)[:total_len]
    h = 2.0 * conv_r[2 * n - 2 - np.arange(n)]
    grad = h / tau
    return grad, conv

                        trial /= trial.sum()
                        obj = fast_convolve(trial).max()
                        if obj < best_obj - 1e-14:
                            a = trial
                            best_obj = obj
                            best_a = a.copy()
                            improved = True
                            break
            if improved:
                break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                rng = np.random.default_rng(it * 31 + 17)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
                no_improve = 0

        if verbose and it % 50 == 0:
            print(f"    PeakCD iter {it}: obj={2*n*best_obj:.8f}")

    return best_a, 2 * n * best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_initialization(n, shape='gaussian', **kw):
    """Create various initialization shapes."""
    x = np.arange(n, dtype=np.float64)
    if shape == 'gaussian':
        center = kw.get('center', 0.45) * n
        sigma = kw.get('sigma', 0.25) * n
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    elif shape == 'cosine':
        a = 0.5 * (1 + np.cos(np.pi * (x - n / 2) / (n / 2)))
        a = np.maximum(a, 0)
    elif shape == 'flat_top':
        half = n * kw.get('width', 0.55) / 2
        t = np.clip(np.abs(x - n / 2) / half, 0, 1)
        a = (1 - t * t) ** 2
    elif shape == 'sech':
        a = 1.0 / np.cosh((x - n / 2) / (n * 0.2))
    elif shape == 'power':
        p = kw.get('p', 2.0)
        t = np.abs(x - n / 2) / (n / 2)
        a = (1 - t ** 2) ** p
        a = np.maximum(a, 0)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a
