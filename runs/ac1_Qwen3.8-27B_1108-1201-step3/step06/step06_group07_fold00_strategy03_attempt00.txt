def main_candidate_search(budget_s=1800):
    start_time = time.time()
    deadline = start_time + budget_s
    
    # Initial diverse candidate generation
    best_seq = None

    def make_gaussian(n=768, sigma_frac=0.3):
        x = np.linspace(-1.0, 1.0, n)  # time domain
        sigma = n * sigma_frac
        a = np.exp(-0.5 * (x/sigma)**2)
        a = np.maximum(a, 0)
        s = sum(a)
        if s > 1e-15:
            a = a / s
        return a
    
    def make_bell(n=768, width_frac=0.6):
        x = np.arange(n) - n/2.0
        half_width = n * width_frac / 2.0
        t = np.clip(x/half_width, -0.5, 1.0)  # blended domain to avoid extremes
        a = (1 - np.abs(t)) ** 4
        a = np.maximum(a, 0)
        s = sum(a)
        if s > 1e-15:
            a = a / s
        return a
    
    # --- Candidate pool initialization ---
    # Candidate type 1: full length Gaussian
    cand1 = make_gaussian(768, 0.3)
    score1 = evaluate_sequence(cand1.tolist())
    if score1 < best_score:
        best_score = score1
        best_seq = cand1.copy()
    init_scores = []   # This is now not generated due to time limits
    init_candidates = [("Gauss", cand1), ("Bell", make_bell(768, 0.5))]
    
    for (name, seq) in init_candidates:

            best_seq = seq.copy()
    
    # --- Mirror Descent Optimization ---
    # We allocate 40% of time to mirror descent
    t_phase1 = int(budget_s * 0.4)
    
    def project_simplex(v):
        n = len(v)
        if n == 1:
            return np.array([max(v[0], 0.0)])
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.minimum(np.where(u * np.arange(1, n+1) > cssv)[0], n-1) if np.any(u * np.arange(1, n+1) > cssv) else np.array([n-1])
        if len(ind) == 0:
            return np.ones(n) / n
        theta = (cssv[ind] + 1e-10) / (ind + 1.0)
        return np.maximum(v - theta, 0.0)
    
    def compute_subgradient(a, peak_indices, weights):
        n = len(a)
        g = np.zeros(n)
        for idx in range(len(peak_indices)):
            k = int(peak_indices[idx])
            w_k = weights[idx]
            if w_k < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n-1, k)
            if je < js:
                continue
            idx_array = np.arange(js, je+1)
            products = a[idx_array] * a[k - idx_array]
            if len(products) == 0:
                continue
            max_index = idx_array[np.argmax(products)]
            g[max_index] += w_k * 2 * a[k - max_index]
        return g
    
    # Mirror descent with time limit and adaptive learning/temperature
    def mirror_descent(a0, time_limit, top_k=30, temp_init=0.0005, lr_init=0.01, verbose=False, label=""):
        a = np.maximum(a0.astype(float), 1e-15)
