"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LSE-subgradient descent with
flat-spectrum initialization, adaptive temperature, Barzilai-Borwein steps,
and multi-length exploration. Leverages the insight that optimal sequences
have approximately flat power spectra (|FFT(f)| ≈ constant).

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex (sum=1, x>=0)."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_subgradient(f, conv, tau, top_k_frac=0.2):
    """Compute smoothed subgradient of max(conv) using LSE weighting."""
    n = len(f)
    len_c = len(conv)
    k_top = max(5, int(len_c * top_k_frac))
    peaks = np.argpartition(conv, -k_top)[-k_top:]
    pv = conv[peaks]
    scaled = tau * pv
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-15:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        idx = np.arange(js, je + 1)
        contrib = f[k - idx] + f[idx]
        sg[js:je + 1] += w[i] * contrib
    return sg

    n = len(f)

    best_f = f.copy()

    prev_grad = None

        M = conv.max()

        if M < best_obj - 1e-15:
            best_obj = M
            best_f = f.copy()
            no_improve = 0

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex (sum=1, x>=0)."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_gradient(f, conv, tau, top_k_frac=0.2):
    """Compute LSE-weighted subgradient of max(conv)."""
    n = len(f)
    len_c = len(conv)
    k_top = max(5, int(len_c * top_k_frac))
    peaks = np.argpartition(conv, -k_top)[-k_top:]
    pv = conv[peaks]
    scaled = tau * pv
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        idx_max = np.argmax(pv)
        w = np.zeros(len(peaks))
        w[idx_max] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-15:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        idx = np.arange(js, je + 1)
        contrib = f[k - idx] + f[idx]
        sg[js:je + 1] += w[i] * contrib
    return sg

def flat_spectrum_init(n, seed=42):
    """Initialize with approximately flat power spectrum."""
    rng = np.random.default_rng(seed)
    # Random Hermitian phases
    phases = rng.uniform(-np.pi, np.pi, n)
    phases = (phases + phases[::-1].copy()) / 2  # enforce symmetry approximately
    # Actually, for real signal: F[m] = conj(F[n-m])
    # Let's just use random phases and take real part
    F = np.zeros(n, dtype=complex)
    F[0] = 1.0  # DC = sum = 1
    target_mag = 1.0 / np.sqrt(2 * n - 1)
    for m in range(1, n):
        F[m] = target_mag * np.exp(1j * phases[m])
    # Enforce Hermitian symmetry
    F = (F + np.conj(F[np.arange(n)[::-1]])[np.arange(n)[::-1]]) / 2
    F[0] = 1.0
    f = np.fft.irfft(F, n)
    f = np.maximum(f, 0)
    s = f.sum()
    if s > 1e-15:
        f /= s
    else:
        f = np.ones(n) / n
    return f

def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

    n = len(f)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    lr = 0.01
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        if M < best_obj - 1e-16:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Target top peaks
        k_top = max(3, len(conv) // 20)
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        sg = np.zeros(n)
        for k in peaks:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[js:je + 1] += f[k - idx] + f[idx]
        sg /= len(peaks)

        f_new = f * np.exp(-lr * sg)
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()

        obj_new = fast_convolve(f_new).max()
        if obj_new < M:
            f = f_new
            lr = min(lr * 1.01, 0.04)
        else:
            lr *= 0.7
            if lr < 1e-10:
                break

        if no_improve > 150:
            rng = np.random.default_rng(it * 97 + 11)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.003, 1e-15)
            f = f / f.sum()
            lr = 0.01
            no_improve = 0

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization combining LSE-GD, multiplicative descent,
    coordinate descent, peak shaving, and multi-length exploration.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)
    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # Phase 1: Try parent with different optimizers
    print("=== Phase 1: Optimize Parent ===")
    if parent is not None:
        n_p = len(parent)
        # LSE-GD on parent
        rem = deadline - time.monotonic() - 900
        if rem > 60:
            t = min(rem, 200)
            print(f"  LSE-GD on parent n={n_p}, t={t:.0f}s")
            a_opt, _ = lse_gd(parent, time_limit=t, tau_init=30.0, lr_init=0.002, verbose=True)
            try_seq(a_opt, "lse_parent")

        # Multiplicative descent
        rem = deadline - time.monotonic() - 800
        if rem > 60:
            t = min(rem, 150)
            print(f"  MD on parent n={n_p}, t={t:.0f}s")
            a_opt, _ = multiplicative_descent(parent, time_limit=t, lr_init=0.015, verbose=True)
            try_seq(a_opt, "md_parent")

        # Coordinate descent on best so far
        if best_seq is not None:
            rem = deadline - time.monotonic() - 700
            if rem > 30:
                t = min(rem, 100)
                print(f"  CD on best n={len(best_seq)}, t={t:.0f}s")
                a_opt, _ = coordinate_descent(best_seq, time_limit=t, verbose=True)
                try_seq(a_opt, "cd1")

    # Phase 2: Multi-length exploration with diverse inits
    print("\n=== Phase 2: Multi-Length + Diverse Inits ===")
    target_lengths = [512, 640, 768, 896, 1024, 1280]
    if parent is not None:
        target_lengths.insert(0, len(parent))

    for n_try in target_lengths:
        if time.monotonic() > deadline - 200:
            break
        rem = deadline - time.monotonic() - 200
        per_init_time = max(20, min(60, rem / 12))

        # Get initializations for this length
        inits = []
        if parent is not None and len(parent) != n_try:
            inits.append(interpolate_seq(parent, n_try))
        inits.append(parametric_init(n_try, variant=0, seed=seed))
        inits.append(parametric_init(n_try, variant=1, seed=seed))
        inits.append(parametric_init(n_try, variant=3, seed=seed))
        inits.append(flat_spectrum_init(n_try, seed=seed))

        # Score inits quickly
        scored = []
        for idx, a_init in enumerate(inits):
            conv = fast_convolve(a_init)
            obj = 2 * n_try * conv.max()
            scored.append((obj, a_init, f"init_{idx}_n{n_try}"))
        scored.sort(key=lambda x: x[0])

        # Optimize top 2
        for obj_i, a_i, label in scored[:2]:
            if time.monotonic() > deadline - 180:
                break
            rem = deadline - time.monotonic() - 180
            t = min(per_init_time, rem / 2)
            if t < 15:
                break
            print(f"  LSE-GD {label} (fast obj={obj_i:.4f}), t={t:.0f}s")
            a_opt, _ = lse_gd(a_i, time_limit=t, tau_init=20.0, lr_init=0.003)
            sc = try_seq(a_opt, label)

            # Quick CD on result
            if sc < best_score + 0.01 and time.monotonic() < deadline - 150:
                t_cd = min(40, (deadline - time.monotonic() - 150))
                if t_cd > 15:
                    a_opt2, _ = coordinate_descent(a_opt, time_limit=t_cd)
                    try_seq(a_opt2, label + "_cd")

    # Phase 3: Intensive refinement of best
    print("\n=== Phase 3: Intensive Refinement ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Coordinate descent (long)
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t = min(rem, 120)
            print(f"  Long CD n={n_best}, t={t:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t, verbose=True)
            try_seq(a_opt, "cd_long")

        # LSE-GD with high tau (focus on exact max)
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t = min(rem, 80)
            print(f"  High-tau LSE-GD, t={t:.0f}s")
            a_opt, _ = lse_gd(best_seq, time_limit=t, tau_init=500.0, lr_init=0.001,
                              tau_growth=1.001, top_k_frac=0.05)
            try_seq(a_opt, "hitauga")

        # Multiplicative with low lr
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t = min(rem, 50)
            a_opt, _ = multiplicative_descent(best_seq, time_limit=t, lr_init=0.005, tau_init=100.0)
            try_seq(a_opt, "md_low")

    # Phase 4: Perturbation and escape
    print("\n=== Phase 4: Escape Attempts ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rng = np.random.default_rng(seed + 999)

        for attempt in range(5):
            if time.monotonic() > deadline - 80:
                break
            rem = deadline - time.monotonic() - 80
            t = min(rem / max(1, 5 - attempt), 50)
            if t < 15:
                break

            # Different perturbation strategies
            if attempt == 0:
                noise = rng.standard_normal(n_best) * 0.01
                f_init = project_simplex(best_seq + noise)
            elif attempt == 1:
                # Blend with reversed
                f_init = 0.6 * best_seq + 0.4 * best_seq[::-1]
                f_init = f_init / f_init.sum()
            elif attempt == 2:
                # Blend with parametric
                p_init = parametric_init(n_best, variant=attempt, seed=seed + attempt * 17)
                f_init = 0.7 * best_seq + 0.3 * p_init
                f_init = f_init / f_init.sum()
            elif attempt == 3:
                # Sparse perturbation
                f_init = best_seq.copy()
                idx = rng.choice(n_best, size=n_best // 4, replace=False)
                f_init[idx] *= rng.uniform(0.5, 2.0, size=len(idx))
                f_init = np.maximum(f_init, 0)
                f_init = f_init / f_init.sum()
            else:
                # Interpolate to different length and back
                new_n = n_best + rng.integers(-16, 17)
                if new_n > 2 and new_n < 3000:
                    f_interp = interpolate_seq(best_seq, new_n)
                    f_init = interpolate_seq(f_interp, n_best)
                else:
                    f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.02)

            print(f"  Escape {attempt}, t={t:.0f}s")
            a_opt, _ = lse_gd(f_init, time_limit=t, tau_init=30.0, lr_init=0.003)
            try_seq(a_opt, f"escape_{attempt}")

            if best_seq is not None:
                rem2 = deadline - time.monotonic() - 60
                if rem2 > 15:
                    a_opt2, _ = coordinate_descent(best_seq, time_limit=min(30, rem2))
                    try_seq(a_opt2, f"escape_{attempt}_cd")

    # Phase 5: Final polish
    print("\n=== Phase 5: Final Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t = min(rem, 40)
            a_opt, _ = coordinate_descent(best_seq, time_limit=t)
            try_seq(a_opt, "final_cd")

        rem = deadline - time.monotonic() - 10
        if rem > 10:
            t = min(rem, 20)
            a_opt, _ = multiplicative_descent(best_seq, time_limit=t, lr_init=0.003, tau_init=200.0)
            try_seq(a_opt, "final_md")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final check
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
