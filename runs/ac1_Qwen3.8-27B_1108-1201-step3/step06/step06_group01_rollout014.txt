import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    """Compute the score for a non-negative sequence."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    a_norm = a / s
    b = fast_convolve(a_norm)
    return 2.0 * n * b.max()

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def get_lp_solver():
    global _lp_solver
    if '_lp_solver' in globals():
        return _lp_solver
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.CBC, cp.GLPK, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return cp.ECOS

def solve_lp(a_fixed, solver=None, active_constraints=None):
    """Solve the LP: min M s.t. (T_a @ b)_k <= M for k in active, b >= 0, sum(b) = 1"""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    
    if active_constraints is not None:
        constraints = [T[active_constraints] @ b_var <= M_var, cp.sum(b_var) == 1]
    else:
        constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an active-set KKT iteration
combined with LP best-response solving, direct KKT linear system solving,
subgradient polishing, and L-BFGS refinement. The key innovation is a

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score_local(a):
    """Compute score without calling evaluate_sequence."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    a_norm = a / s
    b = fast_convolve(a_norm)
    return float(2.0 * n * b.max())


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    global _lp_solver
    if '_lp_solver' in globals():
        return _lp_solver
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.CBC, cp.GLPK, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return cp.ECOS


def solve_best_response_lp(a_fixed, solver=None, active_k=None):
    """
    Given fixed a (non-negative, sum=1), find b (non-negative, sum=1)
    minimizing max_k conv(a, b)[k]. If active_k given, only enforce those constraints.
    Returns (b, M).
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)

    if active_k is not None and len(active_k) < len(T):
        constraints = [T[active_k] @ b_var <= M_var, cp.sum(b_var) == 1]
    else:
        constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]

    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.ECOS, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def find_active_set(b, n, threshold_frac=0.999, top_k=None):
    """Find active constraints (lag indices near the peak)."""
    b_max = b.max()
    if top_k is not None:
        return np.argpartition(b, -top_k)[-top_k:]
    active = np.where(b >= threshold_frac * b_max)[0]
    if len(active) == 0:
        active = np.array([np.argmax(b)])
    return active


def solve_kkt_linear_system(a, K, S, max_inner=100):
    """
    Solve the KKT conditions via alternating linear system solves.
    Given active set K and support S, solve:
      For i in S: sum_{k in K} w_k * a[k-i] = lambda
      Sum of a over S = 1
      Sum of w over K = 1
    Uses alternating fixed-point: solve for a given w, then for w given a.
    """
    n = len(a)
    nS = len(S)
    nK = len(K)
    if nS == 0 or nK == 0:
        return a, w

    # Initialize w uniformly
    w = np.ones(nK) / nK
    a_S = a[S].copy()
    a_S = np.maximum(a_S, 0)
    s_sum = a_S.sum()
    if s_sum < 1e-15:
        a_S = np.ones(nS) / nS
    else:
        a_S = a_S / s_sum

    lambda_val = 0.0
    prev_a = a_S.copy()

    for outer in range(max_inner):
        # Build matrix M_a: M_a[i_idx, j_idx] = w[k_idx] where k = S[i_idx] + S[j_idx] if in K
        # Equation for index i=S[i_idx]: sum_{k in K} w_k * a[k - S[i_idx]] = lambda
        # a[k - S[i_idx]] = a_S[j_idx] if k - S[i_idx] == S[j_idx], i.e., k = S[i_idx] + S[j_idx]
        M_a = np.zeros((nS, nS))
        for ii in range(nS):
            for jj in range(nS):
                k = S[ii] + S[jj]
                if 0 <= k <= 2 * n - 2:
                    # Check if k is in K
                    for kk in range(nK):
                        if K[kk] == k:
                            M_a[ii, jj] = w[kk]
                            break

        # Solve: M_a @ a_S = lambda * 1, and sum(a_S) = 1
        # From M_a @ a_S = lambda * 1: all components of M_a @ a_S are equal
        # Write as: (M_a - J/nS) @ a_S = 0 (nS-1 independent eqs) + 1^T @ a_S = 1
        J = np.ones((nS, nS)) / nS
        A_sys = M_a - J
        b_sys = np.zeros(nS)
        b_sys[-1] = 1.0
        A_sys[-1, :] = 1.0

        try:
            a_S_new = np.linalg.solve(A_sys, b_sys)
        except np.linalg.LinAlgError:
            try:
                a_S_new = np.linalg.lstsq(A_sys, b_sys, rcond=None)[0]
            except Exception:
                break

        a_S_new = np.maximum(a_S_new, 0)
        s_sum = a_S_new.sum()
        if s_sum < 1e-15:
            a_S_new = np.ones(nS) / nS
            s_sum = 1.0
        else:
            a_S_new = a_S_new / s_sum

        # Check convergence of a
        if np.max(np.abs(a_S_new - prev_a)) < 1e-10:
            a_S = a_S_new
            break
        a_S = a_S_new
        prev_a = a_S_new.copy()

        # Update full a
        a_full = np.zeros(n)
        a_full[S] = a_S

        # Solve for w given a_full
        # Equation for k=K[kk]: sum_{i in S} a_i * a[k - i] should equal lambda
        # Define N[kk, ii] = a_full[S[ii]] * a_full[K[kk] - S[ii]] (if valid)
        N = np.zeros((nK, nS))
        for kk in range(nK):
            for ii in range(nS):
                j = K[kk] - S[ii]
                if 0 <= j < n:
                    N[kk, ii] = a_full[j]
        # We want: N^T @ w_K = lambda * 1_S (for each i in S)
        # i.e., for each ii: sum_{kk} N[kk, ii] * w[kk] = lambda
        # And sum(w) = 1
        # Matrix: N^T is nS x nK. We want N^T @ w = lambda * 1
        # This is underdetermined if nK > nS. Use least squares with constraint sum(w)=1.
        Nt = N.T  # nS x nK
        # (Nt - J/nK * 1) @ w = 0 (nS-1 eqs) + 1^T w = 1
        # Actually: Nt @ w = lambda * ones(nS), sum(w) = 1
        # => (Nt - ones(nS,1) @ ones(1,nK)/nS) @ w = 0 (nS-1 independent)
        # Plus: ones(1,nK) @ w = 1
        Jw = np.ones((nS, nK)) / nK
        A_w = Nt - Jw
        b_w = np.zeros(nS + 1)
        A_w_full = np.vstack([A_w, np.ones((1, nK))])
        b_w_full = np.zeros(nS + 1)
        b_w_full[-1] = 1.0

        try:
            w_new = np.linalg.solve(A_w_full, b_w_full)
        except np.linalg.LinAlgError:
            try:
                w_new = np.linalg.lstsq(A_w_full, b_w_full, rcond=None)[0]
            except Exception:
                break

        w_new = np.maximum(w_new, 0)
        w_sum = w_new.sum()
        if w_sum < 1e-15:
            w_new = np.ones(nK) / nK
            w_sum = 1.0
        else:
            w_new = w_new / w_sum

        w = w_new

    # Update a
    a_full = np.zeros(n)
    a_full[S] = a_S
    a_full = np.maximum(a_full, 0)
    s = a_full.sum()
    if s < 1e-15:
        a_full = np.ones(n) / n
        s = 1.0
    else:
        a_full = a_full / s
    return a_full, w


def active_set_lp_iteration(a0, time_limit, solver=None, max_iters=500, verbose=False):

    t0 = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute auto-correlation
        b = fast_convolve(a)
        b_max = b.max()

        # Find active set
        active = find_active_set(b, n, threshold_frac=0.995)

        # Find support
        S = np.where(a > 1e-10)[0]
        if len(S) < 2:
            S = np.arange(n)

        # LP best response
        b_new, M_lp = solve_best_response_lp(a, solver=solver, active_k=active)
        if b_new is not None:
            # Damped update
            alpha = min(1.0, 3.0 / (it + 1))
            a_trial = (1 - alpha) * a + alpha * b_new
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            score_trial = compute_score_local(a_trial)
            if score_trial < best_score - 1e-12:
                best_score = score_trial
                best_a = a_trial.copy()
            a = a_trial

        # KKT refinement (every few iterations)
        if it % 3 == 2:
            b_curr = fast_convolve(a)
            active_curr = find_active_set(b_curr, n, threshold_frac=0.99)
            S_curr = np.where(a > 1e-8)[0]

                score_kkt = compute_score_local(a_kkt)

                    best_a = a_kkt.copy()
                    a = a_kkt
                else:
                    # Blend
                    a = 0.7 * a + 0.3 * a_kkt
                    a = np.maximum(a, 0)
                    a = a / a.sum()

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AS-LP iter={it} ({elapsed:.0f}s): best={best_score:.6f}")

def subgradient_refine(a0, time_limit, lr_init=0.005, top_k=30, verbose=False):
    """Subgradient method with momentum and adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score_local(a)
    v = np.zeros(n)
    mom = 0.9
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        temp = 0.01 * max(pv.max(), 1e-10)
        scaled = (pv - pv.max()) / temp
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

        v = mom * v - lr * sg

        # Project onto non-negative simplex
        a_candidate = a + v
        u = np.sort(a_candidate)[::-1]
        cssv = np.cumsum(u)
        cond = u * np.arange(1, n + 1) > (cssv - 1)
        if cond.any():
            rho = np.nonzero(cond)[0][-1]
            theta = (cssv[rho] - 1.0) / (rho + 1.0)
            a_new = np.maximum(a_candidate - theta, 0.0)
        else:
            a_new = np.ones(n) / n

        a_new = a_new / a_new.sum()
        score = compute_score_local(a_new)

        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            # Reduce momentum slowly
            mom = max(0.5, mom * 0.999)
        else:
            no_improve += 1

        lr *= 0.999
        if no_improve > 200:
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 1000000)
            a = best_a + rng.standard_normal(n) * 0.002
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            lr = max(lr * 2, 0.003)
            no_improve = 0
            mom = 0.9

        if verbose and (no_improve + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad ({elapsed:.0f}s): score={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def z_from_a(a):
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def lbgfs_refine(a0, time_limit, T=100, maxiter=3000, verbose=False):
    """L-BFGS refinement in log-space with soft-max smoothing."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    stopped = [False]

    def func_and_grad(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15, np.zeros_like(z)
        a = softmax_from_z(z)
        b = fast_convolve(a)
        Tb = T * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        log_sum_exp = max_Tb + np.log(exp_Tb.sum())
        obj = (2.0 * n / T) * log_sum_exp
        w = exp_Tb / exp_Tb.sum()
        # Gradient: cross-correlation of w with a
        size = 4 * n
        fft_size = 1 << (size - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        grad_a = irfft(np.conj(W) * A, fft_size)[:n]
        grad_a = 2.0 * n * grad_a
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - a * S)
        return obj, grad_z

    def func(z):
        obj, _ = func_and_grad(z)
        return obj

    def jac(z):
        _, g = func_and_grad(z)
        return g

    try:
        res = minimize(
            fun=func,
            x0=z0,
            jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12,
                     'maxfun': maxiter * 50, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
    except Exception:
        a_out = a0

    return a_out

                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if not improved:
                # Try a small random perturbation
                rng = np.random.default_rng(step)
                a_trial = a + rng.standard_normal(n) * 0.001
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = compute_score_local(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                if not improved:
                    break

        if verbose and step % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave step={step} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate and return updated best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(budget_s=1800.0):
    """
    Multi-phase optimization combining:
    1. Active-set LP iteration (novel KKT-based approach)
    2. Fictitious play (alternating LP best responses)
    3. Direct KKT linear system solving
    4. L-BFGS soft-max polishing
    5. Subgradient and peak-shaving local search
    6. Multi-length exploration
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score

    # === Phase 1: Active-Set LP at parent length ===
    print("\n=== Phase 1: Active-Set LP ===")
    if best_seq is not None:
        n_par = len(best_seq)
        rem = deadline - time.monotonic() - 600
        if rem > 60:
            t_alloc = min(rem * 0.4, 300.0)
            print(f"  AS-LP n={n_par}, budget={t_alloc:.0f}s")
            a_as, score_as = active_set_lp_iteration(
                best_seq, t_alloc, solver=solver, verbose=True
            )
            best_seq, best_score = check_and_update(a_as, f"ASLP_n{n_par}", best_seq, best_score)

    # === Phase 2: Fictitious Play ===
    print(f"\n=== Phase 2: Fictitious Play (best={best_score:.6f}) ===")
    n_values = [768, 640, 896, 1024, 1280, 512]
    if parent is not None and len(parent) not in n_values:
        n_values.insert(0, len(parent))

    # Prioritize by proximity to current best length
    cur_n = len(best_seq) if best_seq is not None else 768
    n_values_sorted = sorted(n_values, key=lambda x: abs(x - cur_n))
    fp_n_values = n_values_sorted[:4]

    for n_try in fp_n_values:
        rem = deadline - time.monotonic() - 400
        if rem < 60:
            break
        t_alloc = min(rem * 0.5, 200.0)
        if t_alloc < 30:
            continue
        print(f"  FP n={n_try}, budget={t_alloc:.0f}s")
        initial = None
        if best_seq is not None:
            initial = interpolate_sequence(best_seq, n_try)
        a_fp, score_fp, iters = fictitious_play_optimize(
            n_try, t_alloc, initial=initial, solver=solver, verbose=False
        )
        best_seq, best_score = check_and_update(a_fp, f"FP_n{n_try}", best_seq, best_score)

    # === Phase 3: KKT + LP combined refinement ===
    print(f"\n=== Phase 3: KKT Refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_alloc = min(rem * 0.4, 150.0)
            print(f"  KKT n={n_best}, budget={t_alloc:.0f}s")
            a_kkt, score_kkt = active_set_lp_iteration(
                best_seq, t_alloc, solver=solver, verbose=True
            )
            best_seq, best_score = check_and_update(a_kkt, f"KKT_n{n_best}", best_seq, best_score)

    # === Phase 4: L-BFGS Polish ===
    print(f"\n=== Phase 4: L-BFGS Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for T in [50, 200]:
            rem = deadline - time.monotonic() - 100
            if rem < 15:
                break
            t_alloc = min(rem * 0.3, 60.0)
            print(f"  L-BFGS T={T}, n={n_best}, budget={t_alloc:.0f}s")
            a_lb = lbgfs_refine(best_seq, t_alloc, T=T)
            best_seq, best_score = check_and_update(a_lb, f"lbgfs_T{T}", best_seq, best_score)

    # === Phase 5: Subgradient + Peak shaving ===
    print(f"\n=== Phase 5: Local Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t_sg = min(rem * 0.35, 60.0)
            top_k = max(20, n_best // 12)
            a_sg, _ = subgradient_refine(best_seq, t_sg, lr_init=0.003, top_k=top_k)
            best_seq, best_score = check_and_update(a_sg, "subgrad", best_seq, best_score)

        rem = deadline - time.monotonic() - 50
        if rem > 15:
            t_ps = min(rem * 0.5, 40.0)
            a_ps, _ = peak_shave_local(best_seq, t_ps)
            best_seq, best_score = check_and_update(a_ps, "peak_shave", best_seq, best_score)

    # === Phase 6: Second round with refined start ===
    print(f"\n=== Phase 6: Round 2 (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 60
        if rem > 40:
            t_alloc = min(rem * 0.5, 150.0)
            print(f"  AS-LP round 2: n={n_best}, budget={t_alloc:.0f}s")
            a_as2, _ = active_set_lp_iteration(best_seq, t_alloc, solver=solver, verbose=False)
            best_seq, best_score = check_and_update(a_as2, f"ASLP2_n{n_best}", best_seq, best_score)

            # L-BFGS on new best
            rem = deadline - time.monotonic() - 40
            if rem > 20:
                a_lb2 = lbgfs_refine(best_seq, min(rem * 0.5, 40.0), T=100)
                best_seq, best_score = check_and_update(a_lb2, "lbgfs_r2", best_seq, best_score)

    # === Phase 7: Try alternative lengths ===
    print(f"\n=== Phase 7: Alt Lengths (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        alt_lengths = []
        for factor in [0.85, 1.15, 1.3, 0.7]:
            alt_n = int(n_best * factor)
            if 256 <= alt_n <= 2048 and alt_n != n_best and alt_n not in alt_lengths:
                alt_lengths.append(alt_n)

        for alt_n in alt_lengths[:3]:
            rem = deadline - time.monotonic() - 30
            if rem < 30:
                break
            t_alloc = min(rem * 0.4, 80.0)
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, _, _ = fictitious_play_optimize(
                alt_n, t_alloc, initial=init_alt, solver=solver, verbose=False
            )
            best_seq, best_score = check_and_update(a_alt, f"FP_alt_n{alt_n}", best_seq, best_score)

            if best_seq is not None and len(best_seq) == alt_n:
                rem = deadline - time.monotonic() - 20
                if rem > 15:
                    a_kkt_alt, _ = active_set_lp_iteration(best_seq, rem * 0.6, solver=solver)
                    best_seq, best_score = check_and_update(a_kkt_alt, f"kkt_alt_n{alt_n}", best_seq, best_score)

    # === Phase 8: Final micro-polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 8:
            a_final = lbgfs_refine(best_seq, min(rem * 0.4, 20.0), T=300, maxiter=2000)
            best_seq, best_score = check_and_update(a_final, "final_lbgfs", best_seq, best_score)

        rem = deadline - time.monotonic() - 3
        if rem > 5:
            a_ps_final, _ = peak_shave_local(best_seq, min(rem * 0.6, 15.0))
            best_seq, best_score = check_and_update(a_ps_final, "final_ps", best_seq, best_score)

    # Fallback
    if best_seq is None:
        if parent is not None:
            best_seq = parent
        else:
            best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
