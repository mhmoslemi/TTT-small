Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) diverse initialization, (2) Adam with cyclic temperature, 
(3) sparse-constraint LP-Frank-Wolfe, (4) coordinate descent refinement.

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_smooth_grad(f, tau):
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    w_rev = w[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)
    h = conv_result[n - 1:2 * n - 1][::-1]
    grad = (2.0 / tau) * h
    return grad, conv

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations over previous approach:
1. Constraint-generation LP-Frank-Wolfe (much faster than full LP)
2. Cyclic temperature Adam with cosine annealing and warm restarts
3. Multi-length interpolation strategy
4. Targeted coordinate descent using peak structure

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_grad(f, tau):
    """Gradient of log-sum-exp smooth approximation of max-convolution."""
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    w_rev = w[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)
    h = conv_result[n - 1:2 * n - 1][::-1]
    grad = (2.0 / tau) * h
    return grad, conv


def build_constraint_matrix(f, active_indices, n):
    """Build sparse constraint matrix for selected convolution positions."""
    rows = len(active_indices)
    A = np.zeros((rows, n))
    for idx, k in enumerate(active_indices):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A[idx, js:je + 1] = f[je - (k - js):je + 1] if je >= js else 0
    # More carefully:
    for idx, k in enumerate(active_indices):
        k = int(k)
        A[idx, :] = 0.0
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A[idx, j] = f[k - j]
    return A


def fast_lp_direction(f, M, n, max_constraints=30, max_iters=50):
    """
    Constraint-generation LP to find Frank-Wolfe direction.
    Solves: max sum(g) s.t. (f*g)[k] <= M for active k, g >= 0.
    """
    active = list(range(n - 1, min(n + 10, 2 * n - 1)))  # Start near center
    g_prev = np.ones(n) / n

    for it in range(max_iters):
        # Build constraint matrix
        active_arr = np.array(active[:max_constraints])
        A = np.zeros((len(active_arr), n))
        for idx, k in enumerate(active_arr):
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                A[idx, j] = f[k - j]

        c = -np.ones(n)
        b_ub = np.full(len(active_arr), M)
        bounds = [(0, None)] * n

        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 5.0})
            if not res.success:
                return None
            g = res.x
        except Exception:
            return None

        # Check full convolution feasibility
        fg = np.convolve(f, g)
        max_fg = fg.max()

        if max_fg <= M * 1.001:
            return g

        # Add violated constraint
        worst_k = int(np.argmax(fg))
        if worst_k not in active:
            active.append(worst_k)
            if len(active) > max_constraints:
                # Remove the least-tight constraint
                tights = []
                for ak in active:
                    tights.append(fg[ak])
                # Remove the one with smallest value (least tight)
                remove_idx = int(np.argmin(tights))
                active.pop(remove_idx)
        else:
            # Already in active set but violated - LP should be infeasible
            # This means M is too tight; return None
            return None

    return g


def lp_fw_optimize(f0, time_limit, max_constraints=30, verbose=False):
    """LP-based Frank-Wolfe with constraint generation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        # Check time per iteration
        if time.monotonic() - t0 > time_limit:
            break
        it += 1

        g = fast_lp_direction(f, M, n, max_constraints=max_constraints)
        if g is None:
            # Relax M slightly
            M *= 1.05
            if it % 50 == 0:
                M = fast_convolve(f).max() * 1.01
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            continue
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-13:
            no_improve += 1
            if no_improve > 50:
                break
            continue
        else:
            no_improve = 0

        # Backtracking line search
        best_alpha = 0.0
        best_val = M
        alpha = 1.0
        for _ in range(25):
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                alpha *= 0.5
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
            alpha *= 0.7

        if best_alpha < 1e-10:
            no_improve += 1
            if no_improve > 100:
                break
            continue

        f_new = f + best_alpha * (g - f)
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()
        conv_new = fast_convolve(f_new)
        M_new = conv_new.max()
        f = f_new
        M = M_new

        if M_new < best_obj - 1e-14:
            best_obj = M_new
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 10 == 0:
                print(f"    LP-FW iter {it}: obj={2 * n * best_obj:.8f}")
        else:
            no_improve += 1

    return best_f, 2 * n * best_obj


def adam_cyclic_optimize(f0, time_limit, verbose=False):
    """
    Adam optimization with cyclic temperature and cosine learning rate.
    Key: temperature oscillates to escape local minima.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv0 = fast_convolve(f)
    M0 = conv0.max()

    # Temperature schedule: oscillate between tau_min and tau_max
    tau_min = 0.01 * M0
    tau_max = 0.15 * M0
    tau = tau_max
    cycle_len = 800  # iterations per cycle

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    base_lr = 0.005

    best_obj = M0
    best_f = f.copy()
    no_improve = 0
    t0 = time.monotonic()
    restart_count = 0

    while time.monotonic() - t0 < time_limit:
        t += 1

        # Cyclic temperature
        phase = (t % cycle_len) / cycle_len
        tau = tau_min + (tau_max - tau_min) * 0.5 * (1 + np.cos(2 * np.pi * phase))

        # Cosine learning rate with warm restarts
        restart_phase = (t % 600) / 600.0
        lr = base_lr * 0.5 * (1 + np.cos(np.pi * restart_phase))
        lr = max(lr, base_lr * 0.05)

        grad, conv = compute_smooth_grad(f, tau)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new

        # Warm restart with perturbation
        if no_improve > 300:
            restart_count += 1
            rng = np.random.default_rng(t * 7 + restart_count * 13)
            noise_scale = 0.001 + 0.002 * min(restart_count, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            base_lr = max(base_lr * 0.8, 0.001)
            if verbose:
                print(f"    Adam restart {restart_count} at iter {t}: obj={2 * n * best_obj:.8f}")

        if verbose and t % 2000 == 0:
            print(f"    Adam iter {t}: obj={2 * n * best_obj:.8f}, tau={tau:.3e}, lr={lr:.5f}")

    return best_f, 2 * n * best_obj


def subgradient_fw(f0, time_limit, verbose=False):
    """
    Frank-Wolfe with exact subgradient of the max function.
    Very fast per iteration (no LP), but slower convergence.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0
    running_avg = f.copy()
    avg_count = 1

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Active set: positions where conv is within 1% of max
        threshold = M * 0.99
        active = np.where(conv >= threshold)[0]

        # Subgradient: 2 * Q(f) * u where u is uniform on active set
        # Q(f)[j] = sum_k u[k] * f[k-j]
        u = np.zeros(2 * n - 1)
        u[active] = 1.0 / len(active)

        u_rev = u[::-1]
        total_len = n + (2 * n - 1) - 1
        fft_size = 1 << (total_len - 1).bit_length()
        A = np.fft.rfft(f, fft_size)
        U = np.fft.rfft(u_rev, fft_size)
        conv_r = np.fft.irfft(A * U, fft_size)
        h = conv_r[n - 1:2 * n - 1][::-1]
        grad = 2.0 * h

        # FW step: g = e_{argmax grad}
        j_star = int(np.argmax(grad))
        g = np.zeros(n)
        g[j_star] = 1.0

        # Step size: gamma_k = 2/(k+2)
        gamma = 2.0 / (it + 2)

        f_new = (1 - gamma) * f + gamma * g
        f_new = np.maximum(f_new, 0.0)
        st = f_new.sum()
        if st > 1e-15:
            f_new /= st
        f = f_new

        # Running average (Nesterov)
        running_avg = running_avg + (f - running_avg) / avg_count
        avg_count += 1

        conv_new = fast_convolve(f)
        M_new = conv_new.max()

        # Also check the running average
        conv_avg = fast_convolve(running_avg / running_avg.sum())
        M_avg = conv_avg.max()

        if M_new < best_obj - 1e-14:
            best_obj = M_new
            best_f = f.copy()
            if verbose and it % 100 == 0:
                print(f"    SubGrad-FW iter {it}: obj={2 * n * best_obj:.8f}, avg_obj={2 * n * M_avg:.8f}")
        elif M_avg < best_obj - 1e-14:
            best_obj = M_avg
            best_f = running_avg / running_avg.sum()
            if verbose and it % 100 == 0:
                print(f"    SubGrad-FW iter {it} (avg): obj={2 * n * best_obj:.8f}")

        # Check time
        if it % 5000 == 0 and time.monotonic() - t0 > time_limit:
            break

    return best_f, 2 * n * best_obj


def coordinate_descent_peaks(f0, time_limit, verbose=False):
    """
    Coordinate descent that targets the top peaks of the convolution.
    Uses pairwise moves: shift mass from peak-contributing elements to others.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        improved = False

        # Get top peaks
        top_k = min(8, 2 * n - 1)
        peak_indices = np.argsort(conv)[-top_k:]

        for pk in peak_indices:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            if je < js:
                continue

            # Find the pair contributing most to this peak
            idx = np.arange(js, je + 1)
            products = f[idx] * f[pk - idx]
            j_max = idx[np.argmax(products)]
            j_other = pk - j_max

            if j_max == j_other:
                # Diagonal: reduce f[j_max]
                old_val = f[j_max]
                if old_val < 1e-12:
                    continue
                for frac in [0.3, 0.5, 0.7]:
                    trial = f.copy()
                    removed = old_val * (1 - frac)
                    trial[j_max] = old_val * frac
                    # Distribute to elements far from all peaks
                    mask = np.ones(n, dtype=bool)
                    for pk2 in peak_indices[:4]:
                        js2 = max(0, int(pk2) - n + 1)
                        je2 = min(n - 1, int(pk2))
                        mask[js2:je2 + 1] = False
                    if mask.sum() > 0:
                        weights = f[mask].copy()
                        if weights.sum() > 1e-15:
                            trial[mask] += removed * weights / weights.sum()
                    trial = np.maximum(trial, 0)
                    st = trial.sum()
                    if st > 1e-15:
                        trial /= st
                    obj = fast_convolve(trial).max()
                    if obj < best_obj - 1e-15:
                        f = trial
                        best_obj = obj
                        best_f = f.copy()
                        improved = True
                        break
            else:
                # Off-diagonal: reduce the product by adjusting one element
                old_jm = f[j_max]
                old_jo = f[j_other]
                if old_jm < 1e-12 or old_jo < 1e-12:
                    continue
                # Reduce the larger one
                target = j_max if old_jm >= old_jo else j_other
                other = j_other if target == j_max else j_max
                old_target = f[target]
                for frac in [0.4, 0.6, 0.8]:
                    trial = f.copy()
                    removed = old_target * (1 - frac)
                    trial[target] = old_target * frac
                    mask = np.ones(n, dtype=bool)
                    mask[target] = False
                    mask[other] = False
                    for pk2 in peak_indices[:4]:
                        js2 = max(0, int(pk2) - n + 1)
                        je2 = min(n - 1, int(pk2))
                        mask[js2:je2 + 1] = False
                    if mask.sum() > 0:
                        weights = f[mask].copy()
                        if weights.sum() > 1e-15:
                            trial[mask] += removed * weights / weights.sum()
                    trial = np.maximum(trial, 0)
                    st = trial.sum()
                    if st > 1e-15:
                        trial /= st
                    obj = fast_convolve(trial).max()
                    if obj < best_obj - 1e-15:
                        f = trial
                        best_obj = obj
                        best_f = f.copy()
                        improved = True
                        break

        if not improved:
            no_improve += 1
            if no_improve > 20:
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                no_improve = 0
        else:
            no_improve = 0

        if verbose and it % 50 == 0:
            print(f"    CD iter {it}: obj={2 * n * best_obj:.8f}")

    return best_f, 2 * n * best_obj


def interpolate_sequence(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_asymmetric_gaussian(n, center_frac=0.4, sigma_frac=0.3):
    """Asymmetric Gaussian initialization."""
    x = np.arange(n) - n * center_frac
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_power_decay(n, decay=2.0):
    """Power-law decay initialization."""
    x = np.arange(n, dtype=np.float64)
    a = (1.0 + x) ** (-decay)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_two_bump(n, frac1=0.3, frac2=0.7, sigma=0.15):
    """Two-bump initialization for asymmetry."""
    x = np.arange(n, dtype=np.float64) / n
    a = np.exp(-0.5 * ((x - frac1) / sigma) ** 2) + 0.7 * np.exp(-0.5 * ((x - frac2) / sigma) ** 2)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_ramp_cosine(n, ramp_alpha=0.5):
    """Ramp times cosine window."""
    x = np.arange(n, dtype=np.float64) / n
    ramp = (1.0 - ramp_alpha) + ramp_alpha * x
    cos_part = np.cos(np.pi * (x - 0.5)) ** 2
    a = ramp * cos_part
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

        inits.append((make_power_decay(n, 3.0), n))
        inits.append((make_two_bump(n, 0.25, 0.6, 0.12), n))
        inits.append((make_ramp_cosine(n, 0.6), n))
        inits.append((np.ones(n) / n, n))

    # Quick Adam on each (10s each, up to 40s total)
    for i, (a_init, n) in enumerate(inits):
        if time.monotonic() > deadline - 1500:
            break
        sc = try_seq(a_init, f"init_{i}_n{n}")
        # Very quick Adam (10s) to see if this direction is promising
        rem = deadline - time.monotonic() - 1450
        if rem > 15:
            a_opt, obj = adam_cyclic_optimize(a_init, time_limit=min(rem, 12))
            try_seq(a_opt, f"quick_adam_{i}")

    if parent is not None:
        try_seq(parent, "parent")
        # Quick Adam on parent
        rem = deadline - time.monotonic() - 1400
        if rem > 30:
            a_opt, obj = adam_cyclic_optimize(parent, time_limit=min(rem, 40))
            try_seq(a_opt, "adam_parent")

    if best_seq is None:
        best_seq = make_asymmetric_gaussian(768, 0.35, 0.3)
        best_score = try_seq(best_seq, "fallback")

    work_seq = best_seq.copy()
    work_n = len(work_seq)
    print(f"  Best after Phase 1: score={best_score:.6f}, n={work_n}")

    # === Phase 2: Main Adam optimization ===
    print(f"\n=== Phase 2: Adam Cyclic (n={work_n}) ===")
    rem = deadline - time.monotonic() - 900
    t2 = min(rem, 500)
    if t2 > 60:
        print(f"  Time: {t2:.0f}s")
        a_opt, obj = adam_cyclic_optimize(work_seq, time_limit=t2 * 0.6, verbose=True)
        try_seq(a_opt, "adam_main")

        # Second round with different params from result
        if time.monotonic() < deadline - 800:
            rem2 = deadline - time.monotonic() - 800
            t2b = min(rem2, 200)
            a_opt2, obj2 = adam_cyclic_optimize(best_seq, time_limit=t2b, verbose=True)
            try_seq(a_opt2, "adam_round2")

    # === Phase 3: Fast LP-FW ===
    print(f"\n=== Phase 3: Fast LP-FW ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 500
        if rem > 60:
            t3 = min(rem, 450)
            print(f"  Time: {t3:.0f}s")
            # Run with different constraint counts
            for K in [20, 35]:
                if time.monotonic() < deadline - 250:
                    a_opt, obj = lp_fw_optimize(best_seq, time_limit=min(t3 * 0.5, 200),
                                                max_constraints=K, verbose=True)
                    try_seq(a_opt, f"lpfw_K{K}")

            # LP-FW from perturbed start
            if time.monotonic() < deadline - 200:
                rem3 = deadline - time.monotonic() - 200
                perturbed = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * 0.003)
                a_opt2, obj2 = lp_fw_optimize(perturbed, time_limit=min(rem3, 100),
                                              max_constraints=25, verbose=False)
                try_seq(a_opt2, "lpfw_pert")

    # === Phase 4: Subgradient FW ===
    print(f"\n=== Phase 4: Subgradient FW ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 60:
            t4 = min(rem, 200)
            print(f"  Time: {t4:.0f}s")
            a_opt, obj = subgradient_fw(best_seq, time_limit=t4, verbose=True)
            try_seq(a_opt, "subgrad_fw")

    # === Phase 5: Coordinate Descent ===
    print(f"\n=== Phase 5: Coordinate Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 40:
            t5 = min(rem, 150)
            print(f"  Time: {t5:.0f}s")
            a_opt, obj = coordinate_descent_peaks(best_seq, time_limit=t5, verbose=True)
            try_seq(a_opt, "coord_descent")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-Length ===")
    if best_seq is not None:
        base = best_seq
        for target_n in [1024, 1280, 640, 1536]:
            rem = deadline - time.monotonic() - 80
            if rem < 40:
                break
            f_interp = interpolate_sequence(base, target_n)
            try_seq(f_interp, f"interp_n{target_n}")

            # Quick optimization on interpolated
            t6 = min(rem * 0.4, 60)
            if t6 > 15:
                a_opt, obj = adam_cyclic_optimize(f_interp, time_limit=t6)
                try_seq(a_opt, f"adam_n{target_n}")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t7 = min(rem, 60)
            a_opt, obj = adam_cyclic_optimize(best_seq, time_limit=t7, verbose=False)
            try_seq(a_opt, "final_polish")

        # Final LP-FW squeeze
        rem = deadline - time.monotonic() - 15
        if rem > 15:
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=min(rem, 30),
                                        max_constraints=20, verbose=False)
            try_seq(a_opt, "final_lpfw")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    elapsed = time.monotonic() - (deadline - float(budget_s) + 30.0)
    print(f"Total time: {elapsed:.1f}s")
    return [float(x) for x in best_seq]
