Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based exchange algorithm.

The key insight is that at each iteration, we solve an LP to find the optimal
direction to move: maximize sum(b) subject to b >= 0 and (f*b)[k] <= M for 
active peaks k. This gives the exact best direction, unlike subgradient methods.

Additional improvements:
- Active set management (only include binding + near-binding constraints)

import numpy as np
from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def eval_local(a):
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S * S))

    for k in top_idx:
        # (f*b)[k] = sum_j f[k-j]*b[j] where 0 <= k-j < n

        hi = min(n - 1, k)

        row = np.zeros(n)
        for j in range(lo, hi + 1):
            row[j] = f[k - j]
        constraints.append(row @ b <= M)

                 min_step=1e-4, solver=cp.CLARABEL):

    f: input array (will be normalized internally)
    Returns: (best_f, best_M)
    """

    f = f.copy().astype(np.float64)
    f = np.maximum(f, 0)
    S = f.sum()
    if S < 1e-15:
        return f, float('inf')
    f = f / S
    
    best_f = f.copy()
    best_M = fast_convolve(f).max()
    
    step = step_init
    t0 = time.monotonic()
    
    for it in range(max_iters):
        ff = fast_convolve(f)
        M = ff.max()
        
        if M < best_M - 1e-12:

            best_f = f.copy()

            break
        b = b / Sb

        accepted = False
        for t in [step, step*0.5, step*0.25, step*0.1, step*0.01, min_step]:

            f_new = np.maximum(f_new, 0)
            S_new = f_new.sum()
            if S_new < 1e-15:

            M_new = fast_convolve(f_new).max()

        else:

            step = min(step * 1.05, 1.0)

    M_final = fast_convolve(f).max()
    if M_final < best_M:
        best_M = M_final
        best_f = f.copy()
    
    return best_f, best_M

The LP approach from the state-of-the-art:

<= (1-t)^2 conv(f,f)[k] + 2t(1-t) M + t^2 conv(b,b)[k]

For this to not exceed M, we need:
(1-t)^2 conv(f,f)[k] + 2t(1-t) M + t^2 conv(b,b)[k] <= M

For small t, the LHS ≈ conv(f,f)[k] + 2t(M - conv(f,f)[k]) + t^2(conv(b,b)[k] - 2M + conv(f,f)[k])

The first-order term is 2t(M - conv(f,f)[k]) > 0 for non-peak positions. So for small t, non-peak positions INCREASE. But they start below M, so for sufficiently small t, they stay below M.

The constraint is: for all k, (1-t)^2 c_k + 2t(1-t) d_k + t^2 e_k <= M
where c_k = conv(f,f)[k], d_k = conv(f,b)[k] <= M, e_k = conv(b,b)[k]

The worst case is when c_k is closest to M but not equal. For these, t needs to be small.

"""
LP-based exchange algorithm for minimizing 2n*max(conv(a,a))/(sum(a))^2.

Uses linear programming to find the exact optimal descent direction at each
iteration, rather than subgradient approximations. This gives much faster
convergence and better final solutions.

Algorithm:
1. Normalize f to simplex (sum=1, f>=0)
2. Compute M = max(conv(f,f))
3. Solve LP: max sum(b) s.t. conv(f,b) <= M, b >= 0
4. Move toward b with line search
5. Repeat until convergence or time limit
6. Multiple restarts with diverse initializations
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def eval_local(a):
    """Compute 2n*max(conv)/(sum)^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S * S))


def build_conv_matrix(f):
    """Build convolution matrix A where conv(f, x) = A @ x.
    A has shape (2n-1, n) and A[k,j] = f[k-j] if 0 <= k-j < n else 0."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp(f, M, active_fracs=None, solver=cp.CLARABEL):
    """
    Solve: max sum(b) s.t. conv(f,b)[k] <= M for active k, b >= 0.
    
    active_fracs: if given, only include constraints where conv(f,f)[k] > M * active_fracs
    """
    n = len(f)
    ff = fast_convolve(f)
    
    # Determine active constraints
    if active_fracs is not None:
        threshold = M * active_fracs
        active_mask = ff >= threshold
        # Always include the global max
        active_mask[ff.argmax()] = True
    else:
        active_mask = np.ones(2 * n - 1, dtype=bool)
    
    # Build sub-matrix
    A = build_conv_matrix(f)[active_mask, :]
    
    b = cp.Variable(n, nonneg=True)
    constraints = [A @ b <= M]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=solver)
        if b.value is not None and prob.status in ['optimal', 'optimal_inaccurate']:
            return np.asarray(b.value, dtype=np.float64)
    except Exception as e:
        pass
    
    # Fallback solvers
    for fallback in [cp.SCS, cp.ECOS, cp.GLOP]:
        try:
            prob.solve(solver=fallback)
            if b.value is not None and prob.status in ['optimal', 'optimal_inaccurate']:
                return np.asarray(b.value, dtype=np.float64)
        except Exception:
            continue
    
    return None


def lp_exchange(f0, time_limit, active_fracs=0.95, top_k_fallback=200,
                max_iters=5000, solver=cp.CLARABEL):
    """
    LP-based exchange algorithm.
    
    Parameters:
    - f0: initial point (will be normalized)
    - time_limit: seconds to run
    - active_fracs: include constraints where conv > M * this fraction
    - Returns: (best_f, best_M)
    """
    n = len(f0)
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 0)
    S = f.sum()
    if S < 1e-15:
        return f, float('inf')
    f = f / S
    
    best_f = f.copy()
    best_M = fast_convolve(f).max()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and it < max_iters:
        it += 1
        ff = fast_convolve(f)
        M = ff.max()
        
        if M < best_M - 1e-14:
            best_M = M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Solve LP
        if active_fracs is not None:
            b = solve_lp(f, M, active_fracs=active_fracs, solver=solver)
        else:
            b = solve_lp(f, M, solver=solver)
        
        if b is None:
            break
        
        b = np.maximum(b, 0)
        Sb = b.sum()
        if Sb < 1e-15:
            break
        b = b / Sb
        
        # Line search
        accepted = False
        for t in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            f_new = (1 - t) * f + t * b
            f_new = np.maximum(f_new, 0)
            S_new = f_new.sum()
            if S_new < 1e-15:
                continue
            f_new = f_new / S_new
            M_new = fast_convolve(f_new).max()
            if M_new < M - 1e-14:
                f = f_new
                accepted = True
                break
        
        if not accepted:
            # Try smaller active set (more constraints might be needed)
            if active_fracs is not None and active_fracs > 0.8:
                active_fracs = max(0.5, active_fracs - 0.05)
            elif no_improve > 100:
                break
            
        # Periodic convergence check
        if no_improve > 500:
            break
    
    # Final check
    M_final = fast_convolve(f).max()
    if M_final < best_M:
        best_M = M_final
        best_f = f.copy()
    
    return best_f, best_M


def make_gaussian_init(n, sigma=0.3):

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def coordinate_polish(a0, time_limit, n_multipliers=10):
    """Simple coordinate descent polish."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 0)
    S = a.sum()
    if S < 1e-15:
        return a
    a = a / S
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    multipliers = np.array([0.5, 0.7, 0.85, 1.15, 1.3, 1.5])
    
    for sweep in range(100):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-14:
                continue
            best_j_obj = best_obj
            best_j_val = old_val
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        if not improved:
            break
    
    return best_a, best_obj

"""
LP-based exchange algorithm for minimizing 2n*max(conv(a,a))/(sum(a))^2.

Key innovation: Uses linear programming to find the exact optimal descent
direction at each iteration (the "state of the art" LP from the problem
description). At each step, given current f with peak M, we solve:
  max sum(b) s.t. b >= 0, conv(f,b)[k] <= M for active k
This gives the exact best direction, unlike subgradient approximations.

Additional strategies:
- Adaptive active-set management for LP speed
- Multiple diverse initializations and restarts
- Multi-length exploration
- Coordinate descent final polish
- Fallback subgradient for robustness
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def eval_local(a):
    """Compute 2n*max(conv)/(sum)^2 locally."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S * S))


def build_conv_matrix(f):
    """Build convolution matrix A where conv(f, x) = A @ x.
    A shape (2n-1, n), A[k,j] = f[k-j] if 0 <= k-j < n else 0."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_direction(f, M, active_fracs, solver):
    """
    Solve LP: max sum(b) s.t. conv(f,b)[k] <= M for active k, b >= 0.
    Returns b (normalized to sum=1) or None on failure.
    """
    n = len(f)
    ff = fast_convolve(f)

    if active_fracs is not None:
        threshold = M * active_fracs
        active_mask = ff >= threshold
        active_mask[ff.argmax()] = True
        n_active = active_mask.sum()
        if n_active > 2 * n - 1:
            active_mask = np.ones(2 * n - 1, dtype=bool)
    else:
        active_mask = np.ones(2 * n - 1, dtype=bool)

    A_full = build_conv_matrix(f)
    A = A_full[active_mask, :]

    b = cp.Variable(n, nonneg=True)
    constraints = [A @ b <= M]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)

    result = None
    solvers_to_try = [solver, cp.SCS, cp.CLARABEL]
    for s in solvers_to_try:
        try:
            prob.solve(solver=s)
            if b.value is not None and prob.status in ['optimal', 'optimal_inaccurate']:
                result = np.asarray(b.value, dtype=np.float64)
                break
        except Exception:
            continue

    if result is None:
        return None

    result = np.maximum(result, 0)
    S = result.sum()
    if S < 1e-15:
        return None
    return result / S


def lp_exchange(f0, time_limit, active_fracs=0.95, solver=cp.CLARABEL,
                min_step=1e-6, max_iters=10000):
    """
    LP-based exchange algorithm to minimize max(conv(f,f)) on the simplex.
    Returns (best_f, best_M).
    """
    n = len(f0)
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 0)
    S = f.sum()
    if S < 1e-15:
        return f, float('inf')
    f = f / S

    best_f = f.copy()
    best_M = fast_convolve(f).max()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    cur_af = active_fracs

    while time.monotonic() - t0 < time_limit and it < max_iters:
        it += 1
        ff = fast_convolve(f)
        M = ff.max()

        if M < best_M - 1e-14:
            best_M = M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Solve LP for direction
        b = solve_lp_direction(f, M, cur_af, solver)
        if b is None:
            # Try with looser active set
            if cur_af is not None and cur_af < 0.99:
                cur_af = min(0.99, cur_af + 0.05)
                continue
            break

        # Line search: try decreasing step sizes
        accepted = False
        for t in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05, 0.01]:
            f_new = (1.0 - t) * f + t * b
            f_new = np.maximum(f_new, 0)
            S_new = f_new.sum()
            if S_new < 1e-15:
                continue
            f_new = f_new / S_new
            M_new = fast_convolve(f_new).max()
            if M_new < M - 1e-14:
                f = f_new
                accepted = True
                break

        if not accepted:
            # Tighten active set (include more constraints)
            if cur_af is not None and cur_af > 0.5:
                cur_af = max(0.5, cur_af - 0.05)
            if no_improve > 200:
                break

        if no_improve > 500:
            break

    M_final = fast_convolve(f).max()
    if M_final < best_M:
        best_M = M_final
        best_f = f.copy()

    return best_f, best_M


def subgradient_refine(f0, time_limit, lr=0.002, momentum=0.9, seed=0):
    """Fallback subgradient method for fine-tuning."""
    rng = np.random.default_rng(seed)
    n = len(f0)
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 0)
    f = f / f.sum()

    best_f = f.copy()
    best_M = fast_convolve(f).max()
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        ff = fast_convolve(f)
        M = ff.max()
        if M < best_M - 1e-14:
            best_M = M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Peak-based subgradient
        peak_idx = np.argmax(ff)
        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)
        g = np.zeros(n)
        g[lo:hi+1] = f[peak_idx - hi:peak_idx - lo + 1][::-1]

        v = momentum * v - lr * g
        f_new = project_simplex(f + v)
        f = f_new

        if no_improve > 3000:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            v = np.zeros(n)
            no_improve = 0
        if no_improve > 10000:
            break

    return best_f, best_M

def make_banded_init(n, band_frac=0.3, seed=0):
    """Initialize with mass concentrated in a band."""
    rng = np.random.default_rng(seed)
    a = np.zeros(n)
    center = n // 2

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
