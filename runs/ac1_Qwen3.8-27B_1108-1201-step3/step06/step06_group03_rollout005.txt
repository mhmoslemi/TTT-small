"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Nesterov-accelerated mirror descent,
exact coordinate descent with golden-section line search, LP-Frank-Wolfe refinement,
and multi-length exploration. Key improvement over previous: exact 1D coordinate

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_value(f):
    """Compute 2*n*max(conv(f)) for f normalized to sum=1."""
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(f, k_peak):
    """Compute gradient of b[k_peak] with respect to f.
    ∂b[k]/∂f[j] = 2*f[k-j] if 0 <= k-j < n, else 0."""
    n = len(f)
    grad = np.zeros(n)
    js = max(0, k_peak - n + 1)
    je = min(n - 1, k_peak)
    if je >= js:
        idx = np.arange(js, je + 1)
        grad[js:je + 1] = 2.0 * f[k_peak - idx]
    return grad


def subgradient_multi_peak(f, top_k=5, smooth_temp=0.0):
    """Compute weighted subgradient using top peaks with soft weights."""
    n = len(f)
    conv = fast_convolve(f)
    top_idx = np.argsort(conv)[-top_k:]
    top_vals = conv[top_idx]
    
    if smooth_temp > 0:
        scaled = (top_vals - top_vals.max()) / smooth_temp
        w = np.exp(scaled)
        w /= w.sum()
    else:
        # Use top-3 with decay
        w = np.zeros(len(top_idx))
        order = np.argsort(top_vals)[::-1]
        for i, rank in enumerate(order):
            w[rank] = 1.0 / (i + 1)
        w /= w.sum()
    
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        if w[i] < 1e-12:
            continue
        grad += w[i] * compute_gradient(f, int(k))
    return grad, conv.max()


def nesterov_mirror_descent(f0, time_limit, lr_init=0.05, momentum=0.9):
    """Nesterov-accelerated mirror descent on the simplex."""
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    y = np.zeros(n)  # dual variable
    t = 1
    t0 = time.monotonic()
    momentum = momentum
    
    while time.monotonic() - t0 < time_limit:
        # Nesterov step: evaluate gradient at extrapolated point
        f_y = np.exp(-y / (lr_init * t))
        f_y = np.maximum(f_y, 1e-15)
        f_y = f_y / f_y.sum()
        
        conv_working = fast_convolve(f_y)
        M = conv_working.max()
        
        # Subgradient at extrapolated point
        top_k = max(5, int(len(conv_working) * 0.1))
        peaks = np.argpartition(conv_working, -top_k)[-top_k:]
        pv = conv_working[peaks]
        scaled = pv / max(M * 0.01, 1e-10)
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je >= js:
                idx = np.arange(js, je + 1)
                sg[js:je + 1] += w[i] * 2.0 * f_y[k - idx]
        
        # Update dual
        y = y + lr_init * t * sg
        
        # Compute new primal
        f_new = np.exp(-y / (lr_init * (t + 1)))
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()
        
        # Check objective
        obj_new = fast_convolve(f_new).max()
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
        
        t += 1
        # Decay momentum / learning rate
        if t % 100 == 0 and lr_init > 0.001:
            lr_init *= 0.98
    
    return best_f, best_obj


def coordinate_descent_exact(f0, time_limit, verbose=False):
    """
    Exact coordinate descent: for each coordinate, find optimal value
    via golden-section search on the 1D objective.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    pass_count = 0
    no_improve_passes = 0
    
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        pass_improvement = 0.0
        rng = np.random.default_rng(pass_count * 1000 + 42)
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj
            
            old_val = f[j]
            
            # Define 1D objective: change a[j] to new_val, rescale rest
            def obj_1d(delta):
                new_val = old_val + delta
                if new_val < -1e-15 or new_val > 1.0 + 1e-15:
                    return 1e10
                new_val = max(0.0, new_val)
                denom = 1.0 - old_val
                if denom < 1e-12:
                    return 1e10
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    return 1e10
                f_trial = f_trial / ss
                return fast_convolve(f_trial).max()
            
            # Golden section search on delta in [-old_val, 1 - old_val]
            lo = -old_val * 0.999
            hi = (1.0 - old_val) * 0.999
            
            # First do a coarse scan
            n_scan = 8
            best_delta = 0.0
            best_val_1d = obj_1d(0.0)
            if hi > lo:
                for s_i in range(n_scan + 1):
                    d = lo + (hi - lo) * s_i / n_scan
                    v = obj_1d(d)
                    if v < best_val_1d:
                        best_val_1d = v
                        best_delta = d
                
                # Refine with golden section around best
                if best_delta > lo and best_delta < hi:
                    span = (hi - lo) / n_scan
                    lo_gs = max(lo, best_delta - span)
                    hi_gs = min(hi, best_delta + span)
                    invphi = (np.sqrt(5.0) - 1.0) / 2.0
                    invphi2 = invphi * invphi
                    c = hi_gs - invphi * (hi_gs - lo_gs)
                    d_pt = lo_gs + invphi * (hi_gs - lo_gs)
                    fc = obj_1d(c)
                    fd = obj_1d(d_pt)
                    for _ in range(12):
                        if fc < fd:
                            hi_gs = d_pt
                            d_pt = c
                            fd = fc
                            c = hi_gs - invphi * (hi_gs - lo_gs)
                            fc = obj_1d(c)
                        else:
                            lo_gs = c
                            c = d_pt
                            fc = fd
                            d_pt = lo_gs + invphi * (hi_gs - lo_gs)
                            fd = obj_1d(d_pt)
                    best_delta = (lo_gs + hi_gs) / 2.0
                    best_val_1d = min(fc, fd)
            
            if best_val_1d < best_obj - 1e-14:
                # Apply the update
                new_val = max(0.0, old_val + best_delta)
                denom = 1.0 - old_val
                if denom > 1e-12:
                    scale = (1.0 - new_val) / denom
                    f[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                
                pass_improvement += (best_obj - best_val_1d)
                best_obj = fast_convolve(f).max()
                best_f = f.copy()
        
        if verbose and pass_count % 10 == 0:
            print(f"      CD pass {pass_count}: obj={best_obj:.10f}, improvement={pass_improvement:.2e}")
        
        if pass_improvement < 1e-12:
            no_improve_passes += 1
            if no_improve_passes > 3:
                break
        else:
            no_improve_passes = 0
    
    return best_f, best_obj


def lp_fw_step(f, M, n):
    """Single LP-Frank-Wolfe step: solve LP and do line search."""
    # LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None, None
    g = res.x
    gs = g.sum()
    if gs < 1e-15:
        return None, None
    g = g / gs
    
    # Line search
    best_alpha = 0.0
    best_val = fast_convolve(f).max()
    for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        ft = f + alpha * (g - f)
        ft = np.maximum(ft, 0)
        st = ft.sum()
        if st < 1e-15:
            continue
        ft = ft / st
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha
    
    if best_alpha > 1e-10:
        # Refine
        lo = max(0.0, best_alpha - 0.1)
        hi = min(1.0, best_alpha + 0.1)
        for _ in range(8):
            mid = (lo + hi) / 2.0
            ft = f + mid * (g - f)
            ft = np.maximum(ft, 0)
            st = ft.sum()
            if st < 1e-15:
                break
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = mid
                lo = mid
            else:
                hi = mid
        f_new = f + best_alpha * (g - f)
        f_new = np.maximum(f_new, 0)
        f_new = f_new / f_new.sum()
        return f_new, fast_convolve(f_new).max()
    
    return None, None


def block_coordinate_descent(f0, time_limit, block_size=4, verbose=False):
    """Block coordinate descent: optimize a small block of coordinates jointly via Nelder-Mead."""
    from scipy.optimize import minimize
    
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Pick a random block
        rng = np.random.default_rng(it * 777 + 13)
        start = rng.integers(0, max(1, n - block_size))
        indices = list(range(start, start + block_size))
        
        # Current values
        x0 = f[indices].copy()
        rest_sum = 1.0 - x0.sum()
        if rest_sum < 1e-12:
            continue
        
        def obj_block(deltas):
            new_vals = x0 + deltas
            if np.any(new_vals < -1e-12):
                return 1e10
            new_vals = np.maximum(new_vals, 0)
            new_rest_sum = 1.0 - new_vals.sum()
            if new_rest_sum < -1e-12:
                return 1e10
            if new_rest_sum < 1e-12:
                # All mass in the block
                f_trial = np.zeros(n)
                f_trial[indices] = new_vals / new_vals.sum()
            else:
                f_trial = f.copy()
                f_trial[indices] = new_vals
                mask = np.ones(n, dtype=bool)
                mask[indices] = False
                f_trial[mask] *= new_rest_sum / rest_sum
                f_trial = np.maximum(f_trial, 0)
                f_trial = f_trial / f_trial.sum()
            return fast_convolve(f_trial).max()
        
        try:
            res = minimize(obj_block, np.zeros(block_size), method='Nelder-Mead',
                          options={'maxiter': 200, 'xatol': 1e-10, 'fatol': 1e-12})
            if res.success and res.fun < best_obj - 1e-13:
                new_vals = np.maximum(x0 + res.x, 0)
                new_rest_sum = 1.0 - new_vals.sum()
                if new_rest_sum > 1e-12:
                    f[indices] = new_vals
                    mask = np.ones(n, dtype=bool)
                    mask[indices] = False
                    f[mask] *= new_rest_sum / rest_sum
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                    best_obj = fast_convolve(f).max()
                    best_f = f.copy()
                elif new_vals.sum() > 1e-12:
                    f = np.zeros(n)
                    f[indices] = new_vals / new_vals.sum()
                    best_obj = fast_convolve(f).max()
                    best_f = f.copy()
        except Exception:
            pass
        
        if verbose and it % 20 == 0:
            print(f"      BCD iter {it}: obj={best_obj:.10f}")
    
    return best_f, best_obj


def smoothed_mirror_descent(f0, time_limit, verbose=False,
                            tau_init=5.0, lr_init=0.02,
                            tau_decay=1.005, lr_decay=0.9997,
                            restart_every=150):
    """Adaptive smoothed mirror descent with multiplicative updates."""
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    tau = tau_init
    lr = lr_init
    no_improve = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:

        M = conv.max()

        pv = conv[peaks]
        scaled = tau * pv
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je >= js:
                idx = np.arange(js, je + 1)
                sg[js:je + 1] += w[i] * 2.0 * f[k - idx]
        
        # Multiplicative update
        f_new = f * np.exp(-lr * sg)
        f_new = np.maximum(f_new, 1e-15)
        f_new = f_new / f_new.sum()
        
        # Line search
        obj_new = fast_convolve(f_new).max()
        if obj_new > M * 1.02:
            found = False
            for damp in [0.3, 0.15, 0.08, 0.04]:
                f_trial = f * (1.0 - damp) + f_new * damp
                f_trial = np.maximum(f_trial, 1e-15)
                f_trial = f_trial / f_trial.sum()
                obj_trial = fast_convolve(f_trial).max()
                if obj_trial < M:
                    f_new = f_trial
                    obj_new = obj_trial
                    found = True
                    break
            if not found:
                lr *= 0.5
                continue
        
        f = f_new
        
        if obj_new < best_obj - 1e-14:

    return best_f, best_obj


def make_init(n, variant, seed=42):

        a = np.linspace(0.1, 1.0, n)
    elif variant == 4:
        a = np.exp(-50 * (x - 0.3) ** 2) + 0.5 * np.exp(-50 * (x - 0.7) ** 2)

        a += rng.standard_normal(n) * 0.1
        a = np.maximum(a, 0)
    elif variant == 7:
        a = np.log1p(np.arange(n) + 1)
    elif variant == 8:

    a = np.maximum(a, 0)
    s = a.sum()

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def trim_sequence(f, threshold=1e-6):
    """Remove leading and trailing values below threshold."""
    f = np.asarray(f, dtype=np.float64)
    mask = f > threshold
    if not mask.any():
        return f
    first = np.argmax(mask)
    last = len(f) - np.argmax(mask[::-1])
    trimmed = f[first:last]
    s = trimmed.sum()
    if s > 1e-15:
        trimmed = trimmed / s
    return trimmed

    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        # Normalize for evaluation (score is scale-invariant)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score
    
    # Try parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass
    
    # If no parent, create initial candidates
    if parent is None:
        n_init = 768
        a_init = make_init(n_init, 0, seed)
        best_seq = a_init
        best_score = float(evaluate_sequence(a_init.tolist()))
        print(f"Initial: n={len(a_init)}, score={best_score:.6f}")
    
    base_n = len(best_seq)
    
    # === Phase 1: Quick SMD to improve initial point ===
    print("\n=== Phase 1: SMD ===")
    rem = deadline - time.monotonic() - 800
    if rem > 30:
        t_alloc = min(rem, 120)
        print(f"  SMD (n={base_n}): t={t_alloc:.0f}s")
        a_opt, obj = smoothed_mirror_descent(best_seq, time_limit=t_alloc, verbose=True,
                                             tau_init=3.0, lr_init=0.015,
                                             tau_decay=1.004, lr_decay=0.9998)
        sc = try_seq(a_opt, "smd_p1")
        if sc < best_score:
            best_seq = a_opt
    
    # === Phase 2: Coordinate Descent (main workhorse) ===
    print("\n=== Phase 2: Coordinate Descent ===")
    rem = deadline - time.monotonic() - 400
    if rem > 30:
        t_alloc = min(rem, 500)
        print(f"  CD exact (n={base_n}): t={t_alloc:.0f}s")
        a_opt, obj = coordinate_descent_exact(best_seq, time_limit=t_alloc, verbose=True)
        sc = try_seq(a_opt, "cd_exact")
        if sc < best_score:
            best_seq = a_opt
    
    # === Phase 3: SMD again after CD ===
    print("\n=== Phase 3: SMD Refinement ===")
    rem = deadline - time.monotonic() - 300
    if rem > 20:
        t_alloc = min(rem, 100)
        print(f"  SMD (n={len(best_seq)}): t={t_alloc:.0f}s")
        a_opt, obj = smoothed_mirror_descent(best_seq, time_limit=t_alloc, verbose=True,
                                             tau_init=10.0, lr_init=0.005,
                                             tau_decay=1.002, lr_decay=0.9999,
                                             restart_every=100)
        sc = try_seq(a_opt, "smd_p3")
        if sc < best_score:
            best_seq = a_opt
    
    # === Phase 4: LP-Frank-Wolfe ===
    print("\n=== Phase 4: LP-FW ===")
    rem = deadline - time.monotonic() - 250
    if rem > 60 and len(best_seq) <= 1200:
        t_alloc = min(rem, 150)
        print(f"  LP-FW (n={len(best_seq)}): t={t_alloc:.0f}s")
        f = best_seq / best_seq.sum()
        n = len(f)
        M = fast_convolve(f).max()
        best_f_fw = f.copy()
        best_obj_fw = M
        t0 = time.monotonic()
        no_imp = 0
        it = 0
        while time.monotonic() - t0 < t_alloc:
            it += 1
            g, M_new = lp_fw_step(best_f_fw, M, n)
            if g is not None:
                best_f_fw = g
                M = M_new
                if M < best_obj_fw - 1e-14:
                    best_obj_fw = M
                    no_imp = 0
                else:
                    no_imp += 1
                if no_imp > 30:
                    break
            else:
                no_imp += 1
                if no_imp > 10:
                    # Try relaxed M
                    g, M_new = lp_fw_step(best_f_fw, M * 1.5, n)
                    if g is not None:
                        # Line search from current
                        best_a = 0
                        best_v = M
                        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                            ft = best_f_fw + alpha * (g - best_f_fw)
                            ft = np.maximum(ft, 0)
                            ft = ft / ft.sum()
                            v = fast_convolve(ft).max()
                            if v < best_v:
                                best_v = v
                                best_a = alpha
                        if best_a > 0:
                            best_f_fw = best_f_fw + best_a * (g - best_f_fw)
                            best_f_fw = np.maximum(best_f_fw, 0)
                            best_f_fw = best_f_fw / best_f_fw.sum()
                            M = fast_convolve(best_f_fw).max()
                            if M < best_obj_fw - 1e-14:
                                best_obj_fw = M
                                no_imp = 0
                    no_imp += 1
                    if no_imp > 40:
                        break
        
        try_seq(best_f_fw, "lpfw")
        if best_score < float(evaluate_sequence(best_seq.tolist())):
            best_seq = best_f_fw
    
    # === Phase 5: Coordinate Descent again ===
    print("\n=== Phase 5: CD Polish ===")
    rem = deadline - time.monotonic() - 200
    if rem > 20:
        t_alloc = min(rem, 200)
        print(f"  CD polish (n={len(best_seq)}): t={t_alloc:.0f}s")
        a_opt, obj = coordinate_descent_exact(best_seq, time_limit=t_alloc, verbose=True)
        sc = try_seq(a_opt, "cd_polish")
        if sc < best_score:
            best_seq = a_opt
    
    # === Phase 6: Block Coordinate Descent ===
    print("\n=== Phase 6: Block CD ===")
    rem = deadline - time.monotonic() - 150
    if rem > 20 and len(best_seq) <= 1000:
        t_alloc = min(rem, 100)
        print(f"  BCD (n={len(best_seq)}): t={t_alloc:.0f}s")
        a_opt, obj = block_coordinate_descent(best_seq, time_limit=t_alloc,
                                             block_size=3, verbose=True)
        sc = try_seq(a_opt, "bcd")
        if sc < best_score:
            best_seq = a_opt
    
    # === Phase 7: Try trimming ===
    print("\n=== Phase 7: Trimming ===")
    for thresh in [1e-5, 1e-4, 1e-3, 5e-3]:
        trimmed = trim_sequence(best_seq, threshold=thresh)
        if len(trimmed) < len(best_seq):
            sc = try_seq(trimmed, f"trim_{thresh}")
    
    # === Phase 8: Try different lengths via interpolation ===
    print("\n=== Phase 8: Multi-Length ===")
    rem = deadline - time.monotonic() - 100
    if rem > 30:
        for target_n in [len(best_seq) - 32, len(best_seq) + 32, len(best_seq) + 64]:
            if target_n <= 0 or target_n > 2000:
                continue
            if rem < 15:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            sc = try_seq(f_interp, f"interp_n{target_n}")
            if sc < best_score + 0.01:
                # Quick CD on the interpolated
                t_alloc = min(rem * 0.3, 40)
                a_opt, obj = coordinate_descent_exact(f_interp, time_limit=t_alloc)
                sc = try_seq(a_opt, f"cd_interp_n{target_n}")
                if sc < best_score:
                    best_seq = a_opt
    
    # === Phase 9: Final CD squeeze ===
    print("\n=== Phase 9: Final Squeeze ===")
    rem = deadline - time.monotonic() - 10
    if rem > 10:
        t_alloc = min(rem, 60)
        print(f"  Final CD (n={len(best_seq)}): t={t_alloc:.0f}s")
        a_opt, obj = coordinate_descent_exact(best_seq, time_limit=t_alloc)
        sc = try_seq(a_opt, "final_cd")
        if sc < best_score:
            best_seq = a_opt
    
    # === Phase 10: One more SMD with very low lr ===
    rem = deadline - time.monotonic() - 5
    if rem > 10:
        t_alloc = min(rem, 30)
        a_opt, obj = smoothed_mirror_descent(best_seq, time_limit=t_alloc,
                                             tau_init=20.0, lr_init=0.002,
                                             tau_decay=1.001, lr_decay=0.99999,
                                             restart_every=50)
        sc = try_seq(a_opt, "final_smd")
        if sc < best_score:
            best_seq = a_opt
    
    # Final verification
    if best_seq is None:
        best_seq = np.ones(512) / 512
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
