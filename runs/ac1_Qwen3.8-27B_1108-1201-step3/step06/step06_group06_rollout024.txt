def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_grad_and_conv(f, tau):
    """Compute gradient of log(sum_k exp(conv[k]/tau)) and the convolution."""
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()

    # grad_j = (2/tau) * sum_k w[k] * a[k-j] (with bounds)
    # This equals (2/tau) * (a * w_rev)[n-1:2n-1] reversed
    w_rev = w[::-1]
    total_len = n + len(w) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(f, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv_result = np.fft.irfft(A * W, fft_size)
    h = conv_result[n - 1:2 * n - 1][::-1]

    grad = (2.0 / tau) * h
    return grad, conv


def subgrad_at_max(a, k_max=None):
    """Compute subgradient of max_k (a*a)[k] at position k_max."""
    n = len(a)
    if k_max is None:
        conv = fast_convolve(a)
        k_max = int(np.argmax(conv))
    else:
        conv = None

    j_min = max(0, k_max - n + 1)
    j_max = min(n - 1, k_max)

    sg = np.zeros(n)
    if j_max >= j_min:
        vals = a[k_max - j_max: k_max - j_min + 1]
        indices = np.arange(j_min, j_max + 1)
        sg[indices] = 2.0 * vals
        if k_max % 2 == 0:
            j_diag = k_max // 2
            if j_min <= j_diag <= j_max:
                sg[j_diag] = a[j_diag]

    if conv is None:
        conv = fast_convolve(a)
    return sg, conv


def adam_nesterov_optimize(f0, time_limit, tau_init=None, lr_init=0.005,
                           use_nesterov=True, verbose=False):
    """Adam with optional Nesterov momentum on the smoothed objective."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    if tau_init is None:
        tau = 0.1 * conv.max()
    else:
        tau = max(tau_init, 1e-8)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    t = 0

    best_obj = conv.max()
    best_f = f.copy()
    no_improve = 0
    t0 = time.monotonic()

    # Nesterov state
    f_prev = f.copy()
    beta_nes = 0.0

    while time.monotonic() - t0 < time_limit:
        t += 1

        # Nesterov extrapolation
        if use_nesterov and t > 1:
            beta_nes = (1.0 + np.sqrt(1.0 + 4.0 * beta_nes ** 2)) / 2.0
            f_nes = (1.0 - 1.0 / beta_nes) * f + (1.0 / beta_nes) * f_prev
            f_nes = np.maximum(f_nes, 0.0)
            f_nes_sum = f_nes.sum()
            if f_nes_sum < 1e-15:
                f_nes = f
            else:
                f_nes = f_nes / f_nes_sum
        else:
            f_nes = f

        grad, conv = compute_smooth_grad_and_conv(f_nes, tau)
        M = conv.max()

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)

        f_prev = f
        f = f_new

        conv_new = fast_convolve(f)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive temperature
        tau *= 0.9995
        if tau < 1e-7:
            tau = 1e-7

        # Adaptive learning rate
        if no_improve > 80:
            lr *= 0.9
        if no_improve > 300:
            rng = np.random.default_rng(t * 7 + 3)
            noise_scale = 0.003 + 0.004 * min(no_improve / 500.0, 2.0)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            m = np.zeros(n)
            v = np.zeros(n)
            f_prev = f.copy()
            no_improve = 0
            lr = lr_init * 0.8
            tau = 0.06 * fast_convolve(f).max()

        if verbose and t % 200 == 0:
            print(f"    AdamNest iter {t}: obj={2*n*best_obj:.8f}, tau={tau:.2e}, lr={lr:.5f}")

    return best_f, 2 * n * best_obj


def nesterov_subgrad_optimize(f0, time_limit, verbose=False):
    """
    Nesterov-accelerated subgradient method on the actual max-convolution.
    Uses the true (non-smooth) subgradient for exact tracking.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)

    else:
        f = f / s

    best_f = f.copy()
    t0 = time.monotonic()
    t = 0

    # Estimate Lipschitz constant
    L = 2.0 * n * np.max(f) * 2.0
    alpha = 0.5
    step = alpha / (L + 1e-8)

    f_prev = f.copy()
    t0_nes = 10.0

    while time.monotonic() - t0 < time_limit:
        t += 1

        # Nesterov extrapolation
        beta_nes = (1.0 + np.sqrt(1.0 + 4.0 * (t / t0_nes) ** 2)) / 2.0
        f_nes = (1.0 - 1.0 / beta_nes) * f + (1.0 / beta_nes) * f_prev
        f_nes = np.maximum(f_nes, 0.0)
        sn = f_nes.sum()
        if sn < 1e-15:
            f_nes = f
        else:
            f_nes = f_nes / sn

        # Subgradient at f_nes
        conv_nes = fast_convolve(f_nes)
        k_max = int(np.argmax(conv_nes))
        sg, _ = subgrad_at_max(f_nes, k_max)

        # Step size decay
        step_t = alpha / (L + 1e-8) * 0.5

        f_prev = f
        f_new = f - step_t * sg
        f_new = project_simplex(f_new)

        # Evaluate
        conv_new = fast_convolve(f_new)
        M = conv_new.max()

        f = f_new

        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()

        # Also try a few random peak positions for diversity

            perm = np.random.default_rng(t).permutation(len(conv_nes))
            for pk in perm[:3]:

                trial = project_simplex(f - step_t * sg2)
                M2 = fast_convolve(trial).max()

                    best_obj = M2
                    best_f = trial.copy()

        if verbose and t % 100 == 0:

def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive relaxation and line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    M = conv.max()
    best_obj = M
    best_f = f.copy()

    it = 0
    m_exp = 1.0
    consecutive_fail = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Solve LP: max sum(g) s.t. (f*g)[k] <= M*m_exp for all k, g >= 0
        g = _solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp = min(m_exp * 1.05, 3.0)
            consecutive_fail += 1
            if consecutive_fail > 30:
                rng = np.random.default_rng(it * 13 + 5)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                consecutive_fail = 0
            continue

        g_sum = g.sum()
        if g_sum < 1e-12:
            m_exp = min(m_exp * 1.1, 3.0)
            continue

        g = g / g_sum
        consecutive_fail = 0

        if np.max(np.abs(g - f)) < 1e-10:
            m_exp = min(m_exp * 1.02, 2.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M

        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = np.maximum(f + alpha * (g - f), 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            # Fine bisection
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = np.maximum(f + mid * (g - f), 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-8:
            f_new = np.maximum(f + best_alpha * (g - f), 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new
            m_exp = max(1.0, m_exp * 0.95)

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
        else:
            m_exp = min(m_exp * 1.05, 3.0)

        if verbose and it % 20 == 0:
            print(f"    LP-FW iter {it}: obj={2*n*best_obj:.8f}, m_exp={m_exp:.2f}")

    return best_f, 2 * n * best_obj


def _solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def minimax_mw_optimize(f0, time_limit, verbose=False):
    """Minimax with multiplicative weights for outer and projected gradient for inner."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    p = np.ones(2 * n - 1) / (2 * n - 1)
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    lr_p = 0.2
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Inner: minimize a^T Q(p) a via projected gradient
        a_inner = f.copy()
        inner_iters = min(40, max(15, 3000 // (it + 1)))
        # Lipchitz estimate for inner QP
        p_rev = p[::-1]
        total_len = n + len(p) - 1
        fft_size = 1 << (total_len - 1).bit_length()

        for _ in range(inner_iters):
            A = np.fft.rfft(a_inner, fft_size)
            P = np.fft.rfft(p_rev, fft_size)
            conv_r = np.fft.irfft(A * P, fft_size)
            grad = 2.0 * conv_r[n - 1:2 * n - 1][::-1]
            inner_lr = 2.0 / (n * 1.2)
            a_inner = a_inner - inner_lr * grad
            a_inner = project_simplex(a_inner)

        f = a_inner
        conv = fast_convolve(f)
        M = conv.max()

        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            if verbose and it % 30 == 0:
                print(f"    Minimax iter {it}: obj={2*n*best_obj:.8f}")

        # Outer: multiplicative weight update
        scaled = conv / (0.03 * M + 1e-15)
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        p = (1 - lr_p) * p + lr_p * w
        p = p / p.sum()

        if it % 80 == 0:
            lr_p = max(0.05, lr_p * 0.92)

    return best_f, 2 * n * best_obj


def peak_redistribution_cd(f0, time_limit, verbose=False):
    """
    Peak-targeted coordinate descent: identifies peak-contributing pairs
    and redistributes mass to valley positions.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()
        improved = False

        # Get top peaks
        top_k = min(8, 2 * n - 1)
        peak_indices = np.argsort(conv)[-top_k:]

        # Find valley positions (lowest convolution values)
        valley_indices = np.argsort(conv)[:min(10, 2 * n - 1)]
        valley_positions = set(valley_indices.tolist())

        for pk in peak_indices:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            if je < js:
                continue

            idx = np.arange(js, je + 1)
            products = f[idx] * f[pk - idx]
            j_max = idx[np.argmax(products)]
            j_other = pk - j_max

            if j_max == j_other:
                old_val = f[j_max]
                if old_val < 1e-10:
                    continue
                for factor in [0.3, 0.5, 0.7]:
                    trial = f.copy()
                    trial[j_max] *= factor
                    diff = old_val - trial[j_max]
                    # Redistribute to valley-contributing elements
                    target_mask = np.zeros(n, dtype=bool)
                    for vp in valley_positions:
                        vp = int(vp)
                        vs = max(0, vp - n + 1)
                        ve = min(n - 1, vp)
                        for vi in range(vs, ve + 1):
                            if vi != j_max:
                                target_mask[vi] = True
                    if target_mask.sum() > 0 and f[target_mask].sum() > 1e-10:
                        weights = f[target_mask] / f[target_mask].sum()
                        trial[target_mask] += diff * weights
                    trial = np.maximum(trial, 0)
                    ss = trial.sum()
                    if ss > 1e-10:
                        trial /= ss
                    obj = fast_convolve(trial).max()
                    if obj < best_obj - 1e-12:
                        f = trial
                        best_obj = obj
                        best_f = f.copy()
                        improved = True
                        break
            else:
                old_jm = f[j_max]
                for scale in [0.4, 0.6, 0.8]:
                    new_jm = old_jm * scale
                    diff = old_jm - new_jm
                    trial = f.copy()
                    trial[j_max] = new_jm
                    # Add to positions that contribute to valleys
                    target_mask = np.zeros(n, dtype=bool)
                    target_mask[j_max] = False
                    target_mask[j_other] = False
                    for vp in valley_positions[:5]:
                        vp = int(vp)
                        vs = max(0, vp - n + 1)
                        ve = min(n - 1, vp)
                        target_mask[vs:ve+1] = True
                    if target_mask.sum() > 0 and f[target_mask].sum() > 1e-10:
                        weights = f[target_mask] / f[target_mask].sum()
                        trial[target_mask] += diff * weights
                    trial = np.maximum(trial, 0)
                    ss = trial.sum()
                    if ss > 1e-10:
                        trial /= ss
                    obj = fast_convolve(trial).max()
                    if obj < best_obj - 1e-12:
                        f = trial
                        best_obj = obj
                        best_f = f.copy()
                        improved = True
                        break

        if not improved:
            no_improve += 1
            if no_improve > 10:
                rng = np.random.default_rng(int(time.monotonic() * 10000) % 100000)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                no_improve = 0
        else:
            no_improve = 0

        if verbose and it % 50 == 0:
            print(f"    PeakCD: obj={2*n*best_obj:.8f}")

    return best_f, 2 * n * best_obj


def two_scale_optimize(f0, time_limit, block_size=8, verbose=False):
    """
    Two-scale optimization: optimize a coarse (block-averaged) version,
    then use it to guide fine-scale optimization.
    """
    n = len(f0)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()

    # Create coarse version by averaging blocks
    n_blocks = n // block_size
    if n_blocks < 8:
        block_size = max(1, n // 64)
        n_blocks = n // block_size
    if n_blocks < 8:
        return best_f, 2 * n * best_obj

    pad = n_blocks * block_size - n
    f_padded = np.concatenate([f, np.zeros(pad)])
    f_blocks = f_padded[:n_blocks * block_size].reshape(n_blocks, block_size).mean(axis=1)
    f_coarse = f_blocks / f_blocks.sum()

    # Optimize coarse version
    t_coarse = time_limit * 0.3
    coarse_opt, _ = adam_nesterov_optimize(f_coarse, time_limit=t_coarse, lr_init=0.01)

    # Upsample coarse to fine
    f_upsampled = np.repeat(coarse_opt, block_size)
    if len(f_upsampled) > n:
        f_upsampled = f_upsampled[:n]
    f_upsampled = np.maximum(f_upsampled, 0)
    su = f_upsampled.sum()
    if su > 1e-15:
        f_upsampled /= su

    # Blend with current best
    f_blended = 0.5 * best_f + 0.5 * f_upsampled
    f_blended = project_simplex(f_blended)

    conv_bl = fast_convolve(f_blended)
    if conv_bl.max() < best_obj:
        best_obj = conv_bl.max()
        best_f = f_blended.copy()

    # Fine-scale optimization starting from blend
    t_fine = time_limit - (time.monotonic() - t0) - 5
    if t_fine > 10:
        fine_opt, _ = adam_nesterov_optimize(best_f, time_limit=t_fine, lr_init=0.003)
        conv_fo = fast_convolve(fine_opt)
        if conv_fo.max() < best_obj:
            best_obj = conv_fo.max()
            best_f = fine_opt.copy()
            if verbose:
                print(f"    TwoScale: obj={2*n*best_obj:.8f}")

    return best_f, 2 * n * best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, center_frac=0.4, sigma_frac=0.25):
    """Gaussian initialization."""
    x = np.arange(n) - n * center_frac
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_symmetric_gaussian(n, sigma_frac=0.2):
    """Symmetric Gaussian."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_two_hump(n, separation=0.3):
    """Two-humped initialization."""
    a = make_gaussian(n, 0.5 - separation, 0.12) + make_gaussian(n, 0.5 + separation, 0.12)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_exponential(n, rate=0.5):
    """Exponential decay."""
    a = np.exp(-rate * np.arange(n) / n)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization with Nesterov acceleration, two-scale refinement,
    LP-FW, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
        return score

    # === Load parent ===
    print("=== Loading parent ===")
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_gaussian(768, 0.4, 0.25)
        try_seq(parent, "default_init")

    # === Phase 1: Multi-length exploration (120s) ===
    print("\n=== Phase 1: Multi-Length Exploration ===")
    phase1_budget = 120

    # Try interpolating parent to different lengths
    target_lengths = [512, 768, 1024, 1536]
    if parent is not None:
        for tl in target_lengths:
            if time.monotonic() > deadline - (budget_s - 30 - phase1_budget):
                break
            interp = interpolate_sequence(parent, tl)
            try_seq(interp, f"interp_n{tl}")

    # Try diverse shapes at key lengths
    for n_try in [768, 1024]:
        if time.monotonic() > deadline - (budget_s - 30 - phase1_budget):
            break
        shapes = [
            make_gaussian(n_try, 0.35, 0.22),
            make_symmetric_gaussian(n_try, 0.18),
            make_two_hump(n_try, 0.25),
            make_exponential(n_try, 0.6),
            np.ones(n_try) / n_try,
        ]
        for i, sh in enumerate(shapes):
            try_seq(sh, f"shape_{n_try}_{i}")

    print(f"  After Phase 1: best={best_score:.6f} (n={len(best_seq) if best_seq is not None else 0})")

    # === Phase 2: Nesterov Adam optimization (500s) ===
    print(f"\n=== Phase 2: Nesterov Adam (main workhorse) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t2 = min(rem, 500)
        if t2 > 30:
            print(f"  Time: {t2:.0f}s, n={len(best_seq)}")
            a_opt, obj = adam_nesterov_optimize(
                best_seq, time_limit=t2 * 0.55,
                tau_init=0.08 * fast_convolve(best_seq).max(),
                lr_init=0.005, use_nesterov=True, verbose=True)
            try_seq(a_opt, "adam_nest_main")

            # Second pass with lower temperature
            if time.monotonic() < deadline - 600:
                rem2 = deadline - time.monotonic() - 600
                t2b = min(rem2, 200)
                if t2b > 20:
                    a_opt2, _ = adam_nesterov_optimize(
                        best_seq, time_limit=t2b,
                        tau_init=0.03 * fast_convolve(best_seq).max(),
                        lr_init=0.002, use_nesterov=True, verbose=True)
                    try_seq(a_opt2, "adam_nest_fine")

    # === Phase 3: Nesterov subgradient (200s) ===
    print(f"\n=== Phase 3: Nesterov Subgradient ===")
    if best_seq is not None and time.monotonic() < deadline - 450:
        rem = deadline - time.monotonic() - 450
        t3 = min(rem, 200)
        if t3 > 20:
            print(f"  Time: {t3:.0f}s, n={len(best_seq)}")
            a_opt, _ = nesterov_subgrad_optimize(best_seq, time_limit=t3, verbose=True)
            try_seq(a_opt, "subgrad_nest")

    # === Phase 4: LP-FW refinement (400s) ===
    print(f"\n=== Phase 4: LP-FW Refinement ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 40:
            t4 = min(rem, 400)
            print(f"  Time: {t4:.0f}s, n={len(best_seq)}")
            # Only use LP-FW for n <= 1536 (LP gets too big)
            if len(best_seq) <= 1536:
                a_opt, _ = lp_fw_optimize(best_seq, time_limit=t4, verbose=True)
                try_seq(a_opt, "lpfw")
            else:

                a_opt, _ = adam_nesterov_optimize(best_seq, time_limit=t4, lr_init=0.003, verbose=True)
                try_seq(a_opt, "adam_large")

            # Perturbed LP-FW
            if time.monotonic() < deadline - 250 and len(best_seq) <= 1536:
                rem2 = deadline - time.monotonic() - 250
                t4b = min(rem2, 120)
                if t4b > 20:
                    perturbed = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * 0.004)
                    a_opt2, _ = lp_fw_optimize(perturbed, time_limit=t4b, verbose=False)
                    try_seq(a_opt2, "lpfw_pert")

    # === Phase 5: Minimax MW (200s) ===
    print(f"\n=== Phase 5: Minimax MW ===")
    if best_seq is not None and time.monotonic() < deadline - 200:
        rem = deadline - time.monotonic() - 200
        t5 = min(rem, 200)
        if t5 > 20:
            print(f"  Time: {t5:.0f}s, n={len(best_seq)}")
            a_opt, _ = minimax_mw_optimize(best_seq, time_limit=t5, verbose=True)
            try_seq(a_opt, "minimax")

    # === Phase 6: Two-scale (150s) ===
    print(f"\n=== Phase 6: Two-Scale Optimization ===")
    if best_seq is not None and time.monotonic() < deadline - 150:
        rem = deadline - time.monotonic() - 150
        t6 = min(rem, 150)
        if t6 > 20:
            print(f"  Time: {t6:.0f}s, n={len(best_seq)}")
            a_opt, _ = two_scale_optimize(best_seq, time_limit=t6, block_size=8, verbose=True)
            try_seq(a_opt, "two_scale")

    # === Phase 7: Peak redistribution CD (100s) ===
    print(f"\n=== Phase 7: Peak Redistribution CD ===")
    if best_seq is not None and time.monotonic() < deadline - 80:
        rem = deadline - time.monotonic() - 80
        t7 = min(rem, 100)
        if t7 > 15:
            print(f"  Time: {t7:.0f}s, n={len(best_seq)}")

    # === Phase 8: Multi-length final push (100s) ===
    print(f"\n=== Phase 8: Multi-Length Final Push ===")
    if best_seq is not None:
        base = best_seq
        for target_n in [1024, 1536, 2048]:
            if time.monotonic() > deadline - 40:
                break
            n_cur = len(base)
            if target_n == n_cur:
                continue
            f_interp = interpolate_sequence(base, target_n)
            sc = try_seq(f_interp, f"final_interp_n{target_n}")
            if sc < best_score + 0.01:
                # Worth optimizing
                rem = deadline - time.monotonic() - 30
                t_opt = min(rem, 60)
                if t_opt > 15:
                    a_opt, _ = adam_nesterov_optimize(
                        f_interp, time_limit=t_opt, lr_init=0.004, verbose=False)
                    try_seq(a_opt, f"final_opt_n{target_n}")

    # === Phase 9: Final polish (40s) ===
    print(f"\n=== Phase 9: Final Polish ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        rem = deadline - time.monotonic() - 5
        t9 = min(rem, 40)
        if t9 > 10:
            a_opt, _ = adam_nesterov_optimize(
                best_seq, time_limit=t9,
                tau_init=0.02 * fast_convolve(best_seq).max(),

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
