    # Build constraint matrix
    A_ub = np.zeros((num_constraints, n))
    for j in range(n):
        A_ub[j:j+n, j] = a
    
    c = -np.ones(n)  # minimize -sum(g)
    b_ub = np.full(num_constraints, M)
    
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.status != 0:
        return None, 0.0
    
    g = res.x
    S = -res.fun
    return g, S

def line_search(a, g, max_evals=30):
    """
    Golden section search for t in [0,1] minimizing objective((1-t)*a + t*g).
    Both a and g are on the simplex.
    """
    def obj_at_t(t):
        x = (1 - t) * a + t * g
        conv = fast_convolve(x)
        return 2.0 * len(x) * conv.max()  # sum(x) = 1
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ≈ 0.618
    lo = 0.0
    hi = 1.0
    
    # Initial points
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = obj_at_t(c1)
    f2 = obj_at_t(c2)
    
    for _ in range(max_evals - 2):
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = obj_at_t(c1)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = obj_at_t(c2)
    
    t_best = (lo + hi) / 2
    return t_best

def coordinate_descent_polish(a0, time_limit, verbose=False, label=""):
    """Polish with coordinate descent."""
    a = normalize_simplex(a0)
    n = len(a)
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            if old_val < 1e-14:
                continue
            
            best_j_val = old_val
            best_j_obj = best_obj
            
            for mult in [0.8, 0.9, 0.95, 1.05, 1.1, 1.2]:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 1.0:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                trial = a.copy()
                trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                trial[mask] *= scale
                trial = np.maximum(trial, 0.0)
                ss = trial.sum()
                if ss < 1e-12:
                    continue
                trial /= ss
                obj = 2.0 * n * fast_convolve(trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-12:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0.0)
                    a /= a.sum()
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_a, best_obj / (2.0 * n)


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def make_bell(n, width_frac=0.6):
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def make_uniform(n):
    return np.ones(n) / n

"""
LP-based Frank-Wolfe optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses scipy HiGHS for LP subproblems, golden section line search, diverse
initializations at multiple lengths, symmetrization, and coordinate descent
polishing. Key innovation over previous approach: exact LP direction finding
instead of approximate subgradient methods.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def normalize_simplex(a):
    """Normalize to simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_convolution_matrix(a):
    """Build (2n-1) x n matrix: A[k,j] = a[k-j] if 0<=k-j<=n-1 else 0."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp_direction(a, M):
    """
    Solve LP: max sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0.
    Returns (g, S) where S is optimal value. Returns (None, 0) on failure.
    """
    n = len(a)
    num_constraints = 2 * n - 1

    A_ub = np.zeros((num_constraints, n))
    for j in range(n):
        A_ub[j:j + n, j] = a

    c = -np.ones(n)
    b_ub = np.full(num_constraints, M)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status != 0:
        return None, 0.0

    g = res.x
    S = -res.fun
    return g, S


def line_search_golden(a, g, max_evals=30):

        conv = fast_convolve(x)
        return 2.0 * n * conv.max()

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = obj_at_t(c1)
    f2 = obj_at_t(c2)

    for _ in range(max(0, max_evals - 2)):
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = obj_at_t(c1)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = obj_at_t(c2)

    return (lo + hi) / 2.0


def lp_frank_wolfe(a0, time_limit, verbose=False, label=""):
    """
    LP-based Frank-Wolfe for minimizing max(conv(a,a)) on the simplex.
    Returns (best_a, max_conv_value).
    """
    a = normalize_simplex(a0.astype(np.float64))
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    lp_time_total = 0.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()

        # Solve LP
        lp_start = time.monotonic()
        g, S = solve_lp_direction(a, M)
        lp_time_total += time.monotonic() - lp_start

        if g is None:
            break

        if S <= 1.0 + 1e-8:
            no_improve += 1
            if no_improve > 30:
                # Try relaxed constraint
                g2, S2 = solve_lp_direction(a, M * 1.005)
                if g2 is not None and S2 > 1.0 + 1e-6:
                    g, S = g2, S2
                else:
                    # Perturb and continue
                    if no_improve > 100:
                        break
                    rng = np.random.default_rng(it * 31 + 7)
                    a = normalize_simplex(a + rng.standard_normal(n) * 0.002 * a)
                    continue
            else:
                rng = np.random.default_rng(it * 31 + 7)
                a = normalize_simplex(a + rng.standard_normal(n) * 0.001 * a)
                continue

        g = normalize_simplex(g)

        # Line search
        t_opt = line_search_golden(a, g, max_evals=25)
        a_new = (1.0 - t_opt) * a + t_opt * g
        a_new = normalize_simplex(a_new)

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-11:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
            if verbose and (it % 100 == 0 or it <= 5):
                avg_lp = lp_time_total / max(it, 1) * 1000
                print(f"    {label} it={it}: obj={best_obj:.8f}, S={S:.4f}, "
                      f"t={t_opt:.4f}, LP_avg={avg_lp:.0f}ms")
        else:
            a = a_new
            no_improve += 1
            if no_improve > 60:
                rng = np.random.default_rng(it * 127 + 13)
                noise = rng.standard_normal(n) * 0.003 * best_a
                a = normalize_simplex(best_a + noise)
                no_improve = 0

        # Periodic symmetrization
        if it % 500 == 0:
            a_sym = 0.5 * (a + a[::-1])
            a_sym = normalize_simplex(a_sym)
            conv_sym = fast_convolve(a_sym)
            obj_sym = 2.0 * n * conv_sym.max()
            if obj_sym < best_obj:
                best_obj = obj_sym
                best_a = a_sym.copy()
                a = a_sym.copy()
                if verbose:
                    print(f"    {label} it={it}: SYM improved obj={best_obj:.8f}")

    if verbose:
        print(f"    {label} done: it={it}, best_obj={best_obj:.8f}, "
              f"total_LP_time={lp_time_total:.1f}s")

    return best_a, best_obj / (2.0 * n)


def mirror_descent(a0, time_limit, verbose=False, label=""):
    """
    Multiplicative weight mirror descent on the simplex.
    Returns (best_a, max_conv_value).
    """
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = 0.01
    temp = 0.0005
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()

        K = min(30, len(conv))
        if K < len(conv):
            peak_indices = np.argpartition(conv, -K)[-K:]
        else:
            peak_indices = np.arange(len(conv))

        pv = conv[peak_indices]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peak_indices))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Subgradient
        g = np.zeros(n)
        for idx in range(len(peak_indices)):
            k = int(peak_indices[idx])
            wk = w[idx]
            if wk < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue
            g[js:je + 1] += 2.0 * wk * a[k - np.arange(js, je + 1)]

        # Backtracking
        best_lr = lr * 0.1
        best_trial = M * 1.01
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            if trial_lr < 1e-10:
                continue
            log_upd = np.clip(-trial_lr * g, -50, 50)
            a_t = np.maximum(a * np.exp(log_upd), 1e-15)
            a_t = a_t / a_t.sum()
            obj_t = 2.0 * n * fast_convolve(a_t).max()
            if obj_t <= best_trial:
                best_lr = trial_lr
                best_trial = obj_t
                if obj_t < M * 0.999:
                    break

        log_upd = np.clip(-best_lr * g, -50, 50)
        a = np.maximum(a * np.exp(log_upd), 1e-15)
        a = a / a.sum()

        new_obj = 2.0 * n * fast_convolve(a).max()
        if new_obj < best_obj - 1e-12:
            best_obj = new_obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    {label} MD it={it}: obj={best_obj:.8f}")
        else:
            no_improve += 1

        temp = min(temp * 1.002, 0.05)
        if no_improve > 100:
            lr *= 0.5
            no_improve = 0
        lr = max(lr * 0.9999, 1e-8)

        if no_improve > 500:
            rng = np.random.default_rng(it * 7919 + 13)
            a = np.maximum(best_a + rng.standard_normal(n) * 0.002, 1e-15)
            a = a / a.sum()
            lr = 0.003
            no_improve = 0

        if it % 500 == 0:
            a_sym = 0.5 * (a + a[::-1])
            a_sym = np.maximum(a_sym, 1e-15)
            a_sym = a_sym / a_sym.sum()
            sym_obj = 2.0 * n * fast_convolve(a_sym).max()
            if sym_obj < best_obj:
                best_obj = sym_obj
                best_a = a_sym.copy()
                a = a_sym.copy()

        if lr < 1e-9:
            break

    return best_a, best_obj / (2.0 * n)


def coordinate_descent(a0, time_limit, verbose=False, label=""):
    """Coordinate descent polishing. Returns (best_a, max_conv_value)."""
    a = normalize_simplex(a0)
    n = len(a)
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-13:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in [0.8, 0.9, 0.95, 1.05, 1.1, 1.25]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.9:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-11:
                    continue
                scale = (1.0 - new_val) / denom
                trial = a.copy()
                trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                trial[mask] *= scale
                trial = np.maximum(trial, 0.0)
                ss = trial.sum()
                if ss < 1e-12:
                    continue
                trial /= ss
                obj = 2.0 * n * fast_convolve(trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-11:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0.0)
                    a /= a.sum()
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            break

    return best_a, best_obj / (2.0 * n)


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_triangular(n):
    """Triangular (tent) sequence."""
    a = 1.0 - 2.0 * np.abs(np.arange(n) - (n - 1) / 2.0) / (n - 1)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def symmetrize(a):
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    ss = a_sym.sum()
    if ss > 1e-15:
        a_sym /= ss
    return a_sym


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: LP-based Frank-Wolfe (primary), mirror descent
    (diversity), coordinate descent (polishing), across multiple lengths
    with diverse initializations and symmetrization.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception as e:
        print(f"No parent: {e}")

    # === Phase 1: Generate and evaluate diverse initializations ===
    print("=== Phase 1: Initializations ===")
    n_parent = len(parent) if parent is not None else 768

    length_options = [n_parent, 768, 640, 896, 1024]
    length_options = sorted(set(l for l in length_options if l >= 128))

    init_candidates = []
    for n_try in length_options[:4]:
        for sig in [0.25, 0.30, 0.35, 0.40]:
            init_candidates.append((make_gaussian(n_try, sig), f"g{sig}_n{n_try}"))
        init_candidates.append((make_bell(n_try, 0.5), f"bell_n{n_try}"))
        init_candidates.append((make_triangular(n_try), f"tri_n{n_try}"))

    if parent is not None:
        init_candidates.append((parent, "parent"))
        init_candidates.append((symmetrize(parent), "parent_sym"))

    # Score all initializations
    scored = []
    for a_init, label in init_candidates:
        a_norm = normalize_simplex(a_init)
        obj = 2.0 * len(a_norm) * fast_convolve(a_norm).max()
        scored.append((obj, a_norm, label))
    scored.sort(key=lambda x: x[0])

    for obj, a_init, label in scored[:5]:
        try_seq(a_init, f"init_{label}")

    if best_seq is None:
        best_seq = scored[0][1].copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"  Best init: obj={scored[0][0]:.6f} ({scored[0][2]})")

    # === Phase 2: LP Frank-Wolfe on top initializations ===
    print(f"\n=== Phase 2: LP Frank-Wolfe ===")
    top_k = min(3, len(scored))
    phase2_budget = (deadline - time.monotonic()) * 0.40

    for i in range(top_k):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        obj_init, a_init, label = scored[i]
        t_alloc = min(phase2_budget / top_k, remaining - 30)
        if t_alloc < 20:
            break

        print(f"  LP-FW [{label}] n={len(a_init)}, budget={t_alloc:.0f}s")
        a_opt, mc = lp_frank_wolfe(a_init, time_limit=t_alloc,
                                   verbose=True, label=f"LP_{label}")
        try_seq(a_opt, f"LP_{label}")
        try_seq(symmetrize(a_opt), f"LP_{label}_sym")

    # === Phase 3: Mirror descent for diversity ===
    print(f"\n=== Phase 3: Mirror Descent ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_md = min(remaining * 0.20, 300)
        if t_md > 30:
            a_norm = normalize_simplex(best_seq)
            a_opt, mc = mirror_descent(a_norm, time_limit=t_md,
                                       verbose=True, label="MD_from_best")
            try_seq(a_opt, "MD_best")
            try_seq(symmetrize(a_opt), "MD_best_sym")

    # === Phase 4: Coordinate descent polish ===
    print(f"\n=== Phase 4: CD Polish ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_cd = min(remaining * 0.15, 200)
        if t_cd > 20:
            a_norm = normalize_simplex(best_seq)
            a_opt, mc = coordinate_descent(a_norm, time_limit=t_cd,
                                           verbose=True, label="CD1")
            try_seq(a_opt, "CD1")

    # === Phase 5: Second LP round from best ===
    print(f"\n=== Phase 5: LP Round 2 ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_lp2 = min(remaining * 0.25, 500)
        if t_lp2 > 30:
            a_norm = normalize_simplex(best_seq)
            a_opt, mc = lp_frank_wolfe(a_norm, time_limit=t_lp2,
                                       verbose=True, label="LP2")
            try_seq(a_opt, "LP2")
            try_seq(symmetrize(a_opt), "LP2_sym")

    # === Phase 6: Polish with CD again ===
    print(f"\n=== Phase 6: CD Polish 2 ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        t_cd2 = min(remaining * 0.15, 150)
        if t_cd2 > 15:
            a_norm = normalize_simplex(best_seq)
            a_opt, mc = coordinate_descent(a_norm, time_limit=t_cd2,
                                           verbose=True, label="CD2")
            try_seq(a_opt, "CD2")

    # === Phase 7: Shape transfer to other lengths ===
    print(f"\n=== Phase 7: Shape Transfer ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        a_best = normalize_simplex(best_seq)
        for n_try in [640, 896, 1024, 1280]:
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break
            if n_try == len(a_best):
                continue
            a_interp = interpolate_sequence(a_best, n_try)
            t_alloc = min(remaining * 0.15, 120)
            if t_alloc < 15:
                break
            a_opt, mc = lp_frank_wolfe(a_interp, time_limit=t_alloc,
                                       verbose=False, label=f"xfer_{n_try}")
            try_seq(a_opt, f"xfer_{n_try}")

    # === Phase 8: Final fine polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            t_final = min(remaining - 10, 60)
            if t_final > 5:
                a_norm = normalize_simplex(best_seq)
                a_opt, mc = coordinate_descent(a_norm, time_limit=t_final,
                                               verbose=False, label="final_CD")
                try_seq(a_opt, "final_CD")
                # Try symmetrized final
                try_seq(symmetrize(best_seq), "final_sym")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final check
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
