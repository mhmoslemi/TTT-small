import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u * rho > (cssv - 1)
    rho_max = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_max] - 1.0) / (rho_max + 1.0)
    return np.maximum(v - theta, 0.0)

def lp_direction(f, M):
    """Solve LP: max 1'g s.t. (f*g)_k <= M, g >= 0"""
    n = len(f)
    # Build constraint matrix (Toeplitz structure)
    # A[k, j] = f[k-j] if 0 <= k-j < n else 0
    # Using scipy's toeplitz or manual construction
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def fw_step(f, time_limit, verbose=False):

    f = f / f.sum()
    n = len(f)

    M = c.max()
    best_f = f.copy()
    best_obj = M

    gamma = 0.9

        f_extrap = (1 + gamma) * f - gamma * momentum
        f_extrap = np.maximum(f_extrap, 0)
        s = f_extrap.sum()
        if s < 1e-15:

        else:

        c_extrap = fast_convolve(f_extrap)

        g = lp_direction(f_extrap, M_extrap)
        if g is None:

            g = lp_direction(f_extrap, M_extrap * 1.1)
            if g is None:

        gs = g.sum()
        if gs < 1e-15:
            break
        g = g / gs
        
        # FW step with line search
        if np.linalg.norm(g - f) < 1e-12:

        best_alpha = 0

            if val < best_val:

                best_alpha = alpha

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)

def make_super_gaussian(n, p, width_frac=0.5):
    x = np.arange(n) - n / 2.0
    half_w = n * width_frac / 2.0
    t = np.abs(x) / half_w
    a = np.exp(-(t ** p))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def make_raised_cosine(n):
    x = np.linspace(-1, 1, n)
    a = 0.5 * (1 + np.cos(np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def make_blackman(n):
    a = np.blackman(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

def make_uniform(n):
    return np.ones(n) / n

def get_lp_matrix(f):
    """Build the Toeplitz constraint matrix for the LP."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A

def solve_lp_direction(f, M, A_precomputed=None):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    n = len(f)
    if A_precomputed is None:
        A = get_lp_matrix(f)
    else:
        A = A_precomputed
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def fw_lpfw(f0, time_limit, m_scale=1.0, verbose=False):

    Minimizes max_k (f*f)_k subject to f >= 0, sum(f) = 1.

    M = c.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    v = f.copy()  # momentum variable
    gamma = 1.0 / (1.0 + it)
    
    prev_f = f.copy()
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        
        # Build constraint matrix
        A = get_lp_matrix(f)
        
        # Solve LP with current M

        if g is None:
            # Try slightly larger M
            for scale in [1.01, 1.02, 1.05, 1.1]:
                g = solve_lp_direction(f, M * m_scale * scale, A)
                if g is not None:
                    break
            if g is None:
                break
        
        gs = g.sum()
        if gs < 1e-15:
            break
        g = g / gs
        
        # Check if at fixed point
        diff = g - f
        if np.linalg.norm(diff) < 1e-13:
            no_improve += 1
            if no_improve > 50:
                break
            continue
        else:
            if np.linalg.norm(diff) < 1e-10:
                no_improve += 1
            else:
                no_improve = 0
        
        # Line search with cubic interpolation
        c = fast_convolve(f)
        M_current = c.max()
        
        best_alpha = 0.0
        best_val = M_current
        
        # Coarse search
        alphas = np.concatenate([
            np.linspace(0.01, 0.5, 10),
            np.linspace(0.5, 1.0, 10)
        ])
        
        vals = []
        for alpha in alphas:
            ft = f + alpha * diff
            ft = np.maximum(ft, 0)
            st = ft.sum()
            if st < 1e-15:
                vals.append(np.inf)
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            vals.append(val)
            if val < best_val:
                best_val = val
                best_alpha = alpha

            hi = min(1.0, best_alpha + 0.05)

                ft = f + mid * diff
                ft = np.maximum(ft, 0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-10:
            f_new = f + best_alpha * diff
            f_new = np.maximum(f_new, 0)
            f_new = f_new / f_new.sum()
            c_new = fast_convolve(f_new)
            M_new = c_new.max()
            
            # Nesterov update
            v = f
            f = f_new
            M = M_new
            gamma = 1.0 / (1.0 + it)
            
            if M < best_obj - 1e-15:

        else:

            if no_improve > 30:

                rng = np.random.default_rng(it * 7 + 3)

                c = fast_convolve(f)
                M = c.max()
                m_scale = 1.0
                no_improve = 0

def gradient_descent_smoothed(f0, time_limit, beta_schedule=None, verbose=False):
    """Smoothed gradient descent with log-sum-exp."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    c = fast_convolve(f)
    best_obj = c.max()
    
    t0 = time.monotonic()
    momentum = np.zeros(n)
    mu = 0.85
    
    if beta_schedule is None:
        beta_schedule = [4, 8, 16, 32, 64, 128]
    
    for beta in beta_schedule:
        if time.monotonic() - t0 > time_limit * 0.9:
            break
        
        stage_start = time.monotonic()
        stage_time = min(time_limit * 0.12, (time_limit - (time.monotonic() - t0)) * 0.12)
        
        it = 0
        no_improve = 0
        lr = 1.0 / (n ** 0.5 * beta)
        
        while time.monotonic() - stage_start < stage_time:
            it += 1
            c = fast_convolve(f)
            
            # Compute smoothed gradient
            scaled = beta * c
            scaled -= scaled.max()
            exp_scaled = np.exp(scaled)
            w = exp_scaled / exp_scaled.sum()
            
            # Gradient: g_j = 2 * sum_i f_i * w_{j+i}
            w_rev = w[::-1].copy()
            conv_len = n + len(w_rev) - 1
            fft_size = 1 << (conv_len - 1).bit_length()
            F = np.fft.rfft(f, fft_size)
            W = np.fft.rfft(w_rev, fft_size)
            conv = np.fft.irfft(F * W, fft_size)[:conv_len]
            indices = np.arange(2 * n - 2, 2 * n - 2 - n, -1)
            g = 2.0 * conv[indices]
            
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-15:
                break
            
            lr = min(lr, 0.5 / (g_norm + 1e-10))
            momentum = mu * momentum - lr * g
            f_trial = f + momentum
            f_trial = project_simplex(f_trial)
            
            c_trial = fast_convolve(f_trial)
            obj_trial = c_trial.max()
            
            if obj_trial <= c.max() + 1e-10:
                f = f_trial
                if obj_trial < best_obj - 1e-13:
                    best_obj = obj_trial
                    best_f = f.copy()
                    no_improve = 0
                    lr *= 1.03
                else:
                    no_improve += 1
            else:
                no_improve += 1
                lr *= 0.7
                momentum *= 0.5
            
            if no_improve > 60:
                rng = np.random.default_rng(it * 13 + int(beta))
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                momentum = np.zeros(n)
                lr = 1.0 / (n ** 0.5 * beta * 0.5)
                no_improve = 0
    
    return best_f, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate to new length with smoothing."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    # Light kernel smoothing
    kernel = np.array([0.1, 0.2, 0.4, 0.2, 0.1])
    f_smooth = np.convolve(f_interp, kernel, mode='same')
    f_smooth = np.maximum(f_smooth, 0)
    ss = f_smooth.sum()
    if ss > 1e-15:
        f_smooth /= ss
    return f_smooth

def coordinate_polish(f0, time_limit):
    """Quick coordinate descent for final polish."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f0, fast_convolve(f0).max()
    f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    improved = True
    sweep = 0
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        sweep += 1
        for j in range(0, n, max(1, n // 200)):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue
            
            best_j_val = old_val
            best_j_obj = best_obj
            
            for mult in [0.85, 0.9, 0.95, 1.05, 1.1, 1.15]:
                new_val = old_val * mult
                if new_val < 0:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial = f_trial / ss
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
    
    return best_f, best_obj

def symmetrize(f):
    """Make a sequence symmetric: f_j = f_{n-1-j}."""
    n = len(f)
    f_sym = 0.5 * (f + f[::-1])
    s = f_sym.sum()
    if s > 1e-15:
        f_sym /= s
    return f_sym

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def make_super_gaussian(n, p, width_frac=0.5):
    """Super-Gaussian shape: exp(-|x|^p)."""
    x = np.arange(n) - n / 2.0
    half_w = n * width_frac / 2.0
    t = np.abs(x) / half_w
    a = np.exp(-(t ** p))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_raised_cosine(n):
    """Hann window / raised cosine."""
    x = np.linspace(-1, 1, n)
    a = 0.5 * (1 + np.cos(np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:

    return a

    """Blackman window."""

def get_lp_matrix(f):
    """Build the Toeplitz constraint matrix A where A[k,j] = f[k-j] if valid."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_direction(f, M, A=None):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    n = len(f)
    if A is None:
        A = get_lp_matrix(f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def gradient_descent_smoothed(f0, time_limit, verbose=False):
    """Log-sum-exp smoothed gradient descent with momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    c = fast_convolve(f)
    best_obj = c.max()

    t0 = time.monotonic()
    momentum = np.zeros(n)
    mu = 0.85
    beta_schedule = [4, 8, 16, 32, 64, 128, 256]

    for beta in beta_schedule:
        if time.monotonic() - t0 > time_limit * 0.92:
            break

        stage_start = time.monotonic()
        stage_time = min(time_limit * 0.1, (time_limit - (time.monotonic() - t0)) * 0.1)

        it = 0
        no_improve = 0
        lr = 0.5 / (n ** 0.5 * beta)

        while time.monotonic() - stage_start < stage_time:
            it += 1
            c = fast_convolve(f)

            # Softmax weights
            scaled = beta * c
            scaled -= scaled.max()
            exp_scaled = np.exp(scaled)
            w = exp_scaled / exp_scaled.sum()

            # Gradient via correlation
            w_rev = w[::-1].copy()
            conv_len = n + len(w_rev) - 1
            fft_size = 1 << (conv_len - 1).bit_length()
            F = np.fft.rfft(f, fft_size)
            W = np.fft.rfft(w_rev, fft_size)
            conv = np.fft.irfft(F * W, fft_size)[:conv_len]
            indices = np.arange(2 * n - 2, 2 * n - 2 - n, -1)
            g = 2.0 * conv[indices]

            g_norm = np.linalg.norm(g)
            if g_norm < 1e-15:
                break

            lr = min(lr, 0.3 / (g_norm + 1e-10))
            momentum = mu * momentum - lr * g
            f_trial = f + momentum
            f_trial = project_simplex(f_trial)

            c_trial = fast_convolve(f_trial)

def interpolate_sequence(f, new_n):
    """Interpolate to new length with light smoothing."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    kernel = np.array([0.05, 0.2, 0.5, 0.2, 0.05])
    f_smooth = np.convolve(f_interp, kernel, mode='same')
    f_smooth = np.maximum(f_smooth, 0)
    ss = f_smooth.sum()
    if ss > 1e-15:
        f_smooth /= ss
    return f_smooth


def coordinate_polish(f0, time_limit):
    """Coordinate descent for final refinement."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()

            old_val = f[j]

                if ss < 1e-15:

def symmetrize(f):

    f_sym = 0.5 * (f + f[::-1])
    s = f_sym.sum()

        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # === Phase 1: Multi-shape, multi-length initialization ===
    print("=== Phase 1: Initialization Screening ===")
    t1 = time.monotonic()

    n_parent = len(parent) if parent is not None else 768
    lengths = sorted(set([512, 640, 768, 896, 1024, n_parent]))

    best_init = None
    best_init_obj = float('inf')

    for n in lengths:
        shapes = []
        shapes.append(("uniform", make_uniform(n)))
        shapes.append(("gauss2", make_super_gaussian(n, 2, 0.5)))
        shapes.append(("gauss3", make_super_gaussian(n, 3, 0.45)))
        shapes.append(("gauss4", make_super_gaussian(n, 4, 0.45)))
        shapes.append(("gauss6", make_super_gaussian(n, 6, 0.4)))
        shapes.append(("gauss8", make_super_gaussian(n, 8, 0.35)))
        shapes.append(("raised_cos", make_raised_cosine(n)))
        shapes.append(("blackman", make_blackman(n)))

        if parent is not None:
            if n == n_parent:
                shapes.append(("parent", parent.copy()))
            else:
                p_interp = interpolate_sequence(parent, n)
                shapes.append(("parent_interp", p_interp))
                p_sym = symmetrize(p_interp)
                shapes.append(("parent_sym", p_sym))

        for name, a in shapes:
            a = np.maximum(a, 0)
            ss = a.sum()
            if ss < 1e-10:
                continue
            a = a / ss
            obj = evaluate_obj(a)
            if obj < best_init_obj:
                best_init_obj = obj
                best_init = a.copy()
            if obj < best_score + 0.05:
                update_best(a, f"init_{name}_n{n}")

    # Symmetrize best init
    if best_init is not None:
        sym_best = symmetrize(best_init)
        sym_obj = evaluate_obj(sym_best)
        if sym_obj < best_init_obj:
            best_init = sym_best
            best_init_obj = sym_obj
        update_best(sym_best, "init_sym_best")

    # Add random perturbations to best init
    for trial in range(8):
        noise = rng.standard_normal(len(best_init)) * 0.015
        a_pert = np.maximum(best_init + noise, 0)
        a_pert = a_pert / a_pert.sum()
        obj = evaluate_obj(a_pert)
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a_pert.copy()
        update_best(a_pert, f"init_pert{trial}")

    print(f"  Best init obj={best_init_obj:.8f} (n={len(best_init)})")
    print(f"  Phase 1 time: {time.monotonic()-t1:.1f}s")

    # === Phase 2: Smoothed gradient descent ===
    print(f"\n=== Phase 2: Smoothed GD (n={len(best_init)}) ===")
    t2 = time.monotonic()
    rem = deadline - time.monotonic() - 700
    t_alloc = max(30, min(rem, 200))
    print(f"  Allocation: {t_alloc:.0f}s")

    a_gd, obj_gd = gradient_descent_smoothed(best_init, time_limit=t_alloc, verbose=True)
    update_best(a_gd, "gd_phase2")

    # Try from parent directly
    if parent is not None:
        rem = deadline - time.monotonic() - 600
        if rem > 50:
            t_alloc2 = min(rem * 0.35, 120)
            a_gd2, _ = gradient_descent_smoothed(parent, time_limit=t_alloc2)
            update_best(a_gd2, "gd_parent")
            a_gd2_sym = symmetrize(a_gd2)
            update_best(a_gd2_sym, "gd_parent_sym")

    print(f"  Phase 2 time: {time.monotonic()-t2:.1f}s")

    # === Phase 3: Main LP-FW optimization ===
    print(f"\n=== Phase 3: LP-FW Main ===")
    t3 = time.monotonic()

    if best_seq is None:
        best_seq = best_init

    n_main = len(best_seq)
    rem = deadline - time.monotonic() - 500
    t_alloc = max(60, min(rem, 450))
    print(f"  n={n_main}, allocation={t_alloc:.0f}s")

    # Run LP-FW on current best
    a_lpfw, obj_lpfw = fw_lpfw(best_seq, time_limit=t_alloc, m_scale=1.0, verbose=True)
    update_best(a_lpfw, "lpfw_main")

    # Symmetrize and re-optimize
    a_lpfw_sym = symmetrize(a_lpfw)
    update_best(a_lpfw_sym, "lpfw_sym")

    rem = deadline - time.monotonic() - 400
    if rem > 50 and not np.allclose(a_lpfw_sym, a_lpfw):
        t_sym = min(rem * 0.4, 150)
        a_lpfw_sym2, _ = fw_lpfw(a_lpfw_sym, time_limit=t_sym, m_scale=1.0, verbose=True)
        update_best(a_lpfw_sym2, "lpfw_sym_opt")

    print(f"  Phase 3 time: {time.monotonic()-t3:.1f}s")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-Length Exploration ===")
    t4 = time.monotonic()

    if best_seq is not None:
        base_n = len(best_seq)
        explore_lengths = [n for n in [512, 640, 896, 1024, 1280] if abs(n - base_n) > 50]

        for target_n in explore_lengths:
            rem = deadline - time.monotonic() - 200
            if rem < 50:
                break

            f_interp = interpolate_sequence(best_seq, target_n)
            f_sym = symmetrize(f_interp)
            update_best(f_sym, f"interp_sym_n{target_n}")

            # Quick GD
            t_gd = min(rem * 0.25, 60)
            if t_gd > 20:
                a_opt, _ = gradient_descent_smoothed(f_sym, time_limit=t_gd)
                update_best(a_opt, f"gd_n{target_n}")

            # LP-FW
            rem2 = deadline - time.monotonic() - 100
            if rem2 > 30:
                t_lpfw = min(rem2 * 0.5, 100)
                a_opt2, _ = fw_lpfw(a_opt, time_limit=t_lpfw, verbose=False)
                update_best(a_opt2, f"lpfw_n{target_n}")
                a_opt2_sym = symmetrize(a_opt2)
                update_best(a_opt2_sym, f"lpfw_sym_n{target_n}")

    print(f"  Phase 4 time: {time.monotonic()-t4:.1f}s")

    # === Phase 5: Deep LP-FW with perturbations ===
    print(f"\n=== Phase 5: Deep LP-FW ===")
    t5 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        for restart in range(5):
            rem = deadline - time.monotonic() - 80
            if rem < 25:
                break
            t_alloc_r = min(rem / (5 - restart + 1), 80)

            if restart == 0:
                f_init = best_seq.copy()
            elif restart == 1:
                noise = rng.standard_normal(n_best) * 0.002
                f_init = project_simplex(best_seq + noise)
            elif restart == 2:
                # Blend with symmetric
                f_init = 0.5 * best_seq + 0.5 * best_seq[::-1]
                f_init = f_init / f_init.sum()
            elif restart == 3:
                # Blend with raised cosine
                rc = make_raised_cosine(n_best)
                f_init = 0.7 * best_seq + 0.3 * rc
                f_init = f_init / f_init.sum()
            else:
                # Peak-directed perturbation
                c_best = fast_convolve(best_seq)
                peak_idx = np.argmax(c_best)
                f_init = best_seq.copy()
                # Reduce peak contributors
                js = max(0, peak_idx - n_best + 1)
                je = min(n_best - 1, peak_idx)
                f_init[js:je+1] *= 0.95
                f_init = np.maximum(f_init, 0)
                f_init = f_init / f_init.sum()

            m_scale_r = 1.0 + 0.02 * restart
            a_opt_r, _ = fw_lpfw(f_init, time_limit=t_alloc_r, m_scale=m_scale_r,
                                  verbose=(restart == 0))
            update_best(a_opt_r, f"deep_{restart}")
            a_opt_r_sym = symmetrize(a_opt_r)
            update_best(a_opt_r_sym, f"deep_{restart}_sym")

    print(f"  Phase 5 time: {time.monotonic()-t5:.1f}s")

    # === Phase 6: Final coordinate polish ===
    print(f"\n=== Phase 6: Final Polish ===")
    t6 = time.monotonic()

    if best_seq is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 20:
            t_alloc = min(rem, 50)
            a_poly, _ = coordinate_polish(best_seq, time_limit=t_alloc)
            update_best(a_poly, "coord_final")

        # Final LP-FW squeeze
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            t_squeeze = min(rem, 25)
            a_sq, _ = fw_lpfw(a_poly, time_limit=t_squeeze, verbose=False)
            update_best(a_sq, "final_squeeze")
            a_sq_sym = symmetrize(a_sq)
            update_best(a_sq_sym, "final_squeeze_sym")

    print(f"  Phase 6 time: {time.monotonic()-t6:.1f}s")

    # Fallback
    if best_seq is None:
        best_seq = make_uniform(512)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification with official evaluator
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        # Should not happen, but fallback
        best_seq = make_uniform(384)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
