If the max is achieved at a unique k*, then:
`grad f(a) = 2 * (a convolved with delta_{k*})` where the convolution is symmetric

More precisely, `(a*a)[k] = sum_i a[i]*a[k-i]`, so:
`d/d(a[j]) (a*a)[k] = a[k-j] + a[j-k]` (if both indices are valid)

So the subgradient at the argmax k* is:
`grad[j] = a[k*-j] + a[j-k*]` (with zero padding)

This is a circulant-like vector! We can compute it with FFT.

Let me also think about the coordinate descent more carefully.

**Peak-targeted coordinate descent:**

The idea is:
1. Find the current max peak at lag k*
2. The peak value is `b[k*] = sum_i a[i] * a[k*-i]`
3. The "top contributors" are the pairs (i, k*-i) with the largest product
4. To reduce the peak, we want to decrease a[i] (or a[k*-i]) for the top contributors
5. But we also want to minimize the increase in OTHER peaks
6. So we move mass from position i to a position j that minimizes the maximum increase in other peaks

The "cost" of moving mass from i to j is:
- Decrease in b[k*]: `delta * (a[i] + a[k*-i])` (approximately, for small delta)
  - Actually: b[k*] decreases by `delta * a[k*-i]` (if we reduce a[i] by delta)
  - Plus b[k*-i] changes too...
  
This is getting complicated. Let me just use a simple heuristic:
- Move a small fraction of a[i] to a[j] where j is far from the peak
- Check if the score improved
- If not, try a different j or a smaller fraction

Let me write a simpler coordinate descent:

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

def fast_conv(a, b):

    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:n]

def lp_best_response(a, M, solver, reduced_frac=0.6):
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    center = n - 1
    half_width = max(1, int(size * reduced_frac / 2.0))
    k_start = max(0, center - half_width)
    k_end = min(size - 1, center + half_width)
    constraints = []
    for k in range(k_start, k_end + 1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        coeffs = a[i_min:i_max + 1]
        b_slice = b_var[i_min:i_max + 1]
        constraints.append(coeffs @ b_slice <= M)
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, use_geometric=False,
                         restart_threshold=15.0, reduced_frac=0.6):

    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    t0 = time.monotonic()

    x_prev = a.copy()
    n_iters = 0
    last_improve = t0
    stall = 0
    t_counter = 0

    while time.monotonic() - t0 < time_limit:
        t_counter += 1
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        if t_counter < 30:
            b = lp_best_response(y, M_curr, solver, reduced_frac=1.0)

            b = lp_best_response(y, M_curr, solver, reduced_frac=reduced_frac)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t_counter + 2.0)

        if use_geometric:
            y_safe = np.maximum(y, 1e-12)
            g_safe = np.maximum(g, 1e-12)
            y_new = np.exp((1.0 - alpha) * np.log(y_safe) + alpha * np.log(g_safe))
        else:
            y_new = (1.0 - alpha) * y + alpha * g

        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t_counter == 1:
            x_new = y_new.copy()
        else:
            momentum = (t_counter - 1.0) / (t_counter + 2.0)
            x_new = y_new + momentum * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new * norm / s_x

        y = y_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()

        score_x = fast_autocorr(x_new).max()
        if score_x < best_score:

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    [LP] iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lse_objective_and_grad(a_flat, n, T_temp):
    a = np.maximum(a_flat.reshape(n), 1e-15)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    exp_vals = np.exp(T_temp * (b - max_b))
    exp_sum = exp_vals.sum()
    w = exp_vals / exp_sum
    lse = max_b + np.log(exp_sum) / T_temp
    obj = 2.0 * n * lse / (s * s)

    fft_size = 1 << (3 * n - 2).bit_length()
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    w_padded = np.zeros(fft_size)
    w_padded[:2 * n - 1] = w
    A_fft = rfft(a_padded, fft_size)
    W_fft = rfft(w_padded, fft_size)
    cross = irfft(np.conj(A_fft) * W_fft, fft_size)[:n]
    conv_grad = 4.0 * cross
    grad = conv_grad / (s * s) - 2.0 * obj * a / s
    return obj, grad

        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        stage_time = min(remaining / max(1, n_stages - si), 120.0)

        def obj_func(x):
            return lse_objective_and_grad(x, n, T)[0]

            return lse_objective_and_grad(x, n, T)[1]

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
