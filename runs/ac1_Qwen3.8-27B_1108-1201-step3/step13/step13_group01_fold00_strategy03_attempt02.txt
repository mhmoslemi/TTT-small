def evaluate_safe(a):
    if not isinstance(a, list):
        return np.inf
    sequence = a.copy()
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):
            return np.inf
    sequence = [float(x) for x in sequence]
    sequence = [max(0, x) for x in sequence]
    s = np.sum(sequence)
    if s < 0.01:
        return np.inf
    fft_size = 1 << (len(sequence) - 1).bit_length()
    a_padded = np.zeros(fft_size)
    a_padded[:len(sequence)] = sequence
    A_fft = np.fft.rfft(a_padded)
    W_fft = np.fft.rfft(a_padded)
    cross = np.fft.irfft(A_fft * np.conj(W_fft), fft_size)[:2*len(sequence)-1]
    max_conv = np.max(cross)
    return 2 * len(sequence) * max_conv / (s ** 2)
