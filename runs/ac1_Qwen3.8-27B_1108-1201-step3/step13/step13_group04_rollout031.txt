"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using ramp-based initialization
exploiting asymmetry to reduce the central convolution peak.
Key insight: one-sided (ramp-like) sequences achieve score ~1.33 vs ~1.50 for symmetric ones.
Uses LSE gradient descent with Nesterov acceleration, peak-targeting local search,
and multi-parameter initialization sweeping.

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Compute full auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_gradient(a, T):
    """Gradient of LSE surrogate T*log(sum exp(b_k/T)) of auto-convolution."""
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    w = np.exp((b - max_b) / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[n - 1] = 1.0
    else:
        w /= w_sum

    fft_size = 1 << (3 * n - 2).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    A_fwd = rfft(a, fft_size)
    W = rfft(w, fft_size)

    conv1 = irfft(A_rev * W, fft_size)
    conv2 = irfft(A_fwd * W, fft_size)

    r1 = conv1[n - 1: n - 1 + n]
    r2 = conv2[:n]

    return r1 + r2


def nesterov_lse(a0, time_limit, T_start=15.0, T_end=0.2, n_stages=16):
    """Nesterov-accelerated LSE gradient descent with geometric cooling."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    y_prev = a.copy()

    temps = np.geomspace(T_start, T_end, n_stages)
    it_count = 0

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        n_iters_stage = max(30, int(remaining * 4.0 / max(1, n_stages - si)))
        for it in range(n_iters_stage):
            if time.monotonic() - t0 > time_limit:
                break

            if it == 0:
                grad = lse_gradient(y, T)
                y_new = y - (1.0 / 3.0) * grad
            else:
                momentum = (it - 1.0 + np.sqrt(1.0 + 4.0 * (it - 1.0) ** 2)) / (2.0 * (it + 2.0))
                y_ext = y + momentum * (y - y_prev)
                grad = lse_gradient(y_ext, T)
                y_new = y_ext - (1.0 / (it + 3.0)) * grad

            y_prev = y.copy()
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y
            y = y_new
            it_count += 1

            score = fast_autocorr(y).max()
            if score < best_score:
                best_score = score
                best_a = y.copy()

            if it_count % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"      LSE stage={si}/{n_stages} T={T:.3f} it={it_count} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def ramp_init(n, p=1.0, decay_q=0.0, orientation=1):
    """Generate a ramp-like initialization with power and optional decay.
    p: exponent on the linear ramp
    decay_q: additional decay at the end (0 = no decay)
    orientation: 1 = increasing, -1 = decreasing
    """
    if orientation == 1:
        x = np.arange(1, n + 1, dtype=np.float64)
        a = x ** p
        if decay_q > 0:
            a *= ((n - np.arange(n)) / n) ** decay_q
    else:
        x = np.arange(n, 0, -1, dtype=np.float64)
        a = x ** p
        if decay_q > 0:
            a *= ((np.arange(n) + 1) / n) ** decay_q
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def exp_init(n, alpha=1.0, orientation=1):
    """Exponential initialization."""
    if orientation == 1:
        x = np.arange(n, dtype=np.float64)
        a = np.exp(alpha * x / n)
    else:
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-alpha * x / n)
    s = a.sum()
    return a / s

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

        do_try_update(a_out, f"peak_n{n_best}")

    # Phase 4: Final LSE polish
    print(f"\n=== Phase 4: Final LSE polish ===")
    n_best = len(best_seq)
    remaining_t = time_budget - (time.monotonic() - t_global)
    t4 = min(200.0, remaining_t - 50.0)
    if t4 > 10:
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        print(f"  LSE polish n={n_best} (budget={t4:.0f}s)")
        a_out, _ = nesterov_lse(a_work, t4, T_start=3.0, T_end=0.05, n_stages=8)
        do_try_update(a_out, "lse_polish")

    # Phase 5: Final peak targeting
    remaining_t = time_budget - (time.monotonic() - t_global)
    if remaining_t > 30:
        n_best = len(best_seq)
        t5 = min(150.0, remaining_t - 10.0)
        if t5 > 10:
            print(f"\n=== Phase 5: Final peak targeting (budget={t5:.0f}s) ===")
            a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = peak_targeting_search(a_work, t5, seed=4242)
            do_try_update(a_out, "peak_final")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence! Fallback to ramp.")
        best_seq = ramp_init(1600, p=1.0)
        best_score = evaluate_safe(best_seq)

    return [float(x) for x in best_seq]
