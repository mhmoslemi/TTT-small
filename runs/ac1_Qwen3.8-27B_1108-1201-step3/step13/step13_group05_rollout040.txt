def build_T(a):
    """Build dense constraint matrix T where (T @ b)[k] = conv(a,b)[k]."""
    n = len(a)
    T = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1)
        T[k, idx] = a[k - idx]
    return T

def lp_solve(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    T = build_T(a)

        if res.success:
            b = np.maximum(res.x, 0)
            if b.sum() > 1e-15:
                return b
    except Exception:
        pass
    return None
