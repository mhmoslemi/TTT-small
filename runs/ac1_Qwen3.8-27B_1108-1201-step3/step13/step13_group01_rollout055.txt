def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def lp_best_response_reduced(a, M, top_lags, solver):
    """
    Solve LP: max sum(b) s.t. b>=0, (a*b)_k <= M for k in top_lags.
    top_lags: list of lag indices to include as constraints.
    Returns (b, dual_values) or (None, None).
    """
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    
    # Build constraint matrix for just the top lags
    K = len(top_lags)
    T = np.zeros((K, n))
    for row, k in enumerate(top_lags):
        # (a*b)_k = sum_{j: 0<=k-j<n} a[k-j] * b[j]
        # So T[row, j] = a[k-j] if 0 <= k-j < n
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            T[row, j] = a[k - j]
    
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None, None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    
    dual_val = None
    try:
        dual_val = np.asarray(prob.constraints[0].dual_value, dtype=np.float64)
    except Exception:
        pass
    
    return b_val, dual_val

The log-sum-exp objective: `f(a) = (1/T) * log(sum_k exp(T * b_k))` where `b = conv(a,a)` and T is temperature.

The gradient of this with respect to `a_i`:
