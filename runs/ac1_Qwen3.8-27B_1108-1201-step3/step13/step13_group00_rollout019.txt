            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        y = y_new
        a = a_new
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        avg_a += a
        n_avg += 1
        if n_avg % 30 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pair_cd_fast(a_input, time_limit, rng, verbose=True):
    n = len(a_input)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a_input, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    g_buf = np.zeros(size)

    n_passes = 0
    no_improve = 0
    n_evals = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            for _ in range(3):
                j = int(rng.integers(n))
                if j == i:
                    continue

                g_buf[:] = 0.0
                g_buf[i:i + n] = 2.0 * a
                g_buf[j:j + n] -= 2.0 * a

                h_val_2i = 1.0 if 2 * i < size else 0.0
                h_val_2j = 1.0 if 2 * j < size else 0.0

                n_samples = 12
                deltas = np.linspace(0.0, ai, n_samples + 1)[1:]
                if len(deltas) == 0:
                    continue

                base = conv[None, :] + deltas[:, None] * g_buf[None, :]
                if h_val_2i > 0:
                    base[:, 2 * i] += deltas ** 2
                if h_val_2j > 0:
                    base[:, 2 * j] += deltas ** 2
                F = base.max(axis=1)

                best_d_idx = int(np.argmin(F))
                if F[best_d_idx] >= best_score - 1e-13:
                    continue

                best_d = deltas[best_d_idx]
                lo = deltas[max(0, best_d_idx - 1)]
                hi = deltas[min(len(deltas) - 1, best_d_idx + 1)]
                for _ in range(4):
                    mid = (lo + hi) * 0.5
                    f_mid = conv.max() if mid == 0 else (conv + mid * g_buf + mid * mid * np.where(
                        np.arange(size) == 2 * i, mid, 0)).max()
                    f_lo = (conv + lo * g_buf).max()
                    if h_val_2i > 0:
                        f_lo = max(f_lo, conv[2 * i] + lo * g_buf[2 * i] + lo * lo)
                    if h_val_2j > 0:
                        f_lo = max(f_lo, conv[2 * j] + lo * g_buf[2 * j] + lo * lo)
                    f_mid = (conv + mid * g_buf).max()
                    if h_val_2i > 0:
                        f_mid = max(f_mid, conv[2 * i] + mid * g_buf[2 * i] + mid * mid)
                    if h_val_2j > 0:
                        f_mid = max(f_mid, conv[2 * j] + mid * g_buf[2 * j] + mid * mid)
                    if f_mid < f_lo:
                        lo = mid
                    else:
                        hi = mid
                best_d = (lo + hi) * 0.5
                n_evals += 1

                a[i] -= best_d
                a[j] += best_d
                s = a.sum()
                if s < 1e-15:
                    a[i] += best_d
                    a[j] -= best_d
                    continue
                a = a * norm / s
                conv = fast_autocorr(a)
                cur_max = conv.max()

                if cur_max < best_score - 1e-13:
                    best_score = cur_max
                    best_a = a.copy()
                    pass_improved = True
                else:
                    a[j] -= best_d
                    a[i] += best_d
                    a = a * norm / a.sum()
                    conv = fast_autocorr(a)

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}, evals={n_evals}")

    return best_a, best_score


def structured_init(n, rng):
    choice = int(rng.integers(0, 6))
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    else:
        x = np.linspace(0.0, 1.0, n)
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def anneal_kicks(a_input, time_limit, rng, verbose=True):
    n = len(a_input)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a_input, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    g_buf = np.zeros(size)

    kick_size = 0.15
    n_kicks = 0

    while time.monotonic() - t0 < time_limit:
        remaining_single = time_limit - (time.monotonic() - t0)
        if remaining_single < 5:
            break

        frac = (time.monotonic() - t0) / time_limit
        kick_scale = kick_size * (0.3 + 0.7 * (1.0 - frac))

        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if j == i:
            continue
        d = rng.random() * kick_scale * norm
        d = min(d, a[i] * 0.9)
        if d < norm * 1e-10:
            continue

        a[i] -= d
        a[j] += d
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            continue
        a = a * norm / s
        n_kicks += 1

        local_time = min(remaining_single * 0.4, 3.0)
        a, local_score = pair_cd_fast(a, local_time, rng, verbose=False)
        if local_score < best_score:
            best_score = local_score
            best_a = a.copy()
            a = a.copy()
        else:
            a = best_a.copy()

        if verbose and n_kicks % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Anneal kick={n_kicks} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LP Nesterov (15% of budget)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.15, 250.0)
    if t1 > 30:
        a_norm = best_seq.copy()
        s_a = a_norm.sum()
        if s_a > 1e-15:
            a_norm = a_norm * np.sqrt(2.0 * n_base) / s_a
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n_base}")

    # Phase 2: Fast Pair CD (main workhorse, 55% of budget)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Fast Pair CD n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.55, 900.0)
    if t2 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pair_cd_fast(a_norm, t2, rng, verbose=True)
        do_try_update(a_out, f"pcd_n{n_best}")

    # Phase 3: Another LP round for potential improvement
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP round 2 n={n_best} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=True)
        do_try_update(a_out, f"lp2_n{n_best}")

    # Phase 4: More pair CD
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Pair CD round 2 n={n_best} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.50, 400.0)
    if t4 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pair_cd_fast(a_norm, t4, rng, verbose=True)
        do_try_update(a_out, f"pcd2_n{n_best}")

    # Phase 5: Annealing kicks to escape local optima
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Anneal kicks n={n_best} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.40, 200.0)
    if t5 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = anneal_kicks(a_norm, t5, rng, verbose=True)
        do_try_update(a_out, f"anneal_n{n_best}")

    # Phase 6: Final pair CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final PCF n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.85, 150.0)
    if t6 > 5:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pair_cd_fast(a_norm, t6, rng, verbose=True)
        do_try_update(a_out, f"pcd_final_n{n_best}")

    # Phase 7: Try longer length if time permits
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 120:
        n_long = n_base + 400
        print(f"\n=== Phase 7: Try n={n_long} (budget={remaining:.0f}s) ===")
        t7 = min(remaining * 0.6, 200.0)
        init_long = interpolate_sequence(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = pair_cd_fast(a_norm, t7, rng, verbose=True)
        do_try_update(a_out, f"long_n{n_long}")

    # Final check
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
