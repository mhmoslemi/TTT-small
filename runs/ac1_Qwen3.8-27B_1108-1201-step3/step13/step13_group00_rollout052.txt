def normalize(a, target_sum=None):
    """Normalize a to have the given target sum (default: sqrt(2n))."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    n = len(a)

def build_conv_matrix_sparse(a):
    """Build the convolution matrix as a sparse-like structure.
    Returns a function that computes T @ b efficiently."""
    n = len(a)
    # T[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # This is a banded matrix. For efficiency, we'll use FFT-based multiplication.
    
    def matvec(b):
        """Compute T @ b = conv(a, b) (full convolution)."""
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        A = rfft(a, fft_size)
        B = rfft(b, fft_size)
        return irfft(A * B, fft_size)[:size]
    
    def matvec_transpose(y):
        """Compute T^T @ y.
        (T^T y)_j = sum_k T[k,j] y_k = sum_k a[k-j] y_k (valid k)
        = sum_k a[k-j] y_k for k in [j, j+n-1]
        This is the cross-correlation of y with a, evaluated at lag j."""
        n = len(a)
        size = len(y)
        fft_size = 1 << (size - 1).bit_length()
        # (T^T y)_j = sum_k a[k-j] * y_k for valid k
        # = sum_k a[k-j] * y_k
        # Let m = k-j: sum_m a[m] * y[m+j] = cross-correlation of y with a at lag j
        # Using FFT: ifft(FFT(a) * conj(FFT(y))) gives cross-correlation
        A = rfft(a, fft_size)
        Y = rfft(y, fft_size)
        corr = irfft(A * np.conj(Y), fft_size)
        # corr[j] = sum_k a[k] * y[k-j] ... hmm, need to be careful
        # Actually: ifft(FFT(a) * conj(FFT(y))) gives the correlation
        # corr[m] = sum_k a[k+m] * y[k] (circular)
        # We want (T^T y)_j = sum_k a[k-j] * y_k
        # Let l = k-j: sum_l a[l] * y[l+j]
        # This is corr at... if corr[m] = sum_l a[l] * y[l-m], then we want corr at m = -j
        # For circular: corr[-j] = corr[fft_size - j]
        # But we need the linear (non-circular) version.
        # If fft_size >= 2n, the linear and circular are the same for valid outputs.
        return corr[fft_size - n + 1:fft_size + 1]  # Hmm, this might not be right
        
    return matvec, matvec_transpose


def lp_best_response(a, M, solver, active_set=None):
    """Solve the LP: max sum(b) s.t. conv(a,b) <= M, b >= 0.
    Uses constraint screening for efficiency."""
    n = len(a)
    size = 2 * n - 1
    
    b_var = cp.Variable(n, nonneg=True)
    
    # Compute current conv to identify active constraints
    # conv(a, b) at position k = sum_j a_j * b[k-j]
    # We need to build the constraint matrix rows for the active set
    
    # For screening: use the current a to identify which constraints are likely binding
    # A constraint k is "likely binding" if conv(a,a)_k is close to M
    conv_aa = fast_autocorr(a)
    
    if active_set is not None and len(active_set) > 0:
        # Use the provided active set
        ks = active_set
    else:
        # Find constraints where conv(a,a) is close to max
        threshold = M * 0.8
        ks = np.where(conv_aa >= threshold)[0]
        if len(ks) < 5:
            threshold = M * 0.5
            ks = np.where(conv_aa >= threshold)[0]
        if len(ks) < 3:
            ks = np.arange(size)  # Fallback to all
    
    # Build constraint matrix for selected rows
    # Row k of T: T[k, j] = a[k-j] if 0 <= k-j < n
    # T_sub is (len(ks), n)
    T_sub = np.zeros((len(ks), n))
    for idx, k in enumerate(ks):
        for j in range(n):
            val = k - j
            if 0 <= val < n:
                T_sub[idx, j] = a[val]
    
    constraints = [T_sub @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    
    # Verify: check if any dropped constraints are violated
    full_conv = fast_autocorr_helper(a, b_val, size)
    violated = np.where(full_conv > M * (1 + 1e-6))[0]
    
    if len(violated) > 0:
        # Need to re-solve with expanded active set
        new_ks = np.sort(np.union1d(ks, violated))
        if len(new_ks) > 0.8 * size:
            # Just use all constraints
            T_full = np.zeros((size, n))
            for j in range(n):
                T_full[j:j+n, j] = a
            b_var2 = cp.Variable(n, nonneg=True)
            prob2 = cp.Problem(cp.Maximize(cp.sum(b_var2)), [T_full @ b_var2 <= M])
            try:
                prob2.solve(solver=solver, verbose=False)
            except Exception:
                return b_val, ks
            if prob2.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.maximum(np.asarray(b_var2.value, dtype=np.float64), 0.0)
            return b_val, None
        else:
            T_sub2 = np.zeros((len(new_ks), n))
            for idx, k in enumerate(new_ks):
                for j in range(n):
                    val = k - j
                    if 0 <= val < n:
                        T_sub2[idx, j] = a[val]
            b_var2 = cp.Variable(n, nonneg=True)
            prob2 = cp.Problem(cp.Maximize(cp.sum(b_var2)), [T_sub2 @ b_var2 <= M])
            try:
                prob2.solve(solver=solver, verbose=False)
            except Exception:
                return b_val, ks
            if prob2.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.maximum(np.asarray(b_var2.value, dtype=np.float64), 0.0)
            return b_val, new_ks
    
    return b_val, ks


def fast_autocorr_helper(a, b, size=None):
    """Compute full convolution of a and b."""
    n = len(a)
    if size is None:

    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)

def lp_fw_nesterov(a0, time_limit, solver, verbose=True):
    """LP best-response Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = normalize(a0)
    
    best_a = a.copy()
    best_max = fast_autocorr(a).max()
    
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    active_set = None
    iter_count = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute M at current y
        conv_y = fast_autocorr(y)
        M = conv_y.max()
        if M < 1e-12:
            break
        
        # Solve LP best-response
        b, active_set = lp_best_response(y, M, solver, active_set)
        if b is None or b.sum() < 1e-15:
            active_set = None
            continue
        
        b = normalize(b)
        iter_count += 1
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * b
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize(y_new)
        
        if t == 1:
            a_new = b.copy()
        else:
            theta = (t - 1.0) / (t + 2.0)
            a_new = y_new + theta * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new)
        
        y = y_new
        a = a_new
        
        # Check improvement
        score_y = fast_autocorr(y).max()
        if score_y < best_max:
            best_max = score_y
            best_a = y.copy()
        
        score_a = fast_autocorr(a).max()
        if score_a < best_max:
            best_max = score_a
            best_a = a.copy()
        
        # Running average
        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = normalize(avg_a / n_avg)
            score_avg = fast_autocorr(avg_check).max()
            if score_avg < best_max:
                best_max = score_avg
                best_a = avg_check.copy()
        
        if verbose and iter_count % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): best_max={best_max:.8f}")
    
    return best_a, best_max

**Idea: Use the LP to get a direction, then do a line search**

Instead of the FW step (go towards the LP solution with diminishing step size), we can:

3. Update: a <- (1-alpha)*a + alpha*b

This is the "accelerated conditional gradient" (ACGBL) or "Greedy_FW" approach. The line search can be done with a few evaluations of f (which is just computing the max of the conv).

Actually, even better: since b is the best response (maximizes sum(b) subject to conv(a,b) <= M), and we normalize both a and b to sum = sqrt(2n), the line search is:
minimize f((1-alpha)*a + alpha*b) over alpha in [0, 1]

where f(x) = max_k conv(x,x)_k.

This is a 1D problem and can be solved with golden section search or just evaluating at a few points.

Actually, the standard Nesterov FW (also known as FISTA-FW or accelerated FW) works as follows:
- y_t = x_t + (t-1)/(t+2) * (x_t - x_{t-1})  [extrapolation]
- s_t = argmin_{s in C} <∇f(y_t), s>  [linearized step, here LP oracle]
- x_{t+1} = (1 - 2/(t+2)) * y_t + 2/(t+2) * s_t  [convex combination]

The key property is that this gives O(1/t^2) convergence for smooth objectives. For the max-of-linear (convex but not smooth) case, the convergence is O(1/t) for the gap, but the Nesterov variant still helps in practice.

The issue with the LP oracle FW is that the "subproblem" (LP) might not give a good direction if the active set is wrong. Let me make sure the active set handling is robust.

**Let me now think about the pair CD more carefully:**

After the LP-FW converges, we're at a point where no single LP step improves. But coordinated changes to multiple coordinates might help.

The most natural "coordinated change" is the pair swap: move mass from coordinate i to coordinate j. The optimal swap can be found by 1D optimization.

For a systematic approach:
1. Compute the subgradient g (at the max convolution point)
2. Coordinates with high g "want" to decrease, low g can increase
3. Try pairs (i, j) where g_i is high and g_j is low
4. For each such pair, find the optimal delta

But trying all O(n^2) pairs is too expensive. Instead:
- For each i (in order of decreasing g_i), try the top K candidates for j (in order of increasing g_j)
- K can be 10-50

This gives O(n*K) pair evaluations per sweep, which for n=1600 and K=20 is 32000 evaluations. Each evaluation involves computing the change in conv at a few positions, which is O(n) if done naively. Total: O(n^2 * K) per sweep = 1600^2 * 20 = 51M operations. That's fast in numpy.

Actually, for each pair (i,j), we need to evaluate max_k (conv_k + delta * g_k) at the optimal delta. If we precompute g_k for the pair, the evaluation is O(2n) (just finding the max). So total is O(n * K * 2n) = O(n^2 * K) which is fine.

def systematic_pair_cd(a0, time_limit, verbose=True):
    """Systematic pair coordinate descent guided by subgradient."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_max = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    norm_val = np.sqrt(2.0 * n)
    
    sweep = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        sweep_improved = False
        
        conv = fast_autocorr(a)
        max_pos = np.argmax(conv)
        M = conv[max_pos]
        
        # Compute subgradient at max
        g = np.zeros(n)
        for i in range(n):
            v = 0.0
            j1 = max_pos - i
            if 0 <= j1 < n:
                v += a[j1]
            j2 = i - max_pos
            if 0 <= j2 < n:
                v += a[j2]
            g[i] = v
        # g[i] is the subgradient component (times 2, but we just use it for ranking)
        
        # Rank: high g -> wants to lose mass, low g -> can gain
        i_order = np.argsort(-g)  # decreasing g
        j_order = np.argsort(g)   # increasing g
        
        K = min(30, n // 2)  # Try top K j's for each i
        
        for ii in range(min(200, n)):  # Limit number of i's per sweep for time
            if time.monotonic() - t0 > time_limit:
                break
            i = i_order[ii]
            if a[i] < 1e-10:
                continue
            
            for jj in range(K):
                if time.monotonic() - t0 > time_limit:
                    break
                j = j_order[jj]
                if j == i:
                    continue
                
                # Compute the gain vector for transferring from i to j
                # delta_k = 2*delta*(a_{k-j}^+ - a_{k-i}^+) + second_order
                gk = np.zeros(size)
                # a_{k-j}^+ for k in [j, j+n-1]: a[k-j]
                lo1 = j
                hi1 = min(size, j + n)
                gk[lo1:hi1] += a[0:hi1-lo1]
                # a_{k-i}^+ for k in [i, i+n-1]: a[k-i]
                lo2 = i
                hi2 = min(size, i + n)
                gk[lo2:hi2] -= a[0:hi2-lo2]
                gk *= 2.0  # The factor of 2
                
                # Second order: +delta^2 at k=2i, k=2j; -2*delta^2 at k=i+j
                c = np.zeros(size)
                if 2*i < size:
                    c[2*i] += 1.0
                if 2*j < size:
                    c[2*j] += 1.0
                if i+j < size:
                    c[i+j] -= 2.0
                
                # Find optimal delta in [0, a[i]]
                # Objective: max_k (conv_k + delta * gk_k + delta^2 * c_k)
                # Try a few values
                ai = a[i]
                best_d = 0.0
                best_val = M
                
                # Sample delta values
                n_samples = 15
                for d in np.linspace(ai * 0.05, ai * 0.95, n_samples):
                    new_max = (conv + d * gk + d * d * c).max()
                    if new_max < best_val:
                        best_val = new_max
                        best_d = d
                
                if best_d > 1e-12:
                    a[i] -= best_d
                    a[j] += best_d
                    a = np.maximum(a, 0.0)
                    a = normalize(a)
                    new_conv_max = fast_autocorr(a).max()
                    if new_conv_max < best_max - 1e-12:
                        best_max = new_conv_max
                        best_a = a.copy()
                        sweep_improved = True
                    else:
                        # Revert
                        a[j] -= best_d
                        a[i] += best_d
                        a = normalize(a)
        
        if not sweep_improved:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0
        
        if verbose and sweep % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"  PCD sweep={sweep} ({elapsed:.0f}s): best={best_max:.8f}")
    
    return best_a, best_max

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def full_conv(a, b):
    n1 = len(a)
    n2 = len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def normalize_to(a, target):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * target / len(a)
    return a * target / s


def build_conv_matrix(a, rows=None):
    """Build T matrix where T[k,j] = a[k-j] if 0<=k-j<n else 0.
    If rows given, only build those rows."""
    n = len(a)
    size = 2 * n - 1
    if rows is None:
        rows = np.arange(size)
    T = np.zeros((len(rows), n))
    for idx, k in enumerate(rows):
        for j in range(n):
            val = k - j
            if 0 <= val < n:
                T[idx, j] = a[val]
    return T


def lp_best_response(a, M, solver, active_set=None):
    """Solve LP: max sum(b) s.t. conv(a,b)_k <= M for all k, b >= 0."""
    n = len(a)
    size = 2 * n - 1

    b_var = cp.Variable(n, nonneg=True)

    conv_aa = fast_autocorr(a)

    if active_set is not None and len(active_set) > 0 and len(active_set) < size:
        ks = active_set
    else:
        # Dynamic threshold: start tight, expand if needed
        threshold = M * 0.7
        ks = np.where(conv_aa >= threshold)[0]
        if len(ks) < 5:
            threshold = M * 0.3
            ks = np.where(conv_aa >= threshold)[0]
        if len(ks) < 3:
            ks = np.arange(size)

    T_sub = build_conv_matrix(a, ks)
    constraints = [T_sub @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None

    # Verify: check if dropped constraints are violated
    if len(ks) < size:
        full_cb = full_conv(a, b_val)
        violated = np.where(full_cb > M * (1 + 1e-5))[0]
        if len(violated) > 0:
            new_ks = np.sort(np.union1d(ks, violated))
            if len(new_ks) >= size * 0.9:
                T_full = build_conv_matrix(a)
                b_var2 = cp.Variable(n, nonneg=True)
                prob2 = cp.Problem(cp.Maximize(cp.sum(b_var2)), [T_full @ b_var2 <= M])
                try:
                    prob2.solve(solver=solver, verbose=False)
                except Exception:
                    return b_val, None
                if prob2.status in ('optimal', 'optimal_inaccurate'):
                    b_val = np.maximum(np.asarray(b_var2.value, dtype=np.float64), 0.0)
                return b_val, None
            else:
                T_sub2 = build_conv_matrix(a, new_ks)
                b_var2 = cp.Variable(n, nonneg=True)
                prob2 = cp.Problem(cp.Maximize(cp.sum(b_var2)), [T_sub2 @ b_var2 <= M])
                try:
                    prob2.solve(solver=solver, verbose=False)
                except Exception:
                    return b_val, new_ks
                if prob2.status in ('optimal', 'optimal_inaccurate'):
                    b_val = np.maximum(np.asarray(b_var2.value, dtype=np.float64), 0.0)
                return b_val, new_ks

    return b_val, ks


def lp_fw_nesterov(a0, time_limit, solver, verbose=True):
    """LP best-response Frank-Wolfe with Nesterov acceleration and line search."""
    n = len(a0)
    target = np.sqrt(2.0 * n)
    a = normalize_to(a0, target)
    best_a = a.copy()
    best_max = fast_autocorr(a).max()

    y = a.copy()
    prev_a = a.copy()
    active_set = None
    t0 = time.monotonic()
    iter_count = 0
    avg_a = np.zeros(n)
    n_avg = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        b, active_set = lp_best_response(y, M, solver, active_set)
        if b is None:
            active_set = None
            continue
        b = normalize_to(b, target)
        iter_count += 1

        # Nesterov extrapolation for the step
        if t == 1:
            alpha_fw = 1.0
            y_step = b
        else:
            theta = (t - 1.0) / (t + 2.0)
            y_step = y + theta * (y - prev_a)
            y_step = np.maximum(y_step, 0.0)
            y_step = normalize_to(y_step, target)

            # Re-solve LP with extrapolated y for better direction
            conv_ys = fast_autocorr(y_step)
            M_ys = conv_ys.max()
            b2, active_set = lp_best_response(y_step, M_ys, solver, active_set)
            if b2 is not None and b2.sum() > 1e-15:
                b = normalize_to(b2, target)

        alpha_fw = 2.0 / (t + 2.0)
        a_new = (1.0 - alpha_fw) * y + alpha_fw * b
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_to(a_new, target)

        # Line search refinement: try a few step sizes
        best_alpha = alpha_fw
        best_f = fast_autocorr(a_new).max()
        for test_alpha in [min(1.0, alpha_fw * 2), min(1.0, alpha_fw * 3), 0.5, 1.0]:
            if abs(test_alpha - alpha_fw) < 1e-6:
                continue
            candidate = (1.0 - test_alpha) * y + test_alpha * b
            candidate = np.maximum(candidate, 0.0)
            candidate = normalize_to(candidate, target)
            f_val = fast_autocorr(candidate).max()
            if f_val < best_f:
                best_f = f_val
                best_alpha = test_alpha
                a_new = candidate

        prev_a = a
        a = a_new
        y = a.copy()

        # Track best
        score_a = fast_autocorr(a).max()
        if score_a < best_max:
            best_max = score_a
            best_a = a.copy()

        # Running average
        avg_a += a
        n_avg += 1
        if n_avg % 100 == 0:
            avg_check = normalize_to(avg_a / n_avg, target)
            sc = fast_autocorr(avg_check).max()
            if sc < best_max:
                best_max = sc
                best_a = avg_check.copy()

        if verbose and iter_count % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FW iter={iter_count} ({elapsed:.0f}s): best={best_max:.8f}")

    return best_a, best_max


def systematic_pair_cd(a0, time_limit, verbose=True):
    """Subgradient-guided systematic pair coordinate descent."""
    n = len(a0)
    target = np.sqrt(2.0 * n)
    a = normalize_to(a0, target)
    best_a = a.copy()
    best_max = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    sweep = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        if no_improve >= 8:
            break
        sweep_improved = False

        conv = fast_autocorr(a)
        M = conv.max()
        max_pos = int(np.argmax(conv))

        # Compute subgradient at max
        g = np.zeros(n)
        for i in range(n):
            v = 0.0
            j1 = max_pos - i
            if 0 <= j1 < n:
                v += a[j1]
            j2 = i - max_pos
            if 0 <= j2 < n:
                v += a[j2]
            g[i] = v

        # Also consider all near-max positions for a richer subgradient
        near_max_thresh = M * 0.95
        near_max = np.where(conv >= near_max_thresh)[0]
        g_multi = np.zeros(n)
        for kp in near_max:
            for i in range(n):
                v = 0.0
                j1 = kp - i
                if 0 <= j1 < n:
                    v += a[j1]
                j2 = i - kp
                if 0 <= j2 < n:
                    v += a[j2]
                g_multi[i] += v
        g = g_multi / max(1, len(near_max))

        # Rank coordinates
        i_order = np.argsort(-g)
        j_order = np.argsort(g)

        K = min(25, n // 3)
        max_i_per_sweep = min(300, n)

        for ii in range(max_i_per_sweep):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(i_order[ii])
            if a[i] < target * 1e-9:
                continue

            ai = a[i]
            # Precompute gain vector for removing from i
            gain_i = np.zeros(size)
            lo = i
            hi = min(size, i + n)
            gain_i[lo:hi] = a[0:hi - lo]
            # Mirror: a_{i-k} for k in [0, n-1] -> position i-k... wait
            # The change in conv_k from a_i -= delta:
            # -2*delta * a_{k-i}^+ (for k != 2i)
            # At k=2i: -2*delta*a_i + delta^2
            # So gain from removing i: -a_{k-i}^+ - a_{i-k}^+ (factor of 2 absorbed)
            # a_{k-i}^+: nonzero for k in [i, i+n-1], value a[k-i]
            # a_{i-k}^+: nonzero for k in [i-n+1, i], value a[i-k]
            # Since k >= 0: a_{i-k}^+ nonzero for k in [max(0,i-n+1), min(i, n-1)]
            lo2 = max(0, i - n + 1)
            hi2 = min(i + 1, n)
            if hi2 > lo2:
                gain_i[lo2:hi2] += a[i - np.arange(lo2, hi2)]

            for jj in range(K):
                if time.monotonic() - t0 > time_limit:
                    break
                j = int(j_order[jj])
                if j == i:
                    continue

                # Gain from adding to j (positive contribution)
                gain_j = np.zeros(size)
                lo = j
                hi = min(size, j + n)
                gain_j[lo:hi] = a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(j + 1, n)
                if hi2 > lo2:
                    gain_j[lo2:hi2] += a[j - np.arange(lo2, hi2)]

                # Net gain vector: delta * (gain_j - gain_i)
                gk = gain_j - gain_i

                # Second order corrections
                c = np.zeros(size)
                if 2 * i < size:
                    c[2 * i] = 1.0
                if 2 * j < size:
                    c[2 * j] += 1.0
                if i + j < size:
                    c[i + j] -= 2.0

                # Find best delta in [0, ai]
                best_d = 0.0
                best_val = M
                # Coarse grid
                n_samples = 12
                for d in np.linspace(ai * 0.1, ai * 0.9, n_samples):
                    new_max = (conv + d * gk + d * d * c).max()
                    if new_max < best_val:
                        best_val = new_max
                        best_d = d

                if best_d > 1e-10:
                    # Refine around best_d
                    d_lo = max(0, best_d * 0.8)
                    d_hi = min(ai, best_d * 1.2)
                    for d in np.linspace(d_lo, d_hi, 8):
                        new_max = (conv + d * gk + d * d * c).max()
                        if new_max < best_val:
                            best_val = new_max
                            best_d = d

                    if best_d > 1e-10 and best_val < M - 1e-13:
                        a[i] -= best_d
                        a[j] += best_d
                        a = np.maximum(a, 0.0)
                        a = normalize_to(a, target)
                        new_conv_max = fast_autocorr(a).max()
                        if new_conv_max < best_max - 1e-13:
                            best_max = new_conv_max
                            best_a = a.copy()
                            sweep_improved = True
                            conv = fast_autocorr(a)
                            M = conv.max()
                            # Recompute subgradient occasionally
                            if sweep_improved and (ii + jj) % 50 == 0:
                                max_pos = int(np.argmax(conv))
                                for i2 in range(n):
                                    v = 0.0
                                    j1 = max_pos - i2
                                    if 0 <= j1 < n:
                                        v += a[j1]
                                    j2 = i2 - max_pos
                                    if 0 <= j2 < n:
                                        v += a[j2]
                                    g[i2] = v

        if not sweep_improved:
            no_improve += 1
        else:
            no_improve = 0

        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD sweep={sweep} ({elapsed:.0f}s): best={best_max:.8f}")

    return best_a, best_max
