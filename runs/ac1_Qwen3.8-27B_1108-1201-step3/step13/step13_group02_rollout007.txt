    fft_size = 1 << (2*n - 1).bit_length()
    a = a.copy()
    for i in range(n_iter):

        # Blend: A_new = sqrt(target) * exp(1j*angle(A))

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Water filling
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - 1) / rho > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho_hat = np.max(rho[cond])
    theta = (cssv[rho_hat - 1] - 1) / rho_hat
    return np.maximum(v - theta, 0)


def frequency_flatten(a0, n_iter=300, relaxation=0.3, verbose=False):
    """Gerchberg-Saxton style: iterate between flat spectrum target and non-negativity."""
    a = normalize_sum1(a0)
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    for i in range(n_iter):
        A = rfft(a, fft_size)
        mag = np.abs(A)
        # Target: flat magnitude
        target_mag = np.full_like(mag, mag.mean())
        # Relaxed update
        new_mag = (1 - relaxation) * mag + relaxation * target_mag
        # Handle zeros in phase
        phase = np.ones_like(A)
        mask = mag > 1e-15
        phase[mask] = A[mask] / mag[mask]
        A_new = new_mag * phase
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:

        a = a_new
        if verbose and i % 100 == 0:
            print(f"    FreqFlatten iter={i}: score={score:.8f}, best={best_score:.8f}")
    
    return best_a, best_score

    """
    n = len(y)
    row = np.zeros(n)
    # j ranges from max(0, k-n+1) to min(k, n-1)
    j_start = max(0, k - n + 1)
    j_end = min(k, n - 1)
    if j_start <= j_end:
        row[j_start:j_end+1] = y[k - j_end:k - j_start + 1][::-1]
    return row


def build_full_conv_matrix(y):
    """Build the full (2n-1) x n matrix T_y where (T_y @ b) = conv(y, b)."""
    n = len(y)
    T = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            T[k, j_start:j_end+1] = y[k - j_end:k - j_start + 1][::-1]
    return T


def lp_best_response(y, M, solver, max_constraints=None):
    """Solve LP: max sum(b) s.t. conv(y,b) <= M, b >= 0."""
    n = len(y)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    T = build_full_conv_matrix(y)
    
    if max_constraints is not None and max_constraints < size:
        # Use only the top constraints (where y's autocorrelation is largest)
        y_ac = fast_autocorr(y)
        top_idx = np.argsort(y_ac)[::-1][:max_constraints]
        constraints = [T[top_idx, :] @ b_var <= M[top_idx]]
    else:
        constraints = [T @ b_var <= M]
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """LP Nesterov acceleration for minimizing max autocorrelation."""
    n = len(a0)
    a = normalize_sum1(a0)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    y = a.copy()

    n_iters = 0

        M_curr = fast_autocorr(y)
        max_M = M_curr.max()
        if max_M < 1e-12:
            break
        
        # First 30 iters: full LP; after: reduced
        if t < 30:
            b = lp_best_response(y, M_curr, solver)
        else:
            # Reduced: use only constraints where y_ac is close to max
            tol = 0.5 * max_M
            active = M_curr >= tol
            n_active = max(int(active.sum()), 50)
            b = lp_best_response(y, M_curr, solver, max_constraints=n_active)
        
        if b is None:
            stall += 1
            if stall > 100:
                # Perturb
                rng = np.random.default_rng(t * 13 + 7)
                y = normalize_sum1(y + rng.normal(0, 0.01 * y.mean(), n))
                x = y.copy()
                stall = 0
            continue
        
        b = normalize_sum1(b)
        n_iters += 1
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * b
        y_new = normalize_sum1(y_new)
        
        if t == 1:
            x_new = b.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            x_new = y_new + momentum * (y_new - y)
            x_new = normalize_sum1(x_new)
        
        y = y_new
        x = x_new
        
        # Track best
        score_y = fast_autocorr(y).max()
        score_x = fast_autocorr(x).max()
        best_iter_score = min(score_y, score_x)
        
        if best_iter_score < best_score:
            best_score = best_iter_score
            best_a = y.copy() if score_y < score_x else x.copy()
            stall = 0
        else:
            stall += 1
        
        # Iterate averaging (for subgradient-like convergence)
        if t % 50 == 0:
            # Check a running average
            pass  # The Nesterov x already provides averaging-like behavior
        
        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, stall={stall}")

def compute_subgradient(a, k):
    """Compute db[k]/da where b = conv(a,a).
    db[k]/da[j] = a[k-j] + a[j-k] (with a[m]=0 for m outside [0,n-1])
    """
    n = len(a)
    grad = np.zeros(n)
    # a[k-j] for j=0..n-1: non-zero when 0 <= k-j < n => k-n+1 <= j <= k
    j_start1 = max(0, k - n + 1)
    j_end1 = min(n - 1, k)
    if j_start1 <= j_end1:
        grad[j_start1:j_end1+1] += a[k - j_end1 : k - j_start1 + 1][::-1]
    
    # a[j-k] for j=0..n-1: non-zero when 0 <= j-k < n => k <= j <= k+n-1
    j_start2 = max(0, k)
    j_end2 = min(n - 1, k + n - 1)
    if j_start2 <= j_end2:
        grad[j_start2:j_end2+1] += a[j_start2 - k : j_end2 - k + 1]
    
    return grad


def subgradient_descent(a0, time_limit, verbose=True):
    """Subgradient descent with iterate averaging."""
    n = len(a0)
    a = normalize_sum1(a0)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    t0 = time.monotonic()
    avg_a = np.zeros(n)
    n_avg = 0
    iterates = []
    
    # Use polyak step size approximation
    f_star_est = best_score * 0.5  # crude lower bound
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(a)
        max_b = b.max()
        
        # Use top-3 peaks for multi-subgradient
        top_k = np.argsort(b)[::-1][:3]
        weights = np.exp(-(b[top_k] - max_b) / (max_b * 0.1 + 1e-10))
        weights = weights / weights.sum()
        
        grad = np.zeros(n)
        for w, k in zip(weights, top_k):
            grad += w * compute_subgradient(a, k)
        
        # Step size
        grad_norm_sq = np.dot(grad, grad)
        if grad_norm_sq < 1e-30:
            break
        eta = (max_b - f_star_est) / grad_norm_sq
        eta = min(eta, 10.0 / t)  # cap step size
        
        a_new = a - eta * grad
        a_new = project_simplex(a_new)
        a = a_new
        
        # Update average
        avg_a += a
        n_avg += 1
        avg_a_norm = avg_a / n_avg
        
        score = fast_autocorr(a).max()
        avg_score = fast_autocorr(avg_a_norm).max()
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        if avg_score < best_score:
            best_score = avg_score
            best_a = avg_a_norm.copy()
        
        # Update f_star estimate
        f_star_est = min(f_star_est, best_score * 0.9)
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad iter={t} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def lse_gradient_descent(a0, time_limit, verbose=True):
    """LSE smooth gradient descent."""
    n = len(a0)
    a = normalize_sum1(a0)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0
    
    T_start = max(0.5, best_score * 0.1)
    T_end = max(0.01, best_score * 0.005)
    
    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress
        
        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break
        
        w = np.exp((b - max_b) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum
        
        # Gradient using FFT
        # grad[j] = 2 * sum_k w[k] * (a[k-j] + a[j-k]) / 2 = sum_k w[k] * (a[k-j] + a[j-k])
        # = conv(w, a_flip) + conv(w_flip, a) ... let me use the direct formula
        # grad[j] = sum_k w[k] * (a[k-j] + a[j-k])
        # = sum_k w[k] * a[k-j] + sum_k w[k] * a[j-k]
        # First term: (w * a)[j] where we define a[m]=0 outside... 
        # Actually: sum_k w[k] * a[k-j] = (w convolved with a_reversed)[j + n - 1]... 
        # Let me just use the FFT approach
        # sum_k w[k] * a[k-j]: let m = k-j, so k = m+j. sum_m w[m+j] * a[m]
        #   = cross-correlation of w and a at lag j
        # sum_k w[k] * a[j-k]: let m = j-k, so k = j-m. sum_m w[j-m] * a[m]
        #   = convolution of w and a at position j
        
        # Using FFT:
        fft_size = 1 << (2 * n - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        # conv(w, a) = ifft(W * A)
        conv_wa = irfft(W * A, fft_size)  # length 2n-1
        # cross_correlation: sum_k w[k] * a[k-j] = (w * a_flipped)[j + n - 1]
        a_flip = a[::-1]
        AF = rfft(a_flip, fft_size)
        corr = irfft(W * AF, fft_size)  # this gives sum_k w[k] * a_flip[k-j]... 
        # Actually let me just compute the gradient directly for the top peaks
        # Since w is concentrated on a few peaks, this is efficient
        
        # Direct computation using top peaks
        n_top = min(n, max(10, int(0.1 * len(w))))
        top_idx = np.argsort(w)[::-1][:n_top]
        grad = np.zeros(n)
        for k in top_idx:
            if w[k] < 1e-15:
                continue
            grad += w[k] * compute_subgradient(a, k)
        
        # Step size
        grad_norm_sq = np.dot(grad, grad)
        if grad_norm_sq < 1e-30:
            break
        eta = 1.0 / (T * np.sqrt(n) * 0.5)
        eta = min(eta, max_b / (grad_norm_sq + 1e-30))
        
        a_new = a - eta * grad
        a_new = project_simplex(a_new)
        a = a_new
        steps += 1
        
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if verbose and steps % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-GD step={steps} ({elapsed:.0f}s): T={T:.4f}, best={best_score:.8f}")
    
    return best_a, best_score


def peak_shave(a0, time_limit, seed=42, verbose=True):
    """Peak shaving with simulated annealing."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize_sum1(a0)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    T_start = max(0.3, best_score * 0.05)
    T_end = best_score * 0.001
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** elapsed_frac
        
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:5]
        
        improved = False
        for k in top_k:
            # Find the pair (i, j) that contributes most to b[k]
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-18:
                continue
            src_local = np.argmax(contrib)
            src = idx[src_local]
            dst_in_pair = k - src
            
            # Find a good destination (low autocorrelation contribution)
            # Prefer positions that don't contribute to any top peak
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js = max(0, k2 - n + 1)
                je = min(n - 1, k2)
                peak_mask[js:je+1] = True
            
            non_peak = np.where(~peak_mask)[0]
            if len(non_peak) == 0:
                non_peak = np.where(np.arange(n) != src)[0]
            
            # Pick destination with lowest value
            dst = non_peak[np.argmin(a[non_peak])]
            
            # Transfer fraction
            frac = rng.choice([0.05, 0.1, 0.15, 0.2, 0.3])
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            
            score = fast_autocorr(a_trial).max()
            
            if score < best_score:
                a = a_trial
                best_score = score
                best_a = a_trial.copy()
                improved = True
                break
            elif rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                a = a_trial
                improved = True
                break
        
        if not improved:
            stall += 1
            if stall > 100:
                # Larger perturbation
                n_pert = max(2, n // 100)
                idx_pert = rng.choice(n, size=n_pert, replace=False)
                for idx in idx_pert:
                    if a[idx] < 1e-15:
                        continue
                    j = rng.integers(n)
                    amt = a[idx] * rng.uniform(0.2, 0.5)
                    a[idx] -= amt
                    a[j] += amt
                a = normalize_sum1(a)
                stall = 0
        
        if verbose:
            n_done = int((time.monotonic() - t0) * 10)
            if n_done % 5000 == 0 and n_done > 0:
                elapsed = time.monotonic() - t0
                print(f"    PeakShave ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def coordinate_descent(a0, n_sweeps=10, verbose=False):
    """Coordinate descent: for each coordinate, find the optimal value."""
    n = len(a0)
    a = normalize_sum1(a0)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    for sweep in range(n_sweeps):
        for i in range(n):
            # Try different values for a[i]
            # Keep all other coordinates fixed
            a_save = a[i]
            # The contribution of a[i] to b[k] is: a[i] * (a[k-i] + a[i-k]) for each k
            # We want to choose a[i] to minimize max_k (b[k] + a[i] * (a[k-i] + a[i-k] - a_save * (a[k-i] + a[i-k])))
            # = minimize max_k (b[k] + (a[i] - a_save) * (a[k-i] + a[i-k]))
            
            # Try a few candidate values
            candidates = [0.0, a_save * 0.5, a_save * 0.8, a_save, a_save * 1.2, a_save * 1.5, a_save * 2.0]
            candidates = [max(0.0, c) for c in candidates]
            
            best_val = a_save
            best_local_score = fast_autocorr(a).max()
            
            for cv in candidates:
                a[i] = cv
                # Normalize
                s = a.sum()
                if s < 1e-15:
                    a[i] = a_save
                    break
                a_norm = a / s
                score = fast_autocorr(a_norm).max()
                if score < best_local_score:
                    best_local_score = score
                    best_val = cv
            a[i] = best_val
            a = normalize_sum1(a)
        
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if verbose:
            print(f"    CoordDescent sweep={sweep}: best={best_score:.8f}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    return normalize_sum1(f_interp)
