def _get_solver():

    if _global_solver is not None:
        return _global_solver

def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def _proj_simplex(v):
    n = len(v)
    if n <= 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 0:
        out /= s
    else:
        out = np.ones(n) / n
    return out


def _lse_fista(a0, T_schedule, time_limit, verbose=False):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()

    for stage_i, T in enumerate(T_schedule):
        if time.monotonic() - t0 > time_limit:
            break
        L = max(n / (T * 2.0), n / 50.0)
        y = a.copy()
        t_fista = 1
        staging = a.copy()
        max_iters = 20000

        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_score

            grad = _lse_grad(y, T)
            step = 1.0 / L
            z = y - step * grad
            a_new = _proj_simplex(z)

            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - staging)
                y_new = np.maximum(y_new, 0.0)
                s_y = y_new.sum()
                if s_y > 1e-15:
                    y_new = y_new / s_y
                else:
                    y_new = a_new.copy()

            staging = a_new.copy()
            a = a_new
            y = y_new
            t_fista += 1

            if it % 200 == 0:
                sc = _fast_autocorr(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

        sc = _fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE-FISTA T={T:.2f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def _lse_grad(y, T):
    n = len(y)
    ac = _fast_autocorr(y)
    ac_max = ac.max()
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()

    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(y, fft_size)
    W = rfft(w, fft_size)
    cross = irfft(np.conj(A) * W, fft_size)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad


def _build_conv_matrix(a, n):
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j: j + n, j] = a
    return T


def _lp_best_response(a, M, solver, prev_b=None):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = _build_conv_matrix(a, n)
    constraints = [T @ b_var <= M]
    objective = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(objective, constraints)
    try:
        if prev_b is not None:
            b_var.value = prev_b
        prob.solve(solver=solver, verbose=False, warm_start=True)
    except Exception:
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def _lp_nesterov(a0, time_limit, solver, verbose=True, use_adaptive_lpfreq=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    prev_b = None
    lp_count = 0
    subgrad_count = 0
    stagnant = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Decide: LP or subgradient step
        b_curr = _fast_autocorr(y)
        M_curr = b_curr.max()
        if M_curr < 1e-12:
            break

        # Adaptive LP frequency: solve LP more often early, less often later
        if use_adaptive_lpfreq:
            if n_iters < 50:
                do_lp = True
            elif n_iters < 200:
                do_lp = (n_iters % 2 == 0)
            elif n_iters < 500:
                do_lp = (n_iters % 3 == 0)
            else:
                do_lp = (n_iters % 5 == 0)
        else:
            do_lp = True

        if do_lp:
            b = _lp_best_response(y, M_curr, solver, prev_b)
            if b is not None:
                prev_b = b.copy()
                lp_count += 1
            else:
                do_lp = False

        if do_lp and b is not None:
            b_sum = b.sum()
            if b_sum < 1e-15:
                g = y.copy()
            else:
                g = b * norm / b_sum
        else:
            # Subgradient direction (Frank-Wolfe style)
            k_star = int(np.argmax(b_curr))
            grad = np.zeros(n)
            for i in range(n):
                j1 = k_star - i
                if 0 <= j1 < n:
                    grad[i] += 2.0 * y[j1]
                j2 = i - k_star
                if 0 <= j2 < n:
                    grad[i] += 2.0 * y[j2]
            i_min = int(np.argmin(grad))
            v = np.zeros(n)
            v[i_min] = norm
            g = v
            subgrad_count += 1

        n_iters += 1
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = _fast_autocorr(y).max()
        if score_y < best_score - 1e-10:
            best_score = score_y
            best_a = y.copy()
            stagnant = 0
        else:
            stagnant += 1

        score_a = _fast_autocorr(a).max()
        if score_a < best_score - 1e-10:
            best_score = score_a
            best_a = a.copy()
            stagnant = 0

        # Running average
        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = _fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-NS iter={n_iters} (lp={lp_count}, sg={subgrad_count}) ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def _spectral_equalize(a0, time_limit, verbose=False):

    """

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()

        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99)

    # FFT size for circular = linear autocorrelation
    fft_size = 1 << (2 * n - 1).bit_length()

    for iteration in range(50000):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute power spectrum
        A = rfft(a, fft_size)
        P = np.abs(A) ** 2

        # Target: flat at the mean level
        target = P.mean()
        # Focus on the peaks (top 20% of power spectrum values)
        threshold = np.percentile(P, 80)

        # For each "hot" frequency bin, the corresponding time-domain
        # modulation is a cosine. We want to reduce those.
        # Simple approach: find the autocorrelation peak and do local adjustment
        b = _fast_autocorr(a)
        k_star = int(np.argmax(b))
        M_curr = b[k_star]

        # Identify contributing pairs for the peak
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        if len(idx) == 0:
            idx = np.array([k_star % n])
        contrib = a[idx] * a[k_star - idx]

        if contrib.max() < 1e-15:
            # Try random perturbation

            j = rng.integers(n)
            if i == j:
                continue
            frac = rng.uniform(0.005, 0.02)
            amt = a[i] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[i] -= amt
            a_trial[j] += amt
            s_t = a_trial.sum()
            if s_t < 1e-15:

            a_trial = a_trial * norm / s_t
            score = _fast_autocorr(a_trial).max()
            if score < best_score:
                a = a_trial
                best_a = a.copy()
                best_score = score
            continue

        src = idx[np.argmax(contrib)]

            peak_contrib[idx2] += a[idx2] * a[k2 - idx2]

        # Destinations with low peak contribution and low current value
        safe_score = peak_contrib + 0.1 * a
        candidates = np.argsort(safe_score)[:max(1, n // 5)]
        dst = candidates[rng.integers(len(candidates))]

        # Try moving mass
        improved = False
        for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
            amt = a[src] * frac
            if amt < 1e-18:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial * norm / s_t
            score = _fast_autocorr(a_trial).max()
            if score < best_score - 1e-12:
                a = a_trial
                best_a = a.copy()
                best_score = score
                improved = True
                break

        if not improved:

                i = rng.integers(n)
                j = rng.integers(n)
                if i == j:
                    continue
                frac = rng.uniform(0.001, 0.005)
                amt = a[i] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = _fast_autocorr(a_trial).max()

def _peak_shave(a0, time_limit, seed=42):
    """Targeted peak reduction via adaptive mass redistribution."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = _fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:10]
        improved = False

        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]

            # Sort by low autocorrelation contribution
            cand_contrib = np.zeros(len(candidates))
            for ci, c in enumerate(candidates):
                for k2 in top_k[:3]:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    if js2 <= c <= je2:
                        j_partner = k2 - c
                        if 0 <= j_partner < n:
                            cand_contrib[ci] += a[c] * a[j_partner]
            candidates_sorted = candidates[np.argsort(cand_contrib)][:max(1, len(candidates) // 3)]

            dst = candidates_sorted[rng.integers(len(candidates_sorted))]

            for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.05]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = _fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random kick
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-8:
                frac = rng.uniform(0.002, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= -1e-15:
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = _fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0

            if stall > 1000:
                # Bigger random restart from best
                a = best_a.copy() + norm * 0.01 * rng.standard_normal(n)
                a = np.maximum(a, 0.0)
                s_a = a.sum()
                a = a * norm / s_a
                stall = 0

    return best_a, best_score


def _interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def _evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def _subgrad_nesterov(a0, time_limit, verbose=False):
    """Pure subgradient Nesterov (fast, no LP) for quick polishing."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    y = a.copy()
    t0 = time.monotonic()
    step_scale = 4.0 / (n ** 0.5)

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = _fast_autocorr(y)
        k_star = int(np.argmax(b))

        grad = np.zeros(n)
        for i in range(n):
            j1 = k_star - i
            if 0 <= j1 < n:
                grad[i] += 2.0 * y[j1]
            j2 = i - k_star
            if 0 <= j2 < n:
                grad[i] += 2.0 * y[j2]

        i_min = int(np.argmin(grad))
        v = np.zeros(n)
        v[i_min] = norm

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * v
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = v.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_a = _fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        if verbose and t % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad-NS t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search: multi-phase optimization with LP, spectral equalization, and peak shaving."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 100.0, 1680.0)

    solver = _get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = _evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def _try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = _evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        # Try a few different initializations
        inits = []
        x = np.linspace(0.0, 1.0, n_base)
        inits.append(np.sin(np.pi * x) ** 1.5)
        inits.append(np.sin(np.pi * x))
        inits.append(1.0 + 0.3 * np.cos(2 * np.pi * x))
        inits.append(np.random.default_rng(0).dirichlet(np.ones(n_base)))

        best_init = None
        best_init_score = float('inf')
        for init in inits:
            init = np.maximum(init, 0.0)
            s_i = init.sum()
            if s_i > 1e-15:
                init = init / s_i
                sc = _evaluate_safe(init)
                if sc < best_init_score:
                    best_init_score = sc
                    best_init = init.copy()

        best_seq = best_init
        best_score = best_init_score
        print(f"Best init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    # === Phase 1: LSE-FISTA (fast warm-up) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.06, 60.0)
    if t1 > 10:
        print(f"\n=== Phase 1: LSE-FISTA n={n_base} (budget={t1:.0f}s) ===")
        T_sched = [10.0, 3.0, 1.0, 0.3, 0.1]
        a_out, _ = _lse_fista(best_seq, T_sched, t1, verbose=True)
        _try_update(a_out, f"lse1_n{n_base}")

    # === Phase 2: LP+Nesterov (main) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.35, 500.0)
    if t2 > 30:
        print(f"\n=== Phase 2: LP-Nesterov n={n_cur} (budget={t2:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _lp_nesterov(a_norm, t2, solver, verbose=True, use_adaptive_lpfreq=True)
        _try_update(a_out, f"lp1_n{n_cur}")

    # === Phase 3: Spectral equalization ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.10, 120.0)
    if t3 > 15:
        print(f"\n=== Phase 3: Spectral eq n={n_cur} (budget={t3:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _spectral_equalize(a_norm, t3, verbose=False)
        _try_update(a_out, f"spectral_n{n_cur}")

    # === Phase 4: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 10:
        print(f"\n=== Phase 4: Peak shave n={n_cur} (budget={t4:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _peak_shave(a_norm, t4, seed=12345)
        _try_update(a_out, f"shave1_n{n_cur}")

    # === Phase 5: Interpolate to longer + LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq) + 300
    t5 = min(remaining * 0.20, 250.0)
    if t5 > 30:
        print(f"\n=== Phase 5: LP-Nesterov n={n2} (budget={t5:.0f}s) ===")
        init_long = _interpolate_sequence(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = _lp_nesterov(a_norm, t5, solver, verbose=True, use_adaptive_lpfreq=True)
        _try_update(a_out, f"lp2_n{n2}")

    # === Phase 6: Subgradient Nesterov (fast, many steps) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t6 = min(remaining * 0.08, 80.0)
    if t6 > 10:
        print(f"\n=== Phase 6: SubGrad-NS n={n_cur} (budget={t6:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _subgrad_nesterov(a_norm, t6, verbose=False)
        _try_update(a_out, f"sg_n{n_cur}")

    # === Phase 7: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t7 = min(remaining * 0.25, 200.0)
    if t7 > 20:
        print(f"\n=== Phase 7: LP polish n={n_cur} (budget={t7:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _lp_nesterov(a_norm, t7, solver, verbose=True, use_adaptive_lpfreq=False)
        _try_update(a_out, f"lp_final_n{n_cur}")

    # === Phase 8: Final peak shave ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t8 = min(remaining * 0.30, 100.0)
    if t8 > 10:
        print(f"\n=== Phase 8: Final shave n={n_cur} (budget={t8:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _peak_shave(a_norm, t8, seed=99999)
        _try_update(a_out, f"shave_final_n{n_cur}")

    # Final check
    if best_seq is not None:
        fs = _evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
