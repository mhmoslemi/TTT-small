def fast_best_response(a_fixed, n_iters=200, T_lse=50.0):
    """
    Compute the best response b to a_fixed in the zero-sum game.
    min_b max_k (a*b)_k  s.t. sum(b)=1, b>=0
    Using mirror descent with LogSumExp smoothing.
    """
    n = len(a_fixed)
    b = np.ones(n) / n
    fft_size_conv = 1 << (2*n-1).bit_length()  # for a*b
    fft_size_corr = 1 << (3*n-2).bit_length()  # for cross-correlation
    
    A = rfft(a_fixed, fft_size_conv)
    A_corr = rfft(a_fixed[::-1].copy(), fft_size_corr)
    
    for t in range(n_iters):
        # Compute g = a * b
        B = rfft(b, fft_size_conv)
        g = irfft(A * B, fft_size_conv)[:2*n-1]
        
        # LogSumExp weights
        max_g = g.max()
        exp_Tg = np.exp(T_lse * (g - max_g))
        w = exp_Tg / exp_Tg.sum()
        
        # Gradient: grad_b[j] = sum_m a[m] * w[j+m]
        # = (a_rev * w)[n-1+j]
        W = rfft(w, fft_size_corr)
        conv = irfft(A_corr * W, fft_size_corr)
        grad_b = conv[n-1:n-1+n]

        lr_t = 1.0 / (np.sqrt(t + 1))
        b = b * np.exp(-lr_t * (grad_b - grad_b.mean()))
        b = np.maximum(b, 1e-300)

    # Final evaluation
    B = rfft(b, fft_size_conv)
    g = irfft(A * B, fft_size_conv)[:2*n-1]
    return b, g.max()

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def internal_score(a):
    """Compute 2*n*max(a*a) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * float(fast_autocorr(a).max())


def evaluate_safe(arr):
    """Safely evaluate a sequence using the real evaluator."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        if len(a_list) == 0 or sum(a_list) < 0.01:
            return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0} using bisection."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= -1e-15) and abs(v.sum() - target_sum) < 1e-10:
        v = np.maximum(v, 0)
        s = v.sum()
        if s > 1e-15:
            return v / s * target_sum
        return np.ones(n) / n * target_sum

    v = np.maximum(v, 0)
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho = np.nonzero(v_sorted * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.ones(n) / n * target_sum
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-12:
        idx = np.argmax(result)
        result[idx] = max(0, result[idx] + diff)
        s = result.sum()
        if s < 1e-15:
            return np.ones(n) / n * target_sum
        result = result / s * target_sum
    return result


def spectral_shaping(a, n_iter=300, alpha=0.3):
    """Iteratively flatten power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = internal_score(a)

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 50.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def mirror_descent_best_response(a_fixed, b_init, n_iters=50, T_lse=80.0, lr_base=0.01):
    """
    Fast best response computation via multiplicative weights (mirror descent).
    Solves: min_b max_k (a*b)_k s.t. sum(b)=1, b>=0
    """
    n = len(a_fixed)
    b = np.maximum(np.asarray(b_init, dtype=np.float64), 0)
    s = b.sum()
    if s < 1e-15:
        b = np.ones(n) / n
    else:
        b = b / s

    fft_size_conv = 1 << (2 * n - 1).bit_length()
    A = rfft(a_fixed, fft_size_conv)

    for t in range(n_iters):
        # Compute g = a * b
        B = rfft(b, fft_size_conv)
        g = irfft(A * B, fft_size_conv)[:2 * n - 1]

        # LSE weights on g
        max_g = g.max()
        exp_Tg = np.exp(T_lse * (g - max_g))
        w = exp_Tg / exp_Tg.sum()

        # Gradient: grad_b[j] = sum_k w_k * a_{k-j}
        # = cross_corr(w, a)[j] = (a_rev * w)[n-1+j]
        fft_size_corr = 1 << (3 * n - 2).bit_length()
        A_rev = rfft(a_fixed[::-1].copy(), fft_size_corr)
        W = rfft(w, fft_size_corr)
        conv = irfft(A_rev * W, fft_size_corr)
        grad_b = conv[n - 1:n - 1 + n]

        # Multiplicative weights update
        lr_t = lr_base / (1.0 + 0.05 * t)
        b = b * np.exp(-lr_t * (grad_b - grad_b.mean()))
        b = np.maximum(b, 1e-300)
        s = b.sum()
        if s < 1e-300:
            b = np.ones(n) / n
        else:
            b = b / s

    # Final evaluation
    B = rfft(b, fft_size_conv)
    g = irfft(A * B, fft_size_conv)[:2 * n - 1]
    return b, float(g.max()), w


def nesterov_md_optimize(a0, time_limit, n_inner=50, T_lse_inner=80.0, lr_base=0.01, verbose=True):
    """
    Nesterov-accelerated zero-sum game with mirror descent inner loop.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    y = a.copy()
    y_prev = a.copy()
    b_curr = np.ones(n) / n  # warm start for inner loop
    t = 0

    # Track dual for diagnostics
    val_primal = best_score
    val_dual = 0.0

    while time.monotonic() - t0 < time_limit:
        t += 1

        # Inner loop: best response to y
        b_curr, M_val, w_dual = mirror_descent_best_response(
            y, b_curr, n_iters=n_inner, T_lse=T_lse_inner, lr_base=lr_base
        )

        # Dual value: min_j sum_k w_dual[k] * T[k,j] = min_j cross_corr(w_dual, y)[j]
        fft_size_corr = 1 << (3 * n - 2).bit_length()
        Y_rev = rfft(y[::-1].copy(), fft_size_corr)
        W = rfft(w_dual, fft_size_corr)
        conv = irfft(Y_rev * W, fft_size_corr)
        dual_vec = conv[n - 1:n - 1 + n]
        val_dual = float(dual_vec.min())

        # Nesterov outer update
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha_t) * y + alpha_t * b_curr
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s

        if t == 1:
            a_new = b_curr.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = project_simplex(a_new)

        y_prev = y
        y = y_new
        a = a_new

        # Track best
        score_a = internal_score(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = internal_score(y)
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Time-averaged iterate (converges at O(1/t))
        if t % 200 == 0:
            # Quick check of a few iterates
            pass

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            gap = val_primal if val_primal > 0 else best_score / (2 * n)
            print(f"    NMD t={t} ({elapsed:.0f}s): primal={best_score:.6f}, "
                  f"M_bresp={M_val:.8f}, dual={val_dual:.8f}, gap~{best_score/(2*n)-val_dual:.6e}")

    return best_a, best_score


def lse_gradient(a, T, fft_size_conv, fft_size_corr):
    """Compute gradient of (1/T)*log(sum exp(T*(a*a)_k)) w.r.t. a."""
    n = len(a)
    c = fast_autocorr(a)
    max_c = c.max()
    exp_Tc = np.exp(T * (c - max_c))
    w = exp_Tc / exp_Tc.sum()

    A_rev = rfft(a[::-1].copy(), fft_size_corr)
    W = rfft(w, fft_size_corr)
    conv = irfft(A_rev * W, fft_size_corr)
    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad, w, c


def fista_lse_optimize(a0, time_limit, T_start=100.0, T_end=5.0, n_stages=20, verbose=True):
    """
    FISTA (accelerated proximal gradient) on LogSumExp-smoothed objective.
    Slow temperature annealing for fine convergence.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    fft_size_conv = 1 << (2 * n - 1).bit_length()
    fft_size_corr = 1 << (3 * n - 2).bit_length()

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        # FISTA state
        x = a.copy()
        y = a.copy()
        t_fista = 1.0
        t_iter = 0

        max_iters = int(remaining / 0.05)  # aim for ~50ms per iteration
        max_iters = max(max_iters, 50)
        max_iters = min(max_iters, 5000)

        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                break

            t_iter += 1
            # Gradient at y
            grad_y, w_y, c_y = lse_gradient(y, stage_T, fft_size_conv, fft_size_corr)

            # Step size: 1/L where L ~ 2*T (conservative)
            step = 1.0 / (stage_T + 1.0)

            # Proximal step (project to simplex)
            x_new = y - step * grad_y
            x_new = project_simplex(x_new)

            # FISTA update
            t_next = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_fista ** 2))
            y_new = x_new + ((t_fista - 1.0) / t_next) * (x_new - x)
            y_new = project_simplex(y_new)

            x = x_new
            y = y_new
            t_fista = t_next

            # Check best every few iterations
            if (it + 1) % 10 == 0:
                score_x = internal_score(x)
                if score_x < best_score:
                    best_score = score_x
                    best_a = x.copy()
                score_y = internal_score(y)
                if score_y < best_score:
                    best_score = score_y
                    best_a = y.copy()

        # Update a for next stage
        a = x
        score_end = internal_score(a)
        if score_end < best_score:
            best_score = score_end
            best_a = a.copy()

        if verbose and si % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA stage {si}/{n_stages} (T={stage_T:.1f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score

    t0 = time.monotonic()

    y = a.copy()
    y_prev = a.copy()
    t = 0

    while time.monotonic() - t0 < time_limit:
        t += 1

        c = fast_autocorr(y)
        max_c = c.max()

        # Adaptive temperature: decrease over time
        frac = min(1.0, t / 5000.0)
        T_cur = 200.0 * (5.0 / 200.0) ** frac

        max_Tc = T_cur * max_c
        exp_Tc = np.exp(T_cur * (c - max_c))
        w = exp_Tc / exp_Tc.sum()

        # Gradient: 2 * cross_corr(w, y)
        Y_rev = rfft(y[::-1].copy(), fft_size_corr)
        W = rfft(w, fft_size_corr)
        conv = irfft(Y_rev * W, fft_size_corr)
        grad = 2.0 * conv[n - 1:n - 1 + n]

        # Nesterov step with adaptive learning rate
        alpha_t = 2.0 / (t + 2.0)
        # Gradient norm for adaptive step
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            gnorm = 1.0
        step = alpha_t / (T_cur * gnorm)
        step = min(step, 0.5 / (n * 0.01))  # cap step

        y_new = y - step * grad
        y_new = project_simplex(y_new)

        if t == 1:
            a_new = y_new
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = project_simplex(a_new)

        y_prev = y
        y = y_new
        a = a_new

        score_a = internal_score(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad t={t} ({elapsed:.0f}s): T={T_cur:.1f}, best={best_score:.6f}")

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=42):
    """Coordinate-descent peak shaving with targeted moves."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:15]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]

            src_local = np.argmax(contrib)
            src = idx[src_local]

            # Find destination: low-contribution positions
            peak_areas = set()
            for k2 in top_k[:8]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_areas.update(range(js2, je2 + 1))

            candidates = [j for j in range(n) if j not in peak_areas and j != src]
            if not candidates:
                candidates = [j for j in range(n) if j != src]

            # Sort by current value (move from high to low)
            candidates_sorted = sorted(candidates, key=lambda j: a[j])
            n_dst = max(1, len(candidates_sorted) // 4)
            dst_pool = candidates_sorted[:n_dst]
            dst = dst_pool[rng.integers(len(dst_pool))]

            tried = False
            for frac in [0.001, 0.003, 0.008, 0.015, 0.03, 0.06]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-12:
                    a_trial[src] = 0.0
                    a_trial[dst] = a[dst] + a[src]
                s2 = a_trial.sum()
                if s2 < 1e-15:
                    continue
                a_trial = a_trial / s2
                score = internal_score(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    tried = True
                    break

            if improved or tried:
                break

        if not improved:
            # Random perturbation to escape local minima
            stall += 1
            for _ in range(5):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-12:
                    continue
                frac = rng.uniform(0.002, 0.04)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < 0:
                    a_trial[i] = 0
                s2 = a_trial.sum()
                if s2 < 1e-15:
                    continue
                a_trial = a_trial / s2
                score = internal_score(a_trial)
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

            if stall > 300:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def adam_lse_optimize(a0, time_limit, T_start=50.0, T_end=0.5, n_stages=15, lr=0.003, verbose=True):
    """Adam optimizer on LSE-smoothed objective with z-parameterization."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))

    fft_size_conv = 1 << (2 * n - 1).bit_length()
    fft_size_corr = 1 << (3 * n - 2).bit_length()

    best_a = a_init.copy()
    best_score = internal_score(a_init)
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_adam = 0

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        iters = max(20, min(300, int(remaining * 30 / max(1, n_stages - si))))

        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break

            a = softmax_fn(z)
            grad_a, w, c = lse_gradient(a, stage_T, fft_size_conv, fft_size_corr)
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)

            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)

            lr_t = lr / np.sqrt(1 + 0.005 * it)
            z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)

            if (it + 1) % 30 == 0:
                a_check = softmax_fn(z)
                score = internal_score(a_check)
                if score < best_score:
                    best_score = score
                    best_a = a_check.copy()

        a = softmax_fn(z)
        score = internal_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and si % 4 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam-LSE stage {si}/{n_stages} (T={stage_T:.1f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization targeting score < 1.5027.
    Phase 1: Spectral shaping (initialization)
    Phase 2: Nesterov with mirror descent (primary optimizer, fast)
    Phase 3: FISTA LSE (fine optimization with slow annealing)
    Phase 4: Subgradient Nesterov (complementary direction)
    Phase 5: Adam LSE (polish)
    Phase 6: Peak shaving (final refinement)
    Phase 7: Multi-length exploration if time remains
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 80.0, 1720.0)

    best_seq = None
    best_score = float('inf')

    # Load starting point
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = internal_score(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = internal_score(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Spectral shaping (fast initialization)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: Spectral shaping n={n_base} (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.03, 35.0)
    if t1 > 5:
        for alpha_val in [0.5, 0.3, 0.15]:
            a_out = spectral_shaping(best_seq, n_iter=250, alpha=alpha_val)
            sc = internal_score(a_out)
            if sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  Spectral alpha={alpha_val}: {best_score:.6f}")
    print(f"  After Phase 1: best={best_score:.6f} (t={time.monotonic()-t_global_start:.0f}s)")

    # Phase 2: Nesterov MD (main workhorse - allocate most time)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: Nesterov MD n={n_base} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.38, 650.0)
    if t2 > 30:
        a_out, sc_out = nesterov_md_optimize(best_seq, t2, n_inner=40, T_lse=60.0, lr_base=0.008)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [NMD_n{n_base}]: {best_score:.6f}")
    print(f"  After Phase 2: best={best_score:.6f} (t={time.monotonic()-t_global_start:.0f}s)")

    # Phase 3: FISTA LSE (fine optimization)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: FISTA LSE n={n_base} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.22, 370.0)
    if t3 > 20:
        a_out, sc_out = fista_lse_optimize(best_seq, t3, T_start=150.0, T_end=10.0, n_stages=15)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [FISTA_n{n_base}]: {best_score:.6f}")
        # Second FISTA round with lower temperatures
        t3b = min(remaining * 0.10, 150.0)
        if t3b > 15:
            a_out2, _ = fista_lse_optimize(best_seq, t3b, T_start=30.0, T_end=2.0, n_stages=12)
            sc2 = evaluate_safe(a_out2)
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_seq = np.maximum(a_out2, 0)
                s = best_seq.sum()
                if s > 0:
                    best_seq = best_seq / s
                print(f"  *** NEW BEST [FISTA_low_n{n_base}]: {best_score:.6f}")
    print(f"  After Phase 3: best={best_score:.6f} (t={time.monotonic()-t_global_start:.0f}s)")

    # Phase 4: Subgradient Nesterov
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: SubGrad Nesterov n={n_base} (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 220.0)
    if t4 > 15:
        a_out, sc_out = subgradient_nesterov(best_seq, t4)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [SubGrad_n{n_base}]: {best_score:.6f}")
    print(f"  After Phase 4: best={best_score:.6f} (t={time.monotonic()-t_global_start:.0f}s)")

    # Phase 5: Adam LSE polish
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: Adam LSE n={n_base} (remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.10, 130.0)
    if t5 > 10:
        a_out, _ = adam_lse_optimize(best_seq, t5, T_start=50.0, T_end=1.0, n_stages=12, lr=0.002)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [Adam_n{n_base}]: {best_score:.6f}")
    print(f"  After Phase 5: best={best_score:.6f} (t={time.monotonic()-t_global_start:.0f}s)")

    # Phase 6: Peak shaving
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: Peak shaving n={n_base} (remaining={remaining:.0f}s) ===")
    t6 = min(remaining * 0.12, 150.0)
    if t6 > 5:
        a_out, _ = peak_shaving(best_seq, t6, seed=456)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [peak1_n{n_base}]: {best_score:.6f}")
    print(f"  After Phase 6: best={best_score:.6f} (t={time.monotonic()-t_global_start:.0f}s)")

    # Phase 7: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 100:
        n_try = [int(n_base * 0.85), int(n_base * 1.15), int(n_base * 1.3)]
        n_try = [x for x in n_try if 500 <= x <= 3000]
        print(f"\n=== Phase 7: Multi-length exploration (remaining={remaining:.0f}s) ===")
        for n_new in n_try:
            remaining = time_budget - (time.monotonic() - t_global_start)
            if remaining < 40:
                break
            t7i = min(remaining * 0.3, 100.0)
            if t7i < 10:
                continue
            init_new = interpolate_sequence(best_seq, n_new)
            # Quick spectral
            a_spec = spectral_shaping(init_new, n_iter=150, alpha=0.2)
            sc = evaluate_safe(a_spec)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = np.maximum(a_spec, 0)
                s = best_seq.sum()
                if s > 0:
                    best_seq = best_seq / s
                print(f"  *** NEW BEST [spec_n{n_new}]: {best_score:.6f}")

            remaining = time_budget - (time.monotonic() - t_global_start)
            if remaining < 20:
                continue
            # Nesterov MD at new length
            t7i2 = min(remaining * 0.5, 120.0)
            if t7i2 > 15:
                init_md = interpolate_sequence(best_seq, n_new)
                a_out, _ = nesterov_md_optimize(init_md, t7i2, n_inner=30, T_lse=50.0, lr_base=0.006)
                sc = evaluate_safe(a_out)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = np.maximum(a_out, 0)
                    s = best_seq.sum()
                    if s > 0:
                        best_seq = best_seq / s
                    print(f"  *** NEW BEST [NMD_n{n_new}]: {best_score:.6f}")

    # Phase 8: Final peak shaving
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 10:
        t8 = min(remaining * 0.5, 60.0)
        if t8 > 5:
            print(f"\n=== Phase 8: Final peak shave n={len(best_seq)} ===")
            a_out, _ = peak_shaving(best_seq, t8, seed=789)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = np.maximum(a_out, 0)
                s = best_seq.sum()
                if s > 0:
                    best_seq = best_seq / s
                print(f"  *** NEW BEST [peak_final]: {best_score:.6f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        best_seq = np.maximum(best_seq, 0)
        s = best_seq.sum()
        if s > 0:
            best_seq = best_seq / s
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
