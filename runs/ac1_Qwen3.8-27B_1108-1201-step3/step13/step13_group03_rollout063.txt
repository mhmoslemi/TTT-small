For approach 2 (cutting plane):
- Start with a relaxation (e.g., just a >= 0, sum(a) = 1).
- Solve the LP: maximize 0 (or minimize something) subject to the cuts.
- Check if the solution satisfies all quadratic constraints.
- If not, add a linearization of the violated constraint as a new cut.
- Repeat.

The linearization of a^T Q_k a <= M at point a_0 is:
2 * (Q_k * a_0)^T * (a - a_0) + a_0^T Q_k a_0 <= M
i.e., 2 * (Q_k * a_0)^T * a <= M - a_0^T Q_k a_0 + 2 * a_0^T Q_k a_0 = M + a_0^T Q_k a_0

So the cut is: 2 * (Q_k * a_0)^T * a <= M + a_0^T Q_k a_0

This is a valid outer approximation of the convex set {a: a^T Q_k a <= M}.

def _fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def _project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n <= 1:
        return np.array([1.0]) if n == 1 else np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 1e-15:
        out /= s
    else:
        out = np.ones(n) / n
    return out

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP best-response with Nesterov acceleration (exact, primary)
2. Multi-length strategy with warm-starting (n=1600 -> 2500 -> 4000 -> 6000)
3. LSE-smoothed FISTA for large-n scalability and final polish
4. Coordinate descent micro-optimization
5. Optional direct QP verification

tolerance, symmetry-aware initialization, and aggressive final polishing.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix

_global_solver = None


def _get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            v = cp.Variable(4, nonneg=True)
            p = cp.Problem(cp.Minimize(cp.sum(v)), [cp.sum(v) == 1, v <= 0.4])
            p.solve(solver=s)
            if p.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  Using LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.CLARABEL
    print(f"  Fallback solver: CLARABEL")
    return _global_solver


def _fft_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def _eval_autocorr(a):
    """Return max autocorrelation for normalized sequence (sum=sqrt(2n))."""
    return _fft_autocorr(a).max()


def _proj_simplex(v):
    n = len(v)
    if n <= 1:
        return np.ones(max(n, 1)) / max(n, 1)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 1e-15:
        out /= s
    else:
        out = np.ones(n) / n
    return out


def _build_conv_T(a):
    """Build sparse matrix T (2n-1 x n) where (T@b)[k] = conv(a,b)[k]."""
    n = len(a)
    rows = np.zeros(n * n, dtype=np.int32)
    cols = np.zeros(n * n, dtype=np.int32)
    vals = np.zeros(n * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        length = j_max - j_min + 1
        if length > 0:
            rows[idx:idx + length] = k
            cols[idx:idx + length] = range(j_min, j_max + 1)
            vals[idx:idx + length] = a[k - np.arange(j_min, j_max + 1)]
            idx += length
    return csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(2 * n - 1, n))


def _lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. b>=0, conv(a,b)<=M elementwise."""
    n = len(a)
    T = _build_conv_T(a)
    b = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(b)), [T @ b <= M])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def _nesterov_br(a0, time_budget, solver, verbose=True):
    """Nesterov-accelerated best-response dynamics."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s

    best_a = a.copy()
    best_M = _fft_autocorr(a).max()
    y = a.copy()
    t0 = time.monotonic()
    it_count = 0
    best_gap = float('inf')
    best_gap_iter = 0

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_budget:
            break

        M_curr = _fft_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = _lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        it_count += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = _fft_autocorr(y).max()
        if score_y < best_M:
            best_M = score_y
            best_a = y.copy()
            best_gap = score_y
            best_gap_iter = it_count

        score_a = _fft_autocorr(a).max()
        if score_a < best_M:
            best_M = score_a
            best_a = a.copy()

        # Check convergence
        gap = np.linalg.norm(a - y) / max(norm, 1e-15)
        if it_count % 50 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"    BR iter={it_count} ({elapsed:.0f}s): M={best_M:.8f}, gap={gap:.6f}")

        # Early stopping if converged
        if it_count > 50 and gap < 1e-9:
            break

    return best_a, best_M


def _lse_grad(y, T_temp):
    """Gradient of LSE-smoothed max autocorrelation."""
    n = len(y)
    ac = _fft_autocorr(y)
    ac_max = ac.max()
    shifted = (ac - ac_max) / T_temp
    shifted = np.clip(shifted, -50.0, 50.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()

    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(y, fft_size)
    W = rfft(w, fft_size)
    cross = irfft(np.conj(A) * W, fft_size)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad


def _lse_fista(a0, T_schedule, time_budget, verbose=True):
    """FISTA on LSE-smoothed max autocorrelation with temperature annealing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = _fft_autocorr(a).max()
    t0 = time.monotonic()
    L_base = 4.0  # Lipschitz constant for LSE-smoothed objective

    for stage_i, T_val in enumerate(T_schedule):
        L = L_base / max(T_val, 0.01)
        t_fista = 1
        a_prev = a.copy()

        max_iters = 50000
        for it in range(max_iters):
            if time.monotonic() - t0 > time_budget:
                return best_a, best_score

            grad = _lse_grad(y, T_val)
            step = 1.0 / L
            z = y - step * grad
            a_new = _proj_simplex(z)

            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - a_prev)
                y_new = np.maximum(y_new, 0.0)
                s_y = y_new.sum()
                if s_y > 1e-15:
                    y_new /= s_y
                else:
                    y_new = a_new.copy()

            a_prev = a_new
            a = a_new
            y = y_new
            t_fista += 1

            if it % 2000 == 0:
                sc = _fft_autocorr(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

        sc = _fft_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE-FISTA stage={stage_i} T={T_val:.3f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def _interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def _coordinate_descent(a0, time_budget, seed=42):
    """Targeted coordinate descent for peak reduction."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fft_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_budget:
        b = _fft_autocorr(a)
        top_k = np.argsort(b)[::-1][:12]
        improved = False

        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-18:
                continue

            src = idx[np.argmax(contrib)]

            # Find good destination: low contribution to all peaks
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                if je2 >= js2:
                    peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]
            if len(candidates) == 0:
                continue

            # Pick from lowest-valued candidates
            c_sorted = candidates[np.argsort(a[candidates])]
            c_pool = c_sorted[:max(1, len(c_sorted) // 5)]

            peak_idx = rng.integers(len(c_pool))
            dst = c_pool[peak_idx]

            for frac in [0.0005, 0.001, 0.002, 0.005, 0.01, 0.02]:
                amt = a[src] * frac
                if amt < 1e-20:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = _fft_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-12:
                frac = rng.uniform(0.002, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= -1e-15:
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = _fft_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
            if stall > 1000:
                break

    return best_a, best_score


def _make_symmetric(a):
    """Make a sequence symmetric: a[i] = a[n-1-i]."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    return a_sym


def _init_shaped(n, shape='cosine'):
    """Create diverse initial sequences."""
    x = np.linspace(0, 1, n)
    if shape == 'cosine':
        f = np.sin(np.pi * x) ** 1.2
    elif shape == 'gaussian':
        f = np.exp(-4 * (x - 0.5) ** 2)
    elif shape == 'triangular':
        f = 1 - 2 * np.abs(x - 0.5)
    elif shape == 'plateau':
        f = np.ones(n)
        f[:n//4] *= 0.5
        f[3*n//4:] *= 0.5
    else:
        f = np.sin(np.pi * x)
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n)
        s = n
    return f / s

    if t8 > 30:
        print(f"\n=== Phase 8: LP-Nesterov final n={n_best} (budget={t8:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = _nesterov_br(a_norm, t8, solver, verbose=True)
        _try_update(a_out, "lp_final")

    # === Phase 9: LSE-FISTA with very low temperature ===
    remaining = time_budget - (time.monotonic() - t_global)
    t9 = min(remaining * 0.15, 100.0)
    if t9 > 10:
        print(f"\n=== Phase 9: LSE-FISTA low-T n={len(best_seq)} (budget={t9:.0f}s) ===")
        T_sched = [1.0, 0.2, 0.05, 0.01]
        a_out, _ = _lse_fista(best_seq, T_sched, t9, verbose=True)
        _try_update(a_out, "lse_final")

    # === Phase 10: Second coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    t10 = min(remaining * 0.20, 100.0)
    if t10 > 10:
        print(f"\n=== Phase 10: Coord descent 2 n={len(best_seq)} (budget={t10:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * len(best_seq)) / best_seq.sum()
        a_out, _ = _coordinate_descent(a_norm, t10, seed=99999)
        _try_update(a_out, "coord2")

    # === Phase 11: Try symmetry enforcement + re-optimization ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        a_sym = _make_symmetric(best_seq)
        a_sym = a_sym / a_sym.sum()
        _try_update(a_sym, "symmetric")
        if a_sym.sum() > 0.01:
            t11 = min(remaining * 0.15, 100.0)
            if t11 > 20:
                print(f"\n=== Phase 11: LP on symmetric n={len(best_seq)} (budget={t11:.0f}s) ===")
                a_norm_sym = a_sym * np.sqrt(2.0 * len(a_sym)) / a_sym.sum()
                a_out, _ = _nesterov_br(a_norm_sym, t11, solver, verbose=False)
                _try_update(a_out, "lp_sym")

    # Final verification
    if best_seq is not None:
        fs = _evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
