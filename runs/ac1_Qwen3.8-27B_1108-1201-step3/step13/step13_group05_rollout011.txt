def fast_autocorr(a):
    """Compute linear auto-convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def softmax_subgrad(a, ac, tau):
    """
    Compute smoothed subgradient using log-sum-exp weighting.
    grad[i] = sum_k p_k * (a[k-i] + a[i]) for valid indices,
    where p_k = exp(ac[k]/tau) / sum(exp(ac[j]/tau)).
    """
    n = len(a)
    # Stable log-sum-exp
    max_ac = ac.max()
    log_p = ac - max_ac
    p = np.exp(log_p)
    p_sum = p.sum()
    if p_sum < 1e-30:
        p = np.ones(len(ac)) / len(ac)
    else:
        p = p / p_sum

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-12:
            continue
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start > i_end:
            continue
        idx = np.arange(i_start, i_end + 1)
        j = k - idx
        grad[idx] += p[k] * (a[j] + a[idx])
    return grad


def topk_subgrad(a, ac, k=5, beta=3.0):
    """Weighted subgradient using top-k peaks with power weighting."""
    n = len(a)
    k_max = min(k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx].copy(), 0)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    grad = np.zeros(n)
    for wi, kk in zip(w, top_idx):
        if wi < 1e-12:
            continue
        i_start = max(0, int(kk) - n + 1)
        i_end = min(n - 1, int(kk))
        if i_start > i_end:
            continue
        idx = np.arange(i_start, i_end + 1)
        j = int(kk) - idx
        grad[idx] += wi * (a[j] + a[idx])
    return grad

def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. conv(a,b)_k <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)

    # Build constraint matrix T where (T @ b)_k = (a*b)_k
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a

    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * c / n
    else:
        a = a * c / s0

    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, solver)
        if b is None:
            # Fallback: use gradient step
            ac = fast_autocorr(y)
            g = softmax_subgrad(y, ac, tau=M_curr * 0.1)
            g = g - g.mean()
            b = np.maximum(y - g / max(np.abs(g).max(), 1e-10), 0.0)
            if b.sum() < 1e-15:
                b = y.copy()

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g_dir = b * c / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g_dir
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * c / s_y

        if t == 1:
            x_new = g_dir.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new * c / s_x

        y = y_new
        x = x_new

        score_y = fast_autocorr(y).max()
        if score_y < best_m:
            best_m = score_y
            best_a = y.copy()

        score_x = fast_autocorr(x).max()
        if score_x < best_m:
            best_m = score_x
            best_a = x.copy()

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c * c)
            print(f"    LP iter={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def coordinate_descent_polish(a0, time_limit, verbose=True, rng_seed=123):
    """
    2-coordinate descent: pick the pair (i, j) that most reduces the peak,
    move mass between them via golden-section search.
    """
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * c / n
    else:
        a = a * c / s0

    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 100:
        ac = fast_autocorr(a)
        M = ac.max()

        # Find the peak lag
        k_peak = int(np.argmax(ac))

        # Compute gradient at peak
        i_start = max(0, k_peak - n + 1)
        i_end = min(n - 1, k_peak)
        idx = np.arange(i_start, i_end + 1)
        j = k_peak - idx
        # grad_i = a[j] + a[i] (contribution to peak)
        contrib = a[j] + a[idx]

        # Source: element with largest contribution to peak and largest value
        src_score = contrib * a[idx]
        src_pos = idx[np.argmax(src_score)]

        # Destination: element NOT contributing to any top peak, with room to grow
        top_peaks = np.argsort(ac)[::-1][:5]
        peak_mask = np.zeros(n, dtype=bool)
        for kp in top_peaks:
            ps = max(0, int(kp) - n + 1)
            pe = min(n - 1, int(kp))
            peak_mask[ps:pe + 1] = True

        candidates = np.where(~peak_mask)[0]
        if len(candidates) == 0:
            candidates = np.where(np.arange(n) != src_pos)[0]
        if len(candidates) == 0:
            stall += 1
            continue

        # Pick destination: lowest value among candidates
        dst_pos = candidates[np.argmin(a[candidates])]

        if dst_pos == src_pos:
            # Try random pair
            all_idx = np.arange(n)
            pair = rng.choice(all_idx, size=2, replace=False)
            src_pos, dst_pos = pair[0], pair[1]

        # Golden section search for optimal transfer amount
        max_amt = a[src_pos] * 0.5
        if max_amt < 1e-18:
            stall += 1
            continue

        def trial_score(amt):
            a_t = a.copy()
            a_t[src_pos] -= amt
            a_t[dst_pos] += amt
            if a_t[src_pos] < 0 or a_t[dst_pos] < 0:
                return np.inf
            s_t = a_t.sum()
            if s_t < 1e-15:
                return np.inf
            a_t = a_t * c / s_t
            return fast_autocorr(a_t).max()

        # Try a few amounts
        best_amt = 0.0
        best_trial_m = best_m
        for frac in [0.01, 0.03, 0.08, 0.15, 0.3, 0.5]:
            amt = max_amt * frac
            m_t = trial_score(amt)
            if m_t < best_trial_m - 1e-14:
                best_trial_m = m_t
                best_amt = amt

        if best_amt > 0:
            a[src_pos] -= best_amt
            a[dst_pos] += best_amt
            s_a = a.sum()
            if s_a > 1e-15:
                a = a * c / s_a
            m_new = fast_autocorr(a).max()
            if m_new < best_m:
                best_m = m_new
                best_a = a.copy()
                stall = 0
            else:
                # Revert
                a[src_pos] += best_amt
                a[dst_pos] -= best_amt
                a = a * c / a.sum()
                stall += 1
        else:
            # Random perturbation to escape local minimum
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > c * 1e-8:
                frac = rng.uniform(0.02, 0.1)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                s_a = a.sum()
                if s_a > 1e-15:
                    a = a * c / s_a
                m_new = fast_autocorr(a).max()
                if m_new < best_m:
                    best_m = m_new
                    best_a = a.copy()
                    stall = 0
                else:
                    a[i] += amt
                    a[j] -= amt
                    a = a * c / a.sum()
                    stall += 1
            else:
                stall += 1

        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c * c)
            print(f"    CD stall={stall} ({elapsed:.1f}s): J={J:.6f}")

    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m / (c * c)
        print(f"    CD done ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_parametric_init(n, kind='sine'):
    """Generate parametric initial sequences."""
    x = np.linspace(0, 1, n)
    if kind == 'uniform':
        a = np.ones(n)
    elif kind == 'sine':
        a = np.sin(np.pi * (x + 0.5 / n)) ** 2
    elif kind == 'cosine':
        # Concentrated in middle
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif kind == 'gaussian':
        a = np.exp(-4.0 * (x - 0.5) ** 2)
    elif kind == 'power_low':
        # More mass at edges
        a = (x + 1e-6) ** 0.5 * (1 - x + 1e-6) ** 0.5
        a = a + 0.3 * (1 - np.sin(np.pi * x))
    elif kind == 'bimodal':
        a = np.exp(-8.0 * (x - 0.25) ** 2) + np.exp(-8.0 * (x - 0.75) ** 2)
    elif kind == 'triangle':
        a = 1.0 - np.abs(2.0 * x - 1.0)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def propose_candidate(budget_s=1800.0):
    """Main search: multi-phase optimization with adaptive time allocation."""
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    safe_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Total time budget: {safe_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # --- Load parent ---
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent / parent.sum()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # --- Try parametric initializations if no parent ---
    if best_seq is None:
        n_base = 1600
        best_score = float('inf')
        for kind in ['sine', 'cosine', 'gaussian', 'bimodal', 'triangle', 'uniform']:
            a = make_parametric_init(n_base, kind)
            sc = evaluate_safe(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"  Init '{kind}': score={sc:.6f}")
        print(f"Best init: score={best_score:.6f}")

    n0 = len(best_seq)

    # === Phase 1: Smoothed Mirror Descent (global exploration) ===
    remaining = safe_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Smoothed MD, n={n0} ===")
    t1 = min(remaining * 0.12, 180.0)
    if t1 > 20:
        c0 = np.sqrt(2.0 * n0)
        a_norm = best_seq * c0 / best_seq.sum()
        a_out, _ = mirror_descent_smoothed(a_norm, t1, tau_start=0.05,
                                           tau_end=0.005, verbose=True)
        try_update(a_out, f"md_smooth_n{n0}")

    # === Phase 2: LP Nesterov (primary global optimizer) ===
    remaining = safe_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP Nesterov, n={n0} ===")
    t2 = min(remaining * 0.35, 500.0)
    if t2 > 30:
        c0 = np.sqrt(2.0 * n0)
        a_norm = best_seq * c0 / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t2, solver, verbose=True)
        try_update(a_out, f"lp_nesterov_n{n0}")

    # === Phase 3: Sharper Mirror Descent (local refinement) ===
    remaining = safe_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: Sharp MD, n={n_cur} ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        c_cur = np.sqrt(2.0 * n_cur)
        a_norm = best_seq * c_cur / best_seq.sum()
        a_out, _ = mirror_descent_smoothed(a_norm, t3, tau_start=0.01,
                                           tau_end=0.001, verbose=True,
                                           top_k_switch=3000)
        try_update(a_out, f"md_sharp_n{n_cur}")

    # === Phase 4: Second LP pass (further polish) ===
    remaining = safe_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: LP polish, n={n_cur} ===")
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 30:
        c_cur = np.sqrt(2.0 * n_cur)
        a_norm = best_seq * c_cur / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, verbose=True)
        try_update(a_out, f"lp_polish_n{n_cur}")

    # === Phase 5: Coordinate descent polish ===
    remaining = safe_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: Coord descent, n={n_cur} ===")
    t5 = min(remaining * 0.15, 180.0)
    if t5 > 15:
        c_cur = np.sqrt(2.0 * n_cur)
        a_norm = best_seq * c_cur / best_seq.sum()
        a_out, _ = coordinate_descent_polish(a_norm, t5, verbose=True)
        try_update(a_out, f"cd_n{n_cur}")

    # === Phase 6: Try longer length ===
    remaining = safe_budget - (time.monotonic() - t_global)
    n_long = min(n_cur + 800, 3200)
    if n_long > n_cur and remaining > 60:
        print(f"\n=== Phase 6: Extend to n={n_long} ===")
        t6 = min(remaining * 0.25, 250.0)
        a_long = interpolate_sequence(best_seq, n_long)
        # LP at longer length
        c_long = np.sqrt(2.0 * n_long)
        a_norm = a_long * c_long / a_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t6, solver, verbose=True)
        try_update(a_out, f"lp_long_n{n_long}")

    # === Phase 7: Final MD + CD on best ===
    remaining = safe_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 40:
        print(f"\n=== Phase 7: Final refinement, n={n_cur} ===")
        c_cur = np.sqrt(2.0 * n_cur)
        a_norm = best_seq * c_cur / best_seq.sum()
        t7a = min(remaining * 0.5, 100.0)
        if t7a > 15:
            a_out, _ = mirror_descent_smoothed(a_norm, t7a, tau_start=0.005,
                                               tau_end=0.0005, verbose=False,
                                               top_k_switch=1000)
            try_update(a_out, f"md_final_n{n_cur}")

        remaining = safe_budget - (time.monotonic() - t_global)
        n_cur = len(best_seq)
        if remaining > 15:
            t7b = min(remaining * 0.7, 100.0)
            c_cur = np.sqrt(2.0 * n_cur)
            a_norm = best_seq * c_cur / best_seq.sum()
            a_out, _ = coordinate_descent_polish(a_norm, t7b, verbose=True,
                                                 rng_seed=999)
            try_update(a_out, f"cd_final_n{n_cur}")

    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
