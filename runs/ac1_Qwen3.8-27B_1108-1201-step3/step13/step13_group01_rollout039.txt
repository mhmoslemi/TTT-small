def fast_autocorr(a):
    """Compute autocorrelation conv(a,a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)

    # = 2n * [grad_a_i * (1/s - a_norm_i/s^2)]
    # = 2n * grad_a_i * (s - a_i) / s^2  ... where a_i here means max(x_i, 0)
    # But since obj doesn't depend on scale (it's scale invariant), the (s-a_i)/s^2 term is the projection.
    # Actually for our purpose, we just need the direction. Let me use:
    grad_x = grad_a * (s - a) / (s * s)
    grad_x[x < 0] = 0.0
    # Remove mean component (since scale doesn't matter)
    grad_x = grad_x - a * np.dot(grad_x, a) / (s * s)
    return obj, grad_x


def lse_scipy_optimize(a0, time_limit, T_start=50.0, T_end=1.0, n_stages=6):
    n = len(a0)
    a = normalize(a0.copy(), 1.0)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        stage_time = min(remaining / (n_stages - si), 300.0)

        def obj_func(x):
            return lse_objective_and_grad(x, n, T)[0]

        def grad_func(x):
            return lse_objective_and_grad(x, n, T)[1]

        try:
            res = scipy_minimize(obj_func, a.copy(), method='L-BFGS-B',
                                 jac=grad_func,
                                 options={'maxiter': 100, 'maxfun': 300,
                                          'time_limit': stage_time})
            if res.x is not None:
                a_cand = normalize(res.x, 1.0)
                score_c = fast_autocorr(a_cand).max()
                if score_c < best_score:
                    best_score = score_c
                    best_a = a_cand.copy()
                    a = a_cand
        except Exception:
            pass

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, seed=42):
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = normalize(a0.copy(), 1.0)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a)
        top_peaks = np.argsort(c)[::-1][:12]
        mean_c = c.mean()

        improved = False
        for kp in top_peaks:
            kp = int(kp)
            if c[kp] <= mean_c:
                break
            # Find pairs (i, j) with i+j = kp (or i+j = kp in the convolution sense)
            # conv(a,a)[kp] = sum_i a[i]*a[kp-i]
            lo = max(0, kp - n + 1)
            hi = min(n - 1, kp)
            if lo > hi:
                continue
            idx = np.arange(lo, hi + 1)
            contrib = a[idx] * a[kp - idx]
            # Sort by contribution
            order = np.argsort(contrib)[::-1][:8]
            top_src = idx[order]

            # Find target positions (away from all peaks)
            peak_region = np.zeros(n, dtype=bool)
            for kp2 in top_peaks:
                lo2 = max(0, int(kp2) - 5)
                hi2 = min(n - 1, int(kp2) + 5)
                # Also mark the anti-diagonal
                lo3 = max(0, int(kp2) - n + 1 - 5)
                hi3 = min(n - 1, int(kp2) - n + 1 + 5)
                if lo3 <= hi3:
                    peak_region[lo3:hi3 + 1] = True
                peak_region[lo2:hi2 + 1] = True

            candidates = np.where(~peak_region)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)

            for src in top_src:
                if a[src] < 1e-10:
                    continue
                # Move mass from src to a low-value position
                cand_sorted = candidates[np.argsort(a[candidates])][:5]
                for dst in cand_sorted:
                    if dst == src:
                        continue
                    for frac in [0.02, 0.05, 0.1, 0.2]:
                        amt = a[src] * frac
                        if amt < 1e-12:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = normalize(a_trial, 1.0)
                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-12:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation when stalled
            if stall % 5 == 0:
                for _ in range(30):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i == j or a[i] < 1e-10:
                        continue
                    frac = rng.uniform(0.01, 0.15)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = normalize(a_trial, 1.0)
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                        break
            if stall > 100:
                break

    return best_a, best_score


def structured_asymmetric_init(n, rng, mode=0):
    """Create asymmetric initializations that break the symmetry barrier."""
    x = np.arange(n) / n
    if mode == 0:
        # Linear ramp
        a = 1.0 + 0.7 * (2 * x - 1)
    elif mode == 1:
        # Exponential ramp
        beta = 2.0
        a = np.exp(beta * x)
    elif mode == 2:
        # Cosine asymmetric
        a = 0.5 + 0.4 * np.cos(2 * np.pi * x + np.pi / 3) + 0.1 * np.cos(4 * np.pi * x)
    elif mode == 3:
        # Two asymmetric bumps
        a = np.exp(-4.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-4.0 * (x - 0.8) ** 2)
    elif mode == 4:
        # Sine wave (naturally asymmetric on [0,1])
        a = np.abs(np.sin(np.pi * x)) ** 1.5 + 0.3 * np.abs(np.sin(2 * np.pi * x)) ** 2
    elif mode == 5:
        # Random smooth
        a = np.random.default_rng(rng.integers(10000)).uniform(0.5, 1.5, n)
        # Smooth it
        kernel = np.ones(10) / 10
        a = np.convolve(a, kernel, mode='same')
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def update_best(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = normalize(a.copy(), 1.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 0: Quick multi-n scan to determine best length
    print("\n=== Phase 0: Multi-length scan ===")
    scan_ns = [1200, 1600, 2000, 2400]
    scan_results = {}
    scan_time_total = min(time_budget * 0.12, 200.0)

    for n_scan in scan_ns:
        remaining_scan = scan_time_total - (time.monotonic() - t_global)
        if remaining_scan < 15:
            break
        scan_time = min(remaining_scan / max(1, len(scan_ns) - scan_ns.index(n_scan)), 60.0)
        if scan_time < 10:
            break

        if parent is not None:
            init_seq = interpolate_sequence(parent, n_scan)
        else:
            init_seq = structured_asymmetric_init(n_scan, rng, mode=0)

        a_norm = normalize(init_seq, 1.0)
        a_out, sc_out = nesterov_subgradient(a_norm, scan_time, k_peaks=12, verbose=False)
        scan_results[n_scan] = (a_out, sc_out)
        print(f"    n={n_scan}: score={sc_out:.8f} ({time.monotonic()-t_global:.0f}s)")

    # Pick best scan result
    if scan_results:
        best_n = min(scan_results, key=lambda k: scan_results[k][1])
        scan_best_a, scan_best_sc = scan_results[best_n]
        update_best(scan_best_a, f"scan_n{best_n}")
        n_base = best_n
    else:
        n_base = 1600

    if best_seq is None:
        best_seq = structured_asymmetric_init(n_base, rng, mode=0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Deep subgradient optimization at base length
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: SubGrad N={n_base} (budget={remaining:.0f}s) ===")
    t1 = remaining * 0.40
    a_init = normalize(best_seq, 1.0)
    a_out, _ = nesterov_subgradient(a_init, t1, k_peaks=16, verbose=True, eta_scale=1.0)
    update_best(a_out, f"subgrad_n{n_base}")

    # Phase 1b: Try different k_peaks and eta
    remaining = time_budget - (time.monotonic() - t_global)
    t1b = min(remaining * 0.10, 120.0)
    if t1b > 15:
        a_init = normalize(best_seq, 1.0)
        # Add small perturbation to escape local optimum
        pert = rng.normal(0, 0.01, len(a_init))
        a_init = normalize(a_init + pert, 1.0)
        a_out, _ = nesterov_subgradient(a_init, t1b, k_peaks=32, verbose=False, eta_scale=0.5)
        update_best(a_out, f"subgrad_k32_n{n_base}")

    # Phase 2: LSE + L-BFGS-B
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: LSE+LBFGBS n={n_cur} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.18, 250.0)
    if t2 > 20:
        a_init = normalize(best_seq, 1.0)
        a_out, _ = lse_scipy_optimize(a_init, t2, T_start=80.0, T_end=2.0, n_stages=5)
        update_best(a_out, f"lse_n{n_cur}")

    # Phase 3: Try longer length
    n_long = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: SubGrad n={n_long} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        init_long = interpolate_sequence(best_seq, n_long)
        a_init = normalize(init_long, 1.0)
        a_out, _ = nesterov_subgradient(a_init, t3, k_peaks=16, verbose=True)
        update_best(a_out, f"subgrad_long_n{n_long}")

    # Phase 4: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: CoordDescent n={n_cur} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 10:
        a_init = normalize(best_seq, 1.0)
        a_out, _ = coordinate_descent_polish(a_init, t4, seed=12345)
        update_best(a_out, "coord_polish")

    # Phase 5: Final LSE polish with low temperature
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: LSE polish n={n_cur} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.12, 150.0)
    if t5 > 10:
        a_init = normalize(best_seq, 1.0)
        a_out, _ = lse_scipy_optimize(a_init, t5, T_start=10.0, T_end=0.5, n_stages=4)
        update_best(a_out, "lse_polish")

    # Phase 6: Final coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: Final CoordDescent n={n_cur} (budget={remaining:.0f}s) ===")
    t6 = max(remaining * 0.4, 20.0)
    if t6 > 10:
        a_init = normalize(best_seq, 1.0)
        a_out, _ = coordinate_descent_polish(a_init, t6, seed=99999)
        update_best(a_out, "coord_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0] * 100

    return [float(x) for x in best_seq]
