    n = len(a0)
    S = np.sqrt(2.0 * n)  # target sum
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    
    # FFT setup
    size = 2 * n - 1
    fft_size = 1 << (3*n - 2 - 1).bit_length()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    n_stages = 10
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, T in enumerate(temps):
        stage_time = (time_limit * 0.9 - (time.monotonic() - t0)) / (n_stages - si)
        for it in range(100000):
            if time.monotonic() - t0 > time_limit * 0.95:
                break

            conv = fast_autocorr(a)  # length 2n-1
            max_conv = conv.max()
            
            # Compute LSE weights
            w = np.exp((conv - max_conv) / T)
            w /= w.sum()
            
            # Compute gradient via FFT
            # U = w (length 2n-1), V = a (length n)
            # c2[i] = (U * V)[i] for i=0..n-1
            # c1[i] = (U_rev * V)[2n-2-i] for i=0..n-1
            
            U = w  # length 2n-1
            V = a  # length n
            
            FV = rfft(V, fft_size)
            FU = rfft(U, fft_size)
            U_rev = U[::-1]
            FU_rev = rfft(U_rev, fft_size)
            
            # c2 = IFFT(FU * FV)[:n]
            conv_UV = irfft(FU * FV, fft_size)
            c2 = conv_UV[:n]
            
            # c1[i] = conv_UV_rev[2n-2-i]
            conv_Urev_V = irfft(FU_rev * FV, fft_size)
            c1 = conv_Urev_V[n-1:2*n-1][::-1]  # indices n-1 to 2n-2, reversed
            
            grad = 2.0 * (c1 + c2)
            
            # Adam update
            t += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad * grad
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            
            step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

            a = a_new

            if score < best_score:
                best_score = score
                best_a = a.copy()
    
    return best_a, best_score

import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def lse_gradient(a, T, fft_size):
    """Compute gradient of LSE approximation using FFT."""
    n = len(a)
    size = 2 * n - 1
    conv = fast_autocorr(a)
    max_conv = conv.max()
    w = np.exp((conv - max_conv) / T)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(size)
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= w_sum
    
    # Compute gradient via FFT
    FV = rfft(a, fft_size)
    FU = rfft(w, fft_size)
    w_rev = w[::-1]
    FU_rev = rfft(w_rev, fft_size)
    
    conv_wa = irfft(FU * FV, fft_size)
    term2 = conv_wa[:n]
    
    conv_wrev_a = irfft(FU_rev * FV, fft_size)
    term1 = conv_wrev_a[n-1:2*n-1][::-1]
    
    grad = 2.0 * (term1 + term2)
    return grad, conv, max_conv

def project_simplex(x, S):
    """Project x onto {y >= 0, sum(y) = S}."""
    n = len(x)
    if n == 0:
        return x
    # Sort in decreasing order
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - S))[0]
    if len(rho) == 0:
        return np.full(n, S / n)
    rho = rho[-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(x - theta, 0.0)

            break

        lr_min = lr_max * 0.01

def subgrad_momentum_optimize(a0, time_limit, verbose=True):

    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

    v = np.zeros(n)

    n_peaks = 5
    
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_autocorr(a)
        
        # Multi-peak weights
        max_conv = conv.max()
        # Use top peaks with softmax weights
        # Find top n_peaks
        if n_peaks < len(conv):
            top_idx = np.argpartition(conv, -n_peaks)[-n_peaks:]
            top_vals = conv[top_idx]
            weights = np.exp((top_vals - top_vals.max()) / (max_conv * 0.1 + 1e-10))
            weights /= weights.sum()
        else:
            top_idx = np.arange(len(conv))
            weights = np.ones(len(conv)) / len(conv)
        
        # Compute subgradient as weighted combination
        grad = np.zeros(n)
        for pi in range(len(top_idx)):
            k = int(top_idx[pi])
            w = weights[pi]
            # g_i = 2 * (a[k-i] + a[i-k])
            # a[k-i] valid when 0 <= k-i < n, i.e., k-n+1 <= i <= k
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_hi >= i_lo:
                grad[i_lo:i_hi+1] += 2 * w * a[k - np.arange(i_lo, i_hi+1)]
            # a[i-k] valid when 0 <= i-k < n, i.e., k <= i <= k+n-1
            i_lo2 = max(0, k)
            i_hi2 = min(n - 1, k + n - 1)
            if i_hi2 >= i_lo2:
                grad[i_lo2:i_hi2+1] += 2 * w * a[np.arange(i_lo2, i_hi2+1) - k]
        
        # Step size
        step = S / (np.sqrt(t) * max_conv)
        
        # Heavy ball momentum
        v = mu * v - step * grad
        a_new = a + v
        a_new = project_simplex(a_new, S)
        
        a = a_new
        
        # Check improvement

            if score < best_score:

            elapsed = time.monotonic() - t0
            print(f"    Subgrad t={t} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def lse_gradient(a, T, fft_size):
    """Compute gradient of LSE(T * log sum exp(conv/T)) using FFT."""
    n = len(a)
    size = 2 * n - 1
    conv = fast_autocorr(a)

    """Project x onto {y >= 0, sum(y) = S} in O(n log n)."""
    n = len(x)
