    return g


def _find_optimal_delta(conv, g, h, a_i, n_bisect=25):
    """Find delta in [0, a_i] minimizing max_k(conv_k + delta*g_k + h_k*delta^2).
    Uses bisection on the subgradient."""
    if a_i <= 0:
        return 0.0
    size = len(conv)

    def f(delta):
        return (conv + delta * g + h * delta * delta).max()

    # Bisection: find where the subgradient changes sign
    lo = 0.0
    hi = a_i

    # Check if f(hi) < f(0), if so try hi
    f_lo = f(0)
    f_hi = f(hi)
    if f_hi >= f_lo:
        return 0.0

    for _ in range(n_bisect):
        mid = (lo + hi) / 2.0
        f_mid = f(mid)
        if f_mid >= f_lo:
            hi = mid
        else:
            # Check derivative at mid
            vals = conv + mid * g + h * mid * mid
            k_star = vals.argmax()
            grad_at_mid = g[k_star] + 2 * h[k_star] * mid
            if grad_at_mid >= 0:
                hi = mid
            else:
                lo = mid
        if hi - lo < a_i * 1e-10:
            break

    best_delta = (lo + hi) / 2.0
    # Refine with a few more evaluations
    candidates = [lo, (lo + hi) / 2, hi, best_delta * 0.9, best_delta * 1.1]
    best_val = f(best_delta)
    for d in candidates:
        d = max(0.0, min(d, a_i))
        v = f(d)
        if v < best_val:
            best_val = v
            best_delta = d
    return best_delta


def _pair_cd(a, time_limit, rng, verbose=True):
    """Improved pair coordinate descent with bisection."""
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = _fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    h = np.zeros(size)  # second-order coefficients (computed per pair)

    n_passes = 0
    no_improve_count = 0
    pair_attempts_per_pass = 20000

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        n_tried = 0

        for _ in range(pair_attempts_per_pass):
            if time.monotonic() - t0 > time_limit:
                break
            # Pick random i and j with i having significant mass
            i = int(rng.integers(n))
            if a[i] < 1e-12:
                continue
            j = int(rng.integers(n))
            if j == i:
                continue

            n_tried += 1
            g = _compute_gain_vector_fast(a, i, j)

            # Second-order terms
            h = np.zeros(size)
            if 2 * i < size:
                h[2 * i] = 1.0
            if 2 * j < size:
                h[2 * j] = 1.0
            if i != j:
                kj = i + j
                if kj < size:
                    h[kj] = -2.0

            best_delta = _find_optimal_delta(conv, g, h, a[i])

            if best_delta > 1e-15:
                a[i] -= best_delta
                a[j] += best_delta
                s = a.sum()
                if s < 1e-15:
                    a[i] += best_delta
                    a[j] -= best_delta
                    continue
                a = a * norm / s
                conv = _fast_autocorr(a)
                cur_max = conv.max()
                if cur_max < best_score - 1e-14:
                    best_score = cur_max
                    best_a = a.copy()
                    pass_improved = True
                else:
                    # Revert
                    a[j] -= best_delta
                    a[i] += best_delta
                    a = a * norm / a.sum()
                    conv = _fast_autocorr(a)

        if not pass_improved:
            no_improve_count += 1
            if no_improve_count >= 5:
                break
        else:
            no_improve_count = 0

        if verbose and n_passes % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def _pair_cd_full_sweep(a, time_limit, rng, verbose=True):
    """Full-sweep pair CD: try all j for each i in order."""
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = _fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1

    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < 1e-12:
                continue

            for j in range(n):
                if j == i:
                    continue
                g = _compute_gain_vector_fast(a, i, j)
                h = np.zeros(size)
                if 2 * i < size:
                    h[2 * i] = 1.0
                if 2 * j < size:
                    h[2 * j] = 1.0
                if i != j:
                    kj = i + j
                    if kj < size:
                        h[kj] = -2.0

                best_delta = _find_optimal_delta(conv, g, h, ai, n_bisect=20)

                if best_delta > 1e-14:
                    a[i] -= best_delta
                    a[j] += best_delta
                    s = a.sum()
                    a = a * norm / s
                    conv = _fast_autocorr(a)
                    cur_max = conv.max()
                    if cur_max < best_score - 1e-14:
                        best_score = cur_max
                        best_a = a.copy()
                        pass_improved = True
                    else:
                        a[j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()
                        conv = _fast_autocorr(a)
                    ai = a[i]
                    if ai < 1e-12:
                        break

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD-full pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def _peak_targeting(a, time_limit, rng, verbose=True):
    """Target the top peaks specifically: for each top peak, find moves that reduce it."""
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = _fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    top_k = 10

    while time.monotonic() - t0 < time_limit:
        conv = _fast_autocorr(a)
        cur_max = conv.max()
        # Get top peak positions
        top_indices = np.argsort(conv)[-top_k:]
        top_indices = top_indices[::-1]  # descending order of peak

        improved = False
        for k_star in top_indices:
            if time.monotonic() - t0 > time_limit:
                break
            # Find which (i,j) pairs, if we move mass from i to j, reduce conv_k_star most
            # The gain at k_star for moving from i to j is:
            # 2*(a_{k_star - j} - a_{k_star - i}) (first order)
            # We want this to be negative (reduce the peak)
            # So we want a_{k_star - j} < a_{k_star - i}, i.e., a_j (at position k_star-j) < a_i (at position k_star-i)

            # Find the i that maximally contributes to peak k_star
            contrib = np.zeros(n)
            for i in range(n):
                v = 0.0
                j1 = k_star - i
                if 0 <= j1 < n:
                    v += a[j1]
                j2 = i - k_star
                if 0 <= j2 < n:
                    v += a[j2]
                contrib[i] = v

            # Try moving mass from high-contrib positions to low-contrib positions
            high_i = int(np.argmax(contrib))
            if a[high_i] < 1e-12:
                continue

            # Find best j: minimize the increase in max conv
            for attempt in range(50):
                if time.monotonic() - t0 > time_limit:
                    break
                j = int(rng.integers(n))
                if j == high_i:
                    continue
                g = _compute_gain_vector_fast(a, high_i, j)
                h = np.zeros(size)
                if 2 * high_i < size:
                    h[2 * high_i] = 1.0
                if 2 * j < size:
                    h[2 * j] = 1.0
                if high_i != j:
                    kj = high_i + j
                    if kj < size:
                        h[kj] = -2.0

                best_delta = _find_optimal_delta(conv, g, h, a[high_i], n_bisect=20)
                if best_delta > 1e-14:
                    a[high_i] -= best_delta
                    a[j] += best_delta
                    a = a * norm / a.sum()
                    conv = _fast_autocorr(a)
                    if conv.max() < best_score - 1e-14:
                        best_score = conv.max()
                        best_a = a.copy()
                        improved = True
                    else:
                        a[j] -= best_delta
                        a[high_i] += best_delta
                        a = a * norm / a.sum()
                        conv = _fast_autocorr(a)

        if not improved:
            # Try all-pairs for the top peak
            k_star = int(np.argmax(conv))
            n_pairs = min(5000, n * n // 10)
            for _ in range(n_pairs):
                if time.monotonic() - t0 > time_limit:
                    break
                i = int(rng.integers(n))
                if a[i] < 1e-12:
                    continue
                j = int(rng.integers(n))
                if j == i:
                    continue
                g = _compute_gain_vector_fast(a, i, j)
                h = np.zeros(size)
                if 2 * i < size:
                    h[2 * i] = 1.0
                if 2 * j < size:
                    h[2 * j] = 1.0
                if i != j:
                    kj = i + j
                    if kj < size:
                        h[kj] = -2.0
                best_delta = _find_optimal_delta(conv, g, h, a[i], n_bisect=20)
                if best_delta > 1e-14:
                    a[i] -= best_delta
                    a[j] += best_delta
                    a = a * norm / a.sum()
                    conv = _fast_autocorr(a)
                    if conv.max() < best_score - 1e-14:
                        best_score = conv.max()
                        best_a = a.copy()
                        improved = True
                    else:
                        a[j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()
                        conv = _fast_autocorr(a)

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    PeakTarget ({elapsed:.0f}s): best={best_score:.10f}")
            if elapsed > time_limit * 0.8:
                break

    return best_a, best_score


def _lp_best_response(a, M, solver, active_set=None):
    """Solve the LP best-response problem."""
    n = len(a)
    size = 2 * n - 1
    conv_current = _fast_autocorr(a)

    if active_set is None:
        # Select active constraints: those where conv is close to max
        threshold = M * 0.7
        active = np.where(conv_current >= threshold)[0]
        if len(active) < 5:
            active = np.where(conv_current >= M * 0.3)[0]
        if len(active) < 3:
            active = np.arange(size)
        active_set = active

    # Build constraint matrix for active set
    T_active = np.zeros((len(active_set), n))
    a_arr = np.asarray(a, dtype=np.float64)
    for idx, k in enumerate(active_set):
        for i in range(n):
            coeff = 0.0
            j1 = k - i
            if 0 <= j1 < n:
                coeff += a_arr[j1]
            j2 = i - k
            if 0 <= j2 < n:
                coeff += a_arr[j2]
            T_active[idx, i] = coeff

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_active @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, active_set

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, active_set

    b_val = np.maximum(np.asarray(b_val := b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None, active_set

    # Check if dropped constraints are violated
    full_conv = np.zeros(size)
    for i in range(n):
        if b_val[i] > 1e-12:
            # Add contribution of b_i to full_conv
            for k in range(i, i + n):
                if k < size:
                    full_conv[k] += b_val[i] * a_arr[k - i]
            for k in range(max(0, i - n + 1), i):
                full_conv[k] += b_val[i] * a_arr[i - k]

    violated = np.where(full_conv > M * 1.001)[0]
    if len(violated) > 0:
        new_active = np.union1d(active_set, violated)
        if len(new_active) > 0.95 * size:
            # Too many constraints, just return what we have
            pass
        else:
            # Re-solve with expanded active set
            T_new = np.zeros((len(new_active), n))
            for idx, k in enumerate(new_active):
                for i in range(n):
                    coeff = 0.0
                    j1 = k - i
                    if 0 <= j1 < n:
                        coeff += a_arr[j1]
                    j2 = i - k
                    if 0 <= j2 < n:
                        coeff += a_arr[j2]
                    T_new[idx, i] = coeff
            b_var2 = cp.Variable(n, nonneg=True)
            prob2 = cp.Problem(cp.Maximize(cp.sum(b_var2)), [T_new @ b_var2 <= M])
            try:
                prob2.solve(solver=solver, verbose=False)
            except Exception:
                active_set = new_active
                return b_val, active_set
            if prob2.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.maximum(np.asarray(b_var2.value, dtype=np.float64), 0.0)
            active_set = new_active

    return b_val, active_set


def _nesterov_lp(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    active_set = None

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    n_iters = 0
    stall_count = 0
    stall_best = best_score

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = _fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b, active_set = _lp_best_response(y, M_curr, solver, active_set)
        if b is None:
            # Retry with all constraints
            active_set = None
            b, _ = _lp_best_response(y, M_curr, solver, None)
            if b is None:
                continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = _fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        score_a = _fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        avg_a += a
        n_avg += 1
        if n_avg % 30 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                sc = _fast_autocorr(avg_check).max()
                if sc < best_score:
                    best_score = sc
                    best_a = avg_check.copy()
                    stall_count = 0

        # Restart mechanism
        if n_iters % 100 == 0:
            if best_score < stall_best - 1e-8:
                stall_best = best_score
                stall_count = 0
            else:
                stall_count += 1
            if stall_count >= 5:
                # Restart: perturb the current best
                y = best_a + 0.1 * np.random.default_rng(n_iters).standard_normal(n)
                y = np.maximum(y, 0)
                y = y * norm / y.sum()
                a = y.copy()
                avg_a = np.zeros(n)
                n_avg = 0
                stall_count = 0
                stall_best = best_score
                active_set = None

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def _lse_gradient(a0, time_limit, verbose=True):
    """LSE smooth gradient descent with momentum (Adam)."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    temps = np.geomspace(50.0, 1.0, 12)
    step_scales = np.geomspace(0.5, 0.01, 12)

    for si, (T, step_scale) in enumerate(zip(temps, step_scales)):
        if time.monotonic() - t0 > time_limit:
            break
        n_steps = max(20, min(200, int((time_limit - (time.monotonic() - t0)) * 5)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            conv = _fast_autocorr(a)
            max_conv = conv.max()
            exp_conv = np.exp((conv - max_conv) / T)
            w = exp_conv / exp_conv.sum()

            # Compute gradient: grad_i = 2 * (sum_k w_k * a_{k-i} + sum_k w_k * a_{i-k})
            # This is 2 * (cross_corr(w, a))[i] essentially
            # Use FFT for efficiency
            fft_sz = 1 << (size - 1).bit_length()
            W_fft = rfft(w, fft_sz)
            A_fft = rfft(a, fft_sz)
            # cross1[i] = sum_k w[k] * a[k+i]
            cross1 = irfft(W_fft * np.conj(A_fft), fft_sz)
            g1 = cross1[:n]
            # cross2: sum_k w[k] * a[i-k] = sum_k a[i-k] * w[k]
            # = cross_corr(a, w) at lag i
            cross2 = irfft(A_fft * np.conj(W_fft), fft_sz)
            g2 = np.zeros(n)
            for i in range(n):
                # sum_k a[i-k] * w[k]: valid k where 0<=i-k<n and 0<=k<n
                # i-k >= 0 => k <= i; i-k < n => k > i-n
                k_lo = max(0, i - n + 1)
                k_hi = min(n, i + 1)
                if k_hi > k_lo:
                    g2[i] = np.dot(w[k_lo:k_hi], a[i - np.arange(k_lo, k_hi)])

            grad = 2.0 * (g1 + g2)

            # Adam update
            t_adam = it + 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)

            lr = step_scale * norm / np.sqrt(n)
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s
            a = a_new

            score = _fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE T={T:.1f} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function: combines LP Nesterov, pair CD, peak targeting, and LSE."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1720.0)

    solver = _get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {_lp_solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = _evaluate_safe(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None
        best_seq = None

    def do_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = _evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize if no parent
    if best_seq is None:
        n_base = 1600
        best_seq = _structured_init(n_base, rng)
        best_score = _evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n_base = len(best_seq)

    # === Phase 1: LP Nesterov for global structure (if parent is not already good) ===
    remaining = time_budget - (time.monotonic() - t_global)
    # If we have a parent with score < 1.51, skip most of the LP phase
    lp_time = min(remaining * 0.25, 400.0) if best_score > 1.51 else min(remaining * 0.10, 150.0)
    if lp_time > 20:
        print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget={lp_time:.0f}s) ===")
        a_norm = _normalize(best_seq)
        a_out, _ = _nesterov_lp(a_norm, lp_time, solver, verbose=True)
        do_update(a_out, f"lp_n{n_base}")

    # === Phase 2: Random pair CD (main local optimizer) ===
    remaining = time_budget - (time.monotonic() - t_global)
    nc = len(best_seq)
    pcd_time = min(remaining * 0.35, 500.0)
    if pcd_time > 20:
        print(f"\n=== Phase 2: Random Pair CD n={nc} (budget={pcd_time:.0f}s) ===")
        a_norm = _normalize(best_seq)
        a_out, _ = _pair_cd(a_norm, pcd_time, rng, verbose=True)
        do_update(a_out, f"rcd_n{nc}")

    # === Phase 3: Peak targeting ===
    remaining = time_budget - (time.monotonic() - t_global)
    nc = len(best_seq)
    pt_time = min(remaining * 0.20, 300.0)
    if pt_time > 20:
        print(f"\n=== Phase 3: Peak Targeting n={nc} (budget={pt_time:.0f}s) ===")
        a_norm = _normalize(best_seq)
        a_out, _ = _peak_targeting(a_norm, pt_time, rng, verbose=True)
        do_update(a_out, f"pt_n{nc}")

    # === Phase 4: More pair CD after peak targeting ===
    remaining = time_budget - (time.monotonic() - t_global)
    nc = len(best_seq)
    pcd2_time = min(remaining * 0.20, 300.0)
    if pcd2_time > 20:
        print(f"\n=== Phase 4: Pair CD round 2 n={nc} (budget={pcd2_time:.0f}s) ===")
        a_norm = _normalize(best_seq)
        a_out, _ = _pair_cd(a_norm, pcd2_time, rng, verbose=True)
        do_update(a_out, f"rcd2_n{nc}")

    # === Phase 5: LSE gradient for smooth refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    nc = len(best_seq)
    lse_time = min(remaining * 0.10, 150.0)
    if lse_time > 15:
        print(f"\n=== Phase 5: LSE Gradient n={nc} (budget={lse_time:.0f}s) ===")
        a_norm = _normalize(best_seq)
        a_out, _ = _lse_gradient(a_norm, lse_time, verbose=True)
        do_update(a_out, f"lse_n{nc}")

    # === Phase 6: Try a different length ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        n_alt = n_base + 400
        print(f"\n=== Phase 6: Try n={n_alt} (budget={remaining:.0f}s) ===")
        init_alt = _interpolate(best_seq, n_alt)
        a_norm = _normalize(init_alt, target_sum=np.sqrt(2.0 * n_alt))
        # Quick LP
        t_lp = min(remaining * 0.15, 120.0)
        if t_lp > 20:
            a_out, _ = _nesterov_lp(a_norm, t_lp, solver, verbose=False)
            do_update(a_out, f"lp_n{n_alt}")
        # Pair CD on new length
        remaining = time_budget - (time.monotonic() - t_global)
        nc_alt = len(best_seq)
        t_pcd = min(remaining * 0.40, 200.0)
        if t_pcd > 20:
            a_norm2 = _normalize(best_seq)
            a_out2, _ = _pair_cd(a_norm2, t_pcd, rng, verbose=False)
            do_update(a_out2, f"rcd_n{nc_alt}")

    # === Phase 7: Final intensive pair CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    nc = len(best_seq)
    final_time = max(remaining * 0.5, 30.0) if remaining > 60 else remaining
    if final_time > 20:
        print(f"\n=== Phase 7: Final Pair CD n={nc} (budget={final_time:.0f}s) ===")
        a_norm = _normalize(best_seq)
        a_out, _ = _pair_cd(a_norm, final_time, rng, verbose=True)
        do_update(a_out, f"final_rcd_n{nc}")

    # === Final verification ===
    if best_seq is not None:
        fs = _evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0] * 100

    return [float(x) for x in best_seq]
