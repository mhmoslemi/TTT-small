So T[k, i] = a[k-i] if 0 <= k-i < n, else 0.

Let me verify with a small example:
a = [a0, a1, a2], b = [b0, b1, b2]
conv(a, b) = [a0*b0, a0*b1+a1*b0, a0*b2+a1*b1+a2*b0, a1*b2+a2*b1, a2*b2]

Constraint 0: a0*b0 <= M → T[0,:] = [a0, 0, 0]
Constraint 1: a0*b1 + a1*b0 <= M → T[1,:] = [a1, a0, 0]
Constraint 2: a0*b2 + a1*b1 + a2*b0 <= M → T[2,:] = [a2, a1, a0]
Constraint 3: a1*b2 + a2*b1 <= M → T[3,:] = [0, a2, a1]
Constraint 4: a2*b2 <= M → T[4,:] = [0, 0, a2]

So T[k, i] = a[k-i] if 0 <= k-i < n.

For k=0: T[0,i] = a[0-i] → only i=0 gives a[0]. ✓
For k=1: T[1,i] = a[1-i] → i=0 gives a[1], i=1 gives a[0]. ✓
For k=2: T[2,i] = a[2-i] → i=0 gives a[2], i=1 gives a[1], i=2 gives a[0]. ✓

def fast_autocorr(a):

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix_sparse(a):
    """Build sparse convolution matrix T where T[k,i] = a[k-i] if 0<=k-i<n."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            rows.append(k)
            for i in idx:
                cols.append(i)
                vals.append(a[k - i])
    # Vectorized build
    row_idx = []
    col_idx = []
    val = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            row_idx.extend([k] * len(idx))
            col_idx.extend(idx.tolist())
            val.extend((a[k - idx]).tolist())
    return csr_matrix((val, (row_idx, col_idx)), shape=(2 * n - 1, n))

def lp_best_response(a, M, solver, prev_b=None):
    """Solve LP: max sum(b) s.t. conv(a,b)_k <= M for all k, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    
    # Build constraint matrix
    T = build_conv_matrix_sparse(a)
    
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    if b_var.value is None:
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def lp_best_response_dual(a, M, solver):
    """Solve LP and return both primal and dual solution."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_conv_matrix_sparse(a)
    
    constraints = [T @ b_var <= M]
    con = constraints[0]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    if b_var.value is None:
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    
    # Get dual values
    dual_val = None
    try:
        if con.dual_value is not None:
            dual_val = np.maximum(np.asarray(con.dual_value, dtype=np.float64), 0.0)
    except Exception:
        pass
    
    if b_val.sum() < 1e-15:
        return None, dual_val
    return b_val, dual_val

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, use_dual=False):
    """Nesterov-accelerated LP best-response with optional dual-based gradient."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * c / n
    else:
        a = a * c / s0
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    avg_a = a.copy()
    n_avg = 1
    
    # Track improvement ratio for convergence
    prev_ratio = float('inf')
    stall_count = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        
        if use_dual:
            b, dual = lp_best_response_dual(y, M_curr, solver)
        else:
            b = lp_best_response(y, M_curr, solver)
            dual = None
        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        
        # Improvement ratio
        ratio = b_sum / c
        if ratio > prev_ratio * (1 + 1e-6):
            stall_count += 1
        else:
            stall_count = 0
        prev_ratio = max(prev_ratio, ratio)
        
        # If improvement is tiny, we might be converged
        if ratio < 1.0 + 1e-8 and t > 50:
            if stall_count > 30:
                break
            stall_count += 1
        
        g = b * c / b_sum
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * c / s_y
        
        if t == 1:
            x_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new * c / s_x
        
        y = y_new
        x = x_new
        
        # Track averages (Nesterov's averaged iterate)
        avg_a += x
        n_avg += 1
        
        # Check all iterates
        for cand, label in [(y, 'y'), (x, 'x')]:
            M_cand = fast_autocorr(cand).max()
            if M_cand < best_m:
                best_m = M_cand
                best_a = cand.copy()
        
        # Check average every 10 steps
        if n_avg % 10 == 0:
            avg_check = avg_a / n_avg
            avg_check = np.maximum(avg_check, 0.0)
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * c / s_avg
                M_avg = fast_autocorr(avg_check).max()
                if M_avg < best_m:
                    best_m = M_avg
                    best_a = avg_check.copy()
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c * c)
            print(f"    LP iter={t} ({elapsed:.1f}s): J={J:.6f}, ratio={ratio:.6f}")
    
    # Final check of average
    if n_avg > 1:
        avg_final = avg_a / n_avg
        avg_final = np.maximum(avg_final, 0.0)
        s_af = avg_final.sum()
        if s_af > 1e-15:
            avg_final = avg_final * c / s_af
            M_af = fast_autocorr(avg_final).max()
            if M_af < best_m:
                best_m = M_af
                best_a = avg_final.copy()
    
    return best_a, best_m

def coord_descent_polish(a0, time_limit, verbose=False):
    """Coordinate descent: move mass between pairs to reduce peak convolution."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * c / n
    else:
        a = a * c / s0
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    ac = fast_autocorr(a)
    M = ac.max()
    
    max_iters = 50000
    iter_count = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 500 and iter_count < max_iters:
        iter_count += 1
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:5]
        improved = False
        
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]
            
            # Find good destination: minimize impact on top peaks
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                k2_int = int(k2)
                js = max(0, k2_int - n + 1)
                je = min(n - 1, k2_int)
                if js <= je:
                    peak_mask[js:je + 1] = True
            
            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]
            if len(candidates) == 0:
                continue
            
            # Try several fractions
            best_trial = None
            best_trial_m = best_m
            for frac in [0.01, 0.02, 0.05, 0.1]:
                amt = a[src] * frac
                if amt < 1e-20:
                    continue
                for dst in candidates[np.argsort(a[candidates])[:5]]:
                    if dst == src:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-15:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial * c / s_t
                    ac_trial = fast_autocorr(a_trial)
                    M_trial = ac_trial.max()
                    if M_trial < best_trial_m - 1e-14:
                        best_trial_m = M_trial
                        best_trial = a_trial.copy()
            
            if best_trial is not None:
                a = best_trial
                best_m = best_trial_m
                best_a = a.copy()
                improved = True
                stall = 0
                break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            if a.max() > c * 1e-6:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > c * 1e-8:
                    frac = rng.uniform(0.02, 0.1)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    if s_t > 1e-15:
                        a = a * c / s_t
                        M_new = fast_autocorr(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0.0)
                            a = a * c / a.sum()
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    Coord polish ({elapsed:.1f}s, {iter_count} iters): best_J={2.0*n*best_m/(c*c):.6f}")
    
    return best_a, best_m

def mirror_descent_fast(a0, time_limit, top_k=4, beta=2.5, verbose=False):
    """Fast mirror descent with exponential weights."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * c / n
    else:
        a = a * c / s0
    a = np.maximum(a, 1e-15)
    a = a * c / a.sum()
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    avg_a = a.copy()
    n_avg = 1
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        step = 1.0 / (np.sqrt(t) + 1.0)
        
        ac = fast_autocorr(a)
        M = ac.max()
        
        # Compute subgradient from top-k peaks
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 0)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n, k_int + 1)
            idx = np.arange(i_start, i_end)
            j = k_int - idx
            grad[idx] += 2.0 * wi * a[j]
        
        # Mirror descent update
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - step * grad
        max_log = log_a_new.max()
        a_new = np.exp(log_a_new - max_log)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * c / s_new
        a = a_new
        
        M_new = fast_autocorr(a).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a.copy()
        
        avg_a += a
        n_avg += 1
        if n_avg % 200 == 0:
            avg_check = avg_a / n_avg
            avg_check = np.maximum(avg_check, 0.0)
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * c / s_avg
                M_avg = fast_autocorr(avg_check).max()
                if M_avg < best_m:
                    best_m = M_avg
                    best_a = avg_check.copy()
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c * c)
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def raised_cosine(n):
    """Raised cosine (Hanning) window."""
    x = np.arange(n)
    return 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))

def proakis_window(n, alpha=0.5):
    """Proakis window."""
    x = np.arange(n)
    x_norm = x / (n - 1) - 0.5
    return np.cos(np.pi * (2 * alpha / (1 + alpha)) * x_norm + (np.pi / 2) * (1 - alpha) / (1 + alpha))

            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent / parent.sum()
            best_score = p_score
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 1600
        best_seq = raised_cosine(n_base)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    
    # Phase 1: LP optimization at parent/best length (main phase, 50% of budget)
    n0 = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP Nesterov n={n0} (main) ===")
    t1 = remaining * 0.45
    if t1 > 30:
        c0 = np.sqrt(2.0 * n0)
        a_norm = best_seq * c0 / best_seq.sum()
        a_out, m_out = nesterov_lp_optimize(a_norm, t1, solver, verbose=True, use_dual=False)
        try_update(a_out, f"lp_main_n{n0}")
    
    # Phase 2: Mirror descent at best length (10%)
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: Mirror Descent n={n_cur} ===")
    t2 = remaining * 0.15
    if t2 > 15:
        c_cur = np.sqrt(2.0 * n_cur)
        a_norm = best_seq * c_cur / best_seq.sum()
        a_out, _ = mirror_descent_fast(a_norm, t2, top_k=4, beta=2.5, verbose=True)
        try_update(a_out, f"md_n{n_cur}")
    
    # Phase 3: Try a longer length (15%)
    n_long = min(n_cur + 400, 2400)
    remaining = time_budget - (time.monotonic() - t_global)
    if n_long > n_cur:
        print(f"\n=== Phase 3: LP Nesterov n={n_long} (longer) ===")
        t3 = remaining * 0.25
        if t3 > 30:
            a_long = interpolate_sequence(best_seq, n_long)
            c_long = np.sqrt(2.0 * n_long)
            a_norm = a_long * c_long / a_long.sum()
            a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=True, use_dual=False)
            try_update(a_out, f"lp_long_n{n_long}")
    
    # Phase 4: Coordinate polish at best (10%)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Coord Polish n={n_best} ===")
    t4 = remaining * 0.20
    if t4 > 10:
        c_best = np.sqrt(2.0 * n_best)
        a_norm = best_seq * c_best / best_seq.sum()
        a_out, _ = coord_descent_polish(a_norm, t4, verbose=True)
        try_update(a_out, f"coord_n{n_best}")
    
    # Phase 5: Final LP polish (20%)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Final LP Polish n={n_best} ===")
    t5 = remaining * 0.60
    if t5 > 20:
        c_best = np.sqrt(2.0 * n_best)
        a_norm = best_seq * c_best / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_out if 'a_out' in dir() else a_norm, t5, solver, verbose=True, use_dual=False)
        try_update(a_out, f"lp_final_n{n_best}")
    
    # Phase 6: Final coord polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 10:
        t6 = min(remaining * 0.5, 60.0)
        if t6 > 5:
            print(f"\n=== Phase 6: Final Coord Polish n={n_best} ===")
            c_best = np.sqrt(2.0 * n_best)
            a_norm = best_seq * c_best / best_seq.sum()
            a_out, _ = coord_descent_polish(a_norm, t6, verbose=True)
            try_update(a_out, f"coord_final_n{n_best}")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
