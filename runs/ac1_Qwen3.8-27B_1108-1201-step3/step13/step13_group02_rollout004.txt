  import numpy as np
  import time
  from scipy.fft import rfft, irfft

  def fast_autocorr(a):
      n = len(a)
      size = 2 * n - 1
      fft_size = 1 << (size - 1).bit_length()
      A = rfft(np.asarray(a, dtype=np.float64), fft_size)
      return irfft(A * A, fft_size)[:size]

  def obj_value(a):
      """Compute 2*n*max(ac)/(sum(a))^2"""
      n = len(a)
      ac = fast_autocorr(a)
      s = a.sum()
      if s < 0.01:
          return float('inf')
      return 2.0 * n * ac.max() / (s * s)
