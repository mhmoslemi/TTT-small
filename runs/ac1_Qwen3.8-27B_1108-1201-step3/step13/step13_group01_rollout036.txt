    for s in [cp.CLARABEL, cp.ECOS, cp.SCS]:

def fast_autocorr(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64).ravel()]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:

def build_T_matrix(a):
    """Build the Toeplitz-like matrix T where (T@b)[k] = (a*b)[k]."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        T[start:end, i] = a[:end - start]
    return T

def lp_best_response(a, M, solver):
    """Solve: max sum(b) s.t. (a*b) <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_T_matrix(a)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)

        return None

        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:

    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:

def nesterov_lp(a0, time_limit, solver, verbose=True):
    """LP-based Nesterov acceleration for min max_k (a*a)[k]."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_peak = fast_autocorr(a).max()
    
    y = a.copy()  # Nesterov's y (extrapolated point)
    t0 = time.monotonic()
    last_improve = t0
    stall = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute current peak at y
        b_y = fast_autocorr(y)
        M_curr = b_y.max()
        if M_curr < 1e-12:
            break

        # Solve LP best response
        b_lp = lp_best_response(y, M_curr, solver)
        if b_lp is None:
            continue

        b_sum = b_lp.sum()
        if b_sum < 1e-15:
            continue

        # Normalize g to have the same "norm"
        g = b_lp * norm / b_sum

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        # Nesterov extrapolation for a
        if t == 1:
            a_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            a_new = y_new + beta * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        # Check improvement
        peak_y = fast_autocorr(y).max()
        if peak_y < best_peak:
            best_peak = peak_y
            best_a = y.copy()
            last_improve = time.monotonic()

        peak_a = fast_autocorr(a).max()
        if peak_a < best_peak:
            best_peak = peak_a
            best_a = a.copy()
            last_improve = time.monotonic()

        # Stall detection
        if time.monotonic() - last_improve > 60.0:
            stall += 1

                break

            stall = 0

            print(f"    NLP t={t} ({time.monotonic()-t0:.0f}s): peak={best_peak:.8f}")

def lse_objective_and_grad(x, n, T_temp):
    """Compute LSE surrogate objective and gradient."""
    a = np.maximum(x.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    
    exp_vals = np.exp(T_temp * (b - max_b))
    exp_sum = exp_vals.sum()
    w = exp_vals / exp_sum
    lse = max_b + np.log(exp_sum) / T_temp
    
    # Gradient: grad[j] = 2 * sum_l a[l] * w[j+l]
    # Compute efficiently
    grad = np.zeros(n)
    for j in range(n):
        # sum over l: a[l] * w[j+l] where 0 <= j+l < 2n-1
        l_max = min(n, 2*n-1-j)
        if l_max > 0:
            grad[j] = 2.0 * np.dot(a[:l_max], w[j:j+l_max])
    
    # Full objective gradient (with normalization)
    obj = 2.0 * n * lse / (s * s)
    full_grad = grad / (s * s) - 2.0 * obj * a / s
    
    return obj, full_grad.ravel()

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):

def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        end = min(n, size - i)
        T[i:i+end, i] = a[:end]
    return T

def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_T(a)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def nesterov_lp(a0, time_limit, solver, verbose=False):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_peak = fast_autocorr(a).max()
    y = a.copy()
    t0 = time.monotonic()
    last_improve = t0
    stall = 0
    avg_a = np.zeros(n)
    n_avg = 0
    iters_done = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b_lp = lp_best_response(y, M_curr, solver)
        if b_lp is None:
            stall += 1
            if stall > 20:
                break
            continue
        stall = 0

        b_sum = b_lp.sum()
        if b_sum < 1e-15:
            continue
        g = b_lp * norm / b_sum
        iters_done += 1

        # Adaptive alpha: slower when improvement is small
        alpha = 2.0 / (t + 2.0)
        if t > 100:
            alpha = max(alpha, 0.005)  # floor to prevent too-slow decay

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        peak_y = fast_autocorr(y).max()
        if peak_y < best_peak:
            best_peak = peak_y
            best_a = y.copy()
            last_improve = time.monotonic()

        peak_a = fast_autocorr(a).max()
        if peak_a < best_peak:
            best_peak = peak_a
            best_a = a.copy()
            last_improve = time.monotonic()

        # Polyak averaging
        avg_a += a
        n_avg += 1
        if n_avg % 30 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                peak_avg = fast_autocorr(avg_check).max()
                if peak_avg < best_peak:
                    best_peak = peak_avg
                    best_a = avg_check.copy()
                    last_improve = time.monotonic()

        if time.monotonic() - last_improve > 90.0:
            stall += 1
            if stall > 10:
                if verbose:
                    print(f"    NLP stall at t={t} ({iters_done} LPs), best={best_peak:.8f}")
                break
        else:
            stall = 0

        if verbose and iters_done % 50 == 0:
            print(f"    NLP iter={iters_done} ({time.monotonic()-t0:.0f}s): best={best_peak:.8f}")

    return best_a, best_peak

def lse_obj(x, n, T_temp):
    a = np.maximum(x.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10
    ev = np.exp(T_temp * (b - max_b))
    lse = max_b + np.log(ev.sum()) / T_temp
    return 2.0 * n * lse / (s * s)

def lse_grad(x, n, T_temp):
    a = np.maximum(x.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return np.zeros(n)
    ev = np.exp(T_temp * (b - max_b))
    w = ev / ev.sum()
    lse = max_b + np.log(ev.sum()) / T_temp

    # grad[j] = 2 * sum_l a[l] * w[j+l] for valid indices
    # = 2 * convolve(a, w_flipped)[2n-2-j]
    w_flipped = w[::-1]
    conv_res = np.convolve(a, w_flipped, mode='full')
    # conv_res has length n + (2n-1) - 1 = 3n-2
    # grad[j] = 2 * conv_res[2n-2-j] for j=0,...,n-1
    grad = 2.0 * conv_res[2*n-2 : n-2 : -1]

    obj_val = 2.0 * n * lse / (s * s)
    full_grad = grad / (s * s) - 2.0 * obj_val * a / s
    return full_grad.ravel()

def lse_optimize(a0, time_limit, T_start=30.0, T_end=2.0, n_stages=5):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_peak = fast_autocorr(a).max()
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        stage_time = min(remaining / (n_stages - si), 100.0)

        def obj_func(x):
            return lse_obj(x, n, T)

        def grad_func(x):
            return lse_grad(x, n, T)

        try:
            res = scipy_minimize(obj_func, a.copy(), method='L-BFGS-B',
                                  jac=grad_func, bounds=[(0, 1000)] * n,
                                  options={'maxiter': 300, 'maxfun': 800,
                                           'time_limit': stage_time, 'gtol': 1e-6})
            if res.x is not None:
                a_cand = np.maximum(res.x, 0.0)
                s_c = a_cand.sum()
                if s_c > 1e-10:
                    a_cand = a_cand * norm / s_c
                    peak_c = fast_autocorr(a_cand).max()
                    if peak_c < best_peak:
                        best_peak = peak_c
                        best_a = a_cand.copy()
                        a = a_cand
        except Exception:
            pass

    return best_a, best_peak

def coord_descent_polish(a0, time_limit, seed=42):
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_peak = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_peaks = np.argsort(b)[::-1][:12]

        improved = False
        for k_peak in top_peaks:
            # Which indices contribute to this peak?
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_peak - idx]
            # Top contributors
            top_c = idx[np.argsort(contrib)[::-1][:8]]

            # Build mask of indices that affect any top peak
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:6]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            for src in top_c:
                if a[src] < 1e-12:
                    continue
                # Find good destinations: low value, outside peak mask
                candidates = np.where(~peak_mask)[0]
                if len(candidates) == 0:
                    candidates = np.where(np.arange(n) != src)[0]
                # Sort by value (prefer moving to low-value positions)
                candidates_sorted = candidates[np.argsort(a[candidates])][:max(4, len(candidates) // 4)]

                for dst in candidates_sorted[:4]:
                    for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-10:
                            continue
                        a_trial = a_trial * norm / s_t
                        peak = fast_autocorr(a_trial).max()
                        if peak < best_peak - 1e-14:
                            a = a_trial
                            best_peak = peak
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random exploration when stuck
            if stall % 5 == 0:
                for _ in range(30):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i == j or a[i] < 1e-12:
                        continue
                    frac = rng.uniform(0.005, 0.08)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm / s_t
                    peak = fast_autocorr(a_trial).max()
                    if peak < best_peak:
                        a = a_trial
                        best_peak = peak
                        best_a = a.copy()
                        stall = 0
                        break
            if stall > 300:
                break

    return best_a, best_peak

        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_star - idx]
        top_c = idx[np.argsort(contrib)[::-1][:6]]

        improved = False
        for src in top_c[:4]:
            if a[src] < 1e-12:
                continue
            # Try 2D moves: reduce src, adjust another peak contributor
            for src2 in top_c:
                if src2 == src or a[src2] < 1e-12:
                    continue
                for f1 in [0.01, 0.03]:
                    for f2 in [0.01, 0.03]:
                        amt1 = a[src] * f1
                        amt2 = a[src2] * f2
                        a_trial = a.copy()
                        a_trial[src] -= amt1
                        a_trial[src2] -= amt2
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-10:
                            continue
                        # Redistribute to random low positions
                        a_trial += (amt1 + amt2) * np.ones(n) / n
                        a_trial = a_trial * norm / a_trial.sum()
                        peak = fast_autocorr(a_trial).max()
                        if peak < best_peak - 1e-14:
                            a = a_trial
                            best_peak = peak
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 200:
                break

    return best_a, best_peak

def subgrad_nesterov_fast(a0, time_limit, verbose=False):
    """Fast subgradient Nesterov (no LP, just peak gradient)."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_peak = fast_autocorr(a).max()
    t0 = time.monotonic()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    last_improve = t0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(y)
        k_star = int(np.argmax(b))
        M_curr = b[k_star]

        # Subgradient: grad[j] = 2*a[k_star-j] if valid
        grad = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        grad[js:je+1] = 2.0 * a[k_star - np.arange(js, je + 1)]

        # Frank-Wolfe: find argmin grad, put all mass there
        i_min = int(np.argmin(grad))
        v = np.zeros(n)
        v[i_min] = norm

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * v
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = v.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        peak_a = fast_autocorr(a).max()
        if peak_a < best_peak:
            best_peak = peak_a
            best_a = a.copy()
            last_improve = time.monotonic()

        avg_a += a
        n_avg += 1
        if n_avg % 500 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                peak_avg = fast_autocorr(avg_check).max()
                if peak_avg < best_peak:
                    best_peak = peak_avg
                    best_a = avg_check.copy()
                    last_improve = time.monotonic()

        if time.monotonic() - last_improve > 120.0:
            break

    return best_a, best_peak

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

def structured_init(n, rng):
    choice = rng.integers(0, 8)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x)
    elif choice == 3:
        a = np.exp(-4 * (x - 0.3) ** 2) + 0.7 * np.exp(-4 * (x - 0.7) ** 2)
    elif choice == 4:
        a = np.exp(-3 * (x - 0.5) ** 2)
    elif choice == 5:
        a = 1.0 + 0.3 * np.sin(3 * np.pi * x)
    elif choice == 6:
        a = (1 - x) * x + 0.1
    else:
        a = 1.0 + 0.2 * np.cos(5 * np.pi * x) + 0.1 * np.cos(11 * np.pi * x)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def propose_candidate(seed=42, budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)
    rng = np.random.default_rng(seed)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        best_seq = structured_init(1600, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 0: Multi-length scan
    print("\n=== Phase 0: Quick multi-length scan ===")
    scan_lengths = [1600, 2000, 2400]
    scan_results = {}

    scan_time_total = time_budget * 0.12
    per_scan = scan_time_total / len(scan_lengths)

    for n_scan in scan_lengths:
        if time.monotonic() - t_global > time_budget * 0.15:
            break
        if best_seq is not None:
            init_seq = interpolate_seq(best_seq, n_scan)
        else:
            init_seq = structured_init(n_scan, rng)

        a_norm = init_seq * np.sqrt(2.0 * n_scan) / init_seq.sum()
        a_out, sc_out = nesterov_lp(a_norm, per_scan, solver, verbose=False)
        scan_results[n_scan] = (a_out, sc_out)
        print(f"    n={n_scan}: peak={sc_out:.8f} (t={time.monotonic()-t_global:.0f}s)")

    # Pick best from scan
    if scan_results:
        best_scan_n = min(scan_results, key=lambda k: scan_results[k][1])
        try_update(scan_results[best_scan_n][0], f"scan_n{best_scan_n}")

    n_base = len(best_seq)

    # Phase 1: Deep LP-Nesterov at best length
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.35
    print(f"\n=== Phase 1: Deep LP n={n_base} ({t1:.0f}s) ===")
    a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
    a_out, _ = nesterov_lp(a_norm, t1, solver, verbose=True)
    try_update(a_out, f"lp_deep_n{n_base}")

    # Phase 2: Try a longer length
    n_long = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 30:
        print(f"\n=== Phase 2: LP n={n_long} ({t2:.0f}s) ===")
        init_long = interpolate_seq(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = nesterov_lp(a_norm, t2, solver, verbose=True)
        try_update(a_out, f"lp_long_n{n_long}")

    # Phase 3: LSE optimization
    n_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        print(f"\n=== Phase 3: LSE n={n_best} ({t3:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_optimize(a_norm, t3, T_start=40.0, T_end=3.0, n_stages=5)
        try_update(a_out, f"lse_n{n_best}")

    # Phase 4: Coordinate descent polish
    n_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.20, 250.0)
    if t4 > 15:
        print(f"\n=== Phase 4: Coord descent n={n_best} ({t4:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coord_descent_polish(a_norm, t4, seed=12345)
        try_update(a_out, "coord1")

    # Phase 5: 2D coordinate descent
    n_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.12, 150.0)
    if t5 > 10:
        print(f"\n=== Phase 5: 2D coord n={n_best} ({t5:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coord_2d_polish(a_norm, t5, seed=777)
        try_update(a_out, "coord2d")

    # Phase 6: Subgradient Nesterov (different trajectory)
    n_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.10, 120.0)
    if t6 > 10:
        print(f"\n=== Phase 6: SubGrad n={n_best} ({t6:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = subgrad_nesterov_fast(a_norm, t6, verbose=False)
        try_update(a_out, f"subgrad_n{n_best}")

    # Phase 7: Final coordinate descent
    n_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    t7 = remaining * 0.30
    if t7 > 10:
        print(f"\n=== Phase 7: Final coord n={n_best} ({t7:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coord_descent_polish(a_norm, t7, seed=55555)
        try_update(a_out, "coord_final")

    # Final check
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
