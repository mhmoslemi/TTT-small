```python
def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}"""
    n = len(v)
    # Water-filling
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - S))[0][-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
```

```python
def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}"""
    n = len(v)
```

            y = a.copy()
        else:

            y = np.maximum(y, 0.0)

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val
