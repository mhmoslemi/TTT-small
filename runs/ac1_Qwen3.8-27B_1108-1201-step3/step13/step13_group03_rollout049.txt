def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def _proj_simplex(v):
    n = len(v)
    if n <= 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 0:
        out /= s
    else:
        out = np.ones(n) / n
    return out


def _lse_grad(y, T):
    n = len(y)
    ac = _fast_autocorr(y)
    ac_max = ac.max()
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(y, fft_size)
    W = rfft(w, fft_size)
    cross = irfft(np.conj(A) * W, fft_size)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad

def _subgradient_method(a0, time_limit, verbose=True):

    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

    x = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = _fast_autocorr(x)
        k_star = int(np.argmax(ac))
        M = ac[k_star]

        # d/dx_i [(a*a)_k] = a_{k-i} + a_{i-k} (when indices valid)

            if 0 <= j2 < n:
                grad[i] += 2.0 * x[j2]

        grad += 1e-8 * x

        # Step size
        step = norm / (np.sqrt(t) * (np.linalg.norm(grad) + 1e-15))
        x_new = x - step * grad
        x_new = np.maximum(x_new, 0.0)
        s_x = x_new.sum()
        if s_x < 1e-15:
            x_new = x.copy()
        else:
            x_new = x_new * norm / s_x

        x = x_new
        avg_a += x
        n_avg += 1

        if t % 10 == 0:
            avg_a = avg_a / n_avg
            s_avg = avg_a.sum()
            if s_avg > 1e-15:
                avg_norm = avg_a * norm / s_avg
                score = 2.0 * n * _fast_autocorr(avg_norm).max() / (avg_norm.sum() ** 2)
                if score < best_score - 1e-10:
                    best_score = score
                    best_a = avg_norm.copy()

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def _peak_shave(a, time_limit, seed=42):
    """Targeted peak reduction via mass redistribution."""
    n = len(a)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = _fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:8]
        improved = False

        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2: je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]

            candidates_sorted = candidates[np.argsort(a[candidates])][:max(1, len(candidates) // 4)]
            dst = candidates_sorted[rng.integers(len(candidates_sorted))]

            for frac in [0.001, 0.003, 0.007, 0.015, 0.03]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = _fast_autocorr(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-10:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = _fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
            if stall > 500:
                break

    return best_a, best_score


def _frank_wolfe_nesterov(a0, time_limit, verbose=False):
    """Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    y = a.copy()
    t0 = time.monotonic()
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        b = _fast_autocorr(y)
        k_star = int(np.argmax(b))
        M_curr = b[k_star]
        grad = np.zeros(n)
        for i in range(n):
            j1 = k_star - i
            if 0 <= j1 < n:
                grad[i] += 2.0 * y[j1]
            j2 = i - k_star
            if 0 <= j2 < n:
                grad[i] += 2.0 * y[j2]
        i_min = int(np.argmin(grad))
        v = np.zeros(n)
        v[i_min] = norm
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * v
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        if t == 1:
            a_new = v.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        y = y_new
        a = a_new
        score_a = _fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW-Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}")
    return best_a, best_score


def _interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def _evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def _spectral_init(n, seed=0):
    """Initialize using spectral considerations - aim for flat power spectrum."""
    rng = np.random.default_rng(seed)
    # Start with a random phase signal, take real part, rectify
    freqs = np.fft.rfftfreq(n)
    # Target: flat spectrum -> white noise -> but we need non-negative
    # Use a cosine sum with random frequencies
    a = np.ones(n) / n
    for _ in range(5):
        freqs_rand = rng.uniform(0.5, n // 2 - 1, size=50)
        phases = rng.uniform(0, 2 * np.pi, 50)
        x = np.linspace(0, 1, n, endpoint=False)
        signal = np.zeros(n)
        for f, p in zip(freqs_rand, phases):
            signal += np.cos(2 * np.pi * f * x + p)
        a = np.maximum(signal, 0)
        s = a.sum()
        if s > 1e-15:
            a = a / s
        else:
            a = np.ones(n) / n
    return a


def _spectral_refine(a0, time_limit, verbose=False):
    """Refine using spectral domain gradient information."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    y = a.copy()
    t_fista = 1

    for it in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = _fast_autocorr(y)
        k_star = int(np.argmax(ac))
        M = ac[k_star]

        # Gradient of (a*a)_k_star
        grad = np.zeros(n)
        for i in range(n):
            j1 = k_star - i
            if 0 <= j1 < n:
                grad[i] += 2.0 * y[j1]
            j2 = i - k_star
            if 0 <= j2 < n:
                grad[i] += 2.0 * y[j2]

        # Spectral correction: add penalty for concentration in frequency domain
        A = rfft(y, 2 * n)
        power = np.abs(A) ** 2
        power_max = power.max()
        if power_max > 0:
            power_rel = power / power_max
            # Transform back to get spectral gradient
            spec_grad = irfft(power_rel, 2 * n)[:n]
            grad += 0.1 * spec_grad

        # Add smoothness penalty
        if n > 2:
            grad[1:-1] += 0.01 * (2 * y[1:-1] - y[:-2] - y[2:])
            grad[0] += 0.01 * (2 * y[0] - y[1])
            grad[-1] += 0.01 * (2 * y[-1] - y[-2])

        L = np.linalg.norm(grad) + 1e-15
        step = 1.0 / L

        z = y - step * grad
        a_new = _proj_simplex(z)
        a_new = a_new * norm / a_new.sum()

        if t_fista == 1:
            y_new = a_new.copy()
        else:
            mom = (t_fista - 1.0) / (t_fista + 2.0)
            y_new = a_new + mom * (a_new - y)
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y > 1e-15:
                y_new = y_new * norm / s_y
            else:
                y_new = a_new.copy()

        y = y_new
        t_fista += 1

        if it % 100 == 0:
            sc = _fast_autocorr(a_new).max()
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_new.copy()

        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Spectral it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: Exploits the fact that the all-ones sequence achieves
    the theoretical minimum J=2. Then uses optimization to verify and
    potentially improve via spectral refinement and peak shaving.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)

    solver = _get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    best_seq = None
    best_score = float('inf')

    # Try to load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = _evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def _try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = _evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    n_base = 1600

        best_score = ones_score
        best_seq = ones_seq.copy()

    if best_seq is None:

            best_seq = spec_init.copy()

    n_best = len(best_seq)
    print(f"\nStarting optimization from n={n_best}, score={best_score:.6f}")

    # Phase 1: LSE-FISTA (fast, smooth approximation)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.15, 200.0)
    if t1 > 20:
        print(f"\n=== Phase 1: LSE-FISTA n={n_best} (budget={t1:.0f}s) ===")
