    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    return _global_solver

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def lp_best_response(a, M, solver, active_set=None):
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    
    if active_set is not None and len(active_set) < size:
        constraints = [T[active_set, :] @ b_var <= M]
    else:
        conv_current = T @ a
        active_set = np.where(conv_current > M * 0.95)[0]
        if len(active_set) < 10:
            active_set = np.where(conv_current > M * 0.5)[0]
        if len(active_set) > 0.95 * size:
            constraints = [T @ b_var <= M]
        else:
            constraints = [T[active_set, :] @ b_var <= M]
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, active_set
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, active_set
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, active_set
    
    # Check dropped constraints
    if active_set is not None and len(active_set) < size:
        full_conv = T @ b_val
        violated = np.where(full_conv > M * (1 + 1e-6))[0]
        if len(violated) > 0:
            new_active = np.union1d(active_set, violated)
            b_var2 = cp.Variable(n, nonneg=True)
            constraints2 = [T[new_active, :] @ b_var2 <= M]
            prob2 = cp.Problem(cp.Maximize(cp.sum(b_var2)), constraints2)
            try:
                prob2.solve(solver=solver, verbose=False)
            except Exception:
                pass
            else:
                if prob2.status in ('optimal', 'optimal_inaccurate'):
                    b_val = np.maximum(np.asarray(b_var2.value, dtype=np.float64), 0.0)
            active_set = new_active
    return b_val, active_set

def nesterov_lp(a0, time_limit, solver, verbose=False):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    active_set = None
    n_iters = 0
    stall_count = 0
    prev_best = best_score
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        
        b, active_set = lp_best_response(y, M_curr, solver, active_set)
        if b is None:
            active_set = None
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        
        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    stall_count = 0
        
        # If stalled for too long, add small perturbation
        if stall_count > 200 and n_iters > 100:
            rng_local = np.random.default_rng(n_iters)
            pert = rng_local.normal(0, 0.01 * norm / n, n)
            a_pert = a + pert
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert * norm / a_pert.sum()
            y = a_pert.copy()
            stall_count = 0
            
        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def pair_cd_exact(a0, time_limit, rng, verbose=False):
    """Pair coordinate descent with exact 1D line search via grid + refinement."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0
    
    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    n_passes = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-9:
                continue
            
            # Try a few random j's
            for attempt in range(5):
                j = int(rng.integers(n))
                if j == i:
                    continue
                
                # Compute g and h vectors for pair (i,j)
                # new_conv_k = conv_k + delta * g_k + delta^2 * h_k
                g = np.zeros(size)
                h = np.zeros(size)
                
                # g_k = 2 * a_{k-i} (where 0 <= k-i < n)
                if i < size:
                    g[i:i+n] += 2.0 * a  # a_{k-i} for k in [i, i+n-1]
                
                # h_k = [k==2i] + [k==2j] - 2*[k==i+j]
                if 2 * i < size:
                    h[2 * i] += 1.0
                if 2 * j < size:
                    h[2 * j] += 1.0
                if i + j < size:
                    h[i + j] -= 2.0
                
                # Now subtract the contribution from j (since we're moving mass from j to i)
                # Wait, I defined a' = a + delta*(e_i - e_j), so:
                # new_conv_k = conv_k + 2*delta*a_{k-i} + delta^2*([k=2i]+[k=2j]-2*[k=i+j])
                # But this doesn't account for the change at position k-i when k-i = j
                # Let me re-derive...
                # Actually my derivation above is correct. The formula is:
                # new_conv_k = conv_k + 2*delta*a_{k-i} + delta^2*([k=2i]+[k=2j]-2*[k=i+j])
                # where a_{k-i} uses the ORIGINAL a (including a_j).
                # This is because we're computing the FIRST-ORDER change in a_i (adding delta)
                # and the SECOND-ORDER correction.
                # Wait no, let me re-derive from scratch.
                
                # a'_i = a_i + delta, a'_j = a_j - delta, a'_m = a_m for m != i,j
                # conv(a',a')_k = sum_{p=0}^{n-1} a'_p * a'_{k-p}
                # = sum_p a_p * a_{k-p}  [base]
                #   + delta * a'_{k-i}    [p=i term with new a_i]
                #   - a_{k-i} * delta     [k-p=i term with new a_i, if k-i != i]
                #   ... this is getting complicated. Let me use the linearity trick.
                
                # Let u = e_i, v = -e_j. Then a' = a + delta*(u+v) = a + delta*(e_i - e_j).
                # conv(a',a') = conv(a,a) + delta*(conv(a, e_i-e_j) + conv(e_i-e_j, a))
                #                + delta^2 * conv(e_i-e_j, e_i-e_j)
                # conv(a, e_i)_k = a_{k-i} (for 0<=k-i<n)
                # conv(a, e_j)_k = a_{k-j} (for 0<=k-j<n)
                # So conv(a, e_i-e_j)_k = a_{k-i} - a_{k-j}
                # Similarly conv(e_i-e_j, a)_k = a_{k-i} - a_{k-j}
                # conv(e_i-e_j, e_i-e_j)_k = [k=2i] - [k=i+j] - [k=i+j] + [k=2j]
                #                            = [k=2i] + [k=2j] - 2[k=i+j]
                
                # So: new_conv_k = conv_k + 2*delta*(a_{k-i} - a_{k-j}) + delta^2*([k=2i]+[k=2j]-2[k=i+j])
                
                # I made an error before! The g vector should be 2*(a_{k-i} - a_{k-j}), not just 2*a_{k-i}.
                # Let me fix this.
                
                g = np.zeros(size)
                h = np.zeros(size)
                
                # g_k = 2*(a_{k-i} - a_{k-j})
                if i < size:
                    g[i:i+n] += 2.0 * a
                if j < size:
                    g[j:j+n] -= 2.0 * a
                
                # h_k = [k==2i] + [k==2j] - 2*[k==i+j]
                if 2 * i < size:
                    h[2 * i] += 1.0
                if 2 * j < size:
                    h[2 * j] += 1.0
                if i + j < size:
                    h[i + j] -= 2.0
                
                # Valid delta range: a_i + delta >= 0 and a_j - delta >= 0
                # So delta in [-a_i, a_j]
                d_lo = -a[i]
                d_hi = a[j]
                
                if d_hi <= d_lo:
                    continue
                
                # Find optimal delta via grid search + refinement
                # Coarse grid
                n_coarse = 50
                deltas = np.linspace(d_lo, d_hi, n_coarse)
                
                # Vectorized evaluation: for each delta, compute max(conv + delta*g + delta^2*h)
                # Shape: (n_coarse, size)
                # new_conv[d, k] = conv[k] + deltas[d]*g[k] + deltas[d]^2*h[k]
                # = conv[k] + deltas[d]*g[k] + (deltas[d]^2)*h[k]
                
                # For efficiency, compute in batches
                best_d = 0.0
                best_val = best_score  # don't worsen
                
                # Use chunked evaluation to save memory
                chunk_size = min(n_coarse, max(1, 2000000 // size))
                for d_start in range(0, n_coarse, chunk_size):
                    d_end = min(d_start + chunk_size, n_coarse)
                    d_chunk = deltas[d_start:d_end]  # shape (chunk,)
                    # new_conv shape: (chunk, size)
                    new_vals = conv[None, :] + d_chunk[:, None] * g[None, :] + (d_chunk ** 2)[:, None] * h[None, :]
                    row_max = new_vals.max(axis=1)
                    # Find best in this chunk
                    local_best = np.argmin(row_max)
                    if row_max[local_best] < best_val - 1e-14:
                        best_val = row_max[local_best]
                        best_d = d_chunk[local_best]
                
                if best_d == 0.0:
                    continue
                
                # Refine around best_d with finer grid
                d_margin = (d_hi - d_lo) / n_coarse * 2
                d_ref_lo = max(d_lo, best_d - d_margin)
                d_ref_hi = min(d_hi, best_d + d_margin)
                n_fine = 50
                deltas_fine = np.linspace(d_ref_lo, d_ref_hi, n_fine)
                
                for d in deltas_fine:
                    new_max = (conv + d * g + d * d * h).max()
                    if new_max < best_val - 1e-14:
                        best_val = new_max
                        best_d = d
                
                if best_d == 0.0:
                    continue
                
                # Apply the update
                a[i] += best_d
                a[j] -= best_d
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s < 1e-15:
                    a[i] -= best_d
                    a[j] += best_d
                    continue
                a = a * norm / s
                conv = fast_autocorr(a)
                cur_max = conv.max()
                
                if cur_max < best_score - 1e-12:
                    best_score = cur_max
                    best_a = a.copy()
                    pass_improved = True
                    no_improve = 0
                else:
                    # Revert
                    a[j] += best_d
                    a[i] -= best_d
                    s = a.sum()
                    if s > 1e-15:
                        a = a * norm / s
                    conv = fast_autocorr(a)
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
        
        if verbose and n_passes % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def lse_adam(a0, time_limit, verbose=False):
    """LSE-smoothed objective with Adam optimizer."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    
    # Anneal temperature
    n_stages = 10
    temps = np.geomspace(50.0, 0.3, n_stages)
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        n_steps = max(20, min(200, int(remaining * 2)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            conv = fast_autocorr(a)
            max_conv = conv.max()
            exp_conv = np.exp((conv - max_conv) / T)
            w = exp_conv / exp_conv.sum()
            
            # Compute gradient via FFT
            # grad_i = 2 * sum_k w_k * (a_{k-i} + a_{i-k} - [k==2i]*a_i)
            # Actually: d/d a_i conv_k = a_{k-i} + a_{i-k} (for k != 2i)
            # and d/d a_i conv_{2i} = 2*a_i
            # So grad_i = sum_k w_k * (a_{k-i} + a_{i-k})
            # = sum_k w_k * a_{k-i} + sum_k w_k * a_{i-k}
            # First term: cross-correlation of w and a at lag i
            # Second term: cross-correlation of w and a at lag -i (or correlation)
            
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            
            # sum_k w_k * a_{k-i}: this is (w * a_flipped)[n-1+i]... 
            # Let me think again. w_k * a_{k-i}: let m = k-i, then k = m+i
            # sum_m w_{m+i} * a_m = sum_m a_m * w_{m+i}
            # This is the cross-correlation of a and w at lag i.
            # In FFT: ifft(conj(FFT(a)) * FFT(w)) gives... 
            # Actually: cross_correlation(a, w)[i] = sum_m a[m] * w[m+i]
            # = ifft(conj(FFT(a)) * FFT(w))[i] (with circular correlation)
            
            # For our case: sum_k w_k * a_{k-i}
            # Let's define c[i] = sum_k w_k * a_{k-i}
            # c[i] = sum_k w[k] * a[k-i] (valid k, k-i)
            # This is the cross-correlation of w and a, evaluated at... 
            # cross_corr(w, a)[d] = sum_k w[k] * a[k+d]
            # So c[i] = cross_corr(w, a)[-i] = cross_corr(a, w)[i]
            
            # Using FFT: cross_corr(w,a)[d] = ifft(FFT(w) * conj(FFT(a)))[d]
            # (circular correlation)
            
            W_A = W * np.conj(A)
            cross1 = irfft(W_A, fft_size)
            # cross1[d] = sum_k w[k] * a[(k+d) mod fft_size]
            # We want c[i] = sum_k w[k] * a[k-i] = cross1[-i mod fft_size] = cross1[fft_size - i]
            
            c = np.zeros(n)
            for i in range(n):
                c[i] = cross1[fft_size - i] if i > 0 else cross1[0]
            
            # Second term: sum_k w_k * a_{i-k}
            # Let m = i-k, then k = i-m
            # sum_m w[i-m] * a[m] = sum_m a[m] * w[i-m]
            # This is cross_corr(a, w)[-i]... or we can compute directly
            # cross_corr(a, w)[d] = sum_k a[k] * w[k+d]
            # sum_m a[m] * w[i-m]: let d = i-m, m = i-d
            # = sum_d a[i-d] * w[d]
            # = cross_corr(w, a)[i] (checking: cross_corr(w,a)[d] = sum_k w[k]*a[k+d], so at d=i: sum_k w[k]*a[k+i])
            # Hmm, that's not quite right either.
            
            # Let me just compute it as: sum_k w[k] * a[i-k]
            # = (w * a_rev)[i + n - 1] where a_rev = a[::-1]
            # Because (w * a_rev)[m] = sum_k w[k] * a_rev[m-k] = sum_k w[k] * a[n-1-(m-k)]
            # = sum_k w[k] * a[n-1-m+k]
            # Setting n-1-m+k = i-k doesn't work simply...
            
            # Alternative: just compute both terms together
            # grad_i = sum_k w_k * (a_{k-i} + a_{i-k}) for k != 2i part
            # + 2*w_{2i} * a_i for the k=2i part (since d/d a_i a_i^2 = 2*a_i)
            # Wait, actually d/d a_i of conv_{2i} = d/d a_i (a_i^2 + sum_{j!=i} a_j * a_{2i-j})
            # = 2*a_i + sum_{j!=i, 2i-j != i} a_{2i-j} * 0 + ... 
            # No, conv_{2i} = sum_j a_j * a_{2i-j}. The terms with a_i:
            # j=i: a_i * a_i = a_i^2
            # j=2i-i=i: same term
            # So d/d a_i conv_{2i} = 2*a_i
            # For k != 2i: d/d a_i conv_k = a_{k-i} + a_{i-k} (the two cross terms)
            
            # So grad_i = sum_{k != 2i} w_k * (a_{k-i} + a_{i-k}) + w_{2i} * 2*a_i
            # = sum_k w_k * (a_{k-i} + a_{i-k}) + w_{2i} * a_i  [since a_{2i-i} = a_i, the formula gives a_i + a_i = 2*a_i for k=2i, which matches]
            
            # So actually: grad_i = sum_k w_k * (a_{k-i} + a_{i-k}) works for all k!
            # (With the convention that a_m = 0 for m < 0 or m >= n)
            
            # Let's compute sum_k w_k * a_{k-i} for all i using FFT:
            # This is the circular cross-correlation, which for fft_size > 2n-1 gives the linear version.
            
            # Method: pad w and a to fft_size, compute ifft(conj(FFT(w)) * FFT(a))
            # This gives corr[d] = sum_k w[k] * a[k+d] (circular, but linear if fft_size > 2n)
            # We want s1[i] = sum_k w[k] * a[k-i] = corr[-i] = corr[fft_size - i]
            
            # And s2[i] = sum_k w[k] * a[i-k]
            # Let m = i-k: sum_m w[i-m] * a[m] = sum_m a[m] * w[i-m]
            # This is the cross-correlation of a and w at lag i: 
            # cross(a,w)[i] = sum_m a[m] * w[m+i]... no
            # Actually: sum_m a[m] * w[i-m] = (a * w_rev)[i] where w_rev[m] = w[-m mod fft_size]
            # In FFT: FFT(a * w_rev) = FFT(a) * FFT(w_rev) = A * conj(W) (since FFT of reversed = conj of FFT for real signals... not exactly)
            
            # Simpler: s2[i] = sum_k w[k] * a[i-k]
            # Let's define b = w[::-1] (reversed). Then:
            # s2[i] = sum_k w[k] * a[i-k] = sum_k b[n-1-k] * a[i-k]
            # Let j = n-1-k: k = n-1-j
            # = sum_j b[j] * a[i-n+1+j]
            # = (b * a)[i-n+1] (convolution)
            # In FFT: ifft(FFT(b) * FFT(a))[i-n+1]
            
            # This is getting too complicated for the FFT approach. Let me just use a direct but
            # vectorized computation for n <= 4000.
            
            # Direct computation:
            grad = np.zeros(n)
            # For each i: grad[i] = sum_k w[k] * a[k-i] + sum_k w[k] * a[i-k]
            # = dot(w[i:min(n, size)], a[:min(n, size-i)]) + dot(w[max(0,i-n+1):i+1], a[i - np.arange(...)])
            
            # Vectorized version using the fact that this is a Toeplitz system:
            # First term: for i=0: dot(w[:n], a[:n])
            #            for i=1: dot(w[1:n], a[:n-1])
            #            ...
            #            for i=n-1: dot(w[n-1:], a[:1])
            # Second term: for i=0: dot(w[:1], a[0])  [only k=0: w[0]*a[0]]
            #            for i=1: dot(w[:2], a[1:0:-1]) = w[0]*a[1] + w[1]*a[0]
            #            ...
            
            # Actually, let me just use the convolution matrix approach for the gradient:
            # grad = 2 * (T_w.T @ a + T_w @ a) where T_w is the conv matrix of w...
            # No, that's not right either.
            
            # Let me just do it the brute force way but vectorized:
            # s1[i] = sum_{k=i}^{n-1} w[k] * a[k-i]  (terms where k-i >= 0 and k < n)
            #       + sum_{k=n}^{min(size-1, 2n-1)} w[k] * a[k-i]  (terms where k >= n, but w only has n elements)
            # Actually w has length size = 2n-1, and a has length n.
            # s1[i] = sum_k w[k] * a[k-i] where 0 <= k < size and 0 <= k-i < n
            #       = sum_{k=max(0,i)}^{min(size-1, i+n-1)} w[k] * a[k-i]
            
            # For efficiency with n up to 4000, let me use the FFT approach properly:
            # s1[i] = sum_k w[k] * a[k-i] for valid indices
            # Pad w to length fft_size, pad a to length fft_size
            # Compute circular cross-correlation: c = ifft(conj(FFT(w)) * FFT(a))
            # c[d] = sum_k w[k] * a[(k+d) mod fft_size]
            # For d = -i (mod fft_size) = fft_size - i:
            # c[fft_size-i] = sum_k w[k] * a[(k - i) mod fft_size]
            # For i in [0, n-1] and fft_size > 2n-1, the wrap-around doesn't affect valid indices
            # So s1[i] = c[fft_size-i] for i > 0, c[0] for i = 0
            
            # s2[i] = sum_k w[k] * a[i-k] for valid indices
            # = sum_k a[i-k] * w[k] where 0 <= k < size and 0 <= i-k < n
            # = sum_{k=max(0,i-n+1)}^{min(i, size-1)} a[i-k] * w[k]
            # Let m = i-k: = sum_m a[m] * w[i-m] where 0 <= m < n and 0 <= i-m < size
            # = sum_{m=max(0,i-size+1)}^{min(n-1, i)} a[m] * w[i-m]
            # This is the convolution (a * w)[i] (truncated)
            # In FFT: ifft(FFT(a) * FFT(w))[i] gives circular conv, which equals linear conv if fft_size > n + size - 1 = 3n - 2
            # But our fft_size is only > 2n-1, so there might be wrap-around.
            # For safety, use a larger FFT size for this computation.
            
            # You know what, for n <= 4000, let me just use a direct O(n^2) computation
            # with numpy tricks. It'll be fast enough.
            
            # s1[i] = dot(w[i:i+n], a[:n-i]) for i < n (adjusted for w length)
            # Actually w has length 2n-1, so:
            # s1[i] = dot(w[i:min(i+n, size)], a[:min(n, size-i)])
            # But size = 2n-1, and i < n, so i+n < 2n < size+1, and size-i > n-1 >= 0
            # So s1[i] = dot(w[i:i+n], a[:n]) ... wait, that's only n elements from w
            # w[i:i+n] has min(n, size-i) elements, and a[:min(n, size-i)] has the same
            # Since size = 2n-1 and i < n: size - i > n-1, so min(n, size-i) = n
            # So s1[i] = dot(w[i:i+n], a)
            
            # s2[i]: sum_k w[k] * a[i-k] where 0<=k<size, 0<=i-k<n
            # i-k in [0,n-1] means k in [i-n+1, i]
            # k in [0, size-1] means k in [0, 2n-2]
            # So k in [max(0, i-n+1), min(i, 2n-2)]
            # Since i < n: i-n+1 <= 0, so k in [0, i]
            # s2[i] = dot(w[0:i+1], a[i:0:-1])  [reversed]
            # = dot(w[:i+1], a[i::-1])
            
            # Great! So:
            # s1[i] = dot(w[i:i+n], a)  (for i in 0..n-1)
            # s2[i] = dot(w[:i+1], a[i::-1])  (for i in 0..n-1)
            # grad[i] = s1[i] + s2[i]
            
            # s1 can be computed as: for each i, dot(w[i:i+n], a)
            # This is like a sliding dot product, which can be done with FFT or directly.
            # For n=1600, it's 1600 dots of length 1600 = ~2.5M ops. Fine.
            
            # Let me just compute directly:
            # s1: w[i:i+n] dot a, for i=0..n-1
            # This is the (n x n) matrix W_sliding @ a where W_sliding[i] = w[i:i+n]
            # We can build this matrix or use FFT.
            
            # For n=1600, building the matrix is 1600*1600 = 2.56M entries. Fine.
            W_mat = np.zeros((n, n))
            for i in range(n):
                W_mat[i] = w[i:i+n]
            s1 = W_mat @ a
            
            # s2[i] = dot(w[:i+1], a[i::-1])
            # This is: sum_{k=0}^{i} w[k] * a[i-k] = (w * a)[i] (convolution at position i)
            # Can compute via FFT:
            s2 = irfft(rfft(w, fft_size) * A, fft_size)[:n]
            # Wait, that gives the full convolution of w and a, not what we want.
            # (w * a)[i] = sum_k w[k] * a[i-k] which is exactly s2[i]!
            # But we need w to be treated as a sequence of length n (or size?) for the convolution.
            # Actually, the convolution (w * a) of two sequences of lengths len(w) and len(a) has length len(w)+len(a)-1.
            # w has length size = 2n-1, a has length n.
            # (w*a)[i] for i in 0..n-1: sum_{k=0}^{i} w[k] * a[i-k] (since i < n <= len(a), and i < size)
            # This is exactly s2[i].
            
            # Using FFT with sufficient length:
            conv_size = size + n  # 3n-1
            fft2_size = 1 << (conv_size - 1).bit_length()
            s2_full = irfft(rfft(w, fft2_size) * rfft(a, fft2_size), fft2_size)[:n]
            
            grad = s1 + s2_full
            
            # Adam update
            lr = 2.0 * norm / (T * (1.0 + 0.1 * it))
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** (it + 1))
            v_hat = v / (1 - beta2 ** (it + 1))
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new * norm / s
            else:
                a_new = a.copy()
            a = a_new
            
            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE-Adam T={T:.1f} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        x = np.linspace(0.0, 1.0, n)
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.5 * np.sin(4.0 * np.pi * x) + 0.2 * np.cos(8.0 * np.pi * x)
        a = np.maximum(a, 0.0)
    else:
        # Random smooth
        freqs = rng.integers(1, 10, 5)
        coeffs = rng.uniform(0.1, 1.0, 5)
        phases = rng.uniform(0, 2 * np.pi, 5)
        x = np.linspace(0, 2 * np.pi, n)
        a = np.zeros(n)
        for f, c, p in zip(freqs, coeffs, phases):
            a += c * np.cos(f * x + p)
        a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss
