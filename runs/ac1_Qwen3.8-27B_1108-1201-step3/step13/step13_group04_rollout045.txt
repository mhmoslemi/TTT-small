    This is the cross-correlation of w (shifted) and a.

    Second term: sum_k w[k]*a[m-k] where a[m-k] is defined for 0 <= m-k < n, i.e., m-n+1 <= k <= m.
    = sum_{k=max(0,m-n+1)}^{m} w[k]*a[m-k]
    = sum_{j=0}^{m} w[m-j]*a[j] (substituting j = m-k)
    = (w * a)[m] in the standard convolution sense (truncated to valid range).
    
    So g[m] = sum_j w[j+m]*a[j] + (w*a)[m]
    
    For the first term, note that sum_j w[j+m]*a[j] = <w[m:], a[:n-m]> (dot product of w[m:] and first n-m elements of a).
    
    Actually, let me just use the FFT approach:
    
    First term for all m: it's the cross-correlation of w and a, but with w being longer than a.
    Specifically, for m = 0, ..., n-1:
    f[m] = sum_{j=0}^{n-1} w[j+m] * a[j]
    
    This is the "unreversed" cross-correlation. In FFT terms:
    f = ifft(fft(w) * conj(fft(a, len(w))))[:n]
    Wait, I need to be more careful.
    
    The cross-correlation R[m] = sum_j w[j+m]*a[j] (for m >= 0) is related to the convolution:
    R[m] = (w * a_rev)[m + n - 1] where a_rev[j] = a[n-1-j]
    
    Let me verify: (w * a_rev)[k] = sum_j w[j] * a_rev[k-j] = sum_j w[j] * a[n-1-k+j]
    For k = m + n - 1: = sum_j w[j] * a[n-1-m-n+1+j] = sum_j w[j] * a[j-m]
    Hmm, that's sum_j w[j]*a[j-m], not sum_j w[j+m]*a[j].
    
    Let me try: (w * a_rev)[k] = sum_j w[j] * a_rev[k-j]
    a_rev[k-j] = a[n-1-(k-j)] = a[n-1-k+j]
    
    For this to equal a[j'], we need n-1-k+j = j', so j' = n-1-k+j.
    When j'=0: j = k-n+1. When j'=n-1: j = k.
    
    So (w * a_rev)[k] = sum_{j=k-n+1}^{k} w[j] * a[n-1-k+j]
    Let j' = n-1-k+j, so j = j' + k - n + 1:
    = sum_{j'=0}^{n-1} w[j'+k-n+1] * a[j']
    
    For k = m + n - 1:
    = sum_{j'=0}^{n-1} w[j'+m] * a[j']
    
    Yes! So first term f[m] = (w * a_rev)[m + n - 1].
    
    Second term: (w * a)[m] (standard convolution, truncated).
    
    So total gradient g[m] = (w * a_rev)[m+n-1] + (w*a)[m]
    
    This matches what the previous code had! Good.
    """
    n = len(a)
    c = fast_autocorr(a)
    max_c = c.max()
    w = np.exp((c - max_c) / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[n - 1] = 1.0
    else:
        w /= w_sum

    # Compute (w * a_rev)[m+n-1] for m = 0,...,n-1 and (w*a)[m] for m = 0,...,n-1
    fft_size = 1 << (3 * n - 2).bit_length()
    a_rev = a[::-1]
    W = rfft(w, fft_size)
    A_rev = rfft(a_rev, fft_size)
    A = rfft(a, fft_size)
    
    conv1 = irfft(W * A_rev, fft_size)  # w * a_rev, length 3n-2
    conv2 = irfft(W * A, fft_size)       # w * a, length 3n-2
    
    term1 = conv1[n-1 : 2*n-1]  # (w * a_rev)[m+n-1] for m=0,...,n-1
    term2 = conv2[0 : n]        # (w * a)[m] for m=0,...,n-1
    
    return term1 + term2


def project_to_simplex(a, target_sum):
    """Project a onto {x >= 0, sum(x) = target_sum}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-300:
        return np.full_like(a, target_sum / len(a))
    return a * target_sum / s


def lse_value(a, T):
    """Compute LSE value: T * log(sum exp(c_k / T))."""
    c = fast_autocorr(a)
    max_c = c.max()
    return T * (max_c / T + np.log(np.sum(np.exp((c - max_c) / T))))


def nesterov_lse_optimize(a0, time_limit, T_start=20.0, T_end=0.1, n_stages=15, verbose=True):
    """Nesterov-accelerated LSE gradient descent with adaptive cooling."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    it_count = 0
    prev_y = None

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2.0:
            break

        n_iters = max(100, int(remaining * 4.0 / max(1, n_stages - si)))
        y = a.copy()
        y_prev = a.copy() if si == 0 else (best_a * norm / best_a.sum()).copy()
        
        for it in range(n_iters):
            if time.monotonic() - t0 > time_limit:
                break

            # Nesterov extrapolation
            if it == 0:
                y_eval = y
            else:
                beta = (1.0 + np.sqrt(1.0 + 4.0 * (it - 1) ** 2)) / (2.0 * (it + 1))
                y_eval = y + beta * (y - y_prev)
                y_eval = project_to_simplex(y_eval, norm)

            grad = lse_gradient(y_eval, T)
            
            # Step size: inverse of iteration count (standard for convex problems)
            step = 1.0 / (it + 3.0)
            # Scale step by T to account for the T factor in LSE
            step *= T
            
            y_new = y - step * grad
            y_new = project_to_simplex(y_new, norm)
            
            y_prev = y.copy()
            y = y_new
            it_count += 1

            # Track best
            if it_count % 10 == 0:
                score = fast_autocorr(y).max()
                if score < best_score:
                    best_score = score
                    best_a = y.copy()

            if verbose and it_count % 2000 == 0:
                elapsed = time.monotonic() - t0
                cur_score = fast_autocorr(y).max()
                print(f"    LSE-Nest stage={si}/{n_stages} T={T:.2f} it={it_count} ({elapsed:.0f}s): cur={cur_score:.6f} best={best_score:.6f}")

    return best_a, best_score


def mirror_descent_optimize(a0, time_limit, T_start=10.0, T_end=0.1, n_stages=12, verbose=True):
    """Mirror descent (multiplicative weights) on LSE surrogate."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    it_count = 0

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2.0:
            break

        n_iters = max(100, int(remaining * 3.0 / max(1, n_stages - si)))
        for it in range(n_iters):
            if time.monotonic() - t0 > time_limit:
                break

            grad = lse_gradient(a, T)
            # Mirror descent step: a <- a * exp(-eta * grad)
            eta = 1.0 / (T * np.sqrt(it + 1))
            log_a = np.log(a + 1e-30) - eta * grad
            log_a -= log_a.max()
            a_new = np.exp(log_a)
            a_new = project_to_simplex(a_new, norm)
            a = a_new
            it_count += 1

            if it_count % 10 == 0:
                score = fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()

            if verbose and it_count % 2000 == 0:
                elapsed = time.monotonic() - t0
                cur_score = fast_autocorr(a).max()
                print(f"    Mirror stage={si}/{n_stages} T={T:.2f} ({elapsed:.0f}s): cur={cur_score:.6f} best={best_score:.6f}")

    return best_a, best_score


def lp_best_response_fast(a, M, solver, max_active=50):
    """
    Fast LP best-response using active set method.
    max sum(b) s.t. (a*b)_j <= M for all j, b >= 0
    """
    n = len(a)
    
    # Build the constraint matrix T where T[j,k] = a[j-k] if 0 <= j-k < n
    # We'll use a sparse matrix
    rows = []
    cols = []
    vals = []
    for j in range(2 * n - 1):
        for k in range(n):
            i = j - k
            if 0 <= i < n:
                if a[i] > 1e-15:
                    rows.append(j)
                    cols.append(k)
                    vals.append(a[i])
    
    T_mat = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Start with a small active set
    # Find the current violation
    b_init = np.ones(n)  # initial guess
    ab_init = T_mat @ b_init
    active = list(np.argsort(-ab_init)[:max_active])
    
    best_b = b_init
    best_obj = b_init.sum()
    
    for iteration in range(30):
        b_var = cp.Variable(n, nonneg=True)
        constraints = []
        for j in active:
            row = T_mat.getrow(j)
            constraints.append(row @ b_var <= M)
        
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        if b_val.sum() < 1e-15:
            break
        
        # Check all constraints
        ab = T_mat @ b_val
        violations = ab - M
        max_viol = violations.max()
        
        if max_viol < 1e-6 * M:
            break
        
        # Add most violated constraint
        worst = int(np.argmax(violations))
        if worst not in active:
            active.append(worst)
            # Keep active set size bounded
            if len(active) > max_active * 2:
                # Remove least violated from active
                active_violations = [(j, ab[j] - M) for j in active]
                active_violations.sort(key=lambda x: x[1])
                active = [j for j, _ in active_violations[-max_active:]]
        
        if ab[_].max() < best_obj:  # this doesn't make sense, let me fix
            pass
    
    return b_val


def lp_nesterov_optimize(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    iter_count = 0

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break
        if time.monotonic() - t0 > time_limit - 10:  # leave time for final eval
            break

        c = fast_autocorr(y)
        M_curr = c.max()
        if M_curr < 1e-10:
            break
        
        b = lp_best_response_fast(y, M_curr, solver, max_active=100)
        if b is None or b.sum() < 1e-15:
            break

        g = b * norm / b.sum()
        iter_count += 1

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        if t == 1:
            y_new = g.copy()
        else:
            y_new = (1.0 - alpha) * y + alpha * g
            y_new = project_to_simplex(y_new, norm)
            # Nesterov extrapolation
            beta = ((t - 1.0 + np.sqrt(1.0 + 4.0 * (t - 1.0) ** 2)) / (2.0 * (t + 1.0)))
            y_new = y_new + beta * (y_new - y)
            y_new = project_to_simplex(y_new, norm)

        y = y_new

        if iter_count % 5 == 0:
            score = fast_autocorr(y).max()
            if score < best_score:
                best_score = score
                best_a = y.copy()

            if verbose and iter_count % 50 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP-Nest iter={iter_count} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def peak_targeting_2swap(a0, time_limit, seed=42, verbose=True):
    """
    2-swap local search targeting the peak of the convolution.
    At each step, identify the peak and try to reduce it by moving mass
    from high-contributing positions to low-impact positions.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    it_count = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a)
        # Get top peaks
        top_ks = np.argsort(c)[::-1][:8]

        improved = False
        for k_star in top_ks:
            # Find contributing pairs
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_star - idx]
            
            if contrib.max() < 1e-15:
                continue
            
            # Get top contributors
            n_top = min(5, len(idx))
            top_contrib = idx[np.argsort(contrib)[::-1][:n_top]]
            
            for src in top_contrib:
                if a[src] < norm * 1e-10:
                    continue

                # Find dst positions that minimize impact on top peaks
                # A position d impacts peak k if d is in [max(0,k-n+1), min(n-1,k)]
                impact = np.zeros(n)
                for k2 in top_ks:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    impact[js2:je2+1] += a[k2 - np.arange(js2, je2+1)]
                
                # Choose dst with low impact and low current value
                impact[src] = np.inf  # don't move to self
                n_candidates = min(10, n // 4)
                # Sort by (impact + a) to find low-impact, low-value positions
                scores = impact + 0.1 * a
                dst_candidates = np.argsort(scores)[:n_candidates]
                
                for dst in dst_candidates:
                    if dst == src:
                        continue
                    fracs = [0.01, 0.03, 0.07, 0.15, 0.3]
                    for frac in fracs:
                        amt = a[src] * frac
                        if amt < norm * 1e-12:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-10:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            it_count += 1
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape local minima
            if rng.integers(5) == 0:
                n_perturb = rng.integers(2, 8)
                for _ in range(n_perturb):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a[i] > norm * 1e-8:
                        frac = rng.uniform(0.01, 0.1)
                        amt = a[i] * frac
                        a_trial = a.copy()
                        a_trial[i] -= amt
                        a_trial[j] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t > 1e-15:
                            a_trial = a_trial * norm / s_t
                            score = fast_autocorr(a_trial).max()
                            if score < best_score:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                stall = 0

            if stall > 300:
                if verbose:
                    print(f"    PeakSearch: stalled at {best_score:.6f} ({time.monotonic()-t0:.0f}s)")
                break

        if verbose and it_count % 500 == 0:
            print(f"    PeakSearch it={it_count} ({time.monotonic()-t0:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def structured_init(n, seed):
    """Generate diverse structured initializations."""
    rng = np.random.default_rng(seed)
    choice = rng.integers(0, 8)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.25 * np.cos(2 * np.pi * 2 * x) + 0.1 * np.cos(2 * np.pi * 5 * x)
    elif choice == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        a = np.exp(-4.0 * (x - 0.5) ** 2 / 0.1)
    elif choice == 6:
        # Piecewise polynomial
        a = np.where(x < 0.5, (2*x)**2, (2*(1-x))**2)
    else:
        # Random smooth
        a = np.ones(n) * 0.5
        for _ in range(rng.integers(3, 8)):
            center = rng.uniform(0.1, 0.9)
            width = rng.uniform(0.05, 0.25)
            amp = rng.uniform(-0.3, 0.5)
            a += amp * np.exp(-0.5 * ((x - center) / width) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n)
    return a / s


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def multi_scale_optimize(a_seed, n_target, time_limit, solver, verbose=True):
    """
    Multi-scale optimization: solve at small scale, interpolate up, refine.
    """
    t0 = time.monotonic()
    
    # Start at a small scale and upscale
    scales = []
    s = 64
    while s < n_target:
        scales.append(s)
        s = min(s * 2, n_target)
    scales.append(n_target)
    
    # Remove duplicates
    scales = list(dict.fromkeys(scales))
    
    if verbose:
        print(f"  Multi-scale: {scales}")
    
    a_curr = None
    for i, n_scale in enumerate(scales):
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 10:
            break
        
        # Allocate time: more time for larger scales
        frac = n_scale / n_target
        t_alloc = remaining * frac * 0.7
        
        if a_curr is None:
            a_curr = structured_init(n_scale, seed=42 + i)
        else:
            a_curr = interpolate_sequence(a_curr, n_scale)
        
        a_curr = a_curr * np.sqrt(2.0 * n_scale) / a_curr.sum()
        
        # Run LSE optimization at this scale
        a_out, score = nesterov_lse_optimize(a_curr, t_alloc, T_start=10.0, T_end=0.3,
                                              n_stages=max(5, n_scale // 20), verbose=False)
        a_curr = a_out
        a_curr = a_curr * np.sqrt(2.0 * n_scale) / a_curr.sum()
        
        if verbose:
            sc = fast_autocorr(a_curr).max()
            print(f"  Multi-scale n={n_scale}: score={sc:.6f} ({time.monotonic()-t0:.0f}s)")
    
    return a_curr
