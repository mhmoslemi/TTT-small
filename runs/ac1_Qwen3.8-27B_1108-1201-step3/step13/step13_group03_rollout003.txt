import numpy as np

from scipy.fft import rfft, irfft

_global_solver = None

def _get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return _global_solver
        except Exception:
            continue
    _global_solver = cp.CLARABEL
    print(f"  LP solver (fallback): CLARABEL")
    return _global_solver

def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def _proj_simplex(v):
    n = len(v)
    if n <= 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 0:
        out /= s
    else:
        out = np.ones(n) / n
    return out

def _lse_grad(y, T):
    n = len(y)
    ac = _fast_autocorr(y)
    ac_max = ac.max()
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(y, fft_size)
    W = rfft(w, fft_size)
    cross = irfft(np.conj(A) * W, fft_size)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad

def _lse_fista(a0, T_schedule, iters_per_stage, time_limit, verbose=False):

        a = a / s
    a = a / a.sum()  # normalize to sum=1 for FISTA on simplex
    
    y = a.copy()
    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    L_init = n / 50.0

    for stage_i, T in enumerate(T_schedule):
        L = max(L_init / (T * 10.0), n / (T * 2.0))
        t_fista = 1
        staging = a.copy()
        max_iters = max(int(iters_per_stage * (T_schedule[0] / T) ** 0.3), 2000)

        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_score

            grad = _lse_grad(y, T)
            step = 1.0 / L
            z = y - step * grad
            a_new = _proj_simplex(z)

            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - staging)
                y_new = np.maximum(y_new, 0.0)
                s_y = y_new.sum()
                if s_y > 1e-15:

                else:
                    y_new = a_new.copy()

            staging = a_new.copy()
            a = a_new
            y = y_new
            t_fista += 1

            if it % 300 == 0:
                sc = _fast_autocorr(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

def _build_conv_matrix(a, n):
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j: j + n, j] = a
    return T

def _lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = _build_conv_matrix(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def _lp_nesterov_linesearch(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration and exact line search."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = _fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        b = _lp_best_response(y, M_curr, solver)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new_nest = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new_nest = y_new + momentum * (y_new - y)
            a_new_nest = np.maximum(a_new_nest, 0.0)
            s_a = a_new_nest.sum()
            if s_a < 1e-15:
                a_new_nest = y_new.copy()
            else:
                a_new_nest = a_new_nest * norm / s_a

        # Line search: try different step sizes between y and g
        best_local_a = None
        best_local_score = best_score
        
        for frac in [1.0, 0.7, 0.5, 0.3, 0.1]:
            candidate = (1.0 - frac) * y + frac * g
            candidate = np.maximum(candidate, 0.0)
            s_c = candidate.sum()
            if s_c < 1e-15:
                continue
            candidate = candidate * norm / s_c
            score = _fast_autocorr(candidate).max()
            if score < best_local_score:
                best_local_score = score
                best_local_a = candidate.copy()

        if best_local_a is not None:
            a_new = best_local_a
        else:
            a_new = a_new_nest

        y = y_new
        a = a_new

        score_y = _fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = _fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 10 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = _fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nesterov+LS iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def _subgradient_avg(a0, time_limit, verbose=True):
    """Subgradient method with Polyak-Rubinstein averaging for max(a*a)."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Polyak-Rubinstein: need an upper bound on optimal value
    # Use the current value as a rough upper bound
    f_star_est = best_score * 0.99  # optimistic estimate
    step_param = 0.5
    
    avg_a = np.zeros(n)
    n_avg = 0
    y = a.copy()
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute subgradient at y
        ac = _fast_autocorr(y)
        k_star = int(np.argmax(ac))
        M_curr = ac[k_star]
        
        # Subgradient: grad[k] = 2 * y[k*-j] (valid indices)
        # More precisely: d/dy[j] of c[k*] = sum_i y[i]*y[k*-i]
        # = y[k*-j] + y[j-k*] (for valid indices)
        # But since it's the self-convolution, 
        # c[k*] = sum_i y[i]*y[k*-i]
        # d/dy[j] = y[k*-j] (if 0<=k*-j<n) + y[j-(k*-j)] ... 
        # Actually: c[k] = sum_i y[i]*y[k-i]
        # d/dy[j] = sum_i [δ(i,j)*y[k-i] + y[i]*δ(k-i,j)]
        #         = y[k-j] + y[k-j]  (when both k-j and j are valid)
        # Wait: the first term gives y[k-j] (when i=j, need 0<=k-j<n)
        # The second term: k-i=j means i=k-j, so we get y[k-j] (need 0<=k-j<n)
        # So d/dy[j] = 2*y[k*-j] if 0<=k*-j<n, else y[k*-j] if only one is valid...
        # Actually for the full convolution, both terms are valid iff 0<=k*-j<n AND 0<=j<n
        # Since j is always in [0,n), we just need 0<=k*-j<n, i.e., j in [k*-n+1, k*]
        
        grad = np.zeros(n)
        j_range = np.arange(max(0, k_star - n + 1), min(n - 1, k_star) + 1)
        grad[j_range] = 2.0 * y[k_star - j_range]
        
        # Check if the gradient is zero (shouldn't happen for non-negative a)
        if np.max(np.abs(grad)) < 1e-15:
            break
        
        # Polyak-Rubinstein step size
        # gamma_t = (f(y_t) - f*) / (||g_t||^2)
        f_curr = M_curr
        grad_norm_sq = np.sum(grad**2)
        if grad_norm_sq < 1e-30:
            break
        gamma = (f_curr - f_star_est) / (grad_norm_sq + 1e-15)
        gamma = max(gamma, 1e-10)
        gamma = min(gamma, step_param / np.sqrt(t))  # cap by Polyak-Rubinstein
        
        # Actually, for Polyak-Rubinstein, the step is:
        # y_{t+1} = y_t - gamma_t * g_t
        # where gamma_t = (f(y_t) - f*) / ||g_t||^2
        # But we also need to project to the constraint set.
        
        # Simpler: use a fixed decay step size with projection
        step_size = step_param / np.sqrt(t + 1.0)
        y_new = y - step_size * grad
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        y = y_new
        
        # Track best
        score = _fast_autocorr(y).max()
        if score < best_score:
            best_score = score
            best_a = y.copy()
        
        # Average
        avg_a += y
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = _fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad-Avg t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M_curr:.6f}")

    return best_a, best_score

def _freq_init(n, rng):
    """Generate initial sequence with approximately flat magnitude spectrum."""
    # Random phase, flat magnitude
    freq_size = 1 << (2 * n - 1).bit_length()
    phi = rng.uniform(0, 2 * np.pi, freq_size // 2 + 1)
    mag = np.ones(freq_size // 2 + 1)
    A = mag * np.exp(1j * phi)
    # Make it Hermitian for real signal
    A_full = np.zeros(freq_size, dtype=complex)
    A_full[:freq_size // 2 + 1] = A
    for i in range(1, freq_size - freq_size // 2 - 1 + 1):
        A_full[freq_size - i] = np.conj(A[i])
    
    a = irfft(A_full, freq_size)[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def _interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def _coordinate_refine(a0, time_limit, rng, verbose=True):
    """Targeted coordinate-wise refinement focusing on peak contributors."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        ac = _fast_autocorr(a)
        k_star = int(np.argmax(ac))
        
        # Find entries contributing to the peak
        j_low = max(0, k_star - n + 1)
        j_high = min(n - 1, k_star)
        contrib_idx = np.arange(j_low, j_high + 1)
        contrib = a[contrib_idx] * a[k_star - contrib_idx]
        
        if contrib.max() < 1e-15:
            break
        
        # Try to reduce the peak by moving mass from high-contributing entries
        # to low-contributing entries (far from the peak position)
        for _ in range(50):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Pick source from peak contributors
            src_idx = np.argsort(contrib)[::-1][:max(1, len(contrib_idx) // 3)]
            src = contrib_idx[src_idx[rng.integers(len(src_idx))]]
            
            # Pick destination far from peak
            peak_mask = np.zeros(n, dtype=bool)
            # Mark positions that contribute to top peaks
            top_peaks = np.argsort(ac)[::-1][:10]
            for kp in top_peaks:
                jl = max(0, kp - n + 1)
                jh = min(n - 1, kp)
                peak_mask[jl:jh+1] = True
            
            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]
            
            # Pick from low-mass candidates
            candidates_low = candidates[np.argsort(a[candidates])][:max(1, len(candidates)//4)]
            dst = candidates_low[rng.integers(len(candidates_low))]
            
            improved = False
            for frac in [0.0005, 0.001, 0.002, 0.005, 0.01, 0.02]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = _fast_autocorr(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            
            if not improved and rng.random() < 0.3:
                # Random perturb
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > norm * 1e-10:
                    frac = rng.uniform(0.001, 0.01)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        s_t = a_trial.sum()
                        if s_t > 1e-15:
                            a_trial = a_trial * norm / s_t
                            score = _fast_autocorr(a_trial).max()
                            if score < best_score:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                stall = 0
        
        if time.monotonic() - t0 > time_limit:
            break

        # Check stall
        stall += 1
        if stall > 200:
            break
