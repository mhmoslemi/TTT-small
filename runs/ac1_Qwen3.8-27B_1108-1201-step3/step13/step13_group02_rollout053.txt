    # Adaptive step size
    base_lr = 0.5 / n
    lr = base_lr
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        p = p_start + (p_end - p_start) * elapsed_frac
        p = max(p, 10.0)
        
        c = fast_autocorr(a)
        max_c = c.max()
        if max_c < 1e-15:
            break
        
        grad = pnorm_gradient(a, p)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Adaptive step: target reduction in the objective
        # f ~ max_c, we want to reduce it by a small fraction
        step = lr * max_c / (grad_norm + 1e-15)
        step = min(step, 1.0)
        
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        score = fast_autocorr(a_new).max()
        
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            # Decrease learning rate slightly
            lr = max(lr * 0.999, base_lr * 0.01)
        elif score < max_c * 1.01:
            a = a_new  # accept small increases
            lr = max(lr * 0.99, base_lr * 0.01)
        else:
            lr = max(lr * 0.9, base_lr * 0.01)
        
        steps += 1
        
        if verbose and steps % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    pnorm step={steps} p={p:.0f} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def lp_best_response_fast(a, M, solver, max_active=200):
    """LP with constraint reduction using only active constraints."""
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    
    # Start with all constraints
    active = np.ones(size, dtype=bool)
    
    for round_idx in range(4):
        T_active = T[active, :]
        M_active = M  # M is scalar, all constraints have the same RHS
        
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T_active @ b_var <= M_active])
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            return None
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        
        # Check which constraints are active
        circ = T @ b_val
        tol = 1e-7 * M + 1e-12
        active_new = circ >= M - tol
        
        if round_idx == 0:
            # First round: keep a generous set
            n_active = int(active_new.sum())
            if n_active > max_active:
                # Keep only the most violated
                violate = circ - M
                top_idx = np.argsort(violate)[::-1][:max_active]
                active_new = np.zeros(size, dtype=bool)
                active_new[top_idx] = True
            active = active_new
        else:
            if active_new.sum() == active.sum() and np.allactive_equal(active, active_new):
                break
            active = active_new
            if active.sum() < 10:
                break
    
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def allactive_equal(a, b):
    return np.array_equal(a, b)

def nesterov_lp(a0, time_limit, solver, verbose=True):
    """LP Nesterov acceleration for minimizing max autocorrelation."""
    n = len(a0)
    a = normalize(a0, 1.0)
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    best_stall = 0
    
    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        
        b = lp_best_response_fast(y, M_curr, solver, max_active=150)
        if b is None:
            if n_iters > 1000:
                # Try with full constraints
                T = build_conv_matrix(y)
                b_var = cp.Variable(n, nonneg=True)
                prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M_curr])
                try:
                    prob.solve(solver=solver, verbose=False)
                    if prob.status in ('optimal', 'optimal_inaccurate'):
                        b = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
                except Exception:
                    b = None
            if b is None:
                continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b / b_sum
        n_iters += 1
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_a
        
        y = y_new
        a = a_new
        
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
            best_stall = 0
        else:
            stall += 1
            best_stall += 1
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
            best_stall = 0
        
        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, stall={stall}")
        
        # Adaptive restart
        if stall > 300:
            rng = np.random.default_rng(n_iters * 13 + 7)
            perturb_scale = 0.01 * a.mean()
            y = np.maximum(y + rng.normal(0, perturb_scale, n), 0)
            y = y / y.sum()
            a = np.maximum(a + rng.normal(0, perturb_scale, n), 0)
            a = a / a.sum()
            stall = 0
            if verbose:
                print(f"    [Perturb restart at iter {n_iters}]")
        
        if best_stall > 2000:
            break
    
    return best_a, best_score

def local_refinement(a0, time_limit, verbose=True):
    """Local refinement via pair swaps and small perturbations."""
    n = len(a0)
    rng = np.random.default_rng(12345)
    a = normalize(a0, 1.0)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_improve = 0
    n_trials = 0
    T = best_score * 0.005
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = (time.monotonic() - t0) / time_limit
        T = best_score * 0.005 * (1.0 - elapsed_frac * 0.5)
        
        c = fast_autocorr(a)
        # Find the peak
        peak_idx = np.argmax(c)
        
        # Strategy 1: Move mass from peak contributors to non-peak positions
        if peak_idx < n:
            js = 0
            je = peak_idx
        else:
            js = peak_idx - n + 1
            je = n - 1
        
        idx = np.arange(js, je + 1)
        if len(idx) == 0:
            continue
        contrib = a[idx] * a[peak_idx - idx + (0 if peak_idx < n else n - 1 - peak_idx)]
        # Actually let me just use the contrib from the autocorrelation
        contrib = np.array([a[i] * a[peak_idx - i] for i in idx if 0 <= peak_idx - i < n])
        # Hmm this is getting complicated. Let me simplify.
        
        # Simpler: find the pair (i, j) with i+j = peak_idx (or |i-j| = peak_idx) that contributes most
        # For conv, result[k] = sum_i a[i]*a[k-i] (with valid indices)
        # So for peak at k, the pairs are (i, k-i) for valid i
        
        k = peak_idx
        pairs = []
        for i in range(max(0, k - n + 1), min(n, k + 1)):
            j = k - i
            if 0 <= j < n:
                pairs.append((i, j))
        
        if pairs:
            pairs = np.array(pairs)
            contribs = a[pairs[:, 0]] * a[pairs[:, 1]]
            src_idx = pairs[np.argmax(contribs), 0]
            
            # Find destination: position that doesn't contribute to any top peak
            top_peaks = np.argsort(c)[::-1][:5]
            peak_positions = set()
            for pk in top_peaks:
                for i in range(max(0, pk - n + 1), min(n, pk + 1)):
                    if 0 <= pk - i < n:
                        peak_positions.add(i)
            
            candidates = [i for i in range(n) if i not in peak_positions and i != src_idx]
            if candidates:
                # Pick a candidate with low value
                candidates = np.array(candidates)
                dst = candidates[np.argmin(a[candidates])]
                
                # Move a small fraction
                frac = rng.uniform(0.01, 0.08)
                amt = a[src_idx] * frac
                if amt > 1e-15:
                    a_trial = a.copy()
                    a_trial[src_idx] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial / s_t
                        score = fast_autocorr(a_trial).max()
                        n_trials += 1
                        
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            n_improve += 1
                        elif rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                            a = a_trial
        
        # Strategy 2: Random pair swap
        if rng.random() < 0.3:
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j:
                a_trial = a.copy()
                a_trial[i], a_trial[j] = a_trial[j], a_trial[i]
                score = fast_autocorr(a_trial).max()
                n_trials += 1
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    n_improve += 1
                elif rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                    a = a_trial
        
        # Strategy 3: Small random perturbation
        if rng.random() < 0.2:
            n_pert = max(1, n // 100)
            idx_pert = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for idx in idx_pert:
                amt = a[idx] * rng.uniform(0.05, 0.2)
                j = rng.integers(n)
                a_trial[idx] -= amt
                a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial / s_t
                score = fast_autocorr(a_trial).max()
                n_trials += 1
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    n_improve += 1
                elif rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                    a = a_trial
        
        if verbose and n_trials % 5000 == 0 and n_trials > 0:
            elapsed = time.monotonic() - t0
            print(f"    LocalRef trials={n_trials} improve={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

def make_initialization(n, method='uniform'):
    if method == 'uniform':
        return np.ones(n) / n
    elif method == 'half_sine':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x)
        return a / a.sum()
    elif method == 'trapezoid':
        x = np.linspace(0, 1, n)
        a = np.maximum(0, 1 - 2 * np.abs(x - 0.5))
        if a.sum() < 1e-15:
            return np.ones(n) / n
        return a / a.sum()
    elif method == 'edge_bump':
        x = np.linspace(0, 1, n)
        a = 1 + 0.5 * (np.cos(2 * np.pi * x) + 1)
        a = np.maximum(a, 0)
        return a / a.sum()
    elif method == 'random':
        rng = np.random.default_rng(42)
        a = rng.random(n)
        return a / a.sum()
    elif method == 'two_peak':
        x = np.linspace(0, 1, n)
        a = np.exp(-50 * (x - 0.3)**2) + np.exp(-50 * (x - 0.7)**2) + 0.1
        return a / a.sum()
    else:
        return np.ones(n) / n
