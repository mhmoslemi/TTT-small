**Bisection on M:**
We want to find the minimum M such that there exists a >= 0, sum a = 1, (a*a)[k] <= M for all k.

For a given M, the feasibility problem is:
Find a >= 0, sum a = 1, (a*a)[k] <= M for all k.

This is a non-convex feasibility problem (because (a*a)[k] is not concave in a). However, we can solve it iteratively:

1. Start with a feasible point (e.g., uniform)
2. If (a*a)[k] > M for some k, find a direction that reduces (a*a)[k]
3. Move in that direction while maintaining feasibility for other constraints

This is essentially a "constraint satisfaction" approach.

Alternatively, we can use the "multiplicative weights" approach:
- Maintain a "dual variable" for each constraint k
- Update: a[i] = a[i] * product_k (1 - eta*indicator[k*(i)])
- This is like a "repulsion" from the peaks

Let me think about a concrete algorithm:

**Algorithm: "Repulsive force" method**

1. Initialize a uniformly (or from previous best)
2. At each step:
   a. Compute (a*a)[k] for all k
   b. For each k, compute the "force" on a[i] due to constraint k: F_k[i] = a[k-i] (the derivative of (a*a)[k] w.r.t. a[i])
   c. Compute the "pressure" at each k: P_k = max(0, (a*a)[k] - M_target)
   d. The total force on a[i] is: F[i] = sum_k P_k * F_k[i]
   e. Update: a[i] = a[i] / (1 + eta * F[i]) (multiplicative update that reduces a[i] where it contributes to peaks)
   f. Renormalize: a = a / sum(a)
3. Decrease M_target over time (annealing)

This is like a "soft" version of the constraint satisfaction approach.

Actually, this is very similar to the "Multiplicative Weights Update" (MWU) or "Exponential Weights" algorithm used in learning theory.

"""

2. LP best-response with Nesterov acceleration - polish

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None

def _get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return _global_solver
        except Exception:
            continue
    _global_solver = cp.CLARABEL
    print(f"  LP solver (fallback): CLARABEL")
    return _global_solver

def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def _lse_gradient(a, w):
    """Compute gradient of LSE-smoothed max autocorrelation.
    grad[i] = 2 * sum_j a[j] * w[i+j] for i=0,...,n-1
    """
    n = len(a)
    # a_rev[j] = a[n-1-j]
    # cross[i] = conv_full[i+n-1] where conv_full = convolve(a_rev, w)
    a_rev = a[::-1]
    # w has length 2n-1
    total_len = n + (2*n - 1) - 1  # = 3n-2
    fft_size = 1 << (total_len).bit_length()
    A_rev = rfft(a_rev, fft_size)
    W = rfft(w, fft_size)
    conv_full = irfft(A_rev * W, fft_size)
    grad = 2.0 * conv_full[n-1:2*n-1]
    return grad

def _mirror_descent_lse(a0, time_limit, T_schedule, verbose=True):
    """Mirror descent (multiplicative updates) with LSE smoothing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    
    for stage_i, T in enumerate(T_schedule):
        # Step size: smaller when T is small (sharper peaks need smaller steps)
        eta = 50.0 / (n * max(T, 0.1))
        eta = min(eta, 2.0)  # cap for stability
        
        # Number of iterations proportional to 1/T (more iters for lower T)
        max_iters = int(5000 * (T_schedule[0] / T) ** 0.4)
        max_iters = max(max_iters, 500)
        
        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_score
            
            conv = _fast_autocorr(a)
            conv_max = conv.max()
            
            # LSE weights
            shifted = (conv - conv_max) / T
            shifted = np.clip(shifted, -60.0, 60.0)
            exp_s = np.exp(shifted)
            w = exp_s / exp_s.sum()
            
            # Gradient
            grad = _lse_gradient(a, w)
            
            # Multiplicative update: a_new = a * exp(-eta * grad) / Z
            log_update = -eta * grad
            log_update = log_update - log_update.max()  # stability
            a_new = a * np.exp(log_update)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / s
            
            a = a_new
            
            # Periodic check
            if it % 200 == 0:
                cm = _fast_autocorr(a).max()
                if cm < best_score:
                    best_score = cm
                    best_a = a.copy()
            
        # End of stage check
        cm = _fast_autocorr(a).max()
        if cm < best_score:
            best_score = cm
            best_a = a.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    MD-LSE stage={stage_i} T={T:.2f} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def _build_conv_matrix(a, n):
    """Build the matrix T such that T @ b = a * b (convolution)."""
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def _lp_best_response(a, M, solver):
    """Solve: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = _build_conv_matrix(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def _lp_nesterov(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = _fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        b = _lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        score_y = _fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        score_a = _fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        avg_a += a
        n_avg += 1
        if n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = _fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
        
        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nesterov iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def _peak_shave(a, time_limit, seed=42):
    """Targeted peak reduction via mass redistribution."""
    n = len(a)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = _fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:10]
        improved = False
        
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]
            
            # Find destination: index that doesn't contribute to any top peak
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2+1] = True
            
            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]
            
            # Prefer candidates with low current value
            candidates_sorted = candidates[np.argsort(a[candidates])]
            n_try = max(3, len(candidates_sorted) // 5)
            
            for dst_idx in range(min(n_try, len(candidates_sorted))):
                dst = candidates_sorted[dst_idx]
                for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = _fast_autocorr(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-10:
                frac = rng.uniform(0.003, 0.04)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = _fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
            
            if stall > 1000:
                break
    
    return best_a, best_score

def _interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def _mirror_descent_lse_v2(a0, time_limit, T_schedule, eta_schedule, verbose=True):
    """Improved mirror descent with adaptive step size."""
    n = len(a0)

            conv = _fast_autocorr(a)
            conv_max = conv.max()

            shifted = (conv - conv_max) / T

            w = exp_s / exp_s.sum()
            
            # Gradient
            grad = _lse_gradient(a, w)

            a = a_new
            
            if it % 100 == 0:
                cm = _fast_autocorr(a).max()
                if cm < best_score:
                    best_score = cm
                    best_a = a.copy()
        
        cm = _fast_autocorr(a).max()
        if cm < best_score:
            best_score = cm
            best_a = a.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    MD-LSE v2 stage={stage_i} T={T:.2f} ({elapsed:.0f}s): best={best_score:.8f}")
