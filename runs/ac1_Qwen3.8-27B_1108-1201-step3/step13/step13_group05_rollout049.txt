def normalize_to_sum(a, target_sum):
    """Normalize a non-negative vector to have the target sum."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * target_sum / len(a)
    return a * target_sum / s


def exact_subgrad(a, k_star):
    """
    Compute exact subgradient of f(a) = max_k conv(a,a)_k at point a
    where the maximum is achieved at lag k_star.
    
    Returns gradient vector g of same length as a.
    g[i] = 2*a[k_star - i] if 0 <= k_star - i < n and 2*i != k_star
           a[k_star - i] if 0 <= k_star - i < n and 2*i == k_star
           0 otherwise
    """
    n = len(a)
    g = np.zeros(n)
    i_min = max(0, k_star - n + 1)
    i_max = min(n - 1, k_star)
    for i in range(i_min, i_max + 1):
        j = k_star - i
        if 0 <= j < n:
            if 2 * i == k_star:
                g[i] = a[j]
            else:
                g[i] = 2.0 * a[j]
    return g


def exact_subgrad_fast(a, k_star):
    """
    Vectorized exact subgradient computation.
    """
    n = len(a)
    g = np.zeros(n)
    i_min = max(0, k_star - n + 1)
    i_max = min(n - 1, k_star)
    idx = np.arange(i_min, i_max + 1)
    j_idx = k_star - idx
    # All valid since i_min ensures k_star - i_max >= 0 and i_max ensures k_star - i_min < n
    vals = 2.0 * a[j_idx]
    # Fix the middle term if k_star is even and i = k_star/2 is in range
    if k_star % 2 == 0:
        mid = k_star // 2
        if i_min <= mid <= i_max:
            pos = mid - i_min
            vals[pos] = a[j_idx[pos]]
    g[i_min:i_max + 1] = vals
    return g

def fista_optimize(a0, time_limit, verbose=False):
    """
    FISTA (accelerated proximal gradient) for minimizing max conv(a,a)
    on the simplex {a >= 0, sum(a) = sqrt(2n)}.
    """
    n = len(a0)
    c = np.sqrt(2.0 * n)
    
    def proj_simplex(v, s):
        """Project v onto {x >= 0, sum(x) = s}."""
        n_v = len(v)
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u)
        cond = u * np.arange(1, n_v + 1) + s - cssv > 0
        rho = u[np.argmax(cond)]
        return np.maximum(v - rho, 0.0)
    
    y = normalize_to_sum(a0, c)
    x = y.copy()
    
    best_a = x.copy()
    best_m = fast_autocorr(x).max()
    t0 = time.monotonic()
    
    # Lipschitz constant estimate
    ac_y = fast_autocorr(y)
    k_star = int(np.argmax(ac_y))
    g_y = exact_subgrad_fast(y, k_star)
    L = np.linalg.norm(g_y, ord=np.inf) + 1e-10
    step = 1.0 / L
    
    t = 1
    while time.monotonic() - t0 < time_limit:
        ac_y = fast_autocorr(y)
        k_star = int(np.argmax(ac_y))
        g_y = exact_subgrad_fast(y, k_star)
        
        # Update step if needed
        L_new = np.linalg.norm(g_y, ord=np.inf) + 1e-10
        if L_new < L * 0.9:
            L = L_new
            step = 1.0 / L
        
        # Proximal step
        z = y - step * g_y
        x_new = proj_simplex(z, c)
        
        # Nesterov acceleration
        t_next = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y_new = x_new + ((t - 1.0) / t_next) * (x_new - x)
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = x_new.copy()
        else:
            y_new = y_new * c / s_y
        
        x = x_new
        y = y_new
        t = t_next
        
        # Track best
        M_new = fast_autocorr(x).max()
        if M_new < best_m:
            best_m = M_new
            best_a = x.copy()
        
        M_y = fast_autocorr(y).max()
        if M_y < best_m:
            best_m = M_y
            best_a = y.copy()
        
        if verbose and int(t) % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c * c)
            print(f"    FISTA t={int(t)} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m


def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. conv(a,b)_k <= M for all k, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    
    # Build constraint matrix: T[i,j] = a[j] if i-j is in [0, n-1]
    # conv(a,b)_k = sum_j a_j * b_{k-j}
    # For efficiency, use structure: T is banded
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_lp(a0, time_limit, solver, verbose=False, lp_every=5):
    """
    Nesterov-accelerated LP best-response with periodic LP solves.
    Between LP solves, uses exact subgradient for fast updates.
    """
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = normalize_to_sum(a0, c)
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    t = 1
    
    for it in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac_y = fast_autocorr(y)
        M_curr = ac_y.max()
        k_star = int(np.argmax(ac_y))
        
        # Best response direction
        if it % lp_every == 1:
            # Solve LP for exact best response
            b = lp_best_response(y, M_curr, solver)
            if b is not None:
                b_sum = b.sum()
                if b_sum > 1e-15:
                    g = b * c / b_sum
                else:
                    g = normalize_to_sum(exact_subgrad_fast(y, k_star), c)
            else:
                g = normalize_to_sum(exact_subgrad_fast(y, k_star), c)
        else:
            # Use subgradient direction as approximation
            g_dir = exact_subgrad_fast(y, k_star)
            g = normalize_to_sum(g_dir + 0.01, c)
        
        # Nesterov update
        alpha = 2.0 / (it + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize_to_sum(y_new, c)
        
        if it == 1:
            x_new = g.copy()
        else:
            beta = (it - 1.0) / (it + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            x_new = normalize_to_sum(x_new, c)
        
        y = y_new
        x = x_new
        
        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_m:
            best_m = score_y
            best_a = y.copy()
        
        score_x = fast_autocorr(x).max()
        if score_x < best_m:
            best_m = score_x
            best_a = x.copy()
        
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c * c)
            print(f"    NLP it={it} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m


def coordinate_polish(a0, time_limit, n_trials_per_peak=5, verbose=False):

def generate_starting_points(n, target_sum):

    points.append(s); names.append("flat")
    
    # Two-lobe
    s = np.sin(np.pi * x) ** 2 + 0.3 * np.sin(2 * np.pi * x) ** 2
    s = np.maximum(s, 0)
    points.append(s); names.append("two_lobe")

    s = (x * (1 - x)) ** 0.5

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
