import numpy as np

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print(f"  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response_warm(a, M, solver, warm_b=None):
    """LP best response with warm start."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_conv_matrix(a)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def lp_best_response_reduced(a, M, solver):
    """LP with constraint reduction for speed."""
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)

    for _ in range(4):
        circ = T @ b_val
        tol = 0.001 * M
        active = circ >= M - tol
        n_active = int(active.sum())
        if n_active >= size - 5 or n_active < 10:
            break
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T[active, :] @ b_var <= M[active]])
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        b_new = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        circ_new = T @ b_new
        if circ_new.max() <= M * (1 + 1e-5):
            b_val = b_new
        else:
            break

    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """Nesterov LP acceleration with improved convergence."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_M = fast_autocorr(a).max()

    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        if t < 30:
            b = lp_best_response_warm(y, M_curr, solver)
        elif t < 80:
            b = lp_best_response_reduced(y, M_curr, solver)
        else:
            b = lp_best_response_reduced(y, M_curr, solver)

        if b is None:
            M_curr = fast_autocorr(y).max()
            b = lp_best_response_warm(y, M_curr, solver)
            if b is None:
                continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_M:
            best_M = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_M:
            best_M = score_a
            best_a = a.copy()
            stall = 0

        if verbose and n_iters % 500 == 0:

def compute_lse_gradient(a, w):
    n = len(a)
    a_rev = a[::-1]
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)
    conv = irfft(W * AR, fft_size)
    return 2.0 * conv[n - 1:2 * n - 1]


def lse_gradient_descent(a0, time_limit, verbose=True):
    """LSE gradient descent with adaptive step size."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0

    T_start = max(0.5, best_score * 0.05)
    T_end = max(0.001, best_score * 0.002)

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress

        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break

        w = np.exp((b - max_b) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        grad = compute_lse_gradient(a, w)

        # Adaptive step: try a few step sizes
        best_step_score = best_score
        best_step_a = a
        for step_mult in [1.0, 0.5, 2.0, 0.25]:
            step = step_mult / (max(T, 0.01) * n * 0.3)
            a_trial = a - step * grad
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial * norm / s_t
            sc = fast_autocorr(a_trial).max()
            if sc < best_step_score:
                best_step_score = sc
                best_step_a = a_trial

        a = best_step_a
        steps += 1

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and steps % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-GD step={steps} ({elapsed:.0f}s): T={T:.4f}, best={best_score:.8f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, verbose=True):
    """Coordinate descent with exact 1D subproblem solving."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_passes = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        improved = False
        rng = np.random.default_rng(n_passes * 1000 + 42)

        # Process positions in random order for diversity
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[i]
            if old_val < 1e-15:
                continue

            # Compute base convolution (without a_i's contribution)
            a_zero = a.copy()
            a_zero[i] = 0.0
            c_base = fast_autocorr(a_zero)

            # Determine which lags are affected by a_i
            # c_k = c_base_k + 2*a_i*a_{k-i} (for k != 2i, 0<=k-i<n)
            # c_k = c_base_k + a_i^2 (for k = 2i)

            # Candidate values for a_i
            candidates = [old_val, 0.0, old_val * 0.5, old_val * 1.5]

            # Add candidates that equalize with current max
            c_full = fast_autocorr(a)
            M_curr = c_full.max()

            # For k = 2i: a_i^2 + base_{2i} = M_curr => a_i = sqrt(M_curr - base_{2i})
            k_center = 2 * i
            if 0 <= k_center < 2 * n - 1:
                base_at_center = c_base[k_center]
                if M_curr > base_at_center:
                    candidates.append(np.sqrt(M_curr - base_at_center))

            # For linear terms: 2*a_i*a_{k-i} + base_k = M_curr
            # => a_i = (M_curr - base_k) / (2*a_{k-i})
            top_lags = np.argsort(c_full)[::-1][:5]
            for k in top_lags:
                k_minus_i = k - i
                if 0 <= k_minus_i < n and k != 2 * i and a[k_minus_i] > 1e-15:
                    base_at_k = c_base[k]
                    if M_curr > base_at_k:
                        candidates.append((M_curr - base_at_k) / (2.0 * a[k_minus_i]))

            # Try each candidate
            best_val = old_val
            best_candidate_M = M_curr
            for cand in candidates:
                if cand < 0 or cand > norm:
                    continue
                a_trial = a.copy()
                a_trial[i] = cand
                c_trial = c_base.copy()
                # Update affected lags
                for k in range(2 * n - 1):
                    k_minus_i = k - i
                    if 0 <= k_minus_i < n:
                        if k == 2 * i:
                            c_trial[k] = c_base[k] + cand * cand
                        else:
                            c_trial[k] = c_base[k] + 2.0 * cand * a[k_minus_i]
                trial_M = c_trial.max()
                if trial_M < best_candidate_M - 1e-10:
                    best_candidate_M = trial_M
                    best_val = cand

            if best_val != old_val and best_val >= 0:
                a[i] = best_val
                s_a = a.sum()
                if s_a > 1e-15:
                    a = a * norm / s_a
                    new_M = fast_autocorr(a).max()
                    if new_M < best_M:
                        best_M = new_M
                        best_a = a.copy()
                        improved = True

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc pass={n_passes} ({elapsed:.0f}s): best_M={best_M:.8f}, improved={improved}")

        if not improved and n_passes > 20:
            break

    return best_a, best_M


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def peak_shave_refine(a0, time_limit, seed=42, verbose=True):
    """Targeted peak shaving with single-source redistribution."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_improve = 0
    n_stall = 0

    T = max(0.1, best_M * 0.01)

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        M_curr = b.max()
        top_k = np.argsort(b)[::-1][:5]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            # Find best destination: position that minimizes new max
            non_src = np.where(np.arange(n) != src)[0]
            # Focus on positions not contributing to any top peak
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True
            non_peak = non_src[~peak_mask]
            if len(non_peak) == 0:
                non_peak = non_src

            # Try a few destinations
            candidates_dst = non_peak[np.argsort(a[non_peak])][:10]
            if len(candidates_dst) == 0:
                continue

            frac = rng.choice([0.01, 0.03, 0.05, 0.1])
            amt = a[src] * frac
            if amt < 1e-15:
                continue

            for dst in candidates_dst[:5]:
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_M:
                    a = a_trial
                    best_M = score
                    best_a = a.copy()
                    improved = True
                    n_improve += 1
                    break
                elif rng.random() < np.exp(-(score - best_M) / max(T, 1e-10)):
                    a = a_trial
                    improved = True
                    break

        if not improved:
            n_stall += 1
            if n_stall > 50:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > norm * 1e-10:
                    frac = rng.uniform(0.02, 0.1)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_M * 1.01:
                            a = a_trial
                            if score < best_M:
                                best_M = score
                                best_a = a.copy()
                                n_improve += 1
                n_stall = 0
                T *= 0.95

        if verbose and n_improve % 1000 == 0 and n_improve > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave improve={n_improve} ({elapsed:.0f}s): best_M={best_M:.8f}")

    return best_a, best_M

        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    # Phase 1: Nesterov LP at base length
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Nesterov LP n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.30, 480.0)
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n_base}")

    # Phase 2: Peak shave at base length
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: PeakShave n={n_cur} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.10, 120.0)
    if t2 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = peak_shave_refine(a_norm, t2, seed=99, verbose=True)
        do_try_update(a_out, f"pd_n{n_cur}")

    # Phase 3: Interpolate to n=2000, Nesterov
    n2 = max(n_base + 200, 2000)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Nesterov LP n={n2} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.25, 350.0)
    if t3 > 30:
        init_long = interpolate_sequence(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n2}")

    # Phase 4: Interpolate to n=2400, Nesterov
    n3 = max(n_base + 400, 2400)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Nesterov LP n={n3} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.20, 280.0)
    if t4 > 30:
        init_long3 = interpolate_sequence(best_seq, n3)
        a_norm = init_long3 * np.sqrt(2.0 * n3) / init_long3.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n3}")

    # Phase 5: LSE gradient descent on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LSE-GD n={n_best} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 180.0)
    if t5 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_gradient_descent(a_norm, t5, verbose=True)
        do_try_update(a_out, f"lse_n{n_best}")

        # Coordinate descent right after LSE
        remaining = time_budget - (time.monotonic() - t_global)
        t5b = min(remaining * 0.10, 100.0)
        if t5b > 10:
            n_cur = len(best_seq)
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = coordinate_descent(a_norm, t5b, verbose=True)
            do_try_update(a_out, f"cd_n{n_cur}")

    # Phase 6: Final peak shave
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final PeakShave n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.15, 150.0)
    if t6 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_shave_refine(a_norm, t6, seed=2024, verbose=True)
        do_try_update(a_out, f"pd_final_n{n_best}")

    # Phase 7: Final coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final CoordDesc n={n_best} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.15, 150.0)
    if t7 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent(a_norm, t7, verbose=True)
        do_try_update(a_out, f"cd_final_n{n_best}")

    # Phase 8: Final Nesterov polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final Nesterov n={n_best} (budget={remaining:.0f}s) ===")
    t8 = min(remaining * 0.30, 200.0)
    if t8 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t8, solver, verbose=True)
        do_try_update(a_out, f"lp_final_n{n_best}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
