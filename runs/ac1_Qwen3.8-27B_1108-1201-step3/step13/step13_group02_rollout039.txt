def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_solve_reduced(a, M, solver, n_keep=100):
    """LP best response with constraint reduction."""
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    
    # First solve with all constraints
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    
    # Now reduce: keep only the tightest constraints
    for _ in range(5):
        circ = T @ b_val
        # Keep the n_keep tightest constraints
        idx = np.argsort(circ)[::-1][:max(n_keep, 50)]
        if len(idx) >= size:
            break
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T[idx, :] @ b_var <= M[idx]])
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        b_new = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        # Verify feasibility
        circ_new = T @ b_new
        if circ_new.max() <= M * (1 + 1e-4):
            b_val = b_new
        else:
            # Infeasible, add back violated constraints
            violated = np.where(circ_new > M)[0]
            idx = np.unique(np.concatenate([idx, violated]))
    
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def lse_momentum_descent(a0, time_limit, verbose=True):

    n = len(a0)
    norm = np.sqrt(2.0 * n)

    s0 = a.sum()

        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0
    
    # Nesterov momentum state
    y = a.copy()
    prev_a = a.copy()
    
    T_start = max(0.5, best_score * 0.1)
    T_end = max(0.001, best_score * 0.003)
    
    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)

        AR = rfft(a_rev, fft_size)
        conv = irfft(W * AR, fft_size)
        grad = 2.0 * conv[n - 1:2 * n - 1]
        
        # Nesterov step
        t = steps + 1
        beta = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / (2.0 * t + 1.0)
        
        # Adaptive step size
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:

        step = (y.sum() / (n * grad_norm)) * 0.1

        a = a_new

def peak_directed_search(a0, time_limit, seed=42, verbose=True):
    """Targeted peak reduction: identify top peaks and reduce them specifically."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    T = best_score * 0.01
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        # Get top 5 peaks
        top_k = np.argsort(b)[::-1][:5]
        
        improved = False
        for k in top_k:
            # Find which pairs (i, j) with i+j=k contribute most
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            
            if contrib.max() < 1e-18:
                continue
            
            # The dominant contributor
            src = idx[np.argmax(contrib)]
            partner = k - src
            
            # Try reducing src and adding to a position that doesn't
            # contribute to any top peak
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True
            
            non_peak = np.where(~peak_mask)[0]
            if len(non_peak) == 0:
                non_peak = np.where(a < a.mean())[0]
            
            # Try several fractions
            for frac in [0.05, 0.1, 0.2, 0.3]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                
                # Pick destination
                if len(non_peak) > 0:
                    dst = non_peak[rng.integers(len(non_peak))]
                else:
                    dst = rng.integers(n)
                
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                
                score = fast_autocorr(a_trial).max()
                
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
                elif rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                    a = a_trial
                    improved = True
                    break
            
            if improved:
                break
        
        if not improved:
            # Random pair swap
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-10:
                frac = rng.uniform(0.1, 0.3)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                    elif rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                        a = a_trial
        
        # Decrease temperature
        T = max(T * 0.999, best_score * 0.001)
        
        if verbose:
            elapsed = time.monotonic() - t0
            if int(elapsed) % 10 == 0:
                print(f"    PeakDir ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def try_different_init(n, current_best):
    """Try a few different initialization shapes."""
    candidates = []
    
    # Current best (interpolated)
    candidates.append(interpolate_sequence(current_best, n))
    
    # Raised cosine
    x = np.linspace(0, 1, n)
    c = np.cos(np.pi * (x - 0.5)) ** 2
    c = np.maximum(c, 0) / c.sum()
    candidates.append(c)
    
    # Gaussian
    x = np.linspace(0, 1, n)
    c = np.exp(-20 * (x - 0.5) ** 2)
    c = c / c.sum()
    candidates.append(c)
    
    # Sine

    c = c / c.sum()
    candidates.append(c)
    
    # Bimodal
    c = np.exp(-50 * (x - 0.3) ** 2) + np.exp(-50 * (x - 0.7) ** 2)
    c = c / c.sum()
    candidates.append(c)
    
    return candidates

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print("  LP solver: " + str(s))
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def get_pause(a, n):
    """Return the (2n-1, n) convolution matrix T where (T @ b)[k] = (a*b)[k]."""
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def lp_best_response(a, M, solver, n_keep=100):
    """Solve the LP: max sum(b) s.t. (a*b)(k) <= M for all k, b >= 0.
    Uses constraint reduction for speed."""
    n = len(a)
    size = 2 * n - 1
    T = get_pause(a, n)
    b_var = cp.Variable(n, nonneg=True)

    # First solve with all constraints
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)

    # Iteratively reduce to most binding constraints
    for _ in range(4):
        circ = T @ b_val
        # Keep top n_keep constraints by tightness
        idx = np.argsort(circ)[::-1][:max(n_keep, 40)]
        if len(idx) >= size:
            break
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T[idx, :] @ b_var <= M[idx]])
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        b_new = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        # Verify full feasibility
        circ_new = T @ b_new
        if circ_new.max() <= M * (1 + 1e-4):
            b_val = b_new
        else:
            # Add back violated constraints
            violated = np.where(circ_new > M)[0]
            idx = np.unique(np.concatenate([idx, violated]))
            if len(idx) >= size:
                break

    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def lp_dual_solve(a, M, solver):
    """Solve the dual LP: min M*sum(y) s.t. T^T y >= 1, y >= 0.
    Returns the optimal sum(b) value (strong duality)."""
    n = len(a)
    T = get_pause(a, n)
    y_var = cp.Variable(2 * n - 1, nonneg=True)
    prob = cp.Problem(cp.Minimize(M * cp.sum(y_var)), [T.T @ y_var >= 1.0])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    return float(prob.value)


def nesterov_lp_optimize(a0, time_limit, solver, n_keep=100, verbose=True):
    """LP-based Nesterov method with adaptive restarts."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    global_stall = 0

    y = a.copy()
    prev_y = a.copy()

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, solver, n_keep=n_keep)
        if b is None:
            n_iters += 1
            stall += 1
            global_stall += 1
            if stall > 100:
                break
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov step with mild adaptation
        alpha = 2.0 / (t + 2.0)
        if t <= 50:
            alpha = min(alpha * 1.5, 1.0)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        # Nesterov extrapolation
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - prev_y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        prev_y = y
        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
            global_stall = 0
        else:
            stall += 1
            global_stall += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
            global_stall = 0

        # Restart at different scales
        if stall > 150:
            rng = np.random.default_rng(n_iters * 7 + 3)
            ps = 0.02 * a.mean()
            y = np.maximum(y + rng.normal(0, ps, n), 0)
            y = y * norm / y.sum()
            a = np.maximum(a + rng.normal(0, ps, n), 0)
            a = a * norm / a.sum()
            prev_y = a.copy()
            stall = 0

        if global_stall > 400:
            rng = np.random.default_rng(n_iters * 13 + 7)
            ps = 0.08 * a.mean()
            y = np.maximum(y + rng.normal(0, ps, n), 0)
            y = y * norm / y.sum()
            a = np.maximum(a + rng.normal(0, ps, n), 0)
            a = a * norm / a.sum()
            prev_y = a.copy()
            global_stall = 0

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print("    NLP iter=%d (%.0fs): best=%.8f, stall=%d" % (n_iters, elapsed, best_score, stall))

    return best_a, best_score


def lse_momentum_descent(a0, time_limit, verbose=True):
    """LSE-smoothed gradient descent with Nesterov momentum."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0

    y = a.copy()
    prev_y = a.copy()

    T_start = max(0.5, best_score * 0.08)
    T_end = max(0.001, best_score * 0.002)
    fft_size = 1 << (3 * n - 2).bit_length()

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress

        b = fast_autocorr(y)
        max_b = b.max()
        if max_b < 1e-15:
            break

        w = np.exp((b - max_b) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        # FFT-based gradient
        a_rev = y[::-1]
        W = rfft(w, fft_size)
        AR = rfft(a_rev, fft_size)
        conv = irfft(W * AR, fft_size)
        grad = 2.0 * conv[n - 1:2 * n - 1]

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Adaptive step
        step = 0.1 * y.sum() / (n * grad_norm)
        y_new = y - step * grad
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        # Nesterov momentum
        t = steps + 1
        beta = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / (2.0 * t + 1.0)
        a_new = beta * y_new + (1.0 - beta) * prev_y
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        if s_a < 1e-15:
            a_new = y_new.copy()
        else:
            a_new = a_new * norm / s_a

        prev_y = y_new
        y = a_new
        a = a_new
        steps += 1

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and steps % 200 == 0:
            elapsed = time.monotonic() - t0
            print("    LSE-Mom step=%d (%.0fs): T=%.4f, best=%.8f" % (steps, elapsed, T, best_score))

    return best_a, best_score


def peak_directed_search(a0, time_limit, seed=42, verbose=True):
    """Target the top autocorrelation peaks for reduction."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    T = best_score * 0.01
    n_improve = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:5]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            if contrib.max() < 1e-18:
                continue

            src = idx[np.argmax(contrib)]

            # Build peak mask
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            non_peak = np.where(~peak_mask)[0]
            if len(non_peak) == 0:
                non_peak = np.where(a < a.mean())[0]
            if len(non_peak) == 0:
                non_peak = np.array([0])

            for frac in [0.05, 0.1, 0.15, 0.25]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue

                dst = non_peak[rng.integers(len(non_peak))]
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t

                score = fast_autocorr(a_trial).max()

                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    n_improve += 1
                    improved = True
                    break
                elif rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                    a = a_trial
                    improved = True
                    break

            if improved:
                break

        if not improved:
            # Random pair perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-10:
                frac = rng.uniform(0.05, 0.2)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        n_improve += 1
                    elif rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                        a = a_trial

        T = max(T * 0.999, best_score * 0.001)

        if verbose and n_improve % 2000 == 0 and n_improve > 0:
            elapsed = time.monotonic() - t0
            print("    PeakDir improve=%d (%.0fs): best=%.8f" % (n_improve, elapsed, best_score))

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def try_update(state, a, label):
    """Update the best result stored in state dict."""
    if a is None:
        return
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < state['score']:
        state['score'] = sc
        state['seq'] = a.copy()
        state['n'] = len(a)
        print("  *** NEW BEST [" + label + "]: " + "%.6f" % sc + " (n=" + str(len(a)) + ")")


def init_from_parent(n, parent_seq):
    """Create an initial sequence of length n from the parent."""
    return interpolate_sequence(parent_seq, n)


def init_rainbow(n):
    """Raised cosine initialization."""
    x = np.linspace(0.0, 1.0, n)
    c = np.cos(np.pi * (x - 0.5)) ** 2
    c = np.maximum(c, 0)
    c = c / c.sum()
    return c


def init_gaussian(n):
    """Gaussian initialization."""
    x = np.linspace(0.0, 1.0, n)
    c = np.exp(-20.0 * (x - 0.5) ** 2)
    c = c / c.sum()
    return c


def init_sine(n):
    """Sine initialization."""
    x = np.linspace(0.0, 1.0, n)
    c = np.sin(np.pi * x) ** 2
    c = c / c.sum()
    return c


def init_bimodal(n):
    """Bimodal initialization."""
    x = np.linspace(0.0, 1.0, n)
    c = np.exp(-40.0 * (x - 0.3) ** 2) + np.exp(-40.0 * (x - 0.7) ** 2)
    c = c / c.sum()
    return c


def best_init_for_length(n, parent_seq):
    """Pick the best initialization for a given length by quick evaluation."""
    candidates = [
        init_from_parent(n, parent_seq),
        init_rainbow(n),
        init_gaussian(n),
        init_sine(n),
        init_bimodal(n),
    ]
    best_c = candidates[0]
    best_sc = np.inf
    for c in candidates:
        cn = c * np.sqrt(2.0 * n)
        sc = fast_autocorr(cn).max()
        if sc < best_sc:
            best_sc = sc
            best_c = c
    return best_c

        # Quick LSE polish at new length
        remaining = time_budget - (time.monotonic() - t_global)
        t4b = min(remaining * 0.05, 60.0)
        if t4b > 10:
            n_cur = state['n']
            a_norm = state['seq'] * np.sqrt(2.0 * n_cur) / state['seq'].sum()
            a_out, _ = lse_momentum_descent(a_norm, t4b, verbose=False)
            try_update(state, a_out, "lse2_n%d" % n_cur)

    # === Phase 5: Explore n+400 ===
    n3 = state['n'] + 400
    remaining = time_budget - (time.monotonic() - t_global)
    print("\n=== Phase 5: LP Nesterov n=%d (budget=%.0fs) ===" % (n3, remaining))
    t5 = min(remaining * 0.12, 160.0)
    if t5 > 30:
        init3 = best_init_for_length(n3, state['seq'])
        a_norm = init3 * np.sqrt(2.0 * n3) / init3.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t5, solver, n_keep=100, verbose=True)
        try_update(state, a_out, "lp_n%d" % n3)

    # === Phase 6: Final LP polish on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = state['n']
    print("\n=== Phase 6: Final LP n=%d (budget=%.0fs) ===" % (n_best, remaining))
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 20:
        a_norm = state['seq'] * np.sqrt(2.0 * n_best) / state['seq'].sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t6, solver, n_keep=80, verbose=True)
        try_update(state, a_out, "lpfin_n%d" % n_best)

    # === Phase 7: Final peak-directed ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = state['n']
    print("\n=== Phase 7: Final PeakDir n=%d (budget=%.0fs) ===" % (n_best, remaining))
    t7 = min(remaining * 0.08, 80.0)
    if t7 > 10:
        a_norm = state['seq'] * np.sqrt(2.0 * n_best) / state['seq'].sum()
        a_out, _ = peak_directed_search(a_norm, t7, seed=555, verbose=False)
        try_update(state, a_out, "pd_fin_n%d" % n_best)

    # === Phase 8: Final LSE ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = state['n']
    print("\n=== Phase 8: Final LSE n=%d (budget=%.0fs) ===" % (n_best, remaining))
    t8 = min(remaining * 0.12, 120.0)
    if t8 > 10:
        a_norm = state['seq'] * np.sqrt(2.0 * n_best) / state['seq'].sum()
        a_out, _ = lse_momentum_descent(a_norm, t8, verbose=True)
        try_update(state, a_out, "lse_fin_n%d" % n_best)

    # Final verification
    if state['seq'] is not None:
        fs = evaluate_safe(state['seq'])
        if np.isfinite(fs):
            state['score'] = fs
        print("\nFINAL: score=%.6f, n=%d, time=%.0fs" % (state['score'], len(state['seq']), time.monotonic() - t_global))
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in state['seq']]
