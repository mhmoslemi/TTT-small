from scipy.fft import rfft, irfft

_global_solver = None

def _get_solver():

    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])

def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

def _proj_simplex(v, target_sum=1.0):
    """Project vector v onto {x: sum(x)=target_sum, x>=0}."""
    n = len(v)
    if n <= 1:
        return np.array([target_sum])
    # First project onto sum=1, then scale
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 1e-15:
        out = out * target_sum / s
    else:
        out = np.full(n, target_sum / n)
    return out

def _fista_lse(a0, T_schedule, iters_per_stage, time_limit, verbose=False):
    """FISTA on LSE-smoothed max autocorrelation with backtracking."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    best_a = a.copy()
    best_val = _fast_autocorr(a).max()
    t0 = time.monotonic()
    t_fista = 1.0

    for stage_i, T in enumerate(T_schedule):
        # Initial Lipschitz estimate
        L = max(n / (T * 1.0), 10.0)
        
        # Local best tracking
        local_best = best_val
        local_stall = 0
        
        max_iters = max(int(iters_per_stage * (T_schedule[0] / T) ** 0.3), 3000)
        
        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_val
            
            # Backtracking line search
            val_y, grad_y = _lse_value_and_grad(y, T)
            step = 1.0 / L
            
            for _bt in range(20):
                z = y - step * grad_y
                a_new = _proj_simplex(z)
                val_new, _ = _lse_value_and_grad(a_new, T)
                # Check sufficient decrease
                if val_new <= val_y - 0.5 * step * np.dot(grad_y, y - a_new):
                    break
                step *= 0.5
                L *= 2.0
            
            # Nesterov update
            if t_fista == 1.0:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - y)
                y_new = np.maximum(y_new, 0.0)
                s_y = y_new.sum()
                if s_y > 1e-15:
                    y_new = y_new / s_y
                else:
                    y_new = a_new.copy()
            
            prev_y = y
            y = y_new
            a = a_new
            t_fista += 1.0
            
            # Track best
            if it % 100 == 0 or it == max_iters - 1:
                sc = _fast_autocorr(a).max()
                if sc < best_val - 1e-12:
                    best_val = sc
                    best_a = a.copy()
                    local_stall = 0
                elif sc < local_best:
                    local_best = sc
                else:
                    local_stall += 100
            
            if local_stall > 3000 and it > 100:
                # Restart from best
                y = best_a.copy()
                a = best_a.copy()
                t_fista = 1.0
                L = max(n / (T * 1.0), 10.0)
                local_stall = 0
        
        sc = _fast_autocorr(a).max()
        if sc < best_val:
            best_val = sc
            best_a = a.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    FISTA-LSE stage={stage_i} T={T:.2f} ({elapsed:.0f}s): best={best_val:.8f}")

    return best_a, best_val
