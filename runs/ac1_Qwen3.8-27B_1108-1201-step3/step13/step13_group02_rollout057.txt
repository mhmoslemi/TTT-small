def build_conv_matrix_from_a(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def lp_best_response(a, M, solver):
    """Solve the LP: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_conv_matrix_from_a(a)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def lp_best_response_reduced(a, M, solver, n_rounds=5):
    """LP with constraint reduction for speed."""
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix_from_a(a)
    b_var = cp.Variable(n, nonneg=True)
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    
    for _ in range(n_rounds):
        circ = T @ b_val
        tol = 1e-7 * M
        active = circ >= M - tol
        n_active = int(active.sum())
        if n_active >= size - 10:
            break
        if n_active < 30:
            break
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T[active, :] @ b_var <= M[active]])
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        b_new = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        circ_new = T @ b_new
        if circ_new.max() <= M * (1 + 1e-5):
            b_val = b_new

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """LP Nesterov acceleration with optimal averaging."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    
    # Store all iterates for optimal averaging
    all_ys = [y.copy()]
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        if t < 30:
            b = lp_best_response(y, M_curr, solver)
        else:
            b = lp_best_response_reduced(y, M_curr, solver)

        if b is None:
            stall += 1
            if stall > 50:
                break
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new
        all_ys.append(y.copy())
        
        # Optimal averaging: average of all iterates
        if len(all_ys) <= 500:
            avg_y = np.mean(all_ys, axis=0)
            avg_y = np.maximum(avg_y, 0)
            s_avg = avg_y.sum()
            if s_avg > 1e-15:
                avg_y = avg_y * norm / s_avg
                score_avg = fast_autocorr(avg_y).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_y.copy()
                    stall = 0
                else:
                    stall += 1
        else:
            # Just check current and Nesterov
            score_y = fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()
                stall = 0
            else:
                stall += 1
            
            score_a = fast_autocorr(a).max()
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()
                stall = 0

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, stall={stall}")

        if stall > 500:
            rng = np.random.default_rng(n_iters * 13 + 7)
            perturb_scale = 0.03 * a.mean()
            y = np.maximum(y + rng.normal(0, perturb_scale, n), 0)
            y = y * norm / y.sum()
            a = np.maximum(a + rng.normal(0, perturb_scale, n), 0)
            a = a * norm / a.sum()
            stall = 0
            all_ys = [y.copy()]
            if verbose:
                print(f"    [LP Restart at iter {n_iters}]")

def lse_adam(a0, time_limit, lr=0.5, verbose=True):
    """Adam optimizer on LSE-smoothed max autocorrelation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0

    b = fast_autocorr(a)
    max_b = b.max()
    T_start = max(0.1, max_b * 0.05)
    T_end = max(0.001, max_b * 0.002)

    steps = 0
    no_improve = 0
    best_lr = lr

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** elapsed_frac
        
        # Decay learning rate
        lr_eff = lr * (1.0 - 0.3 * elapsed_frac)

        b = fast_autocorr(a)
        max_b = b.max()
        
        # LSE weights (shifted for numerical stability)
        w = np.exp((b - max_b) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        # Gradient of LSE: d/d a_i [sum_k w_k * (a*a)_k]
        # (a*a)_k = sum_j a_j * a_{k-j}
        # d(a*a)_k / da_i = a_{k-i} + a_{i-k} (if in range)
        # = 2 * (w * a_rev) offset
        a_rev = a[::-1]
        size_grad = (2*n-1) + n - 1  # = 3n-2
        fft_size = 1 << (size_grad - 1).bit_length()
        W = rfft(w, fft_size)
        AR = rfft(a_rev, fft_size)
        conv = irfft(W * AR, fft_size)
        grad = 2.0 * conv[n - 1:2 * n - 1]

        # Project gradient onto sum=0 subspace (for the normalization constraint)
        grad = grad - grad.mean()

        # Adam update
        t_count += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t_count)
        v_hat = v / (1 - beta2 ** t_count)

        step = lr_eff * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        a = a_new
        steps += 1

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and steps % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam step={steps} ({elapsed:.0f}s): T={T:.4f}, lr={lr_eff:.4f}, best={best_score:.8f}, no_impr={no_improve}")

        # Adaptive restart
        if no_improve > 1500:
            rng = np.random.default_rng(steps * 3 + 1)
            scale = 0.02 * a.mean()
            a = np.maximum(a + rng.normal(0, scale, n), 0)
            a = a * norm / a.sum()
            m = np.zeros(n)
            v = np.zeros(n)
            t_count = 0
            no_improve = 0
            if verbose:
                print(f"    [Adam Restart at step {steps}]")

    return best_a, best_score

def spectral_balance(a0, time_limit, verbose=True):
    """Iteratively balance the spectrum to flatten autocorrelation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Work in log-domain for better conditioning
    # a = exp(v), optimize v
    v = np.log(a + 1e-10)
    
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_mom = np.zeros(n)
    t_count = 0
    lr = 0.05
    
    steps = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        
        a = np.exp(v)
        s = a.sum()
        if s < 1e-15:
            break
        a_norm = a * norm / s
        
        b = fast_autocorr(a_norm)
        max_b = b.max()
        
        # Use LSE with moderate temperature
        T = max(0.01, max_b * 0.01)
        w = np.exp((b - max_b) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum
        
        # Gradient in v-space: d LSE / dv_i = (d LSE / da_i) * a_i
        a_rev = a_norm[::-1]
        size_grad = (2*n-1) + n - 1
        fft_size = 1 << (size_grad - 1).bit_length()
        W = rfft(w, fft_size)
        AR = rfft(a_rev, fft_size)
        conv = irfft(W * AR, fft_size)
        grad_a = 2.0 * conv[n - 1:2 * n - 1]
        grad_v = grad_a * a_norm  # Chain rule: da_i/dv_i = a_i
        
        # Project
        grad_v = grad_v - grad_v.mean()
        
        # Adam
        t_count += 1
        m = beta1 * m + (1 - beta1) * grad_v
        v_mom = beta2 * v_mom + (1 - beta2) * grad_v ** 2
        m_hat = m / (1 - beta1 ** t_count)
        v_hat = v_mom / (1 - beta2 ** t_count)
        
        lr_eff = lr * (1.0 - 0.5 * elapsed_frac)
        v = v - lr_eff * m_hat / (np.sqrt(v_hat) + eps)
        
        steps += 1
        
        score = fast_autocorr(a_norm).max()
        if score < best_score:
            best_score = score
            best_a = a_norm.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if verbose and steps % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SpectralAdam step={steps} ({elapsed:.0f}s): best={best_score:.8f}")
        
        if no_improve > 500:
            break
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

def spectral_interpolate(f, new_n):
    """Interpolate using spectral methods (better preserves structure)."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    
    # Pad with zeros, take FFT, zero out high frequencies, IFFT
    fft_size = max(2 * old_n, 2 * new_n)
    fft_size = 1 << (fft_size - 1).bit_length()
    F = rfft(f_norm, fft_size)
    
    # Keep only the first new_n/2 + 1 frequency bins (low-pass)
    n_keep = new_n // 2 + 1
    F_high = np.zeros_like(F)
    F_high[:min(n_keep, len(F))] = F[:min(n_keep, len(F))]
    
    new_f = irfft(F_high, fft_size)[:new_n]
    new_f = np.maximum(new_f, 0.0)
    ss = new_f.sum()
    if ss < 1e-15:
        return interpolate_sequence(f, new_n)
    return new_f / ss

        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1

def build_conv_matrix_from_a(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_conv_matrix_from_a(a)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def lp_best_response_reduced(a, M, solver, n_rounds=5):

    try:

def lse_adam(a0, time_limit, lr=0.5, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0
    b = fast_autocorr(a)
    max_b = b.max()
    T_start = max(0.1, max_b * 0.05)
    T_end = max(0.001, max_b * 0.002)
    steps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** elapsed_frac
        lr_eff = lr * (1.0 - 0.4 * elapsed_frac)
        b = fast_autocorr(a)
        max_b = b.max()
        w = np.exp((b - max_b) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum
        a_rev = a[::-1]
        size_grad = (2 * n - 1) + n - 1
        fft_size = 1 << (size_grad - 1).bit_length()
        W = rfft(w, fft_size)
        AR = rfft(a_rev, fft_size)
        conv = irfft(W * AR, fft_size)
        grad = 2.0 * conv[n - 1:2 * n - 1]
        grad = grad - grad.mean()
        t_count += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t_count)
        v_hat = v / (1 - beta2 ** t_count)
        step = lr_eff * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        a = a_new
        steps += 1
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if verbose and steps % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam step={steps} ({elapsed:.0f}s): T={T:.4f}, lr={lr_eff:.4f}, best={best_score:.8f}")
        if no_improve > 2000:
            rng = np.random.default_rng(steps * 3 + 1)
            scale = 0.02 * a.mean()
            a = np.maximum(a + rng.normal(0, scale, n), 0)
            a = a * norm / a.sum()
            m = np.zeros(n)
            v = np.zeros(n)
            t_count = 0
            no_improve = 0
            if verbose:
                print(f"    [Adam Restart at step {steps}]")
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

def spectral_interpolate(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    fft_size = 1 << (2 * max(old_n, new_n) - 1).bit_length()
    F = rfft(f_norm, fft_size)
    n_keep = new_n // 2 + 1
    F_low = np.zeros_like(F)
    F_low[:min(n_keep, len(F))] = F[:min(n_keep, len(F))]
    new_f = irfft(F_low, fft_size)[:new_n]
    new_f = np.maximum(new_f, 0.0)
    ss = new_f.sum()
    if ss < 1e-15:
        return interpolate_sequence(f, new_n)
    return new_f / ss

            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s}")

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)
    print(f"Starting length: {n_base}")

    # Phase 1: LP Nesterov at base (workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.30, 500.0)
    if t1 > 30:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n_base}")

    # Phase 2: LSE-Adam at base
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LSE-Adam n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 20:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lse_adam(a_norm, t2, lr=0.5, verbose=True)
        do_try_update(a_out, f"adam_n{n_cur}")

    # Phase 3: Log-domain Adam
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LogAdam n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.10, 150.0)
    if t3 > 20:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = log_domain_adam(a_norm, t3, verbose=True)
        do_try_update(a_out, f"logadam_n{n_cur}")

    # Phase 4: LP Nesterov again (benefits from Adam's progress)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LP Nesterov n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 30:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, verbose=True)
        do_try_update(a_out, f"lp2_n{n_cur}")

    # Phase 5: Try longer length
    n2 = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: LSE-Adam n={n2} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.12, 200.0)
    if t5 > 30:
        init_long = spectral_interpolate(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = lse_adam(a_norm, t5, lr=0.5, verbose=True)
        do_try_update(a_out, f"adam_n{n2}")
        # LP polish at new length
        remaining = time_budget - (time.monotonic() - t_global)
        t5b = min(remaining * 0.10, 150.0)
        if t5b > 20:
            n_cur = len(best_seq)
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = nesterov_lp_optimize(a_norm, t5b, solver, verbose=True)
            do_try_update(a_out, f"lp_n{n_cur}")

    # Phase 6: Final alternating polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final alternating polish n={n_best} (budget={remaining:.0f}s) ===")
    if remaining > 60:
        for round_idx in range(3):
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 30:
                break
            tb = remaining / (3 - round_idx)
            n_cur = len(best_seq)
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            if round_idx % 2 == 0:
                a_out, _ = lse_adam(a_norm, min(tb * 0.5, 80.0), lr=0.3, verbose=False)
                do_try_update(a_out, f"final_adam_{round_idx}")
            else:
                a_out, _ = nesterov_lp_optimize(a_norm, min(tb * 0.7, 100.0), solver, verbose=False)
                do_try_update(a_out, f"final_lp_{round_idx}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    return [float(x) for x in best_seq]
