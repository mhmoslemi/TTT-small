            # Convert to the actual score
            score = best_obj  # since 2n * max / (sum)^2 and sum=1, score = 2n*max

def adam_logspace(a0, time_limit, verbose=True):
    """Adam optimizer in log-space with adaptive logsumexp smoothing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    a = a / a.sum()
    x = np.log(a)  # log-space parameterization
    
    best_a = a.copy()
    best_obj = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Adam hyperparameters
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    
    # Adaptive temperature for logsumexp
    beta_temp = 10.0  # Start smooth
    beta_growth = 1.001  # Grow each iteration
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Convert x to a via softmax
        x_max = x.max()
        exp_x = np.exp(x - x_max)
        a = exp_x / exp_x.sum()
        
        ac = fast_autocorr(a)
        
        if t % 100 == 0:
            # Use logsumexp for smooth gradient
            M = ac.max()
            shifted = beta_temp * (ac - M)
            shifted = np.maximum(shifted, -100)  # Clip for stability
            weights = np.exp(shifted)
            weights = weights / weights.sum()
            
            # Gradient of logsumexp(beta * ac) w.r.t. a
            # = beta * sum_k w_k * d(ac_k)/da
            # For each k, dac_k/da[j] = a[k-j] + a[j-k] (valid range)
            g = np.zeros(n)
            # Use top contributing lags (where weights are significant)
            sig_idx = np.where(weights > weights.max() * 1e-3)[0]
            for k in sig_idx:
                wk = weights[k] * beta_temp
                if wk < 1e-10:
                    continue
                js = max(0, int(k) - (n - 1))
                je = min(n - 1, int(k))
                for i in range(js, je + 1):
                    g[i] += wk * a[int(k) - i]
                    j2 = int(k) - i
                    if j2 != i:
                        g[j2] += wk * a[i]
            
            # Convert gradient from a-space to x-space
            # a = softmax(x), so da[i]/dx[j] = a[i](δ_{ij} - a[j])
            # dx gradient = (1/a[i]) * sum_j (da[j]/dx[i]) * g_a[j]
            # = (1/a[i]) * (g_a[i] - a[i] * sum_j g_a[j])
            # = g_a[i]/a[i] - sum_j g_a[j]
            
            g_x = g / (a + 1e-30) - g.sum()
            
        else:
            # Use exact subgradient (faster)
            k_star = int(np.argmax(ac))
            g_a = np.zeros(n)
            js = max(0, k_star - (n - 1))
            je = min(n - 1, k_star)
            for i in range(js, je + 1):
                g_a[i] += a[k_star - i]
                j2 = k_star - i
                if j2 != i:
                    g_a[j2] += a[i]
            g_x = g_a / (a + 1e-30) - g_a.sum()
        
        # Adam update
        t1 = t + 1
        m = beta1 * m + (1 - beta1) * g_x
        v = beta2 * v + (1 - beta2) * g_x ** 2
        m_hat = m / (1 - beta1 ** t1)
        v_hat = v / (1 - beta2 ** t1)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Periodically check and track best
        if t % 200 == 0:
            x_max = x.max()
            exp_x = np.exp(x - x_max)
            a_check = exp_x / exp_x.sum()
            obj_check = 2.0 * n * fast_autocorr(a_check).max()
            if obj_check < best_obj:
                best_obj = obj_check
                best_a = a_check.copy()
        
        # Grow temperature
        if t % 100 == 0:
            beta_temp *= beta_growth
            if beta_temp > 1000:
                beta_temp = 1000
        
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam-Log t={t} ({elapsed:.1f}s): score={best_obj:.6f}, beta={beta_temp:.1f}")
    
    return best_a, best_obj

def coordinate_descent_smart(a0, time_limit, verbose=True):
    """Smart coordinate descent: move mass from peak contributors to quiet positions."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    stall = 0
    iteration = 0
    
    while time.monotonic() - t0 < time_limit and stall < 500:
        iteration += 1
        ac = fast_autocorr(a)
        M = ac.max()
        avg_ac = 1.0 / (2 * n - 1)  # sum of ac = 1 (since sum a = 1)
        
        # Find hot lags (above average by some margin)
        threshold = avg_ac * 1.2
        hot_lags = np.where(ac > threshold)[0]
        
        if len(hot_lags) == 0:
            hot_lags = np.array([int(np.argmax(ac))])
        
        # For each position, compute its "cost" = total contribution to hot lags
        cost = np.zeros(n)
        for k in hot_lags[:20]:  # Limit to top 20 hot lags
            k = int(k)
            excess = ac[k] - avg_ac
            if excess <= 0:
                continue
            js = max(0, k - (n - 1))
            je = min(n - 1, k)
            for i in range(js, je + 1):
                # a[i] * a[k-i] contributes to ac[k]
                # "Cost" of position i is proportional to a[k-i] (marginal contribution)
                cost[i] += excess * a[k - i]
                j2 = k - i
                if j2 != i:
                    cost[j2] += excess * a[i]
        
        # Find source: highest cost position with significant mass
        src_candidates = np.where(a > 1e-6)[0]
        if len(src_candidates) == 0:
            break
        src = src_candidates[np.argmax(cost[src_candidates])]
        
        # Find destination: lowest cost position
        dst_candidates = np.where(np.arange(n) != src)[0]
        if len(dst_candidates) == 0:
            break
        dst = dst_candidates[np.argmin(cost[dst_candidates])]
        
        if src == dst:
            stall += 1
            continue
        
        # Try different amounts to move
        improved = False
        for frac in [0.02, 0.01, 0.005, 0.05, 0.001]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            if a_trial[src] < -1e-15:
                continue
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            ac_trial = fast_autocorr(a_trial)
            M_trial = ac_trial.max()
            obj_trial = 2.0 * n * M_trial
            if obj_trial < best_obj - 1e-12:
                a = a_trial
                best_obj = obj_trial
                best_a = a.copy()
                improved = True
                stall = 0
                break
        
        if not improved:
            stall += 1
            if stall > 50:
                # Random perturbation
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-5:
                    frac = rng.uniform(0.01, 0.1)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    obj_new = 2.0 * n * fast_autocorr(a).max()
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a.copy()
                        stall = 0
        
        if verbose and iteration % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Coord t={iteration} ({elapsed:.1f}s): score={best_obj:.6f}, stall={stall}")
    
    return best_a, best_obj

        k_star = int(np.argmax(ac))

        je = min(n - 1, k_star)
        for i in range(js, je + 1):

            j2 = k_star - i

        g = np.zeros(n)
        for i in range(js, je + 1):
            g[i] += y[k_star - i]

            if j2 != i:
                g[j2] += y[i]

        step = 2.0 / (2.0 + t)

        y_new = proj_simplex(y - step * g, 1.0)

        beta_t = (t - 1.0) / (t + 2.0) if t > 1 else 0.0
        x_new = y_new + beta_t * (y_new - y)
        x_new = proj_simplex(x_new, 1.0)
        
        y = y_new

        for candidate in [y, x]:

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def random_init(n, rng, seed_type='uniform'):
    """Generate random initialization."""
    if seed_type == 'uniform':
        a = rng.random(n) + 0.1
    elif seed_type == 'gaussian':
        x = np.linspace(0, 1, n)
        a = np.exp(-2 * (x - 0.5) ** 2 / (0.1 + 0.3 * rng.random()))
        a += 0.05 * rng.random(n)
    elif seed_type == 'bimodal':
        x = np.linspace(0, 1, n)
        c1 = rng.uniform(0.2, 0.4)
        c2 = 1 - c1
        a = np.exp(-10 * (x - c1) ** 2) + np.exp(-10 * (x - c2) ** 2)
        a += 0.05 * rng.random(n)
    elif seed_type == 'triangular':
        x = np.linspace(0, 1, n)
        a = 1 - np.abs(2 * x - 1)
        a += 0.05 * rng.random(n)
    else:
        a = rng.random(n) + 0.1
    a = np.maximum(a, 0)
    return a / a.sum()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Mirror descent with exact subgradient and Polyak averaging
2. Adam optimizer in log-space with adaptive logsumexp smoothing
3. Vectorized coordinate descent targeting peak lags
4. Nesterov-accelerated projected gradient
5. Multi-length exploration (1600-3200) with interpolation
6. Multiple initialization strategies

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def proj_simplex(v, s):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) + s - cssv > 0
    if not np.any(cond):
        return np.zeros(n)
    rho = u[np.argmax(cond)]
    return np.maximum(v - rho, 0.0)

def compute_subgrad_at_lag(a, k):
    """Compute subgradient of (a*a)[k] w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    js = max(0, k - (n - 1))
    je = min(n - 1, k)
    if js > je:
        return g
    idx = np.arange(js, je + 1)
    other = k - idx
    g[idx] += a[other]
    mask = other != idx
    if np.any(mask):
        g[other[mask]] += a[idx[mask]]
    return g

def compute_weighted_subgrad(a, ac, top_k=10, beta=4.0):
    """Compute weighted subgradient using top-k peaks with power weighting."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 0)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        if wi < 1e-12:
            continue
        k = int(k)
        js = max(0, k - (n - 1))
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        other = k - idx
        grad[idx] += wi * a[other]
        mask = other != idx
        if np.any(mask):
            grad[other[mask]] += wi * a[idx[mask]]
    return grad

def mirror_descent_weighted(a0, time_limit, top_k=5, beta=3.0, verbose=True):
    """Mirror descent with weighted soft subgradient and Polyak averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    avg_a = a.copy()
    n_avg = 1
    gamma = 1.0
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        ac = fast_autocorr(a)
        g = compute_weighted_subgrad(a, ac, top_k=top_k, beta=beta)
        step = gamma / (np.sqrt(t) + 1.0)
        log_ratio = -step * g
        log_ratio -= log_ratio.max()
        a_new = a * np.exp(log_ratio)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new
        obj_new = 2.0 * n * fast_autocorr(a).max()
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a.copy()
        avg_a += a
        n_avg += 1
        if n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check / s_avg
                obj_avg = 2.0 * n * fast_autocorr(avg_check).max()
                if obj_avg < best_obj:
                    best_obj = obj_avg
                    best_a = avg_check.copy()
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD-W t={t} ({elapsed:.1f}s): score={best_obj:.6f}")
    return best_a, best_obj

def adam_logspace(a0, time_limit, verbose=True):
    """Adam optimizer in log-space with smooth max surrogate."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    a = a / a.sum()
    x = np.log(a)
    best_a = a.copy()
    best_obj = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    beta_temp = 5.0
    beta_growth = 1.002
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        x_max = x.max()
        exp_x = np.exp(x - x_max)
        a = exp_x / exp_x.sum()
        ac = fast_autocorr(a)
        if t % 50 == 0:
            M = ac.max()
            shifted = beta_temp * (ac - M)
            shifted = np.clip(shifted, -100, 0)
            weights = np.exp(shifted)
            w_sum = weights.sum()
            if w_sum < 1e-30:
                weights = np.zeros_like(ac)
                weights[int(np.argmax(ac))] = 1.0
            else:
                weights = weights / w_sum
            g_a = np.zeros(n)
            sig_idx = np.where(weights > weights.max() * 1e-4)[0]
            for k in sig_idx[:30]:
                wk = weights[int(k)] * beta_temp
                if wk < 1e-10:
                    continue
                ki = int(k)
                js = max(0, ki - (n - 1))
                je = min(n - 1, ki)
                idx = np.arange(js, je + 1)
                other = ki - idx
                g_a[idx] += wk * a[other]
                mask = other != idx
                if np.any(mask):
                    g_a[other[mask]] += wk * a[idx[mask]]
        else:
            k_star = int(np.argmax(ac))
            g_a = compute_subgrad_at_lag(a, k_star)
        g_x = g_a / (a + 1e-30) - g_a.sum()
        t1 = t + 1
        m = beta1 * m + (1 - beta1) * g_x
        v = beta2 * v + (1 - beta2) * g_x ** 2
        m_hat = m / (1 - beta1 ** t1)
        v_hat = v / (1 - beta2 ** t1)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        if t % 200 == 0:
            x_max = x.max()
            exp_x = np.exp(x - x_max)
            a_check = exp_x / exp_x.sum()
            obj_check = 2.0 * n * fast_autocorr(a_check).max()
            if obj_check < best_obj:
                best_obj = obj_check
                best_a = a_check.copy()
        if t % 100 == 0:
            beta_temp = min(beta_temp * beta_growth, 2000.0)
        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam t={t} ({elapsed:.1f}s): score={best_obj:.6f}, beta={beta_temp:.1f}")
    return best_a, best_obj

def nesterov_proj(a0, time_limit, verbose=True):
    """Nesterov-accelerated projected gradient on simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    y = a.copy()
    x = a.copy()
    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break
        ac = fast_autocorr(y)
        k_star = int(np.argmax(ac))
        g = compute_subgrad_at_lag(y, k_star)
        step = 2.0 / (2.0 + t)
        y_new = proj_simplex(y - step * g, 1.0)
        if t > 1:
            beta_t = (t - 1.0) / (t + 2.0)
        else:
            beta_t = 0.0
        x_new = y_new + beta_t * (y_new - y)
        x_new = proj_simplex(x_new, 1.0)
        y = y_new
        x = x_new
        for candidate in [y, x]:
            obj_c = 2.0 * n * fast_autocorr(candidate).max()
            if obj_c < best_obj:
                best_obj = obj_c
                best_a = candidate.copy()
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Nest t={t} ({elapsed:.1f}s): score={best_obj:.6f}")
    return best_a, best_obj

def coordinate_descent_vec(a0, time_limit, verbose=True):
    """Vectorized coordinate descent: move mass from high-cost to low-cost positions."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_obj = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(9999)
    stall = 0
    iteration = 0
    while time.monotonic() - t0 < time_limit and stall < 300:
        iteration += 1
        ac = fast_autocorr(a)
        avg_ac = 1.0 / (2 * n - 1)
        threshold = avg_ac * 1.1
        hot_lags = np.where(ac > threshold)[0]
        if len(hot_lags) == 0:
            hot_lags = np.array([int(np.argmax(ac))])
        top_hot = hot_lags[np.argsort(ac[hot_lags])[::-1]][:15]
        cost = np.zeros(n)
        for k in top_hot:
            ki = int(k)
            excess = ac[ki] - avg_ac
            if excess <= 0:
                continue
            js = max(0, ki - (n - 1))
            je = min(n - 1, ki)
            idx = np.arange(js, je + 1)
            other = ki - idx
            contrib = a[idx] * a[other]
            cost[idx] += excess * a[other]
            mask = other != idx
            if np.any(mask):
                cost[other[mask]] += excess * a[idx[mask]]
        src_candidates = np.where(a > 1e-7)[0]
        if len(src_candidates) == 0:
            break
        src = src_candidates[np.argmax(cost[src_candidates])]
        dst_mask = np.ones(n, dtype=bool)
        dst_mask[src] = False
        dst = np.where(dst_mask)[0][np.argmin(cost[dst_mask])]
        if src == dst:
            stall += 1
            continue
        improved = False
        for frac in [0.01, 0.005, 0.02, 0.002, 0.05]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            if a_trial[src] < -1e-15:
                continue
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            obj_trial = 2.0 * n * fast_autocorr(a_trial).max()
            if obj_trial < best_obj - 1e-12:
                a = a_trial
                best_obj = obj_trial
                best_a = a.copy()
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
            if stall > 30:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-5:
                    amt = a[i] * rng.uniform(0.02, 0.1)
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0)
                    s2 = a.sum()
                    if s2 > 1e-15:
                        a = a / s2
                        obj_new = 2.0 * n * fast_autocorr(a).max()
                        if obj_new < best_obj:
                            best_obj = obj_new
                            best_a = a.copy()
                            stall = 0
        if verbose and iteration % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Coord it={iteration} ({elapsed:.1f}s): score={best_obj:.6f}, stall={stall}")
    return best_a, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def random_init(n, rng, seed_type):
    """Generate structured random initialization."""
    if seed_type == 'uniform':
        a = rng.random(n) + 0.1
    elif seed_type == 'gaussian':
        x = np.linspace(0, 1, n)
        sigma = 0.15 + 0.2 * rng.random()
        a = np.exp(-2.0 * (x - 0.5) ** 2 / sigma)
        a += 0.05 * rng.random(n)
    elif seed_type == 'bimodal':
        x = np.linspace(0, 1, n)
        c1 = rng.uniform(0.25, 0.4)
        c2 = 1.0 - c1
        a = np.exp(-15.0 * (x - c1) ** 2) + np.exp(-15.0 * (x - c2) ** 2)
        a += 0.05 * rng.random(n)
    elif seed_type == 'flat_noise':
        a = np.ones(n) * 0.5 + 0.5 * rng.random(n)
    else:
        a = rng.random(n) + 0.1
    a = np.maximum(a, 0)
    return a / a.sum()

def propose_candidate(budget_s=1800.0):
    """
    Main search: multi-method optimization at multiple lengths.
    Uses mirror descent (exact + weighted), Adam in log-space,
    Nesterov projected gradient, and vectorized coordinate descent.
    Explores lengths 1600-3200 with interpolation.
    """
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    print(f"Time budget: {time_budget:.0f}s")

    # Track best globally
    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent / parent.sum()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0, 1, n_base)
        best_seq = (1.0 - np.abs(2.0 * x - 1.0)) ** 0.7
        best_seq = np.maximum(best_seq, 1e-6)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    # Helper to check and update best (inline, no closure)
    # We'll just compute score and compare directly

    # === Phase 1: Optimize at parent length with multiple methods ===
    n0 = len(best_seq)

    remaining = time_budget - (time.monotonic() - t_global)
    t_alloc = min(remaining * 0.10, 150.0)
    if t_alloc > 10:
        print(f"\n=== Phase 1a: MD-Weighted n={n0} (budget {t_alloc:.0f}s) ===")
        a_out, _ = mirror_descent_weighted(best_seq, t_alloc, top_k=5, beta=3.0, verbose=True)
        sc = evaluate_safe(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  NEW BEST: {best_score:.6f}")

    remaining = time_budget - (time.monotonic() - t_global)
    t_alloc = min(remaining * 0.10, 150.0)
    if t_alloc > 10:
        print(f"\n=== Phase 1b: MD-Exact n={n0} (budget {t_alloc:.0f}s) ===")
        a_out, _ = mirror_descent_exact(best_seq, t_alloc, verbose=True)
        sc = evaluate_safe(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  NEW BEST: {best_score:.6f}")

    remaining = time_budget - (time.monotonic() - t_global)
    t_alloc = min(remaining * 0.10, 150.0)
    if t_alloc > 10:
        print(f"\n=== Phase 1c: Adam-Log n={n0} (budget {t_alloc:.0f}s) ===")
        a_out, _ = adam_logspace(best_seq, t_alloc, verbose=True)
        sc = evaluate_safe(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  NEW BEST: {best_score:.6f}")

    remaining = time_budget - (time.monotonic() - t_global)
    t_alloc = min(remaining * 0.08, 120.0)

    n_best = len(best_seq)

    if remaining > 30:
        t_alloc = min(remaining * 0.25, 200.0)
        if t_alloc > 10:
            print(f"\n=== Phase 5b: Final MD-W n={n_best} (budget {t_alloc:.0f}s) ===")
            a_out, _ = mirror_descent_weighted(best_seq, t_alloc, top_k=3, beta=2.0, verbose=True)
            sc = evaluate_safe(a_out)
            if sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  NEW BEST: {best_score:.6f}")

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)

    if remaining > 30:
        t_alloc = min(remaining * 0.25, 200.0)
        if t_alloc > 10:
            print(f"\n=== Phase 5c: Final Adam n={n_best} (budget {t_alloc:.0f}s) ===")
            a_out, _ = adam_logspace(best_seq, t_alloc, verbose=True)
            sc = evaluate_safe(a_out)
            if sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  NEW BEST: {best_score:.6f}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
