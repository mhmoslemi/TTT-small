        w /= w_sum
    
    fft_size = 1 << (3 * n - 2).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    A_fwd = rfft(a, fft_size)
    W = rfft(w, fft_size)
    
    conv1 = irfft(A_rev * W, fft_size)
    conv2 = irfft(A_fwd * W, fft_size)
    
    r1 = conv1[n - 1: n - 1 + n]
    r2 = conv2[:n]
    
    return r1 + r2

def lse_value(a, T):
    """Compute LSE surrogate value."""
    b = fast_autocorr(a)
    max_b = b.max()
    log_sum = max_b / T + np.log(np.sum(np.exp((b - max_b) / T)))
    return T * log_sum

def nesterov_lse(a0, time_limit, T_start=20.0, T_end=0.5, n_stages=15, verbose=True):
    """Nesterov-accelerated LSE gradient descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    y_prev = a.copy()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    it_count = 0
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        
        n_iters_stage = max(20, int(remaining * 2.5 / max(1, n_stages - si)))
        for it in range(n_iters_stage):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Nesterov extrapolation
            if it == 0:
                y_ext = y
            else:
                momentum = (np.sqrt(1 + 4 * (it-1)**2) + (it-1)) / (2*(it+1))
                y_ext = y + momentum * (y - y_prev)
                y_ext = np.maximum(y_ext, 0)
                s_ext = y_ext.sum()
                if s_ext > 1e-15:
                    y_ext = y_ext * norm / s_ext
                else:
                    y_ext = y.copy()
            
            grad = lse_gradient(y_ext, T)
            lr = norm / (T * (it + 10))
            
            y_new = y - lr * grad
            y_new = np.maximum(y_new, 0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y
            
            y_prev = y.copy()
            y = y_new
            it_count += 1
            
            score = fast_autocorr(y).max()
            if score < best_score:
                best_score = score
                best_a = y.copy()
            
            if it_count % 100 == 0:
                elapsed = time.monotonic() - t0
                if verbose:
                    print(f"    LSE-N stage={si+1}/{n_stages} T={T:.2f} it={it_count} ({elapsed:.0f}s): best={best_score:.8f}")

def log_space_descent(a0, time_limit, verbose=True):

        grad_a = lse_gradient(a_curr, T=1.0)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

def random_perturbation(a, seed, scale=0.05):

    pert = rng.normal(0, scale, size=n)

    if s < 1e-15:

    return a_new * norm / s

def fast_autocorr(a):

    """Compute gradient of LSE surrogate: T*log(sum exp(b_k/T))."""

    score_history = []
    stale_count = 0

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        n_iters_stage = max(30, int(remaining * 3.0 / max(1, n_stages - si)))
        for it in range(n_iters_stage):
            if time.monotonic() - t0 > time_limit:
                break

            # Nesterov extrapolation with standard momentum
            if it == 0:
                beta = 0.0
            else:
                beta = (1.0 + np.sqrt(1.0 + 4.0 * (it - 1) ** 2)) / (2.0 * (it + 1.0))
                beta = min(beta, 0.999)

            if beta > 0:
                y_ext = y + beta * (y - y_prev)
                y_ext = np.maximum(y_ext, 0.0)
                s_ext = y_ext.sum()
                if s_ext < 1e-15:
                    y_ext = y.copy()
                else:
                    y_ext = y_ext * norm / s_ext
            else:
                y_ext = y.copy()

            grad = lse_gradient(y_ext, T)

            # Learning rate: adapt based on gradient magnitude
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                lr = 0.0
            else:
                lr = norm / (T * np.sqrt(it + 10.0))
                lr = min(lr, 0.5 * norm / grad_norm)

            y_new = y - lr * grad
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y

            y_prev = y.copy()
            y = y_new
            it_count += 1

            score = fast_autocorr(y).max()
            if score < best_score:
                best_score = score
                best_a = y.copy()
                stale_count = 0
            else:
                stale_count += 1

            # Adaptive: if stuck, reduce temperature faster
            if stale_count > 100 and si < n_stages - 1:
                T = max(T * 0.7, T_end)
                stale_count = 0

            if it_count % 150 == 0:
                elapsed = time.monotonic() - t0
                if verbose:
                    print(f"    LSE-Adv stage={si+1}/{n_stages} T={T:.2f} it={it_count} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def log_space_descent(a0, time_limit, verbose=True):
    """Optimize in log-space with adaptive step size."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15).copy()
    s0 = a.sum()
    a = a * norm / s0

    x = np.log(a)
    x_mean_centered = x - x.mean()
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    T = 5.0

    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break

        a_curr = np.exp(x - x.max())
        a_curr = a_curr * norm / a_curr.sum()

        grad_a = lse_gradient(a_curr, T)
        grad_x = a_curr * grad_a
        grad_x -= np.mean(grad_x)

        grad_norm = np.linalg.norm(grad_x)
        if grad_norm < 1e-15:
            T *= 0.8
            continue

        lr = 0.5 * norm / grad_norm / np.sqrt(it + 1.0)
        lr = min(lr, 1.0)

        x_new = x - lr * grad_x
        a_new = np.exp(x_new - x_new.max())
        a_new = a_new * norm / a_new.sum()

        score = fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            x = x_new
        else:
            T *= 0.95

        if (it + 1) % 300 == 0:
            elapsed = time.monotonic() - t0
            if verbose:
                print(f"    LogSpace it={it+1} T={T:.2f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def analytical_gradient_descent(a0, time_limit, verbose=True):
    """
    Use the analytical gradient of 2*n*max(c)/S^2.
    When max is at a unique position k*, the subgradient w.r.t. a_i is:
    2*n*(2*c_kstar_i * S - max_c * 2*S) / S^4 ... 
    Actually let's use the LSE with very small T as approximation to true max.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    T = 0.5

    for it in range(50000):
        if time.monotonic() - t0 > time_limit:
            break

        # Use very small T to approximate the true max gradient
        grad = lse_gradient(a, T)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Project gradient to tangent space of {sum(a)=norm, a>=0}
        # Simple: just use gradient descent with projection
        lr = norm / (grad_norm * np.sqrt(it + 1))
        lr = min(lr, 0.3)

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        score = fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            T = max(T * 0.98, 0.05)
        else:
            T = max(T * 0.9, 0.05)

        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            if verbose:
                print(f"    AnalyticGD it={it+1} T={T:.3f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, verbose=True):
    """Coordinate descent cycling through all coordinates."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    # For each coordinate i, try a few multipliers
    multipliers = [0.5, 0.8, 0.9, 1.1, 1.2, 1.5]

    for epoch in range(500):
        if time.monotonic() - t0 > time_limit:
            break

        improved = False
        for i in range(0, n, max(1, n // 400)):  # Subsample if n is large
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[i]
            for mult in multipliers:
                a_trial = a.copy()
                a_trial[i] = old_val * mult
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            # Try with more coordinates
            if epoch % 10 == 0:
                for i in range(n):
                    if time.monotonic() - t0 > time_limit:
                        break
                    old_val = a[i]
                    for mult in [0.7, 1.3, 0.5, 2.0]:
                        a_trial = a.copy()
                        a_trial[i] = old_val * mult
                        s_t = a_trial.sum()
                        if s_t < 1e-15 or s_t > 1e10:
                            continue
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
            if not improved:
                break

        if verbose and (epoch + 1) % 20 == 0:
            print(f"    CoordDesc epoch={epoch+1} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def peak_targeting_search(a0, time_limit, seed=42, verbose=True):
    """Targeted local search focusing on reducing correlation peaks."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    it_count = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_ks = np.argsort(b)[::-1][:8]

        improved = False
        for k_star in top_ks:
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k_star - idx]
            top_idx = idx[np.argsort(contrib)[::-1][:5]]

            for src in top_idx:
                if a[src] < norm * 1e-9:
                    continue

                # Find low-contribution positions
                peak_mask = np.zeros(n, dtype=bool)
                for k2 in top_ks[:4]:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    if je2 >= js2:
                        peak_mask[js2:je2 + 1] = True

                candidates = np.where(~peak_mask)[0]
                if len(candidates) == 0:
                    candidates = np.where(np.arange(n) != src)[0]
                candidates = candidates[np.argsort(a[candidates])[:max(5, len(candidates) // 4)]]

                for dst in candidates[:5]:
                    for frac in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
                        amt = a[src] * frac
                        if amt < 1e-16:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < -1e-16:
                            continue
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-13:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            it_count += 1
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape
            if rng.integers(15) == 0:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > norm * 1e-8:
                    frac = rng.uniform(0.01, 0.1)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score + 0.01:  # Accept slightly worse
                            a = a_trial
                            stall = 0
                        if score < best_score:
                            best_score = score
                            best_a = a.copy()

            if stall > 300:
                break

        if verbose and it_count % 1000 == 0:
            print(f"    PeakSearch it={it_count} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def multi_scale_optimize(a_init, time_limit, n_targets, verbose=True):
    """Optimize at multiple scales: start small, grow, refine."""
    t0 = time.monotonic()
    best_a = a_init.copy()
    best_score = evaluate_safe(a_init)

    for target_n in sorted(n_targets):
        if time.monotonic() - t0 > time_limit * 0.85:
            break
        remaining = time_limit * 0.85 - (time.monotonic() - t0)
        if remaining < 10:
            break

        if len(best_a) != target_n:
            current = interpolate_sequence(best_a, target_n)
        else:
            current = best_a.copy()

        t_sub = min(remaining * 0.8, 200.0)
        a_out, _ = nesterov_lse_adaptive(current, t_sub, T_start=10.0, T_end=0.3,
                                          n_stages=10, verbose=verbose)
        sc = evaluate_safe(a_out)
        if sc < best_score:
            best_score = sc
            best_a = a_out.copy()
            elapsed = time.monotonic() - t0
            if verbose:
                print(f"    MultiScale n={target_n} ({elapsed:.0f}s): score={sc:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Combines multiple optimization strategies:
    - Nesterov LSE with adaptive temperature
    - Log-space optimization
    - Analytical gradient descent (small T)
    - Coordinate descent
    - Peak-targeting local search
    - Multi-scale approach
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    best_seq = None
    best_score = float('inf')

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize from parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init (uniform): n={n_base}, score={best_score:.8f}")
    else:
        n_base = len(best_seq)

    # Try uniform as baseline
    uniform = np.ones(n_base) / n_base
    u_score = evaluate_safe(uniform)
    print(f"Uniform baseline: n={n_base}, score={u_score:.8f}")
    if u_score < best_score:
        best_score = u_score
        best_seq = uniform.copy()
        print("  Uniform is better, using as starting point.")

    # ============ Phase 1: LSE Nesterov adaptive ============
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LSE-Nesterov-Adaptive n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.22, 380.0)
    if t1 > 30:
        a_out, _ = nesterov_lse_adaptive(best_seq, t1, T_start=15.0, T_end=0.3,
                                         n_stages=16, verbose=True)
        do_try_update(a_out, f"lse_adv_n{n_base}")

    # ============ Phase 2: Analytical gradient (small T) ============
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: AnalyticGD n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.10, 160.0)
    if t2 > 10:
        a_out, _ = analytical_gradient_descent(best_seq, t2, verbose=True)
        do_try_update(a_out, f"analytic_n{len(best_seq)}")

    # ============ Phase 3: Log-space ============
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LogSpace n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.08, 120.0)
    if t3 > 10:
        a_out, _ = log_space_descent(best_seq, t3, verbose=True)
        do_try_update(a_out, f"log_n{len(best_seq)}")

    # ============ Phase 4: Multi-scale ============
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: MultiScale (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.18, 300.0)
    if t4 > 30:
        n_targets = [len(best_seq), len(best_seq) + 400, len(best_seq) + 800, len(best_seq) + 1400]
        a_out, _ = multi_scale_optimize(best_seq, t4, n_targets, verbose=True)
        do_try_update(a_out, "multiscale")

    # ============ Phase 5: Peak targeting ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: PeakTarget n={n_best} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 10:
        a_out, _ = peak_targeting_search(best_seq, t5, seed=7777, verbose=True)
        do_try_update(a_out, f"peak_n{n_best}")

    # ============ Phase 6: Coordinate descent ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: CoordDesc n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.08, 100.0)
    if t6 > 5:
        a_out, _ = coordinate_descent(best_seq, t6, verbose=True)
        do_try_update(a_out, f"coord_n{n_best}")

    # ============ Phase 7: Random restarts ============
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Random restarts (budget={remaining:.0f}s) ===")
    n_restarts = 4
    t_per = remaining / (n_restarts + 1)
    for r in range(n_restarts):
        if time.monotonic() - t_global > time_budget - 30:
            break
        t_r = min(t_per, 50.0)
        if t_r < 5:
            break
        # Create perturbed version
        rng_r = np.random.default_rng(seed + r * 137 + 1)
        n_r = len(best_seq)
        norm_r = np.sqrt(2.0 * n_r)
        a_base = best_seq * norm_r / best_seq.sum()
        pert = rng_r.normal(0, 0.08, size=n_r)
        a_init = np.maximum(a_base + pert, 0)
        s_init = a_init.sum()
        if s_init > 1e-15:
            a_init = a_init * norm_r / s_init
        a_out, _ = nesterov_lse_adaptive(a_init, t_r, T_start=8.0, T_end=0.5,
                                         n_stages=8, verbose=False)
        do_try_update(a_out, f"restart{r+1}")

    # ============ Phase 8: Final LSE polish ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final polish n={n_best} (budget={remaining:.0f}s) ===")
    t8 = min(remaining * 0.25, 120.0)
    if t8 > 5:
        a_out, _ = nesterov_lse_adaptive(best_seq, t8, T_start=5.0, T_end=0.1,
                                         n_stages=10, verbose=True)
        do_try_update(a_out, "final_lse")

    # ============ Phase 9: Final peak targeting ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 10:
        t9 = min(remaining * 0.5, 80.0)
        if t9 > 5:
            print(f"\n=== Phase 9: Final peak target n={n_best} (budget={t9:.0f}s) ===")
            a_out, _ = peak_targeting_search(best_seq, t9, seed=99991, verbose=True)
            do_try_update(a_out, "final_peak")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
