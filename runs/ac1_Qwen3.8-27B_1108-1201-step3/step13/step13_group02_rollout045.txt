    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def build_conv_matrix_subset(a, indices):
    """Build convolution matrix only for specified row indices."""
    n = len(a)
    T = np.zeros((len(indices), n))
    for row_idx, k in enumerate(indices):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            T[row_idx, j] = a[k - j]
    return T

def lp_best_response_topk(a, M, solver, top_k=100):
    """LP best response using only top-K constraints."""
    n = len(a)
    size = 2 * n - 1
    b = fast_autocorr(a)
    # Get top K lag indices
    top_idx = np.argsort(b)[::-1][:top_k]
    
    # Build constraint matrix for top K
    T = build_conv_matrix_subset(a, top_idx)
    M_vals = M  # All constraints have the same bound M
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M_vals]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    
    # Verify feasibility with ALL constraints
    full_cross = fast_cross_corr(a, b_val)
    if full_cross.max() > M * (1 + 1e-4):
        # Infeasible - try with more constraints
        top_idx2 = np.argsort(b)[::-1][:min(size, top_k * 3)]
        T2 = build_conv_matrix_subset(a, top_idx2)
        b_var2 = cp.Variable(n, nonneg=True)
        constraints2 = [T2 @ b_var2 <= M_vals]
        prob2 = cp.Problem(cp.Maximize(cp.sum(b_var2)), constraints2)
        try:
            prob2.solve(solver=solver, verbose=False)
        except Exception:
            return None
        if prob2.status not in ('optimal', 'optimal_inaccurate'):
            return None
        b_val = np.maximum(np.asarray(b_var2.value, dtype=np.float64), 0.0)
        full_cross2 = fast_cross_corr(a, b_val)
        if full_cross2.max() > M * (1 + 1e-3):
            return None
    
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def fast_cross_corr(a, b):
    """Compute cross-correlation a * b (full)."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def nesterov_lp_fast(a0, time_limit, solver, verbose=True):
    """LP Nesterov with top-K constraint reduction and dual tracking."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    # Adaptive top-K
    top_k = 50
    top_k_max = min(2 * n - 1, 500)
    
    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        
        # Adaptive top-K: increase if we're making slow progress
        if n_iters % 200 == 0 and n_iters > 0:
            top_k = min(top_k_max, top_k + 25)
        
        b = lp_best_response_topk(y, M_curr, solver, top_k=min(top_k, 200))
        if b is None:
            top_k = min(top_k_max, top_k + 50)
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
            # Decrease top_k if improving (can use fewer constraints)
            top_k = max(30, top_k - 10)
        else:
            stall += 1
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        
        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, top_k={top_k}, stall={stall}")
        
        # Adaptive restart
        if stall > 300:
            rng = np.random.default_rng(n_iters * 13 + 7)
            # Structured perturbation: perturb positions that contribute to peaks
            b_full = fast_autocorr(y)
            peak_idx = np.argsort(b_full)[::-1][:20]
            # Get positions that contribute to peaks
            contrib = set()
            for k in peak_idx:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                contrib.update(range(js, je + 1))
            contrib = list(contrib[:n // 4]) if len(contrib) > n // 4 else list(contrib)
            if contrib:
                perturb_scale = 0.03 * y.mean()
                for idx in contrib:
                    y[idx] = max(0, y[idx] + rng.normal(0, perturb_scale))
                s_y = y.sum()
                if s_y > 1e-15:
                    y = y * norm / s_y
            stall = 0
            top_k = min(top_k_max, top_k + 50)
    
    return best_a, best_score

def fista_lse(a0, time_limit, verbose=True):
    """FISTA (Nesterov accelerated gradient) on log-sum-exp approximation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0
    
    # FISTA parameters
    k_counter = 1.0
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        
        # Anneal temperature
        T_start = max(0.5, best_score * 0.05)
        T_end = max(0.001, best_score * 0.001)
        T = T_start * (T_end / T_start) ** elapsed_frac
        
        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break
        
        # Sharp edge detection: identify near-maximal peaks
        threshold = max_b * 0.99
        active_mask = b >= threshold
        n_active = active_mask.sum()
        
        # Use only active peaks for gradient (when sharp edge is small)
        if n_active < 50:
            b_eff = np.where(active_mask, b, -np.inf)
        else:
            b_eff = b
        
        # LSE gradient on active peaks
        w = np.exp((b_eff - max_b) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum
        
        # Compute gradient using FFT
        n = len(a)
        a_rev = a[::-1]
        fft_size = 1 << (3 * n - 2).bit_length()
        W = rfft(w, fft_size)
        AR = rfft(a_rev, fft_size)
        conv = irfft(W * AR, fft_size)
        grad = 2.0 * conv[n - 1:2 * n - 1]
        
        # FISTA step with adaptive step size
        # Lipschitz constant estimate
        L = 2.0 * max_b * n_active / (T * T)  # Rough estimate
        step = 1.0 / max(L, 1e-10)
        step = min(step, 1.0 / (n * 0.1))  # Cap step size
        
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        # FISTA momentum
        k_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * k_counter * k_counter))
        a_momentum = a_new + ((k_counter - 1.0) / k_new) * (a_new - a)
        a_momentum = np.maximum(a_momentum, 0.0)
        s_m = a_momentum.sum()
        if s_m > 1e-15:
            a_momentum = a_momentum * norm / s_m
        
        # Backtracking line search (simplified)
        score_new = fast_autocorr(a_new).max()
        if score_new > max_b * 1.01:
            # Step was too large, reduce
            step *= 0.5
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new * norm / s_new
            score_new = fast_autocorr(a_new).max()
        
        a = a_new
        k_counter = k_new
        steps += 1
        
        if score_new < best_score:
            best_score = score_new
            best_a = a.copy()
        
        # Also check momentum point
        score_mom = fast_autocorr(a_momentum).max()
        if score_mom < best_score:
            best_score = score_mom
            best_a = a_momentum.copy()
            a = a_momentum.copy()
        
        if verbose and steps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA step={steps} ({elapsed:.0f}s): T={T:.4f}, n_active={n_active}, best={best_score:.8f}")
    
    return best_a, best_score

def shattering_polish(a0, time_limit, seed=42, verbose=True):
    """Targeted shattering: solve small local optimizations to reduce top peaks."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        
        # Find top peaks
        top_k = np.argsort(b)[::-1][:5]
        
        improved = False
        for k_star in top_k:
            if b[k_star] < max_b * 0.95:
                break
            
            # Find positions contributing to this peak
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_star - idx]
            
            if contrib.max() < 1e-18:
                continue
            
            # Find the dominant contributor
            src = idx[np.argmax(contrib)]
            partner = k_star - src
            
            # Find good destinations (low autocorrelation impact)
            # Compute "impact" of each position: how much it contributes to top peaks
            impact = np.zeros(n)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                impact[js2:je2 + 1] += a[js2:je2 + 1] * a[k2 - np.arange(js2, je2 + 1)]
            
            # Best destinations are those with low impact and low value
            score_dst = impact + 0.1 * a
            # Exclude positions near the source
            exclude = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                exclude.update(range(max(0, src - 5), min(n, src + 6)))
            
            candidates = [i for i in range(n) if i not in exclude and a[i] < a[src] * 0.5]
            if not candidates:
                candidates = [i for i in range(n) if i != src]
            
            # Pick the best candidate
            candidates = np.array(candidates)
            cand_scores = score_dst[candidates]
            best_dst = candidates[np.argmin(cand_scores)]
            
            # Try transferring mass
            for frac in [0.01, 0.03, 0.05, 0.1]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = fast_autocorr(a_trial).max()
                
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    n_improve += 1
                    break
                elif b[k_star] * (1 - frac) < score < b[k_star]:
                    # Partial improvement on this peak
                    a = a_trial
                    improved = True
                    break
        
        if not improved:
            # Try random pair swaps
            for _ in range(10):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j:
                    continue
                amt = min(a[i], a[j]) * rng.uniform(0.05, 0.2)
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    n_improve += 5
                    break
        
        if verbose and n_improve % 500 == 0 and n_improve > 0:
            elapsed = time.monotonic() - t0
            print(f"    Shatter improve={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")
        
        # Early stop if no improvement for a long time
        if n_improve > 0 and (time.monotonic() - t0) > time_limit * 0.5:
            # Check if we're stuck
            b_check = fast_autocorr(a)
            if b_check.max() > best_score * 0.999:
                break
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss
