    n = len(a)

    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

    return irfft(A * B, fft_size)[:size]

    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def project_simplex(v, target_sum=1.0):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and abs(v.sum() - target_sum) < 1e-12:
        return v
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho = np.nonzero(v_sorted * np.arange(1, n + 1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-15:
        idx = np.argmax(result)
        result[idx] += diff
        result = np.maximum(result, 0)
        result = result / result.sum() * target_sum
    return result

def fast_autocorr(a):
    """Full autocorrelation a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_conv(a, b):
    """Compute cross-correlation/convolution a*b using FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def evaluate_safe(arr):
    """Safely evaluate a sequence via the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def internal_score(a):
    """Compute 2*n*max_b assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()


def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if n == 0:
        return v
    if np.all(v >= 0) and abs(v.sum() - target_sum) < 1e-13:
        return v
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho_mask = v_sorted * np.arange(1, n + 1) > (cssv - target_sum)
    rho_idx = np.nonzero(rho_mask)[0]
    if len(rho_idx) == 0:
        return np.full(n, target_sum / n)
    rho = rho_idx[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-15:
        idx = np.argmax(result)
        result[idx] += diff
        result = np.maximum(result, 0)
        s = result.sum()
        if s > 1e-15:
            result = result / s * target_sum
    return result


def spectral_shaping(a, n_iter=300, alpha=0.3):
    """Iteratively flatten power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = internal_score(a)

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 50.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a

def solve_lp_best_response(a_fixed, solver, warm_start=None, max_iter=100):
    """
    Solve: min M s.t. (T_a b)_k <= M for all k, sum(b)=1, b>=0.
    With warm-start support.
    """
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    # Build constraints using the Toeplitz/convolution structure
    # (T_a @ b)_k = sum_j a_fixed[k-j] * b_j for valid j
    # This equals cross_conv(a_fixed, b)_k
    # In cvxpy, we express this as: for each k, sum over j of a_fixed[k-j]*b_j <= M

    # Efficient: use the fact that (a*b)_k = sum_j a_{k-j} b_j
    # We can write this as a matrix multiplication with a Toeplitz matrix
    # But for cvxpy, let's use the expression directly
    # (a*b) = cp.convolve doesn't exist, so we build the matrix
    
    # Actually, let's use a different formulation:
    # (T_a b)_k = sum_j T_a[k,j] b_j where T_a[k,j] = a[k-j] if valid
    # We can express (T_a b) as the first 2n-1 elements of convolve(a, b)
    # In cvxpy: we need linear expressions
    
    # Build the Toeplitz matrix explicitly (for n up to ~3000 this is fine)
    # T has shape (2n-1, n)
    # T[k, j] = a[k-j] if 0 <= k-j < n else 0
    
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed

    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    
    # Warm start
    if warm_start is not None:
        b_var.value = warm_start
        M_val_est = (T @ warm_start).max()
        M_var.value = M_val_est + 1e-6

    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, max_iter=max_iter)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iter=200)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=300)
            except Exception:
                return None, None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None, None
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, warm_start_b=None, verbose=True):
    """Nesterov-accelerated LP best-response for the zero-sum game."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0
    stall_count = 0
    stall_threshold = 3000

    # Build T matrix once (it changes with a, so rebuild periodically)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a

    prev_b = warm_start_b

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Solve LP best response
        b_star, M_val = solve_lp_best_response(y, solver, warm_start=prev_b)
        if b_star is None:
            continue
        prev_b = b_star

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = project_simplex(a_new)

        y = y_new
        a = a_new

        # Check score every few iterations (autocorr is expensive)
        if t % 3 == 0 or t < 10:
            score_a = internal_score(a)
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()
                stall_count = 0
            else:
                stall_count += 3

            score_y = internal_score(y)
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()
                stall_count = 0

        # Time-averaged iterate
        avg_a += a
        n_avg += 1
        if n_avg % 500 == 0:
            avg_check = avg_a / n_avg
            score_avg = internal_score(avg_check)
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
                stall_count = 0

        if stall_count > stall_threshold:
            break

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}")

    return best_a, best_score


def peak_shaving_advanced(a0, time_limit, seed=42):
    """Advanced peak shaving with contribution analysis and multiple strategies."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        peak_val = b.max()
        peak_indices = np.where(b > peak_val * 0.999)[0]

        improved = False
        for k in peak_indices[:20]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            total_contrib = contrib.sum()
            if total_contrib < 1e-15:
                continue

            # Find the pair with largest contribution
            src_local = np.argmax(contrib)
            src = idx[src_local]
            partner = k - src

            # Find destination: coordinate far from all peak areas
            peak_areas = set()
            for k2 in peak_indices[:10]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                for j2 in range(js2, je2 + 1, max(1, n // 100)):
                    peak_areas.add(j2)

            candidates = [j for j in range(n) if j not in peak_areas and j != src and j != partner]
            if not candidates:
                candidates = [j for j in range(n) if j != src]

            # Pick destination with lowest "peak involvement"
            candidate_scores = np.array([a[j] for j in candidates])
            # Prefer low-a destinations
            sorted_idx = np.argsort(candidate_scores)[:max(1, len(candidates) // 5)]
            dst = candidates[sorted_idx[rng.integers(len(sorted_idx))]]

            # Try moving mass from src (or partner, whichever is larger)
            for source in [src, partner]:
                if source < 0 or source >= n:
                    continue
                for frac in [0.001, 0.003, 0.008, 0.015, 0.03, 0.05]:
                    amt = a[source] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[source] -= amt
                    a_trial[dst] += amt
                    if a_trial[source] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    score = internal_score(a_trial)
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Random perturbation to escape
            stall += 1
            for _ in range(5):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.02)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        a_trial = a_trial / a_trial.sum()
                        score = internal_score(a_trial)
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
                            break
            if stall > 300:
                break

    return best_a, best_score


def mirror_descent_optimize(a0, time_limit, verbose=True):
    """
    Mirror descent (multiplicative weights) for the min-max problem.
    Uses softmax-weighted subgradient with Nesterov acceleration.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    # Work in log-space: a = softmax(z)
    z = np.log(np.maximum(a, 1e-30))
    z_y = z.copy()  # Nesterov ghost

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t_adam = 0
    t = 0
    stall = 0

    fft_size = 1 << ((3*n-1) - 1).bit_length()

    while time.monotonic() - t0 < time_limit:
        t += 1
        a_curr = np.exp(z - z.max())
        a_curr /= a_curr.sum()

        b = fast_autocorr(a_curr)
        max_b = b.max()

        # Softmax weights for smoothing
        T_smooth = max(1.0, 100.0 * np.exp(-t / 5000.0))
        Tb = T_smooth * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()

        # Gradient: grad_j = 2 * sum_k w_k * a_{k-j}
        A_rev = rfft(a_curr[::-1].copy(), fft_size)
        W = rfft(w, fft_size)
        conv = irfft(A_rev * W, fft_size)
        grad = 2.0 * conv[n - 1:n - 1 + n]

        # Adam update in z-space
        t_adam += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t_adam)
        v_hat = v / (1 - beta2 ** t_adam)

        lr_t = 0.5 / np.sqrt(1 + 0.001 * t)
        z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        # Center z to prevent overflow
        z = z - z.mean()

        # Nesterov
        alpha_t = 2.0 / (t + 2.0)
        z_y = (1 - alpha_t) * z_y + alpha_t * z

        # Check score periodically
        if t % 50 == 0:
            a_check = np.exp(z_y - z_y.max())
            a_check /= a_check.sum()
            score = internal_score(a_check)
            if score < best_score:
                best_score = score
                best_a = a_check.copy()
                stall = 0
            else:
                stall += 50

            a_check2 = np.exp(z - z.max())
            a_check2 /= a_check2.sum()
            score2 = internal_score(a_check2)
            if score2 < best_score:
                best_score = score2
                best_a = a_check2.copy()
                stall = 0

            if verbose and t % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"      MirrorDesc t={t} ({elapsed:.0f}s): best={best_score:.6f}")

        if stall > 5000:
            break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def random_spectral_init(n, seed=0, n_trials=20):
    """Generate good initial sequences via random spectral shaping."""
    rng = np.random.default_rng(seed)
    best_a = np.ones(n) / n
    best_score = internal_score(best_a)
    fft_size = 1 << (2 * n - 1).bit_length()

    for trial in range(n_trials):
        # Random phase in frequency domain
        A = np.ones(fft_size // 2 + 1) / np.sqrt(fft_size)
        for i in range(1, len(A)):
            A[i] *= np.exp(1j * rng.uniform(0, 2 * np.pi))
        # Add some spectral shaping toward flatness
        power = np.abs(A) ** 2
        target = np.exp(np.mean(np.log(power[1:] + 1e-30)))
        ratio = np.sqrt(target / (power[1:] + 1e-30))
        ratio = np.clip(ratio, 0.1, 10.0)
        A[1:] *= ratio
        a = irfft(A, fft_size)[:n]
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            continue
        a = a / s
        score = internal_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_a, best_score
