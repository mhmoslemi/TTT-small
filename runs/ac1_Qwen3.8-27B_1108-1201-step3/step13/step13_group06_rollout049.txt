```python
import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_conv(a, b=None):
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]

def evaluate_local(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    c = fast_conv(a, a)
    return 2.0 * n * c.max() / (s * s)
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral (DFT-based) initialization for low-autocorrelation sequences
2. Adam optimizer on log-space (softmax parameterization) with top-K LogSumExp
3. Direct subgradient with Nesterov momentum on the hard-max objective
4. Multi-peak shaving with bisection on transfer amounts
5. Multi-length exploration with interpolation
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_conv(a, b=None):
    """FFT-based linear convolution."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def score_from_conv(a, c):
    """Compute score given sequence and its autocorrelation."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    return 2.0 * n * c.max() / (s * s)


def get_score(a):
    """Compute the objective value."""
    a = np.maximum(a, 0)
    c = fast_conv(a, a)
    return score_from_conv(a, c)


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def spectral_init(n, rng):
    """
    Generate initial sequence using DFT-based method.
    Target: flat autocorrelation → |DFT(a)| should be as uniform as possible.
    """
    # Target autocorrelation: uniform
    L = 1 << (2 * n - 1).bit_length()
    target = np.zeros(L)
    target[:2*n-1] = 1.0
    
    # DFT of target
    T = rfft(target, L)
    
    # We want |A|^2 ≈ T, so A ≈ sqrt(T) with appropriate phase
    # T should be non-negative real (since target is symmetric and non-negative)
    # Take magnitude and use linear phase
    mag = np.sqrt(np.maximum(T, 0))
    
    # Use zero phase (centered)
    A = mag
    a_full = irfft(A, L)
    
    # Take first n elements
    a = a_full[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def structured_inits(n, rng, k=5):
    """Generate k diverse structured initializations."""
    inits = []
    
    # 1. Spectral
    inits.append(spectral_init(n, rng))
    
    # 2. Uniform with noise
    a = np.ones(n) + rng.normal(0, 0.1, n)
    a = np.maximum(a, 0)
    a = a / a.sum()
    inits.append(a)
    
    # 3. Edge-weighted
    x = np.linspace(0, 1, n)
    a = np.exp(-20 * x**2) + np.exp(-20 * (x-1)**2) + 0.1
    a = a / a.sum()
    inits.append(a)
    
    # 4. Central peak
    x = np.linspace(0, 1, n)
    a = np.exp(-30 * (x - 0.5)**2) + 0.05
    a = a / a.sum()
    inits.append(a)
    
    # 5. Cosine window
    x = np.linspace(0, np.pi, n)
    a = np.cos(x/2)**4 + 0.01
    a = a / a.sum()
    inits.append(a)
    
    return inits


def adam_lse_optimize(a0, time_limit, T_start=100.0, T_end=1.0, n_stages=20, topk=30, seed=42):
    """
    Adam optimizer on log-space with top-K LogSumExp smoothing.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    
    best_a = a_init.copy()
    best_score = get_score(a_init)
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        # Learning rate: start moderate, decrease over stages
        lr = 0.01 * min(1.0, 500.0 / n) * (0.9 ** si)
        
        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.85, 0.999, 1e-8
        
        # Number of iterations per stage
        iters_per_stage = max(50, int(remaining * 2))
        max_it = min(iters_per_stage, 5000)
        
        stall = 0
        best_stage_score = best_score
        
        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 2:
                break
            
            a = softmax_fn(z)
            c = fast_conv(a, a)
            
            # Top-K LogSumExp
            k_use = min(topk, len(c))
            top_idx = np.argsort(c)[::-1][:k_use]
            top_vals = c[top_idx]
            
            # Softmax weights over top-K
            T_eff = stage_T
            shifted = top_vals - top_vals.max()
            exp_shifted = np.exp(T_eff * shifted)
            w = exp_shifted / exp_shifted.sum()
            
            # Compute gradient of LSE w.r.t. a
            # LSE = (1/T) * log(sum exp(T*c_k)) for k in top-K
            # dLSE/da[j] = (1/T) * sum_{k in topK} w_k * dc_k/da[j]
            # dc_k/da[j] = a[k-j] (for valid j)
            grad_a = np.zeros(n)
            for idx in range(k_use):
                K = top_idx[idx]
                wk = w[idx] / T_eff
                js = max(0, K - n + 1)
                je = min(n - 1, K)
                if js <= je:
                    idx_range = np.arange(js, je + 1)
                    grad_a[idx_range] += wk * a[K - idx_range]
            
            # Convert to z-space gradient
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)
            
            # Adam update
            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z**2
            m_hat = m_z / (1 - beta1**t_adam)
            v_hat = v_z / (1 - beta2**t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Check score periodically
            if (it + 1) % 10 == 0:
                a_check = softmax_fn(z)
                sc = get_score(a_check)
                if sc < best_score:
                    best_score = sc
                    best_a = a_check.copy()
                    stall = 0
                    best_stage_score = sc
                else:
                    stall += 1
                
                # Reset if stuck
                if stall > 200:
                    z = np.log(np.maximum(best_a, 1e-30))
                    m_z = np.zeros(n)
                    v_z = np.zeros(n)
                    t_adam = 0
                    lr *= 0.7
                    stall = 0
        
        # End of stage
        a_stage = softmax_fn(z)
        sc_stage = get_score(a_stage)
        if sc_stage < best_score:
            best_score = sc_stage
            best_a = a_stage.copy()
        
        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {si}/{n_stages} (T={stage_T:.1f}, {elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score


def subgradient_momentum(a0, time_limit, seed=42):
    """
    Direct subgradient descent with Nesterov momentum on the hard-max objective.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = get_score(a)
    
    # Nesterov momentum
    y = a.copy()
    v = np.zeros(n)
    mu = 0.9  # momentum
    step = 0.005
    
    t0 = time.monotonic()
    stall = 0
    K = 10  # number of top peaks to consider
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov lookahead
        y = a + mu * v
        
        # Compute autocorrelation at y
        c = fast_conv(y, y)
        cmax = c.max()
        
        # Get top K peaks
        top_idx = np.argsort(c)[::-1][:K]
        
        # Compute subgradient (average of top K peak gradients)
        grad = np.zeros(n)
        for k in top_idx:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                idx_range = np.arange(js, je + 1)
                grad[idx_range] += y[k - idx_range]
        grad /= K
        
        # Normalize
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        # Momentum update
        v = mu * v - step * grad
        a_new = a + v
        
        # Project to simplex
        a_new = project_simplex(a_new)
        
        # Check improvement
        sc_new = get_score(a_new)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
            # Slowly decay step
            step = max(step * 0.9999, 0.0001)
        else:
            stall += 1
            # Backtrack
            step *= 0.95
            if stall % 50 == 0:
                # Reset from best
                a = best_a.copy()
                v = np.zeros(n)
                # Add small random perturbation
                pert = rng.normal(0, 0.001, n)
                a = a + pert * a
                a = project_simplex(a)
            
            if stall > 500:
                break
        
        if it % 5000 == 0 and it > 0:
            elapsed = time.monotonic() - t0
            print(f"    SG-mom iter {it} ({elapsed:.0f}s): best={best_score:.6f}, step={step:.5f}")
    
    return best_a, best_score


def multi_peak_shaving(a0, time_limit, seed=42):
    """
    Multi-peak shaving: simultaneously reduce the top K peaks by
    finding optimal mass transfers via bisection.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = get_score(a)
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit - 2:
        c = fast_conv(a, a)
        cmax = c.max()
        
        # Find peaks within 5% of max
        threshold = cmax * 0.95
        peak_positions = np.where(c >= threshold)[0]
        if len(peak_positions) == 0:
            peak_positions = np.argsort(c)[::-1][:5]
        
        # For each peak, find the most contributing source
        improved = False
        for k in peak_positions[:8]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            
            # Contribution from each pair
            contrib = a[idx] * a[k - idx]
            total_contrib = contrib.sum()
            if total_contrib < 1e-15:
                continue
            
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-10:
                continue
            
            # Find safe destinations (positions not contributing to any top peak)
            danger = np.zeros(n, dtype=bool)
            for kp in peak_positions[:10]:
                js2 = max(0, kp - n + 1)
                je2 = min(n - 1, kp)
                danger[js2:je2+1] = True
            
            safe = np.where(~danger)[0]
            safe = safe[safe != src]
            if len(safe) == 0:
                safe = np.where(np.arange(n) != src)[0]
            
            # Sort safe positions by current value (prefer moving to low positions)
            safe_sorted = safe[np.argsort(a[safe])]
            dst_pool = safe_sorted[:max(3, len(safe_sorted) // 8)]
            
            if len(dst_pool) == 0:
                continue
            
            # For each candidate destination, find optimal transfer via bisection
            found = False
            for dst in dst_pool[:5]:
                # Bisection to find the max transfer that reduces the peak
                lo, hi = 0.0, src_val * 0.5
                best_frac = 0
                best_cmax = cmax
                
                for _ in range(15):
                    mid = (lo + hi) / 2
                    a_trial = a.copy()
                    a_trial[src] -= mid
                    a_trial[dst] += mid
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        hi = mid
                        continue
                    a_trial = a_trial / s_t
                    c_trial = fast_conv(a_trial, a_trial)
                    cm = c_trial.max()
                    if cm < best_cmax:
                        best_cmax = cm
                        best_frac = mid
                        lo = mid
                    else:
                        hi = mid
                
                if best_frac > 1e-6:
                    a_trial = a.copy()
                    a_trial[src] -= best_frac
                    a_trial[dst] += best_frac
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    a_trial = a_trial / s_t
                    sc = get_score(a_trial)
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        found = True
                        break
            
            if found:
                break
        
        if not improved:
            stall += 1
            # Try random perturbations
            if stall % 5 == 0:
                n_pert = rng.integers(2, 6)
                a_trial = a.copy()
                for _ in range(n_pert):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a_trial[i] > 1e-8:
                        frac = rng.uniform(0.002, 0.02)
                        amt = a_trial[i] * frac
                        a_trial[i] -= amt
                        a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t > 1e-8:
                    a_trial = a_trial / s_t
                    sc = get_score(a_trial)
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        stall = 0
            
            if stall > 100:
                break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
