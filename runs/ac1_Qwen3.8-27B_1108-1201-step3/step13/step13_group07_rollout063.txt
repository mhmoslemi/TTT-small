By the rearrangement inequality, for a fixed set of values {a[j]}, the sum sum_j a[j]*a[n-1-j] is:
- Maximized when a and its reverse are similarly sorted (i.e., a is symmetric)

Since a >= 0, we can't make it truly antisymmetric, but we can make it "one-sided" - e.g., increasing or decreasing.

        a = a / s

    y = a.copy()
    y_prev = a.copy()
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    iterate_sum = np.zeros(n)
    # Store recent iterates for local halo
    recent_window = []
    window_size = 500

    t = 0
    lp_time_total = 0.0

    while time.monotonic() - t0 < time_limit:
        t += 1
        lp_t0 = time.monotonic()
        b_star, M_val = solve_lp_best_response(y, solver)
        lp_time_total += time.monotonic() - lp_t0
        if b_star is None:
            continue

        alpha_t = 2.0 / (t + 2.0)
        y = (1 - alpha_t) * y + alpha_t * b_star
        y = np.maximum(y, 0)
        s = y.sum()
        if s < 1e-15:
            y = a.copy()
        else:
            y = y / s

        # Nesterov extrapolation
        if t == 1:
            a = b_star.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            a = y + beta * (y - y_prev)
            a = project_simplex(a)

        y_prev = y.copy()

        # Accumulate for time-average
        iterate_sum += a

        # Keep recent window
        recent_window.append(a.copy())
        if len(recent_window) > window_size:
            recent_window.pop(0)

        # Check scores (a, y, time-average)
        score_a = internal_score(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = internal_score(y)
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Halo checks
        if t % 50 == 0:
            # Full time average
            halo_full = iterate_sum / t
            sc = internal_score(halo_full)
            if sc < best_score:
                best_score = sc
                best_a = halo_full.copy()

            # Recent window average
            if len(recent_window) >= 50:
                halo_recent = np.mean(recent_window[-min(200, len(recent_window)):], axis=0)
                halo_recent = np.maximum(halo_recent, 0)
                s_h = halo_recent.sum()
                if s_h > 1e-15:
                    halo_recent = halo_recent / s_h
                    sc = internal_score(halo_recent)
                    if sc < best_score:
                        best_score = sc
                        best_a = halo_recent.copy()

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            avg_lp = lp_time_total / t * 1000
            print(f"    NL t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}, avg_LP={avg_lp:.0f}ms")

    # Final halo check with various windows
    if t > 0:
        for frac in [1.0, 0.5, 0.3, 0.2, 0.1]:
            k = max(1, int(t * frac))
            # Approximate using total sum (we don't store all iterates)
            pass
        halo = iterate_sum / t
        sc = internal_score(halo)
        if sc < best_score:
            best_score = sc
            best_a = halo.copy()

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    NL done: t={t}, time={elapsed:.0f}s, best={best_score:.6f}")

    return best_a, best_score


def spectral_shaping(a0, n_iter=300, alpha=0.25):
    """Flatten power spectrum to reduce autocorrelation peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = internal_score(a)

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2

        non_dc = B[1:]
        # Target: geometric mean of non-DC power
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        scale = np.ones(len(A))
        ratio = target / (on_dc + 1e-30) if False else target / (non_dc + 1e-30)
        ratio = np.clip(ratio, 0.05, 20.0)
        scale[1:] = np.power(ratio, alpha / 2.0)

        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def subgradient_polish(a0, time_limit, verbose=True):
    """Subgradient descent with Nesterov momentum on max_k (a*a)_k."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    v = np.zeros(n)
    beta = 0.7  # momentum
    base_lr = 0.05

    conv_len = 2 * n - 1
    fft_size = 1 << (conv_len - 1).bit_length()

    t = 0
    while time.monotonic() - t0 < time_limit:
        t += 1
        b = fast_autocorr(a)

        # Softmax weights on top peaks (smoothed subgradient)
        T_smooth = max(5.0, 100.0 * max(0, 1.0 - t / 500.0))
        Tb = T_smooth * (b - b.min())
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()

        # Gradient: grad_j = 2 * sum_k w_k * a_{k-j}
        A_rev = rfft(a[::-1].copy(), fft_size)
        W = rfft(w, fft_size)
        conv = irfft(A_rev * W, fft_size)
        grad = 2.0 * conv[n - 1:n - 1 + n]

        # Learning rate
        lr_t = base_lr / (1.0 + 0.005 * t)

        # Nesterov momentum
        v = beta * v + (1 - beta) * grad
        a_new = a - lr_t * v
        a_new = project_simplex(a_new)

        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad t={t} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, seed=42):
    """Coordinate descent: move mass from peak-contributing positions to low-impact positions."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        # Find top contributing lag
        k_peak = int(np.argmax(b))
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]

        if contrib.max() < 1e-15:
            break

        # Find the pair (i, j) with max contribution at the peak
        pair_idx = int(np.argmax(contrib))
        i = int(idx[pair_idx])
        j = int(k_peak - idx[pair_idx])

        # Try moving mass from max(a[i], a[j]) to a random low-index position
        if a[i] >= a[j]:
            src = i
        else:
            src = j

        # Find a good destination: low contribution to all top peaks
        top_lags = np.argsort(b)[::-1][:5]
        dest_candidates = []
        for d in range(n):
            if d == src:
                continue
            # Check contribution to top peaks
            max_contrib = 0
            for k2 in top_lags:
                if 0 <= k2 - d < n:
                    max_contrib = max(max_contrib, a[k2 - d])
            dest_candidates.append((max_contrib, d))

        dest_candidates.sort()
        dst = dest_candidates[0][1]

        # Try different fractions
        improved = False
        for frac in [0.01, 0.03, 0.05, 0.1, 0.2, 0.3]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            if a_trial[src] < 0:
                continue
            s_t = a_trial.sum()
            a_trial = a_trial / s_t
            score = internal_score(a_trial)
            if score < best_score - 1e-12:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                stall = 0
                break

        if not improved:
            # Random perturbation
            i_r = int(rng.integers(n))
            j_r = int(rng.integers(n))
            if i_r != j_r and a[i_r] > 1e-10:
                frac = rng.uniform(0.01, 0.1)
                amt = a[i_r] * frac
                a_trial = a.copy()
                a_trial[i_r] -= amt
                a_trial[j_r] += amt
                if a_trial[i_r] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = internal_score(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            stall += 1
            if stall > 200:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(arr):
    """Safely evaluate a sequence using the provided evaluate_sequence function."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')
