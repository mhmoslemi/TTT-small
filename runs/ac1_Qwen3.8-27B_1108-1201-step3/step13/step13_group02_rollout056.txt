`f_T(a) = T * log(sum_k exp(c_k / T))`
`df_T/da_i = (1/sum_k exp(c_k/T)) * sum_k exp(c_k/T) * dc_k/da_i`
`= sum_k w_k * dc_k/da_i`

where `w_k = exp(c_k/T) / sum_j exp(c_j/T)`.

And `dc_k/da_i = a_{k-i}` (if `0 <= k-i < n`).

So: `g_i = sum_k w_k * a_{k-i}` (summed over valid k).

This is the same gradient as before, but we apply it multiplicatively:
`a_i^{new} = a_i^{old} * exp(-eta * g_i)`
Then normalize: `a_i^{new} = a_i^{new} / sum(a^{new})`.

The key advantage of multiplicative updates:
- Automatically maintains non-negativity
- The step size is relative, so large components are reduced more
- Convergence can be faster for problems with non-negativity constraints

The choice of `eta` is critical. A good heuristic is:
`eta = 1 / (max(g) - min(g))` or `eta = 1 / std(g)`.

    top_idx = np.argsort(c)[::-1][:n_sel]
    T_sub = build_T_rows(a, top_idx)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_sub @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    # Verify against all constraints via FFT
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b_val, fft_size)
    full_conv = irfft(A * B, fft_size)[:size]
    if full_conv.max() > M * (1 + 1e-5):
        # Add violated constraints
        violated = full_conv > M * (1 + 1e-5)
        all_idx = np.unique(np.concatenate([top_idx, np.where(violated)[0]]))
        T_full = build_T_rows(a, all_idx)
        constraints = [T_full @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            pass
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
            if b_val.sum() < 1e-15:
                return None
            # Final check
            B2 = rfft(b_val, fft_size)
            full_conv2 = irfft(A * B2, fft_size)[:size]
            if full_conv2.max() > M * (1 + 1e-3):
                return None
    return b_val

def nesterov_lp(a0, time_limit, solver, verbose=True):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    best_stall = 0
    n_constraints = 150

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        c_y = fast_autocorr(y)
        M_curr = c_y.max()
        if M_curr < 1e-15:
            break

        b = lp_solve(y, M_curr, solver, n_constraints=n_constraints)
        if b is None:
            continue
        b = b / b.sum()
        n_iters += 1

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * b
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y

        if t == 1:
            a_new = b.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
            best_stall = 0
        else:
            stall += 1
            best_stall += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
            best_stall = 0

        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, stall={stall}")

        # Adaptive restart
        if stall > 300 and best_stall > 500:
            rng = np.random.default_rng(n_iters * 13 + 7)
            scale = 0.03 * a.mean()
            y = np.maximum(y + rng.normal(0, scale, n), 0)
            y = y / y.sum()
            a = np.maximum(a + rng.normal(0, scale, n), 0)
            a = a / a.sum()
            stall = 0
            best_stall = 0
            if verbose:
                print(f"    [Restart at iter {n_iters}]")
            # Increase constraints after restart
            n_constraints = min(300, n_constraints + 50)

        # Gradually increase constraints for accuracy
        if n_iters % 1000 == 0:
            n_constraints = min(400, n_constraints + 25)

    return best_a, best_score

def compute_lse_gradient(a, w):
    n = len(a)
    a_rev = a[::-1]
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)
    conv = irfft(W * AR, fft_size)
    return 2.0 * conv[n - 1:2 * n - 1]

def lse_gd(a0, time_limit, verbose=True):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0

    T_start = max(0.5, best_score * 0.1)
    T_end = max(0.01, best_score * 0.005)

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress

        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break

        w = np.exp((b - max_b) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        grad = compute_lse_gradient(a, w)
        step = 1.0 / (T * n * 0.3)
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new
        steps += 1

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and steps % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-GD step={steps} ({elapsed:.0f}s): T={T:.5f}, best={best_score:.8f}")

    return best_a, best_score

def mirror_descent(a0, time_limit, verbose=True):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0

    T_start = max(0.5, best_score * 0.15)
    T_end = max(0.01, best_score * 0.008)

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress

        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break

        w = np.exp((b - max_b) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        grad = compute_lse_gradient(a, w)
        # Multiplicative update
        g_centered = grad - grad.mean()
        eta = 1.0 / (max(np.abs(g_centered).max(), 1e-10) * n * 0.5 + 1e-10)
        a_new = a * np.exp(-eta * g_centered)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new
        steps += 1

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and steps % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDescent step={steps} ({elapsed:.0f}s): T={T:.5f}, best={best_score:.8f}")

    return best_a, best_score

def peak_shave(a0, time_limit, seed=42, verbose=True):
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_improve = 0
    n_stall = 0

    T_sa = max(0.1, best_score * 0.02)

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_sa * (1.0 - elapsed_frac) * 5 + best_score * 0.001

        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:10]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            if contrib.max() < 1e-18:
                continue

            src = idx[np.argmax(contrib)]

            # Build peak mask
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            non_peak = np.where(~peak_mask)[0]
            if len(non_peak) == 0:
                non_peak = np.where(np.arange(n) != src)[0]

            n_dest = min(6, len(non_peak))
            sorted_dst = non_peak[np.argsort(a[non_peak])][:n_dest * 3]
            selected_dst = sorted_dst[rng.choice(len(sorted_dst), min(n_dest, len(sorted_dst)), replace=False)]

            frac_total = rng.choice([0.03, 0.06, 0.1, 0.15, 0.25])
            amt_total = a[src] * frac_total
            if amt_total < 1e-18:
                continue

            dst_vals = a[selected_dst] + 1e-10
            weights = 1.0 / dst_vals
            weights = weights / weights.sum()
            amounts = amt_total * weights

            a_trial = a.copy()
            a_trial[src] -= amt_total
            for di, dst in enumerate(selected_dst):
                a_trial[dst] += amounts[di]

            if a_trial[src] < -1e-12:
                continue
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t

            score = fast_autocorr(a_trial).max()

            if score < best_score:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                n_improve += 1
                break
            elif rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                a = a_trial
                improved = True
                break

        if not improved:
            n_stall += 1
            # Random pair swap
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.05, 0.2)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        n_improve += 1
                    elif rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                        a = a_trial

            if n_stall > 150:
                # Larger perturbation
                idx_pert = rng.choice(n, size=max(2, n // 40), replace=False)
                a_trial = a.copy()
                for idx in idx_pert:
                    amt = a[idx] * rng.uniform(0.1, 0.5)
                    j = rng.integers(n)
                    a_trial[idx] -= amt
                    a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score * 1.02:
                        a = a_trial
                        if score < best_score:
                            best_score = score
                            best_a = a.copy()
                n_stall = 0

        if verbose and n_improve > 0 and n_improve % 3000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave improve={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    # Phase 1: LP Nesterov at base length
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.30, 450.0)
    if t1 > 30:
        a_out, _ = nesterov_lp(best_seq, t1, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n_base}")

    # Phase 2: Mirror descent
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: MirrorDescent n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.10, 120.0)
    if t2 > 10:
        a_out, _ = mirror_descent(best_seq, t2, verbose=True)
        do_try_update(a_out, f"md_n{len(best_seq)}")

    # Phase 3: LSE-GD polish
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LSE-GD n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.08, 100.0)
    if t3 > 10:
        a_out, _ = lse_gd(best_seq, t3, verbose=True)
        do_try_update(a_out, f"lse_n{len(best_seq)}")

    # Phase 4: LP Nesterov at n+200
    n2 = n_base + 200
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LP Nesterov n={n2} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.20, 280.0)
    if t4 > 30:
        init_long = interpolate_seq(best_seq, n2)
        a_out, _ = nesterov_lp(init_long, t4, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n2}")

    # Phase 5: Mirror descent at n+200
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: MirrorDescent n={n_cur} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.08, 100.0)
    if t5 > 10:
        a_out, _ = mirror_descent(best_seq, t5, verbose=False)
        do_try_update(a_out, f"md_n{n_cur}")

    # Phase 6: LP Nesterov at n+400
    n3 = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: LP Nesterov n={n3} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 30:
        init_long3 = interpolate_seq(best_seq, n3)
        a_out, _ = nesterov_lp(init_long3, t6, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n3}")

    # Phase 7: Final LP on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP n={n_best} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.20, 200.0)
    if t7 > 20:
        a_out, _ = nesterov_lp(best_seq, t7, solver, verbose=True)
        do_try_update(a_out, f"lp_final_n{n_best}")

    # Phase 8: Final peak shaving
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final PeakShave n={n_best} (budget={remaining:.0f}s) ===")
    t8 = min(remaining * 0.15, 120.0)
    if t8 > 10:
        a_out, _ = peak_shave(best_seq, t8, seed=999, verbose=True)
        do_try_update(a_out, f"pd_final_n{n_best}")

    # Phase 9: Final LSE polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final LSE n={n_best} (budget={remaining:.0f}s) ===")
    t9 = min(remaining * 0.10, 80.0)
    if t9 > 10:
        a_out, _ = lse_gd(best_seq, t9, verbose=False)
        do_try_update(a_out, f"lse_final_n{n_best}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
