import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, minimize_scalar
import cvxpy as cp

def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def internal_score(a):
    """Compute 2*n*max_b assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def softmax_fn(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def project_simplex(v, target_sum=1.0):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho = np.nonzero(v_sorted * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.full(n, target_sum/n)
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-12:
        result = result / result.sum() * target_sum
    return result

def lse_objective_and_grad(z, T):
    """LogSumExp objective and gradient for z-parameterization."""
    a = softmax_fn(z)
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    obj = n * (max_Tb + np.log(exp_Tb.sum())) / T  # times 2 for the score
    
    # Gradient
    w = exp_Tb / exp_Tb.sum()
    conv_len = n + (2*n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    grad_a = 2.0 * conv[n-1:n-1+n]
    
    # Chain rule: d/dz_j = a_j * (grad_a_j - sum_i a_i * grad_a_i)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return 2.0 * obj, 2.0 * grad_z

def lp_best_response_fixed_a(a_fixed, solver):
    """Given fixed a, find b that minimizes max_k (a*b)_k."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build convolution matrix T where (T @ b)_k = (a*b)_k
    # (a*b)_k = sum_j a_j * b_{k-j}
    # For efficiency, use the structure
    
    T = np.zeros((2*n-1, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver, max_iter=1000)
    except Exception:
        pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    b_val = b_val / s
    return b_val

def coordinate_ascent_peak_reduction(a0, time_limit, seed=42):
    """
    Direct coordinate manipulation to reduce the peak autocorrelation.
    Idea: identify which pairs (i,j) contribute most to the peak,
    and redistribute mass from those positions.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    stall_count = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        peak_idx = np.argmax(b)
        peak_val = b[peak_idx]
        
        # Find which pairs contribute most to the peak
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_idx - idx]
        
        # Top contributors
        n_top = min(5, len(idx))
        top_local = np.argsort(contrib)[::-1][:n_top]
        
        improved = False
        for li in top_local:
            src = idx[li]
            dst_partner = peak_idx - src
            
            # Try moving mass from src to a random position
            # that doesn't contribute to the peak
            peak_areas = set()
            for k in np.argsort(b)[::-1][:20]:
                js2 = max(0, k - n + 1)
                je2 = min(n - 1, k)
                peak_areas.update(range(js2, je2 + 1))
            
            candidates = [j for j in range(n) if j not in peak_areas and j != src]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            
            # Try several fractions
            for frac in [0.001, 0.002, 0.005, 0.01, 0.02]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                dst = candidates[rng.integers(len(candidates))]
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                
                score = internal_score(a_trial)
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                stall_count = 0
                break
        
        if not improved:
            # Random perturbation
            stall_count += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:

    return best_a, best_score

def spectral_shaping_v2(a, n_iter=1000, alpha=0.2, verbose=False):
    """Spectral shaping with adaptive parameters."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()
    
    best_a = a.copy()
    best_score = internal_score(a)
    
    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        
        # Multi-scale target: flatten toward geometric mean
        non_dc = B[1:]
        # Use a softer target: geometric mean raised to a power
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 100.0)
        scale = np.ones(len(A))
        scale[1:] = np.power(ratio, alpha / 2.0)
        
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s2 = a_new.sum()
        if s2 < 1e-15:
            break
        a_new = a_new / s2
        
        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    
    return best_a

def window_functions(n):
    """Generate various window functions as starting points."""
    x = np.linspace(0, 1, n)
    windows = {}
    windows['uniform'] = np.ones(n)
    windows['sine'] = np.sin(np.pi * x)**2
    windows['hann'] = 0.5 * (1 + np.cos(2*np.pi*x - 2*np.pi))
    windows['hann'] = 0.5 - 0.5*np.cos(2*np.pi*(x - 0.5))
    windows['hamming'] = 0.54 - 0.46*np.cos(2*np.pi*(x - 0.5))
    windows['blackman'] = 0.42 - 0.5*np.cos(2*np.pi*(x-0.5)) + 0.08*np.cos(4*np.pi*(x-0.5))
    windows['ninewp'] = 0.424 - 0.4972*np.cos(2*np.pi*(x-0.5)) + 0.0812*np.cos(4*np.pi*(x-0.5) - 2*np.pi)
    # Raised cosine with different parameters
    for beta in [0.5, 0.7, 0.9, 1.0]:
        windows[f'raised_cos_{beta}'] = np.clip(np.cos(np.pi*(x - 0.5)/beta), 0, 1)**2
    for name in windows:
        w = np.maximum(windows[name], 0)
        sw = w.sum()
        if sw > 0:
            windows[name] = w / sw
    return windows

def get_lp_solver():
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.CBC]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-initialization (parent, windows, random)
2. scipy L-BFGS-B on LogSumExp-smoothed objective (primary)
3. Hand-rolled Adam on LSE (secondary, different params)

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def internal_score(a):
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()

def softmax_fn(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def project_simplex(v, target_sum=1.0):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if np.all(v >= 0) and abs(v.sum() - target_sum) < 1e-12:
        return v
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho = np.nonzero(v_sorted * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-10:
        result = np.maximum(result, 0)
        result = result / result.sum() * target_sum
    return result

def lse_obj_grad(z, T):
    """LogSumExp objective and gradient w.r.t. z."""
    a = softmax_fn(z)
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    exp_sum = exp_Tb.sum()
    obj = (max_Tb + np.log(exp_sum)) / T
    
    w = exp_Tb / exp_sum
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    grad_a = 2.0 * conv[n - 1:n - 1 + n]
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return obj, grad_z

def lse_obj_grad_wrapper(args):
    z, T = args
    return lse_obj_grad(z, T)

def scipy_lbfgs_lse(a0, time_limit, T_start=200.0, T_end=0.05, n_stages=30, verbose=True):
    """Use scipy L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = internal_score(a_init)
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        
        def func(z_curr):
            obj, _ = lse_obj_grad(z_curr, stage_T)
            return obj
        
        def grad(z_curr):
            _, g = lse_obj_grad(z_curr, stage_T)
            return g
        
        res = minimize(func, z, jac=grad, method='L-BFGS-B',
                       options={'maxiter': 200, 'gtol': 1e-8, 'maxfun': 500})
        
        if res.success or res.fun < 1e10:
            z = res.x
        
        a = softmax_fn(z)
        score = internal_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if verbose and si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      L-BFGS stage {si}/{n_stages} (T={stage_T:.1f}, {elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score

def adam_lse_optimize(a0, time_limit, T_start=500.0, T_end=0.05, n_stages=40, lr=0.008, verbose=True):
    """Adam optimizer on LSE-smoothed objective."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)

    if s < 1e-15:

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = internal_score(a_init)
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_adam = 0
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        iters = max(30, min(300, int(remaining * 100 / max(1, n_stages - si))))

            t_adam += 1

                score = internal_score(a)

    t = 0
