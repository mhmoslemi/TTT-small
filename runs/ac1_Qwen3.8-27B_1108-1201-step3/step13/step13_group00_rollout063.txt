            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 10 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lse_gradient_optimize(a0, time_limit, verbose=True):
    """LSE gradient descent with FFT-accelerated gradients and annealing."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    # Compute initial max for temperature
    conv_init = fast_autocorr(a)
    max_init = conv_init.max()

    # Annealing schedule: geometric from T_high to T_low
    T_high = max_init / 5.0
    T_low = max_init / 10000.0
    n_stages = 12
    temps = np.geomspace(T_high, T_low, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        # More steps at lower temperatures
        base_steps = max(50, min(300, int(remaining * 2)))
        lr = 0.5 / (n * T + 1e-10)

        for it in range(base_steps):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_autocorr(a)
            max_conv = conv.max()
            # Compute LSE weights
            exp_conv = np.exp((conv - max_conv) / T)
            w = exp_conv / exp_conv.sum()

            # Compute gradient using FFT
            # grad_i = 2 * sum_k w[k] * a[k-i]
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            C = irfft(W * np.conj(A), fft_size)
            g = C[:n]
            grad = 2.0 * g

            # Adaptive learning rate with backtracking
            step = lr / (1.0 + 0.01 * it)
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new

            # Check if objective improved (use max of conv as proxy)
            new_conv = fast_autocorr(a_new)
            new_max = new_conv.max()
            if new_max < max_conv + 1e-10:
                a = a_new
            else:
                # Backtracking: reduce step
                step *= 0.5
                a_new = a - step * grad
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new > 1e-15:
                    a_new = a_new * norm / s_new
                    a = a_new

            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {si+1}/{n_stages} T={T:.4f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pair_coordinate_descent(a0, time_limit, rng, verbose=True):
    """Efficient pair coordinate descent with random pairs."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False

        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-8:
                continue

            for attempt in range(2):
                j = int(rng.integers(n))
                if j == i:
                    continue

                ai = a[i]
                # Compute first-order gain: change in conv if we move delta from i to j
                # d(conv_k)/d(delta) = a[k-j] + a[j-k] - a[k-i] - a[i-k] (with valid indices)
                # For the peak, we mainly care about the current max location
                k_max = int(np.argmax(conv))

                # First-order change at k_max
                gain = 0.0
                # Contribution from j (positive, we're adding to j)
                if 0 <= k_max - j < n:
                    gain += a[k_max - j]
                if 0 <= j - k_max < n:
                    gain += a[j - k_max]
                # Contribution from i (negative, we're removing from i)
                if 0 <= k_max - i < n:
                    gain -= a[k_max - i]
                if 0 <= i - k_max < n:
                    gain -= a[i - k_max]

                if gain >= 0:
                    # Moving mass from i to j would increase the peak at k_max
                    # Try small move to see if other peaks decrease enough
                    pass
                else:
                    # Moving helps reduce the peak at k_max
                    # Find optimal delta
                    best_d = 0.0
                    best_val = conv.max()
                    # Try several delta values
                    n_samples = 10
                    deltas = np.linspace(0.01 * ai, 0.9 * ai, n_samples)
                    for d in deltas:
                        # Update conv approximately (first order)
                        # Conv change: for each k, delta * (a[k-j]+a[j-k]-a[k-i]-a[i-k])
                        # Just check the peak
                        new_conv_kmax = conv[k_max] + d * gain
                        # Also check second-order at k=2i and k=2j
                        extra = 0.0
                        if 2 * i < size:
                            extra = max(extra, -2 * ai * d + d * d)
                        if 2 * j < size:
                            extra = max(extra, 2 * a[j] * d + d * d)
                        new_max = max(new_conv_kmax, conv[k_max] + d * gain * 0.5 + abs(extra) * 0.1)
                        # Simple: just check if the reduction at k_max is significant
                        if new_conv_kmax < best_val - 1e-12:
                            best_val = new_conv_kmax
                            best_d = d

                    if best_d > 0:
                        a[i] -= best_d
                        a[j] += best_d
                        s = a.sum()
                        if s < 1e-15:
                            a[i] += best_d
                            a[j] -= best_d
                            continue
                        a = a * norm / s
                        conv = fast_autocorr(a)
                        cur_max = conv.max()
                        if cur_max < best_score - 1e-12:
                            best_score = cur_max
                            best_a = a.copy()
                            pass_improved = True
                            no_improve = 0
                        else:
                            a[j] -= best_d
                            a[i] += best_d
                            a = a * norm / a.sum()
                            conv = fast_autocorr(a)

        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def subgradient_fw(a0, time_limit, verbose=False):
    """Fast Frank-Wolfe with subgradient - O(n) per step."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0

    for t in range(1, 200000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        k_star = int(np.argmax(conv_y))

        # Subgradient: grad_i = 2 * y[k_star - i] (for valid k_star - i)
        grad = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n, k_star + 1)
        for i in range(lo, hi):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * y[j]

        i_min = int(np.argmin(grad))
        v = np.zeros(n)
        v[i_min] = norm

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * v
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = v.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        if t % 100 == 0:
            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 200 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                sc = fast_autocorr(avg_check).max()
                if sc < best_score:
                    best_score = sc
                    best_a = avg_check.copy()

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss
