```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based Nesterov optimization
with adaptive parameters, multi-length search, and coordinate polish.
"""
import numpy as np
import cvxpy as cp
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:size]
    return c

def proj_simplex(v, s):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) + s - cssv > 0
    rho = u[np.argmax(cond)]
    return np.maximum(v - rho, 0.0)

def lp_best_response(a, M, solver):
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    # conv(a,b)[k] = sum_j a_j * b_{k-j} (with bounds)
    # This is a linear function of b
    # T @ b where T[k,j] = a_j if 0 <= k-j < n, else 0
    # T is (2n-1) x n
    T = np.zeros((2*n-1, n))
    for j in range(n):
        T[j:j+n, j] = a
    prob = cp.Problem(cp.Maximize(cp.sum(b)), [T @ b <= M])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b.value, dtype=np.float64)
    val = np.maximum(val, 0)
    if val.sum() < 1e-15:
        return None
    return val
```

        if Mx < best_m:
            best_m = Mx
            best_a = x.copy()
        
        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c * c)
            print(f"    SD t={t} ({elapsed:.0f}s): J={J:.6f}")
    
    return best_a.copy(), best_m

def coordinate_polish(a0, time_limit, verbose=False):
    """Coordinate descent: move mass to reduce peak autocorrelation."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.full(n, c / n)
    else:
        a = a * c / s
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    
    stall = 0
    max_stall = 300
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        M = ac.max()
        # Top 5 peaks
        top_peaks = np.argsort(ac)[::-1][:5]
        
        improved = False
        for k in top_peaks:
            ki = int(k)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[ki - idx]
            src = idx[np.argmax(contrib)]
            
            # Find destination: minimize impact on top peaks
            peak_involvement = np.zeros(n)
            for k2 in top_peaks:
                k2i = int(k2)
                js2 = max(0, k2i - n + 1)
                je2 = min(n - 1, k2i)
                peak_involvement[js2:je2 + 1] += 1
            
            # Exclude indices near the source from top peaks
            candidates = np.where(peak_involvement == 0)[0]
            if len(candidates) < 10:
                candidates = np.where(np.abs(np.arange(n) - src) > n // 4)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            
            # Pick a random candidate with low value
            cands = candidates[np.argsort(a[candidates])[:min(20, len(candidates))]]
            if len(cands) == 0:
                continue
            dst = cands[rng.integers(len(cands))]
            if dst == src:
                continue
            
            for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                if amt < 1e-20:
                    break
                a_t = a.copy()
                a_t[src] -= amt
                a_t[dst] += amt
                if a_t[src] < 0:
                    continue
                st = a_t.sum()
                if st < 1e-15:
                    continue
                a_t = a_t * c / st
                Mt = fast_autocorr(a_t).max()
                if Mt < best_m - 1e-15:
                    a = a_t
                    best_m = Mt
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                amt = a[i] * rng.uniform(0.01, 0.05)
                a[i] -= amt
                a[j] += amt
                if a[i] >= 0:
                    s2 = a.sum()
                    if s2 > 1e-15:
                        a = a * c / s2
                        M2 = fast_autocorr(a).max()
                        if M2 < best_m:
                            best_m = M2
                            best_a = a.copy()
                            stall = 0
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = np.maximum(a, 0)
                            a = a * c / a.sum()
    
    if verbose:
        print(f"    Coord ({time.monotonic()-t0:.0f}s): J={2.0*n*best_m/(c*c):.6f}")
    return best_a.copy(), best_m

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    ox = np.linspace(0, 1, old_n)
    nx = np.linspace(0, 1, new_n)
    fi = np.interp(nx, ox, fn)
    fi = np.maximum(fi, 0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss

def find_solver():
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.3])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    print("  Fallback solver: ECOS")
    return cp.ECOS
