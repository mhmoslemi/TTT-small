Actually, the gradient of c[k] w.r.t. a[i] is:

For the hinge loss L = sum_k max(0, c[k] - thresh), the subgradient is:
- dL/da[i] = sum_{k: c[k] > thresh} (a[k-i] + a[i-k])

Note that a[k-i] + a[i-k] = a[k-i] + a[(-k)%... ] 

Wait, since the convolution is of a with itself, c[k] = sum_j a[j] * a[k-j]. The derivative d c[k] / d a[i] = a[k-i] + a[i-k] (the two terms come from the two ways i can appear in the sum).

But a[k-i] is only valid when 0 ≤ k-i < n, and a[i-k] is valid when 0 ≤ i-k < n.

So for a given k:
- a[k-i] is valid for i in [max(0, k-n+1), min(n-1, k)]
- a[i-k] is valid for i in [max(0, k), min(n-1, n-1+k)]

    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

    else:

        score_a = fast_autocorr(a).max()

def hinge_loss_optimize(a0, time_limit, verbose=False):
    """
    New: Minimize max_k c[k] via hinge loss L = sum_k max(0, c[k] - thresh).
    Gradient computed in O(n log n) via FFT cross-correlation.
    Anneal threshold downward.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    initial_max = fast_autocorr(a).max()
    # Anneal from slightly below current max down to target
    n_stages = 20
    # Start near current max, go below
    start_thresh = initial_max * 0.999
    end_thresh = initial_max * 0.93
    thresholds = np.linspace(start_thresh, end_thresh, n_stages)

    base_lr = norm / (n * 4.0)

    for si, thresh in enumerate(thresholds):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break

        # Adaptive number of steps
        frac_remaining = (time_limit - (time.monotonic() - t0)) / time_limit
        n_steps = max(30, int(frac_remaining * 200))

        # Adaptive learning rate: decrease as stages progress
        lr = base_lr * (1.0 + 0.5 * (si / n_stages))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_autocorr(a)
            w = (conv > thresh).astype(np.float64)
            n_active = int(w.sum())
            if n_active == 0:
                break

            # Gradient via cross-correlation: grad[i] = 2 * sum_m a[m] * w[i+m]
            A = rfft(a, fft_size)
            W = rfft(w, fft_size)
            cross = irfft(A * np.conj(W), fft_size)
            grad = 2.0 * cross[:n]

            # Projected gradient step with backtracking
            a_trial = a - lr * grad
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-15:
                # Retry with smaller step
                a_trial = a - 0.1 * lr * grad
                a_trial = np.maximum(a_trial, 0.0)
                s_trial = a_trial.sum()
                if s_trial < 1e-15:
                    break
                a_trial = a_trial * norm / s_trial
            else:
                a_trial = a_trial * norm / s_trial

            new_max = fast_autocorr(a_trial).max()
            if new_max < best_score:
                best_score = new_max
                best_a = a_trial.copy()

            # Accept if it doesn't make things too much worse
            if new_max <= initial_max * 1.05:
                a = a_trial
            else:
                # Halve learning rate
                lr *= 0.5
                if lr < base_lr * 0.01:
                    break
                a_trial = a - lr * grad
                a_trial = np.maximum(a_trial, 0.0)
                s_trial = a_trial.sum()
                if s_trial < 1e-15:
                    break
                a_trial = a_trial * norm / s_trial
                a = a_trial

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    Hinge stage {si}/{n_stages} thresh={thresh:.4f} active={n_active} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    base_lr = norm / (n * 10.0)
    n_iters = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_autocorr(a)
        cur_max = conv.max()

        # Gradient of mean(c^2): (4/size) * sum_k c[k] * dc[k]/da[i]
        # dc[k]/da[i] = 2*a[k-i] (for valid k-i)
        # grad[i] = (4/size) * sum_k c[k] * 2*a[k-i] = (8/size) * sum_m a[m] * c[i+m]
        # = (8/size) * (a * c)[i] (cross-correlation)
        A = rfft(a, fft_size)
        C = rfft(conv, fft_size)
        cross = irfft(A * np.conj(C), fft_size)
        grad = (8.0 / size) * cross[:n]

        # Center the gradient (subtract mean to preserve sum better)
        grad -= grad.mean()

        # Adaptive step size
        elapsed = time.monotonic() - t0
        progress = elapsed / time_limit
        lr = base_lr * (1.0 + progress)

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new * norm / s

        new_max = fast_autocorr(a_new).max()
        if new_max < best_score:
            best_score = new_max
            best_a = a_new.copy()

        if new_max <= cur_max * 1.01:
            a = a_new
        else:
            # Backtrack
            lr *= 0.5
            a_new = a - lr * grad
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                break
            a_new = a_new * norm / s
            a = a_new

        n_iters += 1
        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    2ndMom iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def peak_guided_pair_cd(a0, time_limit, rng, verbose=False):
    """
    Improved pair CD: uses peak positions to guide which pairs to try.
    For peak at k*, pairs (i, k*-i) are prioritized.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    top_k = 10  # Number of top peaks to consider

    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False

        # Get top peak positions
        top_peaks = np.argsort(conv)[-top_k:][::-1]

        # Build a list of candidate (i,j) pairs guided by peaks
        candidate_pairs = []
        for k_star in top_peaks:
            # Pairs that contribute to this peak: (i, k_star - i)
            for i in range(n):
                j = k_star - i
                if 0 <= j < n and j != i:
                    if a[i] * a[j] > 0:
                        candidate_pairs.append((i, j, k_star))

        # Also add random pairs for exploration
        for _ in range(n // 2):
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j:
                candidate_pairs.append((i, j, -1))

        # Shuffle
        rng.shuffle(candidate_pairs)

        # Process a subset (time-limited)
        max_pairs_to_try = min(len(candidate_pairs), 2000)
        for idx in range(max_pairs_to_try):
            if time.monotonic() - t0 > time_limit:
                break

            i, j, _ = candidate_pairs[idx]
            if a[i] < norm * 1e-9:
                continue

            # Try transferring mass from i to j (and vice versa)
            # Use a few trial deltas
            ai = a[i]
            best_d = 0.0
            best_local_max = conv.max()

            # Try transferring from i to j
            for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                d = ai * frac
                # Approximate new conv
                # Change in conv[k] = d * (contribution of j - contribution of i)
                # For k: delta_c[k] = d * (a[k-j] + a[j-k] - a[k-i] - a[i-k])
                # Quick check: just evaluate at the top few positions
                new_conv_approx = conv.copy()
                # Update affected positions
                for k in range(max(0, i), min(size, i + n)):
                    if 0 <= k - i < n:
                        new_conv_approx[k] -= d * a[k - i]
                    if 0 <= i - k < n:
                        new_conv_approx[k] -= d * a[i - k]
                for k in range(max(0, j), min(size, j + n)):
                    if 0 <= k - j < n:
                        new_conv_approx[k] += d * a[k - j]
                    if 0 <= j - k < n:
                        new_conv_approx[k] += d * a[j - k]

                approx_max = new_conv_approx.max()
                if approx_max < best_local_max:
                    # Verify with exact computation
                    a[i] -= d
                    a[j] += d
                    conv_test = fast_autocorr(a)
                    test_max = conv_test.max()
                    if test_max < best_local_max - 1e-12:
                        best_local_max = test_max
                        best_d = d
                        conv = conv_test
                    else:
                        a[i] += d
                        a[j] -= d
            # Also try transferring from j to i
            if a[j] > norm * 1e-9:
                aj = a[j]
                for frac in [0.05, 0.1, 0.2, 0.5]:
                    d = aj * frac
                    a[i] += d
                    a[j] -= d
                    conv_test = fast_autocorr(a)
                    test_max = conv_test.max()
                    if test_max < best_local_max - 1e-12:
                        best_local_max = test_max
                        best_d = -d  # negative means j -> i
                        conv = conv_test
                    else:
                        a[i] -= d
                        a[j] += d

            if best_d != 0.0:
                s = a.sum()
                if s > 1e-15:
                    a = a * norm / s
                if best_local_max < best_score - 1e-12:
                    best_score = best_local_max
                    best_a = a.copy()
                    pass_improved = True
                    no_improve = 0

        if not pass_improved:
            # Try random perturbation to escape
            if no_improve >= 2:
                pert = rng.normal(0, norm * 0.01, n)
                a_pert = a + pert
                a_pert = np.maximum(a_pert, 0.0)
                s = a_pert.sum()
                if s > 1e-15:
                    a_pert = a_pert * norm / s
                    conv_pert = fast_autocorr(a_pert).max()
                    if conv_pert < best_score * 1.1:
                        a = a_pert
                        conv = fast_autocorr(a)
                break
            no_improve += 1

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
        a = np.maximum(a, 0)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        x = np.linspace(0.0, 1.0, n)
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        # Number-theoretic: quadratic residue pattern
        a = np.zeros(n)
        for i in range(n):
            a[i] = 1.0 if ((i * i) % n) < n // 2 else 0.3
    else:
        # Multi-scale
        x = np.linspace(0.0, 1.0, n)
        a = 0.4 + 0.3 * np.sin(2 * np.pi * x) ** 2 + 0.2 * np.sin(6 * np.pi * x) ** 2
        a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function. Tries multiple strategies to minimize the objective."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # === Phase 1: LP Nesterov at base length ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.22, 350.0)
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n_base}")

    # === Phase 2: Hinge-loss gradient (NEW - main improvement) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Hinge-loss gradient n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = hinge_loss_optimize(a_norm, t2, verbose=True)
        do_try_update(a_out, f"hinge_n{n_best}")

    # === Phase 3: Second-moment smoothing (NEW) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: 2nd moment n={n_best} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = second_moment_optimize(a_norm, t3, verbose=True)
        do_try_update(a_out, f"2ndmom_n{n_best}")

    # === Phase 4: Peak-guided pair CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Peak-guided PCD n={n_best} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_guided_pair_cd(a_norm, t4, rng, verbose=True)
        do_try_update(a_out, f"pcd_n{n_best}")

    # === Phase 5: Hinge loss again (refinement after CD) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Hinge refinement n={n_best} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.20, 250.0)
    if t5 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = hinge_loss_optimize(a_norm, t5, verbose=True)
        do_try_update(a_out, f"hinge2_n{n_best}")

    # === Phase 6: Try different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    lengths_to_try = []
    for delta in [200, -200, 400, -400, 600]:
        n_try = n_best + delta
        if n_try >= 400 and n_try <= 4000:
            lengths_to_try.append(n_try)

    for n_try in lengths_to_try[:3]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        print(f"\n=== Phase 6: Multi-length n={n_try} (budget={remaining:.0f}s) ===")
        t6 = min(remaining * 0.25, 200.0)
        if t6 > 30:
            init_try = interpolate_sequence(best_seq, n_try)
            a_norm = init_try * np.sqrt(2.0 * n_try) / init_try.sum()
            # Quick hinge loss
            a_out, _ = hinge_loss_optimize(a_norm, t6 * 0.5, verbose=False)
            do_try_update(a_out, f"hinge_n{n_try}")
            # Quick LP
            remaining = time_budget - (time.monotonic() - t_global)
            t6b = min(remaining * 0.20, 100.0)
            if t6b > 20:
                a_out2, _ = nesterov_lp_optimize(a_norm, t6b, solver, verbose=False)
                do_try_update(a_out2, f"lp_n{n_try}")

    # === Phase 7: Final intensive pair CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final PCD n={n_best} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.30, 200.0)
    if t7 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_guided_pair_cd(a_norm, t7, rng, verbose=True)
        do_try_update(a_out, f"pcd_final_n{n_best}")

    # === Phase 8: Final hinge polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final hinge polish n={n_best} (budget={remaining:.0f}s) ===")
    t8 = min(remaining * 0.50, 150.0)
    if t8 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = hinge_loss_optimize(a_norm, t8, verbose=True)
        do_try_update(a_out, f"hinge_final_n{n_best}")

    # === Phase 9: Second moment final ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        print(f"\n=== Phase 9: Final 2ndMom n={n_best} (budget={remaining:.0f}s) ===")
        t9 = min(remaining * 0.60, 100.0)
        if t9 > 10:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = second_moment_optimize(a_norm, t9, verbose=False)
            do_try_update(a_out, f"2ndmom_final_n{n_best}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
