def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

    n = len(a_fixed)

    return T

                    best_score = score_l
                    best_a = avg_l.copy()
                    stall_count = 0

            # Also check geometric mean of a and b_star
            geo = np.sqrt(np.maximum(a * b_star, 0))
            gs = geo.sum()
            if gs > 1e-15:
                geo = geo / gs
                score_g = 2.0 * n * fast_autocorr(geo).max()
                if score_g < best_score:
                    best_score = score_g
                    best_a = geo.copy()
                    stall_count = 0

        # Restart if stalled
        if stall_count > 400 and t > 100:
            a = best_a.copy()
            y = best_a.copy()
            avg_short = np.zeros(n)
            n_short = 0
            avg_long = np.zeros(n)
            n_long = 0
            short_buf = [None] * WINDOW_SHORT
            short_idx = 0
            long_buf = [None] * WINDOW_LONG
            long_idx = 0
            stall_count = 0
            warm_b = None

        if t - last_print > 1000:
            elapsed = time.monotonic() - t0
            print(f"  LP t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}")
            last_print = t

    return best_a, best_score


def direct_subgradient_max(a0, time_limit, seed=123):
    """
    Direct subgradient method on the max autocorrelation peak.
    Uses multiple peak averaging and adaptive step sizes.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    eta = 0.02
    patience = 0
    consecutive_no_improve = 0
    n_peaks = 5  # number of peaks to average over

    # Nesterov momentum
    v = np.zeros(n)

    while time.monotonic() - t0 < time_limit - 2:
        b = fast_autocorr(a)
        bmax = b.max()

        # Get top peaks
        top_k = np.argsort(b)[::-1][:n_peaks]
        # Weight peaks by their values (normalized)
        w = b[top_k].copy()
        w = w / w.sum()

        # Compute averaged subgradient
        subgrad = np.zeros(n)
        for i, k in enumerate(top_k):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            subgrad[idx] += w[i] * 2.0 * a[k - idx]

        # Nesterov step
        v = 0.9 * v - eta * subgrad
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        score_new = 2.0 * n * fast_autocorr(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            patience = 0
            consecutive_no_improve = 0
        else:
            a = a_new
            patience += 1
            consecutive_no_improve += 1

        # Adaptive step size
        if consecutive_no_improve % 50 == 0:
            eta *= 0.92
            if eta < 1e-5:
                eta = 0.02
                # Perturb
                noise = rng.normal(0, 0.005, n)
                a = a * (1 + noise)
                a = normalize(a)
                v = np.zeros(n)
                consecutive_no_improve = 0

        if patience > 2000:
            break

    return best_a, best_score


def multi_scale_peak_shave(a0, time_limit, seed=42):
    """
    Multi-scale peak shaving: tries moves at different granularities.
    Includes swap moves and block moves in addition to transfer moves.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 3:
        b = fast_autocorr(a)
        bmax = b.max()
        threshold = bmax * 0.985
        top_peaks = np.where(b >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(b)[::-1][:8]

        improved = False

        # Strategy 1: Transfer from high-contributing source to safe destination
        for k in top_peaks[:8]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            # Danger zone
            danger = np.zeros(n, dtype=bool)
            for kp in top_peaks[:12]:
                js2 = max(0, kp - n + 1)
                je2 = min(n - 1, kp)
                danger[js2:je2 + 1] = True

            safe = np.where(~danger)[0]
            safe = safe[safe != src]
            if len(safe) == 0:
                safe = np.where(np.arange(n) != src)[0]
            safe_sorted = safe[np.argsort(a[safe])]
            dst_pool = safe_sorted[:max(3, len(safe_sorted) // 5)]

            found = False
            for dst in dst_pool[:6]:
                for frac in [0.003, 0.008, 0.015, 0.03, 0.05, 0.08]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-12:
                        continue
                    a_trial = a_trial / st
                    new_max = fast_autocorr(a_trial).max()
                    if new_max < bmax * (1 - 1e-9):
                        a = a_trial
                        new_score = 2.0 * n * new_max
                        if new_score < best_score:
                            best_score = new_score
                            best_a = a.copy()
                        improved = True
                        stall = 0
                        found = True
                        break
                if found:
                    break
            if improved:
                break

        # Strategy 2: Swap move
        if not improved:
            for k in top_peaks[:5]:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                if len(idx) < 2:
                    continue
                contrib = a[idx] * a[k - idx]
                src = idx[np.argmax(contrib)]

                # Find a position with similar value to swap with
                potential = np.where(np.abs(a - a[src]) < 0.3 * a[src])[0]
                potential = potential[potential != src]
                rng.shuffle(potential)
                for dst in potential[:5]:
                    a_trial = a.copy()
                    a_trial[src], a_trial[dst] = a_trial[dst], a_trial[src]
                    new_max = fast_autocorr(a_trial).max()
                    if new_max < bmax * (1 - 1e-9):
                        a = a_trial
                        new_score = 2.0 * n * new_max
                        if new_score < best_score:
                            best_score = new_score
                            best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            if stall % 15 == 0:
                # Random perturbation
                n_pert = rng.integers(1, 4)
                a_trial = a.copy()
                for _ in range(n_pert):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a_trial[i] > 1e-10:
                        frac = rng.uniform(0.002, 0.05)
                        amt = a_trial[i] * frac
                        a_trial[i] -= amt
                        a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-10:
                    a_trial = a_trial / st
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > 300:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length using cubic spline."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    # Use linear interpolation (cubic can introduce negatives)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def fourier_informed_init(n, base_seq=None, seed=42):
    """
    Create an initialization that tries to have flat power spectrum.
    Uses the DFT to guide the construction.
    """
    rng = np.random.default_rng(seed)
    if base_seq is not None:
        a = normalize(interpolate_sequence(base_seq, n))
    else:
        # Start with a slightly modulated uniform
        x = np.linspace(0, 2 * np.pi, n)
        a = 1.0 + 0.1 * np.cos(x) + 0.05 * np.sin(2 * x)
        a = np.maximum(a, 0)
        a = a / a.sum()

    # Try to flatten the power spectrum by adjusting phases in Fourier domain
    # For real signals, we can't directly control phases, but we can
    # use the DFT magnitude as a guide
    for _ in range(5):
        A = rfft(a)
        mag = np.abs(A)
        # The power spectrum is mag^2; we want it flat
        # Gradient of var(mag^2) w.r.t. a can be computed via FFT
        # Simple approach: multiply by inverse-magnitude in Fourier domain
        target_mag = np.ones_like(mag)
        target_mag[0] = mag[0]  # keep DC
        # Scale factor
        scale = mag.sum() / target_mag.sum()
        # Adjust: a_new = ifft(A * (target_mag/mag)^alpha)
        alpha = 0.3
        ratio = target_mag / np.maximum(mag, 1e-10)
        ratio = ratio ** alpha
        A_new = A * ratio
        a_new = irfft(A_new, n)
        # Project to non-negative simplex
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a = a_new / s
        else:
            break

    return a


def combined_polish(a0, time_limit, seed=777):
    """
    Combined polishing: alternates between subgradient steps and peak shaving.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    eta = 0.01
    phase = 0  # 0 = subgradient, 1 = peak shave
    phase_counter = 0
    patience = 0

    while time.monotonic() - t0 < time_limit - 3:
        if phase == 0:
            # Subgradient step
            b = fast_autocorr(a)
            bmax = b.max()
            top_k = np.argsort(b)[::-1][:3]
            w = b[top_k].copy()
            w = w / w.sum()

            subgrad = np.zeros(n)
            for i, k in enumerate(top_k):
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                subgrad[idx] += w[i] * 2.0 * a[k - idx]

            a_new = a - eta * subgrad
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new / s
                score = 2.0 * n * fast_autocorr(a_new).max()
                if score < best_score:
                    best_score = score
                    best_a = a_new.copy()
                    a = a_new
                    patience = 0
                else:
                    patience += 1
            else:
                patience += 1

            if patience > 50:
                phase = 1
                patience = 0
                phase_counter = 0

        else:
            # Quick peak shave (one iteration)
            b = fast_autocorr(a)
            bmax = b.max()
            top_peaks = np.argsort(b)[::-1][:5]
            improved = False

            for k in top_peaks[:3]:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                if len(idx) == 0:
                    continue
                contrib = a[idx] * a[k - idx]
                if contrib.max() < 1e-15:
                    continue
                src = idx[np.argmax(contrib)]
                # Move to lowest-value position not in danger
                dst = np.argmin(a)
                if dst == src:
                    dst = np.argmin(np.where(np.arange(n) == src, 1e10, a))

                for frac in [0.01, 0.02, 0.04]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-12:
                        continue
                    a_trial = a_trial / st
                    new_max = fast_autocorr(a_trial).max()
                    if new_max < bmax * (1 - 1e-9):
                        a = a_trial
                        new_score = 2.0 * n * new_max
                        if new_score < best_score:
                            best_score = new_score
                            best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

            phase_counter += 1
            if phase_counter >= 3 or not improved:
                phase = 0
                phase_counter = 0
                if not improved:
                    patience += 1

        if patience > 500:
            break

    return best_a, best_score


def propose_candidate():
    """
    Multi-phase optimization:
    Phase 1: LP Nesterov at n=1600 (refine parent)
    Phase 2: LP Nesterov at n=1800 (primary exploration)
    Phase 3: LP Nesterov at n=2000 (extended exploration)
    Phase 4: Direct subgradient at best length
    Phase 5: Multi-scale peak shaving
    Phase 6: Combined polish
    """
    t_global = time.monotonic()
    TIME_BUDGET = 1700.0  # Leave 100s buffer

    solver = find_solver()
    print(f"Solver: {solver}, budget: {TIME_BUDGET:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Get parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None
        n_default = 1600
        best_seq = np.ones(n_default) / n_default
        best_score = 2.0 * n_default * fast_autocorr(best_seq).max()
        print(f"Default init: n={n_default}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    elapsed = time.monotonic() - t_global
    remaining = TIME_BUDGET - elapsed

    # === Phase 1: Quick LP at parent length (15%) ===
    n_base = len(best_seq)
    t1 = min(remaining * 0.12, 180.0)
    if t1 > 20:
        print(f"\n=== Phase 1: LP Nesterov n={n_base} ({t1:.0f}s) ===")
        a_out, _ = nesterov_lp_optimize(best_seq, t1, solver)
        try_update(a_out, f"lp1_n{n_base}")

    elapsed = time.monotonic() - t_global
    remaining = TIME_BUDGET - elapsed

    # === Phase 2: LP at n=1800 (30%) ===
    n_1800 = 1800
    t2 = min(remaining * 0.30, 500.0)
    if t2 > 30:
        print(f"=== Phase 2: LP Nesterov n={n_1800} ({t2:.0f}s) ===")
        # Initialize with Fourier-informed approach
        init_1800 = interpolate_sequence(best_seq, n_1800)
        a_out, _ = nesterov_lp_optimize(init_1800, t2, solver)
        try_update(a_out, f"lp2_n{n_1800}")

    elapsed = time.monotonic() - t_global
    remaining = TIME_BUDGET - elapsed

    # === Phase 3: Subgradient at best length (15%) ===
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 20:
        print(f"=== Phase 3: Subgradient n={len(best_seq)} ({t3:.0f}s) ===")
        a_out, _ = direct_subgradient_max(best_seq, t3, seed=456)
        try_update(a_out, f"subgrad_n{len(best_seq)}")

    elapsed = time.monotonic() - t_global
    remaining = TIME_BUDGET - elapsed

    # === Phase 4: Peak shaving at best length (15%) ===
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 15:
        print(f"=== Phase 4: Peak shave n={len(best_seq)} ({t4:.0f}s) ===")
        a_out, _ = multi_scale_peak_shave(best_seq, t4, seed=888)
        try_update(a_out, f"peak_n{len(best_seq)}")

    elapsed = time.monotonic() - t_global
    remaining = TIME_BUDGET - elapsed

    # === Phase 5: LP at n=2000 (15%) ===
    n_2000 = 2000
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 30:
        print(f"=== Phase 5: LP Nesterov n={n_2000} ({t5:.0f}s) ===")
        init_2000 = interpolate_sequence(best_seq, n_2000)
        a_out, _ = nesterov_lp_optimize(init_2000, t5, solver)
        try_update(a_out, f"lp5_n{n_2000}")

    elapsed = time.monotonic() - t_global
    remaining = TIME_BUDGET - elapsed

    # === Phase 6: Combined polish (15%) ===
    t6 = min(remaining * 0.20, 200.0)
    if t6 > 15:
        print(f"=== Phase 6: Combined polish n={len(best_seq)} ({t6:.0f}s) ===")
        a_out, _ = combined_polish(best_seq, t6, seed=999)
        try_update(a_out, f"polish_n{len(best_seq)}")

    elapsed = time.monotonic() - t_global
    remaining = TIME_BUDGET - elapsed

    # === Phase 7: Final LP refinement at best length (remaining) ===
    t7 = min(remaining * 0.50, 150.0)
    if t7 > 15:
        print(f"=== Phase 7: Final LP polish n={len(best_seq)} ({t7:.0f}s) ===")
        a_out, _ = nesterov_lp_optimize(best_seq, t7, solver)
        try_update(a_out, f"final_lp_n{len(best_seq)}")

    elapsed = time.monotonic() - t_global
    remaining = TIME_BUDGET - elapsed

    # === Phase 8: Final peak shave ===
    t8 = min(remaining * 0.50, 60.0)
    if t8 > 5:
        print(f"=== Phase 8: Final peak shave ({t8:.0f}s) ===")
        a_out, _ = multi_scale_peak_shave(best_seq, t8, seed=1234)
        try_update(a_out, "final_peak")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        total_time = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={total_time:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
