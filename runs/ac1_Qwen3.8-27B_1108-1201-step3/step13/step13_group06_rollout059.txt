```python
def subgradient_step(a, top_k_peaks, step_size):
    """Compute and apply subgradient step."""
    n = len(a)
    b = fast_autocorr(a)
    # Get top peaks
    top_idx = np.argsort(b)[::-1][:top_k_peaks]
    
    # Compute subgradient: average of 2*Q_k*a over top peaks
    grad = np.zeros(n)
    for k in top_idx:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        # (Q_k * a)[j] = a[k-j] for j in [js, je]
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * a[k - idx]
    grad /= len(top_idx)
    
    # Step
    a_new = a - step_size * grad
    
    # Project onto simplex (sum=1, a>=0)
    a_new = project_simplex(a_new)
    return a_new
```

The simplex projection can be done as:
```python
def project_simplex(v, target_sum=1.0):
    """Project v onto {x : sum(x) = target_sum, x >= 0}."""
    n = len(v)
    # Lagrangian: min 0.5*||x-v||^2 s.t. sum(x)=s, x>=0
    # Solution: x = max(v - t, 0) where t is chosen so sum(x) = s
    # Use bisection or sorting
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cv - target_sum))[0][-1]
    tau = (cv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - tau, 0)
```

The learning rate for projected gradient on a convex quadratic is `lr = 2 / L` where L is the Lipschitz constant of the gradient, which is `2 * lambda_max(S)`. Since S is PSD and `S = sum q_k Q_k` with each Q_k having eigenvalues at most 1 (actually, let me check)...

Since each `Q_k` is a submatrix of a permutation matrix (it has at most one 1 in each row and column), its spectral norm is at most 1. So `lambda_max(S) <= 1`.

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov LP zero-sum game (quick start)
2. Multiplicative Weights game with Toeplitz QP (main driver)
3. Nesterov-accelerated subgradient with active set tracking
4. Greedy peak shaving with quadratic move optimization
5. Multi-length exploration with interpolation
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    """Project v onto {x : sum(x) = target_sum, x >= 0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def toep_mult(q, a):
    """Compute S @ a where S[j,l] = q[j+l]. q has length 2n-1, a has length n."""
    n = len(a)
    # convolve(q_full, a[::-1])[n-1 : 2n-1]
    a_rev = a[::-1].copy()
    fft_size = 1 << (len(q) + n - 1).bit_length()
    Q = rfft(q, fft_size)
    A = rfft(a_rev, fft_size)
    result = irfft(Q * A, fft_size)
    return result[n-1:2*n-1]

def solve_qp_toep(q, a_init, n_iters=30, lr=0.9):
    """Solve min a^T S a s.t. sum(a)=1, a>=0 where S[j,l]=q[j+l]."""
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    for _ in range(n_iters):
        grad = 2.0 * toep_mult(q, a)
        a_new = a - lr * grad
        a_new = project_simplex(a_new, 1.0)
        if np.allclose(a_new, a, atol=1e-12):
            a = a_new
            break
        a = a_new
    return a

def mw_game(a_init, time_limit, eta=None, max_iters=100000, n_qp_iters=30):
    """Multiplicative weights zero-sum game."""
    n = len(a_init)
    K = 2 * n - 1  # number of positions
    t0 = time.monotonic()
    
    if eta is None:
        eta = np.sqrt(2.0 * np.log(K) / 1000.0)
    
    # Initialize q uniformly
    q = np.ones(K) / K
    
    # Initialize a
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    # Running average
    a_avg = np.zeros(n)
    N = 0
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    
    last_print = 0
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve QP with current q
        a_new = solve_qp_toep(q, a, n_iters=n_qp_iters)
        a = a_new
        
        # Update running average
        a_avg = a_avg * N / (N + 1) + a / (N + 1)
        N += 1
        
        # Compute payoff
        b = fast_autocorr(a)
        bmax = b.max()
        score_a = 2.0 * n * bmax
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        score_avg = 2.0 * n * fast_autocorr(a_avg).max()
        if score_avg < best_score:
            best_score = score_avg
            best_a = a_avg.copy()
        
        # Update q with multiplicative weights
        # Normalize b for numerical stability
        b_norm = b / b.max()
        q = q * np.exp(eta * b_norm)
        qs = q.sum()
        if qs < 1e-300:
            q = np.ones(K) / K
        else:
            q = q / qs
        
        if t - last_print > 2000:
            elapsed = time.monotonic() - t0
            print(f"      MW t={t} ({elapsed:.0f}s): best={best_score:.6f}, q_max={q.max():.4e}")
            last_print = t
    
    return best_a, best_score

def subgradient_nesterov(a_init, time_limit, topk=20, max_iters=50000):
    """Nesterov-accelerated subgradient method for max of quadratic forms."""
    n = len(a_init)
    t0 = time.monotonic()
    
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    y = a.copy()  # Nesterov lookup point
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    
    a_avg = np.zeros(n)
    N = 0
    stall = 0
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute subgradient at y (lookup point)
        b = fast_autocorr(y)
        top_idx = np.argsort(b)[::-1][:topk]
        
        # Subgradient: average of 2*Q_k*a for top k's
        grad = np.zeros(n)
        for k in top_idx:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * y[k - idx]
        grad /= len(top_idx)
        
        # Nesterov step
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * a  # wait, this isn't right
        
        # Actually, standard Nesterov:
        # g_t = subgradient at y_t
        # a_{t+1} = project(a_t - lr_t * g_t)
        # y_{t+1} = a_{t+1} + ((t)/(t+2)) * (a_{t+1} - a_t)
        
        lr_t = 1.0 / np.sqrt(t + 1)  # or some other schedule
        a_candidate = a - lr_t * grad
        a_candidate = project_simplex(a_candidate, 1.0)
        
        momentum = (t - 1.0) / (t + 1.0) if t > 1 else 0.0
        y = a_candidate + momentum * (a_candidate - a)
        y = project_simplex(y, 1.0)  # keep y in the feasible set
        
        a = a_candidate
        
        # Update average
        a_avg = a_avg * N / (N + 1) + a / (N + 1)
        N += 1
        
        # Check score
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        score_avg = 2.0 * n * fast_autocorr(a_avg).max()
        if score_avg < best_score:
            best_score = score_avg
            best_a = a_avg.copy()
            stall = 0
        
        # Restart if stalled
        if stall > 1000:
            a = best_a.copy()
            y = best_a.copy()
            a_avg = np.zeros(n)
            N = 0
            stall = 0
    
    return best_a, best_score

def peak_shaving_quadratic(a_init, time_limit, seed=42):
    """Peak shaving with quadratic move optimization."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit - 2:
        b = fast_autocorr(a)
        bmax = b.max()
        
        # Find top peaks
        top_peaks = np.argsort(b)[::-1][:10]
        
        improved = False
        for k in top_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) < 2:
                continue
            
            # Find the pair (i, j) that maximally contributes to peak k
            contrib = a[idx] * a[k - idx]
            
            # Source: position with highest contribution
            src = idx[np.argmax(contrib)]
            
            # Destination: position where a[k-j] is smallest (reduces peak k most)
            # But also need to not increase other peaks too much
            # Simple heuristic: move to a position far from the peak
            candidates = np.arange(n)
            candidates = candidates[candidates != src]
            # Score each candidate by how much it reduces peak k
            scores = a[k - src] - np.where((k - candidates >= 0) & (k - candidates < n), a[k - candidates], a[k - src])
            # Sort by reduction
            cand_sorted = candidates[np.argsort(scores)]
            
            for dst in cand_sorted[:5]:
                # Optimal d: minimize (a[src]-d)*(a[k-src]-d) + (a[dst]+d)*(a[k-dst]+d) + const
                # = const + d*(a[k-dst] - a[k-src]) + d^2*(1 - 1) [if k-src != src and k-dst != dst]
                # First order: d* = (a[k-src] - a[k-dst]) / 2 (for the pair (src, dst))
                # But need 0 < d < a[src]
                
                aks = a[k - src] if 0 <= k - src < n else 0.0
                akd = a[k - dst] if 0 <= k - dst < n else 0.0
                
                # Linear approximation: d* = (aks - akd) / 2
                d_opt = (aks - akd) / 2.0
                d_opt = max(d_opt, 0.0)
                d_opt = min(d_opt, a[src] * 0.5)
                
                if d_opt < 1e-12:
                    continue
                
                # Try the optimal d and a few nearby values
                for d in [d_opt, d_opt * 0.5, d_opt * 0.2]:
                    if d < 1e-12 or d >= a[src]:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= d
                    a_trial[dst] += d
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-12:
                        continue
                    a_trial = a_trial / st
                    new_max = fast_autocorr(a_trial).max()
                    if new_max < bmax * (1 - 1e-10):
                        a = a_trial
                        new_score = 2.0 * n * new_max
                        if new_score < best_score:
                            best_score = new_score
                            best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall % 20 == 0:
                # Random perturbation
                n_pert = rng.integers(1, 4)
                a_trial = a.copy()
                for _ in range(n_pert):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a_trial[i] > 1e-8:
                        frac = rng.uniform(0.001, 0.02)
                        d = a_trial[i] * frac
                        a_trial[i] -= d
                        a_trial[j] += d
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-10:
                    a_trial = a_trial / st
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score * 1.01:  # allow slight increase to escape
                        a = a_trial
                        stall = 0
            if stall > 200:
                break
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
```

For a palindromic sequence, the center peak is `sum(a[j]^2) >= 1/n`.

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multiplicative Weights zero-sum game with Toeplitz QP (main driver)
2. Nesterov-accelerated subgradient with active set tracking
3. Greedy peak shaving with quadratic move optimization
4. Multi-length exploration with interpolation
5. Multi-restart MW game with annealed learning rates
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Project v onto {x : sum(x) = target_sum, x >= 0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def toep_mult(q, a):
    """Compute S @ a where S[j,l] = q[j+l]. q has length 2n-1, a has length n."""
    n = len(a)
    a_rev = a[::-1].copy()
    conv_len = len(q) + n - 1
    fft_size = 1 << (conv_len).bit_length()
    Q = rfft(q, fft_size)
    A = rfft(a_rev, fft_size)
    result = irfft(Q * A, fft_size)
    return result[n - 1:2 * n - 1]


def solve_qp_toep(q, a_init, n_iters=40, lr=0.85):
    """Solve min a^T S a s.t. sum(a)=1, a>=0 where S[j,l]=q[j+l] using projected gradient."""
    a = np.maximum(a_init.copy(), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    for _ in range(n_iters):
        grad = 2.0 * toep_mult(q, a)
        a_new = a - lr * grad
        a_new = project_simplex(a_new, 1.0)
        diff = np.max(np.abs(a_new - a))
        a = a_new
        if diff < 1e-13:
            break
    return a


def mw_game(a_init, time_limit, eta_init=None, eta_end=None, max_iters=200000, n_qp_iters=35):
    """
    Multiplicative weights zero-sum game.
    Player 1 (a): minimize max_k conv(a,a)[k]
    Player 2 (q): choose distribution over peak positions
    """
    n = len(a_init)
    K = 2 * n - 1
    t0 = time.monotonic()

    if eta_init is None:
        eta_init = np.sqrt(2.0 * np.log(K) / 500.0)
    if eta_end is None:
        eta_end = eta_init * 0.1

    q = np.ones(K) / K
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    a_avg = np.zeros(n)
    N = 0
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    last_print = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive eta: anneal from eta_init to eta_end
        frac = min(1.0, t / max(t0_iters, 1)) if 't0_iters' in dir() else t / max(max_iters, 1)
        frac = min(1.0, t / 5000.0)
        eta_t = eta_init * (eta_end / eta_init) ** frac

        # Solve QP with current q
        a_new = solve_qp_toep(q, a, n_iters=n_qp_iters)
        a = a_new

        # Update running average
        a_avg = a_avg * N / (N + 1) + a / (N + 1)
        N += 1

        # Compute payoff
        b = fast_autocorr(a)
        bmax = b.max()
        score_a = 2.0 * n * bmax
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        if N % 5 == 0:
            score_avg = 2.0 * n * fast_autocorr(a_avg).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = a_avg.copy()

        # Update q with multiplicative weights (Player 2 maximizes)
        b_norm = b / max(bmax, 1e-300)
        q = q * np.exp(eta_t * b_norm)
        qs = q.sum()
        if qs < 1e-300 or not np.isfinite(qs):
            q = np.ones(K) / K
        else:
            q = q / qs

        if t - last_print > 3000:
            elapsed = time.monotonic() - t0
            print(f"      MW t={t} ({elapsed:.0f}s): best={best_score:.6f}, q_max={q.max():.2e}, eta={eta_t:.4e}")
            last_print = t

    return best_a, best_score


def subgradient_nesterov(a_init, time_limit, topk=15, max_iters=200000):
    """Nesterov-accelerated subgradient for max of quadratic forms."""
    n = len(a_init)
    t0 = time.monotonic()

    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    a_avg = np.zeros(n)
    N = 0
    stall = 0
    last_print = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute subgradient at y
        b = fast_autocorr(y)
        top_idx = np.argsort(b)[::-1][:topk]

        # Subgradient: average of 2*Q_k*y for top k's
        grad = np.zeros(n)
        for k in top_idx:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * y[k - idx]
        grad /= len(top_idx)

        # Nesterov step
        lr_t = 0.5 / np.sqrt(t + 1.0)
        a_candidate = a - lr_t * grad
        a_candidate = project_simplex(a_candidate, 1.0)

        momentum = (t - 1.0) / (t + 1.0) if t > 1 else 0.0
        y = a_candidate + momentum * (a_candidate - a)
        y = np.maximum(y, 0)
        ys = y.sum()
        if ys < 1e-15:
            y = a_candidate.copy()
        else:
            y = y / ys

        a = a_candidate

        # Update average
        a_avg = a_avg * N / (N + 1) + a / (N + 1)
        N += 1

        # Check score (every few iterations for speed)
        if N % 3 == 0 or N < 10:
            score_a = 2.0 * n * fast_autocorr(a).max()
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()
                stall = 0
            else:
                stall += 1

            if N % 10 == 0:
                score_avg = 2.0 * n * fast_autocorr(a_avg).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = a_avg.copy()
                    stall = 0

        # Restart if stalled
        if stall > 500 and t > 50:
            a = best_a.copy()
            y = best_a.copy()
            a_avg = np.zeros(n)
            N = 0
            stall = 0

        if t - last_print > 5000:
            elapsed = time.monotonic() - t0
            print(f"      SG t={t} ({elapsed:.0f}s): best={best_score:.6f}")
            last_print = t

    return best_a, best_score


def peak_shaving_quadratic(a_init, time_limit, seed=42):

    n = len(a_init)

    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0)
    s = a.sum()

    last_print = 0

    while time.monotonic() - t0 < time_limit - 2:
        iter_count += 1
        b = fast_autocorr(a)
        bmax = b.max()

        # Find top peaks to target
        top_peaks = np.argsort(b)[::-1][:8]

        improved = False
        for k in top_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) < 2:
                continue

            # Contribution at each position
            contrib = a[idx] * a[k - idx]
            src = idx[np.argmax(contrib)]

            # Find best destination
            # We want dst such that a[k-dst] is small (reduces this peak)
            # and moving there doesn't increase other peaks too much
            candidates = np.setdiff1d(np.arange(n), [src])
            # Score: reduction in peak k = a[k-src] - a[k-dst]
            k_minus = k - candidates
            valid = (k_minus >= 0) & (k_minus < n)
            scores = np.zeros(len(candidates))
            scores[valid] = a[k - src] - a[k_minus[valid]]
            scores[~valid] = a[k - src]  # moving to invalid position has no benefit for this peak
            cand_sorted = candidates[np.argsort(scores)[::-1]]

            for dst in cand_sorted[:6]:
                akd = a[k - dst] if 0 <= k - dst < n else 0.0
                aks = a[k - src] if 0 <= k - src < n else 0.0

                # Optimal d from linearization: d* = (aks - akd) / 2
                d_opt = (aks - akd) / 2.0
                d_opt = max(d_opt, 0.0)
                d_opt = min(d_opt, a[src] * 0.4)

                for d in [d_opt, d_opt * 0.5, d_opt * 0.25]:
                    if d < 1e-12 or d >= a[src] - 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= d
                    a_trial[dst] += d
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-12:
                        continue
                    a_trial = a_trial / st
                    new_max = fast_autocorr(a_trial).max()
                    if new_max < bmax * (1 - 1e-10):
                        a = a_trial
                        new_score = 2.0 * n * new_max
                        if new_score < best_score:
                            best_score = new_score
                            best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 15 == 0 and stall < 300:
                # Random perturbation to escape local minimum
                n_pert = rng.integers(1, 5)
                a_trial = a.copy()
                for _ in range(n_pert):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a_trial[i] > 1e-8:

                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-10:
                    a_trial = a_trial / st
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score * 1.005:
                        a = a_trial
                        stall = 0
            if stall > 300:
                break

        if iter_count - last_print > 5000:
            elapsed = time.monotonic() - t0
            print(f"      PS iter={iter_count} ({elapsed:.0f}s): best={best_score:.6f}")

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n, endpoint=False)
    new_x = np.linspace(0, 1, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

                a_out2, _ = subgradient_nesterov(best_seq, t2b, topk=25)
                try_update(a_out2, "subgrad_25k")

    # === Phase 3: Peak Shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Peak Shaving n={n_base} ===")
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 15:
        a_out, _ = peak_shaving_quadratic(best_seq, t3, seed=777)
        try_update(a_out, "peak_shave_1")

    # === Phase 4: Mirror Descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Mirror Descent n={n_base} ===")
    t4 = min(remaining * 0.08, 100.0)
    if t4 > 10:
        a_out, _ = mirror_descent_refine(best_seq, t4, eta=0.008, seed=555)
        try_update(a_out, "mirror_1")

    # === Phase 5: Longer length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Length Exploration ===")
    for delta_n in [100, 200]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 40:
            break
        n_long = n_base + delta_n
        t5 = min(remaining * 0.12, 150.0)
        if t5 > 30:
            print(f"  Trying n={n_long}...")
            init_long = interpolate_sequence(best_seq, n_long)
            a_out, _ = mw_game(init_long, t5, eta_init=0.04, eta_end=0.003)
            try_update(a_out, f"mw_n{n_long}")

            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 30:
                t5b = min(remaining * 0.05, 60.0)
                if t5b > 15:
                    a_out2, _ = peak_shaving_quadratic(best_seq if len(best_seq) == n_base else a_out, t5b, seed=123)
                    if len(a_out2) == n_long:
                        try_update(a_out2, f"ps_n{n_long}")

    # === Phase 6: Shorter length (sometimes shorter is better) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        n_short = max(800, n_base - 100)
        t6 = min(remaining * 0.08, 100.0)
        if t6 > 20:
            print(f"\n=== Phase 6: Shorter length n={n_short} ===")
            init_short = interpolate_sequence(best_seq, n_short)
            a_out, _ = mw_game(init_short, t6, eta_init=0.05, eta_end=0.005)
            try_update(a_out, f"mw_n{n_short}")

    # === Phase 7: Final MW polish with very low eta ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final MW Polish ===")
    t7 = min(remaining * 0.20, 200.0)
    if t7 > 20:
        a_out, _ = mw_game(best_seq, t7, eta_init=0.01, eta_end=0.001, n_qp_iters=50)
        try_update(a_out, "mw_final")

    # === Phase 8: Final peak shave ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        t8 = min(remaining * 0.30, 100.0)
        if t8 > 10:
            print(f"\n=== Phase 8: Final Peak Shave ===")
            a_out, _ = peak_shaving_quadratic(best_seq, t8, seed=2001)
            try_update(a_out, "ps_final")

    # === Phase 9: Final subgradient micro-polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        t9 = min(remaining * 0.40, 80.0)
        if t9 > 8:
            print(f"\n=== Phase 9: Final Subgradient ===")
            a_out, _ = subgradient_nesterov(best_seq, t9, topk=8)
            try_update(a_out, "sg_final")

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence([float(x) for x in best_seq]))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
```
