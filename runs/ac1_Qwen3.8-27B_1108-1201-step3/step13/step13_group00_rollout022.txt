    n = len(a)
    norm = np.sqrt(2.0 * n)

    c = np.zeros(n + 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0

    T = build_T_rows(a, active_indices)

    bounds = [(0, None)] * (n + 1)

        return None, float('inf')
    
    b_val = np.maximum(res.x[:n], 0.0)
    t_val = max(float(res.x[n]), 0.0)

    c = np.zeros(n + 1)
    c[n] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([norm])
    A_ub = np.hstack([T, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    bounds = [(0, None)] * (n + 1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5.0})
    except Exception:
        return None, float('inf')
    if res.status != 0:
        return None, float('inf')
    b_val = np.maximum(res.x[:n], 0.0)
    t_val = max(float(res.x[n]), 0.0)
    if b_val.sum() < 1e-15:
        return None, float('inf')
    return b_val, t_val

def nesterov_lp_optimize(a0, time_limit, verbose=True):
    """Main LP-based Nesterov optimization."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    y = a.copy()
    a_curr = a.copy()
    active_indices = None
    stagnation = 0

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M = conv_y.max()

        # Constraint screening
        if t < 50:
            active_indices = np.where(conv_y >= M * 0.9)[0]
        elif active_indices is None:
            active_indices = np.where(conv_y >= M * 0.95)[0]
        else:
            # Add newly active, prune old
            new = np.where(conv_y >= M * 0.92)[0]
            active_indices = np.union1d(active_indices, new)
            if len(active_indices) > 500:
                active_indices = np.where(conv_y[active_indices] >= M * 0.90)[0]
                if len(active_indices) == 0:
                    active_indices = np.where(conv_y >= M * 0.85)[0]

        # Solve LP
        b, t_val = lp_min_peak(y, active_indices)
        if b is None:
            # Fallback: use top 200
            top_k = np.argsort(conv_y)[-200:]
            b, t_val = lp_min_peak(y, top_k)
            if b is None:
                b = y.copy()
                t_val = M

        # Check violations
        cross = cross_conv(y, b)
        violated = np.where(cross > t_val * (1.0 + 1e-5) + 1e-12)[0]
        if len(violated) > 0:
            active_indices = np.union1d(active_indices, violated)
            if len(active_indices) < 800:
                b2, t2 = lp_min_peak(y, active_indices)
                if b2 is not None:
                    b = b2
                    t_val = t2

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * b
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = b.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        prev_best = best_score
        y = y_new
        a_curr = a_new

        sc_y = fast_autocorr(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            stagnation = 0
        else:
            stagnation += 1

        sc_a = fast_autocorr(a_curr).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a_curr.copy()
            stagnation = 0

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP t={t} ({elapsed:.0f}s): best={best_score:.8f} M={M:.6f} |A|={len(active_indices)}")

        # Early stopping if very stagnant
        if stagnation > 5000 and t > 5000:
            # Reset y to best and continue with tighter active set
            y = best_a.copy()
            a_curr = best_a.copy()
            active_indices = np.where(fast_autocorr(y) >= best_score * 0.99)[0]
            stagnation = 0

    return best_a, best_score

def pair_cd_polish(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent polish."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a * norm / s

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    n_passes = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-6:
                continue

            # Try 5 random j's
            js = rng.integers(0, n, size=5)
            for j in js:
                j = int(j)
                if j == i:
                    continue
                ai = a[i]
                if ai < 1e-12:
                    continue

                # Compute gradient of max conv w.r.t. transfer i->j
                # First order: change in conv[k] = d * (a[k-j]+a[j-k'] - a[k-i]-a[i-k''])
                # Use a few sample delta values
                best_d = 0.0
                best_v = conv.max()
                fracs = [0.1, 0.2, 0.35, 0.5, 0.7, 0.85]
                for f in fracs:
                    d = f * ai
                    a[i] -= d
                    a[j] += d
                    a = a * norm / a.sum()
                    cn = fast_autocorr(a)
                    v = cn.max()
                    if v < best_v - 1e-13:
                        best_v = v
                        best_d = d
                    a[i] += d
                    a[j] -= d
                    a = a * norm / a.sum()

                if best_d > 0:
                    a[i] -= best_d
                    a[j] += best_d
                    a = a * norm / a.sum()
                    conv = fast_autocorr(a)
                    cur = conv.max()
                    if cur < best_score - 1e-13:
                        best_score = cur
                        best_a = a.copy()
                        pass_improved = True

        if not pass_improved:
            no_improve += 1
            if no_improve >= 4:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

def structured_init(n, rng):
    choice = rng.integers(0, 8)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(4.0 * np.pi * x) + 0.15 * np.cos(10.0 * np.pi * x) + 0.05 * np.cos(18.0 * np.pi * x)
    elif choice == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.8 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2.5
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.6
    elif choice == 6:
        a = np.sin(np.pi * x) ** 2 * (1.0 + 0.3 * np.cos(6.0 * np.pi * x))
    else:
        a = 0.4 + 0.6 * np.cos(2.0 * np.pi * (x - 0.5)) ** 2
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def critical_reduction(a, time_limit, verbose=False):
    """Targeted reduction of critical convolution positions."""
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    a = a * norm / s

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()

    for outer in range(200):
        if time.monotonic() - t0 > time_limit:
            break

        critical = np.where(conv >= best_score * 0.999)[0]
        improved = False

        for k_star in critical:
            if time.monotonic() - t0 > time_limit:
                break

            # Gradient of conv[k_star] w.r.t. a[i]
            grad = np.zeros(n)
            for i in range(n):
                v = 0.0
                j1 = k_star - i
                if 0 <= j1 < n:
                    v += a[j1]
                j2 = i - k_star
                if 0 <= j2 < n:
                    v += a[j2]
                grad[i] = v

            # Find position to decrease (highest grad contribution)
            i_dec = int(np.argmax(grad))
            if grad[i_dec] < 1e-10:
                continue

            # Find position to increase (lowest grad, to minimize impact)
            i_inc = int(np.argmin(grad))
            if i_inc == i_dec:
                i_inc = (i_dec + 1) % n

            # Try small transfer
            d = 0.01 * a[i_dec]
            if d < 1e-12:
                continue
            a[i_dec] -= d
            a[i_inc] += d
            a = a * norm / a.sum()
            conv_new = fast_autocorr(a)
            if conv_new.max() < best_score - 1e-13:
                best_score = conv_new.max()
                best_a = a.copy()
                conv = conv_new
                improved = True
            else:
                a[i_dec] += d
                a[i_inc] -= d
                a = a * norm / a.sum()

        if not improved:
            # Try random perturbations
            for _ in range(50):
                if time.monotonic() - t0 > time_limit:
                    break
                i = int(np.random.randint(n))
                j = int(np.random.randint(n))
                if i == j:
                    continue
                d = np.random.uniform(0.001, 0.02) * a[i]
                a[i] -= d
                a[j] += d
                a = np.maximum(a, 0.0)
                a = a * norm / a.sum()
                conv_new = fast_autocorr(a)
                if conv_new.max() < best_score - 1e-13:
                    best_score = conv_new.max()
                    best_a = a.copy()
                    conv = conv_new
                    improved = True
                else:
                    a[j] -= d
                    a[i] += d
                    a = a * norm / a.sum()

            if not improved:
                break

        if verbose and outer % 10 == 0:
            print(f"    CR outer={outer}: best={best_score:.8f}, |crit|={len(critical)}")

    return best_a, best_score

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Default init if no parent
    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)
    print(f"\nStarting: n={n_base}, score={best_score:.6f}")

    # Phase 1: LP Nesterov at base length
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.45, 700.0)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 30:
        norm = np.sqrt(2.0 * n_base)
        a_norm = best_seq * norm / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, verbose=True)
        try_update(a_out, f"lp_n{n_base}")

    # Phase 2: Pair CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 2: Pair CD n={n_best} (budget={t2:.0f}s) ===")
    if t2 > 10:
        norm = np.sqrt(2.0 * n_best)
        a_norm = best_seq * norm / best_seq.sum()
        a_out, _ = pair_cd_polish(a_norm, t2, rng, verbose=True)
        try_update(a_out, f"pcd_n{n_best}")

    # Phase 3: Try longer length
    n_long = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.18, 300.0)
    print(f"\n=== Phase 3: LP Nesterov n={n_long} (budget={t3:.0f}s) ===")
    if t3 > 30:
        init_long = interpolate_sequence(best_seq, n_long)
        norm = np.sqrt(2.0 * n_long)
        a_norm = init_long * norm / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t3, verbose=True)
        try_update(a_out, f"lp_n{n_long}")
        # Quick PCD on longer
        remaining = time_budget - (time.monotonic() - t_global)
        t3b = min(remaining * 0.05, 50.0)
        if t3b > 5:
            n_cur = len(best_seq)
            norm_cur = np.sqrt(2.0 * n_cur)
            a_norm2 = best_seq * norm_cur / best_seq.sum()
            a_out2, _ = pair_cd_polish(a_norm2, t3b, rng, verbose=False)
            try_update(a_out2, f"pcd_long_n{n_cur}")

    # Phase 4: Try shorter length
    n_short = max(800, n_base - 400)
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.12, 150.0)
    print(f"\n=== Phase 4: LP Nesterov n={n_short} (budget={t4:.0f}s) ===")
    if t4 > 20:
        init_short = interpolate_sequence(best_seq, n_short)
        norm = np.sqrt(2.0 * n_short)
        a_norm = init_short * norm / init_short.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t4, verbose=False)
        try_update(a_out, f"lp_n{n_short}")

    # Phase 5: Critical reduction on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.15, 150.0)
    print(f"\n=== Phase 5: Critical reduction n={n_best} (budget={t5:.0f}s) ===")
    if t5 > 10:
        norm = np.sqrt(2.0 * n_best)
        a_norm = best_seq * norm / best_seq.sum()
        a_out, _ = critical_reduction(a_norm, t5, verbose=True)
        try_update(a_out, f"crit_n{n_best}")

    # Phase 6: Final LP + PCD on absolute best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final polish n={n_best} ===")
    t6 = min(remaining * 0.3, 100.0)
    if t6 > 10:
        norm = np.sqrt(2.0 * n_best)
        a_norm = best_seq * norm / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t6, verbose=False)
        try_update(a_out, f"final_lp_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)
    t7 = min(remaining * 0.5, 80.0)
    if t7 > 5:
        n_best = len(best_seq)
        norm = np.sqrt(2.0 * n_best)
        a_norm = best_seq * norm / best_seq.sum()
        a_out, _ = pair_cd_polish(a_norm, t7, rng, verbose=True)
        try_update(a_out, f"final_pcd_n{n_best}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
