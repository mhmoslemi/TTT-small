def normalize_seq(a, target_sum):

    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()

def softmax_subgrad(a, conv_val, beta=8.0, min_weight=1e-8):
    """Compute softmax-weighted subgradient of max(conv). Uses all significant peaks."""
    n = len(a)
    max_cv = conv_val.max()
    if max_cv < 1e-15:
        return np.zeros(n)
    scaled = beta * conv_val / max_cv
    scaled -= scaled.max()
    weights = np.exp(scaled)
    total_w = weights.sum()
    if total_w < 1e-30:
        return np.zeros(n)
    weights = weights / total_w
    active = weights > min_weight
    if not np.any(active):
        k_max = int(np.argmax(conv_val))
        active = np.zeros(len(conv_val), dtype=bool)
        active[k_max] = True
        weights = np.zeros(len(conv_val))
        weights[k_max] = 1.0
    grad = np.zeros(n)
    for k in np.where(active)[0]:
        w = weights[k]
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n - 1, k_int)
        if i_start > i_end:
            continue
        idx = np.arange(i_start, i_end + 1)
        j = k_int - idx
        grad[idx] += 2.0 * w * a[j]
    return grad


def nesterov_softmax_optimize(a0, time_limit, beta_init=8.0, beta_decay=0.999, verbose=True):
    """Nesterov-accelerated softmax subgradient with simplex projection."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = normalize_seq(a0, c)
    if a is None:
        a = np.ones(n) * c / n
    
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    v = a.copy()
    x_prev = a.copy()
    beta_nesterov = 0.0
    beta_target = 0.7
    step_scale = 1.0
    beta_sm = beta_init
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_conv(v)
        M = ac.max()
        
        g = softmax_subgrad(v, ac, beta=beta_sm, min_weight=1e-8)
        
        # Adaptive step size
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            step = 0.0
        else:
            step = step_scale / (np.sqrt(t) * g_norm + 1e-10)
            step = min(step, 0.5 / g_norm if g_norm > 0 else 0.5)
        
        # Gradient step
        v_new = v - step * g
        v_new = np.maximum(v_new, 0.0)
        s_new = v_new.sum()
        if s_new < 1e-15:
            v_new = v.copy()
        else:
            v_new = v_new * c / s_new
        
        # Nesterov momentum
        if t == 1:
            x_new = v_new.copy()
        else:
            beta_nesterov = (1.0 + np.sqrt(1.0 + 4.0 * beta_target * beta_target)) / 2.0
            momentum = (beta_nesterov - 1.0) / (1.0 + beta_nesterov)
            x_new = v_new + momentum * (v_new - x_prev)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:
                x_new = v_new.copy()
            else:
                x_new = x_new * c / s_x
        
        x_prev = v_new
        v = x_new
        
        # Decay softmax parameter
        beta_sm *= beta_decay
        beta_sm = max(beta_sm, 2.0)
        
        # Track best
        M_v = fast_conv(v).max()
        if M_v < best_m:
            best_m = M_v
            best_a = v.copy()
        
        # Periodic check on x_prev trajectory average
        if t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c * c)
            if verbose and t % 2000 == 0:
                print(f"    NSO t={t} ({elapsed:.1f}s): J={J:.6f}, M={best_m:.6f}, beta={beta_sm:.2f}")
    
    return best_a, best_m


def lp_best_response_warm(a, M, solver, warm_b=None):
    """Solve LP: max sum(b) s.t. conv(a,b)_k <= M, b >= 0. With warm start."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    
    # Build constraint matrix T where (T@b)_k = conv(a,b)_k
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        if warm_b is not None and prob.status in (None, 'unstarted'):
            b_var.value = warm_b
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    return b_val, M

def freq_domain_smooth(a, n_smooth=5):
    """Apply frequency domain smoothing to reduce spectral peaks.
    Ideas: 
    1. Slightly flatten the magnitude of the DFT
    2. Phase adjustment to reduce time-domain peaks
    """
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = rfft(a, fft_size)
    
    # Compute current convolution peak locations
    conv_val = irfft(A * A, fft_size)[:2*n-1]
    conv_max = conv_val.max()
    
    # In frequency domain, A^2 determines the convolution.
    # We want to reduce the peak of IFFT(A^2).
    # Strategy: slightly reduce |A|^2 at frequencies where A^2 has large positive real part
    # at the peak locations.
    
    # Simpler approach: apply a mild spectral flattening
    # Reduce the dynamic range of |A| slightly
    mag = np.abs(A)
    mag_max = mag.max()
    mag_min = mag[mag > 1e-10].min() if np.any(mag > 1e-10) else 1e-10
    
    # Gentle compression: new_mag = mag^alpha where alpha < 1
    alpha = 0.95  # mild compression
    new_mag = mag ** alpha
    new_mag *= mag_max / new_mag.max()  # preserve DC
    
    # Reconstruct with original phase
    phase = np.angle(A)
    A_new = new_mag * np.exp(1j * phase)
    
    # Enforce Hermitian symmetry (rfft already handles this)
    a_new = irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * a.sum() / s

        A_sq = A * A

def coordinate_polish(a0, time_limit, verbose=False):
    """Aggressive coordinate descent: move mass from high-peak contributors."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = normalize_seq(a0, c)
    if a is None:
        return a, c
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    stall = 0
    while time.monotonic() - t0 < time_limit and stall < 500:
        ac = fast_conv(a)
        top_k = np.argsort(ac)[::-1][:10]
        improved = False
        
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]
            
            # Find destination: minimize impact on all top peaks
            peak_contrib = np.zeros(n)
            for k2 in top_k:
                k2_int = int(k2)
                js2 = max(0, k2_int - n + 1)
                je2 = min(n - 1, k2_int)
                idx2 = np.arange(js2, je2 + 1)
                peak_contrib[idx2] += a[k2_int - idx2]
            
            # Prefer positions with low peak_contrib and currently low a value
            score = peak_contrib + 10.0 * a
            score[src] = np.inf
            candidates = np.argsort(score)[:max(1, n // 4)]
            dst = candidates[rng.integers(len(candidates))]
            
            if dst == src:
                continue
            
            for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial * c / a_trial.sum()
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_m - 1e-14:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        
        if not improved:
            stall += 1
            # Random perturbation
            num_perturb = rng.integers(1, 5)
            for _ in range(num_perturb):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > c * 1e-9:
                    frac = rng.uniform(0.005, 0.03)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    if a[i] >= 0:
                        a = a * c / a.sum()
                        M_new = fast_conv(a).max()
                        if M_new < best_m:
                            best_m = M_new
                            best_a = a.copy()
                            stall = 0
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = a * c / a.sum()
    
    if verbose:
        elapsed = time.monotonic() - t0
        J = 2.0 * n * best_m / (c * c)
        print(f"    Coord ({elapsed:.1f}s): J={J:.6f}")
    return best_a.copy(), best_m


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def mirror_descent_optimize(a0, time_limit, top_k=8, beta=4.0, verbose=True):
    """Mirror descent with exponential weights on simplex."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = normalize_seq(a0, c)
    if a is None:
        a = np.ones(n) * c / n
    a = np.maximum(a, 1e-15)
    a = a * c / a.sum()
    
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    gamma = 1.0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        step = gamma / (np.sqrt(t) + 1.0)
        ac = fast_conv(a)
        M = ac.max()
        
        # Soft subgradient with top-k
        k_max = min(top_k, len(ac))
        top_idx = np.argsort(ac)[::-1][:k_max]
        top_vals = np.maximum(ac[top_idx], 0)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(n - 1, k_int)
            idx = np.arange(i_start, i_end + 1)
            j = k_int - idx
            grad[idx] += 2.0 * wi * a[j]
        
        # Mirror descent update
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - step * grad
        max_log = log_a_new.max()
        a_new = np.exp(log_a_new - max_log)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * c / s_new
        a = a_new
        
        M_new = fast_conv(a).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a.copy()
        
        if verbose and t % 3000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c * c)
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m


def propose_candidate(budget_s=1800.0):
    """Main search: multi-phase optimization with novel frequency-domain methods."""
    rng = np.random.default_rng(2024)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)
    
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent / parent.sum()
            best_score = p_score
    except Exception:
        pass
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    if best_seq is None:
        n_base = 1600
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    
    n0 = len(best_seq)
    print(f"Starting: n={n0}, score={best_score:.6f}")
    
    # === Phase 1: Nesterov softmax gradient (main driver) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Nesterov Softmax n={n0} ===")
    t1 = min(remaining * 0.20, 300.0)
    if t1 > 30:
        c0 = np.sqrt(2.0 * n0)
        a_norm = best_seq * c0 / best_seq.sum()
        a_out, _ = nesterov_softmax_optimize(a_norm, t1, beta_init=10.0, beta_decay=0.9995, verbose=True)
        try_update(a_out, f"nesterov_softmax_n{n0}")
    
    # === Phase 2: Mirror descent (complementary) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Mirror Descent n={n0} ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 20:
        c0 = np.sqrt(2.0 * n0)
        a_norm = best_seq * c0 / best_seq.sum()
        a_out, _ = mirror_descent_optimize(a_norm, t2, top_k=10, beta=3.0, verbose=True)
        try_update(a_out, f"md_n{n0}")
    
    # === Phase 3: LP best-response ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LP Best-Response n={n0} ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 30:
        c0 = np.sqrt(2.0 * n0)
        a_norm = best_seq * c0 / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=True)
        try_update(a_out, f"lp_n{n0}")

    remaining = time_budget - (time.monotonic() - t_global)

    t5 = min(remaining * 0.15, 200.0)

        a_long = interpolate_sequence(best_seq, n1)
        c1 = np.sqrt(2.0 * n1)
        a_norm = a_long * c1 / a_long.sum()
        # Run Nesterov softmax at new length

        try_update(a_out, f"ns_n{n1}")

            try_update(a_pol, f"coord_n{n1}")

    t6 = min(remaining * 0.12, 150.0)

    # === Phase 7: Random restarts at best length ===
    n_best = len(best_seq)

    t7 = min(remaining * 0.15, 180.0)
    if t7 > 30:
        c_best = np.sqrt(2.0 * n_best)
        n_restarts = min(5, int(t7 // 40) + 1)
        for r in range(n_restarts):
            if time.monotonic() - t_global > time_budget - 60:
                break
            # Perturb best sequence
            a_pert = best_seq.copy()
            noise_scale = 0.02 * (r + 1)
            a_pert += rng.normal(0, noise_scale, size=n_best)
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            a_pert = a_pert * c_best / a_pert.sum()
            
            t_sub = min((time_budget - (time.monotonic() - t_global)) * 0.15, 50.0)
            if t_sub < 10:
                break
            a_out, _ = nesterov_softmax_optimize(a_pert, t_sub, beta_init=6.0, beta_decay=0.999, verbose=False)
            try_update(a_out, f"restart{r}_n{n_best}")
    
    # === Phase 8: Final coordinate polish ===
    n_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Final Coordinate Polish n={n_best} ===")
    t8 = min(remaining * 0.25, 200.0)
    if t8 > 15:
        c_best = np.sqrt(2.0 * n_best)
        a_norm = best_seq * c_best / best_seq.sum()
        a_out, _ = coordinate_polish(a_norm, t8, verbose=True)
        try_update(a_out, f"coord_final_n{n_best}")
    
    # === Phase 9: Final Nesterov softmax ===
    n_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"\n=== Phase 9: Final Nesterov Softmax n={n_best} ===")
        t9 = min(remaining * 0.4, 100.0)
        c_best = np.sqrt(2.0 * n_best)
        a_norm = best_seq * c_best / best_seq.sum()
        a_out, _ = nesterov_softmax_optimize(a_norm, t9, beta_init=5.0, beta_decay=0.999, verbose=True)
        try_update(a_out, f"ns_final_n{n_best}")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
