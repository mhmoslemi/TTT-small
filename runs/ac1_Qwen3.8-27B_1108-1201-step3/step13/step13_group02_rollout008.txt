```python
def cross_corr_fft(b, a):
    """Compute np.convolve(b, a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    B = np.fft.rfft(b, fft_size)
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(B * A, fft_size)[:size]
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Proximal-Accelerated Nesterov (PAN)
with FFT-based fast best-response approximation, followed by L-BFGS-B polish
on the LogSumExp-smoothed objective.
"""

import numpy as np
import time
import scipy.optimize
from scipy.fft import rfft, irfftn

# Precomputed FFT sizes
def next_pow2(n):
    return 1 << (n - 1).bit_length()

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = next_pow2(size)
    A = rfft(a.astype(np.float64), fft_size)
    return irfftn(A * A, fft_size)[:size]

def fast_convolve(a, b):
    """Full convolution of a and b using FFT."""
    na, nb = len(a), len(b)
    size = na + nb - 1
    fft_size = next_pow2(size)
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfftn(A * B, fft_size)[:size]

def fast_convolve_transpose(v, a):
    """Compute T^T @ v where T is the convolution matrix with kernel a.
    v has length 2*len(a)-1, result has length len(a).
    Uses the identity: (T^T v)[j] = conv(v, a_rev)[j + n - 1]
    """
    n = len(a)
    a_rev = a[::-1]
    size = len(v) + n - 1  # = (2n-1) + n - 1 = 3n - 2
    fft_size = next_pow2(size)
    V = rfft(v, fft_size)
    AR = rfft(a_rev, fft_size)
    conv = irfftn(V * AR, fft_size)
    return conv[n-1:2*n-1]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def normalize_a(a, norm_val):
    """Normalize a to have sum = norm_val, clipping to [0, inf)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * norm_val / len(a)
    return a * norm_val / s

def proximal_best_response(y, M, n_inner=30, eta_scale=1.0):
    """Approximate the LP best response using proximal gradient.
    Solves: max sum(b) s.t. conv(b, y) <= M, b >= 0
    Using proximal gradient on the penalty: 0.5*sum(max(conv(b,y)-M,0)^2)
    """
    n = len(y)
    b = y.copy()
    # Step size: eta = 1 / ||T^T T|| <= 1 / ||T||^2 <= 1/||y||_1^2
    # With our normalization, ||y||_1 = sqrt(2n), so eta <= 1/(2n)
    eta = eta_scale / (np.sum(y) ** 2)
    
    for i in range(n_inner):
        # Compute residual = max(T@b - M, 0)
        c = fast_convolve(b, y)
        residual = np.maximum(c - M, 0.0)
        res_max = residual.max()
        if res_max < 1e-12 * M:
            break  # Nearly feasible
        
        # Gradient = T^T @ residual
        grad = fast_convolve_transpose(residual, y)
        
        # Proximal gradient step
        b = np.maximum(b - eta * grad, 0.0)
    
    # Final feasibility check: scale down if needed
    c = fast_convolve(b, y)
    c_max = c.max()
    if c_max > M * (1 + 1e-10):
        b = b * M / c_max
    
    b_sum = b.sum()
    if b_sum < 1e-15:
        return None
    return b

def pan_optimize(a0, time_limit, n_inner=30, verbose=True):
    """Proximal-Accelerated Nesterov optimization."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = normalize_a(a0.copy(), norm_val)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    stall_best = best_score
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-14:
            break
        
        # Approximate best response
        b = proximal_best_response(y, M_curr, n_inner=n_inner)
        if b is None:
            continue
        
        b = normalize_a(b, norm_val)
        n_iters += 1
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * b
        y_new = normalize_a(y_new, norm_val)
        
        momentum = (t - 1.0) / (t + 2.0)
        a_new = y_new + momentum * (y_new - y)
        a_new = normalize_a(a_new, norm_val)
        
        y = y_new
        a = a_new
        
        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        
        # Adaptive restart
        if stall > 500:
            rng = np.random.default_rng(n_iters * 13 + 7)
            scale = 0.05 * a.mean()
            y = normalize_a(np.maximum(y + rng.normal(0, scale, n), 0), norm_val)
            a = normalize_a(np.maximum(a + rng.normal(0, scale, n), 0), norm_val)
            stall = 0
            if verbose:
                print(f"    [PAN Restart at iter {n_iters}]")
        
        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PAN iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, stall={stall}")
    
    return best_a, best_score

def lse_objective_and_grad(x, T_temp, n):
    """Compute LSE objective and gradient for the normalized sequence."""
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    a = a * norm_val / s  # normalize
    
    c = fast_autocorr(a)
    c_max = c.max()
    
    # LSE
    shifted = (c - c_max) / T_temp
    exp_shifted = np.exp(shifted)
    lse_val = c_max + T_temp * np.log(exp_shifted.sum())
    
    # Gradient of LSE w.r.t. a
    w = exp_shifted / exp_shifted.sum()
    # d(LSE)/d(a_i) = (2/T) * sum_k w_k * (a_{i+k} + a_{i-k}) [with bounds]
    # = (2/T) * (w * a_rev)[i + n - 1]
    a_rev = a[::-1]
    fft_size = next_pow2(3 * n - 2)
    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)
    conv = irfftn(W * AR, fft_size)
    grad_a = (2.0 / T_temp) * conv[n-1:2*n-1]
    
    # Chain rule through normalization: a = x * norm_val / sum(x)
    # d(a_i)/d(x_j) = norm_val * (delta_ij * sum(x) - x_i) / sum(x)^2
    # = (delta_ij - a_i/norm_val) / (s/norm_val)
    # = norm_val * (delta_ij/s - x_i/s^2)
    # grad_x_j = sum_i grad_a_i * d(a_i)/d(x_j)
    # = (norm_val/s) * (grad_a_j - a_j * sum(grad_a) / s)
    
    # Actually: a_i = norm_val * x_i / s where s = sum(x)
    # da_i/dx_j = norm_val * (delta_ij * s - x_i) / s^2
    # = (norm_val/s) * (delta_ij - x_i/s)
    # = (1/s) * (delta_ij - a_i/norm_val) * norm_val... 
    
    # Let me redo: a_i = norm_val * x_i / s
    # da_i/dx_j = norm_val * (delta_ij * s - x_i * 1) / s^2
    # = (norm_val / s^2) * (delta_ij * s - x_i)
    # = (norm_val / s) * delta_ij - (norm_val * x_i) / s^2
    # = (norm_val / s) * delta_ij - a_i / s
    
    # grad_x_j = sum_i grad_a_i * (norm_val/s * delta_ij - a_i/s)
    # = (norm_val/s) * grad_a_j - (1/s) * sum_i(grad_a_i * a_i)
    
    factor = norm_val / s
    grad_x = factor * grad_a - (1.0 / s) * np.dot(grad_a, a)
    
    # Also need to handle the max(x, 0) part (subgradient)
    grad_x = np.where(x >= 0, grad_x, 0.0)
    
    return lse_val, grad_x

def lse_lbfgsb_optimize(a0, time_limit, verbose=True):
    """Polish using L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = normalize_a(a0.copy(), norm_val)
    
    best_a = a.copy()
    best_score = evaluate_safe(a)
    t0 = time.monotonic()
    
    # Start with relatively high temperature, anneal
    c0 = fast_autocorr(a).max()
    T_start = max(0.01, c0 * 0.1)
    T_end = max(0.001, c0 * 0.01)
    
    n_total = 0
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        T_current = T_start * (T_end / T_start) ** elapsed_frac
        
        # Define objective for scipy
        def obj_func(x):
            val, grad = lse_objective_and_grad(x, T_current, n)
            return val, grad
        
        try:
            result = scipy.optimize.minimize(
                obj_func, a, method='L-BFGS-B',
                jac=True,
                bounds=[(0, 1000.0)] * n,
                options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-8, 'maxfun': 500}
            )
            a_candidate = normalize_a(result.x, norm_val)
            score = evaluate_safe(a_candidate)
            n_total += 1
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_a = a_candidate.copy()
                a = a_candidate.copy()
        except Exception as e:
            if verbose:
                print(f"    L-BFGS-B error: {e}")
            break
        
        if verbose and n_total % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-LBFGS round={n_total} ({elapsed:.0f}s): T={T_current:.4f}, best={best_score:.6f}")
    
    return best_a, best_score
