"""

1. Smooth log-sum-exp gradient descent with Adam and adaptive temperature annealing
2. Direct subgradient peak-attacking with simplex projection
3. LP zero-sum game (simplified Nesterov) for global convergence
4. Iterative peak-shaving with multi-position moves
5. Multi-length exploration with interpolation
Key innovation: exact gradient computation for the smooth max objective,
combined with aggressive simplex projection and multi-scale temperature annealing.

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def project_simplex(v):
    """Project v onto the unit simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_gradient(a, conv, T):
    """
    Compute gradient of smooth max: (1/T)*log(sum_k exp(T*conv[k]))
    with respect to a.
    grad[j] = 2 * sum_k w_k * a[k-j] where 0 <= k-j < n
    """
    n = len(a)
    max_conv = conv.max()
    exp_conv = np.exp(T * (conv - max_conv))
    w = exp_conv / exp_conv.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[js:je + 1] += 2.0 * wk * a[k - idx]
    return grad


def compute_gradient_sparse(a, conv, T, topk):
    """
    Compute gradient using only top-k peaks (sparse approximation).
    Much faster when T is low.
    """
    n = len(a)
    k_use = min(topk, 2 * n - 1)
    top_idx = np.argsort(conv)[::-1][:k_use]
    
    max_conv = conv[top_idx].max()
    exp_top = np.exp(T * (conv[top_idx] - max_conv))
    w_top = exp_top / exp_top.sum()
    
    grad = np.zeros(n)
    for i in range(k_use):
        k = top_idx[i]
        wk = w_top[i]
        if wk < 1e-12:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[js:je + 1] += 2.0 * wk * a[k - idx]
    return grad


def smooth_max_value(conv, T):
    """Compute smooth max value."""
    max_conv = conv.max()
    exp_conv = np.exp(T * (conv - max_conv))
    return (max_conv + np.log(exp_conv.sum())) / T


def adam_optimize(a0, time_limit, T_start=50.0, T_end=0.1, n_stages=None,
                  base_lr=0.01, topk_threshold=50, use_sparse=False, seed=42):
    """
    Adam optimization with temperature annealing on the smooth max objective.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    if n_stages is None:
        n_stages = max(20, int(time_limit * 2))
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit - 2:
            break
        
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        lr = base_lr * min(1.0, 50.0 / n) * (0.95 ** si)
        m_a = np.zeros(n)
        v_a = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.8, 0.999, 1e-8
        
        max_it = max(10, int(min(remaining, 30.0) * 8))
        
        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 1:
                break
            
            conv = fast_autocorr(a)
            bmax = conv.max()
            
            # Choose sparse or full gradient
            n_above = np.sum(conv > bmax * (1.0 - 1.0 / stage_T))
            if use_sparse and stage_T < topk_threshold:
                k_eff = min(max(5, int(n_above) + 5), 2 * n - 1)
                grad = compute_gradient_sparse(a, conv, stage_T, k_eff)
            else:
                grad = compute_gradient(a, conv, stage_T)
            
            # Adam update
            t_adam += 1
            m_a = beta1 * m_a + (1 - beta1) * grad
            v_a = beta2 * v_a + (1 - beta2) * grad ** 2
            m_hat = m_a / (1 - beta1 ** t_adam)
            v_hat = v_a / (1 - beta2 ** t_adam)
            
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = project_simplex(a_new)
            
            # Check improvement
            score = 2.0 * n * fast_autocorr(a_new).max()
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
            
            a = a_new
            
            # Periodic checkpoint
            if (it + 1) % 20 == 0 and score > best_score * 1.01:
                a = best_a.copy()
                m_a = np.zeros(n)
                v_a = np.zeros(n)
                t_adam = 0
        
        a_stage = project_simplex(a)
        score_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()
        
        if si % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Adam stage {si}/{n_stages} (T={stage_T:.2f}, {elapsed:.1f}s): best={best_score:.6f}")
    
    return best_a, best_score


def subgradient_peak_attack(a0, time_limit, seed=42):
    """
    Direct subgradient method: attack the top peaks directly.
    At each step, compute subgradient at all peaks above threshold,
    take a step, and project onto simplex.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    step_size = 0.002 * min(1.0, 100.0 / n)
    stall = 0
    step_scale = 1.0
    
    for it in range(50000):
        if time.monotonic() - t0 > time_limit - 1:
            break
        
        conv = fast_autocorr(a)
        bmax = conv.max()
        
        # Find peaks above 95% of max
        threshold = bmax * 0.95
        peak_indices = np.where(conv >= threshold)[0]
        if len(peak_indices) == 0:
            peak_indices = np.array([np.argmax(conv)])
        
        # Compute subgradient: average gradient at active peaks
        grad = np.zeros(n)
        for k in peak_indices[:20]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[js:je + 1] += 2.0 * a[k - idx]
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Normalized step
        a_new = a - step_size * step_scale * grad / grad_norm
        a_new = project_simplex(a_new)
        
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall % 50 == 0:
                step_scale *= 0.8
                if step_scale < 0.01:
                    step_scale = 1.0
                    step_size = min(step_size * 1.5, 0.01)
                    # Random perturbation
                    a = best_a.copy()
                    noise = rng.normal(0, 0.005, n)
                    a = a * (1 + noise)
                    a = project_simplex(a)
            
            if stall > 500:
                break
    
    return best_a, best_score

                        a_trial = np.maximum(a_trial, 0)
                        st = a_trial.sum()
                        if st < 1e-12:
                            continue
                        a_trial = a_trial / st
                        new_max = fast_autocorr(a_trial).max()
                        if new_max < bmax * (1 - 1e-9):
                            a = a_trial
                            score = 2.0 * n * new_max
                            if score < best_score:
                                best_score = score
                                best_a = a.copy()
                            improved = True
                            found = True
                            stall = 0
                            break
                    if found:
                        break
                if found:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall % 10 == 0 and stall <= 100:
                n_pert = rng.integers(1, 4)
                a_trial = a.copy()
                for _ in range(n_pert):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a_trial[i] > 1e-10:
                        frac = rng.uniform(0.002, 0.02)
                        amt = a_trial[i] * frac
                        a_trial[i] -= amt
                        a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-10:
                    a_trial = a_trial / st
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            if stall > 200:
                break
    
    return best_a, best_score


def build_toeplitz(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def find_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    for solver_name in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.CLARABEL


def lp_game_optimize(a0, time_limit, solver, max_iters=20000):
    """
    LP zero-sum game (Nesterov accelerated) for the convex relaxation.
    """
    import cvxpy as cp
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    warm_b = None
    stall = 0
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit - 1:
            break
        
        # Solve LP: min M s.t. conv(y, b) <= M, sum(b)=1, b>=0
        T_mat = build_toeplitz(y)
        
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver)
        except Exception:
            try:
                prob.solve(solver=cp.SCS)
            except Exception:
                try:
                    prob.solve(solver=cp.ECOS)
                except Exception:
                    continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        b_val = b_val / b_sum
        warm_b = b_val
        
        # Nesterov update
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_val
        y_new = np.maximum(y_new, 0)
        y_sum = y_new.sum()
        if y_sum < 1e-15:
            y_new = b_val.copy()
        else:
            y_new = y_new / y_sum
        
        if t == 1:
            a_new = b_val.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new
        
        y = y_new
        a = a_new
        
        # Check improvement
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        
        if stall > 300 and t > 50:
            a = best_a.copy()
            y = best_a.copy()
            stall = 0
            warm_b = None
        
        if t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP t={t} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = int(rng.integers(0, 8))
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.8 + 0.15 * np.cos(2 * x) + 0.05 * np.cos(7 * x)
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-6 * (x - 0.3) ** 2) + 0.7 * np.exp(-6 * (x - 0.7) ** 2)
    elif choice == 5:
        x = np.linspace(-1, 1, n)
        a = np.maximum(0, 1 - np.abs(x) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(4 * np.pi * x)
    else:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * (x - 0.5) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, time_budget=1800.0):
    """
    Multi-phase optimization combining smooth gradient descent,
    subgradient peak attack, LP game, and peak shaving.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(time_budget) - 120.0, 1680.0)
    
    solver = find_solver()
    print(f"LP solver: {solver}, budget: {time_budget:.0f}s")
    
    # Get parent
    best_seq = None
    best_score = float('inf')
    
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # === Phase 1: Adam smooth gradient at parent length ===
    n_base = len(best_seq) if best_seq is not None else 1600
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Adam smooth gradient n={n_base} ===")
    t1 = min(remaining * 0.30, 500.0)
    if t1 > 60:
        a_out, _ = adam_optimize(best_seq, t1, T_start=80.0, T_end=0.05,
                                  n_stages=40, base_lr=0.008, use_sparse=True)
        try_update(a_out, f"adam_n{n_base}")
    
    # === Phase 2: Subgradient peak attack ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 2: Subgradient peak attack n={n_base} ===")
    t2 = min(remaining * 0.10, 120.0)
    if t2 > 20:
        a_out, _ = subgradient_peak_attack(best_seq, t2, seed=123)
        try_update(a_out, f"subgrad_n{n_base}")
    
    # === Phase 3: LP game (focused, fewer iterations) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 3: LP game n={n_base} ===")
    t3 = min(remaining * 0.25, 400.0)
    if t3 > 60:
        a_out, _ = lp_game_optimize(best_seq, t3, solver, max_iters=10000)
        try_update(a_out, f"lp_n{n_base}")
    
    # === Phase 4: Adam again (warm start from LP) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 4: Adam polish n={n_base} ===")
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 20:
        a_out, _ = adam_optimize(best_seq, t4, T_start=20.0, T_end=0.01,
                                  n_stages=20, base_lr=0.003, use_sparse=True)
        try_update(a_out, f"adam_polish_n{n_base}")
    
    # === Phase 5: Multi-position peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 5: Peak shaving n={n_base} ===")
    t5 = min(remaining * 0.10, 100.0)
    if t5 > 10:
        a_out, _ = multi_position_peak_shave(best_seq, t5, seed=777)
        try_update(a_out, "peak_shave_1")
    
    # === Phase 6: Try slightly longer length ===
    n_long = n_base + 150
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 6: Optimize at n={n_long} ===")
    t6 = min(remaining * 0.12, 150.0)
    if t6 > 40:
        init_long = interpolate_sequence(best_seq, n_long)

                                  n_stages=20, base_lr=0.006)

        if remaining2 > 30:
            a_out, _ = multi_position_peak_shave(
                best_seq if len(best_seq) == n_long else init_long,
                min(remaining2 * 0.3, 40.0), seed=456)
            try_update(a_out, f"peak_n{n_long}")
    
    # === Phase 7: Multi-start with different initializations ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 7: Multi-start ===")
    t7_total = min(remaining * 0.08, 80.0)
    if t7_total > 30:
        for si in range(2):
            if time.monotonic() - t_global > t_global + time_budget - 60:
                break
            t7_each = t7_total / 2
            init = structured_init(n_base, rng)
            a_out, _ = adam_optimize(init, t7_each, T_start=30.0, T_end=0.5,
                                      n_stages=15, base_lr=0.01)
            try_update(a_out, f"multi_{si}")
    
    # === Phase 8: Final Adam very low temperature ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t8 = min(remaining * 0.10, 60.0)
        if t8 > 10:
            print(f"=== Phase 8: Final low-T Adam ===")
            a_out, _ = adam_optimize(best_seq, t8, T_start=3.0, T_end=0.005,

    # === Phase 9: Final peak shave ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 8:
        t9 = min(remaining * 0.30, 40.0)
        if t9 > 5:
            print(f"=== Phase 9: Final peak shave ===")
            a_out, _ = multi_position_peak_shave(best_seq, t9, seed=2001)
            try_update(a_out, "final_peak")
    
    # Final check
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
