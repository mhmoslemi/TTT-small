def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_of(a):
    """Compute the score 2*n*max_b/sum_a^2"""
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (s * s)

def build_T(a):
    """Build the convolution constraint matrix T (2n-1 x n).
    T[j,k] = a[j-k] if 0 <= j-k < n, else 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(n):
        # Column k has nonzeros at rows k, k+1, ..., k+n-1
        # T[j,k] = a[j-k]
        for j in range(k, min(k + n, 2 * n - 1)):
            rows.append(j)
            cols.append(k)
            vals.append(a[j - k])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_best_response(a, M, solver, b_init=None):
    """Solve LP: max sum(b) s.t. T@b <= M, b >= 0."""
    n = len(a)
    T = build_T(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        if b_init is not None:
            b_var.value = b_init
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    if b_var.value is None:
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def local_search_polish(a0, time_limit, seed=42, verbose=True):

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    return _global_solver


def fast_autocorr(a):
    """Compute full convolution (linear autocorrelation) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_T_sparse(a):
    """Build the convolution constraint matrix T (2n-1 x n) as sparse CSR.
    T[j,k] = a[j-k] if 0 <= j-k < n, else 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(n):
        j_start = k
        j_end = min(k + n, 2 * n - 1)
        for j in range(j_start, j_end):
            rows.append(j)
            cols.append(k)
            vals.append(a[j - k])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, solver, b_init=None):
    """Solve LP: max sum(b) s.t. T@b <= M, b >= 0."""
    n = len(a)
    T = build_T_sparse(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        if b_init is not None:
            b_var.value = b_init
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    if b_var.value is None:
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, window_size=100):
    """LP best-response with Nesterov + sliding-window Polyak averaging."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    b_prev = None
    n_iters = 0
    t_last = t0

    # Sliding window for Polyak averaging
    window = []
    running_avg = np.zeros(n)

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, solver, b_init=b_prev)
        if b is None:
            continue
        b_prev = b

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        # Sliding window Polyak average
        window.append(a.copy())
        if len(window) > window_size:
            oldest = window.pop(0)
            running_avg -= oldest / window_size
        running_avg += a / window_size

        # Track best (current, Nesterov, and averaged)
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        if n_iters % 20 == 0:
            s_avg = running_avg.sum()
            if s_avg > 1e-15:
                avg_check = running_avg * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and (time.monotonic() - t_last > 15.0 or n_iters % 300 == 0):
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")
            t_last = time.monotonic()

    return best_a, best_score


def fft_subgradient(z, active_weights):
    """Compute weighted subgradient: sum_k w_k * grad_k(z) using FFT.
    z has length n, active_weights has length 2n-1.
    grad_k(z)[x] = z[k-x] (if 0<=k-x<n) + z[x-k] (if 0<=x-k<n)
    """
    n = len(z)
    fft_size = 1 << (3 * n - 2).bit_length()

    W = rfft(active_weights, fft_size)
    Z = rfft(z, fft_size)

    # Part 2: (z * w)[x] for x = 0, ..., n-1
    conv = irfft(Z * W, fft_size)
    part2 = conv[:n]

    # Part 1: sum_j z[j] * w[j+x] = cross-correlation (w cross z)[x]
    # = ifft(conj(fft(z)) * fft(w))[0:n]
    # Actually: cross_corr = ifft(fft(w, N) * conj(fft(z, N)))
    # Let me verify: (w cross z)[x] = sum_j w[j+x] * conj(z[j])
    # With real z: = sum_j w[j+x] * z[j] = Part1[x] ✓
    cross = irfft(W * np.conj(Z), fft_size)
    part1 = cross[:n]

    return part1 + part2


def subgradient_polish(a0, time_limit, verbose=True):
    """Polish using Nesterov-accelerated subgradient on L-infinity of autocorrelation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    y_prev = a.copy()
    it = 0
    h0 = 0.5 / max(np.max(a), 1e-10) * (1.0 / np.sqrt(n)) * 10.0

    while time.monotonic() - t0 < time_limit:
        if it == 0:
            z = y
        else:
            beta = (1.0 + np.sqrt(1.0 + 4.0 * (it - 1) ** 2)) / (2.0 * (it + 1))
            z = y + beta * (y - y_prev)
            z = np.maximum(z, 0.0)
            sz = z.sum()
            if sz > 1e-15:
                z = z * norm / sz

        b_z = fast_autocorr(z)
        max_val = b_z.max()
        if max_val < 1e-15:
            break

        # Active set: lags within 1% of max
        threshold = max_val * 0.99
        active_mask = b_z >= threshold
        n_active = active_mask.sum()
        if n_active == 0:
            active_mask[np.argmax(b_z)] = True
            n_active = 1

        # Compute weights (uniform over active set)
        w = np.zeros(2 * n - 1)
        w[active_mask] = 1.0 / n_active

        # Compute subgradient using FFT
        grad = fft_subgradient(z, w)

        # Adaptive step size
        h = h0 / (1.0 + 0.005 * it)

        y_prev = y.copy()
        y_new = y - h * grad
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        y = y_new

        it += 1
        score = fast_autocorr(y).max()
        if score < best_score:
            best_score = score
            best_a = y.copy()

        if verbose and it % 2000 == 0:
            print(f"    SubGrad it={it} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lse_gradient(a, T):
    """Compute LSE surrogate gradient using FFT."""
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    w = np.exp((b - max_b) / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[n - 1] = 1.0
    else:
        w /= w_sum

    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)

    conv = irfft(A * W, fft_size)
    part2 = conv[:n]

    cross = irfft(W * np.conj(A), fft_size)
    part1 = cross[:n]

    return part1 + part2


def nesterov_lse(a0, time_limit, T_start=10.0, T_end=0.2, n_stages=10, verbose=True):
    """Nesterov-accelerated LSE gradient descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    y_prev = a.copy()
    it_total = 0

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2.0:
            break

        n_iters = max(30, int(remaining * 2.5 / max(1, n_stages - si)))
        for it in range(n_iters):
            if time.monotonic() - t0 > time_limit:
                break

            if it == 0:
                z = y
            else:
                beta = (1.0 + np.sqrt(1.0 + 4.0 * (it - 1) ** 2)) / (2.0 * (it + 1))
                z = y + beta * (y - y_prev)
                z = np.maximum(z, 0.0)
                sz = z.sum()
                if sz > 1e-15:
                    z = z * norm / sz

            grad = lse_gradient(z, T)
            h = 1.0 / (T * (it + 3.0) ** 0.5)

            y_prev = y.copy()
            y_new = y - h * grad
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y
            y = y_new
            it_total += 1

            score = fast_autocorr(y).max()
            if score < best_score:
                best_score = score
                best_a = y.copy()

            if it_total % 500 == 0 and verbose:
                print(f"    LSE stage={si} T={T:.3f} it={it_total} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def local_search(a0, time_limit, seed=42, verbose=True):
    """Peak-targeting local search with random pair swaps."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        k_peak = np.argmax(b)

        # Target the peak: find which positions contribute most
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]
        top_idx = idx[np.argsort(contrib)[::-1][:5]]

        # Try moving mass from top contributors to low positions
        improved = False
        for src in top_idx:
            if a[src] < norm * 1e-9:
                continue
            # Find destination: random low-value position
            for _ in range(10):
                dst = rng.integers(n)
                if dst == src:
                    continue
                for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-14:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        it += 1
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 500:
                break
        else:
            stall = 0

        if verbose and it % 500 == 0:
            print(f"    Local it={it} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving relative shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search: LP best-response + subgradient + local search at multiple lengths."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1650.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver={solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def do_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)}, t={time.monotonic()-t_global:.0f}s)")

    if best_seq is None:
        n_init = 1600
        best_seq = np.ones(n_init) / n_init
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    # ===== Phase 1: LP best-response (main workhorse) =====
    remaining = time_budget - (time.monotonic() - t_global)
    n1 = len(best_seq)
    print(f"\n=== Phase 1: LP-Nesterov n={n1} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.40, 600.0)
    if t1 > 60:
        a_work = best_seq * np.sqrt(2.0 * n1) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_work, t1, solver, verbose=True, window_size=100)
        do_update(a_out, f"lp1_n{n1}")

    # ===== Phase 2: Subgradient polish =====
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq)
    print(f"\n=== Phase 2: SubGrad n={n2} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 20:
        a_work = best_seq * np.sqrt(2.0 * n2) / best_seq.sum()
        a_out, _ = subgradient_polish(a_work, t2, verbose=True)
        do_update(a_out, f"subgrad_n{n2}")

    # ===== Phase 3: LP round 2 =====
    remaining = time_budget - (time.monotonic() - t_global)
    n3 = len(best_seq)
    print(f"\n=== Phase 3: LP round 2 n={n3} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 60:
        a_work = best_seq * np.sqrt(2.0 * n3) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_work, t3, solver, verbose=True, window_size=50)
        do_update(a_out, f"lp2_n{n3}")

    # ===== Phase 4: LSE refinement =====
    remaining = time_budget - (time.monotonic() - t_global)
    n4 = len(best_seq)
    print(f"\n=== Phase 4: LSE n={n4} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 15:
        a_work = best_seq * np.sqrt(2.0 * n4) / best_seq.sum()
        a_out, _ = nesterov_lse(a_work, t4, T_start=5.0, T_end=0.1, n_stages=8, verbose=True)
        do_update(a_out, f"lse_n{n4}")

    # ===== Phase 5: Local search =====
    remaining = time_budget - (time.monotonic() - t_global)
    n5 = len(best_seq)
    print(f"\n=== Phase 5: LocalSearch n={n5} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 10:
        a_work = best_seq * np.sqrt(2.0 * n5) / best_seq.sum()
        a_out, _ = local_search(a_work, t5, seed=42, verbose=True)
        do_update(a_out, f"local_n{n5}")

    # ===== Phase 6: Try longer length =====
    n6 = n5 + 200
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: LP on n={n6} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.25, 300.0)
    if t6 > 60:
        init6 = interpolate_sequence(best_seq, n6)
        a_work = init6 * np.sqrt(2.0 * n6) / init6.sum()
        a_out, _ = nesterov_lp_optimize(a_work, t6, solver, verbose=True, window_size=80)
        do_update(a_out, f"lp_n{n6}")

    # ===== Phase 7: Final subgradient + local search =====
    remaining = time_budget - (time.monotonic() - t_global)
    n7 = len(best_seq)
    t7a = min(remaining * 0.4, 100.0)
    if t7a > 10:
        print(f"\n=== Phase 7a: SubGrad n={n7} (budget={t7a:.0f}s) ===")
        a_work = best_seq * np.sqrt(2.0 * n7) / best_seq.sum()
        a_out, _ = subgradient_polish(a_work, t7a, verbose=True)
        do_update(a_out, "subgrad_final")

    remaining = time_budget - (time.monotonic() - t_global)
    n7b = len(best_seq)
    t7b = min(remaining * 0.5, 100.0)
    if t7b > 10:
        print(f"\n=== Phase 7b: Local n={n7b} (budget={t7b:.0f}s) ===")
        a_work = best_seq * np.sqrt(2.0 * n7b) / best_seq.sum()
        a_out, _ = local_search(a_work, t7b, seed=999, verbose=True)
        do_update(a_out, "local_final")

    # Final
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
