def build_conv_matrix(a, n):
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    return T


def lp_best_response_with_dual(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    size = 2 * n - 1
    T = build_conv_matrix(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None, None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    dual_val = None

        dual_val = np.asarray(prob.constraints[0].dual_value, dtype=np.float64)
        dual_val = np.maximum(dual_val, 0.0)
    except Exception:
        pass
    return b_val, dual_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):

    stall_count = 0

            break

        if M_curr < 1e-12:

        if b is None:

        g = b * norm / b_sum
        n_iters += 1

            # If dual is concentrated (low entropy), we're close to optimum -> smaller step
            # If dual is spread (high entropy), more room to move -> larger step
            d_factor = 0.5 + 0.5 * d_entropy
        else:
            d_factor = 1.0

        alpha = min(alpha * d_factor, 1.0)
        # Ensure minimum progress

        y_new = np.maximum(y_new, 0.0)

        else:
            y_new = y_new * norm / s_y

            a_new = g.copy()

        a = a_new

            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()

        avg_a += a
        n_avg += 1

def lse_objective_and_grad(a_flat, n, T_temp, top_k=5):
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    # Use top-k peaks for the LSE to focus on the important constraints
    if top_k < len(b):
        top_idx = np.argsort(b)[::-1][:top_k]
        b_sorted = b[top_idx]
    else:
        b_sorted = b

    max_b = b_sorted.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)

    shifted = T_temp * (b_sorted - max_b)
    exp_vals = np.exp(shifted)
    lse = max_b + np.log(exp_vals.sum()) / T_temp
    obj = 2.0 * n * lse / (s * s)

    # Weight for gradient
    w = exp_vals / exp_vals.sum() * top_k  # rescale to approximate full LSE weight

    # Reconstruct full weight vector
    w_full = np.zeros(2 * n - 1)
    w_full[top_idx] = w if top_k < len(b) else w

    # Gradient via FFT cross-correlation
    fft_size = 1 << (3 * n - 2).bit_length()
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    w_padded = np.zeros(fft_size)
    w_padded[:2 * n - 1] = w_full
    A_fft = rfft(a_padded, fft_size)
    W_fft = rfft(w_padded, fft_size)
    cross = irfft(A_fft * np.conj(W_fft), fft_size)[:n]

    conv_grad = 4.0 * cross
    grad = conv_grad / (s * s) - 2.0 * obj * a / s
    return obj, grad.ravel()

        top_peaks = np.argsort(b)[::-1][:10]
        improved = False

        for k_peak in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k_peak - idx]
            top_src = idx[np.argsort(contrib)[::-1][:10]]

            for i in top_src:
                if a[i] < 1e-10:
                    continue
                # Try swapping mass between i and a random position
                for _ in range(20):
                    j = rng.integers(n)
                    if j == i:
                        continue
                    for frac in [0.05, 0.15, 0.3, 0.5]:
                        amt = min(a[i], a[j]) * frac if frac < 1.0 else a[i] * 0.5
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[i] -= amt
                        a_trial[j] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-10:
                            continue
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-13:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 300:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(x) + 0.2 * np.cos(3.0 * x) + 0.1 * np.cos(7.0 * x)
    elif choice == 2:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 4
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 2.0 * x) + 0.15 * np.cos(2.0 * np.pi * 5.0 * x)
    elif choice == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 4.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 6:
        # Fourier-anisotropic: modulate with multiple frequencies
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.2 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x) - 0.05 * np.cos(6 * np.pi * x)
    else:
        a = np.ones(n) * (1.0 + 0.1 * rng.standard_normal(n))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to use parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 0: Multi-length diverse initialization scan ===
    print("\n=== Phase 0: Multi-length scan ===")
    scan_lengths = [1600, 2000, 2400]
    scan_results = {}

    for n_scan in scan_lengths:
        if time.monotonic() - t_global > time_budget * 0.12:
            break
        scan_time = min((time_budget * 0.12 - (time.monotonic() - t_global)) /
                        max(1, len(scan_lengths) - scan_lengths.index(n_scan)), 60.0)
        if scan_time < 8:
            break

        # Use parent if available, otherwise structured init
        if best_seq is not None:
            init_seq = interpolate_sequence(best_seq, n_scan)
        else:
            init_seq = structured_init(n_scan, rng)

        a_norm = init_seq * np.sqrt(2.0 * n_scan) / init_seq.sum()
        a_out, sc_out = nesterov_lp_optimize(a_norm, scan_time, solver, verbose=False)
        scan_results[n_scan] = (a_out, sc_out)
        print(f"    n={n_scan}: raw_score={sc_out:.8f} (t={time.monotonic()-t_global:.0f}s)")

    # Pick best from scan
    if scan_results:
        best_scan_n = min(scan_results, key=lambda k: scan_results[k][1])
        scan_best_a, _ = scan_results[best_scan_n]
        do_try_update(scan_best_a, f"scan_n{best_scan_n}")

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    # === Phase 1: Deep LP-Nesterov (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Deep LP n={n_base} (budget={remaining:.0f}s) ===")
    t1 = remaining * 0.40
    a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
    a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
    do_try_update(a_out, f"lp_deep_n{n_base}")

    # === Phase 2: Mirror descent (complementary trajectory) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: Mirror descent n={n_cur} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.12, 150.0)
    if t2 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = mirror_descent_optimize(a_norm, t2, verbose=False)
        do_try_update(a_out, f"mirror_n{n_cur}")

    # === Phase 3: LSE + L-BFGS (smoothing) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: LSE scipy n={n_cur} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lse_scipy_optimize(a_norm, t3, T_start=30.0, T_end=2.0, n_stages=5, top_k=5)
        do_try_update(a_out, f"lse_n{n_cur}")

    # === Phase 4: Try longer sequence ===
    n_long = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LP n={n_long} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 30:
        init_long = interpolate_sequence(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, verbose=True)
        do_try_update(a_out, f"lp_long_n{n_long}")

    # === Phase 5: Peak-targeted coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: Coord descent n={n_cur} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = peak_coord_descent(a_norm, t5, seed=99999)
        do_try_update(a_out, "coord_polish")

    # === Phase 6: Two-point swap descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: Swap descent n={n_cur} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.10, 120.0)
    if t6 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = two_point_swap_descent(a_norm, t6, seed=55555)
        do_try_update(a_out, "swap_descent")

    # === Phase 7: Final LSE polish with low temperature ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: LSE polish n={n_cur} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.15, 150.0)
    if t7 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lse_scipy_optimize(a_norm, t7, T_start=5.0, T_end=0.5, n_stages=4, top_k=3)
        do_try_update(a_out, "lse_polish")

    # === Phase 8: Final coordinate descent (long) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 8: Final coord n={n_cur} (budget={remaining:.0f}s) ===")
    t8 = remaining * 0.6
    if t8 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = peak_coord_descent(a_norm, t8, seed=77777)
        do_try_update(a_out, "coord_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
