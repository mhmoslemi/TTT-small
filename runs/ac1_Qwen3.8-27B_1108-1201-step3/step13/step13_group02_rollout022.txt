def lse_lbfgs(a_init, n, T, maxiter=1000):
    """Minimize LSE-smoothed autocorrelation peak using L-BFGS."""
    norm = np.sqrt(2.0 * n)
    # Initial z from a_init
    a_init = np.maximum(a_init, 1e-10)  # Ensure positivity for log
    z_init = np.log(a_init / norm)  # Since a = norm * softmax(z), z = log(a/norm) + const
    # Actually: a_i = norm * exp(z_i) / S, so a_i/norm = exp(z_i)/S
    # log(a_i/norm) = z_i - log(S)
    # We can set z_init = log(a_i/norm) (the constant -log(S) doesn't matter for softmax)
    
    def obj_grad(z):
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a = norm * exp_z / exp_z.sum()
        
        b = fast_autocorr(a)
        b_max = b.max()
        
        if T < 1e-8:
            # Just return the max
            return b_max, np.zeros(n)
        
        shifted = (b - b_max) / T
        shifted_max = shifted.max()
        log_sum_exp = np.log(np.exp(shifted - shifted_max).sum())
        value = b_max + T * log_sum_exp
        
        p = np.exp(shifted - shifted_max)
        p = p / p.sum()
        
        grad_a = np.correlate(p, a, mode='valid')
        dot_ga = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot_ga / norm)
        
        return value, grad_z
    
    result = minimize(obj_grad, z_init, method='L-BFGS-B', jac=True, options={'maxiter': maxiter, 'gtol': 1e-8})
    
    # Convert back to a
    z_opt = result.x
    z_max = z_opt.max()
    exp_z = np.exp(z_opt - z_max)
    a_opt = norm * exp_z / exp_z.sum()
    
    return a_opt, value, result

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_solve(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_conv_matrix(a)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    return np.maximum(b_val, 0.0)


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0

    y = a.copy()

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

                break
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue

        n_iters += 1

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)

            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0

        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, stall={stall}")

            if s_y > 1e-15:

            if s_a > 1e-15:

            stall = 0
            if verbose:

def lse_lbfgs_optimize(a0, time_limit, verbose=True):
    """L-BFGS on LSE-smoothed autocorrelation with adaptive temperature annealing."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    def make_obj_grad(T):
        def obj_grad(z):
            z_clip = np.clip(z, -30, 30)
            z_max = z_clip.max()
            exp_z = np.exp(z_clip - z_max)
            a_local = norm * exp_z / exp_z.sum()

            b = fast_autocorr(a_local)
            b_max = b.max()

            if T < 1e-10:
                shifted = np.zeros_like(b)
                shifted[np.argmax(b)] = 1.0
                p = shifted
                value = b_max
            else:
                shifted = (b - b_max) / T
                shifted_max = shifted.max()
                log_sum_exp = np.log(np.exp(shifted - shifted_max).sum())
                value = b_max + T * log_sum_exp
                p = np.exp(shifted - shifted_max)
                p = p / p.sum()

            a_rev = a_local[::-1]
            fft_size = 1 << (3 * n - 2).bit_length()
            P = rfft(p, fft_size)
            AR = rfft(a_rev, fft_size)
            conv = irfft(P * AR, fft_size)
            grad_a = 2.0 * conv[n - 1:2 * n - 1]

            dot_ga = np.dot(grad_a, a_local)
            grad_z = a_local * (grad_a - dot_ga / norm)

            return value, grad_z
        return obj_grad

    # Temperature annealing schedule
    T_start = max(0.5, best_score * 0.15)
    T_schedule = []
    T = T_start
    for i in range(30):
        T_schedule.append(T)
        T *= 0.7
        if T < 0.001:
            break

    z_init = np.log(a / norm + 1e-30)

    for idx, T in enumerate(T_schedule):
        if time.monotonic() - t0 > time_limit:
            break

        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        maxiter = max(50, int(remaining / 0.01))
        obj_grad = make_obj_grad(T)

        try:
            result = minimize(obj_grad, z_init, method='L-BFGS-B', jac=True,
                             options={'maxiter': maxiter, 'gtol': 1e-10,
                                      'maxls': 50, 'maxcor': 20})
        except Exception as e:
            if verbose:
                print(f"    LSE-LBFGS T={T:.4f}: error {e}")
            break

        z_opt = result.x
        z_opt = np.clip(z_opt, -30, 30)
        z_max = z_opt.max()
        exp_z = np.exp(z_opt - z_max)
        a_opt = norm * exp_z / exp_z.sum()

        score = fast_autocorr(a_opt).max()
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    LSE-LBFGS T={T:.4f} ({elapsed:.0f}s): best={best_score:.8f}, "
                      f"obj={result.fun:.8f}, status={result.status if hasattr(result, 'status') else '?'}")

        z_init = z_opt

        # Check if we've converged (objective barely changing)
        if hasattr(result, 'fun') and idx > 0:
            pass  # Continue annealing

    return best_a, best_score


def peak_shave_optimize(a0, time_limit, seed=42, verbose=True):
    """Multi-destination peak shaving with adaptive step sizes."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_improve = 0
    n_stall = 0

    T_start = max(0.3, best_score * 0.08)
    T_end = max(0.01, best_score * 0.003)

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = (time.monotonic() - t0) / time_limit
        T = T_start * (T_end / T_start) ** min(elapsed_frac, 1.0)

        b = fast_autocorr(a)
        b_max = b.max()
        top_k = np.argsort(b)[::-1][:min(12, 2 * n - 1)]

        improved = False
        for k in top_k:
            if b[k] < b_max * 0.85:
                break

            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            if contrib.max() < 1e-18:
                continue

            src = idx[np.argmax(contrib)]

            # Find positions that DON'T contribute to top peaks
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k[:6]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            non_peak = np.where(~peak_mask)[0]
            if len(non_peak) < 3:
                non_peak = np.setdiff1d(np.arange(n), [src])

            n_dest = min(6, len(non_peak))
            # Prefer low-value, non-peak positions
            sorted_dst = non_peak[np.argsort(a[non_peak])]
            candidates = sorted_dst[:min(n_dest * 4, len(sorted_dst))]
            selected_dst = candidates[rng.choice(len(candidates),
                                                  min(n_dest, len(candidates)),
                                                  replace=False)]

            # Try multiple redistribution fractions
            for frac in [0.03, 0.06, 0.1, 0.15, 0.25]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue

                dst_vals = a[selected_dst] + 1e-10
                weights = 1.0 / dst_vals
                weights = weights / weights.sum()
                amounts = amt * weights

                a_trial = a.copy()
                a_trial[src] -= amt
                for di, dst in enumerate(selected_dst):
                    a_trial[dst] += amounts[di]

                if a_trial[src] < -1e-12:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t

                score = fast_autocorr(a_trial).max()

                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    n_improve += 1
                    break
                elif rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                    a = a_trial
                    improved = True
                    break

        if not improved:
            n_stall += 1
            # Random pair swap
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-10:
                frac = rng.uniform(0.05, 0.2)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial * norm / a_trial.sum()
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        n_improve += 1
                    elif rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                        a = a_trial

            if n_stall > 300:
                # Larger perturbation
                n_pert = max(3, n // 30)
                idx_pert = rng.choice(n, size=n_pert, replace=False)
                a_trial = a.copy()
                for idx in idx_pert:
                    amt = a[idx] * rng.uniform(0.1, 0.5)
                    j = rng.integers(n)
                    a_trial[idx] -= amt
                    a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score * 1.02:
                        a = a_trial
                        if score < best_score:
                            best_score = score
                            best_a = a.copy()
                n_stall = 0

        if verbose and n_improve % 3000 == 0 and n_improve > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave improve={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
