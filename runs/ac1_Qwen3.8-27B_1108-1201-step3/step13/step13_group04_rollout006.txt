def fast_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

    v = np.asarray(v, dtype=np.float64)

    u = np.sort(v)[::-1]

    if s > 0:

    return w

5. Improved peak-targeting local search with simulated annealing
6. Mirror descent as complementary optimizer

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    return s

def fast_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * (np.arange(1, n + 1) - 1) > (cssv - target_sum)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        return np.full(n, target_sum / n)
    rho = rho_idx[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 1e-15:
        w = w * target_sum / s
    else:
        w = np.full(n, target_sum / n)
    return w

def lse_gradient(a, T):
    """Compute gradient of LSE surrogate T*log(sum exp(b_k/T)) where b = conv(a,a)."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    w = np.exp((b - max_b) / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[n - 1] = 1.0
    else:
        w /= w_sum

    fft_size = 1 << (3 * n - 2).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    A_fwd = rfft(a, fft_size)
    W = rfft(w, fft_size)

    conv1 = irfft(A_rev * W, fft_size)
    conv2 = irfft(A_fwd * W, fft_size)

    r1 = conv1[n - 1:n - 1 + n]
    r2 = conv2[:n]

    return r1 + r2

def lse_value(a, T):
    """Compute LSE surrogate value."""
    b = fast_conv(a)
    max_b = b.max()
    log_sum = max_b / T + np.log(np.sum(np.exp((b - max_b) / T)))
    return T * log_sum

def nesterov_lse(a0, time_limit, T_start=15.0, T_end=0.2, n_stages=14, verbose=True):
    """Nesterov-accelerated LSE gradient descent with geometric cooling."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()

    y = a.copy()
    y_prev = a.copy()

    temps = np.geomspace(T_start, T_end, n_stages)
    it_count = 0

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        n_iters_stage = max(30, int(remaining * 2.5 / max(1, n_stages - si)))
        for it in range(n_iters_stage):
            if time.monotonic() - t0 > time_limit:
                break

            if it == 0:
                y_new = y - (1.0 / 3.0) * lse_gradient(y, T)
                y_prev = y.copy()
            else:
                momentum = (it - 1.0 + np.sqrt(1.0 + 4.0 * (it - 1.0) ** 2)) / (2.0 * (it + 2.0))
                y_extrap = y + momentum * (y - y_prev)
                y_extrap = np.maximum(y_extrap, 0.0)
                s_e = y_extrap.sum()
                if s_e > 1e-15:
                    y_extrap = y_extrap * norm / s_e
                grad = lse_gradient(y_extrap, T)
                y_prev = y.copy()
                y_new = y_extrap - (1.0 / (it + 2.0)) * grad

            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y
            y = y_new
            it_count += 1

            score = fast_conv(y).max()
            if score < best_score:
                best_score = score
                best_a = y.copy()

            if it_count % 100 == 0:
                elapsed = time.monotonic() - t0
                if verbose:
                    print(f"    LSE-N it={it_count} T={T:.3f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def adam_lse(a0, time_limit, T_start=10.0, T_end=0.1, n_stages=10, lr=0.01, beta1=0.9, beta2=0.999, verbose=True):
    """Adam optimizer on LSE surrogate."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()

    m = np.zeros(n)
    v = np.zeros(n)
    it_count = 0

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        n_iters_stage = max(30, int(remaining * 2.0 / max(1, n_stages - si)))
        for it in range(n_iters_stage):
            if time.monotonic() - t0 > time_limit:
                break

            grad = lse_gradient(a, T)
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** (it + 1))
            v_hat = v / (1 - beta2 ** (it + 1))
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)

            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_a
            a = a_new
            it_count += 1

            score = fast_conv(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

            if it_count % 100 == 0:
                elapsed = time.monotonic() - t0
                if verbose:
                    print(f"    Adam-LSE it={it_count} T={T:.3f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def peak_subgradient_nesterov(a0, time_limit, n_active=8, verbose=True):
    """Peak-active subgradient with Nesterov acceleration and simplex projection."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_peak = fast_conv(a).max()
    t0 = time.monotonic()

    y = a.copy()
    y_prev = a.copy()
    it_count = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_conv(y)
        max_b = b.max()
        threshold = max_b * 0.90
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        active = active[np.argsort(b[active])[::-1][:n_active]]

        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / len(active)

        fft_size = 1 << (3 * n - 2).bit_length()
        if t == 1:
            y_extrap = y
        else:
            momentum = (t - 1.0) / (t + 2.0)
            y_extrap = y + momentum * (y - y_prev)
            y_extrap = np.maximum(y_extrap, 0.0)
            s_e = y_extrap.sum()
            if s_e > 1e-15:
                y_extrap = y_extrap / s_e

        A_rev = rfft(y_extrap[::-1].copy(), fft_size)
        A_fwd = rfft(y_extrap, fft_size)
        W = rfft(w, fft_size)
        conv1 = irfft(A_rev * W, fft_size)
        conv2 = irfft(A_fwd * W, fft_size)
        grad = conv1[n - 1:n - 1 + n] + conv2[:n]

        step_size = 0.5 / np.sqrt(t + 1.0)
        y_prev = y.copy()
        y_new = y_extrap - step_size * grad
        y_new = project_simplex(y_new, 1.0)
        y = y_new
        it_count += 1

        score = fast_conv(y).max()
        if score < best_peak:
            best_peak = score
            best_a = y.copy()

        if it_count % 50 == 0:
            elapsed = time.monotonic() - t0
            if verbose and it_count % 200 == 0:
                print(f"    SubGrad it={it_count} ({elapsed:.0f}s): peak={best_peak:.10f}")

        if it_count % 2000 == 0 and best_peak < 1e-6:
            break

    norm = np.sqrt(2.0 * n)
    return best_a * norm, best_peak

def mirror_descent_lse(a0, time_limit, T_start=8.0, T_end=0.15, n_stages=10, verbose=True):
    """Mirror descent (multiplicative weights) on LSE surrogate."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    it_count = 0

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        n_iters_stage = max(30, int(remaining * 1.5 / max(1, n_stages - si)))
        for it in range(n_iters_stage):
            if time.monotonic() - t0 > time_limit:
                break
            grad = lse_gradient(a, T)
            eta = 1.0 / (T * (it + 1) ** 0.4)
            log_a = np.log(a + 1e-30) - eta * grad
            log_a -= log_a.max()
            a_new = np.exp(log_a)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s
            a = a_new
            it_count += 1
            score = fast_conv(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
            if it_count % 200 == 0:
                elapsed = time.monotonic() - t0
                if verbose:
                    print(f"    MirrorDesc it={it_count} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. (a*b)_j <= M for all j, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    from scipy.sparse import lil_matrix
    T = lil_matrix((2 * n - 1, n))
    for j in range(2 * n - 1):
        for k in range(n):
            i = j - k
            if 0 <= i < n:
                T[j, k] = a[i]
    T = T.tocsr()
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()

    y = a.copy()
    a_nes = np.zeros(n)
    n_iters = 0

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break
        M_curr = fast_conv(y).max()
        if M_curr < 1e-12:
            break
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_nes = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_nes = y_new + momentum * (y_new - y)
            a_nes = np.maximum(a_nes, 0.0)
            s_a = a_nes.sum()
            if s_a < 1e-15:
                a_nes = y_new.copy()
            else:
                a_nes = a_nes * norm / s_a

        y = y_new

        score_y = fast_conv(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        score_a = fast_conv(a_nes).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a_nes.copy()

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nesh iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def peak_targeting_search(a0, time_limit, seed=42, verbose=True):
    """Improved peak-targeting local search with larger steps and SA acceptance."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_conv(a).max()
    t0 = time.monotonic()
    temp_sa = 0.01 * best_score
    it_count = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 < time_limit * 0.5:
            temp_sa = max(0.001 * best_score, temp_sa * 0.999)
        else:
            temp_sa = 0.0

        b = fast_conv(a)
        top_ks = np.argsort(b)[::-1][:8]

        improved = False
        for _ in range(20):
            k_star = top_ks[rng.integers(len(top_ks))]
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_star - idx]
            top_contrib_idx = idx[np.argsort(contrib)[::-1][:5]]

            src = top_contrib_idx[rng.integers(len(top_contrib_idx))]
            if a[src] < norm * 1e-8:
                continue

            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_ks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]

            if len(candidates) > 10:
                candidates = candidates[np.argsort(a[candidates])[:10]]

            dst = candidates[rng.integers(len(candidates))]
            frac = rng.choice([0.01, 0.03, 0.05, 0.1, 0.2, 0.3])
            amt = a[src] * frac
            if amt < 1e-16:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            if a_trial[src] < -1e-16:
                continue
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial * norm / s_t
            score = fast_conv(a_trial).max()

            accept = False
            if score < best_score:
                accept = True
            elif temp_sa > 0 and score < best_score + temp_sa:
                accept = True

            if accept:
                a = a_trial
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                improved = True
                stall = 0
                it_count += 1
                if best_score < 1e-6:
                    break

        if not improved:
            stall += 1
            if rng.integers(20) == 0:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > norm * 1e-6:
                    frac = rng.uniform(0.01, 0.15)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = fast_conv(a_trial).max()
                        if score < best_score or (temp_sa > 0 and score < best_score + temp_sa):
                            a = a_trial
                            if score < best_score:
                                best_score = score
                                best_a = a.copy()
                            stall = 0

            if stall > 300:
                if verbose:
                    print(f"    PeakSearch: stalled at {best_score:.8f} ({time.monotonic()-t0:.0f}s)")
                break

        if verbose and it_count % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakSearch it={it_count} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

def structured_init(n, seed):
    """Generate a diverse structured initialization."""
    rng = np.random.default_rng(seed)
    choice = rng.integers(0, 7)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(2.0 * x) + 0.1 * np.cos(7.0 * x) + 0.05 * np.cos(13.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        a = np.ones(n) * 0.5
        for _ in range(rng.integers(3, 8)):
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.05, n * 0.3)
            amp = rng.uniform(-0.3, 0.5)
            a += amp * np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
    else:
        freqs = rng.integers(2, 20, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        amps = rng.uniform(0.1, 0.5, size=5)
        x = np.linspace(0, 2 * np.pi, n)
        a = np.ones(n) * 0.8
        for f_val, p_val, a_val in zip(freqs, phases, amps):
            a += a_val * np.cos(f_val * x + p_val)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # ============ Phase 1: Nesterov LSE at base length ============
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LSE-Nesterov n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.25, 400.0)
    if t1 > 30:
        a_work = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lse(a_work, t1, T_start=12.0, T_end=0.15, n_stages=16, verbose=True)
        do_try_update(a_out, f"lse_n{n_base}")

    # ============ Phase 2: Adam LSE at base length ============
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Adam-LSE n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 20:
        n_best = len(best_seq)
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = adam_lse(a_work, t2, T_start=8.0, T_end=0.1, n_stages=10, lr=0.05, verbose=True)
        do_try_update(a_out, f"adam_n{n_best}")

    # ============ Phase 3: Peak subgradient Nesterov ============
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: PeakSubGrad n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 250.0)
    if t3 > 20:
        n_best = len(best_seq)
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_subgradient_nesterov(a_work, t3, n_active=10, verbose=True)
        do_try_update(a_out, f"subgrad_n{n_best}")

    # ============ Phase 4: LP Nesterov ============
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LP-Nesterov n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 30:
        n_best = len(best_seq)
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_work, t4, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n_best}")

    # ============ Phase 5: LSE at longer length ============
    n2 = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: LSE-Nesterov n={n2} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 250.0)
    if t5 > 30:
        init_long = interpolate_sequence(best_seq, n2)
        a_work = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = nesterov_lse(a_work, t5, T_start=15.0, T_end=0.2, n_stages=14, verbose=True)
        do_try_update(a_out, f"lse_n{n2}")

    # ============ Phase 6: LSE at even longer length ============
    n3 = n_base + 800
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: LSE-Nesterov n={n3} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.12, 200.0)
    if t6 > 20:
        init_long3 = interpolate_sequence(best_seq, n3)
        a_work = init_long3 * np.sqrt(2.0 * n3) / init_long3.sum()
        a_out, _ = nesterov_lse(a_work, t6, T_start=20.0, T_end=0.3, n_stages=12, verbose=True)
        do_try_update(a_out, f"lse_n{n3}")

    # ============ Phase 7: Peak targeting on best ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: PeakTargeting n={n_best} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.15, 250.0)
    if t7 > 10:
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_targeting_search(a_work, t7, seed=77777, verbose=True)
        do_try_update(a_out, f"peak_n{n_best}")

    # ============ Phase 8: Final LSE polish ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: LSE polish n={n_best} (budget={remaining:.0f}s) ===")
    t8 = min(remaining * 0.20, 200.0)
    if t8 > 10:
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lse(a_work, t8, T_start=3.0, T_end=0.05, n_stages=10, verbose=True)
        do_try_update(a_out, "lse_polish")

    # ============ Phase 9: Final peak targeting ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 10:
        t9 = min(remaining * 0.4, 120.0)
        if t9 > 5:
            print(f"\n=== Phase 9: Final peak targeting n={n_best} (budget={t9:.0f}s) ===")
            a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = peak_targeting_search(a_work, t9, seed=123456, verbose=True)
            do_try_update(a_out, "peak_final")

    # ============ Phase 10: Second LP round if time remains ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 60:
        t10 = min(remaining * 0.5, 150.0)
        print(f"\n=== Phase 10: LP round 2 n={n_best} (budget={t10:.0f}s) ===")
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_work, t10, solver, verbose=True)
        do_try_update(a_out, "lp_round2")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
