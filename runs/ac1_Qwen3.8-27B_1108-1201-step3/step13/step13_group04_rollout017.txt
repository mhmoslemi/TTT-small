def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)

def nesterov_lse(a0, time_limit, T_start=30.0, T_end=0.1, n_stages=20, verbose=True):
    """
    Nesterov-accelerated LSE gradient descent with improved temperature schedule.
    Includes Polyak averaging.
    """
    n = len(a0)
    a = normalize_nonneg(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Polyak averaging
    avg_a = np.zeros(n)
    n_avg = 0
    
    y = a.copy()
    y_prev = a.copy()
    global_it = 0
    
    # Temperature schedule: slow start, fast end
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2.0:
            break
        
        # More iterations at high T (exploration), fewer at low T (exploitation)
        if si < n_stages // 3:
            n_iters_stage = max(30, int(remaining * 2.5 / max(1, n_stages - si)))
        elif si < 2 * n_stages // 3:
            n_iters_stage = max(30, int(remaining * 3.0 / max(1, n_stages - si)))
        else:
            n_iters_stage = max(20, int(remaining * 4.0 / max(1, n_stages - si)))
        
        for it in range(n_iters_stage):
            if time.monotonic() - t0 > time_limit:
                break
            global_it += 1
            
            # Nesterov extrapolation
            if global_it <= 2:
                y_eval = y
            else:
                momentum = (global_it - 2.0 + np.sqrt(1.0 + 4.0*(global_it-2.0)**2)) / (2.0*(global_it+1.0))
                y_eval = y + momentum * (y - y_prev)
                y_eval = np.maximum(y_eval, 0.0)
                s_ve = y_eval.sum()
                if s_ve > 1e-15:
                    y_eval = y_eval / s_ve
                else:
                    y_eval = y.copy()
            
            grad = lse_gradient(y_eval, T)
            
            # Step size
            step = 1.0 / (global_it + 3.0)
            y_new = y - step * grad
            y_new = np.maximum(y_new, 0.0)
            s_yn = y_new.sum()
            if s_yn < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / s_yn
            
            y_prev = y.copy()
            y = y_new
            
            # Track best
            score = fast_autocorr(y).max()
            if score < best_score:
                best_score = score
                best_a = y.copy()
            
            # Polyak averaging (start after first few iterations)
            if global_it > n_stages * 5:

                n_avg += 1
                if n_avg % 100 == 0:

                    score_avg = fast_autocorr(avg_check).max()
                    if score_avg < best_score:
                        best_score = score_avg
                        best_a = avg_check.copy()
            
            if verbose and global_it % 300 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LSE-Nest stage={si}/{n_stages} T={T:.3f} it={global_it} ({elapsed:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score

def multiplicative_weights_nesterov(a0, time_limit, verbose=True):

    Faster than LSE when we can use the active-set approximation.
    """

    best_a = a.copy()

    u_prev = u.copy()

    n_avg = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < best_score:
            best_score = max_b
            best_a = a.copy()
        
        # Soft active set with adaptive temperature
        # Use a relatively low T to focus on the peaks
        T = max(0.005, 0.05 * max_b)
        w = np.exp((b - max_b) / T)

        if w_sum < 1e-300:
            w = np.zeros(2*n-1)
            w[n-1] = 1.0
        else:
            w /= w_sum
        
        # Compute pressure (same as LSE gradient)
        fft_size = 1 << (3*n-2).bit_length()
        A_rev = rfft(a[::-1].copy(), fft_size)
        A_fwd = rfft(a, fft_size)
        W = rfft(w, fft_size)
        conv1 = irfft(A_rev * W, fft_size)
        conv2 = irfft(A_fwd * W, fft_size)
        p = conv1[n-1:n-1+n] + conv2[:n]
        
        # Normalize pressure
        p_mean = p.mean()
        if p_mean < 1e-15:
            break
        p_norm = p / p_mean
        
        # Learning rate
        eta = 0.8 / (it ** 0.6)
        
        # Nesterov in log-space
        if it <= 2:
            u_eval = u
        else:
            momentum = (it - 2.0 + np.sqrt(1.0 + 4.0*(it-2.0)**2)) / (2.0*(it+1.0))
            u_eval = u + momentum * (u - u_prev)
        
        u_new = u_eval - eta * p_norm
        u_prev = u.copy()
        
        # Convert back: a = exp(u) / sum(exp(u))
        u_new -= u_new.max()
        a_new = np.exp(u_new)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new /= s
        a = a_new
        
        # Polyak averaging
        if it > 100:
            avg_a += a
            n_avg += 1
            if n_avg % 200 == 0:
                avg_check = normalize_nonneg(avg_a / n_avg)
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MW-Nest it={it} ({elapsed:.0f}s): best={best_score:.10f}")
        
        if best_score < 1e-12:
            break
    
    return best_a, best_score

def peak_targeting_search(a0, time_limit, seed=42, verbose=True):
    """
    Peak-targeting local search with adaptive move sizes.
    Tries to reduce the peak of the autocorrelation by moving mass.
    """

    rng = np.random.default_rng(seed)
    a = normalize_nonneg(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    it_count = 0

        b = fast_autocorr(a)
        # Get top peaks
        top_k = 8
        peak_indices = np.argsort(b)[::-1][:top_k]
        
        improved = False
        for k_star in peak_indices:
            # Find which (i, k-i) pairs contribute most to this peak
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue

            # Top contributors
            n_top = min(5, len(idx))
            top_contrib_idx = idx[np.argsort(contrib)[::-1][:n_top]]
            
            for src in top_contrib_idx:
                if a[src] < best_a.sum() * 1e-10:
                    continue
                
                # Find source of mass to move: positions that contribute to multiple peaks
                peak_mask = np.zeros(n, dtype=bool)
                for k2 in peak_indices:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    if js2 <= n-1 and je2 >= 0:
                        peak_mask[js2:je2+1] = True
                
                # Candidates: positions NOT in peak regions (or with low values)
                candidates = np.where(~peak_mask)[0]
                if len(candidates) == 0:
                    candidates = np.where(np.arange(n) != src)[0]
                
                # Sort by value (prefer low-value positions)
                candidates = candidates[np.argsort(a[candidates])]
                # Take a spread of candidates

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Top-K subgradient Nesterov (fast O(nK) per step, no FFT)
2. LSE (log-sum-exp) Nesterov gradient descent with improved temperature schedule
3. Polyak averaging of iterates for variance reduction
4. Peak-targeting local search with adaptive moves and random restarts
5. Multi-length exploration with interpolation transfer
Key improvements: top-K subgradient as fast pre-optimizer, better temperature
schedule with more stages, Polyak averaging, and more aggressive peak targeting.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Compute full autocorrelation a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_gradient(a, T):
    """
    Compute gradient of LSE surrogate: T*log(sum_k exp(b_k/T)) where b = autocorr(a).
    dL/da[m] = sum_k w[k] * (a[k-m] + a[m-k])
    """
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    w = np.exp((b - max_b) / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[n - 1] = 1.0
    else:
        w /= w_sum

    fft_size = 1 << (3 * n - 2).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    A_fwd = rfft(a, fft_size)
    W = rfft(w, fft_size)
    conv1 = irfft(A_rev * W, fft_size)
    conv2 = irfft(A_fwd * W, fft_size)
    return conv1[n - 1:n - 1 + n] + conv2[:n]


def top_k_subgradient(a, K):
    """
    Compute a subgradient using the top K lags of the autocorrelation.
    Much faster than LSE gradient (O(n*K) vs O(n log n)).
    Returns the average subgradient over the K most active lags.
    """
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    
    # Find top K lags
    top_k = np.argsort(b)[::-1][:K]
    
    # Soft weights: larger weight for lags closer to max
    w = np.zeros(2 * n - 1)
    for k in top_k:
        w[k] = np.exp((b[k] - max_b) / (max_b * 0.1 + 1e-15))
    w_sum = w.sum()
    if w_sum < 1e-300:
        return np.ones(n) / n
    w /= w_sum
    
    # Compute gradient using the top-k weights (sparse convolution)
    # grad[i] = sum_k w[k] * (a[k-i] + a[i-k]) for valid indices
    grad = np.zeros(n)
    for k in top_k:
        if w[k] < 1e-15:
            continue
        # Contribution from lag k: a[k-i] + a[i-k] for valid i
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        # a[k - i] for i in idx
        grad[idx] += w[k] * a[k - idx]
        # a[i - k] for i in idx where i-k >= 0, i.e., i >= k
        valid = idx >= k
        if np.any(valid):
            grad[idx[valid]] += w[k] * a[idx[valid] - k]
    return grad


def normalize_nonneg(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def evaluate_safe(a):
    """Safely evaluate a sequence using the provided evaluate_sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving relative shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def nesterov_subgradient(a0, time_limit, K=16, verbose=True):
    """
    Nesterov-accelerated top-K subgradient descent.
    Fast: O(n*K) per step, no FFT for gradient (only for autocorrelation check).
    """
    n = len(a0)
    a = normalize_nonneg(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    y_prev = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov extrapolation
        if it <= 2:
            y_eval = y
        else:
            momentum = (it - 2.0 + np.sqrt(1.0 + 4.0 * (it - 2.0) ** 2)) / (2.0 * (it + 1.0))
            y_eval = y + momentum * (y - y_prev)
            y_eval = np.maximum(y_eval, 0.0)
            s_ve = y_eval.sum()
            if s_ve > 1e-15:
                y_eval = y_eval / s_ve
            else:
                y_eval = y.copy()

        # Compute top-K subgradient
        grad = top_k_subgradient(y_eval, K)

        # Step size: 1/sqrt(it) for subgradient methods
        step = 1.0 / np.sqrt(it + 1.0)
        y_new = y - step * grad
        y_new = np.maximum(y_new, 0.0)
        s_yn = y_new.sum()
        if s_yn < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_yn

        y_prev = y.copy()
        y = y_new

        # Track best
        score = fast_autocorr(y).max()
        if score < best_score:
            best_score = score
            best_a = y.copy()

        # Polyak averaging (after warmup)
        if it > 50:
            avg_a += y
            n_avg += 1
            if n_avg % 200 == 0:
                avg_check = normalize_nonneg(avg_a / n_avg)
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad-Nest it={it} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score

    """

    y_prev = a.copy()
    global_it = 0

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2.0:
            break

        # More iterations at low T (exploitation phase)

        # Cap per-stage iterations

        for it in range(n_iters_stage):
            if time.monotonic() - t0 > time_limit:
                break
            global_it += 1

            # Nesterov extrapolation

                y_eval = y
            else:

                y_eval = y + momentum * (y - y_prev)

    t2 = min(remaining * 0.42, 700.0)
    if t2 > 30:
        a_out, _ = nesterov_lse(best_seq, t2, T_start=25.0, T_end=0.2, n_stages=24, verbose=True)
        do_try_update(a_out, f"lse_n{n_best}")

    # ============ Phase 3: LSE polish (very low T) ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LSE polish n={n_best} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.18, 280.0)
    if t3 > 15:
        a_out, _ = nesterov_lse(best_seq, t3, T_start=3.0, T_end=0.05, n_stages=12, verbose=True)
        do_try_update(a_out, f"lse_polish_n{n_best}")

    # ============ Phase 4: Peak targeting ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: PeakTargeting n={n_best} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.25, 350.0)
    if t4 > 10:
        a_out, _ = peak_targeting_search(best_seq, t4, seed=12345, verbose=True)
        do_try_update(a_out, f"peak_n{n_best}")

    # ============ Phase 5: Try slightly different length ============
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_alt = n_best + 100
        print(f"\n=== Phase 5: LSE at n={n_alt} (budget={remaining:.0f}s) ===")
        t5 = min(remaining * 0.5, 250.0)
        if t5 > 20:
            init_alt = interpolate_sequence(best_seq, n_alt)
            a_out, _ = nesterov_lse(init_alt, t5, T_start=15.0, T_end=0.3, n_stages=12, verbose=True)
            do_try_update(a_out, f"lse_n{n_alt}")

    # ============ Phase 6: Final peak targeting polish ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 20:
        t6 = min(remaining * 0.7, 150.0)
        if t6 > 10:
            print(f"\n=== Phase 6: Final peak targeting n={n_best} (budget={t6:.0f}s) ===")
            a_out, _ = peak_targeting_search(best_seq, t6, seed=99999, verbose=True)
            do_try_update(a_out, "peak_final")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
