def normalize(a, S):
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) * S / len(a)
    return a * S / s

def lse_gradient(a, w, N):
    """Compute gradient of LSE_T(conv(a,a)) w.r.t. a using FFT."""
    Aw = rfft(a, N)
    Ww = rfft(w, N)
    F = irfft(Ww * np.conj(Aw), N)[:len(a)]
    G = irfft(Ww * Aw, N)[:len(a)]
    return F + G

def lse_mirror_descent(a_init, time_limit, T_start=50.0, T_end=0.1, 
                       n_stages=12, eta_scale=1.0, verbose=True):

    a = normalize(a, S)

    temps = np.geomspace(T_start, T_end, n_stages)

        remaining = time_limit - elapsed
        if remaining < 1:
            break

            w = np.exp(shifted)
            w_sum = w.sum()
            if w_sum < 1e-300:
                break
            w /= w_sum

            exponent = np.clip(exponent, -50.0, 50.0)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss
