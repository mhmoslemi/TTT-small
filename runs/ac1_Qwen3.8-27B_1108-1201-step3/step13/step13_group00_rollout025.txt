def solve_full_lp(a, M, solver):
    """Solve: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_conv_matrix(a)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, None
    # Try to extract dual variables
    dual_vals = None
    try:
        cv = prob.constraints[0].dual_value
        if cv is not None:
            dual_vals = np.maximum(np.asarray(cv, dtype=np.float64), 0.0)
    except Exception:
        pass
    return b_val, dual_vals

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, use_full_lp=True):
    """Nesterov-accelerated LP best-response method."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    y_prev = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    n_iters = 0
    restart_count = 0
    best_stagnation = 0
    last_best_update = 0.0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        if use_full_lp:
            b, duals = solve_full_lp(y, M_curr, solver)
        else:
            b = solve_screened_lp(y, M_curr, solver)
            duals = None

        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov step with restart
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        # Check for restart: if score at y_new > score at y, restart momentum
        score_y_new = fast_autocorr(y_new).max()
        score_y = fast_autocorr(y).max()

        if t > 10 and score_y_new > score_y * 1.001:
            # Restart: reset momentum
            y_new = y.copy()
            y_prev = y_new.copy()
            restart_count += 1
            if verbose and restart_count <= 10:
                print(f"    NLP restart #{restart_count} at t={t}")
        else:
            y_prev = y.copy()

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            if restart_count > 0 and (t % 50 == 0):
                # After restart, use less momentum initially
                momentum = min(momentum, 0.5)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            best_stagnation = 0
            last_best_update = time.monotonic()
        else:
            best_stagnation += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            best_stagnation = 0
            last_best_update = time.monotonic()

        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:

            if s_avg > 1e-15:

                    last_best_update = time.monotonic()

            # To reduce conv[pk], we want a[pk-j] < a[pk-i], i.e., move from high to low sensitivity.

            # Sort by sensitivity: remove from high, add to low
            # But we also need to consider ALL conv positions, not just pk.
            # For now, focus on pk and check the global max afterward.

            # Find best donor (high sensitivity) and receiver (low sensitivity)
            # Only consider positions with significant mass
            min_mass = norm * 1e-6
            donors = np.where((sensit > np.median(sensit)) & (a > min_mass))[0]
            receivers = np.where((sensit < np.median(sensit)) & (a < sens_init_median(a)))[0]

            if len(donors) == 0 or len(receivers) == 0:
                no_improve_rounds += 1
                continue

            # Try top few donor-receiver pairs
            best_delta = 0.0
            best_new_max = conv.max()
            best_i = -1
            best_j = -1

            # Sort donors by descending sensitivity, receivers by ascending
            donors_sorted = donors[np.argsort(sensit[donors])[::-1]][:10]
            receivers_sorted = receivers[np.argsort(sensit[receivers])][:10]

            for i in donors_sorted:
                if time.monotonic() - t0 > time_limit:
                    break
                for j in receivers_sorted:
                    if j == i:
                        continue
                    # Compute the gain vector for moving delta from i to j
                    # Change in conv[k] = delta * (a[k-j]+a[j-k] - a[k-i]+a[i-k]) ... 
                    # More carefully:
                    # conv_new[k] = conv[k] + delta*(a[k-j] + a[j-k] - a[k-i] - a[i-k])
                    # where a[x] = 0 if x not in [0,n-1]
                    # (the + terms come from the new a[j]+delta, the - terms from a[i]-delta)
                    # Plus second order: at k=2i: (a[i]-d)^2 - a[i]^2 = -2a[i]d + d^2
                    #                    at k=2j: (a[j]+d)^2 - a[j]^2 = 2a[j]d + d^2

                    ai = a[i]
                    if ai < min_mass:
                        continue

                    # Compute gain vector g where conv_new[k] ≈ conv[k] + delta * g[k]
                    g = np.zeros(size)
                    # Contribution from +delta at position j:
                    # a[j] appears in conv[k] at k = j+r for r in [0,n-1] with weight a[r]
                    # and at k = j-r for r in [1,n-1] with weight a[r] (by symmetry of conv)
                    # Actually conv(a,b)[k] = sum_r a[r]*b[k-r]
                    # If b = a + delta*(e_j - e_i), then:
                    # conv(a, delta*e_j)[k] = delta * a[k-j] (if 0<=k-j<n)
                    # conv(delta*e_j, a)[k] = delta * a[j-k] (if 0<=j-k<n, i.e., j-n<k<=j)
                    # Wait, conv is not symmetric in general... but conv(a,a) IS symmetric.
                    # conv(a + d*(ej-ei), a + d*(ej-ei))[k]
                    # = conv(a,a)[k] + d*(conv(ej,a)[k] + conv(a,ej)[k] - conv(ei,a)[k] - conv(a,ei)[k])
                    #   + d^2*conv(ej-ei, ej-ei)[k]
                    # conv(ej, a)[k] = a[k-j] (if 0<=k-j<n)
                    # conv(a, ej)[k] = a[j-k] (if 0<=j-k<n) -> wait no
                    # conv(a, b)[k] = sum_i a[i]*b[k-i]
                    # conv(a, ej)[k] = a[j-k]... no. conv(a,ej)[k] = sum_i a[i]*ej[k-i] = a[k-j] (if 0<=k-j<n)
                    # conv(ej, a)[k] = sum_i ej[i]*a[k-i] = a[k-j] (if j is in [0,n-1] and the sum picks i=j)
                    # Wait: conv(ej,a)[k] = sum_i ej[i]*a[k-i] = a[k-j] (only i=j contributes)
                    # conv(a,ej)[k] = sum_i a[i]*ej[k-i] = a[i] where k-i=j, i=k-j, so = a[k-j] (if 0<=k-j<n)
                    # So both are a[k-j]! That makes sense since conv(a,a) is symmetric.
                    # So: conv(ej,a)[k] + conv(a,ej)[k] = 2*a[k-j] (if 0<=k-j<n)

                    # Therefore: g[k] = 2*(a[k-j] - a[k-i]) where valid, else just one side.
                    # More carefully:
                    # g[k] = 2*a[k-j] (if 0<=k-j<n) + 2*a[j-k] (if 0<=j-k<n)
                    #        - 2*a[k-i] (if 0<=k-i<n) - 2*a[i-k] (if 0<=i-k<n)
                    # But since a is defined on [0,n-1], a[x]=0 for x outside.
                    # And a[k-j] is valid when 0<=k-j<n, i.e., j<=k<=j+n-1.
                    # a[j-k] is valid when 0<=j-k<n, i.e., j-n+1<=k<=j.
                    # Note these ranges overlap at k=j (where both give a[0]=a[0]... wait a[j-j]=a[0])
                    # At k=j: a[k-j]=a[0] and a[j-k]=a[0]. But conv(ej,a)[j] = a[0] and conv(a,ej)[j] = a[0].
                    # So total is 2*a[0]. But wait, that's double-counting? No, because in conv(a,a),
                    # the term a[j]*a[0] appears once (when i=j, k-i=0) and the term a[0]*a[j] appears
                    # once (when i=0, k-i=j). They're different terms of the sum.
                    # So yes, g[k] = 2*a[max(0,k-j)] if j<=k<j+n, plus 2*a[max(0,j-k)] if j-n<k<=j,
                    # minus similar for i.
                    # Actually let me just use: g[k] = 2*(indicator(0<=k-j<n)*a[k-j] + indicator(0<=j-k<n)*a[j-k])
                    #                                  - 2*(indicator(0<=k-i<n)*a[k-i] + indicator(0<=i-k<n)*a[i-k])
                    # But at k=j: indicator(0<=0<n)=1, so first term is 2*a[0]. And indicator(0<=0<n)=1 for j-k=0.
                    # Second term: 2*a[0]. Total at k=j: 2*a[0]+2*a[0] - (similar for i) = 4*a[0] - ...
                    # Hmm, that doesn't seem right. Let me just compute it directly.

                    # Direct computation of g:
                    # For +delta at j:
                    #   conv gain at k = delta * (a[k-j] if 0<=k-j<n else 0) + delta * (a[j-k] if 0<=j-k<n else 0)
                    #   Wait, I think I'm confusing myself. Let me just use the matrix T.
                    # conv(a, delta*(ej-ei)) + conv(delta*(ej-ei), a)
                    # = delta * (T @ (ej-ei)) + delta * (T^T @ (ej-ei))  ... no that's not right either.
                    # 
                    # OK let me just compute it numerically. The gain from adding delta*e_j to a:
                    # new_a = a + delta*e_j
                    # new_conv = conv(new_a, new_a) = conv(a,a) + delta*conv(a,ej) + delta*conv(ej,a) + delta^2*conv(ej,ej)
                    # conv(a,ej)[k] = sum_i a[i]*ej[k-i] = a[k-j] if 0<=k-j<n else 0
                    # conv(ej,a)[k] = sum_i ej[i]*a[k-i] = a[k-j] if 0<=k-j<n else 0
                    # So conv(a,ej) = conv(ej,a) = a shifted. They're the same! (This is because ej is a delta.)
                    # So the linear gain from +delta at j is: 2*delta*a[k-j] (shifted a, valid range)
                    # Similarly, the linear gain from -delta at i is: -2*delta*a[k-i]

                    # So: g[k] = 2*(a[k-j] if 0<=k-j<n else 0) - 2*(a[k-i] if 0<=k-i<n else 0)

                    # Let me compute g directly:
                    g = np.zeros(size)
                    # +2*a shifted by j:
                    lo_j = max(0, j)
                    hi_j = min(size, j + n)
                    if hi_j > lo_j:
                        g[lo_j:hi_j] += 2.0 * a[lo_j - j:hi_j - j]
                    # -2*a shifted by i:
                    lo_i = max(0, i)
                    hi_i = min(size, i + n)
                    if hi_i > lo_i:
                        g[lo_i:hi_i] -= 2.0 * a[lo_i - i:hi_i - i]

                    # Second order corrections at k=2j and k=2i:
                    # At k=2j: extra delta^2 (from (a[j]+d)^2 - a[j]^2 = 2a[j]d + d^2, the d^2 part)
                    # At k=2i: extra delta^2 (from (a[i]-d)^2 - a[i]^2 = -2a[i]d + d^2, the d^2 part)
                    # Wait, the linear terms are already captured in g. The quadratic is just d^2 at those positions.
                    # Actually no: the linear term at k=2j is 2*a[0]*d from both sides = 4*a[0]*d? No...
                    # Let me just be careful. At k=2j:
                    # conv_new[2j] = conv[2j] + 2*a[j]*d (from linear, both sides) + d^2 (from (a[j]+d)^2)
                    # But 2*a[j]*d is already in g[2j]*d. And g[2j] = 2*a[2j-j] - 2*a[2j-i] (if valid)
                    # = 2*a[j] - 2*a[2j-i]. So the linear part gives (2*a[j] - 2*a[2j-i])*d.
                    # The actual change should be: 2*a[j]*d + d^2 (from the quadratic at 2j)
                    # So there's an extra d^2 at position 2j, and an extra d^2 at position 2i (from (a[i]-d)^2).
                    # Wait: (a[i]-d)^2 - a[i]^2 = -2a[i]d + d^2. The -2a[i]d is in the linear term.
                    # The d^2 is extra.
                    # Similarly at 2j: (a[j]+d)^2 - a[j]^2 = 2a[j]d + d^2. The 2a[j]d is in linear. Extra d^2.
                    # So total quadratic correction: d^2 at k=2j and d^2 at k=2i.
                    # (If 2j == 2i, i.e., i==j, skip since we skip i==j.)

                    # For the 1D optimization, I'll evaluate at several delta values and pick the best.
                    # To be efficient, precompute the current max and use the structure.

                    # Bisection / sampling for optimal delta in [0, ai]:
                    # f(d) = max_k (conv[k] + d*g[k] + quadratic_correction[d][k])
                    # where quadratic_correction is d^2 at k=2i and k=2j (if in range)

                    # Sample at a few key deltas
                    n_samples = 16
                    deltas = np.linspace(0, ai, n_samples + 1)[1:]
                    cur_max = conv.max()
                    for d in deltas:
                        new_conv_est = conv + d * g
                        # Add quadratic corrections
                        if 2 * j < size:
                            new_conv_est[2 * j] += d * d
                        if 2 * i < size:
                            new_conv_est[2 * i] += d * d
                        est_max = new_conv_est.max()
                        if est_max < best_new_max - 1e-13:
                            best_new_max = est_max
                            best_delta = d
                            best_i = i
                            best_j = j

            if best_i >= 0 and best_delta > 0:
                # Apply the move
                a[best_i] -= best_delta
                a[best_j] += best_delta
                s = a.sum()
                a = a * norm / s
                conv = fast_autocorr(a)
                cur_max = conv.max()
                if cur_max < best_score - 1e-13:
                    best_score = cur_max
                    best_a = a.copy()
                    round_improved = True
                else:
                    # Revert
                    a[best_i] += best_delta
                    a[best_j] -= best_delta
                    a = a * norm / a.sum()
                    conv = fast_autocorr(a)

        if not round_improved:
            no_improve_rounds += 1
        else:
            no_improve_rounds = 0

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    PTC round={no_improve_rounds} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def sens_init_median(a):
    """Just a placeholder - use median of a."""
    return np.median(a)


def lse_gradient_optimize(a0, time_limit, verbose=True):
    """LSE smooth gradient descent with FT-based gradient computation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    # Use annealed temperature
    n_stages = 6
    temps = np.geomspace(10.0, 0.3, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(20, min(80, int(remaining * 2)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_autocorr(a)
            max_c = conv.max()
            # LSE weights (softmax of conv/T)
            exp_c = np.exp((conv - max_c) / T)
            w = exp_c / exp_c.sum()

            # Compute gradient: grad_i = 2 * sum_k w_k * (a[k-i] + a[i-k]) (valid indices)
            # = 2 * (conv(w, a))[i] + 2 * (conv(w, a))[size-1-i] ... 
            # Actually: sum_k w_k * a[k-i] for valid k-i is the i-th element of full conv(w, a)
            # And sum_k w_k * a[i-k] is the (size-1-i)-th element... let me just use FFT.
            # grad_i = 2 * sum_k w[k] * a[k-i] (where 0<=k-i<n, 0<=k<size)
            #        + 2 * sum_k w[k] * a[i-k] (where 0<=i-k<n, 0<=k<size)
            # First term: this is cross-correlation of w and a at lag i
            # = (w * a_rev)[n-1+i] ... let me just compute via FFT
            # sum_k w[k]*a[k-i]: let m = k-i, k = m+i. sum_m w[m+i]*a[m] for valid m.
            # This is the inner product of w[shifted by i] with a.
            # In FFT: ifft(FFT(w) * conj(FFT(a))) gives cross-correlation.
            # cross[i] = sum_k w[k]*a[k+i] (with circular wrap)
            # We want sum_k w[k]*a[k-i] = cross[-i] = cross[fft_size - i] (for i > 0)
            
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            # cross_corr[i] = sum_k w[k]*a[k+i] (circular)
            cross1 = irfft(W * np.conj(A), fft_size)
            # We want sum_k w[k]*a[k-i] for i=0..n-1
            # = cross1[fft_size - i] for i>0, cross1[0] for i=0
            term1 = np.zeros(n)
            term1[0] = cross1[0]
            if n > 1:
                term1[1:] = cross1[fft_size - 1:fft_size - n:-1]

            # Second term: sum_k w[k]*a[i-k] = sum_k a[i-k]*w[k]
            # Let m = i-k: sum_m a[m]*w[i-m] = cross-correlation of a and w at lag i
            # = cross_corr_aw[i] where cross_corr_aw[i] = sum_m a[m]*w[m+i]
            cross2 = irfft(A * np.conj(W), fft_size)
            # cross2[i] = sum_m a[m]*w[m+i]
            # We want sum_k a[i-k]*w[k]. Let m=i-k: sum_m a[m]*w[i-m] = sum_m a[m]*w[-(m-i)]
            # = cross2[fft_size - i] for i>0... 
            # Actually: cross2[m] = sum_k a[k]*w[k+m]. We want sum_k a[k]*w[i-k] = sum_k a[k]*w[-(k-i)]
            # = cross2 evaluated at m = -(k-i)... this is getting circular (pun intended).
            # Let me just use: sum_k w[k]*a[i-k] = (w * a_rev)[i] where a_rev = a[::-1]
            # Actually the simplest: it's the same as term1 but with a reversed.
            # sum_k w[k]*a[i-k] = sum_k w[k]*a_rev[n-1-(i-k)] = sum_k w[k]*a_rev[n-1-i+k]
            # = cross-correlation of w and a_rev at lag (n-1-i)
            
            A_rev = rfft(a[::-1], fft_size)
            cross3 = irfft(W * np.conj(A_rev), fft_size)
            term2 = np.zeros(n)
            # term2[i] = sum_k w[k]*a[i-k]
            # = sum_k w[k]*a_rev[n-1-i+k]
            # Let this be the cross-correlation of w and a_rev at lag (n-1-i)
            # cross3[l] = sum_k w[k]*a_rev[k+l]
            # We want sum_k w[k]*a_rev[k+(n-1-i)] = cross3[n-1-i]
            for i in range(n):
                term2[i] = cross3[(n - 1 - i) % fft_size]

            grad = 2.0 * (term1 + term2)

            # Normalized gradient step
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            step = norm / (grad_norm * (1.0 + 0.1 * it))
            # Cap the step
            step = min(step, 0.5 * norm / n)

            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s
            a = a_new

            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE T={T:.2f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    """Generate structured initial sequences."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        x = np.linspace(0.0, 1.0, n)
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        # Fourier-based: low-pass filtered random
        freqs = np.arange(n)
        mask = freqs < n * 0.1
        spectrum = np.zeros(n)
        spectrum[mask] = rng.standard_normal(mask.sum())
        a = irfft(np.abs(spectrum) + 1.0, n)
    else:
        # Broad peak
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.5) ** 2) + 0.3 * np.sin(np.pi * x)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def single_coord_optimize(a, time_limit, rng, verbose=False):
    """Optimize one coordinate at a time (cyclic coordinate descent)."""
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1

    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 3:
        n_passes += 1
        pass_improved = False
        no_improve += 1

        # Cyclic order (or random)
        order = np.arange(n)
        if n_passes > 1:
            rng.shuffle(order)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = order[idx]

            # Optimize a[i] given all others fixed
            # conv[k] as function of a[i]:
            # conv[k] = sum_j a[j]*a[k-j]
            # Terms involving a[i]:
            #   j=i: a[i]*a[k-i] (linear in a[i])
            #   k-j=i, j=k-i: a[k-i]*a[i] (linear in a[i])
            #   j=i and k-j=i (k=2i): a[i]^2 (quadratic)
            # So conv[k] = c_k + a[i]*d_k + (a[i]^2 if k=2i else 0)
            # where c_k is the part not involving a[i], d_k = 2*a[k-i] (if valid) + (a[i] if k=2i)
            # Wait: if k=2i, the term is a[i]^2, derivative is 2*a[i].
            # If k!=2i, the terms are 2*a[i]*a[k-i], derivative is 2*a[k-i].
            # So d_k = 2*a[k-i] if 0<=k-i<n (this covers both cases since at k=2i, a[k-i]=a[i])
            # And the quadratic term is a[i]^2 at k=2i.
            # conv[k] = base_k + a[i]*2*a[k-i] + (a[i]^2 if k==2i else 0)
            # where base_k = conv[k] - a[i]*2*a[k-i] - (a[i]^2 if k==2i else 0)
            #         = the part of conv[k] that doesn't involve a[i]

            ai = a[i]
            if ai < 1e-15:
                # a[i] = 0, try making it positive
                continue

            # Compute base and linear coefficient
            # base_k = conv[k] - contribution of a[i]
            # Contribution of a[i] to conv[k]:
            #   = a[i]*a[k-i] (from j=i) + a[k-i]*a[i] (from k-j=i) - a[i]^2 (if k=2i, subtract the double count)
            #   = 2*a[i]*a[k-i] - a[i]^2 (if k=2i)
            #   = 2*a[i]*a[k-i] (if k!=2i)
            # So: base_k = conv[k] - 2*a[i]*a[k-i] + (a[i]^2 if k==2i else 0)
            # And linear coeff = 2*a[k-i]
            # And quadratic coeff = 1 if k==2i else 0

            # For the max over k of (base_k + 2*a[i]*a[k-i] + (a[i]^2 if k==2i)):
            # This is a 1D convex optimization (max of convex functions of a[i]).
            # The minimum of the max is where the top constraints cross.

            # Sample the objective at several values of a[i] and pick the best
            # Since the domain is [0, infinity) but we want sum(a) = norm,
            # increasing a[i] means we must decrease others proportionally.
