def lse_grad(a, T):
    """
    Compute gradient of (1/T)*log(sum exp(T*(a*a)_k)) w.r.t. a.
    Returns gradient and the weighting vector w.
    """
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    
    # grad_j = 2 * sum_k w_k * a_{k-j} for valid indices
    # This is 2 * (w * a_reversed) convolved, extracted at the right offset
    conv_len = n + (2 * n - 1) - 1  # = 3n - 2
    fft_size = 1 << (conv_len - 1).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad, w, exp_Tb.max()

def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def spectral_flattening(a0, n_iter=200, alpha=0.25):
    """Iteratively flatten the power spectrum."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()
    
    best_a = a.copy()
    best_score = internal_score(a)
    
    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 50.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    
    return best_a

def fista_subgradient(a0, time_limit):
    """
    FISTA acceleration for the subgradient method on the max of convolution.
    Uses the exact subgradient (pick argmax) with FISTA step sizes.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = internal_score(a)
    y = a.copy()
    t_fista = 1.0
    t0 = time.monotonic()
    step = 1.0 / n  # initial step
    
    for t in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(y)
        k_max = np.argmax(b)
        
        # Subgradient at y: grad_j = 2 * y_{k_max - j} for valid indices
        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        grad = np.zeros(n)
        for j in range(js, je + 1):
            grad[j] = 2.0 * y[k_max - j]
        
        # Gradient step
        a_new = y - step * grad
        a_new = project_simplex(a_new)
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_fista ** 2)) / 2.0
        y = a_new + ((t_fista - 1.0) / t_new) * (a_new - a)
        y = project_simplex(y)
        t_fista = t_new
        
        # Decay step
        step = 1.0 / (np.sqrt(t_fista) * np.sqrt(n))
        
        a = a_new
        score = internal_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Also check y
        score_y = internal_score(y)
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
    
    return best_a, best_score

def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0 : sum(x) = target_sum}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if n == 0:
        return v
    if np.all(v >= 0) and abs(v.sum() - target_sum) < 1e-14:
        return v
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    idx = np.nonzero(v_sorted * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(idx) == 0:
        return np.full(n, target_sum / n)
    rho = idx[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0.0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-14:
        idx_max = np.argmax(result)
        result[idx_max] += diff
        result = np.maximum(result, 0.0)
        total = result.sum()
        if total > 0:
            result = result / total * target_sum
    return result

    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    try:
        prob.solve(solver=solver)
    except Exception:
        pass

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.SCS, cp.CLARABEL, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated LP best-response with time-averaged tracking."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0
    stall_count = 0
    stall_threshold = 4000

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0.0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = project_simplex(a_new)

        y = y_new
        a = a_new

        score_a = internal_score(a)
        if score_a < best_score:
            if score_a < best_score - 1e-10:
                stall_count = 0
            best_score = score_a
            best_a = a.copy()

        score_y = internal_score(y)
        if score_y < best_score:
            if score_y < best_score - 1e-10:
                stall_count = 0
            best_score = score_y
            best_a = y.copy()

        avg_a += a
        n_avg += 1
        if n_avg >= 50 and n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            score_avg = internal_score(avg_check)
            if score_avg < best_score - 1e-12:
                best_score = score_avg
                best_a = avg_check.copy()
                stall_count = 0

        stall_count += 1
        if stall_count > stall_threshold:
            break

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NL t={t} ({elapsed:.0f}s): best={best_score:.8f}, LP_M={M_val:.10f}")

    return best_a, best_score


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def lse_objective_grad_z(z, T):
    """
    Compute LSE_T objective and gradient w.r.t. z (where a = softmax(z)).
    LSE_T = (1/T) * log(sum_k exp(T * (a*a)_k))
    """
    a = softmax_fn(z)
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    lse_val = (max_Tb + np.log(exp_Tb.sum())) / T

    w = exp_Tb / exp_Tb.sum()

    # grad of LSE w.r.t. a: grad_a_j = 2 * sum_k w_k * a_{k-j}
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    grad_a = 2.0 * conv[n - 1: n - 1 + n]

    # Chain rule: grad_z_j = a_j * (grad_a_j - sum_i a_i * grad_a_i)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    return lse_val, grad_z, a


def adam_lse_optimize(a0, time_limit, verbose=True):
    """
    Multi-stage LogSumExp Adam optimization.
    Uses adaptive temperature and learning rate scheduling.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s

    z = np.log(np.maximum(a_init, 1e-30))
    z -= z.mean()

    t0 = time.monotonic()
    best_a = a_init.copy()
    best_score = internal_score(a_init)

    # Multi-stage with decreasing temperature
    stages = [
        (50.0, 10.0, 20, 0.005),
        (10.0, 3.0, 25, 0.003),
        (3.0, 1.0, 25, 0.002),
        (1.0, 0.3, 25, 0.001),
        (0.3, 0.05, 30, 0.0005),
    ]

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_adam = 0

    for si, (T_start, T_end, n_inner, lr) in enumerate(stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        temps = np.geomspace(T_start, T_end, n_inner)
        for it in range(3000):
            if time.monotonic() - t0 > time_limit:
                break
            T_cur = temps[min(it, n_inner - 1)] if it >= n_inner else temps[it % n_inner]
            # Actually use a smooth schedule
            T_cur = T_start * (T_end / T_start) ** (it / max(1, n_inner * 2))

            lse_val, grad_z, a_cur = lse_objective_grad_z(z, T_cur)

            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)

            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            if it % 30 == 0:
                a_check = softmax_fn(z)
                score = internal_score(a_check)
                if score < best_score:
                    best_score = score
                    best_a = a_check.copy()

        a_check = softmax_fn(z)
        score = internal_score(a_check)
        if score < best_score:
            best_score = score
            best_a = a_check.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"      Adam-LSE stage {si} done (T: {T_start}->{T_end}, {elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def fista_smooth_optimize(a0, time_limit, verbose=True):
    """
    FISTA-accelerated projected gradient on LSE-smoothed objective.
    Works directly in a-space (simplex constraint).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = internal_score(a)
    y = a.copy()
    t_fista = 1.0
    t0 = time.monotonic()

    n_total_steps = max(1, int(time_limit * 40))

    for t in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        frac = (t + 1) / n_total_steps
        T_cur = 20.0 * (0.05 / 20.0) ** frac
        step = 1.0 / (np.sqrt(t_fista) * 2.0 * np.sqrt(n))

        yy = np.maximum(y, 0.0)
        ss = yy.sum()
        if ss > 0:
            yy = yy / ss

        b = fast_autocorr(yy)
        Tb = T_cur * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()

        conv_len = n + (2 * n - 1) - 1
        fft_size = 1 << (conv_len - 1).bit_length()
        A_rev = rfft(yy[::-1].copy(), fft_size)
        W = rfft(w, fft_size)
        conv = irfft(A_rev * W, fft_size)
        grad = 2.0 * conv[n - 1: n - 1 + n]

        a_new = yy - step * grad
        a_new = project_simplex(a_new)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_fista ** 2)) / 2.0
        y = a_new + ((t_fista - 1.0) / t_new) * (a_new - a)
        y = project_simplex(y)
        t_fista = t_new
        a = a_new

        score = internal_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        score_y = internal_score(y)
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      FISTA t={t} (T={T_cur:.3f}, {elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def targeted_peak_reduction(a0, time_limit, seed=42):
    """
    Quadratic-line-search peak reduction.
    Identify top peaks, find responsible elements, try targeted moves
    with optimal step size from quadratic approximation.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        peak_val = b.max()
        threshold = peak_val * 0.92
        peak_indices = np.where(b >= threshold)[0]

        improved = False

        for k in peak_indices[:8]:
            if time.monotonic() - t0 > time_limit:
                break

            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue

            # Top contributing pairs
            top_pairs = np.argsort(contrib)[::-1][:5]
            for pi in top_pairs:
                src = idx[pi]
                partner = k - src

                # Find safe destinations (not in any top peak's support)
                dangerous = set()
                for k2 in peak_indices[:5]:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    dangerous.update(range(js2, je2 + 1))

                safe = [j for j in range(n) if j not in dangerous and j != src]
                if not safe:
                    safe = [j for j in range(n) if j != src]

                safe_sorted = sorted(safe, key=lambda j: a[j])
                dst_pool = safe_sorted[:max(2, len(safe_sorted) // 5)]

                for dst in dst_pool[:3]:
                    # Try quadratic-optimal step: minimize (a_src - d)(a_dst + d) contribution
                    # The change in (a*a)_k from moving d from src to dst:
                    # delta_k = -d*(a[partner] + a[k-dst]) + d^2*(if src==dst... not possible)
                    # More precisely:
                    # Old contribution at k from src: a[src]*a[partner]
                    # New: (a[src]-d)*a[partner]
                    # New contribution at k from dst: d*a[k-dst] (if k-dst valid and != dst)
                    # Total change at k: -d*a[partner] + d*a[k-dst] = d*(a[k-dst] - a[partner])
                    # For other lags, similar analysis
                    # Simple approach: try a few fractions and pick best
                    for frac in [0.0005, 0.001, 0.002, 0.005, 0.01, 0.02, 0.03]:
                        d = a[src] * frac
                        if d < 1e-16:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= d
                        a_trial[dst] += d
                        if a_trial[src] < -1e-15:
                            continue
                        a_trial[src] = max(0.0, a_trial[src])
                        a_trial[:] /= a_trial.sum()
                        score = internal_score(a_trial)
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random exploration
            for _ in range(3):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-12:
                    continue
                d = a[i] * rng.uniform(0.001, 0.008)
                a_trial = a.copy()
                a_trial[i] -= d
                a_trial[j] += d
                if a_trial[i] >= 0:
                    a_trial[:] /= a_trial.sum()
                    score = internal_score(a_trial)
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                        improved = True
                        break
            if stall > 800:
                break

    return best_a, best_score


def spectral_shaping(a0, n_iter=200, alpha=0.25):
    """Iteratively flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = internal_score(a)

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        scale = np.ones(len(A))
        ratio = np.clip(target / (B[1:] + 1e-30), 0.01, 50.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization.
    Phase 1: Parent validation + spectral pre-shaping
    Phase 2: Nesterov LP (main optimizer, ~45% time)
    Phase 3: FISTA smooth refinement (~15% time)
    Phase 4: Adam LSE polish (~15% time)
    Phase 5: Targeted peak reduction (~15% time)
    Phase 6: Second LP round + final peak shave (~remaining time)
    """
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    # --- Phase 0: Initialize ---
    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = internal_score(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.8f}")
    except Exception as e:
        print(f"Parent load failed: {e}")

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq /= best_seq.sum()
        best_score = internal_score(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n = len(best_seq)

    # --- Phase 1: Quick spectral pre-shaping ---
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: Spectral pre-shaping (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.03, 30.0)
    if t1 > 3:
        for alpha_val in [0.3, 0.15]:
            a_spec = spectral_shaping(best_seq, n_iter=150, alpha=alpha_val)
            sc = internal_score(a_spec)
            if sc < best_score:
                best_score = sc
                best_seq = a_spec.copy()
                print(f"  Spectral alpha={alpha_val}: {best_score:.8f}")
    print(f"  After Phase 1: best={best_score:.8f}")

    # --- Phase 2: Main Nesterov LP ---
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: Nesterov LP (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.40, 600.0)
    if t2 > 30:
        a_out, sc_out = nesterov_lp_optimize(best_seq, t2, solver, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0.0)
            best_seq /= best_seq.sum()
            print(f"  *** NEW BEST [LP]: {best_score:.8f}")
        # Also try the internal score (might differ slightly due to normalization)
        sc_int = internal_score(best_seq)
        if sc_int < best_score:
            best_score = sc_int
    print(f"  After Phase 2: best={best_score:.8f} (t={time.monotonic()-t_global_start:.0f}s)")

    # --- Phase 3: FISTA smooth refinement ---
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: FISTA smooth (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 10:
        a_out, sc_out = fista_smooth_optimize(best_seq, t3, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0.0)
            best_seq /= best_seq.sum()
            print(f"  *** NEW BEST [FISTA]: {best_score:.8f}")
        sc_int = internal_score(best_seq)
        if sc_int < best_score:
            best_score = sc_int
    print(f"  After Phase 3: best={best_score:.8f} (t={time.monotonic()-t_global_start:.0f}s)")

    # --- Phase 4: Adam LSE polish ---
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: Adam LSE (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 10:
        a_out, sc_out = adam_lse_optimize(best_seq, t4, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0.0)
            best_seq /= best_seq.sum()
            print(f"  *** NEW BEST [LSE]: {best_score:.8f}")
        sc_int = internal_score(best_seq)
        if sc_int < best_score:
            best_score = sc_int
    print(f"  After Phase 4: best={best_score:.8f} (t={time.monotonic()-t_global_start:.0f}s)")

    # --- Phase 5: Targeted peak reduction ---
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: Peak reduction (remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.18, 250.0)
    if t5 > 5:
        a_out, sc_out = targeted_peak_reduction(best_seq, t5, seed=123)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0.0)
            best_seq /= best_seq.sum()
            print(f"  *** NEW BEST [Peak1]: {best_score:.8f}")
        sc_int = internal_score(best_seq)
        if sc_int < best_score:
            best_score = sc_int
    print(f"  After Phase 5: best={best_score:.8f} (t={time.monotonic()-t_global_start:.0f}s)")

    # --- Phase 6: Second LP round ---
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: Second LP round (remaining={remaining:.0f}s) ===")
    t6 = min(remaining * 0.40, 250.0)
    if t6 > 20:
        a_out, sc_out = nesterov_lp_optimize(best_seq, t6, solver, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0.0)
            best_seq /= best_seq.sum()
            print(f"  *** NEW BEST [LP2]: {best_score:.8f}")
        sc_int = internal_score(best_seq)
        if sc_int < best_score:
            best_score = sc_int
    print(f"  After Phase 6: best={best_score:.8f} (t={time.monotonic()-t_global_start:.0f}s)")

    # --- Phase 7: Final peak shave ---
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 7: Final peak shave (remaining={remaining:.0f}s) ===")
    t7 = min(remaining * 0.50, 100.0)
    if t7 > 5:
        a_out, sc_out = targeted_peak_reduction(best_seq, t7, seed=789)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0.0)
            best_seq /= best_seq.sum()
            print(f"  *** NEW BEST [Peak2]: {best_score:.8f}")
        sc_int = internal_score(best_seq)
        if sc_int < best_score:
            best_score = sc_int

    # --- Final verification ---
    if best_seq is not None:
        final_eval = evaluate_safe(best_seq)
        if np.isfinite(final_eval) and final_eval < best_score:
            best_score = final_eval
        best_seq = np.maximum(best_seq, 0.0)
        s = best_seq.sum()
        if s > 0:
            best_seq = best_seq / s
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
