def fast_autocorr(a):

    return irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    """Project onto {x: sum(x)=target_sum, x>=0}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= -1e-15) and abs(v.sum() - target_sum) < 1e-10:
        return np.maximum(v, 0)
    v = np.maximum(v, 0)
    if v.sum() < 1e-30:
        return np.full(n, target_sum / n)
    # Binary search on threshold
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    rho = np.nonzero(v_sorted * np.arange(1, n + 1) > (css - target_sum))[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho = rho[-1]
    theta = (css[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-12:
        idx = np.argmax(result)
        result[idx] += diff
        result = np.maximum(result, 0)
        result = result / result.sum() * target_sum
    return result

def fista_lse(a0, time_limit, T_schedule=None):
    """
    FISTA with LSE-smoothed objective.
    Minimize f_T(a) = (1/T)*log(sum exp(T*c[k])) subject to a in simplex.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    
    # FISTA parameters
    y = a.copy()
    t_k = 1.0
    
    if T_schedule is None:
        T_schedule = np.geomspace(50.0, 0.5, 30)
    
    total_iters = 0
    for T in T_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        # Adaptive learning rate based on T
        # The LSE objective has smoothness ~ T * ||a||_inf (roughly)
        # Use lr = 1/sqrt(T) as a starting point
        lr = 1.0 / np.sqrt(T * 2.0)
        
        # Reset FISTA state for new T
        y = a.copy()
        t_k = 1.0
        m = np.zeros(n)  # momentum accumulator
        
        stage_iters = max(50, int(min(300, remaining * 100 / max(1, len(T_schedule)))))
        
        for it in range(stage_iters):
            if time.monotonic() - t0 > time_limit:
                break
            
            grad, obj_val, w, c = lse_grad_and_obj(y, T)
            
            # Gradient step
            a_new = y - lr * grad
            a_new = project_simplex(a_new)
            
            # FISTA update
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            y_new = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
            y = project_simplex(np.maximum(y_new, 0))
            t_k = t_new
            a = a_new
            
            # Check score
            score = internal_score(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
            
            total_iters += 1
        
        if total_iters % 500 < 50:
            elapsed = time.monotonic() - t0
            print(f"      FISTA-LSE T={T:.2f} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score

def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where (T @ b)[k] = sum_j a_fixed[max(0,k-n+1):min(n,k+1)-j] * b[j]... 
    Actually T[k,j] = a_fixed[k-j] if 0<=k-j<n else 0"""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = max(0, 0 - j)  # k-j >= 0 => k >= j
        hi = min(2*n-2, n-1 + j)  # k-j < n => k < j+n
        # T[k,j] = a_fixed[k-j] for k in [j, j+n-1]
        T[j:j+n, j] = a_fixed
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Solve: min M s.t. (a_fixed*b)_k <= M for all k, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, max_iter=500)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.PDLP, cp.CLARABEL, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, max_iter=500)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp(a0, time_limit, solver):

    n = len(a0)

        if b_star is None:

        y = np.maximum(y, 0)
        s = y.sum()
        if s < 1e-15:

        else:
            y = y / s

            a = b_star.copy()

            a = project_simplex(a_new)

        if score_a < best_score:
            if score_a < best_score - 1e-8:

            best_score = score_a
            best_a = a.copy()
        
        score_y = internal_score(y)
        if score_y < best_score:
            if score_y < best_score - 1e-8:

def spectral_shaping(a, n_iter=300, alpha=0.3):
    """Flatten power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()
    
    best_a = a.copy()
    best_score = internal_score(a)
    
    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 50.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def peak_shaving_analytical(a0, time_limit, seed=42):
    """
    Peak shaving using the analytical structure.
    Key insight: to reduce peak at lag k, reduce a[i]*a[k-i] for the contributing pairs.
    We do this by moving mass from high-contributing positions to low-contributing ones.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    stall = 0
    max_stall = 800
    
    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a)
        # Find top peaks
        top_k = np.argsort(c)[::-1][:15]
        peak_val = c[top_k[0]]
        
        improved = False
        for k in top_k[:8]:
            # Find contributing indices
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            
            if contrib.max() < 1e-15:
                continue
            
            # Source: index with highest contribution
            src_local = np.argmax(contrib)
            src = idx[src_local]
            partner = k - src
            
            # Destination candidates: indices NOT contributing to any top peak
            peak_areas = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                for j2 in range(js2, je2 + 1):
                    peak_areas.add(j2)
                    peak_areas.add(k2 - j2)
            
            # Also add the partner to peak areas (it contributes to this peak)
            peak_areas.add(partner)
            
            candidates = [j for j in range(n) if j not in peak_areas and a[j] < a[src] * 0.5]
            if not candidates:
                candidates = [j for j in range(n) if j not in (src, partner)]
            
            if not candidates:
                continue
            
            # Try to move a small amount from src to a random candidate
            # The gain is roughly: reducing c[k] by 2*amt*a[partner] while 
            # possibly increasing other c[k'] values
            for frac in [0.01, 0.005, 0.02, 0.003, 0.03]:
                amt = a[src] * frac
                if amt < 1e-14:
                    continue
                
                # Try a few destinations
                n_tries = min(5, len(candidates))
                for _ in range(n_tries):
                    dst = candidates[rng.integers(len(candidates))]
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    score = internal_score(a_trial)
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = internal_score(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                    elif score < best_score + 0.001:
                        # Accept with some probability (simulated annealing style)
                        if rng.random() < 0.1:
                            a = a_trial
            if stall > max_stall:
                break
    
    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral shaping initialization
2. FISTA with LSE-smoothed objective (primary optimizer with multi-stage temperature)
3. Mirror descent (exponential gradient) as complementary approach
4. LP-based Nesterov method for polishing
5. Analytical peak shaving using the structure: dc[k]/da[i] = 2*a[k-i]
Key insight: the gradient of each autocorrelation lag is a shifted copy of a itself.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def internal_score(a):
    """Compute 2*n*max_b assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()


def project_simplex(v, target_sum=1.0):
    """Project onto {x: sum(x)=target_sum, x>=0}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= -1e-15) and abs(v.sum() - target_sum) < 1e-12:
        return np.maximum(v, 0)
    v = np.maximum(v, 0)
    if v.sum() < 1e-30:
        return np.full(n, target_sum / n)
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    rho_idx = np.nonzero(v_sorted * np.arange(1, n + 1) > (css - target_sum))[0]
    if len(rho_idx) == 0:
        return np.full(n, target_sum / n)
    rho = rho_idx[-1]
    theta = (css[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-12:
        idx = np.argmax(result)
        result[idx] += diff
        result = np.maximum(result, 0)
        ss = result.sum()
        if ss > 1e-30:
            result = result / ss * target_sum
    return result


def lse_gradient_and_obj(a, T):
    """
    Compute gradient and objective of LSE-smoothed max autocorrelation.
    f_T(a) = (1/T)*log(sum_k exp(T*c[k]))
    grad f_T[i] = (2/T) * sum_k w[k] * a[k-i] (valid indices)
    where w = softmax(T*c)
    """
    n = len(a)
    c = fast_autocorr(a)
    Tc = T * c
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    w = exp_Tc / exp_Tc.sum()
    obj_val = (max_Tc + np.log(exp_Tc.sum())) / T

    # h[i] = sum_m a[m] * w[m+i] using FFT cross-correlation
    N = 1 << (3 * n).bit_length()
    A = rfft(a, N)
    W = rfft(w, N)
    h = irfft(np.conj(A) * W, N)[:n]
    grad = (2.0 / T) * h
    return grad, obj_val, w, c


def fista_lse(a0, time_limit, T_list=None, verbose=True):
    """
    FISTA with LSE-smoothed objective and simplex projection.
    Multi-stage: high T (smooth, fast convergence) -> low T (sharp, accurate).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    if T_list is None:
        T_list = np.geomspace(100.0, 0.1, 30)

    total_iters = 0
    for stage_idx, T in enumerate(T_list):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        # Adaptive iteration count: more iterations at high T, fewer at low T
        frac = (stage_idx + 1.0) / len(T_list)
        stage_time = remaining * (0.6 * (1 - frac) + 0.4 * frac)
        stage_iters = max(20, int(stage_time * 80))

        # Step size: for LSE with temperature T, smoothness ~ O(T)
        # Use lr = c / sqrt(T) with c tuned empirically
        lr = 2.0 / (T + 1.0)

        # FISTA state
        y = a.copy()
        t_k = 1.0

        for it in range(stage_iters):
            if time.monotonic() - t0 > time_limit:
                break

            grad, obj_val, w, c = lse_gradient_and_obj(y, T)

            # Gradient step + projection
            a_new = y - lr * grad
            a_new = project_simplex(a_new)

            # FISTA momentum
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            y_new = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
            y = np.maximum(y_new, 0)
            ss = y.sum()
            if ss < 1e-15:
                y = a_new.copy()
            else:
                y = y / ss
            t_k = t_new
            a = a_new

            # Track best
            score = internal_score(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()

            total_iters += 1

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"      FISTA-LSE T={T:.2f} ({elapsed:.0f}s, iter={total_iters}): best={best_score:.6f}")

    return best_a, best_score


def mirror_descent_smooth(a0, time_limit, verbose=True):
    """
    Mirror descent (exponential gradient) with LSE-smoothed objective.
    Update: a_new[i] proportional to a[i] * exp(-lr * grad[i])
    Naturally maintains non-negativity.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    T = 10.0
    lr = 0.5
    t = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break

        # Anneal temperature
        T = max(0.5, T * 0.9995)
        lr_t = lr / (1 + 0.02 * t)

        grad, obj_val, w, c = lse_gradient_and_obj(a, T)

        # Exponential update
        log_a = np.log(np.maximum(a, 1e-30)) - lr_t * grad
        log_a = log_a - log_a.max()
        a_new = np.exp(log_a)
        ss = a_new.sum()
        if ss < 1e-30:
            a_new = a.copy()
        else:
            a_new = a_new / ss

        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            t = max(0, t - 10)  # reset patience-like counter
        a = a_new

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Mirror t={t} T={T:.2f} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix: (T @ b)[k] = sum_j T[k,j]*b[j] = (a*b)[k]."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Solve: min M s.t. (a*b)[k] <= M for all k, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, max_iter=1000)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.PDLP, cp.CLARABEL, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, max_iter=1000)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated LP best-response (zero-sum game)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0
    stall = 0

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue

        alpha_t = 2.0 / (t + 2.0)
        y = (1 - alpha_t) * y + alpha_t * b_star
        y = np.maximum(y, 0)
        s = y.sum()
        if s < 1e-15:
            y = best_a.copy()
        else:
            y = y / s

        if t == 1:
            a = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y + momentum * (y - a)
            a = project_simplex(a_new)

        score_a = internal_score(a)
        if score_a < best_score:
            if score_a < best_score - 1e-8:
                stall = 0
            best_score = score_a
            best_a = a.copy()

        score_y = internal_score(y)
        if score_y < best_score:
            if score_y < best_score - 1e-8:
                stall = 0
            best_score = score_y
            best_a = y.copy()

        avg_a += a
        n_avg += 1
        if n_avg >= 100 and n_avg % 200 == 0:
            avg_check = avg_a / n_avg
            score_avg = internal_score(avg_check)
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
                stall = 0

        stall += 1
        if stall > 6000:
            break

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}")

    return best_a, best_score


def spectral_shaping(a, n_iter=300, alpha=0.3):
    """Flatten power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = internal_score(a)

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 50.0)
        scale[1:] = np.power(ratio, alpha / 2.0)

        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def peak_shaving_analytical(a0, time_limit, seed=42):
    """
    Peak shaving using analytical structure: dc[k]/da[i] = 2*a[k-i].
    Move mass from high-contributing positions to low-contributing ones.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    stall = 0
    max_stall = 600

    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a)
        top_k = np.argsort(c)[::-1][:20]

        improved = False
        for k in top_k[:10]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            if contrib.max() < 1e-15:
                continue

            src_local = np.argmax(contrib)
            src = int(idx[src_local])
            partner = k - src

            # Build set of "hot" indices (contributing to any top peak)
            peak_areas = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                for j2 in range(js2, je2 + 1):
                    peak_areas.add(j2)
                    peak_areas.add(k2 - j2)

            candidates = [j for j in range(n) if j not in peak_areas]
            if not candidates:
                candidates = list(range(n))

            # Try moving from src to a cool destination
            n_tries = min(8, len(candidates))
            for _ in range(n_tries):
                dst = int(candidates[rng.integers(len(candidates))])
                for frac in [0.01, 0.005, 0.02, 0.003, 0.015, 0.008]:
                    amt = a[src] * frac
                    if amt < 1e-14:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = internal_score(a_trial)
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                    elif score < best_score + 0.0005 and rng.random() < 0.05:
                        a = a_trial
            if improved:
                break

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.01)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    ss = a_trial.sum()
                    if ss > 1e-15:
                        a_trial = a_trial / ss
                        score = internal_score(a_trial)
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
                        elif score < best_score + 0.001 and rng.random() < 0.08:
                            a = a_trial
            if stall > max_stall:
                break

    return best_a, best_score

        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = internal_score(best_seq)
        print(f"Default: n={len(best_seq)}, score={best_score:.6f}")

    n = len(best_seq)

    # Phase 1: Spectral shaping (quick, multiple alphas)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Spectral ({remaining:.0f}s remaining) ===")
    t1 = min(remaining * 0.03, 30.0)
    if t1 > 5:
        for alpha_val in [0.4, 0.25, 0.15]:
            if time.monotonic() - t_global > time_budget:
                break
            a_out = spectral_shaping(best_seq, n_iter=200, alpha=alpha_val)
            sc = internal_score(a_out)
            if sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  Spectral a={alpha_val}: {best_score:.6f}")
    print(f"  After P1: {best_score:.6f} ({time.monotonic()-t_global:.0f}s)")

    # Phase 2: FISTA-LSE (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: FISTA-LSE ({remaining:.0f}s remaining) ===")
    t2 = min(remaining * 0.30, 500.0)
    if t2 > 30:
        T_sched = np.geomspace(80.0, 0.05, 40)
        a_out, sc_out = fista_lse(best_seq, t2, T_list=T_sched)
        sc = internal_score(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** FISTA: {best_score:.6f}")
    print(f"  After P2: {best_score:.6f} ({time.monotonic()-t_global:.0f}s)")

    # Phase 3: Mirror descent (complementary direction)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Mirror Descent ({remaining:.0f}s remaining) ===")
    t3 = min(remaining * 0.10, 120.0)
    if t3 > 15:
        a_out, sc_out = mirror_descent_smooth(best_seq, t3)
        sc = internal_score(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** Mirror: {best_score:.6f}")
    print(f"  After P3: {best_score:.6f} ({time.monotonic()-t_global:.0f}s)")

    # Phase 4: LP-based Nesterov (strong but slow)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LP Nesterov ({remaining:.0f}s remaining) ===")
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 30:
        a_out, sc_out = nesterov_lp(best_seq, t4, solver)
        sc = internal_score(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** LP: {best_score:.6f}")
    print(f"  After P4: {best_score:.6f} ({time.monotonic()-t_global:.0f}s)")

    # Phase 5: Multi-start refinement
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Multi-start ({remaining:.0f}s remaining) ===")
    t5 = min(remaining * 0.12, 150.0)
    if t5 > 20:
        a_out, sc_out = multi_start_refine(best_seq, t5, n_starts=4)
        sc = internal_score(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** Multi-start: {best_score:.6f}")
    print(f"  After P5: {best_score:.6f} ({time.monotonic()-t_global:.0f}s)")

    # Phase 6: FISTA-LSE round 2 (lower temperatures)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: FISTA-LSE fine ({remaining:.0f}s remaining) ===")
    t6 = min(remaining * 0.12, 120.0)
    if t6 > 10:
        T_sched2 = np.geomspace(5.0, 0.02, 20)
        a_out, sc_out = fista_lse(best_seq, t6, T_list=T_sched2)
        sc = internal_score(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** FISTA-fine: {best_score:.6f}")
    print(f"  After P6: {best_score:.6f} ({time.monotonic()-t_global:.0f}s)")

    # Phase 7: Peak shaving
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Peak Shaving ({remaining:.0f}s remaining) ===")
    t7 = min(remaining * 0.15, 120.0)
    if t7 > 10:
        a_out, sc_out = peak_shaving_analytical(best_seq, t7, seed=777)
        sc = internal_score(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** PeakShave: {best_score:.6f}")
    print(f"  After P7: {best_score:.6f} ({time.monotonic()-t_global:.0f}s)")

    # Phase 8: Final LP polish if time allows
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Final LP ({remaining:.0f}s remaining) ===")
    t8 = min(remaining * 0.30, 150.0)
    if t8 > 30:
        a_out, sc_out = nesterov_lp(best_seq, t8, solver, verbose=False)
        sc = internal_score(a_out)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** LP-final: {best_score:.6f}")
    print(f"  After P8: {best_score:.6f} ({time.monotonic()-t_global:.0f}s)")

    # Phase 9: Final peak shave
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t9 = min(remaining * 0.40, 60.0)
        if t9 > 5:
            a_out, sc_out = peak_shaving_analytical(best_seq, t9, seed=31415)
            sc = internal_score(a_out)
            if sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  *** PeakShave-final: {best_score:.6f}")

    # Final: normalize and verify
    if best_seq is not None:
        best_seq = np.maximum(best_seq, 0)
        s = best_seq.sum()
        if s > 1e-15:
            best_seq = best_seq / s
        # Verify with actual evaluator
        try:
            eval_score = evaluate_sequence([float(x) for x in best_seq])
            if np.isfinite(eval_score):
                best_score = float(eval_score)
        except Exception:
            pass
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}")
    else:
        print("ERROR: no valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
