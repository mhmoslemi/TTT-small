"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

2. LP best-response with Nesterov acceleration and aggressive constraint screening

5. Final polish with combined CD + LSE

Key improvements over previous:

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print(f"  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def full_convolution(u, v):
    """Full linear convolution of u and v via FFT."""
    n1, n2 = len(u), len(v)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    U = rfft(u, fft_size)
    V = rfft(v, fft_size)
    return irfft(U * V, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def build_conv_matrix(a, k_indices=None):
    """Build convolution matrix T where (T @ b)[k] = sum_j a[j]*b[k-j]."""
    n = len(a)
    size = 2 * n - 1
    if k_indices is None:
        k_indices = np.arange(size)
    T = np.zeros((len(k_indices), n))
    for idx, k in enumerate(k_indices):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            T[idx, j] = a[k - j]
    return T


def lp_solve_best_response(a, M, solver, active_set=None, tol_frac=0.05):
    """Solve LP: max sum(b) s.t. T(a)@b <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    conv_cur = T @ a

    if active_set is not None and len(active_set) < size:
        T_sub = T[active_set, :]
    else:
        threshold = M * (1.0 - tol_frac)
        active_set = np.where(conv_cur > threshold)[0]
        if len(active_set) < 20:
            threshold = M * 0.5
            active_set = np.where(conv_cur > threshold)[0]
        T_sub = T[active_set, :]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_sub @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, active_set

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, active_set

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None, active_set

    # Check for violated dropped constraints
    if active_set is not None and len(active_set) < size:
        full_conv = T @ b_val
        violated = np.where(full_conv > M * (1.0 + 1e-6))[0]
        if len(violated) > 0:
            new_active = np.union1d(active_set, violated)
            if len(new_active) > 0.85 * size:
                # Fallback to full
                T_full = T
                b_var2 = cp.Variable(n, nonneg=True)
                prob2 = cp.Problem(cp.Maximize(cp.sum(b_var2)), [T_full @ b_var2 <= M])
                try:
                    prob2.solve(solver=solver, verbose=False)
                    if prob2.status in ('optimal', 'optimal_inaccurate'):
                        b_val = np.maximum(np.asarray(b_var2.value, dtype=np.float64), 0.0)
                except Exception:
                    pass
                active_set = new_active
            else:
                T_sub2 = T[new_active, :]
                b_var2 = cp.Variable(n, nonneg=True)
                prob2 = cp.Problem(cp.Maximize(cp.sum(b_var2)), [T_sub2 @ b_var2 <= M])
                try:
                    prob2.solve(solver=solver, verbose=False)
                    if prob2.status in ('optimal', 'optimal_inaccurate'):
                        b_val = np.maximum(np.asarray(b_var2.value, dtype=np.float64), 0.0)
                except Exception:
                    pass
                active_set = new_active

    return b_val, active_set


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated LP best-response optimization."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    active_set = None

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b, active_set = lp_solve_best_response(y, M_curr, solver, active_set)
        if b is None:
            active_set = None
            b, _ = lp_solve_best_response(y, M_curr, solver, None, tol_frac=0.0)
            if b is None:
                continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 15 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP t={t} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def lse_gradient_step(a, T_temp, fft_size):
    """Compute gradient of LSE approximation using FFT. O(n log n)."""
    n = len(a)
    conv = fast_autocorr(a)
    max_conv = conv.max()
    exp_conv = np.exp((conv - max_conv) / T_temp)
    w = exp_conv / exp_conv.sum()

    # grad[i] = 2 * (sum_k w[k]*a[i-k] + sum_k w[k]*a[k-i])
    # First term: c[i] where c = full_conv(w, a)
    # Second term: e[n-1-i] where e = full_conv(a, w[::-1])
    c = full_convolution(w, a)
    e = full_convolution(a, w[::-1].copy())
    grad = 2.0 * (c[:n] + e[n - 1:n - 1 + n][::-1])
    return grad


def lse_optimize(a0, time_limit, verbose=True):
    """LSE gradient descent with Barzilai-Borwein step size."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    # Anneal temperature in stages
    n_stages = 10
    temps = np.geomspace(50.0, 0.3, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        max_steps = max(5, min(200, int(remaining * 5)))

        a_prev = a.copy()
        grad_prev = None
        step_bb = norm / (n * T)

        for it in range(max_steps):
            if time.monotonic() - t0 > time_limit:
                break

            grad = lse_gradient_step(a, T, fft_size)

            # Barzilai-Borwein step size
            if grad_prev is not None:
                delta_a = a - a_prev
                delta_g = grad - grad_prev
                s_dot = np.dot(delta_a, delta_a)
                y_dot = np.dot(delta_a, delta_g)
                if s_dot > 1e-15 and abs(y_dot) > 1e-15:
                    step_bb = s_dot / abs(y_dot)
                    step_bb = np.clip(step_bb, norm / (100 * n), norm * 10 / n)

            # Projected gradient step
            a_new = a - step_bb * grad
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s

            a_prev = a.copy()
            grad_prev = grad.copy()
            a = a_new

            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage T={T:.2f} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def fast_pair_cd(a0, time_limit, rng, verbose=True):
    """
    Fast pair coordinate descent with peak-focused selection.
    Uses precomputed influence vectors for O(1) per-pair delta evaluation.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1

    # Precompute influence matrix: infl[j, k] = a[k-j] + a[j-k] (contribution of position j to conv[k])
    # This is the convolution matrix T(a)
    def compute_influence(a_arr):
        """Compute full influence matrix T where T[k,j] = a[k-j] (with bounds)."""
        T_mat = np.zeros((size, n))
        for j in range(n):
            # T[k, j] = a[k-j] for valid k-j
            k_min = j
            k_max = j + n - 1
            T_mat[k_min:k_max + 1, j] = a
            # Also the "reverse" part: for conv[k] = sum_j a[j]*a[k-j]
            # The influence of a[j] on conv[k] is a[k-j] (if valid)
            # But we also need the symmetric part... 
            # Actually conv[k] = sum_j a[j] * a[k-j], so d conv[k] / d a[j] = a[k-j]
            # But since conv is a*a, we have d conv[k] / d a[j] = a[k-j] + a[j-k] (from both terms)
            # Wait no: conv[k] = sum_i a[i] * a[k-i]
            # d/d a[j]: term i=j gives a[k-j], term k-i=j gives i=k-j, so a[k-j] * a[j] contributes a[j]... 
            # Actually: conv[k] = sum_i a[i] * a[k-i]
            # d conv[k] / d a[j] = a[k-j] (from i=j) + a[j-k] (from k-i=j, i.e., i=k-j)
            # But a[j-k] is only valid if 0 <= j-k < n
            # So the full influence is a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
            pass
        # Simpler: compute via the fact that T[k,j] = a[k-j] + a[j-k] for valid indices
        # This equals: for each j, the j-th column of T is a "V-shape" centered at j
        # T[k,j] = a[|k-j|] if |k-j| < n... no that's not right either.
        # Let me just compute it directly.
        T_mat = np.zeros((size, n))
        for j in range(n):
            for k in range(size):
                v = 0.0
                d1 = k - j
                if 0 <= d1 < n:
                    v += a[d1]
                d2 = j - k
                if 0 <= d2 < n:
                    v += a[d2]
                T_mat[k, j] = v
        return T_mat

    # For efficiency, compute influence matrix using FFT-like approach
    # T[k, j] = a[k-j] (if valid) + a[j-k] (if valid)
    # The first part: for fixed j, as k varies, a[k-j] is a shifted version of a
    # The second part: for fixed j, as k varies, a[j-k] is a reversed shifted version
    
    def compute_influence_fast(a_arr):
        T_mat = np.zeros((size, n))
        for j in range(n):
            # First part: a[k-j] for k in [j, j+n-1]
            T_mat[j:j+n, j] += a_arr
            # Second part: a[j-k] for k in [max(0, j-n+1), j]
            # a[j-k]: when k=0, a[j]; when k=j, a[0]
            lo = max(0, j - n + 1)
            hi = j + 1
            T_mat[lo:hi, j] += a_arr[j-lo:j-hi+n+1][::-1] if hi > lo else 0
        return T_mat

    # Even faster: use the convolution structure
    # T[k,j] for the "forward" part: shift a by j, place at row j onwards
    # T[k,j] for the "backward" part: reverse a, shift appropriately
    def compute_influence_fft(a_arr):
        T_mat = np.zeros((size, n))
        # Forward: T_fwd[k,j] = a[k-j] if 0 <= k-j < n
        for j in range(n):
            T_mat[j:j+n, j] = a_arr
        # Backward: T_bwd[k,j] = a[j-k] if 0 <= j-k < n, i.e., k in [max(0,j-n+1), min(j, size-1)]
        for j in range(n):
            lo = max(0, j - n + 1)
            hi = min(j + 1, size)
            idx = np.arange(lo, hi)
            # a[j-k] for k in idx: a[j-lo], a[j-(lo+1)], ..., a[j-(hi-1)]
            # = a[j-lo : j-hi+1] reversed... let me think
            # j-k for k=lo: j-lo; for k=hi-1: j-hi+1
            # So values are a[j-lo], a[j-lo-1], ..., a[j-hi+1]
            # That's a_arr[j-lo : j-hi+1 : -1] if j-lo >= j-hi+1
            # j-lo to j-hi+1 (decreasing): a_arr[j-hi+1 : j-lo+1][::-1]
            T_mat[idx, j] += a_arr[j - hi + 1: j - lo + 1][::-1]
        return T_mat

    # Use the direct method (fast enough for n <= 4000)
    T_mat = compute_influence_fft(a)
    
    n_passes = 0
    no_improve = 0
    max_passes = 500

    while time.monotonic() - t0 < time_limit and n_passes < max_passes:
        n_passes += 1
        pass_improved = False

        # Find current peak and hot positions
        M_curr = conv.max()
        hot_thresh = M_curr * (1.0 - 0.01)
        hot = np.where(conv > hot_thresh)[0]
        
        # Prioritize indices by their contribution to hot positions
        # hotness[i] = sum_{k in hot} T_mat[k,i] * (conv[k] - hot_thresh + 1e-10)
        hot_weights = (conv[hot] - hot_thresh + 1e-10)
        hotness = T_mat[hot, :] @ hot_weights

        # Get indices sorted by hotness (process hottest first)
        perm = np.argsort(-hotness)
        # Add some randomness to the tail
        tail_start = int(0.7 * n)
        perm[tail_start:] = rng.permutation(perm[tail_start:])

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-9:
                continue

            # Find best j: minimize hotness[j] / (some measure of how much j "fills a gap")
            # Simple strategy: pick j with minimum hotness (among those with low current value)
            # But we also want j to not be the same as i
            j_candidates = np.where(hotness < hotness[i] * 0.8)[0]
            j_candidates = j_candidates[j_candidates != i]
            
            if len(j_candidates) == 0:
                j_candidates = np.arange(n)
                j_candidates = j_candidates[j_candidates != i]
            
            # Pick the top few candidates with lowest hotness
            if len(j_candidates) > 10:
                j_candidates = j_candidates[np.argsort(hotness[j_candidates])[:10]]
            
            best_delta = 0.0
            best_val = conv.max()
            best_j = -1

            T_i = T_mat[:, i]
            for j in j_candidates:
                T_j = T_mat[:, j]
                g = T_j - T_i  # change in conv per unit delta transferred from i to j

                # First-order: new_conv = conv + delta * g
                # Second-order: at k=2i: -2*ai*delta + delta^2; at k=2j: 2*a[j]*delta + delta^2
                # Find optimal delta in [0, ai]
                
                # Quick check: if g at the peak is positive, this transfer makes things worse
                k_star = int(np.argmax(conv))
                if g[k_star] > 0 and a[j] < ai * 0.1:
                    continue

                # Find optimal delta using the PWL structure
                # f(delta) = max_k(conv[k] + delta * g[k]) + second_order
                # Upper bound: find delta where the max line changes
                # For simplicity, use a small number of cleverly chosen delta values
                
                # The optimal delta is where the gradient changes sign
                # At the peak, if g[k*] < 0, increasing delta helps
                # The max will be limited by the "second highest" line
                
                # Use ternary-like search with 5 points
                deltas = np.array([0.0, ai * 0.25, ai * 0.5, ai * 0.75, ai])
                # Add the delta where conv + delta*g first exceeds M_curr
                pos_g = g > 0
                if np.any(pos_g):
                    cross_deltas = (M_curr - conv[pos_g]) / g[pos_g]
                    cross_deltas = cross_deltas[(cross_deltas > 0) & (cross_deltas < ai)]
                    if len(cross_deltas) > 0:
                        deltas = np.concatenate([deltas, cross_deltas[:3]])
                    deltas = np.clip(deltas, 0, ai)
                    deltas = np.unique(np.round(deltas, 10))

                for d in deltas:
                    if d <= 0:
                        continue
                    new_conv_val = (conv + d * g).max()
                    # Second order corrections
                    if 2 * i < size:
                        so_2i = conv[2 * i] - 2 * ai * d + d * d
                        new_conv_val = max(new_conv_val, so_2i)
                    if 2 * j < size:
                        so_2j = conv[2 * j] + 2 * a[j] * d + d * d
                        new_conv_val = max(new_conv_val, so_2j)
                    if new_conv_val < best_val - 1e-13:
                        best_val = new_conv_val
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                i2 = i
                j2 = best_j
                d2 = best_delta
                a[i2] -= d2
                a[j2] += d2
                s = a.sum()
                if s < 1e-15:
                    a[i2] += d2
                    a[j2] -= d2
                    continue
                a = a * norm / s

                # Update conv (fast: only changed portion)
                conv = fast_autocorr(a)
                cur_max = conv.max()
                
                if cur_max < best_score - 1e-13:
                    best_score = cur_max
                    best_a = a.copy()
                    pass_improved = True
                    no_improve = 0
                    # Update T_mat (approximate - recompute periodically)
                    if n_passes % 5 == 0:
                        T_mat = compute_influence_fft(a)
                else:
                    # Revert
                    a[j2] -= d2
                    a[i2] += d2
                    a = a * norm / a.sum()
                    conv = fast_autocorr(a)

        if not pass_improve:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def quick_pair_cd(a0, time_limit, rng, verbose=True):
    """
    Faster pair CD: recomputes influence every pass, uses simpler heuristics.
    Better suited for longer runs with moderate n.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1

    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False

        M_curr = conv.max()
        
        # Compute hotness for all positions
        # hotness[i] = T[i, :] weighted by proximity to peak
        # T[i, j] = a[i-j] + a[j-i]
        # Approximate: hotness[i] = sum over hot k of a[k-i] (rough proxy)
        hot_mask = conv > M_curr * 0.95
        hot_idx = np.where(hot_mask)[0]
        
        # Compute contribution of each position to the peak region
        contrib = np.zeros(n)
        for k in hot_idx[:50]:  # Limit to top 50 hot positions
            for j in range(n):
                d1 = k - j
                if 0 <= d1 < n:
                    contrib[j] += a[d1]
        
        perm = np.argsort(-contrib)
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-9:
                continue

            # Try to find best j
            # j should have low contribution
            candidate_j = perm[perm != i][:15]
            
            best_d = 0.0
            best_val = M_curr
            best_j = -1

            for j in candidate_j:
                if j == i:
                    continue
                # Compute g = influence_j - influence_i at each k
                # g[k] = (a[k-j] + a[j-k]) - (a[k-i] + a[i-k])
                # Only compute at the top few hot positions for speed
                test_ks = hot_idx[:20]
                g_vals = np.zeros(len(test_ks))
                for t_idx, k in enumerate(test_ks):
                    v = 0.0
                    d1 = k - j
                    if 0 <= d1 < n:
                        v += a[d1]
                    d2 = j - k
                    if 0 <= d2 < n:
                        v += a[d2]
                    d3 = k - i
                    if 0 <= d3 < n:
                        v -= a[d3]
                    d4 = i - k
                    if 0 <= d4 < n:
                        v -= a[d4]
                    g_vals[t_idx] = v

                # If all g_vals at hot positions are positive, skip
                if np.all(g_vals > 0):
                    continue

                # Try a few delta values
                for frac in [0.1, 0.3, 0.5, 0.7, 1.0]:
                    d = ai * frac
                    # Estimate new max at hot positions
                    new_hot_vals = conv[test_ks] + d * g_vals
                    new_est = new_hot_vals.max()
                    # Also check the diagonal terms
                    if 2 * i < size:
                        new_est = max(new_est, conv[2 * i] - 2 * ai * d + d * d)
                    if 2 * j < size:
                        new_est = max(new_est, conv[2 * j] + 2 * a[j] * d + d * d)
                    
                    if new_est < best_val - 1e-13:
                        # Verify with full conv
                        a[i] -= d
                        a[j] += d
                        s = a.sum()
                        a = a * norm / s
                        new_conv = fast_autocorr(a)
                        actual_max = new_conv.max()
                        if actual_max < best_val - 1e-13:
                            best_val = actual_max
                            best_d = d
                            best_j = j
                            conv = new_conv
                        else:
                            # Revert
                            a[j] -= d
                            a[i] += d
                            a = a * norm / a.sum()
                            conv = fast_autocorr(a)

            if best_j >= 0 and best_d > 0 and best_val < best_score - 1e-13:
                best_score = best_val
                best_a = a.copy()
                pass_improved = True
                no_improve = 0
            elif best_j >= 0 and best_d > 0:
                # Applied but didn't improve global best
                pass_improved = True

        if not pass_improve:
            no_improve += 1
            if no_improve >= 8:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    QPCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    else:
        x = np.linspace(0.0, 1.0, n)
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize if no parent
    if best_seq is None:
        n_base = 1200
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # === Phase 1: LP Nesterov at base length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n1 = len(best_seq)
    print(f"\n=== Phase 1: LP Nesterov n={n1} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.25, 400.0)
    if t1 > 30:
        a_work = best_seq * np.sqrt(2.0 * n1) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_work, t1, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n1}")

    # === Phase 2: Fast pair CD (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Fast PCD n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.30, 500.0)
    if t2 > 20:
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = fast_pair_cd(a_work, t2, rng, verbose=True)
        do_try_update(a_out, f"pcd_n{n_best}")

    # === Phase 3: LSE refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LSE n={n_best} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 10:
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_optimize(a_work, t3, verbose=True)
        do_try_update(a_out, f"lse_n{n_best}")

    # === Phase 4: Pair CD again after LSE ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: PCD after LSE n={n_best} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 15:
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = fast_pair_cd(a_work, t4, rng, verbose=True)
        do_try_update(a_out, f"pcd2_n{n_best}")

    # === Phase 5: Try different lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        # Try a slightly different length
        n_alt = n_best + 200
        print(f"\n=== Phase 5a: LP n={n_alt} (budget={remaining:.0f}s) ===")
        t5a = min(remaining * 0.15, 150.0)
        if t5a > 20:
            init_alt = interpolate_sequence(best_seq, n_alt)
            a_work = init_alt * np.sqrt(2.0 * n_alt) / init_alt.sum()
            a_out, _ = nesterov_lp_optimize(a_work, t5a, solver, verbose=False)
            do_try_update(a_out, f"lp_n{n_alt}")
            # Quick CD on it
            remaining2 = time_budget - (time.monotonic() - t_global)
            t5b = min(remaining2 * 0.10, 80.0)
            if t5b > 10 and best_seq is not None:
                n_curr = len(best_seq)
                a_work2 = best_seq * np.sqrt(2.0 * n_curr) / best_seq.sum()
                a_out2, _ = fast_pair_cd(a_work2, t5b, rng, verbose=False)
                do_try_update(a_out2, f"pcd_n{n_curr}_alt")

    # === Phase 6: Try shorter length (easier to optimize) ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_short = max(600, int(len(best_seq) * 0.7))
        print(f"\n=== Phase 6: LP+CD n={n_short} (budget={remaining:.0f}s) ===")
        t6 = min(remaining * 0.15, 120.0)
        if t6 > 20:
            init_short = interpolate_sequence(best_seq, n_short)
            a_work = init_short * np.sqrt(2.0 * n_short) / init_short.sum()
            a_out, _ = nesterov_lp_optimize(a_work, min(t6 * 0.6, 70), solver, verbose=False)
            do_try_update(a_out, f"lp_n{n_short}")
            remaining2 = time_budget - (time.monotonic() - t_global)
            if remaining2 > 20:
                n_curr = len(best_seq)
                a_work2 = best_seq * np.sqrt(2.0 * n_curr) / best_seq.sum()
                a_out2, _ = quick_pair_cd(a_work2, min(remaining2 * 0.3, 60), rng, verbose=False)
                do_try_update(a_out2, f"qpcd_n{n_curr}")

    # === Phase 7: Final intensive CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final CD n={n_best} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.5, 200.0)
    if t7 > 10:
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = fast_pair_cd(a_work, t7, rng, verbose=True)
        do_try_update(a_out, f"final_pcd_n{n_best}")

    # === Phase 8: LSE polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        print(f"\n=== Phase 8: LSE polish n={n_best} (budget={remaining:.0f}s) ===")
        t8 = min(remaining * 0.5, 80.0)
        if t8 > 5:
            a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = lse_optimize(a_work, t8, verbose=False)
            do_try_update(a_out, f"lse_polish_n{n_best}")

    # Final
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
