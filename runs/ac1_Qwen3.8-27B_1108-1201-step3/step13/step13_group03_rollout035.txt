def _lp_best_response_adaptive(a, y_ref, n, max_constraints=200, solver=None):
    """LP best response with adaptive constraint selection."""
    solver = solver or _get_solver()
    b_var = cp.Variable(n, nonneg=True)
    
    # Find the most active constraints using y_ref
    xy = _cross_corr(a, y_ref)  # (a*y_ref)(k) for k=0,...,2n-2
    # Select top K constraints
    K = min(max_constraints, 2*n-1)
    top_k_idx = np.argsort(xy)[::-1][:K]
    
    # Build constraint matrix for selected constraints
    T_sel = np.zeros((K, n))
    for row, k in enumerate(top_k_idx):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        if j_start <= j_end:
            T_sel[row, j_start:j_end+1] = a[j_start:j_end+1]
    
    M = xy.max()
    constraints = [T_sel @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    return np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)

_global_solver = None


def _get_solver():

def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def _cross_corr(a, b):
    """(a*b)(k) = sum_i a_i * b_{k-i} (linear cross-correlation)."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    result = irfft(A * np.conj(B), fft_size)[:size]
    return result


def _fast_conv(a, b):
    """Full linear convolution a*b using FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def _proj_simplex(v, target_sum=1.0):
    n = len(v)
    if n <= 1:
        return np.array([max(0.0, target_sum)])
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 1e-15:
        out *= target_sum / s
    else:
        out = np.full(n, target_sum / n)
    return out


def _lse_grad(y, T):
    n = len(y)
    ac = _fast_autocorr(y)
    ac_max = ac.max()
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(y, fft_size)
    W = rfft(w, fft_size)
    cross = irfft(np.conj(A) * W, fft_size)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad


def _lse_fista(a0, T_schedule, iters_per_stage, time_limit, verbose=False):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    L_init = n / 50.0

    for stage_i, T in enumerate(T_schedule):
        L = max(L_init / (T * 10.0), n / (T * 2.0))
        t_fista = 1
        staging = a.copy()
        max_iters = max(int(iters_per_stage * (T_schedule[0] / T) ** 0.3), 2000)

        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_score

            grad = _lse_grad(y, T)
            step = 1.0 / L
            z = y - step * grad
            a_new = _proj_simplex(z)

            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - staging)
                y_new = np.maximum(y_new, 0.0)
                s_y = y_new.sum()
                if s_y > 1e-15:
                    y_new = y_new / s_y
                else:
                    y_new = a_new.copy()

            staging = a_new.copy()
            a = a_new
            y = y_new
            t_fista += 1

            if it % 500 == 0:
                sc = _fast_autocorr(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

        sc = _fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE-FISTA stage={stage_i} T={T:.2f} ({elapsed:.0f}s): best={best_score:.8f}")

    sc = _fast_autocorr(a).max()
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

def _build_constraint_matrix(a, n, k_indices):
    """Build constraint matrix T where T[i,j] = a[j] if 0 <= k_i - j <= n-1."""
    K = len(k_indices)
    T = np.zeros((K, n))
    for i, k in enumerate(k_indices):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        if j_start <= j_end:
            T[i, j_start:j_end + 1] = a[j_start:j_end + 1]
    return T


def _lp_adaptive(a, y_ref, n, max_K, solver):
    """Fast LP best-response using only top-K most active constraints."""
    K = min(max_K, 2 * n - 1)
    # Find most active constraints
    xy = _fast_conv(a, y_ref)
    top_idx = np.argsort(xy)[::-1][:K]
    top_idx = np.sort(top_idx)
    
    M = xy[top_idx].max()
    if M < 1e-15:
        return np.ones(n) / n
    
    T = _build_constraint_matrix(a, n, top_idx)
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def _lp_full(a, y_ref, n, solver):
    """Full LP best-response with all constraints (slower but exact)."""
    num_cons = 2 * n - 1
    xy = _fast_conv(a, y_ref)
    M = xy.max()
    if M < 1e-15:
        return np.ones(n) / n
    
    # Build full constraint matrix efficiently
    T = np.zeros((num_cons, n))
    for k in range(num_cons):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        if j_start <= j_end:
            T[k, j_start:j_end + 1] = a[j_start:j_end + 1]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def _lp_nesterov_adaptive(a0, time_limit, solver, verbose=True, use_full=False, full_start_iter=500):

    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()

        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    y = a.copy()
    polyak_avg = np.zeros(n)
    n_polyak = 0
    t0 = time.monotonic()
    n_iters = 0
    stall_count = 0
    stall_threshold = 200

    # Adaptive K schedule: start small, grow over time
    K_schedule = [(0, 50), (100, 100), (300, 200), (600, 400), (1000, 2*n-1 if use_full else 400)]
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Determine K for this iteration
        curr_K = K_schedule[-1][1]
        for threshold, K_val in K_schedule:
            if t <= threshold:
                curr_K = K_val
                break
        if use_full and t >= full_start_iter:
            curr_K = 2 * n - 1

        if curr_K < 2 * n - 1:
            b = _lp_adaptive(a, y, n, curr_K, solver)
        else:
            b = _lp_full(a, y, n, solver)
        
        if b is None:
            # Fallback: use the current y
            b = y.copy()
            s_b = b.sum()
            if s_b < 1e-15:
                b = np.ones(n) / n
            else:
                b = b * norm / s_b
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        # Polyak averaging
        polyak_avg += a
        n_polyak += 1

        # Check scores
        score_y = _fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = _fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        # Check Polyak average periodically
        if n_polyak % 50 == 0:
            avg_check = polyak_avg / n_polyak
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = _fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    stall_count = 0

        # Restart if stalled
        if stall_count > stall_threshold:
            # Restart from best_a
            y = best_a.copy()
            a = best_a.copy()
            polyak_avg = np.zeros(n)
            n_polyak = 0
            stall_count = 0
            if verbose:

def _peak_shave(a, time_limit, seed=42):
    n = len(a)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = _fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:6]
        improved = False

        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            # Find destination: element that's in peak regions but small
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]

            candidates_sorted = candidates[np.argsort(a[candidates])][:max(1, len(candidates) // 4)]
            dst = candidates_sorted[rng.integers(len(candidates_sorted))]

            for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.05]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = _fast_autocorr(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-10:
                frac = rng.uniform(0.003, 0.05)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = _fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0

            if stall > 300:
                break

    return best_a, best_score


def _interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

def _fast_conv(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def _proj_simplex(v, target_sum=1.0):
    n = len(v)
    if n <= 1:
        return np.array([max(0.0, target_sum)])
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 1e-15:
        out *= target_sum / s
    else:
        out = np.full(n, target_sum / n)
    return out


def _lse_grad(y, T):
    n = len(y)
    ac = _fast_autocorr(y)
    ac_max = ac.max()
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(y, fft_size)
    W = rfft(w, fft_size)
    cross = irfft(np.conj(A) * W, fft_size)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad

def _build_constraint_matrix_fast(a, n, k_indices):

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = _fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:6]
        improved = False

        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            if contrib.max() < 1e-15:
                continue
            src = int(idx[np.argmax(contrib)])

            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]
            if len(candidates) == 0:
                continue

            candidates_sorted = candidates[np.argsort(a[candidates])][:max(1, len(candidates) // 4)]
            dst = int(candidates_sorted[rng.integers(len(candidates_sorted))])

            for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.05, 0.08]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = _fast_autocorr(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > norm * 1e-10:
                frac = rng.uniform(0.003, 0.05)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = _fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0

            if stall > 400:
                break

    return best_a, best_score


def _interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def _evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def _spectral_init(n, seed=0):
    """Generate initial sequence using spectral design (inverse of desired autocorrelation)."""
    rng = np.random.default_rng(seed)
    # Start with a target autocorrelation shape (flat top)
    target_ac = np.ones(2 * n - 1)
    # Taper at edges to ensure positive definiteness
    taper = np.ones(2 * n - 1)
    edge = n // 4
    for i in range(edge):
        taper[i] = 0.5 * (1 + np.cos(np.pi * i / edge))
        taper[2 * n - 2 - i] = 0.5 * (1 + np.cos(np.pi * i / edge))
    target_ac *= taper

    # Find a signal with this autocorrelation via IFFT
    # Power spectrum = FFT of autocorrelation
    fft_size = 1 << (2 * n - 2).bit_length()
    P = np.zeros(fft_size, dtype=np.complex128)
    # Wrap the autocorrelation to get the power spectrum
    P[:n] += target_ac[:n]
    P[fft_size - (2*n-2-n):] += target_ac[n:]
    # Ensure non-negativity of power spectrum
    P = np.maximum(np.real(P), 1e-10)

    # Take square root and IFFT to get the signal
    signal = irfft(np.sqrt(P), fft_size)
    # Fold to length n (take first n and last n-1, average)
    a = np.zeros(n)
    for i in range(n):
        a[i] = 0.5 * (signal[i] + signal[fft_size - (n - 1 - i)] if (fft_size - (n - 1 - i)) < fft_size else signal[i])
    a = np.abs(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search: adaptive LP-Nesterov with progressive constraint selection."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1680.0)

    solver = _get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    # Try to load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = _evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    # Determine base length
    n_base = 2000

    # Build initial sequence
    if parent is not None and len(parent) >= 1000:
        init = _interpolate_sequence(best_seq, n_base)
    else:
        # Mix of raised cosine and spectral init
        x = np.linspace(0.0, 1.0, n_base)
        init1 = 0.5 * (1 + np.cos(2 * np.pi * x))
        init1 = np.maximum(init1, 1e-6)
        init1 = init1 / init1.sum()
        init2 = _spectral_init(n_base, seed=seed)
        init = 0.5 * (init1 + init2)
        init = np.maximum(init, 1e-8)
        init = init / init.sum()

    best_seq = init.copy()
    init_score = _evaluate_safe(best_seq)
    if init_score < best_score:
        best_score = init_score
    print(f"Init: n={n_base}, score={best_score:.6f}")

    def _try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = _evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: LSE-FISTA (quick refinement) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.08, 120.0)
    if t1 > 20:
        print(f"\n=== Phase 1: LSE-FISTA n={n_base} (budget={t1:.0f}s) ===")
        T_sched = [8.0, 2.5, 0.8, 0.25, 0.08]
        a_out, _ = _lse_fista(best_seq, T_sched, 8000, t1, verbose=True)
        _try_update(a_out, f"lse1_n{n_base}")

    # === Phase 2: Main LP-Nesterov at n_base ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.38, 550.0)
    if t2 > 60:
        print(f"\n=== Phase 2: LP-Nesterov n={n_cur} (budget={t2:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _lp_nesterov_v2(a_norm, t2, solver, verbose=True)
        _try_update(a_out, f"lp1_n{n_cur}")

    # === Phase 3: Try n+400 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = len(best_seq) + 400
    t3 = min(remaining * 0.18, 280.0)
    if t3 > 40:
        print(f"\n=== Phase 3: LP-Nesterov n={n_long} (budget={t3:.0f}s) ===")
        init_long = _interpolate_sequence(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = _lp_nesterov_v2(a_norm, t3, solver, verbose=True)
        _try_update(a_out, f"lp2_n{n_long}")

    # === Phase 4: LSE-FISTA polish at best length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.10, 150.0)
    if t4 > 15:
        print(f"\n=== Phase 4: LSE-FISTA polish n={n_best} (budget={t4:.0f}s) ===")
        T_sched = [3.0, 1.0, 0.3, 0.1, 0.03]
        a_out, _ = _lse_fista(best_seq, T_sched, 15000, t4, verbose=True)
        _try_update(a_out, f"lse2_n{n_best}")

    # === Phase 5: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 10:
        print(f"\n=== Phase 5: Peak shave n={n_best} (budget={t5:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = _peak_shave(a_norm, t5, seed=12345)
        _try_update(a_out, "peak_shave")

    # === Phase 6: Final LP-Nesterov at best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t6 = min(remaining * 0.28, 350.0)
    if t6 > 30:
        print(f"\n=== Phase 6: Final LP n={n_best} (budget={t6:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = _lp_nesterov_v2(a_norm, t6, solver, verbose=True)
        _try_update(a_out, "lp_final")

    # === Phase 7: Final peak shave ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        n_best = len(best_seq)
        t7 = min(remaining * 0.5, 50.0)
        if t7 > 5:
            print(f"\n=== Phase 7: Final peak shave n={n_best} (budget={t7:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = _peak_shave(a_norm, t7, seed=99999)
            _try_update(a_out, "peak_shave_final")

    # Final verification
    if best_seq is not None:
        fs = _evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
