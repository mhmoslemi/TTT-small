"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast LP zero-sum game with constraint generation (active set)
2. Nesterov momentum with adaptive restarts

4. Top-k LSE gradient descent with Adam

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def toep_memcpy(a_fixed, b):
    """Compute T @ b where T is the Toeplitz-like matrix for convolution with a_fixed.
    i.e., (a_fixed * b)_k = sum_j a_fixed_j * b_{k-j}
    Using FFT for O(n log n).
    """
    n = len(a_fixed)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a_fixed, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def find_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.SCS, cp.PDLP, cp.ECOS, cp.GLPK]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_active_lp(a_fixed, active_set, solver, warm_b=None, max_constraints=200):
    """
    Solve LP: min M s.t. (a_fixed*b)_k <= M for k in active_set, sum(b)=1, b>=0.
    Uses constraint generation internally.
    Returns (b_optimal, M_optimal).
    """
    n = len(a_fixed)
    
    # Build constraint matrix for active set
    # T[k, j] = a_fixed[j] if 0 <= k-j < n else 0
    # Actually: (a_fixed * b)_k = sum_j a_fixed[j] * b[k-j]
    # So coefficient of b[j] in constraint k is a_fixed[k-j] if 0 <= k-j < n
    
    K = list(active_set)
    if len(K) > max_constraints:
        K = K[:max_constraints]
    
    # Build the constraint matrix
    m = len(K)
    T = np.zeros((m, n))
    for i, k in enumerate(K):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            T[i, j] = a_fixed[k - j]
    
    # Set up LP
    if warm_b is not None and np.all(warm_b >= -1e-10) and warm_b.sum() > 1e-10:
        b_var = cp.Variable(n, nonneg=True, value=warm_b)
    else:
        b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable(value=1.0)
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.CLARABEL, cp.PDLP, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, active_set
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, active_set
    b_val = b_val / s
    M_val = float(M_var.value)
    
    # Check for violated constraints (constraint generation)
    # Compute full convolution a_fixed * b_val
    full_conv = toep_memcpy(a_fixed, b_val)
    
    # Find most violated constraint not in active set
    active_set_arr = np.array(active_set)
    mask = np.ones(len(full_conv), dtype=bool)
    mask[active_set_arr] = False
    
    violated = full_conv[mask]
    if len(violated) > 0:
        max_violated_val = violated.max()
        if max_violated_val > M_val * (1 + 1e-6):
            # Find the index
            violated_idx = np.where(mask)[0]
            worst_k = violated_idx[np.argmax(violated)]
            new_active_set = list(active_set) + [worst_k]
            return b_val, M_val, new_active_set
        else:
            # No significant violation
            return b_val, M_val, active_set
    else:
        return b_val, M_val, active_set


def nesterov_lp_fast(a0, time_limit, solver, max_iters=50000):
    """

    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    y = a.copy()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Initialize active set with top peaks
    c0 = fast_autocorr(y)
    initial_active = np.argsort(c0)[::-1][:30].tolist()
    active_set = list(initial_active)
    
    warm_b = None
    stall_count = 0
    last_print = 0
    
    # Running averages
    avg_long = np.zeros(n)
    n_long = 0
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP with constraint generation
        b_star, M_val, active_set = solve_active_lp(y, active_set, solver, warm_b)
        
        if b_star is None:
            warm_b = None
            # Try with fewer constraints
            active_set = np.argsort(fast_autocorr(y))[-5:].tolist()
            continue
        
        warm_b = b_star
        
        # Nesterov update
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_sum = y_new.sum()
        if y_sum < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / y_sum
        
        if t == 1:
            a_new = b_star.copy()
            a_prev = y
        else:
            a_prev = a
            # Nesterov extrapolation
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new
        
        y = y_new
        a = a_new
        
        # Track running average
        avg_long += a
        n_long += 1
        
        # Check scores
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        
        # Check running average every 50 iterations
        if n_long % 50 == 0 and n_long >= 100:
            avg_check = avg_long / n_long
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
                stall_count = 0
        
        # Adaptive restart
        if stall_count > 300 and t > 50:
            a = best_a.copy()
            y = best_a.copy()
            avg_long = np.zeros(n)
            n_long = 0
            stall_count = 0
            warm_b = None
            c_restart = fast_autocorr(y)
            active_set = np.argsort(c_restart)[::-1][:30].tolist()
        
        if t - last_print > 500:
            elapsed = time.monotonic() - t0
            print(f"      LP t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}, |K|={len(active_set)}")
            last_print = t
    
    return best_a, best_score


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def compute_topk_lse_grad(a, T_temp, topk):
    """Compute top-k LSE objective gradient."""
    n = len(a)
    c = fast_autocorr(a)
    k_use = min(topk, len(c))
    top_idx = np.argsort(c)[::-1][:k_use]
    top_vals = c[top_idx]
    
    max_Tc = (T_temp * top_vals).max()
    exp_Tc = np.exp(T_temp * top_vals - max_Tc)
    lse_val = (max_Tc + np.log(exp_Tc.sum())) / T_temp
    w = exp_Tc / exp_Tc.sum()
    
    grad_a = np.zeros(n)
    for idx in range(k_use):
        K = top_idx[idx]
        wK = w[idx]
        js = max(0, K - (n - 1))
        je = min(n - 1, K)
        if js <= je:
            idx_range = np.arange(js, je + 1)
            grad_a[idx_range] += 2.0 * wK * a[K - idx_range]
    
    return lse_val, grad_a

            grad_z = a * (grad_a - S)
            
            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            if (it + 1) % 5 == 0:
                a = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
        
        a_stage = softmax_fn(z)
        score_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()
        
        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE stage {si}/{n_stages} (T={stage_T:.2f}, {elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast LP zero-sum game with constraint generation (active set) - key improvement
2. Nesterov momentum with adaptive restarts
3. Multi-length exploration with interpolation

5. Iterative peak shaving with paired moves and adaptive step sizes

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_pair(a, b):
    """Compute a*b (full convolution) using FFT."""
    n = max(len(a), len(b))
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def find_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.SCS, cp.PDLP, cp.ECOS, cp.GLPK]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_constraint_matrix(a_fixed, active_indices):
    """
    Build constraint matrix T where T[i,j] gives coefficient of b_j in constraint i.
    (a_fixed * b)_k = sum_j a_fixed[k-j] * b_j  (for valid indices)
    So T[i,j] = a_fixed[k_j - j] where k_j is the index for row i.
    """
    n = len(a_fixed)
    m = len(active_indices)
    T = np.zeros((m, n))
    for i, k in enumerate(active_indices):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            T[i, j] = a_fixed[k - j]
    return T


def solve_active_lp(a_fixed, active_set, solver, warm_b=None, max_constraints=150):
    """
    Solve LP with constraint generation.
    min M s.t. (a_fixed*b)_k <= M for k in active_set, sum(b)=1, b>=0
    Returns (b_val, M_val, new_active_set) or (None, None, active_set) on failure.
    """
    n = len(a_fixed)
    
    if len(active_set) > max_constraints:
        # Keep the most important constraints (those with highest RHS in current solution)
        active_set = active_set[:max_constraints]
    
    T = build_constraint_matrix(a_fixed, active_set)
    m = len(T)
    
    if warm_b is not None and np.all(warm_b >= -1e-10) and warm_b.sum() > 1e-10:
        b_var = cp.Variable(n, nonneg=True, value=warm_b)
    else:
        b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable(value=1.0 / max(1, min(n, 10)))
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.CLARABEL, cp.PDLP, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, active_set
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, active_set
    b_val = b_val / s
    M_val = float(M_var.value)
    
    # Constraint generation: check if any non-active constraint is violated
    full_conv = convolve_pair(a_fixed, b_val)
    
    active_arr = np.array(active_set, dtype=int)
    mask = np.ones(len(full_conv), dtype=bool)
    valid_mask = (active_arr >= 0) & (active_arr < len(full_conv))
    if valid_mask.any():
        mask[active_arr[valid_mask]] = False
    
    violated_vals = full_conv[mask]
    if len(violated_vals) > 0:
        max_viol = violated_vals.max()
        if max_viol > M_val * (1 + 1e-5) + 1e-12:
            violated_idx = np.where(mask)[0]
            worst_k = int(violated_idx[np.argmax(violated_vals)])
            new_active = list(active_set) + [worst_k]
            return b_val, M_val, new_active
    return b_val, M_val, active_set

    Nesterov-accelerated zero-sum game with constraint generation LP.
    """

    if s < 1e-15:

    else:

    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    c0 = fast_autocorr(y)
    active_set = np.argsort(c0)[::-1][:40].tolist()
    
    warm_b = None
    stall_count = 0
    last_print = 0
    
    # Running averages with different windows
    avg_long = np.zeros(n)
    n_long = 0
    avg_med = np.zeros(n)
    n_med = 0
    MED_WINDOW = 40
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_star, M_val, active_set = solve_active_lp(y, active_set, solver, warm_b)
        
        if b_star is None:
            warm_b = None
            continue
        
        warm_b = b_star

        alpha_t = 2.0 / (t + 2.0)
        y_prev = y.copy()
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_sum = y_new.sum()
        if y_sum < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / y_sum
        
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new
        
        y = y_new
        a = a_new
        
        # Update running averages
        avg_long += a
        n_long += 1
        avg_med += a
        n_med += 1
        if n_med > MED_WINDOW:
            # Simple exponential-ish: just keep tracking
            pass
        
        # Check scores
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        
        # Check long average periodically
        if n_long % 20 == 0 and n_long >= 50:
            avg_check = avg_long / n_long
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
                stall_count = 0
        
        # Adaptive restart
        if stall_count > 250 and t > 50:
            a = best_a.copy()
            y = best_a.copy()
            avg_long = np.zeros(n)
            n_long = 0
            stall_count = 0
            warm_b = None
            c_restart = fast_autocorr(y)
            active_set = np.argsort(c_restart)[::-1][:40].tolist()
        
        if t - last_print > 200:
            elapsed = time.monotonic() - t0
            print(f"      LP t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.10f}, |K|={len(active_set)}")
            last_print = t
    
    return best_a, best_score


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def lse_grad_step(z, T_temp, topk):
    """One gradient step for top-k LSE in z-space."""
    a = softmax_fn(z)
    n = len(a)
    c = fast_autocorr(a)
    k_use = min(topk, len(c))
    top_idx = np.argsort(c)[::-1][:k_use]
    top_vals = c[top_idx]
    
    scaled = T_temp * top_vals
    max_s = scaled.max()
    exp_s = np.exp(scaled - max_s)
    w = exp_s / exp_s.sum()
    
    grad_a = np.zeros(n)
    for idx in range(k_use):
        K = int(top_idx[idx])
        wK = w[idx]
        js = max(0, K - (n - 1))
        je = min(n - 1, K)
        if js <= je:
            idx_range = np.arange(js, je + 1)
            grad_a[idx_range] += 2.0 * wK * a[K - idx_range]
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    score = 2.0 * n * c.max()
    return score, grad_z, a


def adam_lse_optimize(a0, time_limit, T_start=30.0, T_end=0.05, n_stages=20, topk=15):
    """Adam-optimized top-k LSE with multi-stage temperature annealing."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        base_lr = 5e-3 * min(1.0, 300.0 / n)
        lr = base_lr * (0.92 ** si)
        
        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.75, 0.999, 1e-8
        
        max_it = max(10, int(remaining * 4))
        
        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 2:
                break
            
            score, grad_z, a_curr = lse_grad_step(z, stage_T, topk)
            
            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            if (it + 1) % 4 == 0:
                a_check = softmax_fn(z)
                sc = 2.0 * n * fast_autocorr(a_check).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a_check.copy()
        
        a_stage = softmax_fn(z)
        score_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()
        
        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE {si}/{n_stages} (T={stage_T:.2f}, {elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score


def peak_shaving_advanced(a0, time_limit, seed=42):
    """
    Advanced peak shaving with:
    - Multiple peaks considered simultaneously
    - Paired and triple moves
    - Adaptive step sizes based on peak structure
    - Broader destination search
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit - 2:
        b = fast_autocorr(a)
        bmax = b.max()
        
        # Find peaks within 0.5% of max
        threshold = bmax * 0.998
        top_peaks = np.where(b >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(b)[::-1][:5]
        
        # Compute danger zones (positions affecting any top peak)
        danger = np.zeros(n, dtype=bool)
        for k in top_peaks[:25]:
            js = max(0, int(k) - n + 1)
            je = min(n - 1, int(k))
            danger[js:je + 1] = True
        
        improved = False
        
        # Single moves
        for k in top_peaks[:20]:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-18:
                continue
            
            # Top contributing source positions
            top_src = np.argsort(contrib)[::-1][:8]
            
            for si in range(min(4, len(top_src))):
                src = int(idx[top_src[si]])
                
                # Destinations: prefer safe, low-value positions
                safe = np.where(~danger)[0]
                safe = safe[safe != src]
                if len(safe) == 0:
                    safe = np.where(np.arange(n) != src)[0]
                safe_sorted = safe[np.argsort(a[safe])][:25]
                
                found = False
                for dst in safe_sorted[:12]:
                    for frac in [0.002, 0.004, 0.008, 0.015, 0.03, 0.06, 0.10]:
                        amt = a[src] * frac
                        if amt < 1e-16:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[int(dst)] += amt
                        if a_trial[src] < -1e-12:
                            continue
                        a_trial = np.maximum(a_trial, 0)
                        st = a_trial.sum()
                        if st < 1e-12:
                            continue
                        a_trial = a_trial / st
                        new_max = fast_autocorr(a_trial).max()
                        if new_max < bmax * (1 - 1e-10):
                            a = a_trial
                            new_score = 2.0 * n * new_max
                            if new_score < best_score:
                                best_score = new_score
                                best_a = a.copy()
                            improved = True
                            found = True
                            break
                    if found:
                        break
                if found:
                    break
            if improved:
                break
        
        # Paired moves (if single moves didn't work)
        if not improved:
            for k in top_peaks[:8]:
                k = int(k)
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                if len(idx) < 2:
                    continue
                contrib = a[idx] * a[k - idx]
                top_src = np.argsort(contrib)[::-1][:4]
                
                if len(top_src) < 2:
                    continue
                
                src1 = int(idx[top_src[0]])
                src2 = int(idx[top_src[1]])
                
                safe = np.where(~danger)[0]
                safe = safe[(safe != src1) & (safe != src2)]
                if len(safe) < 2:
                    continue
                safe_sorted = safe[np.argsort(a[safe])][:12]
                
                found = False
                for d1_i in range(min(6, len(safe_sorted))):
                    d1 = int(safe_sorted[d1_i])
                    for d2_i in range(min(6, len(safe_sorted))):
                        if d2_i == d1_i:
                            continue
                        d2 = int(safe_sorted[d2_i])
                        for frac in [0.003, 0.008, 0.015, 0.03]:
                            amt1 = a[src1] * frac
                            amt2 = a[src2] * frac
                            a_trial = a.copy()
                            a_trial[src1] -= amt1
                            a_trial[src2] -= amt2
                            a_trial[d1] += amt1
                            a_trial[d2] += amt2
                            if a_trial[src1] < -1e-12 or a_trial[src2] < -1e-12:
                                continue
                            a_trial = np.maximum(a_trial, 0)
                            st = a_trial.sum()
                            if st < 1e-12:
                                continue
                            a_trial = a_trial / st
                            new_max = fast_autocorr(a_trial).max()
                            if new_max < bmax * (1 - 1e-10):
                                a = a_trial
                                new_score = 2.0 * n * new_max
                                if new_score < best_score:
                                    best_score = new_score
                                    best_a = a.copy()
                                improved = True
                                found = True
                                break
                        if found:
                            break
                    if found:
                        break
                if found:
                    break
            if improved:
                continue
        
        if not improved:
            stall += 1
            # Escalating random perturbations
            if stall % 3 == 0:
                scale = min(0.05, 0.005 * (stall // 3))
                n_pert = rng.integers(2, max(3, n_pert if False else 5))
                a_trial = a.copy()
                for _ in range(n_pert):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a_trial[i] > 1e-10:
                        frac = rng.uniform(0.001, scale)
                        amt = a_trial[i] * frac
                        a_trial[i] -= amt
                        a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-10:
                    a_trial = a_trial / st
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            
            if stall > 80:
                break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, time_budget=1800.0):
    """
    Main search function. Uses fast constraint-generation LP game as the core,
    supplemented by LSE gradient polish and advanced peak shaving.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(time_budget) - 80.0, 1720.0)
    
    solver = find_solver()
    print(f"LP solver: {solver}, budget: {time_budget:.0f}s")
    
    # Get parent sequence
    best_seq = None
    best_score = float('inf')
    
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if best_seq is None:
        n_init = 1600
        best_seq = np.ones(n_init) / n_init
        best_score = evaluate_safe(best_seq)
        print(f"Using uniform init: n={n_init}, score={best_score:.6f}")
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    n_base = len(best_seq)
    
    # === Phase 1: Main LP game ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP game n={n_base} ===")
    t1 = min(remaining * 0.45, 750.0)
    if t1 > 60:
        a_out, _ = nesterov_lp_game(best_seq, t1, solver)
        try_update(a_out, f"lp_n{n_base}")
    
    # === Phase 2: LP game with different momentum (restart) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.18, 300.0)
    if t2 > 40:
        print(f"\n=== Phase 2: LP game restart n={n_base} ===")
        # Slight perturbation to escape potential local structure
        pert = best_seq + rng.normal(0, 0.01, n_base)
        pert = np.maximum(pert, 0)
        pert = pert / pert.sum()
        a_out, _ = nesterov_lp_game(pert, t2, solver)
        try_update(a_out, f"lp2_n{n_base}")
    
    # === Phase 3: LSE gradient polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.12, 200.0)
    if t3 > 20:
        print(f"\n=== Phase 3: LSE polish n={len(best_seq)} ===")
        nn = len(best_seq)
        a_out, _ = adam_lse_optimize(best_seq, t3, T_start=20.0, T_end=0.05, n_stages=18, topk=12)
        try_update(a_out, f"lse_n{nn}")
    
    # === Phase 4: Try longer length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_base + 150
    t4 = min(remaining * 0.12, 180.0)
    if t4 > 50:

        init_long = interpolate_sequence(best_seq, n_long)
        a_out, _ = nesterov_lp_game(init_long, t4, solver)
        try_update(a_out, f"lp_n{n_long}")
    
    # === Phase 5: Peak shaving on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.10, 120.0)

        print(f"\n=== Phase 5: Peak shaving n={len(best_seq)} ===")
        a_out, _ = peak_shaving_advanced(best_seq, t5, seed=777)

    # Final evaluation
    if best_seq is not None:

            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
