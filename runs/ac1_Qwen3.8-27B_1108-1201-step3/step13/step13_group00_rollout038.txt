_global_solver = None

def _get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    return _global_solver


def _fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def _norm_seq(a, target_sum):
    """Normalize a to have the given sum, keeping non-negativity."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) * target_sum / len(a)
    return a * target_sum / s


def _structured_init(n, rng, mode=0):
    """Generate a structured non-negative initialization."""
    if mode == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif mode == 1:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif mode == 2:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2 * x) + 0.15 * np.cos(5 * x) + 0.05 * np.cos(9 * x)
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-8 * (x - 0.3) ** 2) + 0.7 * np.exp(-8 * (x - 0.7) ** 2)
    elif mode == 4:
        x = np.linspace(0, 1, n)
        a = (1 - np.abs(2 * x - 1)) ** 3
    elif mode == 5:
        # Sum of cosines with random phases
        a = np.ones(n)
        for _ in range(5):
            freq = rng.uniform(1, 10)
            phase = rng.uniform(0, 2 * np.pi)
            a += 0.3 * np.cos(2 * np.pi * freq * np.linspace(0, 1, n) + phase)
    elif mode == 6:
        # Random smooth (low-pass filtered noise)
        a = np.random.randn(n)
        # Simple exponential smoothing
        kernel = np.exp(-np.arange(20) / 5.0)
        a = np.convolve(a, kernel, mode='same')
    else:
        x = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def _interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def _lse_gradient_step(a, T, fft_size):
    """One LSE gradient step. Returns (new_a, gradient, b)."""
    n = len(a)
    size = 2 * n - 1
    A = rfft(a, fft_size)
    b = irfft(A * A, fft_size)[:size]
    b_max = b.max()
    if b_max < 1e-15:
        return a, np.zeros(n), b
    exp_b = np.exp((b - b_max) / T)
    w = exp_b / exp_b.sum()
    
    W = rfft(w, fft_size)
    # s1[i] = sum_k w[k]*a[k-i] = cross-correlation at negative lag
    c = irfft(W * np.conj(A), fft_size)
    s1 = np.zeros(n)
    s1[0] = c[0]
    if n > 1:
        s1[1:] = c[fft_size-1 : fft_size-n : -1]
    # s2[i] = sum_k w[k]*a[i-k] = convolution at position i
    conv_wa = irfft(W * A, fft_size)
    s2 = conv_wa[:n]
    
    g = 2.0 * (s1 + s2)
    return g, b


def _lse_optimize(a0, time_limit, n_steps=10000, verbose=False):
    """LSE gradient descent with temperature annealing."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = _norm_seq(a0, norm)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    
    T_init = norm / n * 5.0
    T_final = norm / n * 0.1
    
    for it in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        frac = it / n_steps
        T = T_init * (T_final / T_init) ** frac
        
        g, b = _lse_gradient_step(a, T, fft_size)
        
        # Adaptive step: proportional to norm / (gradient norm * sqrt(n))
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        step = 0.8 * norm / (g_norm * np.sqrt(n))
        # Cap the step to avoid oscillation
        step = min(step, 0.5 / np.max(np.abs(g)) if np.max(np.abs(g)) > 1e-15 else 1.0)
        
        a_new = a - step * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a = a_new * norm / s_new
        
        if it % 100 == 0:
            score = _fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
        
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE it={it} ({elapsed:.1f}s): best={best_score:.8f}")
    
    # Final check
    score = _fast_autocorr(a).max()
    if score < best_score:
        best_score = score
        best_a = a.copy()
    
    return best_a, best_score


def _lp_solve(a, M, solver, tol=0.05):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0. Uses screening."""
    n = len(a)
    size = 2 * n - 1
    
    # Compute current convolution
    conv_a = _fast_autocorr(a)
    
    # Screen: keep constraints near the bound
    active = np.where(conv_a > M * (1.0 - tol))[0]
    if len(active) < 5:
        active = np.where(conv_a > M * 0.5)[0]
    
    # Build constraint matrix for active rows only
    T_active = np.zeros((len(active), n))
    for idx, k in enumerate(active):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            T_active[idx, j] = a[k - j]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_active @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    
    # Verify no dropped constraints are violated
    full_conv = np.zeros(size)
    # Compute conv(a, b_val) via FFT
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b_val, fft_size)
    full_conv = irfft(A * B, fft_size)[:size]
    
    if full_conv.max() > M * (1 + 1e-4):
        # Some constraints violated; try with more active set
        active2 = np.where(full_conv > M * 0.9)[0]
        if len(active2) > 0.95 * size:
            # Just use all constraints
            T_all = np.zeros((size, n))
            for k in range(size):
                j_lo = max(0, k - n + 1)
                j_hi = min(n - 1, k)
                T_all[k, j_lo:j_hi+1] = a[k - np.arange(j_lo, j_hi+1)]
            b_var2 = cp.Variable(n, nonneg=True)
            prob2 = cp.Problem(cp.Maximize(cp.sum(b_var2)), [T_all @ b_var2 <= M])
            try:
                prob2.solve(solver=solver, verbose=False)
            except Exception:
                return b_val
            if prob2.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.maximum(np.asarray(b_var2.value, dtype=np.float64), 0.0)
            return b_val
    
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def _nesterov_lp(a0, time_limit, solver, verbose=False):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = _norm_seq(a0, norm)
    
    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    n_iter = 0
    last_print = 0
    
    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M = _fast_autocorr(y).max()
        if M < 1e-12:
            break
        
        b = _lp_solve(y, M, solver)
        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iter += 1
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / sy
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            sa = a_new.sum()
            if sa < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / sa
        
        y = y_new
        a = a_new
        
        if n_iter % 50 == 0:
            score_y = _fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()
            score_a = _fast_autocorr(a).max()
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()
        
        if verbose and n_iter - last_print >= 200:
            last_print = n_iter
            elapsed = time.monotonic() - t0
            print(f"    LP it={n_iter} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score
