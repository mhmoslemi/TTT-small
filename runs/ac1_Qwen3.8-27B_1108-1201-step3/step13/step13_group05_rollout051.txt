def _proj_simplex(v, s):
    """Project v onto {x >= 0, sum(x) = s} using Richardson's algorithm."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho + s - cssv > 0
    if not np.any(cond):
        return np.zeros(n)
    rho_val = u[np.max(np.where(cond)[0])]
    return np.maximum(v - rho_val, 0.0)

def _normalize(a, target_sum):

    if s < 1e-15:
        n = len(a)
        return np.ones(n) * target_sum / n
    return a * target_sum / s

def _smooth_max_and_grad(a, ac, beta):
    """Compute smooth max approximation and its gradient."""
    n = len(a)

    if p_sum < 1e-30:

        p = p / p_sum

        idx = np.arange(i_start, i_end)

def _lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. conv(a,b)_k <= M for all k, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    
    # Build constraint matrix T where T[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # conv(a, b)_k = sum_j a[j] * b[k-j]
    # For efficiency, use the fact that each column of T is a shifted version of a
    
    T = cp.ndarray((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def _interpolate(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def _coordinate_polish(a0, time_limit, verbose=False):
    """Coordinate descent: move mass between pairs to reduce max autocorrelation."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = _normalize(a0, c)
    
    best_a = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    stall = 0
    max_stall = 100
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = _fast_autocorr(a)
        # Find the lag with maximum autocorrelation
        k_max = int(np.argmax(ac))
        
        # Find the pair (i, j) that contributes most to this peak
        i_start = max(0, k_max - n + 1)
        i_end = min(n, k_max + 1)
        idx = np.arange(i_start, i_end)
        contrib = a[idx] * a[k_max - idx]
        
        if contrib.max() < 1e-15:
            stall += 10
            continue
        
        src = idx[np.argmax(contrib)]
        
        # Try moving a small amount from src to various destinations
        best_trial = None
        best_trial_m = best_m
        
        # Try random destinations
        for _ in range(20):
            dst = rng.integers(n)
            if dst == src:
                continue
            for frac in [0.01, 0.02, 0.05, 0.1]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = _normalize(a_trial, c)
                ac_trial = _fast_autocorr(a_trial)
                M_trial = ac_trial.max()
                if M_trial < best_trial_m:
                    best_trial_m = M_trial
                    best_trial = a_trial
        
        if best_trial is not None:
            a = best_trial
            best_m = best_trial_m
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            # Random perturbation to escape
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > c * 1e-8:
                frac = rng.uniform(0.01, 0.05)
                amt = min(a[i] * frac, a[i] * 0.5)
                a[i] -= amt
                a[j] += amt
                a = _normalize(a, c)
                M_new = _fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    Coord ({elapsed:.1f}s): score={2.0*n*best_m/(c*c):.6f}")
    
    return best_a, best_m

def _simulated_annealing(a0, time_limit, n_trials=500, verbose=False):
    """Simulated annealing: random perturbations with acceptance criterion."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = _normalize(a0, c)
    
    best_a = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    
    T = 0.01  # initial temperature
    T_min = 1e-6
    cooling = 0.999
    
    for t in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break
        if T < T_min:
            break
        
        # Generate perturbation: move mass between 2-4 random positions
        n_move = rng.integers(2, 5)
        srcs = rng.choice(n, size=n_move, replace=False)
        dsts = rng.choice(n, size=n_move, replace=False)
        
        a_trial = a.copy()
        for s, d in zip(srcs, dsts):
            if s == d:
                continue
            frac = rng.uniform(0.01, 0.1)
            amt = a_trial[s] * frac
            if amt < 1e-18:
                continue
            a_trial[s] -= amt
            a_trial[d] += amt
        
        a_trial = _normalize(a_trial, c)
        M_trial = _fast_autocorr(a_trial).max()
        M_curr = _fast_autocorr(a).max()
        
        delta = M_trial - M_curr
        if delta < 0 or rng.random() < np.exp(-delta / T):
            a = a_trial
            if M_trial < best_m:
                best_m = M_trial
                best_a = a.copy()
        
        T *= cooling
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    SA ({elapsed:.1f}s): score={2.0*n*best_m/(c*c):.6f}")
    
    return best_a, best_m

def _topk_subgrad(a, ac, top_k=5, beta=3.0):
    """Compute subgradient using top-k peaks with power weighting."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 0)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        i_start = max(0, int(k) - n + 1)
        i_end = min(n, int(k) + 1)
        idx = np.arange(i_start, i_end)
        j = int(k) - idx
        grad[idx] += 2.0 * wi * a[j]
    return grad

def _mirror_descent_topk(a0, time_limit, top_k=5, beta=3.0, verbose=True):
    """Mirror descent with top-k power-weighted subgradient."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = _normalize(a0, c)
    
    best_a = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.monotonic()
    
    gamma = 2.0
    avg_a = a.copy()
    n_avg = 1
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = _fast_autocorr(a)
        g = _topk_subgrad(a, ac, top_k=top_k, beta=beta)
        
        step = gamma / np.sqrt(t + 1.0)
        
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - step * g
        max_log = log_a_new.max()
        a_new = np.exp(log_a_new - max_log)
        a_new = _normalize(np.maximum(a_new, 0.0), c)
        a = a_new
        
        M_new = _fast_autocorr(a).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a.copy()
        
        avg_a += a
        n_avg += 1
        if n_avg % 100 == 0:
            avg_check = _normalize(avg_a / n_avg, c)
            M_avg = _fast_autocorr(avg_check).max()
            if M_avg < best_m:
                best_m = M_avg
                best_a = avg_check.copy()
        
        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_m / (c * c)
            print(f"    MD-topk t={t} ({elapsed:.1f}s): score={score:.6f}")
    
    return best_a, best_m

def _proj_simplex(v, s):
    n = len(v)
    if n == 0:

    u = np.sort(v)[::-1]

    rho = np.arange(1, n + 1)

    if not np.any(cond):

    rho_val = u[np.max(np.where(cond)[0])]

def _build_conv_matrix(a):
    n = len(a)
    nnz = n * n
    rows = np.zeros(nnz, dtype=np.int32)
    cols = np.zeros(nnz, dtype=np.int32)
    vals = np.zeros(nnz, dtype=np.float64)
    for j in range(n):
        rows[j*n:(j+1)*n] = np.arange(j, j + n)
        cols[j*n:(j+1)*n] = j
        vals[j*n:(j+1)*n] = a
    return sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

        b = _lp_best_response(y, M_curr, solver, T_cache)
        if b is None:
            stale += 1
            if stale > 50:
                break
            continue
        stale = 0
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = _normalize(b, c)
        if t == 1:
            y_new = g.copy()
        else:
            beta_mom = (t - 0.5) / (t + 1.5)
            y_extrap = y + beta_mom * (y - x_prev)
            y_extrap = np.maximum(y_extrap, 0.0)
            alpha = 2.0 / (t + 2.0)
            y_new = (1.0 - alpha) * y_extrap + alpha * g
            y_new = _normalize(y_new, c)
        x_prev = y
        y = y_new
        avg_y += y
        n_avg += 1
        M_y = _fast_autocorr(y).max()
        if M_y < best_m:
            best_m = M_y
            best_a = y.copy()
        if t % 5 == 0:
            avg_check = _normalize(avg_y / n_avg, c)
            M_avg = _fast_autocorr(avg_check).max()
            if M_avg < best_avg_m:
                best_avg_m = M_avg
                best_avg_a = avg_check.copy()
            if M_avg < best_m:
                best_m = M_avg
                best_a = avg_check.copy()
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_m / (c * c)
            print(f"    LP t={t} ({elapsed:.1f}s): score={score:.6f}, M={M_curr:.6f}")
    if best_avg_m < best_m:
        best_m = best_avg_m
        best_a = best_avg_a
    return best_a, best_m

def _interpolate(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def _anneal(a0, time_limit, verbose=False):
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = _normalize(a0, c)
    best_a = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(4242)
    T = 0.005
    T_min = 1e-7
    cooling = 0.9995

    while time.monotonic() - t0 < time_limit and T > T_min:
        n_move = rng.integers(1, 4)
        srcs = rng.choice(n, size=n_move, replace=False)
        dsts = rng.choice(n, size=n_move, replace=False)
        a_trial = a.copy()
        for s_i, d_i in zip(srcs, dsts):
            if s_i == d_i:
                continue
            frac = rng.uniform(0.01, 0.08)
            amt = a_trial[s_i] * frac
            if amt < 1e-18:
                continue
            a_trial[s_i] -= amt
            a_trial[d_i] += amt
        a_trial = _normalize(a_trial, c)
        M_trial = _fast_autocorr(a_trial).max()
        M_curr = _fast_autocorr(a).max()
        delta = M_trial - M_curr
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            a = a_trial
            if M_trial < best_m:
                best_m = M_trial
                best_a = a.copy()
        T *= cooling
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    SA ({elapsed:.1f}s): score={2.0*n*best_m/(c*c):.6f}")
    return best_a, best_m

def _coord_polish(a0, time_limit, verbose=False):
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = _normalize(a0, c)
    best_a = a.copy()
    best_m = _fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    stall = 0

    while time.monotonic() - t0 < time_limit and stall < 150:
        ac = _fast_autocorr(a)
        k_max = int(np.argmax(ac))
        i_start = max(0, k_max - n + 1)
        i_end = min(n, k_max + 1)
        idx = np.arange(i_start, i_end)
        contrib = a[idx] * a[k_max - idx]
        if len(contrib) == 0 or contrib.max() < 1e-15:
            stall += 5
            continue
        src = int(idx[np.argmax(contrib)])
        improved = False
        for attempt in range(15):
            dst = int(rng.integers(n))
            if dst == src:
                continue
            for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = _normalize(a_trial, c)
                M_trial = _fast_autocorr(a_trial).max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        if improved:
            stall = 0
        else:
            stall += 1
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > c * 1e-8:
                frac = rng.uniform(0.01, 0.05)
                amt = min(a[i] * frac, a[i] * 0.5)
                a[i] -= amt
                a[j] += amt
                a = _normalize(a, c)
                M_new = _fast_autocorr(a).max()
                if M_new < best_m:
                    best_m = M_new
                    best_a = a.copy()
                    stall = 0
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    Coord ({elapsed:.1f}s): score={2.0*n*best_m/(c*c):.6f}")
    return best_a, best_m

def propose_candidate(budget_s=1800.0):
    rng = np.random.default_rng(42)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)
    solver = _get_lp_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = _evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent / parent.sum()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = _evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = _evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n0 = len(best_seq)

    # Phase 1: Quick MD top-k at parent length
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: MD-topk n={n0} ===")
    t1 = min(remaining * 0.12, 180.0)
    if t1 > 10:
        c0 = np.sqrt(2.0 * n0)
        a_norm = best_seq * c0 / best_seq.sum()
        a_out, _ = _md_topk(a_norm, t1, top_k=5, beta=3.0, gamma=2.0, verbose=True)
        try_update(a_out, f"md_topk_n{n0}")
        if best_score < 1.504:
            a_out2, _ = _md_topk(a_norm, t1 * 0.5, top_k=3, beta=2.0, gamma=3.0, verbose=False)
            try_update(a_out2, f"md_topk3_n{n0}")

    # Phase 2: LP Nesterov at parent length
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP Nesterov n={n0} ===")
    t2 = min(remaining * 0.25, 420.0)
    if t2 > 30:
        c0 = np.sqrt(2.0 * n0)
        a_norm = best_seq * c0 / best_seq.sum()
        a_out, _ = _nesterov_lp(a_norm, t2, solver, verbose=True)
        try_update(a_out, f"lp_n{n0}")

    # Phase 3: Smooth continuation at parent length
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Smooth continuation n={n0} ===")
    t3 = min(remaining * 0.12, 200.0)
    if t3 > 15:
        c0 = np.sqrt(2.0 * n0)
        a_norm = best_seq * c0 / best_seq.sum()
        a_out, _ = _md_smooth(a_norm, t3, beta_start=3.0, beta_end=60.0, verbose=True)
        try_update(a_out, f"smooth_n{n0}")

    # Phase 4: Multi-length exploration
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Multi-length exploration ===")
    lengths = [1200, 2000, 2400, 3200, 4000]
    lengths = [L for L in lengths if L != n0]
    per_length = min(remaining * 0.25 / max(len(lengths), 1), 120.0)
    for L in lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 30:
            break
        t_alloc = min(per_length, remaining * 0.8)
        if t_alloc < 15:
            continue
        print(f"  Exploring n={L} (t={t_alloc:.0f}s)")
        a_long = _interpolate(best_seq, L)
        c_L = np.sqrt(2.0 * L)
        a_norm = a_long * c_L / a_long.sum()
        # Short MD first
        t_md = t_alloc * 0.4
        a_out, _ = _md_topk(a_norm, t_md, top_k=5, beta=2.5, gamma=2.0, verbose=False)
        try_update(a_out, f"md_n{L}")
        # Then short LP if time remains
        remaining_sub = time_budget - (time.monotonic() - t_global)
        t_lp = min(remaining_sub * 0.5, t_alloc * 0.5)
        if t_lp > 20:
            a_curr = best_seq if len(best_seq) == L else a_out / a_out.sum()
            c_L2 = np.sqrt(2.0 * L)
            a_lp = a_curr * c_L2 / a_curr.sum()
            a_out2, _ = _nesterov_lp(a_lp, t_lp, solver, verbose=False, max_iters=200)
            try_update(a_out2, f"lp_n{L}")

    # Phase 5: Intensive LP at best length
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Intensive LP n={n_best} ===")
    t5 = min(remaining * 0.25, 350.0)
    if t5 > 30:
        c_best = np.sqrt(2.0 * n_best)
        a_norm = best_seq * c_best / best_seq.sum()
        a_out, _ = _nesterov_lp(a_norm, t5, solver, verbose=True, max_iters=5000)
        try_update(a_out, f"lp_final_n{n_best}")

    # Phase 6: Simulated annealing
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Annealing n={n_best} ===")
    t6 = min(remaining * 0.10, 120.0)
    if t6 > 10:
        c_best = np.sqrt(2.0 * n_best)
        a_norm = best_seq * c_best / best_seq.sum()
        a_out, _ = _anneal(a_norm, t6, verbose=True)
        try_update(a_out, f"anneal_n{n_best}")

    # Phase 7: Coordinate polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Coordinate polish n={n_best} ===")
    t7 = min(remaining * 0.20, 200.0)
    if t7 > 10:
        c_best = np.sqrt(2.0 * n_best)
        a_norm = best_seq * c_best / best_seq.sum()
        a_out, _ = _coord_polish(a_norm, t7, verbose=True)
        try_update(a_out, f"coord_n{n_best}")

    # Phase 8: Final MD with different params
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final MD variants n={n_best} ===")
    t8 = min(remaining * 0.30, 200.0)
    if t8 > 10:
        c_best = np.sqrt(2.0 * n_best)
        a_norm = best_seq * c_best / best_seq.sum()
        # Variant 1: fewer top-k, higher beta
        t8a = t8 * 0.5
        a_out, _ = _md_topk(a_norm, t8a, top_k=3, beta=4.0, gamma=2.5, verbose=False)
        try_update(a_out, f"md3_n{n_best}")
        # Variant 2: more top-k, lower beta
        t8b = min(t8 * 0.5, time_budget - (time.monotonic() - t_global))
        if t8b > 5:
            a_out2, _ = _md_topk(a_norm, t8b, top_k=8, beta=1.5, gamma=1.5, verbose=False)
            try_update(a_out2, f"md8_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = _evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    return [float(x) for x in best_seq]
