            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    last_improve = time.monotonic()

        if time.monotonic() - last_improve > 45.0:
            stall_count += 1
            if stall_count > 8:
                if verbose:
                    print(f"    NLP Stalling at iter={n_iters}, best={best_score:.8f}")
                break
        else:
            stall_count = 0

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): max_conv={best_score:.8f}")

    return best_a, best_score


def lse_objective_and_grad(a_flat, n, T_temp):
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(a_flat)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros_like(a_flat)
    shifted = T_temp * (b - max_b)
    shifted = np.minimum(shifted, 500.0)
    exp_vals = np.exp(shifted)
    exp_sum = exp_vals.sum()
    if exp_sum < 1e-300:
        return 1e10, np.zeros_like(a_flat)
    w = exp_vals / exp_sum
    lse = max_b + np.log(exp_sum) / T_temp
    obj = 2.0 * n * lse / (s * s)

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    w_padded = np.zeros(fft_size)
    w_padded[:size] = w
    A_fft = rfft(a_padded, fft_size)
    W_fft = rfft(w_padded, fft_size)
    cross = irfft(A_fft * W_fft, fft_size)[:n]
    cross2 = irfft(A_fft * np.conj(W_fft), fft_size)[:n]
    conv_grad = 4.0 * (cross + cross2 - a * (w.sum() * 2.0))
    conv_grad = 4.0 * cross
    grad = conv_grad / (s * s) - 2.0 * obj * a / s
    return obj, grad.ravel()


def lse_scipy_optimize(a0, time_limit, T_start=50.0, T_end=1.0, n_stages=6):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        stage_time = remaining / (n_stages - si)
        max_stage = min(stage_time, 180.0)

        x0 = a.copy()

        def obj_func(x, n=n, T=T_temp_holder[0]):
            val, _ = lse_objective_and_grad(x, n, T)
            return val

        def grad_func(x, n=n, T=T_temp_holder[0]):
            _, g = lse_objective_and_grad(x, n, T)
            return g

        T_temp_holder[0] = T
        try:
            res = scipy_minimize(obj_func, x0, method='L-BFGS-B',
                                  jac=grad_func,
                                  options={'maxiter': 300, 'maxfun': 800})
            if res.x is not None:
                a_candidate = np.maximum(res.x, 0.0)
                s_c = a_candidate.sum()
                if s_c > 1e-10:
                    a_candidate = a_candidate * norm / s_c
                    score_c = fast_autocorr(a_candidate).max()
                    if score_c < best_score:
                        best_score = score_c
                        best_a = a_candidate.copy()
                        a = a_candidate
        except Exception:
            pass

    return best_a, best_score


T_temp_holder = [0.0]


def soft_constraint_descent(a0, M_target, time_limit, n_inner=2000):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    lr = 0.5
    best_violation = fast_autocorr(a).max() - M_target
    no_improve = 0

    for it in range(n_inner):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        violations = np.maximum(b - M_target, 0.0)
        violation_sum = violations.sum()

        if violation_sum < 1e-10:
            break

        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        a_padded = np.zeros(fft_size)
        a_padded[:n] = a
        v_padded = np.zeros(fft_size)
        v_padded[:size] = violations
        A_fft = rfft(a_padded, fft_size)
        V_fft = rfft(v_padded, fft_size)
        cross = irfft(A_fft * V_fft, fft_size)[:n]
        grad = 4.0 * cross
        grad -= 2.0 * np.linalg.norm(grad) * a / np.linalg.norm(a) * 0.01

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        a_new = a - lr * grad / grad_norm
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        score_new = fast_autocorr(a_new).max()
        violation_new = score_new - M_target

        if violation_new < best_violation:
            a = a_new
            best_violation = violation_new
            if score_new < best_score:
                best_score = score_new
                best_a = a_new.copy()
            no_improve = 0
            lr = min(lr * 1.05, 5.0)
        else:
            no_improve += 1
            lr = max(lr * 0.8, 0.01)
            if no_improve > 50:
                break

    return best_a, best_score


def bisection_soft_constraint(a0, time_limit):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    current_peak = fast_autocorr(a).max()
    best_a = a.copy()
    best_score = current_peak
    t0 = time.monotonic()

    M_low = current_peak * 0.95
    M_high = current_peak

    for bisect_iter in range(20):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit * 0.85:
            break
        remaining = time_limit * 0.85 - elapsed
        if remaining < 10:
            break

        M_mid = (M_low + M_high) / 2.0
        stage_time = min(remaining / max(1, 20 - bisect_iter), 60.0)

        a_candidate, score_candidate = soft_constraint_descent(a, M_mid, stage_time)
        peak_candidate = fast_autocorr(a_candidate).max()

        if score_candidate < best_score:
            best_score = score_candidate
            best_a = a_candidate.copy()
            a = a_candidate

        if peak_candidate <= M_mid * 1.01:
            M_high = M_mid
        else:
            M_low = M_mid

        elapsed = time.monotonic() - t0
        if elapsed > time_limit * 0.85:
            break
        print(f"    Bisection iter={bisect_iter}: M_mid={M_mid:.8f}, peak={peak_candidate:.8f} ({elapsed:.0f}s)")

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, seed=42):
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        n_peaks = min(12, 2 * n - 1)
        top_peaks = np.argsort(b)[::-1][:n_peaks]
        peak_vals = b[top_peaks]

        improved = False
        for pk in range(min(4, len(top_peaks))):
            k_peak = top_peaks[pk]
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k_peak - idx]
            top_contrib = idx[np.argsort(contrib)[::-1][:8]]

            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:5]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            for src in top_contrib:
                if a[src] < 1e-12:
                    continue
                candidates = np.where(~peak_mask)[0]
                if len(candidates) == 0:
                    candidates = np.where(np.arange(n) != src)[0]
                if len(candidates) == 0:
                    continue
                candidates_sorted = candidates[np.argsort(-a[candidates])][:max(1, len(candidates) // 4)]

                for dst in candidates_sorted[:4]:
                    for frac in [0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-10:
                            continue
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-13:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall >= 5 and stall % 5 == 0:
                for _ in range(50):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i == j or a[i] < 1e-12:
                        continue
                    frac = rng.uniform(0.01, 0.15)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                        break
            if stall > 300:
                break

    return best_a, best_score


def spectral_flatten(a0, time_limit, n_iters=500):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        A_fft = rfft(a, fft_size)
        mag = np.abs(A_fft)
        phase = np.angle(A_fft)
        target_mag = np.ones_like(mag)
        target_mag[0] = mag[0]
        target_mag = target_mag / target_mag.max()

        blend = 0.1
        new_mag = (1 - blend) * mag + blend * mag.max()
        new_fft = new_mag * np.exp(1j * phase)
        a_new = irfft(new_fft, fft_size, n=fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new * norm / s_new

        score_new = fast_autocorr(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 1: LP-Nesterov deep optimization at base length
    n_base = len(best_seq)

    a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
    a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
    do_try_update(a_out, f"lp_deep_n{n_base}")

    # Phase 2: Soft-constraint bisection
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Soft-constraint bisection n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = bisection_soft_constraint(a_norm, t2)
        do_try_update(a_out, f"soft_bisect_n{n_best}")

    # Phase 3: LSE + L-BFGS
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)

    remaining = time_budget - (time.monotonic() - t_global)

    return [float(x) for x in best_seq]
