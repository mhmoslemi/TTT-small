def fast_conv(a):
    """Fast linear convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def objective_smoothed(a, tau=50.0):
    """Compute smoothed objective using log-sum-exp."""
    c = fast_conv(a)
    c_max = c.max()
    # Numerically stable log-sum-exp
    shifted = tau * (c - c_max)
    log_sum = np.log(np.sum(np.exp(shifted))) + tau * c_max
    return log_sum / tau

def gradient_smoothed(a, tau=50.0):

    # For each k, a[k-i] is non-zero when 0 <= k-i < n, i.e., i <= k < i+n
    # So g[i] = 2 * sum_{k=max(0,i)}^{min(2n-2, i+n-1)} w[k] * a[k-i]
    
    # This is equivalent to: for each i, take the slice w[i:i+n] (padded) and dot with a.
    # We can do this efficiently with FFT.

    # Hmm, that gives sum_j w[j]*a[j-i] which is what we want!

def objective_exact(a):
    """Compute the exact objective value."""
    c = fast_conv(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    return 2.0 * len(a) * c.max() / (s * s)

def softmax(x):
    """Stable softmax."""
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()

def adam_softmax_optimize(a0, time_limit, tau_start=100.0, tau_end=500.0, lr=0.01, verbose=True):
    """
    Optimize using Adam with softmax reparameterization and log-sum-exp smoothing.
    a = softmax(x) * S where S is a fixed scale (we optimize over x).
    The objective is scale-invariant so we can just use a = softmax(x).
    """
    n = len(a0)
    # Initialize x from a0
    a_init = np.maximum(a0, 1e-10)
    a_init = a_init / a_init.sum()
    x = np.log(a_init)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t = 0
    
    best_a = a_init.copy()
    best_obj = objective_exact(best_a)
    t0 = time.monotonic()
    
    # Learning rate schedule: start larger, decay
    base_lr = lr
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive tau: start small (smooth), increase (sharp)
        progress = min(1.0, t / 5000)
        tau = tau_start + (tau_end - tau_start) * progress
        
        # Current a
        a = softmax(x)
        
        # Compute gradient of smoothed objective w.r.t. x
        # d/dx_i [f(a)] = sum_j (da_j/dx_i) * df/da_j
        # da_j/dx_i = a_j * (δ(j,i) - a_i)  (softmax Jacobian)
        # So d/dx_i f = a_i * (df/da_i - sum_j a_j * df/da_j)
        
        g_a = gradient_smoothed(a, tau)  # df/da
        # Project gradient: g_x[i] = a[i] * (g_a[i] - sum(j) a[j]*g_a[j])
        g_mean = np.dot(a, g_a)
        g_x = a * (g_a - g_mean)
        
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * g_x
        v = beta2 * v + (1 - beta2) * g_x ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr_t = base_lr * (1.0 - 0.5 * progress)  # slight decay
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        # Evaluate
        a_new = softmax(x)
        obj = objective_exact(a_new)
        
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam t={t} ({elapsed:.1f}s): obj={best_obj:.6f}, tau={tau:.0f}")
    
    return best_a, best_obj

def subgradient_optimize(a0, time_limit, verbose=True):
    """
    Projected gradient descent with exact subgradient and Nesterov momentum.
    """
    n = len(a0)
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = objective_exact(best_a)
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_conv(y)
        M = c.max()
        
        # Find active lags
        active = np.where(c >= M - 1e-10 * max(1, M))[0]
        # Limit to top 20 for stability
        if len(active) > 20:
            top_idx = np.argsort(c)[::-1][:20]
            active = top_idx
        
        # Subgradient: g[i] = sum_{k in active} 2*a[k-i] (with a out of range = 0)
        g = np.zeros(n)
        for k in active:
            k = int(k)
            # g[i] += 2*a[k-i] for valid i where 0 <= k-i < n
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j = k - np.arange(i_start, i_end + 1)
                g[i_start:i_end+1] += 2.0 * a[j]
        
        # Step size
        alpha = 2.0 / (t + 2.0)
        
        # Update y
        y_new = y - alpha * g
        y_new = np.maximum(y_new, 0.0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s
        
        # Nesterov momentum for x
        if t == 1:
            x_new = y_new.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / s_x
        
        y = y_new
        x = x_new
        
        obj_y = objective_exact(y)
        if obj_y < best_obj:
            best_obj = obj_y
            best_a = y.copy()
        
        obj_x = objective_exact(x)
        if obj_x < best_obj:
            best_obj = obj_x
            best_a = x.copy()
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG t={t} ({elapsed:.1f}s): obj={best_obj:.6f}")
    
    return best_a, best_obj

def mirror_descent_v2(a0, time_limit, top_k=10, beta=2.0, verbose=True):
    """Improved mirror descent with running average and adaptive parameters."""
    n = len(a0)
    a = np.maximum(a0, 1e-15)

    for t in range(1, 2000000):

            break
        
        c = fast_conv(a)
        M = c.max()
        
        # Top-k weighted subgradient
        k_max = min(top_k, len(c))
        top_idx = np.argsort(c)[::-1][:k_max]
        top_vals = np.maximum(c[top_idx], 0)
        w = top_vals ** beta
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / w_sum
        
        g = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j = k - np.arange(i_start, i_end + 1)
                g[i_start:i_end+1] += 2.0 * wi * a[j]
        
        # Mirror descent step
        step = gamma / np.sqrt(t)
        log_a = np.log(a)
        log_a_new = log_a - step * g
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new
        
        # Update best
        obj = objective_exact(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        # Running average
        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            obj_avg = objective_exact(avg_check)
            if obj_avg < best_obj:
                best_obj = obj_avg
                best_a = avg_check.copy()
        
        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD2 t={t} ({elapsed:.1f}s): obj={best_obj:.6f}")
    
    return best_a, best_obj

def coordinate_descent(a0, time_limit, verbose=True):
    """Coordinate descent: try moving mass between pairs of positions."""
    n = len(a0)
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = objective_exact(best_a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        M = c.max()
        
        # Find top contributing positions to the peak
        peak_k = int(np.argmax(c))
        i_start = max(0, peak_k - n + 1)
        i_end = min(n - 1, peak_k)
        idx = np.arange(i_start, i_end + 1)
        contrib = a[idx] * a[peak_k - idx]
        
        # Source: position with highest contribution to peak
        src = idx[np.argmax(contrib)]
        
        # Destination: position that minimizes impact on top peaks
        top_peaks = np.argsort(c)[::-1][:5]
        impact = np.zeros(n)
        for k in top_peaks:
            k = int(k)
            i_s = max(0, k - n + 1)
            i_e = min(n - 1, k)
            ii = np.arange(i_s, i_e + 1)
            impact[ii] += a[k - ii]
        
        # Exclude positions that are high-impact
        candidate_mask = impact < np.percentile(impact, 30)
        if ~candidate_mask[src]:
            candidate_mask[src] = False
        candidates = np.where(candidate_mask)[0]
        
        if len(candidates) == 0:
            candidates = np.where(np.arange(n) != src)[0]
        
        # Try several random destinations
        tries = min(20, len(candidates))
        dst_indices = rng.choice(candidates, size=tries, replace=False)
        
        improved = False
        for dst in dst_indices:
            if int(dst) == src:
                continue
            for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                obj_trial = objective_exact(a_trial)
                if obj_trial < best_obj - 1e-12:
                    a = a_trial
                    best_obj = obj_trial
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation to escape local minimum
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0)
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                    obj = objective_exact(a)
                    if obj < best_obj:
                        best_obj = obj
                        best_a = a.copy()
                        stall = 0
                    else:
                        # Revert
                        a[i] += amt
                        a[j] -= amt
                        a = a / a.sum()
        
        if verbose and (stall == 0 or stall % 50 == 0):
            elapsed = time.monotonic() - t0
            print(f"    CD stall={stall} ({elapsed:.1f}s): obj={best_obj:.6f}")
    
    return best_a, best_obj

def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def generate_initializations(n, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.linspace(0, 1, n)
    
    # Sine-squared
    f = np.sin(np.pi * x) ** 2
    inits.append(f / f.sum())
    
    # Cosine window
    f = np.cos(np.pi * (x - 0.5)) ** 2
    inits.append(f / f.sum())
    
    # Gaussian
    f = np.exp(-20 * (x - 0.5) ** 2)
    inits.append(f / f.sum())
    
    # Bimodal
    f = np.exp(-50 * (x - 0.25) ** 2) + np.exp(-50 * (x - 0.75) ** 2)
    inits.append(f / f.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Random smooth
    for _ in range(3):
        freqs = rng.integers(1, 20, size=10)
        coeffs = rng.normal(0, 1, size=10)
        phases = rng.uniform(0, 2*np.pi, size=10)
        f = np.ones(n)
        for freq, coeff, phase in zip(freqs, coeffs, phases):
            f += coeff * np.sin(2 * np.pi * freq * x + phase)
        f = np.maximum(f, 0.01)
        inits.append(f / f.sum())
    
    # Triangular
    f = np.abs(1 - 2 * x)
    inits.append(f / f.sum())
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth L2/L4 objective (sum of squared/fourth powers of convolution)
   with FFT-based gradient computation
2. Adam optimizer with softmax reparameterization (handles a>=0, sum=1)
3. FISTA ( accelerated proximal gradient) on the simplex
4. Coordinate descent polish targeting peak lags
5. Multi-length, multi-initialization search strategy

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_conv(a):
    """Fast linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def gradient_sum_sq(a):
    """
    Gradient of F(a) = sum_k c[k]^2 where c = conv(a,a).
    g[i] = 4 * sum_k c[k] * a[k-i] = 4 * conv(c, a_rev)[i+n-1]
    """
    n = len(a)
    c = fast_conv(a)
    a_rev = a[::-1]
    total_size = (2 * n - 1) + n - 1  # 3n-2
    fft_size = 1 << (total_size - 1).bit_length()
    C = rfft(c, fft_size)
    AR = rfft(a_rev, fft_size)
    conv_ca = irfft(C * AR, fft_size)
    return 4.0 * conv_ca[n - 1: 2 * n - 1]


def gradient_sum_p(a, p):
    """
    Gradient of F(a) = sum_k c[k]^p where c = conv(a,a).
    g[i] = 2p * sum_k c[k]^(p-1) * a[k-i] = 2p * conv(c^(p-1), a_rev)[i+n-1]
    """
    n = len(a)
    c = fast_conv(a)
    cp = c ** (p - 1)
    a_rev = a[::-1]
    total_size = (2 * n - 1) + n - 1
    fft_size = 1 << (total_size - 1).bit_length()
    CP = rfft(cp, fft_size)
    AR = rfft(a_rev, fft_size)
    conv_ca = irfft(CP * AR, fft_size)
    return 2.0 * p * conv_ca[n - 1: 2 * n - 1]


def objective_value(a):
    """Compute the exact objective: 2n * max(c) / sum(a)^2."""
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    c = fast_conv(a)
    return 2.0 * len(a) * c.max() / (s * s)


def softmax(x):
    """Numerically stable softmax."""
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()


def proj_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) + s - cssv > 0
    if not np.any(cond):
        return np.full(n, s / n)
    rho = u[np.argmax(cond)]
    return np.maximum(v - rho, 0.0)


def adam_softmax_optimize(a0, time_limit, objective='l2', p=4,
                          lr=0.05, verbose=True):
    """
    Adam optimizer with softmax reparameterization.
    Optimizes F(a) = sum(c[k]^p) with a = softmax(x).
    """
    n = len(a0)
    a_init = np.maximum(a0, 1e-10)
    a_init = a_init / a_init.sum()
    x = np.log(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t = 0

    best_a = a_init.copy()
    best_obj = objective_value(best_a)
    t0 = time.monotonic()

    for it in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        t = it

        a = softmax(x)

        # Compute gradient of F w.r.t. a
        if objective == 'l2':
            g_a = gradient_sum_sq(a)
        else:
            g_a = gradient_sum_p(a, p)

        # Transform to x-space: g_x[i] = a[i] * (g_a[i] - dot(a, g_a))
        g_mean = np.dot(a, g_a)
        g_x = a * (g_a - g_mean)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * g_x ** 2
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        # Learning rate decay
        lr_t = lr * (1.0 + t) ** (-0.5)
        x = x - lr_t * m_hat / (np.sqrt(v_hat) + eps)

        # Evaluate every 10 steps (save compute)
        if it % 10 == 0:
            a_new = softmax(x)
            obj = objective_value(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Adam t={it} ({elapsed:.1f}s): obj={best_obj:.6f}")

def fista_optimize(a0, time_limit, p=4, verbose=True):
    """
    FISTA (accelerated proximal gradient) minimizing sum(c[k]^p)
    subject to a >= 0, sum(a) = 1.
    """
    n = len(a0)
    a = np.maximum(a0, 0.0)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = objective_value(best_a)
    t0 = time.monotonic()

    y = a.copy()
    z = a.copy()
    tk = 1.0

    # Estimate Lipschitz constant
    c = fast_conv(y)
    g = gradient_sum_p(y, p) if p != 2 else gradient_sum_sq(y)
    L = max(1.0, np.linalg.norm(g, ord=np.inf) * n ** 0.5)

    for it in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_conv(y)
        g = gradient_sum_p(y, p) if p != 2 else gradient_sum_sq(y)

        # Proximal step
        step = 1.0 / L
        y_new = y - step * g
        y_new = proj_simplex(y_new, 1.0)

        # Check if we need to increase L
        if it % 100 == 0:
            L = max(L * 1.01, np.linalg.norm(g, ord=np.inf) * n ** 0.5)

        # FISTA update
        tk_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * tk * tk))
        z_new = y_new + ((tk - 1.0) / tk_new) * (y_new - y)
        z_new = np.maximum(z_new, 0.0)
        sz = z_new.sum()
        if sz < 1e-15:
            z_new = y_new.copy()
        else:
            z_new = z_new / sz

        y = y_new
        z = z_new
        tk = tk_new

        # Evaluate
        if it % 5 == 0:
            obj_y = objective_value(y)
            if obj_y < best_obj:
                best_obj = obj_y
                best_a = y.copy()
            obj_z = objective_value(z)
            if obj_z < best_obj:
                best_obj = obj_z
                best_a = z.copy()

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      FISTA t={it} ({elapsed:.1f}s): obj={best_obj:.6f}")

def subgradient_nesterov(a0, time_limit, verbose=True):
    """
    Nesterov-accelerated subgradient method on the exact max objective.
    """
    n = len(a0)
    a = np.maximum(a0, 0.0)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = objective_value(best_a)
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()

    for it in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_conv(y)
        M = c.max()

        # Active lags (top 10)
        top_idx = np.argsort(c)[::-1][:10]
        top_vals = np.maximum(c[top_idx], 0)
        w = top_vals / top_vals.sum()

        # Weighted subgradient
        g = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j = k - np.arange(i_start, i_end + 1)
                g[i_start:i_end + 1] += 2.0 * wi * a[j]

        alpha = 2.0 / (it + 2.0)

        # Update y
        y_new = y - alpha * g
        y_new = np.maximum(y_new, 0.0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s

        # Nesterov momentum
        if it == 1:
            x_new = y_new.copy()
        else:
            beta = (it - 1.0) / (it + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new

        if it % 5 == 0:
            obj_y = objective_value(y)
            if obj_y < best_obj:
                best_obj = obj_y
                best_a = y.copy()
            obj_x = objective_value(x)
            if obj_x < best_obj:
                best_obj = obj_x
                best_a = x.copy()

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SG-Nest t={it} ({elapsed:.1f}s): obj={best_obj:.6f}")

    return best_a, best_obj


def coordinate_descent(a0, time_limit, verbose=True):
    """
    Coordinate descent: move mass from high-contribution positions to low-impact ones.
    """
    n = len(a0)
    a = np.maximum(a0, 0.0)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = objective_value(best_a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    stall = 0
    max_stall = 300

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        M = c.max()
        peak_k = int(np.argmax(c))

        # Identify contributors to peak
        i_start = max(0, peak_k - n + 1)
        i_end = min(n - 1, peak_k)
        idx = np.arange(i_start, i_end + 1)
        contrib = a[idx] * a[peak_k - idx]
        src = idx[np.argmax(contrib)]

        # Compute impact of each position on top peaks
        top_peaks = np.argsort(c)[::-1][:8]
        impact = np.zeros(n)
        for k in top_peaks:
            k = int(k)
            i_s = max(0, k - n + 1)
            i_e = min(n - 1, k)
            ii = np.arange(i_s, i_e + 1)
            impact[ii] += 2.0 * a[k - ii]

        # Find low-impact destinations
        threshold = np.percentile(impact, 20)
        candidates = np.where(impact <= threshold)[0]
        candidates = candidates[candidates != src]

        if len(candidates) == 0:
            candidates = np.where(np.arange(n) != src)[0]

        improved = False
        n_tries = min(30, len(candidates))
        dst_pool = rng.choice(candidates, size=n_tries, replace=False)

        for dst in dst_pool:
            if time.monotonic() - t0 > time_limit:
                break
            dst = int(dst)
            for frac in [0.001, 0.003, 0.01, 0.03, 0.08]:
                amt = a[src] * frac
                if amt < 1e-14:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                obj_trial = objective_value(a_trial)
                if obj_trial < best_obj - 1e-12:
                    a = a_trial
                    best_obj = obj_trial
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.005, 0.05)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0)
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                    obj = objective_value(a)
                    if obj < best_obj:
                        best_obj = obj
                        best_a = a.copy()
                        stall = max(0, stall - 50)
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = a / a.sum()

        if verbose and (stall == 0 or stall % 100 == 0):
            elapsed = time.monotonic() - t0
            print(f"      CD stall={stall} ({elapsed:.1f}s): obj={best_obj:.6f}")

    return best_a, best_obj


def mirror_descent_opt(a0, time_limit, top_k=8, beta=2.0, verbose=True):
    """Mirror descent (exponential weights) with running average."""
    n = len(a0)
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = objective_value(best_a)
    t0 = time.monotonic()

    avg_a = a.copy()
    n_avg = 1
    gamma = 0.3

    for it in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_conv(a)
        M = c.max()

        # Top-k weighted subgradient
        k_max = min(top_k, len(c))
        top_idx = np.argsort(c)[::-1][:k_max]
        top_vals = np.maximum(c[top_idx], 0)
        w = top_vals ** beta
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(k_max) / k_max
        else:
            w = w / ws

        g = np.zeros(n)
        for wi, k in zip(w, top_idx):
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j = k - np.arange(i_start, i_end + 1)
                g[i_start:i_end + 1] += 2.0 * wi * a[j]

        # Mirror descent step
        step = gamma / np.sqrt(it)
        log_a = np.log(a)
        log_a_new = log_a - step * g
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new

        # Track best
        if it % 5 == 0:
            obj = objective_value(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

        # Running average
        avg_a += a
        n_avg += 1
        if n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            obj_avg = objective_value(avg_check)
            if obj_avg < best_obj:
                best_obj = obj_avg
                best_a = avg_check.copy()

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MD t={it} ({elapsed:.1f}s): obj={best_obj:.6f}")

    return best_a, best_obj


def interpolate_to_length(f, new_n):
    """Interpolate sequence to new length, preserving sum normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def generate_diverse_inits(n, rng):
    """Generate diverse initial sequences."""
    inits = []
    x = np.linspace(0, 1, n)
    safe = 1e-6

    # Sine-squared
    f = np.sin(np.pi * x) ** 2
    f = np.maximum(f, safe)
    inits.append(f / f.sum())

    # Cosine bell
    f = np.cos(np.pi * (x - 0.5)) ** 2
    f = np.maximum(f, safe)
    inits.append(f / f.sum())

    # Gaussian
    f = np.exp(-30 * (x - 0.5) ** 2) + safe
    inits.append(f / f.sum())

    # Bimodal
    f = np.exp(-80 * (x - 0.25) ** 2) + np.exp(-80 * (x - 0.75) ** 2) + safe
    inits.append(f / f.sum())

    # Trimodal
    f = (np.exp(-100 * (x - 0.2) ** 2) + np.exp(-100 * (x - 0.5) ** 2)
         + np.exp(-100 * (x - 0.8) ** 2)) + safe
    inits.append(f / f.sum())

    # Uniform
    inits.append(np.ones(n) / n)

    # Triangle
    f = np.abs(1.0 - 2.0 * x) + safe
    inits.append(f / f.sum())

    # Sine with phase
    f = np.sin(np.pi * x + np.pi / 4) ** 2 + safe
    inits.append(f / f.sum())

    # Random smooth (2 samples)
    for _ in range(2):
        n_fourier = 8
        freqs = rng.integers(1, 15, size=n_fourier)
        amps = rng.uniform(0.1, 1.0, size=n_fourier)
        phases = rng.uniform(0, 2 * np.pi, size=n_fourier)
        f = np.ones(n) * 0.5
        for fr, amp, ph in zip(freqs, amps, phases):
            f += amp * np.sin(2 * np.pi * fr * x + ph)
        f = np.maximum(f, safe)
        inits.append(f / f.sum())

    return inits


def propose_candidate(budget_s=1800.0):
    """
    Main search: multi-phase, multi-length, multi-init optimization.
    Combines smooth L4 gradient (Adam), FISTA, subgradient Nesterov,
    mirror descent, and coordinate descent.
    """
    rng = np.random.default_rng(2024)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            p_score = evaluate_sequence([float(xi) for xi in parent])
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent / parent.sum()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = np.maximum(best_seq, 1e-10)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_sequence([float(xi) for xi in best_seq])
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_sequence([float(xi) for xi in a])
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # ============ PHASE 1: Smooth L4 optimization at parent length ============
    n0 = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Adam L4 at n={n0} (remaining {remaining:.0f}s) ===")
    t1 = min(remaining * 0.20, 250.0)
    if t1 > 30:
        a_out, _ = adam_softmax_optimize(best_seq, t1, objective='lp', p=4,
                                         lr=0.08, verbose=True)
        try_update(a_out, f"adamL4_n{n0}")

    # ============ PHASE 2: FISTA L4 at parent length ============
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: FISTA L4 at n={n0} (remaining {remaining:.0f}s) ===")
    t2 = min(remaining * 0.20, 200.0)
    if t2 > 20:
        a_start = best_seq.copy()
        a_out, _ = fista_optimize(a_start, t2, p=4, verbose=True)
        try_update(a_out, f"fistaL4_n{n0}")

    # ============ PHASE 3: Subgradient Nesterov at parent length ============
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Subgrad Nesterov at n={n0} (remaining {remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 150.0)
    if t3 > 20:
        a_start = best_seq.copy()
        a_out, _ = subgradient_nesterov(a_start, t3, verbose=True)
        try_update(a_out, f"sgnest_n{n0}")

    # ============ PHASE 4: Mirror descent at parent length ============
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Mirror Descent at n={n0} (remaining {remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 150.0)
    if t4 > 20:
        a_start = best_seq.copy()
        a_out, _ = mirror_descent_opt(a_start, t4, top_k=10, beta=2.5, verbose=True)
        try_update(a_out, f"md_n{n0}")

    # ============ PHASE 5: Coordinate descent polish ============
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Coordinate Descent at n={n0} (remaining {remaining:.0f}s) ===")
    t5 = min(remaining * 0.20, 250.0)
    if t5 > 20:
        a_start = best_seq.copy()
        a_out, _ = coordinate_descent(a_start, t5, verbose=True)
        try_update(a_out, f"cd_n{n0}")

    # ============ PHASE 6: Try longer lengths ============
    for n_long in [2000, 2400]:
        if n_long == n0:
            continue
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        print(f"\n=== Phase 6: Long seq n={n_long} (remaining {remaining:.0f}s) ===")
        a_long = interpolate_to_length(best_seq, n_long)

        # Adam L4
        t6a = min(remaining * 0.30, 200.0)
        if t6a > 20:
            a_out, _ = adam_softmax_optimize(a_long, t6a, objective='lp', p=4,
                                             lr=0.06, verbose=True)
            try_update(a_out, f"adamL4_n{n_long}")

        # Subgradient
        remaining = time_budget - (time.monotonic() - t_global)
        t6b = min(remaining * 0.30, 150.0)
        if t6b > 20:
            a_start = best_seq.copy() if len(best_seq) == n_long else interpolate_to_length(best_seq, n_long)
            a_out, _ = subgradient_nesterov(a_start, t6b, verbose=True)
            try_update(a_out, f"sgnest_n{n_long}")

        # Coordinate polish
        remaining = time_budget - (time.monotonic() - t_global)
        t6c = min(remaining * 0.40, 200.0)
        if t6c > 20:
            a_start = best_seq.copy() if len(best_seq) == n_long else interpolate_to_length(best_seq, n_long)
            a_out, _ = coordinate_descent(a_start, t6c, verbose=True)
            try_update(a_out, f"cd_n{n_long}")

    # ============ PHASE 7: Diverse initializations at best length ============
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_best = len(best_seq)
        print(f"\n=== Phase 7: Diverse inits at n={n_best} (remaining {remaining:.0f}s) ===")
        inits = generate_diverse_inits(n_best, rng)
        t_per = max(15.0, (remaining - 20) / (len(inits) * 2))

        for idx, init in enumerate(inits):
            if time.monotonic() - t_global > time_budget - 20:
                break
            init_obj = objective_value(init)
            # Only bother if init is reasonably good
            if init_obj < best_score + 0.3:
                # Quick Adam
                a_out, _ = adam_softmax_optimize(init, min(t_per, remaining * 0.12),
                                                 objective='lp', p=4, lr=0.1,
                                                 verbose=False)
                try_update(a_out, f"div_adam_{idx}_n{n_best}")

                # Quick CD
                remaining = time_budget - (time.monotonic() - t_global)
                if remaining > 20:
                    a_out2, _ = coordinate_descent(a_out, min(t_per * 0.7, remaining * 0.4),
                                                   verbose=False)
                    try_update(a_out2, f"div_cd_{idx}_n{n_best}")

    # ============ PHASE 8: Final focused polish ============
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_best = len(best_seq)
        print(f"\n=== Phase 8: Final polish at n={n_best} (remaining {remaining:.0f}s) ===")

        # Adam with smaller lr for fine tuning
        t8a = min(remaining * 0.4, 150.0)
        if t8a > 15:
            a_out, _ = adam_softmax_optimize(best_seq, t8a, objective='lp', p=4,
                                             lr=0.01, verbose=True)
            try_update(a_out, f"final_adam_n{n_best}")

        # More coordinate descent
        remaining = time_budget - (time.monotonic() - t_global)
        t8b = min(remaining * 0.5, 200.0)
        if t8b > 15:
            a_out, _ = coordinate_descent(best_seq, t8b, verbose=True)
            try_update(a_out, f"final_cd_n{n_best}")

        # One more subgradient pass
        remaining = time_budget - (time.monotonic() - t_global)
        t8c = min(remaining * 0.5, 100.0)
        if t8c > 15:
            a_out, _ = subgradient_nesterov(best_seq, t8c, verbose=True)
            try_update(a_out, f"final_sg_n{n_best}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_sequence([float(xi) for xi in best_seq])
        if np.isfinite(fs):
            best_score = fs
        print(f"\n{'='*50}")
        print(f"FINAL RESULT: score={best_score:.6f}, n={len(best_seq)}")
        print(f"Total time: {time.monotonic() - t_global:.0f}s")
        print(f"{'='*50}")
    else:
        print("ERROR: No valid sequence found!")
        n = 1600
        x = np.linspace(0, 1, n)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = np.maximum(best_seq, 1e-10)
        best_seq = best_seq / best_seq.sum()

    return [float(xi) for xi in best_seq]
