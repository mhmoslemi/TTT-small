3. Novel direct gradient descent with adaptive multi-peak targeting
4. Periodic restarts from diverse perturbations to escape local minima
5. Peak shaving with multi-direction mass redistribution
Key innovation: exploits the simple gradient structure (gradient of autocorr peak
is a shifted copy of a) for extremely fast iterations, combined with periodic
LP-based global steps to guarantee progress.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None


def _get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return _global_solver
        except Exception:
            continue
    _global_solver = cp.CLARABEL
    print(f"  LP solver (fallback): CLARABEL")
    return _global_solver


def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def _proj_simplex(v, target_sum=1.0):
    n = len(v)
    if n <= 1:
        return np.array([target_sum])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rhs = cssv - target_sum
    rho = np.nonzero(u * np.arange(1, n + 1) > rhs)[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 1e-15:
        out = out * target_sum / s
    else:
        out = np.full(n, target_sum / n)
    return out


def _lse_grad(y, T):
    n = len(y)
    ac = _fast_autocorr(y)
    ac_max = ac.max()
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(y, fft_size)
    W = rfft(w, fft_size)
    cross = irfft(np.conj(A) * W, fft_size)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad


def _lse_fista(a0, T_schedule, iters_per_stage, time_limit, verbose=False):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    L_init = n / 50.0

    for stage_i, T in enumerate(T_schedule):
        L = max(n / (T * 0.8), L_init / (T * 10.0))
        t_fista = 1
        staging = a.copy()
        max_iters = int(iters_per_stage * (T_schedule[0] / T) ** 0.25)
        max_iters = max(max_iters, 3000)

        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_score

            grad = _lse_grad(y, T)
            step = 1.0 / L
            z = y - step * grad
            a_new = _proj_simplex(z, 1.0)

            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - staging)
                y_new = np.maximum(y_new, 0.0)
                s_y = y_new.sum()
                if s_y > 1e-15:
                    y_new = y_new / s_y
                else:
                    y_new = a_new.copy()

            staging = a_new.copy()
            a = a_new
            y = y_new
            t_fista += 1

            if it % 300 == 0:
                sc = _fast_autocorr(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

        sc = _fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE-FISTA stage={stage_i} T={T:.2f} ({elapsed:.0f}s): best={best_score:.8f}")

    sc = _fast_autocorr(a).max()
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def _build_conv_matrix(a, n):
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        js = max(0, j - n + 1)
        T[j: j + n - j, js: js + (j + n - j)] = a
    # Actually let's just build it properly
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        for m in range(2 * n - 1):
            i = m - j
            if 0 <= i < n:
                T[m, j] = a[i]
    return T


def _build_conv_matrix_fast(a, n):
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        start = 0
        end = n - j
        T[j: j + end, j] = a[:end]
        T[j: j + j + 1, j] = a[j:]
    return T


def _build_conv_matrix_v2(a, n):
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j: j + n - j, j] = a[:n - j]
        T[j: j + j + 1, j] = a[j:]
    return T


def _lp_best_response(a, M, n, solver):
    b_var = cp.Variable(n, nonneg=True)
    # Constraint: a^T B_m b <= M for m = 0,...,2n-2
    # (a*b)[m] = sum_j a[m-j]*b[j] for valid indices
    # Build the constraint matrix efficiently
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        # (a*b)[m] = sum_j a[m-j]*b[j], valid when 0<=m-j<n
        # For fixed j, m ranges from j to j+n-1
        T[j: j + n, j] = a
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def _lp_nesterov(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = _fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        b = _lp_best_response(y, M_curr, n, solver)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = _fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = _fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 10 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = _fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nesterov iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def _direct_gradient_descent(a0, time_limit, verbose=False):
    """Direct gradient descent on max autocorrelation with adaptive step size."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    step_size = 1.0 / n
    stall_count = 0

    for it in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = _fast_autocorr(a)
        peak_idx = int(np.argmax(ac))
        peak_val = ac[peak_idx]

        # Gradient of ac[peak_idx] w.r.t. a[j] = 2*a[peak_idx - j] if valid
        grad = np.zeros(n)
        for j in range(n):
            idx = peak_idx - j
            if 0 <= idx < n:
                grad[j] = 2.0 * a[idx]

        # Subgradient of max: use gradient at the peak
        # We want to decrease the peak, so move in negative gradient direction
        # But we need to stay on the simplex
        a_new = a - step_size * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        score_new = _fast_autocorr(a_new).max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
            # Decrease step size slightly on success
            step_size *= 0.995
        else:
            a = a_new  # still move, but don't update best
            stall_count += 1
            step_size *= 0.9

        if stall_count > 200:
            # Restart with perturbation
            rng = np.random.default_rng(it)
            a = best_a + rng.normal(0, 0.01, n) * norm
            a = np.maximum(a, 0.0)
            a = a * norm / a.sum()
            stall_count = 0
            step_size = 1.0 / n

        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GradDesc it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def _multi_peak_gradient_descent(a0, time_limit, verbose=False):
    """Gradient descent targeting multiple peaks simultaneously."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    step_size = 2.0 / n
    top_k = 5

    for it in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = _fast_autocorr(a)
        # Get top K peaks
        top_peaks = np.argsort(ac)[::-1][:top_k]
        peak_max = ac[top_peaks[0]]

        # Weight peaks by their relative height
        weights = np.zeros(len(ac))
        for k_idx in top_peaks:
            weights[k_idx] = ac[k_idx] / peak_max

        # Combined gradient: sum of weighted gradients at top peaks
        grad = np.zeros(n)
        for k_idx in top_peaks:
            w = weights[k_idx]
            for j in range(n):
                idx = k_idx - j
                if 0 <= idx < n:
                    grad[j] += w * 2.0 * a[idx]

        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Move in negative gradient direction with simplex projection
        a_trial = a - step_size * grad
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            continue
        a_trial = a_trial * norm / s_trial

        score = _fast_autocorr(a_trial).max()
        if score < best_score:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            step_size *= 1.002
        else:
            step_size *= 0.95
            a = a + step_size * 0.5 * grad  # small step opposite to undo
            a = np.maximum(a, 0.0)
            s_a = a.sum()
            if s_a > 1e-15:
                a = a * norm / s_a

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MultiPeak it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def _peak_shave(a, time_limit, seed=42):
    n = len(a)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = _fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:10]
        improved = False

        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2: je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]

            candidates_sorted = candidates[np.argsort(a[candidates])][:max(1, len(candidates) // 4)]
            if len(candidates_sorted) == 0:
                continue
            dst = candidates_sorted[rng.integers(len(candidates_sorted))]

            for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.05]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = _fast_autocorr(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-10:
                frac = rng.uniform(0.005, 0.05)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = _fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0

            if stall > 800:
                break

    return best_a, best_score


def _spectral_flat_init(n, rng):
    """Initialize with a sequence whose DFT magnitude is as flat as possible."""
    # Generate random phases
    phases = rng.uniform(0, 2 * np.pi, n)
    # Create a sequence with flat magnitude spectrum
    # a = IDFT of exp(i*phases) - but this gives complex values
    # Instead, use the real part and take absolute value
    a = np.cos(phases) + 0.5
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def _structured_init(n, rng):
    """Generate a structured initial sequence."""
    method = rng.integers(3)
    x = np.linspace(0, 1, n)
    if method == 0:
        # Sinusoidal with harmonics
        a = np.sin(np.pi * x) ** 1.2 + 0.3 * np.sin(3 * np.pi * x) ** 2
    elif method == 1:
        # Exponential decay from edges
        d = np.minimum(x, 1 - x) * 2  # 0 at edges, 1 at center
        a = np.exp(-2 * (1 - d) ** 2) + 0.5 * d
    else:
        # Random smooth
        k = rng.integers(3, 10)
        a = np.ones(n)
        for i in range(k):
            center = rng.uniform(0, 1)
            width = rng.uniform(0.05, 0.2)
            a += np.exp(-((x - center) / width) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def _interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def _evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search: multi-method optimization with restarts and diverse strategies."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 100.0, 1700.0)

    solver = _get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to use parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = _evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def _try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = _evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        n_base = 1200
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5 + 0.3 * np.sin(2 * np.pi * x) ** 2
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = _evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # === Phase 1: LSE-FISTA at base length ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.15, 200.0)
    if t1 > 20:
        print(f"\n=== Phase 1: LSE-FISTA n={n_base} (budget={t1:.0f}s) ===")
        T_sched = [30.0, 8.0, 2.0, 0.5, 0.15, 0.05]
        a_out, _ = _lse_fista(best_seq, T_sched, 10000, t1, verbose=True)
        _try_update(a_out, f"lse1_n{n_base}")

    # === Phase 2: Multi-peak gradient descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.10, 120.0)
    if t2 > 15:
        print(f"\n=== Phase 2: MultiPeak GradDesc n={n_best} (budget={t2:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = _multi_peak_gradient_descent(a_norm, t2, verbose=True)
        _try_update(a_out, f"grad_n{n_best}")

    # === Phase 3: LP + Nesterov ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.25, 400.0)
    if t3 > 30:
        print(f"\n=== Phase 3: LP-Nesterov n={n_best} (budget={t3:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = _lp_nesterov(a_norm, t3, solver, verbose=True)
        _try_update(a_out, f"lp1_n{n_best}")

    # === Phase 4: LSE-FISTA with lower temperatures ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 15:
        print(f"\n=== Phase 4: LSE-FISTA low-T n={n_best} (budget={t4:.0f}s) ===")
        T_sched = [2.0, 0.5, 0.15, 0.04, 0.01]
        a_out, _ = _lse_fista(best_seq, T_sched, 12000, t4, verbose=True)
        _try_update(a_out, f"lse2_n{n_best}")

    # === Phase 5: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.12, 100.0)
    if t5 > 10:
        print(f"\n=== Phase 5: Peak shave n={n_best} (budget={t5:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = _peak_shave(a_norm, t5, seed=999)
        _try_update(a_out, "peak_shave1")

    # === Phase 6: Try longer sequence ===
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq) + 400
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 30:
        print(f"\n=== Phase 6: LP-Nesterov n={n2} (budget={t6:.0f}s) ===")
        init_long = _interpolate_sequence(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = _lp_nesterov(a_norm, t6, solver, verbose=True)
        _try_update(a_out, f"lp2_n{n2}")

    # === Phase 7: LSE on best (possibly new length) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t7 = min(remaining * 0.10, 120.0)
    if t7 > 10:
        print(f"\n=== Phase 7: LSE-FISTA final n={n_best} (budget={t7:.0f}s) ===")
        T_sched = [1.0, 0.3, 0.08, 0.02]
        a_out, _ = _lse_fista(best_seq, T_sched, 15000, t7, verbose=True)
        _try_update(a_out, f"lse3_n{n_best}")

    # === Phase 8: Multi-peaked gradient descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t8 = min(remaining * 0.08, 80.0)
    if t8 > 10:
        print(f"\n=== Phase 8: MultiPeak n={n_best} (budget={t8:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = _multi_peak_gradient_descent(a_norm, t8, verbose=True)
        _try_update(a_out, f"grad2_n{n_best}")

    # === Phase 9: Another LP pass ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t9 = min(remaining * 0.15, 200.0)
    if t9 > 20:
        print(f"\n=== Phase 9: LP polish n={n_best} (budget={t9:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = _lp_nesterov(a_norm, t9, solver, verbose=True)
        _try_update(a_out, "lp_final")

    # === Phase 10: Final peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t10 = min(remaining * 0.25, 150.0)
    if t10 > 10:
        print(f"\n=== Phase 10: Final peak shave n={n_best} (budget={t10:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = _peak_shave(a_norm, t10, seed=77777)
        _try_update(a_out, "peak_shave_final")

    # === Phase 11: Final LSE with very low T ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t11 = min(remaining * 0.30, 200.0)
    if t11 > 15:
        print(f"\n=== Phase 11: LSE ultra-low-T n={n_best} (budget={t11:.0f}s) ===")
        T_sched = [0.3, 0.08, 0.02, 0.005]
        a_out, _ = _lse_fista(best_seq, T_sched, 20000, t11, verbose=True)
        _try_update(a_out, "lse_ultra")

    # Final verification
    if best_seq is not None:
        fs = _evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
