def _fast_autocorr(a):
    """Compute full convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def _proj_simplex(v, target_sum=1.0):
    """Project vector v onto {x: sum(x)=target_sum, x>=0}."""
    n = len(v)
    if n <= 1:
        return np.array([max(0, target_sum)])
    # Scale to work with sum=1, then rescale
    v1 = v / target_sum
    u = np.sort(v1)[::-1]

    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]

        return np.clip(v, 0, None)
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v1 - theta, 0.0)
    return out * target_sum

Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Sparse LSE-subgradient with adaptive temperature and backtracking line search
2. Multiple diverse initializations (parametric, smoothed random, parent)
3. Periodic LP best-response corrections via cvxpy
4. Final peak-directed local search
Key: Focuses computation on the few dominant autocorrelation peaks,
uses fast O(k*n) gradient computation, and explores multiple basins.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None


def _get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return _global_solver
        except Exception:
            continue
    _global_solver = cp.CLARABEL
    print(f"  LP solver (fallback): CLARABEL")
    return _global_solver


def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def _proj_simplex(v, target_sum=1.0):
    n = len(v)
    if n <= 1:
        return np.array([max(0.0, target_sum)])
    v_scale = v / target_sum
    u = np.sort(v_scale)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        out = np.maximum(v, 0.0)
        s = out.sum()
        if s > 1e-15:
            out = out * target_sum / s
        else:
            out = np.full(n, target_sum / n)
        return out
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    out = np.maximum(v_scale - theta, 0.0)
    return out * target_sum


def _sparse_lse_grad(a, w, top_k=20):
    """Compute sparse LSE gradient using only top-k autocorrelation positions."""
    n = len(a)
    grad = np.zeros(n)
    top_idx = np.argsort(w)[::-1][:top_k]
    for k in top_idx:
        wk = w[k]
        if wk < 1e-12:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        i_range = np.arange(js, je + 1)
        grad[i_range] += 2.0 * wk * a[k - i_range]
    return grad


def _lse_weights(ac, T):
    """Compute LSE softmax weights from autocorrelation."""
    ac_max = ac.max()
    shifted = np.clip((ac - ac_max) / T, -50.0, 50.0)
    exp_s = np.exp(shifted)
    return exp_s / exp_s.sum()


def _lse_subgrad_opt(a0, time_limit, T_start=10.0, T_end=0.05, decay_factor=0.95,
                     top_k=20, lp_every=200, solver=None, verbose=True):
    """
    LSE-subgradient with backtracking line search and periodic LP corrections.
    Optimizes a on the simplex (sum=1, a>=0) to minimize max(conv(a,a)).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_maxb = _fast_autocorr(a).max()
    t0 = time.monotonic()
    T = T_start
    it = 0
    stall_count = 0

    while time.monotonic() - t0 < time_limit:
        ac = _fast_autocorr(a)
        cur_max = ac.max()
        if cur_max < best_maxb:
            best_maxb = cur_max
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        w = _lse_weights(ac, T)
        grad = _sparse_lse_grad(a, w, top_k)

        # Backtracking line search
        step = 1.0 / (np.linalg.norm(grad, ord=np.inf) + 1e-10)
        step = min(step, 0.5)
        improved = False
        for _ in range(20):
            a_trial = a - step * grad
            a_trial = _proj_simplex(a_trial, 1.0)
            ac_trial = _fast_autocorr(a_trial)
            maxb_trial = ac_trial.max()
            if maxb_trial < cur_max * (1.0 + 1e-10):
                a = a_trial
                improved = True
                break
            step *= 0.5
        if not improved:
            a = _proj_simplex(a - step * grad, 1.0)

        it += 1
        T = max(T_end, T * decay_factor)

        # Periodic LP correction
        if solver is not None and it % lp_every == 0:
            M = _fast_autocorr(a).max()
            b = _lp_solve(a, M, solver)
            if b is not None:
                b_sum = b.sum()
                if b_sum > 1e-15:
                    b_norm = b / b_sum
                    # Blend
                    a_blend = 0.7 * a + 0.3 * b_norm
                    a_blend = np.maximum(a_blend, 0.0)
                    sb = a_blend.sum()
                    if sb > 1e-15:
                        a_blend /= sb
                        ac_blend = _fast_autocorr(a_blend).max()
                        if ac_blend < _fast_autocorr(a).max():
                            a = a_blend

        # Stall restart: small perturbation
        if stall_count > 500:
            rng_local = np.random.default_rng(it)
            pert = rng_local.normal(0, 0.01, n)
            a_pert = a + pert
            a_pert = _proj_simplex(np.maximum(a_pert, 0), 1.0)
            ac_pert = _fast_autocorr(a_pert).max()
            if ac_pert < cur_max:
                a = a_pert
            stall_count = 0

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-subgrad it={it} T={T:.3f} ({elapsed:.0f}s): best_maxb={best_maxb:.8f}")

    ac_final = _fast_autocorr(a).max()
    if ac_final < best_maxb:
        best_maxb = ac_final
        best_a = a.copy()
    return best_a, best_maxb


def _lp_solve(a, M, solver):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    # Build constraint: conv(a, b)[k] <= M for all k
    # conv(a,b)[k] = sum_j a[j]*b[k-j]
    # We build this efficiently using the band structure
    constraints = []
    # For each k in 0..2n-2, conv(a,b)[k] = sum over valid j of a[j]*b[k-j]
    # This is (A_k) @ b where A_k has at most min(n, k+1, 2n-1-k) nonzeros
    # For efficiency, build the full matrix (sparse)
    # Actually for n=1600, building 3199 rows x 1600 cols is fine
    T_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        T_mat[j:j + n, j] = a
    constraints.append(T_mat @ b_var <= M)
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def _lp_nesterov_polish(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration for final polish."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_maxb = _fast_autocorr(a).max()
    y = a.copy()
    t0 = time.monotonic()

    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break

        M = _fast_autocorr(y).max()
        if M < 1e-12:
            break
        b = _lp_solve(y, M, solver)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = _proj_simplex(np.maximum(y_new, 0), 1.0)

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = _proj_simplex(np.maximum(a_new, 0), 1.0)

        y = y_new
        a = a_new

        score_y = _fast_autocorr(y).max()
        if score_y < best_maxb:
            best_maxb = score_y
            best_a = y.copy()

        score_a = _fast_autocorr(a).max()
        if score_a < best_maxb:
            best_maxb = score_a
            best_a = a.copy()

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest t={t} ({elapsed:.0f}s): best_maxb={best_maxb:.8f}")

    return best_a, best_maxb


def _peak_directed_search(a0, time_limit, seed=42):
    """
    Direct peak-reduction: for each of the top peaks, try moving mass
    from the highest-contributing source position to a low-impact destination.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_maxb = _fast_autocorr(a).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        ac = _fast_autocorr(a)
        top_peaks = np.argsort(ac)[::-1][:10]
        improved = False

        for k in top_peaks:
            if ac[k] < best_maxb - 1e-10:
                continue
            # Find which positions contribute most to peak k
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            # Find safe destinations (not involved in top peaks)
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:5]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) < 10:
                candidates = np.where(np.arange(n) != src)[0]

            # Try moving small amounts to low-a positions
            candidates_sorted = candidates[np.argsort(a[candidates])]
            n_try = min(5, len(candidates_sorted))

            for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                for _ in range(n_try):
                    dst = candidates_sorted[rng.integers(len(candidates_sorted))]
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    ac_trial = _fast_autocorr(a_trial).max()
                    if ac_trial < best_maxb - 1e-12:
                        a = a_trial
                        best_maxb = ac_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Random perturbation to escape
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-8:
                amt = a[i] * rng.uniform(0.01, 0.05)
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    ac_trial = _fast_autocorr(a_trial).max()
                    if ac_trial < best_maxb:
                        a = a_trial
                        best_maxb = ac_trial
                        best_a = a.copy()
            else:
                # Small random jitter
                jitter = rng.normal(0, 0.005, n)
                a_trial = np.maximum(a + jitter, 0)
                st = a_trial.sum()
                if st > 1e-15:
                    a_trial /= st
                    ac_trial = _fast_autocorr(a_trial).max()
                    if ac_trial < best_maxb:
                        a = a_trial
                        best_maxb = ac_trial
                        best_a = a.copy()

    return best_a, best_maxb


def _parametric_init(n, p, alpha, eps, phase=0.0):
    """Generate a parametric initialization: ((1 + eps*cos(2*pi*i*p/n + phase))^alpha + 0.01)/n"""
    x = np.arange(n)
    base = 1.0 + eps * np.cos(2.0 * np.pi * p * x / n + phase)
    a = np.maximum(base, 0.0) ** alpha + 0.01
    a = a / a.sum()
    return a


def _smoothed_random_init(n, seed, sigma_factor=0.05):
    """Generate a smoothed random initialization."""
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    # Gaussian smoothing
    sigma = max(1, int(sigma_factor * n))
    kernel = np.exp(-0.5 * (np.arange(-2*sigma, 2*sigma+1) / sigma) ** 2)
    kernel /= kernel.sum()
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    else:
        a = np.ones(n) / n
    return a

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: multi-initialization, sparse LSE-subgradient optimization,
    LP corrections, and peak-directed local search.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1650.0)
    solver = _get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    # Get parent sequence
    n_target = 1600
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            n_target = len(parent)
            p_score = _evaluate_safe(parent)
            print(f"Parent: n={n_target}, score={p_score:.6f}")
    except Exception:
        pass

    best_a = None
    best_score = float('inf')

    def _try_update(a, label):
        nonlocal best_a, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = _evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Generate diverse initializations
    inits = []
    if parent is not None:
        inits.append(('parent', parent))

    # Parametric inits
    for p_freq in [1, 2, 3, 5]:
        for alpha in [1.0, 1.5, 2.0, 3.0]:
            for eps in [0.3, 0.6, 0.9]:
                a_init = _parametric_init(n_target, p_freq, alpha, eps)
                inits.append((f"param_p{p_freq}_a{alpha}_e{eps}", a_init))

    # Smoothed random inits
    for s in [42, 123, 456, 789]:
        a_init = _smoothed_random_init(n_target, s)
        inits.append((f"rand_s{s}", a_init))

    # Evaluate all inits and pick top 3
    init_scores = []
    for label, a_init in inits:
        sc = _evaluate_safe(a_init)
        if np.isfinite(sc):
            init_scores.append((sc, label, a_init))
    init_scores.sort(key=lambda x: x[0])
    print(f"\nTop initializations: {[(l, f'{s:.4f}') for s, l, _ in init_scores[:5]]}")

    if not init_scores:
        best_a = np.ones(n_target) / n_target
        best_score = _evaluate_safe(best_a)
    else:
        best_a = init_scores[0][2].copy()
        best_score = init_scores[0][0]

    top_inits = [a for _, _, a in init_scores[:3]]
    if best_a not in top_inits:
        top_inits.insert(0, best_a.copy())
    top_inits = top_inits[:3]

    # Phase 1: LSE-subgradient on top-3 initializations (split time)
    n_init = len(top_inits)
    remaining = time_budget - (time.monotonic() - t_global)
    t_per_init = min(remaining * 0.35 / n_init, 200.0)
    t_per_init = max(t_per_init, 30.0)

    for i, a_init in enumerate(top_inits):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 30:
            break
        t_alloc = min(t_per_init, remaining * 0.3)
        print(f"\n=== Phase 1.{i}: LSE-subgrad init {i} (budget={t_alloc:.0f}s) ===")
        a_out, _ = _lse_subgrad_opt(a_init, t_alloc,
                                     T_start=10.0, T_end=0.05,
                                     decay_factor=0.97, top_k=20,
                                     lp_every=300, solver=solver, verbose=True)
        _try_update(a_out, f"lse_init{i}")

    # Phase 2: LP-Nesterov polish on current best
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.25, 300.0)
    if t2 > 30:
        print(f"\n=== Phase 2: LP-Nesterov (budget={t2:.0f}s) ===")
        a_out, _ = _lp_nesterov_polish(best_a, t2, solver, verbose=True)
        _try_update(a_out, "lp_nest")

    # Phase 3: LSE-subgradient on LP result (different T schedule)
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.20, 250.0)
    if t3 > 30:
        print(f"\n=== Phase 3: LSE-subgrad round 2 (budget={t3:.0f}s) ===")
        a_out, _ = _lse_subgrad_opt(best_a, t3,
                                     T_start=5.0, T_end=0.02,
                                     decay_factor=0.96, top_k=10,
                                     lp_every=200, solver=solver, verbose=True)
        _try_update(a_out, "lse_r2")

    # Phase 4: LP-Nesterov again
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.20, 250.0)
    if t4 > 30:
        print(f"\n=== Phase 4: LP-Nesterov round 2 (budget={t4:.0f}s) ===")
        a_out, _ = _lp_nesterov_polish(best_a, t4, solver, verbose=True)
        _try_update(a_out, "lp_r2")

    # Phase 5: Peak-directed local search
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.15, 150.0)
    if t5 > 15:
        print(f"\n=== Phase 5: Peak-directed search (budget={t5:.0f}s) ===")
        a_out, _ = _peak_directed_search(best_a, t5, seed=999)
        _try_update(a_out, "peak_direct")

    # Phase 6: Final LSE with very low temperature
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.10, 120.0)
    if t6 > 15:
        print(f"\n=== Phase 6: LSE low-T final (budget={t6:.0f}s) ===")
        a_out, _ = _lse_subgrad_opt(best_a, t6,
                                     T_start=2.0, T_end=0.01,
                                     decay_factor=0.94, top_k=5,
                                     lp_every=100, solver=solver, verbose=True)
        _try_update(a_out, "lse_lowT")

    # Phase 7: Final peak-directed
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t7 = min(remaining * 0.5, 100.0)
        if t7 > 10:
            print(f"\n=== Phase 7: Final peak-directed (budget={t7:.0f}s) ===")
            a_out, _ = _peak_directed_search(best_a, t7, seed=31415)
            _try_update(a_out, "peak_final")

    # Final verification
    if best_a is not None:
        fs = _evaluate_safe(best_a)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_a)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_a]
