    shifted = np.clip((C - C_max) / T, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(y.astype(np.float64), fft_size)
    W = rfft(w.astype(np.float64), fft_size)
    cross = irfft(np.conj(A) * W, fft_size)
    # CORRECT: cross[m] = sum_j y[j]*w[j+m], so grad[i] = 2*cross[i]
    return 2.0 * cross[:n]


def _proj_simplex(v, S=1.0):
    n = len(v)
    if n <= 1:
        return np.array([S])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - S))[0]
    if len(rho) == 0:
        return np.full(n, S / n)
    rho = rho[-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 1e-15:
        out *= S / s
    else:
        out = np.full(n, S / n)
    return out


def _lse_fista(a0, T_schedule, iters_per_stage, time_limit, verbose=False):
    """FISTA on LSE-smoothed max with corrected gradient."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = _fast_conv(a).max()
    t0 = time.monotonic()

    for stage_i, T in enumerate(T_schedule):
        L = max(n / (T * 1.5), n / 30.0)
        t_fista = 1
        staging = a.copy()
        max_iters = max(int(iters_per_stage * (T_schedule[0] / T) ** 0.25), 1000)

        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_score

            grad = _lse_grad(y, T)
            z = y - (1.0 / L) * grad
            a_new = _proj_simplex(z)

            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - staging)
                y_new = np.maximum(y_new, 0.0)
                s_y = y_new.sum()
                if s_y > 1e-15:
                    y_new /= s_y
                else:
                    y_new = a_new.copy()

            staging = a_new.copy()
            a = a_new
            y = y_new
            t_fista += 1

            if it % 400 == 0:
                sc = _fast_conv(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

        sc = _fast_conv(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE-FISTA st={stage_i} T={T:.2f} ({elapsed:.0f}s): best={best_score:.8f}")

    sc = _fast_conv(a).max()
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def _get_lp_structure(n):
    if n in _lp_structure_cache:
        return _lp_structure_cache[n]
    rows = []
    cols = []
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for i in range(js, je + 1):
            rows.append(k)
            cols.append(i)
    r = np.array(rows, dtype=np.int32)
    c = np.array(cols, dtype=np.int32)
    _lp_structure_cache[n] = (r, c)
    return r, c


def _lp_best_response(a, M, solver):
    n = len(a)
    if M < 1e-15:
        return a.copy()
    rows, cols = _get_lp_structure(n)
    vals = a[cols]
    T_mat = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def _lp_nesterov(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov + adaptive restart."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_conv(a).max()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    restart_count = 0
    stall_counter = 0
    stall_threshold = 80
    best_stall_score = best_score

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = _fast_conv(y).max()
        if M_curr < 1e-14:
            break
        b = _lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Adaptive step: use 2/(t+c) with c adapting
        c = 2.0 + 2.0 * np.sin(0.1 * np.log1p(t))  # mild oscillation
        alpha = 2.0 / (t + c)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + c)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = _fast_conv(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_counter = 0
            best_stall_score = best_score
        else:
            stall_counter += 1

        score_a = _fast_conv(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 25 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = _fast_conv(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    stall_counter = 0

        # Adaptive restart
        if stall_counter > stall_threshold and restart_count < 8:
            restart_count += 1
            # Perturb y with small noise
            rng_r = np.random.default_rng(42 + restart_count * 1000)
            noise = rng_r.normal(0, 0.02, n)
            y_pert = y * (1 + noise)
            y_pert = np.maximum(y_pert, 0)
            s_p = y_pert.sum()
            if s_p > 1e-15:
                y = y_pert * norm / s_p
            avg_a = np.zeros(n)
            n_avg = 0
            stall_counter = 0
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    LP-Nest restart #{restart_count} ({elapsed:.0f}s): score={best_score:.8f}")

        if verbose and n_iters % 150 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nesterov it={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    # Check average at end
    if n_avg > 0:
        avg_check = avg_a / n_avg
        s_avg = avg_check.sum()
        if s_avg > 1e-15:
            avg_check = avg_check * norm / s_avg
            score_avg = _fast_conv(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()

    return best_a, best_score


def _local_search(a, time_limit, seed=42):
    """Multi-coordinate local search targeting convolution peaks."""
    n = len(a)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_conv(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        C = _fast_conv(a)
        top_k = np.argsort(C)[::-1][:10]
        improved = False

        for k in top_k[:5]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            if contrib.max() < 1e-15:
                continue
            # Find the most contributing source
            src = idx[np.argmax(contrib)]

            # Build peak mask: positions that contribute to top peaks
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k[:5]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            # Candidate destinations: low mass, outside peak region
            candidates = np.where(~peak_mask)[0]
            if len(candidates) < 3:
                candidates = np.where(np.arange(n) != src)[0]

            # Try moving to a few best candidates
            cand_scores = C.copy()
            for dst in candidates[np.argsort(a[candidates])[:max(3, len(candidates) // 8)]]:
                for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-16:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = _fast_conv(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break

        if not improved:
            # Random perturbation to escape local minimum
            stall += 1
            if stall % 20 == 0:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > norm * 1e-10:
                    frac = rng.uniform(0.003, 0.02)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        s_t = a_trial.sum()
                        if s_t > 1e-15:
                            a_trial = a_trial * norm / s_t
                            score = _fast_conv(a_trial).max()
                            if score < best_score:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                stall = 0

                # Multi-coordinate move: swap mass between 3 positions
                if stall > 50:
                    idx3 = rng.choice(n, size=4, replace=False)
                    a_trial = a.copy()
                    for trial in range(5):
                        s_i, d_i = idx3[trial], idx3[(trial + 1) % 4]
                        if a_trial[s_i] > norm * 1e-10:
                            amt = a_trial[s_i] * rng.uniform(0.01, 0.04)
                            a_trial[s_i] -= amt
                            a_trial[d_i] += amt
                    s_t = a_trial.sum()
                    if s_t > 1e-15 and (a_trial >= 0).all():
                        a_trial = a_trial * norm / s_t
                        score = _fast_conv(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0

            if stall > 800:
                break

    return best_a, best_score


def _frank_wolfe_fast(a0, time_limit, verbose=False):
    """FAST Frank-Wolfe: single-coordinate FW (no LP needed)."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_conv(a).max()
    y = a.copy()
    t0 = time.monotonic()

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        C = _fast_conv(y)
        k_star = int(np.argmax(C))

        # Gradient of C[k_star]
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        grad = np.zeros(n)
        for i in range(js, je + 1):
            grad[i] = 2.0 * y[k_star - i]
        # Also the symmetric part (same since it's self-convolution)

        i_min = int(np.argmin(grad))
        v = np.zeros(n)
        v[i_min] = norm

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * v
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = v.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_a = _fast_conv(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        if verbose and t % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def _interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def _evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

            print(f"\n=== Phase 9: Final local search n={n_cur} (t={t9:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = _local_search(a_norm, t9, seed=99999)
            _try_update(a_out, "local_final")

        # Last resort: quick FW
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            t10 = min(remaining * 0.5, 30.0)
            if t10 > 5:
                print(f"\n=== Phase 10: FW n={n_cur} (t={t10:.0f}s) ===")
                a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
                a_out, _ = _frank_wolfe_fast(a_norm, t10, verbose=False)
                _try_update(a_out, "fw_final")

    # Final verification
    if best_seq is not None:
        fs = _evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
