from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_gradient_topk(a, b, k):
    """Compute gradient using top-k peaks with uniform weights."""
    n = len(a)
    # Find top-k peaks
    if k >= len(b):
        k = len(b)
    top_idx = np.argsort(b)[::-1][:k]
    w = np.zeros(len(b))
    w[top_idx] = 1.0 / k
    
    # Compute gradient via convolution
    a_rev = a[::-1]
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)
    conv = irfft(W * AR, fft_size)
    return 2.0 * conv[n - 1:2 * n - 1]

def compute_gradient_lse(a, b, T):
    """Compute gradient using LSE weights."""
    n = len(a)
    max_b = b.max()
    w = np.exp((b - max_b) / T)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(b))
        w[np.argmax(b)] = 1.0
    else:
        w = w / w_sum
    
    a_rev = a[::-1]
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)
    conv = irfft(W * AR, fft_size)
    return 2.0 * conv[n - 1:2 * n - 1]

                break
            progress = min(1.0, (time.monotonic() - phase_t0) / phase_time)
            if use_topk:
                T = 1e-10
            else:
                T = T_from * max(T_to / T_from, 0.01) ** progress

            b = fast_autocorr(a)
            if use_topk:
                g = grad_topk(a, b, k)
            else:
                g = grad_lse(a, b, T)

            g_norm = np.linalg.norm(g)
            if g_norm < 1e-15:
                break

            # Adaptive step size: scale by inverse gradient norm
            a_norm = np.linalg.norm(a)
            eta = weight * a_norm / (g_norm * 1.0 + 1e-15)
            eta = min(eta, 2.0)

            # Heavy ball update
            v = beta * v - eta * g
            a_trial = a + v
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = normalize(a_trial, norm)
            steps_in_phase += 1

            sc = fast_autocorr(a_trial).max()
            if sc < best_score:
                best_score = sc
                best_a = a_trial.copy()
                stall = 0
            else:
                stall += 1

            a = a_trial

            if stall > 150:
                # Restart with perturbation
                rng = np.random.default_rng(steps_in_phase * 13 + 7)
                pert_scale = 0.03 * a.mean()
                a = np.maximum(a + rng.normal(0, pert_scale, n), 0)
                a = normalize(a, norm)
                v = np.zeros(n)
                stall = 0
                if verbose:
                    elapsed = time.monotonic() - t0
                    print(f"    GDescent phase={pi} restart at step={steps_in_phase} ({elapsed:.0f}s): best={best_score:.8f}")

            if verbose and steps_in_phase % 200 == 0:
                elapsed = time.monotonic() - t0
                label = f"topk{k}" if use_topk else f"T={T:.4f}"
                print(f"    GDescent phase={pi} step={steps_in_phase} ({elapsed:.0f}s) [{label}]: best={best_score:.8f}")

        total_steps += steps_in_phase

    return best_a, best_score


def local_search(a0, time_limit, verbose=True):
    """Coordinate-pair local search: move mass from high-contributing positions to low ones."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(a0, 0.0)
    a = normalize(a, norm)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    improves = 0
    stalls = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        peak_k = int(np.argmax(b))

        # Find top contributing pairs to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_k - idx]

        if contrib.max() < 1e-18:
            stalls += 1
            break

        # Try moving mass from the top contributor to a random low position
        src_local = idx[np.argmax(contrib)]
        # Also consider second and third contributors
        top_contrib_idx = idx[np.argsort(contrib)[::-1][:5]]

        improved_this_round = False
        for src in top_contrib_idx[:3]:
            if a[src] < 1e-12:
                continue
            # Pick a destination: prefer positions not in any top-5 peak
            top_peaks = np.argpartition(b, -5)[-5:]
            peak_mask = np.zeros(n, dtype=bool)
            for pk in top_peaks:
                js2 = max(0, pk - n + 1)
                je2 = min(n - 1, pk)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]
            # Pick from lowest-valued candidates
            sorted_cand = candidates[np.argsort(a[candidates])[:20]]
            dst = sorted_cand[rng.integers(len(sorted_cand))]

            # Try different fractions
            for frac in [0.01, 0.02, 0.05, 0.1, 0.15]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = normalize(a_trial, norm)
                sc = fast_autocorr(a_trial).max()
                if sc < best_score:
                    a = a_trial
                    best_score = sc
                    best_a = a_trial.copy()
                    improved_this_round = True
                    improves += 1
                    break

        if improved_this_round:
            stalls = 0
        else:
            stalls += 1
            if stalls > 100:
                # Larger perturbation
                n_pert = max(2, n // 80)
                idx_pert = rng.choice(n, size=n_pert, replace=False)
                a_trial = a.copy()
                for p in idx_pert:
                    if a_trial[p] < 1e-12:
                        continue
                    frac = rng.uniform(0.05, 0.2)
                    amt = a_trial[p] * frac
                    j = rng.integers(n)
                    a_trial[p] -= amt
                    a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = normalize(a_trial, norm)
                sc = fast_autocorr(a_trial).max()
                if sc < best_score * 1.005:
                    a = a_trial
                    if sc < best_score:
                        best_score = sc
                        best_a = a_trial.copy()
                stalls = 0

        if verbose and improves % 500 == 0 and improves > 0:
            elapsed = time.monotonic() - t0
            print(f"    LocalSearch improves={improves} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1720.0)
    solver = get_solver()
    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_eval_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_eval_score = true_score(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_eval_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_eval_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = true_score(a)
        if np.isfinite(sc) and sc < best_eval_score:
            best_eval_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize if no parent
    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = best_seq / best_seq.sum()
        best_eval_score = true_score(best_seq)
        print(f"Default init: n={n_base}, score={best_eval_score:.6f}")

    # Phase 1: LP warmup
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.18, 250.0)
    n_init = len(best_seq)
    print(f"\n=== Phase 1: LP warmup n={n_init} (budget={t1:.0f}s) ===")
    if t1 > 20:
        a_norm = normalize(best_seq, np.sqrt(2.0 * n_init))
        a_out, _ = lp_best_response_iteration(a_norm, t1, solver, verbose=True)
        try_update(a_out, f"lp{n_init}")

    # Phase 2: Multi-resolution gradient descent
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.40, 500.0)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: MultiRes GDescent n={n_cur} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_norm = normalize(best_seq, np.sqrt(2.0 * n_cur))
        a_out, _ = multi_res_grad_descent(a_norm, t2, verbose=True)
        try_update(a_out, f"grad{n_cur}")

    # Phase 3: Local search
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 180.0)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: Local Search n={n_cur} (budget={t3:.0f}s) ===")
    if t3 > 10:
        a_norm = normalize(best_seq, np.sqrt(2.0 * n_cur))
        a_out, _ = local_search(a_norm, t3, verbose=True)
        try_update(a_out, f"local{n_cur}")

    # Phase 4: Another gradient round (different momentum)
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.20, 250.0)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: GDescent round 2 n={n_cur} (budget={t4:.0f}s) ===")
    if t4 > 20:
        # Slightly perturb before restart
        rng = np.random.default_rng(77)
        a_pert = best_seq + rng.normal(0, 0.01 * best_seq.mean(), len(best_seq))
        a_pert = np.maximum(a_pert, 0.0)
        a_norm = normalize(a_pert, np.sqrt(2.0 * n_cur))
        a_out, _ = multi_res_grad_descent(a_norm, t4, verbose=True)
        try_update(a_out, f"grad2{n_cur}")

    # Phase 5: Local search again
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.10, 120.0)
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: Local Search 2 n={n_cur} (budget={t5:.0f}s) ===")
    if t5 > 10:
        a_norm = normalize(best_seq, np.sqrt(2.0 * n_cur))
        a_out, _ = local_search(a_norm, t5, verbose=True)
        try_update(a_out, f"local2{n_cur}")

    # Phase 6: Try longer length
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_init + 400
    t6 = min(remaining * 0.35, 300.0)
    print(f"\n=== Phase 6: Long n={n_long} (budget={t6:.0f}s) ===")
    if t6 > 60:
        init_long = interpolate_sequence(best_seq, n_long)
        a_norm = normalize(init_long, np.sqrt(2.0 * n_long))
        # Quick LP
        t6a = t6 * 0.3
        a_out, _ = lp_best_response_iteration(a_norm, t6a, solver, verbose=False)
        try_update(a_out, f"lp_long{n_long}")
        # Gradient
        t6b = t6 * 0.5
        n_cur = len(best_seq)
        a_norm = normalize(best_seq, np.sqrt(2.0 * n_cur))
        a_out, _ = multi_res_grad_descent(a_norm, t6b, verbose=False)
        try_update(a_out, f"grad_long{n_cur}")
        # Local
        t6c = t6 * 0.2
        n_cur = len(best_seq)
        a_norm = normalize(best_seq, np.sqrt(2.0 * n_cur))
        a_out, _ = local_search(a_norm, t6c, verbose=False)
        try_update(a_out, f"local_long{n_cur}")

    # Phase 7: Final polish on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t7 = max(remaining, 30.0)
    print(f"\n=== Phase 7: Final Polish n={n_best} (budget={t7:.0f}s) ===")
    if t7 > 20:
        a_norm = normalize(best_seq, np.sqrt(2.0 * n_best))
        a_out, _ = multi_res_grad_descent(a_norm, t7 * 0.5, verbose=False)
        try_update(a_out, f"final_grad{n_best}")
        a_norm = normalize(best_seq, np.sqrt(2.0 * n_best))
        a_out, _ = local_search(a_norm, t7 * 0.5, verbose=False)
        try_update(a_out, f"final_local{n_best}")

    # Final output
    if best_seq is not None:
        fs = true_score(best_seq)
        best_eval_score = fs
        print(f"\nFINAL: score={best_eval_score:.6f}, n={len(best_seq)}, time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
