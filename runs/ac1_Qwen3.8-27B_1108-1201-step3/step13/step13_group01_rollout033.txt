    g = 2.0 * c[2*n-2-i for i in range(n)]  # g_i = 2*c[2n-2-i]
    
    s = a.sum()
    grad = 2.0*n * g / (s*s) - 4.0*n*L*a / (s**3)
    
    return grad

        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        a = a_new
        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score
