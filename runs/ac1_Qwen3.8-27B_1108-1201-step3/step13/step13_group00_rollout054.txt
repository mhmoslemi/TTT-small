#   Actually for j=0: hi = min(2n-1, n) = n, lo = 0, length = n

#   for j in [0, n-1]: length = n always (since j < n, j+n <= 2n-1 when j <= n-1)

def compute_T_v2(a):
    """Efficiently compute T such that T @ b = conv(a,b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def lse_gradient(a, T_temp, fft_size=None):
    """Compute gradient of LSE(T_temp, conv(a,a))."""
    n = len(a)
    size = 2 * n - 1
    if fft_size is None:
        fft_size = 1 << ((3 * n - 2) - 1).bit_length()
    
    conv = fast_autocorr(a)
    max_conv = conv.max()
    exp_conv = np.exp((conv - max_conv) / T_temp)
    w = exp_conv / exp_conv.sum()
    
    # grad[i] = (2/T) * sum_k w[k] * a[k-i] (valid k-i)
    # = (2/T) * corr[N-i] where corr = ifft(FFT(w_padded) * conj(FFT(a_padded)))
    w_pad = np.zeros(fft_size)
    w_pad[:size] = w
    a_pad = np.zeros(fft_size)
    a_pad[:n] = a
    
    W = rfft(w_pad, fft_size)
    A = rfft(a_pad, fft_size)
    corr = irfft(W * np.conj(A), fft_size)
    
    # grad[i] = (2/T) * corr[fft_size - i] for i=1..n-1, corr[0] for i=0
    grad = np.zeros(n)
    grad[0] = (2.0 / T_temp) * corr[0]
    if n > 1:
        grad[1:] = (2.0 / T_temp) * corr[fft_size - 1 : fft_size - n : -1]
    
    return grad, w

def mirror_descent_optimize(a0, time_limit, verbose=True):
    """Mirror descent (multiplicative weights) on the simplex."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    fft_size = 1 << ((3 * n - 2) - 1).bit_length()
    
    # Anneal temperature
    n_stages = 10
    temps = np.geomspace(50.0, 1.0, n_stages)
    
    for si, T_temp in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        n_steps = max(20, int(remaining * 2))
        
        step = 1.0 / (T_temp * max(1, int(np.sqrt(n))))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            grad, w = lse_gradient(a, T_temp, fft_size)
            
            # Multiplicative update
            log_a = np.log(np.maximum(a, 1e-300))
            new_log_a = log_a - step * grad
            # Normalize on simplex (sum = norm)

            a_new = np.exp(new_log_a) * norm
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:

            a = a_new

            if score < best_score:

                best_a = a.copy()
        
        if verbose:

            print(f"    MD stage T={T_temp:.1f} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score
