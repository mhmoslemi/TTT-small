The subgradient of `f(a) = max_k c[k]` is the convex hull of `{2 * a[k-·] : k ∈ argmax}` (where `a[k-·]` denotes the vector with components `a[k-j]` for valid j, and 0 otherwise).

    avg_a = a.copy()
    n_avg = 1
    
    step = 1.0
    step_min = 0.1
    step_max = 10.0
    
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_conv(a)
        M = c.max()
        
        if use_exact:
            g = compute_subgrad(a, c, mode='exact')
        else:
            g = compute_subgrad(a, c, mode='soft', top_k=top_k, beta=beta)
        
        # Mirror descent update
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - step * g
        max_log = log_a_new.max()
        a_new = np.exp(log_a_new - max_log)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
            step = max(step * 0.5, step_min)
        else:
            a_new = a_new * c_norm / s_new
            M_new = fast_conv(a_new).max()
            
            if M_new > M * 1.05:
                # Step too large
                step = max(step * 0.5, step_min)
            else:
                a = a_new
                if M_new < M * 0.995:
                    step = min(step * 1.1, step_max)
                else:
                    step = max(step * 0.97, step_min)
                
                if M_new < best_m:
                    best_m = M_new
                    best_a = a_new.copy()
        
        # Running average
        avg_a += a
        n_avg += 1
        if n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * c_norm / s_avg
                M_avg = fast_conv(avg_check).max()
                if M_avg < best_m:
                    best_m = M_avg
                    best_a = avg_check.copy()
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c_norm * c_norm)
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.6f}, step={step:.3f}")
    
    return best_a, best_m

def lp_best_response(a, M, solver, max_iters=50):
    """
    LP: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0.
    Uses constraint generation: start with few constraints, add violated ones.
    """
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    
    # Build constraint matrix (sparse would be better but dense is OK for moderate n)
    # (a*b)[k] = sum_j a[k-j] * b[j] for valid j
    # We only enforce a subset of constraints initially
    
    # Start with all constraints (for simplicity, constraint generation is complex)
    # Actually for n up to 4000, 2n-1 ≈ 8000 constraints is manageable
    
    constraints = []
    
    # Build the Toeplitz-like matrix
    # T[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # (T @ b)[k] = sum_j T[k,j] * b[j] = (a*b)[k]
    
    # For efficiency, use the fact that T is a convolution matrix
    # But cvxpy might not handle this efficiently, so let's build it explicitly
    # for n <= 4000
    
    if n <= 4000:
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j+n, j] = a
        constraints = [T @ b_var <= M]
    else:
        # For larger n, use a subset of constraints
        # Only enforce constraints at positions where (a*a) is large
        c_self = fast_conv(a)
        top_positions = np.argsort(c_self)[::-1][:min(200, 2*n-1)]
        T = np.zeros((len(top_positions), n))
        for ii, k in enumerate(top_positions):
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            T[ii, idx] = a[k - idx]
        constraints = [T @ b_var <= M]
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def nesterov_lp(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    c_norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * c_norm / n
    else:
        a = a * c_norm / s0
    
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_conv(y).max()
        if M_curr < 1e-12:
            break
        
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * c_norm / b_sum
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * c_norm / s_y
        
        if t == 1:
            x_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new * c_norm / s_x
        
        y = y_new
        x = x_new
        
        score_y = fast_conv(y).max()
        if score_y < best_m:
            best_m = score_y
            best_a = y.copy()
        
        score_x = fast_conv(x).max()
        if score_x < best_m:
            best_m = score_x
            best_a = x.copy()
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c_norm * c_norm)
            print(f"    LP iter={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def coordinate_polish(a0, time_limit, verbose=True):
    """
    Coordinate descent: transfer mass from peak-contributing positions
    to low-impact positions.
    """
    n = len(a0)
    c_norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * c_norm / n
    else:
        a = a * c_norm / s0
    
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    stall = 0
    max_stall = 300
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        c = fast_conv(a)
        top_k = np.argsort(c)[::-1][:10]
        improved = False
        
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            src = idx[np.argmax(contrib)]
            
            # Find destination: position with minimal impact on all top peaks
            impact = np.zeros(n)
            for k2 in top_k:
                k2 = int(k2)
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                idx2 = np.arange(js2, je2 + 1)
                # Impact of increasing a[i] on c[k2]: a[k2-i]
                j2 = k2 - idx2
                impact[idx2] += a[j2]
            
            # Exclude positions already in peak-affected regions
            src_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                k2 = int(k2)
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                src_mask[js2:je2+1] = True
            
            candidates = np.where(~src_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]
            
            # Pick candidate with lowest impact and lowest value
            scores = impact[candidates] + 0.01 * a[candidates]
            cands_sorted = candidates[np.argsort(scores)]
            
            # Try several candidates
            for dst in cands_sorted[:5]:
                dst = int(dst)
                if dst == src:
                    continue
                for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial * c_norm / s_t
                    c_trial = fast_conv(a_trial)
                    M_trial = c_trial.max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > c_norm * 1e-6:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                if a[i] >= 0:
                    a = a * c_norm / a.sum()
                    M_new = fast_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = a * c_norm / a.sum()
        
        if verbose and stall % 50 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c_norm * c_norm)
            print(f"    Coord stall={stall} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m
