So (M_k a)[j] = a[k-j] (if valid), (M_k a)[i] = a[k-i] (if valid).

δ^T M_k δ = (-t)^2 * M_k[i,i] + 2*(-t)*t*M_k[i,j] + t^2 * M_k[j,j]
= t^2 * M_k[i,i] - 2*t^2 * M_k[i,j] + t^2 * M_k[j,j]
= t^2 * (M_k[i,i] - 2*M_k[i,j] + M_k[j,j])

M_k[i,i] = 1 if 2i = k, else 0
M_k[i,j] = 1 if i+j = k, else 0
M_k[j,j] = 1 if 2j = k, else 0

So:
Δ(a*a)[k] = t * ((M_k a)[j] - (M_k a)[i]) + t^2 * (M_k[i,i] - 2*M_k[i,j] + M_k[j,j])

This is a quadratic in t for each k. The constraint is:
(a*a)[k] + t * ((M_k a)[j] - (M_k a)[i]) + t^2 * (M_k[i,i] - 2*M_k[i,j] + M_k[j,j]) <= V for all k.

For a given target V (the current max), we want to find the maximum t >= 0 that keeps all constraints satisfied. This is a one-dimensional feasibility problem - we can binary search on t!

This gives us an exact coordinate-descent-like step for pairwise moves.

**Algorithm:**
1. Start with current a, compute (a*a), find current max V.
2. For each pair (i, j) [or a subset of promising pairs]:
   a. Compute the linear and quadratic coefficients for all k
   b. Binary search for the largest t in [0, a[i]] such that all (a*a)[k] + linear*t + quad*t^2 <= V * (1 + epsilon) [or <= V for strict improvement]
   c. If t > 0, apply the move
3. Repeat until no improvement

This is exact (in the sense that each step is optimal for the chosen pair) and guaranteed to make progress if the current point is not a local minimum.

The issue is computational cost: for each pair, we need to check all 2n-1 lags. With n=1600, that's 3200 lags per pair check. If we try, say, 100 pairs per iteration, that's 320,000 operations per iteration, which is fast.

Let me also think about which pairs to try. The most promising pairs are:
- (i, j) where i is a "peak contributor" (high a[i] that contributes to the max lag) and j is a "low" position (low a[j])
- Pairs where the linear term is negative (meaning moving mass from i to j initially decreases the max)

Actually, a simpler heuristic: for the current max lag k*, the positions that contribute most are those where a[j]*a[k*-j] is large. Moving mass from these positions to positions that don't contribute to any peak would be beneficial.

1. **LP Nesterov** (improved from previous)

3. **Subgradient method with dual distribution** (new)

5. **Final polish**

```python
import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def build_toeplitz(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def find_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(y, T, solver, warm_b=None):
    """Solve LP: min M s.t. T@b <= M, sum(b)=1, b>=0."""
    n = len(y)
    if warm_b is not None and np.all(warm_b >= -1e-10) and warm_b.sum() > 1e-10:
        b_var = cp.Variable(n, nonneg=True, value=warm_b)
    else:
        b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable(value=1.0 / n)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, max_iters=100000):
    """
    Nesterov-accelerated LP best-response with adaptive restart.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    avg_long = np.zeros(n)
    n_long = 0
    WINDOW_MED = 50
    avg_med = np.zeros(n)
    n_med = 0
    med_buf = [None] * WINDOW_MED
    med_idx = 0

    stall_count = 0
    warm_b = None
    last_print = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        T = build_toeplitz(y)
        b_star, M_val = solve_lp_best_response(y, T, solver, warm_b)
        if b_star is None:
            warm_b = None
            continue
        warm_b = b_star

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_sum = y_new.sum()
        if y_sum < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / y_sum

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new

        y = y_new
        a = a_new

        avg_long += a
        n_long += 1
        if med_buf[med_idx] is not None:
            avg_med -= med_buf[med_idx]
        med_buf[med_idx] = a.copy()
        avg_med += a
        n_med += 1
        med_idx = (med_idx + 1) % WINDOW_MED

        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if n_long % 20 == 0 and n_long >= 40:
            avg_check = avg_long / n_long
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
                stall_count = 0

        if n_med >= WINDOW_MED and n_long % 20 == 0:
            avg_m = avg_med / WINDOW_MED
            score_m = 2.0 * n * fast_autocorr(avg_m).max()
            if score_m < best_score:
                best_score = score_m
                best_a = avg_m.copy()
                stall_count = 0

        if stall_count > 800 and t > 200:
            a = best_a.copy()
            y = best_a.copy()
            avg_long = np.zeros(n)
            n_long = 0
            avg_med = np.zeros(n)
            n_med = 0
            med_buf = [None] * WINDOW_MED
            med_idx = 0
            stall_count = 0
            warm_b = None

        if t - last_print > 2000:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}")
            last_print = t

    return best_a, best_score
```

```python
def pairwise_move_descent(a0, time_limit, seed=42):
    """
    Exact pairwise move coordinate descent.
    For each pair (i,j), find the optimal transfer amount t that minimizes
    the maximum convolution value.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit - 2:
        b = fast_autocorr(a)
        bmax = b.max()
        
        # Find top peaks
        top_k = min(5, len(b))
        top_peaks = np.argsort(b)[::-1][:top_k]
        
        # Compute "danger" positions: those that contribute to top peaks
        danger = np.zeros(n, dtype=float)
        for k in top_peaks:
            js = max(0, k - (n - 1))
            je = min(n - 1, k)
            for j in range(js, je + 1):
                danger[j] += a[j] * a[k - j] / (b[k] + 1e-30)
        
        # Find promising sources: high danger + high value
        sources_score = danger * (a + 1e-10)
        # Find promising destinations: low danger + low value
        dests_score = (1.0 - danger) * (1.0 - a + 1e-10)
        
        # Get candidate sources and destinations
        n_sources = min(20, n)
        n_dests = min(20, n)
        src_candidates = np.argsort(sources_score)[::-1][:n_sources]
        dst_candidates = np.argsort(dests_score)[::-1][:n_dests]
        
        # Also add some random pairs
        if n > 2:
            for _ in range(5):
                src_candidates = np.append(src_candidates, rng.integers(n))
                dst_candidates = np.append(dst_candidates, rng.integers(n))
        
        improved = False
        for src in src_candidates:
            if time.monotonic() - t0 > time_limit - 3:
                break
            for dst in dst_candidates:
                if src == dst:
                    continue
                if a[src] < 1e-12:
                    continue
                
                # Compute coefficients for the quadratic in t
                # Δ(a*a)[k] = t * (a[k-dst] - a[k-src]) + t^2 * (M[k,src,src] - 2*M[k,src,dst] + M[k,dst,dst])
                # where M[k,i,j] = 1 if i+j=k, else 0
                
                # For each lag k, compute linear and quadratic coefficients
                # Linear: L_k = a[k-dst] (if valid) - a[k-src] (if valid)
                # Quadratic: Q_k = [2*src==k] - 2*[src+dst==k] + [2*dst==k]
                
                best_t = 0.0
                best_new_max = bmax
                
                # Binary search for optimal t
                t_lo = 0.0
                t_hi = a[src]
                
                # First check if any t > 0 helps
                # Evaluate at a small t
                t_test = t_hi * 0.01
                new_b = _eval_conv_after_transfer(a, src, dst, t_test, n)
                new_max = new_b.max()
                if new_max >= bmax:
                    continue
                
                # Binary search for the largest t that keeps max <= bmax
                for _ in range(30):
                    t_mid = (t_lo + t_hi) / 2
                    new_b = _eval_conv_after_transfer(a, src, dst, t_mid, n)
                    new_max = new_b.max()
                    if new_max <= bmax * (1 + 1e-12):
                        t_lo = t_mid
                    else:
                        t_hi = t_mid
                
                t_opt = t_lo * 0.99  # Slightly less to ensure strict improvement
                if t_opt < 1e-12:
                    continue
                
                # Apply the move
                a[src] -= t_opt
                a[dst] += t_opt
                a = np.maximum(a, 0)
                sn = a.sum()
                if sn < 1e-12:
                    a[src] += t_opt
                    a[dst] -= t_opt
                    continue
                a = a / sn
                
                new_score = 2.0 * n * fast_autocorr(a).max()
                if new_score < best_score:
                    best_score = new_score
                    best_a = a.copy()
                    improved = True
                    no_improve = 0
                else:
                    # Revert
                    a[dst] -= t_opt
                    a[src] += t_opt
                    no_improve += 1
        
        if not improved:
            # Try random perturbation
            if no_improve % 10 == 0:
                n_pert = rng.integers(2, 8)
                a_trial = a.copy()
                for _ in range(n_pert):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a_trial[i] > 1e-10:
                        frac = rng.uniform(0.005, 0.05)
                        amt = a_trial[i] * frac
                        a_trial[i] -= amt
                        a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-10:
                    a_trial = a_trial / st
                    sc = 2.0 * n * fast_autocorr(a_trial).max()
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        no_improve = 0
        
        if no_improve > 300:
            break
    
    return best_a, best_score


def _eval_conv_after_transfer(a, src, dst, t, n):
    """Evaluate convolution after transferring t from src to dst."""
    a_new = a.copy()
    a_new[src] -= t
    a_new[dst] += t
    return fast_autocorr(a_new)
```

Hmm, the `_eval_conv_after_transfer` function is expensive because it does a full FFT each time. For the binary search, that's 30 FFTs per pair, which is too slow.

Let me optimize: instead of recomputing the full convolution, I can compute the change analytically.

When we transfer t from src to dst:
new_a = a - t*e_src + t*e_dst

new_conv = (a - t*e_src + t*e_dst) * (a - t*e_src + t*e_dst)
= a*a - t*(e_src*a + a*e_src) + t*(e_dst*a + a*e_dst) + t^2*(e_src*e_dst + e_dst*e_src) - t^2*e_src*e_src + t^2*e_dst*e_dst

Wait, let me be more careful:
(a + δ) * (a + δ) where δ = -t*e_src + t*e_dst

= a*a + 2*(δ * a) + δ * δ

(δ * a)[k] = sum_j δ[j] * a[k-j]
= -t * a[k-src] + t * a[k-dst] (with valid indices)

(δ * δ)[k] = sum_j δ[j] * δ[k-j]
= (-t)^2 * [k-src-src is valid... ]

Actually, δ has only two non-zero entries: δ[src] = -t, δ[dst] = t.

(δ * δ)[k] = δ[src]*δ[k-src] + δ[dst]*δ[k-dst]
= (-t)*δ[k-src] + t*δ[k-dst]

δ[k-src] is non-zero only if k-src = dst, i.e., k = src+dst. Then δ[k-src] = δ[dst] = t.
δ[k-dst] is non-zero only if k-dst = src, i.e., k = src+dst. Then δ[k-dst] = δ[src] = -t.

So (δ*δ)[k] = 0 for all k, except possibly k = src+dst:
If k = src+dst: (δ*δ)[k] = (-t)*t + t*(-t) = -2t^2

Wait, that gives -2t^2 at k = src+dst. Let me double-check:
(δ*δ)[src+dst] = sum_j δ[j]*δ[src+dst-j]
Non-zero when j=src (then src+dst-j = dst, δ[dst]=t): term = (-t)*t = -t^2
Non-zero when j=dst (then src+dst-j = src, δ[src]=-t): term = t*(-t) = -t^2
Total: -2t^2

And for other k values, (δ*δ)[k] = 0.

So:
new_conv[k] = a*a[k] + 2*t*(a[k-dst] - a[k-src]) - 2*t^2*[k == src+dst]

where a[k-src] is 0 if k-src is out of [0, n-1], and similarly for a[k-dst].

This is much faster to compute! For a given pair (src, dst), I can compute the change in O(n) instead of O(n log n).

```python
def _conv_change_linear(a, src, dst, n):
    """Compute L[k] = 2*(a[k-dst] - a[k-src]) for all k (the linear part of conv change)."""
    L = np.zeros(2 * n - 1)
    # a[k-dst] is valid when 0 <= k-dst < n, i.e., dst <= k <= dst+n-1
    L[dst:dst+n] += 2.0 * a
    # a[k-src] is valid when 0 <= k-src < n, i.e., src <= k <= src+n-1
    L[src:src+n] -= 2.0 * a
    return L


def _conv_change_quad(src, dst, n):
    """The quadratic part is -2*t^2 at k = src+dst, else 0."""
    # Returns the index and value
    k = src + dst
    if 0 <= k <= 2 * n - 2:
        return k, -2.0
    return -1, 0.0

    """
    Exact pairwise move coordinate descent with O(n) change evaluation.
    """
    n = len(a0)

        contrib = np.zeros(n)

            for j in range(js, je + 1):

        dst_set = np.argsort(contrib)[:n_try]

                    continue

                L = _conv_change_linear(a, src, dst, n)
                qk, qv = _conv_change_quad(src, dst, n)

                # i.e., L[k]*t + qv*t^2*[k==qk] <= bmax - b[k] for all k

                # L[qk]*t + qv*t^2 <= R[qk]
                # For linear constraints at other k:
                # L[k]*t <= R[k]
                
                # Find the max t that satisfies all constraints
                t_max = a[src]  # Can't transfer more than we have

                # L[k]*t <= R[k] for all k
                # If L[k] > 0: t <= R[k]/L[k]
                # If L[k] < 0: always satisfied (for t >= 0)
                # If L[k] == 0: need R[k] >= 0 (always true)
                
                pos_L = L > 1e-15
                if np.any(pos_L):
                    ratio = R[pos_L] / L[pos_L]  # Wait, R = bmax - b
                    # Actually R[k] = bmax - b[k]
                    # For k where L[k] > 0: t <= (bmax - b[k]) / L[k]
                    # But bmax - b[k] could be 0 for the max lag
                    valid = (bmax - b) > -1e-15  # Should always be true
                    ratios = (bmax - b) / L
                    # For positions where L > 0, we need t <= ratios
                    # For the max position(s), bmax - b[k] = 0, so if L[k] > 0 at those positions, t must be 0
                    # That's a problem - it means we can't increase the value at the max position
                    
                    # Actually, we need (bmax - b[k]) / L[k] >= 0, which requires bmax - b[k] >= 0
                    # This is always true. But if bmax - b[k] = 0 and L[k] > 0, then t <= 0.
                    # This means if the linear term increases the value at the current max, we can't do any transfer.
                    
                    t_max = min(t_max, np.minimum(ratios[pos_L], t_max))
                
                # Check quadratic constraint
                if qk >= 0 and qv != 0:
                    # L[qk]*t + qv*t^2 <= bmax - b[qk]
                    # qv = -2, so: -2t^2 + L[qk]*t <= bmax - b[qk]
                    # 2t^2 - L[qk]*t + (bmax - b[qk]) >= 0
                    # This is always true for t=0 (since bmax-b[qk] >= 0)
                    # For large t, 2t^2 dominates, so it's satisfied
                    # The constraint might be active in between
                    # Actually since qv < 0, the quadratic opens downward
                    # L[qk]*t - 2t^2 <= bmax - b[qk]
                    # Let f(t) = L[qk]*t - 2t^2 - (bmax - b[qk])
                    # We need f(t) <= 0
                    # f(0) = -(bmax - b[qk]) <= 0 ✓
```

```python
import numpy as np

from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def build_toeplitz(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def find_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.SCS, cp.PDLP, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(y, T, solver, warm_b=None):
    """Solve LP: min M s.t. T@b <= M, sum(b)=1, b>=0. Returns (b, M, dual_weights)."""
    n = len(y)
    if warm_b is not None and np.all(warm_b >= -1e-10) and warm_b.sum() > 1e-10:
        b_var = cp.Variable(n, nonneg=True, value=warm_b)
    else:
        b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable(value=1.0 / n)
    cons = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    cons_ref = None
    prob = cp.Problem(cp.Minimize(M_var), cons)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, None
    b_val = b_val / s
    M_val = float(M_var.value)
    # Try to get dual variables for the constraint T@b <= M
    dual_w = None
    try:
        if cons[0].dual_value is not None:
            dual_w = np.asarray(cons[0].dual_value, dtype=np.float64)
            dual_w = np.maximum(dual_w, 0)
            ds = dual_w.sum()
            if ds > 1e-15:
                dual_w = dual_w / ds
    except Exception:
        dual_w = None
    return b_val, M_val, dual_w

def conv_change_after_transfer(a, src, dst):
    """
    Compute L and q such that new_conv[k] = old_conv[k] + L[k]*t + q*t^2
    after transferring t from a[src] to a[dst].
    L[k] = 2*(a[k-dst] - a[k-src]) (with out-of-range as 0)
    q = -2 if 0 <= src+dst <= 2n-2, else 0 (at lag src+dst)
    """
    n = len(a)
    conv_len = 2 * n - 1
    L = np.zeros(conv_len)
    L[dst:dst + n] += 2.0 * a
    L[src:src + n] -= 2.0 * a
    qk = src + dst
    if 0 <= qk < conv_len:
        q = -2.0
    else:
        q = 0.0
        qk = -1
    return L, qk, q


def pairwise_transfer_descent(a0, time_limit, seed=42):
    """
    Fast pairwise transfer descent using analytical change computation.
    For each promising (src, dst) pair, compute the maximum safe transfer.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit - 3:
        b = fast_autocorr(a)
        bmax = b.max()

        # Find top peaks
        n_peaks = min(8, 2 * n - 1)
        top_peaks = np.argsort(b)[::-1][:n_peaks]

        # Compute contribution of each position to top peaks
        contrib = np.zeros(n)
        for k in top_peaks:
            js = max(0, k - (n - 1))
            je = min(n - 1, k)
            contrib[js:je + 1] += a[js:je + 1] * a[k - js:k - je - 1 + 1]

        # Also add a global "importance" based on all lags
        total_contrib = np.zeros(n)
        for j in range(n):
            # a[j] appears in conv[k] for k in [j, j+n-1] as a[j]*a[k-j]
            total_contrib[j] = a[j] * (a[j:min(2 * n - 1, j + n)].sum())

        # Select sources (high contribution to peaks) and destinations (low)
        n_src = min(20, n)
        n_dst = min(20, n)
        src_set = np.argsort(contrib)[::-1][:n_src]
        dst_set = np.argsort(contrib)[:n_dst]

        # Add some diversity
        if n > 10:
            rand_src = rng.choice(n, size=5, replace=False)
            rand_dst = rng.choice(n, size=5, replace=False)
            src_set = np.concatenate([src_set, rand_src])
            dst_set = np.concatenate([dst_set, rand_dst])

        improved = False
        for src in src_set:
            if time.monotonic() - t0 > time_limit - 4:
                break
            if a[src] < 1e-12:
                continue

            for dst in dst_set:
                if src == dst:
                    continue
                if a[src] < 1e-12:
                    continue

                # Compute change coefficients
                L, qk, q = conv_change_after_transfer(a, src, dst)

                # Find max t from linear constraints: L[k]*t <= bmax - b[k] for all k
                # For L[k] > 0: t <= (bmax - b[k]) / L[k]
                pos_mask = L > 1e-14
                if np.any(pos_mask):
                    margins = (bmax - b) / L[pos_mask]
                    margins = margins[margins > 0]
                    if len(margins) > 0:
                        t_bound = margins.min()
                    else:
                        t_bound = 0.0
                else:
                    t_bound = a[src]

                # Also respect: t <= a[src]
                t_bound = min(t_bound, a[src])

                if t_bound < 1e-14:
                    continue

                # Try a fraction of the bound
                for frac in [0.95, 0.7, 0.5, 0.3, 0.1]:
                    t_try = t_bound * frac
                    if t_try < 1e-14:
                        continue

                    # Check quadratic constraint at qk
                    if qk >= 0 and q != 0:
                        new_val_qk = b[qk] + L[qk] * t_try + q * t_try * t_try
                        if new_val_qk > bmax * (1 + 1e-10):
                            continue

                    # Apply and verify
                    a_trial = a.copy()
                    a_trial[src] -= t_try
                    a_trial[dst] += t_try
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-12:
                        continue
                    a_trial = a_trial / st

                    new_b = fast_autocorr(a_trial)
                    new_max = new_b.max()
                    if new_max < bmax * (1 - 1e-10):
                        a = a_trial
                        new_score = 2.0 * n * new_max
                        if new_score < best_score:
                            best_score = new_score
                            best_a = a.copy()
                        improved = True
                        no_improve = 0
                        break

        if not improved:
            no_improve += 1
            # Random perturbation to escape
            if no_improve % 15 == 0:
                n_pert = rng.integers(2, 10)
                a_trial = a.copy()
                for _ in range(n_pert):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a_trial[i] > 1e-10:
                        frac = rng.uniform(0.005, 0.08)
                        amt = a_trial[i] * frac
                        a_trial[i] -= amt
                        a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-10:
                    a_trial = a_trial / st
                    sc = 2.0 * n * fast_autocorr(a_trial).max()
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        no_improve = 0

            if no_improve > 400:
                break

    return best_a, best_score


def subgradient_descent(a0, time_limit, eta0=0.05, seed=123):

        a = np.ones(n) / n
    else:
        a = a / s

    t0 = time.monotonic()

            break

        # w_k = exp(-(bmax - b[k]) / scale)

            js = max(0, k - (n - 1))
            je = min(n - 1, k)
            # grad[j] += 2 * wk * a[k-j] for j in [js, je]
            grad[js:je + 1] += 2.0 * wk * a[k - js:k - je]

        # Step size: use Polyak-like rule
        # eta = (f(a) - f*) / ||grad||^2, but we don't know f*
        # Use diminishing: eta = eta0 / (1 + it * 0.01)
        eta = eta0 / (1.0 + it * 0.005)
        
        # Gradient step
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            eta = eta * 0.5
            a_new = a - eta * grad
            a_new = np.maximum(a_new, 0)
            sn = a_new.sum()
            if sn < 1e-15:
                continue
        a_new = a_new / sn

        score_new = 2.0 * n * fast_autocorr(a_new).max()
        if score_new < best_score:
            a = a_new
            best_score = score_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall % 100 == 0:
                eta = max(eta0 * 0.01, eta * 0.5)
            if stall > 1000:
                # Random restart
                a = best_a.copy()
                noise = rng.normal(0, 0.02, n)
                a = a + noise * a
                a = np.maximum(a, 0)
                sn = a.sum()
                if sn > 1e-10:
                    a = a / sn
                elif sn > 0:
                    a = a / sn
                stall = 0
                eta = eta0

        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SGD it={it} ({elapsed:.0f}s): best={best_score:.6f}, eta={eta:.6f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng, seed_type=0):
    """Generate diverse structured initializations."""
    choice = seed_type % 10
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 4)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 3
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 2 + 0.1
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.7 + 0.2 * np.cos(2 * x) + 0.1 * np.cos(5 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-8 * (x - 0.25) ** 2) + 0.6 * np.exp(-8 * (x - 0.75) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * np.cos(np.pi * x / 3) ** 2 + 0.05
    elif choice == 7:
        x = np.linspace(-1, 1, n)
        a = np.maximum(0, 1 - np.abs(x) ** 1.5) + 0.05
    elif choice == 8:
        x = np.linspace(0, 1, n)
        a = np.exp(-30 * x ** 2) + 0.5 * np.exp(-30 * (x - 1) ** 2) + 0.01
    else:
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * (x - 0.3) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def multi_start_optimize(n, time_limit, solver, rng):
    """Try multiple initializations and keep the best."""
    best_a = None
    best_score = float('inf')
    n_starts = max(2, int(time_limit / 80))
    per_start = time_limit / n_starts
    t0 = time.monotonic()

    for si in range(n_starts):
        if time.monotonic() - t0 > time_limit - 10:
            break
        init = structured_init(n, rng, si)
        a_out, sc = nesterov_lp_optimize(init, per_start, solver)
        if sc < best_score:
            best_score = sc
            best_a = a_out.copy()
            elapsed = time.monotonic() - t0
            print(f"      Multi-start {si+1}/{n_starts} ({elapsed:.0f}s): score={sc:.6f}")

    if best_a is None:
        best_a = structured_init(n, rng, 0)
        best_score = 2.0 * n * fast_autocorr(best_a).max()
    return best_a, best_score


def fine_local_search(a0, time_limit, seed=777):
    """
    Very fine-grained local search: try tiny adjustments to individual
    coordinates and pairs, accepting any improvement.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 2:
        b = fast_autocorr(a)
        bmax = b.max()

        # Find the top contributing pair to the max lag
        k_max = int(np.argmax(b))

        # Try moving small amounts from high contributors at k_max to low positions
        js = max(0, k_max - (n - 1))
        je = min(n - 1, k_max)
        idx = np.arange(js, je + 1)
        if len(idx) == 0:
            break
        contrib = a[idx] * a[k_max - idx]
        top_srcs = idx[np.argsort(contrib)[::-1][:10]]

        # Find safe destinations (positions not contributing much to any peak)
        # Compute total contribution of each position to all peaks
        total_contribution = np.zeros(n)
        for k in range(2 * n - 1):
            js2 = max(0, k - (n - 1))
            je2 = min(n - 1, k)
            total_contribution[js2:je2 + 1] += b[k] * a[js2:je2 + 1] * a[k - js2:k - je2] / (b[k] + 1e-30)

        safe_dsts = np.argsort(total_contribution)[:15]

        improved = False
        for src in top_srcs:
            if time.monotonic() - t0 > time_limit - 3:
                break
            if a[src] < 1e-12:
                continue
            for dst in safe_dsts:
                if src == dst:
                    continue
                # Try several small transfers
                for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.05]:
                    t_move = a[src] * frac
                    if t_move < 1e-14:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= t_move
                    a_trial[dst] += t_move
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-12:
                        continue
                    a_trial = a_trial / st
                    new_max = fast_autocorr(a_trial).max()
                    if new_max < bmax * (1 - 1e-9):
                        a = a_trial
                        new_score = 2.0 * n * new_max
                        if new_score < best_score:
                            best_score = new_score
                            best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 50 == 0:
                # Try a random pair swap
                for _ in range(5):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.02)
                        amt = a[i] * frac
                        a_trial = a.copy()
                        a_trial[i] -= amt
                        a_trial[j] += amt
                        a_trial = np.maximum(a_trial, 0)
                        st = a_trial.sum()
                        if st > 1e-10:
                            a_trial = a_trial / st
                            sc = 2.0 * n * fast_autocorr(a_trial).max()
                            if sc < best_score:
                                a = a_trial
                                best_score = sc
                                best_a = a.copy()
                                improved = True
                                stall = 0
            if stall > 500:
                break

    return best_a, best_score


def propose_candidate(seed=42, time_budget=1800.0):
    """
    Multi-phase optimization combining LP game, pairwise transfer,
    subgradient descent, and local polish.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(time_budget) - 80.0, 1700.0)

    solver = find_solver()
    print(f"LP solver: {solver}, budget: {time_budget:.0f}s")

    # Get parent
    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: LP Nesterov at parent length ===
    n_base = len(best_seq) if best_seq is not None else 1600
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} ===")
    t1 = min(remaining * 0.35, 550.0)
    if t1 > 60:
        a_out, _ = nesterov_lp_optimize(best_seq, t1, solver)
        try_update(a_out, f"lp_n{n_base}")

    # === Phase 2: Pairwise transfer descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 2: Pairwise transfer n={n_base} ===")
    t2 = min(remaining * 0.20, 250.0)
    if t2 > 30:
        a_out, _ = pairwise_transfer_descent(best_seq, t2, seed=111)
        try_update(a_out, f"pairwise_n{n_base}")

    # === Phase 3: Subgradient descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 3: Subgradient n={n_base} ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 30:
        a_out, _ = subgradient_descent(best_seq, t3, eta0=0.03, seed=222)
        try_update(a_out, f"sgd_n{n_base}")

    # === Phase 4: Fine local search ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 4: Fine local n={n_base} ===")
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 20:
        a_out, _ = fine_local_search(best_seq, t4, seed=333)
        try_update(a_out, f"fine_n{n_base}")

    # === Phase 5: LP at slightly different length ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_long = n_base + 100
        print(f"=== Phase 5: LP n={n_long} ===")
        t5 = min(remaining * 0.15, 150.0)
        if t5 > 30:
            init_long = interpolate_sequence(best_seq, n_long)
            a_out, _ = nesterov_lp_optimize(init_long, t5, solver)
            try_update(a_out, f"lp_n{n_long}")

    # === Phase 6: Pairwise at longer length ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        print(f"=== Phase 6: Pairwise n={n_long if remaining > 0 else n_base} ===")
        t6 = min(remaining * 0.15, 100.0)
        if t6 > 20:
            a_out, _ = pairwise_transfer_descent(best_seq, t6, seed=444)
            try_update(a_out, "pairwise_2")

    # === Phase 7: Multi-start exploration at base length ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        print(f"=== Phase 7: Multi-start n={n_base} ===")
        t7 = min(remaining * 0.25, 250.0)
        if t7 > 60:
            a_out, _ = multi_start_optimize(n_base, t7, solver, rng)
            try_update(a_out, f"multistart_n{n_base}")

    # === Phase 8: Final polish sequence ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        print(f"=== Phase 8: Final polish ===")
        t8a = min(remaining * 0.4, 80.0)
        if t8a > 10:
            a_out, _ = pairwise_transfer_descent(best_seq, t8a, seed=555)
            try_update(a_out, "polish_pairwise")
        
        remaining = time_budget - (time.monotonic() - t_global)
        t8b = min(remaining * 0.3, 60.0)
        if t8b > 10:
            a_out, _ = fine_local_search(best_seq, t8b, seed=666)
            try_update(a_out, "polish_fine")

        remaining = time_budget - (time.monotonic() - t_global)
        t8c = min(remaining * 0.3, 40.0)
        if t8c > 10:
            a_out, _ = subgradient_descent(best_seq, t8c, eta0=0.01, seed=777)
            try_update(a_out, "polish_sgd")

    # === Phase 9: Ultimate final ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t9 = min(remaining * 0.8, 30.0)
        if t9 > 5:
            print(f"=== Phase 9: Ultimate ===")
            a_out, _ = fine_local_search(best_seq, t9, seed=888)
            try_update(a_out, "ultimate")

    # Final check
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
```
