def sca_lp_step(a0, solver):
    """One SCA iteration: solve the linearized LP and return the new point."""
    n = len(a0)
    b = fast_autocorr(a0)  # conv(a0, a0)
    
    # Build the LP
    # Variables: delta (n), M (1)
    delta = cp.Variable(n)
    M = cp.Variable()
    
    constraints = [
        cp.sum(delta) == 0,  # maintain sum(a) = 1
        a0 + delta >= 0,     # non-negativity
        M >= 0               # M is non-negative
    ]
    
    # For each lag k: b[k] + 2 * sum_j a0[k-j] * delta[j] <= M
    # The coefficient of delta[j] in constraint k is 2 * a0[k-j] (if 0 <= k-j < n)
    # We can vectorize this using the Toeplitz-like structure
    
    for k in range(2*n - 1):
        # Coefficient vector for delta in constraint k
        # coeff[j] = 2 * a0[k-j] if 0 <= k-j < n, else 0
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        coeff = np.zeros(n)
        if js <= je:
            coeff[js:je+1] = 2 * a0[k - (js:je+1)]  # a0[k-j] for j in js..je
        constraints.append(b[k] + coeff @ delta <= M)
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    prob.solve(solver=solver)
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    delta_val = np.asarray(delta.value, dtype=np.float64)
    M_val = float(M.value)
    return a0 + delta_val, M_val

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def find_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP, cp.GLPK]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_sca_lp(a0, b0, M_target, active_lags, solver):
    """
    Solve the SCA linearized LP.
    Minimize M s.t. b0[k] + 2*sum_j a0[k-j]*delta[j] <= M for active k
                   sum(delta) = 0, a0 + delta >= 0
    Returns (delta, M) or (None, None) on failure.
    """
    n = len(a0)
    delta = cp.Variable(n)
    M = cp.Variable()

    constraints = [
        cp.sum(delta) == 0,
        a0 + delta >= 0,
        M >= 0
    ]

    # Build constraints for active lags
    for k in active_lags:
        # Coefficient of delta[j] is 2*a0[k-j] if 0 <= k-j < n
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            constraints.append(b0[k] <= M)
            continue
        # coeff[j] = 2*a0[k-j] for j in [js, je]
        coeff = np.zeros(n)
        idx = np.arange(js, je + 1)
        coeff[idx] = 2.0 * a0[k - idx]
        constraints.append(b0[k] + coeff @ delta <= M)

    prob = cp.Problem(cp.Minimize(M), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    delta_val = np.asarray(delta.value, dtype=np.float64)
    M_val = float(M.value)
    return delta_val, M_val


def sca_optimize(a0, time_limit, solver, max_iters=5000):
    """
    Successive Convex Approximation optimization.
    At each step, linearize the constraint at the current point and solve the LP.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    step_size = 0.5
    stall_count = 0
    consecutive_failures = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        M_curr = b.max()

        # Select active lags (within 5% of max)
        threshold = M_curr * 0.95
        active_lags = np.where(b >= threshold)[0]
        if len(active_lags) == 0:
            active_lags = np.argsort(b)[::-1][:20]
        # Limit to at most 300 active lags for LP efficiency
        if len(active_lags) > 300:
            top_idx = np.argsort(b[active_lags])[::-1][:300]
            active_lags = active_lags[top_idx]

        delta, M_lp = solve_sca_lp(a, b, M_curr, active_lags, solver)
        if delta is None:
            consecutive_failures += 1
            if consecutive_failures > 10:
                break
            continue
        consecutive_failures = 0

        # Damped step
        a_new = a + step_size * delta
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score_new = 2.0 * n * fast_autocorr(a_new).max()

        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            stall_count = 0
            # Increase step size slightly
            step_size = min(1.0, step_size * 1.1)
        else:
            stall_count += 1
            # Decrease step size
            step_size = max(0.01, step_size * 0.8)

        a = a_new

        # Periodic check of average
        if t % 50 == 0:
            if stall_count > 200:
                # Restart from best
                a = best_a.copy()
                step_size = 0.5
                stall_count = 0

        if t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SCA t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_curr:.8f}, step={step_size:.3f}")

    return best_a, best_score


def coordinate_descent_optimize(a0, time_limit, seed=42, n_sweeps=1000):
    """
    Exact coordinate descent: for each coordinate, find the maximum feasible value
    given the current max constraint, and update.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    b = fast_autocorr(a)
    M = b.max()

    # We work with the constraint max_k conv(a,a)[k] <= M
    # and try to reduce M iteratively
    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit - 2:
            break

        # Every 20 sweeps, recompute M and check best
        if sweep % 20 == 0:
            b = fast_autocorr(a)
            M = b.max()
            score = 2.0 * n * M
            if score < best_score:
                best_score = score
                best_a = a.copy()

        # Random or cyclic order
        if sweep % 50 == 0:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for i in order:
            x = a[i]
            if x < 1e-15:
                continue

            # Compute the maximum feasible x for coordinate i
            # Constraints: for each k, conv(a,a)[k] with a[i] changed to x' must be <= M
            # conv(a,a)[k] - 2*x*a[k-i] + 2*x'*a[k-i] <= M (for k != 2i, k-i valid)
            # conv(a,a)[k] - x^2 + x'^2 <= M (for k = 2i)

            x_max = 1000.0  # upper bound

            # Constraint from k = 2i (if valid, i.e., 2i <= 2n-2)
            k_diag = 2 * i
            if k_diag <= 2 * n - 2:
                rhs = M - b[k_diag] + x * x
                if rhs >= 0:
                    x_max = min(x_max, np.sqrt(rhs))
                else:
                    # Current point is infeasible (shouldn't happen)
                    x_max = min(x_max, x)

            # Constraints from k where 0 <= k-i < n and k != 2i
            # conv[k] - 2*x*a[k-i] + 2*x'*a[k-i] <= M
            # x' <= (M - conv[k] + 2*x*a[k-i]) / (2*a[k-i])
            js = max(0, i - (n - 1))  # k - i >= 0 => k >= i, and k <= 2n-2
            # Actually k ranges from 0 to 2n-2, and k-i must be in [0, n-1]
            # So k ranges from i to i+n-1, intersected with [0, 2n-2]
            k_start = max(0, i)
            k_end = min(2 * n - 2, i + n - 1)

            for k in range(k_start, k_end + 1):
                if k == 2 * i:
                    continue  # already handled
                kj = k - i  # 0 <= kj < n guaranteed by range
                aj = a[kj]
                if aj < 1e-15:
                    continue
                rhs = M - b[k] + 2.0 * x * aj
                if rhs < 0:
                    x_max = min(x_max, x)  # infeasible, keep current
                else:
                    x_max = min(x_max, rhs / (2.0 * aj))

            x_max = max(0.0, x_max)

            # Also need to consider: if we increase a[i], we should decrease others
            # but for now, just clamp to x_max
            # Try to set a[i] to x_max (or close to it)
            x_new = x_max
            if x_new > x * 1.01:
                # We can increase; but we need to maintain sum = 1
                # For now, just set it and renormalize later
                a[i] = x_new
            elif x_new < x * 0.99:
                # Must decrease
                a[i] = x_new

        # Renormalize
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
            continue
        a = a / s
        a = np.maximum(a, 0)

    # Final score
    s = a.sum()
    if s > 1e-15:
        a = a / s
        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_a, best_score


def lse_gradient_descent(a0, time_limit, T_start=50.0, T_end=0.5, n_stages=20, topk=30):
    """
    Top-k LogSumExp gradient descent with Adam, multi-stage temperature annealing.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        # Adaptive learning rate
        base_lr = 5e-3 * min(1.0, 200.0 / n)
        lr = base_lr * (0.9 ** si)

        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.8, 0.999, 1e-8

        max_it = max(30, int(min(remaining, 60.0) * 8))

        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 2:
                break

            a = softmax_fn(z)
            b = fast_autocorr(a)

            # Top-k LSE
            k_use = min(topk, len(b))
            top_idx = np.argsort(b)[::-1][:k_use]
            top_vals = b[top_idx]

            # LSE value
            max_tv = top_vals.max()
            exp_tv = np.exp(top_vals - max_tv)
            lse_val = (max_tv + np.log(exp_tv.sum())) / stage_T
            w = exp_tv / exp_tv.sum()

            # Gradient
            grad_a = np.zeros(n)
            for idx in range(k_use):
                K = top_idx[idx]
                wK = w[idx]
                js = max(0, K - (n - 1))
                je = min(n - 1, K)
                if js <= je:
                    idx_range = np.arange(js, je + 1)
                    grad_a[idx_range] += 2.0 * wK * a[K - idx_range]

            # Transform to z-space
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)

            # Adam update
            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            if (it + 1) % 10 == 0:
                a_check = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a_check).max()
                if score < best_score:
                    best_score = score
                    best_a = a_check.copy()
                    if (it + 1) % 50 == 0:
                        z = np.log(np.maximum(best_a, 1e-30))
                        m_z = np.zeros(n)
                        v_z = np.zeros(n)
                        t_adam = 0

        a_stage = softmax_fn(z)
        score_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()

        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE stage {si}/{n_stages} (T={stage_T:.1f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def peak_equalization(a0, time_limit, seed=123):
    """
    Iterative peak equalization: identify the top peaks and redistribute mass
    from contributing elements to flat elements.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 3:
        b = fast_autocorr(a)
        bmax = b.max()

        # Find top peaks
        n_peaks = min(20, len(b))
        top_peaks = np.argsort(b)[::-1][:n_peaks]

        # For each peak, find the most contributing element pair
        improved = False
        for k in top_peaks[:8]:
            js = max(0, k - (n - 1))
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-18:
                continue

            # The most contributing element at src
            src = idx[np.argmax(contrib)]
            partner = k - src

            # Find safe destination (low a value, not contributing to any top peak)
            danger = np.zeros(n, dtype=bool)
            for kp in top_peaks[:10]:
                jsp = max(0, kp - (n - 1))
                jep = min(n - 1, kp)
                danger[jsp:jep + 1] = True

            safe = np.where(~danger)[0]
            if len(safe) == 0:
                safe = np.where(np.arange(n) != src)[0]
            safe_sorted = safe[np.argsort(a[safe])]
            dst_pool = safe_sorted[:max(3, len(safe_sorted) // 5)]
            if len(dst_pool) == 0:
                continue

            # Try redistributing from src to dst
            found = False
            for dst in dst_pool[:5]:
                if dst == src:
                    continue
                for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-12:
                        continue
                    a_trial = a_trial / st
                    new_max = fast_autocorr(a_trial).max()
                    if new_max < bmax * (1 - 1e-9):
                        a = a_trial
                        score = 2.0 * n * new_max
                        if score < best_score:
                            best_score = score
                            best_a = a.copy()
                        improved = True
                        found = True
                        stall = 0
                        break
                if found:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 50 == 0:
                # Random perturbation
                n_pert = rng.integers(1, 4)
                a_trial = a.copy()
                for _ in range(n_pert):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a_trial[i] > 1e-10:
                        frac = rng.uniform(0.002, 0.02)
                        amt = a_trial[i] * frac
                        a_trial[i] -= amt
                        a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-10:
                    a_trial = a_trial / st
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > 300:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 10)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 4)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 3
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 2
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.7 + 0.2 * np.cos(2 * x) + 0.1 * np.cos(5 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-8 * (x - 0.25) ** 2) + 0.6 * np.exp(-8 * (x - 0.75) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * np.cos(np.pi * x / 3) ** 2
    elif choice == 7:
        x = np.linspace(-1, 1, n)
        a = np.maximum(0, 1 - np.abs(x) ** 1.5)
    elif choice == 8:
        x = np.linspace(0, 1, n)
        a = np.exp(-50 * x ** 2) + np.exp(-50 * (x - 1) ** 2)
    else:
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * (x - 0.5) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def perturbed_init(a0, rng, scale=0.08):
    """Create a perturbed version of a sequence."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a0)) / len(a0)
    a = a / s
    noise = rng.normal(0, scale, len(a))
    a_new = a + noise * (a + 1e-6)
    a_new = np.maximum(a_new, 0)
    sn = a_new.sum()
    if sn < 1e-15:
        return a.copy()
    return a_new / sn


def mirror_descent_refine(a0, time_limit, eta=0.01, seed=999):
    """
    Multiplicative weights (mirror descent) refinement.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    current_eta = eta
    while time.monotonic() - t0 < time_limit - 3:
        b = fast_autocorr(a)
        bmax = b.max()
        threshold = bmax * 0.98
        top_peaks = np.where(b >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(b)[::-1][:5]

        importance = np.zeros(n)
        for k in top_peaks[:12]:
            js = max(0, k - (n - 1))
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            importance[idx] += a[k - idx]

        imax = importance.max()
        if imax > 1e-15:
            importance = importance / imax

        updates = 1.0 - current_eta * importance
        updates = np.maximum(updates, 0.01)
        updates = updates / updates.mean()
        a_new = a * updates
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new / sn

        score_new = 2.0 * n * fast_autocorr(a_new).max()
        if score_new < best_score:
            a = a_new
            best_score = score_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall % 50 == 0:
                current_eta /= 2
                if current_eta < 1e-6:
                    current_eta = eta
                    a = best_a.copy()
                    noise = rng.normal(0, 0.005, n)
                    a = a * (1 + noise)
                    a = np.maximum(a, 0)
                    sn = a.sum()
                    if sn > 1e-10:
                        a = a / sn

            if stall > 400:
                break

    return best_a, best_score


def propose_candidate(seed=42, time_budget=1800.0):
    """
    Multi-phase optimization combining SCA, coordinate descent, LSE gradient,
    peak equalization, mirror descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(time_budget) - 120.0, 1680.0)

    solver = find_solver()
    print(f"LP solver: {solver}, budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    n_base = len(best_seq) if best_seq is not None else 1600

    # === Phase 1: SCA at parent length ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.30, 450.0)
    if t1 > 60:
        print(f"\n=== Phase 1: SCA n={n_base} (budget={t1:.0f}s) ===")
        a_out, _ = sca_optimize(best_seq, t1, solver)
        try_update(a_out, f"sca_n{n_base}")

    # === Phase 2: LSE gradient at parent length ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.18, 250.0)
    if t2 > 30:
        print(f"\n=== Phase 2: LSE n={n_base} (budget={t2:.0f}s) ===")
        a_out, _ = lse_gradient_descent(best_seq, t2, T_start=40.0, T_end=0.3, n_stages=25, topk=30)
        try_update(a_out, f"lse_n{n_base}")

    # === Phase 3: Coordinate descent at parent length ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 30:
        print(f"\n=== Phase 3: CoordDescent n={n_base} (budget={t3:.0f}s) ===")
        a_out, _ = coordinate_descent_optimize(best_seq, t3, seed=777)
        try_update(a_out, f"coord_n{n_base}")

    # === Phase 4: Peak equalization ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 15:
        print(f"\n=== Phase 4: PeakEq n={n_base} (budget={t4:.0f}s) ===")
        a_out, _ = peak_equalization(best_seq, t4, seed=333)
        try_update(a_out, "peq_1")

    # === Phase 5: Mirror descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.08, 100.0)
    if t5 > 15:
        print(f"\n=== Phase 5: Mirror n={n_base} (budget={t5:.0f}s) ===")
        a_out, _ = mirror_descent_refine(best_seq, t5, eta=0.005, seed=555)
        try_update(a_out, "mirror_1")

    # === Phase 6: SCA at longer length ===
    n_long = n_base + 200
    remaining = time_budget - (time.monotonic() - t_global)
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 40:
        print(f"\n=== Phase 6: SCA n={n_long} (budget={t6:.0f}s) ===")
        init_long = interpolate_sequence(best_seq, n_long)
        a_out, _ = sca_optimize(init_long, t6, solver)
        try_update(a_out, f"sca_n{n_long}")

    # === Phase 7: LSE at longer length ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 40:
        n_long2 = n_base + 350
        t7 = min(remaining * 0.10, 130.0)
        if t7 > 20:
            print(f"\n=== Phase 7: LSE n={n_long2} (budget={t7:.0f}s) ===")
            init_long2 = interpolate_sequence(best_seq, n_long2)
            a_out, _ = lse_gradient_descent(init_long2, t7, T_start=30.0, T_end=1.0, n_stages=15, topk=25)
            try_update(a_out, f"lse_n{n_long2}")

    # === Phase 8: Coordinate descent at best found so far ===
    remaining = time_budget - (time.monotonic() - t_global)
    t8 = min(remaining * 0.10, 120.0)
    if t8 > 20:
        print(f"\n=== Phase 8: CoordDescent final n={len(best_seq)} (budget={t8:.0f}s) ===")
        a_out, _ = coordinate_descent_optimize(best_seq, t8, seed=999)
        try_update(a_out, f"coord_final_n{len(best_seq)}")

    # === Phase 9: Multi-start with perturbation ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 9: Multi-start ===")
    t9_total = min(remaining * 0.12, 100.0)
    if t9_total > 30:
        for si in range(3):
            if time.monotonic() - t_global > t_global + time_budget - 50:
                break
            t9_each = t9_total / 3
            pert = perturbed_init(best_seq, rng, scale=0.05 * (si + 1))
            a_out, _ = lse_gradient_descent(pert, t9_each, T_start=15.0, T_end=2.0, n_stages=8, topk=20)
            try_update(a_out, f"multi_{si}")

    # === Phase 10: Final SCA polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        t10 = min(remaining * 0.25, 80.0)
        if t10 > 20:
            print(f"\n=== Phase 10: Final SCA polish ===")
            a_out, _ = sca_optimize(best_seq, t10, solver)
            try_update(a_out, "sca_final")

    # Final check
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
