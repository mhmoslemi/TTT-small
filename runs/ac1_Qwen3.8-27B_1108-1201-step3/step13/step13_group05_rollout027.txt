def fast_autocorr(a):

def compute_exact_subgrad(a, ac, tol=1e-10):
    """Compute exact subgradient using all lags where ac[k] >= M - tol."""
    n = len(a)
    M = ac.max()
    active = np.where(ac >= M - tol)[0]
    if len(active) == 0:
        active = np.argmax(ac)
        active = [active]
    
    grad = np.zeros(n)
    for k in active:
        k = int(k)
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            grad[j] += 2.0 * a[k - j]
    grad /= len(active)
    return grad, M

def compute_weighted_subgrad(a, ac, top_k=10, beta=5.0):
    """Compute weighted subgradient using top-k lags with exponential weighting."""
    n = len(a)
    M = ac.max()
    # Use lags within a tolerance of the max
    tol = 0.01 * M
    candidates = np.where(ac >= M - tol)[0]
    if len(candidates) == 0:
        candidates = np.argmax(ac)
        candidates = np.array([candidates])
    
    # Weight by (ac[k] - M + tol) or just by ac[k]
    w = ac[candidates] ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(candidates)) / len(candidates)
    else:
        w = w / w_sum
    
    grad = np.zeros(n)

        k = int(k)
        for j in range(max(0, k - n + 1), min(n, k + 1)):

def simulated_annealing(a0, time_limit, verbose=True):
    """Simulated annealing with pairwise mass transfers."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * c / n
    else:
        a = a * c / s0
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    T_start = 0.05 * best_m
    T_end = 0.001 * best_m
    total_time = time_limit
    
    ac = fast_autocorr(a)
    M = ac.max()
    
    n_steps = 0
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = elapsed / time_limit
        T = T_start * (T_end / T_start) ** frac
        
        # Pick random positions
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j or a[i] < 1e-15:
            continue
        
        # Pick amount to transfer
        frac_amt = rng.uniform(0.001, 0.02)
        amt = min(a[i] * frac_amt, a[i] * 0.5)
        
        if amt < 1e-20:
            continue
        
        a[i] -= amt
        a[j] += amt
        a = a * c / a.sum()
        
        ac = fast_autocorr(a)
        M_new = ac.max()
        delta = M_new - M
        
        if delta < 0 or rng.uniform() < np.exp(-delta / max(T, 1e-30)):
            M = M_new
            if M_new < best_m:
                best_m = M_new
                best_a = a.copy()
                if verbose and n_steps % 10000 == 0:
                    print(f"    SA step={n_steps} ({elapsed:.1f}s): J={2.0*n*best_m/(c*c):.6f}, T={T:.6f}")
        else:
            a[i] += amt
            a[j] -= amt
            a = a * c / a.sum()
        
        n_steps += 1
    
    if verbose:
        print(f"    SA done ({time.monotonic()-t0:.1f}s): steps={n_steps}, J={2.0*n*best_m/(c*c):.6f}")
    
    return best_a, best_m

def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. conv(a,b)_k <= M for all k, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    
    # Build constraint matrix efficiently using FFT trick
    # T @ b = conv(a, b) where T is a Toeplitz-like matrix
    # For large n, we can use the structure
    
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated LP best-response."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * c / n
    else:
        a = a * c / s0
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * c / b_sum
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * c / s_y
        
        if t == 1:
            x_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new * c / s_x
        
        y = y_new
        x = x_new
        
        score_y = fast_autocorr(y).max()
        if score_y < best_m:
            best_m = score_y
            best_a = y.copy()
        
        score_x = fast_autocorr(x).max()
        if score_x < best_m:
            best_m = score_x
            best_a = x.copy()
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c * c)
            print(f"    LP iter={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def spatial_sa(a0, time_limit, verbose=True):
    """Spatial simulated annealing: prefer neighboring positions for exchanges."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * c / n
    else:
        a = a * c / s0
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    
    T_start = 0.02 * best_m
    T_end = 0.0005 * best_m
    
    ac = fast_autocorr(a)
    M = ac.max()
    n_steps = 0
    n_accept = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(elapsed / time_limit, 1.0)
        T = T_start * (T_end / T_start) ** frac
        
        # Pick position i weighted by value
        # Use a medium-range spatial distance
        i = rng.integers(n)
        if a[i] < 1e-15:
            # Pick a random non-zero position
            nonzero = np.where(a > 1e-15)[0]
            if len(nonzero) == 0:
                break
            i = nonzero[rng.integers(len(nonzero))]
        
        # Pick j with spatial preference (closer positions more likely)
        dist = rng.exponential(scale=n // 10)
        j = (i + int(dist) * (1 if rng.uniform() < 0.5 else -1)) % n
        
        if j == i:
            continue
        
        frac_amt = rng.uniform(0.002, 0.015)
        amt = a[i] * frac_amt
        
        if amt < 1e-20:
            continue
        
        a[i] -= amt
        a[j] += amt
        a = a * c / a.sum()
        
        ac = fast_autocorr(a)
        M_new = ac.max()
        delta = M_new - M
        
        if delta < 0 or rng.uniform() < np.exp(-delta / max(T, 1e-30)):
            M = M_new
            n_accept += 1
            if M_new < best_m:
                best_m = M_new
                best_a = a.copy()
        else:
            a[i] += amt
            a[j] -= amt
            a = a * c / a.sum()
        
        n_steps += 1
        
        if verbose and n_steps % 20000 == 0:
            print(f"    SpSA step={n_steps} ({elapsed:.1f}s): J={2.0*n*best_m/(c*c):.6f}, T={T:.6f}, acc={n_accept/n_steps:.3f}")
    
    if verbose:
        print(f"    SpSA done ({time.monotonic()-t0:.1f}s): steps={n_steps}, acc={n_accept/max(n_steps,1):.3f}, J={2.0*n*best_m/(c*c):.6f}")
    
    return best_a, best_m

def greedy_coord_descent(a0, time_limit, verbose=True):
    """Greedy coordinate descent: for each worst lag, find best single-coordinate move."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * c / n
    else:
        a = a * c / s0
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(555)
    
    ac = fast_autocorr(a)
    M = ac.max()
    
    stall = 0
    while time.monotonic() - t0 < time_limit and stall < 100:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:12]
        improved = False
        
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            
            # Find source: position with highest contribution to this lag
            src = idx[np.argmax(contrib)]
            
            # Find destination: position that minimally contributes to all top lags
            total_contrib = np.zeros(n)
            for k2 in top_k:
                k2 = int(k2)
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                idx2 = np.arange(js2, je2 + 1)
                if len(idx2) > 0:
                    total_contrib[idx2] += a[k2 - idx2]
            
            # Exclude positions near the source
            exclude = set(range(max(0, src - 5), min(n, src + 6)))
            exclude.add(src)
            
            candidates = [j for j in range(n) if j not in exclude and a[j] < a[src]]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            
            # Sort by total contribution (want low)
            candidates.sort(key=lambda j: total_contrib[j])
            
            # Try a few candidates
            tried = 0
            for dst in candidates[:5]:
                if dst == src:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.04, 0.06, 0.1]:
                    amt = a[src] * frac
                    if amt < 1e-20:
                        break
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        break
                    a_trial = a_trial * c / a_trial.sum()
                    ac_trial = fast_autocorr(a_trial)
                    M_trial = ac_trial.max()
                    if M_trial < best_m - 1e-15:
                        a = a_trial
                        best_m = M_trial
                        best_a = a.copy()
                        ac = ac_trial
                        M = M_trial
                        improved = True
                        stall = 0
                        tried += 1
                        break
                if improved:
                    break
        
        if not improved:
            stall += 1
            # Random perturbation to escape local minimum
            for _ in range(20):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > c * 1e-6:
                    amt = a[i] * rng.uniform(0.01, 0.05)
                    a[i] -= amt
                    a[j] += amt
                    a = a * c / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        break
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = a * c / a.sum()
        
        if verbose and (stall == 1 or stall % 20 == 0):
            elapsed = time.monotonic() - t0
            print(f"    GCD ({elapsed:.1f}s): J={2.0*n*best_m/(c*c):.6f}, stall={stall}")
    
    if verbose:
        print(f"    GCD done ({time.monotonic()-t0:.1f}s): J={2.0*n*best_m/(c*c):.6f}")
    
    return best_a, best_m

def spectral_interpolate(f, new_n):
    """Interpolate using spectral method (preserves frequency content better)."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    
    # Use DFT-based interpolation
    fft_size_old = 1 << (2 * old_n).bit_length()
    F = rfft(f_norm, fft_size_old)
    # Zero-pad in frequency domain
    fft_size_new = 1 << (2 * new_n).bit_length()
    F_new = np.zeros(fft_size_new // 2 + 1, dtype=np.complex128)
    F_new[:len(F)] = F
    f_interp = irfft(F_new, fft_size_new)[:new_n]
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

            avg_check = np.maximum(avg_check, 0.0)
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * c / s_avg
                M_avg = fast_autocorr(avg_check).max()
                if M_avg < best_m:
                    best_m = M_avg
                    best_a = avg_check.copy()
        
        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c * c)
            print(f"    MD-exact t={t} ({elapsed:.1f}s): J={J:.6f}, |active|={len(active)}")
    
    return best_a, best_m

def spatial_sa(a0, time_limit, seed=42, verbose=True):
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * c / n
    else:
        a = a * c / s0
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    T_start = 0.03 * best_m
    T_end = 0.0005 * best_m
    
    ac = fast_autocorr(a)
    M = ac.max()
    n_steps = 0
    n_accept = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(elapsed / time_limit, 1.0)
        T = T_start * (T_end / T_start) ** frac
        
        i = rng.integers(n)
        if a[i] < 1e-15:
            nonzero = np.where(a > 1e-15)[0]
            if len(nonzero) == 0:
                break
            i = nonzero[rng.integers(len(nonzero))]
        
        # Spatial distance: exponential distribution
        dist = int(rng.exponential(scale=n // 8))
        direction = 1 if rng.uniform() < 0.5 else -1
        j = (i + dist * direction) % n
        
        if j == i:
            continue
        
        frac_amt = rng.uniform(0.001, 0.01)
        amt = a[i] * frac_amt
        if amt < 1e-20:
            continue
        
        a[i] -= amt
        a[j] += amt
        a = a * c / a.sum()
        
        ac_new = fast_autocorr(a)
        M_new = ac_new.max()
        delta = M_new - M
        
        if delta < 0 or rng.uniform() < np.exp(-delta / max(T, 1e-30)):
            ac = ac_new
            M = M_new
            n_accept += 1
            if M_new < best_m:
                best_m = M_new
                best_a = a.copy()
        else:
            a[i] += amt
            a[j] -= amt
            a = a * c / a.sum()
        
        n_steps += 1
        
        if verbose and n_steps % 10000 == 0:
            J_cur = 2.0 * n * best_m / (c * c)
            print(f"    SA step={n_steps} ({elapsed:.1f}s): J={J_cur:.6f}, T={T:.8f}, acc={n_accept/max(n_steps,1):.3f}")
    
    if verbose:
        J_final = 2.0 * n * best_m / (c * c)
        print(f"    SA done ({time.monotonic()-t0:.1f}s): steps={n_steps}, acc={n_accept/max(n_steps,1):.3f}, J={J_final:.6f}")
    
    return best_a, best_m

def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * c / n
    else:
        a = a * c / s0
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    y = a.copy()
    x = a.copy()
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * c / b_sum
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * c / s_y
        
        if t == 1:
            x_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new * c / s_x
        
        y = y_new
        x = x_new
        
        score_y = fast_autocorr(y).max()
        if score_y < best_m:
            best_m = score_y
            best_a = y.copy()
        score_x = fast_autocorr(x).max()
        if score_x < best_m:
            best_m = score_x
            best_a = x.copy()
        
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c * c)
            print(f"    LP iter={t} ({elapsed:.1f}s): J={J:.6f}")
    
    return best_a, best_m

def greedy_cd(a0, time_limit, verbose=True):
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * c / n
    else:
        a = a * c / s0
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    stall = 0
    
    while time.monotonic() - t0 < time_limit and stall < 150:
        ac = fast_autocorr(a)
        top_k_idx = np.argsort(ac)[::-1][:10]
        improved = False
        
        for k in top_k_idx:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            src = idx[np.argmax(contrib)]
            
            # Compute total contribution to top lags for each position
            total_contrib = np.zeros(n)
            for k2 in top_k_idx:
                k2 = int(k2)
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                idx2 = np.arange(js2, je2 + 1)
                if len(idx2) > 0:
                    total_contrib[idx2] += a[k2 - idx2]
            
            # Find best destination
            exclude_set = set(range(max(0, src - 3), min(n, src + 4)))
            best_dst = -1
            best_tc = float('inf')
            for j in range(n):
                if j in exclude_set or a[j] >= a[src]:
                    continue
                if total_contrib[j] < best_tc:
                    best_tc = total_contrib[j]
                    best_dst = j
            
            if best_dst < 0:
                continue
            
            for frac in [0.003, 0.005, 0.01, 0.02, 0.03, 0.05, 0.08]:
                amt = a[src] * frac
                if amt < 1e-20:
                    break
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                if a_trial[src] < 0:
                    break
                a_trial = a_trial * c / a_trial.sum()
                ac_trial = fast_autocorr(a_trial)
                M_trial = ac_trial.max()
                if M_trial < best_m - 1e-15:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        
        if not improved:
            stall += 1
            # Random perturbation
            for _ in range(30):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > c * 1e-6:
                    amt = a[i] * rng.uniform(0.005, 0.03)
                    a[i] -= amt
                    a[j] += amt
                    a = a * c / a.sum()
                    M_new = fast_autocorr(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
                        break
                    else:
                        a[i] += amt
                        a[j] -= amt
                        a = a * c / a.sum()
        
        if verbose and (stall == 1 or stall % 30 == 0):
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m / (c * c)
            print(f"    GCD ({elapsed:.1f}s): J={J:.6f}, stall={stall}")
    
    return best_a, best_m

def spectral_interp(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    fft_size_old = 1 << (2 * old_n).bit_length()
    F = rfft(f_norm, fft_size_old)
    fft_size_new = 1 << (2 * new_n).bit_length()
    F_new = np.zeros(fft_size_new // 2 + 1, dtype=np.complex128)
    F_new[:len(F)] = F
    f_interp = irfft(F_new, fft_size_new)[:new_n]
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def update_best(a, best_seq, best_score, label):
    """Return updated (best_seq, best_score) if a is better."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_safe(a_norm)
    if np.isfinite(sc) and sc < best_score:
        elapsed = 0  # will be printed by caller
        return a_norm.copy(), sc
    return best_seq, best_score
