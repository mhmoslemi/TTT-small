def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    if not _HAS_CVXPY:
        return None
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64).ravel()
        a_list = [float(x) for x in a_arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def build_conv_matrix(a, n):
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    return T


def lp_best_response(a, M, solver):
    n = len(a)
    if solver is None or not _HAS_CVXPY:
        return None
    b_var = cp.Variable(n, nonneg=True)
    size = 2 * n - 1
    T = build_conv_matrix(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def qp_best_response(a, M, mu, solver):
    """Solve proximal QP: min ||b - (a+mu)||^2/(2mu) s.t. T@b <= M, b>=0"""
    n = len(a)
    if solver is None or not _HAS_CVXPY:
        return None
    b_var = cp.Variable(n, nonneg=True)
    size = 2 * n - 1
    T = build_conv_matrix(a, n)
    target = a + mu
    obj = cp.quad_form(b_var - target, mu)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Minimize(obj), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, step_mode='standard'):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    last_improve = t0
    stall_count = 0
    best_M = best_score

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        if step_mode == 'adaptive':
            ratio = min(M_curr / max(best_M, 1e-10), 2.0)
            alpha = (2.0 / (t + 2.0)) * ratio
            alpha = min(alpha, 1.0)
        elif step_mode == 'slow':
            alpha = 1.0 / np.sqrt(t + 1.0)
            alpha = min(alpha, 1.0)
        else:
            alpha = 2.0 / (t + 2.0)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()
        if score_y < best_M:
            best_M = score_y

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()

        avg_a += a
        n_avg += 1
        if n_avg % 10 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    last_improve = time.monotonic()

        if time.monotonic() - last_improve > 45.0:
            stall_count += 1
            if stall_count > 15:
                break
        else:
            stall_count = 0

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, M={M_curr:.8f}")

    return best_a, best_score


def swap_polish(a0, time_limit, seed=42):
    """Fast O(n) pairwise swap coordinate descent."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    conv = fast_autocorr(a)
    best_score = conv.max()
    t0 = time.monotonic()
    stall = 0
    n_peaks = min(12, n)

    while time.monotonic() - t0 < time_limit:
        peak_indices = np.argsort(conv)[::-1][:n_peaks]
        current_max = best_score
        improved = False

        for pk in peak_indices:
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[pk - idx]
            top_src = idx[np.argsort(contrib)[::-1][:6]]

            # Build peak mask (lags where conv is close to max)
            threshold = current_max * 0.95
            peak_mask = conv > threshold

            for src in top_src:
                if a[src] < 1e-10:
                    continue
                # Compute delta_conv for moving from src to dst
                # delta_conv[k] = a[k-src] + a[src-k] - a[k-dst] - a[dst-k]
                dst_candidates = np.where(~peak_mask)[0]
                if len(dst_candidates) == 0:
                    dst_candidates = np.arange(n)
                # Prioritize low values (destinations with small mass)
                dst_candidates = dst_candidates[np.argsort(a[dst_candidates])][:min(20, len(dst_candidates))]

                for dst in dst_candidates:
                    if dst == src:
                        continue
                    # Compute the change vector d_k
                    d_k = np.zeros(2 * n - 1)
                    # Contribution from removing at src: -a[k-src] - a[src-k]
                    valid1 = (0 <= pk - src + np.arange(2*n-1) - src) & (pk - src + np.arange(2*n-1) - src >= 0) & (pk - src + np.arange(2*n-1) - src < n)
                    # Simpler: just compute directly
                    for k in range(len(conv)):
                        v = 0.0
                        j1 = k - src
                        if 0 <= j1 < n:
                            v += a[j1]
                        j2 = src - k
                        if 0 <= j2 < n:
                            v += a[j2]
                        j3 = k - dst
                        if 0 <= j3 < n:
                            v -= a[j3]
                        j4 = dst - k
                        if 0 <= j4 < n:
                            v -= a[j4]
                        d_k[k] = v

                    # Find best delta
                    # We want: max_k (conv[k] + delta * d_k[k]) < current_max
                    # For each k: delta * d_k[k] < current_max - conv[k]
                    positive_mask = d_k > 0
                    if positive_mask.any():
                        max_allowed = (current_max - conv[positive_mask] - 1e-12) / d_k[positive_mask]
                        max_allowed = max_allowed[max_allowed > 0]
                        if len(max_allowed) == 0:
                            continue
                        delta_max_allowed = min(max_allowed) * 0.99
                    else:
                        delta_max_allowed = a[src]

                    delta_max_allowed = min(delta_max_allowed, a[src] * 0.8)
                    if delta_max_allowed < 1e-10:
                        continue

                    # Also check it actually helps at the peak
                    if d_k[pk] >= 0:
                        continue

                    for frac in [0.5, 0.3, 0.15]:
                        delta = delta_max_allowed * frac
                        if delta < 1e-12:
                            continue
                        # Check new max
                        new_conv = conv + delta * d_k
                        new_max = new_conv.max()
                        if new_max < current_max - 1e-10:
                            a[src] -= delta
                            a[dst] += delta
                            a = np.maximum(a, 0.0)
                            s_a = a.sum()
                            a = a * norm / s_a
                            conv = fast_autocorr(a)
                            current_max = conv.max()
                            best_score = min(best_score, current_max)
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 5 == 0:
                # Random perturbation to escape
                for _ in range(50):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i == j or a[i] < 1e-10:
                        continue
                    frac = rng.uniform(0.01, 0.05)
                    delta = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= delta
                    a_trial[j] += delta
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    a_trial = a_trial * norm / s_t
                    sc = fast_autocorr(a_trial).max()
                    if sc < best_score:
                        a = a_trial
                        conv = fast_autocorr(a)
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
            if stall > 300:
                break

    return best_a, best_score


def lse_objective(a_flat, n, T_temp):
    a = np.maximum(a_flat, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10
    exp_vals = np.exp(T_temp * (b - max_b))
    lse = max_b + np.log(exp_vals.sum()) / T_temp
    return 2.0 * n * lse / (s * s)


def lse_gradient(a_flat, n, T_temp):
    a = np.maximum(a_flat, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return np.zeros(n)
    exp_vals = np.exp(T_temp * (b - max_b))
    w = exp_vals / exp_vals.sum()

    # Compute cross-correlation of a and w via FFT
    fft_size = 1 << ((2 * n - 1) - 1).bit_length()
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    w_padded = np.zeros(fft_size)
    w_padded[:2 * n - 1] = w
    A_fft = rfft(a_padded, fft_size)
    W_fft = rfft(w_padded, fft_size)
    cross = irfft(A_fft * np.conj(W_fft), fft_size)[:n]

    conv_grad = 4.0 * cross
    obj_val = 2.0 * n * (max_b + np.log(exp_vals.sum()) / T_temp) / (s * s)
    grad = conv_grad / (s * s) - 2.0 * obj_val * a / s
    return grad


def lsc_scipy_optimize(a0, time_limit, T_start=30.0, T_end=1.0, n_stages=6):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        stage_time = min(remaining / max(1, n_stages - si), 180.0)

        def obj_func(x):
            return lse_objective(x, n, T)

        def grad_func(x):
            return lse_gradient(x, n, T)

        try:
            res = scipy_minimize(obj_func, a.copy(), method='L-BFGS-B',
                                  jac=grad_func, bounds=[(0, None)] * n,
                                  options={'maxiter': 300, 'maxfun': 1000})
            if res.x is not None:
                a_candidate = np.maximum(res.x, 0.0)
                s_c = a_candidate.sum()
                if s_c > 1e-10:
                    a_candidate = a_candidate * norm / s_c
                    score_c = fast_autocorr(a_candidate).max()
                    if score_c < best_score:
                        best_score = score_c
                        best_a = a_candidate.copy()
                        a = a_candidate
        except Exception:
            pass

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def structured_init(n, rng):
    choice = rng.integers(0, 5)
    if choice == 0:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.2
    elif choice == 1:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x)
    elif choice == 2:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-3.0 * (x - 0.7) ** 2)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * 3 * x)
    else:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 4.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver={solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            norm_p = np.sqrt(2.0 * len(parent))
            parent_normed = parent * norm_p / parent.sum()
            p_score = fast_autocorr(parent_normed).max()
            print(f"Parent: n={len(parent)}, max_conv={p_score:.8f}")
            best_seq = parent_normed.copy()
            best_score = p_score
    except Exception:
        parent = None
        print("No parent available, starting fresh")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        n_a = len(a)
        a_normed = a * np.sqrt(2.0 * n_a) / s
        sc = fast_autocorr(a_normed).max()
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_normed.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={n_a}, t={elapsed:.0f}s)")

    # Phase 1: LP-Nesterov on parent (main workhorse)
    n_base = len(best_seq) if best_seq is not None else 1600
    if best_seq is None:
        best_seq = structured_init(n_base, rng)
        best_seq = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        best_score = fast_autocorr(best_seq).max()
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP-Nesterov n={n_base} (budget={remaining:.0f}s) ===")
    t1 = remaining * 0.40
    a_out, _ = nesterov_lp_optimize(best_seq, t1, solver, verbose=True, step_mode='adaptive')
    try_update(a_out, "lp_nesterov")

    # Phase 2: QP variant (different trajectory)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: QP variant n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 30 and solver is not None:
        n_q = len(best_seq)
        a_q = best_seq.copy()
        best_q = fast_autocorr(a_q).max()
        best_q_a = a_q.copy()
        y_q = a_q.copy()
        mu = 1.0
        for t_iter in range(1, 200000):
            if time.monotonic() - t_global > time_budget - 60:
                break
            M_curr = fast_autocorr(y_q).max()
            b = qp_best_response(y_q, M_curr, mu, solver)
            if b is None:
                continue
            g = b * np.sqrt(2.0 * n_q) / b.sum()
            alpha = 2.0 / (t_iter + 2.0)
            y_q = np.maximum((1 - alpha) * y_q + alpha * g, 0.0)
            s_y = y_q.sum()
            y_q = y_q * np.sqrt(2.0 * n_q) / max(s_y, 1e-15)
            sc = fast_autocorr(y_q).max()
            if sc < best_q:
                best_q = sc
                best_q_a = y_q.copy()
        try_update(best_q_a, "qp_variant")

    # Phase 3: LSE optimization
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LSE scipy n={n_best} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 20:
        a_out, _ = lsc_scipy_optimize(best_seq, t3, T_start=40.0, T_end=2.0, n_stages=5)
        try_update(a_out, "lse_scipy")

    # Phase 4: Try longer sequence
    n_long = n_base + 600
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LP n={n_long} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 250.0)
    if t4 > 30:
        init_long = interpolate_sequence(best_seq, n_long)
        init_long = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(init_long, t4, solver, verbose=True, step_mode='adaptive')
        try_update(a_out, f"lp_long_n{n_long}")

    # Phase 5: Fast swap polish on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Swap polish n={n_best} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.20, 300.0)
    if t5 > 10:
        a_out, _ = swap_polish(best_seq, t5, seed=12345)
        try_update(a_out, "swap_polish")

    # Phase 6: LSE with low temperature for final polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: LSE polish n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.12, 120.0)
    if t6 > 15:
        a_out, _ = lsc_scipy_optimize(best_seq, t6, T_start=8.0, T_end=0.5, n_stages=4)
        try_update(a_out, "lse_polish_lowT")

    # Phase 7: Final swap polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final swap n={n_best} (budget={remaining:.0f}s) ===")
    t7 = remaining * 0.45
    if t7 > 10:
        a_out, _ = swap_polish(best_seq, t7, seed=99999)
        try_update(a_out, "swap_final")

    # Final verification
    if best_seq is not None:
        final_score = evaluate_safe(best_seq)
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
