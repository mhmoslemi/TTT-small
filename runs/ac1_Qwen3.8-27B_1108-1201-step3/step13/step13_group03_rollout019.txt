    A = rfft(a.astype(np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def eval_safe(a):
    """Safely evaluate a sequence using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        s = sum(a_list)
        if s < 0.01:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def build_conv_matrix(a):
    """Build sparse convolution matrix T where (T@b)[k] = conv(a,b)[k]."""
    n = len(a)
    diags_data = []
    for lag in range(n):
        length = 2 * n - 1 - lag
        diags_data.append(a[lag] * np.ones(length, dtype=np.float64))
    offsets = list(range(n))
    T = sparse_diags(diags_data, offsets, shape=(2*n-1, n))
    return T.tocsr()

def lp_best_response(a, M, solver):
    """Solve LP: maximize sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_conv_matrix(a)
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

    s0 = a.sum()

        a = a * norm / s0
