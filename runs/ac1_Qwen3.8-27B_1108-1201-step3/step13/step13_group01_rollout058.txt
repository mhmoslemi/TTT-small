def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def active_set_lp_best_response(a, M, solver, k_constraints=None):
    """
    LP best response using only top-k convolution constraints.
    This is faster than using all 2n-1 constraints.
    """
    n = len(a)
    size = 2 * n - 1
    b = fast_autocorr(a)
    
    if k_constraints is None or k_constraints >= size:
        # Use all constraints
        top_k = np.arange(size)
    else:
        # Use only top-k constraints (by value)
        top_k = np.argsort(b)[::-1][:k_constraints]
    
    b_var = cp.Variable(n, nonneg=True)
    
    # Build constraint matrix only for selected rows
    T_sel = np.zeros((len(top_k), n))
    for row, k in enumerate(top_k):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                T_sel[row, i] = a[j]
    
    constraints = [T_sel @ b_var <= M * np.ones(len(top_k))]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def full_lp_best_response(a, M, solver):
    """LP best response with all constraints (slower but more accurate)."""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, k_constraints=500, use_full_lp_every=100):
    """
    Nesterov-accelerated Frank-Wolfe with active-set LP.
    Uses top-k constraints for speed, periodically switches to full LP.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    last_improve = 0.0
    stall_count = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        # Use active-set LP most of the time, full LP periodically
        if t % use_full_lp_every == 0:
            b = full_lp_best_response(y, M_curr, solver)
        else:
            b = active_set_lp_best_response(y, M_curr, solver, k_constraints)
        
        if b is None:
            b = full_lp_best_response(y, M_curr, solver)
            if b is None:
                continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Adaptive step size based on duality gap
        if best_score > 0 and M_curr > 0:
            # Gap: M_curr is the current "primal" value
            # A lower bound would be the average of conv
            avg_conv = (fast_autocorr(y).sum()) / (2 * n - 1)
            gap_ratio = M_curr / max(avg_conv, 1e-15)
            # Larger gap → can take bigger steps
            alpha = 2.0 / (t + 2.0)
            alpha *= min(gap_ratio / 2.0, 1.0)  # Scale by gap
            alpha = max(alpha, 0.01)  # Minimum step
        else:
            alpha = 2.0 / (t + 2.0)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()

        avg_a += a
        n_avg += 1
        if n_avg % 20 == 0:

            s_avg = avg_check.sum()

def lse_objective_and_grad(a_flat, n, T):
    """Compute LSE-smoothed objective and gradient."""
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    
    exp_vals = np.exp(T * (b - max_b))
    exp_sum = exp_vals.sum()
    w = exp_vals / exp_sum
    lse = max_b + np.log(exp_sum) / T
    
    # Gradient of LSE w.r.t. a
    # d(lse)/d(a_i) = sum_k w[k] * d(conv(a,a)[k])/d(a_i)
    # d(conv(a,a)[k])/d(a_i) = a[k-i] + a[i-k] (when in range)
    # This is equivalent to: (w * a + w_flipped * a) restricted to [0, n-1]
    # Using FFT:
    fft_size = 1 << (3 * n - 2).bit_length()
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    w_padded = np.zeros(fft_size)
    w_padded[:2 * n - 1] = w
    A_fft = rfft(a_padded, fft_size)
    W_fft = rfft(w_padded, fft_size)
    cross = irfft(A_fft * np.conj(W_fft), fft_size)[:n]
    
    # d(conv(a,a))/d(a_i) = 2 * cross[i] (by symmetry of autocorrelation)
    conv_grad = 2.0 * cross
    
    where_a_positive = a > 1e-10
    grad = np.zeros(n)
    grad[where_a_positive] = conv_grad[where_a_positive]
    
    # Objective gradient (including the 1/s^2 factor)
    obj_val = 2.0 * n * lse / (s * s)
    obj_grad = 2.0 * n * conv_grad / (s * s) - 2.0 * obj_val * a / s
    
    return obj_val, obj_grad.ravel()


def lse_scipy_optimize(a0, time_limit, T_start=30.0, T_end=1.0, n_stages=6):
    """Optimize using LSE-smoothed surrogate with L-BFGS-B."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        stage_time = min(remaining / (n_stages - si), 100.0)

        x0 = a.copy()

        def obj_func(x):
            val, _ = lse_objective_and_grad(x, n, T)
            return val

        def grad_func(x):
            _, g = lse_objective_and_grad(x, n, T)
            return g

        try:
            res = scipy_minimize(obj_func, x0, method='L-BFGS-B',
                                  jac=grad_func,
                                  bounds=[(0, None)] * n,
                                  options={'maxiter': 300, 'maxfun': 1000,
                                           'ftol': 1e-12, 'gtol': 1e-10})
            if res.x is not None:
                a_candidate = np.maximum(res.x, 0.0)
                s_c = a_candidate.sum()
                if s_c > 1e-10:
                    a_candidate = a_candidate * norm / s_c
                    score_c = fast_autocorr(a_candidate).max()
                    if score_c < best_score:
                        best_score = score_c
                        best_a = a_candidate.copy()
                        a = a_candidate
        except Exception as e:
            print(f"    LSE stage {si} error: {e}")

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, seed=42):
    """
    FFT-accelerated coordinate descent targeting top peaks.
    Moves mass from high-contributing positions to low-peak positions.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    def try_move(a, src, dst, amt):
        """Try moving amt from src to dst. Returns (new_a, score) or (None, None)."""
        if a[src] < amt + 1e-15:
            return None, None
        a_trial = a.copy()
        a_trial[src] -= amt
        a_trial[dst] += amt
        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-10:
            return None, None
        a_trial = a_trial * norm / s_t
        score = fast_autocorr(a_trial).max()
        return a_trial, score

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_peaks = np.argsort(b)[::-1][:10]
        
        improved = False
        
        # Try to reduce each top peak
        for k_peak in top_peaks:
            if b[k_peak] <= best_score * 1.001:
                continue  # Already below best
            
            # Find which pairs contribute most to this peak
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_peak - idx]
            top_contrib = np.argsort(contrib)[::-1][:8]
            
            # Build mask of positions involved in any top peak
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True
            
            # Try moving from high-contributing to low-involvement positions
            for ci in top_contrib:
                src = idx[ci]
                if a[src] < 1e-10:
                    continue
                
                # Find good destination: not in peak mask, has low value
                candidates = np.where(~peak_mask)[0]
                if len(candidates) == 0:
                    candidates = np.where(~peak_mask | (np.arange(n) != src))[0]
                if len(candidates) == 0:
                    continue
                
                # Sort by current value (prefer moving to low-value positions)
                candidates_sorted = candidates[np.argsort(a[candidates])][:20]
                
                for dst in candidates_sorted[:10]:
                    for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial, score = try_move(a, src, dst, amt)
                        if a_trial is not None and score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            
            # Random perturbations when stalled
            for _ in range(50):
                if time.monotonic() - t0 > time_limit:
                    break
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a_trial, score = try_move(a, i, j, amt)
                if a_trial is not None and score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
                    break
            
            if stall > 300:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def structured_init(n, rng):

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize as scipy_minimize

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64).ravel()
        a_list = [float(x) for x in a_arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def build_T_active(a, selected_k):
    """Build constraint matrix rows for selected lags only (vectorized)."""
    n = len(a)
    T = np.zeros((len(selected_k), n))
    for row, k in enumerate(selected_k):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            T[row, i_start:i_end + 1] = a[k - i_start:k - i_end - 1:-1]
    return T


def active_set_lp_best_response(a, M, solver, k_constraints):
    """LP best response using only top-k convolution constraints."""
    n = len(a)
    size = 2 * n - 1
    b = fast_autocorr(a)

    if k_constraints >= size:
        top_k = np.arange(size)
    else:
        top_k = np.argsort(b)[::-1][:k_constraints]

    b_var = cp.Variable(n, nonneg=True)
    T_sel = build_T_active(a, top_k)
    constraints = [T_sel @ b_var <= M * np.ones(len(top_k))]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def full_lp_best_response(a, M, solver):
    """LP best response with all 2n-1 constraints."""
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    T = build_T_active(a, np.arange(size))
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, k_constraints=400,
                         use_full_lp_every=80, stall_thresh=40.0, stall_patience=12):
    """
    Nesterov-accelerated Frank-Wolfe with active-set LP.
    Uses top-k constraints for speed, periodically switches to full LP.
    Adaptive step size based on duality gap.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    last_improve = 0.0
    stall_count = 0

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-14:
            break

        if t % use_full_lp_every == 0:
            b = full_lp_best_response(y, M_curr, solver)
        else:
            b = active_set_lp_best_response(y, M_curr, solver, k_constraints)

        if b is None:
            b = full_lp_best_response(y, M_curr, solver)
            if b is None:
                continue

        # Safety check: verify the solution doesn't violate full constraint too badly
        b_check = fast_autocorr(y * 0 + b)  # placeholder, skip for speed
        # Actually verify: max(a_new * b) should not be too much larger than M
        # conv(new_a, b) where new_a = y, b = solution
        # This is expensive so we skip it most of the time

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Adaptive step size
        if best_score > 0 and M_curr > 0:
            # Use gap ratio to modulate step
            avg_conv = M_curr * (2 * n - 1) / (2 * n - 1)  # just M_curr as proxy
            # Better: use the ratio of current peak to theoretical lower bound
            # Lower bound: sum(conv)/(2n-1)
            conv_sum = (y.sum()) ** 2  # sum of full conv = (sum a)^2
            lb = conv_sum / (2 * n - 1)
            gap_ratio = M_curr / max(lb, 1e-15)
            alpha = 2.0 / (t + 2.0)
            # Modulate: if gap is large, take bigger steps
            alpha *= min(max(gap_ratio / 1.5, 1.0), 3.0)
            alpha = min(alpha, 1.0)
        else:
            alpha = 2.0 / (t + 2.0)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()

        avg_a += a
        n_avg += 1
        if n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    last_improve = time.monotonic()

        if time.monotonic() - last_improve > stall_thresh:
            stall_count += 1
            if stall_count > stall_patience:
                if verbose:
                    print(f"    Stalling at iter={n_iters}, best={best_score:.8f}")
                break
        else:
            stall_count = 0

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lse_objective_and_grad(a_flat, n, T):
    """Compute LSE-smoothed objective and gradient for the score function."""
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)

    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)

    exp_vals = np.exp(T * (b - max_b))
    exp_sum = exp_vals.sum()
    w = exp_vals / exp_sum
    lse = max_b + np.log(exp_sum) / T

    # Gradient: d(conv(a,a))/d(a_i) computed via cross-correlation with w
    fft_size = 1 << (3 * n - 2).bit_length()
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    w_padded = np.zeros(fft_size)
    w_padded[:2 * n - 1] = w
    A_fft = rfft(a_padded, fft_size)
    W_fft = rfft(w_padded, fft_size)
    cross = irfft(A_fft * np.conj(W_fft), fft_size)[:n]

    conv_grad = 2.0 * cross
    obj_val = 2.0 * n * lse / (s * s)
    obj_grad = 2.0 * n * conv_grad / (s * s) - 2.0 * obj_val * a / s

    return obj_val, obj_grad.ravel()


def lse_scipy_optimize(a0, time_limit, T_start=40.0, T_end=1.0, n_stages=5):
    """Optimize using LSE-smoothed surrogate with L-BFGS-B."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        stage_time = min(remaining / (n_stages - si), 120.0)

        x0 = a.copy()

        def obj_func(x):
            val, _ = lse_objective_and_grad(x, n, T)
            return val

        def grad_func(x):
            _, g = lse_objective_and_grad(x, n, T)
            return g

        try:
            res = scipy_minimize(obj_func, x0, method='L-BFGS-B',
                                  jac=grad_func,
                                  bounds=[(0, None)] * n,
                                  options={'maxiter': 200, 'maxfun': 500,
                                           'ftol': 1e-14, 'gtol': 1e-12})
            if res.x is not None:
                a_candidate = np.maximum(res.x, 0.0)
                s_c = a_candidate.sum()
                if s_c > 1e-10:
                    a_candidate = a_candidate * norm / s_c
                    score_c = fast_autocorr(a_candidate).max()
                    if score_c < best_score:
                        best_score = score_c
                        best_a = a_candidate.copy()
                        a = a_candidate
        except Exception:
            pass

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, seed=42):
    """
    Aggressive FFT-accelerated coordinate descent targeting top peaks.
    Uses multi-fraction search and randomized restarts.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    fractions = [0.003, 0.007, 0.013, 0.023, 0.041, 0.072, 0.12, 0.2]

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_peaks = np.argsort(b)[::-1][:12]

        improved = False

        for k_peak in top_peaks:
            if b[k_peak] <= best_score * 1.0005:
                continue

            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_peak - idx]
            top_contrib = np.argsort(contrib)[::-1][:10]

            # Build peak involvement mask
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            for ci in top_contrib:
                src = idx[ci]
                if a[src] < 1e-12:
                    continue

                candidates = np.where(~peak_mask)[0]
                if len(candidates) == 0:
                    candidates = np.where(np.arange(n) != src)[0]
                if len(candidates) == 0:
                    continue

                # Prefer low-value, non-peak positions
                candidates_sorted = candidates[np.argsort(a[candidates])][:25]

                for dst in candidates_sorted[:15]:
                    for frac in fractions:
                        amt = a[src] * frac
                        if amt < 1e-14:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-10:
                            continue
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1

            # Random perturbations
            n_trials = 80 if stall < 50 else 40
            for _ in range(n_trials):
                if time.monotonic() - t0 > time_limit:
                    break
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-12:
                    continue
                frac = fractions[rng.integers(len(fractions))]
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial * norm / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
                    break

            # Occasional larger perturbation to escape
            if stall > 100 and stall % 50 == 0:
                n_perturb = rng.integers(5, 20)
                a_pert = a.copy()
                for _ in range(n_perturb):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a_pert[i] > 1e-10:
                        amt = a_pert[i] * rng.uniform(0.05, 0.15)
                        a_pert[i] -= amt
                        a_pert[j] += amt
                a_pert = np.maximum(a_pert, 0.0)
                s_p = a_pert.sum()
                if s_p > 1e-10:
                    a_pert = a_pert * norm / s_p
                    score_p = fast_autocorr(a_pert).max()
                    if score_p < best_score * 1.05:  # Allow slightly worse
                        a = a_pert  # Accept to explore

            if stall > 400:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main optimization: active-set LP+Nesterov, LSE gradient, coordinate descent.
    Multi-length exploration with adaptive time allocation.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 0: Multi-length scan
    print("\n=== Phase 0: Multi-length scan ===")
    scan_lengths = [1200, 1600, 2000, 2400]
    scan_results = {}
    phase0_budget = time_budget * 0.10

    for i_scan, n_scan in enumerate(scan_lengths):
        elapsed = time.monotonic() - t_global
        if elapsed > time_budget * 0.12:
            break
        remaining = phase0_budget - elapsed
        n_left = len(scan_lengths) - i_scan
        scan_time = min(remaining / max(1, n_left), 50.0)
        if scan_time < 8:
            break

        if best_seq is not None:
            init_seq = interpolate_sequence(best_seq, n_scan)
        else:
            x = np.linspace(0.0, 1.0, n_scan)
            init_seq = np.sin(np.pi * x) ** 1.5
            init_seq = np.maximum(init_seq, 0.0)
            init_seq = init_seq / init_seq.sum()

        a_norm = init_seq * np.sqrt(2.0 * n_scan) / init_seq.sum()
        a_out, sc_out = nesterov_lp_optimize(a_norm, scan_time, solver,
                                             verbose=False, k_constraints=250)
        scan_results[n_scan] = (a_out, sc_out)
        print(f"    n={n_scan}: score={sc_out:.8f} ({time.monotonic()-t_global:.0f}s)")

    if scan_results:
        best_scan_n = min(scan_results, key=lambda k: scan_results[k][1])
        do_a, do_sc = scan_results[best_scan_n]
        try_update(do_a, f"scan_n{best_scan_n}")

    if best_seq is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Deep LP Nesterov at best length
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 1: Deep LP n={n_cur} (budget={remaining:.0f}s) ===")

    remaining = time_budget - (time.monotonic() - t_global)

        try_update(a_out, f"lse_n{n_cur}")

        try_update(a_out, f"lp_long_n{n_long}")

    # Phase 5: LP Nesterov continued at best length
    remaining = time_budget - (time.monotonic() - t_global)

    t5 = min(remaining * 0.15, 200.0)
    if t5 > 15:

        a_out, _ = nesterov_lp_optimize(a_norm, t5, solver, verbose=True,
                                         k_constraints=300, use_full_lp_every=40,
                                         stall_thresh=30.0, stall_patience=8)

    # Phase 6: LSE low-temperature polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: LSE polish n={n_cur} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.10, 100.0)
    if t6 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lse_scipy_optimize(a_norm, t6, T_start=10.0, T_end=0.5, n_stages=4)
        try_update(a_out, "lse_polish")

    # Phase 7: Final aggressive coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: Final coord n={n_cur} (budget={remaining:.0f}s) ===")
    t7 = remaining * 0.55
    if t7 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = coordinate_descent_polish(a_norm, t7, seed=77777)
        try_update(a_out, "coord_final")

    # Phase 8: Final LSE + coordinate
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 8: Final LSE+coord n={n_cur} (budget={remaining:.0f}s) ===")
        t8a = remaining * 0.4
        t8b = remaining * 0.5
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lse_scipy_optimize(a_norm, t8a, T_start=5.0, T_end=0.3, n_stages=3)
        try_update(a_out, "lse_final")
        if time.monotonic() - t_global < time_budget - 10:
            a_norm = best_seq * np.sqrt(2.0 * len(best_seq)) / best_seq.sum()
            a_out, _ = coordinate_descent_polish(a_norm, t8b, seed=31337)
            try_update(a_out, "coord_final2")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
