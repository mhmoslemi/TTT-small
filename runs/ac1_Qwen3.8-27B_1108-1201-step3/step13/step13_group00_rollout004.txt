                else:

                if hi1 > lo1:
                    g[lo1:hi1] += a[:hi1-lo1]
                # a[k-i] for valid k: k-i in [0,n-1] means k in [i, i+n-1]
                lo2 = i
                hi2 = min(size, i + n)
                if hi2 > lo2:
                    g[lo2:hi2] -= a[:hi2-lo2]
                
                # Second order: h[k] = 1_{k=2j} - 2*1_{k=i+j} + 1_{k=2i}
                # Only nonzero at k=2i, k=2j, k=i+j
                
                # Define f(delta) = max_k [conv[k] + 2*delta*g[k] + delta^2*h[k]]
                # Use ternary search on [0, a[i]]
                def f_of_delta(d):
                    val = (conv + 2.0 * d * g).max()
                    # Add quadratic corrections
                    if 2 * i < size:
                        val = max(val, conv[2*i] + 2.0*d*g[2*i] + d*d)
                    if 2 * j < size:
                        val = max(val, conv[2*j] + 2.0*d*g[2*j] + d*d)
                    if (i + j) < size:
                        val = max(val, conv[i+j] + 2.0*d*g[i+j] - 2.0*d*d)
                    return val
                
                # Check if f(0) is already the max (i.e., no improvement possible)
                f0 = f_of_delta(0)
                if f0 >= best_score + 1e-10:
                    # Even at delta=0, we can't do better (but we should check if
                    # moving in this direction helps)
                    pass
                
                # Ternary search
                lo = 0.0
                hi = a[i]
                if hi < 1e-12:
                    continue
                
                # First check endpoints
                f_lo = f_of_delta(lo)
                f_hi = f_of_delta(hi)
                
                # Check if moving helps at all
                # Evaluate at midpoint
                mid = (lo + hi) / 2
                f_mid = f_of_delta(mid)
                
                if f_mid >= f_lo and f_hi >= f_lo:
                    # The function is increasing, try the other direction
                    # Actually, if both endpoints are higher than the start,
                    # we want to go to the lower one
                    if f_hi < f_lo:
                        # Go to hi
                        best_d = hi
                        best_val = f_hi
                    else:
                        # Go to lo (i.e., don't move)
                        continue
                else:
                    # There's a minimum inside; do ternary search
                    for _ in range(30):
                        m1 = lo + (hi - lo) / 3
                        m2 = hi - (hi - lo) / 3
                        f_m1 = f_of_delta(m1)
                        f_m2 = f_of_delta(m2)
                        if f_m1 <= f_m2:
                            hi = m2
                        else:
                            lo = m1
                    best_d = (lo + hi) / 2
                    best_val = f_of_delta(best_d)
                
                if best_val < best_score - 1e-12 and best_d > 1e-10:
                    a[i] -= best_d
                    a[j] += best_d
                    s = a.sum()
                    if s < 1e-15:
                        a[i] += best_d
                        a[j] -= best_d
                        continue
                    a = a * norm / s
                    conv = fast_autocorr(a)
                    cur = conv.max()
                    if cur < best_score - 1e-12:
                        best_score = cur
                        best_a = a.copy()
                        pass_improved = True
                    else:
                        # Revert
                        a[j] -= best_d
                        a[i] += best_d
                        s = a.sum()
                        a = a * norm / s
                        conv = fast_autocorr(a)
            # End for attempt
        
        if not pass_improved:
            no_improve_passes += 1
            if no_improve_passes >= 3:
                break
        else:
            no_improve_passes = 0
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    PCD ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def project_simplex(v, target_sum):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    # We want to find the closest point to v on the simplex
    # Using the standard algorithm:
    # x_i = max(v_i - tau, 0) where tau is chosen so sum(x) = target_sum
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho = np.nonzero(v_sorted * np.arange(1, n + 1) > (cssv - target_sum))[0][-1]
    tau = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - tau, 0.0)

def nesterov_subgrad(a0, time_limit, verbose=True):
    """Nesterov-accelerated subgradient method with simplex projection."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    
    y = a.copy()
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_autocorr(y)
        M = conv_y.max()
        
        # Subgradient at peak positions
        # Use top-k peaks for a smoother gradient
        k_peaks = min(10, size)
        peak_vals = np.sort(conv_y)[::-1][:k_peaks]
        max_val = peak_vals[0]
        # Weight by exp((val - max)/T) for smoothness
        T_smooth = max(0.1, M * 0.01)
        weights = np.exp((conv_y - max_val) / T_smooth)
        weights /= weights.sum()
        
        # Gradient: grad_i = 2 * sum_k weights[k] * (a[k-i] + a[i-k])
        # = 2 * (cross_corr(weights, a) + cross_corr(a, weights))[i]
        # Use FFT
        fft_size = 1 << (size - 1).bit_length()
        W = rfft(weights, fft_size)
        A = rfft(y, fft_size)
        # cross_corr(w, a)[i] = sum_k w[k] * a[k+i] (circular)
        # = ifft(FFT(w) * conj(FFT(a)))
        cc1 = irfft(W * np.conj(A), fft_size)[:n]
        cc2 = irfft(A * np.conj(W), fft_size)[:n]
        grad = 2.0 * (cc1 + cc2)
        
        # Nesterov step
        lr = norm * 2.0 / (t * np.sqrt(t))
        step = y - lr * grad
        y_new = project_simplex(step, norm)
        
        if t == 1:
            a_new = y_new.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = project_simplex
