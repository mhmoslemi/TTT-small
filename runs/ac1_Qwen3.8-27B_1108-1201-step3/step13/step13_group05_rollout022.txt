The LP is:
Maximize $\sum_i b_i$
Subject to: $\sum_i b_i a_{k-i} \leq M$ for all $k$
$b_i \geq 0$

where $M = \max_k (a*a)_k$ (the current peak).

The solution $b^*$ gives the "best response" direction. Then we update:
$a_{new} = (1-t) \cdot a + t \cdot \frac{b^*}{\sum b^*}$

def lp_best_response(a, M, top_k=100, solver=None):
    n = len(a)
    q = fast_autocorr(a)
    
    # Select top K lag constraints
    lag_order = np.argsort(q)[::-1]
    selected_lags = lag_order[:top_k]
    
    # Build constraint matrix for selected lags
    # (a*b)_k = sum_i b_i * a_{k-i}
    # For lag k: b_i * a_{k-i} for valid i
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = []
    
    for k in selected_lags:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        # (a*b)_k = sum_{j=js}^{je} a_{k-j} * b_j

    # Actually, (a*b)_k = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a_{k-i} * b_i

    # Let me build the matrix properly
    # T[i, k] = a_{k-i} if 0 <= k-i < n, else 0
    # But we only want selected lags
    
    # For efficiency, build a sparse matrix
    rows = []  # row indices (within selected lags)
    cols = []
    vals = []
    for idx, k in enumerate(selected_lags):
        js = max(0, int(k) - n + 1)
        je = min(n - 1, int(k))
        for j in range(js, je + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])

    T = csr_matrix((vals, (rows, cols)), shape=(len(selected_lags), n))

    prob.solve(solver=solver, verbose=False)

        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
