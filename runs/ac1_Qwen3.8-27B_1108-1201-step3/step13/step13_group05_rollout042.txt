def fast_conv(a, b=None):
    """Compute linear convolution a*b (or a*a if b=None) using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]

def fast_autocorr(a):
    """Compute linear convolution a*a using FFT."""
    return fast_conv(a, a)

def compute_soft_subgrad(a, ac, top_k=10, beta=3.0):

        idx = np.arange(i_start, i_end)
        j = k_int - idx

    w = exp_vals / exp_vals.sum()
    
    a_rev = a[::-1]
    w_conv = fast_conv(w, a_rev)
    grad_a = 2.0 * w_conv[n-1 : 2*n-1]
    
    # Chain rule: grad_x_k = a_k * (grad_a_k - sum_j a_j * grad_a_j)
    weighted_sum = np.dot(a, grad_a)
    grad_x = a * (grad_a - weighted_sum)
    
    return grad_x, a, logsumexp_val

def anneal_smooth_max(a0, time_limit, T0=None, T_min=None, decay=0.995, verbose=True):
    """Anneal the log-sum-exp approximation with Nesterov gradient descent."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    # Work in log-parameter space
    x = np.log(a + 1e-30)

        ac = fast_autocorr(a)
        T0 = ac.max() / 5.0
    if T_min is None:
        T_min = 1e-6
    
    T = T0

    best_obj = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Nesterov momentum
    x_prev = x.copy()
    mu = 0.9
    
    # Step size: start with 1/L estimate
    gx, a_curr, fval = softmax_param_grad(x, T)
    L = np.max(np.abs(gx)) + 1e-10
    step = 1.0 / L
    
    iter_count = 0
    last_print = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        if T < T_min:
            break
        
        iter_count += 1
        
        # Nesterov lookahead
        beta = (1.0 + np.sqrt(1.0 + 4.0 * mu * mu)) / 2.0
        x_look = x + mu * (x - x_prev)
        
        # Compute gradient at lookahead point
        gx, a_look, fval = softmax_param_grad(x_look, T)
        
        # Gradient step
        x_new = x_look - step * gx
        x_new = x_new - x_new.max()  # numerical stability
        
        # Update momentum
        x_prev = x.copy()
        x = x_new
        
        # Recover a
        x_shifted = x - x.max()
        exp_x = np.exp(x_shifted)
        a_curr = exp_x / exp_x.sum()
        
        # Check exact objective periodically
        if iter_count % 50 == 0:
            ac = fast_autocorr(a_curr)
            obj = 2.0 * n * ac.max()
            if obj < best_obj:
                best_obj = obj
                best_a = a_curr.copy()
            
            if verbose and time.monotonic() - last_print > 10:
                last_print = time.monotonic()
                elapsed = time.monotonic() - t0
                print(f"      Anneal it={iter_count}, T={T:.6f}, obj={best_obj:.6f}, t={elapsed:.0f}s")
        
        # Decay temperature
        T *= decay
        
        # Adjust step size based on gradient magnitude
        gx_mag = np.max(np.abs(gx))
        if gx_mag > 0:
            step = min(step * 1.01, 2.0 / gx_mag)
            step = max(step, 1e-6)
    
    # Final recovery
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    a_final = exp_x / exp_x.sum()
    ac_final = fast_autocorr(a_final)
    if 2.0 * n * ac_final.max() < best_obj:
        best_obj = 2.0 * n * ac_final.max()
        best_a = a_final.copy()
    
    return best_a, best_obj

def mirror_descent_exact(a0, time_limit, top_k=10, beta=3.0, verbose=True):
    """Mirror descent with exact top-k subgradient on simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

        log_a_new = log_a - step * g
        max_log = log_a_new.max()
        a_new = np.exp(log_a_new - max_log)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:

def lp_best_response(a, M, solver=None):
    """Solve LP: max sum(b) s.t. conv(a,b)_k <= M for all k, b >= 0."""
    try:
        import cvxpy as cp
    except ImportError:
        return None
    
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    
    # Build constraint: conv(a, b) <= M
    # conv(a, b)_k = sum_j a * b_{k-j} ... this is hard to express directly
    # Use the matrix form: T @ b <= M where T is the convolution matrix
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    if solver is None:
        solver = cp.ECOS
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def nesterov_lp_optimize(a0, time_limit, solver=None, verbose=True):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    x = a.copy()
    last_print = time.monotonic()
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-15:
            break
        
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b / b_sum
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        
        if t == 1:
            x_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / s_x
        
        y = y_new
        x = x_new
        
        obj_y = 2.0 * n * fast_autocorr(y).max()
        if obj_y < best_obj:
            best_obj = obj_y
            best_a = y.copy()
        
        obj_x = 2.0 * n * fast_autocorr(x).max()
        if obj_x < best_obj:
            best_obj = obj_x
            best_a = x.copy()
        
        if verbose and time.monotonic() - last_print > 15:
            last_print = time.monotonic()
            elapsed = time.monotonic() - t0
            print(f"      LP it={t} ({elapsed:.0f}s): obj={best_obj:.6f}")
    
    return best_a, best_obj

def coordinate_polish(a0, time_limit, verbose=True):
    """Coordinate descent: move mass from high-peak contributors to low-peak positions."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    stall = 0
    last_print = time.monotonic()
    
    while time.monotonic() - t0 < time_limit and stall < 300:
        ac = fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:10]
        improved = False
        
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            js = max(0, int(k) - n + 1)
            je = min(n - 1, int(k))
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[int(k) - idx]
            src = idx[np.argmax(contrib)]
            
            # Find positions that don't contribute to top peaks
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, int(k2) - n + 1)
                je2 = min(n - 1, int(k2))
                peak_mask[js2:je2 + 1] = True
            
            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]
            
            cands_sorted = candidates[np.argsort(a[candidates])]
            # Pick a low-value candidate
            pick_range = max(1, len(cands_sorted) // 4)
            dst = cands_sorted[rng.integers(pick_range)]
            
            if dst == src:
                continue
            
            for frac in [0.003, 0.007, 0.015, 0.03, 0.06, 0.1]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                M_trial = fast_autocorr(a_trial).max()
                obj_trial = 2.0 * n * M_trial
                if obj_trial < best_obj - 1e-15:
                    a = a_trial
                    best_obj = obj_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        
        if not improved:
            stall += 1
            # Random perturbation to escape local minima
            if stall % 10 == 0:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.02, 0.08)
                    amt = a[i] * frac
                    a[i] -= amt
                    a[j] += amt
                    if a[i] >= 0:
                        a = a / a.sum()
                        M_new = fast_autocorr(a).max()
                        if 2.0 * n * M_new < best_obj:
                            best_obj = 2.0 * n * M_new
                            best_a = a.copy()
                            stall = 0
                        else:
                            a[i] += amt
                            a[j] -= amt
                            a = a / a.sum()
        
        if verbose and time.monotonic() - last_print > 20:
            last_print = time.monotonic()
            elapsed = time.monotonic() - t0
            print(f"      Coord ({elapsed:.0f}s): obj={best_obj:.6f}, stall={stall}")
    
    return best_a, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

"""

2. Exact top-k subgradient mirror descent as complement

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_conv(a, b=None):
    """Compute linear convolution a*b (or a*a if b=None) using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]

def fast_autocorr(a):
    """Compute linear convolution a*a using FFT."""
    return fast_conv(a, a)

def compute_topk_subgrad(a, ac, top_k=10, beta=3.0):
    """Compute weighted subgradient using top-k active lags with power weighting."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = ac[top_idx].copy()
    top_vals = np.maximum(top_vals, 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        i_start = max(0, k_int - n + 1)
        i_end = min(n, k_int + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        j = k_int - idx
        grad[idx] += 2.0 * wi * a[j]
    return grad

def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. conv(a,b)_k <= M for all k, b >= 0."""
    import cvxpy as cp
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def interp_seq(f, new_n):
    """Interpolate sequence to new length preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
