    excess = b_autocorr - M * 0.9  # use 90% of M as threshold
    w_soft = np.exp(temperature * np.maximum(excess, 0))
    w_soft /= w_soft.sum() if w_soft.sum() > 0 else 1.0
    
    # Pressure on b[i] = sum_k w_soft[k] * a[k-i]
    # This is the cross-correlation
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w_soft, fft_size)
    cross = irfft(A * W, fft_size)[:n]
    
    # b[i] = a[i] * exp(-eta * pressure[i]) ... but we need to be more careful
    # The LP solution satisfies: b[i] > 0 only if the constraint is "tight" for some k
    # A soft approximation: b[i] ∝ exp(-eta * max(0, pressure[i] - threshold))
    
    eta = 1.0 / max(M, 1e-10)
    b = a * np.exp(-eta * cross)
    b = np.maximum(b, 0)
    s = b.sum()
    if s < 1e-15:
        return a.copy()
    return b / s

def cross_corr(a, b):

def build_conv_matrix(a, n):
    """Build the sparse-ish convolution matrix T where T @ b = conv(a, b)[:n]
    Actually T has shape (2n-1, n) where (T @ b)[k] = sum_i a[max(0,min(i,k))...]
    For efficiency, we build it as a dense matrix for LP."""
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    return T


def lp_best_response_with_dual(a, M, solver):
    """Solve LP best-response and return (b, dual_values)."""
    import cvxpy as cp

    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)

        return None, None

def lse_objective_and_grad(x, n, T_temp):
    """Compute LSE objective and gradient in one call."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)

    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)

    exp_vals = np.exp(T_temp * (b - max_b))
    exp_sum = exp_vals.sum()
    w = exp_vals / exp_sum
    lse = max_b + np.log(exp_sum) / T_temp

    # Gradient: d/d a[i] of lse
    # = (1/T) * sum_k w[k] * d(b[k])/d(a[i])
    # d(b[k])/d(a[i]) = a[k-i] + a[i-k] (when indices valid)
    # This is 2 * cross_corr(a, w)[:n] approximately
    size = 2 * n - 1
    fft_size = 1 << (size + n - 1).bit_length()
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    w_padded = np.zeros(fft_size)
    w_padded[:size] = w
    A_fft = rfft(a_padded, fft_size)
    W_fft = rfft(w_padded, fft_size)
    cross = irfft(A_fft * W_fft, fft_size)[:n]

    conv_grad = (2.0 / T_temp) * cross
    obj_val = 2.0 * n * lse / (s * s)
    grad = conv_grad / (s * s) - 2.0 * obj_val * a / s
    return obj_val, grad

def coordinate_descent_polish(a0, time_limit, seed=42):

                if len(candidates) < 10:
                    candidates = np.where(np.arange(n) != src)[0]
                # Sort by a value (move from high to low)
                candidates_sorted = candidates[np.argsort(a[candidates])][:max(2, len(candidates) // 4)]

                for dst in candidates_sorted[:5]:
                    for frac in [0.005, 0.01, 0.02, 0.05, 0.1]:
                        amt = a[src] * frac
                        if amt < 1e-15:

                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < -1e-15:
                            continue
                        a_trial = np.maximum(a_trial, 0.0)

                        if s_t < 1e-10:

            for _ in range(30):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-10:

            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new
        avg_a += a
        n_avg += 1

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 1:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 2:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.5) ** 2)
    elif choice == 3:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-4.0 * (x - 0.7) ** 2)
    elif choice == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 4.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 6:
        x = np.linspace(0.0, 1.0, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    else:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-2.0 * (x - 0.5) ** 2) + 0.3 * np.exp(-8.0 * (x - 0.2) ** 2) + 0.3 * np.exp(-8.0 * (x - 0.8) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function. Returns the best sequence found."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 0: Quick multi-length scan with soft method (fast)
    print("\n=== Phase 0: Multi-length scan ===")
    scan_results = {}
    scan_lengths = [1600, 2000, 2400, 2800]

    if best_seq is not None:
        for n_scan in scan_lengths:
            if time.monotonic() - t_global > time_budget * 0.12:
                break
            remaining_scan = time_budget * 0.12 - (time.monotonic() - t_global)
            if remaining_scan < 5:
                break
            scan_time = min(remaining_scan / max(1, len(scan_lengths) - scan_lengths.index(n_scan)), 40.0)
            if scan_time < 3:
                break

            if n_scan == len(best_seq):
                continue  # Skip same length, already have it

            init_seq = interpolate_sequence(best_seq, n_scan)
            a_norm = init_seq * np.sqrt(2.0 * n_scan) / init_seq.sum()
            a_out, sc_out = soft_multiplicative_optimize(a_norm, scan_time, verbose=False)
            scan_results[n_scan] = (a_out, sc_out)
            print(f"    n={n_scan}: score={sc_out:.8f} ({time.monotonic()-t_global:.0f}s)")
            do_try_update(a_out, f"scan_soft_n{n_scan}")
    else:
        # No parent, use structured init
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
        n_base = 1600

    # Determine best length to focus on
    n_focus = len(best_seq)
    if scan_results:
        best_scan_n = min(scan_results, key=lambda k: scan_results[k][1])
        if scan_results[best_scan_n][1] < best_score * 1.001:
            n_focus = best_scan_n
            print(f"  Focusing on n={n_focus} from scan")

    # Phase 1: Deep LP+Nesterov at focus length
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Deep LP n={n_focus} (budget={remaining:.0f}s) ===")
    t1 = remaining * 0.40
    if t1 > 20:
        a_init = best_seq
        if n_focus != len(a_init):
            a_init = interpolate_sequence(a_init, n_focus)
        a_norm = a_init * np.sqrt(2.0 * n_focus) / a_init.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True, use_dual=True)
        do_try_update(a_out, f"lp_deep_n{n_focus}")

    # Phase 2: LSE + L-BFGS-B
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LSE scipy n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.12, 150.0)
    if t2 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_scipy_optimize(a_norm, t2, T_start=25.0, T_end=1.5, n_stages=4)
        do_try_update(a_out, f"lse_n{n_best}")

    # Phase 3: Try a different length with LP
    n_alt = n_focus + 400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LP n={n_alt} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.18, 250.0)
    if t3 > 20:
        init_alt = interpolate_sequence(best_seq, n_alt)
        a_norm = init_alt * np.sqrt(2.0 * n_alt) / init_alt.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=True, use_dual=True)
        do_try_update(a_out, f"lp_alt_n{n_alt}")

    # Phase 4: Subgradient Nesterov (different trajectory)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: SubGrad n={n_best} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.08, 100.0)
    if t4 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = subgrad_projected_optimize(a_norm, t4, verbose=False)
        do_try_update(a_out, f"subgrad_n{n_best}")

    # Phase 5: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Coord descent n={n_best} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.25, 300.0)
    if t5 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent_polish(a_norm, t5, seed=12345)
        do_try_update(a_out, "coord_polish_1")

    # Phase 6: LSE polish with low temperature
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: LSE polish n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.15, 120.0)
    if t6 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_scipy_optimize(a_norm, t6, T_start=5.0, T_end=0.5, n_stages=3)
        do_try_update(a_out, "lse_polish")

    # Phase 7: Final coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final coord n={n_best} (budget={remaining:.0f}s) ===")
    t7 = remaining * 0.6
    if t7 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = coordinate_descent_polish(a_norm, t7, seed=77777)
        do_try_update(a_out, "coord_final")

    # Phase 8: One more LP run if time permits
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        print(f"\n=== Phase 8: Final LP n={n_best} (budget={remaining:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, remaining * 0.7, solver, verbose=False, use_dual=True)
        do_try_update(a_out, "lp_final")
        remaining2 = time_budget - (time.monotonic() - t_global)
        if remaining2 > 15:
            a_norm2 = best_seq * np.sqrt(2.0 * len(best_seq)) / best_seq.sum()
            a_out2, _ = coordinate_descent_polish(a_norm2, remaining2 * 0.8, seed=99999)
            do_try_update(a_out2, "coord_very_final")

    # Final check
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
