"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated LP zero-sum game with long averaging windows
2. Top-k LogSumExp gradient descent with adaptive temperature
3. Targeted peak reduction via local search
4. Multi-length and multi-start exploration

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]

        return float(sc)
    except Exception:
        return float('inf')


def build_toeplitz(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a

def find_solver():
    for solver_name in [cp.CLARABEL, cp.SCS, cp.PDLP, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(y, T, solver, warm_b=None):
    n = len(y)
    b_var = cp.Variable(n, nonneg=True)
    if warm_b is not None and np.all(warm_b >= -1e-10) and warm_b.sum() > 1e-10:
        b_var.value = warm_b
    M_var = cp.Variable(value=1.0/n)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        for alt in [cp.SCS, cp.CLARABEL, cp.ECOS, cp.PDLP]:
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(np.max(T @ b_val))


def nesterov_lp_optimize(a0, time_limit, solver, max_iters=100000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Multiple averaging windows
    avg_short = np.zeros(n)
    n_short = 0
    avg_long = np.zeros(n)
    n_long = 0
    # Rolling window
    WINDOW = 50
    roll_buf = [None] * WINDOW
    roll_idx = 0
    roll_sum = np.zeros(n)
    
    stall_count = 0
    warm_b = None
    best_M = float('inf')
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        T = build_toeplitz(y)
        b_star, M_val = solve_lp_best_response(y, T, solver, warm_b)
        if b_star is None:
            warm_b = None
            continue
        warm_b = b_star
        best_M = min(best_M, M_val)
        
        # Nesterov step
        eta = 2.0 / (t + 2.0)
        y_new = (1 - eta) * y + eta * b_star
        y_new = np.maximum(y_new, 0)
        ys = y_new.sum()
        if ys < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / ys
        
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new
        
        y = y_new
        a = a_new
        
        # Update averages
        avg_short += a
        n_short += 1
        avg_long += a
        n_long += 1
        
        # Rolling window
        if roll_buf[roll_idx] is not None:
            roll_sum -= roll_buf[roll_idx]
        roll_buf[roll_idx] = a.copy()
        roll_sum += a
        roll_idx = (roll_idx + 1) % WINDOW
        
        # Check best
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Check running averages (every 10 iterations)
        if n_long % 10 == 0:
            for avg, cnt in [(avg_short, n_short), (avg_long, n_long)]:
                if cnt > 5:
                    avg_check = avg / cnt
                    score_avg = 2.0 * n * fast_autocorr(avg_check).max()
                    if score_avg < best_score:
                        best_score = score_avg
                        best_a = avg_check.copy()
                        stall_count = 0
            
            if n_long >= WINDOW:
                roll_avg = roll_sum / WINDOW
                score_roll = 2.0 * n * fast_autocorr(roll_avg).max()
                if score_roll < best_score:
                    best_score = score_roll
                    best_a = roll_avg.copy()
                    stall_count = 0
        
        # Restart if stalled
        if stall_count > 300 and t > 50:
            a = best_a.copy()
            y = best_a.copy()
            avg_short = np.zeros(n)
            n_short = 0
            avg_long = np.zeros(n)
            n_long = 0
            roll_buf = [None] * WINDOW
            roll_idx = 0
            roll_sum = np.zeros(n)
            stall_count = 0
            warm_b = None
        
        if t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}")
    
    return best_a, best_score


def softmax_fn(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def compute_topk_lse_grad(a, T_temp, topk):
    n = len(a)
    c = fast_autocorr(a)
    k_use = min(topk, len(c))
    top_idx = np.argsort(c)[::-1][:k_use]
    top_vals = c[top_idx]
    
    # LSE objective
    max_v = top_vals.max()
    exp_v = np.exp(top_vals - max_v)
    lse_val = (max_v + np.log(exp_v.sum())) / T_temp
    w = exp_v / exp_v.sum()
    
    # Gradient
    grad_a = np.zeros(n)
    for idx in range(k_use):
        K = top_idx[idx]
        wK = w[idx]
        js = max(0, K - (n - 1))
        je = min(n - 1, K)
        if js <= je:
            idx_range = np.arange(js, je + 1)
            grad_a[idx_range] += 2.0 * wK * a[K - idx_range]
    
    return lse_val, grad_a


def topk_lse_grad_z(z, T_temp, topk):
    a = softmax_fn(z)
    n = len(a)
    obj, grad_a = compute_topk_lse_grad(a, T_temp, topk)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def adam_topk_lse(a0, time_limit, T_start=100.0, T_end=0.1, n_stages=40, topk=20):
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        base_lr = 5e-3 * min(1.0, 200.0 / n)
        lr = base_lr * (0.9 ** si)
        
        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.8, 0.999, 1e-8
        
        # More iterations per stage for lower temperatures
        iters_per_stage = max(30, int(min(remaining, 30.0) * 8))
        
        for it in range(iters_per_stage):
            if time.monotonic() - t0 > time_limit - 2:
                break
            
            obj, grad_z = topk_lse_grad_z(z, stage_T, topk)
            
            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            if (it + 1) % 5 == 0:
                a = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    # Reset momentum occasionally
                    if (it + 1) % 25 == 0:
                        m_z = np.zeros(n)
                        v_z = np.zeros(n)
                        t_adam = 0
        
        a_stage = softmax_fn(z)
        score_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()
    
    return best_a, best_score


def peak_shaving(a0, time_limit, seed=42):
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit - 2:
        b = fast_autocorr(a)
        bmax = b.max()
        
        # Find top peaks
        top_k = min(20, len(b))
        top_peaks = np.argsort(b)[::-1][:top_k]
        
        improved = False
        for k in top_peaks[:8]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]
            
            # Find safe destinations (positions that don't contribute to top peaks)
            danger = np.zeros(n, dtype=bool)
            for kp in top_peaks[:12]:
                jsp = max(0, kp - n + 1)
                jep = min(n - 1, kp)
                danger[jsp:jep + 1] = True
            
            safe = np.where(~danger)[0]
            safe = safe[safe != src]
            if len(safe) == 0:
                safe = np.where(np.arange(n) != src)[0]
            safe_sorted = safe[np.argsort(a[safe])]
            dst_pool = safe_sorted[:max(10, len(safe_sorted) // 3)]
            if len(dst_pool) == 0:
                continue
            
            for dst in dst_pool[:10]:
                for frac in [0.003, 0.005, 0.01, 0.02, 0.04, 0.06, 0.10]:
                    amt = a[src] * frac
                    if amt < 1e-15 or a[src] - amt < -1e-12:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-12:
                        continue
                    a_trial = a_trial / st
                    new_max = fast_autocorr(a_trial).max()
                    if new_max < bmax * (1 - 1e-10):
                        a = a_trial
                        new_score = 2.0 * n * new_max
                        if new_score < best_score:
                            best_score = new_score
                            best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall % 10 == 0:
                # Random perturbation
                n_pert = rng.integers(2, 8)
                a_trial = a.copy()
                for _ in range(n_pert):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a_trial[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.05)
                        amt = a_trial[i] * frac
                        a_trial[i] -= amt
                        a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-10:
                    a_trial = a_trial / st
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            
            if stall > 300:
                break
    
    return best_a, best_score


def mirror_descent_refine(a0, time_limit, eta=0.01, seed=999):
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    current_eta = eta
    
    while time.monotonic() - t0 < time_limit - 2:
        b = fast_autocorr(a)
        bmax = b.max()
        threshold = bmax * 0.98
        top_peaks = np.where(b >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(b)[::-1][:5]
        
        # Compute importance
        importance = np.zeros(n)
        for k in top_peaks[:15]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            importance[idx] += a[k - idx]
        
        imax = importance.max()
        if imax > 1e-15:
            importance = importance / imax
        
        updates = 1.0 - current_eta * importance
        updates = np.maximum(updates, 0.01)
        updates = updates / updates.mean()
        a_new = a * updates
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new / sn
        
        score_new = 2.0 * n * fast_autocorr(a_new).max()
        if score_new < best_score:
            a = a_new
            best_score = score_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall % 30 == 0:
                current_eta *= 0.8
                if current_eta < 1e-6:
                    current_eta = eta
                    a = best_a.copy()
                    noise = rng.normal(0, 0.02, n)
                    a = a * (1 + noise)
                    a = np.maximum(a, 0)
                    sn = a.sum()
                    if sn > 1e-10:
                        a = a / sn
            if stall > 400:
                break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
