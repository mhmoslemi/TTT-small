def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0} using bisection."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.ones(n) * target_sum / n
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    # Fix numerical issues
    s = result.sum()
    if s > 0:
        result = result / s * target_sum
    else:
        result = np.ones(n) * target_sum / n
    return result

    v = v / s
    # Proper simplex projection via bisection on theta
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho_arr = np.nonzero(cond)[0]
    if len(rho_arr) == 0:
        return np.ones(n) * target_sum / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    s2 = result.sum()
    if s2 > 1e-15:
        result = result / s2 * target_sum
    else:
        result = np.ones(n) * target_sum / n
    return result


def lse_obj_and_grad(a, T):
    """Compute LSE objective and gradient: (1/T)*log(sum exp(T*c_k)) where c=a*a."""
    n = len(a)
    c = fast_autocorr(a)  # length 2n-1
    Tc = T * c
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    Z = exp_Tc.sum()
    obj = (max_Tc + np.log(Z)) / T

    w = exp_Tc / Z  # weights

    # Compute gradient: grad[m] = T * sum_k w[k] * (a[k-m] + a[m-k])
    # Using FFT cross-correlation
    a_pad = np.zeros(2 * n - 1)
    a_pad[:n] = a

    fft_size = 1 << (2 * (2 * n - 1) - 1).bit_length()
    W_fft = rfft(w, fft_size)
    A_fft = rfft(a_pad, fft_size)

    # S1[m] = sum_j w[j+m] * a_pad[j] = cross_correlation(w, a_pad) at lag m
    S1 = irfft(W_fft * np.conj(A_fft), fft_size)[:n]
    # S2[m] = (w * a_pad)[m] = convolution at index m
    S2 = irfft(W_fft * A_fft, fft_size)[:n]

    grad = T * (S1 + S2)
    return obj, grad


def nesterov_lse_optimize(a0, time_limit, T_start=200.0, T_end=0.5, n_phases=30, verbose=True):
    """
    Nesterov-accelerated LSE optimization with adaptive temperature.
    Much faster per iteration than LP (O(n log n) vs O(n^2) or more).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()  # Nesterov lookahead
    v = np.zeros(n)  # momentum velocity

    best_a = a.copy()
    best_score = internal_score(a)

    t0 = time.monotonic()
    t_iter = 0
    base_lr = 0.05
    momentum_beta = 0.85

    # Temperature schedule: geometric
    temps = np.geomspace(T_start, T_end, n_phases)
    phase_len = max(20, int(time_limit * 50 / n_phases))

    stall = 0
    stall_limit = 8000

    for phase_i, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break

        for it in range(phase_len):
            if time.monotonic() - t0 > time_limit:
                break
            t_iter += 1

            # Compute gradient at lookahead point y
            obj_y, grad_y = lse_obj_and_grad(y, T)

            # Nesterov update
            momentum = (t_iter - 1.0) / (t_iter + 2.0) if t_iter > 1 else 0.0
            lr_t = base_lr / (1.0 + 0.005 * t_iter)

            y_new = y - lr_t * (momentum * v + (1 - momentum) * grad_y)
            y_new = project_simplex(y_new)

            # Projected a update (Nesterov step)
            a_new = y_new + momentum * (y_new - y)
            a_new = project_simplex(a_new)

            v = y_new - y
            y = y_new
            a = a_new

            # Check score periodically
            if t_iter % 50 == 0 or it == phase_len - 1:
                score_a = internal_score(a)
                if score_a < best_score:
                    best_score = score_a
                    best_a = a.copy()
                    stall = 0
                else:
                    stall += 50

                score_y = internal_score(y)
                if score_y < best_score:
                    best_score = score_y
                    best_a = y.copy()
                    stall = 0

            if stall > stall_limit and phase_i > n_phases // 2:
                break

        if verbose and phase_i % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Nesterov-LSE phase {phase_i}/{n_phases} (T={T:.1f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def spectral_shaping(a, n_iter=500, alpha=0.3):
    """Iteratively flatten power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = internal_score(a)

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2

        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 50.0)
        scale[1:] = np.power(ratio, alpha / 2.0)

        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where T[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.SCS, cp.ECOS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def solve_lp_best_response(a_fixed, M, solver, T=None):
    """Solve: max sum(b) s.t. (a_fixed*b)_k <= M for all k, b >= 0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    T_mat = T if T is not None else build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M, b_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, max_iter=300, verbose=False)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, max_iter=500, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    b_val = b_val / s
    return b_val


def frank_wolfe_lp(a0, time_limit, solver, verbose=True):
    """
    Frank-Wolfe / LP best-response iteration.
    Given a, solve LP to find b maximizing sum(b) s.t. (a*b)_k <= M.
    Then move a toward b.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    y_old = a.copy()

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    T_mat = build_conv_matrix(a)
    stall = 0
    stall_limit = 3000

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        # Recompute T and M periodically (every 100 iters or when a changes significantly)
        if t % 100 == 1:
            M = fast_autocorr(y).max()
            T_mat = build_conv_matrix(y)
        else:
            M = fast_autocorr(y).max()

        b_star = solve_lp_best_response(y, M, solver, T_mat)
        if b_star is None:
            continue

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = project_simplex(a_new)

        y_old = y
        y = y_new
        a = a_new

        if t % 200 == 0:
            score_a = internal_score(a)
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()
                stall = 0
            else:
                stall += 200

            score_y = internal_score(y)
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()
                stall = 0

        # Time-averaged
        if t % 1000 == 0:
            avg_candidate = project_simplex((a + y) / 2)
            score_avg = internal_score(avg_candidate)
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_candidate.copy()
                stall = 0

        stall += 1
        if stall > stall_limit:
            break

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      FW-LP t={t} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def peak_targeting_descent(a0, time_limit, seed=42):
    """
    Directed peak reduction: identify contributing coordinates to peaks,
    transfer mass to low-impact positions. Uses exact computation of
    the impact of each transfer.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a)
        c_max = c.max()
        # Find top peaks
        top_k = np.argsort(c)[::-1][:20]

        improved = False
        # For each peak, find the coordinate that contributes most
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            src = idx[np.argmax(contrib)]

            # Find destination: minimize impact on all peaks
            # Approximate: destination j that minimizes sum of a[j']*a[k2-j'] over peaks
            peak_impact = np.zeros(n)
            for k2 in top_k[:10]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                if js2 > je2:
                    continue
                for j in range(js2, je2 + 1):
                    # If we put mass at j, it contributes a[j]*a[k2-j] to c[k2]
                    partner = a[k2 - j] if 0 <= k2 - j < n else 0
                    peak_impact[j] += partner

            # Choose destination with lowest peak_impact
            candidates = [j for j in range(n) if j != src]
            if not candidates:
                continue
            cand_arr = np.array(candidates)
            sorted_cand = cand_arr[np.argsort(peak_impact[cand_arr])[:50]]
            dst = sorted_cand[rng.integers(len(sorted_cand))]

            # Try different transfer amounts
            for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = internal_score(a_trial)
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            # Random perturbation to escape local minimum
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.002, 0.03)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = internal_score(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            stall += 1
            if stall > 1000:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def subgradient_optimize(a0, time_limit, verbose=True):
    """
    Smoother subgradient method with Nesterov momentum.
    Uses softmax-weighted peaks for a smooth subgradient of max(c).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    v = np.zeros(n)
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    t = 0
    momentum_beta = 0.9
    base_lr = 0.08

    while time.monotonic() - t0 < time_limit:
        t += 1

        # Smooth subgradient at y
        T_smooth = max(1.0, 80.0 * np.exp(-t / 5000.0))
        obj_y, grad_y = lse_obj_and_grad(y, T_smooth)

        # Nesterov step
        lr_t = base_lr / (1.0 + 0.01 * t)
        momentum = min(1.0, 0.85 + 0.1 * t / 10000.0)

        y_new = y - lr_t * (momentum * v + (1 - momentum) * grad_y)
        y_new = project_simplex(y_new)

        a_new = y_new + momentum * (y_new - y)
        a_new = project_simplex(a_new)

        v = y_new - y
        y = y_new
        a = a_new

        if t % 100 == 0:
            score = internal_score(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()

            score_y = internal_score(y)
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SubGrad t={t} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)

            best_score = sc
            best_seq = a_out.copy()
            print(f"  *** NEW BEST [FinalPeak]: {best_score:.6f}")

    # Final verification
    if best_seq is not None:
        best_seq = np.maximum(best_seq, 0)
        s = best_seq.sum()
        if s > 0:
            best_seq = best_seq / s
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
