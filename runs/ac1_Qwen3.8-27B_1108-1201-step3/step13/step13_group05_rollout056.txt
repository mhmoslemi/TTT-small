def fast_conv(a):
    """Fast linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def eval_score(a):
    """Compute the score without calling evaluate_sequence (faster)."""

    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_conv(a)
    return float(2.0 * n * c.max() / (s * s))

def normalize(a, scale=None):
    """Normalize a to have sum = scale (default sqrt(2n))."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    n = len(a)
    if scale is None:
        scale = np.sqrt(2.0 * n)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * scale / n
    return a * scale / s

def proj_simplex(v, s):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) + s - cssv > 0
    if not np.any(cond):
        return np.zeros(n)
    rho = u[np.argmax(cond)]
    return np.maximum(v - rho, 0.0)

def subgrad_conv(a, ac, top_k=5, beta=3.0):
    """Compute soft subgradient of max(conv(a,a)) using top-k lags."""
    n = len(a)
    k_idx = np.argsort(ac)[::-1][:min(top_k, len(ac))]
    vals = np.maximum(ac[k_idx], 0)
    w = vals ** beta
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(len(k_idx)) / len(k_idx)
    else:
        w = w / ws
    g = np.zeros(n)
    for wi, k in zip(w, k_idx):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            g[j] += 2.0 * wi * a[k - j]
    return g

def mirror_descent(a0, time_limit, top_k=5, beta=3.0, verbose=True):
    """Mirror descent on the simplex for minimizing max(conv(a,a))."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = normalize(a0, c)
    best_a = a.copy()
    best_m = fast_conv(a).max()

    avg_a = a.copy()
    n_avg = 1

        if time.monotonic() - t0 > time_limit:
            break
        ac = fast_conv(a)
        g = subgrad_conv(a, ac, top_k, beta)
        step = gamma / (np.sqrt(t) + 1.0)
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - step * g
        max_log = log_a_new.max()
        a_new = np.exp(log_a_new - max_log)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new * c / s
            a = a_new
        M = fast_conv(a).max()
        if M < best_m:
            best_m = M
            best_a = a.copy()
        avg_a += a
        n_avg += 1
        if n_avg % 200 == 0:
            avg = normalize(avg_a, c)
            M_avg = fast_conv(avg).max()
            if M_avg < best_m:
                best_m = M_avg
                best_a = avg.copy()
        if verbose and t % 5000 == 0:
            print(f"    MD t={t}: J={2.0*n*best_m/c**2:.6f}")
    return best_a

def lp_solve(a, M, solver):
    """Solve LP: max sum(b) s.t. conv(a,b)_k <= M, b >= 0."""
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    prob = cp.Problem(cp.Maximize(cp.sum(b)), [T @ b <= M])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def lp_optimize(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = normalize(a0, c)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    y = a.copy()
    x = a.copy()
    stall = 0
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        ac = fast_conv(y)
        M = ac.max()
        if M < 1e-12:
            break
        b = lp_solve(y, M, solver)
        if b is None:
            stall += 1
            if stall > 5:
                break
            continue
        stall = 0
        g = normalize(b, c)
        alpha = 2.0 / (t + 2.0)
        y_new = normalize((1 - alpha) * y + alpha * g, c)
        if t == 1:
            x_new = g.copy()
        else:
            beta_nc = (t - 1.0) / (t + 2.0)
            x_new = normalize(y_new + beta_nc * (y_new - y), c)
        y = y_new
        x = x_new

            Mc = fast_conv(cand).max()
            if Mc < best_m:
                best_m = Mc
                best_a = cand.copy()
        if verbose and t % 10 == 0:
            print(f"    LP t={t}: J={2.0*n*best_m/c**2:.6f}")
    return best_a

def coord_polish(a0, time_limit, verbose=True):
    """Coordinate descent polish."""
    n = len(a0)
    c = np.sqrt(2.0 * n)
    a = normalize(a0, c)
    best_a = a.copy()
    best_m = fast_conv(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    stall = 0
    while time.monotonic() - t0 < time_limit and stall < 100:
        ac = fast_conv(a)
        top_k = np.argsort(ac)[::-1][:5]
        improved = False
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            src = idx[np.argmax(contrib)]
            # Find destination with low contribution to top peaks
            low_mask = np.ones(n, dtype=bool)
            for k2 in top_k:
                k2 = int(k2)
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                low_mask[js2:je2+1] = False
            cands = np.where(low_mask)[0]
            if len(cands) == 0:
                cands = np.where(np.arange(n) != src)[0]
            cands = cands[np.argsort(a[cands])]
            dst = cands[rng.integers(min(10, len(cands)))]
            if dst == src:
                continue
            for frac in [0.01, 0.02, 0.05, 0.1]:
                amt = a[src] * frac
                if amt < 1e-20:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = normalize(a_trial, c)
                M_trial = fast_conv(a_trial).max()
                if M_trial < best_m - 1e-12:
                    a = a_trial
                    best_m = M_trial
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
        if not improved:
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > c * 1e-6:
                frac = rng.uniform(0.02, 0.1)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                if a[i] >= 0:
                    a = normalize(a, c)
                    M_new = fast_conv(a).max()
                    if M_new < best_m:
                        best_m = M_new
                        best_a = a.copy()
                        stall = 0
    if verbose:
        print(f"    Coord: J={2.0*n*best_m/c**2:.6f} ({time.monotonic()-t0:.0f}s)")
    return best_a

def interp_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    fn = f / f.sum()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    fi = np.interp(x_new, x_old, fn)
    fi = np.maximum(fi, 0)
    s = fi.sum()
    if s < 1e-15:
        fi = np.ones(new_n) / new_n
    else:
        fi = fi / s
    return fi

def lp_solve_topleaks(a, M, solver, top_lags=20):
    """Solve LP with only top-k lag constraints."""
    n = len(a)
    ac = fast_conv(a)
    top_indices = np.argsort(ac)[::-1][:top_lags]
    b = cp.Variable(n, nonneg=True)
    T = np.zeros((len(top_indices), n))
