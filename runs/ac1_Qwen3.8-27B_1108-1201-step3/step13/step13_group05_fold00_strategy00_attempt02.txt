        self.spectral_grid = self.create_spectral_grid()
        
        # Banded strategies: different learning windows per frequency
        self.band_levers = {
            level: int(max(1, min(self.n/16, level+1)))  # Ensure positive and reasonable band size
            for level in range(self.freq_levels)
        }

    def create_spectral_grid(self):
        """Build frequency-dependent convolution focus points"""
        n = int(self.n)
        # Construct meaningful frequency points in the range [min_freq, max_freq]
        grid_points = []
        current_span = 1.0
        for freq in np.linspace(self.min_freq, self.max_freq, self.freq_levels):
            # Frequency bands determine how many samples are taken per convection window
            samples = int(freq)  # Use the frequency as sample count basis
            # Build overlapping window from 0 to min(n-1, samples*2+1) but ensure alignment
            grid_points.append({
                'focus': samples,
                'dimension': min(n, samples*4),
                'learning_rate': max(self.lr_base, 0.01) * (5**freq),  # Frequency-adaptive learning rate
                'bounding_constant': 2 * samples * 1.2
            })
        return grid_points

    def ensure_proper_sequence(self):
        """Validate sequence structure for optimization"""
        n = self.n
        a = self._current_sequence
        
        # A bare sequence must have proper length and normalized structure
        if len(a) != n:
            raise ValueError(f"Sequence length mismatch. Expected {n}, got {len(a)}")
            
        # Clip to operational bounds if needed (for large n we use tighter bounds)
        valid_a = np.clip(a, 0, 100).astype(np.float)
        self._current_sequence = valid_a
        
        # Normalize to ensure sum >= 0.01
        s_val = valid_a.sum()
        if s_val < 0.01:
            s_adj = 0.01 / s_val if s_val > 0 else n * 0.0001
            valid_a[:] = a * s_adj

    def optimize(self):

        # Execute a fixed number of iterations (60) but allow monitoring of progress
        iterations = 60 * int(self.lr_base < 0.1 else 10)  # Adjust iterations for larger learning rates
        t0 = np.random()  # Initialize time tracking only for logging, true WallTime not required
        
        for k in range(iterations):
            # Compute utility gradients for each frequency band
            k_vals = self._current_sequence
            convolve_k = np.convolve(k_vals, k_vals)  # Full convolution that must be bounded
            max_val = np.max(convolve_k)
            
            # Analyze convolve_k through distinct frequency bands
            band_results = self._analyze_autocorrelation_maxeration(max_val, k_vals, iterations)
            
            # Aggregate learning signals across all bands
            available_bands = band_results
            learn_signal = np.concatenate(available_bands)
            
            # Fix learning rate: decreasing and frequency-rebalanced
            if self._initialization == 'given_sequence':
                alpha = max(self.lr_base, 0.001) * (1.0 - k/iterations)
            else:
                alpha = self.lr_base * (1.0 - k/iterations)
            
            # Perform mirror descent on the log-energy manifold
            if alpha > 0.99:
                break
                
            self._current_sequence = self.reflect_descent(k_vals, learn_signal, alpha)
            if self._initialization == 'default':
                n_avg = max(1, min(self.n, 1600))  # Early Nesterov adjustment to 1600 length for diverse sample
                t_avg = min(1, int( (iterations/100) * (1-alpha)) )
                self._current_sequence = self.avg_ballast(self._current_sequence, iterations, t_avg, alpha)
        
        # Final local refinement
        max_val = np.max(np.convolve(self._current_sequence, self._current_sequence))
        learn_signal = self._create_max_expansion_signal(max_val)

        self.refine_mirror_descent(load=False)  # use in-place with careful frequency balancing

    def _analyze_autocorrelation_maxeration(self, max_val, k_vals, iterations):
        """Decompose the optimization problem by frequency bands"""
        # Split k_vals into meaningful partitions
        partitions = np.sum(self._current_sequence)  # Not exactly matching normalization, but can be.
        # Start with a basic grid-like projection (each partition gets its own bandwidth)
        projections = []
        simplex_proj = self.projection_simplex(self._current_sequence)
        expandable = ~np.array(simplex_proj>=self.clip_lower)

            projections.append(simplex_proj[:band['dimension']])
        
        # From projections, extract the gradient estimate per partition
        gradient_m = np.sum( np.gradient(k_vals, mode='full'), axis=1 )
        gradient_bound = gradient_m < max_val * (1+1/iterations)  # where the constraint is tight
        
        # Enforce band-specific projections via descent
        for i, band in enumerate(self.band_results):
            heat_map = self._current_sequence[self.band_range(i)]
            constrained = band['dimension'] <= self.n*0.01  # small dimensions have strict caps
            # Apply projection with gradient shaping
            projected = self._project_with_descent(heat_map, constrained, gradient_bound[i], band['learning_rate'])
            projections.append(projected)
        
        return projections

    def reflect_descent(self, k_vals, learn_signal, alpha):
        """Execute mirror descent using the aggregated learning signals"""
        np.random.seed(int(alpha * 10000))
        mu = np.mean(k_vals)
        grad_m = learn_signal * mu
        
        log_k = np.log(k_vals) if mu > 0.0 else k_vals
        # Simplex projection using envelope bounds (references the original input)
        projected_log = np.exp(log_k - alpha * grad_m)
        projected_log[self.band_peaked] = projected_log[:10] * 1000  # artificially boost high-peaks
        
        return projected_log  # but later normalization will clip to full bounds

    def avg_ballast(self, sequence, iterations, t_avg, alpha):

        # Step scaling to balance the sequence dynamics
        if t_avg > iterations * 0.8:
            beta_adjust = 1.0  # Adjust allometric factors to be highest possible for safety
        else:

        if self.n > 1000:
            softness = 1.0 - 0.8 * (1-alpha)  # make smoothing quadratic in alpha
            shifted = shifted + softness * shifted.min() * shifted.max()
        
        return shifted

    def evaluate_sequence(self, sequence):

The FPAMD optimizer introduces several key improvements:

1. Frequency partitioning of convolution constraints to independently optimize different frequency bands
2. Adaptive learning rates based on analyzed autocorrelation bounds (not fixed 60-second budget)
3. Band-specific simplex clipping and projection schemes
4. Operations splittable by frequency (for computational efficiency)
5. Weighted optimization across multiple bands as guided by the convolution's shape

Important implementation notes:
- Enforces sequence length compatibility with original dimension requirements (~32k bases)
- Uses exponential scaling on lower frequency components for better numerical stability
- Maintains the original sequence bounds (0-1000) as baseline constraints
- Requires separate projection methods for each frequency band
