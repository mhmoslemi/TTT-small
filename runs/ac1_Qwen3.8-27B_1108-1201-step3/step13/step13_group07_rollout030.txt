            avg_b_n = avg_b / n_avg
            # Cross-correlation via FFT
            fft_size = 1 << (2 * n - 1).bit_length()
            A_f = rfft(avg_a_n, fft_size)
            B_f = rfft(avg_b_n, fft_size)
            cross = irfft(A_f * B_f, fft_size)[:2*n-1]
            cross_max = cross.max()
            gap = abs(M_val - cross_max) if M_val is not None else 0
            
            if gap < best_gap:
                best_gap = gap
            
            # Track values for convergence
            col_val = M_val
            # Row value: max_k (a_n * b_n)_k
            row_val = cross_max
            
            if verbose and t % 1000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NL t={t} ({elapsed:.0f}s): best={best_score:.7f}, "
                      f"row={row_val:.8f}, col={col_val:.8f}, gap={gap:.2e}")
        
        if stall_count > stall_threshold:
            if verbose:
                print(f"    NL: stall detected at t={t}, best={best_score:.7f}")
            break
    
    elapsed = time.monotonic() - t0
    if verbose:
        print(f"    NL finished: t={t}, {elapsed:.0f}s, best={best_score:.7f}")
    
    return best_a, best_score


def lse_smoothed_grad(z, a, T_temp):
    """Compute gradient of LSE-smoothed objective w.r.t. a."""
    n = len(a)
    b = fast_autocorr(a)
    Tb = T_temp * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    
    # Gradient: 2 * (w * a_rev) cross-correlation, extract middle n
    fft_size = 1 << (n + 2*n-2).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    grad = 2.0 * conv[n-1:n-1+n]
    return grad, w, b


def adam_lse_optimize(a0, time_limit, T_start=50.0, T_end=0.01, n_stages=30, lr=0.004):
    """LogSumExp-smoothed optimization with Adam optimizer."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = internal_score(a_init)
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_adam = 0
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        # Adaptive iterations per stage
        iters_per_stage = max(20, min(300, int(50 * remaining / (n_stages - si))))
        
        for it in range(iters_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            
            a = softmax_of(z)
            grad_a, w, b = lse_smoothed_grad(z, a, stage_T)
            
            # Convert to z-gradient (Jacobian of softmax)
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)
            
            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)
            
            lr_t = lr / (1.0 + 0.005 * t_adam * 0.01)
            z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            
            if it % 30 == 0:
                a_check = softmax_of(z)
                score = internal_score(a_check)
                if score < best_score:
                    best_score = score
                    best_a = a_check.copy()
        
        a_check = softmax_of(z)
        score = internal_score(a_check)
        if score < best_score:
            best_score = score
            best_a = a_check.copy()
        
        if si % 8 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {si}/{n_stages} T={stage_T:.2f} ({elapsed:.0f}s): best={best_score:.7f}")
    
    return best_a, best_score


def softmax_of(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def coordinate_descent_optimize(a0, time_limit, seed=123):
    """
    Aggressive coordinate descent: identify peak-contributing coordinates
    and redistribute mass to minimize the maximum autocorrelation.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / a.sum()
    
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    stall = 0
    iteration = 0
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        b = fast_autocorr(a)
        
        # Find top peaks
        n_peaks = min(15, 2 * n - 1)
        top_peaks = np.argsort(b)[::-1][:n_peaks]
        current_max = b[top_peaks[0]]
        
        # For each peak, find the coordinate with maximum contribution
        best_move_score = best_score
        best_move = None
        
        for trial in range(3):
            k = top_peaks[rng.integers(len(top_peaks))]
            
            # Find which coordinates contribute most to this peak
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            
            if contrib.max() < 1e-15:
                continue
            
            src_local = np.argmax(contrib)
            src = idx[src_local]
            
            # Find destination: minimize impact on all peaks
            # Prefer coordinates that don't contribute to any top peak
            peak_areas = set()
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                for j2 in range(js2, je2 + 1):
                    peak_areas.add(j2)
            
            # Candidates: not in peak areas, sorted by current value (prefer small)
            candidates = [j for j in range(n) if j not in peak_areas]
            if len(candidates) < 5:
                candidates = [j for j in range(n) if j != src]
            
            # Try different amounts
            for frac in [0.003, 0.007, 0.015, 0.03, 0.05]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                
                # Pick destination that's "safe" (low contribution to peaks)
                # Score each candidate by impact
                best_dst = None
                best_dst_impact = float('inf')
                
                # Sample candidates for speed
                n_sample = min(20, len(candidates))
                sample_idx = rng.choice(len(candidates), size=n_sample, replace=False)
                
                for si in sample_idx:
                    dst = candidates[si]
                    # Estimate impact: how much would adding to dst increase peaks?
                    impact = 0.0
                    for k2 in top_peaks[:5]:
                        if 0 <= k2 - dst < n:
                            impact += a[k2 - dst]
                        if 0 <= dst - k2 < n:
                            impact += a[dst - k2]
                    if impact < best_dst_impact:
                        best_dst_impact = impact
                        best_dst = dst
                
                if best_dst is None:
                    continue
                
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                
                score = internal_score(a_trial)
                if score < best_move_score:
                    best_move_score = score
                    best_move = (src, best_dst, amt)
        
        if best_move is not None and best_move_score < best_score - 1e-12:
            src, dst, amt = best_move
            a[src] -= amt
            a[dst] += amt
            a = np.maximum(a, 0)
            a = a / a.sum()
            best_score = best_move_score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            
            # Random perturbation to escape local minima
            if stall % 50 == 0:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.005, 0.03)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0)
                    if a_trial.sum() > 1e-15:
                        a_trial = a_trial / a_trial.sum()
                        score = internal_score(a_trial)
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
            
            if stall > 2000:
                break
        
        if iteration % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD iter={iteration} ({elapsed:.0f}s): best={best_score:.7f}")
    
    return best_a, best_score


def spectral_shaping(a, n_iter=300, alpha=0.3):
    """Iteratively flatten power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()
    
    best_a = a.copy()
    best_score = internal_score(a)
    
    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 50.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def perturbation_restart(a0, n_perturbations=5, seed=99):
    """Generate diverse perturbations of a good sequence."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    best_a = a0.copy()
    best_score = internal_score(a0)
    
    a = a0.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    
    for p in range(n_perturbations):
        # Add structured perturbation
        noise_scale = 0.1 + 0.1 * p
        perturbation = rng.standard_normal(n) * noise_scale
        # Make it sum-zero to preserve normalization
        perturbation -= perturbation.mean()
        
        a_pert = a + perturbation
        a_pert = np.maximum(a_pert, 0)
        s = a_pert.sum()
        if s < 1e-10:
            continue
        a_pert = a_pert / s
        
        # Quick spectral shaping on the perturbed version
        a_shaped = spectral_shaping(a_pert, n_iter=100, alpha=0.2)
        score = internal_score(a_shaped)
        if score < best_score:
            best_score = score
            best_a = a_shaped.copy()
            print(f"      Perturbation {p}: {best_score:.7f}")
    
    return best_a, best_score
