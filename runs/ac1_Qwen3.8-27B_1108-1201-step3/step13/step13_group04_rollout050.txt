def nesterov_topk_lse(a0, time_limit, T_start=20.0, T_end=0.5, n_stages=15, K=50, verbose=True):
    """Nesterov-accelerated topk-LSE gradient descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    y_prev = a.copy()

    temps = np.geomspace(T_start, T_end, n_stages)
    it_count = 0

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        n_iters_stage = max(20, int(remaining * 3.0 / max(1, n_stages - si)))
        for it in range(n_iters_stage):
            if time.monotonic() - t0 > time_limit:
                break

            # Nesterov extrapolation
            if it == 0:
                y_eval = y
            else:
                momentum = (it - 1.0 + np.sqrt(1.0 + 4.0 * (it - 1.0) ** 2)) / (2.0 * (it + 2.0))
                y_eval = y + momentum * (y - y_prev)
                y_eval = np.maximum(y_eval, 0.0)
                s_ye = y_eval.sum()
                if s_ye < 1e-15:
                    y_eval = y.copy()
                else:
                    y_eval = y_eval * norm / s_ye

            grad = topk_lse_gradient(y_eval, T, K)
            if grad is None:
                y_prev = y.copy()
                continue

            # Normalize gradient
            g_norm = grad / (np.linalg.norm(grad) + 1e-30)
            
            # Adaptive step size
            lr = 0.1 / (1.0 + it * 0.01)
            
            y_new = y - lr * g_norm * y_eval.sum()  # scale with sum for invariance
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y

            y_prev = y.copy()
            y = y_new
            it_count += 1

            if it_count % 10 == 0:
                score = fast_autocorr(y).max()
                if score < best_score:
                    best_score = score
                    best_a = y.copy()
                    if verbose and it_count % 100 == 0:
                        elapsed = time.monotonic() - t0
                        print(f"    TopK-LSE stage={si}/{n_stages} T={T:.3f} it={it_count} K={K} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def simulated_annealing_lse(a0, time_limit, T_start=5.0, T_end=0.1, alpha=0.999, verbose=True):
    """Simulated annealing with mass-transfer moves."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    rng = np.random.default_rng(42)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    curr_score = best_score
    t0 = time.monotonic()
    it_count = 0
    accept_count = 0

    while time.monotonic() - t0 < time_limit:
        T_curr = T_start * (alpha ** it_count)
        if T_curr < T_end:
            T_curr = T_end

        # Pick a random move
        move_type = rng.integers(0, 4)
        a_trial = a.copy()

        if move_type == 0:
            # Single mass transfer
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            frac = rng.uniform(0.01, 0.1)
            amt = a[i] * frac
            if amt < 1e-16:
                continue
            a_trial[i] -= amt
            a_trial[j] += amt
        elif move_type == 1:
            # Two-source to one-target (merge)
            i = rng.integers(n)
            j = rng.integers(n)
            k = rng.integers(n)
            if len(set([i, j, k])) < 2:
                continue
            amt1 = a[i] * rng.uniform(0.01, 0.05)
            amt2 = a[j] * rng.uniform(0.01, 0.05)
            if amt1 < 1e-16 or amt2 < 1e-16:
                continue
            a_trial[i] -= amt1
            a_trial[j] -= amt2
            a_trial[k] += amt1 + amt2
        elif move_type == 2:
            # One-source to two-targets (split)
            i = rng.integers(n)
            j = rng.integers(n)
            k = rng.integers(n)
            if len(set([i, j, k])) < 2:
                continue
            frac = rng.uniform(0.02, 0.15)
            amt = a[i] * frac
            if amt < 1e-16:
                continue
            frac2 = rng.uniform(0.3, 0.7)
            a_trial[i] -= amt
            a_trial[j] += amt * frac2
            a_trial[k] += amt * (1 - frac2)
        else:
            # Gaussian perturbation on a random subset
            n_perturb = rng.integers(5, max(6, n // 100))
            idx = rng.choice(n, size=n_perturb, replace=False)
            a_trial[idx] += rng.normal(0, a[idx].mean() * 0.1, size=n_perturb)

        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-15:
            continue
        a_trial = a_trial * norm / s_t

        trial_score = fast_autocorr(a_trial).max()

        # Accept?
        delta = trial_score - curr_score
        if delta < 0 or rng.random() < np.exp(-delta / T_curr):
            a = a_trial
            curr_score = trial_score
            accept_count += 1
            if curr_score < best_score:
                best_score = curr_score
                best_a = a.copy()
                if verbose and it_count % 500 == 0:
                    elapsed = time.monotonic() - t0
                    print(f"    SA it={it_count} T={T_curr:.4f} ({elapsed:.0f}s): best={best_score:.8f}")

        it_count += 1

        if verbose and it_count % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SA it={it_count} T={T_curr:.4f} accept={accept_count}/{it_count} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def directed_local_search(a0, time_limit, verbose=True):

    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:

    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    it_count = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        peak_val = b.max()
        
        # Find top peaks
        top_k = np.argsort(b)[::-1][:10]
        
        improved = False
        for k_star in top_k[:5]:
            # Find contributing (i, k_star-i) pairs
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_star - idx]
            
            # Sort by contribution

            src_positions = idx[order[:5]]
            
            for src in src_positions:
                if a[src] < norm * 1e-9:
                    continue
                
                # Find positions that minimize impact on top peaks

                    # Position src contributes to peak k2 if a[src]*a[k2-src] is significant

                    # We want dst such that these peaks are not in top_k
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)

                        if a[k2 - p] > 0:
                            bad_positions.add(p)
                
                # Also avoid positions that would create new high peaks
                # Positions near existing large values are risky
                top_vals = np.argsort(a)[::-1][:n//4]
                bad_positions.update(top_vals.tolist())
                
                candidates = [p for p in range(n) if p not in bad_positions and p != src]
                if not candidates:
                    candidates = [p for p in range(n) if p != src]
                
                # Pick from candidates with lowest current value
                candidates = sorted(candidates, key=lambda p: a[p])[:20]

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def topk_lse_gradient(a, T, K):
    """
    Compute gradient of T*log(sum_{k in topK} exp(b_k/T)) where b = autocorr(a).
    Only considers the K largest peaks for computational efficiency.
    Returns gradient vector of shape (n,).
    """
    n = len(a)
    b = fast_autocorr(a)
    
    if K >= 2 * n - 1:
        top_idx = np.arange(2 * n - 1)
    else:
        top_idx = np.argpartition(b, -K)[-K:]
    
    b_k = b[top_idx]
    max_bk = b_k.max()
    w = np.exp((b_k - max_bk) / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w_full = np.zeros(2 * n - 1)
        w_full[n - 1] = 1.0
    else:
        w_full = np.zeros(2 * n - 1)
        w_full[top_idx] = w / w_sum
    
    fft_size = 1 << (3 * n - 2).bit_length()
    A_fwd = rfft(a, fft_size)
    W = rfft(w_full, fft_size)
    a_rev = a[::-1].copy()
    A_rev = rfft(a_rev, fft_size)
    
    conv_fwd = irfft(A_fwd * W, fft_size)
    conv_rev = irfft(A_rev * W, fft_size)
    
    r2 = conv_fwd[:n]
    r1 = conv_rev[n-1:n-1+n]
    
    return r1 + r2

    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    y_prev = a.copy()

    it_count = 0

            break
        remaining = time_limit - (time.monotonic() - t0)

        n_iters = max(30, int(remaining * 3.5 / max(1, n_stages - si)))
        for it in range(n_iters):

            else:
                momentum = (it - 1.0 + np.sqrt(1.0 + 4.0 * (it - 1.0) ** 2)) / (2.0 * (it + 2.0))
                y_eval = y + momentum * (y - y_prev)
                y_eval = np.maximum(y_eval, 0.0)
                s_ye = y_eval.sum()
                if s_ye < 1e-15:
                    y_eval = y.copy()
                else:
                    y_eval = y_eval * norm / s_ye

            grad = topk_lse_gradient(y_eval, T, K)
            
            # Step size: use a fixed fraction approach
            lr = 0.5 / (T + 1.0)
            
            y_new = y - lr * grad
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y

            y_prev = y.copy()
            y = y_new
            it_count += 1

            if it_count % 5 == 0:
                score = fast_autocorr(y).max()
                if score < best_score:
                    best_score = score
                    best_a = y.copy()
                    if verbose and it_count % 200 == 0:
                        elapsed = time.monotonic() - t0
                        print(f"    LSE stage={si}/{n_stages} T={T:.3f} it={it_count} K={K} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def directed_local_search(a0, time_limit, seed=42, verbose=True):
    """
    Directed local search targeting the top peaks.
    For each top peak, identifies high-contributing source positions and
    tries transferring mass to low-impact target positions.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    it_count = 0
    stall = 0
    b = fast_autocorr(a)

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:8]
        peak_val = b.max()
        
        improved = False
        
        for k_star in top_k[:5]:
            # Find contributing positions to this peak
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k_star - idx]
            
            # Top contributors
            top_contrib = idx[np.argsort(contrib)[::-1][:8]]
            
            for src in top_contrib:
                if a[src] < norm * 1e-10:
                    continue
                
                # Find good targets: positions that don't contribute to top peaks
                # and have low current values
                # Precompute which positions are "dangerous" (contribute to top peaks)
                dangerous = np.zeros(n, dtype=bool)
                for k2 in top_k[:5]:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    for p in range(js2, je2 + 1):
                        if a[k2 - p] > norm * 0.01:
                            dangerous[p] = True
                
                safe_positions = np.where(~dangerous & (np.arange(n) != src))[0]
                if len(safe_positions) < 10:
                    safe_positions = np.where(np.arange(n) != src)[0]
                
                # Sample from safe positions, preferring low values
                n_sample = min(30, len(safe_positions))
                # Weight by inverse value (prefer low values)
                weights = 1.0 / (a[safe_positions] + norm * 1e-6)
                weights /= weights.sum()
                chosen = rng.choice(safe_positions, size=n_sample, replace=False, p=weights)
                
                best_move = None
                best_move_score = best_score - 1e-15  # need to beat current best
                
                for dst in chosen:
                    for frac in [0.01, 0.02, 0.05, 0.1, 0.2]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < -1e-15:
                            continue
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_move_score:
                            best_move_score = score
                            best_move = (src, dst, amt, a_trial.copy())
                
                if best_move is not None and best_move_score < best_score:
                    a = best_move[3]
                    best_score = best_move_score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    it_count += 1
                    b = fast_autocorr(a)
                    break
            
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation
            if stall % 50 == 0:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > norm * 1e-10:
                    frac = rng.uniform(0.01, 0.2)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score * 1.001:  # allow slight degradation
                            a = a_trial
                            if score < best_score:
                                best_score = score
                                best_a = a.copy()
                                stall = 0
            
            if stall > 2000:
                if verbose:
                    print(f"    DLS: stalled at {best_score:.8f} ({time.monotonic()-t0:.0f}s)")
                # Larger random perturbation
                n_pert = rng.integers(max(5, n//100), max(6, n//30))
                idx = rng.choice(n, size=n_pert, replace=False)
                a[idx] += rng.normal(0, a[idx].mean() * 0.5, size=n_pert)
                a = np.maximum(a, 0.0)
                s_a = a.sum()
                if s_a > 1e-15:
                    a = a * norm / s_a
                stall = 0

        if verbose and it_count % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    DLS it={it_count} stall={stall} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def simulated_annealing(a0, time_limit, T_start=3.0, T_end=0.01, seed=99, verbose=True):
    """Simulated annealing with diverse move types for global exploration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    curr_score = best_score
    t0 = time.monotonic()
    it_count = 0
    accept = 0
    cooling_rate = (T_end / T_start) ** (1.0 / max(1, int(time_limit * 500)))

    while time.monotonic() - t0 < time_limit:
        T_curr = T_start * (cooling_rate ** it_count)
        if T_curr < T_end:
            T_curr = T_end

        # Choose move type
        mt = rng.integers(0, 5)
        a_trial = a.copy()

        if mt == 0:
            # Single transfer
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            frac = rng.uniform(0.01, 0.15)
            amt = a[i] * frac
            if amt < 1e-16:
                continue
            a_trial[i] -= amt
            a_trial[j] += amt
        elif mt == 1:
            # Single transfer (larger)
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            frac = rng.uniform(0.05, 0.4)
            amt = a[i] * frac
            if amt < 1e-16:
                continue
            a_trial[i] -= amt
            a_trial[j] += amt
        elif mt == 2:
            # Two-to-one
            i = rng.integers(n)
            j = rng.integers(n)
            k = rng.integers(n)
            if len(set([i, j, k])) < 2:
                continue
            a_trial[i] -= a[i] * rng.uniform(0.01, 0.05)
            a_trial[j] -= a[j] * rng.uniform(0.01, 0.05)
            a_trial[k] += a[i] * 0.03 + a[j] * 0.03
        elif mt == 3:
            # One-to-two (split)
            i = rng.integers(n)
            j = rng.integers(n)
            k = rng.integers(n)
            if len(set([i, j, k])) < 2:
                continue
            amt = a[i] * rng.uniform(0.02, 0.1)
            if amt < 1e-16:
                continue
            f = rng.uniform(0.3, 0.7)
            a_trial[i] -= amt
            a_trial[j] += amt * f
            a_trial[k] += amt * (1 - f)
        else:
            # Gaussian perturbation on subset
            m = rng.integers(3, max(4, n // 80))
            idx = rng.choice(n, size=m, replace=False)
            a_trial[idx] += rng.normal(0, a[idx].mean() * 0.2, size=m)

        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-15:
            continue
        a_trial = a_trial * norm / s_t

        trial_score = fast_autocorr(a_trial).max()
        delta = trial_score - curr_score

        if delta < 0 or rng.random() < np.exp(-delta * norm / T_curr):
            a = a_trial
            curr_score = trial_score
            accept += 1
            if curr_score < best_score:
                best_score = curr_score
                best_a = a.copy()

        it_count += 1

        if verbose and it_count % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SA it={it_count} T={T_curr:.4f} acc={accept}/{it_count} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score
