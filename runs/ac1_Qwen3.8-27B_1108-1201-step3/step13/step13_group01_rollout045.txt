def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def _fast_score(a):
    s = a.sum()
    if s < 1e-15:
        return 1e10
    n = len(a)
    b = _fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)

def _lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        s_i = i
        e_i = min(i + n, size)
        T[s_i:e_i, i] = a[:e_i - s_i]
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None, None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    dual_val = None
    try:
        dual_val = np.asarray(prob.constraints[0].dual_value, dtype=np.float64)
        if dual_val is not None:
            dual_val = np.maximum(dual_val, 0.0)
    except Exception:
        pass
    return b_val, dual_val

def _lse_obj_and_grad(x_flat, n, T_temp, w_vec):
    a = np.maximum(x_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = _fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    shifted = T_temp * (b - max_b)
    shifted = np.minimum(shifted, 500.0)
    exp_vals = np.exp(shifted)
    weighted = w_vec * exp_vals
    lse_val = max_b + np.log(weighted.sum()) / T_temp
    obj = 2.0 * n * lse_val / (s * s)
    w = weighted / weighted.sum()
    fft_size = 1 << (3 * n - 2).bit_length()
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    w_padded = np.zeros(fft_size)
    w_padded[:2 * n - 1] = w
    A_fft = rfft(a_padded, fft_size)
    W_fft = rfft(w_padded, fft_size)
    cross = irfft(A_fft * np.conj(W_fft), fft_size)[:n]
    conv_grad = 4.0 * cross
    grad = conv_grad / (s * s) - 2.0 * obj * a / s
    return obj, grad.ravel()


def _lse_fista(a0, time_limit, T_start=50.0, T_end=1.0, n_stages=8, w_vec=None):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_score(a)
    t0 = time.monotonic()

    if w_vec is None:
        w_vec = np.ones(2 * n - 1)

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        stage_time = min(remaining / (n_stages - si), 180.0)
        if stage_time < 5:
            break

        def obj_cb(x):
            val, _ = _lse_obj_and_grad(x, n, T, w_vec)
            return val

        def grad_cb(x):
            _, g = _lse_obj_and_grad(x, n, T, w_vec)
            return g

        x0 = a.copy()
        bounds = [(0.0, 1000.0) for _ in range(n)]
        try:
            res = scipy_minimize(obj_cb, x0, jac=grad_cb, method='L-BFGS-B',
                                 bounds=bounds,
                                 options={'maxiter': 300, 'maxfun': 1000,
                                          'gtol': 1e-8, 'ftol': 1e-10})
            if res.x is not None:
                a_cand = np.maximum(res.x, 0.0)
                s_c = a_cand.sum()
                if s_c > 1e-10:
                    a_cand = a_cand * norm / s_c
                    sc = _fast_score(a_cand)
                    if sc < best_score:
                        best_score = sc
                        best_a = a_cand.copy()
                        a = a_cand
                        print(f"  LSE stage {si} T={T:.2f}: score={sc:.8f}")
        except Exception as e:
            print(f"  LSE stage {si} error: {e}")

    return best_a, best_score


def _coord_descent(a0, time_limit, seed=42):

        a = a * norm / s0

    while time.monotonic() - t0 < time_limit:
        b = _fast_autocorr(a)
        top_k = 6

        for k_peak in top_peaks:

            idx = np.arange(js, je + 1)

                continue

            peak_mask = np.zeros(n, dtype=bool)

                    dst_pool = candidates[np.argsort(a[candidates])][:5]

                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 5 == 0:
                for _ in range(30):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i == j or a[i] < 1e-10:
                        continue
                    frac = rng.uniform(0.05, 0.3)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = _fast_score(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                        improved = True
                        break
            if stall > 300:
                break

    return best_a, best_score


def _interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def _structured_init(n, rng):
    choice = int(rng.integers(0, 8))
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.5) ** 2)
    elif choice == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 6:
        a = np.random.default_rng(int(rng.integers(0, 10**6))).random(n)
    else:
        x = np.linspace(0.0, 1.0, n)
        a = 0.3 + 0.5 * np.sin(2 * np.pi * x * 3) ** 2
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def _dual_guided_lse(a0, time_limit, solver, w_vec=None):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_score(a)
    t0 = time.monotonic()

    for it in range(20):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 20:
            break

        b = _fast_autocorr(a)
        M_curr = b.max()
        b_resp, dual = _lp_best_response(a, M_curr, solver)
        if b_resp is None:
            continue

        if dual is not None and dual.sum() > 1e-15:
            w = dual / dual.sum()
            w = w ** 0.5
            w = w / w.sum()
        else:
            w = np.ones(2 * n - 1) / (2 * n - 1)

        stage_time = min(remaining * 0.4, 60.0)
        a_out, sc = _lse_fista(a, stage_time, T_start=20.0, T_end=2.0,
                               n_stages=4, w_vec=w)
        if sc < best_score:
            best_a = a_out
            best_score = sc
            a = a_out

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    solver = _get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = _eval_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = _eval_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 0: Multi-length quick scan
    print("\n=== Phase 0: Multi-length scan ===")
    if best_seq is not None:
        base_n = len(best_seq)
    else:
        base_n = 1600
        best_seq = _structured_init(base_n, rng)
        best_score = _fast_score(best_seq)

    scan_lengths = []
    for dn in [0, 200, 400, 600]:
        nn = base_n + dn
        if nn >= 800 and nn <= 4000:
            scan_lengths.append(nn)

    scan_time_total = time_budget * 0.12
    per_scan = scan_time_total / max(1, len(scan_lengths))
    scan_results = {}

    for n_scan in scan_lengths:
        if time.monotonic() - t_global > t_global + scan_time_total:
            break
        st = time.time()
        if n_scan == base_n and best_seq is not None:
            init_seq = best_seq.copy()
        else:
            init_seq = _interpolate_seq(best_seq, n_scan) if best_seq is not None else _structured_init(n_scan, rng)
        a_norm = init_seq * np.sqrt(2.0 * n_scan) / init_seq.sum()
        alloc = min(per_scan, 80.0)
        if alloc < 10:
            break
        a_out, sc_out = _nesterov_lp(a_norm, alloc, solver, verbose=False)
        scan_results[n_scan] = (a_out, sc_out)
        print(f"  n={n_scan}: score={sc_out:.8f} ({time.monotonic() - t_global:.0f}s)")

    if scan_results:
        best_scan_n = min(scan_results, key=lambda k: scan_results[k][1])
        try_update(scan_results[best_scan_n][0], f"scan_n{best_scan_n}")

    # Determine best working length
    n_work = len(best_seq)
    if scan_results:
        best_scan_n = min(scan_results, key=lambda k: scan_results[k][1])
        if scan_results[best_scan_n][1] < _fast_score(best_seq):
            n_work = best_scan_n

    # Phase 1: Deep LP-Nesterov at best length
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.40
    print(f"\n=== Phase 1: Deep LP n={n_work} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_init = best_seq.copy()
        if n_work != len(a_init):
            a_init = _interpolate_seq(a_init, n_work)
        a_norm = a_init * np.sqrt(2.0 * n_work) / a_init.sum()
        a_out, _ = _nesterov_lp(a_norm, t1, solver, verbose=True)
        try_update(a_out, f"lp_deep_n{n_work}")

    # Phase 2: Dual-guided LSE
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.18, 240.0)
    print(f"\n=== Phase 2: Dual-guided LSE n={n_best} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_init = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = _dual_guided_lse(a_init, t2, solver)
        try_update(a_out, f"dual_lse_n{n_best}")

    # Phase 3: Try slightly different length
    n_alt = n_work + 200 if n_work < 3000 else n_work - 200
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 3: LP n={n_alt} (budget={t3:.0f}s) ===")
    if t3 > 30 and n_alt >= 800:
        init_alt = _interpolate_seq(best_seq, n_alt)
        a_norm = init_alt * np.sqrt(2.0 * n_alt) / init_alt.sum()
        a_out, _ = _nesterov_lp(a_norm, t3, solver, verbose=True)
        try_update(a_out, f"lp_alt_n{n_alt}")

    # Phase 4: LSE polish with low temperature
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.12, 120.0)
    print(f"\n=== Phase 4: LSE polish n={n_best} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_init = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = _lse_fista(a_init, t4, T_start=10.0, T_end=0.5, n_stages=5)
        try_update(a_out, f"lse_polish_n{n_best}")

    # Phase 5: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.25, 300.0)
    print(f"\n=== Phase 5: Coord descent n={n_best} (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_init = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = _coord_descent(a_init, t5, seed=12345)
        try_update(a_out, "coord_polish")

    # Phase 6: Final coord descent with different seed
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t6 = min(remaining * 0.5, 300.0)
    print(f"\n=== Phase 6: Final coord n={n_best} (budget={t6:.0f}s) ===")
    if t6 > 15:
        a_init = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = _coord_descent(a_init, t6, seed=99999)
        try_update(a_out, "coord_final")

    # Final verification
    if best_seq is not None:
        fs = _eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
