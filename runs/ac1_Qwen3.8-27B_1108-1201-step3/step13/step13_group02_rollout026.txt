def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    s = a.sum()

        return float('inf')
    b = fast_autocorr(a)
    return float(2.0 * n * b.max() / (s * s))

def project_simplex(v):
    """Project v onto the simplex {x: x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_weights(b, T):
    """Compute softmax weights for LSE approximation."""
    b_shifted = b - b.max()
    w = np.exp(b_shifted / T)
    s = w.sum()
    if s < 1e-30:
        w = np.zeros_like(b)
        w[b.argmax()] = 1.0
    else:
        w = w / s
    return w


def lse_gradient(a, w, n):
    """Compute gradient of sum_k w_k * conv(a,a)[k] w.r.t. a.
    result[i] = 2 * sum_k w[k] * a[k-i] for valid k-i.
    This equals 2 * conv(w, a_rev)[i + n - 1].
    """
    a_rev = a[::-1]
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)
    conv = irfft(W * AR, fft_size)
    return 2.0 * conv[n - 1:2 * n - 1]


def adaptive_T(b, n):
    """Choose LSE temperature adaptively based on the spread of b."""
    b_max = b.max()
    b_mean = b.mean()
    spread = b_max - b_mean
    if spread < 1e-15:
        return 1.0
    # Target: T * spread ~ 5 for good approximation
    T_target = 5.0 / spread
    return min(max(T_target, 1.0 / (spread * 20)), 10.0 / max(spread, 1e-15))


def projected_momentum_descent(a0, time_limit, verbose=True, max_iter=50000):
    """Projected gradient with momentum, adaptive LSE, and backtracking."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    v = np.zeros(n)
    lr = 1.0 / (2.0 * n)
    beta = 0.9
    n_iter = 0
    stall = 0

    while time.monotonic() - t0 < time_limit and n_iter < max_iter:
        # Compute LSE weights
        b = fast_autocorr(a)
        T = adaptive_T(b, n)
        w = compute_lse_weights(b, T)

        # Compute gradient
        grad = lse_gradient(a, w, n)

        # Momentum step
        v = beta * v - lr * grad
        a_new = project_simplex(a + v)

        # Backtracking line search
        score_new = compute_score(a_new)
        contracts = 0
        while score_new > best_score * 1.001 and contracts < 10:
            lr *= 0.5
            v = beta * v - lr * grad
            a_new = project_simplex(a + v)
            score_new = compute_score(a_new)
            contracts += 1

        if score_new < best_score:
            a = a_new
            best_score = score_new
            best_a = a.copy()
            stall = 0
            # Occasionally increase lr
            if n_iter % 100 == 0:
                lr = min(lr * 1.1, 1.0 / (n * 0.5))
        else:
            a = a_new
            stall += 1
            if stall > 50:
                # Restart momentum
                v = np.zeros(n)
                lr = lr * 0.5
                stall = 0

        n_iter += 1

        if verbose and n_iter % 500 == 0:

            print(f"    PMD iter={n_iter} ({elapsed:.0f}s): T={T:.4f}, lr={lr:.6f}, best={best_score:.8f}")

    return best_a, best_score

                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            non_peak = np.where(~peak_mask)[0]
            if len(non_peak) < 3:
                non_peak = np.where(np.arange(n) != src)[0]

            # Select destinations
            n_dest = min(6, len(non_peak))
            sorted_dst = non_peak[np.argsort(a[non_peak])]
            candidates = sorted_dst[:min(n_dest * 4, len(sorted_dst))]
            selected_dst = candidates[rng.choice(len(candidates), min(n_dest, len(candidates)), replace=False)]

            frac_total = rng.choice([0.03, 0.05, 0.08, 0.12, 0.18])
            amt_total = a[src] * frac_total
            if amt_total < 1e-18:
                continue

            dst_vals = a[selected_dst] + 1e-10
            weights = 1.0 / dst_vals
            weights = weights / weights.sum()
            amounts = amt_total * weights

            a_trial = a.copy()
            a_trial[src] -= amt_total
            for di, dst in enumerate(selected_dst):
                a_trial[dst] += amounts[di]
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t

            score = compute_score(a_trial)

            if score < best_score:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                n_improve += 1
                break
            elif rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                a = a_trial
                improved = True
                break

        if not improved:
            n_stall += 1
            # Random pair swap
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.05, 0.2)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial / s_t
                    score = compute_score(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        n_improve += 1
                    elif rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                        a = a_trial

            if n_stall > 300:
                idx_pert = rng.choice(n, size=max(3, n // 40), replace=False)
                a_trial = a.copy()
                for idx in idx_pert:
                    amt = a[idx] * rng.uniform(0.1, 0.3)
                    j = rng.integers(n)
                    a_trial[idx] -= amt
                    a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial / s_t
                    score = compute_score(a_trial)
                    if score < best_score * 1.005:
                        a = a_trial
                        if score < best_score:
                            best_score = score
                            best_a = a.copy()
                n_stall = 0

        if verbose and n_improve % 2000 == 0 and n_improve > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakRedist improve={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_raised_cosine(n, p=1.5):
    x = np.linspace(0.0, 1.0, n)
    a = np.sin(np.pi * x) ** p
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_gaussian(n, sigma=0.3):
    x = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_bell(n, width=0.4):
    x = np.linspace(0.0, 1.0, n)
    a = 1.0 - np.abs(x - 0.5) / width
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def make_multi_mode(n, n_modes=3, decay=1.5):
    x = np.linspace(0.0, 1.0, n)
    a = np.zeros(n)
    for i in range(n_modes):
        center = (i + 0.5) / n_modes
        a += np.exp(-decay * ((x - center) / 0.1) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = compute_score(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = compute_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize if no parent
    if best_seq is None:
        n_base = 1600
        best_seq = make_raised_cosine(n_base, 1.5)
        best_score = compute_score(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # Phase 1: Try diverse initializations at n=1600 with quick optimization
    n_base = 1600
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Initialization search n={n_base} (budget={remaining:.0f}s) ===")

    inits = []
    if best_seq is not None and len(best_seq) == n_base:
        inits.append(('parent', best_seq.copy()))
    inits.append(('cos1.0', make_raised_cosine(n_base, 1.0)))
    inits.append(('cos1.5', make_raised_cosine(n_base, 1.5)))
    inits.append(('cos2.0', make_raised_cosine(n_base, 2.0)))
    inits.append(('cos3.0', make_raised_cosine(n_base, 3.0)))
    inits.append(('gauss0.25', make_gaussian(n_base, 0.25)))
    inits.append(('gauss0.35', make_gaussian(n_base, 0.35)))
    inits.append(('bell', make_bell(n_base, 0.45)))
    inits.append(('multi2', make_multi_mode(n_base, 2, 2.0)))
    inits.append(('multi3', make_multi_mode(n_base, 3, 2.0)))

    # Quick evaluate all inits
    init_results = []
    for name, a in inits:
        sc = compute_score(a)
        init_results.append((sc, name, a))
        print(f"    Init '{name}': score={sc:.6f}")

    # Sort by score and take top 4
    init_results.sort(key=lambda x: x[0])
    top_inits = init_results[:4]

    # Quick PMD on top initializations
    t_per_init = min(remaining * 0.08, 120.0) / len(top_inits)
    for sc, name, a in top_inits:
        if time.monotonic() - t_global > time_budget - 300:
            break
        a_out, _ = projected_momentum_descent(a, t_per_init, verbose=False)
        do_try_update(a_out, f"pmd_init_{name}")

    # Phase 2: LP Nesterov at n_base on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_base = len(best_seq)
    print(f"\n=== Phase 2: LP Nesterov n={n_base} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.30, 500.0)
    if t2 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t2, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n_base}")

    # Phase 3: PMD on the best
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: PMD n={n_cur} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.10, 120.0)
    if t3 > 10:
        a_out, _ = projected_momentum_descent(best_seq.copy(), t3, verbose=True)
        do_try_update(a_out, f"pmd_n{n_cur}")

    # Phase 4: Peak redistribution
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: PeakRedist n={n_cur} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 10:
        a_out, _ = peak_redistribution(best_seq.copy(), t4, seed=99, verbose=True)
        do_try_update(a_out, f"peak_n{n_cur}")

    # Phase 5: Try larger n = n_base + 200
    n2 = n_base + 200

        init_long = interpolate_sequence(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t5, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n2}")

        # PMD polish at new length
        remaining = time_budget - (time.monotonic() - t_global)
        t5b = min(remaining * 0.06, 80.0)
        if t5b > 10:
            n_cur2 = len(best_seq)
            a_out, _ = projected_momentum_descent(best_seq.copy(), t5b, verbose=False)
            do_try_update(a_out, f"pmd_n{n_cur2}")

    # Phase 6: Try even larger n = n_base + 400
    n3 = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: LP Nesterov n={n3} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 30:
        init_long3 = interpolate_sequence(best_seq, n3)
        a_norm = init_long3 * np.sqrt(2.0 * n3) / init_long3.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t6, solver, verbose=True)

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP n={n_best} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.25, 250.0)
    if t7 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t7, solver, verbose=True)
        do_try_update(a_out, f"lp_final_n{n_best}")

    # Phase 8: Final PMD + PeakRedist
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final PMD+Peak n={n_best} (budget={remaining:.0f}s) ===")
    t8a = min(remaining * 0.4, 100.0)
    if t8a > 10:
        a_out, _ = projected_momentum_descent(best_seq.copy(), t8a, verbose=False)
        do_try_update(a_out, f"pmd_final_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)
    t8b = min(remaining * 0.4, 100.0)

    # Final verification
    if best_seq is not None:

        try:

            if np.isfinite(fs2):

    else:

        return [1.0]

    return [float(x) for x in best_seq]
