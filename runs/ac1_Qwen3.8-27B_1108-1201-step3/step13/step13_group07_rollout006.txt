    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if np.all(v >= -1e-15) and abs(v.sum() - target_sum) < 1e-12:
        v = np.maximum(v, 0)
        s = v.sum()
        if s > 0:
            v = v / s * target_sum
        return v
    v = np.maximum(v, 0)
    s = v.sum()
    if s < 1e-300:
        return np.ones(n) * target_sum / n
    v = v / s * target_sum
    if abs(v.sum() - target_sum) < 1e-10:
        return v
    # Proper simplex projection
    v_orig = np.asarray(v, dtype=np.float64)
    n2 = len(v_orig)
    u = np.sort(v_orig)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n2 + 1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.ones(n2) * target_sum / n2
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v_orig - theta, 0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-14:
        idx = np.argmax(result)
        result[idx] += diff
        result = np.maximum(result, 0)
        s = result.sum()
        if s > 0:
            result = result / s * target_sum
    return result


def spectral_shaping(a, n_iter=400, alpha=0.3):
    """Iteratively flatten power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = internal_score(a)

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 50.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where T[k,j] = a_fixed[k-j] if 0<=k-j<n else 0."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def get_lp_solver():
    """Find the fastest working LP solver."""
    best_solver = cp.ECOS
    best_time = float('inf')
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(100, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            t0 = time.monotonic()
            prob.solve(solver=solver_name)
            t1 = time.monotonic() - t0
            if prob.status in ('optimal', 'optimal_inaccurate') and t1 < best_time:
                best_time = t1
                best_solver = solver_name
        except Exception:
            continue
    return best_solver


def solve_lp_best_response(a_fixed, solver, max_iters=50):
    """Solve: min M s.t. (a_fixed*b)_k <= M for all k, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, max_iter=max_iters)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, max_iter=100)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, max(0.0, float(M_var.value))


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated LP best-response with running average tracking."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0
    stall_count = 0
    stall_threshold = 8000
    it_count = 0

    while True:
        if time.monotonic() - t0 > time_limit:
            break
        it_count += 1

        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue

        alpha_t = 2.0 / (it_count + 2.0)
        y_prev = y.copy()
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s

        if it_count == 1:
            a_new = b_star.copy()
        else:
            momentum = (it_count - 1.0) / (it_count + 2.0)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = project_simplex(a_new)

        y = y_new
        a = a_new

        # Track scores
        score_a = internal_score(a)
        if score_a < best_score:
            if score_a < best_score - 1e-10:
                stall_count = 0
            best_score = score_a
            best_a = a.copy()

        score_y = internal_score(y)
        if score_y < best_score:
            if score_y < best_score - 1e-10:
                stall_count = 0
            best_score = score_y
            best_a = y.copy()

        # Running average
        avg_a += a
        n_avg += 1
        if n_avg % 200 == 0:
            avg_check = avg_a / n_avg
            score_avg = internal_score(avg_check)
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
                stall_count = 0

        stall_count += 1
        if stall_count > stall_threshold:
            if verbose:
                print(f"      Stalled at iter {it_count}, score={best_score:.8f}")
            break

        if verbose and it_count % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NL iter={it_count} ({elapsed:.0f}s): best={best_score:.8f}, M={M_val:.10f}")

    # Final check of average
    if n_avg > 0:
        avg_final = avg_a / n_avg
        sc = internal_score(avg_final)
        if sc < best_score:
            best_score = sc
            best_a = avg_final.copy()

    return best_a, best_score


def mwu_optimize(a0, time_limit, eta=None, verbose=True):
    """
    Multiplicative Weight Updates for the zero-sum game.
    Much faster per iteration than LP (O(n log n) vs O(n^1.5)).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    b = a.copy()  # Second player's strategy
    if eta is None:
        eta = np.sqrt(4.0 * np.log(n) / 1.0)

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    avg_b = np.zeros(n)
    n_avg = 0
    stall_count = 0
    fft_size = 1 << (2 * n - 1).bit_length()

    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1

        # Update a: best response to b
        # Cost for each position j: (a * b)[j]... wait, we need the "row costs"
        # For player a choosing position i, the cost is sum_j b[j] * T[i,j]
        # T[i,j] = a[i-j] if valid... but this depends on a!
        # 
        # Actually for MWU in this non-bilinear setting, we use the current
        # a to define the game matrix, then update b, then use current b
        # to define costs for a.
        
        # Compute (a * b) via FFT
        A_fft = rfft(a, fft_size)
        B_fft = rfft(b, fft_size)
        conv_ab = irfft(A_fft * B_fft, fft_size)[:2*n-1]
        
        # For player b: cost of position j given a is sum_k a[k]*a[k-j]... 
        # Actually the payoff is (a*a)[k] and both players control a.
        # Let's use the approach: fix a, find best response b (LP → approximate with MWU)
        
        # Simple MWU: update b to minimize max_k (a*b)[k]
        # Cost of b_j given a: sum_k y_k * T[k,j] where y_k are dual variables
        # Approximate: use softmax on the convolution
        
        # For b update: want to minimize max_k sum_j b_j * a[k-j]
        # This is equivalent to: b_j proportional to exp(-eta * sum_k w_k * a[k-j])
        # where w_k are weights on lags (use uniform or peak-weighted)
        
        # Use peak-weighted: w_k = softmax(conv_ab / temp)
        temp = max(1e-6, conv_ab.max() * 0.5)
        w = np.exp(conv_ab / temp - conv_ab.max() / temp)
        w = w / w.sum()
        
        # Compute cost for each b_j: sum_k w_k * a[k-j] (valid)
        # This is the convolution of w with a (reversed)
        W_fft = rfft(w, fft_size)
        cost_b = irfft(W_fft * A_fft, fft_size)
        # Cost for position j: we need sum_k w[k] * a[k-j] for valid k-j
        # cost_b[j] = sum_k w[k] * a[k-j]... let me use direct computation
        # Actually: (w * a)[j] = sum_k w[k] * a[j-k], we want sum_k w[k] * a[k-j]
        # That's (w * a_rev)[n-1-j]... let me just compute it directly via FFT
        # a_rev[i] = a[n-1-i]
        # (w * a_rev)[j] = sum_k w[k] * a_rev[j-k] = sum_k w[k] * a[n-1-j+k]
        # We want for each j: sum_k w[k] * a[k-j] (where 0<=k-j<n)
        # = sum_{m} w[m+j] * a[m] (where m = k-j, 0<=m<n, 0<=m+j<2n-1)
        # This is the cross-correlation of w and a at lag j
        # = (w * a_flip)[n-1+j] where a_flip[i] = a[n-1-i]... 
        
        # Simpler: just compute cost_j = sum_k w[k] * a[k-j] using FFT
        # Let a_pad be a padded to length 2n-1, w_pad be w padded
        # cross_corr[j] = sum_k w[k] * a[k-j] = (w * a_flipped)[something]
        
        # Direct: cost_j = sum_{k=max(j,0)}^{min(2n-2, j+n-1)} w[k] * a[k-j]
        # = sum_{m=0}^{n-1} w[m+j] * a[m] (where m+j < 2n-1)
        # This is the inner product of w[j:j+n] (padded) with a
        # Which is the convolution (w * a_flipped) evaluated at appropriate indices
        
        a_flip = a[::-1]
        AF_fft = rfft(a_flip, fft_size)
        cross = irfft(W_fft * AF_fft, fft_size)
        # cross[k] = sum_i w[i] * a_flip[k-i] = sum_i w[i] * a[n-1-k+i]
        # For position j, we want sum_i w[i] * a[i-j] (valid i-j)
        # = sum_m w[m+j] * a[m]
        # cross[n-1+j] = sum_i w[i] * a_flip[n-1+j-i] = sum_i w[i] * a[n-1-(n-1+j-i)] = sum_i w[i] * a[i-j]
        # Yes! So cost_j = cross[n-1+j]
        
        cost_b = cross[n-1:n-1+n]
        cost_b = np.maximum(cost_b, 1e-30)
        
        # MWU update for b
        b_new = b * np.exp(-eta * cost_b)
        s_b = b_new.sum()
        if s_b < 1e-300:
            b_new = b.copy()
        else:
            b_new = b_new / s_b
        
        # Similarly update a using current b
        conv_ab2 = irfft(rfft(a, fft_size) * rfft(b_new, fft_size), fft_size)[:2*n-1]
        temp2 = max(1e-6, conv_ab2.max() * 0.5)
        w2 = np.exp(conv_ab2 / temp2 - conv_ab2.max() / temp2)
        w2 = w2 / w2.sum()
        W2_fft = rfft(w2, fft_size)
        cross2 = irfft(W2_fft * AF_fft, fft_size)
        cost_a = cross2[n-1:n-1+n]
        cost_a = np.maximum(cost_a, 1e-30)
        
        a_new = a * np.exp(-eta * cost_a)
        s_a = a_new.sum()
        if s_a < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / s_a
        
        b = b_new
        a = a_new
        
        # Track
        score_a = internal_score(a)
        if score_a < best_score:
            if score_a < best_score - 1e-10:
                stall_count = 0
            best_score = score_a
            best_a = a.copy()
        
        score_b = internal_score(b)
        if score_b < best_score:
            if score_b < best_score - 1e-10:
                stall_count = 0
            best_score = score_b
            best_a = b.copy()
        
        avg_a += a
        avg_b += b
        n_avg += 1
        if n_avg % 500 == 0:
            sc_avg = internal_score(avg_a / n_avg)
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = (avg_a / n_avg).copy()
                stall_count = 0
            sc_avg_b = internal_score(avg_b / n_avg)
            if sc_avg_b < best_score:
                best_score = sc_avg_b
                best_a = (avg_b / n_avg).copy()
                stall_count = 0
        
        stall_count += 1
        if stall_count > 100000:
            break
        
        if verbose and it % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MWU it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    # Final average check
    if n_avg > 0:
        sc = internal_score(avg_a / n_avg)
        if sc < best_score:
            best_score = sc
            best_a = (avg_a / n_avg).copy()
        sc = internal_score(avg_b / n_avg)
        if sc < best_score:
            best_score = sc
            best_a = (avg_b / n_avg).copy()

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=123):
    """Coordinate-descent peak shaving with adaptive step sizes."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    stall = 0
    fft_size = 1 << (2 * n - 1).bit_length()

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:15]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            if contrib.max() < 1e-15:
                continue
            src_local = np.argmax(contrib)
            src = idx[src_local]

            # Find destination: low-contributing positions
            peak_areas = set()
            for k2 in top_k[:5]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_areas.update(range(js2, je2 + 1))

            candidates = [j for j in range(n) if j not in peak_areas and j != src]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            
            # Sort by current value (move from high to low)
            candidates_sorted = sorted(candidates, key=lambda j: a[j])
            dst = candidates_sorted[rng.integers(len(candidates_sorted))]

            for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = internal_score(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.002, 0.03)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    s_t = a_trial.sum()
                    a_trial = a_trial / s_t
                    score = internal_score(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > 800:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def adam_lse_optimize(a0, time_limit, T_start=50.0, T_end=0.05, n_stages=20, lr=0.003):
    """LogSumExp-smoothed optimization with Adam."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = internal_score(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_adam = 0
    fft_size = 1 << (3 * n - 1).bit_length()
    a_rev = a_init[::-1].copy()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        iters = max(20, min(150, int(remaining * 80 / max(1, n_stages - si))))

        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break
            a = softmax_fn(z)
            b = fast_autocorr(a)
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()

            # Gradient of smooth objective w.r.t. a
            W = rfft(w, fft_size)
            A_rev_fft = rfft(a[::-1].copy(), fft_size)
            conv_grad = irfft(A_rev_fft * W, fft_size)
            grad_a = 2.0 * conv_grad[n - 1:n - 1 + n]

            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)

            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)

            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            if it % 10 == 0:
                a_check = softmax_fn(z)
                score = internal_score(a_check)
                if score < best_score:
                    best_score = score
                    best_a = a_check.copy()

        a = softmax_fn(z)
        score = internal_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Spectral shaping init
    2. Nesterov LP (primary)
    3. MWU (fast refinement)
    4. Longer n exploration
    5. Adam LSE polish
    6. Peak shaving
    7. Final LP polish
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load starting point
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = internal_score(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = internal_score(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Spectral shaping
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: Spectral shaping n={n_base} (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.03, 30.0)
    if t1 > 3:
        for alpha_val in [0.4, 0.2, 0.1]:
            a_out = spectral_shaping(best_seq, n_iter=300, alpha=alpha_val)
            sc = internal_score(a_out)
            if sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"  Spectral alpha={alpha_val}: {best_score:.8f}")
    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 1: best={best_score:.8f} (t={elapsed:.0f}s)")

    # Phase 2: Nesterov LP at base length (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: Nesterov LP n={n_base} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.30, 500.0)
    if t2 > 20:
        a_out, sc_out = nesterov_lp_optimize(best_seq, t2, solver, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [LP_n{n_base}]: {best_score:.8f}")
    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 2: best={best_score:.8f} (t={elapsed:.0f}s)")

    # Phase 3: MWU (fast, many iterations)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: MWU n={n_base} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.10, 120.0)
    if t3 > 10:
        a_out, sc_out = mwu_optimize(best_seq, t3, eta=0.5, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [MWU_n{n_base}]: {best_score:.8f}")
    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 3: best={best_score:.8f} (t={elapsed:.0f}s)")

    # Phase 4: Try longer length n=2000
    n_long = 2000
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: Longer n={n_long} (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 20:
        init_long = interpolate_sequence(best_seq, n_long)
        # Quick spectral
        a_spec = spectral_shaping(init_long, n_iter=200, alpha=0.2)
        sc = internal_score(a_spec)
        if sc < best_score:
            best_score = sc
            best_seq = a_spec.copy()
            print(f"  *** NEW BEST [spec_n{n_long}]: {best_score:.8f}")
        
        # LP at longer length
        t4_lp = t4 * 0.85
        a_out, _ = nesterov_lp_optimize(best_seq, t4_lp, solver, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [LP_n{n_long}]: {best_score:.8f}")
    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 4: best={best_score:.8f} (t={elapsed:.0f}s)")

    # Phase 5: Try n=2400
    n_long2 = 2400
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: Longer n={n_long2} (remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 15:
        init_long2 = interpolate_sequence(best_seq, n_long2)
        a_spec2 = spectral_shaping(init_long2, n_iter=150, alpha=0.2)
        sc = internal_score(a_spec2)
        if sc < best_score:
            best_score = sc
            best_seq = a_spec2.copy()
            print(f"  *** NEW BEST [spec_n{n_long2}]: {best_score:.8f}")
        
        t5_lp = t5 * 0.7
        a_out, _ = nesterov_lp_optimize(best_seq, t5_lp, solver, verbose=False)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [LP_n{n_long2}]: {best_score:.8f}")
    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 5: best={best_score:.8f} (t={elapsed:.0f}s)")

    # Phase 6: Adam LSE polish at best length
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: Adam LSE polish n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t6 = min(remaining * 0.08, 80.0)
    if t6 > 5:
        a_out, _ = adam_lse_optimize(best_seq, t6, T_start=30.0, T_end=0.1, n_stages=15, lr=0.002)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [Adam_LSE]: {best_score:.8f}")
    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 6: best={best_score:.8f} (t={elapsed:.0f}s)")

    # Phase 7: Peak shaving
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 7: Peak shaving n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t7 = min(remaining * 0.12, 100.0)
    if t7 > 5:
        a_out, _ = peak_shaving(best_seq, t7, seed=456)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [peak_shave]: {best_score:.8f}")
    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 7: best={best_score:.8f} (t={elapsed:.0f}s)")

    # Phase 8: Final LP polish
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 8: Final LP n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t8 = min(remaining * 0.35, 300.0)
    if t8 > 15:
        a_out, _ = nesterov_lp_optimize(best_seq, t8, solver, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [LP_final]: {best_score:.8f}")
    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 8: best={best_score:.8f} (t={elapsed:.0f}s)")

    # Phase 9: Final peak shave
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 8:
        t9 = min(remaining * 0.4, 40.0)
        if t9 > 3:
            a_out, _ = peak_shaving(best_seq, t9, seed=789)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = np.maximum(a_out, 0)
                s = best_seq.sum()
                if s > 0:
                    best_seq = best_seq / s
                print(f"  *** NEW BEST [peak_shave_final]: {best_score:.8f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        best_seq = np.maximum(best_seq, 0)
        s = best_seq.sum()
        if s > 0:
            best_seq = best_seq / s
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
