                    if new_max < best_v * (1.0 - 1e-12):
                        best_v = new_max
                        best_d = d
                        best_i = i
                        best_j = j

        if best_d > 0 and best_i >= 0:
            a[best_i] -= best_d
            a[best_j] += best_d
            s = a.sum()
            if s < 1e-15:
                a[best_i] += best_d
                a[best_j] -= best_d
            else:
                a = a * norm / s
                conv = fast_autocorr(a)
                if conv.max() < best_score:
                    best_score = conv.max()
                    best_a = a.copy()
                    pass_improved = True
                    improvements += 1
                    no_improve_count = 0

        if not pass_improved:
            no_improve_count += 1
            if no_improve_count >= 5:
                break

        if verbose and n_passes % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}, impr={improvements}")

    return best_a, best_score


def simulated_annealing(a0, time_limit, rng, verbose=True):
    """Simulated annealing with structured perturbations."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    current_score = conv.max()
    best_a = a.copy()
    best_score = current_score
    t0 = time.monotonic()

    T_start = 0.05
    T_end = 0.001
    n_total = max(100, int(time_limit * 50))
    cooling = (T_end / T_start) ** (1.0 / max(1, n_total - 1))

    T = T_start
    iter_count = 0
    stall_count = 0

    while time.monotonic() - t0 < time_limit:
        T = max(T * cooling, T_end)
        old_a = a.copy()
        old_score = current_score
        pert_type = rng.random()

        if pert_type < 0.4:
            # Swap: move mass from i to j
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            max_d = 0.1 * a[i]
            if max_d < 1e-15:
                continue
            d = rng.random() * max_d
            if d > a[i]:
                d = a[i]
            a[i] -= d
            a[j] += d
        elif pert_type < 0.7:
            # Block reweighting
            block_len = int(rng.integers(5, min(100, n // 4)))
            start = int(rng.integers(0, n - block_len + 1))
            factor = 1.0 + rng.random() * 0.3 - 0.15
            a[start:start + block_len] *= factor
        else:
            # Multi-point perturbation
            n_pts = int(rng.integers(3, 8))
            pts = rng.integers(0, n, n_pts)
            noise = rng.normal(0, 0.05 * norm / n, n_pts)
            for pt in pts:
                a[pt] = max(0.0, a[pt] + noise[pts.tolist().index(int(pt))] if pt in pts else 0)
            # Simpler: just add small random noise to random points
            a[pts] = np.maximum(0.0, a[pts] + rng.normal(0, 0.02 * norm / n, n_pts))

        s = a.sum()
        if s < 1e-15:
            a = old_a.copy()
            continue
        a = a * norm / s
        conv = fast_autocorr(a)
        new_score = conv.max()
        delta = new_score - old_score

        if delta < 0 or rng.random() < np.exp(-delta / T):
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                stall_count = 0
            else:
                stall_count += 1
        else:
            a = old_a.copy()
            conv = fast_autocorr(a)
            current_score = old_score
            stall_count += 1

        if stall_count > 500:
            a = best_a.copy()
            conv = fast_autocorr(a)
            current_score = best_score
            stall_count = 0

        iter_count += 1
        if verbose and iter_count % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SA iter={iter_count} T={T:.5f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def multi_peak_fw(a0, time_limit, n_peaks=5, verbose=True):
    """Frank-Wolfe with multi-peak subgradient."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        M = conv_y.max()

        # Find top-k peaks
        k = min(n_peaks, len(conv_y))
        top_k = np.argpartition(conv_y, -k)[-k:]
        top_k = top_k[conv_y[top_k] > 0.5 * M]
        if len(top_k) == 0:
            top_k = np.array([np.argmax(conv_y)])

        # Compute combined subgradient
        grad = np.zeros(n)
        for kk in top_k:
            w = conv_y[kk] / M
            for i in range(n):
                v = 0.0
                j1 = kk - i
                if 0 <= j1 < n:
                    v += y[j1]
                j2 = i - kk
                if 0 <= j2 < n:
                    v += y[j2]
                grad[i] += w * v
        grad *= 2.0 / len(top_k)

        # FW step
        i_min = int(np.argmin(grad))
        v = np.zeros(n)
        v[i_min] = norm

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * v
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = v.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                sc = fast_autocorr(avg_check).max()
                if sc < best_score:
                    best_score = sc
                    best_a = avg_check.copy()

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score
