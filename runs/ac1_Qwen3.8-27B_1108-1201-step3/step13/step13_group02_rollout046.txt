        T_mat = np.zeros((2*n-1, n))
        for j in range(n):

        constraints = [T_mat @ b_var <= M]

            prob.solve(solver=solver, verbose=False)
        except Exception:
            break
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break

        b_sum = b.sum()
        if b_sum < 1e-15:
            break

        iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        y_new = y_new * norm_val / s_y
        
        momentum = (t - 1.0) / (t + 2.0)
        a_new = y_new + momentum * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s_a = a_new.sum()
        a_new = a_new * norm_val / s_a
        
        y = y_new
        a = a_new
        
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0

            best_score = score_a
            best_a = a.copy()

def peak_flatten(a0, time_limit, seed=42, verbose=True):
    """Iterative peak flattening with smart mass redistribution."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm_val = np.sqrt(2.0 * n)
    a = normalize(a0, norm_val).copy()
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    T = best_score * 0.02
    
    while time.monotonic() - t0 < time_limit:
        frac = min(1.0, (time.monotonic() - t0) / time_limit)
        T_curr = T * (0.01) ** frac  # Cool down
        
        R = fast_autocorr(a)
        max_R = R.max()
        
        # Find peak lags
        peak_threshold = max_R * 0.95
        peak_lags = np.where(R >= peak_threshold)[0]
        
        if len(peak_lags) == 0:
            break
        
        improved = False
        
        for k in peak_lags:
            # Find contributing pairs for lag k
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            
            if contrib.max() < 1e-18:
                continue
            
            # Sort by contribution
            order = np.argsort(contrib)[::-1]
            
            for oi in order[:3]:  # Try top 3 contributors
                src = idx[oi]
                other = k - src  # The paired index
                
                if src == other:
                    continue
                
                # Find a good destination
                # Want a position that doesn't contribute to any peak
                peak_mask = np.zeros(n, dtype=bool)
                for k2 in peak_lags:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    peak_mask[js2:je2+1] = True
                
                non_peak = np.where(~peak_mask)[0]
                if len(non_peak) == 0:
                    non_peak = np.where(np.arange(n) != src)[0]
                
                if len(non_peak) == 0:
                    continue
                
                # Pick destination with lowest value
                dst = non_peak[np.argmin(a[non_peak])]
                
                # Move some mass
                frac = rng.choice([0.05, 0.1, 0.15, 0.2])
                amt = a[src] * frac * 0.5  # Move half to destination
                
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm_val / s_t
                
                R_trial = fast_autocorr(a_trial)
                score_trial = R_trial.max()
                
                if score_trial < best_score:
                    a = a_trial
                    best_score = score_trial
                    best_a = a.copy()
                    improved = True
                    break
                elif rng.random() < np.exp(-(score_trial - best_score) / max(T_curr, 1e-10)):
                    a = a_trial
                    improved = True
                    break
        
        if not improved:
            # Random perturbation
            n_pert = max(1, n // 100)
            srcs = rng.choice(n, size=n_pert, replace=False)
            dsts = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for s, d in zip(srcs, dsts):
                if a_trial[s] > 1e-10:
                    amt = a_trial[s] * rng.uniform(0.1, 0.3)
                    a_trial[s] -= amt
                    a_trial[d] += amt
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial * norm_val / s_t
                score_trial = fast_autocorr(a_trial).max()
                if score_trial < best_score * 1.05:
                    a = a_trial
                    if score_trial < best_score:
                        best_score = score_trial
                        best_a = a.copy()
        
        if verbose and (time.monotonic() - t0) % 10 < 1:
            print(f"    PeakFlatten ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def coordinate_descent(a0, time_limit, verbose=True):
    """Coordinate descent: optimize each element individually."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = normalize(a0, norm_val).copy()
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # For each coordinate i, the optimal value given all others can be
    # found by solving a 1D optimization.
    # R_k = a_i * (sum of relevant others) + (contribution from others)
    # We want to minimize max_k R_k.
    
    # Precompute the "background" contributions
    # For coordinate i, R_k = a_i * w_k + bg_k where bg_k doesn't depend on a_i
    # w_k = a_{k-i} + a_{i-k} (with bounds)
    
    epochs = 0
    while time.monotonic() - t0 < time_limit:
        epochs += 1
        improved = False
        
        # Run through coordinates (randomized order for stochastic CD)
        if epochs % 10 == 1:
            order = np.random.default_rng(epochs).permutation(n)
        else:
            order = np.arange(n)
        
        for i in order:
            # Compute w_k for this coordinate
            # R_k depends on a_i through: a_i * a_{k-i} + a_{k-i} * a_i = 2*a_i*a_{k-i} (when valid)
            # Plus the "background" from other terms
            
            # Actually, let's just do a simple search:
            # Try a few values of a_i and pick the best
            current_val = a[i]
            candidates = [
                current_val * 0.5,
                current_val * 0.8,
                current_val * 1.2,
                current_val * 1.5,
                0.0,
                current_val,
            ]
            
            best_val = current_val
            best_sc = fast_autocorr(a).max()
            
            R = fast_autocorr(a)
            
            for cv in candidates:
                if cv < 0 or cv > 1000:
                    continue
                a[i] = cv
                # Renormalize
                s = a.sum()
                if s < 1e-15:
                    a[i] = current_val
                    continue
                a_norm = a * norm_val / s
                sc = fast_autocorr(a_norm).max()
                if sc < best_sc:
                    best_sc = sc
                    best_val = cv
                a[i] = current_val  # Restore
            
            if best_val != current_val:
                a[i] = best_val
                s = a.sum()
                if s > 1e-15:
                    a = a * norm_val / s
                improved = True
                if fast_autocorr(a).max() < best_score:
                    best_score = fast_autocorr(a).max()
                    best_a = a.copy()
        
        if not improved and epochs > 3:
            break
    
    return best_a, best_score

def interpolate_seq(f, new_n):
    """Interpolate sequence to new length, preserving normalized shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
