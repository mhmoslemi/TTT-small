def build_conv_matrix_rows(a, k_indices):
    """Build only the rows of the convolution matrix for given lag indices."""
    n = len(a)
    rows = []
    for k in k_indices:
        row = np.zeros(n)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for i in range(lo, hi):
            j = k - i
            if 0 <= j < n:
                row[i] += a[j]
        rows.append(row)
    return np.array(rows)


def lp_best_response(a, M, solver, active_indices=None):
    """Solve LP: max sum(b) s.t. b>=0, conv(a,b)[k]<=M for k in active_indices."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    size = 2 * n - 1
    if active_indices is None:
        active_indices = np.arange(size)
    # Build constraint matrix rows
    T = build_conv_matrix_rows(a, active_indices)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-8:
                continue

            for attempt in range(2):
                j = int(rng.integers(n))
                if j == i or a[i] < norm * 1e-8:
                    continue

                # Compute gain vector for transferring delta from i to j
                ai = a[i]
                # First-order change in conv[k] when moving delta from i to j:
                # delta * (a[k-j] + a[j-k] - a[k-i] - a[i-k])
                g = np.zeros(size)
                # +a[k-j] for k in [j, j+n-1]
                lo = j
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[:hi - lo]
                # +a[j-k] for k in [0, j] where j-k in [0,n-1], i.e., k in [max(0,j-n+1), j]
                lo2 = max(0, j - n + 1)
                hi2 = j + 1
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - lo2: j - (hi2 - 1) - 1: -1]

                # -a[k-i] for k in [i, i+n-1]
                lo3 = i
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[:hi3 - lo3]
                # -a[i-k] for k in [max(0,i-n+1), i]
                lo4 = max(0, i - n + 1)
                hi4 = i + 1
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - lo4: i - (hi4 - 1) - 1: -1]

                # Find best delta in [0, ai]
                best_d = 0.0
                best_val = conv.max()
                n_samples = 15
                deltas = np.linspace(0.0, ai * 0.8, n_samples)
                for d in deltas[1:]:  # skip d=0
                    new_max = (conv + d * g).max()
                    # Second order at k=i (removal) and k=j (addition)
                    if i < size:
                        new_max = max(new_max, conv[i] - 2 * ai * d + d * d)
                    if j < size:
                        new_max = max(new_max, conv[j] + 2 * a[j] * d + d * d)
                    if new_max < best_val - 1e-13:
                        best_val = new_max
                        best_d = d

                if best_d > 1e-10:
                    a[i] -= best_d
                    a[j] += best_d
                    s_a = a.sum()
                    if s_a < 1e-15:
                        a[i] += best_d
                        a[j] -= best_d
                        continue
                    a = a * norm / s_a
                    conv = fast_autocorr(a)
                    cur_max = conv.max()
                    if cur_max < best_score - 1e-13:
                        best_score = cur_max
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0
                        conv = fast_autocorr(a)
                    else:
                        a[j] -= best_d
                        a[i] += best_d
                        a = a * norm / a.sum()
                        conv = fast_autocorr(a)

        if not pass_improved:
            no_improve += 1
            if no_improve >= 2:
                break
        if verbose and pass_num % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={pass_num} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def ramp_init(n, exponent=1.0, side='right'):
    """Create an asymmetric ramp initialization."""
    x = np.linspace(0.0, 1.0, n)
    if side == 'right':
        a = x ** exponent
    else:
        a = (1.0 - x) ** exponent
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def exp_init(n, rate=2.0, side='right'):
    """Exponential decay initialization."""
    x = np.linspace(0.0, 1.0, n)
    if side == 'right':
        a = np.exp(rate * x)
    else:
        a = np.exp(rate * (1.0 - x))
    return a / a.sum()


def power_law_init(n, p=2.0):
    """Power law initialization."""
    x = np.linspace(0.0, 1.0, n)
    a = (x + 0.01) ** p
    return a / a.sum()


def cosine_asym_init(n, alpha=1.0):
    """Asymmetric cosine initialization."""
    x = np.linspace(0.0, 1.0, n)
    a = np.cos(np.pi * x * alpha / 2.0) ** 2
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def gaussian_asym_init(n, center=0.3, sigma=0.2):
    """Asymmetric Gaussian (peak off-center)."""
    x = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    return a / a.sum()


def structured_init(n, rng):
    """Try multiple asymmetric inits and return the best."""
    candidates = [
        ramp_init(n, 1.0, 'right'),
        ramp_init(n, 1.5, 'right'),
        exp_init(n, 2.0, 'right'),
        exp_init(n, 3.0, 'right'),
        power_law_init(n, 2.0),
        cosine_asym_init(n, 1.0),
        gaussian_asym_init(n, 0.3, 0.2),
        ramp_init(n, 1.0, 'left'),
    ]
    best = None
    best_sc = float('inf')
    for c in candidates:
        sc = fast_autocorr(c).max()
        if sc < best_sc:
            best_sc = sc
            best = c
    return best


def subgradient_fw(a0, time_limit, verbose=False):
    """Frank-Wolfe with subgradient (fast, no LP)."""
    n = len(a0)
    norm = math.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        k_star = int(np.argmax(conv_y))

        # Subgradient at k_star
        grad = np.zeros(n)
        for i in range(n):
            v = 0.0
            j1 = k_star - i
            if 0 <= j1 < n:
                v += y[j1]
            j2 = i - k_star
            if 0 <= j2 < n:
                v += y[j2]
            grad[i] = v
        grad *= 2.0

        # FW: move toward vertex that minimizes grad
        i_min = int(np.argmin(grad))
        v = np.zeros(n)
        v[i_min] = norm

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * v
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = v.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                sc = fast_autocorr(avg_check).max()
                if sc < best_score:
                    best_score = sc
                    best_a = avg_check.copy()

        if verbose and t % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if parent is None:
        n_base = 1600
        a_init = structured_init(n_base, rng)
        best_seq = a_init
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    print(f"Starting: n={n_base}, score={best_score:.6f}")

    # === Phase 1: LSE Nesterov on best sequence (primary, 40% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.40, 600.0)
    print(f"\n=== Phase 1: LSE-Nesterov n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 30:
        norm = math.sqrt(2.0 * n_base)
        a_norm = best_seq * norm / best_seq.sum()
        a_out, _ = lse_nesterov_optimize(a_norm, t1, verbose=True)
        do_try_update(a_out, f"lse_n{n_base}")

    # === Phase 2: LP Nesterov (20% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 2: LP-Nesterov n={n_cur} (budget={t2:.0f}s) ===")
    if t2 > 30:
        norm = math.sqrt(2.0 * n_cur)
        a_norm = best_seq * norm / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t2, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n_cur}")

    # === Phase 3: LSE Nesterov again (refinement, 15% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 3: LSE-Nesterov round 2 n={n_cur} (budget={t3:.0f}s) ===")
    if t3 > 20:
        norm = math.sqrt(2.0 * n_cur)
        a_norm = best_seq * norm / best_seq.sum()
        a_out, _ = lse_nesterov_optimize(a_norm, t3, verbose=True)
        do_try_update(a_out, f"lse2_n{n_cur}")

    # === Phase 4: Pair CD (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t4 = min(remaining * 0.10, 150.0)
    print(f"\n=== Phase 4: Pair CD n={n_cur} (budget={t4:.0f}s) ===")
    if t4 > 10:
        norm = math.sqrt(2.0 * n_cur)
        a_norm = best_seq * norm / best_seq.sum()
        a_out, _ = pair_coordinate_descent(a_norm, t4, rng, verbose=True)
        do_try_update(a_out, f"pcd_n{n_cur}")

    # === Phase 5: Try different asymmetric inits (10% time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.10, 120.0)
    if t5 > 20:
        print(f"\n=== Phase 5: Multi-start n={n_base} (budget={t5:.0f}s) ===")
        n_starts = max(2, int(t5 / 30))
        per_start = t5 / n_starts
        inits = [
            ramp_init(n_base, 1.0, 'right'),
            exp_init(n_base, 2.5, 'right'),
            power_law_init(n_base, 1.5),
            ramp_init(n_base, 2.0, 'right'),
            gaussian_asym_init(n_base, 0.4, 0.15),
            exp_init(n_base, 4.0, 'right'),
        ]
        for si in range(min(n_starts, len(inits))):
            if time.monotonic() - t_global > time_budget - 60:
                break
            a_s = inits[si]
            norm_s = math.sqrt(2.0 * n_base)
            a_s = a_s * norm_s / a_s.sum()
            t_rem = time_budget - (time.monotonic() - t_global) - 60
            t_alloc = min(per_start, t_rem * 0.3)
            if t_alloc < 10:
                continue
            a_out, _ = lse_nesterov_optimize(a_s, t_alloc, verbose=False)
            do_try_update(a_out, f"start{si}_n{n_base}")

    # === Phase 6: Final LSE polish (remaining time) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    remaining = max(0, remaining - 30)
    t6 = min(remaining * 0.5, 100.0)
    if t6 > 10:
        print(f"\n=== Phase 6: Final LSE polish n={n_cur} (budget={t6:.0f}s) ===")
        norm = math.sqrt(2.0 * n_cur)
        a_norm = best_seq * norm / best_seq.sum()
        a_out, _ = lse_nesterov_optimize(a_norm, t6, verbose=True)
        do_try_update(a_out, f"final_lse_n{n_cur}")

    # === Phase 7: Final pair CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    remaining = max(0, remaining - 20)
    t7 = min(remaining * 0.5, 60.0)
    if t7 > 5:
        print(f"\n=== Phase 7: Final PCD n={n_cur} (budget={t7:.0f}s) ===")
        norm = math.sqrt(2.0 * n_cur)
        a_norm = best_seq * norm / best_seq.sum()
        a_out, _ = pair_coordinate_descent(a_norm, t7, rng, verbose=True)
        do_try_update(a_out, f"final_pcd_n{n_cur}")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
