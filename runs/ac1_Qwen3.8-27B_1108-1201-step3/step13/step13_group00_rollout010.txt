                # d(conv)/d(a[i]) = a[k-i] + a[i-k] (both terms valid)
                # So change from da[i]=-d, da[j]=+d:
                # delta_k = -d*(a[k-i]+a[i-k]) + d*(a[k-j]+a[j-k]) [with 0 for invalid indices]
                
                ai = a[i]
                # Precompute the gain vector g = (a_shifted_j - a_shifted_i) + (a_rev_shifted_j - a_rev_shifted_i)
                g = np.zeros(size)
                # a[k-i] for k in [i, i+n-1]
                g[i:i+n] -= a  # a[k-i] = a[0],a[1],... for k=i,i+1,...
                # a[i-k] for k in [0, i+n-1] (i-k in [0,n-1] → k in [i-n+1, i])
                k_lo = max(0, i - n + 1)
                k_hi = min(size, i + 1)
                if k_hi > k_lo:
                    g[k_lo:k_hi] -= a[i - np.arange(k_lo, k_hi)]
                # a[k-j] for k in [j, j+n-1]
                g[j:j+n] += a
                # a[j-k] for k in [0, j+n-1]
                k_lo2 = max(0, j - n + 1)
                k_hi2 = min(size, j + 1)
                if k_hi2 > k_lo2:
                    g[k_lo2:k_hi2] += a[j - np.arange(k_lo2, k_hi2)]
                
                # Find best delta in [0, ai] minimizing max(conv + delta*g)
                # Sample a few values
                n_samples = 15
                deltas = np.linspace(0, ai, n_samples)[1:]  # skip 0
                best_d = 0.0
                best_val = conv.max()
                for d in deltas:
                    new_max = (conv + d * g).max()
                    # Second order corrections at k=i (a[i]^2 → (a[i]-d)^2) and k=j (a[j]^2 → (a[j]+d)^2)
                    if 2*i < size:
                        new_max = max(new_max, conv[2*i] - 2*ai*d + d*d)
                    if 2*j < size:
                        new_max = max(new_max, conv[2*j] + 2*a[j]*d + d*d)
                    if new_max < best_val - 1e-13:
                        best_val = new_max
                        best_d = d
                
                if best_d > 0:
                    a[i] -= best_d
                    a[j] += best_d
                    s = a.sum()
                    a = a * norm / s
                    conv = fast_autocorr(a)
                    cur_max = conv.max()
                    if cur_max < best_score - 1e-13:
                        best_score = cur_max
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0
                    else:
                        # Revert
                        a[j] -= best_d
                        a[i] += best_d
                        a = a * norm / a.sum()
                        conv = fast_autocorr(a)
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
        
        if verbose and n_passes % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def smooth_gradient_step(a, time_limit, verbose=False):
    """Use log-sum-exp smoothed gradient with adaptive step size."""
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    # Anneal temperature
    temps = np.geomspace(5.0, 0.3, 10)
    
    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(5, int(min(50, remaining * 2)))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            conv = fast_autocorr(a)
            max_c = conv.max()
            exp_c = np.exp((conv - max_c) / T)
            w = exp_c / exp_c.sum()
            
            # Gradient via FFT
            A = np.fft.rfft(a, fft_size)
            W = np.fft.rfft(w, fft_size)
            # grad_i = 2 * sum_k w[k] * (a[k-i] + a[i-k])
            # = 2 * (cross_corr(w,a) + cross_corr(a,w))[i]
            # cross_corr(w,a)[i] = sum_k w[k]*a[k+i] = ifft(FFT(w)*conj(FFT(a)))
            cc1 = np.fft.irfft(W * np.conj(A), fft_size)[:n]
            cc2 = np.fft.irfft(A * np.conj(W), fft_size)
            # cc2[m] = sum_k a[k]*w[k+m], we want sum_k w[k]*a[i-k] = sum_k a[i-k]*w[k]
            # = sum_j a[j]*w[i-j] = conv(a,w)[i]
            conv_aw = np.fft.irfft(A * W, fft_size)[:2*n-1]
            
            grad = 2.0 * (cc1 + conv_aw[:n])
            
            # Adaptive step size
            step = norm / (T * 10.0 * (1.0 + 0.1 * it))
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-14:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s
            a = a_new
            
            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE T={T:.2f} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def structured_init(n, rng):
    choice = rng.integers(0, 8)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.3
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(6 * np.pi * x) + 0.05 * np.cos(14 * np.pi * x)
        a = np.maximum(a, 0.0)
    elif choice == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.6 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2.5
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.5
    elif choice == 6:
        a = np.maximum(0.0, 1.0 - 3.0 * x * (1.0 - x))
    else:
        a = 0.2 + 0.8 * np.exp(-4.0 * (x - 0.5) ** 2 / (0.1 + 0.3 * x * (1-x)))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
