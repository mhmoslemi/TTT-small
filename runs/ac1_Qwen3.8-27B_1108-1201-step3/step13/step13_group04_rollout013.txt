"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LSE-Adam optimizer with adaptive temperature scheduling
2. LP best-response with Nesterov acceleration (using scipy HiGHS)
3. Multi-length exploration with interpolation transfer
4. Local search polish with adaptive moves

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def lse_gradient(a, T):
    """Compute gradient of T*log(sum exp(b_k/T)) where b = autocorr(a)."""
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    w = np.exp((b - max_b) / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[n - 1] = 1.0
    else:
        w /= w_sum
    
    # dL/da[m] = sum_k w[k] * (a[k-m] + a[n-1-k+m] ... )
    # Actually: d(a*a)_k/da_m = a[k-m] + a[m-k] (with valid indices)
    # So gradient[m] = sum_k w[k] * (a_rev * w)[m] + (a * w)[m]
    # where a_rev = a[::-1]
    
    fft_size = 1 << (3 * n - 2).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    A_fwd = rfft(a, fft_size)
    W = rfft(w, fft_size)
    
    conv1 = irfft(A_rev * W, fft_size)
    conv2 = irfft(A_fwd * W, fft_size)
    
    r1 = conv1[n - 1:n - 1 + n]
    r2 = conv2[:n]
    
    return r1 + r2

def lse_value(a, T):
    """Compute LSE surrogate value."""
    b = fast_autocorr(a)
    max_b = b.max()
    log_sum = max_b / T + np.log(np.sum(np.exp((b - max_b) / T)))
    return T * log_sum

def lse_adam(a0, time_limit, T_start=20.0, T_end=0.3, n_stages=15, verbose=True):
    """LSE gradient descent with Adam optimizer and geometric cooling."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.8  # Lower beta1 for more aggressive updates
    beta2 = 0.999
    eps = 1e-8
    t_step = 0
    
    # Learning rate schedule
    base_lr = norm * 0.1
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        
        n_iters = max(100, int(remaining * 4.0 / max(1, n_stages - si)))
        for it in range(n_iters):
            if time.monotonic() - t0 > time_limit:
                break
            
            grad = lse_gradient(a, T)
            t_step += 1
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad**2
            m_hat = m / (1 - beta1**t_step)
            v_hat = v / (1 - beta2**t_step)
            
            # Adaptive learning rate
            lr = base_lr / (1 + 0.01 * it)
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Project to simplex (non-negative, fixed sum)
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s
            
            a = a_new
            
            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
            
            if verbose and it % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"  LSE-Adam stage={si}/{n_stages} T={T:.3f} it={it} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def lp_best_response_fast(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b)_k <= M, b >= 0 using scipy linprog."""
    n = len(a)
    
    # Build constraint matrix A_eq for linprog: A_ub @ x <= b_ub
    # (a*b)[k] = sum_i a[i] * b[k-i] <= M
    # For linprog: c @ x is minimized, so we minimize -sum(b)
    c = -np.ones(n)
    
    # Build A_ub: (2n-1) x n matrix
    # A_ub[k, i] = a[k-i] if 0 <= k-i < n, else 0
    # This is a "convolution matrix"
    
    # Efficient construction using the banded structure
    # Row k: [a[k], a[k-1], ..., a[k-(n-1)]] with zeros padding
    # Actually: A_ub[k, i] = a[k-i] if max(0, k-n+1) <= i <= min(n-1, k)
    
    # For large n, building this matrix explicitly is O(n^2) which is fine for n~2000
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1)
        A_ub[k, idx] = a[k - idx]
    
    b_ub = np.full(2 * n - 1, M)
    
    # Bounds: b >= 0
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if result.status == 0:
            b_val = np.maximum(result.x, 0.0)
            s = b_val.sum()
            if s > 1e-15:
                return b_val
    except Exception as e:
        pass
    return None

def nesterov_lp(a0, time_limit, verbose=True):
    """LP best-response with Nesterov acceleration and averaging."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    n_iters = 0
    start_avg = 50  # Start averaging after 50 iterations
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_autocorr(y).max()
        b = lp_best_response_fast(y, M_curr)
        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        
        # Nesterov step
        alpha = 2.0 / (t + 2.0)
        if t == 1:
            y_new = g.copy()
        else:
            y_new = (1 - alpha) * y + alpha * g
            # Nesterov extrapolation
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y
        
        # Averaged sequence (Polyak-Ruppert)
        a_avg = (1 - alpha) * y + alpha * g
        a_avg = np.maximum(a_avg, 0.0)
        s_avg_iter = a_avg.sum()
        if s_avg_iter > 1e-15:
            a_avg = a_avg * norm / s_avg_iter
        
        y = y_new
        
        if t >= start_avg:

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LSE-Adam optimizer with geometric temperature scheduling
2. LP best-response with Nesterov acceleration (scipy HiGHS)
3. Multi-length exploration with interpolation
4. Peak-targeting local search polish
5. Adaptive time allocation across phases
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_gradient(a, T):
    """Compute gradient of T*log(sum exp(b_k/T)) where b = autocorr(a)."""
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    w = np.exp((b - max_b) / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[n - 1] = 1.0
    else:
        w /= w_sum

    fft_size = 1 << (3 * n - 2).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    A_fwd = rfft(a, fft_size)
    W = rfft(w, fft_size)

    conv1 = irfft(A_rev * W, fft_size)
    conv2 = irfft(A_fwd * W, fft_size)

    r1 = conv1[n - 1: n - 1 + n]
    r2 = conv2[:n]

    return r1 + r2


def lse_adam(a0, time_limit, T_start=20.0, T_end=0.3, n_stages=15, verbose=True):
    """LSE gradient descent with Adam optimizer and geometric cooling."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.75
    beta2 = 0.999
    eps = 1e-8
    t_step = 0
    base_lr = norm * 0.08

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        n_iters = max(100, int(remaining * 4.0 / max(1, n_stages - si)))
        for it in range(n_iters):
            if time.monotonic() - t0 > time_limit:
                break

            grad = lse_gradient(a, T)
            t_step += 1

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad ** 2
            m_hat = m / (1.0 - beta1 ** t_step)
            v_hat = v / (1.0 - beta2 ** t_step)

            lr = base_lr / (1.0 + 0.005 * it)
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()

                a_new = a_new * norm / s

            a = a_new

            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

            if verbose and it % 1000 == 0:
                elapsed = time.monotonic() - t0
                print(f"  LSE-Adam stage={si}/{n_stages} T={T:.3f} it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def build_conv_matrix(a):
    """Build the (2n-1) x n convolution constraint matrix efficiently."""
    n = len(a)
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A_ub[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A_ub


def lp_best_response_fast(a, M, timeout=30.0):
    """Solve LP: max sum(b) s.t. conv(a,b)_k <= M, b >= 0."""
    n = len(a)
    c = -np.ones(n)
    A_ub = build_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    t_start = time.monotonic()
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'time_limit': timeout})
        if time.monotonic() - t_start > timeout + 5:
            return None
        if result.status == 0:
            b_val = np.maximum(result.x, 0.0)
            s = b_val.sum()
            if s > 1e-15:
                return b_val
    except Exception:
        pass
    return None


def nesterov_lp(a0, time_limit, verbose=True):
    """LP best-response with Nesterov acceleration and Polyak averaging."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    n_iters = 0
    start_avg = 30

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response_fast(y, M_curr, timeout=20.0)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        if t == 1:
            y_new = g.copy()
        else:
            y_new = (1.0 - alpha) * y + alpha * g
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y

        a_avg = (1.0 - alpha) * y + alpha * g
        a_avg = np.maximum(a_avg, 0.0)
        s_avg_it = a_avg.sum()
        if s_avg_it > 1e-15:
            a_avg = a_avg * norm / s_avg_it

        y = y_new

        if t >= start_avg:
            avg_a += a_avg
            n_avg += 1
            if n_avg % 5 == 0:
                avg_check = avg_a / n_avg
                s_c = avg_check.sum()
                if s_c > 1e-15:
                    avg_check = avg_check * norm / s_c
                    score_avg = fast_autocorr(avg_check).max()
                    if score_avg < best_score:
                        best_score = score_avg
                        best_a = avg_check.copy()

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if verbose and n_iters % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LP-Nest iter={n_iters} ({elapsed:.0f}s) M={M_curr:.6f}: best={best_score:.8f}")

    return best_a, best_score


def local_search_polish(a0, time_limit, seed=42, verbose=True):
    """Local search with peak-targeting and random perturbations."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    it_count = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        k_peak = np.argmax(b)

        i_start = max(0, k_peak - n + 1)
        i_end = min(n - 1, k_peak)
        idx = np.arange(i_start, i_end + 1)
        contrib = a[idx] * a[k_peak - idx]

        top_src_list = idx[np.argsort(contrib)[::-1][:5]]

        improved = False
        for top_src in top_src_list:
            if a[top_src] < norm * 1e-10:
                continue

            top_peaks = np.argsort(b)[::-1][:5]
            quiet_mask = np.ones(n, dtype=bool)
            for kp in top_peaks:
                js = max(0, kp - n + 1)
                je = min(n - 1, kp)
                quiet_mask[js:je + 1] = False

            quiet_positions = np.where(quiet_mask)[0]
            if len(quiet_positions) == 0:
                quiet_positions = np.where(np.arange(n) != top_src)[0]

            quiet_positions = quiet_positions[np.argsort(a[quiet_positions])[:8]]

            for dst in quiet_positions:
                for frac in [0.0005, 0.002, 0.005, 0.01, 0.02, 0.04]:
                    amt = a[top_src] * frac
                    if amt < 1e-16:
                        continue
                    a_trial = a.copy()
                    a_trial[top_src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        it_count += 1
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if rng.integers(3) == 0:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > norm * 1e-10:
                    frac = rng.uniform(0.001, 0.02)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
            if stall > 1000:
                break

        if verbose and it_count > 0 and it_count % 500 == 0:
            print(f"  LocalSearch it={it_count} ({time.monotonic() - t0:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def structured_init(n, seed):
    """Generate a diverse structured initialization."""
    rng = np.random.default_rng(seed)
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 0.7
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(2.0 * x) + 0.1 * np.cos(7.0 * x) + 0.05 * np.cos(13.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-10.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-10.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 5
    elif choice == 5:
        x = np.linspace(-1, 1, n)
        a = np.sqrt(np.maximum(1.0 - x ** 2, 0.0))
    elif choice == 6:
        x = np.linspace(0.001, 0.999, n)
        a = x * (1.0 - x)
    else:
        a = np.ones(n) * 0.5
        for _ in range(rng.integers(3, 8)):
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.05, n * 0.3)
            amp = rng.uniform(-0.3, 0.5)
            a += amp * np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate():
    """
    Main search combining LSE-Adam, LP-Nesterov, and local search
    at multiple lengths with adaptive time budgeting.
    """
    t_global = time.monotonic()
    time_budget = 1680.0

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize if needed
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, 42)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)
    print(f"Starting: n={n_base}, score={best_score:.6f}")

    # Phase 1: LSE-Adam at base length
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LSE-Adam n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.22, 350.0)
    if t1 > 30:
        a_work = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = lse_adam(a_work, t1, T_start=15.0, T_end=0.3, n_stages=16, verbose=True)
        do_try_update(a_out, f"adam_n{n_base}")

    # Phase 2: LP-Nesterov at base length
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: LP-Nesterov n={n_cur} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.30, 450.0)
    if t2 > 30:
        a_work = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp(a_work, t2, verbose=True)
        do_try_update(a_out, f"lp_n{n_cur}")

    # Phase 3: Local search on current best
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: LocalSearch n={n_cur} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 10:
        a_work = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = local_search_polish(a_work, t3, seed=777, verbose=True)
        do_try_update(a_out, f"local_n{n_cur}")

    # Phase 4: Try longer length
    n2 = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LSE-Adam n={n2} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.18, 250.0)
    if t4 > 20:
        init_long = interpolate_sequence(best_seq, n2)
        a_work = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = lse_adam(a_work, t4, T_start=15.0, T_end=0.5, n_stages=12, verbose=True)
        do_try_update(a_out, f"adam_n{n2}")

    # Phase 5: LP at current best length
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: LP-Nesterov n={n_cur} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.22, 300.0)
    if t5 > 20:
        a_work = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp(a_work, t5, verbose=True)
        do_try_update(a_out, f"lp_n{n_cur}_2")

    # Phase 6: Low-temperature LSE polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: LSE-Adam low-T n={n_cur} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.15, 150.0)
    if t6 > 10:
        a_work = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lse_adam(a_work, t6, T_start=5.0, T_end=0.15, n_stages=10, verbose=True)
        do_try_update(a_out, "adam_lowT")

    remaining = time_budget - (time.monotonic() - t_global)

        print(f"\n=== Phase 7: Final LocalSearch n={n_cur} (budget={remaining:.0f}s) ===")
        t7 = min(remaining * 0.85, 100.0)
        if t7 > 5:
            a_work = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = local_search_polish(a_work, t7, seed=424242, verbose=True)
            do_try_update(a_out, "local_final")

    # Final
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
