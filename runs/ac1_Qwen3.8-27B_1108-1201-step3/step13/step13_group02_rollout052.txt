Now, for the projection: after taking a gradient step, I need to project onto {a >= 0, sum(a) = 1}. This is the Euclidean projection onto the probability simplex, which has a well-known algorithm (sort + threshold).

Let me code this up:

```python
def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast LSE gradient descent
with simplex projection, multi-restart, length exploration, and local peak shaving.
Key improvements over previous: no LP dependency, faster iterations, better
temperature/step-size schedules, more aggressive length exploration.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Compute full autocorrelation (convolution with self) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_lse_gradient_weights(b, T):
    """Compute softmax weights for LSE gradient."""
    b_shifted = (b - b.max()) / T
    w = np.exp(b_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return np.full(len(b), 1.0 / len(b))
    return w / w_sum


def lse_gradient(a, w):
    """Compute gradient of LSE(b, T) w.r.t. a, given weights w on b.
    grad_i = 2 * sum_j w[i+j] * a[j] for valid indices.
    """
    n = len(a)
    # We need (w * a_rev)[i + n - 1] where a_rev = a[::-1]
    # Actually: grad_i = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    # This is the cross-correlation of w (length 2n-1) with a (length n) at lag i
    # = (w * a_rev)[i + n - 1] where * is full convolution
    a_rev = a[::-1]
    size = (2 * n - 1) + n - 1  # full conv length = 3n - 2
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)
    conv = irfft(W * AR, fft_size)
    return 2.0 * conv[n - 1:2 * n - 1]


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_gd(a_init, time_budget, T_ratio_start=0.1, T_ratio_end=0.005, verbose=False, seed=42):
    """LSE gradient descent with simplex projection and adaptive step size."""
    n = len(a_init)
    a = project_simplex(a_init.copy())
    b = fast_autocorr(a)
    max_b = b.max()
    best_a = a.copy()
    best_max_b = max_b
    t0 = time.monotonic()
    
    T_start = max_b * T_ratio_start
    T_end = max_b * T_ratio_end
    
    lr = 1.0 / (n * max_b)  # initial step size
    lr_min = lr * 1e-6
    lr_max = lr * 10.0
    prev_grad = None
    prev_a = None
    
    iters = 0
    stalled = 0
    
    while time.monotonic() - t0 < time_budget:
        progress = min(1.0, (time.monotonic() - t0) / time_budget)
        T = T_start * (T_end / T_start) ** progress
        T = max(T, 1e-12)
        
        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break
        
        w = compute_lse_gradient_weights(b, T)
        grad = lse_gradient(a, w)
        
        # BB step size
        if prev_grad is not None and prev_a is not None:
            dy = a - prev_a
            dg = grad - prev_grad
            dy_dg = dy @ dg
            dg_dg = dg @ dg
            if dg_dg > 1e-30:
                bb_lr = (dy @ dy) / max(dy_dg, 1e-30)
                if 0 < bb_lr < lr_max:
                    lr = bb_lr
        
        # Take step
        a_new = a - lr * grad
        a_new = project_simplex(a_new)
        
        b_new = fast_autocorr(a_new)
        max_b_new = b_new.max()
        
        if max_b_new < max_b:
            # Improvement
            a = a_new
            max_b = max_b_new
            lr = min(lr * 1.2, lr_max)
            if max_b_new < best_max_b:
                best_max_b = max_b_new
                best_a = a.copy()
                stalled = 0
            else:
                stalled += 1
        else:
            # No improvement, reduce step
            lr *= 0.5
            stalled += 1
        
        prev_grad = grad
        prev_a = a
        iters += 1
        
        # Random restart if stalled too long
        if stalled > 200:
            rng = np.random.default_rng(iters * 13 + 7)
            perturb = rng.normal(0, 0.01 / n, n)
            a = project_simplex(a + perturb)
            stalled = 0
            lr = 1.0 / (n * fast_autocorr(a).max())
        
        if verbose and iters % 100 == 0:
            elapsed = time.monotonic() - t0
            score = 2 * n * best_max_b
            print(f"      LSE-GD iter={iters} ({elapsed:.0f}s): best_score={score:.6f}, T={T:.2e}")
    
    score = 2 * n * best_max_b
    if verbose:
        print(f"      LSE-GD done: iters={iters}, score={score:.6f}")
    return best_a, score


def peak_shave_local(a_init, time_budget, verbose=False, seed=42):
    """Local peak shaving: reduce the largest contributors to peaks."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    a = project_simplex(a_init.copy())
    b = fast_autocorr(a)
    best_a = a.copy()
    best_max_b = b.max()
    t0 = time.monotonic()
    
    iters = 0
    T = max(0.01, best_max_b * n * 0.01)
    
    while time.monotonic() - t0 < time_budget:
        progress = min(1.0, (time.monotonic() - t0) / time_budget)
        T = best_max_b * n * 0.01 * (0.1 ** progress)
        
        b = fast_autocorr(a)
        max_b = b.max()
        
        # Find the peak
        k_peak = np.argmax(b)
        
        # Find contributors to this peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]
        
        if contrib.max() < 1e-20:
            break
        
        src = idx[np.argmax(contrib)]
        amt = a[src] * rng.uniform(0.05, 0.2)
        
        # Find good destinations (low value, not contributing to peaks)
        # Compute a "pressure" map: how much each position contributes to top peaks
        top_k = np.argsort(b)[::-1][:5]
        pressure = np.zeros(n)
        for k in top_k:
            js2 = max(0, k - n + 1)
            je2 = min(n - 1, k)
            for j in range(js2, je2 + 1):
                pressure[j] += a[j] * a[k - j]
        
        # Destinations: low pressure, not the source
        candidates = np.where(np.arange(n) != src)[0]
        if len(candidates) == 0:
            break
        # Sort by pressure (prefer low pressure)
        sorted_cand = candidates[np.argsort(pressure[candidates])]
        n_dest = min(3, len(sorted_cand))
        selected = sorted_cand[rng.choice(n_dest, min(1, n_dest), replace=False)]
        dst = selected[0]
        
        a_new = a.copy()
        a_new[src] -= amt
        a_new[dst] += amt
        a_new = project_simplex(a_new)
        
        b_new = fast_autocorr(a_new)
        max_b_new = b_new.max()
        
        if max_b_new < best_max_b:
            a = a_new
            best_max_b = max_b_new
            best_a = a.copy()
        elif max_b_new < max_b * 1.01 and rng.random() < np.exp(-(max_b_new - best_max_b) / max(T, 1e-15)):
            a = a_new
        else:
            # Try smaller step
            if iters % 50 == 0:
                # Random perturbation
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-10:
                    amt2 = a[i] * rng.uniform(0.01, 0.05)
                    a_new2 = a.copy()
                    a_new2[i] -= amt2
                    a_new2[j] += amt2
                    a_new2 = project_simplex(a_new2)
                    b_new2 = fast_autocorr(a_new2)
                    if b_new2.max() < best_max_b:
                        a = a_new2
                        best_max_b = b_new2.max()
                        best_a = a.copy()
        
        iters += 1
        if iters % 5000 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"      PeakShave iter={iters} ({elapsed:.0f}s): best={2*n*best_max_b:.6f}")
    
    score = 2 * n * best_max_b
    return best_a, score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_initial_shape(n, shape_type='sin15', seed=42):
    """Create an initial shape of length n."""
    x = np.linspace(0.0, 1.0, n)
    if shape_type == 'sin15':
        a = np.sin(np.pi * x) ** 1.5
    elif shape_type == 'sin2':
        a = np.sin(np.pi * x) ** 2
    elif shape_type == 'sin3':
        a = np.sin(np.pi * x) ** 3
    elif shape_type == 'cos':
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    elif shape_type == 'trap':
        # Trapezoid: ramp up, flat, ramp down
        w = 0.2
        a = np.zeros(n)
        a[:int(w*n)] = np.linspace(0, 1, int(w*n))
        a[int(w*n):n-int(w*n)] = 1.0
        a[n-int(w*n):] = np.linspace(1, 0, int(w*n))
    elif shape_type == 'gauss':
        a = np.exp(-((x - 0.5) ** 2) / (2 * 0.15 ** 2))
    elif shape_type == 'random_smooth':
        rng = np.random.default_rng(seed)
        # Generate random smooth function
        freqs = rng.integers(1, 5, size=8)
        amps = rng.uniform(0.5, 1.0, size=8)
        phases = rng.uniform(0, 2*np.pi, size=8)
        a = 0.5 + sum(amps[i] * np.sin(2*np.pi*freqs[i]*x + phases[i]) for i in range(8))
        a = np.maximum(a, 0.01)
    else:
        a = np.sin(np.pi * x) ** 1.5
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

We want grad of LSE(b, T) = T * log(sum_k exp(b_k/T)) with respect to a.

d/da_i [T * log(sum_k exp(b_k/T))] = T * (1/sum_k exp(b_k/T)) * sum_k exp(b_k/T) * (1/T) * db_k/da_i
= (1/sum_k exp(b_k/T)) * sum_k exp(b_k/T) * db_k/da_i
= sum_k w_k * db_k/da_i

where w_k = exp(b_k/T) / sum_j exp(b_j/T)

And db_k/da_i = 2*a_{k-i} if 0 <= k-i < n, else 0.

So grad_i = sum_{k=0}^{2n-2} w_k * 2*a_{k-i} * [0<=k-i<n]
= 2 * sum_{j=0}^{n-1} w_{i+j} * a_j  (where j = k-i, so k = i+j, and we need 0 <= i+j <= 2n-2)

Since i is in [0, n-1] and j is in [0, n-1], i+j is in [0, 2n-2], which is always valid. So:

grad_i = 2 * sum_{j=0}^{n-1} w_{i+j} * a_j

This is the cross-correlation of w (length 2n-1) with a (length n) at lag i, times 2.

To compute this efficiently:
- rev(a) = a[n-1], a[n-2], ..., a[0] (length n)
- full_conv(w, rev(a)) has length (2n-1) + n - 1 = 3n-2
- The cross-correlation at lag i is full_conv(w, rev(a))[i + n - 1]

Let me verify with a small example:

full_conv(w, rev(a)):
[0] = w0*a1
[1] = w0*a0 + w1*a1
[2] = w1*a0 + w2*a1
[3] = w2*a0

grad_0 should be 2*(w0*a0 + w1*a1) = 2*full_conv[1] = 2*full_conv[0+2-1] = 2*full_conv[1] ✓
grad_1 should be 2*(w1*a0 + w2*a1) = 2*full_conv[2] = 2*full_conv[1+2-1] = 2*full_conv[2] ✓

So the formula is: grad = 2 * full_conv(w, rev(a))[n-1 : 2n-1]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast LSE gradient descent
with simplex projection, multi-restart from diverse shapes, adaptive step-size
(Barzilai-Borwein), length exploration (1600/2000/2500), and local peak shaving.
Avoids LP dependency for speed; uses FFT-based gradient throughout.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full autocorrelation a*a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_grad(a, w):
    """Gradient of LSE(b,T) wrt a given softmax weights w on b.
    grad_i = 2 * sum_j w[i+j] * a[j]."""
    n = len(a)
    a_rev = a[::-1]
    size = (2 * n - 1) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)
    conv = irfft(W * AR, fft_size)
    return 2.0 * conv[n - 1:2 * n - 1]


def softmax_weights(b, T):
    """Numerically stable softmax weights for LSE."""
    b_shift = (b - b.max()) / T
    w = np.exp(b_shift)
    s = w.sum()
    if s < 1e-300:
        return np.full(len(b), 1.0 / len(b))
    return w / s


def proj_simplex(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_gd(a_init, time_budget, T_start_frac=0.1, T_end_frac=0.003, verbose=False, seed=42):
    """LSE gradient descent with simplex projection and BB step size."""
    n = len(a_init)
    a = proj_simplex(np.asarray(a_init, dtype=np.float64).copy())
    b = fast_autocorr(a)
    max_b = b.max()
    best_a = a.copy()
    best_max_b = max_b
    t0 = time.monotonic()

    T_start = max(max_b * T_start_frac, 1e-12)
    T_end = max(max_b * T_end_frac, 1e-15)

    lr = 2.0 / (n * max_b + 1e-15)
    lr_min = lr * 1e-8
    lr_max = lr * 20.0
    prev_grad = None
    prev_a = None
    iters = 0
    stall = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_budget:
        progress = min(1.0, (time.monotonic() - t0) / time_budget)
        T = T_start * (T_end / T_start) ** progress
        T = max(T, 1e-15)

        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-18:
            break

        w = softmax_weights(b, T)
        grad = lse_grad(a, w)

        # Barzilai-Borwein step size
        if prev_grad is not None and prev_a is not None:
            dy = a - prev_a
            dg = grad - prev_grad
            dy_dg = dy @ dg
            dg_dg = dg @ dg
            if dg_dg > 1e-30 and abs(dy_dg) > 1e-30:
                bb = (dy @ dy) / abs(dy_dg)
                if lb_min := lr_min:
                    pass
                if 0 < bb < lr_max:
                    lr = bb
                else:
                    lr = min(lr * 1.5, lr_max)
            else:
                lr = min(lr * 1.1, lr_max)

        a_new = a - lr * grad
        a_new = proj_simplex(a_new)

        b_new = fast_autocorr(a_new)
        max_b_new = b_new.max()

        if max_b_new < max_b * 0.9999:
            a = a_new
            max_b = max_b_new
            lr = min(lr * 1.15, lr_max)
            stall = 0
            if max_b_new < best_max_b:
                best_max_b = max_b_new
                best_a = a.copy()
        else:
            lr = max(lr * 0.5, lr_min)
            stall += 1

        prev_grad = grad
        prev_a = a
        iters += 1

        if stall > 300:
            scale = 0.02 / np.sqrt(n)
            a = proj_simplex(a + rng.normal(0, scale, n))
            b = fast_autocorr(a)
            best_max_b_check = b.max()
            if best_max_b_check < best_max_b:
                best_max_b = best_max_b_check
                best_a = a.copy()
            stall = 0
            lr = 2.0 / (n * b.max() + 1e-15)

        if verbose and iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE-GD iter={iters} ({elapsed:.0f}s): score={2*n*best_max_b:.8f}, T={T:.2e}, lr={lr:.2e}")

    return best_a, 2.0 * n * best_max_b


def peak_shave(a_init, time_budget, verbose=False, seed=42):
    """Local peak shaving with simulated annealing."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    a = proj_simplex(np.asarray(a_init, dtype=np.float64).copy())
    best_a = a.copy()
    best_max_b = fast_autocorr(a).max()
    t0 = time.monotonic()
    iters = 0

    T0 = best_max_b * n * 0.05
    T_end = best_max_b * n * 0.001

    while time.monotonic() - t0 < time_budget:
        progress = min(1.0, (time.monotonic() - t0) / time_budget)
        T = T0 * (T_end / T0) ** progress

        b = fast_autocorr(a)
        max_b = b.max()

        # Target the worst peak
        k_peak = int(np.argmax(b))
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]

        if contrib.max() < 1e-20:
            break

        src = int(idx[np.argmax(contrib)])
        frac = rng.uniform(0.03, 0.15)
        amt = a[src] * frac

        # Compute pressure map from top peaks
        top_k = np.argsort(b)[::-1][:6]
        pressure = np.zeros(n)
        for k in top_k:
            js2 = max(0, k - n + 1)
            je2 = min(n - 1, k)
            i2 = np.arange(js2, je2 + 1)
            pressure[i2] += a[i2] * a[k - i2]

        # Pick destination with low pressure
        cands = np.where(np.arange(n) != src)[0]
        if len(cands) == 0:
            break
        sorted_c = cands[np.argsort(pressure[cands])]
        n_pick = min(4, len(sorted_c))
        dst = int(sorted_c[rng.integers(n_pick)])

        a_new = a.copy()
        a_new[src] -= amt
        a_new[dst] += amt
        a_new = proj_simplex(a_new)
        b_new = fast_autocorr(a_new)
        mb_new = b_new.max()

        if mb_new < best_max_b:
            a = a_new
            best_max_b = mb_new
            best_a = a.copy()
        elif rng.random() < np.exp(-(mb_new - best_max_b) / max(T, 1e-18)):
            a = a_new

        iters += 1

        if iters % 10000 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"      PS iter={iters} ({elapsed:.0f}s): score={2*n*best_max_b:.8f}")

    return best_a, 2.0 * n * best_max_b


def interp_seq(f, new_n):
    """Interpolate sequence to new length preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    ox = np.linspace(0.0, 1.0, old_n)
    nx = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(nx, ox, fn)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def make_shape(n, typ, seed=42):
    """Generate a normalized non-negative initial shape."""
    x = np.linspace(0.0, 1.0, n)
    if typ == 'sin15':
        a = np.sin(np.pi * x) ** 1.5
    elif typ == 'sin2':
        a = np.sin(np.pi * x) ** 2
    elif typ == 'sin3':
        a = np.sin(np.pi * x) ** 3
    elif typ == 'sin4':
        a = np.sin(np.pi * x) ** 4
    elif typ == 'cos2':
        a = (0.5 + 0.5 * np.cos(2 * np.pi * x)) ** 2
    elif typ == 'trap':
        w = 0.15
        nw = max(1, int(w * n))
        a = np.ones(n)
        a[:nw] = np.linspace(0, 1, nw)
        a[n - nw:] = np.linspace(1, 0, nw)
    elif typ == 'gauss':
        a = np.exp(-4.0 * (x - 0.5) ** 2)
    elif typ == 'bump':
        a = np.maximum(0, 1 - (2 * (x - 0.5)) ** 2) ** 2
    else:
        a = np.sin(np.pi * x) ** 1.5
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def eval_local(a):
    """Fast local score computation (sum normalized to 1)."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()


def evaluate_safe(a):
    """Call the external evaluate_sequence safely."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')
