import numpy as np

from scipy.fft import rfft, irfft

def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def _proj_simplex(v):
    n = len(v)
    if n <= 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 0:
        out /= s
    else:
        out = np.ones(n) / n
    return out

def _lse_grad(y, T):
    n = len(y)
    ac = _fast_autocorr(y)
    ac_max = ac.max()
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(y, fft_size)
    W = rfft(w, fft_size)
    cross = irfft(np.conj(A) * W, fft_size)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad

def _build_conv_matrix(a, n):
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j: j + n, j] = a

    b_var = cp.Variable(n, nonneg=True)
    T = _build_conv_matrix(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def _interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

import numpy as np

from scipy.fft import rfft, irfft


def _fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def _proj_simplex(v):
    n = len(v)
    if n <= 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 0:
        out /= s
    else:
        out = np.ones(n) / n
    return out


def _lse_grad(y, T):
    n = len(y)
    ac = _fast_autocorr(y)
    ac_max = ac.max()
    shifted = (ac - ac_max) / T
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(y, fft_size)
    W = rfft(w, fft_size)
    cross = irfft(np.conj(A) * W, fft_size)
    grad = 2.0 * cross[n - 1: n - 1 + n]
    return grad

def _build_conv_matrix(a, n):

    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

                    if s_t < 1e-15:

            i = rng.integers(n)

def _interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    time_budget = min(float(budget_s) - 100.0, 1680.0)

    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = _evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    def _try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = _evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = _evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # === Phase 1: Quick LSE-FISTA ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.06, 90.0)
    if t1 > 10:
        n_cur = len(best_seq)
        print(f"\n=== Phase 1: LSE-FISTA n={n_cur} (budget={t1:.0f}s) ===")
        T_sched = [8.0, 2.5, 0.8, 0.25, 0.08]
        a_out, _ = _lse_fista(best_seq, T_sched, 12000, t1, verbose=True)
        _try_update(a_out, f"lse1_n{n_cur}")

    # === Phase 2: LP at base length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 30:
        print(f"\n=== Phase 2: LP n={n_best} (budget={t2:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = _lp_optimize(a_norm, t2, solver, verbose=True, use_nesterov=True)
        _try_update(a_out, f"lp1_n{n_best}")

    # === Phase 3: Interpolate to n+400 and LP ===
    n2 = len(best_seq) + 400
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.25, 380.0)
    if t3 > 30:
        print(f"\n=== Phase 3: LP n={n2} (budget={t3:.0f}s) ===")
        init_long = _interpolate_sequence(best_seq, n2)
        # Quick LSE pre-process
        t_pre = min(t3 * 0.15, 30.0)
        if t_pre > 5:
            a_pre, _ = _lse_fista(init_long, [4.0, 1.2, 0.4, 0.1], 6000, t_pre, verbose=False)
            a_norm = a_pre * np.sqrt(2.0 * n2) / a_pre.sum()
        else:
            a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        t_lp = t3 - t_pre - 5
        if t_lp > 15:
            a_out, _ = _lp_optimize(a_norm, t_lp, solver, verbose=True, use_nesterov=True)
            _try_update(a_out, f"lp2_n{n2}")

    # === Phase 4: Interpolate to n+800 and LP ===
    n3 = len(best_seq) + 800
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.25, 350.0)
    if t4 > 30:
        print(f"\n=== Phase 4: LP n={n3} (budget={t4:.0f}s) ===")
        init_long3 = _interpolate_sequence(best_seq, n3)
        t_pre = min(t4 * 0.12, 25.0)
        if t_pre > 5:
            a_pre, _ = _lse_fista(init_long3, [4.0, 1.0, 0.3], 4000, t_pre, verbose=False)
            a_norm = a_pre * np.sqrt(2.0 * n3) / a_pre.sum()
        else:
            a_norm = init_long3 * np.sqrt(2.0 * n3) / init_long3.sum()
        t_lp = t4 - t_pre - 5
        if t_lp > 15:
            a_out, _ = _lp_optimize(a_norm, t_lp, solver, verbose=True, use_nesterov=True)
            _try_update(a_out, f"lp3_n{n3}")

    # === Phase 5: LSE-FISTA on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.12, 150.0)
    if t5 > 10:
        print(f"\n=== Phase 5: LSE-FISTA n={n_best} (budget={t5:.0f}s) ===")
        T_sched = [3.0, 0.8, 0.25, 0.08, 0.03]
        a_out, _ = _lse_fista(best_seq, T_sched, 20000, t5, verbose=True)
        _try_update(a_out, f"lse2_n{n_best}")

    # === Phase 6: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t6 = min(remaining * 0.12, 100.0)
    if t6 > 10:
        print(f"\n=== Phase 6: Peak shave n={n_best} (budget={t6:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = _peak_shave(a_norm, t6, seed=42)
        _try_update(a_out, "peak_shave1")

    # === Phase 7: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t7 = min(remaining * 0.25, 200.0)
    if t7 > 20:
        print(f"\n=== Phase 7: LP final n={n_best} (budget={t7:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = _lp_optimize(a_norm, t7, solver, verbose=True, use_nesterov=False)
        _try_update(a_out, "lp_final")

    # === Phase 8: Final peak shave ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        t8 = min(remaining * 0.6, 80.0)
        if t8 > 5:
            print(f"\n=== Phase 8: Peak shave 2 n={n_best} (budget={t8:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = _peak_shave(a_norm, t8, seed=77777)
            _try_update(a_out, "peak_shave2")

    if best_seq is not None:
        fs = _evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
