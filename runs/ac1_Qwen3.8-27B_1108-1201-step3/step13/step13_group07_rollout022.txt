    # cross_corr(a, w)[j] = sum_m a[m] * w[m+j]

    a_pad[:n] = a
    A_f = rfft(a_pad, fft_size)
    W_f = rfft(weights, fft_size)

    # Since a_pad[m] = a[m] for m < n and 0 otherwise:
    # cross_corr[j] = sum_{m=0}^{min(n-1, 2n-2-j)} a[m] * w[m+j]

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) Spectral shaping, (2) Annealed LogSumExp-Adam with Nesterov,
(3) LP polishing via zero-sum game, (4) Peak shaving.
Explores multiple sequence lengths.

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Full autocorrelation (linear convolution a*a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_convolution(a, b):
    """Linear convolution of a and b using FFT."""
    n = len(a) + len(b) - 1
    fft_size = 1 << (n - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:n]

def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if v.sum() < 1e-300:
        return np.full(n, target_sum / n)
    # Standard simplex projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    rho = cond.sum() - 1  # Last index where condition is true
    if rho < 0:
        return np.full(n, target_sum / n)
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0.0)
    s = result.sum()
    if s < 1e-300:
        return np.full(n, target_sum / n)
    result *= target_sum / s
    return result

def lse_gradient(a, T):
    """
    Compute gradient of (1/T)*log(sum(exp(T*conv(a,a)))) w.r.t. a.
    Returns (gradient, lse_value, max_conv).
    """
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    lse_val = (max_Tb + np.log(exp_Tb.sum())) / T
    
    # Gradient: grad[j] = 2 * sum_k w[k] * a[k-j] (valid indices)
    # = 2 * cross_correlation(a_padded, w) [j]
    fft_size = 1 << (2*n - 1).bit_length()
    a_pad = np.zeros(2*n - 1)
    a_pad[:n] = a
    A_f = rfft(a_pad, fft_size)
    W_f = rfft(w, fft_size)
    cross_corr = irfft(A_f * np.conj(W_f), fft_size)
    grad = 2.0 * cross_corr[:n]
    
    return grad, lse_val, b.max()

def annealed_lse_adam(a0, time_limit, T_start=5.0, T_end=500.0, 
                       lr=0.01, beta1=0.9, beta2=0.999, nesterov=True):

    Annealed LogSumExp + Adam optimizer with optional Nesterov momentum.
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    t0 = time.monotonic()
    
    # Total estimated iterations
    # Each iteration: ~2 FFTs of size 8192 (for n=1600) ≈ 0.5ms
    # For n=3000: FFT size 8192, similar
    est_iter_time = 0.001  # ~1ms per iteration estimate
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed_frac = min(1.0, t / max(1, int(time_limit / est_iter_time)))
        # Annealing: T increases over time
        T = T_start * (T_end / T_start) ** elapsed_frac
        
        grad, lse_val, max_conv = lse_gradient(a, T)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        lr_t = lr / (1 + 0.0001 * t)
        
        if nesterov:
            # Nesterov lookahead
            a_lookup = a - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
            a_lookup = project_simplex(a_lookup)
            grad2, _, _ = lse_gradient(a_lookup, T)
            m = beta1 * m + (1 - beta1) * grad2
            v = beta2 * v + (1 - beta2) * (grad2 * grad2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        else:
            a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        
        a_new = project_simplex(a_new)
        
        # Check true score (not just LogSumExp)

            if true_score < best_score:

def reduced_lp_step(a_fixed, b_current, top_k=50, solver=None):
    """
    Given fixed a, find b that minimizes max over TOP K peaks of (a*b)[k].
    This is an LP with only top_k + 2 constraints (plus sum and nonneg).
    """
    n = len(a_fixed)
    b_vals = b_current  # current convolution a*a
    top_idx = np.argsort(b_vals)[::-1][:top_k]
    
    # Build reduced constraint matrix
    # T_reduced[i, j] = a_fixed[k_i - j] if 0 <= k_i - j < n
    T_red = np.zeros((top_k, n))
    for i, k in enumerate(top_idx):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            T_red[i, j] = a_fixed[k - j]
    
    # LP: min M s.t. T_red @ b <= M, sum(b) = 1, b >= 0
    import cvxpy as cp
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T_red @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    if solver:
        prob.solve(solver=solver)

    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val /= s

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) Spectral shaping init, (2) Annealed LogSumExp-Adam with Nesterov,
(3) LP polishing via zero-sum game with reduced constraints, (4) Peak shaving.
Explores multiple sequence lengths (1600, 2000, 2500, 3000).
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full autocorrelation (linear convolution a*a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolution(a, b):
    """Linear convolution of a and b using FFT."""
    n = len(a) + len(b) - 1
    fft_size = 1 << (n - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:n]


def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if v.sum() < 1e-300:
        return np.full(n, target_sum / n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    rho = int(cond.sum()) - 1
    if rho < 0:
        return np.full(n, target_sum / n)
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0.0)
    s = result.sum()
    if s < 1e-300:
        return np.full(n, target_sum / n)
    result *= target_sum / s
    return result


def lse_gradient(a, T):
    """
    Compute gradient of (1/T)*log(sum(exp(T*conv(a,a)))) w.r.t. a.
    Returns (gradient, lse_value, max_conv).
    """
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    lse_val = (max_Tb + np.log(exp_Tb.sum())) / T

    # Gradient: grad[j] = 2 * sum_k w[k] * a[k-j] (valid indices)
    fft_size = 1 << (2 * n - 1).bit_length()
    a_pad = np.zeros(2 * n - 1)
    a_pad[:n] = a
    A_f = rfft(a_pad, fft_size)
    W_f = rfft(w, fft_size)
    cross_corr = irfft(A_f * np.conj(W_f), fft_size)
    grad = 2.0 * cross_corr[:n]

    return grad, lse_val, b.max()


def annealed_lse_adam(a0, time_limit, T_start=5.0, T_end=500.0,
                      lr=0.01, use_nesterov=False, verbose=True):
    """
    Annealed LogSumExp + Adam optimizer.
    T anneals from T_start to T_end over the optimization.
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** elapsed_frac

        grad, lse_val, max_conv = lse_gradient(a, T)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        lr_t = lr / (1 + 0.00005 * t)
        step = lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = project_simplex(a_new)

        if t % 5 == 0:
            true_score = 2.0 * n * fast_autocorr(a_new).max()
            if true_score < best_score:
                best_score = true_score
                best_a = a_new.copy()
                if verbose and true_score < 1.51:
                    elapsed = time.monotonic() - t0
                    print(f"          NEW BEST: score={true_score:.6f} (t={t}, {elapsed:.0f}s)")
            a = a_new
        else:
            a = a_new

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            cur_score = 2.0 * n * fast_autocorr(a).max()
            print(f"          iter={t} ({elapsed:.0f}s): T={T:.1f}, cur={cur_score:.6f}, best={best_score:.6f}")

    return best_a, best_score


def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_rows(a_fixed, row_indices):
    """Build rows of the convolution matrix for specified row indices."""
    n = len(a_fixed)
    T = np.zeros((len(row_indices), n))
    for i, k in enumerate(row_indices):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            T[i, j] = a_fixed[k - j]
    return T


def solve_reduced_lp(a_fixed, top_k=200, solver=None):
    """
    Solve LP: min M s.t. (a*b)[k] <= M for top_k peaks, sum(b)=1, b>=0.
    Uses reduced constraint set for speed.
    """
    import cvxpy as cp
    n = len(a_fixed)
    a_fixed = np.asarray(a_fixed, dtype=np.float64)

    # Find top_k indices from current a*a
    aa = fast_autocorr(a_fixed)
    top_idx = np.argsort(aa)[::-1][:top_k]

    T_red = build_conv_rows(a_fixed, top_idx)

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T_red @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    if solver:
        try:
            prob.solve(solver=solver, max_iter=500)
        except Exception:
            prob.solve()
    else:
        prob.solve()

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val /= s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, top_k=200, verbose=True):
    """Nesterov-accelerated LP best-response with reduced constraints."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0
    stall = 0

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        alpha_t = 2.0 / (t + 2.0)

        b_star, M_val = solve_reduced_lp(y, top_k=top_k, solver=solver)
        if b_star is None:
            continue

        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = project_simplex(a_new)

        y = y_new
        a = a_new

        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0

        avg_a += a
        n_avg += 1
        if n_avg % 200 == 0:
            avg_check = avg_a / n_avg
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
                stall = 0

        if stall > 5000:
            break

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"          LP t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}")

    return best_a, best_score


def spectral_shaping(a, n_iter=300, alpha=0.3):
    """Iteratively flatten power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 50.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def peak_shaving(a0, time_limit, seed=42):
    """Coordinate-descent peak shaving with random restarts."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:5]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            src = idx[np.argmax(contrib)]

            peak_areas = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_areas.update(range(js2, je2 + 1))

            candidates = [j for j in range(n) if j not in peak_areas]
            if not candidates:
                candidates = list(range(n))
            candidates_sorted = sorted(candidates, key=lambda j: a[j])
            dst_pool = candidates_sorted[:max(1, len(candidates_sorted) // 4)]
            dst = dst_pool[rng.integers(len(dst_pool))]

            for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.002, 0.015)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
            if stall > 300:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(arr):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

    print(f"Total time: {time.monotonic() - t_global_start:.1f}s")

    return [float(x) for x in best_seq]
