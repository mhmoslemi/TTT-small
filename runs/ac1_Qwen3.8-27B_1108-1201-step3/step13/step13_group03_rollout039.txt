    b = np.asarray(a, dtype=np.float64)
    # Find top K lags
    conv_val = _fast_autocorr(b)
    top_k = np.argsort(conv_val)[::-1][:K]
    
    # Build constraint matrix for only top K lags
    T = np.zeros((K, n))
    for idx, k in enumerate(top_k):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                T[idx, i] = b[j]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    prob.solve(solver=solver, verbose=False)
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    return np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)

def _get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)

def _fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def _lse_grad(a, T):
    n = len(a)
    ac = _fast_conv(a)
    ac_max = ac.max()
    shifted = np.clip((ac - ac_max) / T, -60.0, 60.0)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    cross = irfft(np.conj(A) * W, fft_size)
    grad = 2.0 * cross[0:n]  # FIXED
    return grad

def _proj_simplex(v, target_sum=1.0):
    n = len(v)
    if n <= 1:
        return np.array([target_sum])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.ones(n) * target_sum / n
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s > 0:
        out *= target_sum / s
    else:
        out = np.ones(n) * target_sum / n
    return out

def _lse_fista(a0, T_schedule, iters_per_stage, time_limit):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = _fast_conv(a).max()
    t0 = time.monotonic()
    
    for stage_i, T in enumerate(T_schedule):
        L = max(n / (T * 2.0), 1.0)
        t_fista = 1
        staging = a.copy()
        max_iters = max(int(iters_per_stage * (T_schedule[0] / T) ** 0.3), 2000)
        
        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_score
            grad = _lse_grad(y, T)
            step = 1.0 / L
            z = y - step * grad
            a_new = _proj_simplex(z, target_sum=1.0)
            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - staging)
                y_new = np.maximum(y_new, 0.0)
                s_y = y_new.sum()
                if s_y > 1e-15:
                    y_new /= s_y
                else:
                    y_new = a_new.copy()
            staging = a_new.copy()
            a = a_new
            y = y_new
            t_fista += 1
            if it % 500 == 0:
                sc = _fast_conv(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
        sc = _fast_conv(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_a, best_score

def _build_T_full(a):
    n = len(a)
    T = np.zeros((2*n-1, n))
    for i in range(n):
        T[i:i+n, i] = a
    return T

def _active_set_lp(a, M, K, T_full, solver):
    n = len(a)
    conv_val = _fast_conv(a)
    top_k = np.argsort(conv_val)[::-1][:K]
    T_sel = T_full[top_k]
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_sel @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    prob.solve(solver=solver, verbose=False)
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val
