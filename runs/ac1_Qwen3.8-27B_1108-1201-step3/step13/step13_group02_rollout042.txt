def _fast_autocorr(a):
    """Compute full autocorrelation (linear convolution a*a) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def _normalize(a, norm_val=None):
    """Normalize sequence to have sum = norm_val."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    if norm_val is None:
        norm_val = np.sqrt(2.0 * len(a))
    return a * norm_val / s

    if old_n == new_n:
        return f.copy()
    f = np.asarray(f, dtype=np.float64)
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, fn)
    fi = np.maximum(fi, 0.0)
    ss = fi.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return fi / ss


def _lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    
    # Build constraint matrix T where (T @ b)[k] = (a*b)[k]
    # T has shape (2n-1, n)
    # T[k, j] = a[k-j] if 0 <= k-j < n, else 0
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def _lp_best_response_active(a, M, solver, max_iter=5):
    """LP best response with active constraint reduction."""
    n = len(a)
    size = 2 * n - 1
    
    # Build constraint matrix
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    
    b_var = cp.Variable(n, nonneg=True)
    
    # Initial full solve
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    
    # Iteratively reduce to active constraints
    for _ in range(max_iter):
        circ = T @ b_val
        tol = 1e-7 * max(M.max() if hasattr(M, 'max') else M, 1e-10)
        active = circ >= M - tol
        n_active = int(active.sum())
        if n_active >= size - 10 or n_active < 30:
            break
        
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T[active, :] @ b_var <= M[active]])
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        b_new = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        circ_new = T @ b_new
        if circ_new.max() <= M * (1 + 1e-5):
            b_val = b_new
        else:
            break
    
    if b_val.sum() < 1e-15:
        return None
    return b_val


def _nesterov_lp(a0, time_limit, solver, verbose=True):
    """LP Nesterov acceleration for minimizing max autocorrelation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_autocorr(y)
        M_max = M_curr.max()
        if M_max < 1e-12:
            break
        
        # Use active constraint LP after initial iterations
        if t < 30:
            b = _lp_best_response(y, M_max, solver)
        else:
            b = _lp_best_response_active(y, M_max, solver)
        
        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y = y_new
        a = a_new
        
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        
        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, stall={stall}")
        
        # Adaptive restart
        if stall > 300:
            rng = np.random.default_rng(n_iters * 13 + 7)
            perturb_scale = 0.03 * a.mean()
            y = np.maximum(y + rng.normal(0, perturb_scale, n), 0)
            y = y * norm / y.sum()
            a = np.maximum(a + rng.normal(0, perturb_scale, n), 0)
            a = a * norm / a.sum()
            stall = 0
            if verbose:
                print(f"    [Restart at iter {n_iters}]")
    
    return best_a, best_score


def _lse_gradient_descent(a0, time_limit, verbose=True):
    """LSE-smoothed gradient descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0
    
    T_start = max(0.5, best_score * 0.1)
    T_end = max(0.001, best_score * 0.003)
    
    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        # Cosine annealing
        T = T_end + 0.5 * (T_start - T_end) * (1 + np.cos(np.pi * progress))
        
        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break
        
        # Softmax weights with numerical stability
        scaled = (b - max_b) / T
        # Clip to avoid overflow/underflow issues
        scaled = np.clip(scaled, -50, 0)
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum
        
        # Gradient: grad_i = 2 * sum_k w[k] * (a_conv_contrib)_k_i
        # (a*b)_k = sum_j a[j] * b[k-j], so grad of (a*a)_k w.r.t. a[i] = 2*a[k-i] (if valid)
        # grad_i = 2 * sum_k w[k] * a[k-i] (where 0 <= k-i < n)
        # This is 2 * (w_reversed * a)[i + n - 1] ... let me think more carefully.
        
        # Actually: d/d a[i] of (a*a)[k] = a[k-i] + a[k-i] = 2*a[k-i] if 0<=k-i<n
        # So grad[i] = 2 * sum_{k: 0<=k-i<n} w[k] * a[k-i]
        # = 2 * sum_{j: 0<=j<n} w[i+j] * a[j] (where k = i+j)
        
        # This is a cross-correlation of w and a!
        # grad[i] / 2 = sum_j w[i+j] * a[j] = (w * a_rev)[i + n - 1]
        # where a_rev = a[::-1]
        
        a_rev = a[::-1]
        fft_size = 1 << (3 * n - 2).bit_length()
        W = rfft(w, fft_size)
        AR = rfft(a_rev, fft_size)
        conv = irfft(W * AR, fft_size)
        grad = 2.0 * conv[n - 1:2 * n - 1]
        
        # Adaptive step size
        # Use a simple backtracking approach
        step = 1.0 / (n * max(T, 1e-6) * 0.3)
        
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        # Check improvement, backtrack if needed
        score_new = fast_autocorr(a_new).max()
        if score_new > best_score * 1.1:
            # Try smaller step
            a_new = a - 0.1 * step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new * norm / s_new
                score_new = fast_autocorr(a_new).max()
                if score_new > best_score * 1.1:
                    a_new = a.copy()
                    score_new = best_score
        
        a = a_new
        steps += 1
        
        if score_new < best_score:
            best_score = score_new
            best_a = a.copy()
        
        if verbose and steps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE step={steps} ({elapsed:.0f}s): T={T:.4f}, best={best_score:.8f}")
    
    return best_a, best_score


def _peak_shaving(a0, time_limit, seed=42, verbose=True):
    """Multi-destination peak shaving with simulated annealing."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_improve = 0
    n_stall = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        T = best_score * 0.02 * (1.0 - elapsed_frac) + best_score * 0.001
        
        b = fast_autocorr(a)
        # Get top peaks
        top_k = np.argsort(b)[::-1][:12]
        
        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            
            if contrib.max() < 1e-18:
                continue
            
            src = idx[np.argmax(contrib)]
            
            # Find non-peak positions
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True
            
            non_peak = np.where(~peak_mask)[0]
            if len(non_peak) < 5:
                non_peak = np.where(np.arange(n) != src)[0]
            
            # Select destinations (prefer low-value positions)
            n_dest = min(8, len(non_peak))
            sorted_dst = non_peak[np.argsort(a[non_peak])][:n_dest * 2]
            selected_dst = sorted_dst[rng.choice(len(sorted_dst), min(n_dest, len(sorted_dst)), replace=False)]
            
            # Try different fractions
            for frac in [0.03, 0.07, 0.12, 0.2]:
                amt_total = a[src] * frac
                if amt_total < 1e-18:
                    continue
                
                # Distribute to destinations
                dst_vals = a[selected_dst] + 1e-10
                weights = 1.0 / dst_vals
                weights = weights / weights.sum()
                amounts = amt_total * weights
                
                a_trial = a.copy()
                a_trial[src] -= amt_total
                a_trial[selected_dst] += amounts
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                
                score = fast_autocorr(a_trial).max()
                
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    n_improve += 1
                    break
                elif rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                    a = a_trial
                    improved = True
                    break
        
        if not improved:
            n_stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-8:
                frac = rng.uniform(0.05, 0.2)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= -1e-12:
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            n_improve += 1
                        elif rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                            a = a_trial
                            improved = True
            
            if n_stall > 150:
                # Larger perturbation
                idx_pert = rng.choice(n, size=max(1, n // 40), replace=False)
                a_trial = a.copy()
                for idx in idx_pert:
                    amt = a[idx] * rng.uniform(0.1, 0.5)
                    j = rng.integers(n)
                    a_trial[idx] -= amt
                    a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score * 1.02:
                        a = a_trial
                        if score < best_score:
                            best_score = score
                            best_a = a.copy()
                n_stall = 0
        
        if verbose and n_improve % 1000 == 0 and n_improve > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave improve={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def _spectral_shape_init(n, rng, norm=None):
    """Create an initial sequence with good spectral properties."""
    if norm is None:
        norm = np.sqrt(2.0 * n)
    
    # Try different parametric families
    options = []
    
    # Raised cosine
    x = np.linspace(0, 1, n)
    options.append(np.sin(np.pi * x))
    
    # Hamming-like
    options.append(0.54 - 0.46 * np.cos(2 * np.pi * x))
    
    # Raised cosine with power
    options.append(np.sin(np.pi * x) ** 1.5)
    
    # Tukey-like
    r = 0.5
    tukey = np.ones(n)
    edge = int(r * n / 2)
    for i in range(edge):
        tukey[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (r * n) - 1)))
        tukey[n-1-i] = tukey[i]
    options.append(tukey)
    
    # Random smooth (low-pass filtered white noise)
    noise = rng.standard_normal(n)
    # Apply a simple low-pass by averaging
    kernel_size = max(3, n // 100)
    kernel = np.ones(kernel_size) / kernel_size
    smooth = np.convolve(noise, kernel, mode='same')
    smooth = np.maximum(smooth, 0)
    options.append(smooth)
    
    # Gaussian window
    center = n / 2
    sigma = n / (4 * rng.uniform(0.8, 1.5))
    options.append(np.exp(-0.5 * ((x * n - center) / sigma) ** 2))
    
    # Add slight random perturbation to each
    best = None
    best_sc = float('inf')
    for opt in options:
        opt = np.maximum(opt, 0)
        s = opt.sum()
        if s < 1e-15:
            continue
        opt = opt * norm / s
        sc = fast_autocorr(opt).max()
        if sc < best_sc:
            best_sc = sc
            best = opt.copy()
    
    if best is None:
        best = np.ones(n) * norm / n
    
    return best


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function to find the best sequence."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    
    solver = _get_solver()
    rng = np.random.default_rng(seed)
    print(f"Time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = _evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass
    
    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = _evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Initialize if no parent
    if best_seq is None:
        n_base = 1600
        best_seq = _spectral_shape_init(n_base, rng)
        best_seq = best_seq / best_seq.sum()
        best_score = _evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")
    
    n_base = len(best_seq)
    
    # Phase 1: LP Nesterov at base length
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.30, 500.0)
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = _nesterov_lp(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n_base}")
    
    # Phase 2: LSE gradient descent
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LSE-GD n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.10, 120.0)
    if t2 > 10:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _lse_gradient_descent(a_norm, t2, verbose=True)
        do_try_update(a_out, f"lse_n{n_cur}")
    
    # Phase 3: Peak shaving
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: PeakShave n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.10, 100.0)
    if t3 > 10:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _peak_shaving(a_norm, t3, seed=123, verbose=True)
        do_try_update(a_out, f"ps_n{n_cur}")
    
    # Phase 4: Longer length n+200
    n2 = n_base + 200
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LP Nesterov n={n2} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.18, 250.0)
    if t4 > 30:
        init_long = _interpolate_to_length(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = _nesterov_lp(a_norm, t4, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n2}")
        
        # LSE polish at new length
        remaining = time_budget - (time.monotonic() - t_global)
        t4b = min(remaining * 0.06, 60.0)
        if t4b > 10:
            n_cur = len(best_seq)
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = _lse_gradient_descent(a_norm, t4b, verbose=False)
            do_try_update(a_out, f"lse_n{n_cur}")
    
    # Phase 5: Even longer n+400
    n3 = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: LP Nesterov n={n3} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.12, 180.0)
    if t5 > 30:
        init_long3 = _interpolate_to_length(best_seq, n3)
        a_norm = init_long3 * np.sqrt(2.0 * n3) / init_long3.sum()
        a_out, _ = _nesterov_lp(a_norm, t5, solver, verbose=True)

        alt_init = _spectral_shape_init(n_alt, rng)
        # Blend with the current best (interpolated)
        blend = 0.5 * _interpolate_to_length(best_seq, n_alt) + 0.5 * (alt_init / alt_init.sum())
        blend = blend / blend.sum()
        
        t6 = min(remaining * 0.12, 120.0)

            a_out, _ = _nesterov_lp(a_norm, t6, solver, verbose=True)
            do_try_update(a_out, f"blend_n{n_alt}")
    
    # Phase 7: Final peak shaving on best

def _interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length using cubic-like smoothing."""
    f = np.asarray(f, dtype=np.float64)
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    fn = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)

def _build_conv_matrix(a):

    for j in range(n):

    return T

    """Solve LP: max sum(b) s.t. conv(a,b) <= M_val, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = _build_conv_matrix(a)
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M_val])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def _lse_gd(a0, time_limit, verbose=True):
    """LSE-smoothed gradient descent with adaptive step and cosine annealing."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0

    T_start = max(0.3, best_score * 0.08)
    T_end = max(0.0005, best_score * 0.002)

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_end + 0.5 * (T_start - T_end) * (1.0 + np.cos(np.pi * progress))

        b = _fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break

        scaled = (b - max_b) / T
        scaled = np.clip(scaled, -60.0, 0.0)
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        # Gradient: grad[i] = 2 * sum_j w[i+j] * a[j] = 2 * (w * a_rev)[i+n-1]
        a_rev = a[::-1]
        fft_size = 1 << (3 * n - 2).bit_length()
        W = rfft(w, fft_size)
        AR = rfft(a_rev, fft_size)
        conv = irfft(W * AR, fft_size)
        grad = 2.0 * conv[n - 1:2 * n - 1]

        # Adaptive step size
        step = 1.0 / (n * max(T, 1e-8) * 0.4)
        step = min(step, 2.0)

        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        score_new = _fast_autocorr(a_new).max()

        # Backtracking if terrible
        if score_new > best_score * 1.5:
            for shrink in [0.5, 0.1, 0.01]:
                a_trial = a - step * shrink * grad
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial * norm / st
                sc_try = _fast_autocorr(a_trial).max()
                if sc_try < score_new:
                    a_new = a_trial
                    score_new = sc_try
                    break
            else:
                a_new = a.copy()
                score_new = best_score

        a = a_new
        steps += 1

        if score_new < best_score:
            best_score = score_new
            best_a = a.copy()

        if verbose and steps % 80 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE step={steps} ({elapsed:.0f}s): T={T:.5f}, best={best_score:.8f}")

    return best_a, best_score


def _peak_shaving(a0, time_limit, seed=42, verbose=True):
    """Multi-destination peak shaving with simulated annealing."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    t0 = time.monotonic()
    n_improve = 0
    n_stall = 0

    while time.monotonic() - t0 < time_limit:
        ef = min(1.0, (time.monotonic() - t0) / time_limit)
        T = best_score * (0.03 * (1.0 - ef) + 0.001)

        b = _fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:10]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            if contrib.max() < 1e-18:
                continue

            src = idx[np.argmax(contrib)]

            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            non_peak = np.where(~peak_mask)[0]
            if len(non_peak) < 5:
                non_peak = np.where(np.arange(n) != src)[0]

            n_dest = min(6, len(non_peak))
            sorted_dst = non_peak[np.argsort(a[non_peak])][:n_dest * 3]
            sel = sorted_dst[rng.choice(len(sorted_dst), min(n_dest, len(sorted_dst)), replace=False)]

            for frac in [0.04, 0.08, 0.15, 0.25]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue

                dv = a[sel] + 1e-12
                wts = 1.0 / dv
                wts = wts / wts.sum()
                amts = amt * wts

                a_t = a.copy()
                a_t[src] -= amt
                a_t[sel] += amts
                a_t = np.maximum(a_t, 0.0)
                st = a_t.sum()
                if st < 1e-15:
                    continue
                a_t = a_t * norm / st

                sc = _fast_autocorr(a_t).max()

                if sc < best_score:
                    a = a_t
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    n_improve += 1
                    break
                elif rng.random() < np.exp(-(sc - best_score) / max(T, 1e-12)):
                    a = a_t
                    improved = True
                    break

        if not improved:
            n_stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-8:
                amt = a[i] * rng.uniform(0.05, 0.2)
                a_t = a.copy()
                a_t[i] -= amt
                a_t[j] += amt
                a_t = np.maximum(a_t, 0)
                st = a_t.sum()
                if st > 1e-15:
                    a_t = a_t * norm / st
                    sc = _fast_autocorr(a_t).max()
                    if sc < best_score:
                        a = a_t
                        best_score = sc
                        best_a = a.copy()
                        n_improve += 1
                    elif rng.random() < np.exp(-(sc - best_score) / max(T, 1e-12)):
                        a = a_t
                        improved = True

            if n_stall > 120:
                np_ = rng.choice(n, size=max(2, n // 50), replace=False)
                a_t = a.copy()
                for idx in np_:
                    amt = a_t[idx] * rng.uniform(0.1, 0.5)
                    j2 = rng.integers(n)
                    a_t[idx] -= amt
                    a_t[j2] += amt
                a_t = np.maximum(a_t, 0)
                st = a_t.sum()
                if st > 1e-15:
                    a_t = a_t * norm / st
                    sc = _fast_autocorr(a_t).max()
                    if sc < best_score * 1.03:
                        a = a_t
                        if sc < best_score:
                            best_score = sc
                            best_a = a.copy()
                n_stall = 0

        if verbose and n_improve > 0 and n_improve % 800 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave imp={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def _parametric_init(n, rng):
    """Generate diverse parametric initializations and return the best one."""
    norm = np.sqrt(2.0 * n)
    x = np.linspace(0.0, 1.0, n)
    candidates = []

    # Raised cosine
    candidates.append(np.sin(np.pi * x))
    # Power-modified cosine
    candidates.append(np.sin(np.pi * x) ** 1.5)
    candidates.append(np.sin(np.pi * x) ** 2.0)
    # Hamming
    candidates.append(0.54 - 0.46 * np.cos(2 * np.pi * x))
    # Hann
    candidates.append(0.5 - 0.5 * np.cos(2 * np.pi * x))
    # Tukey r=0.4
    r = 0.4
    tuk = np.ones(n)
    edge = max(1, int(r * n / 2))
    for i in range(edge):
        val = 0.5 * (1 + np.cos(np.pi * (2 * i / (r * n) - 1)))
        tuk[i] = val
        tuk[n - 1 - i] = val
    candidates.append(tuk)
    # Gaussian
    for sig in [0.25, 0.35, 0.45]:
        candidates.append(np.exp(-((x - 0.5) / sig) ** 2))
    # Spline-like (sum of shifted cosines)
    f = np.sin(np.pi * x) + 0.3 * np.sin(2 * np.pi * x * 0.7 + 0.5)
    candidates.append(np.maximum(f, 0))

    best_c = None
    best_sc = float('inf')
    for c in candidates:
        c = np.maximum(c, 0)
        s = c.sum()
        if s < 1e-15:
            continue
        c = c * norm / s
        sc = _fast_autocorr(c).max()
        if sc < best_sc:
            best_sc = sc
            best_c = c.copy()

    if best_c is None:
        best_c = np.ones(n) * norm / n
    return best_c


def propose_candidate(seed=42, budget_s=1800.0):
    """Search for the best non-negative sequence minimizing the autocorrelation objective."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    solver = _get_solver()
    rng = np.random.default_rng(seed)
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = _evaluate_safe(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def do_try(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = _evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # If no parent, use parametric init
    if best_seq is None:
        n_init = 1600
        best_seq = _parametric_init(n_init, rng)
        best_seq = best_seq / best_seq.sum()
        best_score = _evaluate_safe(best_seq)
        print(f"Parametric init: n={n_init}, score={best_score:.6f}")

    n_base = len(best_seq)

    # === Phase 1: LP Nesterov at base length ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.22, 400.0)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} ({t1:.0f}s) ===")
    if t1 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = _nesterov_lp(a_norm, t1, solver, verbose=True)
        do_try(a_out, f"lp_n{n_base}")

    # === Phase 2: LSE polish at base ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.06, 80.0)
    print(f"\n=== Phase 2: LSE n={len(best_seq)} ({t2:.0f}s) ===")
    if t2 > 10:
        n_c = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_c) / best_seq.sum()
        a_out, _ = _lse_gd(a_norm, t2, verbose=True)
        do_try(a_out, f"lse_n{n_c}")

    # === Phase 3: Expand to n_base+400 with LP ===
    n2 = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.25, 420.0)
    print(f"\n=== Phase 3: LP Nesterov n={n2} ({t3:.0f}s) ===")
    if t3 > 30:
        init2 = _interpolate_to_length(best_seq, n2)
        # Blend with parametric for diversity
        param2 = _parametric_init(n2, rng)
        blend2 = 0.6 * (init2 / init2.sum()) + 0.4 * (param2 / param2.sum())
        blend2 = blend2 / blend2.sum()
        a_norm = blend2 * np.sqrt(2.0 * n2) / blend2.sum()
        a_out, _ = _nesterov_lp(a_norm, t3, solver, verbose=True)
        do_try(a_out, f"lp_n{n2}")

        # LSE polish at n2
        remaining = time_budget - (time.monotonic() - t_global)
        t3b = min(remaining * 0.05, 60.0)
        if t3b > 10 and len(best_seq) == n2:
            a_norm = best_seq * np.sqrt(2.0 * n2) / best_seq.sum()
            a_out, _ = _lse_gd(a_norm, t3b, verbose=False)
            do_try(a_out, f"lse_n{n2}")

    # === Phase 4: Expand to n_base+800 ===
    n3 = n_base + 800
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.18, 300.0)
    print(f"\n=== Phase 4: LP Nesterov n={n3} ({t4:.0f}s) ===")
    if t4 > 30:
        # Interpolate from current best (which may be at n2)
        if len(best_seq) == n2:
            init3 = _interpolate_to_length(best_seq, n3)
        else:
            init3 = _interpolate_to_length(best_seq, n3)
        param3 = _parametric_init(n3, rng)
        blend3 = 0.7 * (init3 / init3.sum()) + 0.3 * (param3 / param3.sum())
        blend3 = blend3 / blend3.sum()
        a_norm = blend3 * np.sqrt(2.0 * n3) / blend3.sum()
        a_out, _ = _nesterov_lp(a_norm, t4, solver, verbose=True)
        do_try(a_out, f"lp_n{n3}")

    # === Phase 5: Peak shaving on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.12, 120.0)
    print(f"\n=== Phase 5: PeakShave n={n_best} ({t5:.0f}s) ===")
    if t5 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = _peak_shaving(a_norm, t5, seed=999, verbose=True)
        do_try(a_out, f"ps_n{n_best}")

    # === Phase 6: Final LP on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t6 = min(remaining * 0.25, 250.0)
    print(f"\n=== Phase 6: Final LP n={n_best} ({t6:.0f}s) ===")
    if t6 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = _nesterov_lp(a_norm, t6, solver, verbose=True)
        do_try(a_out, f"lpfin_n{n_best}")

    # === Phase 7: Final LSE + PeakShave ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t7 = min(remaining * 0.10, 80.0)
    if t7 > 10:
        print(f"\n=== Phase 7: LSE n={n_best} ({t7:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = _lse_gd(a_norm, t7, verbose=True)
        do_try(a_out, f"lsefin_n{n_best}")

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t8 = min(remaining * 0.20, 100.0)
    if t8 > 10:
        print(f"\n=== Phase 8: PeakShave n={n_best} ({t8:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = _peak_shaving(a_norm, t8, seed=31415, verbose=True)
        do_try(a_out, f"psfin_n{n_best}")

    # Final check
    if best_seq is not None:
        fs = _evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        fallback = np.ones(100)
        return [float(x) for x in fallback]

    return [float(x) for x in best_seq]
