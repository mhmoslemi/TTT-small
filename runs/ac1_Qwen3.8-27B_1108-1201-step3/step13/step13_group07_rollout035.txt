def fast_conv(a, b=None):
    """Compute full convolution a * b (or a * a if b is None)."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def internal_score(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_conv(a).max()

def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho = np.nonzero(v_sorted * np.arange(1, n + 1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-15:
        idx = np.argmax(result)
        result[idx] += diff
        result = np.maximum(result, 0)
        result = result / result.sum() * target_sum
    return result

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an improved multi-phase strategy:

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(arr):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def internal_score(a):
    """Compute 2*n*max_b assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho_candidates = np.nonzero(v_sorted * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho_candidates) == 0:
        return np.full(n, target_sum / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-15:
        idx = np.argmax(result)
        result[idx] += diff
        result = np.maximum(result, 0)
        rsum = result.sum()
        if rsum > 1e-300:
            result = result / rsum * target_sum
        else:
            result = np.full(n, target_sum / n)
    return result


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where T[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver, max_iter=200):
    """Solve: min M s.t. (a_fixed*b)_k <= M for all k, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, max_iter=max_iter)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, max_iter=300)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def zsg_multiplicative_weights(a0, time_limit, verbose=True):
    """
    Fast zero-sum game using multiplicative weights (entropy mirror descent).
    Row minimizes max_k (a*a)_k, column plays softmax weights on peaks.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    a_avg = a.copy()
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()
    conv_fft_size = 1 << (3 * n - 2).bit_length()

    eta_base = 0.15
    beta_start = 2.0
    beta_end = 50.0

    t = 0
    stall = 0
    stall_limit = 200000

    while time.monotonic() - t0 < time_limit:
        t += 1

        # Compute convolution a*a
        A = rfft(a, fft_size)
        b = irfft(A * A, fft_size)[:2 * n - 1]

        # Column strategy: softmax with cooling
        progress = min(1.0, t / max(1, int(time_limit * 100)))
        beta_t = beta_start + (beta_end - beta_start) * progress
        pb = beta_t * (b - b.max())
        p = np.exp(pb)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros(2 * n - 1)
            p[b.argmax()] = 1.0
        else:
            p = p / p_sum

        # Compute gradient: g_j = 2 * sum_k p_k * a_{k-j}
        a_rev = a[::-1]
        P = rfft(p, conv_fft_size)
        AR = rfft(a_rev, conv_fft_size)
        conv = irfft(P * AR, conv_fft_size)[:3 * n - 2]
        Pa = conv[n - 1:2 * n - 1]
        g = 2.0 * Pa

        # Adaptive step size
        g_norm = np.max(np.abs(g))
        if g_norm < 1e-15:
            g_norm = 1.0
        eta_t = eta_base / (np.sqrt(1.0 + 0.005 * t) * g_norm) * 2.0

        # Multiplicative weights update
        log_update = -eta_t * g
        log_update -= log_update.max()
        weights = np.exp(log_update)
        a_new = a * weights
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-300:
            break
        a_new = a_new / a_new_sum

        # Update time average
        a_avg = (1.0 - 1.0 / t) * a_avg + (1.0 / t) * a_new
        a = a_new

        # Check scores periodically
        if t % 100 == 0:
            score = internal_score(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                stall = 0
            score_avg = internal_score(a_avg)
            if score_avg < best_score:
                best_score = score_avg
                best_a = a_avg.copy()
                stall = 0

        stall += 1
        if stall > stall_limit:
            break

        if verbose and t % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ZSG-MW t={t} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated LP best-response for the zero-sum game."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0
    stall_count = 0
    stall_threshold = 8000

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = project_simplex(a_new)

        y = y_new
        a = a_new

        # Check scores
        if t % 5 == 0:
            score_a = internal_score(a)
            if score_a < best_score:
                if score_a < best_score - 1e-9:
                    stall_count = 0
                best_score = score_a
                best_a = a.copy()

            score_y = internal_score(y)
            if score_y < best_score:
                if score_y < best_score - 1e-9:
                    stall_count = 0
                best_score = score_y
                best_a = y.copy()

        # Time-averaged iterate
        avg_a += a
        n_avg += 1
        if n_avg >= 100 and n_avg % 200 == 0:
            avg_check = avg_a / n_avg
            score_avg = internal_score(avg_check)
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
                stall_count = 0

        stall_count += 1
        if stall_count > stall_threshold:
            break

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NL t={t} ({elapsed:.0f}s): best={best_score:.6f}, LP_M={M_val:.8f}")

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=123):
    """Coordinate-descent peak shaving with more aggressive moves."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    stall = 0
    fft_size = 1 << (2 * n - 1).bit_length()

    while time.monotonic() - t0 < time_limit:
        A = rfft(a, fft_size)
        b = irfft(A * A, fft_size)[:2 * n - 1]
        top_k = np.argsort(b)[::-1][:15]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]

            src_local = np.argmax(contrib)
            src = idx[src_local]

            peak_areas = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_areas.update(range(js2, je2 + 1))

            candidates = [j for j in range(n) if j not in peak_areas]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            if not candidates:
                continue
            candidates_sorted = sorted(candidates, key=lambda j: a[j])
            dst_pool = candidates_sorted[:max(1, len(candidates_sorted) // 4)]
            dst = dst_pool[rng.integers(len(dst_pool))]

            for frac in [0.001, 0.003, 0.005, 0.01, 0.015, 0.02, 0.03, 0.05]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                A_t = rfft(a_trial, fft_size)
                b_t = irfft(A_t * A_t, fft_size)[:2 * n - 1]
                score = 2.0 * n * b_t.max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.002, 0.03)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= -1e-15:
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    A_t = rfft(a_trial, fft_size)
                    b_t = irfft(A_t * A_t, fft_size)[:2 * n - 1]
                    score = 2.0 * n * b_t.max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > 800:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def adam_lse_optimize(a0, time_limit, T_start=50.0, T_end=0.02, n_stages=20, lr=0.004):
    """LogSumExp-smoothed optimization with Adam optimizer."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()
    conv_fft_size = 1 << (3 * n - 2).bit_length()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = internal_score(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_adam = 0

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        iters = max(20, min(150, int(remaining * 40 / max(1, n_stages - si))))

        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break

            a = softmax_fn(z)
            A = rfft(a, fft_size)
            b = irfft(A * A, fft_size)[:2 * n - 1]
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()

            # Gradient via FFT
            a_rev = a[::-1]
            W = rfft(w, conv_fft_size)
            AR = rfft(a_rev, conv_fft_size)
            conv = irfft(W * AR, conv_fft_size)[:3 * n - 2]
            grad_a = 2.0 * conv[n - 1:2 * n - 1]

            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)

            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)

            lr_t = lr / np.sqrt(1 + 0.005 * it)
            z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)

            if it % 10 == 0:
                a = softmax_fn(z)
                score = internal_score(a)
                if score < best_score:
                    best_score = score
                    best_a = a.copy()

        a = softmax_fn(z)
        score = internal_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_a, best_score

    Phase 4: Adam LSE polish
    Phase 5: Peak shaving (aggressive local search)

    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()

        if s > 0:
            best_seq = best_seq / s
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
