For the center: `(a*a)_{n-1} = sum_j a_j * a_{n-1-j}`. By AM-GM or other inequalities, this is maximized when `a_j = a_{n-1-j}` (symmetric), and the max is `sum a_j^2` (when a is symmetric).

The minimum of `(a*a)_{n-1}` subject to `sum(a)=1, a>=0` is... if we make `a` very asymmetric (e.g., all mass at one end), then `(a*a)_{n-1} ≈ 0`. But then other lags become large.

The total sum of autocorrelations is 1 (as I computed). So if we minimize one lag, we must increase others.

The optimal strategy is to "spread" the autocorrelation as evenly as possible across all 2n-1 lags. The theoretical minimum max is `1/(2n-1)` (if we could make all lags equal). For a non-negative sequence, this might not be achievable, but we should be able to get close.

If we achieve `max((a*a)_k) ≈ 1/(2n-1)`, the score would be `2n/(2n-1) ≈ 1 + 1/(2n-1)`. For n=1600, that's about 1.0003. We're at 1.5, so there's significant room for improvement in theory.

But in practice, the non-negativity constraint makes it much harder to achieve a flat autocorrelation. The "flat spectrum" for a non-negative sequence is the "Fejér" type construction, which gives a triangular autocorrelation.

I think the practical limit for non-negative sequences is around 1.4-1.5 for large n, which is consistent with our current result.

OK, let me stop theorizing and just write the code. I'll focus on:

1. Making the LP-based game iteration more efficient
2. Adding a "KKT projection" step that uses the structure of the KKT conditions
3. Multi-n exploration
4. Better peak shaving

Here's my implementation:

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Aggressive spectral shaping initialization
2. Game-theoretic LP iteration with adaptive steps
3. KKT projection refinement (new)
4. Multi-length exploration
5. Peak shaving polish
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def internal_score(a):
    """Compute 2*n*max_b assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()


def evaluate_safe(arr):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def project_simplex(v, target_sum=1.0):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    v = v.copy()
    if np.all(v >= 0) and abs(v.sum() - target_sum) < 1e-12:
        return v
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho = np.nonzero(v_sorted * np.arange(1, n + 1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    s = result.sum()
    if s > 1e-15:
        result = result / s * target_sum
    else:
        result = np.ones(n) / n * target_sum
    return result


def spectral_shaping(a, n_iter=500, alpha=0.3):
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = internal_score(a)

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2

        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 50.0)
        scale[1:] = np.power(ratio, alpha / 2.0)

        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def get_lp_solver():
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build constraint: (a_fixed * b)_k <= M for all k
    # (a_fixed * b)_k = sum_j a_fixed[k-j] * b_j
    # This is a linear function of b.
    
    # Use the structure: the convolution matrix T has T[k,j] = a_fixed[k-j] if valid
    # For efficiency, build it as a banded matrix
    
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver, max_iter=500)
    except Exception:
        pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, max_iter=500)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    y = a.copy()
    
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    
    avg_a = np.zeros(n)
    n_avg = 0
    stall_count = 0
    stall_threshold = 8000
    
    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue
        
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s
        
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = project_simplex(a_new)
        
        y = y_new
        a = a_new
        
        score_a = internal_score(a)
        if score_a < best_score:
            stall_count = 0
            best_score = score_a
            best_a = a.copy()
        
        score_y = internal_score(y)
        if score_y < best_score:
            stall_count = 0
            best_score = score_y
            best_a = y.copy()
        
        avg_a += a
        n_avg += 1
        if n_avg >= 100 and n_avg % 200 == 0:
            avg_check = avg_a / n_avg
            score_avg = internal_score(avg_check)
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
                stall_count = 0
        
        stall_count += 1
        if stall_count > stall_threshold:
            break
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NL t={t} ({elapsed:.0f}s): best={best_score:.6f}, LP_M={M_val:.8f}")
    
    return best_a, best_score


def kkt_refine(a0, time_limit, m_active=20, verbose=True):
    """
    Refine using KKT conditions.
    At the optimum with active peaks K:
      2 * sum_{k in K} lambda_k * a_{k-j} = mu for all j
    This means the "weighted cross-correlation" is flat.
    We use this to guide updates.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    
    # Iteratively:
    # 1. Identify active peaks
    # 2. Compute the "KKT gradient" (how non-flat the weighted correlation is)
    # 3. Adjust a to make it flatter
    
    lr = 0.01
    decay = 0.999
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(a)
        
        # Active peaks: top m by value
        top_k = np.argsort(b)[::-1][:m_active]
        
        # Softmax weights on active peaks
        b_active = b[top_k]
        T_smooth = 50.0
        w_active = softmax_fn(T_smooth * b_active)
        
        # Full weight vector
        w = np.zeros(2 * n - 1)
        w[top_k] = w_active
        
        # Compute the "KKT residual": for each j, compute sum_k w_k * a_{k-j}
        # This is the cross-correlation of w and a
        conv_len = n + (2 * n - 1) - 1
        fft_size = 1 << (conv_len - 1).bit_length()
        A_rev = rfft(a[::-1].copy(), fft_size)
        W = rfft(w, fft_size)
        conv = irfft(A_rev * W, fft_size)
        corr = conv[n - 1: n - 1 + n]  # cross-correlation at lags 0..n-1
        
        # The KKT condition says corr should be constant (= mu/2)
        # The "violation" is how non-constant corr is
        corr_mean = corr.mean()
        residual = corr - corr_mean
        
        # Gradient of the "KKT violation" w.r.t. a:
        # d/d a_j [sum_k w_k * a_{k-l}] for each l
        # This is like a "deconvolution" step
        # Simple approach: move a in the direction that makes corr flatter
        # grad_j = sum_l residual_l * (d/d a_j)(sum_k w_k * a_{k-l})
        #        = sum_l residual_l * w_{j+l} (approximately)
        
        # Actually, let's use a simpler approach:
        # The gradient of the smoothed objective is:
        # grad_j = 2 * sum_k w_k * a_{k-j} = 2 * corr_j (approximately)
        # We want to decrease the peaks of the autocorrelation
        # So we move in the direction: a_j -= lr * grad_j
        
        # Actually, the subgradient of max_k (a*a)_k is:
        # sg_j = 2 * sum_k w_k * a_{k-j}
        # where w_k are weights on the active peaks
        
        # This is exactly 2 * corr (up to the index convention)
        # Let me recompute: (a*a)_k = sum_j a_j * a_{k-j}
        # d(a*a)_k / d a_j = 2 * a_{k-j} (with boundary conditions)
        # subgrad_j = sum_k w_k * 2 * a_{k-j}
        
        # So subgrad = 2 * corr (where corr_j = sum_k w_k * a_{k-j})
        # My corr computation above should give this.
        
        sg = 2.0 * corr
        
        # Normalize the subgradient (so that 1^T sg = 0 for the projection to work well)
        sg = sg - sg.mean()
        
        # Update
        a_new = a - lr * sg
        a_new = project_simplex(a_new)
        
        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            a_new = a - lr * 0.1 * sg
            a_new = project_simplex(a_new)
            score = internal_score(a_new)
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
                a = a_new
            lr *= 0.95
        
        lr = max(lr * decay, 1e-6)
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      KKT it={it} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score
```
