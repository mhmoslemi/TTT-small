**Approach: Direct optimization via projected gradient with multiple peaks**

The objective is `F(a) = 2n * max_j b_j / (sum a)^2` where `b = conv(a,a)`.

If we fix the normalization `sum(a) = 1`, this becomes `2n * max_j b_j`.

The subgradient of `max_j b_j` with respect to `a[i]` is:
`∂/∂a[i] b_j = a[j-i] + a[i-j]` (for valid indices)

So the subgradient of the objective is `2n * (a[j*-i] + a[i-j*])` where j* is (one of) the argmax of b.

A projected subgradient descent would be:
1. Find j* = argmax b
2. Compute the subgradient: for each i, `g[i] = a[j*-i] + a[i-j*]` (with appropriate bounds)
3. Update: `a_new = max(0, a - lr * g)`, then renormalize

But this only uses one peak at a time. A better approach is to use all peaks that are close to the maximum (the "active set").

**Better approach: Use the "all near-max" peaks**

If multiple peaks are near the maximum, we should subtract contributions from all of them. The "smoothed" subgradient using softmax weighting:
`g[i] = sum_j w_j * (a[j-i] + a[i-j])` where `w_j = exp(b_j/T) / sum_k exp(b_k/T)`.

def compute_lse_gradient(a, w):
    n = len(a)
    a_rev = a[::-1]
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)
    conv = irfft(W * AR, fft_size)
    return 2.0 * conv[n - 1:2 * n - 1]


def lse_gradient_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0

    T_start = max(0.5, best_score * 0.15)
    T_end = max(0.001, best_score * 0.003)

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress

        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break

        w = np.exp((b - max_b) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        grad = compute_lse_gradient(a, w)

        # Adaptive step size
        step = 0.8 / (np.max(np.abs(grad)) + 1e-15)
        step = min(step, 1.0 / (n * 0.1))

        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        a = a_new
        steps += 1

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and steps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE-GD step={steps} ({elapsed:.0f}s): T={T:.4f}, best={best_score:.8f}")

    return best_a, best_score


def direct_subgradient_step(a, n_top=5, lr=None):
    """Take a step in the direction that reduces the top peaks simultaneously."""
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()

    # Find top peaks
    top_k = np.argsort(b)[::-1][:n_top]
    # Weight by proximity to max
    weights = np.zeros(len(b))
    for k in top_k:
        weights[k] = np.exp(-(b.max() - b[k]) / max(0.01 * b.max(), 1e-10))
    weights = weights / weights.sum()

    # Gradient of weighted sum of peaks
    grad = compute_lse_gradient(a, weights)

    if lr is None:
        lr = 0.5 / (np.max(np.abs(grad)) + 1e-15)
        lr = min(lr, 1.0 / n)

    return a, grad, lr

        T = max_b * 0.05 * (1 - progress) + max_b * 0.001
        w = np.exp((b - max_b) / T)
        w = w / w.sum()
        grad = compute_lse_gradient(a, w)

        # Evading gradient (for step size)
        avg_grad = 0.9 * avg_grad + 0.1 * grad
        g_norm = np.max(np.abs(avg_grad)) + 1e-15
        lr = 1.0 / (g_norm * np.sqrt(steps + 1))
        lr = min(lr, 2.0 / n)

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        a = a_new
        steps += 1

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and steps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SubGrad step={steps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def multi_peak_shave(a0, time_limit, seed=42, verbose=True):
    """Peak shaving with simulated annealing and multi-destination moves."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_improve = 0
    n_stall = 0

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        T = best_score * (0.05 * (1 - elapsed_frac) + 0.001)

        b = fast_autocorr(a)
        max_b = b.max()
        top_k = np.argsort(b)[::-1][:min(10, 2 * n // 10 + 1)]

        improved = False
        for k in top_k:
            if b[k] < max_b * 0.95:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            if contrib.max() < 1e-18:
                continue

            src = idx[np.argmax(contrib)]

            # Find good destinations
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k[:5]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            non_peak = np.where(~peak_mask)[0]
            if len(non_peak) < 5:
                non_peak = np.where(np.arange(n) != src)[0]

            # Select destinations preferentially from low-value positions
            n_dest = min(4, len(non_peak))
            sorted_dst = non_peak[np.argsort(a[non_peak])][:max(n_dest * 2, 5)]
            selected_dst = sorted_dst[rng.choice(len(sorted_dst), min(n_dest, len(sorted_dst)), replace=False)]

            frac_total = rng.choice([0.03, 0.05, 0.1, 0.15, 0.2, 0.3])
            amt_total = a[src] * frac_total
            if amt_total < 1e-18:
                continue

            dst_vals = a[selected_dst] + 1e-12
            weights = 1.0 / dst_vals
            weights = weights / weights.sum()
            amounts = amt_total * weights

            a_trial = a.copy()
            a_trial[src] -= amt_total
            for di, dst in enumerate(selected_dst):
                a_trial[dst] += amounts[di]
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial * norm / s_t

            score = fast_autocorr(a_trial).max()

            if score < best_score:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                n_improve += 1
                break
            elif rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                a = a_trial
                improved = True
                break

        if not improved:
            n_stall += 1
            # Random pair swap
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-10:
                frac = rng.uniform(0.05, 0.2)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            n_improve += 1
                        elif rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                            a = a_trial

            if n_stall > 100:
                idx_pert = rng.choice(n, size=max(2, n // 80), replace=False)
                a_trial = a.copy()
                for idx in idx_pert:
                    amt = a[idx] * rng.uniform(0.1, 0.5)
                    j = rng.integers(n)
                    a_trial[idx] -= amt
                    a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score * 1.02:
                        a = a_trial
                        if score < best_score:
                            best_score = score
                            best_a = a.copy()
                n_stall = 0

        if verbose and n_improve > 0 and n_improve % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      PeakShave improve={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def gaussian_smooth(a, sigma):
    n = len(a)
    if sigma < 0.5:
        return a.copy()
    kernel_size = int(6 * sigma) | 1
    x = np.arange(kernel_size) - kernel_size // 2
    kernel = np.exp(-0.5 * (x / sigma) ** 2)
    kernel = kernel / kernel.sum()
    padded = np.concatenate([np.zeros(kernel_size), a, np.zeros(kernel_size)])
    smoothed = np.convolve(padded, kernel, mode='valid')
    return smoothed[:n]


def make_init_variants(parent, n_target, seed=42):
    """Create multiple initialization variants."""
    rng = np.random.default_rng(seed)
    variants = []

    # Base: interpolate parent to target length
    if len(parent) != n_target:
        base = interpolate_sequence(parent, n_target)
    else:
        base = parent / parent.sum()

    # Variant 1: Smoothed
    smooth_sigma = max(2.0, n_target * 0.02)
    v = gaussian_smooth(base, smooth_sigma)
    v = v / v.sum()
    variants.append(v)

    # Variant 2: Asymmetric (linear ramp weighted)
    ramp = np.linspace(0.3, 1.7, n_target)
    v = base * ramp
    v = v / v.sum()
    variants.append(v)

    # Variant 3: Asymmetric (exponential decay)
    decay = np.exp(-np.linspace(0, 2.0, n_target))
    v = base * decay
    v = v / v.sum()
    variants.append(v)

    # Variant 4: Reverse exponential
    decay_r = np.exp(-np.linspace(0, 2.0, n_target)[::-1])
    v = base * decay_r
    v = v / v.sum()
    variants.append(v)

    # Variant 5: Random perturbation
    pert = rng.normal(0, 0.1, n_target)
    v = base * (1 + pert)
    v = np.maximum(v, 0)
    if v.sum() > 1e-10:
        v = v / v.sum()
    variants.append(v)

    # Variant 6: Bump shifted left
    center_shift = int(n_target * 0.3)
    x = np.linspace(0, 1, n_target)
    bump = np.exp(-((x - center_shift / n_target) ** 2) / (2 * 0.15 ** 2))
    v = bump / bump.sum()
    variants.append(v)

    # Variant 7: Bump shifted right
    center_shift = int(n_target * 0.7)
    bump = np.exp(-((x - center_shift / n_target) ** 2) / (2 * 0.15 ** 2))
    v = bump / bump.sum()
    variants.append(v)

    return variants


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1680.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
    except Exception:
        pass

    if parent is None:
        n_init = 1600
        x = np.linspace(0, 1, n_init)
        parent = np.sin(np.pi * x) ** 1.5
        parent = parent / parent.sum()
        print(f"Using default init: n={n_init}")

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 1: Evaluate initializations at different lengths
    print("\n=== Phase 1: Initialization ===")
    n_parent = len(parent)
    candidate_lengths = [n_parent, max(800, n_parent - 200), n_parent + 200, n_parent + 400]
    candidate_lengths = [n for n in candidate_lengths if 400 <= n <= 3000]

    inits = {}
    for n_target in candidate_lengths:
        variants = make_init_variants(parent, n_target, seed=seed)
        best_v = None
        best_v_score = float('inf')
        for i, v in enumerate(variants):
            sc = evaluate_local(v)
            if sc < best_v_score:
                best_v_score = sc
                best_v = v
        if best_v is not None:
            inits[n_target] = (best_v, best_v_score)
            print(f"  n={n_target}: best init score={best_v_score:.6f}")

    # Pick best initialization
    if inits:
        best_n = min(inits.keys(), key=lambda x: inits[x][1])
        best_seq = inits[best_n][0].copy()
        best_score = evaluate_safe(best_seq)
        print(f"  Best init: n={best_n}, score={best_score:.6f}")
    else:
        best_seq = parent.copy()
        best_score = evaluate_safe(best_seq)

    n_current = len(best_seq)

    # Phase 2: LP Nesterov at best length
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LP Nesterov n={n_current} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.30, 500.0)
    if t2 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_current) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t2, solver, verbose=True, restart_threshold=250)
        do_try_update(a_out, f"lp_n{n_current}")

    # Phase 3: LSE gradient
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LSE-GD n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.10, 120.0)
    if t3 > 10:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lse_gradient_descent(a_norm, t3, verbose=True)
        do_try_update(a_out, f"lse_n{n_cur}")

    # Phase 4: Subgradient descent
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: SubGrad n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.08, 100.0)
    if t4 > 10:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = subgradient_descent(a_norm, t4, verbose=True)
        do_try_update(a_out, f"subgrad_n{n_cur}")

    # Phase 5: Peak shaving
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: PeakShave n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.10, 100.0)
    if t5 > 10:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = multi_peak_shave(a_norm, t5, seed=999, verbose=True)
        do_try_update(a_out, f"peaks_n{n_cur}")

    # Phase 6: Try another length
    other_lengths = [n for n in candidate_lengths if n != n_current]
    if other_lengths:
        # Pick the next best length
        best_other_n = min(other_lengths, key=lambda x: inits.get(x, (None, float('inf')))[1])
        remaining = time_budget - (time.monotonic() - t_global)
        print(f"\n=== Phase 6: LP Nesterov n={best_other_n} (budget={remaining:.0f}s) ===")
        t6 = min(remaining * 0.20, 300.0)
        if t6 > 30:
            init_long = interpolate_sequence(best_seq, best_other_n)
            a_norm = init_long * np.sqrt(2.0 * best_other_n) / init_long.sum()
            a_out, _ = nesterov_lp_optimize(a_norm, t6, solver, verbose=True, restart_threshold=200)
            do_try_update(a_out, f"lp_n{best_other_n}")

            # Quick LSE polish at new length
            remaining = time_budget - (time.monotonic() - t_global)
            t6b = min(remaining * 0.05, 60.0)
            if t6b > 10:
                n_cur = len(best_seq)
                a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
                a_out, _ = lse_gradient_descent(a_norm, t6b, verbose=False)
                do_try_update(a_out, f"lse_n{n_cur}")

    # Phase 7: Second LP round on best (with different restart threshold)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: LP Nesterov (2nd) n={n_best} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.18, 250.0)
    if t7 > 20:
        # Add small perturbation to escape potential local minimum
        pert = rng.normal(0, 0.01 * best_seq.mean(), n_best)
        a_pert = np.maximum(best_seq + pert, 0)
        a_pert = a_pert / a_pert.sum()
        a_norm = a_pert * np.sqrt(2.0 * n_best) / a_pert.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t7, solver, verbose=True, restart_threshold=150)
        do_try_update(a_out, f"lp2_n{n_best}")

    # Phase 8: Final peak shaving
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final PeakShave n={n_best} (budget={remaining:.0f}s) ===")
    t8 = min(remaining * 0.15, 150.0)
    if t8 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = multi_peak_shave(a_norm, t8, seed=31415, verbose=True)
        do_try_update(a_out, f"peaks_final_n{n_best}")

    # Phase 9: Final LSE polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final LSE n={n_best} (budget={remaining:.0f}s) ===")
    t9 = min(remaining * 0.12, 100.0)
    if t9 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_gradient_descent(a_norm, t9, verbose=True)
        do_try_update(a_out, f"lse_final_n{n_best}")

    # Phase 10: Final short LP
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t10 = min(remaining * 0.25, 200.0)
    if t10 > 15:
        print(f"\n=== Phase 10: Final LP n={n_best} (budget={t10:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t10, solver, verbose=True, restart_threshold=100)
        do_try_update(a_out, f"lp_final_n{n_best}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
