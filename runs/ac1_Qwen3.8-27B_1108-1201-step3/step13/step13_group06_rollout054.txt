        for k in range(0, 2 * n - 1, max(1, (2 * n - 1) // 30)):
            js = max(0, k - (n - 1))
            je = min(n - 1, k)
            if js <= je:
                js_arr = np.arange(js, je + 1)
                row = np.zeros(n, dtype=np.float64)
                row[js_arr] = y[k - js_arr]
                constraints.append(row @ b_var <= M_var)
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solved = False
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            solved = True
    except Exception:
        pass
    if not solved:
        for alt in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue
    
    if not solved:
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None, None
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else 1.0/n
    return b_val, M_val


def get_active_set(y, init_count=40):
    """Initialize active set from top peaks of autocorrelation of y."""
    n = len(y)
    b = fast_autocorr(y)
    top_idx = np.argsort(b)[::-1][:init_count]
    # Always include some spread-out positions
    spread = np.linspace(0, 2*n-2, init_count//2).astype(int)
    combined = set(list(top_idx) + list(spread))
    return list(combined)[:init_count + 10]


def add_violations(y, b_star, M_val, active_set, max_constr=250, tol=1e-7):
    """Check for violated constraints and add them to active set."""
    conv_val = fast_conv(y, b_star)
    violated = np.where(conv_val > M_val + tol)[0]
    if len(violated) == 0:
        return active_set
    
    overshoot = conv_val[violated] - M_val
    order = np.argsort(overshoot)[::-1]
    new_idx = violated[order]
    
    active_set_set = set(active_set)
    for idx in new_idx:
        if len(active_set_set) >= max_constr:
            break
        active_set_set.add(int(idx))
    
    return list(active_set_set)


def nesterov_lp_optimize(a0, time_limit, solver, max_iters=50000):
    """
    Nesterov-accelerated LP zero-sum game with constraint generation.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    y = a.copy()
    x = a.copy()  # Primal iterate (average)
    S = 0.0
    
    active_set = get_active_set(y)
    warm_b = None
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    last_check = 0
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP for best response
        b_star, M_val = lp_solve(y, active_set, solver, warm_b)
        if b_star is None:
            warm_b = None
            continue
        warm_b = b_star.copy()
        
        # Update active set every 10 iterations
        if t % 10 == 0:
            active_set = add_violations(y, b_star, M_val, active_set)
        
        # Nesterov update
        S_new = S + t
        x_new = (S / S_new) * x + (t / S_new) * b_star
        x_new = np.maximum(x_new, 0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = b_star.copy()
        else:
            x_new = x_new / sx
        
        # y is a "lookahead" point (Nesterov)
        y_new = x_new.copy()
        
        S = S_new
        x = x_new
        y = y_new
        
        # Check improvement
        if t % 5 == 0:
            score_x = 2.0 * n * fast_autocorr(x).max()
            if score_x < best_score:
                best_score = score_x
                best_a = x.copy()
                stall = 0
            else:
                stall += 1
            
            score_b = 2.0 * n * fast_autocorr(b_star).max()
            if score_b < best_score:
                best_score = score_b
                best_a = b_star.copy()
                stall = 0
        
        # Restart if stalled
        if stall > 300 and t > 50:
            x = best_a.copy()
            y = best_a.copy()
            S = 0
            warm_b = None
            active_set = get_active_set(y)
            stall = 0
        
        if t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}, |active|={len(active_set)}, M={M_val:.8f}")
    
    return best_a, best_score


def lse_grad_obj(a, T, topk):
    """
    Compute top-k log-sum-exp objective and gradient.
    Objective: logsumexp(T * b[0:topk_sorted]) / T
    Gradient wrt a.
    """
    n = len(a)
    b = fast_autocorr(a)
    k_use = min(topk, len(b))
    top_idx = np.argsort(b)[::-1][:k_use]
    top_vals = b[top_idx]
    
    # LSE value
    max_tv = top_vals.max()
    exp_vals = np.exp(T * (top_vals - max_tv))
    lse_val = (max_tv + np.log(exp_vals.sum()) / T)  # Wait, this isn't right
    
    # Actually: logsumexp(T * top_vals) / T
    # = (T * max_tv + log(sum(exp(T*(top_vals - max_tv)))))/T
    # = max_tv + log(sum(exp(T*(top_vals - max_tv))))/T
    lse_val = max_tv + np.log(exp_vals.sum()) / T
    
    # Weights
    w = exp_vals / exp_vals.sum()
    
    # Gradient: grad[j] = (1/T) * sum_k w_k * T * (a[k-j] + a[j-k])
    #           = sum_k w_k * (a[k-j] + a[j-k])
    grad = np.zeros(n)
    for idx in range(k_use):
        K = int(top_idx[idx])
        wk = w[idx]
        js = max(0, K - (n - 1))
        je = min(n - 1, K)
        if js <= je:
            idx_range = np.arange(js, je + 1)
            # a[K - idx_range] + a[idx_range - K] (where valid)
            # For j in [js, je]: a[K-j] + a[j-K] (second term only if j-K >= 0)
            contrib = np.zeros(n)
            contrib[idx_range] = a[K - idx_range]  # a[K-j] for j in range
            # Second term: a[j-K] for j > K
            js2 = max(K + 1, 0)
            je2 = min(n - 1, K + (n - 1))
            if js2 <= je2:
                idx_range2 = np.arange(js2, je2 + 1)
                contrib[idx_range2] += a[idx_range2 - K]
            grad += wk * contrib
    
    return lse_val, grad / T


def adam_lse_optimize(a0, time_limit, T_start=50.0, T_end=0.3, n_stages=20, topk=30):
    """Adam-optimized top-k LSE with multi-stage annealing."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        base_lr = 2e-3 * min(1.0, 200.0 / n)
        lr = base_lr * (0.9 ** si)
        
        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.8, 0.999, 1e-8
        
        max_it = max(10, int(min(remaining, 90.0) * 8))
        
        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 2:
                break
            
            a = softmax_fn(z)
            lse_val, grad_a = lse_grad_obj(a, stage_T, topk)
            
            # Convert to z-gradient
            S_dot = np.dot(grad_a, a)
            grad_z = a * (grad_a - S_dot)
            
            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            if (it + 1) % 5 == 0:
                a_check = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a_check).max()
                if score < best_score:
                    best_score = score
                    best_a = a_check.copy()
        
        a_stage = softmax_fn(z)
        score_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()
        
        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {si}/{n_stages} (T={stage_T:.2f}, {elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score

        threshold = bmax * 0.995

        danger = np.zeros(n, dtype=bool)
