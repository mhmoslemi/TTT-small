def project_nonneg_sum(a, c):
    """Euclidean projection onto {x >= 0, sum(x) = c}."""
    n = len(a)
    if c <= 0:
        return np.zeros(n)
    lo = a.min() - c
    hi = a.max()
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.maximum(a - mid, 0.0))
        if s > c:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0

def compute_lse_gradient(a, T):
    """
    Compute gradient of LSE_T(conv(a,a)) w.r.t. a.
    Returns (grad, conv_values, weights).
    Uses FFT cross-correlation for O(n log n) complexity.
    """
    n = len(a)
    c = fast_autocorr(a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    w = exp_c / exp_c.sum()
    
    # Cross-correlation: grad_i = 2 * sum_j a_j * w_{j+i}
    # Via FFT: need padding >= n + (2n-1) - 1 = 3n-2
    N = 1 << (3 * n - 2 - 1).bit_length()
    A = rfft(a, N)
    W = rfft(w, N)
    cross = irfft(A * np.conj(W), N)[:n]
    
    grad = 2.0 * cross
    return grad, c, w

def lse_adam_optimize(a0, time_limit, verbose=True):

    LSE-smoothed Adam gradient descent with temperature annealing.
    Minimizes max(conv(a,a)) subject to a >= 0, sum(a) = sqrt(2n).

    norm = np.sqrt(2.0 * n)

def build_conv_matrix(a):
    """Build the convolution matrix T where T @ b = conv(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def lp_best_response_screened(a, M, solver, active_set=None, tol_frac=0.05):
    """
    LP best response: max sum(b) s.t. conv(a,b) <= M, b >= 0.
    Uses constraint screening for speed.
    """
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    conv_current = T @ a
    
    # Determine active set
    if active_set is None:
        threshold = M * (1.0 - tol_frac)
        active_set = np.where(conv_current > threshold)[0]
        if len(active_set) < 5:
            active_set = np.where(conv_current > M * 0.3)[0]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T[active_set, :] @ b_var <= M]
    # Add a constraint to prevent b from being too concentrated
    # (sum(b) <= 2 * sqrt(2n) as a sanity bound)
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, active_set
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, active_set
    
    if b_var.value is None:
        return None, active_set
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None, active_set
    
    # Check if dropped constraints are violated
    if len(active_set) < size:
        full_conv = T @ b_val
        violated = np.where(full_conv > M * (1.0 + 1e-6))[0]
        if len(violated) > 0:
            new_active = np.union1d(active_set, violated)
            if len(new_active) >= 0.95 * size:
                # Fall back to full LP
                b_var2 = cp.Variable(n, nonneg=True)
                prob2 = cp.Problem(cp.Maximize(cp.sum(b_var2)), [T @ b_var2 <= M])
                try:
                    prob2.solve(solver=solver, verbose=False)
                except Exception:
                    return b_val, active_set
                if prob2.status in ('optimal', 'optimal_inaccurate') and prob2.value is not None:
                    b_val = np.maximum(np.asarray(b_var2.value, dtype=np.float64), 0.0)
                    active_set = np.arange(size)
            else:
                # Re-solve with expanded active set
                b_var2 = cp.Variable(n, nonneg=True)
                prob2 = cp.Problem(cp.Maximize(cp.sum(b_var2)), [T[new_active, :] @ b_var2 <= M])
                try:
                    prob2.solve(solver=solver, verbose=False)
                except Exception:
                    return b_val, active_set
                if prob2.status in ('optimal', 'optimal_inaccurate') and prob2.value is not None:
                    b_val = np.maximum(np.asarray(b_var2.value, dtype=np.float64), 0.0)
                active_set = new_active
    
    return b_val, active_set

        M_curr = conv_y.max()

        if b_sum < 1e-15:

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast LSE (log-sum-exp) smoothed Adam gradient descent with FFT-based O(n log n) gradients
2. Geometric temperature annealing for exploration-to-exploitation transition
3. LP best-response (cvxpy) for precise large-step improvements with active-set screening
4. Nesterov-accelerated LP iteration
5. Multi-length exploration (n from 800 to 3000) with interpolation transfer
6. Noise-injected restarts to escape local optima
Key innovations: vectorized cross-correlation gradient, adaptive step sizes,

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print(f"  LP solver (fallback): ECOS")
    return _global_solver

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def project_nonneg_sum(a, c):
    n = len(a)
    if c <= 0:
        return np.zeros(n)
    lo = a.min() - c
    hi = a.max()
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.maximum(a - mid, 0.0))
        if s > c:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.maximum(a - tau, 0.0)

def compute_lse_gradient(a, T):
    n = len(a)
    c = fast_autocorr(a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    w = exp_c / exp_c.sum()
    N = 1 << (3 * n - 2 - 1).bit_length()
    A = rfft(a, N)
    W = rfft(w, N)
    cross = irfft(A * np.conj(W), N)[:n]
    grad = 2.0 * cross
    return grad, c, w

def lse_adam_optimize(a0, time_limit, verbose=True, restarts=2):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    for restart in range(restarts + 1):
        if restart > 0:
            # Add noise to escape local optimum
            noise = rng.normal(0, 0.05 * norm / n, n)
            a = np.maximum(a + noise, 0.0)
            a = a * norm / a.sum()

        m = np.zeros(n)
        v = np.zeros(n)
        beta1 = 0.85
        beta2 = 0.999
        eps = 1e-10
        step_count = 0

        n_stages = 14
        temps = np.geomspace(60.0, 0.15, n_stages)

        for si, T in enumerate(temps):
            if time.monotonic() - t0 > time_limit:
                break
            remaining = time_limit - (time.monotonic() - t0)
            if remaining < 0.5:
                break
            frac = (si + 1) / n_stages
            n_steps = max(40, int(remaining * (2.0 + 4.0 * frac)))

            for it in range(n_steps):
                if time.monotonic() - t0 > time_limit:
                    break
                grad, c, w = compute_lse_gradient(a, T)
                step_count += 1
                m = beta1 * m + (1.0 - beta1) * grad
                v = beta2 * v + (1.0 - beta2) * grad * grad
                m_hat = m / (1.0 - beta1 ** step_count)
                v_hat = v / (1.0 - beta2 ** step_count)
                lr = 0.8 * norm / np.sqrt(n) / (1.0 + 0.5 * it / max(n_steps, 1))
                a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
                a_new = project_nonneg_sum(a_new, norm)
                a = a_new
                score = c.max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()

        if verbose and restarts <= 2:
            elapsed = time.monotonic() - t0
            print(f"    LSE restart={restart} ({elapsed:.1f}s): best={best_score:.8f}")

    return best_a, best_score

def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def lp_best_response(a, M, solver, active_set=None, tol_frac=0.05):
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    conv_current = T @ a
    if active_set is None:
        threshold = M * (1.0 - tol_frac)
        active_set = np.where(conv_current > threshold)[0]
        if len(active_set) < 5:
            active_set = np.where(conv_current > M * 0.3)[0]
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T[active_set, :] @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, active_set
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, active_set
    if b_var.value is None:
        return None, active_set
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None, active_set
    if len(active_set) < size:
        full_conv = T @ b_val
        violated = np.where(full_conv > M * (1.0 + 1e-6))[0]
        if len(violated) > 0:
            new_active = np.union1d(active_set, violated)
            if len(new_active) >= 0.95 * size:
                b_var2 = cp.Variable(n, nonneg=True)
                prob2 = cp.Problem(cp.Maximize(cp.sum(b_var2)), [T @ b_var2 <= M])
                try:
                    prob2.solve(solver=solver, verbose=False)
                except Exception:
                    return b_val, active_set
                if prob2.status in ('optimal', 'optimal_inaccurate') and b_var2.value is not None:
                    b_val = np.maximum(np.asarray(b_var2.value, dtype=np.float64), 0.0)
                    active_set = np.arange(size)
            else:
                b_var2 = cp.Variable(n, nonneg=True)
                prob2 = cp.Problem(cp.Maximize(cp.sum(b_var2)), [T[new_active, :] @ b_var2 <= M])
                try:
                    prob2.solve(solver=solver, verbose=False)
                except Exception:
                    return b_val, active_set
                if prob2.status in ('optimal', 'optimal_inaccurate') and b_var2.value is not None:
                    b_val = np.maximum(np.asarray(b_var2.value, dtype=np.float64), 0.0)
                active_set = new_active
    return b_val, active_set

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, max_iters=150):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    active_set = None
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        conv_y = fast_autocorr(y)
        M_curr = conv_y.max()
        b, active_set = lp_best_response(y, M_curr, solver, active_set)
        if b is None:
            active_set = None
            b, active_set = lp_best_response(y, M_curr, solver, None, tol_frac=0.0)
            if b is None:
                continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        y = y_new
        a = a_new
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        avg_a += a
        n_avg += 1
        if n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={t} ({elapsed:.0f}s): best={best_score:.8f}")
    return best_a, best_score

def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-5.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-5.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 4
    elif choice == 5:
        x = np.linspace(0.0, 1.0, n)
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.5
    elif choice == 6:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.5) ** 2) * (1.0 + 0.3 * np.cos(20.0 * x))
    else:
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 + 0.2 * np.sin(2.0 * np.pi * x) ** 2
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

def subgradientfw_optimize(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    size = 2 * n - 1

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break
        conv_y = fast_autocorr(y)
        k_star = int(np.argmax(conv_y))
        grad = np.zeros(n)
        for i in range(n):
            v = 0.0
            j1 = k_star - i
            if 0 <= j1 < n:
                v += y[j1]
            j2 = i - k_star
            if 0 <= j2 < n:
                v += y[j2]
            grad[i] = v
        grad *= 2.0
        i_min = int(np.argmin(grad))
        v_vec = np.zeros(n)
        v_vec[i_min] = norm
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * v_vec
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        if t == 1:
            a_new = v_vec.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        y = y_new
        a = a_new
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        avg_a += a
        n_avg += 1
        if n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                sc = fast_autocorr(avg_check).max()
                if sc < best_score:
                    best_score = sc
                    best_a = avg_check.copy()
        if verbose and t % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW t={t} ({elapsed:.0f}s): best={best_score:.8f}")
    return best_a, best_score

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 2000
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LSE Adam at parent length (primary workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.30, 500.0)
    print(f"\n=== Phase 1: LSE Adam n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = lse_adam_optimize(a_norm, t1, verbose=True, restarts=2)
        try_update(a_out, f"lse_n{n_base}")

    # Phase 2: LP Nesterov at best length
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.25, 350.0)
    print(f"\n=== Phase 2: LP Nesterov n={n_best} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t2, solver, verbose=True)
        try_update(a_out, f"lp_n{n_best}")

    # Phase 3: LSE at alternative lengths
    n_alts = [n_base + 400, n_base - 300, n_base + 800]
    n_alts = [n for n in n_alts if 600 <= n <= 5000]
    for n_alt in n_alts:
        remaining = time_budget - (time.monotonic() - t_global)
        t3 = min(remaining * 0.12, 150.0)
        if t3 < 15:
            continue
        print(f"\n=== Phase 3: LSE n={n_alt} (budget={t3:.0f}s) ===")
        init_alt = interpolate_sequence(best_seq, n_alt)
        a_norm = init_alt * np.sqrt(2.0 * n_alt) / init_alt.sum()
        a_out, _ = lse_adam_optimize(a_norm, t3, verbose=False, restarts=1)
        try_update(a_out, f"lse_n{n_alt}")

    # Phase 4: FW subgradient (different dynamics, good for escaping LP traps)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 4: FW subgrad n={n_best} (budget={t4:.0f}s) ===")
    if t4 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = subgradientfw_optimize(a_norm, t4, verbose=True)
        try_update(a_out, f"fw_n{n_best}")

    # Phase 5: LP again on the improved sequence
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.10, 120.0)
    print(f"\n=== Phase 5: LP round 2 n={n_best} (budget={t5:.0f}s) ===")
    if t5 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t5, solver, verbose=True, max_iters=80)
        try_update(a_out, f"lp2_n{n_best}")

    # Phase 6: Final LSE polish (very low T for precision)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t6 = min(remaining * 0.10, 100.0)
    print(f"\n=== Phase 6: Final LSE polish n={n_best} (budget={t6:.0f}s) ===")
    if t6 > 5:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_adam_optimize(a_norm, t6, verbose=False, restarts=3)
        try_update(a_out, f"polish_n{n_best}")

    # Final check
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
