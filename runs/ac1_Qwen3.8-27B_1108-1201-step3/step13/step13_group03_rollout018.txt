import numpy as np
import time
from scipy.fft import rfft, irfft

# Helper functions (all top-level)

def _fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    return irfft(A * np.conj(A), fft_size)[:size]

def _eval(a):
    """Safely evaluate the sequence."""
    try:
        return float(evaluate_sequence([float(x) for x in a]))
    except Exception:
        return float('inf')

def _normalize(a, target_sum=1.0):
    """Normalize to non-negative with given sum."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * target_sum / len(a)
    return a * target_sum / s

def _peak_gradient(a, peaks, weights):
    """Compute gradient of weighted sum of peak autocorrelations."""
    n = len(a)
    grad = np.zeros(n)
    for k, w in zip(peaks, weights):
        k = int(k)
        # Contribution from a[k-i] for valid i
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        # a[k - i] for i in [js, je]
        # a[j] contributes to grad[i] where i = k - j, i.e., j = k - i
        # For the term sum_i a[i]*a[k-i]:
        #   d/di a[i] = a[k-i] (if 0 <= k-i < n)
        #   d/d(k-i) = a[i] (i.e., d/d a[j] where j = k-i, contrib = a[i] = a[k-j])
        # So grad[i] += a[k-i] + a[i-k] (both terms)
        
        for i in range(n):
            j1 = k - i
            j2 = i - k
            val = 0.0
            if 0 <= j1 < n:
                val += a[j1]
            if 0 <= j2 < n:
                val += a[j2]
            if val != 0:

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft


def _fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    return irfft(A * np.conj(A), fft_size)[:size]


def _eval_safe(a):
    """Safely evaluate the sequence using the provided evaluate_sequence."""
    try:
        return float(evaluate_sequence([float(x) for x in a]))
    except Exception:
        return float('inf')


def _normalize(a, target=1.0):
    """Clip to non-negative and scale to target sum."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * target / max(len(a), 1)
    return a * target / s


def _compute_multi_peak_grad(a, peak_lags, weights):
    """
    Compute gradient of sum_t weights[t] * r(k_t) w.r.t. a.
    r(k) = sum_i a[i]*a[i+k], so dr(k)/da[m] = a[m+k] + a[m-k] (with bounds).
    Returns the gradient (without the factor of 2, caller handles it).
    """
    n = len(a)
    grad = np.zeros(n)
    for kk, w in zip(peak_lags, weights):
        kk = int(kk)
        if kk == 0:
            grad += w * 2.0 * a
        else:
            # a[m+kk] for m = 0..n-1-kk
            s1 = np.zeros(n)
            s1[:n - kk] = a[kk:]
            # a[m-kk] for m = kk..n-1
            s2 = np.zeros(n)
            s2[kk:] = a[:n - kk]
            grad += w * (s1 + s2)
    return 2.0 * grad


def _mirror_descent_nesterov(a0, time_limit, top_k=8, T_init=0.5, T_final=0.05,
                              eta_init=0.3, eta_decay=0.9995, verbose=True):
    """
    Nesterov-accelerated mirror descent (multiplicative) on the simplex.
    Minimizes max of top-K autocorrelation peaks.
    """
    n = len(a0)
    a = _normalize(a0.copy())
    a_prev = a.copy()
    y = a.copy()
    best_a = a.copy()
    best_obj = 2.0 * n * _fast_autocorr(a).max()
    t0 = time.monotonic()
    t = 1

    n_iters = 0
    while time.monotonic() - t0 < time_limit:
        # Nesterov momentum
        if t > 1:
            momentum = (t - 1.0) / (t + 2.0)
            y = a + momentum * (a - a_prev)
            y = _normalize(y)
        else:
            y = a.copy()

        a_prev = a.copy()

        # Compute autocorrelation and find peaks
        ac = _fast_autocorr(y)
        peak_val = ac.max()

        # Track best (on y, which is the extrapolated point)
        obj_y = 2.0 * n * peak_val
        if obj_y < best_obj:
            best_obj = obj_y
            best_a = y.copy()

        # Top-K peaks
        k_use = min(top_k, len(ac))
        peak_indices = np.argsort(ac)[::-1][:k_use]
        peak_vals = ac[peak_indices]

        # Adaptive temperature
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_init + (T_final - T_init) * progress
        weights = np.exp(peak_vals / T)
        weights /= weights.sum()

        # Gradient
        grad = _compute_multi_peak_grad(y, peak_indices, weights)

        # Multiplicative update
        eta = eta_init * (eta_decay ** t)
        eta = max(eta, 0.005)
        # Clip for numerical stability
        grad_clipped = np.clip(grad, -5.0 / eta, 5.0 / eta)
        a = y * np.exp(-eta * grad_clipped)
        a = _normalize(a)

        t += 1
        n_iters += 1

        if verbose and n_iters % 2000 == 0:
            elapsed = time.monotonic() - t0
            obj_a = 2.0 * n * _fast_autocorr(a).max()
            if obj_a < best_obj:
                best_obj = obj_a
                best_a = a.copy()
            print(f"    MirrorDescent iter={n_iters} ({elapsed:.0f}s): "
                  f"obj_y={obj_y:.8f}, obj_a={obj_a:.8f}, best={best_obj:.8f}")

    # Final check
    obj_a = 2.0 * n * _fast_autocorr(a).max()
    if obj_a < best_obj:
        best_obj = obj_a
        best_a = a.copy()
    return best_a, best_obj


def _build_conv_constraint_matrix(a, n):
    """Build the constraint matrix T where (T @ b)[k] = conv(a,b)[k]."""
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def _lp_best_response(a, M, solver_name):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    import cvxpy as cp
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = _build_conv_constraint_matrix(a, n)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver_name, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def _lp_nesterov_accelerated(a0, time_limit, solver_name, verbose=True):
    """
    LP best-response with Nesterov acceleration and adaptive M.
    """
    n = len(a0)
    norm_target = np.sqrt(2.0 * n)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_target / n
    else:
        a = a * norm_target / s0

    best_a = a.copy()
    best_score = _fast_autocorr(a).max()
    y = a.copy()
    y_prev = a.copy()
    t0 = time.monotonic()
    t = 1

    while time.monotonic() - t0 < time_limit:
        M_curr = _fast_autocorr(y).max()
        if M_curr < 1e-14:
            break

        b = _lp_best_response(y, M_curr, solver_name)
        if b is None:
            t += 1
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            t += 1
            continue
        g = b * norm_target / b_sum

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_target / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm_target / s_a

        y_prev = y.copy()
        y = y_new
        a = a_new

        # Track best
        sc_y = _fast_autocorr(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        sc_a = _fast_autocorr(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        t += 1
        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M_curr:.8f}")

    return best_a, best_score


def _spectral_flatten(a, time_limit, alpha=0.3, n_iters=200, verbose=True):
    """
    Spectral domain peak reduction: iteratively flatten |A(w)|^2 to reduce
    autocorrelation peaks.
    """
    n = len(a)
    a = _normalize(a.copy())
    best_a = a.copy()
    best_obj = 2.0 * n * _fast_autocorr(a).max()
    t0 = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Full FFT for spectral manipulation
        fft_size = 1 << (2 * n - 1).bit_length()
        A = fft(a.astype(np.float64), fft_size)
        P = np.abs(A) ** 2

        # Compute target: geometric mean of power spectrum
        # (log-flattening)
        log_P = np.log(P + 1e-30)
        mean_log_P = np.mean(log_P)
        target_log_P = (1 - alpha) * log_P + alpha * mean_log_P
        target_P = np.exp(target_log_P)

        # Scale to match total energy
        energy_ratio = P.sum() / (target_P.sum() + 1e-30)
        target_P *= energy_ratio

        # New amplitude, keep phase
        new_amp = np.sqrt(target_P)
        phase = np.angle(A)
        A_new = new_amp * np.exp(1j * phase)

        # Inverse FFT
        a_new = np.real(ifft(A_new, fft_size))[:n]

        # Project onto non-negative simplex
        a_new = _normalize(a_new)

        # Check improvement
        obj = 2.0 * n * _fast_autocorr(a_new).max()
        if obj < best_obj - 1e-12:
            a = a_new
            best_obj = obj
            best_a = a_new.copy()
        else:
            # Try with smaller alpha
            a_new2 = a.copy()
            # Blend
            a_new2 = (1 - 0.5) * a + 0.5 * a_new
            a_new2 = _normalize(a_new2)
            obj2 = 2.0 * n * _fast_autocorr(a_new2).max()
            if obj2 < best_obj - 1e-12:
                a = a_new2
                best_obj = obj2
                best_a = a_new2.copy()

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SpectralFlatten iter={it} ({elapsed:.0f}s): best={best_obj:.8f}")

    return best_a, best_obj


def _local_search_pairwise(a, time_limit, verbose=True, rng=None):
    """

    Targets the top peaks specifically.
    """
    if rng is None:
        rng = np.random.default_rng(12345)
    n = len(a)
    a = _normalize(a.copy())
    best_a = a.copy()
    best_obj = 2.0 * n * _fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        ac = _fast_autocorr(a)
        top_k = np.argsort(ac)[::-1][:5]
        improved = False

        for k in top_k:
            k = int(k)
            # Identify contributing positions for this peak
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue

            # Source: position contributing most to this peak
            src = idx[np.argmax(contrib)]

            # Find target: position that contributes least to ALL top peaks
            peak_mask = np.zeros(n, dtype=float)
            for k2 in top_k:
                k2 = int(k2)
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                idx2 = np.arange(js2, je2 + 1)
                if len(idx2) > 0:
                    c2 = a[idx2] * a[k2 - idx2]
                    peak_mask[idx2] = np.maximum(peak_mask[idx2], c2)

            # Candidates: positions with low peak contribution and low current value
            candidates = np.where(peak_mask < 0.1 * peak_mask.max())[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            candidates = np.delete(candidates, src) if src in candidates else candidates
            if len(candidates) == 0:
                continue

            # Pick a few random candidates and the lowest-value ones
            n_cand = min(len(candidates), 20)
            low_val_cand = candidates[np.argsort(a[candidates])[:n_cand // 2]]
            rand_cand = rng.choice(candidates, size=min(n_cand // 2, len(candidates)), replace=False)
            all_cand = np.concatenate([low_val_cand, rand_cand])

            for dst in all_cand:
                # Binary search for optimal transfer amount
                lo, hi = 0.0, a[src] * 0.1
                best_amt = 0.0
                best_sc = best_obj
                for _ in range(8):
                    mid = (lo + hi) / 2
                    a_trial = a.copy()
                    a_trial[src] -= mid
                    a_trial[dst] += mid
                    if a_trial[src] < -1e-15:
                        hi = mid
                        continue
                    a_trial = _normalize(a_trial)
                    sc = _fast_autocorr(a_trial).max() * 2.0 * n
                    if sc < best_sc:
                        best_sc = sc
                        best_amt = mid
                        lo = mid
                    else:
                        hi = mid

                if best_amt > 1e-10 and best_sc < best_obj - 1e-12:
                    a[src] -= best_amt
                    a[dst] += best_amt
                    a = _normalize(a)
                    best_obj = best_sc
                    best_a = a.copy()
                    improved = True
                    break

            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape local minimum
            if stall > 100:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-8:
                    amt = a[i] * rng.uniform(0.01, 0.05)
                    a[i] -= amt
                    a[j] += amt
                    a = _normalize(a)
                    stall = 0
            if stall > 1000:
                break

        if verbose and (improved) and (int((time.monotonic() - t0)) % 10 == 0):
            elapsed = time.monotonic() - t0
            print(f"    LocalSearch ({elapsed:.0f}s): best={best_obj:.8f}")

    return best_a, best_obj


def _interpolate(a, new_n):
    """Interpolate sequence to new length, preserving shape."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss


def _get_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.3])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                print(f"  Using LP solver: {s}")
                return s
        except Exception:
            continue
    print("  Using LP solver: CLARABEL (fallback)")
    return cp.CLARABEL


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function combining mirror descent, LP best-response,
    spectral flattening, and local search for minimizing
    J(a) = 2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)

    solver_name = _get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # --- Initialization ---
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = _eval_safe(best_seq)
            print(f"Starting from height_sequence_1: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0, 1, n_base)
        best_seq = np.sin(np.pi * x) ** 2.0
        best_seq = np.maximum(best_seq, 0.0) / np.sum(best_seq)
        best_score = _eval_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    def _try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a.astype(np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = _eval_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    n_base = len(best_seq)

    # === Phase 1: Mirror Descent with Nesterov (fast) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.15, 200.0)
    if t1 > 20:
        print(f"\n=== Phase 1: Mirror Descent n={n_base} (budget={t1:.0f}s) ===")
        a_out, _ = _mirror_descent_nesterov(
            best_seq, t1, top_k=10, T_init=0.8, T_final=0.03,
            eta_init=0.5, eta_decay=0.9997, verbose=True)
        _try_update(a_out, f"mirror_n{n_base}")

    # === Phase 2: LP Best Response (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.30, 450.0)
    if t2 > 30:
        print(f"\n=== Phase 2: LP-Nesterov n={n_cur} (budget={t2:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _lp_nesterov_accelerated(a_norm, t2, solver_name, verbose=True)
        _try_update(a_out, f"lp_n{n_cur}")

    # === Phase 3: Spectral Flattening ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.10, 120.0)
    if t3 > 10:
        print(f"\n=== Phase 3: Spectral Flatten n={n_cur} (budget={t3:.0f}s) ===")
        a_out, _ = _spectral_flatten(best_seq, t3, alpha=0.3, n_iters=100, verbose=True)
        _try_update(a_out, f"spectral_n{n_cur}")

    # === Phase 4: Mirror Descent round 2 (refined) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t4 = min(remaining * 0.10, 100.0)
    if t4 > 15:
        print(f"\n=== Phase 4: Mirror Descent 2 n={n_cur} (budget={t4:.0f}s) ===")
        a_out, _ = _mirror_descent_nesterov(
            best_seq, t4, top_k=5, T_init=0.3, T_final=0.02,
            eta_init=0.3, eta_decay=0.9998, verbose=True)
        _try_update(a_out, f"mirror2_n{n_cur}")

    # === Phase 5: LP at longer length ===
    n_long = n_cur + 400
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 30:
        print(f"\n=== Phase 5: LP-Nesterov n={n_long} (budget={t5:.0f}s) ===")
        init_long = _interpolate(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = _lp_nesterov_accelerated(a_norm, t5, solver_name, verbose=True)
        _try_update(a_out, f"lp_long_n{n_long}")

    # === Phase 6: Mirror descent at longer length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    if n_cur != n_long:
        n_try = n_cur + 300
    else:
        n_try = n_cur + 300
    t6 = min(remaining * 0.08, 80.0)
    if t6 > 10:
        print(f"\n=== Phase 6: Mirror Descent n={n_try} (budget={t6:.0f}s) ===")
        init_try = _interpolate(best_seq, n_try)
        a_out, _ = _mirror_descent_nesterov(
            init_try, t6, top_k=8, T_init=0.5, T_final=0.05,
            eta_init=0.4, eta_decay=0.9996, verbose=True)
        _try_update(a_out, f"mirror_long_n{n_try}")

    # === Phase 7: Local search (pairwise) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t7 = min(remaining * 0.15, 180.0)
    if t7 > 15:
        print(f"\n=== Phase 7: Local Search n={n_cur} (budget={t7:.0f}s) ===")
        a_out, _ = _local_search_pairwise(best_seq, t7, verbose=True, rng=rng)
        _try_update(a_out, f"local_n{n_cur}")

    # === Phase 8: Spectral + mirror combined ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t8 = min(remaining * 0.08, 80.0)
    if t8 > 10:
        print(f"\n=== Phase 8: Spectral+Mirror n={n_cur} (budget={t8:.0f}s) ===")
        a_sp, _ = _spectral_flatten(best_seq, t8 * 0.4, alpha=0.2, n_iters=50, verbose=False)
        a_out, _ = _mirror_descent_nesterov(
            a_sp, t8 * 0.6, top_k=5, T_init=0.2, T_final=0.01,
            eta_init=0.2, eta_decay=0.9999, verbose=False)
        _try_update(a_out, f"spec_mirror_n{n_cur}")

    # === Phase 9: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t9 = min(remaining * 0.20, 200.0)
    if t9 > 20:
        print(f"\n=== Phase 9: Final LP n={n_cur} (budget={t9:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = _lp_nesterov_accelerated(a_norm, t9, solver_name, verbose=True)
        _try_update(a_out, f"lp_final_n{n_cur}")

    # === Phase 10: Final local search ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t10 = min(remaining * 0.30, 120.0)
    if t10 > 10:
        print(f"\n=== Phase 10: Final Local Search n={n_cur} (budget={t10:.0f}s) ===")
        a_out, _ = _local_search_pairwise(best_seq, t10, verbose=True, rng=rng)
        _try_update(a_out, f"local_final_n{n_cur}")

    # Final verification
    if best_seq is not None:
        fs = _eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
            print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, "
                  f"time={time.monotonic()-t_global:.0f}s")
        else:
            print("\nWARNING: Final evaluation failed!")
            return [1.0]
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
