import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
from scipy import sparse


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def internal_score(a):
    n = len(a)
    return 2.0 * n * fast_autorr(a).max() if False else 2.0 * n * fast_autocorr(a).max()


def project_simplex(v, target_sum=1.0):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if n == 0:
        return v
    if np.all(v >= 0) and abs(v.sum() - target_sum) < 1e-10:
        return v
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho_candidates = np.nonzero(v_sorted * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho_candidates) == 0:
        return np.full(n, target_sum / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    s = result.sum()
    if s < 1e-15:
        return np.full(n, target_sum / n)
    result = result / s * target_sum
    return result

def build_lp_precompute(n):
    """Precompute row/col indices for the LP constraint matrix."""
    rows_T = np.empty(n * n, dtype=np.int32)
    cols_T = np.empty(n * n, dtype=np.int32)
    for j in range(n):
        rows_T[j*n:(j+1)*n] = np.arange(j, j + n, dtype=np.int32)
        cols_T[j*n:(j+1)*n] = j
    rows_M = np.arange(2*n-1, dtype=np.int32)
    cols_M = np.full(2*n-1, n, dtype=np.int32)
    all_rows = np.concatenate([rows_T, rows_M])
    all_cols = np.concatenate([cols_T, cols_M])
    return all_rows, all_cols


def solve_best_response_precomputed(y, all_rows, all_cols, n, A_eq, c_obj, b_ub):
    """Solve the best response LP using precomputed structure."""
    vals_T = np.tile(y, n)
    vals_M = np.full(2*n-1, -1.0)
    all_vals = np.concatenate([vals_T, vals_M])
    A_ub = sparse.csc_matrix((all_vals, (all_rows, all_cols)), shape=(2*n-1, n+1))
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq,
                  bounds=[(0, None)]*n + [(None, None)], method='highs',
                  options={'presolve': True, 'time_limit': 10.0})
    if res.status != 0:
        return None, None
    b_val = np.maximum(res.x[:n], 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    return b_val / s, float(res.x[n])


def nesterov_game(a0, time_limit, verbose=True):
    """Nesterov-accelerated zero-sum game iteration with precomputed LP structure."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0
    stall_count = 0

    all_rows, all_cols = build_lp_precompute(n)
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    c_obj = np.zeros(n+1)
    c_obj[n] = 1.0
    b_ub = np.zeros(2*n-1)
    bounds = [(0, None)]*n + [(None, None)]

    iter_count = 0
    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        vals_T = np.tile(y, n)
        vals_M = np.full(2*n-1, -1.0)
        all_vals = np.concatenate([vals_T, vals_M])
        A_ub = sparse.csc_matrix((all_vals, (all_rows, all_cols)), shape=(2*n-1, n+1))

        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10.0})

        if res.status != 0:
            continue

        b_star = np.maximum(res.x[:n], 0)
        s_b = b_star.sum()
        if s_b < 1e-15:
            continue
        b_star = b_star / s_b
        M_val = float(res.x[n])
        iter_count += 1

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = project_simplex(a_new)

        y = y_new
        a = a_new

        score_a = internal_score(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        score_y = internal_score(y)
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        avg_a += a
        n_avg += 1
        if n_avg % 500 == 0:
            avg_check = avg_a / n_avg
            score_avg = internal_score(avg_check)
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
                stall_count = 0

        stall_count += 1
        if stall_count > 100000:
            break

        if verbose and iter_count % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Game t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}, iter={iter_count}")

    return best_a, best_score, iter_count


def spectral_shaping(a0, n_iter=300, alpha=0.25):
    """Iteratively flatten power spectrum to reduce autocorrelation peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    fft_size = 1 << (2*n-1).bit_length()

    best_a = a.copy()
    best_score = internal_score(a)

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        ratio = np.clip(target / (non_dc + 1e-30), 0.01, 50.0)
        scale = np.ones(len(A))
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score


def smooth_max_subgrad(a0, time_limit, T_start=50.0, T_end=1.0):
    """Subgradient optimization of smooth max of autocorrelation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    fft_size = 1 << (2*n-1).bit_length()
    t = 0
    base_lr = 0.05
    momentum = np.zeros(n)
    beta = 0.85

    while time.monotonic() - t0 < time_limit:
        t += 1
        b = fast_autocorr(a)
        T_cur = T_start * (T_end / T_start) ** (t / max(1, int(time_limit * 100)))
        T_cur = max(T_cur, T_end)
        Tb = T_cur * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()

        # Gradient via FFT: grad_j = 2 * sum_k w_k * a[k-j] (valid)
        W = rfft(w, fft_size)
        A_rev = rfft(a[::-1].copy(), fft_size)
        conv = irfft(A_rev * W, fft_size)
        grad = 2.0 * conv[n-1:n-1+n]

        lr_t = base_lr / (1 + 0.005 * t)
        momentum = beta * momentum + (1 - beta) * grad
        a_new = a - lr_t * momentum
        a_new = project_simplex(a_new)

        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        if t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SubGrad t={t} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=123):
    """Coordinate descent peak shaving."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:15]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            src_local = np.argmax(contrib)
            src = idx[src_local]

            peak_areas = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_areas.update(range(js2, je2 + 1))

            candidates = [j for j in range(n) if j not in peak_areas]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            candidates_sorted = sorted(candidates, key=lambda j: a[j])
            dst_pool = candidates_sorted[:max(1, len(candidates_sorted) // 4)]
            dst = dst_pool[rng.integers(len(dst_pool))]

            for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = internal_score(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.002, 0.015)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = internal_score(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > 800:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def diverse_init(n, seed=0):
    """Generate diverse initial sequences."""
    rng = np.random.default_rng(seed)
    inits = []

    # Sine
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x) ** 2
    inits.append(a / a.sum())

    # Gaussian
    x = np.linspace(-2, 2, n)
    a = np.exp(-x**2 / 4)
    inits.append(a / a.sum())

    # Triangle
    x = np.linspace(0, 1, n)
    a = 1 - np.abs(2*x - 1)
    inits.append(a / a.sum())

    # Random smooth
    a = np.ones(n)
    for freq in range(1, 5):
        a += rng.uniform(-0.5, 0.5) * np.sin(2*np.pi*freq*x + rng.uniform(0, 2*np.pi))
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        inits.append(a / a.sum())

    # Two bumps
    a = np.exp(-((x - 0.3)**2) / 0.02) + np.exp(-((x - 0.7)**2) / 0.02)
    inits.append(a / a.sum())

    return inits

            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = np.maximum(a_out, 0)
                s = best_seq.sum()
                if s > 0:
                    best_seq = best_seq / s
                print(f"  *** NEW BEST [LP_n{n_try}]: {best_score:.6f}")

    elapsed = time.monotonic() - t_global_start
    print(f"\n  After Phase 4: best={best_score:.6f} (t={elapsed:.0f}s)")

    # Phase 5: More LP at the best length found so far
    n_best = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: More LP n={n_best} (remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.25, 400.0)
    if t5 > 30:
        a_out, sc_out, n_iter = nesterov_game(best_seq, t5, verbose=True)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [LP_round2_n{n_best}]: {best_score:.6f} ({n_iter} iterations)")
    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 5: best={best_score:.6f} (t={elapsed:.0f}s)")

    # Phase 6: Peak shaving
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: Peak shaving n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t6 = min(remaining * 0.15, 100.0)
    if t6 > 5:
        a_out, sc_out = peak_shaving(best_seq, t6, seed=456)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.maximum(a_out, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
            print(f"  *** NEW BEST [peak_shave_1]: {best_score:.6f}")
    elapsed = time.monotonic() - t_global_start
    print(f"  After Phase 6: best={best_score:.6f} (t={elapsed:.0f}s)")

    # Phase 7: Final LP push if time remains
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 30:
        print(f"\n=== Phase 7: Final LP n={len(best_seq)} (remaining={remaining:.0f}s) ===")
        t7 = min(remaining * 0.30, 200.0)
        if t7 > 20:
            a_out, sc_out, _ = nesterov_game(best_seq, t7, verbose=False)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = np.maximum(a_out, 0)
                s = best_seq.sum()
                if s > 0:
                    best_seq = best_seq / s
                print(f"  *** NEW BEST [LP_final]: {best_score:.6f}")

    # Phase 8: Final peak shave
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 10:
        t8 = min(remaining * 0.30, 50.0)
        if t8 > 5:
            a_out, sc_out = peak_shaving(best_seq, t8, seed=789)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = np.maximum(a_out, 0)
                s = best_seq.sum()
                if s > 0:
                    best_seq = best_seq / s
                print(f"  *** NEW BEST [peak_shave_2]: {best_score:.6f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        best_seq = np.maximum(best_seq, 0)
        s = best_seq.sum()
        if s > 0:
            best_seq = best_seq / s
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
