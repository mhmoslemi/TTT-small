    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v

        r = len(active_constraints)

    if b is None:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Active-set Nesterov Frank-Wolfe (constraint generation) as primary method
2. Mirror descent (exponential weights) as fast warm start
3. Coordinate descent for final polish
4. Multi-start with diverse initializations and adaptive length scaling
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_conv(a, b=None):
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), n=fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), n=fft_size)
    return irfft(A * B, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def proj_simplex(v, s):
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) + s - cssv > 0
    if not np.any(cond):
        return np.zeros(n)
    rho = u[np.argmax(cond)]
    return np.maximum(v - rho, 0.0)


def build_conv_row(a, k, n):
    """Build the row of the convolution matrix for lag k."""
    row = np.zeros(n)
    js = max(0, k - (n - 1))
    je = min(n - 1, k)
    for j in range(js, je + 1):
        row[j] = a[k - j]
    return row


def active_set_lp(a, n, S, active_set, max_inner_iter=50):
    """
    Solve: min M s.t. conv(a,b)_k <= M for k in active_set, b >= 0, sum(b) = S
    Returns (b, M) or (None, None) if infeasible.
    """
    r = len(active_set)
    if r == 0:
        b = np.full(n, S / n)
        M = fast_conv(a, b).max()
        return b, M
    
    # Variables: [b_0, ..., b_{n-1}, M] -> n+1 variables
    # Obj: minimize M
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Inequality constraints: for each active k: sum_j A[k,j] b_j - M <= 0
    A_ub = np.zeros((r, n + 1))
    for i, k in enumerate(active_set):
        js = max(0, int(k) - (n - 1))
        je = min(n - 1, int(k))
        for j in range(js, je + 1):
            A_ub[i, j] = a[int(k) - j]
        A_ub[i, n] = -1.0
    b_ub = np.zeros(r)
    
    # Equality: sum(b) = S
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    
    # Bounds
    bounds = [(0, None)] * n + [(-np.inf, np.inf)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None, None
    
    if res.status not in (0, 3):  # 0=optimal, 3=optimal (inaccurate)
        return None, None
    
    b = np.maximum(res.x[:n], 0.0)
    M = res.x[n]
    return b, M


def best_response(a, n, S, max_lags=20, max_iter=80):
    """
    Find best response b: min max_k conv(a,b)_k subject to b>=0, sum(b)=S.
    Uses constraint generation (active set method).
    """
    # Initial active set: top lags from conv(a, uniform)
    b_init = np.full(n, S / n)
    ab_init = fast_conv(a, b_init)
    top_lags = np.argsort(ab_init)[::-1][:5]
    active_set = list(top_lags)
    
    best_b = b_init.copy()
    best_M = ab_init.max()
    
    for _ in range(max_iter):
        b, M = active_set_lp(a, n, S, active_set)
        if b is None:
            break
        
        # Check all constraints
        ab = fast_conv(a, b)
        actual_M = ab.max()
        
        if actual_M < best_M:
            best_M = actual_M
            best_b = b.copy()
        
        # Find violated constraints
        violations = ab - M
        if np.max(violations) < 1e-10 * max(1.0, M):
            break  # Satisfied
        
        worst_k = int(np.argmax(violations))
        if worst_k not in active_set:
            active_set.append(worst_k)
        
        # Keep active set manageable
        if len(active_set) > max_lags:
            # Remove least violated
            viol_in_set = np.array([ab[k] - M for k in active_set])
            order = np.argsort(viol_in_set)
            active_set = [active_set[i] for i in order[len(order)//2:]]
    
    return best_b
