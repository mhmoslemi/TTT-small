def lse_objective_and_grad(a_flat, n, T_param):
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    shifted = T_param * (b - max_b)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    exp_sum = exp_vals.sum()
    if exp_sum < 1e-300:
        return 1e10, np.zeros(n)
    w = exp_vals / exp_sum
    lse = max_b + np.log(exp_sum) / T_param
    obj = 2.0 * n * lse / (s * s)

    # Gradient via cross-correlation
    fft_size = 1 << (3 * n - 2).bit_length()
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    w_padded = np.zeros(fft_size)
    w_padded[:2 * n - 1] = w
    A_fft = rfft(a_padded, fft_size)
    W_fft = rfft(w_padded, fft_size)
    cross = irfft(A_fft * W_fft, fft_size)[:n]
    conv_grad = 4.0 * cross
    grad = conv_grad / (s * s) - 2.0 * obj * a / s
    return obj, grad.ravel()


def lse_scipy_optimize(a0, time_limit, T_start=40.0, T_end=1.0, n_stages=8):

    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        stage_time = min(remaining / max(1, n_stages - si), 100.0)
        if stage_time < 3:
            break

        a_current = a.copy()

        def obj_func(x):
            val, _ = lse_objective_and_grad(x, n, T)
            return val

        def grad_func(x):
            _, g = lse_objective_and_grad(x, n, T)
            return g

        try:
            res = scipy_minimize(obj_func, a_current, method='L-BFGS-B',
                                 jac=grad_func, bounds=[(0.0, None)] * n,
                                 options={'maxiter': 300, 'maxfun': 1000,
                                          'ftol': 1e-12, 'gtol': 1e-10})
            if res.x is not None:
                a_candidate = np.maximum(res.x, 0.0)
                s_c = a_candidate.sum()
                if s_c > 1e-10:
                    a_candidate = a_candidate * norm / s_c
                    score_c = fast_autocorr(a_candidate).max()
                    if score_c < best_score:
                        best_score = score_c
                        best_a = a_candidate.copy()
                        a = a_candidate
        except Exception:
            pass

    return best_a, best_score


def peak_aware_coord_descent(a0, time_limit, seed=42):

    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

    if s0 < 1e-15:

    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        iteration += 1
        b = fast_autocorr(a)
        k_peak = int(np.argmax(b))
        M_curr = b[k_peak]

        if M_curr < best_score - 1e-12:
            best_score = M_curr
            best_a = a.copy()
            stall = 0
            continue

        # Find contributing indices for the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]
        total_contrib = contrib.sum()

        if total_contrib < 1e-15:
            stall += 1
            if stall > 500:
                break
            continue

        # Find top contributors
        top_k = min(10, len(idx))
        top_local = np.argsort(contrib)[::-1][:top_k]
        top_src = idx[top_local]

        # Build peak mask (positions that contribute to top peaks)
        top_peaks = np.argsort(b)[::-1][:12]
        peak_mask = np.zeros(n, dtype=bool)
        for kp in top_peaks:
            js2 = max(0, kp - n + 1)
            je2 = min(n - 1, kp)
            peak_mask[js2:je2 + 1] = True

        improved = False

        # Try moving mass from top contributors to non-peak positions
        for src in top_src:
            if a[src] < 1e-10:
                continue

            # Find good destinations: low contribution to the peak
            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]

            # Score destinations by how much they'd reduce the peak
            # Moving delta from src to dst changes c[k_peak] by approximately:
            # -2*delta*a[k_peak-src] + 2*delta*a[k_peak-dst] (if valid indices)
            src_contrib_val = a[k_peak - src] if 0 <= k_peak - src < n else 0.0

            dst_scores = np.zeros(len(candidates))
            for ci, dst in enumerate(candidates):
                dst_contrib_val = a[k_peak - dst] if 0 <= k_peak - dst < n else 0.0
                dst_scores[ci] = src_contrib_val - dst_contrib_val

            # Try the best destinations
            top_dst = np.argsort(dst_scores)[::-1][:5]
            for di in top_dst:
                if dst_scores[di] <= 0:
                    continue
                dst = candidates[di]
                if a[dst] > a[src] * 3:
                    continue

                # Binary search for optimal delta
                delta_lo = 0.0
                delta_hi = a[src] * 0.5
                best_delta = 0.0
                best_delta_score = M_curr

                for _ in range(6):
                    delta_mid = (delta_lo + delta_hi) / 2
                    if delta_mid < 1e-15:
                        break
                    a_trial = a.copy()
                    a_trial[src] -= delta_mid
                    a_trial[dst] += delta_mid
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        break
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_delta_score:
                        best_delta_score = score
                        best_delta = delta_mid
                        delta_lo = delta_mid
                    else:
                        delta_hi = delta_mid

                if best_delta > 0 and best_delta_score < best_score - 1e-12:
                    a[src] -= best_delta
                    a[dst] += best_delta
                    a = np.maximum(a, 0.0)
                    s_t = a.sum()
                    a = a * norm / s_t
                    best_score = best_delta_score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

            if improved:
                break

        if not improved:
            # Try smaller random perturbations
            stall += 1
            if stall % 5 == 0:
                for _ in range(50):
                    i = rng.integers(n)
                    if a[i] < 1e-10:
                        continue
                    j = rng.integers(n)
                    if i == j:
                        continue
                    frac = rng.uniform(0.005, 0.08)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                        improved = True
                        break

            if stall > 800:
                # Random restart with perturbation
                perturb = rng.random(n) * 0.1 * a.max()
                a = a + perturb
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-10:
                    a = a * norm / s_t
                    sc = fast_autocorr(a).max()
                    if sc > best_score * 1.5:
                        a = best_a.copy()
                    stall = 0

    return best_a, best_score


def water_filling_polish(a0, time_limit, seed=42):
    """Proactively reduce peaks by distributing excess."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        avg_b = (a.sum() ** 2) / (2 * n - 1)
        k_peak = int(np.argmax(b))
        excess = b[k_peak] - avg_b

        if excess < 1e-10:
            break

        if b[k_peak] < best_score - 1e-12:
            best_score = b[k_peak]
            best_a = a.copy()
            stall = 0
            continue

        # Find all top peaks and their excess
        top_peaks = np.argsort(b)[::-1][:15]
        peak_excess = np.array([b[kp] - avg_b for kp in top_peaks])
        total_excess = peak_excess[peak_excess > 0].sum()

        if total_excess < 1e-10:
            break

        # For the worst peak, find the dominant pairs and reduce them
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]
        total_c = contrib.sum()
        if total_c < 1e-15:
            stall += 1
            if stall > 200:
                break
            continue

        # Proportional reduction: reduce each contributing pair by a fraction
        # of its excess contribution
        target_reduction = excess * 0.3  # try to reduce 30% of excess
        reduction_per_unit = target_reduction / total_c if total_c > 1e-15 else 0

        # Apply reduction: for each (i, k-i) pair, reduce the product
        # by reducing min(a[i], a[k-i]) slightly
        applied = np.zeros(n)
        for li in range(len(idx)):
            i = idx[li]
            j = k_peak - i
            pair_product = a[i] * a[j]
            if pair_product < 1e-15:
                continue
            # The contribution to c[k_peak] from this pair is:
            # pair_product if i != j, else pair_product (counted once in sum)
            # Actually c[k_peak] = sum_i a[i]*a[k_peak-i], so each pair (i, k-i)
            # with i != k-i contributes 2*a[i]*a[k-i] to c[k_peak]
            # Wait no, it's sum over i, so each i contributes a[i]*a[k-i]
            # If i != k-i, both i and k-i appear in the sum separately
            # So the total from the pair {i, k-i} is 2*a[i]*a[k-i] (if i != k-i)
            # or a[i]^2 (if i == k-i)

            if i == j:
                # a[i]^2 term: reduce a[i]
                needed = reduction_per_unit * pair_product
                # d(a[i]^2) = 2*a[i]*da = -needed => da = -needed/(2*a[i])
                da = -needed / (2.0 * a[i]) if a[i] > 1e-15 else 0.0
                da = max(da, -a[i] * 0.3)  # limit reduction
                applied[i] += da
            else:
                # Reduce the smaller of a[i], a[j]
                if a[i] <= a[j]:
                    needed = reduction_per_unit * 2.0 * pair_product
                    da = -needed / (2.0 * a[j]) if a[j] > 1e-15 else 0.0
                    da = max(da, -a[i] * 0.2)
                    applied[i] += da
                else:
                    needed = reduction_per_unit * 2.0 * pair_product
                    da = -needed / (2.0 * a[i]) if a[i] > 1e-15 else 0.0
                    da = max(da, -a[j] * 0.2)
                    applied[j] += da

        # Redistribute: add the removed mass to positions far from peaks
        total_removed = -applied[applied < 0].sum()
        if total_removed < 1e-15:
            stall += 1
            if stall > 200:
                break
            continue

        # Find good redistribution targets (low values, far from peaks)
        peak_mask = np.zeros(n, dtype=bool)
        for kp in top_peaks:
            js2 = max(0, kp - n + 1)
            je2 = min(n - 1, kp)
            peak_mask[js2:je2 + 1] = True

        non_peak = np.where(~peak_mask)[0]
        if len(non_peak) == 0:
            non_peak = np.where(np.arange(n) > 0)[0]

        # Distribute to lowest-valued non-peak positions
        n_targets = min(20, len(non_peak))
        targets = non_peak[np.argsort(a[non_peak])[:n_targets]]
        per_target = total_removed / len(targets)

        a_trial = a.copy()
        a_trial += applied  # apply reductions
        for tgt in targets:
            a_trial[tgt] += per_target

        a_trial = np.maximum(a_trial, 0.0)
        s_t = a_trial.sum()
        if s_t < 1e-10:
            stall += 1
            continue
        a_trial = a_trial * norm / s_t
        score = fast_autocorr(a_trial).max()

        if score < best_score - 1e-12:
            a = a_trial
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 300:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def structured_init(n, seed=0):
    rng = np.random.default_rng(seed)
    choice = seed % 5
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 1:
        freq1 = rng.uniform(2, 6)
        freq2 = rng.uniform(5, 15)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * freq1 * x) + 0.2 * np.cos(2 * np.pi * freq2 * x)
    elif choice == 2:
        a = np.exp(-5.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-5.0 * (x - 0.7) ** 2)
    elif choice == 3:
        a = 1.0 / (1.0 + np.exp(-20.0 * (x - 0.5)))
        a = a + 0.3 * np.sin(10 * np.pi * x)
    else:
        a = np.ones(n) / n
        a = a + 0.2 * np.cos(4 * np.pi * x)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 0: Quick multi-length scan
    print("\n=== Phase 0: Multi-length scan ===")
    scan_lengths = [1400, 1600, 1800, 2000]
    scan_results = {}
    scan_budget = time_budget * 0.12

    for n_scan in scan_lengths:
        elapsed = time.monotonic() - t_global
        if elapsed > time_budget * 0.12:
            break
        remaining_for_scan = time_budget * 0.12 - elapsed
        if remaining_for_scan < 15:
            break
        scan_time = min(remaining_for_scan / max(1, len(scan_lengths) - scan_lengths.index(n_scan)), 60.0)
        if scan_time < 10:
            break

        if best_seq is not None:
            init_seq = interpolate_sequence(best_seq, n_scan)
        else:
            init_seq = structured_init(n_scan, seed=n_scan)

        a_norm = init_seq * np.sqrt(2.0 * n_scan) / init_seq.sum()
        a_out, sc_out = nesterov_lp_optimize(a_norm, scan_time, solver, verbose=False)
        scan_results[n_scan] = (a_out, sc_out)
        print(f"    n={n_scan}: score={sc_out:.8f} (t={time.monotonic()-t_global:.0f}s)")

    if scan_results:
        best_scan_n = min(scan_results, key=lambda k: scan_results[k][1])
        scan_best_a, scan_best_sc = scan_results[best_scan_n]
        do_try_update(scan_best_a, f"scan_n{best_scan_n}")

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, seed=42)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    # Phase 1: LP-Nesterov at base length (main driver)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget={remaining:.0f}s) ===")
    t1 = remaining * 0.30
    a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
    a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
    do_try_update(a_out, f"lp_n{n_base}")

    n_base = len(best_seq)

    # Phase 2: LSE optimization
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: LSE n={n_base} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = lse_scipy_optimize(a_norm, t2, T_start=50.0, T_end=1.0, n_stages=8)
        do_try_update(a_out, f"lse_n{n_base}")

    n_base = len(best_seq)

    # Phase 3: Water-filling polish
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Water-filling n={n_base} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.12, 150.0)
    if t3 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = water_filling_polish(a_norm, t3, seed=123)
        do_try_update(a_out, f"waterfill_n{n_base}")

    n_base = len(best_seq)

    # Phase 4: Peak-aware coordinate descent (main polish)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Coord descent n={n_base} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.25, 400.0)
    if t4 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = peak_aware_coord_descent(a_norm, t4, seed=42)
        do_try_update(a_out, f"coord_n{n_base}")

    n_base = len(best_seq)

    # Phase 5: Try a slightly different length
    n_alt = n_base + 200
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Alt length n={n_alt} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 20:
        init_alt = interpolate_sequence(best_seq, n_alt)
        a_norm = init_alt * np.sqrt(2.0 * n_alt) / init_alt.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t5 * 0.4, solver, verbose=False)
        if a_out is not None:
            a_alt = a_out.copy()
            # Quick coord descent on the new length
            t5cd = t5 * 0.6
            if t5cd > 10:
                a_out2, _ = peak_aware_coord_descent(a_alt, t5cd, seed=99)
                do_try_update(a_out2, f"coord_alt_n{n_alt}")
            do_try_update(a_alt, f"lp_alt_n{n_alt}")

    n_base = len(best_seq)

    # Phase 6: Final LSE with low temperature
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: LSE polish n={n_base} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.12, 120.0)
    if t6 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = lse_scipy_optimize(a_norm, t6, T_start=5.0, T_end=0.5, n_stages=5)
        do_try_update(a_out, "lse_polish")

    n_base = len(best_seq)

    # Phase 7: Final aggressive coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    final_time = remaining * 0.7
    print(f"\n=== Phase 7: Final coord n={n_base} (time={final_time:.0f}s) ===")
    if final_time > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = peak_aware_coord_descent(a_norm, final_time, seed=77777)
        do_try_update(a_out, "coord_final")

    # Final water-filling
    remaining = time_budget - (time.monotonic() - t_global)
    n_base = len(best_seq)
    if remaining > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = water_filling_polish(a_norm, remaining * 0.8, seed=31415)
        do_try_update(a_out, "waterfill_final")

    # Very final coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_base = len(best_seq)
    if remaining > 5:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = peak_aware_coord_descent(a_norm, remaining, seed=27182)
        do_try_update(a_out, "coord_veryfinal")

    # Final check
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
