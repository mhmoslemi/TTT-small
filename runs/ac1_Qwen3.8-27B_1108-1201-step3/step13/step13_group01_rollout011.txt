def build_conv_matrix_sparse(a):

            j_hi = min(n - 1, k_peak)
            if j_hi < j_lo:
                continue
            js = np.arange(j_lo, j_hi + 1)
            contrib = a[js] * a[k_peak - js]
            
            # Find the most contributing element
            top_idx = js[np.argsort(contrib)[::-1][:3]]
            
            # Find elements NOT contributing to any top peak
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in peak_indices:
                jl = max(0, k2 - n + 1)
                jh = min(n - 1, k2)
                peak_mask[jl:jh+1] = True
            
            free_indices = np.where(~peak_mask)[0]
            if len(free_indices) == 0:
                free_indices = np.where(np.arange(n) != top_idx[0])[0]
            
            # Sort free indices by value (prefer moving to low-value positions)
            free_sorted = free_indices[np.argsort(a[free_indices])][:max(1, len(free_indices)//4)]
            
            for src in top_idx:
                if a[src] < 1e-10:
                    continue
                for dst in free_sorted[:5]:
                    for frac in [0.005, 0.01, 0.02, 0.05]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-10:
                            continue
                        a_trial = a_trial * norm / s_t
                        sc = fast_autocorr(a_trial).max()
                        if sc < best_score - 1e-14:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Random perturbation to escape local minima
            stall += 1
            if stall > 50:
                break
            for _ in range(10):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-10:
                    continue
                frac = float(rng.uniform(0.01, 0.05))
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial * norm / s_t
                sc = fast_autocorr(a_trial).max()
                if sc < best_score:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    stall = 0
                    break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    return np.maximum(f_interp, 0.0)
