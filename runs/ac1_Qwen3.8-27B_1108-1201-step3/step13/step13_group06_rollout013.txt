```python
import numpy as np
from scipy.fft import rfft, irfft
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def fast_autocorr(a):
    """Compute full autocorrelation a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def full_lse_obj_grad(p, T):
    """Compute full LSE objective and gradient wrt p (on simplex).
    p should sum to 1 and be non-negative.
    Returns (lse_value, grad_p) where grad_p is the gradient on the ambient space.
    """
    n = len(p)
    c = fast_autocorr(p)  # length 2n-1
    Tc = T * c
    Tc_shifted = Tc - Tc.max()
    exp_Tc = np.exp(Tc_shifted)
    lse_val = (Tc.max() + np.log(exp_Tc.sum())) / T
    w = exp_Tc / exp_Tc.sum()
    
    # grad[j] = 2 * sum_m w[j+m] * p[m] for j=0,...,n-1
    # This is 2 * correlate(w, p, mode='valid')
    grad_p = 2.0 * correlate(w, p, mode='valid', method='fft')
    return lse_val, grad_p

def simplex_obj_grad(z, T):
    """Objective and gradient in log-space z (p = softmax(z))."""
    p = softmax_fn(z)
    lse_val, grad_p = full_lse_obj_grad(p, T)
    # dLSE/dz[j] = p[j] * (grad_p[j] - sum_i p[i]*grad_p[i])
    S = np.dot(grad_p, p)
    grad_z = p * (grad_p - S)
    return lse_val, grad_z

def softmax_fn(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s
```

```python
        rng = np.random.default_rng(42)
```

```python
def fast_autocorr(a):
    """Compute full autocorrelation a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def full_lse_obj_grad_z(z, T):
    """
    Compute LSE objective and gradient in z-space.
    p = softmax(z), objective = LSE of autocorrelation of p.
    Returns (objective_value, gradient_wrt_z).
    """
    p = softmax_fn(z)
    n = len(p)
    c = fast_autocorr(p)
    Tc = T * c
    Tc_max = Tc.max()
    exp_Tc = np.exp(Tc - Tc_max)
    lse_val = (Tc_max + np.log(exp_Tc.sum())) / T
    w = exp_Tc / exp_Tc.sum()
    # grad_p[j] = 2 * sum_m w[j+m] * p[m] = 2 * correlate(w, p, 'valid')
    grad_p = 2.0 * correlate(w, p, mode='valid', method='fft')
    S = np.dot(grad_p, p)
    grad_z = p * (grad_p - S)
    return lse_val, grad_z


def lse_wrapper(z, T):
    """Wrapper for scipy minimize: returns (f, grad)."""
    val, grad = full_lse_obj_grad_z(z, T)
    return val, grad


def lse_fun_only(z, T):
    val, _ = full_lse_obj_grad_z(z, T)
    return val


def lse_grad_only(z, T):
    _, grad = full_lse_obj_grad_z(z, T)
    return grad


def spectral_init(n, n_iters=300, seed=42):
    """Initialize using spectral flattening (Gerchberg-Saxton variant)."""
    rng = np.random.default_rng(seed)
    p = rng.exponential(1.0, n)
    p = p / p.sum()
    fft_len = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        A = rfft(p, fft_len)
        P = np.abs(A) ** 2
        pmin = P[P > 1e-15].min() if np.any(P > 1e-15) else 1e-15
        # Flatten: take power 0.7 to reduce dynamic range
        P_target = np.maximum(P, pmin) ** 0.7
        ratio = np.sqrt(P_target / np.maximum(P, 1e-30))
        A_new = A * ratio
        p_new = irfft(A_new, fft_len)[:n]
        p_new = np.maximum(p_new, 0)
        s = p_new.sum()
        if s < 1e-15:
            break
        p_new = p_new / s
        p = p_new

    return p


def structured_init(n, seed=0):
    """Generate a structured initialization."""
    rng = np.random.default_rng(seed)
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.exp(-4 * (x - 0.3) ** 2) + 0.5 * np.exp(-4 * (x - 0.7) ** 2)
    elif choice == 2:
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(6 * np.pi * x) + 0.2 * np.cos(10 * np.pi * x)
    elif choice == 4:
        a = np.exp(-6 * x ** 2) + np.exp(-6 * (x - 1) ** 2)
    elif choice == 5:
        a = (1 - x) ** 1.5 + 1e-6
    elif choice == 6:
        a = np.maximum(0, np.cos(2 * np.pi * x * 3) ** 2 + 0.3)
    else:
        a = np.exp(-3 * (x - 0.5) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def lse_lbfgs_optimize(a0, time_limit, T_stages=None, max_lbfgs_iter=500):
    """
    Multi-stage LSE optimization using L-BFGS-B in softmax space.
    T_stages: list of temperatures from high to low.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    if T_stages is None:
        T_stages = [50000.0, 20000.0, 10000.0, 5000.0, 2000.0, 1000.0, 500.0, 200.0, 100.0, 50.0]

    z = np.log(np.maximum(a, 1e-30))
    t0 = time.monotonic()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    for stage_i, T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit - 5:
            break

        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        def obj_grad(z_val):
            val, grad = full_lse_obj_grad_z(z_val, T)
            return val, grad

        try:
            result = minimize(
                obj_grad, z, jac=True, method='L-BFGS-B',
                options={'maxiter': max_lbfgs_iter, 'maxfun': max_lbfgs_iter * 2,
                         'gtol': 1e-12, 'ftol': 1e-14}
            )
        except Exception as e:
            print(f"      L-BFGS error at T={T}: {e}")
            break

        z = np.asarray(result.x, dtype=np.float64)
        a_stage = softmax_fn(z)
        score = 2.0 * n * fast_autocorr(a_stage).max()

        if score < best_score:
            best_score = score
            best_a = a_stage.copy()

        elapsed = time.monotonic() - t0
        print(f"      LSE stage {stage_i+1}/{len(T_stages)} (T={T:.0f}, {elapsed:.0f}s): score={best_score:.8f}")

    # Final evaluation with actual max
    score_final = 2.0 * n * fast_autocorr(best_a).max()
    if score_final < best_score:
        best_score = score_final

    return best_a, best_score

    best_score = 2.0 * n * fast_autocorr(a).max()

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit - 5:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.8, 0.999, 1e-8
        current_lr = lr * (0.9 ** si)

        max_it = int(min(remaining, 60.0) * 30)
        max_it = max(max_it, 50)

        stall = 0
        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 3:
                break

            obj, grad_z = full_lse_obj_grad_z(z, T)
            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - current_lr * m_hat / (np.sqrt(v_hat) + eps)

            if (it + 1) % 10 == 0:
                a_chk = softmax_fn(z)
                sc = 2.0 * n * fast_autocorr(a_chk).max()

        if si % 5 == 0:

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=42):
    """
    Systematic peak shaving: reduce max autocorrelation by moving mass.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 3:
        b = fast_autocorr(a)
        bmax = b.max()
        # Find all peaks within 0.5% of max
        threshold = bmax * 0.995
        top_peaks = np.where(b >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(b)[::-1][:5]

        # Compute danger zones (positions that contribute to top peaks)
        danger = np.zeros(n, dtype=bool)
        for k in top_peaks[:20]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            danger[js:je + 1] = True

        improved = False
        for k in top_peaks[:15]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            # Find safe destinations (not in danger zone)
            safe = np.where(~danger)[0]
            safe = safe[safe != src]
            if len(safe) == 0:
                safe = np.where(np.arange(n) != src)[0]
            # Sort by current value (move to low-value positions)
            safe_sorted = safe[np.argsort(a[safe])]
            dst_pool = safe_sorted[:max(10, len(safe_sorted) // 3)]
            if len(dst_pool) == 0:
                continue

            found = False
            for dst in dst_pool[:12]:
                for frac in [0.003, 0.005, 0.01, 0.02, 0.03, 0.05, 0.08]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-12:
                        continue
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-12:
                        continue
                    a_trial = a_trial / st
                    new_max = fast_autocorr(a_trial).max()
                    if new_max < bmax * (1 - 1e-9):
                        a = a_trial
                        new_score = 2.0 * n * new_max
                        if new_score < best_score:
                            best_score = new_score
                            best_a = a.copy()
                        improved = True
                        stall = 0
                        found = True
                        break
                if found:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 5 == 0:
                # Random perturbation
                n_pert = rng.integers(1, 6)
                a_trial = a.copy()
                for _ in range(n_pert):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a_trial[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.02)
                        amt = a_trial[i] * frac
                        a_trial[i] -= amt
                        a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-10:
                    a_trial = a_trial / st
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > 100:
                break

    return best_a, best_score


def coordinate_descent_refine(a0, time_limit, seed=123):
    """
    Coordinate descent: try adjusting pairs of coordinates to reduce max autocorrelation.
    More targeted than peak shaving.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 3:
        b = fast_autocorr(a)
        bmax = b.max()
        k_star = b.argmax()

        # The gradient at k_star tells us which positions to adjust
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_star - idx]

        if contrib.max() < 1e-15:
            break

        # Try moving mass from high-contribution to low-contribution (within the peak's support)
        # or to positions outside the peak's influence
        src_candidates = idx[np.argsort(contrib)[::-1][:5]]
        dst_candidates = np.where(~np.isin(np.arange(n), idx))[0]
        dst_candidates = dst_candidates[np.argsort(a[dst_candidates])[:10]]

        improved = False
        for src in src_candidates:
            for dst in dst_candidates:
                for frac in [0.01, 0.02, 0.04, 0.06]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-12:
                        continue
                    a_trial = a_trial / st
                    new_max = fast_autocorr(a_trial).max()
                    if new_max < bmax * (1 - 1e-8):
                        a = a_trial
                        new_score = 2.0 * n * new_max
                        if new_score < best_score:
                            best_score = new_score
                            best_a = a.copy()

                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 50:

def symmetrize(a):
    """Average with reversal to exploit symmetry."""
    return (a + a[::-1]) / 2.0

    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 5: Coord descent n={n_base} ===")
    t5 = min(remaining * 0.08, 80.0)
    if t5 > 10:
        a_out, _ = coordinate_descent_refine(best_seq, t5, seed=999)
        try_update(a_out, "coord_desc")

    # === Phase 6: Try longer length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_base + 200
    print(f"=== Phase 6: LSE L-BFGS n={n_long} ===")
    t6 = min(remaining * 0.15, 180.0)
    if t6 > 40:
        init_long = interpolate_sequence(best_seq, n_long)
        T_stages_long = [80000.0, 30000.0, 10000.0, 5000.0, 2000.0, 1000.0, 500.0, 200.0]
        a_out, _ = lse_lbfgs_optimize(init_long, t6, T_stages=T_stages_long, max_lbfgs_iter=200)
        try_update(a_out, f"lse_lbfgs_n{n_long}")

    # === Phase 7: Peak shaving on longer ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        print(f"=== Phase 7: Peak shave n={n_long} ===")
        t7 = min(remaining * 0.08, 60.0)
        if t7 > 10:
            a_out, _ = peak_shaving(best_seq, t7, seed=1001)
            try_update(a_out, "peak_shave_long")

    # === Phase 8: Final LSE polish at best length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"=== Phase 8: Final LSE polish n={n_best} ===")
    t8 = min(remaining * 0.20, 200.0)
    if t8 > 20:
        # Very low temperature for precise max approximation
        T_stages_final = [5000.0, 2000.0, 1000.0, 500.0, 300.0, 200.0]
        a_out, _ = lse_lbfgs_optimize(best_seq, t8, T_stages=T_stages_final, max_lbfgs_iter=200)
        try_update(a_out, "final_lse")

    # === Phase 9: Final peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"=== Phase 9: Final peak shave ===")
    t9 = min(remaining * 0.30, 150.0)
    if t9 > 10:
        a_out, _ = peak_shaving(best_seq, t9, seed=2000)
        try_update(a_out, "final_peak")

    # === Phase 10: Symmetrization check ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        a_sym = symmetrize(best_seq)
        a_sym = a_sym / a_sym.sum()
        sc_sym = evaluate_safe(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.copy()
            print(f"  *** Symmetrization improved: {best_score:.8f}")

        # LSE on symmetrized
        t10 = min(remaining * 0.3, 60.0)
        if t10 > 10:
            a_out, _ = lse_lbfgs_optimize(a_sym, t10,
                                           T_stages=[3000.0, 1000.0, 500.0, 200.0],
                                           max_lbfgs_iter=100)
            try_update(a_out, "lse_sym")

    # Final
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
```
