Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS-B on LSE-smoothed objective with correct FFT-based gradient
2. Multi-length exploration (n = 1200 to 2400)
3. Adaptive temperature annealing
4. Subgradient Nesterov as complementary method
5. Coordinate descent polish
Key: correct gradient via cross-correlation + convolution in frequency domain.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize as scipy_minimize

# Time management
TOTAL_TIME = 1800.0
START_TIME = None

def check_time(fraction=1.0):
    """Check if we've used more than 'fraction' of total time."""
    global START_TIME
    if START_TIME is None:
        return False
    return (time.monotonic() - START_TIME) > fraction * TOTAL_TIME

def fast_autocorr(a):
    """Full linear autocorrelation via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), N)
    return irfft(A * A, N)[:size]

def lse_obj_grad(x, n, T):
    """
    Objective and gradient of 2*n*LSE(T*conv(a,a)) / (sum(a))^2.
    x: array of length n (non-negative)
    Returns (objective, gradient)
    """
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    
    # LSE computation
    scaled = T * (b - max_b)
    # For numerical stability, clip
    scaled = np.clip(scaled, -500, 500)
    exp_sum = np.exp(scaled).sum()
    lse = max_b + np.log(exp_sum) / T
    
    obj = 2.0 * n * lse / (s * s)
    
    # Gradient
    w = T * np.exp(scaled) / exp_sum  # length 2n-1
    
    # FFT-based gradient
    N = 1 << (3 * n - 2).bit_length()
    A = rfft(a, N)
    W = rfft(w, N)
    corr_part = irfft(W * np.conj(A), N)[:n]
    conv_part = irfft(W * A, N)[:n]
    dLSE_da = corr_part + conv_part
    
    grad = 2.0 * n * dLSE_da / (s * s) - 4.0 * n * lse * a / (s ** 3)
    
    return obj, grad

def lbfgs_optimize(a0, time_limit, T_start=5.0, T_end=100.0, n_stages=8):
    """
    Run L-BFGS-B with adaptive temperature on the LSE-smoothed objective.
    Returns (best_a, best_score).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64).copy(), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    best_a = a / a.sum()
    best_score = 2.0 * n * fast_autocorr(best_a).max()
    
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        stage_time = min(remaining / (n_stages - si), 120.0)
        
        x0 = a.copy()
        
        def func(x):
            return lse_obj_grad(x, n, T)[0]
        
        def jac(x):
            return lse_obj_grad(x, n, T)[1]
        
        try:
            res = scipy_minimize(func, x0, method='L-BFGS-B',
                                  jac=jac, bounds=[(1e-10, None)] * n,
                                  options={'maxiter': 300, 'maxfun': 1000,
                                           'gtol': 1e-8})
            if res.x is not None:
                a_cand = np.maximum(res.x, 0.0)
                s_c = a_cand.sum()
                if s_c > 1e-10:
                    a_cand = a_cand / s_c
                    score_c = 2.0 * n * fast_autocorr(a_cand).max()
                    if score_c < best_score:
                        best_score = score_c
                        best_a = a_cand.copy()
                        a = a_cand
        except Exception as e:
            pass
    
    # Also evaluate the final 'a' (which might be better than the last L-BFGS-B output)
    s_final = a.sum()
    if s_final > 1e-10:
        a_final = a / s_final
        score_final = 2.0 * n * fast_autocorr(a_final).max()
        if score_final < best_score:
            best_score = score_final
            best_a = a_final.copy()
    
    return best_a, best_score

def fast_autocorr(a):
    """Full linear autocorrelation via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), N)
    return irfft(A * A, N)[:size]


def lse_obj_grad(x, n, T):
    """
    Compute objective and gradient of 2*n*LSE(T*conv(a,a)) / (sum(a))^2.
    Returns (objective, gradient_vector_of_length_n).
    """
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    scaled = T * (b - max_b)
    scaled = np.clip(scaled, -500.0, 500.0)
    exp_vals = np.exp(scaled)
    exp_sum = exp_vals.sum()
    if exp_sum < 1e-300:
        return 1e10, np.zeros(n)
    lse = max_b + np.log(exp_sum) / T
    obj = 2.0 * n * lse / (s * s)
    w = T * exp_vals / exp_sum  # length 2n-1
    N = 1 << (3 * n - 2).bit_length()
    A = rfft(a, N)
    W = rfft(w, N)
    corr_part = irfft(W * np.conj(A), N)[:n]
    conv_part = irfft(W * A, N)[:n]
    dLSE_da = corr_part + conv_part
    grad = 2.0 * n * dLSE_da / (s * s) - 4.0 * n * lse * a / (s ** 3)
    return obj, grad

def lbfgs_optimize(a0, time_limit, T_start, T_end, n_stages):

def subgrad_nesterov(a0, time_limit):

            break
        b_y = fast_autocorr(y)
        k_star = int(np.argmax(b_y))
        best_set = np.where(b_y >= 0.95 * b_y[k_star])[0]
        grad = np.zeros(n)
        for k in best_set:
            for j in range(n):
                v1 = k - j
                if 0 <= v1 < n:
                    grad[j] += y[v1]
                v2 = j - k
                if 0 <= v2 < n:
                    grad[j] += y[v2]
        grad[j] /= max(1, len(best_set))
        i_min = int(np.argmin(grad))
        v = np.zeros(n)
        v[i_min] = 1.0
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * v
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s_y
        if t == 1:
            a_new = v.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_a
        y = y_new
        a = a_new
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()

def interpolate_seq(f, new_n):
    """Interpolate (or extrapolate) a sequence to a new length, normalized to sum=1."""
    old_n = len(f)
    if old_n == new_n:
        s = f.sum()
        if s < 1e-15:
            return np.ones(new_n) / new_n
        return f / s
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
