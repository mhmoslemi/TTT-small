2. Multiplicative-weights (mirror descent) with exact subgradient

import numpy as np

from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def compute_subgrad(a, ac, top_k=1):
    """Compute subgradient of max conv(a,a) with respect to a.
    For active lags k where ac[k] is maximal, grad[i] += a[k-i]."""
    n = len(a)
    k_max = min(top_k, len(ac))
    # Find top k lags
    top_idx = np.argsort(ac)[::-1][:k_max]
    # Weight by how close to max
    max_val = ac[top_idx[0]]
    w = np.zeros(k_max)
    for j, k in enumerate(top_idx):
        w[j] = ac[k] / (max_val + 1e-30)
    w /= w.sum()
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        if i_start >= i_end:
            continue
        idx = np.arange(i_start, i_end)
        partner = k - idx
        grad[idx] += wi * a[partner]
    return grad

def mirror_descent(a0, time_limit, eta=0.5, top_k=3, verbose=False):

    Minimizes max_k conv(a,a)_k subject to sum(a)=1, a>=0."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    a = a / a.sum()
    
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Exponential moving average for the iterate
    avg_log_a = np.log(a)
    n_avg = 1
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        ac = fast_autocorr(a)
        M = ac.max()
        
        if M < best_m:
            best_m = M
            best_a = a.copy()
        
        if M < 1e-15:
            break
        
        # Compute subgradient
        g = compute_subgrad(a, ac, top_k=top_k)
        
        # Mirror descent update: a[i] proportional to a[i] * exp(-eta * g[i])
        log_a = np.log(a) - eta * g
        log_a -= log_a.max()  # numerical stability
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-30:
            break
        a = a_new / s
        
        # Track running average in log space

        n_avg += 1
        
        # Periodically check the average

            avg_a = np.maximum(avg_a, 0.0)
            s_avg = avg_a.sum()
            if s_avg > 1e-30:
                avg_a /= s_avg
                M_avg = fast_autocorr(avg_a).max()
                if M_avg < best_m:

def make_ramp(n, power=1.0):
    """Create a ramp-like initial sequence: (i+1)^power normalized."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    s = a.sum()
    if s > 0:
        a /= s
    return a

def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-30:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-30:
        return np.ones(new_n) / new_n
    return f_interp / ss

            s = a.sum()
            if s > 1e-30:
                a /= s
            
            # Quick MD optimization
            a_out, _ = mirror_descent_adaptive(a, min(t_per_shape, remaining * 0.6), verbose=False)
            try_update(a_out, f"shape{shape_i}_n{n_best}")
    
    # === Phase 4: Final coordinate polish ===
    print(f"\n=== Phase 4: Final polish ===")
    remaining = time_budget - (time.monotonic() - t_global)
    
    if remaining > 20:
        t_final = remaining * 0.8
        a_out, _ = coordinate_polish(best_seq, t_final, verbose=False)
        try_update(a_out, "final_polish")
    
    # === Final validation ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
