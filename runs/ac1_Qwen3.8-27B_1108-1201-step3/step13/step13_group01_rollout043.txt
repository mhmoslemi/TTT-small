"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP best-response with Nesterov acceleration.
Key insight: The linearly increasing sequence a[i]=i gives score ~4/3 for large n,
which is far better than the previous initialization. Starting from this and refining
with LP-Nesterov yields near-optimal results.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    return _global_solver

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64).ravel()]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        T[start:end, i] = a[start - i:end - i]
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.arange(n, dtype=np.float64)
        a = a * norm / a.sum()
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    last_improve = 0.0
    stall_count = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()

        if time.monotonic() - last_improve > 60.0:
            stall_count += 1
            if stall_count > 5:
                if verbose:
                    print(f"    Stalling at iter={n_iters}, best={best_score:.8f}")
                break
        else:
            stall_count = 0

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def coordinate_polish(a0, time_limit):
    """Local polish using peak-targeted mass transfer."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a0, fast_autocorr(a0).max()

Key insight: The linearly increasing sequence a[i]=i gives score ~4/3 for large n,

import numpy as np

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), ffmpeg_size) if False else rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        start = j
        end = min(j + n, size)
        T[start:end, j] = a[:end - start]
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def subgrad_nesterov(a0, time_limit, verbose=False):
    """Fast subgradient Nesterov method (no LP)."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.arange(n, dtype=np.float64)
        a = a * norm_val / a.sum()
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(y)
        k_star = int(np.argmax(b))
        M_curr = b[k_star]

        # Subgradient: d(a*a)[k_star]/da[i] = a[k_star-i] + a[i-k_star]
        grad = np.zeros(n)
        for i in range(n):
            val = 0.0
            j1 = k_star - i
            if 0 <= j1 < n:
                val += y[j1]
            j2 = i - k_star
            if 0 <= j2 < n:
                val += y[j2]
            grad[i] = 2.0 * val

        # Move mass to the position with smallest gradient
        i_min = int(np.argmin(grad))
        v = np.zeros(n)
        v[i_min] = norm_val

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * v
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_val / s_y

        if t == 1:
            a_new = v.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm_val / s_a

        y = y_new
        a = a_new

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 500 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm_val / s_avg
                score_avg = fast_autocorr(avg_check).max()

            elapsed = time.monotonic() - t0
            print(f"    SubGrad t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.arange(n, dtype=np.float64)
        a = a * norm_val / a.sum()
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    last_improve = 0.0
    stall_count = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, solver)
        if b is None:
            # Fallback: use current y as best response
            g = y.copy()
        else:
            b_sum = b.sum()
            if b_sum < 1e-15:
                g = y.copy()
            else:
                g = b * norm_val / b_sum

        n_iters += 1

        # Adaptive step size
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_val / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm_val / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()

        if time.monotonic() - last_improve > 90.0:
            stall_count += 1
            if stall_count > 3:
                if verbose:
                    print(f"    Stalling at iter={n_iters}, best={best_score:.8f}")
                break
        else:
            stall_count = 0

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def coordinate_polish(a0, time_limit, seed=12345):
    """Local polish using peak-targeted mass transfer."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()

        return a0, fast_autocorr(a0).max()
    a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        k_star = int(np.argmax(b))

        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_star - idx]
        top_idx = idx[np.argsort(contrib)[::-1][:5]]

        improved = False
        for src in top_idx:
            if a[src] < 1e-10:
                continue
            # Move mass to low-value positions
            low_positions = np.where(a < np.median(a))[0]
            low_positions = low_positions[low_positions != src]
            if len(low_positions) == 0:
                low_positions = np.where(np.arange(n) != src)[0]

            for dst in low_positions[:5]:
                for frac in [0.003, 0.01, 0.03, 0.08]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm_val / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 1000:
                break
            if stall % 100 == 0:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-8:
                    frac = rng.uniform(0.01, 0.05)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t > 1e-10:
                        a_trial = a_trial * norm_val / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0

    return best_a, best_score

    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # ========== Phase 0: Linear initialization ==========
    print("\n=== Phase 0: Linear init ===")
    n_primary = 1600
    a_lin = np.arange(n_primary, dtype=np.float64)
    a_lin = a_lin * (1000.0 / (n_primary - 1))  # max value = 1000
    a_lin_norm = a_lin * np.sqrt(2.0 * n_primary) / a_lin.sum()
    init_score = evaluate_safe(a_lin_norm)
    print(f"  Linear init n={n_primary}: score={init_score:.6f}")
    do_try_update(a_lin_norm, "linear_init")

    # Also try a slightly longer linear sequence
    n_alt = 2000
    a_lin_alt = np.arange(n_alt, dtype=np.float64)
    a_lin_alt = a_lin_alt * (1000.0 / (n_alt - 1))
    a_lin_alt_norm = a_lin_alt * np.sqrt(2.0 * n_alt) / a_lin_alt.sum()
    do_try_update(a_lin_alt_norm, f"linear_n{n_alt}")

    # ========== Phase 1: Fast subgradient Nesterov ==========
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.10
    print(f"\n=== Phase 1: SubGrad Nesterov n={n_primary} (budget={t1:.0f}s) ===")
    if t1 > 10:
        a_out, sc_out = subgrad_nesterov(a_lin_norm, t1, verbose=False)
        do_try_update(a_out, f"subgrad_n{n_primary}")

    # ========== Phase 2: LP Nesterov (primary length) ==========
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq) if best_seq is not None else n_primary
    t2 = remaining * 0.45
    print(f"\n=== Phase 2: LP Nesterov n={n_best} (budget={t2:.0f}s) ===")
    a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
    a_out, sc_out = nesterov_lp_optimize(a_work, t2, solver, verbose=True)
    do_try_update(a_out, f"lp_n{n_best}")

    # ========== Phase 3: LP Nesterov on longer sequence ==========
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = 2400
    t3 = remaining * 0.20
    print(f"\n=== Phase 3: LP Nesterov n={n_long} (budget={t3:.0f}s) ===")
    if t3 > 60:
        a_lin_long = np.arange(n_long, dtype=np.float64)
        a_lin_long = a_lin_long * (1000.0 / (n_long - 1))
        a_lin_long_norm = a_lin_long * np.sqrt(2.0 * n_long) / a_lin_long.sum()
        do_try_update(a_lin_long_norm, f"linear_n{n_long}")
        a_out, sc_out = nesterov_lp_optimize(a_lin_long_norm, t3, solver, verbose=False)
        do_try_update(a_out, f"lp_long_n{n_long}")

    # ========== Phase 4: Coordinate polish ==========
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = remaining * 0.25
    print(f"\n=== Phase 4: Coord polish n={n_best} (budget={t4:.0f}s) ===")
    if t4 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, sc_out = coordinate_polish(a_norm, t4, seed=777)
        do_try_update(a_out, "coord_polish")

    # ========== Phase 5: Final LP polish ==========
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = remaining * 0.50
    print(f"\n=== Phase 5: Final LP n={n_best} (budget={t5:.0f}s) ===")
    if t5 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, sc_out = nesterov_lp_optimize(a_norm, t5, solver, verbose=True)
        do_try_update(a_out, "lp_final")

    # ========== Final output ==========
    if best_seq is not None:
        # Ensure max value <= 1000 to avoid clipping
        max_val = best_seq.max()
        if max_val > 1000.0:
            best_seq = best_seq * (1000.0 / max_val)
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = min(best_score, fs)
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
