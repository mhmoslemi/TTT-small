    """
    LSE objective and gradient w.r.t. z (softmax parametrization).
    f(z) = T * log(sum_k exp(b_k / T)), b = conv(softmax(z), softmax(z))
    Correct gradient: dL/da[m] = 2 * sum_j w[m+j] * a[j]  (NOT r1+r2)
    """
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()

    b = fast_conv(a)
    b_max = b.max()
    log_sum = b_max / T + np.log(np.sum(np.exp((b - b_max) / T)))
    val = T * log_sum

    w = np.exp((b - b_max) / T)
    w /= w.sum()

    # Correct gradient: grad_a[m] = 2 * sum_j w[m+j] * a[j]
    # Computed as 2 * conv(w, a_rev)[n-1 : n-1+n]
    a_rev = a[::-1].copy()
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w, fft_size)
    A_rev = rfft(a_rev, fft_size)
    conv_w_ar = irfft(W * A_rev, fft_size)
    r1 = conv_w_ar[n - 1: n - 1 + n]
    grad_a = 2.0 * r1

    # Chain rule for softmax: grad_z = a * (grad_a - a^T grad_a)
    a_dot_grad = np.dot(a, grad_a)
    grad_z = a * (grad_a - a_dot_grad)

    return val, grad_z


def lbfgs_lse_optimize(a_init, n, time_limit, T_stages=None, max_iter_per_stage=300, verbose=True):
    """
    Multi-stage L-BFGS-B optimization on LSE objective with softmax parametrization.
    """
    if T_stages is None:
        T_stages = [20.0, 12.0, 8.0, 5.0, 3.5, 2.5, 1.8, 1.3, 1.0, 0.7, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1]

    t0 = time.monotonic()
    a_current = np.maximum(a_init, 1e-15).astype(np.float64)
    a_current = a_current / a_current.sum()
    z_current = np.log(a_current)

    best_a = a_current.copy()
    best_score = fast_conv(a_current).max()
    n_evals = 0

    for si, T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2.0:
            break

        # Scale max iterations by remaining time
        iters = max(30, min(max_iter_per_stage, int(remaining * 100)))

        def fun_grad(z):
            nonlocal n_evals
            v, g = lse_obj_and_grad(z, T, n)
            n_evals += 1
            return v, g

        try:
            result = minimize(fun_grad, z_current, method='L-BFGS-B',
                              jac=True,
                              options={'maxiter': iters, 'gtol': 1e-8,
                                       'ftol': 1e-14, 'maxfun': iters * 3})
            z_current = result.x
        except Exception:
            continue

        z_max = z_current.max()
        a_new = np.exp(z_current - z_max)
        s = a_new.sum()
        if s < 1e-30:
            continue
        a_new /= s

        score = fast_conv(a_new).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()

        if verbose and (si % 3 == 0 or si == len(T_stages) - 1):
            elapsed = time.monotonic() - t0
            print(f"      LSE-LBFGS stage={si:2d} T={T:5.2f} ({elapsed:5.0f}s) evals={n_evals:5d}: max_b={best_score:.8f}")

    return best_a, best_score


def subgradient_refine(a_init, n, time_limit, verbose=True):
    """
    Subgradient method on the exact max of conv(a,a).
    Uses adaptive step size and averaging.
    """
    t0 = time.monotonic()
    a = np.maximum(a_init, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s

    best_a = a.copy()
    best_score = fast_conv(a).max()

    avg_a = np.zeros(n)
    n_avg = 0
    it = 0
    eta = 1.0 / best_score

    while time.monotonic() - t0 < time_limit:
        b = fast_conv(a)
        k_star = np.argmax(b)
        max_b = b[k_star]

        # Subgradient: g[m] = 2*a[k_star - m] if 0 <= k_star - m < n
        g = np.zeros(n)
        valid_start = max(0, k_star - n + 1)
        valid_end = min(n - 1, k_star)
        m_range = np.arange(valid_start, valid_end + 1)
        idx = k_star - m_range
        g[m_range] = 2.0 * a[idx]

        # Adaptive step size
        eta = 1.0 / (max_b * (it + 1) ** 0.4)

        a_new = a - eta * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new /= s_new

        a = a_new
        it += 1

        # Track best
        score = fast_conv(a).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()

        # Running average (Polyak averaging)
        avg_a += a
        n_avg += 1
        if n_avg % 20 == 0:
            avg_a_cur = avg_a / n_avg
            s_avg = avg_a_cur.sum()
            if s_avg > 1e-15:
                avg_a_cur /= s_avg
                score_avg = fast_conv(avg_a_cur).max()
                if score_avg < best_score - 1e-12:
                    best_score = score_avg
                    best_a = avg_a_cur.copy()

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SubGrad iter={it:5d} ({elapsed:5.0f}s): max_b={best_score:.8f}")

    return best_a, best_score


def local_search_peak(a_init, n, time_limit, seed=42, verbose=True):
    """
    Gradient-guided peak-targeting local search.
    Moves mass from positions contributing to peaks toward positions that don't.
    """
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s

    best_a = a.copy()
    best_score = fast_conv(a).max()
    stall = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_conv(a)
        # Get top peak positions
        sorted_ks = np.argsort(b)[::-1]
        top_ks = sorted_ks[:min(5, len(sorted_ks))]
        max_b = b[top_ks[0]]

        # Build a "peak contribution" mask: positions that contribute to top peaks
        peak_contrib = np.zeros(n)
        for k in top_ks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for m in range(js, je + 1):
                peak_contrib[m] += a[m] * a[k - m] * b[k] / max_b

        # Source: positions with high peak contribution
        src_candidates = np.argsort(peak_contrib)[::-1][:10]
        # Destination: positions with low peak contribution and low values
        dst_candidates = np.argsort(peak_contrib)[:n // 3]

        improved = False
        for src in src_candidates:
            if a[src] < 1e-12:
                continue
            for dst in dst_candidates:
                if dst == src:
                    continue
                # Try different transfer fractions
                for frac in [0.001, 0.003, 0.008, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-14:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-14:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-14:
                        continue
                    a_trial /= s_t
                    score = fast_conv(a_trial).max()
                    if score < best_score - 1e-10:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        it += 1
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape local minima
            if rng.integers(3) == 0:
                # Gradient-guided random perturbation
                b_curr = fast_conv(a)
                k_star = np.argmax(b_curr)
                g = np.zeros(n)
                js = max(0, k_star - n + 1)
                je = min(n - 1, k_star)
                m_range = np.arange(js, je + 1)
                idx = k_star - m_range
                g[m_range] = 2.0 * a[idx]
                # Move mass opposite to gradient
                perm = rng.permutation(n)
                for idx_m in perm[:5]:
                    if g[idx_m] > 1e-6:
                        dst_idx = perm[len(perm) - 1]
                        amt = a[idx_m] * rng.uniform(0.002, 0.02)
                        a_trial = a.copy()
                        a_trial[idx_m] -= amt
                        a_trial[dst_idx] += amt
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t > 1e-14:
                            a_trial /= s_t
                            score = fast_conv(a_trial).max()
                            if score < best_score:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                stall = 0
                                break

            if stall > 100:
                if verbose:
                    print(f"      LocalSearch: stalled at max_b={best_score:.8f} ({time.monotonic()-t0:.0f}s)")
                break

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LocalSearch it={it:5d} ({elapsed:5.0f}s): max_b={best_score:.8f}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_init(n, seed, parent=None):
    """Generate diverse initializations."""
    rng = np.random.default_rng(seed)
    choice = rng.integers(0, 7)
    if choice == 0 and parent is not None:
        return interpolate_seq(parent, n)
    elif choice == 0:
        return np.ones(n) / n
    elif choice == 1:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 3:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.2 * np.cos(2 * x) + 0.1 * np.cos(5 * x) + 0.05 * np.cos(11 * x)
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-8 * (x - 0.3) ** 2) + 0.8 * np.exp(-8 * (x - 0.7) ** 2)
    elif choice == 5:
        a = rng.dirichlet(np.ones(n) * 0.5)
    else:
        # Smooth random
        a = np.ones(n) * 0.5
        for _ in range(rng.integers(2, 6)):
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.05, n * 0.25)
            amp = rng.uniform(-0.4, 0.6)
            a += amp * np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Uses corrected LSE gradient with L-BFGS-B,
    subgradient refinement, and local search across multiple lengths.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    best_seq = None
    best_score = float('inf')
    parent = None

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: score={sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)
    print(f"Starting optimization. Best so far: {best_score:.6f} at n={n_base}")

    # Define lengths to try
    lengths = sorted(set([
        n_base,
        max(200, n_base - 400),
        n_base + 200,
        n_base + 400,
        max(400, n_base // 2),
    ]))
    # Filter reasonable lengths
    lengths = [l for l in lengths if 100 <= l <= 3500]

    total_phases = len(lengths) * 2 + 2  # L-BFGS + subgrad per length, + final polish
    phase_budget = time_budget / total_phases

    # Phase 1-2N: For each length, run L-BFGS-B then subgradient
    for n in lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 30:
            break

        n_starts = 3 if n <= 2000 else 2
        n_budget_lse = min(remaining * 0.6, 400.0)
        n_budget_sub = min(remaining * 0.3, 150.0)

        print(f"\n=== Length n={n} (remaining={remaining:.0f}s) ===")

        # Try multiple initializations
        for start_i in range(n_starts):
            remaining_inner = n_budget_lse - (n_starts - 1 - start_i) * (n_budget_lse / n_starts * 0.3)
            if remaining_inner < 10:
                break

            seed_i = seed + n * 100 + start_i
            init = make_init(n, seed_i, parent)

            a_out, _ = lbfgs_lse_optimize(
                init, n,
                time_limit=remaining_inner,
                T_stages=[15.0, 8.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08],
                max_iter_per_stage=200,
                verbose=(start_i == 0)
            )
            try_update(a_out, f"lbfgs_n{n}_s{start_i}")

        # Subgradient refinement on the best found at this length
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            n_best = len(best_seq)
            if n_best == n:
                a_work = best_seq.copy()
            else:
                a_work = interpolate_seq(best_seq, n)

            t_sub = min(remaining * 0.2, n_budget_sub)
            if t_sub > 10:
                a_out, _ = subgradient_refine(a_work, n, t_sub, verbose=False)
                try_update(a_out, f"subgrad_n{n}")

    # Final phase: intensive optimization at the best length found
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_final = len(best_seq)
        print(f"\n=== Final intensive optimization at n={n_final} (remaining={remaining:.0f}s) ===")

        # L-BFGS-B with fine temperature schedule
        t_lse_final = min(remaining * 0.4, 400.0)
        if t_lse_final > 10:
            init_f = best_seq.copy()
            a_out, _ = lbfgs_lse_optimize(
                init_f, n_final,
                time_limit=t_lse_final,
                T_stages=[5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.4, 0.3, 0.22, 0.16, 0.12, 0.09, 0.07, 0.05, 0.04, 0.03, 0.02, 0.015, 0.01],
                max_iter_per_stage=150,
                verbose=True
            )
            try_update(a_out, "lbfgs_final")

        # Subgradient refinement
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t_sub_final = min(remaining * 0.3, 200.0)
            if t_sub_final > 10:
                a_out, _ = subgradient_refine(best_seq, len(best_seq), t_sub_final, verbose=True)
                try_update(a_out, "subgrad_final")

        # Local search
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t_ls = min(remaining * 0.5, 200.0)
            if t_ls > 5:
                a_out, _ = local_search_peak(best_seq, len(best_seq), t_ls, seed=77777, verbose=True)
                try_update(a_out, "local_final")

        # One more L-BFGS-B polish
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 10:
            t_polish = min(remaining * 0.6, 100.0)
            if t_polish > 5:
                a_out, _ = lbfgs_lse_optimize(
                    best_seq, len(best_seq),
                    time_limit=t_polish,
                    T_stages=[1.0, 0.5, 0.3, 0.2, 0.15, 0.1, 0.08, 0.06, 0.05],
                    max_iter_per_stage=200,
                    verbose=True
                )
                try_update(a_out, "polish_final")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
