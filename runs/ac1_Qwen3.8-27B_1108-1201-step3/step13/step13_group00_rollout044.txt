"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LSE Mirror Descent (multiplicative weight updates with log-sum-exp smoothing)
2. Smart peak-directed pair coordinate descent (O(n) per step)
3. Multi-length exploration with interpolation-based transfer
4. Perturbation-based restarts to escape local optima
Key innovations: FFT-based gradient for LSE, targeted pair selection at peak lag,
and adaptive temperature annealing for the LSE approximation.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def lse_gradient(a, w, fft_size):
    """
    Compute gradient of LSE objective: sum_k w_k * conv(a,a)[k]
    where w_k = exp(conv_k / T) / sum(exp(conv/T))
    grad_i = 2 * sum_k w_k * (a[k-i] + a[i-k])
    """
    n = len(a)
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    # sum_k w[k]*a[k-i] = cross-correlation = ifft(FFT(w) * conj(FFT(a)))
    corr = irfft(W * np.conj(A), fft_size)[:n]
    # sum_k w[k]*a[i-k] = convolution = ifft(FFT(a) * FFT(w))
    conv = irfft(A * W, fft_size)[:n]
    return 2.0 * (corr + conv)


def lse_mirror_descent(a0, time_limit, verbose=True):
    """
    LSE Mirror Descent: multiplicative weight updates to minimize
    the log-sum-exp of the autocorrelation (smooth approximation of max).
    """
    n = len(a0)
    S = np.sum(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / a.sum()

    size = 2 * n - 1
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    # Actually we need fft_size >= (2n-1) + n - 1 = 3n-2 for the gradient
    # But for autocorr we only need >= 2n-1. Let's use a single size.
    fft_size = 1 << (3 * n - 1).bit_length()

    best_a = a.copy()
    conv = fast_autocorr(a)
    best_score = conv.max()
    t0 = time.monotonic()

    # Temperature schedule: start high (smooth), anneal to low (sharp)
    n_stages = 12
    total_budget = time_limit
    stage_times = []
    # Allocate more time to later (lower temperature) stages
    weights = np.array([1, 1, 1, 1, 1, 1, 2, 2, 3, 3, 4, 5], dtype=np.float64)
    weights /= weights.sum()
    temps = np.geomspace(50.0, 0.3, n_stages)

    for si in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        T = temps[si]
        stage_budget = time_limit * weights[si]
        t_stage = time.monotonic()

            max_conv = conv.max()

            w = exp_scaled / exp_scaled.sum()

            # Clip to avoid overflow

            log_update = np.clip(log_update, -50.0, 5.0)

            a_new = np.maximum(a_new, 0.0)

def smart_pair_cd(a0, time_limit, rng, verbose=True):
    """
    Smart pair coordinate descent: at each step, find the best (i,j) pair
    that most reduces the peak autocorrelation when transferring mass from i to j.
    Uses O(n) sweep to identify candidates near the peak lag.
    """
    n = len(a0)
    S = np.sum(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / a.sum()

    size = 2 * n - 1
    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        k_star = int(np.argmax(conv))

        # Compute contribution to peak: c[i] = a[k_star - i] + a[i - k_star]
        # (with zero padding for invalid indices)
        c = np.zeros(n)
        for i in range(n):
            v = 0.0
            j1 = k_star - i
            if 0 <= j1 < n:
                v += a[j1]
            j2 = i - k_star
            if 0 <= j2 < n:
                v += a[j2]
            c[i] = v

        # Best i to decrease: high c[i] * a[i] (large contribution to peak)
        # Best j to increase: low c[j] (small increase to peak)
        # Effective gain of transferring delta from i to j:
        #   delta * (c[j] - c[i]) should be negative (reduce peak)
        #   So we want c[i] > c[j]

        # Try top candidates
        n_tries = min(20, n // 10)
        i_candidates = np.argsort(-c * a)[:n_tries]
        j_candidates = np.argsort(c)[:n_tries]

        best_gain = 0
        best_i = -1
        best_j = -1
        best_delta = 0

        for i in i_candidates:
            if best_i >= 0 and c[i] * a[i] < c[best_i] * a[best_i] * 0.3:
                break
            ai = a[i]
            if ai < S * 1e-8:
                continue
            for j in j_candidates:
                if j == i:
                    continue
                gain_per_unit = c[i] - c[j]
                if gain_per_unit <= 0:
                    continue
                # Find optimal delta: minimize max(conv + delta*(change_vector))
                # First order: peak changes by delta * (c[j] - c[i])
                # But other conv values also change. Use a simple upper bound:
                # delta_max = ai (can't transfer more than available)
                # The peak reduction is delta * gain_per_unit
                # But we need to check that no other conv value exceeds current peak
                # For simplicity, use a fraction of ai
                delta = min(ai, gain_per_unit * 0.1)  # Conservative
                # Try a few deltas
                for d_frac in [0.1, 0.25, 0.5, 0.75, 1.0]:
                    d = delta * d_frac
                    if d <= 0 or d > ai:
                        continue
                    # Compute new conv (approximate: only update peak region)
                    a[i] -= d
                    a[j] += d
                    # Quick check: just update the peak
                    new_peak_val = conv[k_star] + d * (c[j] - c[i])
                    # Also check second-order effects at k=2i and k=2j
                    if 2 * i < size:
                        new_peak_val = max(new_peak_val,
                                           conv[2 * i] - 2 * a[i] * d + d * d)
                    if 2 * j < size:
                        new_peak_val = max(new_peak_val,
                                           conv[2 * j] + 2 * a[j] * d + d * d)
                    if new_peak_val < conv[k_star] - 1e-10:
                        candidate_gain = conv[k_star] - new_peak_val
                        if candidate_gain > best_gain:
                            best_gain = candidate_gain
                            best_i = int(i)
                            best_j = int(j)
                            best_delta = d
                    # Revert
                    a[i] += d
                    a[j] -= d

        if best_i >= 0 and best_gain > 0:
            a[best_i] -= best_delta
            a[best_j] += best_delta
            s = a.sum()
            if s < 1e-15:
                a[best_i] += best_delta
                a[best_j] -= best_delta
                continue
            a = a * S / s
            conv = fast_autocorr(a)
            cur_max = conv.max()
            if cur_max < best_score - 1e-12:
                best_score = cur_max
                best_a = a.copy()
                pass_improved = True
                no_improve = 0
            else:
                # Revert
                a[best_j] -= best_delta
                a[best_i] += best_delta
                a = a * S / a.sum()
                conv = fast_autocorr(a)
                no_improve += 1
        else:
            # No good pair found, try random pairs
            for _ in range(5):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < S * 1e-9:
                    continue
                delta = a[i] * 0.1
                a[i] -= delta
                a[j] += delta
                a = a * S / a.sum()
                conv_new = fast_autocorr(a)
                if conv_new.max() < best_score - 1e-12:
                    best_score = conv_new.max()
                    best_a = a.copy()
                    conv = conv_new
                    pass_improved = True
                    no_improve = 0
                    break
                else:
                    a[j] -= delta
                    a[i] += delta
                    a = a * S / a.sum()
            else:
                no_improve += 1

        if not pass_improved:
            if no_improve >= 8:
                break

        if verbose and n_passes % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def projected_grad_descent(a0, time_limit, verbose=True):
    """Projected gradient descent on LSE of autocorrelation."""

def _project_simplex(v, S):
    """Project vector v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(1 - u / np.arange(1, n + 1) * 0 + (u - (cssv - S) / np.arange(1, n + 1)) > 0)
    # Standard projection:
    # Find tau such that sum(max(v - tau, 0)) = S
    # Use bisection
    lo = v.min() - S
    hi = v.max()
    for _ in range(50):
        mid = (lo + hi) / 2
        s = np.sum(np.maximum(v - mid, 0))
        if s > S:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2
    return np.maximum(v - tau, 0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def multiplicative_perturb(a, rng, strength=0.1):

    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Smart Pair CD n={n_best} (budget={min(remaining, 250):.0f}s) ===")
    t3 = min(remaining * 0.18, 250.0)
    if t3 > 15:
        a_out, _ = smart_pair_cd(best_seq, t3, rng, verbose=True)
        do_try_update(a_out, f"spcd_n{n_best}")

    # ============ Phase 4: LSE Mirror Descent at longer length ============
    n2 = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LSE Mirror Descent n={n2} (budget={min(remaining, 250):.0f}s) ===")
    t4 = min(remaining * 0.18, 250.0)
    if t4 > 30:
        init_long = interpolate_sequence(best_seq, n2)
        a_out, _ = lse_mirror_descent(init_long, t4, verbose=True)
        do_try_update(a_out, f"lse_md_n{n2}")

    # ============ Phase 5: Pair CD on best ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Pair CD n={n_best} (budget={min(remaining, 200):.0f}s) ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 15:
        a_out, _ = smart_pair_cd(best_seq, t5, rng, verbose=True)
        do_try_update(a_out, f"spcd2_n{n_best}")

    # ============ Phase 6: Perturbation + LSE restart ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Perturbation + LSE n={n_best} (budget={min(remaining, 150):.0f}s) ===")
    t6 = min(remaining * 0.10, 150.0)
    if t6 > 20:
        for attempt in range(3):
            if t6 - (time.monotonic() - t_global) < 5:
                break
            perturbed = multiplicative_perturb(best_seq, rng, strength=0.05 * (attempt + 1))
            a_out, _ = lse_mirror_descent(perturbed, t6 / 3.0, verbose=False)
            do_try_update(a_out, f"pert_lse_{attempt}_n{n_best}")

    # ============ Phase 7: Try shorter length ============
    n_short = max(800, n_base - 400)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: LSE n={n_short} (budget={min(remaining, 100):.0f}s) ===")
    t7 = min(remaining * 0.08, 100.0)
    if t7 > 20:
        init_short = interpolate_sequence(best_seq, n_short)
        a_out, _ = lse_mirror_descent(init_short, t7, verbose=False)
        do_try_update(a_out, f"lse_n{n_short}")
        a_out2, _ = smart_pair_cd(init_short, min(t7 / 2, 40), rng, verbose=False)
        do_try_update(a_out2, f"spcd_n{n_short}")

    # ============ Phase 8: Final polish with pair CD ============
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final Pair CD n={n_best} (budget={min(remaining, 100):.0f}s) ===")
    t8 = min(remaining * 0.15, 100.0)
    if t8 > 10:
        a_out, _ = smart_pair_cd(best_seq, t8, rng, verbose=True)
        do_try_update(a_out, f"spcd_final_n{n_best}")

    # Final check
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
