def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def internal_score(a):
    """Compute 2*n*max_b assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()


def evaluate_safe(arr):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        sc = evaluate_sequence(a_list)

def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if np.all(v >= 0) and abs(v.sum() - target_sum) < 1e-12:
        return v
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho = np.nonzero(v_sorted * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-12:
        idx = np.argmax(result)
        result[idx] += diff
        result = np.maximum(result, 0)
        s = result.sum()
        if s > 0:
            result = result / s * target_sum
    return result


def compute_gradient(a, w):
    """Compute gradient: grad_j = 2 * sum_k w_k * a_{k-j} (valid indices)."""
    n = len(a)
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    if conv_len > fft_size:
        fft_size *= 2
    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad


def softmax_weights(b, T):
    """Compute softmax weights on b with temperature T."""
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    return exp_Tb / exp_Tb.sum()


def spectral_shaping(a, n_iter=400, alpha=0.3):
    """Iteratively flatten power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = internal_score(a)

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 50.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def multiplicative_weights_optimize(a0, time_limit, verbose=True):
    """
    Mirror descent (multiplicative weights) on the simplex.
    Uses adaptive temperature for the peak softmax.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    t = 0
    base_lr = 0.02
    mom = np.zeros(n)
    beta = 0.7

    # Adaptive temperature: start high (smooth), decrease over time
    T_start = 200.0
    T_end = 5.0
    total_est = int(time_limit * 100)  # rough estimate of iterations

    while time.monotonic() - t0 < time_limit:
        t += 1
        b = fast_autocorr(a)

        # Adaptive temperature
        frac = min(1.0, t / max(1, total_est))
        T = T_start * (T_end / T_start) ** frac
        T = max(T, T_end)

        w = softmax_weights(b, T)
        grad = compute_gradient(a, w)

        # Learning rate with decay
        lr_t = base_lr / (1.0 + 0.005 * t)

        # Momentum
        mom = beta * mom + (1 - beta) * grad
        a_new = a - lr_t * mom
        a_new = project_simplex(a_new)

        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MW t={t} ({elapsed:.0f}s, T={T:.1f}): best={best_score:.6f}")

    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated LP best-response for the zero-sum game."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0
    stall_count = 0
    stall_threshold = 3000

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            stall_count += 1
            if stall_count > 20:
                break
            continue
        stall_count = 0

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = project_simplex(a_new)

        y = y_new
        a = a_new

        score_a = internal_score(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = internal_score(y)
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        avg_a += a
        n_avg += 1
        if n_avg >= 50 and n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            score_avg = internal_score(avg_check)
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()

        if verbose and t % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NL t={t} ({elapsed:.0f}s): best={best_score:.6f}, LP_M={M_val:.8f}")

    return best_a, best_score


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where T[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CBC, cp.GLPK, cp.SCS, cp.ECOS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name, max_iter=50)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def solve_lp_best_response(a_fixed, solver):
    """Solve: min M s.t. (a_fixed*b)_k <= M for all k, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, max_iter=200)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.ECOS, cp.PDLP, cp.CLARABEL, cp.CBC, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, max_iter=300)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def peak_equalization(a0, time_limit, T_adapt=True, verbose=True):
    """
    Top-K peak equalization: try to make the top K peaks as equal as possible
    by moving mass from contributors to the highest peak.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    t = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        b = fast_autocorr(a)
        max_b = b.max()

        # Adaptive temperature: focus more on top peaks as we progress
        if T_adapt:
            T = max(10.0, 200.0 * (best_score / max_b) ** 2)
        else:
            T = 50.0

        w = softmax_weights(b, T)
        grad = compute_gradient(a, w)

        # Find where to move mass TO (min gradient = least increase)
        # and WHERE to take it FROM (max gradient * value = most decrease)
        lr = 0.05 / (1.0 + 0.01 * t)

        a_new = a - lr * grad
        a_new = project_simplex(a_new)

        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        # Periodic: try removing small entries
        if t % 50 == 0:
            threshold = best_score * 1e-10
            a = np.where(a < 1e-12, 0, a)
            s = a.sum()
            if s > 1e-15:
                a = a / s

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      PeakEq t={t} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def aggressive_peak_shaving(a0, time_limit, seed=123):
    """Aggressive coordinate-descent peak shaving with multiple strategies."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:15]
        max_b = b.max()

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            if contrib.sum() < 1e-15:
                continue

            # Strategy 1: Move from max contributor to min-value location
            src_local = np.argmax(contrib)
            src = idx[src_local]

            # Find best destination: minimize increase in all top peaks
            peak_areas = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_areas.update(range(js2, je2 + 1))

            candidates = [j for j in range(n) if j not in peak_areas and j != src]
            if not candidates:
                candidates = [j for j in range(n) if j != src]

            # Score each candidate by how much it would increase the top peaks
            best_dst = None
            best_dst_score = float('inf')
            sample = candidates[:min(200, len(candidates))]
            for dst in sample:
                trial_increase = 0.0
                for k2 in top_k[:5]:
                    if 0 <= k2 - dst < n:
                        trial_increase += a[k2 - dst]
                if trial_increase < best_dst_score:
                    best_dst_score = trial_increase
                    best_dst = dst

            if best_dst is None:
                best_dst = sample[rng.integers(len(sample))]

            for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.03, 0.05]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[best_dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = internal_score(a_trial)
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

            if improved:
                stall = 0
                break

        # Strategy 2: Random perturbation
        if not improved:
            stall += 1
            if stall % 5 == 0:
                i1 = rng.integers(n)
                i2 = rng.integers(n)
                if i1 != i2:
                    frac = rng.uniform(0.002, 0.03)
                    amt = a[i1] * frac
                    if amt > 1e-15:
                        a_trial = a.copy()
                        a_trial[i1] -= amt
                        a_trial[i2] += amt
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        score = internal_score(a_trial)
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0

            if stall > 2000:
                break

        if stall % 100 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"      PShave ({elapsed:.0f}s): best={best_score:.6f}, stall={stall}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def generate_initializations(n, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.linspace(0, 1, n)

    # Raised cosine
    a = (1 + np.cos(2 * np.pi * x)) / 2
    a = np.maximum(a, 1e-10) / a.sum()
    inits.append(a)

    # Sine squared
    a = np.sin(np.pi * x) ** 2
    a = np.maximum(a, 1e-10) / a.sum()
    inits.append(a)

    # Gaussian center
    a = np.exp(-4 * (x - 0.5) ** 2)
    a = a / a.sum()
    inits.append(a)

    # Two bumps
    a = np.exp(-20 * (x - 0.3) ** 2) + np.exp(-20 * (x - 0.7) ** 2)
    a = a / a.sum()
    inits.append(a)

    # Flat top
    a = np.ones(n)
    a[n // 4:3 * n // 4] = 2.0
    a = a / a.sum()
    inits.append(a)

    # Random smooth
    a = np.random.default_rng(rng.integers(10000)).random(n)
    # Smooth it
    from scipy.ndimage import uniform_filter1d
    a = uniform_filter1d(a, size=max(3, n // 50))
    a = np.maximum(a, 0) / a.sum()
    inits.append(a)

    # Low-freq random
    k = np.arange(n)
    a = 1.0 + 0.5 * np.sin(2 * np.pi * 3 * k / n + rng.random() * 2 * np.pi)
    a = a + 0.3 * np.sin(2 * np.pi * 7 * k / n + rng.random() * 2 * np.pi)
    a = np.maximum(a, 0) / a.sum()
    inits.append(a)

    return inits
