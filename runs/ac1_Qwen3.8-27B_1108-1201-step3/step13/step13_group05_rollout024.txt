def fast_autocorr(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def softmax_from_theta(theta):
    """Compute softmax of theta (normalized to sum to 1)."""
    theta_max = theta.max()
    exp_theta = np.exp(theta - theta_max)
    return exp_theta / exp_theta.sum()

def compute_soft_max_objective_and_grad(theta, alpha, n=None):
    """
    Compute the soft-max objective and gradient.
    a = softmax(theta), sum(a) = 1.
    Objective: 2n * (1/alpha) * log(sum_k exp(alpha * ac_k))
    """
    if n is None:
        n = len(theta)
    
    a = softmax_from_theta(theta)
    ac = fast_autocorr(a)
    
    # Soft max
    ac_max = ac.max()
    log_sum_exp = ac_max + np.log(np.sum(np.exp(alpha * (ac - ac_max)))) / alpha
    obj = 2.0 * n * log_sum_exp
    
    # Gradient of soft max w.r.t. ac
    w = np.exp(alpha * (ac - ac_max))
    w = w / w.sum()
    
    # g_a[j] = 2 * sum_{m=0}^{n-1} w[j+m] * a[m]
    # Use FFT for efficiency
    fft_size = 1 << (2 * n - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    ga = 2.0 * irfft(W * np.conj(A), fft_size)[:n]
    
    # Chain rule: d(a_i)/d(theta_j) = a_i * (delta_ij - a_j)
    # d(obj)/d(theta_j) = sum_i (d(obj)/d(a_i)) * a_i * (delta_ij - a_j)
    #                   = 2n * [ga_j * a_j - a_j * sum_i ga_i * a_i]
    #                   = 2n * a_j * (ga_j - dot(ga, a))
    
    dot_ga_a = np.dot(ga, a)
    grad = 2.0 * n * a * (ga - dot_ga_a)
    
    return obj, grad

def optimize_with_lbfgs(a0, alphas, max_iter_total=500, verbose=True):
    """
    Optimize using L-BFGS with multi-scale alpha.
    a0: initial sequence (sums to 1).
    alphas: list of alpha values (increasing).
    """
    n = len(a0)
    theta = np.log(a0 + 1e-30)  # Initial theta
    
    best_obj = 2.0 * n * fast_autocorr(a0).max()
    best_a = a0.copy()
    
    for i, alpha in enumerate(alphas):
        # Reinitialize theta from current best
        theta = np.log(best_a + 1e-30)
        
        def obj_and_grad(theta):
            return compute_soft_max_objective_and_grad(theta, alpha, n)
        
        result = minimize(obj_and_grad, theta, method='L-BFGS-B',
                         jac=True, options={'maxiter': max_iter_total, 'gtol': 1e-8})
        
        a_new = softmax_from_theta(result.x)
        ac = fast_autocorr(a_new)
        true_obj = 2.0 * n * ac.max()

            best_obj = true_obj
            best_a = a_new.copy()
        
        if verbose:
            print(f"    alpha={alpha:.1f}: soft_obj={result.fun:.8f}, true_obj={true_obj:.8f}, iters={result.nit}")
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed (soft-max) objective with L-BFGS optimization
2. Multi-scale alpha annealing (coarse to fine)
3. Multiple diverse initializations and lengths
4. FFT-accelerated gradient computation
5. Coordinate descent polish and momentum-based refinement

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_autocorr(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely call the global evaluate_sequence."""
    try:
        a_list = [float(x) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def softmax_from_theta(theta):
    """Compute normalized softmax of theta (sums to 1)."""
    theta_max = theta.max()
    exp_theta = np.exp(theta - theta_max)
    s = exp_theta.sum()
    if s < 1e-300:
        return np.ones(len(theta)) / len(theta)
    return exp_theta / s

def compute_soft_max_obj_grad(theta, alpha, n=None):
    """
    Compute soft-max objective and gradient w.r.t. theta.
    a = softmax(theta), sum(a)=1.
    obj = 2n * soft_max(ac, alpha)
    """
    if n is None:
        n = len(theta)
    
    a = softmax_from_theta(theta)
    ac = fast_autocorr(a)
    
    # Soft max with log-sum-exp trick
    ac_max_val = ac.max()
    shifted = alpha * (ac - ac_max_val)
    # Clip to avoid overflow
    shifted = np.minimum(shifted, 500.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    soft_max_val = ac_max_val + log_sum / alpha
    obj = 2.0 * n * soft_max_val
    
    # Weights
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(len(ac)) / len(ac)
    else:
        w = w / w_sum
    
    # S_j = sum_{m=0}^{n-1} w[j+m] * a[m]  (cross-correlation, first n elements)
    fft_size = 1 << (2 * n - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    S = irfft(W * np.conj(A), fft_size)[:n]
    
    # grad_j = 4n * a_j * (S_j - dot(S, a))
    dot_S_a = np.dot(S, a)
    grad = 4.0 * n * a * (S - dot_S_a)
    
    return obj, grad

def lbfgs_optimize(a_init, alphas, n, max_iter_per_alpha=300, verbose=True):
    """
    L-BFGS optimization with alpha annealing.
    Returns best (a, obj) found.
    """
    best_a = a_init.copy()
    best_obj = 2.0 * n * fast_autocorr(a_init).max()
    
    for idx, alpha in enumerate(alphas):
        theta = np.log(best_a + 1e-30)
        
        def fun(x):
            return compute_soft_max_obj_grad(x, alpha, n)
        
        try:
            result = minimize(fun, theta, method='L-BFGS-B', jac=True,
                             options={'maxiter': max_iter_per_alpha, 'gtol': 1e-10,
                                      'ftol': 1e-12})
        except Exception as e:
            if verbose:
                print(f"      L-BFGS failed at alpha={alpha}: {e}")
            continue
        
        a_new = softmax_from_theta(result.x)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s
        
        ac = fast_autocorr(a_new)
        true_obj = 2.0 * n * ac.max()
        
        if true_obj < best_obj:
            best_obj = true_obj
            best_a = a_new.copy()
        
        if verbose:
            print(f"      alpha={alpha:8.1f}: soft={result.fun:.8f} true={true_obj:.8f} iter={result.nit} status={result.status}")
    
    return best_a, best_obj

            if time.monotonic() - t0 > time_limit:

                a[src] -= amt
                a[best_dst] += amt

                a = a / a.sum()
                best_m = best_trial_m
                best_a = a.copy()
                improved = True

                break

            # Random restart perturbation

                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-10:

                    a[j] += amt

                    M_new = fast_autocorr(a).max()

    a = np.cos(np.pi * (x - 0.5)) ** 2
    a = a + 1e-15
    inits.append(('cosine', a / a.sum()))
    
    # Gaussian
    a = np.exp(-6.0 * (x - 0.5) ** 2)
    inits.append(('gauss6', a / a.sum()))
    
    a = np.exp(-10.0 * (x - 0.5) ** 2)
    inits.append(('gauss10', a / a.sum()))
    
    # Triangle
    a = np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.0)
    a = a + 1e-15
    inits.append(('triangle', a / a.sum()))
    
    # Uniform
    inits.append(('uniform', np.ones(n) / n))
    
    # Hann window
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = a + 1e-15
    inits.append(('hann', a / a.sum()))
    
    # Power-law shapes
    for p in [1.5, 2.5, 4.0]:
        a = np.maximum(1.0 - np.abs(2.0 * x - 1.0) ** p, 1e-15)
        inits.append((f'pow{p}', a / a.sum()))
    
    # Flat-top (constant with smooth edges)
    edge = 0.1
    a = np.ones(n)
    mask_l = x < edge
    mask_r = x > 1 - edge
    a[mask_l] = 0.5 * (1 + np.cos(np.pi * x[mask_l] / edge))
    a[mask_r] = 0.5 * (1 + np.cos(np.pi * (x[mask_r] - 1 + edge) / edge))
    inits.append(('flat_top', a / a.sum()))
    
    return inits

def propose_candidate(budget_s=1800.0):
    """Main search function: multi-length, multi-init L-BFGS with alpha annealing."""
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)
    
    print(f"Time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if best_seq is None:
        n_base = 1600
        x = np.linspace(0, 1, n_base)
        best_seq = np.cos(np.pi * (x - 0.5)) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")
    
    # Main optimization loop over lengths
    lengths = [1600, 2400, 3200, 4800]
    
    for n in lengths:
        if time.monotonic() - t_global > time_budget - 60:
            break
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        
        print(f"\n{'='*60}")
        print(f"=== n={n} (remaining: {remaining:.0f}s) ===")
        print(f"{'='*60}")
        
        # Generate initializations
        inits = generate_initializations(n, parent)
        
        # Score initializations and pick best ones
        init_scored = []
        for name, a in inits:
            ac = fast_autocorr(a)
            sc = 2.0 * n * ac.max()
            init_scored.append((sc, name, a))
        init_scored.sort(key=lambda x: x[0])
        
        # Try top 4 initializations
        n_tries = min(4, len(init_scored))
        time_per_init = (remaining - 30) / (n_tries + 2)  # +2 for polish
        
        for try_idx in range(n_tries):
            if time.monotonic() - t_global > time_budget - 30:
                break
            
            sc_init, name, a_init = init_scored[try_idx]
            label = f"n{n}_{name}"
            
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining < 20:
                break
            
            print(f"\n  [{label}] init_score={sc_init:.8f}")
            
            # Phase A: L-BFGS with alpha annealing
            alphas = [1.0, 3.0, 10.0, 50.0, 200.0, 1000.0]
            max_iter = max(100, int(80 * time_per_init))
            max_iter = min(max_iter, 500)
            
            t_start = time.monotonic()
            a_out, obj_out = lbfgs_optimize(a_init, alphas, n, max_iter_per_alpha=max_iter, verbose=True)
            try_update(a_out, label + "_lbfgs")
            
            # Phase B: Momentum fine-tuning on the best
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 30:
                a_best_local = a_out if obj_out < sc_init else a_init
                # Use the current best at this length
                if best_seq is not None and len(best_seq) == n:
                    a_best_local = best_seq
                t_mom = min(remaining * 0.3, 60.0)
                if t_mom > 10:
                    a_mom, _ = momentum_optimize(a_best_local, 500.0, n, time_limit=t_mom,
                                                 lr=0.002, momentum=0.92, verbose=False)
                    try_update(a_mom, label + "_momentum")
            
            # Phase C: Coordinate polish
            remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 20:
                a_for_polish = best_seq if (best_seq is not None and len(best_seq) == n) else a_out
                t_polish = min(remaining * 0.2, 40.0)
                if t_polish > 5:
                    a_coord, _ = coordinate_polish(a_for_polish, time_limit=t_polish, verbose=True)
                    try_update(a_coord, label + "_coord")
        
        # Post-length polish: coordinate descent on best at this length
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20 and best_seq is not None and len(best_seq) == n:
            t_polish = min(remaining * 0.2, 60.0)
            if t_polish > 5:
                print(f"\n  [n{n}_polish]")
                a_coord, _ = coordinate_polish(best_seq, time_limit=t_polish, verbose=True)
                try_update(a_coord, f"polish_n{n}")
    
    # Final fine-tuning on the overall best
    if time.monotonic() - t_global < time_budget - 20 and best_seq is not None:
        n_best = len(best_seq)
        remaining = time_budget - (time.monotonic() - t_global)
        print(f"\n{'='*60}")
        print(f"=== Final fine-tuning n={n_best} (remaining: {remaining:.0f}s) ===")
        print(f"{'='*60}")
        
        # High-alpha L-BFGS
        alphas_fine = [200.0, 1000.0, 5000.0, 20000.0]
        max_iter = min(300, max(100, int(remaining * 0.3)))
        a_out, _ = lbfgs_optimize(best_seq, alphas_fine, n_best, max_iter_per_alpha=max_iter, verbose=True)
        try_update(a_out, "final_fine")
        
        # Momentum with high alpha
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 30:
            a_mom, _ = momentum_optimize(best_seq, 2000.0, n_best, time_limit=min(remaining * 0.3, 60.0),
                                         lr=0.001, momentum=0.95, verbose=True)
            try_update(a_mom, "final_momentum")
        
        # Final coordinate polish
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_coord, _ = coordinate_polish(best_seq, time_limit=min(remaining * 0.5, 60.0), verbose=True)
            try_update(a_coord, "final_coord")
    
    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\n{'='*60}")
        print(f"FINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
        print(f"{'='*60}")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
    
    return [float(x) for x in best_seq]
