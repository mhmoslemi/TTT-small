"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print(f"  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_lse_gradient(a, conv, T):
    """Compute gradient of LSE approximation using FFT. O(n log n)."""
    n = len(a)
    size = 2 * n - 1
    max_conv = conv.max()
    exp_conv = np.exp((conv - max_conv) / T)
    s = exp_conv.sum()
    if s < 1e-300:
        return np.zeros(n)
    w = exp_conv / s
    
    # grad[i] = 2 * sum_m w[m+i] * a[m] for i=0..n-1
    # This is 2 * cross_correlation(w, a)[i]
    fft_size = 1 << (size + n - 1 - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    cross = irfft(W * np.conj(A), fft_size)
    grad = 2.0 * cross[:n]
    return grad


def evaluate_safe(a):

def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver, active_set=None, tol=0.05):
    """LP: max sum(b) s.t. conv(a,b) <= M, b >= 0. With constraint screening."""
    n = len(a)
    size = 2 * n - 1
    conv_current = fast_autocorr(a)
    
    if active_set is None or len(active_set) < 5:
        # Select active constraints
        if tol >= 1.0:
            active_set = np.arange(size)
        else:
            active_set = np.where(conv_current > M * (1.0 - tol))[0]
            if len(active_set) < 10:
                active_set = np.where(conv_current > M * 0.7)[0]
            if len(active_set) < 5:
                active_set = np.where(conv_current > M * 0.5)[0]
    
    b_var = cp.Variable(n, nonneg=True)
    T = build_conv_matrix(a)
    
    if len(active_set) >= size:
        constraints = [T @ b_var <= M]
    else:
        constraints = [T[active_set, :] @ b_var <= M]
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, active_set
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None, active_set
    
    # Check if dropped constraints are violated
    if len(active_set) < size:
        full_conv = T @ b_val
        violated = np.where(full_conv > M * (1 + 1e-4))[0]
        if len(violated) > 0:
            new_active = np.union1d(active_set, violated)
            if len(new_active) > 0.85 * size:
                # Re-solve with all constraints
                b_var2 = cp.Variable(n, nonneg=True)
                constraints2 = [T @ b_var2 <= M]
                prob2 = cp.Problem(cp.Maximize(cp.sum(b_var2)), constraints2)
                try:
                    prob2.solve(solver=solver, verbose=False)
                except Exception:
                    return b_val, active_set
                if prob2.status in ('optimal', 'optimal_inaccurate'):
                    b_val = np.maximum(np.asarray(b_var2.value, dtype=np.float64), 0.0)
                    active_set = np.arange(size)
                else:
                    active_set = new_active
            else:
                active_set = new_active
    
    return b_val, active_set


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    n_iters = 0
    active_set = None
    t0 = time.monotonic()

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b, active_set = lp_best_response(y, M_curr, solver, active_set)
        if b is None:
            active_set = None
            b, active_set = lp_best_response(y, M_curr, solver, None, tol=1.0)
            if b is None:
                continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if n_iters % 5 == 0:
            score_a = fast_autocorr(a).max()
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def fft_lse_optimize(a0, time_limit, verbose=True):
    """FFT-based LSE smooth gradient descent with adaptive temperature."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    # Anneal temperature from high to low
    n_stages = 12
    temps = np.geomspace(50.0, 0.1, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break
        
        # Adaptive number of steps per stage
        n_steps = max(20, min(200, int(remaining * 4 / (n_stages - si))))
        
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            
            conv = fast_autocorr(a)
            grad = compute_lse_gradient(a, conv, T)
            
            # Adaptive step size
            step = norm / (T * n) / (1.0 + 0.02 * it)
            # Line search: try a few step sizes
            best_step = step
            best_new_score = conv.max()
            for mult in [step, step * 0.5, step * 0.25, step * 2.0]:
                a_try = a - mult * grad
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try * norm / s_try
                sc = fast_autocorr(a_try).max()
                if sc < best_new_score:
                    best_new_score = sc
                    best_step = mult
            
            a = a - best_step * grad
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = a.copy()
            else:
                a = a * norm / s

            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE T={T:.2f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pair_coordinate_descent(a, time_limit, rng, verbose=True):
    """Pair swap coordinate descent for final polish."""
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1

    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-9:
                continue

            # Try a few random partners
            for attempt in range(2):
                j = int(rng.integers(n))
                if j == i:
                    continue

                ai = a[i]
                # Compute gain vector
                g = np.zeros(size)
                # Contribution from a[i]: removing delta from position i
                # g[k] includes -a[k-i] - a[i-k] from position i
                # and +a[k-j] + a[j-k] from position j

                # From position i (subtracting):
                lo1 = i
                hi1 = min(size, i + n)
                if hi1 > lo1:
                    g[lo1:hi1] -= a[:hi1 - lo1]
                lo2 = 0
                hi2 = min(size, i + 1)
                if hi2 > lo2:
                    g[lo2:hi2] -= a[i - lo2:i - lo2 + hi2 - lo2]

                # From position j (adding):
                lo3 = j
                hi3 = min(size, j + n)
                if hi3 > lo3:
                    g[lo3:hi3] += a[:hi3 - lo3]
                lo4 = 0
                hi4 = min(size, j + 1)
                if hi4 > lo4:
                    g[lo4:hi4] += a[j - lo4:j - lo4 + hi4 - lo4]

                # Evaluate at several delta values
                best_d = 0.0
                best_val = conv.max()
                n_samples = 15
                deltas = np.linspace(0.0, ai * 0.8, n_samples)
                for d in deltas[1:]:
                    new_max = (conv + d * g).max()
                    # Second order corrections
                    if 2 * i < size:
                        v2i = conv[2 * i] - 2 * ai * d + d * d
                        new_max = max(new_max, v2i)
                    if 2 * j < size:
                        v2j = conv[2 * j] + 2 * a[j] * d + d * d
                        new_max = max(new_max, v2j)
                    if new_max < best_val - 1e-13:
                        best_val = new_max
                        best_d = d

                if best_d > 0:
                    a[i] -= best_d
                    a[j] += best_d
                    s = a.sum()
                    a = a * norm / s
                    conv = fast_autocorr(a)
                    cur_max = conv.max()
                    if cur_max < best_score - 1e-13:
                        best_score = cur_max
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0
                    else:
                        a[j] -= best_d
                        a[i] += best_d
                        a = a * norm / a.sum()
                        conv = fast_autocorr(a)

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def hybrid_optimize(a0, time_limit, solver, rng, verbose=True):
    """Alternate between LP best-response and FFT gradient steps."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Alternate: 3 LP steps, 10 gradient steps, repeat
    n_lp_per_cycle = 3
    n_grad_per_cycle = 10
    gradient_T = 5.0
    cycle = 0
    active_set = None

    while time.monotonic() - t0 < time_limit:
        cycle += 1

        # LP steps
        for _ in range(n_lp_per_cycle):
            if time.monotonic() - t0 > time_limit:
                break
            M_curr = fast_autocorr(a).max()
            b, active_set = lp_best_response(a, M_curr, solver, active_set)
            if b is not None and b.sum() > 1e-15:
                g_lp = b * norm / b.sum()
                # Small step toward LP solution
                alpha = 0.3
                a_new = (1 - alpha) * a + alpha * g_lp
                a_new = np.maximum(a_new, 0.0)
                s = a_new.sum()
                if s > 1e-15:
                    a_new = a_new * norm / s
                    sc = fast_autocorr(a_new).max()
                    if sc < best_score:
                        best_score = sc
                        best_a = a_new.copy()
                    a = a_new

        # Gradient steps
        for _ in range(n_grad_per_cycle):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_autocorr(a)
            grad = compute_lse_gradient(a, conv, gradient_T)
            step = norm / (gradient_T * n)
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new * norm / s
                sc = fast_autocorr(a_new).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                a = a_new

        # Gradually decrease temperature
        gradient_T *= 0.95

        # Periodic averaging
        if cycle % 10 == 0:
            sc_best = fast_autocorr(best_a).max()
            if verbose and cycle % 20 == 0:
                elapsed = time.monotonic() - t0
                print(f"    Hybrid cycle={cycle} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * x)  # Exponential decay (asymmetric)
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (1.0 - x))  # Reverse exponential
    elif choice == 2:
        x = np.linspace(0.0, 1.0, n)
        a = np.cos(np.pi * x / 2.0) ** 2  # Cosine taper
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 4:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x)
    elif choice == 5:
        x = np.linspace(0.0, 1.0, n)
        a = x ** 0.5 * (1 - x) ** 1.5  # Skewed beta-like
    elif choice == 6:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - x) ** 0.5 * x ** 1.5  # Opposite skew
    else:
        x = np.linspace(0.0, 1.0, n)
        a = 0.3 + 0.7 * np.exp(-3.0 * x) * np.cos(2 * np.pi * x * 3)
        a = np.maximum(a, 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')
    parent = None

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")
    else:
        n_base = len(best_seq)

    # ============================================================
    # Phase 1: LP Nesterov at base length (main workhorse)
    # ============================================================
    remaining = time_budget - (time.monotonic() - t_global)
    n1 = n_base
    print(f"\n=== Phase 1: LP Nesterov n={n1} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.30, 450.0)
    if t1 > 30:
        a_norm = best_seq.copy()
        s_a = a_norm.sum()
        if s_a > 1e-15:
            a_norm = a_norm * np.sqrt(2.0 * n1) / s_a
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n1}")

    # ============================================================
    # Phase 2: FFT-based LSE gradient descent
    # ============================================================
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: FFT LSE gradient n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.18, 250.0)
    if t2 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = fft_lse_optimize(a_norm, t2, verbose=True)
        do_try_update(a_out, f"lse_n{n_best}")

    # ============================================================
    # Phase 3: Hybrid LP + Gradient
    # ============================================================
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Hybrid n={n_best} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = hybrid_optimize(a_norm, t3, solver, rng, verbose=True)
        do_try_update(a_out, f"hybrid_n{n_best}")

    # ============================================================
    # Phase 4: Pair coordinate descent (polish)
    # ============================================================
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Pair CD n={n_best} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pair_coordinate_descent(a_norm, t4, rng, verbose=True)
        do_try_update(a_out, f"pcd_n{n_best}")

    # ============================================================
    # Phase 5: Perturbation + re-optimization (escape local optima)
    # ============================================================
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Perturbation restarts n={n_best} (budget={remaining:.0f}s) ===")
    
    n_perturb = 3
    t_per_perturb = remaining / (n_perturb * 2)
    for pidx in range(n_perturb):
        if time.monotonic() - t_global > time_budget - 60:
            break
        
        # Add noise
        noise_scale = 0.05 + 0.03 * pidx
        noise = rng.normal(0, noise_scale, size=len(best_seq))
        a_pert = best_seq + noise * best_seq.sum() / len(best_seq)
        a_pert = np.maximum(a_pert, 0.0)
        s_p = a_pert.sum()
        if s_p < 1e-15:
            continue
        a_pert = a_pert / s_p
        
        # Quick LP optimization from perturbed point
        t_p = min(t_per_perturb, 80.0)
        a_norm = a_pert * np.sqrt(2.0 * n_best) / a_pert.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t_p, solver, verbose=False)
        do_try_update(a_out, f"perturb{pidx}_lp_n{n_best}")
        
        # Quick gradient from perturbed point
        remaining_p = time_budget - (time.monotonic() - t_global)
        t_gp = min(t_per_perturb * 0.5, 40.0)
        if t_gp > 5:
            a_norm2 = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out2, _ = fft_lse_optimize(a_norm2, t_gp, verbose=False)
            do_try_update(a_out2, f"perturb{pidx}_lse_n{n_best}")

    # ============================================================
    # Phase 6: Try longer length
    # ============================================================
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = n_base + 400
    print(f"\n=== Phase 6: LP Nesterov n={n2} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.12, 120.0)
    if t6 > 20:
        init_long = interpolate_sequence(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t6, solver, verbose=False)
        do_try_update(a_out, f"lp_n{n2}")
        
        # Also try gradient on longer
        remaining_6 = time_budget - (time.monotonic() - t_global)
        t6g = min(remaining_6 * 0.05, 50.0)
        if t6g > 5:
            a_out2, _ = fft_lse_optimize(a_norm, t6g, verbose=False)
            do_try_update(a_out2, f"lse_n{n2}")

    # ============================================================
    # Phase 7: Try shorter length
    # ============================================================
    remaining = time_budget - (time.monotonic() - t_global)
    n3 = max(800, n_base - 400)
    print(f"\n=== Phase 7: LP Nesterov n={n3} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.08, 80.0)
    if t7 > 15:
        init_short = interpolate_sequence(best_seq, n3)
        a_norm = init_short * np.sqrt(2.0 * n3) / init_short.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t7, solver, verbose=False)
        do_try_update(a_out, f"lp_n{n3}")

    # ============================================================
    # Phase 8: Final pair CD polish
    # ============================================================
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final Pair CD n={n_best} (budget={remaining:.0f}s) ===")
    t8 = min(remaining * 0.15, 100.0)
    if t8 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pair_coordinate_descent(a_norm, t8, rng, verbose=True)
        do_try_update(a_out, f"pcd_final_n{n_best}")

    # ============================================================
    # Phase 9: Final LP polish
    # ============================================================
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final LP n={n_best} (budget={remaining:.0f}s) ===")
    t9 = min(remaining * 0.15, 80.0)
    if t9 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t9, solver, verbose=False)
        do_try_update(a_out, f"lp_final_n{n_best}")

    # Final check
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
