import numpy as np
import time

def fast_autocorr(a):
    """Compute full autocorrelation using FFT. O(n log n)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def lse_gradient(a, w):
    """Compute gradient of LSE autocorrelation. a: length n, w: length 2n-1."""
    n = len(a)
    a_rev = a[::-1]
    conv_size = 3 * n - 2
    fft_size = 1 << (conv_size - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    AR = np.fft.rfft(a_rev, fft_size)
    conv = np.fft.irfft(W * AR, fft_size)
    return 2.0 * conv[n - 1: 2 * n - 1]

def normalize(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

    """FISTA with LSE smoothing for minimizing max autocorrelation."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    b0 = fast_autocorr(a)
    best_score = b0.max()
    t0 = time.monotonic()
    
    y = a.copy()
    t_mom = 1.0
    stall = 0
    it = 0
    
    T_start = best_score * T_ratio_start
    T_end = best_score * T_ratio_end
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        T = T_start * (T_end / T_start) ** progress
        
        b = fast_autocorr(y)
        max_b = b.max()
        if max_b < 1e-14:
            break
        
        # Softmax weights
        w = np.exp((b - max_b) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum
        
        # Gradient
        grad = lse_gradient(y, w)
        
        # Adaptive step size
        # Lipschitz constant of LSE gradient: roughly 2 * max(a) / T * n + 2
        # Use a simpler heuristic
        L = 2.0 * n  # conservative
        step = 1.0 / L
        
        # FISTA update

        a_new = normalize(a_new)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom**2)) / 2.0
        y_new = a_new + ((t_mom - 1.0) / t_new) * (a_new - y)
        y = normalize(y_new)
        t_mom = t_new
        
        it += 1
        
        # Track best
        score = fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        # Reset momentum periodically
        if it % 50 == 0:
            t_mom = 1.0
            y = a_new.copy()

            y = normalize(y + rng.normal(0, scale / n, n))
            a_new = normalize(a_new + rng.normal(0, scale / n, n))
            t_mom = 1.0
            stall = 0
        
        if verbose and it % 2000 == 0:
            print(f"    FISTA it={it} ({elapsed:.0f}s): T={T:.6f}, best={best_score:.8f}, stall={stall}")

Actually, one more thought. Let me reconsider the problem from the perspective of **frequency domain**.

By the Wiener-Khinchin theorem:

So the autocorrelation is the inverse DFT of the power spectrum |A(ω)|^2.

The constraint (a*a)_k <= t for all k means that the inverse DFT of |A(ω)|^2 is bounded by t everywhere.

By the uncertainty principle, if |A(ω)|^2 is very concentrated in frequency, then the autocorrelation is spread out in time (and vice versa). To minimize the max autocorrelation, we want |A(ω)|^2 to be as "uniform" as possible in frequency.

But we also need a >= 0, which constrains the phase of A(ω).

This suggests a **frequency-domain optimization**:
1. Start with a candidate a
2. Compute A = FFT(a)
3. Modify |A| to make it more uniform (while keeping a >= 0)
4. Compute a_new = IFFT(A_new)
5. Project a_new onto {a >= 0, sum(a) = 1}
6. Repeat

This is essentially the "magnitude-keeping phase retrieval" approach used in optics and signal processing.

If we modify |A| and keep the phase of A fixed, then a_new = IFFT(|A_new| * e^{i*angle(A)}). This a_new might have negative entries. We can project it onto the non-negative orthant and renormalize.

Alternatively, we can use the "alternating projections" method:
1. Project onto {a: autocorr(a) <= t} (hard projection, solved by water-filling)
2. Project onto {a >= 0, sum(a) = 1} (easy: clip and normalize)

But the hard projection onto {autocorr <= t} is exactly the LP we were trying to avoid!

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA with adaptive LSE smoothing (pure FFT-based, no LP)
2. Multiplicative Weights updates as auxiliary
3. Adaptive temperature schedule (hot→cold)
4. Multiple restarts with random perturbations
5. Multi-length exploration with interpolation
6. Final targeted coordinate descent polish
"""

import numpy as np
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a.astype(np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_grad(a, w):
    """Gradient of LSE. a: length n, w: length 2n-1. Returns length n."""
    n = len(a)
    a_rev = a[::-1]
    conv_len = (2 * n - 1) + n - 1  # = 3n-2
    fft_size = 1 << (conv_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    AR = np.fft.rfft(a_rev, fft_size)
    conv = np.fft.irfft(W * AR, fft_size)
    return 2.0 * conv[n - 1: 2 * n - 1]

def normalize_nonneg(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def fista_lse(a0, time_limit, verbose=True):
    n = len(a0)
    a = normalize_nonneg(a0.astype(np.float64).copy())
    best_a = a.copy()
    b0 = fast_autocorr(a)
    best_score = b0.max()
    t0 = time.monotonic()
    
    y = a.copy()
    t_mom = 1.0
    stall = 0
    it = 0
    
    # Temperature schedule
    T_init = max(best_score * 0.05, 1e-6)
    T_final = max(best_score * 0.001, 1e-9)
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)
        # Exponential cooldown
        T = T_init * (T_final / T_init) ** frac
        
        b = fast_autocorr(y)
        max_b = b.max()
        if max_b < 1e-14:
            break
        
        # Softmax weights (numerically stable)
        log_w = (b - max_b) / T
        log_w_max = log_w.max()
        w = np.exp(log_w - log_w_max)
        w /= w.sum()
        
        # Gradient
        grad = compute_grad(y, w)
        
        # Step size: 1/L where L ~ 2 + c*||a||_2^2/T
        a_norm = np.sqrt(np.sum(y**2))
        L_est = 2.0 + 4.0 * a_norm * a_norm / max(T, 1e-10)
        step = 1.0 / L_est
        
        # FISTA update
        a_new = y - step * grad
        a_new = normalize_nonneg(a_new)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom**2)) / 2.0
        y_new = a_new + ((t_mom - 1.0) / t_new) * (a_new - y)
        y = normalize_nonneg(y_new)
        t_mom = t_new
        it += 1
        
        # Evaluate
        score = fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        # Periodic momentum reset
        if it % 100 == 0:
            t_mom = 1.0
            y = a_new.copy()
        
        # Restart on long stall
        if stall > 300:
            rng = np.random.default_rng(it * 13 + n)
            noise_scale = 0.1 / np.sqrt(n)
            y = normalize_nonneg(y + rng.normal(0, noise_scale, n))
            a_new = normalize_nonneg(a_new + rng.normal(0, noise_scale, n))
            t_mom = 1.0
            stall = 0
        
        if verbose and it % 5000 == 0:
            print(f"    FISTA it={it} ({elapsed:.0f}s): T={T:.2e}, best={best_score:.8f}, stall={stall}")
    
    return best_a, best_score


def mw_optimize(a0, time_limit, verbose=True):
    """Multiplicative Weights (exponentiated gradient) optimization."""
    n = len(a0)
    a = normalize_nonneg(a0.astype(np.float64).copy())
    best_a = a.copy()
    b0 = fast_autocorr(a)
    best_score = b0.max()
    t0 = time.monotonic()
    it = 0
    
    T_init = max(best_score * 0.1, 1e-5)
    T_final = max(best_score * 0.005, 1e-8)
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)
        T = T_init * (T_final / T_init) ** frac
        
        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-14:
            break
        
        w = np.exp((b - max_b) / T)
        w /= w.sum()
        
        grad = compute_grad(a, w)
        
        # Multiplicative update: a_i *= exp(-eta * grad_i)
        # Step size: eta = 1 / max(grad) for stability
        max_grad = np.max(np.abs(grad))
        if max_grad < 1e-15:
            break
        eta = 0.5 / max_grad
        
        log_a = np.log(a + 1e-30) - eta * grad
        # Clip to prevent overflow
        log_a = np.clip(log_a, -50, 50)
        a_new = np.exp(log_a)
        a_new = normalize_nonneg(a_new)
        a = a_new
        it += 1
        
        score = b.max()  # Use the b we already computed (approximate)
        # Actually recompute for accuracy
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if verbose and it % 5000 == 0:
            print(f"    MW it={it} ({elapsed:.0f}s): T={T:.2e}, best={best_score:.8f}")
    
    return best_a, best_score


def coordinate_polish(a0, time_limit, verbose=True):
    """Targeted coordinate descent for final polish."""
    n = len(a0)
    rng = np.random.default_rng(42)
    a = normalize_nonneg(a0.astype(np.float64).copy())
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        
        # Find top peak
        k_star = np.argmax(b)
        
        # Find the dominant contributor to (a*a)_{k_star}
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_star - idx]
        
        if contrib.max() < 1e-20:
            break
        
        src = idx[np.argmax(contrib)]
        
        # Find a good destination (low value, not contributing to top peaks)
        top_peaks = np.argsort(b)[::-1][:5]
        peak_mask = np.zeros(n, dtype=bool)
        for k2 in top_peaks:
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            peak_mask[js2:je2 + 1] = True
        
        candidates = np.where(~peak_mask)[0]
        if len(candidates) == 0:
            candidates = np.arange(n)
        
        # Pick destination with lowest value
        dst = candidates[np.argmin(a[candidates])]
        
        if src == dst:
            dst = (src + n // 2) % n
        
        # Try multiple fractions
        improved = False
        for frac in [0.5, 0.3, 0.1, 0.05, 0.02]:
            amt = a[src] * frac
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            
            score = fast_autocorr(a_trial).max()
            if score < best_score:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                break
        
        it += 1
        if not improved and it % 1000 == 0:
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j:
                amt = a[i] * rng.uniform(0.1, 0.3)
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
        
        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Polish it={it} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def interpolate_to_length(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_local(a):
    """Local evaluation (assumes a is valid, non-negative, sum > 0)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * len(a) * b.max() / (s * s)
