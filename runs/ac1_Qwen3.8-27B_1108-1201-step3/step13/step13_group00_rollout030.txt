"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast subgradient Frank-Wolfe with Nesterov acceleration (primary, no LP)
2. LSE-smoothed projected gradient descent (full gradient via FFT)
3. Pair coordinate descent for final polish
4. Multi-length exploration with interpolation transfer
Key: O(n) per FW iteration enables millions of steps in the time budget.

import numpy as np
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        x = np.linspace(0.0, 1.0, n)
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) * (1 + 0.3 * np.cos(4 * np.pi * x))
        a = np.maximum(a, 0)
    else:
        x = np.linspace(0.0, 1.0, n)
        a = 0.4 + 0.4 * np.sin(np.pi * x) + 0.2 * np.sin(2 * np.pi * x)
        a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def project_onto_simplex(v, c):
    """Project v onto {x >= 0, sum(x) = c}."""
    n = len(v)
    # Binary search for theta: sum(max(v_i - theta, 0)) = c
    lo = np.min(v) - c
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.maximum(v - mid, 0))
        if s > c:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2
    return np.maximum(v - theta, 0.0)

def fw_nesterov_subgrad(a0, time_limit, verbose=True):
    """Fast Frank-Wolfe with subgradient + Nesterov acceleration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    n_iters = 0

    for t in range(1, 50000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute subgradient at argmax of autocorrelation of y
        conv_y = fast_autocorr(y)
        k_star = int(np.argmax(conv_y))

        # Subgradient: g_i = 2 * sum over valid indices of (y_{k-i} + y_{i-k})
        # For the max at k_star: g_i = 2 * (y_{k_star - i} if valid + y_{i - k_star} if valid)
        g = np.zeros(n)
        # y_{k_star - i} is valid when 0 <= k_star - i < n, i.e., k_star - n + 1 <= i <= k_star
        lo1 = max(0, k_star - n + 1)
        hi1 = min(n, k_star + 1)
        if hi1 > lo1:
            idx = np.arange(lo1, hi1)
            g[lo1:hi1] += y[k_star - idx]

        # y_{i - k_star} is valid when 0 <= i - k_star < n, i.e., k_star <= i <= k_star + n - 1
        lo2 = max(0, k_star)
        hi2 = min(n, k_star + n)
        if hi2 > lo2:
            idx = np.arange(lo2, hi2)
            g[lo2:hi2] += y[idx - k_star]

        g *= 2.0

        # FW vertex: e_{i*} * norm where i* = argmin(g)
        i_min = int(np.argmin(g))
        v = np.zeros(n)
        v[i_min] = norm

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * v
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = v.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new
        n_iters += 1

        # Check score periodically (every 100 iters to save time)
        if n_iters % 100 == 0:
            score_y = fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()

            score_a = fast_autocorr(a).max()
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()

            avg_a += a
            n_avg += 1
            if n_avg % 50 == 0:
                avg_check = avg_a / n_avg
                s_avg = avg_check.sum()
                if s_avg > 1e-15:
                    avg_check = avg_check * norm / s_avg
                    sc = fast_autocorr(avg_check).max()
                    if sc < best_score:
                        best_score = sc
                        best_a = avg_check.copy()
                avg_a = np.zeros(n)
                n_avg = 0

            if verbose and n_iters % 10000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    FW t={n_iters} ({elapsed:.0f}s): best_max_conv={best_score:.8f}")

    return best_a, best_score

def lse_projected_grad(a0, time_limit, verbose=True):
    """LSE-smoothed projected gradient descent with full gradient via FFT."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    # Anneal temperature from high to low
    n_stages = 12
    temps = np.geomspace(50.0, 0.3, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break

        # Adaptive learning rate
        lr_base = norm / (n * T) * 0.5
        n_steps = max(20, min(500, int(remaining * 5)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_autocorr(a)
            max_conv = conv.max()
            # LSE weights
            exp_conv = np.exp((conv - max_conv) / T)
            w = exp_conv / exp_conv.sum()

            # Gradient of LSE: grad_i = 2 * sum_k w_k * (a_{k-i} + a_{i-k})
            # = 2 * (cross-corr-like terms)
            # Using the fact that sum_k w_k * a_{k-i} for valid k:
            # Let's compute via the convolution matrix approach but vectorized
            # grad_i = 2 * (sum_k w_k * a_{k-i} + sum_k w_k * a_{i-k})
            # The first term: for each i, sum over k where 0 <= k-i < n
            # = sum over valid (k,i) pairs of w_k * a_{k-i}
            # This is: for the full conv c = conv(w, a) of length 2n-1,
            # c[k] = sum_j w_j * a[k-j]
            # sum_k w_k * a_{k-i} = ? (this is correlation, not convolution)
            # Let's use: sum_k w_k * a_{k-i} = sum_j w_{j+i} * a_j = (w shifted by i) dot a
            # In terms of full conv: if we define r = np.zeros(2n-1), r[j] = w[j] for j < n
            # Then correlation at lag i = sum_j w[j+i] * a[j]
            # This equals the full cross-correlation.
            
            # Simpler approach: use the transpose of T (the convolution matrix for a)
            # If T[i,k] = a[k-i] (when valid), then (T^T @ w)[i] = sum_k T[i,k] * w[k] = sum_k a[k-i] * w[k]
            # And T^T @ w gives us the first part.
            # But building T is O(n^2) which is expensive for large n.
            
            # Alternative: use FFT-based approach
            # sum_k w_k * a_{k-i} for i=0..n-1
            # = cross-correlation of w and a at lag i
            # = ifft(fft(w) * conj(fft(a)))[:n] (circular, but valid for our case with padding)
            
            A_fft = np.fft.rfft(a, fft_size)
            W_fft = np.fft.rfft(w, fft_size)
            
            # cross_corr_1[i] = sum_k w[k] * a[k+i] (for i >= 0)
            # = ifft(fft(w) * conj(fft(a)))[i]
            cc1 = np.fft.irfft(W_fft * np.conj(A_fft), fft_size)
            
            # cross_corr_2[i] = sum_k w[k] * a[i-k] = sum_j a[j] * w[i-j]
            # = ifft(fft(a) * conj(fft(w)))[i]
            cc2 = np.fft.irfft(A_fft * np.conj(W_fft), fft_size)
            
            # For our linear (non-circular) case with fft_size >= 2n-1:
            # cc1[i] = sum_k w[k] * a[k+i] for i = 0, 1, ..., n-1
            # But we want sum_k w[k] * a[k-i], which is cc1[-i] or equivalently:
            # sum_k w[k] * a[k-i] = sum_j w[j+i] * a[j] -- this is cc1 at negative lag
            # Actually: cc1[i] = sum_k w[k] a[k+i], so cc1[-i] (mod fft_size) = sum_k w[k] a[k-i]
            # In the FFT: cc1[-i] = cc1[fft_size - i] for i > 0
            
            # Let me redefine:
            # term1[i] = sum_k w[k] * a[k-i] (valid k: max(0,i) <= k <= min(n-1, size-1-i+1))
            # Actually for k-i to be in [0,n-1], we need i <= k <= i+n-1, and k < n
            # So k in [max(0,i), min(n-1, i+n-1)] = [i, min(n-1, i+n-1)] since i >= 0
            # = [i, n-1] (since i+n-1 >= n-1 always when i >= 0)
            # So term1[i] = sum_{k=i}^{n-1} w[k] * a[k-i]
            
            # This is: (w[i:] dot a[:n-i]) for each i
            # = sum_j w[i+j] * a[j] for j=0..n-i-1
            
            # In cross-correlation terms:
            # cc1[fft_size - i] = sum_k w[(k-i) mod fft_size] * a[k]  -- no this is getting confusing
            
            # Let me just use a direct O(n * avg_support) approach or the matrix trick
            # For n=1600, O(n^2) = 2.5M operations which takes ~2ms in numpy
            # So let's just build the gradient directly:
            
            # Actually, let me use a cleaner FFT approach:
            # Define: B = np.zeros(fft_size); B[:n] = w
            # Then sum_k w[k] * a[k-i] for i=0..n-1
            # = the "anti-convolution" or we can think of it as:
            # Place w in first n positions, a in first n positions
            # The linear cross-correlation R[i] = sum_k w[k] a[k+i] (for i >= 0)
            # = ifft(fft(w) * conj(fft(a)))[:n+1]
            # And sum_k w[k] a[k-i] = R[-i] = ifft(fft(w) * conj(fft(a)))[n-1-i] (for the linear part)
            
            # OK let me just do it the simple way for n <= 2000:
            # grad[i] = 2 * (sum_{k=i}^{n-1} w[k]*a[k-i] + sum_{k=0}^{min(i,n-1)} w[k]*a[i-k])
            
            # Vectorized:
            # Part 1: sum_{k=i}^{n-1} w[k]*a[k-i] = sum_j w[i+j]*a[j] for j=0..n-i-1
            # For all i simultaneously, this is a "reversed" convolution
            # = conv(w[::-1], a[::-1])[...][::-1] or similar
            
            # Let me just use numpy's correlate or a loop with vectorized inner products
            # For n=1600, a Python loop of 1600 iterations with numpy dot is fast enough
            
            grad = np.zeros(n)
            for i in range(n):
                # Part 1: sum_{j=0}^{n-1-i} w[i+j] * a[j]
                if i < n:
                    grad[i] += np.dot(w[i:], a[:n-i])
                # Part 2: sum_{k=0}^{min(i, n-1)} w[k] * a[i-k]
                # = sum_{j=max(0,i-n+1)}^{i} w[j] * a[i-j]
                lo = max(0, i - n + 1)
                hi = min(i + 1, n)
                if hi > lo:
                    grad[i] += np.dot(w[lo:hi], a[i - np.arange(lo, hi)])
            grad *= 2.0

            # Projected gradient step
            lr = lr_base / (1.0 + 0.01 * it)
            a_new = a - lr * grad
            # Project onto {a >= 0, sum(a) = norm}
            a_new = project_onto_simplex(a_new, norm)
            a = a_new

            if it % 10 == 0:
                score = fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {si+1}/{n_stages} T={T:.2f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def pair_cd(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent - move mass between pairs to reduce max autocorrelation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1

    n_passes = 0
    no_improve_count = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-9:
                continue

            # Try moving a small amount from i to a random j
            for _ in range(2):
                j = int(rng.integers(n))
                if j == i:
                    continue

                ai = a[i]
                # First-order change in conv when moving delta from i to j:
                # conv_new[k] = conv[k] + delta * (a[j_side contributions] - a[i_side contributions])
                # Specifically:
                # Removing delta from position i changes conv at positions:
                #   k = i+m for m=0..n-1: conv[k] -= delta * a[m] (from a[i]*a[m] term)
                #     wait, conv[k] = sum_j a[j]*a[k-j]
                # d(conv[k])/d(a[i]) = a[k-i] + a[i-k] (for valid indices)
                # So removing delta from i: conv[k] -= delta * (a[k-i] + a[i-k])
                # Adding delta to j: conv[k] += delta * (a[k-j] + a[j-k])
                # Net change: delta * ((a[k-j] + a[j-k]) - (a[k-i] + a[i-k]))

                # Build the gain vector
                gain = np.zeros(size)
                # Contribution from adding to j:
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    gain[lo:hi] += a[max(0, lo-j):hi-j]
                lo2 = max(0, n-1-j)
                hi2 = min(size, n+j)
                if hi2 > lo2:
                    gain[lo2:hi2] += a[n-1-lo2:n-1-hi2][::-1]

                # Contribution from removing from i:
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    gain[lo3:hi3] -= a[max(0, lo3-i):hi3-i]
                lo4 = max(0, n-1-i)
                hi4 = min(size, n+i)
                if hi4 > lo4:
                    gain[lo4:hi4] -= a[n-1-lo4:n-1-hi4][::-1]

                # Find best delta in [0, ai]
                # conv_new = conv + delta * gain
                # We want to minimize max(conv_new)
                # Try a few candidate deltas
                candidates = [ai * f for f in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5]]
                best_d = 0.0
                best_mv = conv.max()
                for d in candidates:
                    mv = (conv + d * gain).max()
                    if mv < best_mv - 1e-13:
                        best_mv = mv
                        best_d = d

                if best_d > 0:
                    a[i] -= best_d
                    a[j] += best_d
                    s = a.sum()
                    a = a * norm / s
                    conv = fast_autocorr(a)
                    if conv.max() < best_score - 1e-13:
                        best_score = conv.max()
                        best_a = a.copy()
                        pass_improved = True
                    else:
                        # Revert
                        a[j] -= best_d
                        a[i] += best_d
                        a = a * norm / a.sum()
                        conv = fast_autocorr(a)

        no_improve_count = 0 if pass_improved else no_improve_count + 1
        if no_improve_count >= 5:
            break

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

def lse_projected_grad(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 12
    temps = np.geomspace(50.0, 0.3, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break

        lr_base = norm / (n * T) * 0.3
        n_steps = max(20, min(500, int(remaining * 10)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_autocorr(a)
            max_conv = conv.max()
            exp_conv = np.exp((conv - max_conv) / T)
            w = exp_conv / exp_conv.sum()

            A = np.fft.rfft(a, fft_size)
            W = np.fft.rfft(w, fft_size)
            # cross_corr(w,a)[i] = sum_k w[k+i]*a[k] = ifft(W * conj(A))[i]

            cw = np.fft.irfft(A * W, fft_size)[:n]
            grad = 2.0 * (cc + cw)

            # Projected gradient step
            lr = lr_base / (1.0 + 0.01 * it)
            a_new = a - lr * grad
            a_new = project_onto_simplex(a_new, norm)
            a = a_new

                if score < best_score:
                    best_score = score
                    best_a = a.copy()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast subgradient Frank-Wolfe with Nesterov acceleration (O(n) per iter, no LP)
2. FFT-based LSE smoothed projected gradient descent (O(n log n) per step)
3. Adaptive pair coordinate descent for final polish
4. Multi-length exploration with interpolation transfer
Key: FFT-based gradient computation enables thousands of LSE steps; FW is
extremely fast per iteration enabling millions of steps in the time budget.

import numpy as np
import time


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        x = np.linspace(0.0, 1.0, n)
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) * (1.0 + 0.3 * np.cos(4.0 * np.pi * x))
        a = np.maximum(a, 0.0)
    else:
        x = np.linspace(0.0, 1.0, n)
        a = 0.4 + 0.4 * np.sin(np.pi * x) + 0.2 * np.sin(2.0 * np.pi * x)
        a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def project_onto_simplex(v, c):
    n = len(v)
    lo = np.min(v) - c
    hi = np.max(v)
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.maximum(v - mid, 0.0))
        if s > c:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    result = np.maximum(v - theta, 0.0)
    return result


def fw_nesterov_subgrad(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    n_iters = 0
    check_interval = 50

    for t in range(1, 200000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        k_star = int(np.argmax(conv_y))

        g = np.zeros(n)
        lo1 = max(0, k_star - n + 1)
        hi1 = min(n, k_star + 1)
        if hi1 > lo1:
            idx = np.arange(lo1, hi1)
            g[lo1:hi1] += y[k_star - idx]
        lo2 = max(0, k_star)
        hi2 = min(n, k_star + n)
        if hi2 > lo2:
            idx = np.arange(lo2, hi2)
            g[lo2:hi2] += y[idx - k_star]
        g *= 2.0

        i_min = int(np.argmin(g))
        v = np.zeros(n)
        v[i_min] = norm

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * v
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = v.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new
        n_iters += 1

        if n_iters % check_interval == 0:
            score_y = fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()
            score_a = fast_autocorr(a).max()
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()

            avg_a += a
            n_avg += 1
            if n_avg % 100 == 0:
                avg_check = avg_a / n_avg
                s_avg = avg_check.sum()
                if s_avg > 1e-15:
                    avg_check = avg_check * norm / s_avg
                    sc = fast_autocorr(avg_check).max()
                    if sc < best_score:
                        best_score = sc
                        best_a = avg_check.copy()
                avg_a = np.zeros(n)
                n_avg = 0

            if verbose and n_iters % 50000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    FW t={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lse_projected_grad(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 15
    temps = np.geomspace(80.0, 0.2, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        n_steps = max(30, min(300, int(remaining * 8)))
        lr_base = norm / (n * T) * 0.4

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_autocorr(a)
            max_conv = conv.max()
            exp_conv = np.exp((conv - max_conv) / T)
            w = exp_conv / exp_conv.sum()

            A = np.fft.rfft(a, fft_size)
            W = np.fft.rfft(w, fft_size)
            cc = np.fft.irfft(W * np.conj(A), fft_size)[:n]
            cw = np.fft.irfft(A * W, fft_size)[:n]
            grad = 2.0 * (cc + cw)

            lr = lr_base / (1.0 + 0.005 * it)
            a_new = a - lr * grad
            a_new = project_onto_simplex(a_new, norm)
            a = a_new

            if it % 10 == 0:
                score = fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()

        if verbose and (si % 3 == 0 or si == n_stages - 1):
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {si+1}/{n_stages} T={T:.2f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pair_cd(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1

    n_passes = 0
    no_improve_count = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-9:
                continue

            for _ in range(3):
                j = int(rng.integers(n))
                if j == i:
                    continue

                gain = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    gain[lo:hi] += a[max(0, lo - j):hi - j]
                lo2 = max(0, n - 1 - j)
                hi2 = min(size, n + j)
                if hi2 > lo2:
                    src = n - 1 - np.arange(lo2, hi2)
                    valid = (src >= 0) & (src < n)
                    gain[lo2:hi2][valid] += a[src[valid]]

                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    gain[lo3:hi3] -= a[max(0, lo3 - i):hi3 - i]
                lo4 = max(0, n - 1 - i)
                hi4 = min(size, n + i)
                if hi4 > lo4:
                    src4 = n - 1 - np.arange(lo4, hi4)
                    valid4 = (src4 >= 0) & (src4 < n)
                    gain[lo4:hi4][valid4] -= a[src4[valid4]]

                candidates = ai * np.array([0.02, 0.05, 0.1, 0.15, 0.25, 0.4])
                best_d = 0.0
                best_mv = conv.max()
                for d in candidates:
                    mv = (conv + d * gain).max()
                    if mv < best_mv - 1e-13:
                        best_mv = mv
                        best_d = d

                if best_d > 1e-15:
                    a[i] -= best_d
                    a[j] += best_d
                    s = a.sum()
                    a = a * norm / s
                    conv = fast_autocorr(a)
                    if conv.max() < best_score - 1e-13:
                        best_score = conv.max()
                        best_a = a.copy()
                        pass_improved = True
                    else:
                        a[j] -= best_d
                        a[i] += best_d
                        a = a * norm / a.sum()
                        conv = fast_autocorr(a)

        no_improve_count = 0 if pass_improved else no_improve_count + 1
        if no_improve_count >= 8:
            break

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def fine_coordinate_descent(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1

    n_rounds = 0

    while time.monotonic() - t0 < time_limit:
        n_rounds += 1
        improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            # Try small increments and decrements
            deltas = ai * np.array([0.01, 0.03, 0.05, -0.01, -0.03, -0.05])
            best_delta = 0.0
            best_mv = conv.max()

            for d in deltas:
                new_val = ai + d
                if new_val < 0:
                    continue
                a[i] = new_val
                s = a.sum()
                if s < 1e-15:
                    a[i] = ai
                    continue
                a = a * norm / s
                new_conv = fast_autocorr(a)
                mv = new_conv.max()
                if mv < best_mv - 1e-14:
                    best_mv = mv
                    best_delta = d
                    best_conv = new_conv.copy()
                    best_a_local = a.copy()
                a = best_a_local if best_delta != 0 else a0
                a[i] = ai
                a = a * norm / a.sum()

            if best_delta != 0.0:
                a = best_a_local.copy()
                conv = best_conv.copy()
                if best_score - conv.max() > 1e-13:
                    best_score = conv.max()
                    best_a = a.copy()
                    improved = True

        if not improved:
            break

        if verbose and n_rounds % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FCD round={n_rounds} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.30, 450.0)
    print(f"\n=== Phase 1: FW subgrad n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_norm = best_seq.copy()
        s_a = a_norm.sum()
        if s_a > 1e-15:
            a_norm = a_norm * np.sqrt(2.0 * n_base) / s_a
        a_out, _ = fw_nesterov_subgrad(a_norm, t1, verbose=True)
        do_try_update(a_out, f"fw_n{n_base}")

    # Phase 2: LSE projected gradient (FFT-based, fast)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 2: LSE grad n={n_best} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_projected_grad(a_norm, t2, verbose=True)
        do_try_update(a_out, f"lse_n{n_best}")

    # Phase 3: Pair CD
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 3: Pair CD n={n_best} (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pair_cd(a_norm, t3, rng, verbose=True)
        do_try_update(a_out, f"pcd_n{n_best}")

    # Phase 4: Try slightly different length
    n4 = n_base + 200
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 4: FW+LSE n={n4} (budget={t4:.0f}s) ===")
    if t4 > 30:
        init_long = interpolate_sequence(best_seq, n4)
        a_norm = init_long * np.sqrt(2.0 * n4) / init_long.sum()
        t4a = t4 * 0.5
        a_out1, _ = fw_nesterov_subgrad(a_norm, t4a, verbose=False)
        do_try_update(a_out1, f"fw_n{n4}")
        remaining4 = t4 - (time.monotonic() - t_global) + t4 - t4a
        t4b = min(t4 * 0.5, time_budget - (time.monotonic() - t_global) - 60)
        if t4b > 15:
            a_norm2 = best_seq * np.sqrt(2.0 * len(best_seq)) / best_seq.sum() if len(best_seq) != n4 else init_long * np.sqrt(2.0 * n4) / init_long.sum()
            a_out2, _ = lse_projected_grad(a_norm2, t4b, verbose=False)
            do_try_update(a_out2, f"lse_n{n4}")

    # Phase 5: Another FW round on best (with perturbation for diversity)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.15, 150.0)
    print(f"\n=== Phase 5: FW restart n={n_best} (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_pert = best_seq + rng.normal(0, 0.01 * best_seq.mean(), n_best)
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert * np.sqrt(2.0 * n_best) / a_pert.sum()
        a_out, _ = fw_nesterov_subgrad(a_pert, t5, verbose=False)
        do_try_update(a_out, f"fw_restart_n{n_best}")

    # Phase 6: Final LSE polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t6 = min(remaining * 0.25, 200.0)
    print(f"\n=== Phase 6: Final LSE n={n_best} (budget={t6:.0f}s) ===")
    if t6 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_projected_grad(a_norm, t6, verbose=True)
        do_try_update(a_out, f"lse_final_n{n_best}")

    # Phase 7: Final pair CD
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t7 = min(remaining * 0.4, 150.0)
    print(f"\n=== Phase 7: Final PCD n={n_best} (budget={t7:.0f}s) ===")
    if t7 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pair_cd(a_norm, t7, rng, verbose=True)
        do_try_update(a_out, f"pcd_final_n{n_best}")

    # Final check
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
