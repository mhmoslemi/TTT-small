- `2d^2` if m=2i=2j=i+j (i.e., i=j, but we skip this case)
- 0 otherwise

This is a piecewise-convex function (each piece is a quadratic in d, and we're taking the max). The minimum can be found by:
1. Finding all "breakpoints" where the argmax changes
2. Checking the minimum of each quadratic piece at its vertex (if in the valid range)
3. Checking the endpoints d=0 and d=a[i]

For efficiency, I'll use a simplified approach:
1. Compute h(d) at d=0 and d=a[i]
2. Find the top few peaks of conv
3. For each pair of top peaks (p,q), compute the d where conv[p]+d*g[p]+quad_p(d) = conv[q]+d*g[q]+quad_q(d)
4. Evaluate h(d) at all these breakpoints
5. Pick the best

Key innovations: FFT-accelerated LSE gradients, breakpoint line search for pair CD,

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print(f"  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def normalize_a(a, n):
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * norm / n
    return a * norm / s


def lse_mirror_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_a(a0.copy(), n)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    # Use larger FFT for cross-correlation (need 3n-2)
    corr_fft_size = 1 << (3 * n - 1).bit_length()

    n_stages = 10
    temps = np.geomspace(40.0, 0.2, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        # Adaptive steps: more at low temperature
        n_steps = max(30, min(500, int(remaining * 4 * (si + 1) / n_stages)))

        # Precompute A_fft for reuse
        A_fft = rfft(a, corr_fft_size)

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_autocorr(a)
            max_conv = conv.max()
            exp_conv = np.exp((conv - max_conv) / T)
            w = exp_conv / exp_conv.sum()

            # Compute gradient via FFT cross-correlation
            W_fft = rfft(w, corr_fft_size)
            corr = irfft(W_fft * np.conj(A_fft), corr_fft_size)
            # grad[i] = 2 * corr[corr_fft_size - i] for i > 0, 2*corr[0] for i=0
            grad = np.zeros(n)
            grad[0] = corr[0]
            if n > 1:
                grad[1:] = corr[corr_fft_size - np.arange(1, n)]
            grad *= 2.0

            # Mirror descent: a[i] <- a[i] * exp(-eta * grad[i])
            eta = 1.0 / (max_conv + 1e-8)
            grad_clipped = np.clip(grad, -200, 200)
            a_new = a * np.exp(-eta * grad_clipped)
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new = a_new * norm / s
            a = a_new
            # Update A_fft periodically
            if it % 5 == 0:
                A_fft = rfft(a, corr_fft_size)

            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE-MD stage {si} T={T:.2f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def build_conv_matrix_rows(a, row_indices):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((len(row_indices), n))
    for idx, m in enumerate(row_indices):
        lo = max(0, m - n + 1)
        hi = min(n, m + 1)
        if hi > lo:
            T[idx, lo:hi] = a[lo:(hi - m)]
    return T


def lp_best_response(a, M, solver, active_rows=None):
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)

    if active_rows is None:
        conv_current = fast_autocorr(a)
        threshold = M * 0.85
        active_rows = np.where(conv_current > threshold)[0]
        if len(active_rows) < 20:
            active_rows = np.where(conv_current > M * 0.5)[0]

    T = build_conv_matrix_rows(a, active_rows)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None

    # Verify full constraint satisfaction
    full_conv = np.convolve(a, b_val)
    if full_conv.max() > M * (1 + 1e-4):
        # Fallback: full LP
        T_full = build_conv_matrix_rows(a, np.arange(size))
        constraints_full = [T_full @ b_var <= M]
        prob2 = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints_full)
        try:
            prob2.solve(solver=solver, verbose=False)
        except Exception:
            return None
        if prob2.status not in ('optimal', 'optimal_inaccurate'):
            return None
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        s = b_val.sum()
        if s < 1e-15:
            return None

    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_a(a0.copy(), n)
    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def compute_gain_vector(a, i, j, size):
    """Compute first-order gain vector for transfer from i to j."""
    n = len(a)
    # g[m] = 2*(a[m-j] - a[m-i]) for valid indices, 0 otherwise
    g = np.zeros(size)
    # a[m-j] valid when 0 <= m-j < n, i.e., m in [j, j+n-1]
    lo_j = j
    hi_j = min(size, j + n)
    if hi_j > lo_j:
        g[lo_j:hi_j] += 2.0 * a[0:hi_j - lo_j]
    # a[m-i] valid when 0 <= m-i < n, i.e., m in [i, i+n-1]
    lo_i = i
    hi_i = min(size, i + n)
    if hi_i > lo_i:
        g[lo_i:hi_i] -= 2.0 * a[0:hi_i - lo_i]
    return g


def pair_cd_line_search(conv, g, a_i, a_j, i, j, size):
    """Find optimal delta in [0, a_i] to minimize max_k(conv[k] + delta*g[k] + quad_k)."""
    n = len(a) if 'a' in dir() else len(conv) // 2 + 1  # not used, just for clarity
    current_max = conv.max()
    best_d = 0.0
    best_val = current_max

    # Candidate delta values
    candidates = [0.0, a_i]

    # Find breakpoints from top peaks
    top_k = np.argsort(conv)[::-1][:15]
    k_max = int(np.argmax(conv))
    for k in top_k:
        k = int(k)
        if k == k_max:
            continue
        dg = g[k] - g[k_max]
        if abs(dg) > 1e-15:
            delta_bp = (conv[k_max] - conv[k]) / dg
            if 0 < delta_bp < a_i:
                candidates.append(float(delta_bp))

    # Add some uniform samples
    if a_i > 0:
        candidates.extend(np.linspace(0, a_i, 8).tolist())

    # Evaluate at candidates
    for d in candidates:
        d = min(max(d, 0.0), a_i)
        if d == 0:
            continue
        # First order
        new_conv = conv + d * g
        # Second order corrections
        if 2 * i < size:
            new_conv[2 * i] += d * d
        if 2 * j < size:
            new_conv[2 * j] += d * d
        if i != j and (i + j) < size:
            new_conv[i + j] -= 2.0 * d * d
        val = new_conv.max()
        if val < best_val - 1e-12:
            best_val = val
            best_d = d

    return best_d, best_val


def pair_coordinate_descent(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = normalize_a(a0.copy(), n)

    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        n_tried = 0
        n_success = 0

        # Number of pairs to try per pass (adaptive)
        n_pairs = 2000 + n_passes * 500
        n_pairs = min(n_pairs, 20000)

        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break

            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or a[i] < norm * 1e-9:
                continue

            g = compute_gain_vector(a, i, j, size)
            best_d, best_val = pair_cd_line_search(conv, g, a[i], a[j], i, j, size)

            if best_d > 1e-12:
                a[i] -= best_d
                a[j] += best_d
                s = a.sum()
                if s < 1e-15:
                    a[i] += best_d
                    a[j] -= best_d
                    continue

                conv = fast_autocorr(a)
                cur_max = conv.max()
