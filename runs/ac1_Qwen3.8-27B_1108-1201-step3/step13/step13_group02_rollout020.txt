            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0

        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, stall={stall}")

        if stall > 300:
            rng = np.random.default_rng(t * 13 + 7)
            if stall > 600:
                perturb_scale = 0.05 * a.mean()
            else:
                perturb_scale = 0.02 * a.mean()
            y = np.maximum(y + rng.normal(0, perturb_scale, n), 0)
            s_y = y.sum()
            if s_y < 1e-15:
                y = a.copy()
            else:
                y = y * norm / s_y
            a = np.maximum(a + rng.normal(0, perturb_scale, n), 0)
            s_a = a.sum()
            if s_a < 1e-15:
                a = y.copy()
            else:
                a = a * norm / s_a
            stall = 0
            warm_b = None
            if verbose:
                print(f"    [Restart at iter {n_iters}]")

    _warm_start_b = None
    return best_a, best_score


def power_sum_gradient(a, p):
    """Compute gradient of Lp norm of autocorrelation (up to constant)."""
    n = len(a)
    c = fast_autocorr(a)
    c_clipped = np.maximum(c, 1e-30)
    # Weight: proportional to c^(p-1)
    w = c_clipped ** (p - 1)
    a_rev = a[::-1]
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)
    conv = irfft(W * AR, fft_size)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad


def power_sum_gd(a0, time_limit, p_start=20, p_end=500, verbose=True):
    """Power-sum gradient descent with annealed exponent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0
    stagnation = 0

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        # Anneal p from p_start to p_end
        p = p_start * (p_end / p_start) ** elapsed_frac

        grad = power_sum_gradient(a, p)
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-30:
            break

        # Adaptive step size
        a_norm = np.linalg.norm(a)
        step = 0.5 * a_norm / (g_norm + 1e-30)
        # Clip step to prevent instability
        step = min(step, 0.1 * a.max() / (g_norm / n + 1e-30) * 0.1)
        step = min(step, 2.0 / n)

        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        a = a_new
        steps += 1

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1

        if stagnation > 200:
            rng = np.random.default_rng(steps * 3 + 1)
            a = np.maximum(a + rng.normal(0, 0.01 * a.mean(), n), 0)
            s = a.sum()
            if s > 1e-15:
                a = a * norm / s
            stagnation = 0

        if verbose and steps % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PS-GD step={steps} ({elapsed:.0f}s): p={p:.1f}, best={best_score:.8f}")

    return best_a, best_score


def subgradient_method(a0, time_limit, verbose=True):
    """Exact subgradient method for the non-smooth max of autocorrelation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0
    step_size = 1.0 / np.sqrt(n)

    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a)
        max_c = c.max()
        # Subgradient: average over all argmax positions
        tol = max_c * 1e-6 + 1e-12
        argmax_mask = c >= max_c - tol
        w = np.zeros_like(c)
        w[argmax_mask] = 1.0 / argmax_mask.sum()

        # Compute subgradient
        a_rev = a[::-1]
        fft_size = 1 << (3 * n - 2).bit_length()
        W = rfft(w, fft_size)
        AR = rfft(a_rev, fft_size)
        conv = irfft(W * AR, fft_size)
        grad = 2.0 * conv[n - 1:2 * n - 1]

        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-30:
            break

        # Polyak-like step size
        a_val = np.linalg.norm(a)
        step = step_size / (steps + 1.0) ** 0.5
        step = min(step, a_val / (g_norm + 1e-30) * 0.5)

        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        a = a_new
        steps += 1

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and steps % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad step={steps} ({elapsed:.0f}s): step={step:.6f}, best={best_score:.8f}")

    return best_a, best_score


def lp_guided_peak_shave(a0, time_limit, solver, seed=42, verbose=True):
    """Peak shaving guided by LP best response for destination selection."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_improve = 0
    n_stall = 0
    T = max(0.1, best_score * 0.05)
    T_min = best_score * 0.001

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        T = max(T_min, T * (0.9995 ** int(elapsed_frac * 10000)))

        c = fast_autocorr(a)
        max_c = c.max()
        top_k = np.argsort(c)[::-1][:6]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            if contrib.max() < 1e-18:
                continue

            src = idx[np.argmax(contrib)]

            # Use LP to find good destinations
            if n_improve % 10 == 0 and time.monotonic() - t0 < time_limit * 0.7:
                M_curr = max_c
                b = lp_solve_single(a, M_curr, solver)
                if b is not None:
                    # Positions where b is large but a is small are good destinations
                    ratio = b / (a + 1e-10)
                    # Exclude source and positions contributing to top peaks
                    exclude = np.zeros(n, dtype=bool)
                    exclude[src] = True
                    for k2 in top_k:
                        js2 = max(0, k2 - n + 1)
                        je2 = min(n - 1, k2)
                        exclude[js2:je2 + 1] = True
                    ratio[exclude] = -np.inf
                    n_dest = min(4, np.sum(ratio > 0))
                    if n_dest > 0:
                        selected_dst = np.argsort(ratio)[::-1][:n_dest]
                    else:
                        non_excl = np.where(~exclude)[0]
                        if len(non_excl) == 0:
                            continue
                        selected_dst = non_excl[rng.choice(len(non_excl), min(4, len(non_excl)), replace=False)]
                else:
                    non_excl = np.where(np.arange(n) != src)[0]
                    selected_dst = non_excl[rng.choice(len(non_excl), min(4, len(non_excl)), replace=False)]
            else:
                non_excl = np.where(np.arange(n) != src)[0]
                # Prefer low-value positions
                sorted_dst = non_excl[np.argsort(a[non_excl])]
                n_pick = min(4, len(sorted_dst))
                selected_dst = sorted_dst[rng.choice(min(n_pick * 2, len(sorted_dst)), n_pick, replace=False)]

            frac = rng.choice([0.03, 0.05, 0.08, 0.12, 0.18])
            amt = a[src] * frac
            if amt < 1e-18:
                continue

            dst_vals = a[selected_dst] + 1e-12
            weights = 1.0 / dst_vals
            weights = weights / weights.sum()

            a_trial = a.copy()
            a_trial[src] -= amt
            for di, dst in enumerate(selected_dst):
                a_trial[dst] += amt * weights[di]
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial * norm / s_t

            score = fast_autocorr(a_trial).max()

            if score < best_score:
                a = a_trial
                best_score = score
                best_a = a.copy()
                n_improve += 1
                improved = True
                break
            elif rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                a = a_trial
                improved = True
                break

        if not improved:
            n_stall += 1
            # Random pair swap
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-10:
                frac = rng.uniform(0.05, 0.2)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        n_improve += 1
                    elif rng.random() < np.exp(-(score - best_score) / max(T, 1e-10)):
                        a = a_trial

            if n_stall > 150:
                idx_pert = rng.choice(n, size=max(2, n // 40), replace=False)
                a_trial = a.copy()
                for idx in idx_pert:
                    amt = a[idx] * rng.uniform(0.1, 0.3)
                    j = rng.integers(n)
                    a_trial[idx] -= amt
                    a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score * 1.005:
                        a = a_trial
                        if score < best_score:
                            best_score = score
                            best_a = a.copy()
                n_stall = 0

        if verbose and n_improve % 1000 == 0 and n_improve > 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-PeakShave improve={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
        elif np.isfinite(sc):
            pass  # report but no update

    # Initialize if needed
    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    # Phase 1: Quick multi-length LP warmup
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Multi-length warmup (budget={remaining:.0f}s) ===")
    t1_total = min(remaining * 0.08, 80.0)
    if t1_total > 20:
        candidate_lengths = [1200, 1400, 1600, 1800, 2000, 2200]
        # Filter to reasonable ones
        candidate_lengths = [cl for cl in candidate_lengths if 800 <= cl <= 2400]
        t_per = t1_total / len(candidate_lengths)
        for cl in candidate_lengths:
            if time.monotonic() - t_global > time_budget:
                break
            if cl == n_base:
                init_cl = best_seq.copy()
            else:
                init_cl = interpolate_sequence(best_seq, cl)
            a_norm = init_cl * np.sqrt(2.0 * cl) / init_cl.sum()
            a_out, _ = nesterov_lp(a_norm, t_per, solver, verbose=False)
            do_try_update(a_out, f"warmup_n{cl}")

    # Phase 2: Main LP Nesterov on best length
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Main LP Nesterov n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.35, 550.0)
    if t2 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp(a_out if False else a_norm, t2, solver, verbose=True)
        do_try_update(a_out, f"lp_main_n{n_best}")

    # Phase 3: Power-sum gradient descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Power-sum GD n={n_best} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.10, 120.0)
    if t3 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = power_sum_gd(a_norm, t3, p_start=20, p_end=400, verbose=True)
        do_try_update(a_out, f"psgd_n{n_best}")

    # Phase 4: Subgradient method
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Subgradient n={n_best} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.08, 100.0)
    if t4 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = subgradient_method(a_norm, t4, verbose=True)
        do_try_update(a_out, f"subgrad_n{n_best}")

    # Phase 5: LP-guided peak shaving
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP PeakShave n={n_best} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.18, 250.0)
    if t5 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_guided_peak_shave(a_norm, t5, solver, seed=99, verbose=True)
        do_try_update(a_out, f"lpshave_n{n_best}")

    # Phase 6: Try a different length (best + 200 or nearest new length)
    remaining = time_budget - (time.monotonic() - t_global)
    n_alt = n_best + 200
    if n_alt > 2400:
        n_alt = n_best - 200
    print(f"\n=== Phase 6: LP at n={n_alt} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 30:
        init_alt = interpolate_sequence(best_seq, n_alt)
        a_norm = init_alt * np.sqrt(2.0 * n_alt) / init_alt.sum()
        a_out, _ = nesterov_lp(a_norm, t6, solver, verbose=True)
        do_try_update(a_out, f"lp_alt_n{n_alt}")

    # Phase 7: Another LP round on the overall best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: LP round 2 n={n_best} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.20, 250.0)
    if t7 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t7, solver, verbose=True)
        do_try_update(a_out, f"lp_r2_n{n_best}")

    # Phase 8: Final peak shaving
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final PeakShave n={n_best} (budget={remaining:.0f}s) ===")
    t8 = min(remaining * 0.25, 200.0)
    if t8 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_guided_peak_shave(a_norm, t8, solver, seed=2024, verbose=True)
        do_try_update(a_out, f"shave_final_n{n_best}")

    # Phase 9: Final power-sum polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final PS-GD n={n_best} (budget={remaining:.0f}s) ===")
    t9 = min(remaining * 0.20, 150.0)
    if t9 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = power_sum_gd(a_norm, t9, p_start=100, p_end=800, verbose=True)
        do_try_update(a_out, f"psgd_final_n{n_best}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
