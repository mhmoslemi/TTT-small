    run_count = 0

    stall_count = 0

            continue

        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_sum = y_new.sum()
        if y_sum < 1e-15:

        else:

def frequency_domain_refine(a0, time_limit):
    """
    Frequency domain refinement using iterative amplitude and phase adjustment.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a.copy(), 2.0 * n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    fft_size = 1 << (2 * n - 1).bit_length()
    
    # Target: flat power spectrum
    # Start with current power spectrum
    for iteration in range(200):
        if time.monotonic() - t0 > time_limit - 3:
            break
        
        A = rfft(a, fft_size)
        P = np.abs(A) ** 2
        
        # Compute target: geometric mean of current P (flattening)
        P_mean = P.mean()
        # Blend: P_target = P * (P_mean / P)^beta = P^(1-beta) * P_mean^beta
        beta = 0.1
        P_target = P * (P_mean / np.maximum(P, 1e-30)) ** beta
        
        # Reconstruct with new amplitude, keep phase
        mag = np.sqrt(np.maximum(P_target, 0))
        phase = A / np.maximum(np.abs(A), 1e-30)
        A_new = mag * phase
        a_new = irfft(A_new, fft_size)[:n]
        
        # Project to non-negative and normalize
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new / sn
        
        sc = 2.0 * n * fast_autocorr(a_new).max()
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

            y = best_a.copy()
            emas = [best_a.copy() for _ in ema_windows]
            run_sum = best_a.copy()
            run_count = 1
            warm_b = None
            stall_count = 0

        if t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LPC t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M_val:.10f}")

    return best_a, best_score


def spectral_shaping(a0, time_limit):
    """
    Iterative spectral shaping: compress the power spectrum to flatten
    the autocorrelation peak.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()

    # Try different compression levels
    alphas = [0.4, 0.5, 0.6, 0.7, 0.8, 0.85, 0.9]

    for alpha in alphas:
        if time.monotonic() - t0 > time_limit - 2:
            break

        a_curr = best_a.copy()
        for iteration in range(80):
            if time.monotonic() - t0 > time_limit - 2:
                break

            A = rfft(a_curr, fft_size)
            P = np.abs(A) ** 2
            P_max = P.max()
            if P_max < 1e-30:
                break
            P_norm = P / P_max
            P_compressed = P_norm ** alpha
            P_total = np.sum(P)
            P_new_sum = np.sum(P_compressed)
            if P_new_sum < 1e-30:
                break
            P_new = P_compressed * P_total / P_new_sum

            mag_new = np.sqrt(np.maximum(P_new, 0))
            phase = A / np.maximum(np.abs(A), 1e-30)
            A_new = mag_new * phase
            a_new = irfft(A_new, fft_size)[:n]
            a_new = np.maximum(a_new, 0)
            sn = a_new.sum()
            if sn < 1e-15:
                break
            a_new = a_new / sn

            sc = 2.0 * n * fast_autocorr(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a_curr = a_new.copy()

    return best_a, best_score


def freq_domain_refine(a0, time_limit):
    """
    Frequency domain refinement: iteratively flatten the power spectrum
    while maintaining non-negativity.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()

    betas = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4]

    for beta in betas:
        if time.monotonic() - t0 > time_limit - 2:
            break

        a_curr = best_a.copy()
        for iteration in range(100):
            if time.monotonic() - t0 > time_limit - 2:
                break

            A = rfft(a_curr, fft_size)
            P = np.abs(A) ** 2
            P_mean = P.mean()
            if P_mean < 1e-30:
                break

            # Blend toward flat spectrum
            P_target = P * (P_mean / np.maximum(P, 1e-30)) ** beta
            mag = np.sqrt(np.maximum(P_target, 0))
            phase = A / np.maximum(np.abs(A), 1e-30)
            A_new = mag * phase
            a_new = irfft(A_new, fft_size)[:n]
            a_new = np.maximum(a_new, 0)
            sn = a_new.sum()
            if sn < 1e-15:
                break
            a_new = a_new / sn

            sc = 2.0 * n * fast_autocorr(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a_curr = a_new.copy()

    return best_a, best_score


def multi_scale_peak_shave(a0, time_limit, seed=42):
    """
    Multi-scale peak shaving: systematically try to reduce the peak
    by moving mass from high-contributing positions to safe ones.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    scales = [0.001, 0.002, 0.005, 0.01, 0.02, 0.04, 0.07, 0.10]
    stuck = 0

    while time.monotonic() - t0 < time_limit - 3:
        c = fast_autocorr(a)
        bmax = c.max()

        n_peaks = min(15, len(c))
        top_peak_idx = np.argsort(c)[::-1][:n_peaks]

        improved = False
        for scale in scales:
            if time.monotonic() - t0 > time_limit - 3:
                break

            for k in top_peak_idx[:8]:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                if len(idx) == 0:
                    continue
                contrib = a[idx] * a[k - idx]
                if contrib.max() < 1e-15:
                    continue
                top_contrib = np.argsort(contrib)[::-1][:3]

                for ci in top_contrib:
                    src = idx[ci]

                    # Find safe destinations
                    danger_count = np.zeros(n)
                    for pk in top_peak_idx[:8]:
                        ds = max(0, pk - n + 1)
                        de = min(n - 1, pk)
                        danger_count[ds:de + 1] += 1

                    safe = np.where(danger_count <= 2)[0]
                    safe = safe[safe != src]
                    if len(safe) == 0:
                        safe = np.where(np.arange(n) != src)[0]
                    safe_sorted = safe[np.argsort(a[safe])]
                    dst_pool = safe_sorted[:8]

                    for dst in dst_pool:
                        amt = a[src] * scale
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < -1e-12:
                            continue
                        a_trial = np.maximum(a_trial, 0)
                        st = a_trial.sum()
                        if st < 1e-12:
                            continue
                        a_trial = a_trial / st
                        new_max = fast_autocorr(a_trial).max()
                        if new_max < bmax * (1 - 1e-9):
                            a = a_trial
                            new_score = 2.0 * n * new_max
                            if new_score < best_score:
                                best_score = new_score
                                best_a = a.copy()
                            improved = True
                            stuck = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stuck += 1
            # Random perturbation to escape
            if stuck > 5:
                n_pert = rng.integers(1, 5)
                a_trial = a.copy()
                for _ in range(n_pert):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a_trial[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.03)
                        amt = a_trial[i] * frac
                        a_trial[i] -= amt
                        a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-10:
                    a_trial = a_trial / st
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stuck = 0
            if stuck > 300:
                break

    return best_a, best_score


def block_lp_descent(a0, time_limit, solver, block_size=25):
    """
    Block coordinate LP descent: optimize each block with a mini-LP
    while fixing the rest of the sequence.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a.copy(), 2.0 * n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    n_blocks = (n + block_size - 1) // block_size
    rounds = 0
    no_improve_rounds = 0

    while time.monotonic() - t0 < time_limit - 5:
        rounds += 1
        round_improved = False

        for bi in range(n_blocks):
            if time.monotonic() - t0 > time_limit - 5:
                break

            js = bi * block_size
            je = min(n, js + block_size)
            block_len = je - js
            block_sum = a[js:je].sum()

            # Get top peaks for constraints
            c = fast_autocorr(a)
            top_peaks = np.argsort(c)[::-1][:25]

            x = cp.Variable(block_len, nonneg=True)
            M_var = cp.Variable(value=c.max())
            constraints = [cp.sum(x) == block_sum]

            for k in top_peaks:
                # For lag k, the block's contribution:
                # sum_{i=0}^{block_len-1} x[i] * a[k - (js+i)]  (valid range only)
                coeffs = np.zeros(block_len)
                fixed = 0.0
                for i in range(block_len):
                    j = js + i
                    kj = k - j
                    if 0 <= kj < n and not (js <= kj < je):
                        coeffs[i] = a[kj]
                    elif 0 <= kj < n and js <= kj < je:
                        # Both in block - quadratic term, use current value as bound
                        fixed += a[j] * a[kj]

                # Fixed contribution from positions outside block
                # c(k) = sum_j a(j)*a(k-j), the part not involving block
                total_k = c[k]
                block_contrib_current = 0.0
                for i in range(block_len):
                    j = js + i
                    kj = k - j
                    if 0 <= kj < n:
                        block_contrib_current += a[j] * a[kj]

                # Upper bound on new block contribution using coeffs
                if coeffs.max() > 0:
                    constraints.append(cp.sum(coeffs * x) + fixed <= M_var)

            prob = cp.Problem(cp.Minimize(M_var), constraints)
            try:
                prob.solve(solver=solver)
            except Exception:
                continue

            if prob.status not in ('optimal', 'optimal_inaccurate'):
                continue

            x_val = np.asarray(x.value, dtype=np.float64)
            x_val = np.maximum(x_val, 0)
            xs = x_val.sum()
            if xs < 1e-15 or block_sum < 1e-15:
                continue
            x_val = x_val * block_sum / xs

            a_trial = a.copy()
            a_trial[js:je] = x_val
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st

            sc_trial = 2.0 * n * fast_autocorr(a_trial).max()
            if sc_trial < best_score * (1 - 1e-10):
                a = a_trial
                best_score = sc_trial
                best_a = a.copy()
                round_improved = True

        if round_improved:
            no_improve_rounds = 0
        else:
            no_improve_rounds += 1
            if no_improve_rounds > 3:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def fine_local_search(a0, time_limit, seed=77):
    """
    Very fine local search: try tiny perturbations in many directions.
    Uses a pool of candidate directions and evaluates each.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    # Precompute autocorrelation structure for fast evaluation
    # Use the fact that changing a[i] by delta changes c(k) by:
    #   delta * a[k-i] for all valid k-i
    # We'll evaluate changes one at a time

    c = fast_autocorr(a)
    bmax = c.max()
    top_peaks = np.argsort(c)[::-1][:20]

    iter_count = 0
    while time.monotonic() - t0 < time_limit - 3:
        iter_count += 1
        c = fast_autocorr(a)
        bmax = c.max()
        top_peaks = np.argsort(c)[::-1][:20]

        improved = False
        # Try moving mass from various positions
        n_trials_per_iter = 50
        for trial in range(n_trials_per_iter):
            if time.monotonic() - t0 > time_limit - 3:
                break

            # Pick source from positions that contribute to peaks
            k = top_peaks[rng.integers(len(top_peaks))]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            src = rng.integers(js, je + 1)

            # Pick destination
            dst = rng.integers(n)
            if dst == src:
                continue

            # Small fraction
            frac = rng.uniform(0.001, 0.015)
            amt = a[src] * frac
            if amt < 1e-14:
                continue

            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-12:
                continue
            a_trial = a_trial / st

            new_max = fast_autocorr(a_trial).max()
            if new_max < bmax * (1 - 1e-10):
                a = a_trial
                new_score = 2.0 * n * new_max
                if new_score < best_score:
                    best_score = new_score
                    best_a = a.copy()
                improved = True

        if not improved and iter_count > 200:
            break

    return best_a, best_score
