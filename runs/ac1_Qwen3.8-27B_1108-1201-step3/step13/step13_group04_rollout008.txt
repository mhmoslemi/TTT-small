def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_gradient(a, T):
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    w = np.exp((b - max_b) / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[n - 1] = 1.0
    else:
        w /= w_sum
    fft_size = 1 << (3 * n - 2).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    A_fwd = rfft(a, fft_size)
    W = rfft(w, fft_size)
    conv1 = irfft(A_rev * W, fft_size)
    conv2 = irfft(A_fwd * W, fft_size)
    r1 = conv1[n - 1:n - 1 + n]
    r2 = conv2[:n]
    return r1 + r2

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def build_conv_matrix(a):
    n = len(a)
    T = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            T[k, j] = a[k - j]
    return T.tocsr()


def solve_lp_full(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T_mat = build_conv_matrix(a)
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

    for _ in range(5):

        violations = np.where(ab > M * (1 + 1e-6))[0]

def nesterov_dual_lse(a0, time_limit, T1_start=20.0, T1_end=0.5, T2_start=5.0, T2_end=0.1, n_stages=14, verbose=True):

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()

        a = np.ones(n) * norm / n

    t0 = time.monotonic()
    y = a.copy()
    y_prev = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    temps1 = np.geomspace(T1_start, T1_end, n_stages)
    temps2 = np.geomspace(T2_start, T2_end, n_stages)
    it_count = 0
    for si in range(n_stages):
        T1 = temps1[si]
        T2 = temps2[si]
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        n_iters_stage = max(30, int(remaining * 2.5 / max(1, n_stages - si)))
        local_momentum = 0.0
        for it in range(n_iters_stage):
            if time.monotonic() - t0 > time_limit:
                break
            if it == 0:
                y_eval = y
                step_size = 1.0 / 4.0
            else:
                local_momentum = (it * (it + 1)) / ((it - 1) * (it - 1) + 2 * it * (it + 1)) if it > 1 else 0.5
                y_eval = y + local_momentum * (y - y_prev)
                y_eval = np.maximum(y_eval, 0.0)
                s_ye = y_eval.sum()
                if s_ye > 1e-15:
                    y_eval = y_eval * norm / s_ye
                step_size = 2.0 / (it + 4.0)
            grad = dual_lse_gradient(y_eval, T1, T2)
            y_new = y_eval - step_size * grad
            y_prev = y.copy()
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y
            y = y_new
            it_count += 1
            score = fast_autocorr(y).max()
            if score < best_score:
                best_score = score
                best_a = y.copy()
            avg_a += y
            n_avg += 1
            if n_avg % 100 == 0:
                avg_check = avg_a / n_avg
                s_avg = avg_check.sum()
                if s_avg > 1e-15:
                    avg_check = avg_check * norm / s_avg
                    score_avg = fast_autocorr(avg_check).max()
                    if score_avg < best_score:
                        best_score = score_avg
                        best_a = avg_check.copy()
            if verbose and it_count % 200 == 0:
                elapsed = time.monotonic() - t0
                print(f"    DualLSE stage={si}/{n_stages} T1={T1:.2f} T2={T2:.2f} it={it_count} ({elapsed:.0f}s): best={best_score:.8f}")
    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    y = a.copy()
    y_prev = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    n_iters = 0
    use_active = n <= 2000
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        if use_active:
            b = lp_best_response_active(y, M_curr, solver, max_active=25)
        else:
            b = solve_lp_full(y, M_curr, solver)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        y_prev = y.copy()
        y = y_new
        a = a_new
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        avg_a += a
        n_avg += 1
        if n_avg % 30 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nes iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")
    return best_a, best_score


def peak_transfer_search(a0, time_limit, seed=42, verbose=True):
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    it_count = 0
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_ks = np.argsort(b)[::-1][:8]
        improved = False
        for k_star in top_ks:
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_star - idx]
            top_idx = idx[np.argsort(contrib)[::-1][:5]]
            for src in top_idx:
                if a[src] < norm * 1e-9:
                    continue
                peak_mask = np.zeros(n, dtype=bool)
                for k2 in top_ks[:4]:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    peak_mask[js2:je2 + 1] = True
                candidates = np.where(~peak_mask)[0]
                if len(candidates) < 5:
                    candidates = np.where(~peak_mask & (np.arange(n) != src))[0]
                if len(candidates) == 0:
                    candidates = np.where(np.arange(n) != src)[0]
                cand_sorted = candidates[np.argsort(a[candidates])[:min(10, max(5, len(candidates) // 4))]]
                for dst in cand_sorted[:5]:
                    for frac in [0.002, 0.005, 0.01, 0.02, 0.04, 0.07, 0.12]:
                        amt = a[src] * frac
                        if amt < 1e-16:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < -1e-14:
                            continue
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score - 1e-13:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            it_count += 1
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            stall += 1
            if rng.integers(8) == 0:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > norm * 1e-8:
                    frac = rng.uniform(0.003, 0.08)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
            if stall > 300:
                if verbose:
                    print(f"    PeakTransfer: stalled at {best_score:.8f} ({time.monotonic()-t0:.0f}s)")
                break
        if verbose and it_count % 300 == 0:
            print(f"    PeakTransfer it={it_count} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")
    return best_a, best_score


def structured_init(n, seed):
    rng = np.random.default_rng(seed)
    choice = rng.integers(0, 7)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.3
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.2 * np.cos(2.0 * x) + 0.08 * np.cos(5.0 * x) + 0.04 * np.cos(11.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-8.0 * (x - 0.25) ** 2) + 0.8 * np.exp(-8.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2.5
    elif choice == 5:
        x = np.linspace(0.0, 1.0, n)
        a = 0.3 + 0.7 * np.exp(-0.5 * ((x - 0.5) / 0.2) ** 2)
    else:
        a = np.ones(n) * 0.4
        for _ in range(rng.integers(3, 7)):
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.04, n * 0.25)
            amp = rng.uniform(-0.3, 0.4)
            a += amp * np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1650.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    elapsed = time.monotonic() - t_global
    print(f"Starting optimization, base_n={n_base}, t={elapsed:.0f}s")

    # Phase 1: Dual-temperature LSE Nesterov at base length
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 1: Dual-LSE Nesterov n={n_cur} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.22, 360.0)
    if t1 > 20:
        a_work = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_dual_lse(a_work, t1, T1_start=15.0, T1_end=0.5, T2_start=4.0, T2_end=0.1, n_stages=14, verbose=True)
        do_try_update(a_out, f"dulse_n{n_cur}")

    # Phase 2: LP best response at base length
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: LP Nesterov n={n_cur} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.28, 480.0)
    if t2 > 30:
        a_work = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_work, t2, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n_cur}")

    # Phase 3: LSE polish at current best length
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3: LSE polish n={n_cur} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 240.0)
    if t3 > 15:
        a_work = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_dual_lse(a_work, t3, T1_start=5.0, T1_end=0.2, T2_start=2.0, T2_end=0.05, n_stages=10, verbose=True)
        do_try_update(a_out, f"lse_polish_n{n_cur}")

    # Phase 4: Try longer length
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = len(best_seq) + 400
    print(f"\n=== Phase 4: Dual-LSE n={n_long} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.18, 300.0)
    if t4 > 20:
        init_long = interpolate_sequence(best_seq, n_long)
        a_work = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = nesterov_dual_lse(a_work, t4, T1_start=12.0, T1_end=0.3, T2_start=4.0, T2_end=0.1, n_stages=12, verbose=True)
        do_try_update(a_out, f"dulse_n{n_long}")

    # Phase 5: LP at longer length (if improvement found)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if n_best > n_cur:
        print(f"\n=== Phase 5: LP Nesterov n={n_best} (budget={remaining:.0f}s) ===")
        t5 = min(remaining * 0.20, 320.0)
        if t5 > 20:
            a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = nesterov_lp_optimize(a_work, t5, solver, verbose=True)
            do_try_update(a_out, f"lp_n{n_best}")

    # Phase 6: Peak transfer on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: PeakTransfer n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.18, 280.0)
    if t6 > 10:
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_transfer_search(a_work, t6, seed=777, verbose=True)
        do_try_update(a_out, f"peak_n{n_best}")

    # Phase 7: Final LSE with very low temperature
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LSE n={n_best} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.15, 200.0)
    if t7 > 10:
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_dual_lse(a_work, t7, T1_start=3.0, T1_end=0.1, T2_start=1.0, T2_end=0.03, n_stages=8, verbose=True)
        do_try_update(a_out, "final_lse")

    # Phase 8: Final peak transfer
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 15:
        t8 = min(remaining * 0.4, 150.0)
        if t8 > 5:
            print(f"\n=== Phase 8: Final PeakTransfer n={n_best} (budget={t8:.0f}s) ===")
            a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = peak_transfer_search(a_work, t8, seed=31415, verbose=True)
            do_try_update(a_out, "final_peak")

    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = min(best_score, fs)
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
