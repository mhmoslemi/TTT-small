def mirror_descent_step(a, eta, n_top=3):
    """
    Compute a mirror descent (exponentiated gradient) step.
    Uses subgradient of max auto-correlation.
    """
    n = len(a)
    b = fast_autocorr(a)
    # Get top-k peaks
    top_kapeaks = np.argsort(b)[::-1][:n_top]
    # Average subgradient over top peaks
    grad = np.zeros(n)
    for k in top_kapeaks:
        for i in range(n):
            j1 = k - i
            if 0 <= j1 < n:
                grad[i] += a[j1]
            j2 = i - k
            if 0 <= j2 < n:
                grad[i] += a[j2]
    grad /= n_top
    # Exponentiated gradient update
    log_weights = -eta * grad
    log_weights -= log_weights.max()  # numerical stability
    weights = np.exp(log_weights)
    b_new = a * weights
    s = b_new.sum()
    if s < 1e-15:
        return a.copy()
    return b_new / s

def lp_best_response(a, M, solver):
    """Solve LP: max sum(b) s.t. T@b <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    size = 2 * n - 1
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        rows.extend(idx.tolist())
        cols.extend([i] * len(idx))
        vals.extend((a[idx - i]).tolist())
    from scipy.sparse import csr_matrix
    T = csr_matrix((vals, (rows, cols)), shape=(size, n))
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s  # normalized to sum 1

def lse_obj_grad(a_flat, n, T):
    """Compute LSE smooth objective and gradient."""
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    exp_vals = np.exp(T * (b - max_b))
    lse = max_b + np.log(exp_vals.sum()) / T
    
    # Gradient of LSE w.r.t. a
    w = exp_vals / exp_vals.sum()
    # d(lse)/da_i involves the convolution of w with a
    fft_size = 1 << (3 * n - 2).bit_length()
    a_pad = np.zeros(fft_size)
    a_pad[:n] = a
    w_pad = np.zeros(fft_size)
    w_pad[:2 * n - 1] = w
    A_fft = rfft(a_pad, fft_size)
    W_fft = rfft(w_pad, fft_size)
    # The convolution of w with a gives the gradient of sum(b_k * w_k) w.r.t. a_i
    cross = irfft(A_fft * np.conj(W_fft), fft_size)[:n]
    
    obj = 2.0 * n * lse / (s * s)
    # d(obj)/da_i = 2n * d(lse)/da_i / s^2 - 2n * lse * 2a_i / s^3
    # d(lse)/da_i = (1/T) * sum_k w_k * db_k/da_i = (1/T) * 2 * cross_i (approximately)
    # Actually, d(b_k)/da_i = a[k-i] + a[i-k] (for linear conv)
    # And sum_k w_k * d(b_k)/da_i = 2 * (w * a)_i (circular-like convolution)
    # The 'cross' computed above is (w * a) in a circular sense
    conv_grad = 2.0 * cross / T
    grad = conv_grad / (s * s) - 2.0 * obj * a / s
    return obj, grad.ravel()

def lse_optimize(a0, time_limit, verbose=False):
    """LSE optimization with multi-stage temperature annealing."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0
    a = a * norm
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Temperature schedule
    n_stages = 12
    temps = np.geomspace(50.0, 1.0, n_stages)
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        stage_time = min(remaining / (n_stages - si), 180.0)
        
        x0 = a.copy()
        
        def obj_func(x):
            val, _ = lse_obj_grad(x, n, T)
            return val
        
        def grad_func(x):
            _, g = lse_obj_grad(x, n, T)
            return g
        
        try:
            res = scipy_minimize(obj_func, x0, method='L-BFGS-B',
                                  jac=grad_func,
                                  options={'maxiter': 300, 'maxfun': 1000})
            if res.x is not None:
                a_cand = np.maximum(res.x, 0.0)
                s_c = a_cand.sum()
                if s_c > 1e-10:

                        best_a = a_cand.copy()
                        a = a_cand
        except Exception:
            pass
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {si+1}/{n_stages} (T={T:.1f}, {elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def peak_shave(a, time_limit, verbose=False):

        improved = False

            order = np.argsort(contrib)[::-1]

                            s_t = a_trial.sum()

                            score = fast_autocorr(a_trial).max()

                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                improved = True

                        if improved:

                    if improved:
                        break
                    # Try partner too
                    if not improved:
                        pass
                if improved:
                    break
            if improved:
                break

            stall += 1
            # Random exploration
            if stall % 5 == 0:
                for _ in range(30):
                    i = rng.integers(n)
                    j = rng.integers(n)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
