and adaptive time allocation based on observed progress.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_lp_solver = None

def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for s in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = s
                print(f"LP solver: {s}")
                return s
        except Exception:
            continue
    _lp_solver = cp.SCS
    print("LP solver (fallback): SCS")
    return _lp_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    conv = fast_autocorr(a)
    return float(2.0 * n * conv.max() / (s * s))


def build_T(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp(a, M, solver, active=None, min_frac=0.3, max_frac=0.85):
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    conv_a = T @ a

    if active is None or len(active) < 20:
        thresh = M * min_frac
        active = np.where(conv_a > thresh)[0]
        if len(active) < 30:
            thresh = M * 0.1
            active = np.where(conv_a > thresh)[0]

    b = cp.Variable(n, nonneg=True)
    cons = [T[active, :] @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, active

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, active

    b_val = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, active

    # Check violations
    if len(active) < size:
        full_conv = T @ b_val
        violated = np.where(full_conv > M * (1.0 + 1e-3))[0]
        if len(violated) > 0:
            new_active = np.union1d(active, violated)
            if len(new_active) > max_frac * size:
                b2 = cp.Variable(n, nonneg=True)
                cons2 = [T @ b2 <= M]
                prob2 = cp.Problem(cp.Maximize(cp.sum(b2)), cons2)
                try:
                    prob2.solve(solver=solver, verbose=False)
                except Exception:
                    return b_val, np.arange(size)
                if prob2.status in ('optimal', 'optimal_inaccurate'):
                    b_val = np.maximum(np.asarray(b2.value, dtype=np.float64), 0.0)
                return b_val, np.arange(size)
            else:
                b2 = cp.Variable(n, nonneg=True)
                cons2 = [T[new_active, :] @ b2 <= M]
                prob2 = cp.Problem(cp.Maximize(cp.sum(b2)), cons2)
                try:
                    prob2.solve(solver=solver, verbose=False)
                except Exception:
                    return b_val, new_active
                if prob2.status in ('optimal', 'optimal_inaccurate'):
                    b_val = np.maximum(np.asarray(b2.value, dtype=np.float64), 0.0)
                return b_val, new_active

    return b_val, active


def nesterov_lp(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm_t = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm_t / n
    else:
        a = a * norm_t / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    active = None
    no_improve = 0
    patience = 200

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        M = fast_autocorr(y).max()
        if M < 1e-12:
            break

        b, active = solve_lp(y, M, solver, active)
        if b is None:
            active = None
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm_t / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_t / s_y

        if t == 1:
            x_new = g.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new * norm_t / s_x

        y = y_new
        x = x_new

        improved = False
        sc_y = fast_autocorr(y).max()
        if sc_y < best_score - 1e-10:
            best_score = sc_y
            best_a = y.copy()
            improved = True

        sc_x = fast_autocorr(x).max()
        if sc_x < best_score - 1e-10:
            best_score = sc_x
            best_a = x.copy()
            improved = True

        if improved:
            no_improve = 0
        else:
            no_improve += 1

        if verbose and t % 50 == 0:
            print(f"  NLP t={t} ({time.monotonic() - t0:.0f}s): best={best_score:.8f}")

        if no_improve > patience and t > 50:
            break

    return best_a, best_score


def peak_coordinate_descent(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm_t = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a * norm_t / s

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        # Focus on top peaks
        top_k = np.argsort(conv)[-min(10, size):][::-1]
        pass_improved = False

        for k_star in top_k:
            if time.monotonic() - t0 > time_limit:
                break

            # Find contributing indices for this peak
            i_lo = max(0, k_star - n + 1)
            i_hi = min(n - 1, k_star)
            if i_lo > i_hi:
                continue

            # For each contributing i, compute the product a[i]*a[k_star-i]
            contrib_indices = np.arange(i_lo, i_hi + 1)
            products = a[contrib_indices] * a[k_star - contrib_indices]
            max_prod_idx = contrib_indices[np.argmax(products)]
            max_other = k_star - max_prod_idx

            if max_other < 0 or max_other >= n:
                continue

            ai = a[max_prod_idx]
            aj = a[max_other]
            if ai < norm_t * 1e-8 and aj < norm_t * 1e-8:
                continue

            # Try to equalize: transfer from larger to smaller
            if ai >= aj:
                src, dst = max_prod_idx, max_other
            else:
                src, dst = max_other, max_prod_idx

            a_src = a[src]
            a_dst = a[dst]

            # Compute gain vector for transferring delta from src to dst
            g = np.zeros(size)
            # -a shifted by src
            lo1 = max(0, src)
            hi1 = min(size, src + n)
            if hi1 > lo1:
                g[lo1:hi1] -= a[lo1 - src:hi1 - src]
            lo2 = max(0, n - 1 - src)
            hi2 = min(size, n + src)
            if hi2 > lo2:
                idx2 = np.arange(lo2, hi2)
                src2 = n - 1 - idx2
                g[lo2:hi2] -= a[src2]

            # +a shifted by dst
            lo3 = max(0, dst)
            hi3 = min(size, dst + n)
            if hi3 > lo3:
                g[lo3:hi3] += a[lo3 - dst:hi3 - dst]
            lo4 = max(0, n - 1 - dst)
            hi4 = min(size, n + dst)
            if hi4 > lo4:
                idx4 = np.arange(lo4, hi4)
                src4 = n - 1 - idx4
                g[lo4:hi4] += a[src4]

            # Find best delta
            best_d = 0.0
            best_val = conv.max()
            max_transfer = a_src * 0.5  # Don't transfer all
            n_samples = 15
            deltas = np.linspace(0.0, max_transfer, n_samples)
            for d in deltas:
                if d < 1e-10:
                    continue
                new_conv = conv + d * g
                # Second order at 2*src and 2*dst
                if 2 * src < size:
                    new_conv[2 * src] = (a_src - d) ** 2
                if 2 * dst < size:
                    new_conv[2 * dst] = (a_dst + d) ** 2
                nv = new_conv.max()
                if nv < best_val - 1e-12:
                    best_val = nv
                    best_d = d

            if best_d > 1e-10:
                a[src] -= best_d
                a[dst] += best_d
                s_new = a.sum()
                if s_new < 1e-15:
                    a[src] += best_d
                    a[dst] -= best_d
                    continue
                a = a * norm_t / s_new
                conv = fast_autocorr(a)
                cur_max = conv.max()
                if cur_max < best_score - 1e-12:
                    best_score = cur_max
                    best_a = a.copy()
                    pass_improved = True
                else:
                    a[dst] -= best_d
                    a[src] += best_d
                    a = a * norm_t / a.sum()
                    conv = fast_autocorr(a)

        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0

        if verbose and no_improve % 10 == 0:
            print(f"  PCD pass ({time.monotonic() - t0:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lse_gradient(a0, time_limit, verbose=True):
    n = len(a0)
    norm_t = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm_t / n
    else:
        a = a * norm_t / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    temps = [50.0, 20.0, 8.0, 3.0, 1.0, 0.3]
    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        n_steps = max(5, min(80, int(remaining * 2)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_autocorr(a)
            max_c = conv.max()
            exp_c = np.exp((conv - max_c) / T)
            w = exp_c / exp_c.sum()

            # Gradient computation
            grad = np.zeros(n)
            A_fft = rfft(a, 1 << (size - 1).bit_length())
            W_fft = rfft(w, 1 << (size - 1).bit_length())
            # cross1[i] = sum_k w[k] * a[k+i]
            cross1 = irfft(W_fft * np.conj(A_fft), 1 << (size - 1).bit_length())
            g1 = cross1[:n]
            # cross2 for the other term
            cross2 = irfft(np.conj(W_fft) * A_fft, 1 << (size - 1).bit_length())
            g2 = cross2[:n]
            grad = 2.0 * (g1 + g2)

            step = norm_t / (T * np.sqrt(n)) / (1.0 + 0.1 * it)
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm_t / s_new
            a = a_new

            sc = fast_autocorr(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"  LSE T={T} ({time.monotonic() - t0:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.3
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(2.0 * x) + 0.15 * np.cos(6.0 * x) + 0.05 * np.cos(11.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-8.0 * (x - 0.35) ** 2) + 0.8 * np.exp(-8.0 * (x - 0.65) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2.5
    elif choice == 5:
        x = np.linspace(0.0, 1.0, n)
        a = 0.4 + 0.6 * np.sin(np.pi * x) ** 0.6
    elif choice == 6:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * x) * np.sin(2.0 * np.pi * x * 3.0) + 0.3
        a = np.maximum(a, 0.0)
    else:
        x = np.linspace(0.0, 1.0, n)
        a = 0.3 + 0.4 * np.sin(np.pi * x) + 0.2 * np.sin(2.0 * np.pi * x) + 0.1 * np.cos(4.0 * np.pi * x)
        a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1700.0)

    solver = get_lp_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = eval_local(parent)
            print(f"Parent: n={len(parent)}, local_score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None
        print("No parent loaded")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_local(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LP Nesterov at base length (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    n1 = n_base
    print(f"\n=== Phase 1: LP Nesterov n={n1} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.38, 550.0)
    if t1 > 30:
        a_norm = best_seq.copy()
        s_a = a_norm.sum()
        if s_a > 1e-15:
            a_norm = a_norm * np.sqrt(2.0 * n1) / s_a
        a_out, _ = nesterov_lp(a_norm, t1, solver, verbose=True)
        try_update(a_out, f"lp_n{n1}")

    # Phase 2: Peak CD on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Peak CD n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_coordinate_descent(a_norm, t2, rng, verbose=True)
        try_update(a_out, f"pcd_n{n_best}")

    # Phase 3: Try longer length
    n2 = min(n_base + 400, 2400)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LP Nesterov n={n2} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.18, 300.0)
    if t3 > 30:
        init_long = interpolate_seq(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = nesterov_lp(a_norm, t3, solver, verbose=True)
        try_update(a_out, f"lp_n{n2}")

    # Phase 4: Peak CD on longer
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Peak CD n={n_best} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_coordinate_descent(a_norm, t4, rng, verbose=True)
        try_update(a_out, f"pcd2_n{n_best}")

    # Phase 5: LSE gradient refinement
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LSE gradient n={n_best} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.10, 100.0)
    if t5 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_gradient(a_norm, t5, verbose=True)
        try_update(a_out, f"lse_n{n_best}")

    # Phase 6: Shorter length exploration
    n_short = max(1000, n_base - 400)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: LP Nesterov n={n_short} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.10, 120.0)
    if t6 > 20:
        init_short = interpolate_seq(best_seq, n_short)
        a_norm = init_short * np.sqrt(2.0 * n_short) / init_short.sum()
        a_out, _ = nesterov_lp(a_norm, t6, solver, verbose=False)
        try_update(a_out, f"lp_n{n_short}")

    # Phase 7: Final LP Nesterov on best (use remaining time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LP Nesterov n={n_best} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.5, 200.0)
    if t7 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t7, solver, verbose=True)
        try_update(a_out, f"lp_final_n{n_best}")

    # Phase 8: Final peak CD
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final Peak CD n={n_best} (budget={remaining:.0f}s) ===")
    t8 = min(remaining * 0.5, 100.0)
    if t8 > 5:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_coordinate_descent(a_norm, t8, rng, verbose=False)
        try_update(a_out, f"pcd_final_n{n_best}")

    # Final validation
    if best_seq is not None:
        a_final = np.maximum(best_seq, 0.0)
        s_f = a_final.sum()
        if s_f < 0.01:
            a_final = np.ones(100)
        else:
            a_final = a_final / s_f
        try_update(a_final, "final")
        fs = eval_local(best_seq)
        print(f"\nFINAL: local_score={fs:.6f}, n={len(best_seq)}, time={time.monotonic() - t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0] * 100

    # Use the real evaluate_sequence for final check
    try:
        final_check = evaluate_sequence([float(x) for x in best_seq])
        print(f"evaluate_sequence check: {final_check:.6f}")
    except Exception:
        pass

    return [float(x) for x in best_seq]
