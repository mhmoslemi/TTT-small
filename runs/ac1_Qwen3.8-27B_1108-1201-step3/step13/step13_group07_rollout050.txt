import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(arr):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def internal_score(a):
    """Compute 2*n*max_b assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()


def compute_lse_gradient(a, T):
    """Compute gradient of (1/T)*log(sum exp(T*conv(a,a))) w.r.t. a."""
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()

    # Compute d[m] = sum_k w[k] * a[k-m] via FFT
    a_rev = a[::-1].copy()
    # conv(w, a_rev)[n-1+m] for m=0..n-1
    conv_len = len(w) + len(a_rev) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)
    conv = irfft(W * AR, fft_size)
    d = conv[n - 1:n - 1 + n]
    grad = 2.0 * d
    return grad


def compute_lse_value(a, T):
    """Compute (1/T)*log(sum exp(T*conv(a,a)))."""
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    return (max_Tb + np.log(exp_Tb.sum())) / T


def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    v = np.maximum(v, 0)
    v_sum = v.sum()
    if v_sum < 1e-300:
        return np.ones(n) * target_sum / n
    # Binary search for threshold
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho = np.nonzero(v_sorted * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return v / v_sum * target_sum
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    s = result.sum()
    if s < 1e-300:
        return np.ones(n) * target_sum / n
    result = result / s * target_sum
    return result


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def spectral_shaping(a, n_iter=300, alpha=0.3):
    """Iteratively flatten power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = internal_score(a)

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 50.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def adam_lse_optimize(a0, time_limit, T_start=80.0, T_end=0.5, n_stages=20,
                      lr_base=0.03, beta1=0.9, beta2=0.999, eps=1e-8, seed=42):
    """Adam optimization on LogSumExp-smoothed objective with temperature annealing."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s

    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = internal_score(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        # Estimate iterations per stage based on remaining time
        stages_left = n_stages - si
        iters_per_stage = max(20, int(remaining / stages_left / 0.01))
        iters = min(iters_per_stage, 3000)

        local_best = best_score
        local_best_a = best_a.copy()

        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break

            a = softmax_fn(z)
            grad_a = compute_lse_gradient(a, stage_T)
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)

            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)

            # Cosine learning rate within stage
            progress = it / max(1, iters)
            lr_t = lr_base * 0.5 * (1 + np.cos(np.pi * progress))
            z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)

            if it % 30 == 0:
                a = softmax_fn(z)
                score = internal_score(a)
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    local_best = score
                    local_best_a = a.copy()

        a = softmax_fn(z)
        score = internal_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        elapsed = time.monotonic() - t0
        if si % 3 == 0 or si == n_stages - 1:
            print(f"      Adam stage {si}/{n_stages} (T={stage_T:.2f}, {elapsed:.0f}s): best={best_score:.7f}")

    return best_a, best_score


def mirror_descent_optimize(a0, time_limit, T=5.0, lr_base=0.5, seed=42):
    """Mirror descent (multiplicative weights) on smoothed objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    step = 0

    while time.monotonic() - t0 < time_limit:
        step += 1
        grad = compute_lse_gradient(a, T)
        lr_t = lr_base / (1 + 0.05 * step)

        # Multiplicative update
        a_new = a * np.exp(-lr_t * grad)
        s = a_new.sum()
        if s < 1e-300:
            break
        a_new = a_new / s
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-300:
            break
        a_new = a_new / s

        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        if step % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MirrorDescent step={step} ({elapsed:.0f}s): best={best_score:.7f}")

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=123):
    """Coordinate-descent peak shaving with adaptive step sizes."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:15]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            if contrib.max() < 1e-15:
                continue
            src_local = np.argmax(contrib)
            src = idx[src_local]

            # Find destination: minimize impact on other peaks
            peak_areas = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_areas.update(range(js2, je2 + 1))

            candidates = [j for j in range(n) if j not in peak_areas and j != src]
            if not candidates:
                candidates = [j for j in range(n) if j != src]

            # Pick from low-value candidates (least impact)
            candidates_sorted = sorted(candidates, key=lambda j: a[j])
            dst_pool = candidates_sorted[:max(1, len(candidates_sorted) // 4)]
            dst = dst_pool[rng.integers(len(dst_pool))]

            # Try multiple fractions
            best_frac = None
            best_trial_score = best_score
            best_trial = None
            for frac in [0.001, 0.003, 0.005, 0.01, 0.015, 0.02, 0.03, 0.05]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                sc = internal_score(a_trial)
                if sc < best_trial_score:
                    best_trial_score = sc
                    best_trial = a_trial
                    best_frac = frac

            if best_trial is not None and best_trial_score < best_score - 1e-14:
                a = best_trial
                best_score = best_trial_score
                best_a = a.copy()
                improved = True
                stall = 0
                break

        if not improved:
            # Random perturbation to escape
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.002, 0.03)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = internal_score(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > 800:
                break

    return best_a, best_score


def subgradient_step(a, T_smooth=10.0):
    """Single subgradient step with top-peak weighting."""
    n = len(a)
    b = fast_autocorr(a)
    Tb = T_smooth * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()

    # Gradient as before
    a_rev = a[::-1].copy()
    conv_len = len(w) + len(a_rev) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)
    conv = irfft(W * AR, fft_size)
    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad


def subgradient_optimize(a0, time_limit, verbose=True):
    """Subgradient optimization with Nesterov acceleration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = internal_score(a)
    t0 = time.monotonic()

    y = a.copy()
    v = np.zeros(n)
    beta = 0.9
    base_lr = 0.05
    t = 0

    # Periodically recompute T_smooth
    T_max = 30.0
    T_min = 1.0
    total_estimated = int(time_limit / 0.01)  # rough estimate of total steps

    while time.monotonic() - t0 < time_limit:
        t += 1
        T_smooth = T_max * (T_min / T_max) ** (t / max(1, total_estimated))
        T_smooth = max(T_min, T_smooth)

        grad = subgradient_step(y, T_smooth)
        v = beta * v + (1 - beta) * grad
        lr_t = base_lr / (1 + 0.01 * t)
        a_new = y - lr_t * v
        a_new = project_simplex(a_new)

        score = internal_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        # Nesterov: y = a_new + momentum * (a_new - a)
        momentum = (t - 1.0) / (t + 2.0)
        y = a_new + momentum * (a_new - a)
        y = project_simplex(y)
        a = a_new

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SubGrad t={t} ({elapsed:.0f}s): best={best_score:.7f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
