def build_constraint_rows(f, indices, n):

def lp_refine(f_init, time_limit, initial_k=50, max_k=400, label="lp"):
    """LP-based refinement with cutting planes."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    
    b = fast_convolve(f)
    M = b.max()
    best_obj = M
    
    # Initialize active set: top peaks
    if initial_k < len(b):
        active_idx = set(np.argpartition(b, -initial_k)[-initial_k:].tolist())
    else:
        active_idx = set(range(len(b)))
    
    it = 0
    stuck = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 3:
            break
        
        # Solve LP
        g = solve_lp(f, n, M, sorted(active_idx))
        
        if g is None:
            stuck += 1
            if stuck > 10:
                break
            # Shrink active set
            b_cur = fast_convolve(f)
            active_idx = set(np.argpartition(b_cur, -initial_k)[-initial_k:].tolist())
            continue
        
        S_g = g.sum()
        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > 15:
                break
            # Expand active set
            b_cur = fast_convolve(f)
            k_expand = min(max_k, len(b_cur))
            new_top = np.argpartition(b_cur, -k_expand)[-k_expand:]
            active_idx = set(active_idx) | set(new_top.tolist())
            continue
        
        stuck = 0
        h = g / S_g
        
        # Check for violated constraints (cutting plane)
        fg = convolve_fftf(f, h)
        violated = fg > M * (1 + 1e-8)
        if violated.any() and len(active_idx) < max_k:
            v_idx = np.where(violated)[0]
            not_in_active = [i for i in v_idx if i not in active_idx]
            if not_in_active:
                # Add top 20 violated
                fg_vals = fg[not_in_active]
                top_violated = not_in_active[np.argsort(fg_vals)[-20:]]
                active_idx.update(top_violated.tolist())
        
        # Line search
        best_t = 0.1
        best_t_obj = M
        for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7]:
            f_trial = normalize_simplex((1-t)*f + t*h)
            obj_t = fast_convolve(f_trial).max()
            if obj_t < best_t_obj:
                best_t_obj = obj_t
                best_t = t
        
        # Refine with golden section
        if best_t > 0.02:
            lo = max(0.0, best_t - 0.05)
            hi = min(1.0, best_t + 0.1)
            for _ in range(5):
                m1 = lo + (hi - lo) * 0.382
                m2 = hi - (hi - lo) * 0.382
                f1 = normalize_simplex((1-m1)*f + m1*h)
                f2 = normalize_simplex((1-m2)*f + m2*h)
                o1 = fast_convolve(f1).max()
                o2 = fast_convolve(f2).max()
                if o1 < o2:
                    hi = m2
                else:
                    lo = m1
            t_mid = (lo + hi) / 2
            f_mid = normalize_simplex((1-t_mid)*f + t_mid*h)
            o_mid = fast_convolve(f_mid).max()
            if o_mid < best_t_obj:
                best_t_obj = o_mid
                best_t = t_mid
        
        # Update
        f = normalize_simplex((1-best_t)*f + best_t*h)
        cur_obj = fast_convolve(f).max()
        M = cur_obj
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
        
        # Periodically refresh active set
        if it % 20 == 0:
            b = fast_convolve(f)
            keep_k = min(initial_k * 2, max_k)
            active_idx = set(np.argpartition(b, -keep_k)[-keep_k:].tolist())
        
        if it % 50 == 0:
            print(f"    {label} it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} "
                  f"|A|={len(active_idx)} t={time.monotonic()-t0:.0f}s")
    
    return best_f, best_obj

def random_restart(a, n_perturb=50, strength=0.01):
    """Randomly perturb a subset of coordinates."""
    n = len(a)
    rng = np.random.default_rng()
    idx = rng.choice(n, size=min(n_perturb, n), replace=False)
    a_new = a.copy()
    a_new[idx] += rng.standard_normal(len(idx)) * strength
    return normalize_simplex(a_new)

def coordinate_descent(a_init, time_limit, n_sweeps=50):
    """Coordinate descent with multiplicative updates."""
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy()
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    step = max(1, n // 500)
    
    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        for j in range(0, n, step):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            if a[j] < 1e-15:
                continue
            orig = a[j]
            best_val = orig
            best_local = fast_convolve(a).max()
            for factor in [0.8, 0.9, 1.1, 1.2]:
                a[j] = orig * factor
                a = normalize_simplex(a)
                obj = fast_convolve(a).max()
                if obj < best_local:
                    best_local = obj
                    best_val = orig * factor
            a[j] = best_val
            a = normalize_simplex(a)
            if best_local < best_obj - 1e-14:
                best_obj = best_local
                best_a = a.copy()
    
    return best_a, best_obj

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Novel contributions over previous approach:
1. Spectral equalization: iterative frequency-domain method to flatten auto-convolution
2. LP with PDLP solver and adaptive cutting-plane constraint management
3. Multi-length exploration with shape-preserving interpolation
4. Random restart diversification with short LP polishes
5. Coordinate descent with adaptive multiplicative updates
"""

import numpy as np
from scipy.fft import fft, ifft, rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_cross(a, b):
    """Cross-convolution via FFT."""
    n = max(len(a), len(b))
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def normalize_simplex(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def spectral_equalize(a_init, max_iters=500, lam_init=0.05, lam_max=0.4,
                      time_limit=300.0, seed=42):
    """
    Iterative spectral flattening of the auto-convolution.
    In frequency domain, conv(a,a) ~ |A|^2. We iteratively adjust the
    amplitude spectrum to make the time-domain convolution flatter.
    """
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy()
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    lam = lam_init
    no_improve = 0
    rng = np.random.default_rng(seed)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute current |A|^2
        A = fft(a, N)
        B = A * np.conj(A)
        b = ifft(B).real[:size]
        # Normalize b (should sum to ~1)
        b_sum = b.sum()
        if b_sum > 1e-15:
            b = b / b_sum

        # Flatten target
        mean_b = 1.0 / size
        b_target = (1.0 - lam) * b + lam * mean_b

        # Target amplitude spectrum
        B_target = fft(b_target, N)
        amp_target = np.sqrt(np.maximum(B_target.real, 0.0))

        # Preserve phase
        abs_A = np.abs(A)
        phase_A = A / (abs_A + 1e-30)
        A_new = amp_target * phase_A

        # Enforce conjugate symmetry
        A_new[0] = float(np.real(A_new[0]))
        if N % 2 == 0:
            A_new[N // 2] = float(np.real(A_new[N // 2]))
        for k in range(1, (N + 1) // 2):
            A_new[N - k] = np.conj(A_new[k])

        a_new = ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            lam = max(0.01, lam * 0.7)
            no_improve += 1
            if no_improve > 30:
                break
            continue
        a_new /= s

        obj_new = fast_convolve(a_new).max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            no_improve = 0
            lam = min(lam_max, lam * 1.03)
        else:
            # Try blending
            improved = False
            for alpha in [0.2, 0.4, 0.6, 0.8]:
                a_blend = normalize_simplex((1.0 - alpha) * a + alpha * a_new)
                obj_blend = fast_convolve(a_blend).max()
                if obj_blend < best_obj - 1e-14:
                    best_obj = obj_blend
                    best_a = a_blend.copy()
                    a = a_blend.copy()
                    no_improve = 0
                    improved = True
                    break
            if not improved:
                no_improve += 1
                lam = max(0.01, lam * 0.8)
                if no_improve > 25:
                    # Add noise to escape
                    a = normalize_simplex(a + rng.standard_normal(n) * 0.003)
                    no_improve = 0
                if no_improve > 60:
                    break

        if it % 100 == 0:
            print(f"    spec it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} "
                  f"lam={lam:.4f} noimp={no_improve} t={time.monotonic()-t0:.0f}s")

    return best_a, best_obj


def build_constraint_matrix(f, indices, n):
    """Build constraint matrix A where row i encodes (f*g)[k_i] = A[i,:] @ g."""
    indices = np.asarray(indices)
    K = len(indices)
    A = np.zeros((K, n))
    for i in range(K):
        k = int(indices[i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp_direction(f, n, M, active_indices):
    """Solve: max sum(g) s.t. g >= 0, A@g <= M. Returns g or None."""
    K = len(active_indices)
    if K == 0 or K > 3000:
        return None

    A_mat = build_constraint_matrix(f, active_indices, n)

    g_var = cp.Variable(n)
    constraints = [g_var >= 0, A_mat @ g_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)

    g_val = None
    # Try PDLP (fast for large LPs)
    try:
        prob.solve(solver=cp.PDLP, max_iters=100000, eps_abs=1e-9, eps_rel=1e-9)
        if g_var.value is not None:
            g_val = np.array(g_var.value)
    except Exception:
        pass

    if g_val is None:
        try:
            prob.solve(solver=cp.SCS, max_iters=200000)
            if g_var.value is not None:
                g_val = np
