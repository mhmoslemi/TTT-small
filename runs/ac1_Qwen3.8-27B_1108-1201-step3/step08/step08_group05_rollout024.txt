"""
Optimization for minimizing 2*n*max(convolution)/sum^2 using:
- Improved LP with adaptive constraint generation and relaxation
- Peak equalization via gradient steps on the worst lag
- Multi-length exploration with sinusoidal-perturbed initialization
- Subgradient polish with momentum and adaptive restarts

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def score_from_conv(b, n, s):
    if s < 0.01:
        return np.inf
    return float(2.0 * n * b.max() / (s * s))


def lp_solve_adaptive(f, n, M, active_list, relaxation=0.0, max_constraints=500):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)[k] <= M*(1+relaxation) for active k."""
    M_eff = M * (1.0 + relaxation)
    K = len(active_list)
    if K > max_constraints:
        b = fast_convolve(f)
        b_active = b[active_list]
        keep = np.argsort(b_active)[-max_constraints:]
        active_list = [active_list[i] for i in keep]
        K = len(active_list)

    A = np.zeros((K, n))
    for row_i in range(K):
        k = active_list[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]

    g_var = cp.Variable(n)
    prob = cp.Problem(
        cp.Maximize(cp.sum(g_var)),
        [g_var >= 0, A @ g_var <= M_eff]
    )

    g = None
    for solver in [cp.SCS, cp.PDLP, cp.ECOS]:
        try:
            prob.solve(solver=solver, max_iters=500000, eps_abs=1e-9, eps_rel=1e-9)
            if g_var.value is not None:
                g = np.maximum(g_var.value, 0.0)
                if not (np.any(np.isnan(g)) or np.any(np.isinf(g))):
                    break
                g = None
        except Exception:
            continue

    return g


def generate_constraints(f, n, threshold_frac=0.95, max_count=500):
    """Generate constraints: all lags where b_k > threshold_frac * max_b."""
    b = fast_convolve(f)
    M = b.max()
    threshold = M * threshold_frac
    active = np.where(b >= threshold)[0]
    if len(active) > max_count:
        active = np.argsort(b)[-max_count:]
    return sorted(active.tolist()), M


def run_lp_improved(f_init, time_limit, label="lp", initial_constraints=80,
                    max_constraints=500, relaxation_attempts=(0.0, 0.001, 0.005, 0.01)):
    """LP with adaptive constraint generation, relaxation, and KKT-aware step size."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s < 1e-10:
        return f, np.inf
    f = f / s

    b = fast_convolve(f)
    best_obj = b.max()
    best_f = f.copy()
    stuck = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        if M < best_obj - 1e-13:
            best_obj = M
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > 30:
            break

        # Generate constraints
        active_list, _ = generate_constraints(f, n, threshold_frac=0.93,
                                              max_count=min(initial_constraints + it * 3, max_constraints))
        K = len(active_list)
        if K == 0:
            break

        # Try LP with increasing relaxation
        g = None
        for relax in relaxation_attempts:
            g = lp_solve_adaptive(f, n, M, active_list, relaxation=relax,
                                   max_constraints=max_constraints)
            if g is not None and g.sum() > 1.0 + 1e-6:
                break

        if g is None or g.sum() <= 1.0 + 1e-6:
            # Try with more relaxed threshold
            active_list, _ = generate_constraints(f, n, threshold_frac=0.99,
                                                  max_count=max_constraints)
            g = lp_solve_adaptive(f, n, M, active_list, relaxation=0.02,
                                   max_constraints=max_constraints)
            if g is None or g.sum() <= 1.0 + 1e-6:
                if stuck > 5:
                    break
                # Fallback: use steepest descent direction
                peak_idx = b.argmax()
                g = np.zeros(n)
                lo = max(0, peak_idx - n + 1)
                hi = min(n - 1, peak_idx)
                g[lo:hi + 1] = f[hi:lo - 1:-1] if lo < hi else f[:hi + 1]
                if g.sum() <= 1e-10:
                    break

        h = g / g.sum()

        # Line search: try multiple steps
        best_t = 0.05
        best_trial = best_obj
        for t in [0.02, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4]:
            f_trial = (1.0 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            st = f_trial.sum()
            if st < 1e-10:
                continue
            f_trial = f_trial / st
            obj = fast_convolve(f_trial).max()
            if obj < best_trial:
                best_opt = f_trial.copy()
                best_trial = obj
                best_t = t

        if best_trial >= best_obj - 1e-13:
            stuck += 2
            if stuck > 8:
                # Perturb and retry
                f = f + np.random.default_rng(it).standard_normal(n) * 0.002
                f = project_simplex(f)
                best_trial = best_obj
                best_opt = f.copy()
        else:
            f = best_opt.copy() if 'best_opt' in dir() else f
            if best_trial < best_obj:
                best_obj = best_trial
                best_f = f.copy()
                stuck = 0

        if it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} max_b={best_obj:.12f} "
                  f"score={2.0*n*best_obj:.6f} stuck={stuck} t={elapsed:.0f}s")

    return best_f, best_obj


def peak_equalization(f_init, time_limit, label="peak_eq"):
    """Directly reduce the worst lag using gradient steps with BB step sizes."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s < 1e-10:
        return f, np.inf
    f = f / s

    b = fast_convolve(f)
    best_obj = b.max()
    best_f = f.copy()
    prev_grad_norm = None
    prev_f = None
    stuck = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(f)
        M = b.max()
        if M < best_obj - 1e-13:
            best_obj = M
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 40:
                break

        # Find top peaks (top 3)
        top_k = min(3, len(b))
        top_idx = np.argpartition(b, -top_k)[-top_k:]

        # Compute combined gradient for top peaks
        grad = np.zeros(n)
        for k in top_idx:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                grad[j] += f[k - j]

        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        # BB step size if possible
        if prev_grad_norm is not None and prev_f is not None:
            s_vec = f - prev_f
            y_vec = grad - np.zeros(n)  # Inexact gradient, use simple BB
            sy = np.dot(s_vec, y_vec)
            if abs(sy) > 1e-15:
                alpha_bb = np.dot(s_vec, s_vec) / abs(sy)
            else:
                alpha_bb = 0.01
        else:
            alpha_bb = 0.005

        prev_f = f.copy()
        prev_grad_norm = gnorm

        # Try several step sizes
        candidates = [0.001, 0.002, 0.005, alpha_bb, 0.01, 0.02, 0.05]
        best_alpha = None
        best_trial = best_obj + 1e-10
        for alpha in candidates:
            if alpha <= 0:
                continue
            f_trial = f - alpha * grad
            f_trial = project_simplex(f_trial)
            obj = fast_convolve(f_trial).max()
            if obj < best_trial:
                best_trial = obj
                best_alpha = alpha
                best_new = f_trial.copy()

        if best_alpha is None:
            stuck += 3
            if stuck > 15:
                break
            # Random restart from current best
            rng = np.random.default_rng(hash(f.tobytes()[:100]) % 2**31)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.01)
        else:
            f = best_new.copy()

    return best_f, best_obj


def subgradient_polish(f_init, time_limit, label="sg"):
    """Subgradient with momentum, adaptive temperature, and restarts."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s < 1e-10:
        return f, np.inf
    f = f / s

    v = np.zeros(n)
    lr = 0.003
    temp = 0.5
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    rng = np.random.default_rng(12345)
    it = 0

    while time.monotonic() - t0 < time_limit:
        batch = min(50, max(10, int((time_limit - (time.monotonic() - t0)) * 3)))
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit:
                break
            it += 1
            b = fast_convolve(f)

            # Softmax weights
            scaled = b / max(temp, 1e-10)
            scaled -= scaled.max()
            w = np.exp(scaled)
            w /= w.sum()

            # Focus on top lags
            n_top = max(10, n // 10)
            if n_top < len(b):
                top_idx = np.argpartition(b, -n_top)[-n_top:]
                w_sel = w[top_idx].copy()
                ws = w_sel.sum()
                if ws > 1e-30:
                    w_sel /= ws
                else:
                    w_sel = np.ones(len(top_idx)) / len(top_idx)
                w_new = np.zeros(len(b))
                w_new[top_idx] = w_sel
                w = w_new

            # Gradient
            g = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-12:
                    continue
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                idx = np.arange(lo, hi + 1)
                g[idx] += w[k] * f[k - idx]

            v = 0.9 * v - lr * g
            f = project_simplex(f + v)

        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
            lr = min(lr * 1.02, 0.01)
        else:
            no_improve += 1
            if no_improve > 3:
                lr *= 0.9
                no_improve = 0
            if no_improve > 8:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                v = np.zeros(n)
                no_improve = 0
                temp = max(temp * 0.8, 0.1)

        if it % 200 == 0:
            temp = min(temp * 1.2, 3.0)

    return best_f, best_obj


def sinusoidal_init(n, a=0.3, phi=0.0):
    """Create a sinusoidal perturbation of uniform: (1 + a*sin(2*pi*j/n + phi))/n."""
    j = np.arange(n)
    f = (1.0 + a * np.sin(2.0 * np.pi * j / n + phi)) / n
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return f / s


def try_length(target_n, f_best, best_score, time_budget):
    """Try to optimize at a different length by interpolation."""
    n_old = len(f_best)
    old_x = np.linspace(0.0, 1.0, n_old)
    new_x = np.linspace(0.0, 1.0, target_n)
    f_interp = np.interp(new_x, old_x, f_best)
    f_interp = np.maximum(f_interp, 0.0)
    s = f_interp.sum()
    if s > 1e-10:
        f_interp /= s

    # Also try sinusoidal
    f_sine = sinusoidal_init(target_n, a=0.2, phi=0.5)

    results = []
    per_time = time_budget / 2.0

    for f_init in [f_interp, f_sine]:
        if time_budget < 10:
            break
        f_opt, obj = run_lp_improved(f_init, time_limit=per_time,
                                      label=f"n{target_n}",
                                      initial_constraints=50, max_constraints=200)
        score = score_from_conv(fast_convolve(f_opt), target_n, f_opt.sum())
        results.append((score, f_opt))

    if results:
        results.sort(key=lambda x: x[0])
        return results[0]
    return (float('inf'), None)
