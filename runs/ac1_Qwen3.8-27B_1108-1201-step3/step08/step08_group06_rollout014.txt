import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def subgradient_of_max_conv(a, k_star):
    """Compute 2 * S_k*(a) where (S_k a)_j = a_{k-j} if 0<=k-j<n else 0."""
    n = len(a)
    g = np.zeros(n)
    for j in range(n):
        idx = k_star - j
        if 0 <= idx < n:
            g[j] = 2.0 * a[idx]
    return g

def mirror_descent_step(a, g, eta):
    """Multiplicative update: a_new ∝ a * exp(-eta * g), normalized."""
    # For numerical stability
    log_ratio = -eta * g / np.maximum(a, 1e-300)
    log_ratio -= log_ratio.max()
    a_new = a * np.exp(log_ratio)
    s = a_new.sum()
    if s < 1e-300:
        return a.copy()
    return a_new / s

def lbgfs_logspace(a0, time_limit, T, maxiter=3000):
    """L-BFGS in log-space."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-30)
    z0 = np.log(a_safe)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)
    
    def softmax_from_z(z):
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        s = exp_z.sum()
        if s < 1e-300:
            return np.ones(len(z)) / len(z)
        return exp_z / s
    
    def func_and_grad(z):
        a = softmax_from_z(z)
        conv = fast_convolve(a)
        shifted = conv / T
        max_shifted = shifted.max()
        w = np.exp(shifted - max_shifted)
        w = w / w.sum()
        
        log_sum_exp = max_shifted + np.log(w.sum())
        obj = (2.0 * n) * T * log_sum_exp
        
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
        
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        
        return obj, (2.0 * n) * grad_z
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        obj, _ = func_and_grad(z)
        return obj
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        _, g = func_and_grad(z)
        return g
    
    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 30, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        score = compute_score(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception:
        pass
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP fictitious play with Polyak step size and averaging
2. Mirror descent (multiplicative weight updates)
3. Coordinate descent with exact golden-section search
4. L-BFGS in log-space with adaptive temperature
5. Length exploration
Improvements: Polyak step size in LP play, 1D line search for update step,
better time allocation, mirror descent phase.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def get_subgradient(a, k_star):
    n = len(a)
    g = np.zeros(n)
    for j in range(n):
        idx = k_star - j
        if 0 <= idx < n:
            g[j] = 2.0 * a[idx]
    return g


def build_C(f):
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = f
    return C


def solve_lp(f, M):
    n = len(f)
    C = build_C(f)
    c = -np.ones(n)
    A_ub = C
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if res.status != 0:
        return None
    return np.maximum(res.x, 0)


def greedy_lp(f, M):
    n = len(f)
    g = np.zeros(n)
    conv_vals = np.zeros(2 * n - 1)
    for j in range(n):
        k_start = j
        k_end = j + n - 1
        slack = M - conv_vals[k_start:k_end + 1]
        f_slice = f[:n]
        valid = f_slice > 1e-15
        if not valid.any():
            continue
        max_gj = np.inf
        for idx in range(n):
            if f_slice[idx] > 1e-15:
                cand = slack[idx] / f_slice[idx]
                if cand < max_gj:
                    max_gj = cand
        max_gj = max(0.0, max_gj)
        g[j] = max_gj
        conv_vals[k_start:k_end + 1] += f_slice * max_gj
    return g


def lp_fictitious_play(f_init, time_limit, f_star_est=None, verbose=True):
    """LP fictitious play with adaptive step size and Polyak averaging."""
    f = f_init.copy()
    s = f.sum()
    f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()
    
    # Polyak average
    f_avg = f.copy()
    score_avg = best_score
    
    # Track M for F* estimation
    M_best = fast_convolve(f).max()
    
    for k in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_f = fast_convolve(f)
        M = conv_f.max()
        M_best = min(M_best, M)
        
        g = solve_lp(f, M)
        if g is None:
            g = greedy_lp(f, M)
        if g is None or g.sum() < 1e-15:
            break
        g = g / g.sum()
        
        # Adaptive step size
        if k < 10:
            t_step = 0.5
        elif k < 100:
            t_step = 1.0 / (k // 10 + 1)
        else:
            # Polyak-like: based on gap
            f_star_cur = 2.0 * n * M_best
            gap = best_score - f_star_cur
            diff = f - g
            diff_norm_sq = np.dot(diff, diff)
            if diff_norm_sq > 1e-20 and gap > 0:
                t_step = gap / diff_norm_sq
            else:
                t_step = 1.0 / (k + 1)
            t_step = max(0.01, min(t_step, 1.0))
        
        a_new = (1 - t_step) * f + t_step * g
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        
        # Update Polyak average
        f_avg = (k * f_avg + a_new) / (k + 1)
        
        score_new = compute_score(a_new)
        score_avg = compute_score(f_avg)
        
        if score_new < best_score:
            best_score = score_new
            best_f = a_new.copy()
        if score_avg < best_score:
            best_score = score_avg
            best_f = f_avg.copy()
        
        f = a_new
        
        if verbose and k % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LP iter={k}, M={M:.6f}, score={score_new:.8f}, avg={score_avg:.8f}, best={best_score:.8f}, t={elapsed:.0f}s")
    
    return best_f, best_score
