import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def get_lp_solver():
    import cvxpy as cp
    global _lp_solver
    if '_lp_solver' in globals():
        try:
            if _lp_solver is not None:
                return _lp_solver
        except:
            pass
    _lp_solver = None
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.SCS, cp.PDLP, cp.ECOS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except:
            continue
    _lp_solver = cp.SCS
    return cp.SCS

def solve_active_set_lp(a, top_k=100, solver=None):
    """
    Solve LP: minimize M subject to (conv matrix restricted to top-k positions) @ b <= M, sum(b)=1, b>=0
    """
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    
    n = len(a)
    conv = fast_convolve(a)
    # Get top-k peak positions
    k_positions = np.argpartition(conv, -top_k)[-top_k:]
    k_positions = sorted(k_positions)
    
    # Build restricted constraint matrix
    # For each k in k_positions: sum_j a[k-j] * b[j] <= M (where valid)
    # Actually we want: for the CANDIDATE b, sum_j a_fixed[j] * b[k-j] <= M
    # Wait, the LP is: given a_fixed, find b that minimizes max_k sum_j a_fixed[j] * b[k-j]
    # So the constraint for position k is: sum_j a_fixed[j] * b[k-j] <= M
    # where j ranges over valid values (0 <= j < n, 0 <= k-j < n)
    
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    constraints = [cp.sum(b_var) == 1, b_var >= 0]
    
    for k in k_positions:
        # sum_j a[j] * b[k-j] <= M
        # j must satisfy 0 <= j < n and 0 <= k-j < n
        # i.e., max(0, k-n+1) <= j <= min(n-1, k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        expr = a[j_start:j_end+1] @ b_var[j_start:j_end+1]
        constraints.append(expr <= M_var)
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, max_time=30)
    except:
        for alt in [cp.SCS, cp.PDLP, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, max_time=30)
                break
            except:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s

def subgradient_momentum(a0, time_limit, lr_init=0.01, top_k=20, momentum=0.9):
    """Subgradient descent with Nesterov momentum and adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    # AdaGrad accumulator
    G = np.zeros(n)
    v = np.zeros(n)  # momentum
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        score = 2.0 * n * M
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 100:
            lr *= 0.8
            momentum = max(0.5, momentum * 0.95)
            no_improve = 0
            if lr < 1e-8:
                break
        
        # Compute subgradient from top-k peaks
        k_act = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]
        # Softmax weighting
        temp = 0.005 * max(pv.max(), 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        w = w / ws
        
        # Subgradient
        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += 2.0 * w[idx] * a[k - j]
        
        # Update: a_new = a - lr * grad (with momentum)
        v = momentum * v + (1 - momentum) * grad
        a_new = a - lr * v
        
        # Project to simplex
        a = simplex_project(a_new)
        
        # Decay learning rate
        lr *= 0.9995
        
        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGradMomentum it={it} ({elapsed:.0f}s): best={best_score:.6f} lr={lr:.6f} M={M:.8f}")
    
    return best_a, best_score

def spectral_perturb(a0, rng, n_harmonics=10, magnitude=0.05):
    """Perturb the sequence with a random smooth (low-frequency) component."""
    n = len(a0)
    a = a0.copy()
    t = np.arange(n, dtype=np.float64)
    perturb = np.zeros(n)
    for k in range(1, n_harmonics + 1):
        amp = magnitude / k * rng.random()
        phase = rng.uniform(0, 2 * np.pi)
        perturb += amp * np.cos(2 * np.pi * k * t / n + phase)
    a = a + perturb
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a0.copy()
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def init_fourier(n, rng):
    n_harmonics = int(rng.integers(3, 18))
    amplitudes = rng.random(n_harmonics) * 0.35
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = np.arange(1, n_harmonics + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    global _lp_solver
    if '_lp_solver' in globals():
        try:
            if _lp_solver is not None:
                return _lp_solver
        except Exception:
            pass
    _lp_solver = None
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.SCS
    return cp.SCS


def solve_active_set_lp(a_fixed, top_k=100, solver=None, max_time=20):
    """
    Given fixed a (non-negative, sum=1), find b >= 0, sum(b)=1
    minimizing max_k (a*b)[k], using only top_k peak constraints.
    """
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    conv = fast_convolve(a_fixed)
    k_positions = np.argpartition(conv, -top_k)[-top_k:]

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [cp.sum(b_var) == 1]

    for k in k_positions:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        expr = a_fixed[j_start:j_end + 1] @ b_var[j_start:j_end + 1]
        constraints.append(expr <= M_var)

    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, max_time=max_time)
    except Exception:
        for alt in [cp.PDLP, cp.SCS, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, max_time=max_time)
                break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def fictitious_play_dynamic(a_init, time_limit, solver=None, verbose=False):

        if a_new is not None:

def smoothed_objective_and_grad(a, T, top_k=50):
    """
    Compute the smoothed approximation of 2n*max(conv(a,a))/(sum(a))^2
    and its gradient. Uses top-k peaks for efficiency.
    """
    n = len(a)
    conv = fast_convolve(a)
    sum_a = a.sum()

    # Smoothed max: (1/T) * log(sum(exp(T*x)))
    k_idx = np.argpartition(conv, -top_k)[-top_k:]
    Tconv = T * conv[k_idx]
    Tconv_max = Tconv.max()
    W = np.exp(Tconv - Tconv_max)
    W_sum = W.sum()
    W = W / W_sum

    smoothed_max = (conv[k_idx].max() + np.log(W_sum) / T)  # approximation
    # More accurate: log-sum-exp = Tconv_max/T + log(W_sum)/T
    smoothed_max = Tconv_max / T + np.log(W_sum) / T

    # Gradient of smoothed max w.r.t. a
    grad_conv = np.zeros(n)
    for idx in range(len(k_idx)):
        if W[idx] < 1e-14:
            continue
        w = W[idx]
        k = int(k_idx[idx])
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            if i == j:
                grad_conv[i] += w * 2.0 * a[i]
            else:
                grad_conv[i] += w * (a[j] + a[i])

    # Full objective: f = 2n * smoothed_max / sum_a^2
    # grad f = 2n * grad_conv / sum_a^2 - 2 * 2n * smoothed_max * grad(sum_a) / sum_a^3
    # grad(sum_a) = ones(n)
    obj = 2.0 * n * smoothed_max / (sum_a * sum_a)
    grad_obj = (2.0 * n / (sum_a * sum_a)) * grad_conv - (4.0 * n * smoothed_max / (sum_a * sum_a * sum_a)) * np.ones(n)

    return obj, grad_obj

        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            affected = conv[k_start:k_end + 1]
            a_shifted = a

            best_delta = 0.0
            best_new_max = M

            for factor in [0.3, 0.6, 0.8, 0.9, 1.1, 1.25, 1.5, 2.0, 0.0]:
                if factor == 0.0:
                    continue
                delta = (factor - 1.0) * a[i]
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx_2i] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-16:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-16:
                old_ai = a[i]
                a[i] = old_ai + best_delta
                a[i] = max(a[i], 0.0)

                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx_2i] = 2.0 * best_delta * old_ai + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                improved = True

        cur_score = 2.0 * n * M
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        if not improved:
            no_improve += 1
            if no_improve > 8:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, sweeps
