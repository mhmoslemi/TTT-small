Let me implement a **bisection-based coordinate descent**:
1. For coordinate `i`, the function `h(x) = max_k c_k(x)` where `c_k(x)` is the auto-convolution with `a[i] = x` is convex in `x`.
2. Find the minimum by bisection on the subgradient.

The subgradient of `h(x)` is:
- For each `k` where `c_k(x) = h(x)` (active), the derivative is `2*a[k-i]` (or `2*x` if `k=2i`).
- The subgradient is the convex hull of these values: `[min_active deriv, max_active deriv]`.
- The minimum is where 0 is in the subgradient, i.e., `min_active deriv <= 0 <= max_active deriv`.

So: if all active derivatives are positive, decrease `x`. If all are negative, increase `x`. If 0 is in the range, we're at the minimum (or on the boundary).

This gives an exact coordinate minimum (up to the non-negativity constraint)!

def exact_coord_min(i, a, conv, n):
    """Find the exact optimal value of a[i] given all other coordinates fixed."""
    # For each k, c[k] as a function of x = a[i]:
    # c[k] = base[k] + 2*x*a[k-i] (for k != 2i, where 0 <= k-i < n)
    # c[k] = base[k] + x^2 (for k = 2i)
    # where base[k] = conv[k] - current contribution of a[i]
    
    # Current contribution of a[i] to c[k]:
    # If 0 <= k-i < n: contribution = 2*a[i]*a[k-i] (or a[i]^2 if k=2i)
    # If not: contribution = 0
    
    i_val = a[i]
    base = conv.copy()
    for k in range(max(0, i), min(2*n-1, i+n)):
        # c[k] includes a[i]*a[k-i] and a[k-i]*a[i] (if k-i != i)
        if k - i < n and 0 <= k - i:
            if k == 2*i:
                base[k] -= i_val * i_val  # a[i]^2
            elif i < n:
                base[k] -= 2 * i_val * a[k-i]
    
    # Now c[k](x) = base[k] + 2*x*a[k-i] for k where 0<=k-i<n and k!=2i
    # c[2i](x) = base[2i] + x^2
    
    # We want to minimize max_k c[k](x) subject to x >= 0
    
    # Binary search / bisection approach:
    # The function h(x) = max_k c[k](x) is convex in x.
    # Find x* that minimizes h(x).
    
    lo = 0.0
    hi = max(2 * i_val, 0.1)  # upper bound
    
    # Bisection on the subgradient
    for _ in range(50):
        mid = (lo + hi) / 2
        # Compute h and its subgradient at mid
        c_vals = base.copy()
        for k in range(2*n - 1):
            if 0 <= k - i < n:
                if k == 2 * i:
                    c_vals[k] = base[k] + mid * mid
                else:
                    c_vals[k] = base[k] + 2 * mid * a[k - i]
        
        h_val = c_vals.max()
        # Active set
        active = np.where(np.isclose(c_vals, h_val, atol=1e-12))[0]
        # Subgradient at active set
        sgrads = []
        for k in active:
            if 0 <= k - i < n:
                if k == 2 * i:
                    sgrads.append(2 * mid)
                else:
                    sgrads.append(2 * a[k - i])
        
        if not sgrads:
            sgrad = 0
        else:
            sgrad_lo = min(sgrads)
            sgrad_hi = max(sgrads)
            if sgrad_lo <= 0 <= sgrad_hi:
                sgrad = 0  # at optimum
            elif sgrad_lo > 0:
                sgrad = 1  # increasing, go left
            else:
                sgrad = -1  # decreasing, go right
        
        if sgrad > 0:
            hi = mid
        elif sgrad < 0:
            lo = mid
        else:
            break
    
    optimal_x = max(0, (lo + hi) / 2)
    return optimal_x

def spectral_shaping(a0, n_iters=200, eta=0.5, verbose=False):
    """
    TIGER/GIRAFFE-like spectral shaping.
    Iteratively flattens the power spectrum while enforcing
    non-negativity and support constraints.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    fft_size = 1 << (2 * n - 2).bit_length()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t0 > 300:
            break
        # Compute spectrum
        A = rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)

        # Target: flatten the magnitude
        target_mag = np.sqrt(np.mean(mag ** 2))
        # Blend: reduce peaks, boost valleys
        mag_new = (1.0 - eta) * mag + eta * target_mag
        # Ensure minimum magnitude to avoid zeroing
        mag_new = np.maximum(mag_new, 0.01 * target_mag)

        # Reconstruct
        A_new = mag_new * np.exp(1j * phase)
        a_raw = irfft(A_new, fft_size)[:n]

        # Project: non-negative, sum=1
        a = np.maximum(a_raw, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
        else:
            a = a / s

        # Evaluate
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Gradually reduce eta to converge
        eta *= 0.995

        if verbose and it % 50 == 0:
            print(f"      SpectralShape it={it}: score={score:.6f} eta={eta:.4f}")

    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.GLPK, cp.PDLP]:
        try:

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])

            if prob.status in ('optimal', 'optimal_inaccurate'):

def _solve_lp_br(fixed, solver):
    """Solve LP: minimize M s.t. conv(fixed, b) <= M, b in simplex."""
    import cvxpy as cp
    n = len(fixed)

    # Build convolution matrix implicitly via constraint
    # conv(fixed, b)[k] = sum_j fixed[j] * b[k-j]
    # We express this as a linear constraint on b
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    # conv(fixed, b)[k] for k = 0, ..., 2n-2
    # = sum_{j: 0<=k-j<n} fixed[j] * b[k-j]
    # Let's build this as a vector
    conv_expr = np.zeros(2 * n - 1)
    # Actually, cvxpy needs expressions, not numpy arrays
    # Let's build the constraint list

    constraints = [cp.sum(b_var) == 1]

    # For each k, conv(fixed, b)[k] = sum_j fixed[j]*b[k-j]
    # = fixed[0]*b[k] + fixed[1]*b[k-1] + ... (valid terms)
    # This is a linear form in b_var
    for k in range(2 * n - 1):
        expr = 0
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            # expr = sum_{j=j_min}^{j_max} fixed[j] * b_var[k-j]
            for j in range(j_min, j_max + 1):
                if fixed[j] > 1e-15:
                    expr = expr + fixed[j] * b_var[k - j]
        constraints.append(expr <= M_var)

    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, verbose=False)
        except Exception:
            return None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    b_val = b_val / s
    return b_val


def fast_coord_descent(a0, time_limit, n_trials=8, verbose=False, label=""):
    """
    Fast coordinate descent using O(n) sensitivity analysis per coordinate.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    sum_a = 1.0
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve_sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            # Unaffected max
            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, conv[:k_start].max())
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, conv[k_end + 1:].max())

            affected = conv[k_start:k_end + 1]
            a_shifted = a  # a[k-i] for k = k_start..k_end, which is a[0..n-1]

            best_delta = 0.0
            best_new_max = M

            for trial in range(n_trials):
                if trial == 0:
                    delta = -0.3 * a[i]
                elif trial == 1:
                    delta = -0.15 * a[i]
                elif trial == 2:
                    delta = -0.05 * a[i]
                elif trial == 3:
                    delta = -0.02 * a[i]
                elif trial == 4:
                    delta = 0.02 * a[i]
                elif trial == 5:
                    delta = 0.05 * a[i]
                elif trial == 6:
                    delta = 0.15 * a[i]
                else:
                    delta = 0.3 * a[i]

                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx_2i] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_max = max(unaff_max, float(np.max(new_affected)))
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-16:
                a[i] = max(a[i] + best_delta, 0.0)
                sum_a += best_delta

                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx_2i] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(np.max(conv[k_start:k_end + 1])))

                cur_score = 2.0 * n * M / (sum_a * sum_a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > 30:
                break
        else:
            no_improve_sweeps = 0

        if verbose and sweeps % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      FCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, sweeps


def fine_coord_descent(a0, time_limit, verbose=False, label=""):
    """Fine coordinate descent with smaller factors for polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    sum_a = 1.0
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, conv[:k_start].max())
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, conv[k_end + 1:].max())

            affected = conv[k_start:k_end + 1]
            a_shifted = a

            best_delta = 0.0
            best_new_max = M

            for factor in [0.985, 0.97, 0.955, 0.99, 1.01, 1.045, 1.07, 1.085]:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = affected.copy() + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    new_affected[idx_2i] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_max = max(unaff_max, float(np.max(new_affected)))
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-17:
                a[i] = max(a[i] + best_delta, 0.0)
                sum_a += best_delta

                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx_2i] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(np.max(conv[k_start:k_end + 1])))

                cur_score = 2.0 * n * M / (sum_a * sum_a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved and sweeps > 10:
            break

    return best_a, best_score, sweeps


def init_fourier(n, rng):
    """Initialize using random Fourier superposition."""
    n_harmonics = int(rng.integers(5, 25))
    amplitudes = rng.random(n_harmonics) * 0.4
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = rng.permutation(np.arange(1, n_harmonics + 1))
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_cosine(n, rng):
    """Initialize with a cosine bell shape."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * rng.uniform(0.25, 0.6)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_u_shape(n, rng):
    """Initialize with U-shape (more mass at edges)."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    alpha = rng.uniform(0.5, 3.0)
    a = 1.0 + alpha * np.cos(np.pi * t) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_interpolate(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate via evaluate_sequence and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def peak_perturb(a0, rng, magnitude=0.05):
    """Perturb the sequence at peak-contributing positions to escape local optima."""
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    top_k = min(8, len(conv))
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    for k in peaks:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1)
        n_pick = max(2, len(idx) // 5)
        picks = rng.choice(idx, size=min(n_pick, len(idx)), replace=False)
        for i in picks:
            a[i] *= (1.0 + rng.standard_normal() * magnitude)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a0.copy()
    return a / s
