def fast_convolve_full(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)

def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    v = np.maximum(v, 0)
    s = v.sum()
    if s > 1e-15:
        v = v * target_sum / s
    # Use proper simplex projection
    # Project onto {x >= 0, sum(x) = target_sum}
    # This is: target_sum * project_onto_unit_simplex(v)
    n = len(v)

    cond = u * rho > (cssv - 1.0)

def qp_step(f, target_peak=None):
    """
    QP step: minimize (1/2)||g-f||^2 + t*M
    subject to g >= 0, sum(g) = 1, (f*g)[k] <= M for all k.
    """
    n = len(f)
    if target_peak is None:
        C = fast_convolve(f)
        target_peak = C.max()
    
    # Build constraint matrix A: (2n-1) x n
    # A[k, i] = f[k-i] for valid k-i
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi+1] = f[lo:hi+1]
    
    g = cp.Variable(n)
    M_var = cp.Variable()
    t = 0.1  # regularization parameter
    
    constraints = [
        g >= 0,
        cp.sum(g) == 1,
        A @ g <= M_var,
        M_var <= target_peak  # Don't allow peak to increase
    ]
    
    prob = cp.Problem(cp.Minimize(0.5 * cp.sum_squares(g - f) + t * M_var), constraints)
    
    try:
        prob.solve(solver=cp.PDLP, max_iters=100000)
    except:
        try:
            prob.solve(solver=cp.SCS, max_iters=100000)
        except:
            return f
    
    if g.value is None:
        return f
    
    g_val = np.maximum(g.value, 0)
    s = g_val.sum()
    if s < 1e-10:
        return f
    g_val /= s
    return g_val

"""
Auto-convolution peak minimization using log-sum-exp smoothed gradient descent
with adaptive annealing, momentum, QP-based proximal steps, and peak-focused
refinement. Explores multiple sequence lengths and window initializations.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho_idx = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    return np.maximum(v - theta, 0.0) * target_sum


def softmax_stable(x, beta=1.0):
    """Numerically stable softmax."""
    x = np.asarray(x, dtype=np.float64)
    scaled = x * beta
    m = scaled.max()
    if m > 700:
        scaled = scaled - m
    exp_vals = np.exp(scaled)
    s = exp_vals.sum()
    if s < 1e-300:
        return np.full(len(x), 1.0 / len(x))
    return exp_vals / s


def lse_gradient(a, beta):
    """
    Compute gradient of log(sum(exp(beta*C))) w.r.t. a, where C = a*a (auto-conv).
    Returns (gradient, peak, lse_value).
    """
    n = len(a)
    C = fast_convolve(a)
    p = softmax_stable(C, beta)
    # grad[i] = 2 * sum_j p[i+j] * a[j]
    # Compute via FFT: convolve a_rev with p, then extract
    a_rev = a[::-1]
    n_conv = n + len(p) - 1
    fft_sz = 1 << (n_conv - 1).bit_length()
    A_rev_hat = rfft(a_rev, fft_sz)
    P_hat = rfft(p, fft_sz)
    conv = irfft(A_rev_hat * P_hat, fft_sz)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    peak = C.max()
    scaled = C * beta
    sm = scaled.max()
    lse_val = sm + np.log(np.exp(scaled - sm).sum())
    return grad, peak, lse_val


def peak_subgradient(a, top_k=3):
    """
    Compute average subgradient of max_k C[k] over top_k peaks.
    Returns (subgradient, peak_value).
    """
    n = len(a)
    C = fast_convolve(a)
    peak_val = C.max()
    if top_k < len(C):
        top_idx = np.argpartition(C, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(C))
    # Weight by how close each peak is to the max
    w = C[top_idx].copy()
    w = w / w.sum()
    grad = np.zeros(n)
    for idx, wi in zip(top_idx, w):
        k = int(idx)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += wi * 2.0 * f_peak_slice(a, k, lo, hi)
    return grad, peak_val


def f_peak_slice(a, k, lo, hi):
    """Return a[k-hi:k-lo+1][::-1] which is the contribution of lag k to positions lo..hi."""
    return a[k - hi:k - lo + 1][::-1]

def qp_proximal_step(f, time_per_qp=10.0, top_k=200):
    """
    QP proximal step: minimize (1/2)||g-f||^2 + t*M
    subject to g >= 0, sum(g)=1, (f*g)[k] <= M for top-k constraints.
    """
    t0 = time.monotonic()
    n = len(f)
    C = fast_convolve(f)
    target_M = C.max()

    if top_k < len(C):
        top_idx = np.argpartition(C, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(C))

    # Build constraint matrix for selected lags
    K = len(top_idx)
    A = np.zeros((K, n))
    for row_i, k in enumerate(top_idx):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]

    g_var = cp.Variable(n)
    M_var = cp.Variable()
    t_reg = 0.5

    constraints = [
        g_var >= 0,
        cp.sum(g_var) == 1,
        A @ g_var <= M_var,
        M_var <= target_M + 1e-10
    ]

    prob = cp.Problem(cp.Minimize(0.5 * cp.sum_squares(g_var - f) + t_reg * M_var), constraints)

    solved = False
    for solver in [cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver, max_iters=200000)
            solved = True
            break
        except Exception:
            continue

    if not solved or g_var.value is None:
        return f

    g_val = np.maximum(g_var.value, 0.0)
    s = g_val.sum()
    if s < 1e-10:
        return f
    g_val /= s

    # Verify improvement
    new_peak = fast_convolve(g_val).max()
    cur_peak = fast_convolve(f).max()
    if new_peak <= cur_peak + 1e-10:
        return g_val
    return f


def flat_top_window(n):
    """Flat-top window normalized to sum 1."""
    x = np.arange(n)
    w = np.ones(n)
    a_coeff = [0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368]
    for i, ai in enumerate(a_coeff):
        w = w - ai * np.cos(2 * np.pi * i * x / max(1, n - 1) - i * np.pi)
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def hann_window(n):

def super_gaussian_window(n, p=3.0):
    """Super-Gaussian window: exp(-(2x-1)^p) normalized to sum 1."""
    x = 2.0 * np.arange(n) / max(1, n - 1) - 1.0
    w = np.exp(-np.abs(x) ** p)
    s = w.sum()
    if s < 1e-10:

    return w / s


def raised_cosine_window(n):

    t_phase1 = min(250.0, (deadline - time.monotonic()) * 0.15)
    print(f"  Budget: {t_phase1:.0f}s")

    # Candidate lengths
    lengths_to_try = [1024, 1152, 1280, 1536, 1920, 2048, 2560]
    if parent is not None:
        # Add parent length and nearby
        n_p = len(parent)
        lengths_to_try = list(set(lengths_to_try + [n_p, max(512, n_p - 128), min(4096, n_p + 128)]))

    lengths_to_try = [n for n in lengths_to_try if 512 <= n <= 4096]
    lengths_to_try = sorted(set(lengths_to_try))

    # For each length, try a few initializations
    init_types = [
        ("flat_top", flat_top_window),
        ("hann", hann_window),
        ("sg2", lambda n: super_gaussian_window(n, 2.0)),
        ("sg3", lambda n: super_gaussian_window(n, 3.0)),
        ("sg4", lambda n: super_gaussian_window(n, 4.0)),
        ("raised_cos", raised_cosine_window),
    ]

    per_len_budget = t_phase1 / len(lengths_to_try)
    per_init_budget = per_len_budget / len(init_types)

    results = []  # (score, n, best_f, init_name)
    global_best_f = None
    global_best_score = float('inf')

    if parent is not None:
        # Quick optimize parent first
        t_parent = min(30.0, per_len_budget)
        if t_parent > 5:
            fp, pp = quick_optimize(parent, t_parent)
            score_p = float(evaluate_sequence(fp.tolist()))
            if np.isfinite(score_p):
                results.append((score_p, len(fp), fp, "parent"))
                if score_p < global_best_score:
                    global_best_score = score_p
                    global_best_f = fp.copy()
                print(f"  Parent: n={len(fp)}, score={score_p:.8f}")

    for target_n in lengths_to_try:
        rem_phase1 = deadline - time.monotonic() - 20
        if rem_phase1 < 10:
            break
        actual_per_init = min(per_init_budget, rem_phase1 / 3)
        if actual_per_init < 3:
            break

        for init_name, init_func in init_types:
            if time.monotonic() > deadline - 30:
                break
            t_init = min(actual_per_init, deadline - time.monotonic() - 20)
            if t_init < 3:
                continue

            f_init = init_func(target_n)
            f_opt, p_opt = quick_optimize(f_init, t_init)
            score = float(evaluate_sequence(f_opt.tolist()))
            if np.isfinite(score):
                results.append((score, target_n, f_opt, init_name))
                if score < global_best_score:
                    global_best_score = score
                    global_best_f = f_opt.copy()

    if global_best_f is None:
        # Fallback
        global_best_f = flat_top_window(1152)
        global_best_score = float(evaluate_sequence(global_best_f.tolist()))

    results.sort(key=lambda x: x[0])
    print(f"\n  Top results after Phase 1:")
    for i, (sc, nn, _, nm) in enumerate(results[:8]):
        print(f"    {i+1}. n={nn}, score={sc:.8f}, init={nm}")

    # Phase 2: Deep optimization on the best 2-3 candidates
    print(f"\n=== Phase 2: Deep optimization (best={global_best_score:.8f}) ===")
    rem_phase2 = deadline - time.monotonic() - 90
    t_phase2 = min(rem_phase2 * 0.65, 900.0)
    print(f"  Budget: {t_phase2:.0f}s")

    # Optimize top candidates
    top_candidates = []
    seen_n = set()
    for sc, nn, ff, nm in results:
        if nn not in seen_n:
            top_candidates.append((sc, nn, ff))
            seen_n.add(nn)
        if len(top_candidates) >= 3:
            break

    for cand_i, (cand_score, cand_n, cand_f) in enumerate(top_candidates):
        rem_for_cand = t_phase2 / (len(top_candidates) - cand_i)
        rem_for_cand = min(rem_for_cand, deadline - time.monotonic() - 60)
        if rem_for_cand < 20:
            break

        print(f"\n  [Candidate {cand_i+1}] n={cand_n}, start_score={cand_score:.8f}, time={rem_for_cand:.0f}s")

        # Step A: LSE optimization
        t_lse = rem_for_cand * 0.5
        f_lse, p_lse = lse_optimize(cand_f, time_limit=t_lse)
        score_lse = float(evaluate_sequence(f_lse.tolist()))
        print(f"    After LSE: score={score_lse:.8f}")

        # Step B: QP proximal step
        rem_for_qp = rem_for_cand * 0.15
        if rem_for_qp > 5:
            elapsed_qp = time.monotonic()
            f_qp = qp_proximal_step(f_lse, time_per_qp=rem_for_qp, top_k=300)
            score_qp = float(evaluate_sequence(f_qp.tolist()))
            print(f"    After QP: score={score_qp:.8f}")
            if score_qp < score_lse:
                f_lse = f_qp
                p_lse = score_qp

        # Step C: Peak refinement
        rem_for_peak = deadline - time.monotonic() - 40
        rem_for_peak = min(rem_for_peak, rem_for_cand * 0.35)
        if rem_for_peak > 10:
            f_ref, p_ref = peak_refine(f_lse, time_limit=rem_for_peak, n_top=3)
            score_ref = float(evaluate_sequence(f_ref.tolist()))
            print(f"    After peak refine: score={score_ref:.8f}")
            if score_ref < score_lse:
                f_lse = f_ref
                score_lse = score_ref

        if score_lse < global_best_score:
            global_best_score = score_lse
            global_best_f = f_lse.copy()
            print(f"    *** NEW BEST: {global_best_score:.8f} (n={cand_n}) ***")

    # Phase 3: Final refinement on the overall best
    print(f"\n=== Phase 3: Final refinement (best={global_best_score:.8f}) ===")
    rem_final = deadline - time.monotonic() - 10
    t_final = min(rem_final * 0.7, 300.0)
    print(f"  Budget: {t_final:.0f}s")

    if t_final > 20:
        # Peak refinement with more peaks
        t_peak1 = t_final * 0.4
        f_r1, p_r1 = peak_refine(global_best_f, time_limit=t_peak1, n_top=2)
        score_r1 = float(evaluate_sequence(f_r1.tolist()))
        print(f"  After peak refine (2 peaks): {score_r1:.8f}")
        if score_r1 < global_best_score:
            global_best_score = score_r1
            global_best_f = f_r1.copy()

        # LSE polish with high beta
        rem2 = deadline - time.monotonic() - 10
        t_lse2 = min(rem2 * 0.3, t_final * 0.3)
        if t_lse2 > 10:
            f_r2, p_r2 = lse_optimize(global_best_f, time_limit=t_lse2,
                                       beta_start=10.0 * len(global_best_f),
                                       beta_end=50.0 * len(global_best_f))
            score_r2 = float(evaluate_sequence(f_r2.tolist()))
            print(f"  After LSE polish: {score_r2:.8f}")
            if score_r2 < global_best_score:
                global_best_score = score_r2
                global_best_f = f_r2.copy()

        # Final tiny-step peak refinement
        rem3 = deadline - time.monotonic() - 10
        t_rf = min(rem3 * 0.5, t_final * 0.3)
        if t_rf > 5:
            f_r3, p_r3 = peak_refine(global_best_f, time_limit=t_rf, n_top=1)
            score_r3 = float(evaluate_sequence(f_r3.tolist()))
            print(f"  After final peak (1 peak): {score_r3:.8f}")
            if score_r3 < global_best_score:
                global_best_score = score_r3
                global_best_f = f_r3.copy()

    # Final verification
    final_score = float(evaluate_sequence(global_best_f.tolist()))
    if not np.isfinite(final_score):
        # Fallback to parent
        if parent is not None:
            global_best_f = parent.copy()
            final_score = float(evaluate_sequence(global_best_f.tolist()))
        else:
            global_best_f = flat_top_window(1152)
            final_score = float(evaluate_sequence(global_best_f.tolist()))

    print(f"\n{'=' * 60}")
    print(f"FINAL SCORE: {final_score:.8f} (n={len(global_best_f)})")
    print(f"Target: 1.5027")
    print(f"Gap: {final_score - 1.5027:.6f}")
    print(f"{'=' * 60}")

    return [float(x) for x in global_best_f]
