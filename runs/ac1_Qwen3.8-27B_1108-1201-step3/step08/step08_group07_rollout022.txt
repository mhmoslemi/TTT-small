Multi-phase optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses mirror descent (multiplicative weights), LP fictitious play,
L-BFGS smoothing, and coordinate descent in sequence.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.linalg import toeplitz
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_best_response(f, M, prev_g=None):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None))
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return prev_g if prev_g is not None else f.copy()
    if g.value is None:
        return prev_g if prev_g is not None else f.copy()
    return np.maximum(g.value, 0)

    """Mirror descent with multiplicative weights on the probability simplex."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)

    a = a / s

"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Combines mirror descent (multiplicative weights), reduced-constraint LP fictitious play,
L-BFGS log-sum-exp smoothing, and coordinate descent.
Key innovations: Polyak step size for mirror descent, reduced LP constraints,
randomized subgradients for exploration, and multi-length search.

import numpy as np
import time
import cvxpy as cp
from scipy.linalg import toeplitz
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def project_simplex(v):
    """Project v onto the probability simplex {p: sum(p)=1, p>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def mirror_descent(a0, time_limit, verbose=True, label="MD", use_averaging=True):
    """Mirror descent with multiplicative weights and Polyak step size."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0

    # Lower bound: 2n/(2n-1) for sum(a)=1
    lb = 2.0 * n / (2.0 * n - 1.0)

    # Averaging buffer
    avg_a = a.copy()
    total_weight = 1.0

    while time.monotonic() - t0 < time_limit:
        iters += 1

        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))
        M = conv[k_star]

        # Subgradient direction: s_i = a[k_star - i] if 0 <= k_star - i < n
        s = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n, k_star + 1)
        if hi > lo:
            s[lo:hi] = a[k_star - np.arange(lo, hi)]

        # Polyak step size
        gap = best_score - lb
        s_norm_sq = np.dot(s, s)
        if s_norm_sq > 1e-30:
            eta = gap / s_norm_sq
        else:
            eta = 0.01 / np.sqrt(iters)
        eta = min(eta, 1.0)  # cap step size
        eta = max(eta, 1e-8)  # floor

        # Mirror descent update
        log_a = np.log(np.maximum(a, 1e-300)) - eta * s
        log_a -= log_a.max()
        exp_log_a = np.exp(log_a)
        a_new = exp_log_a / exp_log_a.sum()
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss

        score = compute_score(a_new)
        weight = 1.0
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()

        if use_averaging:
            avg_a = (avg_a * total_weight + a_new * weight) / (total_weight + weight)
            total_weight += weight
            avg_score = compute_score(avg_a)
            if avg_score < best_score - 1e-14:
                best_score = avg_score
                best_a = avg_a.copy()

        a = a_new

        if verbose and iters % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, eta={eta:.6f}, "
                  f"time={elapsed:.0f}s/{time_limit:.0f}s")

    # Check averaged solution
    if use_averaging:

        if avg_score < best_score - 1e-14:
            best_score = avg_score
            best_a = avg_a.copy()

    return best_a, best_score

    if prev_g is not None:
        g.value = prev_g

    C, active_ids = build_reduced_conv_matrix(f, M, threshold)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None))
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return (prev_g if prev_g is not None else f.copy()), len(active_ids)
    if g.value is None:
        return (prev_g if prev_g is not None else f.copy()), len(active_ids)
    return np.maximum(g.value, 0), len(active_ids)


def lp_fictitious_play(a0, time_limit, step=0.4, verbose=True, label="LP-FP",
                       threshold=0.05, use_reduced=True):
    """LP fictitious play with reduced constraints and warm-starting."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    prev_g = None
    stall_count = 0
    stall_threshold = 200

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()

        if use_reduced:
            g, n_active = solve_lp_reduced(a, M, prev_g=prev_g, threshold=threshold)
        else:
            g, n_active = solve_lp_reduced(a, M, prev_g=prev_g, threshold=1.0)

        prev_g = g
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g_norm = g / g_sum

        a_new = (1 - step) * a + step * g_norm
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss

        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1

        a = a_new

        if stall_count > stall_threshold:
            step = max(0.05, step * 0.7)
            rng = np.random.default_rng(iters)
            a = best_a + rng.standard_normal(n) * 0.0003
            a = np.maximum(a, 0)
            a = a / a.sum()
            prev_g = None
            stall_count = 0

        if verbose and iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, n_act={n_active}, "
                  f"step={step:.4f}, time={elapsed:.0f}s/{time_limit:.0f}s")

    return best_a, best_score


def softmax_from_z(z):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-30))


def lbgfs_optimize(a0, time_limit, T=500, maxiter=3000):
    """L-BFGS on smoothed log-sum-exp objective in log-space."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)
    fft_size = 1 << (2 * n - 1 - 1).bit_length()

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = softmax_from_z(z)
        conv = fast_convolve(a)
        shifted = conv * T
        max_shifted = shifted.max()
        lse = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
        return (2.0 * n / T) * np.exp(lse)

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        a = softmax_from_z(z)
        conv = fast_convolve(a)
        shifted = conv * T
        max_shifted = shifted.max()
        w = np.exp(shifted - max_shifted)
        w = w / w.sum()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
        S = float(np.dot(grad_a, a))
        grad_z = a * (grad_a - S)
        return (2.0 * n / T) * grad_z

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-15,
                     'maxfun': maxiter * 30, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        score = compute_score(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception:
        pass
    return best_a, best_score


def coordinate_descent(a0, time_limit, n_pairs=300, verbose=True, label="CD"):
    """Coordinate descent with pairwise transfers."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or a[i] < 1e-14:
                continue

            best_local = best_score
            best_t = 0.0
            base = min(a[i], a[j]) if a[j] > 1e-15 else a[i]

            for frac in [0.05, 0.1, 0.2, 0.35, 0.5, 0.7]:
                t = frac * base
                if t >= a[i] - 1e-15:
                    t = a[i] * 0.5
                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_t = t

            if best_t != 0.0:
                a[i] -= best_t
                a[j] += best_t
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = best_local
                best_a = a.copy()
                improved = True

        if not improved and iters > 30:
            noise = rng.standard_normal(n) * 0.001
            a = best_a.copy() + noise
            a = np.maximum(a, 0)
            a = a / a.sum()

        if verbose and iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")

    return best_a, best_score


def bussgang_iteration(a0, time_limit, verbose=True, label="BG"):
    """Bussgang-type spectral flattening."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        A = rfft(a, fft_size)
        power = np.abs(A) ** 2
        total_power = power.sum()
        target = total_power / fft_size
        ratio = np.where(power > 1e-30, target / power, 1.0)
        ratio = np.clip(ratio, 0.1, 10.0)
        tau = 0.3
        A_new = A * (ratio ** tau)
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        if verbose and iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score, verbose=True):
    """Evaluate a candidate and update best if improved."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        score = float('inf')
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Mirror descent (fast, many iterations)
    2. Mirror descent randomized (exploration)
    3. LP fictitious play with reduced constraints (precision)
    4. Bussgang spectral flattening
    5. L-BFGS smoothing polish
    6. Coordinate descent
    7. Final LP polish
    8. Nearby length exploration
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Initialized uniform: n={n_init}, score={best_score:.8f}")

    n_parent = len(parent)

    # Phase 1: Mirror Descent (fast)
    print(f"\n=== Phase 1: Mirror Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.20, 300.0)
    if rem > 60:
        print(f"  Budget: {t1:.0f}s, n={n_parent}")
        a_md, sc_md = mirror_descent(best_seq, t1, verbose=True, label="MD")
        best_seq, best_score = check_and_update(a_md, "mirror_desc", best_seq, best_score)

    # Phase 2: Mirror Descent Randomized
    print(f"\n=== Phase 2: MD Randomized (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.12, 180.0)
    if rem > 60:
        print(f"  Budget: {t2:.0f}s")
        a_mdr, sc_mdr = mirror_descent_randomized(best_seq, t2, alpha=0.3, verbose=True, label="MD-R")
        best_seq, best_score = check_and_update(a_mdr, "md_rand", best_seq, best_score)

    # Phase 3: LP Fictitious Play (reduced constraints)
    print(f"\n=== Phase 3: LP-FP Reduced (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.25, 450.0)
    if rem > 90:
        print(f"  Budget: {t3:.0f}s, n={len(best_seq)}")
        a_lp, sc_lp = lp_fictitious_play(best_seq, t3, step=0.4, verbose=True,
                                         label="LP-Reduced", threshold=0.08)
        best_seq, best_score = check_and_update(a_lp, "lp_reduced", best_seq, best_score)

    # Phase 4: LP with smaller threshold (more constraints = more precise)
    print(f"\n=== Phase 4: LP-FP Tighter (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.15, 250.0)
    if rem > 60:
        print(f"  Budget: {t4:.0f}s")
        a_lp2, sc_lp2 = lp_fictitious_play(best_seq, t4, step=0.2, verbose=True,
                                           label="LP-Tight", threshold=0.02)
        best_seq, best_score = check_and_update(a_lp2, "lp_tight", best_seq, best_score)

    # Phase 5: Bussgang
    print(f"\n=== Phase 5: Bussgang (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.08, 100.0)
    if rem > 40:
        print(f"  Budget: {t5:.0f}s")
        a_bg, sc_bg = bussgang_iteration(best_seq, t5, verbose=True)
        best_seq, best_score = check_and_update(a_bg, "bussgang", best_seq, best_score)

    # Phase 6: L-BFGS
    print(f"\n=== Phase 6: L-BFGS Polish (best={best_score:.8f}) ===")
    for T_val in [300, 1000, 3000]:
        rem = deadline - time.monotonic()
        t6 = min(rem * 0.08, 80.0)
        if rem < 30:
            break
        print(f"  L-BFGS T={T_val}, budget={t6:.0f}s")
        a_lb, sc_lb = lbgfs_optimize(best_seq, t6, T=T_val, maxiter=2000)
        best_seq, best_score = check_and_update(a_lb, f"lbgfs_T{T_val}", best_seq, best_score)

    # Phase 7: Coordinate Descent
    print(f"\n=== Phase 7: Coordinate Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t7 = min(rem * 0.12, 150.0)
    if rem > 30:
        print(f"  Budget: {t7:.0f}s")
        a_cd, sc_cd = coordinate_descent(best_seq, t7, n_pairs=300, verbose=True)
        best_seq, best_score = check_and_update(a_cd, "coord_desc", best_seq, best_score)

    # Phase 8: Final LP polish
    print(f"\n=== Phase 8: Final LP Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t8 = min(rem * 0.15, 200.0)
    if rem > 30:
        print(f"  Budget: {t8:.0f}s")
        a_lp3, sc_lp3 = lp_fictitious_play(best_seq, t8, step=0.1, verbose=True,
                                           label="LP-Final", threshold=0.03)
        best_seq, best_score = check_and_update(a_lp3, "lp_final", best_seq, best_score)

    # Phase 9: Nearby lengths
    print(f"\n=== Phase 9: Nearby Lengths (best={best_score:.8f}) ===")
    alt_lengths = [n_parent + 2, n_parent + 4, n_parent - 2, n_parent + 6, n_parent - 4]
    for alt_n in alt_lengths:
        if alt_n < 128 or alt_n > 4096:
            continue
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t9 = min(rem * 0.10, 60.0)
        init_alt = interpolate_sequence(best_seq, alt_n)
        print(f"  Trying n={alt_n}, budget={t9:.0f}s")
        a_alt, sc_alt = lbgfs_optimize(init_alt, t9, T=500, maxiter=1500)
        best_seq, best_score = check_and_update(a_alt, f"n{alt_n}", best_seq, best_score)

    # Phase 10: Final mirror descent on best
    print(f"\n=== Phase 10: Final MD (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t10 = min(rem * 0.4, 60.0)
        a_final, sc_final = mirror_descent(best_seq, t10, verbose=False, label="MD-end")
        best_seq, best_score = check_and_update(a_final, "md_end", best_seq, best_score)

    # Final verification
    if best_seq is not None:
        try:
            fs = float(evaluate_sequence(best_seq.tolist()))
            if np.isfinite(fs) and fs < best_score:
                best_score = fs
        except Exception:
            pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq) if best_seq is not None else 0})")
    return [float(x) for x in best_seq]
