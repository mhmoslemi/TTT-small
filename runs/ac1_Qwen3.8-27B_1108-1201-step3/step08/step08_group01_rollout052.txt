    return e / e.sum()

    L = 1 << (2*n - 1).bit_length()
    A = rfft(a, L)
    return irfft(A * A, L)[:2*n-1]

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth L-BFGS with 
continuation method on the log-sum-exp parameter, combined with 
coordinate descent polishing and multi-length search.
"""

import numpy as np
from scipy.fft import fft, ifft, rfft, irfft
from scipy.optimize import minimize
import time


def softmax_fn(x):
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()


def fft_conv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    L = 1 << (size - 1).bit_length()
    A = rfft(a, L)
    return irfft(A * A, L)[:size]


def smooth_obj_grad(b, beta, n):
    """
    Compute the smoothed objective 2n * f_beta(softmax(b)) and its gradient.
    f_beta(a) = (1/beta) * log(sum_k exp(beta * c_k)) where c = conv(a,a).
    """
    a = softmax_fn(b)
    conv = fft_conv(a)
    
    # Log-sum-exp for numerical stability
    max_c = conv.max()
    shifted = beta * (conv - max_c)
    sum_exp = np.exp(shifted).sum()
    f_beta = max_c + np.log(sum_exp) / beta
    
    # Softmax weights
    p = np.exp(shifted) / sum_exp
    
    # Cross-correlation: cc[i] = sum_j p[j+i] * a[j]
    L_cc = 1 << (3 * n).bit_length()  # >= 3n-2
    P = fft(np.concatenate([p, np.zeros(L_cc - len(p))]))
    A = fft(np.concatenate([a, np.zeros(L_cc - len(a))]))
    cc = np.real(ifft(P * np.conj(A)))[:n]
    
    # Gradient: dF/db = 4n * a * (cc - dot(p, conv))
    dot_pc = np.dot(p, conv)
    grad = 4.0 * n * a * (cc - dot_pc)
    
    obj = 2.0 * n * f_beta
    return obj, grad


def smooth_LBFGS(n, a_init, beta_list, max_iter_per_beta=300, time_budget=60.0):
    """Run L-BFGS on the smoothed problem with continuation in beta."""
    t0 = time.monotonic()
    b = np.log(np.maximum(a_init, 1e-15))
    best_a = a_init.copy()
    best_true_obj = np.inf
    
    for beta in beta_list:
        if time.monotonic() - t0 > time_budget:
            break
        
        result = minimize(
            lambda b_eval: smooth_obj_grad(b_eval, beta, n)[0],
            b,
            jac=lambda b_eval: smooth_obj_grad(b_eval, beta, n)[1],
            method='L-BFGS-B',
            options={'maxiter': max_iter_per_beta, 'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 50}
        )
        
        if result.x is not None:
            b = result.x
            a_candidate = softmax_fn(b)
            true_obj = 2.0 * n * fft_conv(a_candidate).max()
            if true_obj < best_true_obj:
                best_true_obj = true_obj
                best_a = a_candidate.copy()
    
    return best_a, best_true_obj


def coord_descent_polish(a0, time_budget=60.0, n_sweeps_max=200):
    """Coordinate descent polishing."""
    n = len(a0)
    a = np.maximum(a0, 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a0
    a = a / s
    
    conv = fft_conv(a)
    M = conv.max()
    best_a = a.copy()
    best_score = 2.0 * n * M
    t0 = time.monotonic()
    sweeps = 0
    
    while time.monotonic() - t0 < time_budget and sweeps < n_sweeps_max:
        sweeps += 1
        improved = False
        
        for i in range(0, n, 1):
            if a[i] < 1e-15:
                continue
            
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            
            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, conv[:k_start].max())
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, conv[k_end + 1:].max())
            
            affected = conv[k_start:k_end + 1].astype(np.float64)
            a_shifted = a[0:n]
            
            best_delta = 0.0
            best_new_max = M
            
            # Try several step sizes
            for factor in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue
                
                new_affected = affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx_2i] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
                
                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-16:
                    best_new_max = new_max
                    best_delta = delta
            
            if best_delta != 0.0:
                a[i] += best_delta
                a[i] = max(a[i], 0.0)
                
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx_2i] = 2.0 * best_delta * (a[i] - best_delta) + best_delta ** 2
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if not improved:
            break
    
    return best_a
