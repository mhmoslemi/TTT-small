        f = normalize(f)

from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient_fft(f, w):
    """Compute subgradient P[j] = sum_k w[k] * f[k-j] using FFT cross-correlation."""
    n = len(f)
    L = 1 << (3 * n).bit_length()
    F = rfft(f, L)
    W = rfft(w, L)
    P_full = irfft(np.conj(W) * F, L)
    return P_full[:n]


def solve_lp_direction(f, n, peak_indices, M, eps=1e-8):
    """Solve: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for k in peak_indices."""
    K = len(peak_indices)
    A = np.zeros((K, n))
    for row_i in range(K):
        k = int(peak_indices[row_i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    solved = False
    for solver in [cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver, max_iters=500000, eps=eps)
            solved = True
            break
        except Exception:
            continue

    if not solved or g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None
    return g_val

    best_obj = M_current
    t_grid = np.array([0.02, 0.04, 0.06, 0.08, 0.10, 0.15, 0.20, 0.30, 0.40, 0.50, 0.70])
    for t in t_grid:
        f_trial = (1.0 - t) * f + t * h
        f_trial = np.maximum(f_trial, 0.0)
        s = f_trial.sum()
        if s < 1e-10:

        f_trial = f_trial / s
        obj = fast_convolve(f_trial).max()
        if obj < best_obj - 1e-15:

            best_t = t

            if s < 1e-10:

            obj = fast_convolve(f_trial).max()
            if obj < best_obj - 1e-15:

        if len(peaks) < 5:

        if best_t <= 0:
            stuck += 1
            if stuck > restart_patience:
                break
            continue

        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f = f / s
        else:
            break

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_b - 1e-14:
            best_b = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1

        if it % 20 == 0:
            score = 2.0 * n * best_b
            print(f"    {label} it={it} obj={best_b:.10f} score={score:.6f} "
                  f"peaks={len(peaks)} t={time.monotonic()-t0:.0f}s")

        if it % 25 == 0:
            peak_frac = max(final_peak_frac, peak_frac - 0.004)

        if stuck > restart_patience // 2 and stuck % 4 == 0:
            f = f + rng.standard_normal(n) * 0.0008 * f
            f = np.maximum(f, 0.0)
            f = f / f.sum()

    return best_f, best_b


def mirror_descent_optimize(f_init, time_limit, lr_init=0.06, n_top=150,
                            label="md", seed=555):
    """Mirror descent with Nesterov momentum and adaptive temperature."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    log_f = np.log(f + 1e-30)
    momentum = np.zeros(n)
    best_f = f.copy()
    best_b = fast_convolve(f).max()
    lr = lr_init
    temp = 0.003
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)
    mu = 0.7

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        scaled = b / (temp * M)
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w_sel = w[top_idx].copy()
            ws = w_sel.sum()
            if ws > 1e-30:
                w_sel = w_sel / ws
            w_new = np.zeros(len(b))
            w_new[top_idx] = w_sel
            w = w_new

        grad = compute_subgradient_fft(f, w)

        momentum = mu * momentum - lr * grad
        log_f_new = log_f + momentum
        log_f_new -= log_f_new.max()
        f_new = np.exp(log_f_new)
        s = f_new.sum()
        if s < 1e-30:
            break
        f_new = f_new / s
        f = f_new
        log_f = np.log(f + 1e-30)

        if it % 20 == 0:
            obj = fast_convolve(f).max()
            if obj < best_b - 1e-14:
                best_b = obj
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.02, 0.15)
            else:
                no_improve += 1
                if no_improve > 8:
                    lr *= 0.9
                    no_improve = 0

            if it % 100 == 0:
                temp = min(temp * 0.95, 0.0005)
                score = 2.0 * n * best_b
                print(f"    {label} it={it} obj={best_b:.10f} score={score:.6f} "
                      f"lr={lr:.4f} temp={temp:.6f} t={time.monotonic()-t0:.0f}s")

            if no_improve > 40:
                f = f + rng.standard_normal(n) * 0.002 * f
                f = np.maximum(f, 1e-15)
                f = f / f.sum()
                log_f = np.log(f + 1e-30)
                momentum = np.zeros(n)
                lr = lr_init
                temp = 0.003
                no_improve = 0

    return best_f, best_b


def basin_hopping_lp(f_init, time_limit, label="bh", seed=777,
                     n_restarts=8, restart_frac=0.85, perturb_scale=0.05):
    """Basin hopping: periodically make large perturbations and re-optimize."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-10:
        f = f / s
    best_f = f.copy()
    best_b = fast_convolve(f).max()
    rng = np.random.default_rng(seed)
    n_restarts_done = 0

    while time.monotonic() - t0 < time_limit and n_restarts_done < n_restarts:
        phase_limit = (time_limit - (time.monotonic() - t0)) * restart_frac
        if phase_limit < 10:
            phase_limit = time_limit - (time.monotonic() - t0)

        f_phase, b_phase = lp_optimize(
            f, time_limit=phase_limit, label=label,
            initial_peak_frac=0.95, final_peak_frac=0.85,
            max_peaks=300, restart_patience=20, lp_eps=1e-8,
            seed=seed + n_restarts_done
        )

        if b_phase < best_b - 1e-14:
            best_b = b_phase
            best_f = f_phase.copy()
            print(f"    {label} restart={n_restarts_done} score={2*n*best_b:.6f} "
                  f"t={time.monotonic()-t0:.0f}s")

        n_restarts_done += 1
        if time.monotonic() - t0 >= time_limit:
            break

        f = best_f + rng.standard_normal(n) * perturb_scale * best_f
        f = np.maximum(f, 0.0)
        f = f / f.sum()

    return best_f, best_b


def coordinate_descent(f_init, time_limit, label="cd"):
    """Coordinate descent on individual elements."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_b = fast_convolve(f).max()
    multipliers = np.array([0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0])

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue
            best_j_val = f[j]
            best_j_obj = best_b
            for mult in multipliers:
                f_trial = f.copy()
                f_trial[j] = f[j] * mult
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial = f_trial / s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-14:
                    best_j_obj = obj
                    best_j_val = f[j] * mult
            if abs(best_j_val - f[j]) > 1e-15:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f = f / s
                if best_j_obj < best_b - 1e-14:
                    best_b = best_j_obj
                    improved = True
        if not improved:
            break

    return f, best_b


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    n = len(f)
    if n == 1:
        return np.full(new_n, 1.0 / new_n)
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new = f_new / s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def make_initialization_uniform(n):
    return np.ones(n) / n


def make_initialization_cosine(n):
    x = np.linspace(0, np.pi, n)
    f = np.maximum(np.cos(x / 2) ** 2, 0.0)
    s = f.sum()
    if s > 1e-10:
        f = f / s
    else:
        f = np.ones(n) / n
    return f


def make_initialization_hamming(n):
    f = 0.54 + 0.46 * np.cos(2 * np.pi * np.arange(n) / (n - 1))
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s > 1e-10:
        f = f / s
    else:
        f = np.ones(n) / n
    return f


def make_initialization_blackman(n):
    x = np.arange(n) / (n - 1)
    f = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s > 1e-10:
        f = f / s
    else:
        f = np.ones(n) / n
    return f


def make_initialization_two_bump(n, seed=123):
    """Create a lopsided sequence: larger on one side."""
    rng = np.random.default_rng(seed)
    split = n // 2 + rng.integers(-n // 10, n // 10 + 1)
    f = np.zeros(n)
    f[:split] = 1.0
    f[split:] = 0.3
    s = f.sum()
    if s > 1e-10:
        f = f / s
    return f


def make_initialization_random(n, seed=42):
    rng = np.random.default_rng(seed)
    f = rng.random(n)
    f = f / f.sum()
    return f
