```python
def objective_smoothed(a, beta):
    """Smoothed version of 2*n*max(conv(a,a))/sum(a)^2"""
    n = len(a)
    conv = np.convolve(a, a)
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return 1e10
    # Log-sum-exp for smooth max
    max_conv = np.max(conv)
    # Smoothed max
    shifted = beta * (conv - max_conv)
    log_sum_exp = max_conv + np.log(np.sum(np.exp(shifted))) / beta
    return 2 * n * log_sum_exp / (sum_a ** 2)
```

"""
Spectral initialization + L-BFGS-B + LP refinement for minimizing
2*n*max(conv(a,a))/sum(a)^2.

Key ideas:
1. Spectral initialization: a[k] ∝ (first n coeffs of sqrt(1+x+...+x^{2n-2}))
   This gives an approximately flat auto-convolution.
2. L-BFGS-B with analytical gradient on the smoothed (log-sum-exp) objective.
3. LP-based refinement using PDLP solver.
4. Multi-length exploration.

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def spectral_init(n, fft_size=None):
    """
    Compute the spectral initialization: first n coefficients of
    the polynomial square root of (1 + x + ... + x^{2n-2}).
    """
    if fft_size is None:
        fft_size = 1 << (2 * n - 1).bit_length()
    
    # Target: all-ones vector of length 2n-1
    target = np.zeros(fft_size)
    target[:2 * n - 1] = 1.0
    
    # FFT of target
    T = fft(target)
    
    # Square root in frequency domain
    # Choose branch such that P[0] > 0 (DC component positive)
    mag = np.abs(T)
    phase = np.angle(T)
    P_mag = np.sqrt(mag)
    P_phase = phase / 2.0
    P = P_mag * np.exp(1j * P_phase)
    
    # Ensure DC is positive
    if P[0].real < 0:
        P = -P
    
    # Inverse FFT
    p = ifft(P).real
    
    # Take first n coefficients
    a = p[:n]
    a = np.maximum(a, 0.0)
    
    s = a.sum()
    if s > 1e-10:
        a = a / s
    else:
        a = np.ones(n) / n
    
    return a


def objective_and_grad(a, beta):
    """
    Compute the smoothed objective and its gradient.
    Objective: 2*n*lsmax(conv(a,a))/sum(a)^2
    where lsmax(x) = (1/beta)*log(sum(exp(beta*x)))
    """
    n = len(a)
    conv = fast_convolve(a)
    s = a.sum()
    
    if s < 1e-10:
        return 1e10, np.zeros(n)
    
    # Log-sum-exp with numerical stability
    max_conv = np.max(conv)
    shifted = beta * (conv - max_conv)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    log_sum = max_conv + np.log(np.sum(exp_shifted)) / beta
    
    # Weights (softmax)
    w = exp_shifted / np.sum(exp_shifted)
    
    # Gradient of log-sum-exp w.r.t. a[i]:
    # d(lse)/da[i] = sum_k w[k] * d(conv[k])/da[i] = sum_k w[k] * a[k-i] (if valid)
    # This is a cross-correlation of w and a
    # d(conv[k])/da[i] = a[k-i] if 0 <= k-i < n
    # grad[i] = sum_{k} w[k] * a[k-i] * [0<=k-i<n]
    # = sum_{j=0}^{n-1} w[i+j] * a[j] (where j = k-i, k = i+j)
    # So grad[i] = sum_{j=0}^{min(n-1, 2n-2-i)} w[i+j] * a[j]
    
    # Compute via FFT (cross-correlation)
    # w has length 2n-1, a has length n
    # We want: for each i in 0..n-1, sum_{j=0}^{n-1} w[i+j] * a[j] where i+j < 2n-1
    # This is the cross-correlation of w and a, evaluated at specific lags
    
    # Alternative: just compute directly using the convolution structure
    # grad[i] = sum_{j: 0<=j<n, 0<=i+j<2n-1} w[i+j] * a[j]
    # For i=0: sum_j w[j]*a[j] for j=0..min(n-1,2n-2)
    # For i=n-1: sum_j w[n-1+j]*a[j] for j=0..min(n-1, n-1)
    
    # This is like a "sliding dot product" - can be done with FFT
    # Let's use the fact that it's a cross-correlation:
    # grad = correlate(w, a) evaluated at lags 0, 1, ..., n-1
    # Where correlate(w, a)[k] = sum_j w[j+k] * a[j]
    
    # Using FFT: correlate(w, a) = ifft(fft(w) * conj(fft(a)))
    # But we need to be careful about lengths.
    
    # Simpler approach for moderate n: direct computation
    # For n up to ~2000, O(n^2) is fine for the gradient
    if n <= 2000:
        grad_s = np.zeros(n)
        for i in range(n):
            # sum over j where 0 <= j < n and 0 <= i+j < 2n-1
            j_max = min(n - 1, 2 * n - 2 - i)
            if j_max >= 0:
                grad_s[i] = np.dot(w[i:i + j_max + 1], a[:j_max + 1])
    else:
        # FFT-based computation
        fft_size = 1 << (2 * n - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        # cross-correlation: ifft(W * conj(A)) gives sum_j w[j] * a[j+k]
        # But we want sum_j w[i+j] * a[j] = sum_j w[j'] * a[j'-i] where j' = i+j
        # = correlate(w, a) at lag i
        # correlate(w, a)[i] = sum_j w[j+i] * a[j]
        # = ifft(fft(w) * conj(fft(a))) at appropriate index
        # Actually: ifft(fft(w) * conj(fft(a))) [k] = sum_j w[(j+k) mod N] * a[j]
        # For k = i (small): = sum_j w[j+i] * a[j] (when j+i < N)
        # Which is what we want!
        corr = irfft(W * np.conj(A), fft_size)[:n]
        grad_s = corr
    
    # Full gradient of f = 2*n*lsmax/s^2
    # df/da[i] = 2*n * (grad_s[i] * s - 2*lsmax) / s^3
    coeff = 2.0 * n / (s ** 3)
    grad = coeff * (grad_s * s - 2.0 * log_sum)
    
    obj = 2.0 * n * log_sum / (s ** 2)
    return obj, grad


def lbgfgs_optimize(a_init, time_limit, beta_start=10.0, beta_end=200.0, 
                    max_iter=200, label="lbf"):
    """
    L-BFGS-B optimization with adaptive beta (smoothing parameter).
    """
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    best_score = 2.0 * n * best_obj
    
    # Adaptive beta: start smooth, end sharp
    n_anneal_steps = 5
    beta_schedule = np.geomspace(beta_start, beta_end, n_anneal_steps)
    
    for beta_idx, beta in enumerate(beta_schedule):
        t_remaining = time_limit - (time.monotonic() - t0)
        if t_remaining < 2:
            break
        
        max_this_phase = max(20, max_iter // n_anneal_steps)
        
        def fun(x):
            obj, _ = objective_and_grad(x, beta)
            return obj
        
        def grad(x):
            _, g = objective_and_grad(x, beta)
            return g
        
        try:
            res = minimize(fun, a, jac=grad, method='L-BFGS-B',
                          bounds=[(0, 1000)] * n,
                          options={'maxiter': max_this_phase, 
                                   'ftol': 1e-12, 'gtol': 1e-8,
                                   'maxfun': max_this_phase * 10})
            a = np.maximum(res.x, 0.0)
            s = a.sum()
            if s > 1e-10:
                a = a / s
        except Exception as e:
            print(f"  L-BFGS-B error: {e}")
            break
        
        # Evaluate true objective
        true_obj = fast_convolve(a).max()
        if true_obj < best_obj - 1e-12:
            best_obj = true_obj
            best_score = 2.0 * n * best_obj
            best_a = a.copy()
        
        elapsed = time.monotonic() - t0
        print(f"  {label} beta={beta:.1f} obj={best_obj:.10f} "
              f"score={best_score:.6f} t={elapsed:.0f}s")
    
    return best_a, best_score


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_polish(f_init, time_limit, n_top=100, lr_init=0.002, 
                       momentum=0.9, label="sg"):
    """Subgradient optimization with adaptive parameters."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    f = f / s
    
    v = np.zeros(n)
    lr = lr_init
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    no_improve = 0
    it = 0
    rng = np.random.default_rng(12345)
    temp = 0.001
    
    while time.monotonic() - t0 < time_limit:
        batch_size = max(1, min(10, int((time_limit - (time.monotonic() - t0)) / 0.01)))
        
        for _ in range(batch_size):
            b = fast_convolve(f)
            M = b.max()
            
            # Softmax weighting
            scaled = (b - M) / temp
            scaled = np.clip(scaled, -500, 0)
            w = np.exp(scaled)
            w_sum = w.sum()
            if w_sum < 1e-30:
                w = np.zeros(len(b))
                w[np.argmax(b)] = 1.0
            else:
                w = w / w_sum
            
            # Restrict to top peaks for efficiency
            if n_top < len(b):
                top_idx = np.argpartition(b, -n_top)[-n_top:]
                w_sel = w[top_idx].copy()
                ws = w_sel.sum()
                w_new = np.zeros(len(b))
                w_new[top_idx] = w_sel / ws if ws > 1e-30 else 0
                w = w_new
            
            # Compute subgradient: sum_k w[k] * a[k-i] for valid k-i
            g = np.zeros(n)
            if n <= 1500:
                for i in range(n):
                    j_max = min(n - 1, 2 * n - 2 - i)
                    if j_max >= 0:
                        g[i] = np.dot(w[i:i + j_max + 1], f[:j_max + 1])
            else:
                fft_size = 1 << (2 * n - 1).bit_length()
                W = rfft(w, fft_size)
                F = rfft(f, fft_size)
                g = irfft(W * np.conj(F), fft_size)[:n]
            
            # Update
            v = momentum * v - lr * g
            f_new = project_simplex(f + v)
            
            # Accept if better
            obj_new = fast_convolve(f_new).max()
            if obj_new < M:
                f = f_new
                if obj_new < best_obj - 1e-13:
                    best_obj = obj_new
                    best_score = 2.0 * n * best_obj
                    best_f = f.copy()
                    no_improve = 0
                    lr = min(lr * 1.02, 0.05)
                else:
                    no_improve += 1
            else:
                no_improve += 1
            
            it += 1
        
        if no_improve > 30:
            lr *= 0.9
            temp = min(temp * 1.3, 1.0)
            if no_improve > 60:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                v = np.zeros(n)
                no_improve = 0
        
        if it % 100 == 0:
            print(f"    {label} it={it} obj={best_obj:.10f} "
                  f"score={best_score:.6f} lr={lr:.5f} t={time.monotonic()-t0:.0f}s")
    
    return best_f, best_score


def lp_refine(f_init, time_limit, top_k_init=50, top_k_max=500, label="lp"):
    """LP-based refinement using PDLP solver."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    f = f / s
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    
    it = 0
    stuck = 0
    top_k = top_k_init
    full_len = 2 * n - 1
    momentum_dir = np.zeros(n)
    beta_mom = 0.4
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()
        
        # Peak selection
        if top_k < full_len:
            top_idx = np.argpartition(b, -top_k)[-top_k:]
            # Add boundary peaks
            n_bound = min(n, 5)
            boundary = list(range(min(n_bound, full_len))) + \
                       list(range(max(0, full_len - n_bound), full_len))
            all_idx = list(set(list(top_idx) + boundary))
            all_idx.sort(key=lambda i: b[i], reverse=True)
            indices = np.array(all_idx[:top_k])
        else:
            indices = np.arange(full_len)
        
        # Build constraint matrix
        K = len(indices)
        A_mat = np.zeros((K, n))
        for i in range(K):
            k = indices[i]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A_mat[i, lo:hi + 1] = f[lo:hi + 1]
        
        # Solve LP: max sum(g) s.t. g >= 0, A_mat @ g <= M
        g_var = cp.Variable(n)
        prob = cp.Problem(cp.Maximize(cp.sum(g_var)), 
                         [g_var >= 0, A_mat @ g_var <= M])
        
        solved = False
        for solver in [cp.PDLP, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver, 
                         **({'eps': 1e-9} if solver == cp.PDLP else 
                            {'max_iters': 100000, 'eps_abs': 1e-9, 'eps_rel': 1e-9} if solver == cp.SCS else
                            {'max_iters': 50000}))
                if g_var.value is not None:
                    g_val = np.maximum(g_var.value, 0.0)
                    if np.all(np.isfinite(g_val)):
                        solved = True
                        break
            except Exception:
                continue
        
        if not solved:
            stuck += 2
            if stuck > 50:
                break
            top_k = min(top_k + 20, top_k_max)
            continue
        
        S_g = g_val.sum()
        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > 30:
                # Try perturbation
                rng = np.random.default_rng(it)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                best_obj = fast_convolve(f).max()
                stuck = 0
            continue
        
        h = g_val / S_g
        
        # Momentum
        momentum_dir = beta_mom * momentum_dir + (1 - beta_mom) * h
        mom_sum = momentum_dir.sum()
        h_mom = momentum_dir / mom_sum if mom_sum > 1e-10 else h
        
        # Line search
        best_t = 0.0
        best_trial_obj = M
        best_trial = f.copy()
        
        for direction in [h, h_mom]:
            for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
                trial = (1 - t) * f + t * direction
                trial = np.maximum(trial, 0.0)
                ts = trial.sum()
                if ts < 1e-10:
                    continue
                trial = trial / ts
                obj = fast_convolve(trial).max()
                if obj < best_trial_obj:
                    best_trial_obj = obj
                    best_t = t
                    best_trial = trial.copy()
        
        if best_t == 0.0:
            stuck += 1
            if stuck > 40:
                rng = np.random.default_rng(it + 999)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                stuck = 0
            top_k = min(top_k + 10, top_k_max)
            continue
        
        f = best_trial
        cur_obj = fast_convolve(f).max()
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_score = 2.0 * n * best_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
        
        # Grow top_k
        if it % 10 == 0 and top_k < top_k_max:
            top_k = min(top_k + 10, top_k_max)
        
        if it % 20 == 0:
            print(f"    {label} it={it} obj={best_obj:.10f} "
                  f"score={best_score:.6f} k={K} t={time.monotonic()-t0:.0f}s")
    
    return best_f, best_score


def interp_sequence(src, target_n):
    """Interpolate a sequence to a new length."""
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result

    """

    1. Spectral initialization for multiple n values
    2. L-BFGS-B with adaptive smoothing
    3. LP refinement
    4. Subgradient polish
    5. Final verification
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    # Load parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
    except Exception:
        pass

        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    else:

        print(f"No parent, using uniform n=512, score={p_score:.6f}")

        if time.monotonic() > deadline - 600:
            break
