from scipy.fft import rfft, irfft
import cvxpy as cp
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_direction(f, g):
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]

def build_full_constraint_matrix(f, n):
    """Build matrix where row k encodes (f*g)[k] = A[k,:] @ g"""
    K = 2 * n - 1
    A = np.zeros((K, n))
    for k in range(K):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = f[lo:hi + 1]
    return A

def solve_full_lp(f, n, M, solver='PDLP', eps=1e-8):
    """Solve: max sum(g) s.t. g >= 0, (f*g)(k) <= M for all k"""
    K = 2 * n - 1
    A = build_full_constraint_matrix(f, n)
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    if solver == 'PDLP':
        prob.solve(solver=cp.PDLP, max_iter=100000, eps_abs=eps, eps_rel=eps)
    elif solver == 'MOSEK':
        prob.solve(solver=cp.MOSEK)
    elif solver == 'SCS':
        prob.solve(solver=cp.SCS, max_iters=500000, eps=eps)
    else:
        prob.solve(solver=cp.SCS, max_iters=500000, eps=eps)
    
    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None
    return g_val
