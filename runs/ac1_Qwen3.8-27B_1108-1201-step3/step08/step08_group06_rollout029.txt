from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

                     n_decrease=100, verbose=True):

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score_fast(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def smoothed_gradient(a, T):
    """Compute the gradient of T*logsumexp(conv(a,a)/T) w.r.t. a.
    Returns (gradient, p) where p is the softmax weight vector."""
    conv = fast_convolve(a)
    conv_shifted = conv / T
    conv_max = conv_shifted.max()
    exp_conv = np.exp(conv_shifted - conv_max)
    p = exp_conv / exp_conv.sum()
    # Gradient: g_j = 2 * sum_k p[k] * a[k-j]
    g = 2.0 * np.correlate(p, a, mode='valid')
    return g, p


def mirror_descent(a_init, time_limit, T_start=0.5, T_end=0.001, verbose=True):
    """Mirror descent with continuation method."""
    a = np.maximum(a_init, 1e-15).astype(np.float64)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_score = compute_score_fast(a)
    t0 = time.monotonic()
    
    # Geometric temperature schedule
    n_temps = 12
    T_values = T_start * (T_end / T_start) ** (np.arange(n_temps) / (n_temps - 1))
    # Give more time to smaller T values
    time_weights = np.array([0.5, 0.6, 0.7, 0.8, 0.9, 1.0, 1.0, 1.1, 1.2, 1.3, 1.4, 1.5])
    time_weights = time_weights / time_weights.sum()
    
    for ti, T in enumerate(T_values):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        T_budget = remaining * time_weights[ti] / time_weights[ti:].sum()
        T_budget = min(T_budget, remaining - 5)
        if T_budget < 2:
            T_budget = remaining - 2
        
        if verbose:
            print(f"    T={T:.4f}, budget={T_budget:.0f}s, best={best_score:.8f}")
        
        T_start_time = time.monotonic()
        lr = 0.05  # initial step size
        prev_a = None
        prev_g = None
        iter_count = 0
        stall_count = 0
        last_best = best_score
        
        while time.monotonic() - T_start_time < T_budget:
            g, p = smoothed_gradient(a, T)
            ga = np.dot(g, a)
            z_grad = g - a * ga  # natural gradient (projection to tangent space)
            
            # BB step size
            if prev_a is not None and prev_g is not None:
                s_vec = a - prev_a
                y_vec = g - prev_g
                sy = np.dot(s_vec, y_vec)
                ss = np.dot(s_vec, s_vec)
                if sy > 1e-12 and ss > 1e-12:
                    lr_bb = ss / sy
                    lr = 0.5 * (lr + min(lr_bb, 0.2))
                    lr = max(lr, 0.001)
            
            # Multiplicative update in log-space
            log_a = np.log(np.maximum(a, 1e-30))
            log_a = log_a - lr * z_grad
            log_a = log_a - log_a.max()
            a_new = np.exp(log_a)
            a_new = np.maximum(a_new, 1e-30)
            a_new = a_new / a_new.sum()
            
            prev_a = a
            prev_g = g
            a = a_new
            iter_count += 1
            
            # Periodic score check
            if iter_count % 20 == 0:
                score = compute_score_fast(a)
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    stall_count = 0
                else:
                    stall_count += 1
                
                # Adapt lr if stalling
                if stall_count > 20:
                    lr *= 0.7
                    stall_count = 0
                
                # Reset BB if stalling
                if stall_count > 30:
                    prev_a = None
                    prev_g = None
                    stall_count = 0
                    lr = max(lr * 0.5, 0.001)
    
    return best_a, best_score

def _eval_transfer(a, i, j, t, n):
    """Evaluate the score after transferring t from i to j."""
    a_trial = a.copy()
    a_trial[i] -= t
    a_trial[j] += t
    if a_trial[i] < -1e-15 or a_trial[j] < -1e-15:
        return 1e10
    a_trial[i] = max(0, a_trial[i])
    a_trial[j] = max(0, a_trial[j])
    s = a_trial.sum()
    if s < 1e-15:
        return 1e10
    # Don't normalize for speed (sum is approximately 1)
    conv = fast_convolve(a_trial)
    return 2.0 * n * conv.max() / (s * s)

    We minimize max_k (conv_k + t*delta_k) over t in [-a[j], a[i]].
    """
    # Compute delta_k
    delta = np.zeros(2 * n - 1)
    # a_{k-i} for k in [i, i+n-1]: delta[k] -= 2*a[k-i] → delta[i:i+n] -= 2*a
    # a_{k-j} for k in [j, j+n-1]: delta[k] += 2*a[k-j] → delta[j:j+n] += 2*a
    delta[i:i+n] -= 2.0 * a
    delta[j:j+n] += 2.0 * a
    
    c = conv  # current convolution values
    
    # Binary search for minimum V
    V_lo = 0.0
    V_hi = c.max()  # at t=0, max is c.max()
    
    # Check bounds: the min is at most c.max() (achieved at t=0)
    # and at least c.min() - something
    
    for _ in range(60):
        V_mid = (V_lo + V_hi) / 2.0
        
        # Check feasibility for V_mid
        t_low = -a[j]
        t_high = a[i]
        
        pos_mask = delta > 1e-15
        if pos_mask.any():
            bounds = (V_mid - c[pos_mask]) / delta[pos_mask]
            t_high = min(t_high, bounds.min())
        
        neg_mask = delta < -1e-15
        if neg_mask.any():
            bounds = (V_mid - c[neg_mask]) / delta[neg_mask]
            t_low = max(t_low, bounds.max())
        
        zero_mask = np.abs(delta) <= 1e-15
        if zero_mask.any():
            if c[zero_mask].max() > V_mid:
                V_lo = V_mid
                continue
        
        if t_low <= t_high:
            V_hi = V_mid
        else:
            V_lo = V_mid
    
    V_opt = V_hi
    
    # Find optimal t
    t_low = -a[j]
    t_high = a[i]
    
    pos_mask = delta > 1e-15
    if pos_mask.any():
        bounds = (V_opt - c[pos_mask]) / delta[pos_mask]
        t_high = min(t_high, bounds.min())
    
    neg_mask = delta < -1e-15
    if neg_mask.any():
        bounds = (V_opt - c[neg_mask]) / delta[neg_mask]
        t_low = max(t_low, bounds.max())
    
    if t_low > t_high:
        t_low = t_high = 0.0  # fallback to no transfer
    
    t_opt = (t_low + t_high) / 2.0
    
    return t_opt, 2.0 * n * V_opt

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed mirror descent (multiplicative weight updates) with continuation method
2. Fast coordinate descent with linear-approximation optimal transfer (O(n) per pair)
3. Greedy LP direction finding with geometric mean interpolation
4. Length exploration near the parent
Key innovation: O(n) per-pair optimal transfer via binary search on the min-max of
lines, and O(n log n) per-step smoothed subgradient via FFT cross-correlation.

import numpy as np

from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score_fast(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def smoothed_gradient(a, T):
    """Compute gradient of T*logsumexp(conv(a,a)/T) w.r.t. a."""
    conv = fast_convolve(a)
    conv_shifted = conv / T
    conv_max = conv_shifted.max()
    exp_conv = np.exp(conv_shifted - conv_max)
    p = exp_conv / exp_conv.sum()
    g = 2.0 * np.correlate(p, a, mode='valid')
    return g, p, conv

def build_C_sparse(f):
    """For the LP: the constraint is conv(f,g)_k <= M for all k.
    conv(f,g)_k = sum_j f_{k-j} * g_j (with valid indices).
    We need C[k,j] = f_{k-j} if 0<=k-j<n else 0."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = f
    return C


def greedy_lp_direction(f, M):
    """Greedy LP: find g maximizing sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    g = np.zeros(n)
    conv_vals = np.zeros(2 * n - 1)

    # Process in random order for diversity
    order = np.arange(n)

    for j in order:
        k_start = j
        k_end = j + n - 1
        slack = M - conv_vals[k_start:k_end + 1]
        f_slice = f  # f[k-j] for k in [j, j+n-1] is f[0],...,f[n-1]

        valid = f_slice > 1e-15
        if not np.any(valid):
            continue

        max_gj = np.inf
        idx_valid = np.where(valid)[0]
        for idx in idx_valid:
            if slack[idx] < 0:
                max_gj = 0.0
                break
            cand = slack[idx] / f_slice[idx]
            if cand < max_gj:
                max_gj = cand

        max_gj = max(0.0, max_gj)
        g[j] = max_gj
        conv_vals[k_start:k_end + 1] += f_slice * max_gj

    return g

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
