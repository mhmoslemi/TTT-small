```python
def solve_best_direction_lp(f, n_constraints=None):
    """
    Solve: min M s.t. sum(g)=1, g>=0, (f*g)[k] <= M for all k in selected set.
    Returns (g, M_optimal) or (None, M_current) on failure.
    """
    n = len(f)
    full_len = 2 * n - 1
    
    if n_constraints is None or n_constraints >= full_len:
        indices = np.arange(full_len)
    else:
        b = fast_convolve(f)
        # Select top constraints
        indices = np.argpartition(b, -n_constraints)[-n_constraints:]
        indices = np.sort(indices)
    
    K = len(indices)
    
    # Build constraint matrix A (K x n)
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi+1] = f[lo:hi+1]
    
    # LP: min M
    # Variables: [g_0, ..., g_{n-1}, M]
    # Objective: [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: A@g - M <= 0 → A@g + (-1)*M <= 0
    # sum(g) = 1 → handled as equality
    # g >= 0 → handled by bounds
    
    A_ub = np.hstack([A, -np.ones((K, 1))])
    b_ub = np.zeros(K)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: g_j >= 0 (j < n), M >= 0
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        g = res.x[:n]
        M_opt = res.x[n]
        g = np.maximum(g, 0)
        s = g.sum()

            g = g / s
        return g, M_opt
    return None, fast_convolve(f).max()
```

```python
    full_len = 2 * n - 1
    
    scaled = (b - M) / temp
    scaled = np.clip(scaled, -50.0, 0.0)
    w = np.exp(scaled)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(full_len)
        w[np.argmax(b)] = 1.0
    else:
        w = w / wsum
    
    if n_top < full_len:
        top_idx = np.argpartition(b, -n_top)[-n_top:]
        w_sel = w[top_idx].copy()
        ws = w_sel.sum()
        if ws > 1e-30:
            w_sel = w_sel / ws
        w_new = np.zeros(full_len)
        w_new[top_idx] = w_sel
        w = w_new
    
    # grad[j] = 2 * sum_{k: 0<=k-j<n} w[k] * f[k-j]
    grad = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        j_range = np.arange(lo, hi + 1)
        grad[j_range] += 2.0 * w[k] * f[k - j_range]
    
    return grad, M

def adam_step(f, grad, m, v, t, lr=0.01, b1=0.9, b2=0.999, eps=1e-8):
    """Adam update with simplex projection."""
    m = b1 * m + (1 - b1) * grad
    v = b2 * v + (1 - b2) * grad * grad
    m_hat = m / (1 - b1**t)
    v_hat = v / (1 - b2**t)
    f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
    f_new = project_simplex(f_new)
    return f_new, m, v

def lp_refine(f, time_limit, n_constraints_init=200, n_constraints_max=800,
              label="lp", max_stuck=30):
    """
    LP-based refinement using scipy HiGHS solver.
    """
    t0 = time.monotonic()
    n = len(f)
    f = f.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    
    it = 0
    stuck = 0
    top_k = n_constraints_init
    full_len = 2 * n - 1
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Gradually increase constraints
        if it % 15 == 0 and top_k < n_constraints_max:
            top_k = min(top_k + 20, n_constraints_max)
        
        b = fast_convolve(f)
        M = b.max()
        
        # Solve LP: min M' s.t. sum(g)=1, g>=0, (f*g)[k] <= M' for selected k
        if top_k < full_len:
            idx = np.argpartition(b, -top_k)[-top_k:]
            idx = np.sort(idx)
        else:
            idx = np.arange(full_len)
        
        K = len(idx)
        A = np.zeros((K, n))
        for i, k in enumerate(idx):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A[i, lo:hi+1] = f[lo:hi+1]
        
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.hstack([A, -np.ones((K, 1))])
        b_ub = np.zeros(K)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'maxiter': 10000})
        except Exception:
            stuck += 2
            if stuck > max_stuck:
                break
            continue
        
        if res.status != 0 or res.x is None:
            stuck += 1
            if stuck > max_stuck:
                break
            continue
        
        g = np.maximum(res.x[:n], 0.0)
        S = g.sum()
        if S < 1e-15:
            stuck += 1
            continue
        g = g / S
        
        # Line search: try various t
        best_t_obj = M
        best_t = 0.0
        t_candidates = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9]
        
        for t in t_candidates:
            f_trial = (1 - t) * f + t * g
            f_trial = np.maximum(f_trial, 0.0)
            s_trial = f_trial.sum()
            if s_trial < 1e-15:
                continue
            f_trial = f_trial / s_trial
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t
        
        if best_t == 0.0:
            stuck += 1
            if stuck > max_stuck:
                break
            # Try soft-grad direction as fallback
            grad, _ = softmax_grad(f, 0.05, n_top=100)
            f_trial = project_simplex(f - 0.001 * grad)
            obj = fast_convolve(f_trial).max()
            if obj < M - 1e-12:
                f = f_trial
                if obj < best_obj:
                    best_obj = obj
                    best_f = f.copy()
                    best_score = 2.0 * n * obj
                stuck = 0
            continue
        
        f_new = (1 - best_t) * f + best_t * g
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()
        obj_new = fast_convolve(f_new).max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            best_score = 2.0 * n * obj_new
            f = f_new
            stuck = 0
        else:
            f = f_new
            stuck += 1
            if stuck > max_stuck:
                break
        
        # Periodic perturbation
        if stuck > 15:
            rng = np.random.default_rng(hash((it, n)) % 2**31)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            stuck = 0
        
        if it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} obj={best_obj:.12f} score={best_score:.6f} "
                  f"k={top_k} t={elapsed:.0f}s")
    
    return best_f, best_score

def sg_optimize(f_init, time_limit, lr_init=0.005, temp_init=0.05, temp_min=0.002,
                n_top=200, label="sg"):
    """Soft-max gradient optimization with Adam and temperature annealing."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s
    
    m = np.zeros(n)
    v = np.zeros(n)
    temp = temp_init
    lr = lr_init
    it = 0
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    rng = np.random.default_rng(123)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        grad, M = softmax_grad(f, temp, n_top)
        f, m, v = adam_step(f, grad, m, v, it, lr)
        
        if it % 30 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.03, 0.05)
            else:
                no_improve += 1
                if no_improve > 5:
                    lr *= 0.9
                    no_improve = 0
            
            # Anneal temperature
            temp = max(temp * 0.97, temp_min)
            
            if it % 200 == 0:
                print(f"    {label} it={it} obj={best_obj:.12f} score={2.0*n*best_obj:.6f} "
                      f"temp={temp:.4f} lr={lr:.5f} t={time.monotonic()-t0:.0f}s")
            
            if no_improve > 30:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                m = np.zeros(n)
                v = np.zeros(n)
                no_improve = 0
                # Reset temperature partially
                temp = max(temp * 2, temp_init * 0.5)
    
    return best_f, 2.0 * n * best_obj

def coordinate_descent(f_init, time_limit):
    """Coordinate descent with multiplicative updates."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    best_f = f.copy()
    
    multipliers = [0.5, 0.7, 0.85, 0.9, 1.1, 1.15, 1.3, 1.5, 2.0]
    no_pass = 0
    
    while time.monotonic() - t0 < time_limit:
        any_improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue
            
            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                f_trial = f.copy()
                f_trial[j] = f[j] * mult
                s = f_trial.sum()
                if s < 1e-12:
                    continue
                f_trial = f_trial / s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = f[j] * mult
            
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-12:
                    f = f / s
                if best_obj > best_j_obj:
                    best_obj = best_j_obj
                    best_score = 2.0 * n * best_obj
                    best_f = f.copy()
                    any_improved = True
        
        if not any_improved:
            no_pass += 1
            if no_pass >= 3:
                break
        else:
            no_pass = 0
    
    return best_f, best_score
```

Wait, I realize there might be an issue with the LP. The constraint matrix A can be very large (K x n where K can be up to 2n-1 and n can be up to 3000, so up to 6000 x 3000 = 18 million entries). This might be slow or memory-intensive.

Let me add a cap on the number of constraints to keep things manageable:
- For n=1000: max K = 500 (out of 1999 possible)
- For n=2000: max K = 800 (out of 3999)

Actually, the LP itself should be fine with HiGHS since it's sparse. The constraint matrix A is very sparse (each row has at most n non-zeros, and for the convolution structure, each row has a specific pattern).

But building the full dense matrix might use too much memory for large n. Let me use a CSR sparse matrix instead.

Actually, for the sizes we're dealing with (n up to ~2000, K up to ~800), the dense matrix is at most 800*2000 = 1.6M entries, which is fine.

Let me also think about whether there's a smarter way to select constraints. The most important constraints are those where b[k] is closest to M (the max). For the LP, we want to constrain exactly those points that would otherwise exceed the target.

A better constraint selection:
1. Start with the top K=50 peaks
2. Solve LP
3. Check which constraints are violated in the solution
4. Add violated constraints to the set
5. Repeat

This is the "cutting plane" or "separation oracle" approach, which converges faster.

Let me implement this:

```python
def lp_cutting_plane(f, time_limit, label="cp"):
    """LP with cutting plane: iteratively add violated constraints."""
    t0 = time.monotonic()
    n = len(f)
    f = f.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    full_len = 2 * n - 1
    
    # Start with top 50 peaks
    b = fast_convolve(f)
    constraint_set = set(np.argpartition(b, -50)[-50:].tolist())
    
    it = 0
    stuck = 0
    max_iterations = 200
    
    while time.monotonic() - t0 < time_limit and it < max_iterations:
        it += 1
        stuck += 1
        constraint_set = sorted(constraint_set)
        K = len(constraint_set)
        
        # Build LP
        A = np.zeros((K, n))
        for i, k in enumerate(constraint_set):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A[i, lo:hi+1] = f[lo:hi+1]
        
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.hstack([A, -np.ones((K, 1))])
        b_ub = np.zeros(K)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs', options={'maxiter': 5000})
        except Exception:
            break
        
        if res.status != 0:
            if stuck > 20:
                break
            continue
        
        g = np.maximum(res.x[:n], 0.0)
        S = g.sum()
        if S < 1e-15:
            break
        g = g / S
        M_opt = res.x[n]
        
        # Check if new constraints are violated
        fg = fast_convolve(f * g)  # This is the convolution f*g
        # Actually, (f*g)[k] = sum_j f[j]*g[k-j], which is a cross-correlation
        # Not the same as auto-convolution. Let me compute it directly.
        # (f*g)[k] = sum_j f[j] * g[k-j] for valid j, k-j
        # This is a standard cross-correlation/convolution
        
        # Actually, the LP constraint is A@g <= M', where A[i,j] = f[k_i - j]
        # So (f * g)[k] = sum_j f[k-j] * g[j] = sum_j A[corresponding_row, j] * g[j]
        # The LP ensures this is <= M' for k in constraint_set
        # We need to check ALL k, not just the constraint set
        
        # Compute full cross-convolution f*g
        fg = np.convolve(f, g)  # This gives (f*g)[k] = sum_j f[j]*g[k-j]
        # Wait, numpy's convolve: (a*b)[n] = sum_m a[m]*b[n-m]
        # So (f*g)[k] = sum_j f[j]*g[k-j] ✓
        
        max_fg = fg.max()
        
        if max_fg > M_opt + 1e-8:
            # Find violated constraints
            violated = np.where(fg > M_opt + 1e-8)[0]
            # Add top violations to constraint set
            if len(violated) > 100:
                top_violated = violated[np.argsort(fg[violated])[-100:]]
            else:
                top_violated = violated
            constraint_set.update(top_violated.tolist())
            # Cap the size
            if len(constraint_set) > 1500:
                # Keep only the most violated
                fg_vals = fg[list(constraint_set)]
                keep = np.argsort(fg_vals)[-1500:]
                constraint_set = set([constraint_set[i] for i in keep])
            continue  # Re-solve with new constraints
        
        # No violations: we have the true LP solution
        # Do line search
        best_t_obj = fast_convolve(f).max()
        best_t = 0.0
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7]:
            f_trial = (1 - t) * f + t * g
            f_trial = np.maximum(f_trial, 0.0)
            st = f_trial.sum()
            if st < 1e-15:
                continue
            f_trial = f_trial / st
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t
        
        if best_t > 0:
            f_new = (1 - best_t) * f + best_t * g
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            obj_new = fast_convolve(f_new).max()
            
            if obj_new < best_obj - 1e-14:
```

```python
"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

Key improvements over previous approach:
1. Use scipy.optimize.linprog with HiGHS (10-50x faster than cvxpy)
2. Soft-max gradient descent with Adam for fast initial progress
3. Adaptive constraint selection with cutting-plane refinement
4. Focused multi-phase strategy: SG init -> LP main -> multi-length -> polish
5. Better restart and perturbation strategies
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_convolve(f, g):
    """Cross-convolution (f*g)[k] = sum_j f[j]*g[k-j] via FFT."""
    n = len(f)
    m = len(g)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interp_sequence(src, target_n):
    """Interpolate a sequence to a new length."""
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result


def softmax_subgradient(f, temp, n_top=300):
    """
    Compute weighted subgradient of max(auto-conv) using soft-max.
    Returns (gradient, current_max).
    """
    b = fast_convolve(f)
    M = b.max()
    n = len(f)
    full_len = 2 * n - 1

    scaled = (b - M) / temp
    scaled = np.clip(scaled, -50.0, 0.0)
    w = np.exp(scaled)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(full_len)
        w[np.argmax(b)] = 1.0
    else:
        w = w / wsum

    if n_top < full_len:
        top_idx = np.argpartition(b, -n_top)[-n_top:]
        w_sel = w[top_idx].copy()
        ws = w_sel.sum()
        if ws > 1e-30:
            w_sel = w_sel / ws
        w_new = np.zeros(full_len)
        w_new[top_idx] = w_sel
        w = w_new

    # grad[j] = 2 * sum_{k: 0<=k-j<n} w[k] * f[k-j]
    grad = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        j_range = np.arange(lo, hi + 1)
        grad[j_range] += 2.0 * w[k] * f[k - j_range]

    return grad, M


def adam_update(f, grad, m, v, t, lr=0.01, b1=0.9, b2=0.999, eps=1e-8):
    """Adam optimizer step with simplex projection."""
    m = b1 * m + (1.0 - b1) * grad
    v = b2 * v + (1.0 - b2) * grad * grad
    m_hat = m / (1.0 - b1 ** t)
    v_hat = v / (1.0 - b2 ** t)
    f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
    f_new = project_simplex(f_new)
    return f_new, m, v


def sg_optimize(f_init, time_limit, lr_init=0.01, temp_init=0.05,
                temp_min=0.002, n_top=200, label="sg"):
    """Soft-max gradient optimization with Adam and temperature annealing."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s
    else:

    m = np.zeros(n)
    v = np.zeros(n)
    temp = temp_init
    lr = lr_init
    it = 0
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        it += 1
        grad, M = softmax_subgradient(f, temp, n_top)
        f, m, v = adam_update(f, grad, m, v, it, lr)

        if it % 20 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.04, 0.08)
            else:

                    no_improve = 0

def build_conv_matrix(f, indices, n):
    """Build constraint matrix for convolution constraints."""
    K = len(indices)
    A = np.zeros((K, n))
    for i in range(K):
        k = indices[i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi >= lo:
            A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def lp_direction(f, indices, M_target):
    """
    Solve: max sum(g) s.t. g >= 0, (f*g)[k] <= M_target for k in indices.
    Returns (g_normalized, sum_g) or (None, 0).
    """
    n = len(f)
    K = len(indices)
    A = build_conv_matrix(f, indices, n)

    # max sum(g) s.t. A@g <= M_target, g >= 0
    c = -np.ones(n)
    A_ub = A
    b_ub = np.full(K, M_target)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'maxiter': 10000})
    except Exception:
        return None, 0.0

    if res.status != 0 or res.x is None:
        return None, 0.0

    g = np.maximum(res.x, 0.0)
    S = g.sum()
    if S < 1e-15:
        return None, 0.0
    return g / S, S


def lp_minimize(f, indices, time_budget=5.0):
    """
    Solve: min M s.t. sum(g)=1, g>=0, (f*g)[k] <= M for k in indices.
    Returns (g, M_opt) or (None, None).
    """
    n = len(f)
    K = len(indices)
    A = build_conv_matrix(f, indices, n)

    # Variables: [g_0,...,g_{n-1}, M], total n+1
    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.hstack([A, -np.ones((K, 1))])
    b_ub = np.zeros(K)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'maxiter': 10000})
    except Exception:
        return None, None

    if res.status != 0 or res.x is None:
        return None, None

    g = np.maximum(res.x[:n], 0.0)
    S = g.sum()
    if S < 1e-15:
        return None, None
    g = g / S
    M_opt = res.x[n]
    return g, M_opt

def coordinate_descent(f_init, time_limit):
    """Coordinate descent with multiplicative updates."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    best_f = f.copy()

    multipliers = [0.5, 0.7, 0.85, 0.9, 1.1, 1.15, 1.3, 1.5, 2.0]
    no_pass = 0

    while time.monotonic() - t0 < time_limit:
        any_improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue

            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                f_trial = f.copy()
                f_trial[j] = f[j] * mult
                s = f_trial.sum()
                if s < 1e-12:
                    continue
                f_trial = f_trial / s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = f[j] * mult

            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-12:
                    f = f / s
                if best_obj > best_j_obj:
                    best_obj = best_j_obj
                    best_score = 2.0 * n * best_obj
                    best_f = f.copy()
                    any_improved = True

        if not any_improved:
            no_pass += 1
            if no_pass >= 3:
                break
        else:
            no_pass = 0

    return best_f, best_score

        print(f"\nPhase 4: Deep LP n={n_b}, budget={t4:.0f}s")
        f_res, _ = lp_main_optimize(best_seq, time_limit=t4,
                                    k_init=min(200, 2 * n_b - 1),
                                    k_max=min(800, 2 * n_b - 1),
                                    label="deep", max_stuck=60)
        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = f_res.copy()
            print(f"  NEW BEST [deep]: {real_score:.6f}")

    # ============================================================
    # Phase 5: SG polish
    # ============================================================
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.12, 100.0)
    if t5 > 15:
        n_b = len(best_seq)
        print(f"\nPhase 5: SG polish n={n_b}, budget={t5:.0f}s")
        f_res, _ = sg_optimize(best_seq, time_limit=t5,
                               lr_init=0.003, temp_init=0.02,
                               temp_min=0.001,
                               n_top=min(200, n_b), label="sg_pol")
        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = f_res.copy()
            print(f"  NEW BEST [sg polish]: {real_score:.6f}")

    # ============================================================
    # Phase 6: Coordinate descent
    # ============================================================
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.08, 60.0)
    if t6 > 5:
        print(f"\nPhase 6: CD n={len(best_seq)}, budget={t6:.0f}s")
        f_res, _ = coordinate_descent(best_seq, time_limit=t6)
        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = f_res.copy()
            print(f"  NEW BEST [cd]: {real_score:.6f}")

    # ============================================================
    # Phase 7: Final LP burst with max constraints
    # ============================================================
    rem = deadline - time.monotonic()
    t7 = rem
    if t7 > 20:
        n_b = len(best_seq)
        print(f"\nPhase 7: Final LP n={n_b}, budget={t7:.0f}s")
        f_res, _ = lp_main_optimize(best_seq, time_limit=t7,
                                    k_init=min(300, 2 * n_b - 1),
                                    k_max=min(1000, 2 * n_b - 1),
                                    label="final", max_stuck=80,
                                    use_minimize=True)
        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = f_res.copy()
            print(f"  NEW BEST [final]: {real_score:.6f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        if parent is not None:
            best_seq = parent.copy()
        else:
            best_seq = np.ones(512) / 512.0
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'=' * 60}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'=' * 60}")

    return [float(x) for x in best_seq]
```
