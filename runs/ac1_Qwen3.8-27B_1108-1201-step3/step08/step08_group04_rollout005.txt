Spectral-subgradient optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based cross-correlation for fast gradient computation, combined with 
multiplicative weights LP approximation and periodic spectral flattening.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def cross_corr_fft(w, a, N=None):
    """Compute linear cross-correlation g[j] = sum_k w[k]*a[k-j] for j=0..len(a)-1."""
    n_a = len(a)
    n_w = len(w)
    if N is None:
        N = 1 << (n_a + n_w - 2).bit_length()
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    g_full = np.fft.irfft(np.conj(W) * A, N)[:n_a]
    return g_full

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def eval_obj(f):
    """Compute the objective: 2*n*max(conv(f,f))/(sum(f))^2."""
    n = len(f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fast_convolve(f)
    return 2.0 * n * b.max() / (s * s)

def spectral_flatten(f, alpha=0.5, n_iter=1):
    """
    Spectral flattening: modify the magnitude of FFT to be flatter.
    Keeps the DC component and phases, modifies non-DC magnitudes.
    """
    n = len(f)
    N = 1 << (2*n - 2).bit_length()
    F = np.fft.fft(f, N)
    
    for _ in range(n_iter):
        # Current magnitudes
        mag = np.abs(F)
        # Target: raise non-DC magnitudes to be more uniform
        # Keep DC (index 0 and N/2 for real signals)
        # For non-DC: replace magnitude with a geometric mean of current and some target
        
        # Simple approach: mix current magnitude with its mean (excluding DC)
        non_dc_mag = mag[1:n]  # for real signal, we only need first n/2+1 bins
        # Actually for real signal of length N, FFT has N bins but first half is enough
        # Let's work with the full complex FFT
        
        # Compute target magnitude: geometric mean between current and mean
        non_dc = mag.copy()
        non_dc[0] = 0  # mask DC
        if N > 1:
            non_dc[N // 2] = 0  # Nyquist
        mean_non_dc = non_dc.sum() / max(1, (N - 2))
        
        # New magnitude: interpolate between current and mean
        new_mag = np.power(mag, alpha) * np.power(mean_non_dc + 1e-20, 1 - alpha)
        new_mag[0] = mag[0]  # preserve DC
        if N > 1:
            new_mag[N // 2] = mag[N // 2]
        
        # Reconstruct with new magnitudes and old phases
        phase = np.exp(1j * np.angle(F))
        F_new = new_mag * phase
        
        # Inverse FFT
        f_new = np.fft.ifft(F_new).real[:n]
    
    # Project to simplex
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-15:
        f_new = f_new / s
    else:
        f_new = f.copy()
    
    return f_new

def interp_sequence(src, target_n):
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result
