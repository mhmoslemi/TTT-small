"""
LP-based optimization with PDLP solver for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements: faster LP solves via PDLP, adaptive peak management,
multi-length exploration, subgradient polishing, and symmetry exploitation.

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    v = v * target_sum
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp(f, n, peaks, M):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)[k]<=M for k in peaks."""
    K = len(peaks)
    if K == 0:
        return None
    A = np.zeros((K, n))
    for row_i in range(K):
        k = int(peaks[row_i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    for solver in [cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, max_iters=500000)
            if g.value is not None and np.all(np.isfinite(g.value)):
                return np.maximum(g.value, 0.0)

    best_b = fast_convolve(f).max()

    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        # FFT
        fft_size = 1 << (2 * n - 1).bit_length()
        A = rfft(f, fft_size)
        P = np.abs(A) ** 2

        # Compress dynamic range: P_new = P^alpha (alpha < 1 flattens)
        P_max = P.max()
        P_min = P[P > 1e-30].min() if np.any(P > 1e-30) else 1e-30
        P_norm = P / P_max
        P_compressed = P_max * np.maximum(P_norm, 1e-10) ** alpha

        # Reconstruct with compressed magnitude, keep phase
        A_new = A * np.sqrt(P_compressed / np.maximum(P, 1e-30))
        f_new = irfft(A_new, fft_size)[:n]

        # Project to non-negative simplex
        f_new = np.maximum(f_new, 0.0)
        f_new = project_simplex(f_new)

        f = f_new
        it += 1

        if it % 100 == 0:
            obj = fast_convolve(f).max()
            if obj < best_b - 1e-14:
                best_b = obj
                best_f = f.copy()
            score = 2.0 * n * best_b
            if it % 500 == 0:
                print(f"  {label} it={it} score={score:.6f} wall={time.monotonic()-t0:.0f}s")

    return best_f, best_b


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_x = np.linspace(0.0, 1.0, len(f))
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new
