    for j in range(n):

    return T


def symmetrize(a):
    """Make sequence symmetric: a[i] = a[n-1-i]. Returns normalized."""
    n = len(a)
    a_sym = np.zeros(n)
    half = (n + 1) // 2
    for i in range(half):
        a_sym[i] = (a[i] + a[n - 1 - i]) / 2.0
        a_sym[n - 1 - i] = a_sym[i]
    a_sym = np.maximum(a_sym, 0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym = a_sym / s
    return a_sym


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_sym(a0, time_limit, solver, n_max_iters=80000):
    """Symmetry-enforced Nesterov dual averaging LP refinement."""
    n = len(a0)
    a = symmetrize(np.maximum(np.asarray(a0, dtype=np.float64), 0))
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_score = best_score
    stall_count = 0
    rng = np.random.default_rng(12345)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        b_new_sym = symmetrize(b_new)

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new_sym
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new_sym) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if it % 50 == 0:
            if best_score < prev_score - 1e-8:
                stall_count = 0
            else:
                stall_count += 1
            prev_score = best_score
            if stall_count >= 8:
                # Structured perturbation to escape local min
                half = (n + 1) // 2
                noise_scale = 0.01 + 0.02 * (stall_count % 3)
                noise = rng.random(half) * noise_scale
                y_half = y[:half].copy() + noise
                y_new = np.zeros(n)
                for i in range(half):
                    y_new[i] = y_half[i]
                    y_new[n - 1 - i] = y_half[i]
                y = np.maximum(y_new, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                stall_count = 0

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovLP-Sym iter={it+1} ({elapsed:.0f}s): score={best_score:.7f}")

    return best_a, best_score


def compute_conv_grad(a, b_weights):
    """Compute gradient of weighted sum of conv(a,a) with respect to a."""
    n = len(a)
    grad = np.zeros(n)
    for k in range(len(b_weights)):
        wk = b_weights[k]
        if wk < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * wk * a[k - idx]
    return grad


def lbgfs_full(a0, time_limit, T_start=150.0, T_end=0.3, n_stages=8, maxiter_per_stage=2000):
    """L-BFGS with LogSumExp smoothing on full vector, symmetrized output."""
    n = len(a0)
    a = symmetrize(np.maximum(np.asarray(a0, dtype=np.float64), 0))
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s

    # Work in log-space
    z = np.log(np.maximum(a, 1e-300)).copy()
    t0 = time.monotonic()

    def softmax_z(zv):
        zs = zv - zv.max()
        ev = np.exp(zs)
        sv = ev.sum()
        if sv < 1e-300:
            return np.ones(len(zv)) / len(zv)
        return ev / sv

    temps = np.geomspace(T_start, T_end, n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        T_val = stage_T

        def obj(zv):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            av = softmax_z(zv)
            bv = fast_convolve(av)
            Tb = T_val * bv
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp

        def obj_grad(zv):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(zv)
            av = softmax_z(zv)
            bv = fast_convolve(av)
            Tb = T_val * bv
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = compute_conv_grad(av, w)
            S = np.dot(grad_a, av)
            return av * (grad_a - S)

        try:
            res = minimize(
                fun=obj,
                x0=z,
                jac=obj_grad,
                method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-16, 'gtol': 1e-10,
                         'maxfun': maxiter_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        a_cur = softmax_z(z)
        a_cur = symmetrize(a_cur)
        sc = 2.0 * n * fast_convolve(a_cur).max()
        if (stage_T == temps[0]):
            print(f"    L-BFGS stage 1 done: score={sc:.7f}")

    a_final = softmax_z(z)
    return symmetrize(a_final)


def coordinate_descent_sym(a0, time_limit, n_max_steps=50000):
    """Symmetric coordinate descent: adjust pairs (a[i], a[n-1-i])."""
    n = len(a0)
    a = symmetrize(np.maximum(np.asarray(a0, dtype=np.float64), 0))
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    best_score = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    half = (n + 1) // 2
    rng = np.random.default_rng(999)

    for step in range(n_max_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        # Focus on top convolution peaks
        b_work = b.copy()
        if len(b_work) > 2:
            b_work[0] = -np.inf
            b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))
        peak_val = b_work[peak_k]
        threshold = peak_val - 0.02 * peak_val
        active = np.where(b_work >= threshold)[0]

        # Find top contributing coordinates
        contrib = np.zeros(n)
        for k in active[:8]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib[idx] += a[idx] * a[k - idx]

        # Symmetrize contribution
        contrib_sym = np.zeros(n)
        for i in range(half):
            contrib_sym[i] = contrib[i] + contrib[n - 1 - i]
            contrib_sym[n - 1 - i] = contrib_sym[i]

        top_coords = np.argsort(contrib_sym)[::-1][:min(20, half)]

        improved = False
        for i in top_coords:
            if i >= n:
                continue
            if a[i] < 1e-14:
                continue
            # Try reducing a[i] and a[n-1-i] slightly
            for factor in [0.97, 0.94, 0.90, 0.85]:
                a_trial = a.copy()
                diff = a_trial[i] * (1 - factor)
                a_trial[i] *= factor
                a_trial[n - 1 - i] *= factor
                if diff < 1e-16:
                    continue
                # Redistribute to other coords
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                mask[n - 1 - i] = False
                others = a_trial[mask]
                if others.sum() < 1e-15:
                    continue
                a_trial[mask] = others + 2 * diff * others / others.sum()
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            # Try increasing a low-contribution coordinate
            low_coords = np.argsort(contrib_sym)[:min(5, half)]
            found = False
            for i in low_coords:
                if i >= n:
                    continue
                if a[i] > 0.5 / n:
                    continue
                for factor in [1.1, 1.2, 1.5]:
                    a_trial = a.copy()
                    a_trial[i] *= factor
                    a_trial[n - 1 - i] *= factor
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        found = True
                        break
                if found:
                    break

        if not improved:
            # Random perturbation of a random symmetric pair
            i = rng.integers(0, half)
            factor = 0.95 + 0.1 * rng.random()
            a_trial = a.copy()
            a_trial[i] *= factor
            a_trial[n - 1 - i] *= factor
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-14:
                a = a_trial
                best_score = score
                best_a = a.copy()

    return best_a, best_score


def structured_init(n, rng):
    """Create structured initial sequence with good properties."""
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = (1 + np.cos(2 * np.pi * x)) ** 1.2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif choice == 4:
        x = np.linspace(-1, 1, n)
        a = (1 - x ** 2) ** 0.75
    else:
        # Blackman-like
        x = np.linspace(0, 1, n)
        a = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0)

    a = np.maximum(a, 0)
    a = symmetrize(a)
    return a


def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0)
    s2 = a_interp.sum()
    if s2 > 1e-15:
        a_interp = a_interp / s2
    return a_interp


def evaluate_safe(a):
    """Evaluate sequence with the official evaluator."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase symmetry-enforced optimization:
    1. LP-based Nesterov alternating (main workhorse)
    2. L-BFGS LogSumExp refinement
    3. Second LP round
    4. Coordinate descent polish
    5. Length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_sym = symmetrize(parent)
            p_score = evaluate_safe(p_sym)
            print(f"Parent: n={len(parent)}, score={p_score:.7f}, sym_score={evaluate_safe(p_sym):.7f}")
            best_seq = p_sym
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.7f}")

    n_best = len(best_seq)
    print(f"Starting: n={n_best}, score={best_score:.7f}")

    # === Phase 1: Aggressive Nesterov LP ===
    print(f"\n=== Phase 1: Nesterov LP (n={n_best}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.45, 700.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, sc_lp = nesterov_lp_sym(best_seq, t_alloc, solver=solver, n_max_iters=100000)
        sc_check = evaluate_safe(a_lp)
        if sc_check < best_score:
            best_score = sc_check
            best_seq = a_lp.copy()
            print(f"  NEW BEST [LP1]: {best_score:.7f}")
        else:
            print(f"  LP1 done: {sc_check:.7f} (best remains {best_score:.7f})")

    # === Phase 2: L-BFGS refinement ===
    print(f"\n=== Phase 2: L-BFGS (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 350.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s, T: 150->0.3")
        a_lb = lbgfs_full(best_seq, t_alloc, T_start=150.0, T_end=0.3, n_stages=8, maxiter_per_stage=1500)
        sc_lb = evaluate_safe(a_lb)
        if sc_lb < best_score:
            best_score = sc_lb
            best_seq = a_lb.copy()
            print(f"  NEW BEST [LBFGS]: {best_score:.7f}")
        else:
            print(f"  LBFGS done: {sc_lb:.7f} (best remains {best_score:.7f})")

    # === Phase 3: Second LP round ===
    print(f"\n=== Phase 3: Nesterov LP Round 2 (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.18, 250.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, sc_lp2 = nesterov_lp_sym(best_seq, t_alloc, solver=solver, n_max_iters=50000)
        sc_check = evaluate_safe(a_lp2)
        if sc_check < best_score:
            best_score = sc_check
            best_seq = a_lp2.copy()
            print(f"  NEW BEST [LP2]: {best_score:.7f}")
        else:
            print(f"  LP2 done: {sc_check:.7f}")

    # === Phase 4: Coordinate descent polish ===
    print(f"\n=== Phase 4: Coord Descent (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 10:
        a_cd, sc_cd = coordinate_descent_sym(best_seq, t_alloc, n_max_steps=50000)
        sc_check = evaluate_safe(a_cd)
        if sc_check < best_score:
            best_score = sc_check
            best_seq = a_cd.copy()
            print(f"  NEW BEST [CD]: {best_score:.7f}")
        else:
            print(f"  CD done: {sc_check:.7f}")

    # === Phase 5: Try slightly different lengths ===
    print(f"\n=== Phase 5: Length Exploration (best={best_score:.7f}) ===")
    alt_lengths = []
    for mult in [1.25, 1.5, 0.8, 2.0]:
        alt_n = int(n_best * mult)
        if 256 <= alt_n <= 4096 and alt_n != n_best:
            alt_lengths.append(alt_n)
    alt_lengths = sorted(set(alt_lengths))

    for alt_n in alt_lengths[:2]:
        rem = deadline - time.monotonic()
        if rem < 100:
            break
        t_total = min(rem * 0.35, 250.0)
        if t_total < 50:
            break
        print(f"  Trying n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_to_length(best_seq, alt_n)
        init_alt = symmetrize(init_alt)

        t_lp = t_total * 0.7
        a_alt, _ = nesterov_lp_sym(init_alt, t_lp, solver=solver, n_max_iters=30000)
        sc_alt = evaluate_safe(a_alt)
        if sc_alt < best_score:
            best_score = sc_alt
            best_seq = a_alt.copy()
            print(f"  NEW BEST [n={alt_n}]: {best_score:.7f}")
        else:
            print(f"  n={alt_n}: {sc_alt:.7f} (no improvement)")

        rem = deadline - time.monotonic()
        if rem > 30 and len(best_seq) == alt_n:
            t_lb = min(rem * 0.3, 80.0)
            if t_lb > 15:
                a_alt2 = lbgfs_full(best_seq, t_lb, T_start=50.0, T_end=1.0, n_stages=5, maxiter_per_stage=1000)
                sc_alt2 = evaluate_safe(a_alt2)
                if sc_alt2 < best_score:
                    best_score = sc_alt2
                    best_seq = a_alt2.copy()
                    print(f"  NEW BEST [LBFGS n={alt_n}]: {best_score:.7f}")

    # === Phase 6: Final LP + polish ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.5, 100.0)
        a_fin, _ = nesterov_lp_sym(best_seq, t_fin, solver=solver, n_max_iters=20000)
        sc_fin = evaluate_safe(a_fin)
        if sc_fin < best_score:
            best_score = sc_fin
            best_seq = a_fin.copy()
            print(f"  NEW BEST [Final LP]: {best_score:.7f}")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_cd = min(rem * 0.5, 50.0)
        a_cd2, _ = coordinate_descent_sym(best_seq, t_cd, n_max_steps=20000)
        sc_cd2 = evaluate_safe(a_cd2)
        if sc_cd2 < best_score:
            best_score = sc_cd2
            best_seq = a_cd2.copy()
            print(f"  NEW BEST [Final CD]: {best_score:.7f}")

    # Final verification
    if best_seq is not None:
        final_score = evaluate_safe(best_seq)
        if np.isfinite(final_score):
            best_score = final_score

    print(f"\nFINAL: score={best_score:.7f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
