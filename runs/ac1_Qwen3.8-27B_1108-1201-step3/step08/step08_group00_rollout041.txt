    # grad_a[k] = 2 * sum_{l: 0<=l<n, 0<=l+k<2n-1} w[l+k] * a[l]
    # = 2 * sum_l a[l] * w[l+k] (with w out of bounds = 0)
    
    # This is like a cross-correlation of a and w.
    
    grad_a = np.zeros(n)
    for k in range(n):
        # grad_a[k] = 2 * sum_{l=0}^{n-1} a[l] * w[l+k] (where w is zero outside [0, 2n-2])
        # l+k ranges from k to k+n-1, which is within [0, 2n-2] for k in [0, n-1]
        grad_a[k] = 2.0 * np.dot(a, w[k:k+n])
    
    # Convert gradient from a-space to z-space (log-domain)
    # a = softmax(z), so da_i/dz_j = a_i*(delta_ij - a_j)
    # grad_z_j = sum_i grad_a_i * da_i/dz_j = sum_i grad_a_i * a_i * (delta_ij - a_j)
    # = a_j * (grad_a_j - sum_i a_i * grad_a_i)
    S = np.dot(a, grad_a)
    grad_z = a * (grad_a - S)
    
    return (2.0 * n / T) * grad_z


def adam_logsumexp_optimize(a0, time_limit, T_start=50.0, T_end=1.0, n_stages=8, 
                            lr=0.01, beta1=0.9, beta2=0.999):
    """Adam optimizer with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0.copy())
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    t0 = time.monotonic()
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(100, 100000 // n_stages)
    
    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        for it in range(iters_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Compute gradient
            a = softmax_from_z(z)
            grad_z = compute_grad_logsumexp(z, stage_T)
            
            # Adam update
            t += 1
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            
            z = z - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            
            # Evaluate
            a_new = softmax_from_z(z)
            score = 2.0 * n * fast_convolve(a_new).max()
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
                
            if it % 5000 == 0 and it > 0:
                print(f"      Adam T={stage_T:.1f} iter={it}: score={best_score:.6f}")
    
    return best_a, best_score


def subgradient_optimize(a0, time_limit, lr_init=0.05, n_steps=50000):
    """Subgradient descent with adaptive step size on the simplex."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    avg_lr = lr_init
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Subgradient of max_k (a*a)[k]: pick the achieving k (or average over near-max)
        threshold = max_b - 0.05 * max_b
        active = np.where(b >= threshold)[0]
        
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)
        
        # Polyak step size or fixed with decay
        z = z_from_a(a)
        z = z - lr * g / a  # This is the gradient in z-space: grad_z = a * (grad_a - S), but for subgradient...
        
        # Actually, let's just do projected gradient on a:
        a_new = a - lr * g
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / sn
        
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score - 1e-15:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            # Try smaller step
            a_try = a - lr * 0.5 * g
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            score_try = 2.0 * n * fast_convolve(a_try).max()
            if score_try < score_new:
                a = a_try
                if score_try < best_score:
                    best_score = score_try
                    best_a = a_try.copy()
                    no_improve = 0
        
        if no_improve >= 50:
            lr *= 0.9
            no_improve = 0
        lr = max(lr_init / (1.0 + step * 0.001), 1e-10)
        
        if step % 5000 == 0:
            print(f"      SubGrad {time.monotonic()-t0:.0f}s: score={best_score:.6f} lr={lr:.6f}")
    
    return best_a, best_score

def diverse_init(n, seed=0):
    """Create diverse initial sequences."""
    rng = np.random.default_rng(seed)
    choice = rng.integers(0, 8)
    
    x = np.linspace(0, 1, n)
    
    if choice == 0:
        # Uniform
        a = np.ones(n)
    elif choice == 1:
        # Cosine bell
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        # Two bumps
        a = np.exp(-20 * (x - 0.25) ** 2) + np.exp(-20 * (x - 0.75) ** 2)
    elif choice == 3:
        # Three bumps
        a = np.exp(-30 * (x - 0.2) ** 2) + np.exp(-30 * (x - 0.5) ** 2) + np.exp(-30 * (x - 0.8) ** 2)
    elif choice == 4:
        # Sinusoidal modulation
        a = 1.0 + 0.5 * np.sin(4 * np.pi * x)
    elif choice == 5:
        # Cosine power
        a = np.cos(np.pi * (x - 0.5)) ** 4
    elif choice == 6:
        # Inverted V
        a = 1.0 - np.abs(2 * x - 1)
    else:
        # Random smooth
        freq = rng.integers(2, 8)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * freq * x + rng.random() * 2 * np.pi)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

                return np.zeros_like(z_val)

import numpy as np

from scipy.fft import rfft, irfft
from scipy.optimize import minimize as sp_minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def compute_cross_corr_fft(w, a):
    """Compute C[m] = sum_l a[l]*w[m+l] for m=0,...,n-1 via FFT.
    a has length n, w has length 2n-1."""
    n = len(a)
    a_rev = a[::-1]
    size = len(w) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a_rev, fft_size)
    full_conv = irfft(W * A, fft_size)[:size]
    return full_conv[n - 1:2 * n - 1]


def compute_grad_lse(z, T):
    """Compute exact gradient of (2n/T)*logsumexp(T*conv(softmax(z),softmax(z))) wrt z.
    
    The gradient is: 4n * a * (C - W_b)

    w = softmax(T*b), b = conv(a,a).

    C = compute_cross_corr_fft(w, a)
    W_b = np.dot(w, b)
    grad_z = 4.0 * n * a * (C - W_b)

    return (2.0 * n / T) * log_sum_exp


def adam_lse_optimize(a0, time_limit, T_start=50.0, T_end=0.3, n_stages=10,
                      lr=0.005, beta1=0.9, beta2=0.999, eps=1e-8,
                      max_iters_per_stage=20000):
    """Adam optimizer with LogSumExp smoothing and geometric temperature annealing."""
    n = len(a0)
    z = z_from_a(a0.copy()).astype(np.float64)
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    t0 = time.monotonic()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        iters_this_stage = min(max_iters_per_stage, max(100, int(remaining * 30)))

        for it in range(iters_this_stage):
            if time.monotonic() - t0 > time_limit:
                break

            grad_z = compute_grad_lse(z, stage_T)

            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * grad_z ** 2
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            if (it + 1) % 2000 == 0 or it == iters_this_stage - 1:
                a_new = softmax_from_z(z)
                score = 2.0 * n * fast_convolve(a_new).max()
                if score < best_score:
                    best_score = score
                    best_a = a_new.copy()
                    if it % 10000 == 0:
                        print(f"      Adam T={stage_T:.2f} it={it+1} ({time.monotonic()-t0:.0f}s): {best_score:.6f}")

    a_final = softmax_from_z(z)
    score_final = 2.0 * n * fast_convolve(a_final).max()
    if score_final < best_score:
        best_score = score_final
        best_a = a_final.copy()

    return best_a, best_score


def lbgfs_lse_optimize(a0, time_limit, T_start=200.0, T_end=0.3, n_stages=12):
    """L-BFGS with LogSumExp smoothing using corrected gradient."""
    n = len(a0)
    z = z_from_a(a0.copy()).astype(np.float64)
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        max_it = max(30, min(500, int(remaining * 20)))

        def _func(z_val):
            return compute_eval_lse(z_val, stage_T)

        def _jac(z_val):
            return compute_grad_lse(z_val, stage_T)

        try:
            res = sp_minimize(
                fun=_func, x0=z, jac=_jac,
                method='L-BFGS-B',
                options={'maxiter': max_it, 'ftol': 1e-18, 'gtol': 1e-10, 'disp': False}
            )
            z = res.x
            a_new = softmax_from_z(z)
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        except Exception:
            pass

        if si % 4 == 0:
            print(f"      L-BFGS T={stage_T:.2f} ({time.monotonic()-t0:.0f}s): {best_score:.6f}")

    return best_a, best_score


def subgradient_optimize(a0, time_limit, lr_init=0.04, decay=0.0008, min_lr=1e-9):
    """Subgradient descent on the simplex with adaptive step size."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b - 0.03 * max_b
        active = np.where(b >= threshold)[0]

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        a_new = a - lr * g
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / sn

        score_new = 2.0 * n * fast_convolve(a_new).max()

        if score_new < best_score - 1e-15:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            a_try = a - lr * 0.5 * g
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            score_try = 2.0 * n * fast_convolve(a_try).max()
            if score_try < score_new and score_try < best_score - 1e-15:
                a = a_try
                best_score = score_try
                best_a = a_try.copy()
                no_improve = 0

        if no_improve >= 40:
            lr *= 0.85
            no_improve = 0
        lr = max(lr_init * np.exp(-decay * (no_improve + 100)), min_lr)

    return best_a, best_score


def iterative_peak_shave(a0, time_limit, n_rounds=500, rng_seed=42):
    """Iteratively reduce peak by moving mass from high-contributing to low-impact positions."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        top_i = idx[np.argmax(contribs)]
        partner = peak_k - top_i

        improved = False
        for frac in [0.005, 0.01, 0.02, 0.03, 0.05]:
            delta = a[top_i] * frac
            if delta < 1e-12:
                continue

            low_threshold = np.percentile(a, 30)
            candidates = np.where(a < low_threshold)[0]
            if len(candidates) == 0:
                candidates = np.where(a < np.median(a))[0]
            if len(candidates) == 0:
                continue

            sample_size = min(30, len(candidates))
            sample = candidates[rng.choice(len(candidates), size=sample_size, replace=False)]

            best_pos = -1
            best_sc = float('inf')
            for pos in sample:
                a_trial = a.copy()
                a_trial[top_i] -= delta
                a_trial[pos] += delta
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_sc:
                    best_sc = sc
                    best_pos = pos

            if best_pos >= 0 and best_sc < best_score - 1e-15:
                a[top_i] -= delta
                a[best_pos] += delta
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = 2.0 * n * fast_convolve(a).max()
                best_a = a.copy()
                improved = True
                break

        if not improved:
            top_3 = idx[np.argsort(contribs)[::-1][:3]]
            a_trial = a.copy()
            total_delta = 0.0
            for ti in top_3:
                d = a_trial[ti] * 0.03
                a_trial[ti] -= d
                total_delta += d
            low_positions = np.argsort(a_trial)[:15]
            for pos in low_positions:
                a_trial[pos] += total_delta / len(low_positions)
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 1e-15:
                a_trial = a_trial / st
            sc = 2.0 * n * fast_convolve(a_trial).max()
            if sc < best_score - 1e-15:
                a = a_trial
                best_score = sc
                best_a = a.copy()
            else:
                if rnd > 20:
                    # Try a more aggressive move
                    a_trial2 = a.copy()
                    for ti in top_3:
                        a_trial2[ti] *= 0.85
                    removed = a.sum() - a_trial2.sum()
                    low_pos2 = np.argsort(a_trial2)[:20]
                    for pos in low_pos2:
                        a_trial2[pos] += removed / len(low_pos2)
                    a_trial2 = np.maximum(a_trial2, 0)
                    a_trial2 = a_trial2 / a_trial2.sum()
                    sc2 = 2.0 * n * fast_convolve(a_trial2).max()
                    if sc2 < best_score - 1e-15:
                        a = a_trial2
                        best_score = sc2
                        best_a = a.copy()
                    else:
                        break

    return best_a, best_score


def diverse_init(n, seed=0):
    """Create diverse structured initial sequences."""
    rng = np.random.default_rng(seed)
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)

    if choice == 0:
        a = np.ones(n)
    elif choice == 1:
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        a = np.exp(-30 * (x - 0.25) ** 2) + np.exp(-30 * (x - 0.75) ** 2)
    elif choice == 3:
        a = np.exp(-40 * (x - 0.2) ** 2) + np.exp(-40 * (x - 0.5) ** 2) + np.exp(-40 * (x - 0.8) ** 2)
    elif choice == 4:
        a = 1.0 + 0.4 * np.sin(6 * np.pi * x)
    elif choice == 5:
        a = np.cos(np.pi * (x - 0.5)) ** 4
    elif choice == 6:
        a = 1.0 - 0.5 * np.abs(2 * x - 1)
    elif choice == 7:
        freq = rng.integers(3, 10)
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * freq * x + rng.random() * 6.28)
    elif choice == 8:
        a = np.exp(-5.0 * x * (1.0 - x))
    else:
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * x)
        a = a ** 2.0

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def try_update_candidate(a, label, state):
    """Try to update the best candidate in state dict."""
    if a is None:
        return
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < state['best_score']:
        state['best_score'] = sc
        state['best_seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
