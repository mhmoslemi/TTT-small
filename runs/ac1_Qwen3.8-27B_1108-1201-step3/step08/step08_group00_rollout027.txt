from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_normalized(a):
    """Compute score for normalized (sum=1) sequence."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()

def project_simplex(v):
    """Project onto the probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    """Log-sum-exp smoothed max of convolution."""
    b = fast_convolve(a)
    Tb = T * b
    m = Tb.max()
    return (m + np.log(np.exp(Tb - m).sum())) / T

    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    m = Tb.max()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Fast Nesterov-accelerated projected gradient on log-sum-exp smoothed objective
2. LP-based best response refinement with Nesterov dual averaging
3. Peak shaving coordinate descent
4. Multi-length and multi-start exploration with asymmetric initializations
Key innovations: efficient FFT-based gradient, asymmetric init, adaptive annealing.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score for normalized (sum=1) sequence."""
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def grad_smooth_max(a, T):
    """Gradient of (1/T)*log(sum_k exp(T*conv_k)) w.r.t. a."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    w = e / e.sum()
    # grad[i] = 2 * sum_j a[j] * w[j+i] = 2 * (a correlate w)[i]
    fft_size = 1 << (n + len(w) - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    corr = irfft(np.conj(W) * A, fft_size)[:n]
    return 2.0 * corr


def nesterov_projected_gradient(a0, time_limit, T_start=2.0, T_end=200.0,
                                 lr_init=0.05, n_stages=12, steps_per_stage=2000):
    """Nesterov accelerated projected gradient with temperature annealing."""
    n = len(a0)
    x = project_simplex(a0.copy())
    x_prev = x.copy()
    best_a = x.copy()
    best_score = score_normalized(x)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    total_planned = n_stages * steps_per_stage
    step_count = 0

    for stage in range(n_stages):
        T = temps[stage]
        for step in range(steps_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            global_step = stage * steps_per_stage + step
            lr = lr_init / (1.0 + 0.005 * global_step)

            # Nesterov momentum
            t_curr = 1.0 + np.sqrt(1.0 + 4.0 * step_count * step_count) / 2.0
            t_prev_calc = 1.0 + np.sqrt(1.0 + 4.0 * (step_count - 1) * (step_count - 1)) / 2.0 if step_count > 0 else 1.0
            beta = (t_prev_calc - 1.0) / t_curr if step_count > 0 else 0.0
            y = x + beta * (x - x_prev)
            y = project_simplex(y)

            # Gradient
            grad = grad_smooth_max(y, T)
            x_prev = x.copy()
            x = project_simplex(y - lr * grad)
            step_count += 1

            # Check score periodically
            if step_count % 50 == 0:
                sc = score_normalized(x)
                if sc < best_score:
                    best_score = sc
                    best_a = x.copy()
                    if step_count % 500 == 0:
                        elapsed = time.monotonic() - t0
                        print(f"      NG step={step_count} ({elapsed:.0f}s): score={best_score:.6f} T={T:.1f}")

    return best_a, best_score


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (sum=1, nonneg), find b minimizing max conv(a,b)."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=50000):
    """Nesterov dual averaging LP refinement."""
    n = len(a0)
    a = project_simplex(np.maximum(a0, 0))
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = project_simplex(y)
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = project_simplex(a)

        score_a = score_normalized(a)
        score_y = score_normalized(y)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if it % 500 == 0:
            if best_score < prev_best - 1e-7:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            if stall_count >= 4:
                rng = np.random.default_rng(it)
                y = y + 0.02 * rng.standard_normal(n)
                y = project_simplex(y)
                stall_count = 0

        if (it + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score

        for i in top_coords:

                a_trial[i] *= factor

                    best_a = a.copy()
                    improved = True

            if improved:

        if not improved:
            # Try pair reduction

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def make_asymmetric_init(n, alpha, rng):
    """Create asymmetric initial sequence."""
    choice = rng.integers(0, 5)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = 1.0 + alpha * np.sin(2 * np.pi * x)
    elif choice == 1:
        a = 1.0 + alpha * (2 * x - 1)
    elif choice == 2:
        a = 1.0 + alpha * np.sin(2 * np.pi * x + 0.5)
    elif choice == 3:
        a = np.exp(-3.0 * (x - 0.35) ** 2) + 0.3 * np.exp(-3.0 * (x - 0.65) ** 2)
    else:
        a = 1.0 + alpha * np.sin(4 * np.pi * x)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def make_symmetric_init(n, alpha):
    """Create symmetric initial sequence with shape parameter."""
    x = np.linspace(0, 1, n)
    a = 1.0 + alpha * np.cos(2 * np.pi * x)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a
