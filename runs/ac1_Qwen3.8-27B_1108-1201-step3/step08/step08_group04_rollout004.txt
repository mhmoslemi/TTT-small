def fast_convolve(a):
    """Linear auto-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_gradient(a, T, top_k=None):
    """
    Compute gradient of smooth max: T * log(sum_k exp(b_k/T))
    with respect to a, using FFT-based cross-correlation.
    
    Returns (gradient, weights, b, max_b).
    """
    n = len(a)
    b = fast_convolve(a)
    M = b.max()
    
    # Compute softmax weights
    if top_k is not None and top_k < 2 * n - 1:
        idx = np.argpartition(b, -top_k)[-top_k:]
        b_sel = b[idx]
        w_sel = np.exp((b_sel - b_sel.max()) / T)
        w_sel /= w_sel.sum()
        w = np.zeros(2 * n - 1)
        w[idx] = w_sel
    else:
        w = np.exp((b - b.max()) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(2 * n - 1)
            w[np.argmax(b)] = 1.0
        else:
            w /= w_sum
    
    # Gradient: grad_i = 2 * sum_j w[i+j] * a[j]
    # = 2 * (a_rev * w)[i + n - 1] where a_rev = a[::-1]
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1  # = 3n - 2
    fft_size = 1 << (conv_len - 1).bit_length()
    conv = irfft(rfft(a_rev, fft_size) * rfft(w, fft_size), fft_size)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    
    return grad, w, b, M


def smooth_gradient_optimize(a_init, time_limit, T_init=0.5, T_min=0.01,
                              lr_init=0.02, use_adam=True, label="sg"):
    """
    Optimize using smooth max gradient with multiplicative or Adam updates.
    Returns (best_a, best_score) where score = 2*n*max_conv.
    """
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    best_score = 2.0 * n * best_obj
    
    T = T_init
    lr = lr_init
    it = 0
    no_improve = 0
    total_iters = 0
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Anneal temperature
        frac = it / max(1, it + 10)  # simple progress fraction
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        T = max(T_min, T_init * (T_min / T_init) ** progress)
        
        # Compute gradient
        grad, w, b, M = compute_smooth_gradient(a, T)
        
        if M < best_obj - 1e-14:
            best_obj = M
            best_score = 2.0 * n * best_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Update strategy
        if use_adam:
            it_inner = total_iters + 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad * grad
            m_hat = m / (1 - beta1 ** it_inner)
            v_hat = v / (1 - beta2 ** it_inner)
            
            # Multiplicative (mirror descent) with Adam scaling
            step = m_hat / (np.sqrt(v_hat) + eps)
            a_new = a * np.exp(-lr * step)
            sn = a_new.sum()
            if sn > 1e-30:
                a_new /= sn
            else:
                a_new = a.copy()
        else:
            # Pure multiplicative
            a_new = a * np.exp(-lr * grad)
            sn = a_new.sum()
            if sn > 1e-30:
                a_new /= sn
            else:
                a_new = a.copy()
        
        # Occasionally try Euclidean projection too
        if it % 50 == 0:
            a_eucl = project_simplex(a - lr * grad)
            obj_eucl = fast_convolve(a_eucl).max()
            obj_mult = fast_convolve(a_new).max()
            if obj_eucl < obj_mult:
                a_new = a_eucl
        
        a = a_new
        total_iters += 1
        
        # Adaptive learning rate
        if no_improve > 100:
            lr *= 0.9
            no_improve = 0
            # Try a perturbation to escape
            if total_iters % 500 == 0:
                rng = np.random.default_rng(total_iters)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
        
        if it % 2000 == 0:
            print(f"    {label} it={it} obj={best_obj:.12f} score={best_score:.6f} "
                  f"T={T:.4f} lr={lr:.5f} t={elapsed:.0f}s")
    
    return best_a, best_score


def build_constraint_matrix(f, indices, n):
    """Build K x n constraint matrix for given peak indices."""
    K = len(indices)
    A = np.zeros((K, n))
    for i in range(K):
        k = indices[i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def lp_one_step(f, top_k=100, solver='ECOS'):
    """
    Single LP step: find direction g that maximizes sum(g) subject to
    A@g <= M, g >= 0. Returns blended result or None.
    """
    n = len(f)
    b = fast_convolve(f)
    M = b.max()
    full_len = 2 * n - 1
    
    if top_k >= full_len:
        indices = np.arange(full_len)
    else:
        indices = np.argpartition(b, -top_k)[-top_k:]
        indices = np.sort(indices)
    
    K = len(indices)
    A = build_constraint_matrix(f, indices, n)
    
    g = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])
    
    try:
        if K <= 500 and n <= 2000:
            prob.solve(solver=cp.ECOS, max_iters=20000)
        else:
            prob.solve(solver=cp.SCS, max_iters=100000, eps_abs=1e-8, eps_rel=1e-8)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=50000, eps_abs=1e-7, eps_rel=1e-7)
        except Exception:
            return None, None
    
    if g.value is None:
        return None, None
    
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None, None
    
    S_g = g_val.sum()
    if S_g <= 1.0 + 1e-5:
        return None, None
    
    h = g_val / S_g
    
    # Try line search
    best_f = f.copy()
    best_obj = M
    for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
        f_trial = (1 - t) * f + t * h
        f_trial = np.maximum(f_trial, 0)
        s = f_trial.sum()
        if s < 1e-15:
            continue
        f_trial /= s
        obj = fast_convolve(f_trial).max()
        if obj < best_obj:
            best_obj = obj
            best_f = f_trial.copy()
    
    return best_f, best_obj


def lp_optimize(f_init, time_limit, top_k_start=100, top_k_end=400, label="lp"):
    """LP-based optimization with growing constraint set."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    it = 0
    stuck = 0
    top_k = top_k_start
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        stuck += 0
        
        # Grow top_k
        progress = (time.monotonic() - t0) / time_limit
        top_k = int(top_k_start + progress * (top_k_end - top_k_start))
        
        result_f, result_obj = lp_one_step(f, top_k=top_k)
        
        if result_f is not None and result_obj < best_obj - 1e-12:
            f = result_f
            best_obj = result_obj
            best_score = 2.0 * n * best_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 10:
                # Try perturbation
                rng = np.random.default_rng(it * 7919)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                stuck = 0
            elif stuck > 30:
                break
        
        if it % 10 == 0:
            print(f"    {label} it={it} obj={best_obj:.12f} score={best_score:.6f} "
                  f"k={top_k} t={time.monotonic()-t0:.0f}s")
    
    return best_f, best_score


def coordinate_descent(f_init, time_limit, n_multipliers=7):
    """Coordinate descent with multiplicative multipliers."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    best_f = f.copy()
    
    multipliers = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]
    n_multipliers = len(multipliers)
    
    no_pass = 0
    while time.monotonic() - t0 < time_limit:
        any_improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue
            
            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = f[j] * mult
                if new_val < 0:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-12:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-12:
                    best_j_obj = obj
                    best_j_val = new_val
            
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-12:
                    f = f / s
                if best_j_obj < best_obj - 1e-12:
                    best_obj = best_j_obj
                    best_score = 2.0 * n * best_obj
                    best_f = f.copy()
                    any_improved = True
        
        if not any_improved:
            no_pass += 1
            if no_pass >= 2:
                break
        else:
            no_pass = 0
    
    return best_f, best_score


def interp_sequence(src, target_n):
    """Interpolate a sequence to a new length."""
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result
