        # Compute "negative gradient" direction
        delta = np.zeros(n)
        for k in pk:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            delta[idx] += f[k - idx]

        if delta.max() < 1e-15:
            break

        # Try multiple step sizes
        best_alpha = None
        best_trial = best_obj
        for alpha in [0.001, 0.003, 0.005, 0.01, 0.02, 0.03, 0.05, 0.08]:
            f_trial = project_simplex(f - alpha * delta / delta.max())
            obj = fast_convolve(f_trial).max()
            if obj < best_trial - 1e-14:
                best_trial = obj
                best_alpha = alpha

        if best_alpha is None:
            break

        f_new = project_simplex(f - best_alpha * delta / delta.max())
        obj_new = fast_convolve(f_new).max()
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new

        if it % 50 == 0:
            print(f"    polish iter={it} obj={best_obj:.10f} "
                  f"score={2.0*n*best_obj:.6f} t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def hann_window(n):
    w = 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n) / max(1, n)))
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def flat_top_window(n):
    w = np.ones(n)
    a = [0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368]
    for i, ai in enumerate(a):
        w -= ai * np.cos(2.0 * np.pi * i * np.arange(n) / max(1, n - 1) - i * np.pi)
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def raised_cosine(n):
    x = np.linspace(0.0, 1.0, n)
    w = 0.5 * (1.0 + np.cos(2.0 * np.pi * (x - 0.5)))
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def smoothed_uniform(n, sigma_frac=0.15):
    """Uniform window with slight raised-cosine smoothing at edges."""
    x = np.linspace(-0.5, 0.5, n)
    w = np.maximum(1.0 - (x / sigma_frac) ** 2, 0.0)
    # Blend with uniform
    w = 0.7 * np.ones(n) + 0.3 * w
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s
