Optimized auto-convolution minimization combining:

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(w, a):
    """
    Compute g_i = sum_k w_k * a_{k-i} (the subgradient of sum_k w_k * b_k w.r.t. a)
    where b = a * a (auto-convolution).
    """
    n = len(a)
    # Use FFT: g = ifft(FFT(w) * FFT(a_reversed)) sliced appropriately
    # Actually, g_i = sum_k w_k * a_{k-i} which is cross-correlation
    # = convolve(w, a[::-1]) at positions [n-1, 2n-2]
    fft_size = 1 << ((2 * n - 1) + (n - 1) - 1).bit_length()
    W = rfft(w, fft_size)
    A_rev = rfft(a[::-1], fft_size)
    c = irfft(W * A_rev, fft_size)
    return c[n-1:2*n-1]

def build_constraint_matrix(f, indices, n):
    """Build K x n constraint matrix for given peak indices."""
    K = len(indices)
    A = np.zeros((K, n))
    for i in range(K):
        k = int(indices[i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi+1] = f[lo:hi+1]
    return A


def solve_lp(f, M, indices):
    """
    Solve LP: max sum(g) s.t. A@g <= M, g >= 0.
    Returns g or None.
    """
    n = len(f)
    K = len(indices)
    if K == 0:
        return np.ones(n) / n
    
    A = build_constraint_matrix(f, indices, n)
    
    g_var = cp.Variable(n)
    constraints = [g_var >= 0, A @ g_var <= M]
    objective = cp.Maximize(cp.sum(g_var))
    prob = cp.Problem(objective, constraints)
    
    try:
        if K <= 200 and n <= 2000:
            prob.solve(solver=cp.ECOS, max_iters=50000, abs_acceleration=1e-10)
        else:
            prob.solve(solver=cp.SCS, max_iters=100000, eps_abs=1e-9, eps_rel=1e-9)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=50000, eps_abs=1e-8, eps_rel=1e-8)
        except Exception:
            return None
    
    if g_var.value is None:
        return None
    g_val = np.maximum(g_var.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None
    return g_val

def interp_sequence(src, target_n):
    """Interpolate a sequence to a new length."""
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result


def coordinate_descent(f_init, time_limit):

    t0 = time.monotonic()

    best_f = f.copy()

    no_pass = 0
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue
            best_val = f[j]
            best_obj_j = best_obj
            for m in multipliers:
                f_trial = f.copy()
                f_trial[j] = f[j] * m
                s = f_trial.sum()
                if s < 1e-12:
                    continue
                f_trial = f_trial / s
                obj = fast_convolve(f_trial).max()
                if obj < best_obj_j - 1e-13:
                    best_obj_j = obj
                    best_val = f[j] * m
            if best_val != f[j]:
                f[j] = best_val
                s = f.sum()
                f = f / s
                if best_obj_j < best_obj - 1e-13:
                    best_obj = best_obj_j
                    best_f = f.copy()
                    improved = True
        if not improved:
            no_pass += 1
            if no_pass >= 2:
                break
        else:
            no_pass = 0
    
    return best_f, 2.0 * n * best_obj

"""
Optimized auto-convolution minimization combining:
1. FISTA with log-sum-exp smoothing (fast gradient-based initialization)
2. LP-based direction finding with adaptive peak selection and momentum
3. Multi-scale exploration across sequence lengths
4. Subgradient polishing with FFT-accelerated gradients
5. Coordinate descent for final refinement
6. Final LP burst with maximum constraint count

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(w, a):
    """
    Compute g_i = sum_k w_k * a_{k-i} for i=0..n-1.
    This is the gradient of sum_k w_k * (a*a)_k w.r.t. a.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    # g = convolve(w, a_reversed)[n-1 : 2n-1]
    fft_size = 1 << (3 * n - 3).bit_length()
    W = rfft(w, fft_size)
    Ar = rfft(a[::-1], fft_size)
    c = irfft(W * Ar, fft_size)
    return c[n - 1:2 * n - 1]


def fista_smoothed(f_init, time_limit, T_init=None, T_min=0.005):
    """
    FISTA with log-sum-exp smoothing for fast optimization.
    Minimizes max_k (a*a)_k subject to a >= 0, sum(a) = 1.
    """
    t0 = time.monotonic()
    n = len(f_init)
    a = f_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s

    b0 = fast_convolve(a)
    if T_init is None:
        T_init = b0.max() / 8.0
    T = T_init
    y = a.copy()
    a_prev = a.copy()
    t_mom = 1.0

    # Initial Lipschitz estimate
    b = fast_convolve(y)
    b_shifted = b - b.max()
    w = np.exp(b_shifted / T)
    w_sum = w.sum()
    if w_sum > 1e-30:
        w = w / w_sum
    else:
        w = np.zeros(len(b))
        w[int(np.argmax(b))] = 1.0
    g_init = compute_subgradient(w, y)
    L = max(1.0, 2.0 * float(np.max(np.abs(g_init))) * n)
    lr = 1.0 / L

    best_a = a.copy()
    best_obj = b0.max()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        z = y + ((t_mom - 1.0) / t_new) * (y - a_prev)
        z = np.maximum(z, 0.0)
        z_sum = z.sum()
        if z_sum > 1e-15:
            z = z / z_sum

        # Gradient at y (use y for stability)
        b = fast_convolve(y)
        b_max = b.max()
        b_shifted = b - b_max
        w = np.exp(b_shifted / T)
        w_sum = w.sum()
        if w_sum > 1e-30:
            w = w / w_sum
        else:
            w = np.zeros(len(b))
            w[int(np.argmax(b))] = 1.0

        g = compute_subgradient(w, y)

        # Gradient step with backtracking
        for _ in range(10):
            a_new = z - lr * g
            a_new = project_simplex(a_new)
            obj_new = fast_convolve(a_new).max()
            if obj_new <= b_max + 0.5 * lr * float(np.sum(g * (a_new - z))) + 1e-12:
                break
            lr *= 0.5

        a_prev = y.copy()
        y = a_new.copy()
        t_mom = t_new

        # Adapt T and L periodically
        if it % 30 == 0:
            T = max(T * 0.88, T_min)
            b = fast_convolve(y)
            b_shifted = b - b.max()
            w = np.exp(b_shifted / T)
            ws = w.sum()
            if ws > 1e-30:
                w = w / ws
            g_check = compute_subgradient(w, y)
            L = max(L * 0.95, 2.0 * float(np.max(np.abs(g_check))) * n)
            lr = 1.0 / L

        obj = fast_convolve(y).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = y.copy()

        if it % 80 == 0:
            elapsed = time.monotonic() - t0
            print(f"    fista it={it} obj={best_obj:.10f} T={T:.5f} lr={lr:.6f} t={elapsed:.0f}s")

    return best_a, 2.0 * n * best_obj


def build_constraint_matrix(f, indices, n):
    """Build K x n constraint matrix A where A[k, :] represents (f*g)_k <= M."""
    K = len(indices)
    A = np.zeros((K, n))
    for i in range(K):
        k = int(indices[i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp(f, M, indices):
    """
    Solve LP: max sum(g) s.t. A@g <= M, g >= 0.
    Returns g array or None on failure.
    """
    n = len(f)
    K = len(indices)
    if K == 0:
        return np.ones(n) / n

    A = build_constraint_matrix(f, indices, n)

    g_var = cp.Variable(n)
    constraints = [g_var >= 0, A @ g_var <= M]
    objective = cp.Maximize(cp.sum(g_var))
    prob = cp.Problem(objective, constraints)

    solved = False
    # Try solvers in order of preference
    solvers_to_try = []
    if K <= 300 and n <= 2000:
        solvers_to_try = [
            (cp.ECOS, {'max_iters': 50000, 'abs_acceleration': 1e-10}),
            (cp.SCS, {'max_iters': 100000, 'eps_abs': 1e-9, 'eps_rel': 1e-9}),
        ]
    else:
        solvers_to_try = [
            (cp.PDLP, {'max_iterations': 5000}),
            (cp.SCS, {'max_iters': 200000, 'eps_abs': 1e-9, 'eps_rel': 1e-9}),
            (cp.ECOS, {'max_iters': 100000, 'abs_acceleration': 1e-10}),
        ]

    for solver, kwargs in solvers_to_try:
        try:
            prob.solve(solver=solver, **kwargs)
            solved = True
            break
        except Exception:
            continue

    if not solved or g_var.value is None:
        return None

    g_val = np.maximum(g_var.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None
    return g_val


def lp_iterate(f, time_limit, top_k_init=50, top_k_max=500, momentum_beta=0.35):
    """LP-based optimization with adaptive peak selection and momentum."""
    t0 = time.monotonic()
    n = len(f)
    f = f.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    momentum_dir = np.zeros(n)
    top_k = top_k_init
    stuck = 0
    it = 0
    full_len = 2 * n - 1

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        # Adaptive peak selection
        if top_k >= full_len:
            indices = np.arange(full_len)
        else:
            k_sel = min(top_k, full_len)
            top_idx = np.argpartition(b, -k_sel)[-k_sel:]
            # Add boundary peaks
            n_bnd = min(8, n // 5)
            bnd = list(range(max(0, n_bnd)))
            bnd += list(range(max(0, full_len - n_bnd), full_len))
            # Add center peaks
            center_start = max(0, (full_len - 1) // 2 - 3)
            bnd += list(range(center_start, min(full_len, center_start + 7)))
            all_idx = list(set(list(top_idx) + bnd))
            all_idx.sort(key=lambda i: b[int(i)], reverse=True)
            indices = np.array(all_idx[:top_k])

        # Solve LP
        g = solve_lp(f, M, indices)

        if g is None:
            stuck += 2
            if stuck > 40:
                break
            top_k = max(20, top_k // 2)
            continue

        S_g = g.sum()
        if S_g <= 1.0 + 1e-5:
            stuck += 1
            if stuck > 30:
                break
            top_k = min(top_k + 15, top_k_max)
            continue

        h = g / S_g
        momentum_dir = momentum_beta * momentum_dir + (1.0 - momentum_beta) * h
        mom_sum = momentum_dir.sum()
        if mom_sum > 1e-10:
            h_mom = momentum_dir / mom_sum
        else:
            h_mom = h.copy()

        # Line search over multiple directions and steps
        best_t_obj = M
        best_f_trial = f.copy()
        directions = [h, h_mom]
        if it > 20:
            # Also try a "centroid" direction
            h_cent = (h + h_mom) / 2.0
            cs = h_cent.sum()
            if cs > 1e-10:
                directions.append(h_cent / cs)

        for direction in directions:
            for t_step in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
                f_trial = (1.0 - t_step) * f + t_step * direction
                f_trial = np.maximum(f_trial, 0.0)
                s_trial = f_trial.sum()
                if s_trial < 1e-12:
                    continue
                f_trial = f_trial / s_trial
                obj_trial = fast_convolve(f_trial).max()
                if obj_trial < best_t_obj - 1e-14:
                    best_t_obj = obj_trial
                    best_f_trial = f_trial.copy()

        if best_t_obj >= M - 1e-14:
            stuck += 1
            if stuck > 50:
                # Random restart
                rng = np.random.default_rng(it * 7 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                momentum_dir = np.zeros(n)
                stuck = 0
            top_k = min(top_k + 10, top_k_max)
            continue

        f = best_f_trial
        cur_obj = fast_convolve(f).max()

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
            # Gradually increase constraints
            if it % 10 == 0:
                top_k = min(top_k + 5, top_k_max)
        else:
            stuck += 1

        if it % 25 == 0:
            elapsed = time.monotonic() - t0
            print(f"    lp it={it} obj={best_obj:.10f} score={2*n*best_obj:.6f} k={top_k} stuck={stuck} t={elapsed:.0f}s")

    return best_f, 2.0 * n * best_obj


def subgradient_polish(f_init, time_limit, lr_init=0.002, n_top=80):
    """Subgradient optimization with momentum and FFT-accelerated gradients."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()

        f = f / s

    v = np.zeros(n)
    lr = lr_init
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    temp = 0.0003
    no_improve = 0
    it = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        # Softmax weights on peaks
        scaled = (b - M) / temp
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum > 1e-30:
            w = w / w_sum
        else:
            w = np.zeros(len(b))
            w[int(np.argmax(b))] = 1.0

        # Restrict to top peaks for efficiency
        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w_sel = w[top_idx].copy()
            ws = w_sel.sum()
            if ws > 1e-30:
                w_sel = w_sel / ws
            w = np.zeros(len(b))
            w[top_idx] = w_sel

        # FFT-accelerated subgradient
        g = compute_subgradient(w, f)

        # Adam-like momentum
        v = 0.85 * v - lr * g

        if it % 15 == 0:

            if obj < best_obj - 1e-14:

                lr = lr_init

            if it % 100 == 0:
                elapsed = time.monotonic() - t0

def interp_sequence(src, target_n):
    """Cubic-like interpolation of a sequence to a new length."""
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result


def coordinate_descent(f_init, time_limit):
    """Coordinate descent for final polishing."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    multipliers = [0.5, 0.7, 0.8, 0.85, 0.9, 1.1, 1.15, 1.2, 1.3, 1.5]
    no_pass = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue
            best_val = f[j]
            best_obj_j = best_obj
            for m in multipliers:
                f_trial = f.copy()
                f_trial[j] = f[j] * m
                s = f_trial.sum()
                if s < 1e-12:
                    continue
                f_trial = f_trial / s
                obj = fast_convolve(f_trial).max()
                if obj < best_obj_j - 1e-13:
                    best_obj_j = obj
                    best_val = f[j] * m
            if best_val != f[j]:
                f[j] = best_val
                s = f.sum()
                if s > 1e-12:
                    f = f / s
                if best_obj_j < best_obj - 1e-13:
                    best_obj = best_obj_j
                    best_f = f.copy()
                    improved = True
        if not improved:
            no_pass += 1
            if no_pass >= 2:
                break
        else:
            no_pass = 0

    return best_f, 2.0 * n * best_obj

        print(f"\n{'='*50}")
        print(f"Phase 3: LP n={len(best_seq)}, budget={t3:.0f}s")
        print(f"{'='*50}")
        f_res, _ = lp_iterate(best_seq, time_limit=t3,
                             top_k_init=50,
                             top_k_max=min(600, 2 * len(best_seq) - 1),
                             momentum_beta=0.35)
        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = f_res.copy()
            print(f"  LP improved: {real_score:.6f}")
        else:
            print(f"  LP no improvement (got {real_score:.6f})")

    # ==========================================
    # Phase 4: LP on nearby lengths
    # ==========================================
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.15, 180.0)
    if t4 > 20:
        print(f"\n{'='*50}")
        print(f"Phase 4: LP nearby lengths, budget={t4:.0f}s")
        print(f"{'='*50}")
        n_base = len(best_seq)
        nearby = []
        for factor in [0.85, 0.95, 1.05, 1.15]:
            nt = int(n_base * factor)
            if nt != len(best_seq) and nt not in nearby:
                nt = max(256, min(2500, nt))
                nearby.append(nt)

        if nearby:
            per_n = t4 / len(nearby)
            for nt in nearby:
                rem = deadline - time.monotonic()
                if rem < 15:
                    break
                alloc = min(per_n, rem * 0.85)
                if alloc < 10:
                    break
                f_init = interp_sequence(best_seq, nt)
                print(f"  LP n={nt}, budget={alloc:.0f}s")
                f_res, _ = lp_iterate(f_init, time_limit=alloc,
                                     top_k_init=80,
                                     top_k_max=min(500, 2 * nt - 1),
                                     momentum_beta=0.4)
                real_score = float(evaluate_sequence(f_res.tolist()))
                if np.isfinite(real_score) and real_score < best_score:
                    best_score = real_score
                    best_seq = f_res.copy()
                    print(f"  NEW BEST [LP n={nt}]: {real_score:.6f}")

    # ==========================================
    # Phase 5: Subgradient polish
    # ==========================================
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.08, 80.0)
    if t5 > 12:
        print(f"\n{'='*50}")
        print(f"Phase 5: SG polish n={len(best_seq)}, budget={t5:.0f}s")
        print(f"{'='*50}")
        f_res, _ = subgradient_polish(best_seq, time_limit=t5,
                                     lr_init=0.002,
                                     n_top=max(50, len(best_seq) // 8))
        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = f_res.copy()
            print(f"  SG improved: {real_score:.6f}")

    # ==========================================
    # Phase 6: Coordinate descent
    # ==========================================
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.06, 50.0)
    if t6 > 8:
        print(f"\n{'='*50}")
        print(f"Phase 6: CD n={len(best_seq)}, budget={t6:.0f}s")
        print(f"{'='*50}")
        f_res, _ = coordinate_descent(best_seq, time_limit=t6)
        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = f_res.copy()
            print(f"  CD improved: {real_score:.6f}")

    # ==========================================
    # Phase 7: Final LP burst
    # ==========================================
    rem = deadline - time.monotonic()
    t7 = max(0.0, rem - 10.0)
    if t7 > 12:
        print(f"\n{'='*50}")
        print(f"Phase 7: Final LP n={len(best_seq)}, budget={t7:.0f}s")
        print(f"{'='*50}")
        f_res, _ = lp_iterate(best_seq, time_limit=t7,
                             top_k_init=120,
                             top_k_max=min(1000, 2 * len(best_seq) - 1),
                             momentum_beta=0.3)
        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = f_res.copy()
            print(f"  Final LP improved: {real_score:.6f}")

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        else:
            best_seq = parent.copy()
            best_score = float(evaluate_sequence(best_seq.tolist()))
    except Exception:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL: score={best_score:.6f}, n={len(best_seq)}")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
