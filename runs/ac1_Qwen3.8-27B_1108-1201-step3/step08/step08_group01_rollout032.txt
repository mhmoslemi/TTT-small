def exact_1d_search(a, conv, i, n):
    """Find optimal delta for coordinate i using 1D search."""
    k_start = i
    k_end = min(i + n - 1, 2*n - 2)
    
    # Unaffected max
    if k_start > 0:
        unaff_max = conv[:k_start].max()
    else:
        unaff_max = 0
    if k_end < 2*n-2:
        unaff_right = conv[k_end+1:].max()
        unaff_max = max(unaff_max, unaff_right)
    
    M = conv.max()
    affected_max = max(unaff_max, conv[k_start:k_end+1].max())
    
    # Candidate deltas from critical constraints
    candidates = [0.0]
    for k in range(k_start, k_end+1):
        coeff = 2.0 * a[k-i]  # coefficient of delta in c'_k
        if coeff > 1e-15:
            delta_crit = (M - conv[k]) / coeff
            candidates.append(delta_crit)
    # Quadratic critical point
    if 2*i >= k_start and 2*i <= k_end:
        # c_{2i} + 2*delta*a_i + delta^2 = M
        # delta^2 + 2*a_i*delta + (c_{2i} - M) = 0
        disc = 4*a[i]**2 - 4*(conv[2*i] - M)
        if disc >= 0:
            delta_q = -a[i] + np.sqrt(max(0, a[i]**2 - conv[2*i] + M))
            candidates.append(delta_q)
    
    # Add some heuristics
    candidates.extend([-0.5*a[i], -0.25*a[i], -0.1*a[i], 0.1*a[i], 0.25*a[i], 0.5*a[i]])
    
    # Filter valid candidates
    candidates = [d for d in candidates if d >= -a[i] - 1e-15]
    candidates = [d for d in candidates if abs(d) < 10*a[i] + 1e-6]  # bound
    
    # Evaluate score at each candidate
    best_delta = 0.0
    best_score = 2.0 * n * M  # current score (normalized, sum=1)
    
    for delta in candidates:
        if delta < -a[i] - 1e-15:
            continue
        # Compute new max
        new_max = unaff_max
        for k in range(k_start, k_end+1):
            if k == 2*i:
                new_val = conv[k] + 2*delta*a[i] + delta*delta
            else:
                new_val = conv[k] + 2*delta*a[k-i]
            if new_val > new_max:
                new_max = new_val
        new_sum = 1.0 + delta
        if new_sum < 0.01:
            continue
        score = 2.0 * n * new_max / (new_sum * new_sum)
        if score < best_score:
            best_score = score
            best_delta = delta
    
    return best_delta, best_score

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def normalize_seq(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a), dtype=np.float64) / len(a)
    return a / s


def compute_score(a):
    """Compute 2n*max(conv(a,a)) assuming a is normalized (sum=1)."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    M = fast_convolve(a / s).max()
    return float(2.0 * n * M)


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=np.float64) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def build_convolution_matrix(a):
    """Build T matrix such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a, solver=None):
    """
    Given fixed a (normalized, non-negative), find b >= 0, sum(b)=1
    that minimizes max_k (a*b)_k.
    Returns (b, M_opt) or (None, None).
    """
    import cvxpy as cp
    n = len(a)

    if solver is None:
        try:
            solver = _get_solver()
        except Exception:
            solver = cp.ECOS

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_convolution_matrix(a)

    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    objective = cp.Minimize(M_var)
    prob = cp.Problem(objective, constraints)

    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.GLPK, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def lp_line_search(a, b, n, conv_a, time_per_call=0.5):
    """
    Given current a and LP best response b, do a line search along a -> b.
    Returns the best point found.
    """
    best_a = a.copy()
    best_score = compute_score(a)

    # Try a range of mixing parameters
    alphas = np.array([0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.35, 0.4, 0.5, 0.6, 0.7, 0.8])

    for alpha in alphas:
        candidate = (1.0 - alpha) * a + alpha * b
        candidate = np.maximum(candidate, 0)
        s = candidate.sum()
        if s < 0.01:
            continue
        candidate = candidate / s
        score = compute_score(candidate)
        if score < best_score:
            best_score = score
            best_a = candidate.copy()

    # Also try the "complementary" direction: a - b (if it stays non-negative)
    diff = a - b
    if np.min(diff) > -1e-10:
        for alpha in [0.1, 0.2, 0.3]:
            candidate = a + alpha * diff
            candidate = np.maximum(candidate, 0)
            s = candidate.sum()
            if s < 0.01:
                continue
            candidate = candidate / s
            score = compute_score(candidate)
            if score < best_score:
                best_score = score
                best_a = candidate.copy()

    return best_a, best_score


def block_coord_descent(a0, time_limit, block_size=4, verbose=False, label=""):
    """Coordinate descent that updates blocks of adjacent coordinates."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    stalls = 0

    while time.monotonic() - t0 < time_limit:
        improved = False

        for start in range(0, n, block_size):
            end = min(start + block_size, n)
            block_idx = list(range(start, end))
            n_block = end - start

            # The affected range in convolution
            k_start = start
            k_end = min(end - 1 + n - 1, 2 * n - 2)

            unaff_max = 0.0
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            # Try random perturbations of the block
            ai = a[start:end]
            for trial in range(8):
                if trial == 0:
                    perturbation = np.zeros(n_block)
                else:
                    perturbation = np.random.standard_normal(n_block) * 0.05 * np.sum(ai)

                new_block = np.maximum(ai + perturbation, 0)
                block_sum_change = new_block.sum() - ai.sum()

                # Quick check: compute new score
                a_new = a.copy()
                a_new[start:end] = new_block
                s_new = 1.0 + block_sum_change
                if s_new < 0.01:
                    continue
                a_norm = a_new / s_new
                score = compute_score(a_norm)
                if score < best_score - 1e-14:
                    a = a_norm
                    best_score = score
                    best_a = a.copy()
                    conv = fast_convolve(a)
                    ai = a[start:end]
                    improved = True

        if not improved:
            stalls += 1
            if stalls > 30:
                break
        else:
            stalls = 0

    return best_a, best_score


def peak_targeted_perturb(a0, rng, magnitude=0.03):
    """Perturb coordinates that contribute to the convolution peaks."""
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    top_k = min(10, len(conv))
    peak_positions = np.argpartition(conv, -top_k)[-top_k:]

    for k in peak_positions:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1)
        n_pick = max(1, len(idx) // 5)
        picks = rng.choice(idx, size=min(n_pick, len(idx)), replace=False)
        for i in picks:
            a[i] *= (1.0 + rng.standard_normal() * magnitude)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    return a / s


def mirror_descent_step(a, lr=0.01, top_k=20):
    """Single mirror descent step."""
    n = len(a)
    conv = fast_convolve(a)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    pv = conv[peaks]
    temp = 0.01 * max(pv.max(), 1e-10)
    w = np.exp((pv - pv.max()) / temp)
    w = w / w.sum()

    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-14:
            continue
        k = int(peaks[idx])
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        j = np.arange(i_start, i_end + 1)
        grad[j] += 2.0 * w[idx] * a[k - j]

    scaled = a * np.exp(-lr * grad)
    ss = scaled.sum()
    if ss < 1e-15:
        return a.copy()
    return scaled / ss


def simplex_project(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def init_fourier(n, rng):
    """Initialize using random Fourier superposition."""
    n_harmonics = int(rng.integers(3, 18))
    amplitudes = rng.random(n_harmonics) * 0.35
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = np.arange(1, n_harmonics + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_cosine(n, rng):

        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = normalize_seq(p)
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1024) / 1024
        p_score = 2.0
        print("No parent found, using uniform 1024")

    best_seq = parent.copy()
    best_score = float(evaluate_sequence(parent.tolist()))
    n_parent = len(parent)

    # === Phase 1: Quick multi-length exploration (300s) ===
    print(f"\n=== Phase 1: Multi-length exploration (best={best_score:.6f}) ===")
    t1_start = time.monotonic()
    t1_budget = 280.0

    n_candidates = [n_parent]
    for dn in [-192, -128, -64, 64, 128, 192, 256]:
        nc = n_parent + dn
        if 256 <= nc <= 4096:
            n_candidates.append(nc)
    # Add some "round" numbers
    for nr in [512, 768, 1024, 1280, 1536, 2048]:
        if nr not in n_candidates and 256 <= nr <= 4096:
            n_candidates.append(nr)
    n_candidates = sorted(set(n_candidates))

    phase1_results = []
    for n_try in n_candidates:
        elapsed = time.monotonic() - t1_start
        if elapsed > t1_budget:
            break
        t_left = t1_budget - elapsed
        t_alloc = max(5.0, min(25.0, t_left / max(1, len(n_candidates) - n_candidates.index(n_try))))

        # Interpolate parent
        cand = interpolate_seq(parent, n_try)

        # Quick CD
        a_opt, s_opt, _ = vectorized_coord_descent(cand, t_alloc, n_trials=20, label=f"p1_n{n_try}")
        phase1_results.append((s_opt, a_opt, n_try))

    phase1_results.sort(key=lambda x: x[0])
    print(f"  Phase 1 top: {phase1_results[0][0]:.6f} (n={phase1_results[0][2]})")

    for s_c, a_c, n_c in phase1_results[:3]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_n{n_c}")

    # === Phase 2: Intensive vectorized CD (500s) ===
    print(f"\n=== Phase 2: Intensive CD (best={best_score:.6f}) ===")
    t2_start = time.monotonic()
    t2_budget = 500.0
    n_best = len(best_seq)

    # Multiple rounds with perturbation
    current = best_seq.copy()
    for rnd in range(4):
        rem = t2_budget - (time.monotonic() - t2_start)
        if rem < 20:
            break
        t_alloc = min(rem * 0.4, 130.0)
        print(f"  CD round {rnd+1}, n={n_best}, budget={t_alloc:.0f}s")
        a_opt, s_opt, sweeps = vectorized_coord_descent(current, t_alloc, n_trials=40,
                                                         verbose=True, label=f"ph2r{rnd}")
        best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"ph2_cd_r{rnd}")

        # Perturb to escape
        if best_score > 1.5035:
            current = peak_targeted_perturb(best_seq, rng, magnitude=0.02 + 0.01 * rnd)
        else:
            current = best_seq.copy()

    # === Phase 3: LP-guided optimization (500s) ===
    print(f"\n=== Phase 3: LP-guided (best={best_score:.6f}) ===")
    t3_start = time.monotonic()
    t3_budget = 500.0

    solver = _get_solver()
    print(f"  Using LP solver: {solver}")

    n_best = len(best_seq)
    current = best_seq.copy()
    lp_count = 0

    while time.monotonic() - t3_start < t3_budget:
        rem = t3_budget - (time.monotonic() - t3_start)
        if rem < 10:
            break

        # Solve LP
        t_lp_start = time.monotonic()
        b_opt, M_opt = solve_lp_best_response(current, solver)
        t_lp = time.monotonic() - t_lp_start
        lp_count += 1

        if b_opt is not None:
            # Line search along a -> b
            a_ls, s_ls = lp_line_search(current, b_opt, n_best, None)
            best_seq, best_score = check_and_update(a_ls, best_seq, best_score, f"lp_{lp_count}")
            current = a_ls.copy() if s_ls < compute_score(current) else current.copy()
        else:
            # LP failed, do CD instead
            pass

        # Periodic CD polish
        if lp_count % 5 == 0:
            t_cd = min(rem * 0.2, 30.0)
            a_cd, s_cd, _ = vectorized_coord_descent(current, t_cd, n_trials=30, label=f"ph3_cd{lp_count}")
            best_seq, best_score = check_and_update(a_cd, best_seq, best_score, f"ph3_cd{lp_count}")
            current = a_cd.copy()

        if lp_count % 10 == 0:
            print(f"  LP iter={lp_count}, lp_time={t_lp:.3f}s, best={best_score:.6f}")

    # === Phase 4: Fine polishing (200s) ===
    print(f"\n=== Phase 4: Fine Polish (best={best_score:.6f}) ===")
    t4_start = time.monotonic()
    t4_budget = 200.0

    # Fine CD
    t_fine1 = min(t4_budget * 0.4, 80.0)
    a_fine, s_fine, _ = fine_coord_descent(best_seq, t_fine1, verbose=True, label="fine1")
    best_seq, best_score = check_and_update(a_fine, best_seq, best_score, "fine_cd1")

    # More VCD with many trials
    rem = t4_budget - (time.monotonic() - t4_start)
    if rem > 20:
        t_fine2 = min(rem * 0.5, 60.0)
        a_fine2, s_fine2, _ = vectorized_coord_descent(best_seq, t_fine2, n_trials=50, label="fine2")
        best_seq, best_score = check_and_update(a_fine2, best_seq, best_score, "fine_cd2")

    # Block CD
    rem = t4_budget - (time.monotonic() - t4_start)
    if rem > 20:
        t_block = min(rem * 0.5, 40.0)
        a_blk, s_blk = block_coord_descent(best_seq, t_block, block_size=3, label="block")
        best_seq, best_score = check_and_update(a_blk, best_seq, best_score, "block_cd")

    # Mirror descent
    rem = t4_budget - (time.monotonic() - t4_start)
    if rem > 10:
        a = best_seq.copy()
        lr = 0.005
        t_md_start = time.monotonic()
        for it in range(500):
            if time.monotonic() - t_md_start > rem * 0.3:
                break
            a = mirror_descent_step(a, lr=lr, top_k=20)
            lr *= 0.999
            score = compute_score(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
        print(f"  Mirror descent: best={best_score:.6f}")

    # === Phase 5: Alternative lengths (150s) ===
    print(f"\n=== Phase 5: Alt Lengths (best={best_score:.6f}) ===")
    t5_start = time.monotonic()
    t5_budget = 150.0
    n_best = len(best_seq)

    for dn in [-128, -64, 0, 64, 128]:
        nc = n_best + dn
        if nc < 256 or nc > 4096:
            continue
        if nc == n_best and dn > 0:
            continue
        rem = t5_budget - (time.monotonic() - t5_start)
        if rem < 20:
            break
        t_alloc = min(rem * 0.4, 40.0)
        alt = interpolate_seq(best_seq, nc)
        a_alt, s_alt, _ = vectorized_coord_descent(alt, t_alloc, n_trials=30, label=f"alt_{nc}")
        best_seq, best_score = check_and_update(a_alt, best_seq, best_score, f"alt_n{nc}")

    # Fine polish at new best length
    rem = t5_budget - (time.monotonic() - t5_start)
    if rem > 15:
        a_fp, s_fp, _ = fine_coord_descent(best_seq, min(rem * 0.5, 30.0), label="alt_fine")
        best_seq, best_score = check_and_update(a_fp, best_seq, best_score, "alt_fine")

    # === Phase 6: Final intensive LP + CD (100s) ===
    print(f"\n=== Phase 6: Final LP+CD (best={best_score:.6f}) ===")
    t6_start = time.monotonic()
    t6_budget = 100.0
    current = best_seq.copy()

    while time.monotonic() - t6_start < t6_budget:
        rem = t6_budget - (time.monotonic() - t6_start)
        if rem < 5:
            break

        # Quick LP
        b_opt, M_opt = solve_lp_best_response(current, solver)
        if b_opt is not None:
            a_ls, s_ls = lp_line_search(current, b_opt, len(current), None)
            best_seq, best_score = check_and_update(a_ls, best_seq, best_score, f"final_lp")
            current = a_ls.copy()

        # Quick CD
        t_cd = min(rem * 0.3, 15.0)
        a_cd, s_cd, _ = vectorized_coord_descent(current, t_cd, n_trials=30, label="final_cd")
        best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "final_cd")
        current = a_cd.copy()

    # === Final: Last polish ===
    print(f"\n=== Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 5:
        a_last, s_last, _ = fine_coord_descent(best_seq, min(rem * 0.7, 30.0), label="last")
        best_seq, best_score = check_and_update(a_last, best_seq, best_score, "last_fine")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
