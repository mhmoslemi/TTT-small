"""
FFT-based maximin optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

Uses Sion's minimax theorem: min_f max_k (f*f)(k) = max_p min_f sum_k p_k (f*f)(k).
The inner QP is solved via accelerated projected gradient with FFT-based
Hankel matrix-vector products. Outer loop updates the distribution p over lags.

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mvm(p, f):
    """Compute Q @ f where Q_{ij} = p_{i+j}, using FFT."""
    n = len(f)
    L = 1 << (3 * n).bit_length()
    P = np.zeros(L)
    P[:len(p)] = p
    F = np.zeros(L)
    F[:n] = f[::-1]
    result = irfft(rfft(P, L) * rfft(F, L), L)
    return result[n-1:n-1+n].copy()


def compute_objective(f):
    """Compute 2*n*max(conv(f,f)) for normalized f."""
    n = len(f)
    b = fast_convolve(f)
    return 2.0 * n * b.max()


def run_maximin(f_init, time_limit, alpha_init=2.0, alpha_max=50.0,
                lr=0.8, momentum=0.9, p_update_interval=50,
                label="mm"):
    """
    FFT-based maximin optimization.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    
    # Initialize p
    b = fast_convolve(f)
    alpha = alpha_init
    p = b.copy()
    p = p / p.sum()
    
    v = np.zeros(n)  # momentum velocity
    it = 0
    stuck = 0
    no_p_update = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute gradient: grad = 2 * Q @ f where Q_{ij} = p_{i+j}
        Qf = hankel_mvm(p, f)
        grad = 2.0 * Qf
        
        # Momentum update
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        
        # Check if we made progress
        f = f_new
        
        if it % p_update_interval == 0:
            b = fast_convolve(f)
            cur_obj = b.max()
            
            if cur_obj < best_obj - 1e-14:
                best_obj = cur_obj
                best_score = 2.0 * n * cur_obj
                best_f = f.copy()
                stuck = 0
            else:
                stuck += 1
                if stuck > 20:
                    # Restart from best with perturbation
                    rng = np.random.default_rng(it + 42)
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                    v = np.zeros(n)
                    stuck = 0
                elif stuck > 10:
                    lr = max(lr * 0.9, 0.01)
            
            # Update p (every p_update_interval iterations)
            b = fast_convolve(f)
            # Concentrate p on peaks
            p = np.power(np.maximum(b, 0), alpha)
            ps = p.sum()
            if ps > 1e-30:
                p = p / ps
            else:
                p = b / b.sum()
            
            # Increase alpha gradually
            alpha = min(alpha * 1.05, alpha_max)
            
            if it % (p_update_interval * 10) == 0:
                elapsed = time.monotonic() - t0
                print(f"    {label} it={it} obj={best_obj:.12f} score={best_score:.6f} "
                      f"alpha={alpha:.1f} lr={lr:.4f} t={elapsed:.0f}s")
        
        it += 1
    
    return best_f, best_score


def coordinate_descent(f_init, time_limit):
    """Coordinate descent polish."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    best_f = f.copy()
    
    multipliers = np.array([0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0])
    
    no_pass_improve = 0
    while time.monotonic() - t0 < time_limit:
        any_improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue
            
            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                f_trial = f.copy()
                f_trial[j] = f[j] * mult
                s = f_trial.sum()
                if s < 1e-12:
                    continue
                f_trial = f_trial / s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = f[j] * mult
            
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-12:
                    f = f / s
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_score = 2.0 * n * best_obj
                    best_f = f.copy()
                    any_improved = True
        
        if not any_improved:
            no_pass_improve += 1
            if no_pass_improve >= 2:
                break
        else:
            no_pass_improve = 0
    
    return best_f, best_score


def interp_sequence(src, target_n):
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result

"""
Multi-method optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines FFT-based maximin, cyclical projection, small LP, and coordinate descent.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hankel_mvm(p, f):
    """Compute Q @ f where Q_{ij} = p_{i+j}, using FFT. O(n log n)."""
    n = len(f)
    L = 1 << (3 * n).bit_length()
    P = np.zeros(L)
    P[:len(p)] = p
    F = np.zeros(L)
    F[:n] = f[::-1]
    result = irfft(rfft(P, L) * rfft(F, L), L)
    return result[n-1:n-1+n].copy()


def run_fft_maximin(f_init, time_limit, alpha_init=2.0, alpha_max=80.0,

    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s
    else:
        f = np.ones(n) / n
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    
    b = fast_convolve(f)
    alpha = alpha_init
    p = np.power(np.maximum(b, 0), alpha)
    ps = p.sum()
    p = p / ps if ps > 1e-30 else b / b.sum()
    
    v = np.zeros(n)
    it = 0
    stuck = 0
    cur_lr = lr
    
    while time.monotonic() - t0 < time_limit:
        Qf = hankel_mvm(p, f)
        v = momentum * v - cur_lr * Qf
        f = project_simplex(f + v)
        
        if it % p_interval == 0:
            b = fast_convolve(f)
            obj = b.max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_score = 2.0 * n * obj
                best_f = f.copy()
                stuck = 0
                cur_lr = min(cur_lr * 1.02, lr * 2.0)
            else:

                    v = np.zeros(n)
                    stuck = 0
                    cur_lr = lr
                elif stuck > 10:
                    cur_lr = max(cur_lr * 0.95, 0.01)
            
            # Update p
            p = np.power(np.maximum(b, 0), alpha)
            ps = p.sum()

            alpha = min(alpha * 1.03, alpha_max)

def run_small_lp(f_init, time_limit, K=10, label="slp"):
    """Small LP with scipy for exact direction finding."""
    from scipy.optimize import linprog
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    it = 0
    stuck = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(f)
        M = b.max()
        
        # Select top K peak indices
        top_k = np.argpartition(b, -K)[-K:]
        top_k = np.sort(top_k)
        
        # Build constraint matrix A (K x n): A[k, j] = f[max(0,k-n+1):min(n-1,k)] shifted
        A = np.zeros((K, n))
        for i, k in enumerate(top_k):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A[i, lo:hi+1] = f[lo:hi+1]
        
        # Solve: max sum(g) s.t. A@g <= M, g >= 0
        # In standard form for linprog: min -sum(g) s.t. A@g <= M, -g <= 0
        c = -np.ones(n)
        A_ub = np.vstack([A, -np.eye(n)])
        b_ub = np.concatenate([np.full(K, M), np.zeros(n)])
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
            if res.success:
                g = np.maximum(res.x, 0)
                S_g = g.sum()
                if S_g > 1.0 + 1e-4:
                    h = g / S_g
                    # Line search
                    best_t_obj = M
                    best_t = 0.0
                    for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
                        f_trial = (1-t)*f + t*h
                        f_trial = np.maximum(f_trial, 0)
                        st = f_trial.sum()
                        if st < 1e-15:
                            continue
                        f_trial = f_trial / st
                        obj_t = fast_convolve(f_trial).max()
                        if obj_t < best_t_obj:
                            best_t_obj = obj_t
                            best_t = t
                    
                    if best_t > 0:
                        f = (1-best_t)*f + best_t*h
                        f = np.maximum(f, 0)
                        f = f / f.sum()
                        obj = fast_convolve(f).max()
                        if obj < best_obj - 1e-14:
                            best_obj = obj
                            best_score = 2.0 * n * obj
                            best_f = f.copy()
                            stuck = 0
                        else:
                            stuck += 1
                    else:
                        stuck += 1
                else:
                    stuck += 1
            else:
                stuck += 1
        except Exception:
            stuck += 1
        
        if stuck > 20:
            rng = np.random.default_rng(it + 99)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            stuck = 0
        
        it += 1
        if it % 20 == 0:
            print(f"    {label} it={it} obj={best_obj:.12f} score={best_score:.6f} "
                  f"t={time.monotonic()-t0:.0f}s")
    
    return best_f, best_score


def coordinate_descent(f_init, time_limit):
    """Coordinate descent with multipliers."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    best_f = f.copy()
    
    multipliers = np.array([0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0])
    no_pass = 0
    
    while time.monotonic() - t0 < time_limit:
        any_imp = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue
            best_val = f[j]
            best_obj_j = best_obj
            for mult in multipliers:
                f2 = f.copy()
                f2[j] = f[j] * mult
                sn = f2.sum()
                if sn < 1e-12:
                    continue
                f2 = f2 / sn
                obj2 = fast_convolve(f2).max()
                if obj2 < best_obj_j - 1e-13:
                    best_obj_j = obj2
                    best_val = f[j] * mult
            if best_val != f[j]:
                f[j] = best_val
                sn = f.sum()
                if sn > 1e-12:
                    f = f / sn
                if best_obj_j < best_obj - 1e-13:
                    best_obj = best_obj_j
                    best_score = 2.0 * n * best_obj
                    best_f = f.copy()
                    any_imp = True
        if not any_imp:
            no_pass += 1
            if no_pass >= 3:
                break
        else:
            no_pass = 0
    
    return best_f, best_score


def interp_sequence(src, target_n):
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    sn = result.sum()
    if sn > 1e-15:
        result = result / sn
    else:
        result = np.ones(target_n) / target_n
    return result
