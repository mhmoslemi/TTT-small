def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def compute_score_normalized(a):
    """Assume a is already normalized (sum=1). Score = 2n * max(conv)."""
    n = len(a)

    return 2.0 * n * conv.max()


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def lse_objective_and_grad(a, tau, n):
    """
    Compute the log-sum-exp smoothed objective and its gradient.
    f(a, tau) = 2n * tau * log(sum_k exp(conv_k / tau))
    grad = 2n * sum_k w_k * grad(conv_k) where w_k = softmax(conv/tau)
    """
    b = fast_convolve(a)
    # Log-sum-exp for numerical stability
    b_shifted = b / tau
    b_max = b_shifted.max()
    exp_shifted = np.exp(b_shifted - b_max)
    lse = b_max + np.log(exp_shifted.sum())
    obj = 2.0 * n * tau * lse
    
    # Weights
    w = exp_shifted / exp_shifted.sum()
    
    # Gradient: 2n * sum_k w_k * (2 * shift(a, k))
    grad = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi+1] += 2.0 * w[k] * a[k-hi:k-lo+1][::-1]
    
    grad *= 2.0 * n
    
    return obj, grad


def continuation_lse(a0, time_limit, tau_start=100.0, tau_end=1.0, n_tau_steps=8, maxiter_per_tau=500, verbose=True):
    """
    Continuation method: start with large tau (smooth), decrease to small tau.
    At each tau, run L-BFGS on the smoothed objective.
    """
    n = len(a0)
    a = a0.copy()
    a = project_simplex(a)
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    
    t0 = time.monotonic()
    taus = np.geomspace(tau_start, tau_end, n_tau_steps)
    
    def func(z):
        return lse_objective_and_grad(project_simplex(np.exp(z - z.max())), tau, n)[0]
    
    # Work in log-space to maintain positivity
    # a = softmax(z), so we optimize over z
    
    def softmax(z):
        z_shifted = z - z.max()
        e = np.exp(z_shifted)
        return e / e.sum()
    
    for tau in taus:
        if time.monotonic() - t0 > time_limit:
            break
        
        z = np.log(np.maximum(a, 1e-30))
        
        def obj(z):
            a_trial = softmax(z)
            val, _ = lse_objective_and_grad(a_trial, tau, n)
            return val
        
        def grad(z):
            a_trial = softmax(z)
            _, g_a = lse_objective_and_grad(a_trial, tau, n)
            # Chain rule: d/dz f(softmax(z)) = diag(softmax(z)) @ (g_a - softmax(z) * g_a · softmax(z))
            a_s = a_trial
            S = np.dot(g_a, a_s)
            g_z = a_s * (g_a - S)
            return g_z
        
        if verbose:
            print(f"    LSE: tau={tau:.2f}")
        
        try:
            res = minimize(
                fun=obj, x0=z, jac=grad,
                method='L-BFGS-B',
                options={'maxiter': maxiter_per_tau, 'ftol': 1e-15, 'gtol': 1e-12,
                         'maxfun': maxiter_per_tau * 100, 'disp': False}
            )
            a = softmax(res.x)
            score = compute_score_normalized(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
        except Exception as e:
            if verbose:
                print(f"      L-BFGS error at tau={tau}: {e}")
    
    return best_a, best_score


def build_conv_matrix(f):
    """Build (2n-1) x n matrix C such that C @ g = conv(f, g)."""
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_best_response(f, M, prev_g=None):
    """Solve: max sum(g) s.t. C(f) @ g <= M, g >= 0."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = np.maximum(prev_g, 0)
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None))
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return prev_g if prev_g is not None else f.copy()
    if g.value is None:
        return prev_g if prev_g is not None else f.copy()
    return np.maximum(g.value, 0)


def lp_fictitious_play(a0, time_limit, step=0.3, verbose=True):
    """LP-based fictitious play."""
    n = len(a0)
    a = a0.copy()
    a = project_simplex(a)
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    
    t0 = time.monotonic()
    prev_g = None
    stall_count = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        
        g = solve_best_response(a, M, prev_g=prev_g)
        prev_g = g
        g_sum = g.sum()
        
        if g_sum < 1e-10:
            break
        
        g_norm = g / g_sum
        a_new = (1 - step) * a + step * g_norm
        a_new = project_simplex(a_new)
        
        score = compute_score_normalized(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        a = a_new
        
        if stall_count > 200:
            step = max(0.05, step * 0.7)
            rng = np.random.default_rng(int(time.monotonic() * 1000))
            a = best_a + rng.standard_normal(n) * 0.001
            a = project_simplex(a)
            prev_g = None
            stall_count = 0
    
    return best_a, best_score


def coordinate_descent(a0, time_limit, verbose=True):
    """Coordinate descent: try moving mass between pairs of positions."""
    n = len(a0)
    a = a0.copy()
    a = project_simplex(a)
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        # Try a batch of random moves
        for _ in range(200):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            
            # Try moving mass from i to j (and vice versa)
            for direction in [1, -1]:
                for frac in [0.01, 0.05, 0.1, 0.2, 0.4]:
                    amt = frac * a[i] * direction
                    if a[i] + amt < 0:
                        continue
                    a_trial = a.copy()
                    a_trial[i] += amt
                    a_trial[j] -= amt
                    if a_trial[j] < 0:
                        continue
                    a_trial = np.maximum(a_trial, 0)
                    s = a_trial.sum()
                    if s < 1e-15:
                        continue
                    a_trial = a_trial / s
                    sc = compute_score_normalized(a_trial)
                    if sc < best_score - 1e-14:
                        best_score = sc
                        best_a = a_trial.copy()
                        a = a_trial
                        break
        
        # Periodic: perturb if stuck
        if (int((time.monotonic() - t0) * 10)) % 50 == 0:
            a = best_a + rng.standard_normal(n) * 0.003
            a = project_simplex(a)
    
    return best_a, best_score


def bussgang_iteration(a0, time_limit, tau=0.5, verbose=True):
    """Bussgang spectral flattening."""
    n = len(a0)
    a = a0.copy()
    a = project_simplex(a)
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()
    
    while time.monotonic() - t0 < time_limit:
        A = rfft(a, fft_size)
        power = np.abs(A) ** 2
        total_power = power.sum()
        target = total_power / fft_size
        ratio = np.where(power > 1e-30, target / power, 1.0)
        ratio = np.clip(ratio, 0.1, 10.0)
        A_new = A * (ratio ** tau)
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        score = compute_score_normalized(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    
    return best_a, best_score


def check_and_update(a, label, best_seq, best_score):
    """Evaluate using the actual evaluate_sequence function."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
