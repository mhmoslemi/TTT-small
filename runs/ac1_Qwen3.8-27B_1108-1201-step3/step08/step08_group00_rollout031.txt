            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver, warm_b=None):
    """Given fixed a (sum=1, nonneg), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    # Build constraint matrix: (T @ b)[k] = sum_j a[j] * b[k-j] for valid k-j
    # T is (2n-1) x n
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed

    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=500000):
    """Nesterov-accelerated LP alternating optimization."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, _ = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if (it + 1) % 300 == 0:
            if best_score < prev_best - 1e-8:
                prev_best = best_score
            else:
                stall_count += 1
            if stall_count >= 15:
                rng = np.random.default_rng(it * 7919)
                noise_scale = 0.03
                noise = rng.random(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0
            if (it + 1) % 3000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NLP iter={it+1} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def logsumexp_refine(a0, time_limit, T_start=500.0, T_end=0.3, maxiter=20000):
    """L-BFGS with LogSumExp smoothing and geometric temperature annealing."""
    from scipy.optimize import minimize
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_z = z.copy()
    best_val = float('inf')

    n_stages = 9
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(30, maxiter // n_stages)

    for stage_idx, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time.limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func(z_val, T_val=stage_T):
            a_val = softmax_from_z(z_val)
            b_val = fast_convolve(a_val)
            Tb = T_val * b_val
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp

        def jac(z_val, T_val=stage_T):
            a_val = softmax_from_z(z_val)
            b_val = fast_convolve(a_val)
            Tb = T_val * b_val
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            # Gradient of sum_k w[k] * c[k] w.r.t. a
            grad_a = np.zeros(n)
            for k in range(len(b_val)):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += w[k] * a_val[k - idx]
            # Chain rule: d/dz logsumexp = a * (grad_a - <grad_a, a>)
            S = np.dot(grad_a, a_val)
            return a_val * (grad_a - S) * (2.0 * n / stage_T)

        try:
            res = minimize(
                fun=func,
                x0=z,
                jac=jac,
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            score = 2.0 * n * fast_convolve(a_check).max()
            if score < best_val:
                best_val = score
                best_z = z.copy()
        except Exception:
            pass

        elapsed = time.monotonic() - t0
        if stage_idx % 3 == 0:
            print(f"    LSE stage={stage_idx} T={stage_T:.1f} ({elapsed:.0f}s): best={best_val:.6f}")

    return softmax_from_z(best_z)


def peak_balance_optimize(a0, time_limit, n_steps=50000):
    """Direct peak balancing via coordinate moves and gradient steps."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.02
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        max_c = c.max()
        # Find all near-peak locations
        threshold = max_c - 0.05 * max_c
        active = np.where(c >= threshold)[0]

        # Compute subgradient: average gradient over active peaks
        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                k_minus_j = k - j
                if 0 <= k_minus_j < n:
                    grad[j] += a[k_minus_j]
        grad /= len(active)

        # Projected gradient step on simplex
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score_new = 2.0 * n * fast_convolve(a_new).max()

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive learning rate
        if no_improve >= 50:
            lr *= 0.8
            no_improve = 0
        lr = max(0.005, min(lr * 0.999, 0.05))

        # Periodic coordinate descent on top contributors
        if step % 200 == 199:
            c_curr = fast_convolve(a)
            peak_k = int(np.argmax(c_curr))
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_i = idx[np.argmax(contribs)]
            if a[top_i] > 1e-10:
                for factor in [0.97, 0.95, 0.93]:
                    a_trial = a.copy()
                    diff = a_trial[top_i] * (1 - factor)
                    a_trial[top_i] *= factor
                    mask = np.ones(n, dtype=bool)
                    mask[top_i] = False
                    a_trial[mask] += diff * a_trial[mask] / max(a_trial[mask].sum(), 1e-15)
                    a_trial = a_trial / a_trial.sum()
                    sc = 2.0 * n * fast_convolve(a_trial).max()
                    if sc < best_score - 1e-12:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        break

        if step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakBal {elapsed:.0f}s: score={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score


def dual_guided_step(a0, dual_dist, lr):
    """Take a gradient step guided by the dual distribution from LP."""
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64)
    # Gradient of sum_k p[k] * c[k] where c = a*a
    # d/d a[i] of c[k] = 2 * a[k-i] (if 0 <= k-i < n)
    grad = np.zeros(n)
    for k in range(len(dual_dist)):
        pk = dual_dist[k]
        if pk < 1e-12:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += pk * a[k - idx]
    grad *= 2.0
    # Projected step
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a0.copy()
    return a_new / s


def lp_with_dual_tracking(a0, time_limit, solver, n_max_iters=30000):
    """LP alternating that also tracks dual variables for gradient guidance."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    y = a.copy()
    dual_p = np.ones(2 * n - 1) / (2 * n - 1)  # uniform dual init
    prev_best = best_score
    stall = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, _ = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        # Update dual: uniform over active constraints of the LP
        # The LP constraints are (T_y @ b)[k] <= M. Active where equality holds.
        c_yb = convolve_with(y, b_new)
        M_val = c_yb.max()
        active_tol = 0.01 * M_val
        active_mask = c_yb >= (M_val - active_tol)
        dual_p = np.zeros(len(c_yb))
        dual_p[active_mask] = 1.0
        dual_p /= dual_p.sum()

        # Nesterov update
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        # Also try a dual-guided step
        lr_guide = 0.01 / (1 + it * 0.01)
        a_guide = dual_guided_step(a, dual_p, lr_guide)
        score_guide = 2.0 * n * fast_convolve(a_guide).max()

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        if score_guide < best_score:
            best_score = score_guide
            best_a = a_guide.copy()
            a = a_guide
            stall = 0

        if (it + 1) % 500 == 0:
            if best_score < prev_best - 1e-8:
                prev_best = best_score
            else:
                stall += 1
            if stall >= 20:
                rng = np.random.default_rng(it * 31337)
                y = y + 0.02 * rng.random(n)
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall = 0
            if (it + 1) % 5000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LPD iter={it+1} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def structured_init(n, rng):
    """Create diverse structured initializations."""
    choice = rng.integers(0, 7)
    x = np.linspace(0, 1, n)
    if choice == 0:
        # U-shaped
        a = (x - 0.5) ** 2 + 0.01
    elif choice == 1:
        # W-shaped
        a = (2 * x - 1) ** 4 + 0.01
    elif choice == 2:
        # Gaussian
        a = np.exp(-8 * (x - 0.5) ** 2)
    elif choice == 3:
        # Cosine
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = a ** 2 + 0.01
    elif choice == 4:
        # Double dip
        a = np.exp(-30 * (x - 0.25) ** 2) + np.exp(-30 * (x - 0.75) ** 2) + 0.01
    elif choice == 5:
        # Uniform
        a = np.ones(n)
    else:
        # Power U
        a = ((x - 0.5) * 2) ** 6 + 0.01

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)

            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")

        if s < 0.01:

        try_update(a_pf, "final_pb")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_lpf = min(rem * 0.5, 30.0)
        a_lpf, _ = nesterov_lp_refine(best_seq, t_lpf, solver, n_max_iters=10000)
        try_update(a_lpf, "final_lp")

    # Final check
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
