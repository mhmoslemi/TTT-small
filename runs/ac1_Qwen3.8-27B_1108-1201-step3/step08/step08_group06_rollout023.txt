def mirror_descend_step(a, top_k, eta):
    """One mirror descent step with top-k peak weighting."""
    n = len(a)
    conv = fast_convolve(a)
    
    # Find top-k peaks
    if top_k >= len(conv):
        peak_idx = np.arange(len(conv))
        peak_vals = conv
    else:
        # Partial sort for efficiency
        if top_k <= 20:
            peak_idx = np.argsort(conv)[::-1][:top_k]
        else:
            peak_idx = np.argpartition(conv, -top_k)[::-1][:top_k]
        peak_vals = conv[peak_idx]
    
    # Weight peaks by excess over median
    med = np.median(conv)
    weights = np.maximum(peak_vals - med, 0)
    wsum = weights.sum()
    if wsum < 1e-15:
        weights = np.ones(top_k) / top_k
    else:
        weights = weights / wsum
    
    # Build weight vector
    w = np.zeros(2 * n - 1)
    for k, wt in zip(peak_idx, weights):
        w[int(k)] = wt
    
    # Compute cross-correlation v[j] = sum_k w[k] * a[k-j]
    v = cross_corr_w_a(w, a)
    
    # Responsibility: resp[j] = a[j] * v[j]
    resp = a * v
    
    # Normalize resp for stable exponentiation
    resp_mean = resp.mean()
    if resp_mean < 1e-15:
        return a.copy()
    resp_norm = resp / resp_mean
    
    # Multiplicative update
    factor = np.exp(-eta * resp_norm)
    a_new = a * factor
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    a_new = a_new / s
    return a_new

def build_C(f):
    """Build constraint matrix C where C[k,j] = f[k-j] if 0<=k-j<n else 0."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = f
    return C


def greedy_lp(f, M):
    """Greedy approximation to the LP: build g one coordinate at a time."""
    n = len(f)
    g = np.zeros(n)
    conv_vals = np.zeros(2 * n - 1)
    
    for j in range(n):
        k_start = j
        k_end = j + n - 1
        slack = M - conv_vals[k_start:k_end + 1]
        f_slice = f  # f[k-j] for k in [j, j+n-1] is f[0], f[1], ..., f[n-1]
        valid = f_slice > 1e-15
        if not valid.any():
            continue
        max_gj = np.inf
        for idx in np.where(valid)[0]:
            if slack[idx] >= 0:
                cand = slack[idx] / f_slice[idx]
                if cand < max_gj:
                    max_gj = cand
        max_gj = max(0, max_gj)
        g[j] = max_gj
        conv_vals[k_start:k_end + 1] += f_slice * max_gj
    
    return g

def smoothed_obj_grad(z, T, n):
    """Compute smoothed objective and gradient in log-space (softmax parameterization)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    w = np.exp(shifted - max_shifted)
    w = w / w.sum()
    
    log_sum_exp = max_shifted + np.log(w.sum())
    obj = (2.0 * n) * T * log_sum_exp
    
    # Gradient w.r.t. a
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
    
    # Project to log-space
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return obj, (2.0 * n) * grad_z

    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score_norm(a0)
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        obj, _ = smoothed_obj_grad(z, T, n)
        return obj
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        _, g = smoothed_obj_grad(z, T, n)
        return g
    
    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                     'maxfun': maxiter * 30, 'disp': False}
        )
        z_out = res.x
        z_out = z_out - z_out.max()
        a_out = np.exp(z_out)
        a_out = a_out / a_out.sum()
        score = compute_score_norm(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception:
        pass
    
    return best_a, best_score


def coordinate_descent(a0, time_limit, verbose=True):
    """Coordinate descent with golden-section search for optimal transfer."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score_norm(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    iters = 0
    no_improve_count = 0
    gr = (np.sqrt(5) - 1) / 2
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        
        # Generate pairs: mix of random and peak-focused
        n_pairs = min(n, 300)
        pairs = set()
        
        # Random pairs
        for _ in range(n_pairs // 2):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                pairs.add((min(i, j), max(i, j)))
        
        # Peak-focused pairs
        conv = fast_convolve(a)
        top_peak_idx = int(np.argmax(conv))
        js = max(0, top_peak_idx - n + 1)
        je = min(n - 1, top_peak_idx)
        for _ in range(n_pairs // 4):
            i = rng.integers(js, je + 1)
            j = rng.integers(js, je + 1)
            if i != j:
                pairs.add((min(i, j), max(i, j)))
        
        # Edge pairs
        edge = max(2, n // 10)
        for _ in range(n_pairs // 4):
            i = rng.integers(0, edge)
            j = rng.integers(n - edge, n)
            if i != j:
                pairs.add((min(i, j), max(i, j)))
        
        for (i, j) in list(pairs):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14 and a[j] < 1e-14:
                continue
            
            t_lo = -a[j] + 1e-15
            t_hi = a[i] - 1e-15
            if t_hi <= t_lo:
                continue
            
            x1 = t_hi - gr * (t_hi - t_lo)
            x2 = t_lo + gr * (t_hi - t_lo)
            
            def eval_t(t, i=i, j=j):
                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t
                if a_trial[i] < -1e-15 or a_trial[j] < -1e-15:
                    return 1e10
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    return 1e10
                a_trial = a_trial / ss
                return compute_score_norm(a_trial)
            
            f1 = eval_t(x1)
            f2 = eval_t(x2)
            
            for _ in range(25):
                if time.monotonic() - t0 > time_limit:
                    break
                if f1 < f2:
                    t_hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = t_hi - gr * (t_hi - t_lo)
                    f1 = eval_t(x1)
                else:
                    t_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = t_lo + gr * (t_hi - t_lo)
                    f2 = eval_t(x2)
            
            t_opt = (t_lo + t_hi) / 2
            f_opt = min(f1, f2)
            
            if f_opt < best_score - 1e-13:
                a[i] -= t_opt
                a[j] += t_opt
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = compute_score_norm(a)
                best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve_count += 1
            if no_improve_count > 10:
                noise = rng.standard_normal(n) * 0.001
                a = best_a + noise
                a = np.maximum(a, 0)
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                else:
                    a = best_a.copy()
                no_improve_count = 0
        else:
            no_improve_count = 0
        
        if verbose and iters % 20 == 0:
            print(f"    CD iter={iters}, score={best_score:.8f}")
    
    return best_a, best_score


def peak_targeting(a0, time_limit, verbose=True):
    """Directly target the top peaks by redistributing mass away from them."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score_norm(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(77)
    
    iters = 0
    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        
        # Find top 5 peaks
        top_k = min(5, len(conv))
        if top_k <= 20:
            peak_indices = np.argsort(conv)[::-1][:top_k]
        else:
            peak_indices = np.argpartition(conv, -top_k)[::-1][:top_k]
        
        improved = False
        for pk in peak_indices:
            if time.monotonic() - t0 > time_limit:
                break
            
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            
            if len(idx) == 0:
                continue
            
            # Contributions to this peak
            contribs = np.zeros(len(idx))
            for ii, j in enumerate(idx):
                k2 = pk - j
                if 0 <= k2 < n:
                    contribs[ii] = a[j] * a[k2]
            
            # Try reducing the top contributing positions
            top_contrib = np.argsort(contribs)[::-1][:5]
            
            for ci in top_contrib:
                if contribs[ci] < 1e-14:
                    continue
                i_idx = int(idx[ci])
                if a[i_idx] < 1e-14:
                    continue
                
                # Try different reduction fractions
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    delta = a[i_idx] * frac
                    if delta < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[i_idx] -= delta
                    # Redistribute proportionally to others
                    mask = np.ones(n, dtype=bool)
                    mask[i_idx] = False
                    others = a_trial[mask]
                    os_sum = others.sum()
                    if os_sum < 1e-15:
                        continue
                    a_trial[mask] = others + delta * others / os_sum
                    a_trial = np.maximum(a_trial, 0)
                    s = a_trial.sum()
                    if s < 1e-15:
                        continue
                    a_trial = a_trial / s
                    sc = compute_score_norm(a_trial)
                    if sc < best_score - 1e-13:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved =
                        break
                if improved:
                    break
            
            if not improved:
                # Try adding to a position far from this peak
                noncontrib = np.setdiff1d(np.arange(n), idx)
                if len(noncontrib) > 0:
                    j_idx = noncontrib[rng.integers(0, len(noncontrib))]
                    for frac in [0.01, 0.02, 0.05]:
                        delta = a[j_idx] * frac
                        if delta < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[j_idx] -= delta
                        mask = np.ones(n, dtype=bool)
                        mask[j_idx] = False
                        others = a_trial[mask]
                        os_sum = others.sum()
                        if os_sum < 1e-15:
                            continue
                        a_trial[mask] = others + delta * others / os_sum
                        a_trial = np.maximum(a_trial, 0)
                        s = a_trial.sum()
                        if s < 1e-15:
                            continue
                        a_trial = a_trial / s
                        sc = compute_score_norm(a_trial)
                        if sc < best_score - 1e-13:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
        
        if not improved and iters % 30 == 0:
            a = best_a + rng.standard_normal(n) * 0.003
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            else:
                a = best_a.copy()
        
        if verbose and iters % 20 == 0:
            print(f"    Peak iter={iters}, score={best_score:.8f}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def zero_small_coords(a, threshold):
    """Zero out coordinates below threshold if it improves score."""
    n = len(a)
    mask = a < threshold
    if not mask.any():
        return a.copy()
    a_trial = a.copy()
    a_trial[mask] = 0
    s = a_trial.sum()
    if s < 1e-15:
        return a.copy()
    a_trial = a_trial / s
    return a_trial


def check_and_update(a, best_seq, best_score, verbose=True):
    """Evaluate a candidate and return updated best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score

    except Exception as e:

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, time={total_time:.0f}s)")
    return [float(x) for x in best_seq]
