- This is like the "overlap" of a with its reverse
- If a is a "shifted" version of a smooth function, the overlap reduces

Consider a[i] = f(i/n + δ) for some shift δ. As δ moves away from 0 or 1 (which give symmetric shapes), the overlap at lag n-1 decreases.

For the extreme case, a[i] = f(i/n + 0.5) would be anti-symmetric (a[i] = f((n-1-i)/n + 0.5) = f(1 - i/n) which is the reflection). If f is linear (f(x) = x), then a[i] = i/n + 0.5, and a[n-1-i] = (n-1-i)/n + 0.5. The product a[i]*a[n-1-i] = (i/n + 0.5)*((n-1-i)/n + 0.5).

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, n=fft_size)
    return irfft(A * A, n=fft_size)[:size]

def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_max = n
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def get_lp_solver():
    """Find a working LP solver."""
    try:

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_result = solver_name

    try:
        _lp_solver_result = cp.ECOS

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (sum=1, nonneg), find b (sum=1, nonneg) minimizing max_k conv(a,b)[k]."""
    try:
        import cvxpy as cp
    except ImportError:
        return None, None
    if solver is None:
        solver = get_lp_solver()
    if solver is None:
        return None, None
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def init_anti_palindromic(n, rng, skew=0.1):
    """
    Initialize with an anti-palindromic shape.
    The idea: a[i] is larger on one side to reduce the middle autocorrelation.
    """
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    # Base: gentle slope (anti-palindromic component)
    base = 1.0 + skew * (2.0 * t - 1.0)
    # Add some curvature
    curvature = rng.uniform(0.0, 0.3)
    base += curvature * np.cos(np.pi * t)
    a = np.maximum(base, 0.1)
    a = a / a.sum()
    return a

def init_fourier_flat(n, rng):
    """
    Initialize by designing a sequence with approximately flat power spectrum.
    Start with random phase in frequency domain, then iterate.
    """
    # Start with a random sequence
    a = rng.random(n)
    a = a / a.sum()
    
    # A few iterations of IAA to flatten the spectrum
    for _ in range(50):
        A = fft(a)
        mag = np.abs(A)
        phase = np.angle(A)
        # Compress the magnitude
        target = np.mean(mag)
        mag_new = np.power(mag, 0.7)  # Compression
        mag_new = mag_new / mag_new.mean() * target
        A_new = mag_new * np.exp(1j * phase)
        a = np.real(ifft(A_new))
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s
    
    return a

def iam_iterate(a, n_iter=200, compression=0.5, n=0):
    """
    Iterative Amplitude Modification (IAM) for flattening the autocorrelation.
    Works in the Fourier domain to make |DFT(a)|^2 as flat as possible.
    """
    if n == 0:
        n = len(a)
    a = a.copy()
    
    for it in range(n_iter):
        A = fft(a)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Compress magnitude to flatten the power spectrum
        # mag^2 is the power spectrum; we want it flat
        # Apply compression: mag_new = mag^alpha normalized
        mag_power = np.power(mag, 2.0 * compression)  # = |A|^(2*compression)
        # Normalize to preserve energy
        mag_power = mag_power / np.sqrt(np.mean(mag_power))
        
        A_new = mag_power * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))
        
        # Project to non-negative simplex
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Check improvement
        conv = fast_convolve(a_new)
        conv_old = fast_convolve(a)
        if conv.max() < conv_old.max():
            a = a_new
        else:
            # Damped update
            a = 0.5 * a + 0.5 * a_new
            a = np.maximum(a, 0)
            s = a.sum()
            if s < 1e-15:
                a = np.ones(n) / n
            else:
                a = a / s
    
    return a

def subgradient_method(a0, time_limit, lr_init=0.01, verbose=False):
    """
    Subgradient projection method for minimizing max(conv(a,a)) 
    subject to a >= 0, sum(a) = 1.
    """
    n = len(a0)
    a = np.maximum(a0, 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    lr = lr_init
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        score = 2.0 * n * M
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 50:
            lr *= 0.85
            no_improve = 0
            if lr < 1e-8:
                break
        
        # Find active set (indices where conv is close to max)
        threshold = M * 0.95
        active = np.where(conv >= threshold)[0]
        
        if len(active) == 0:
            active = np.argsort(conv)[-5:]
        
        # Compute subgradient
        # For each active lag k, the gradient contribution to a[j] is:
        #   a[k-j] + a[j] (if both in [0,n))
        grad = np.zeros(n)
        w = np.ones(len(active))
        w = w / w.sum()
        
        for idx, k in enumerate(active):
            k = int(k)
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            for j in range(i_min, i_max + 1):
                kj = k - j
                if 0 <= kj < n:
                    grad[j] += w[idx] * (a[kj] + a[j])
                else:
                    grad[j] += w[idx] * a[j]
        
        # Update: a <- project(a - lr * grad)
        a_new = a - lr * grad
        a_new = simplex_project(a_new)
        
        lr *= 0.999
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad it={it} ({elapsed:.0f}s): best={best_score:.6f} lr={lr:.6f}")
    
    return best_a, best_score

def fast_coord_descent(a0, time_limit, n_trials=10, verbose=False, label=""):
    """
    Fast coordinate descent using O(n) sensitivity analysis per coordinate.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve_sweeps = 0
    
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        
        for i in range(n):
            if a[i] < 1e-15:
                continue
            
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            
            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)
            
            affected = conv[k_start:k_end + 1]
            a_shifted = a  # a[k-i] for k = i, ..., i+n-1 → a[0], a[1], ..., a[n-1]
            
            best_delta = 0.0
            best_new_max = M
            
            for trial in range(n_trials):
                if trial == 0:
                    delta = -0.3 * a[i]
                elif trial == 1:
                    delta = -0.15 * a[i]
                elif trial == 2:
                    delta = -0.05 * a[i]
                elif trial == 3:
                    delta = 0.05 * a[i]
                elif trial == 4:
                    delta = 0.15 * a[i]
                elif trial == 5:
                    delta = 0.3 * a[i]
                elif trial == 6:
                    delta = -0.01 * a[i]
                elif trial == 7:
                    delta = 0.01 * a[i]
                elif trial == 8:
                    delta = -0.5 * a[i]
                else:
                    delta = 0.5 * a[i]
                
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue
                
                new_affected = affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx_2i] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
                
                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_delta = delta
            
            if best_delta != 0.0 and best_new_max < M - 1e-16:
                a[i] += best_delta
                a[i] = max(a[i], 0.0)
                
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx_2i] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > 30:
                break
        else:
            no_improve_sweeps = 0
        
        if verbose and sweeps % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score, sweeps

def simulated_annealing(a0, time_limit, t_init=0.05, t_final=0.001, verbose=False):
    """
    Simulated annealing to escape local minima.
    """
    n = len(a0)
    a = np.maximum(a0, 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    total_time = time_limit
    t = t_init
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = elapsed / total_time
        t = t_init * (t_final / t_init) ** progress
        
        # Random perturbation
        n_perturb = max(1, int(n * 0.01))
        idx = np.random.choice(n, size=n_perturb, replace=False)
        delta = np.random.standard_normal(n_perturb) * t
        a_new = a.copy()
        a_new[idx] += delta
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s
        
        score_new = 2.0 * n * fast_convolve(a_new).max()
        
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
        
        # Accept with probability exp(-(score_new - current)/t)
        current_score = 2.0 * n * fast_convolve(a).max()
        if score_new < current_score or np.random.random() < np.exp(-(score_new - current_score) / max(t, 1e-10)):
            a = a_new
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    SA ({elapsed:.0f}s): best={best_score:.6f} t={t:.6f}")
    
    return best_a, best_score

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

            break
        
        # Multiple initializations
        inits = []
        
        # Anti-palindromic
        for skew in [0.05, 0.1, 0.15, 0.2]:
            inits.append(('apal_%s' % str(skew), init_anti_palindromic(n, rng, skew=skew)))
        
        # Fourier-flat
        inits.append(('fourier', init_fourier_flat(n, rng)))
        
        # Parent-based
        if parent is not None:
            inits.append(('parent', interpolate_seq(parent, n)))
            # Noisy parent
            noisy = interpolate_seq(parent, n) + rng.standard_normal(n) * 0.05
            noisy = np.maximum(noisy, 0)
            noisy = noisy / noisy.sum()
            inits.append(('parent_noise', noisy))
        
        # Uniform
        inits.append(('uniform', np.ones(n) / n))
        
        for name, init_a in inits:
            # Apply IAM to flatten the spectrum
            t_alloc = min(30.0, 300.0 - (time.monotonic() - t1_start))
            if t_alloc < 2:
                break
            a_iam = iam_iterate(init_a, n_iter=min(100, int(t_alloc * 5)))
            
            # Quick CD to polish
            t_cd = min(10.0, 300.0 - (time.monotonic() - t1_start))
            if t_cd < 1:
                break
            a_opt, s_opt, _ = fast_coord_descent(a_iam, t_cd, n_trials=8, label=f"p1_{name}")
            candidates.append((s_opt, a_opt, n, name))
        
        # Sort and keep top
        candidates.sort(key=lambda x: x[0])
    
    print(f"  Generated {len(candidates)} candidates")
    if candidates:
        print(f"  Top: {candidates[0][0]:.6f}")
    
    for s_c, a_c, n_c, name_c in candidates[:5]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_{name_c}_n{n_c}")
    
    # === Phase 2: Intensive coordinate descent on best ===
    print(f"\n=== Phase 2: Intensive CD (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        t2_start = time.monotonic()
        
        current = best_seq.copy()
        for rnd in range(4):
            rem = deadline - time.monotonic() - 600
            if rem < 30:
                break
            t_alloc = min(rem * 0.4, 200.0)
            print(f"  FCD round {rnd+1}, n={n_best}, budget={t_alloc:.0f}s")
            a_opt, s_opt, sweeps = fast_coord_descent(current, t_alloc, n_trials=12,
                                                       verbose=True, label=f"ph2r{rnd}")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"ph2_fcd_r{rnd}")
            current = best_seq.copy()
    
    # === Phase 3: Subgradient method ===
    print(f"\n=== Phase 3: Subgradient (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_sub = min(rem * 0.5, 150.0)
            print(f"  SubGrad n={len(best_seq)}, budget={t_sub:.0f}s")
            a_sub, s_sub = subgradient_method(best_seq, t_sub, lr_init=0.005, verbose=True)
            best_seq, best_score = check_and_update(a_sub, best_seq, best_score, "subgrad")
            
            # CD after subgradient
            rem = deadline - time.monotonic() - 350
            if rem > 20:
                a_cd, s_cd, _ = fast_coord_descent(a_sub, min(rem * 0.5, 60.0), n_trials=10, label="sub_cd")
                best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "sub_cd")
    
    # === Phase 4: Simulated annealing ===
    print(f"\n=== Phase 4: Simulated Annealing (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_sa = min(rem * 0.4, 100.0)
            print(f"  SA n={len(best_seq)}, budget={t_sa:.0f}s")
            a_sa, s_sa = simulated_annealing(best_seq, t_sa, t_init=0.03, verbose=True)
            best_seq, best_score = check_and_update(a_sa, best_seq, best_score, "sa")
            
            # CD after SA
            rem = deadline - time.monotonic() - 250
            if rem > 20:
                a_cd, s_cd, _ = fast_coord_descent(a_sa, min(rem * 0.5, 60.0), n_trials=10, label="sa_cd")
                best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "sa_cd")
    
    # === Phase 5: LP Fictitious Play ===
    print(f"\n=== Phase 5: LP Fictitious Play (best={best_score:.6f}) ===")
    if best_seq is not None and solver is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 200
        if rem > 60:
            t_fp = min(rem * 0.5, 150.0)
            print(f"  FP n={n_best}, budget={t_fp:.0f}s")
            a_fp, score_fp, iters = fictitious_play(
                n_best, t_fp, initial=best_seq, solver=solver, verbose=True
            )
            if a_fp is not None:
                best_seq, best_score = check_and_update(a_fp, best_seq, best_score, "fp")
                # CD after FP
                rem = deadline - time.monotonic() - 150
                if rem > 20:
                    a_cd, s_cd, _ = fast_coord_descent(a_fp, min(rem * 0.5, 60.0), n_trials=10, label="fp_cd")
                    best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "fp_cd")
    
    # === Phase 6: IAM on current best (different compression) ===
    print(f"\n=== Phase 6: IAM Refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            # Try different compression levels
            for comp in [0.3, 0.5, 0.7]:
                rem = deadline - time.monotonic() - 100
                if rem < 10:
                    break
                a_iam = iam_iterate(best_seq, n_iter=min(50, int(rem * 2)), compression=comp)
                a_cd, s_cd, _ = fast_coord_descent(a_iam, min(rem * 0.3, 30.0), n_trials=8, label=f"iam_{comp}")
                best_seq, best_score = check_and_update(a_cd, best_seq, best_score, f"iam_{comp}")
    
    # === Phase 7: Try different length ===
    print(f"\n=== Phase 7: Alt Length (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 80
        if rem > 40:
            for alt_n in [n_best + 128, n_best - 128, n_best + 256]:
                if alt_n < 256 or alt_n > 2560 or alt_n == n_best:
                    continue
                rem = deadline - time.monotonic() - 60
