def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver_cached = None


def get_lp_solver():
    """Find a working LP solver with tight tolerance."""
    global _lp_solver_cached
    if _lp_solver_cached is not None:
        return _lp_solver_cached
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, eps=1e-9, rel tol=1e-9)
    except Exception:
        try:
            prob.solve(solver=solver)
        except Exception:
            pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))

        if b_new is None:

        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

def lbgfs_refine(a0, time_limit, T_start=100.0, T_end=5.0, maxiter=5000):
    """L-BFGS with LogSumExp smoothing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def make_func(T_val):
        def func(z):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    n_stages = 8
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(30, maxiter // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_monotonic_remaining(t0, time_limit)
        if remaining < 3:
            break
        try:
            res = scipy_minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def time_monotonic_remaining(t0, limit):
    return limit - (time.monotonic() - t0)


def active_set_kkt_refine(a0, time_limit, n_iters=500):
    """
    Refine using the KKT conditions with a fixed active set.
    At optimum, for active set S: the "averaged shift" of a is constant on support.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        c_max = c.max()

        # Identify active set: top peaks within 0.5% of max
        threshold = c_max * 0.995
        active = np.where(c >= threshold)[0]
        # Also include the exact argmax
        active = np.unique(np.append(active, np.argmax(c)))

        if len(active) < 1:
            active = np.array([int(np.argmax(c))])
        if len(active) > 50:
            # Keep only top 50
            top_idx = np.argsort(c)[::-1][:50]
            active = top_idx

        # Compute the "effective potential": K[i] = (1/|S|) sum_{k in S} a[k-i]
        K = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            K[idx] += a[k - idx]
        K /= len(active)

        # Find support: where K is above threshold
        K_max = K.max()
        if K_max < 1e-30:
            break
        support_threshold = K_max * 0.5
        support = np.where(K >= support_threshold)[0]

        if len(support) < n // 4:
            # Keep at least n/4 coordinates
            support = np.argsort(K)[::-1][:n // 2]

        # New a: proportional to K on support, zero elsewhere
        a_new = np.zeros(n)
        a_new[support] = K[support]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        # Damped update
        t = 0.3
        a = (1 - t) * a + t * a_new
        a = np.maximum(a, 0)
        a = a / a.sum()

        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()

        if it % 50 == 0:
            print(f"    KKT refine iter={it}: score={best_score:.8f}, |S|={len(active)}, |supp|={len(support)}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create structured initial sequence (asymmetric)."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        # Asymmetric: skewed Gaussian
        a = np.exp(-4 * (x - 0.4) ** 2) + 0.5 * np.exp(-8 * (x - 0.7) ** 2)
    elif choice == 1:
        # Power law (asymmetric)
        a = (x + 0.05) ** 2 + 0.3 * (1 - x + 0.05) ** 3
    elif choice == 2:
        # Cubic with shift
        a = np.sin(np.pi * x ** 0.7) ** 2
    elif choice == 3:
        # Asymmetric raised cosine
        a = 0.5 + 0.5 * np.cos(2 * np.pi * (x - 0.1))
        a = np.maximum(a, 0) ** 1.5
    elif choice == 4:
        # Two-bump asymmetric
        a = np.exp(-9 * (x - 0.3) ** 2) + 0.7 * np.exp(-12 * (x - 0.65) ** 2)
    else:
        # Beta-like distribution
        a = x ** 1.5 * (1 - x) ** 0.8
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def precise_line_search_coord(a, c, i, n):
    """
    For coordinate i, find the best 2-coordinate move.
    Returns (best_j, best_delta, best_score) or None.
    """
    a_i = a[i]
    if a_i < 1e-12:
        return None

    # The affected k range for changing a[i]
    k_lo = i
    k_hi = min(i + n - 1, 2 * n - 2)

    # For a target max M, the constraint on delta (new a[i] = a[i] + delta,
    # new a[j] = a[j] - delta) is:
    # For k in [k_lo, k_hi]: c[k] + 2*delta*a[k-i] (if k != 2i) or c[k] + (delta*(2*a[i]+delta)) (if k==2i) <= M
    # For k in [j, j+n-1]: c[k] - 2*delta*a[k-j] (if k != 2j) or c[k] + (-delta*(2*a[j]-delta)) (if k==2j) <= M

    # This is complex. Let's just do a simple approach:
    # Try several fractions and pick the best.
    best_frac = 0
    best_new_max = c.max()

    for frac in np.concatenate([np.linspace(0.001, 0.1, 20), np.linspace(0.1, 0.5, 10)]):
        d = a_i * frac
        # Compute new max
        new_max = 0.0
        # Check affected positions
        for target_j in [0, n - 1, n // 4, 3 * n // 4, rng_int(n)]:
            if target_j == i:
                target_j = (i + 1) % n
            if a[target_j] < d:
                continue
            # Compute new c max
            max_c = c.max()
            # Update for position i (increase by d)
            for k in range(k_lo, k_hi + 1):
                ki = k - i
                if 0 <= ki < n:
                    if ki == i:
                        new_c = c[k] + d * (2 * a_i + d)
                    elif ki == target_j:
                        old = 2 * a_i * a[target_j]
                        new = 2 * (a_i + d) * (a[target_j] - d)
                        new_c = c[k] + (new - old)
                    else:
                        new_c = c[k] + 2 * d * a[ki]
                    if new_c > max_c:
                        max_c = new_c
            # Update for position target_j (decrease by d)
            k_lo_j = target_j
            k_hi_j = min(target_j + n - 1, 2 * n - 2)
            for k in range(k_lo_j, k_hi_j + 1):
                kj = k - target_j
                if 0 <= kj < n:
                    if kj == target_j:
                        new_c = c[k] + (-d) * (2 * a[target_j] - d)
                    elif kj == i:
                        pass  # already handled
                    else:
                        new_c = c[k] - 2 * d * a[kj]
                    if new_c > max_c:
                        max_c = new_c

            if max_c < best_new_max:
                best_new_max = max_c
                best_frac = frac

    if best_frac > 0:
        return best_frac
    return None


def rng_int(n):
    """Quick random integer (using a simple LCG to avoid rng overhead)."""
    global _rng_state
    if not hasattr(rng_int, '_state'):
        rng_int._state = 123456789
    rng_int._state = (rng_int._state * 1103515245 + 12345) % (2 ** 31)
    return rng_int._state % n
