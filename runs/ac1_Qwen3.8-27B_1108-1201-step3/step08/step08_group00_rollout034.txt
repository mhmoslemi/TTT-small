"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-start L-BFGS with log-sum-exp smoothing and aggressive cooling
2. Subgradient peeling with Nesterov acceleration

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def lse_objective(z, T, n):
    """Log-sum-exp smoothed objective in log-domain."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    return (2.0 * n / T) * log_sum_exp


def lse_gradient(z, T, n):
    """Gradient of log-sum-exp smoothed objective."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    
    S = np.dot(grad_a, a)
    return a * (grad_a - S)


def lse_lbgfs(a0, time_limit, T_start=50.0, T_end=0.1, n_stages=10, maxiter_per_stage=2000):
    """L-BFGS with multi-stage log-sum-exp annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_z = z.copy()
    best_val = lse_objective(z, T_start, n)
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for idx, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        iters = max(50, min(maxiter_per_stage, int(remaining / 0.05)))
        
        try:
            res = minimize(
                fun=lambda z_: lse_objective(z_, stage_T, n),
                x0=z,
                jac=lambda z_: lse_gradient(z_, stage_T, n),
                method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters * 50, 'disp': False}
            )
            new_val = lse_objective(res.x, stage_T, n)
            if new_val < best_val or True:  # Always accept for continuity
                z = res.x
            # Track best in terms of actual max
            a_check = softmax_from_z(z)
            actual = 2.0 * n * fast_convolve(a_check).max()
            if idx % 3 == 0:
                a_best = softmax_from_z(best_z)
                actual_best = 2.0 * n * fast_convolve(a_best).max()
                if actual < actual_best:
                    best_z = z.copy()
        except Exception as e:
            pass
    
    return softmax_from_z(z), softmax_from_z(best_z)


def subgradient_nesterov(a0, time_limit, n_steps=100000):
    """Nesterov-accelerated subgradient method targeting the max peak."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for it in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(y)
        max_b = b.max()
        # Subgradient at the peak
        peak_k = int(np.argmax(b))
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        g = np.zeros(n)
        g[idx] = 2.0 * y[peak_k - idx]
        # Center for simplex
        S = np.dot(g, y)
        g = y * (g / y - S)  # tangent space projection (avoid division by zero)
        # Actually let's use proper projection
        g = g - np.dot(g, y) * y  # project onto tangent space of simplex
        
        alpha = 2.0 / (it + 2)
        z_y = z_from_a(y)
        z_y = z_y - alpha * g
        y_new = softmax_from_z(z_y)
        
        a_momentum = ((it + 1) * a + alpha * (it + 2) * y_new) / (it + 1 + alpha * (it + 2))
        a_momentum = np.maximum(a_momentum, 0)
        sa = a_momentum.sum()
        if sa < 1e-15:
            a_momentum = y_new
        else:
            a_momentum = a_momentum / sa
        
        # Evaluate
        for candidate in [y_new, a_momentum]:
            sc = 2.0 * n * fast_convolve(candidate).max()
            if sc < best_score:
                best_score = sc
                best_a = candidate.copy()
        
        a = a_momentum
        y = y_new
        
        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad iter={it+1} ({elapsed:.1f}s): best={best_score:.6f}")
    
    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, n_steps=50000):
    """Coordinate descent: for each peak-contributing coordinate, find optimal value."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]
        
        # Find coordinates contributing to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Sort by contribution
        top = np.argsort(contribs)[::-1][:20]
        
        improved = False
        for ci in rng.permutation(top):
            i = idx[ci]
            if a[i] < 1e-15:
                continue
            
            # Try various reduction factors
            for factor in [0.99, 0.97, 0.95, 0.92, 0.88, 0.83, 0.75]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-16:
                    continue
                
                a_trial = a.copy()
                a_trial[i] = new_val
                # Redistribute to non-contributing coordinates
                mask = np.ones(n, dtype=bool)
                # Don't redistribute to coordinates that contribute to this peak
                for j in idx:
                    if a[j] > 1e-15:
                        mask[j] = False
                # Actually redistribute to ALL other coordinates proportionally
                mask2 = np.ones(n, dtype=bool)
                mask2[i] = False
                others_sum = a_trial[mask2].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask2] += diff * a_trial[mask2] / others_sum
                a_trial = a_trial / a_trial.sum()
                
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            
            if improved:
                break
        
        if not improved:
            # Try small random perturbations
            for _ in range(3):
                a_trial = a.copy()
                perturbation = rng.standard_normal(n) * 0.02
                a_trial = a_trial + perturbation * a_trial
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial = a_trial / s
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved and step > 1000:
            # Check if we should stop
            if step % 5000 == 0:
                stalled = all(
                    fast_convolve(a).max() >= best_score * (1 - 1e-8)
                    for _ in range(1)
                )
                if stalled:
                    break
    
    return best_a, best_score


def multi_peak_equalization(a0, time_limit, n_steps=30000):
    """Try to equalize the top peaks of the convolution."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(456)
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        # Get top K peaks
        K = min(5, len(b))
        top_peaks = np.argsort(b)[::-1][:K]
        top_vals = b[top_peaks]
        
        # The gap between top peak and second peak
        if len(top_peaks) < 2:
            break
        gap = top_vals[0] - top_vals[1]
        
        if gap < 1e-10:
            # Peaks are already well-equalized, try to reduce overall level
            break
        
        # Find which coordinate most contributes to the top peak but not the second
        peak_k = top_peaks[0]
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs_top = a[idx] * a[peak_k - idx]
        
        # Reduce the top contributors
        top_contrib = np.argsort(contribs_top)[::-1][:5]
        
        improved = False
        for ci in top_contrib:
            i = idx[ci]
            if a[i] < 1e-14:
                continue
            for factor in [0.995, 0.99, 0.985, 0.98]:
                a_trial = a.copy()
                diff = a_trial[i] * (1 - factor)
                a_trial[i] *= factor
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] += diff * a_trial[mask] / others_sum
                a_trial = a_trial / a_trial.sum()
                
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create structured initial sequence designed for low autocorrelation peak."""
    choice = rng.integers(0, 8)
    if choice == 0:
        # Flat-top window (Blackman-Harris-like)

        a = 0.35875 - 0.48829 * np.cos(2*np.pi*x) + 0.14128 * np.cos(4*np.pi*x) - 0.01168 * np.cos(6*np.pi*x)
    elif choice == 1:
        # Raised cosine
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        # Gaussian
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 3:
        # Sinc-like (for low sidelobes)
        x = np.linspace(-2, 2, n)
        a = np.sinc(x) ** 2
        a = np.maximum(a, 0)
    elif choice == 4:
        # Kaiser window
        from scipy.special import i0
        beta = 8.6  # ~21dB sidelobe
        x = np.linspace(-1, 1, n)
        a = i0(beta * np.sqrt(1 - x**2)) / i0(beta)
    elif choice == 5:
        # Random with smoothing
        a = rng.random(n)
        # Smooth
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    elif choice == 6:
        # Bimodal (two peaks)
        x = np.linspace(0, 1, n)
        a = np.exp(-((x-0.25)/0.1)**2) + np.exp(-((x-0.75)/0.1)**2)
    else:
        # Flat with small perturbation
        a = np.ones(n) + 0.1 * rng.standard_normal(n)
        a = np.maximum(a, 0)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-round L-BFGS with log-sum-exp smoothing and fine-grained temperature annealing
2. Multi-peak-aware coordinate descent with smart redistribution
3. Subgradient Nesterov acceleration for escape from local minima
4. Multi-length exploration with interpolation
5. Perturbation-based restarts
Key insight: fine temperature schedule from low T (preserving parent structure) with many stages.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def lse_objective(z, T, n):
    """Log-sum-exp smoothed objective."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    return (2.0 * n / T) * log_sum_exp


def lse_gradient(z, T, n):
    """Gradient of log-sum-exp smoothed objective."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-14:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    return a * (grad_a - S)


def lbgfs_multistage(a0, time_limit, temps, maxiter_per_stage=2000):
    """L-BFGS with multi-stage log-sum-exp annealing given explicit temperature list."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()

    for idx, stage_T in enumerate(temps):
        if time.monotonic() - t0 >= time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        iters = max(30, min(maxiter_per_stage, int(remaining * 0.9 / max(len(temps) - idx, 1) / 0.02)))
        iters = max(30, min(iters, 5000))

        try:
            res = minimize(
                fun=lambda z_: lse_objective(z_, stage_T, n),
                x0=z,
                jac=lambda z_: lse_gradient(z_, stage_T, n),
                method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters * 60, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        a_cur = softmax_from_z(z)
        sc = 2.0 * n * fast_convolve(a_cur).max()
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
        if idx % 3 == 0:
            print(f"    stage {idx+1}/{len(temps)} T={stage_T:.3f}: score={best_score:.6f}")

    return best_a, softmax_from_z(z)


def subgradient_nesterov(a0, time_limit, n_steps=200000):
    """Nesterov-accelerated subgradient method."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(n_steps):
        if time.monotonic() - t0 >= time_limit:
            break

        b = fast_convolve(y)
        peak_k = int(np.argmax(b))
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        g = np.zeros(n)
        g[idx] = 2.0 * y[peak_k - idx]

        # Project gradient onto tangent space of simplex
        g = g - np.dot(g, y) * y

        alpha = 2.0 / (it + 2)
        z_y = z_from_a(y)
        z_y = z_y - alpha * g
        y_new = softmax_from_z(z_y)

        a_momentum = ((it + 1) * a + alpha * (it + 2) * y_new) / (it + 1 + alpha * (it + 2))
        a_momentum = np.maximum(a_momentum, 0)
        sa = a_momentum.sum()
        if sa < 1e-15:
            a_momentum = y_new
        else:
            a_momentum = a_momentum / sa

        for candidate in [y_new, a_momentum]:
            sc = 2.0 * n * fast_convolve(candidate).max()
            if sc < best_score:
                best_score = sc
                best_a = candidate.copy()

        a = a_momentum
        y = y_new

        if (it + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad iter={it+1} ({elapsed:.1f}s): best={best_score:.6f}")

    return best_a, best_score


def coord_descent_multipeak(a0, time_limit, n_steps=100000):

    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)

        multi_contrib = np.zeros(n)

def interpolate_sequence(f, new_n):
    """Interpolate to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        a_cd_fin, _ = coord_descent_multipeak(best_seq, min(rem * 0.4, 50.0), n_steps=50000)
        try_update(a_cd_fin, "final_cd")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
