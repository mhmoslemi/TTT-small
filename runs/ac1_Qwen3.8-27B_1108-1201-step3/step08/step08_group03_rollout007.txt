"""
Optimized search for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
- Refined cutting-plane LP with multi-solver fallback
- Adaptive active set with near-peak constraints
- Peak-directed subgradient polish
- Multi-length exploration with smooth interpolation
- Adaptive time management

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time
import sys


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_cross(f, g):
    """Compute cross-convolution f*g via FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_A_matrix(f, indices, n):
    """Build constraint matrix: row i encodes (f*g)[k_i] = A[i,:] @ g."""
    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp(f, n, M, active_idx):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)[k]<=M for k in active_idx.
    Returns g or None. Tries multiple solvers."""
    K = len(active_idx)
    if K == 0:
        return None

    A_mat = build_A_matrix(f, active_idx, n)
    g = cp.Variable(n)
    obj = cp.Maximize(cp.sum(g))
    cons = [g >= 0, A_mat @ g <= M]

    solvers = [cp.SCS, cp.ECOS, cp.CLARABEL]
    # Try PDDLP for larger problems
    if n >= 512:
        solvers.insert(0, cp.PDLP)

    for solver in solvers:
        try:
            prob = cp.Problem(obj, cons)
            if solver == cp.SCS:
                prob.solve(solver=solver, max_iters=300000, eps=1e-9)
            elif solver == cp.ECOS:
                prob.solve(solver=solver, max_iters=50000)
            elif solver == cp.CLARABEL:
                prob.solve(solver=solver)
            elif solver == cp.PDLP:
                prob.solve(solver=solver, maxiter=300000, requires 계시hip=False)
            else:
                prob.solve(solver=solver)

            if g.value is not None:
                g_val = np.maximum(g.value, 0.0)
                if not np.any(np.isnan(g_val)) and not np.any(np.isinf(g_val)):
                    return g_val
        except Exception:
            continue

    return None


def get_active_set(b, k_init, k_max, fraction=0.95):
    """Get active set: top peaks + near-peak lags."""
    n_b = len(b)
    M = b.max()
    # Include all lags where b[k] > fraction * M, up to k_max
    near_peaks = np.where(b >= fraction * M)[0]
    # Also include top k_init
    top_k = np.argpartition(b, -min(k_init, n_b))[-min(k_init, n_b):]
    combined = np.unique(np.concatenate([near_peaks, top_k]))
    if len(combined) > k_max:
        # Keep the top k_max by value
        order = np.argsort(b[combined])[-k_max:]
        combined = combined[order]
    return np.sort(combined)


def lp_direction_search(f_init, time_limit, k_init=50, k_max=400, label="lp"):
    """Cutting-plane LP direction search with adaptive active set."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    b = fast_convolve(f)
    M = b.max()
    best_obj = M

    active_idx = get_active_set(b, k_init, k_max, fraction=0.95)
    it = 0
    stuck = 0
    no_improve_count = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break

        g = solve_lp(f, n, M, active_idx)

        if g is None:
            stuck += 1
            if stuck > 10:
                break
            # Shrink active set and retry
            if len(active_idx) > k_init:
                active_idx = get_active_set(fast_convolve(f), k_init, len(active_idx) // 2)
            continue

        S_g = g.sum()
        if S_g <= 1.0 + 1e-8:
            stuck += 1
            if stuck > 25:
                break
            # Expand active set
            if len(active_idx) < k_max:
                b_cur = fast_convolve(f)
                new_set = get_active_set(b_cur, k_init, min(k_max, len(active_idx) * 2))
                active_idx = new_set
            elif it % 10 == 0:
                # Perturb and re-evaluate
                b_cur = fast_convolve(f)
                active_idx = get_active_set(b_cur, k_init, k_max, fraction=0.90)
            continue

        # Check for violated constraints (cutting plane)
        fg = convolve_cross(f, g)
        violated = fg > M * (1 + 1e-7)
        if violated.any():
            v_idx = np.where(violated)[0]
            not_active = v_idx[~np.isin(v_idx, active_idx)]
            if len(not_active) > 0 and len(active_idx) < k_max:
                n_add = min(20, len(not_active))
                to_add = not_active[np.argsort(fg[not_active])[-n_add:]]
                active_idx = np.unique(np.concatenate([active_idx, to_add]))
                active_idx = active_idx[np.argsort(fg[active_idx])[-k_max:]]
                # Re-solve
                g = solve_lp(f, n, M, active_idx)
                if g is None:
                    stuck += 1
                    if stuck > 5:
                        break
                    continue
                S_g = g.sum()
                if S_g <= 1.0 + 1e-8:
                    stuck += 1
                    if stuck > 25:
                        break
                    continue

        stuck = 0
        no_improve_count = 0
        h = g / S_g

        # Line search: wide scan
        best_t = 0.05
        best_t_obj = M
        scan_steps = [0.02, 0.05, 0.08, 0.1, 0.12, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5]
        for t in scan_steps:
            f_trial = np.maximum((1.0 - t) * f + t * h, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t

        # Golden section refinement
        if best_t > 0.03 and best_t < 0.95:
            lo = max(0.0, best_t - 0.06)
            hi = min(1.0, best_t + 0.08)
            for _ in range(8):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                fa1 = np.maximum((1 - m1) * f + m1 * h, 0.0)
                sa1 = fa1.sum()
                if sa1 > 1e-10:
                    fa1 /= sa1
                fa2 = np.maximum((1 - m2) * f + m2 * h, 0.0)
                sa2 = fa2.sum()
                if sa2 > 1e-10:
                    fa2 /= sa2
                o1 = fast_convolve(fa1).max()
                o2 = fast_convolve(fa2).max()
                if o1 < o2:
                    hi = m2
                else:
                    lo = m1
            t_mid = (lo + hi) / 2.0
            f_mid = np.maximum((1 - t_mid) * f + t_mid * h, 0.0)
            s_mid = f_mid.sum()
            if s_mid > 1e-10:
                f_mid /= s_mid
                o_mid = fast_convolve(f_mid).max()
                if o_mid < best_t_obj:
                    best_t_obj = o_mid
                    best_t = t_mid

        f = np.maximum((1.0 - best_t) * f + best_t * h, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        else:
            f = np.ones(n) / n
            break

        cur_obj = fast_convolve(f).max()
        M = cur_obj

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count > 15:
                # Refresh active set more aggressively
                b_new = fast_convolve(f)
                active_idx = get_active_set(b_new, k_init, k_max, fraction=0.90)
                no_improve_count = 0

        if it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} obj={best_obj:.12f} score={2.0*n*best_obj:.6f} "
                  f"active={len(active_idx)} t={elapsed:.0f}s")

    return best_f, best_obj


def subgradient_polish(f_init, time_limit, lr_init=0.003, n_top=60, temp_init=0.002):
    """Subgradient method with momentum for final polish."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    v = np.zeros(n)
    lr = lr_init
    temp = temp_init
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(123)
    momentum = 0.8

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)

        # Softmax weights over top peaks
        scale_b = b / temp
        scale_b -= scale_b.max()
        w = np.exp(scale_b)
        w /= w.sum()

        top_idx = np.argpartition(b, -n_top)[-n_top:]
        w_sel = w[top_idx].copy()
        ws = w_sel.sum()
        if ws > 1e-30:
            w_sel /= ws
        else:
            w_sel = np.ones(len(top_idx)) / len(top_idx)
        w = np.zeros(len(b))
        w[top_idx] = w_sel

        # Subgradient computation (vectorized)
        g = np.zeros(n)
        for k in top_idx:
            if w[k] < 1e-12:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += w[k] * 2.0 * f[k - idx]

        v = momentum * v - lr * g
        f = project_simplex(f + v)

        if it % 50 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 10:
                    lr *= 0.85
                    temp *= 1.15
                    no_improve = 0
                if no_improve > 50:
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                    v = np.zeros(n)
                    no_improve = 0

            if it % 400 == 0:
                print(f"    sg it={it} obj={best_obj:.12f} score={2.0*n*best_obj:.6f} "
                      f"t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def peak_shaving(f_init, time_limit, steps=500):
    """Direct peak shaving: identify the peak lag and redistribute mass."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    rng = np.random.default_rng(77)

    for it in range(steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(f)
        peak_val = b.max()
        peak_idx = np.where(b >= peak_val - 1e-12)[0]

        # For each peak lag, find the contributing pairs
        # Move mass from "hot" indices to "cold" indices
        hot_score = np.zeros(n)
        cold_score = np.zeros(n)

        for k in peak_idx[:5]:  # Limit to top 5 peaks
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            contrib = f[idx] * f[k - idx]
            hot_score[idx] += contrib

        # Cold indices: those that don't contribute much to any peak
        total_contribution = np.zeros(n)
        for k in range(0, 2 * n - 1, max(1, (2 * n - 1) // 20)):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            total_contribution[idx] += f[idx] * f[k - idx]

        cold_score = -total_contribution  # More negative = colder

        # Find good exchange pairs
        hot_candidates = np.argsort(hot_score)[-min(20, n // 10):]
        cold_candidates = np.argsort(cold_score)[:min(20, n // 10)]

        best_gain = 0
        best_i = -1
        best_j = -1
        best_eps = 0

        for i in hot_candidates:
            if f[i] < 1e-6:
                continue
            for j in cold_candidates:
                if j == i:
                    continue
                # Try moving a small amount from i to j
                for eps_frac in [0.01, 0.05, 0.1]:
                    eps = f[i] * eps_frac
                    f[j] += eps
                    f[i] -= eps
                    s = f.sum()
                    if s < 1e-10:
                        break
                    f /= s
                    obj = fast_convolve(f).max()
                    gain = peak_val - obj
                    if gain > best_gain:
                        best_gain = gain
                        best_i = i
                        best_j = j
                        best_eps = eps
                    # Undo
                    f[best_j if best_j > 0 else i] 
                    # Just redo:
                    f[i] += eps
                    f[j] -= eps
                    f *= s  # Undo normalization (approximate)
                    break  # Only test one eps per pair for speed

        if best_i >= 0 and best_gain > 0:
            f[best_i] -= best_eps
            f[best_j] += best_eps
            f = np.maximum(f, 0.0)
            s = f.sum()
            if s > 1e-10:
                f /= s
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f.copy()

    return best_f, best_obj


def resample_sequence(f_old, n_new):
    """Resample a sequence to a new length using smooth interpolation."""
    n_old = len(f_old)
    if n_old == n_new:
        return f_old.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    f_new = np.interp(x_new, x_old, f_old)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(n_new) / n_new
    return f_new
