import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_fft_subgradient(f, w):
    """
    Compute subgradient of max(conv(f,f)) w.r.t. f using FFT.
    w: softmax weights on convolution positions (length 2n-1).
    Returns gradient vector of length n.
    """
    n = len(f)
    fft_size = 1 << (3 * n - 2 + 1).bit_length()
    W = rfft(w, fft_size)
    F = rfft(f, fft_size)

    # B[i] = (w * f)[i] for i = 0..n-1 (standard convolution)
    conv_wf = irfft(W * F, fft_size)[:n]

    # A[i] = sum_j f[j] * w[i+j], computed via reversed w
    w_rev = w[::-1]
    W_rev = rfft(w_rev, fft_size)
    conv_fw_rev = irfft(F * W_rev, fft_size)
    # A[i] = conv_fw_rev[2n-2-i], so A = conv_fw_rev[n-1:2n-1][::-1]
    A = conv_fw_rev[n - 1:2 * n - 1][::-1]

    return A + conv_wf


def solve_lp_direction(f, n, peak_indices, M, eps=1e-8):
    """Solve: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for k in peak_indices."""
    K = len(peak_indices)
    if K == 0:
        return None
    A_mat = np.zeros((K, n))
    for row_i in range(K):
        k = int(peak_indices[row_i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A_mat[row_i, lo:hi + 1] = f[lo:hi + 1]

    g = cp.Variable(n)
    constraints = [g >= 0, A_mat @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    solved = False
    for solver in [cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, max_iters=1000000, eps=eps)
            solved = True
            break
        except Exception:
            continue

    if not solved or g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None
    return g_val

def spectral_perturb(f, rng, strength=0.05):
    """
    Frequency-domain structured perturbation.
    Modifies the spectrum to create different "shapes" while keeping f >= 0.
    """
    n = len(f)
    F = rfft(f, n)
    # Add a random frequency component
    k = rng.integers(1, n // 2)
    amp = strength * f.max()
    phase = rng.uniform(0, 2 * np.pi)
    F[k] += amp * np.exp(1j * phase)
    if k < n - k:
        F[n - k] += amp * np.exp(1j * phase)
    f_new = irfft(F, n)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = f.copy()
    return f_new

            if stuck > patience * 3 // 4:
                f = f * (1.0 + 0.01 * rng.standard_normal(n))
                f = np.maximum(f, 0.0)
                f /= f.sum()

    return best_f, best_b


def subgradient_polish(f_init, time_limit, lr_init=0.02, label="sg"):
    """
    FFT-accelerated subgradient descent with adaptive temperature and step size.
    Uses log-sum-exp smoothing of the max.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_b = fast_convolve(f).max()
    lr = lr_init
    temp = 0.003
    no_improve = 0
    it = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        # Softmax weights (smoothed max)
        scaled = b / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w /= w_sum

        # FFT-based subgradient
        grad = compute_fft_subgradient(f, w)

        # Update with projection
        f_new = project_simplex(f - lr * grad)
        f = f_new

        if it % 30 == 0:
            obj = fast_convolve(f).max()
            if obj < best_b - 1e-13:
                best_b = obj
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.05, 0.15)
            else:
                no_improve += 1
                if no_improve > 3:
                    lr *= 0.9
                    no_improve = 0

            if it % 150 == 0:
                temp = min(temp * 1.1, 0.03)
                score = 2.0 * n * best_b
                print(f"    {label} iter={it} obj={best_b:.12f} score={score:.6f} "
                      f"lr={lr:.4f} temp={temp:.5f} t={time.monotonic()-t0:.0f}s")

            if no_improve > 20:
                f = f + rng.standard_normal(n) * 0.002 * f
                f = np.maximum(f, 1e-15)
                f /= f.sum()
                lr = lr_init
                temp = 0.003
                no_improve = 0

    return best_f, best_b


def coordinate_descent(f_init, time_limit, label="cd"):
    """
    Fast coordinate descent with batched evaluations.
    Groups variables and tries multiplicative updates.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_b = fast_convolve(f).max()
    multipliers = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.4, 2.0]
    conv_len = 2 * n - 1
    fft_size = 1 << (conv_len).bit_length()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-14:
                continue
            best_j_val = f[j]
            best_j_obj = best_b
            for mult in multipliers:
                new_val = f[j] * mult
                if new_val < 0:
                    continue
                f[j] = new_val
                s = f.sum()
                if s < 1e-10:
                    f[j] = best_j_val
                    continue
                # Quick eval via FFT
                F = rfft(f, fft_size)
                conv = irfft(F * F, fft_size)[:conv_len]
                obj = conv.max()
                if obj < best_j_obj - 1e-14:
                    best_j_obj = obj
                    best_j_val = new_val
                else:
                    f[j] = best_j_val

            if abs(best_j_val - f[j]) > 1e-15:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                    best_b = fast_convolve(f).max()
                    improved = True

        if not improved:
            break

    return f, fast_convolve(f).max()


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    n = len(f)
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def try_length_optimization(f_base, target_n, time_budget, label):
    """Run full optimization pipeline at a specific length."""
    f_interp = interpolate_to_length(f_base, target_n)
    n = target_n

    # Phase A: LP
    rem = time_budget
    t_lp = rem * 0.6
    if t_lp > 30:
        f_lp, obj_lp = run_lp_optimize(f_interp, time_limit=t_lp, label=f"{label}_lp",
                                        max_peaks_init=15, max_peaks_final=120,
                                        patience=12, lp_eps=1e-8)
    else:
        f_lp = f_interp
        obj_lp = fast_convolve(f_interp).max()

    # Phase B: Subgradient polish
    rem = time_budget - min(time_budget * 0.6, time_budget)
    t_sg = max(rem * 0.7, 10)
    if time.monotonic() < time.time() + t_sg:
        pass  # just time check
    f_sg, obj_sg = subgradient_polish(f_lp, time_limit=min(t_sg, time_budget * 0.3),
                                       lr_init=0.015, label=f"{label}_sg")

    # Phase C: CD
    f_cd, obj_cd = coordinate_descent(f_sg, time_limit=min(20, time_budget * 0.15),
                                       label=f"{label}_cd")

    return f_cd, obj_cd


def propose_candidate(seed=42, budget_s=1800.0):

    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(seed)

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)

            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1024) / 1024
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"No parent, using uniform n=1024, score={p_score:.6f}")

    best_seq = parent.copy()
    best_score = p_score
    n_parent = len(parent)

    # === Phase 1: Main LP optimization on parent ===
    rem = deadline - time.monotonic()

        f1, obj1 = run_lp_optimize(parent, time_limit=t1, label="lp1",
                                    max_peaks_init=15, max_peaks_final=150,

    # === Phase 4: Multi-length exploration ===
