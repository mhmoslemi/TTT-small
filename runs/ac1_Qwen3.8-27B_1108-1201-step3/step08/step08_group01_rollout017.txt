def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_rows(a_fixed, rows):
    """Build rows of the convolution matrix T_a for specified row indices."""
    n = len(a_fixed)
    T = np.zeros((len(rows), n))
    for idx, k in enumerate(rows):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for j in range(i_start, i_end + 1):
            T[idx, j] = a_fixed[k - j]
    return T


def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

import numpy as np
from scipy.fft import rfft, irfft

import cvxpy as cp


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Compute convolve(a, b) using FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_T_rows(a_fixed, rows, n):
    """Build rows of convolution matrix T_a for given row indices."""
    T = np.zeros((len(rows), n))
    for idx, k in enumerate(rows):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        j_range = np.arange(i_start, i_end + 1)
        T[idx, j_range] = a_fixed[k - j_range]
    return T


def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

        T @ b_var <= M_var,

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

Wait, I'm concerned about the efficiency. The coordinate descent with n=1024 does 1024 coordinate updates per sweep, and each update involves computing the max over the affected region. This is O(n) per coordinate, so O(n^2) per sweep. For n=1024, that's about 1 million operations per sweep. If we do 100 sweeps, that's 100 million operations. This should be fine in 200 seconds.

The LP solve is the bottleneck. With column generation, each LP solve should be fast (200-500 constraints, 1024 variables). With a good solver like MOSEK or Gurobi, this should take maybe 0.1-1 seconds per solve. In 300 seconds, we can do 300-3000 iterations.

Let me also think about whether there's a better algorithmic idea.

**Idea: Use the dual of the LP to guide the coordinate descent.**

The dual of the LP (min M s.t. T@b <= M, sum(b)=1, b>=0) is:
- Maximize sum(y)
- Subject to: T^T @ y <= 1 (componentwise)
- y >= 0

The dual solution y* gives weights to each lag position. If we use these weights to define a "loss" function:
L(a) = sum_k y*[k] * conv(a,a)[k]

Then minimizing L(a) (which is a quadratic form) is a convex problem! We can solve it with gradient descent or other methods.

This is actually a great idea. The dual weights y* define a "relaxed" objective that's easier to optimize. And as fictitious play converges, the dual weights converge to the optimal ones, and the relaxed objective converges to the true objective.

So the algorithm becomes:
1. Run a few iterations of fictitious play to get good dual weights y*.
2. Use y* to define a convex quadratic objective: L(a) = a^T (sum_k y*[k] Q_k) a
3. Minimize L(a) subject to sum(a)=1, a>=0 (this is a QP!).
4. Check the true objective. If it's better, keep it; otherwise, go back to step 1.

The QP in step 3 can be solved efficiently with cvxpy.

Let me implement this:

```python
def solve_qp_with_dual_weights(y_weights, n, solver):
    """
    Solve: min a^T B a s.t. sum(a)=1, a>=0
    where B = sum_k y_weights[k] * Q_k
    """
    # Build B = sum_k y[k] * Q_k
    B = np.zeros((n, n))
    for k in range(2 * n - 1):
        if y_weights[k] < 1e-15:
            continue
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            B[i, j] += y_weights[k]
    
    # Make sure B is symmetric
    B = (B + B.T) / 2

"""

Key improvements: efficient LP via constraint generation, dual-guided refinement,

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Compute linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_T_rows(a_fixed, rows, n):
    """Build specified rows of the (2n-1) x n convolution matrix T_a."""
    T = np.zeros((len(rows), n))
    for idx in range(len(rows)):
        k = int(rows[idx])
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        j_range = np.arange(i_start, i_end + 1)
        T[idx, j_range] = a_fixed[k - j_range]
    return T


def simplex_project(v):
    """Project vector v onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def get_lp_solver():

def solve_lp_response(a_fixed, solver, active_rows):
    """
    Solve: min M s.t. T_a[active_rows] @ b <= M, sum(b)=1, b>=0.
    Returns (b, M, status).
    """
    n = len(a_fixed)
    T = build_T_rows(a_fixed, active_rows, n)
    
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    status = None
    try:
        prob.solve(solver=solver)
        status = prob.status
    except Exception:
        pass
    
    if status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                status = prob.status
                if status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if status not in ('optimal', 'optimal_inaccurate'):
        return None, None, status
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, status
    b_val = b_val / s
    
    if M_var.value is not None:
        M_val = float(M_var.value)
    else:
        M_val = float(convolve_ab(a_fixed, b_val).max())
    
    return b_val, M_val, status


def fictitious_play_cg(a_init, n, time_limit, solver, max_active=400, verbose=True):
    """
    Fictitious play with column generation for the LP.
    Alternates between solving the LP best response and updating the sequence.
    """
    a = np.maximum(a_init.astype(np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    conv_init = fast_convolve(a)
    best_score = 2.0 * n * float(conv_init.max())
    
    # Initialize active set with top-k constraints
    top_k = min(max_active, 2 * n - 1)
    active_rows = np.sort(np.argpartition(conv_init, -top_k)[-top_k:])
    
    b_ema = a.copy().astype(np.float64)
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        
        # Solve LP for best response
        b_new, M_lp, status = solve_lp_response(a, solver, active_rows)
        if b_new is None:

            active_rows = np.arange(min(2 * n - 1, 2 * max_active))

            if b_new is None:
                break
        
        # Adaptive step size
        alpha = 2.0 / (it + 1 + 1.0)
        
        # Update EMA
        b_ema = (1.0 - alpha) * b_ema + alpha * b_new

        for cand in candidates:

            cs = cand.sum()
            if cs < 1e-10:

            cconv = fast_convolve(cand)
            score = 2.0 * n * float(cconv.max())
