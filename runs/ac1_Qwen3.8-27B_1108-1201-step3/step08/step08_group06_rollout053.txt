"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the optimization score for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def subgradient_max_conv(a, active_k=None, tol=1e-10):
    """
    Compute subgradient of f(a) = max_k (a*a)_k.
    Returns (gradient, active_set).
    """
    n = len(a)
    conv = fast_convolve(a)
    max_val = conv.max()
    
    if active_k is None:
        active_k = np.where(conv >= max_val - tol * max_val)[0]
    
    # g[j] = 2 * sum_{k in active} a[k-j] for valid k-j
    g = np.zeros(n)
    for k in active_k:
        k = int(k)
        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                g[j] += 2.0 * a[kj]
    
    return g, active_k, max_val


def subgradient_fast(a, top_k=3):
    """
    Faster subgradient using only top-k active constraints.
    """
    n = len(a)
    conv = fast_convolve(a)
    max_val = conv.max()
    # Get top-k positions
    idx = np.argsort(conv)[::-1][:top_k]
    # Only include those close to max
    tol = 1e-8 * max_val
    active = idx[conv[idx] >= max_val - tol]
    
    g = np.zeros(n)
    for k in active:
        k = int(k)
        # g[j] += 2 * a[k-j] for valid j
        # k-j valid means 0 <= k-j < n, i.e., k-n+1 <= j <= k
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        # a[k-j] for j in [j_start, j_end] = a[k-j_end] ... a[k-j_start]
        # = a[max(0,n-1-k)] ... a[min(k, n-1)]
        # Let's just do it directly
        g[j_start:j_end+1] += 2.0 * a[k - np.arange(j_start, j_end+1)]
    
    return g, max_val


def project_simplex(v):
    """Project v onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def frank_wolfe(a0, time_limit, verbose=True, accelerate=True):
    """
    Frank-Wolfe (conditional gradient) method for:
    min f(a) = max_k (a*a)_k  s.t. a >= 0, sum(a) = 1
    
    With optional acceleration (FISTA-style).
    """
    n = len(a0)
    a = a0.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    y = a.copy()  # Ghost point for acceleration
    a_prev = a.copy()
    
    for k in range(10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute subgradient at y
        g, max_val = subgradient_fast(y, top_k=5)
        
        # Linear minimization: s = e_{argmin g}
        s_idx = int(np.argmin(g))
        s = np.zeros(n)
        s[s_idx] = 1.0
        
        # Step size
        if not accelerate:
            t = 2.0 / (k + 2)
        else:
            # Backtracking line search
            t = 1.0
            f_y = fast_convolve(y).max()
            grad_dir = g[s_idx] - np.dot(g, a)  # <g, s - a>
            if grad_dir >= 0:
                # No improvement possible with this s, use full step in FW
                t = 2.0 / (k + 2)
            else:
                # Backtracking
                for _ in range(30):
                    a_trial = (1 - t) * a + t * s
                    a_trial = np.maximum(a_trial, 0)
                    f_trial = fast_convolve(a_trial).max()
                    if f_trial <= f_y + 0.01 * t * grad_dir:
                        break
                    t *= 0.5
        
        a_new = (1 - t) * a + t * s
        a_new = np.maximum(a_new, 0)
        s_sum = a_new.sum()
        if s_sum < 1e-15:
            break
        a_new = a_new / s_sum
        
        # Update
        a_prev = a.copy()
        a = a_new
        
        # Momentum update for ghost point
        if accelerate:
            momentum = (k + 0.5) / (k + 2.5)  # or (k-0.5)/(k+1.5)
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 0)
            y = y / y.sum()
        
        # Track best
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if verbose and k % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  FW[{ 'accel' if accelerate else 'std'}] iter={k}, "
                  f"score={score:.8f}, best={best_score:.8f}, "
                  f"s_idx={s_idx}, t={t:.4f}, elapsed={elapsed:.0f}s")
    
    return best_a, best_score


def subgradient_method(a0, time_limit, verbose=True):
    """
    Subgradient method with simplex projection and iterate averaging.
    """
    n = len(a0)

def linalg_subgradient(a, top_k=5):
    """Compute subgradient efficiently using the structure."""
    n = len(a)
    conv = fast_convolve(a)
    max_val = conv.max()
    idx = np.argsort(conv)[::-1][:top_k]
    active = [int(k) for k in idx if conv[k] >= max_val - 1e-8 * max_val]
    
    g = np.zeros(n)
    for k in active:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        g[j_start:j_end+1] += 2.0 * a[k - np.arange(j_start, j_end + 1)]
    
    return g, max_val


def accelerated_fw(a0, time_limit, verbose=True, top_k=5):
    """
    Accelerated Frank-Wolfe with more careful implementation.
    Uses the structure of the subgradient for efficiency.
    """
    n = len(a0)
    a = a0.copy()
    a = np.maximum(a, 0)
    a_sum = a.sum()
    a = a / a_sum
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    y = a.copy()
    a_prev = a.copy()
    
    for k in range(5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Subgradient at y
        g, max_val = linalg_subgradient(y, top_k=top_k)
        
        # FW vertex: s = e_{argmin g}
        s_idx = int(np.argmin(g))
        
        # Step size: t = 2/(k+2) (standard, safe)
        t = 2.0 / (k + 2)
        
        # New iterate
        a_new = (1 - t) * a + t * 0  # (1-t)*a
        a_new[s_idx] += t  # + t*e_s_idx
        
        # Project: ensure non-negative and sum=1
        a_new = np.maximum(a_new, 0)
        s_sum = a_new.sum()
        a_new = a_new / s_sum
        
        # Momentum
        a_prev = a.copy()
        a = a_new
        
        momentum = (k) / (k + 2.0)  # standard FISTA momentum
        y = a + momentum * (a - a_prev)
        y = np.maximum(y, 0)
        y_sum = y.sum()
        if y_sum > 1e-15:
            y = y / y_sum
        else:
            y = a.copy()
        
        # Track best
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if verbose and k % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  AccFW iter={k}, score={score:.8f}, best={best_score:.8f}, "
                  f"s_idx={s_idx}, elapsed={elapsed:.0f}s")
    
    return best_a, best_score


def coordinate_descent(a0, time_limit, verbose=True):
    """
    Coordinate descent with exact line search for pairs.
    Focus on pairs that affect the peak most.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    iters = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        
        conv = fast_convolve(a)
        peak_idx = int(np.argmax(conv))
        
        # Generate candidate pairs
        n_pairs = min(n, 300)
        pairs = set()
        
        # Peak-focused pairs
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        for _ in range(n_pairs // 2):
            i = int(rng.integers(js, je + 1))
            j = int(rng.integers(0, n))
            if i != j:
                pairs.add((min(i, j), max(i, j)))
        
        # Random pairs
        for _ in range(n_pairs // 2):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                pairs.add((min(i, j), max(i, j)))
        
        for (i, j) in list(pairs):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-15 and a[j] < 1e-15:
                continue
            
            # Golden section search for transfer from i to j
            t_lo = -a[j] + 1e-16
            t_hi = a[i] - 1e-16
            if t_hi <= t_lo:
                continue
            
            gr = (np.sqrt(5) - 1) / 2
            x1 = t_hi - gr * (t_hi - t_lo)
            x2 = t_lo + gr * (t_hi - t_lo)
            
            def eval_transfer(t):
                a_t = a.copy()
                a_t[i] -= t
                a_t[j] += t
                if a_t[i] < -1e-15 or a_t[j] < -1e-15:
                    return 1e10
                a_t = np.maximum(a_t, 0)
                ss = a_t.sum()
                if ss < 1e-15:
                    return 1e10
                a_t = a_t / ss
                return fast_convolve(a_t).max()
            
            f1 = eval_transfer(x1)
            f2 = eval_transfer(x2)
            
            for _ in range(15):
                if f1 < f2:
                    t_hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = t_hi - gr * (t_hi - t_lo)
                    f1 = eval_transfer(x1)
                else:
                    t_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = t_lo + gr * (t_hi - t_lo)
                    f2 = eval_transfer(x2)
            
            t_opt = (t_lo + t_hi) / 2
            f_opt = min(f1, f2)
            
            # Check if this improves the score (which is 2*n*max_conv since sum=1)
            if f_opt < best_score / (2.0 * n) - 1e-14:
                a[i] = max(0, a[i] - t_opt)
                a[j] = max(0, a[j] + t_opt)
                a = a / a.sum()
                best_score = 2.0 * n * fast_convolve(a).max()
                best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 3:
                # Perturb
                noise = rng.standard_normal(n) * 0.001
                a = best_a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
        else:
            no_improve = 0
        
        if verbose and iters % 10 == 0:
            print(f"  CD iter={iters}, score={best_score:.8f}")
    
    return best_a, best_score


def mirror_descent_log(a0, time_limit, verbose=True):
    """
    Mirror descent in log-space (exponential weights).
    Minimize f(a) = max_k (a*a)_k over the simplex.
    """
    n = len(a0)
    a = np.maximum(a0, 1e-30)
    a = a / a.sum()
    z = np.log(a)  # Log-space variables
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    for k in range(2000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        a_curr = np.exp(z - z.max())
        a_curr = a_curr / a_curr.sum()
        
        # Subgradient
        g, max_val = linalg_subgradient(a_curr, top_k=3)
        
        # Mirror descent update in log-space:
        # z_new = z - eta * g (in the dual)
        # In log-space, the mirror map is exp, so:
        # a_new = exp(z - eta * g) / sum(exp(z - eta * g))
        # = a * exp(-eta * g) / sum(a * exp(-eta * g))
        
        eta = 1.0 / np.sqrt(k + 1)
        log_weights = -eta * g
        log_weights -= log_weights.max()
        weights = np.exp(log_weights)
        a_new = a_curr * weights
        a_new = a_new / a_new.sum()
        
        # Update z
        z = np.log(np.maximum(a_new, 1e-30))
        
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        if verbose and k % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  Mirror iter={k}, score={score:.8f}, best={best_score:.8f}, "
                  f"elapsed={elapsed:.0f}s")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def init_fourier(n, mode=1):
    """Fourier-based initialization."""
    j = np.arange(n)
    a = 1.0 + 0.5 * np.cos(2 * np.pi * mode * j / n)
    a = np.maximum(a, 0)
    return a / a.sum()


def init_gaussian(n, sigma_frac=0.3):
    """Gaussian initialization."""
    j = np.arange(n)
    center = n / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((j - center) / sigma) ** 2)
    return a / a.sum()


def init_multi_fourier(n, modes=[1, 2, 3], amps=[0.3, 0.2, 0.1]):
    """Multi-Fourier initialization."""
    j = np.arange(n)
    a = np.ones(n)
    for m, amp in zip(modes, amps):
        a += amp * np.cos(2 * np.pi * m * j / n)
    a = np.maximum(a, 0)
    return a / a.sum()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP Fictitious Play with top-k constraint selection and geometric mean updates
2. Exponentiated Gradient (Mirror Descent with KL divergence) for direct subgradient steps

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    """Cross-convolution f*g via FFT."""
    n = len(f)
    m = len(g)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def compute_score(a):
    """Compute the optimization score for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_C_topk(f, M, k=50):
    """Build constraint matrix using only top-k most active constraints."""
    n = len(f)
    conv = fast_convolve(f)
    # Get top-k positions
    idx = np.argsort(conv)[::-1][:k]
    # Filter to those that are actually near the max
    threshold = M - 1e-6 * M
    idx = idx[conv[idx] >= threshold]
    if len(idx) == 0:
        idx = np.array([int(np.argmax(conv))])
    
    C = np.zeros((len(idx), n))
    for i, k_val in enumerate(idx):
        k_val = int(k_val)
        for j in range(n):
            kj = k_val - j
            if 0 <= kj < n:
                C[i, j] = f[kj]
    return C, idx


def solve_lp_direction(f, M, top_k=100):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for top-k k, g >= 0."""
    n = len(f)
    C, active_idx = build_C_topk(f, M, k=top_k)
    if len(C) == 0:
        return None
    
    c = -np.ones(n)
    b_ub = np.full(len(C), M)
    res = linprog(c, A_ub=C, b_ub=b_ub, bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': 10})
    if res.status != 0 or res.x is None:
        return None
    g = np.maximum(res.x, 0)
    if g.sum() < 1e-15:
        return None
    return g


def geometric_mean_update(f, g, t):
    """Geometric mean interpolation in log-space."""
    f_safe = np.maximum(f, 1e-30)
    g_safe = np.maximum(g, 1e-30)
    log_new = (1 - t) * np.log(f_safe) + t * np.log(g_safe)
    log_new -= log_new - log_new.mean()
    a = np.exp(log_new)
    s = a.sum()
    if s < 1e-30:
        return f.copy()
    return a / s


def lp_fictitious_play(f_init, time_limit, top_k=100, verbose=True):
    """LP-based fictitious play with geometric mean updates."""
    f = f_init.copy()
    s = f.sum()
    f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()
    
    f_avg = f.copy()
    
    for k in range(500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(f)
        M = conv.max()
        
        g = solve_lp_direction(f, M, top_k=top_k)
        if g is None:
            break
        
        g = g / g.sum()
        
        # Adaptive step size
        t_step = 2.0 / (k + 2)
        
        a_new = geometric_mean_update(f, g, t_step)
        
        f_avg = ((k * f_avg) + a_new) / (k + 1)
        
        score_cur = compute_score(a_new)
        score_avg = compute_score(f_avg)
        
        if score_avg < best_score:
            best_score = score_avg
            best_f = f_avg.copy()
        if score_cur < best_score:
            best_score = score_cur
            best_f = a_new.copy()
        
        f = a_new
        
        if verbose and k % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LPFP iter={k}, M={M:.6e}, avg_score={score_avg:.8f}, "
                  f"best={best_score:.8f}, elapsed={elapsed:.0f}s")
    
    return best_f, best_score


def exponentiated_gradient(a0, time_limit, verbose=True, eta_scale=1.0):
    """
    Exponentiated gradient (mirror descent with KL divergence).
    Direct subgradient-based approach.
    """
    n = len(a0)
    a = np.maximum(a0, 1e-30)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    a_avg = a.copy()
    
    for k in range(2000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        M = conv.max()
        
        # Get active set
        top_k = 10
        idx = np.argsort(conv)[::-1][:top_k]
        threshold = M - 1e-8 * M
        active = [int(kk) for kk in idx if conv[kk] >= threshold]
        
        # Compute subgradient: g[j] = sum_{k in active} a[k-j] for valid k-j
        g = np.zeros(n)
        for kk in active:
            j_start = max(0, kk - n + 1)
            j_end = min(n - 1, kk)
            g[j_start:j_end+1] += a[kk - np.arange(j_start, j_end + 1)]
        
        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        
        # Step size
        eta = eta_scale / np.sqrt(k + 1)
        
        # Mirror descent update: a_new = a * exp(-eta * g) / sum(a * exp(-eta * g))
        log_ratio = -eta * g
        log_ratio -= log_ratio.max()
        weights = np.exp(log_ratio)
        a_new = a * weights
        a_sum = a_new.sum()
        if a_sum < 1e-300:
            break
        a_new = a_new / a_sum
        
        # Update average
        a_avg = ((k * a_avg) + a_new) / (k + 1)
        
        score_cur = compute_score(a_new)
        score_avg = compute_score(a_avg)
        
        if score_avg < best_score:
            best_score = score_avg
            best_a = a_avg.copy()
        if score_cur < best_score:
            best_score = score_cur
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and k % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  ExpGrad iter={k}, score_avg={score_avg:.8f}, "
                  f"best={best_score:.8f}, eta={eta:.6f}, elapsed={elapsed:.0f}s")
    
    return best_a, best_score


def coordinate_descent(a0, time_limit, verbose=True):
    """Coordinate descent with golden-section search for pair transfers."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(99)
    
    iters = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        
        conv = fast_convolve(a)
        peak_idx = int(np.argmax(conv))
        
        # Generate pairs
        n_pairs = min(n, 400)
        pairs = set()
        
        # Peak-focused
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        for _ in range(n_pairs // 2):
            i = int(rng.integers(js, je + 1))
            j = int(rng.integers(0, n))
            if i != j:
                pairs.add((min(i, j), max(i, j)))
        
        # Random
        for _ in range(n_pairs // 2):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                pairs.add((min(i, j), max(i, j)))
        
        for (i, j) in list(pairs):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-15 and a[j] < 1e-15:
                continue
            
            t_lo = -a[j] + 1e-16
            t_hi = a[i] - 1e-16
            if t_hi <= t_lo:
                continue
            
            gr = (np.sqrt(5) - 1) / 2
            x1 = t_hi - gr * (t_hi - t_lo)
            x2 = t_lo + gr * (t_hi - t_lo)
            
            def eval_t(t):
                a_t = a.copy()
                a_t[i] -= t
                a_t[j] += t
                if a_t[i] < -1e-14 or a_t[j] < -1e-14:
                    return 1e10
                a_t[i] = max(0, a_t[i])
                a_t[j] = max(0, a_t[j])
                ss = a_t.sum()
                if ss < 1e-15:
                    return 1e10
                a_t = a_t / ss
                return fast_convolve(a_t).max()
            
            f1 = eval_t(x1)
            f2 = eval_t(x2)
            
            for _ in range(18):
                if f1 < f2:
                    t_hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = t_hi - gr * (t_hi - t_lo)
                    f1 = eval_t(x1)
                else:
                    t_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = t_lo + gr * (t_hi - t_lo)
                    f2 = eval_t(x2)
            
            t_opt = (t_lo + t_hi) / 2
            f_opt = min(f1, f2)
            threshold = best_score / (2.0 * n)
            
            if f_opt < threshold - 1e-14:
                a[i] = max(0.0, a[i] - t_opt)
                a[j] = max(0.0, a[j] + t_opt)
                a = a / a.sum()
                best_score = compute_score(a)
                best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 5:
                noise = rng.standard_normal(n) * 0.002
                a = best_a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
        else:
            no_improve = 0
        
        if verbose and iters % 10 == 0:
            print(f"  CD iter={iters}, score={best_score:.8f}")
    
    return best_a, best_score


def block_coordinate_descent(a0, time_limit, verbose=True):
    """
    Block coordinate descent: transfer mass between blocks.
    Divides the sequence into blocks and optimally redistributes between them.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(55)
    
    # Define blocks
    n_blocks = max(4, n // 64)
    block_size = n // n_blocks
    blocks = [np.arange(i * block_size, min((i + 1) * block_size, n)) 
              for i in range(n_blocks)]
    
    iters = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        
        # Pick two random blocks
        b1, b2 = rng.integers(0, n_blocks, size=2)
        if b1 == b2:
            continue
        b1, b2 = int(b1), int(b2)
        
        # Try transferring a fraction from block b1 to block b2
        mass_b1 = a[blocks[b1]].sum()
        mass_b2 = a[blocks[b2]].sum()
        
        if mass_b1 < 1e-15:
            continue
        
        for frac in [0.01, 0.05, 0.1, 0.2, 0.3]:
            delta = mass_b1 * frac
            a_trial = a.copy()
            
            # Reduce block b1 proportionally
            a_trial[blocks[b1]] *= (1 - frac)
            
            # Add to block b2 proportionally (weighted by current values)
            b2_vals = a_trial[blocks[b2]].copy()
            b2_sum = b2_vals.sum()
            if b2_sum > 1e-15:
                a_trial[blocks[b2]] = b2_vals + delta * b2_vals / b2_sum
            else:
                a_trial[blocks[b2]] += delta / len(blocks[b2])
            
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            sc = compute_score(a_trial)
            
            if sc < best_score - 1e-12:
                a = a_trial
                best_score = sc
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 20:
                noise = rng.standard_normal(n) * 0.003
                a = best_a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
        else:
            no_improve = 0
        
        if verbose and iters % 50 == 0:
            print(f"  BCD iter={iters}, score={best_score:.8f}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_fourier(n, modes=[1, 2, 3], amps=[0.3, 0.2, 0.1]):
    """Fourier-based initialization."""
    j = np.arange(n)
    a = np.ones(n)
    for m, amp in zip(modes, amps):
        a += amp * np.cos(2 * np.pi * m * j / n)
    a = np.maximum(a, 0)
    return a / a.sum()


def init_cosine_taper(n, power=2):
    """Cosine taper initialization."""
    j = np.arange(n)
    t = np.pi * j / (n - 1) if n > 1 else np.array([0.0])
    a = np.cos(t / 2) ** power
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def propose_candidate(seed=42, budget_s=1800.0):

    - LP Fictitious Play (top-k constraints, geometric mean updates)
    - Exponentiated Gradient (mirror descent with KL)
    - Coordinate Descent (pair transfers with exact line search)
    - Block Coordinate Descent
    - Multi-start with diverse initializations
    - Length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    if parent is None:
        n_init = 768
        parent = init_fourier(n_init, [1, 2, 3], [0.3, 0.2, 0.1])
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Initialized: n={n_init}, score={best_score:.8f}")
    
    n_parent = len(parent)
    print(f"Starting optimization. Parent n={n_parent}, score={best_score:.8f}")
    print(f"Time budget: {budget_s:.0f}s")
    
    # === Phase 1: LP Fictitious Play (primary method) ===
    print(f"\n{'='*60}")
    print(f"Phase 1: LP Fictitious Play (best={best_score:.8f})")
    print(f"{'='*60}")
    t_alloc = 350.0
    rem = deadline - time.monotonic()
    t_alloc = min(t_alloc, rem * 0.25)
    print(f"  Budget: {t_alloc:.0f}s")
    a1, sc1 = lp_fictitious_play(best_seq, t_alloc, top_k=100, verbose=True)
    best_seq, best_score = check_and_update(a1, "lpfp1", best_seq, best_score)
    
    # === Phase 2: Exponentiated Gradient ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Exponentiated Gradient (best={best_score:.8f})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic() - 600
    if rem > 30:
        t_alloc = min(rem * 0.15, 150.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a2, sc2 = exponentiated_gradient(best_seq, t_alloc, verbose=True, eta_scale=0.5)
        best_seq, best_score = check_and_update(a2, "expgrad", best_seq, best_score)
    
    # === Phase 3: Coordinate Descent ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Coordinate Descent (best={best_score:.8f})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic() - 500
    if rem > 30:
        t_alloc = min(rem * 0.2, 200.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a3, sc3 = coordinate_descent(best_seq, t_alloc, verbose=True)
        best_seq, best_score = check_and_update(a3, "cd1", best_seq, best_score)
    
    # === Phase 4: Second LP FP round (on refined solution) ===
    print(f"\n{'='*60}")
    print(f"Phase 4: LP Fictitious Play Round 2 (best={best_score:.8f})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic() - 400
    if rem > 30:
        t_alloc = min(rem * 0.2, 150.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a4, sc4 = lp_fictitious_play(best_seq, t_alloc, top_k=80, verbose=True)
        best_seq, best_score = check_and_update(a4, "lpfp2", best_seq, best_score)
    
    # === Phase 5: Block Coordinate Descent ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Block Coordinate Descent (best={best_score:.8f})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic() - 300
    if rem > 30:
        t_alloc = min(rem * 0.15, 100.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a5, sc5 = block_coordinate_descent(best_seq, t_alloc, verbose=True)
        best_seq, best_score = check_and_update(a5, "bcd", best_seq, best_score)
    
    # === Phase 6: Multi-start with different initializations ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Multi-Start (best={best_score:.8f})")
    print(f"{'='*60}")
    inits = [
        ("cos_taper", init_cosine_taper(n_parent, 2)),
        ("fourier_1_3", init_fourier(n_parent, [1, 3], [0.25, 0.15])),
        ("fourier_2_5", init_fourier(n_parent, [2, 5], [0.2, 0.1])),
    ]
    
    for label, init_a in inits:
        rem = deadline - time.monotonic() - 200
        if rem < 30:
            break
        t_alloc = min(rem * 0.2, 60.0)
        sc_init = compute_score(init_a)
        print(f"  {label}: init_score={sc_init:.8f}, budget={t_alloc:.0f}s")
        
        # LP FP on this init
        a_new, sc_new = lp_fictitious_play(init_a, t_alloc * 0.7, top_k=50, verbose=False)
        best_seq, best_score = check_and_update(a_new, f"lpfp_{label}", best_seq, best_score)
        
        # Quick CD
        rem2 = deadline - time.monotonic() - 100
        if rem2 > 15:
            a_new2, sc_new2 = coordinate_descent(a_new, min(rem2 * 0.2, 20.0), verbose=False)
            best_seq, best_score = check_and_update(a_new2, f"cd_{label}", best_seq, best_score)
    
    # === Phase 7: Length Exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 7: Length Exploration (best={best_score:.8f})")
    print(f"{'='*60}")
    test_lengths = sorted(set([
        n_parent - 2, n_parent + 2, n_parent + 4, n_parent - 4,
        n_parent // 2 * 3,  # 3n/2
        512, 1024, 1280, 1536, 2048
    ]))
    test_lengths = [l for l in test_lengths if 256 <= l <= 3000 and l != n_parent]
    
    for alt_n in test_lengths:
        rem = deadline - time.monotonic() - 80
        if rem < 20:
            break
        t_alloc = min(rem * 0.12, 50.0)
        if t_alloc < 10:
            break
        
        init_alt = interpolate_sequence(best_seq, alt_n)
        sc_init = compute_score(init_alt)
        print(f"  n={alt_n}: init_score={sc_init:.8f}, budget={t_alloc:.0f}s")
        
        # LP FP
        a_new, sc_new = lp_fictitious_play(init_alt, t_alloc * 0.6, top_k=50, verbose=False)
        best_seq, best_score = check_and_update(a_new, f"lpfp_n{alt_n}", best_seq, best_score)
        
        # Quick CD if promising
        rem2 = deadline - time.monotonic() - 40
        if rem2 > 10 and compute_score(a_new) < 1.51:
            a_new2, sc_new2 = coordinate_descent(a_new, min(rem2 * 0.3, 15.0), verbose=False)
            best_seq, best_score = check_and_update(a_new2, f"cd_n{alt_n}", best_seq, best_score)
    
    # === Phase 8: Final Polish ===
    print(f"\n{'='*60}")
    print(f"Phase 8: Final Polish (best={best_score:.8f})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic() - 20
    if rem > 15:
        t_alloc = rem * 0.6
        a_fin, sc_fin = coordinate_descent(best_seq, t_alloc, verbose=False)
        best_seq, best_score = check_and_update(a_fin, "final_cd", best_seq, best_score)
    
    rem = deadline - time.monotonic() - 5
    if rem > 10:
        a_fin2, sc_fin2 = lp_fictitious_play(best_seq, rem * 0.8, top_k=50, verbose=False)
        best_seq, best_score = check_and_update(a_fin2, "final_lpfp", best_seq, best_score)
    
    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification with the actual evaluator
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    
    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*60}")
    
    return [float(x) for x in best_seq]
