```python
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

    """Smooth approximation of max using log-sum-exp."""
    x_max = x.max()
    return x_max + np.log(np.sum(np.exp(beta * (x - x_max)))) / beta

    """Gradient of softmax_max with respect to x."""
    x_max = x.max()
    y = np.exp(beta * (x - x_max))
    w = y / y.sum()
    return w  # d(softmax_max)/d(x[k]) = w[k]
```

```python
"""
Gradient-based optimization with softmax smoothing, FFT-accelerated gradients,
coordinate descent polishing, random restarts, and length adaptation.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_grad_via_fft(f, w):
    """
    Compute grad[i] = 2 * sum_{j=0}^{n-1} w[i+j] * f[j]
    using FFT (cross-correlation trick).
    w has length 2n-1, f has length n.
    """
    n = len(f)
    f_rev = f[::-1]
    # g[i] = (w * f_rev)[i + n - 1] for i = 0, ..., n-1
    conv_len = len(w) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    F = rfft(f_rev, fft_size)
    conv_result = irfft(W * F, fft_size)[:conv_len]
    # Extract g[i] = conv_result[i + n - 1]
    g = conv_result[n - 1: 2 * n - 1]
    return 2.0 * g


def softmax_approx_max(x, beta):
    """Smooth max approximation via log-sum-exp."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return x_max + log_sum / beta


def softmax_weights(x, beta):
    """Get softmax weights (gradient of smooth max w.r.t. x)."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -50, 50)
    y = np.exp(shifted)
    return y / y.sum()


def project_simplex(v, s=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(s, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - s)
    if not cond.any():
        return np.full(n, s / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def objective(f):
    """Compute the objective value (2*n*max_b/S^2)."""
    n = len(f)
    S = f.sum()
    if S < 1e-15:
        return np.inf
    b = fast_convolve(f)
    max_b = b.max()
    return 2.0 * n * max_b / (S * S)


def smoothed_objective(f, beta):
    """Compute smoothed objective and gradient."""
    n = len(f)
    S = f.sum()
    if S < 1e-15:
        return np.inf, np.zeros(n)
    b = fast_convolve(f)
    smax = softmax_approx_max(b, beta)
    w = softmax_weights(b, beta)
    # Gradient of smooth max w.r.t. f
    grad_conv = compute_grad_via_fft(f, w)
    # Full gradient of 2*n*smax/S^2
    grad = (2.0 * n / (S * S)) * (grad_conv - 2.0 * smax / S)
    obj = 2.0 * n * smax / (S * S)
    return obj, grad


def gradient_descent(f_init, time_limit, beta_init=50.0, lr_init=0.01,
                     label="gd", n_max=0, seed=42):
    """
    Projected gradient descent with softmax smoothing and adaptive parameters.
    f_init is normalized to sum=1.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f /= f.sum()
    best_f = f.copy()
    best_obj = objective(f)
    lr = lr_init
    beta = beta_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)
    momentum = np.zeros(n)
    mu = 0.7  # momentum coefficient

    while time.monotonic() - t0 < time_limit:
        it += 1
        obj, grad = smoothed_objective(f, beta)

        # Momentum update
        momentum = mu * momentum - lr * grad
        f_new = f + momentum
        f_new = project_simplex(f_new, 1.0)

        # Backtracking line search (simplified)
        best_trial_obj = objective(f)
        best_trial_f = f.copy()
        f_new2 = f + 0.5 * momentum
        f_new2 = project_simplex(f_new2, 1.0)
        obj2 = objective(f_new2)
        if obj2 < best_trial_obj:
            best_trial_obj = obj2
            best_trial_f = f_new2
        f_new3 = f + 0.25 * momentum
        f_new3 = project_simplex(f_new3, 1.0)
        obj3 = objective(f_new3)
        if obj3 < best_trial_obj:
            best_trial_obj = obj3
            best_trial_f = f_new3

        f = best_trial_f
        cur_obj = best_trial_obj

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
            # Increase learning rate slightly on progress
            lr = min(lr * 1.02, 0.05)
        else:
            no_improve += 1
            if no_improve >= 5:
                lr *= 0.9
                no_improve = 0
            if no_improve >= 20:
                # Restart with perturbation
                f = best_f + rng.standard_normal(n) * 0.01
                f = np.maximum(f, 0)
                f /= f.sum()
                momentum = np.zeros(n)
                lr = lr_init
                beta = beta_init
                no_improve = 0

        # Slowly increase beta (make max sharper)
        if it % 200 == 0:
            beta = min(beta * 1.3, 500.0)

        if it % 500 == 0:
            print(f"    {label} iter={it} obj={best_obj:.10f} score={best_obj:.6f} "
                  f"beta={beta:.1f} lr={lr:.5f} t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def coordinate_descent(f_init, time_limit, label="cd", n_iterations=100):
    """Coordinate descent on individual elements."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f /= f.sum()
    best_obj = objective(f)
    multipliers = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0]
    passed = 0

    while time.monotonic() - t0 < time_limit and passed < n_iterations:
        passed += 1
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-12:
                # Try setting a small value
                for val in [0.001, 0.005, 0.01]:
                    f_trial = f.copy()
                    f_trial[j] = val
                    f_trial /= f_trial.sum()
                    obj = objective(f_trial)
                    if obj < best_obj - 1e-12:
                        f = f_trial
                        best_obj = obj
                        improved = True
                        break
                continue
            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = objective(f_trial)
                if obj < best_j_obj - 1e-12:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                f /= f.sum()
                if best_j_obj < best_obj - 1e-12:
                    best_obj = best_j_obj
                    improved = True
        if not improved:
            break

    return f, best_obj


def subgradient_step(f, beta, lr, n_top=None):
    """One subgradient step with top-k peak selection."""
    n = len(f)
    b = fast_convolve(f)
    S = f.sum()

    if n_top is not None and n_top < len(b):
        # Focus on top peaks
        top_idx = np.argpartition(b, -n_top)[-n_top:]
        w = np.zeros(len(b))
        # Softmax weights on top peaks
        b_top = b[top_idx]
        b_max = b_top.max()
        shifted = beta * (b_top - b_max)
        shifted = np.clip(shifted, -50, 50)
        w_top = np.exp(shifted)
        w_top /= w_top.sum()
        w[top_idx] = w_top
    else:
        w = softmax_weights(b, beta)

    # Compute gradient
    grad_conv = compute_grad_via_fft(f, w)
    smax = softmax_approx_max(b, beta)
    grad = (2.0 * n / (S * S)) * (grad_conv - 2.0 * smax / S)

    f_new = f - lr * grad
    f_new = project_simplex(f_new, 1.0)
    return f_new


def subgradient_descent(f_init, time_limit, lr_init=0.005, beta=30.0,
                        n_top=50, label="sg", seed=123):
    """Subgradient descent with peak focusing."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f /= f.sum()
    best_f = f.copy()
    best_obj = objective(f)
    lr = lr_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        it += 1
        f = subgradient_step(f, beta, lr, n_top)
        cur_obj = objective(f)

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
            lr = min(lr * 1.01, 0.03)
        else:
            no_improve += 1
            if no_improve >= 10:
                lr *= 0.9
                no_improve = 0
            if no_improve >= 40:
                f = best_f + rng.standard_normal(n) * 0.005
                f = np.maximum(f, 0)
                f /= f.sum()
                lr = lr_init
                no_improve = 0

        if it % 500 == 0:
            print(f"    {label} iter={it} obj={best_obj:.10f} "
                  f"lr={lr:.5f} beta={beta:.1f} t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    n = len(f)
    if new_n <= 0:
        return f
    if new_n == n:
        return f.copy()
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def random_init(n, seed=0, dtype='uniform'):
    """Generate a random initial sequence."""
    rng = np.random.default_rng(seed)
    if dtype == 'uniform':
        f = np.ones(n)
    elif dtype == 'random':
        f = rng.random(n)
    elif dtype == 'two_peak':
        f = np.zeros(n)
        peak1 = n // 4
        peak2 = 3 * n // 4
        width = max(1, n // 20)
        f[max(0, peak1-width):peak1+width] = 1.0
        f[max(0, peak2-width):peak2+width] = 1.0
    elif dtype == 'gaussian_center':
        x = np.linspace(0, 1, n)
        f = np.exp(-20 * (x - 0.5) ** 2)
    elif dtype == 'triangular':
        x = np.linspace(0, 1, n)
        f = 1 - np.abs(2 * x - 1)
    else:
        f = rng.random(n)
    f = np.maximum(f, 0)
    f /= f.sum()
    return f


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main optimization: gradient descent with softmax smoothing,
    coordinate descent, subgradient methods, random restarts, and length adaptation.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    # Get parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    best_seq = parent.copy() if parent is not None else np.ones(1152) / 1152
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"Initial best: score={best_score:.6f}, n={len(best_seq)}")

    # Phase 1: Quick exploration of different initializations and lengths
    rem = deadline - time.monotonic()
    print(f"\n=== Phase 1: Exploration (budget={rem:.0f}s) ===")
    init_budget = min(rem * 0.08, 80.0)
    if init_budget > 10:
        n_base = len(best_seq) if parent is not None else 1152
        candidates = []

        # Try different lengths and initializations
        lengths = [max(512, n_base + d) for d in [-200, -100, 0, 100, 200, 400] if n_base + d >= 256]
        lengths = [l for l in lengths if l <= 3000]
        if len(set(lengths)) < 4:
            for l in [600, 800, 1000, 1200, 1400, 1600]:
                if l not in lengths:
                    lengths.append(l)
        lengths = sorted(set(lengths))

        for n_try in lengths[:6]:
            if time.monotonic() > deadline - init_budget - 5:
                break
            for dtype in ['uniform', 'random', 'gaussian_center']:
                f_try = random_init(n_try, seed=seed + n_try + hash(dtype) % 1000, dtype=dtype)
                score_try = objective(f_try)
                candidates.append((score_try, f_try, n_try, dtype))

        # Sort by score and keep top ones
        candidates.sort(key=lambda x: x[0])
        print(f"  Generated {len(candidates)} candidates, best={candidates[0][0]:.6f}")

        # Try the best few with a quick optimization
        for score_c, f_c, n_c, dtype_c in candidates[:4]:
            if time.monotonic() > deadline - init_budget - 10:
                break
            t_cur = min(15.0, (deadline - time.monotonic()) * 0.1)
            if t_cur < 3:
                break
            f_opt, obj_opt = gradient_descent(f_c, time_limit=t_cur,
                                              beta_init=30, lr_init=0.008,
                                              label=f"exp_{n_c}", seed=seed)
            vs = float(evaluate_sequence(f_opt.tolist()))
            if np.isfinite(vs) and vs < best_score:
                best_score = vs
                best_seq = f_opt.copy()
                print(f"  NEW BEST [explore {n_c} {dtype_c}]: {vs:.6f}")

    # Phase 2: Main gradient descent on best candidate
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.30, 400.0)
    if t2 > 30:
        print(f"\n=== Phase 2: Main GD (n={len(best_seq)}, budget={t2:.0f}s) ===")
        f2, obj2 = gradient_descent(best_seq, time_limit=t2,
                                    beta_init=40, lr_init=0.01,
                                    label="gd_main", seed=seed)
        vs2 = float(evaluate_sequence(f2.tolist()))
        if np.isfinite(vs2) and vs2 < best_score:
            best_score = vs2
            best_seq = f2.copy()
            print(f"  NEW BEST [gd_main]: {vs2:.6f}")

    # Phase 3: Subgradient with peak focusing
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.20, 250.0)
    if t3 > 20:
        print(f"\n=== Phase 3: Subgradient (n={len(best_seq)}, budget={t3:.0f}s) ===")
        f3, obj3 = subgradient_descent(best_seq, time_limit=t3,
                                       lr_init=0.004, beta=25.0,
                                       n_top=40, label="sg", seed=seed+1)
        vs3 = float(evaluate_sequence(f3.tolist()))
        if np.isfinite(vs3) and vs3 < best_score:
            best_score = vs3
            best_seq = f3.copy()
            print(f"  NEW BEST [sg]: {vs3:.6f}")

    # Phase 4: Coordinate descent polishing
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.15, 180.0)
    if t4 > 15:
        print(f"\n=== Phase 4: Coordinate Descent (budget={t4:.0f}s) ===")
        f4, obj4 = coordinate_descent(best_seq, time_limit=t4, label="cd1")
        vs4 = float(evaluate_sequence(f4.tolist()))
        if np.isfinite(vs4) and vs4 < best_score:
            best_score = vs4
            best_seq = f4.copy()
            print(f"  NEW BEST [cd1]: {vs4:.6f}")

    # Phase 5: Random restarts with re-optimization
    rem = deadline - time.monotonic()
    t5_total = min(rem * 0.20, 250.0)
    if t5_total > 30:
        print(f"\n=== Phase 5: Random restarts (budget={t5_total:.0f}s) ===")
        n_best = len(best_seq)
        n_restarts = max(3, int(t5_total / 50))
        seed_base = seed + 100
        for r in range(n_restarts):
            if time.monotonic() > deadline - 30:
                break
            t_r = min(40.0, (deadline - time.monotonic()) / max(1, n_restarts - r))
            if t_r < 10:
                break

            # Perturb best solution
            perturbation_scale = [0.05, 0.1, 0.2, 0.15, 0.08][r % 5]
            rng = np.random.default_rng(seed_base + r)
            f_pert = best_seq + rng.standard_normal(n_best) * perturbation_scale
            f_pert = np.maximum(f_pert, 0)
            f_pert /= f_pert.sum()

            f_r, obj_r = gradient_descent(f_pert, time_limit=t_r,
                                          beta_init=30, lr_init=0.006,
                                          label=f"restart_{r}", seed=seed_base + r)
            vs_r = float(evaluate_sequence(f_r.tolist()))
            if np.isfinite(vs_r) and vs_r < best_score:
                best_score = vs_r
                best_seq = f_r.copy()
                print(f"  NEW BEST [restart {r}]: {vs_r:.6f}")

    # Phase 6: Length adaptation
    rem = deadline - time.monotonic()
    if rem > 60:
        print(f"\n=== Phase 6: Length adaptation (budget={rem:.0f}s) ===")
        n_best = len(best_seq)
        target_lengths = []
        for factor in [0.8, 0.9, 1.1, 1.2, 1.4]:
            tl = int(n_best * factor)
            tl = max(400, min(2500, tl))
            if tl != n_best and tl not in target_lengths:
                target_lengths.append(tl)
        target_lengths.sort(key=lambda x: abs(x - n_best))

        for target_n in target_lengths[:3]:
            if time.monotonic() > deadline - 30:
                break
            t_t = min(60.0, (deadline - time.monotonic()) * 0.3)
            if t_t < 15:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            f_t, obj_t = gradient_descent(f_interp, time_limit=t_t,
                                          beta_init=35, lr_init=0.008,
                                          label=f"len_{target_n}", seed=seed + target_n)
            vs_t = float(evaluate_sequence(f_t.tolist()))
            if np.isfinite(vs_t) and vs_t < best_score:
                best_score = vs_t
                best_seq = f_t.copy()
                print(f"  NEW BEST [len {target_n}]: {vs_t:.6f}")

    # Phase 7: Final polishing - alternate GD and CD
    rem = deadline - time.monotonic()
    if rem > 20:
        print(f"\n=== Phase 7: Final polish (budget={rem:.0f}s) ===")
        # GD
        t7a = min(rem * 0.4, 100.0)
        if t7a > 10:
            f7a, obj7a = gradient_descent(best_seq, time_limit=t7a,
                                          beta_init=60, lr_init=0.003,
                                          label="gd_final", seed=seed+999)
            vs7a = float(evaluate_sequence(f7a.tolist()))
            if np.isfinite(vs7a) and vs7a < best_score:
                best_score = vs7a
                best_seq = f7a.copy()
                print(f"  NEW BEST [gd_final]: {vs7a:.6f}")

        # CD
        rem2 = deadline - time.monotonic()
        t7b = min(rem2 * 0.5, 80.0)
        if t7b > 10:
            f7b, obj7b = coordinate_descent(best_seq, time_limit=t7b, label="cd_final")
            vs7b = float(evaluate_sequence(f7b.tolist()))
            if np.isfinite(vs7b) and vs7b < best_score:
                best_score = vs7b
                best_seq = f7b.copy()
                print(f"  NEW BEST [cd_final]: {vs7b:.6f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        # Fallback to parent
        if parent is not None:
            best_seq = parent.copy()
            best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
```
