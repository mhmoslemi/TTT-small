Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. L-BFGS with continuation (geometric temperature cooling in log-space)
2. LP fictitious play with geometric mean updates (greedy LP)
3. Coordinate descent with golden-section search
4. Length exploration near the parent

import numpy as np

from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def softmax_from_z(z):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-30))


def smoothed_obj_grad(z, T):
    """Same as above but returns (obj, grad)."""
    n = len(z)
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    scaled = conv / T
    shifted = scaled - scaled.max()
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    lse = np.log(exp_shifted.sum())
    full_lse = conv.max() / T + lse
    obj = 2.0 * n * T * full_lse
    grad_a = np.zeros(n)
    for j in range(n):
        grad_a[j] = 2.0 * np.dot(w[j:j+n], a)
    grad_a_full = 2.0 * n * grad_a
    S = np.dot(grad_a_full, a)
    grad_z = a * (grad_a_full - S)
    return obj, grad_z

def greedy_lp(f, M):
    """Greedy approximation to the LP: max sum(g) s.t. C*g <= M, g >= 0."""
    n = len(f)
    g = np.zeros(n)
    conv_vals = np.zeros(2 * n - 1)
    
    for j in range(n):
        k_start = j
        k_end = j + n - 1
        slack = M - conv_vals[k_start:k_end + 1]
        f_slice = f  # f[k-j] for k in [j, j+n-1] is f[0], f[1], ..., f[n-1]
        max_gj = np.inf
        for idx in range(n):
            if f_slice[idx] > 1e-15:
                cand = slack[idx] / f_slice[idx]
                if cand < max_gj:
                    max_gj = cand
        max_gj = max(0.0, max_gj)
        g[j] = max_gj
        conv_vals[k_start:k_end + 1] += f_slice * max_gj
    
    return g


def geometric_mean_update(f, g, t):
    f_safe = np.maximum(f, 1e-30)
    g_safe = np.maximum(g, 1e-30)
    log_new = (1 - t) * np.log(f_safe) + t * np.log(g_safe)
    log_new -= log_new.max()
    a = np.exp(log_new)
    s = a.sum()
    if s < 1e-30:
        return f.copy()
    return a / s
