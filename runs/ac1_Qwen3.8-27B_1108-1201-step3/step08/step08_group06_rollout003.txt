def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the optimization score for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_sparse_C(n):
    """Build sparse convolution constraint matrix C where C[k,j] = f[k-j]."""
    # We'll build this with values; structure is banded
    rows = []
    cols = []
    # C has 2n-1 rows and n cols
    # C[k, j] = 1 if we're going to multiply by f[k-j] later
    # Actually for the LP we need C[k,j] * g[j] = sum_j f[k-j]*g[j]
    # So the matrix depends on f. Let's build it with f values.
    return None  # Will build inline


def build_C_for_f(f):
    """Build the (2n-1) x n matrix C where (C @ g)[k] = (f*g)[k]."""
    n = len(f)
    rows = []
    cols = []
    data = []
    for j in range(n):
        # g[j] contributes to (f*g)[k] for k = j, j+1, ..., j+n-1
        # with coefficient f[k-j] = f[0], f[1], ..., f[n-1]
        for idx in range(n):
            k = j + idx
            data.append(f[idx])
            rows.append(k)
            cols.append(j)
    C = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return C


def solve_lp_for_f(f, M):
    """Solve LP: max sum(g) s.t. (f*g)(k) <= M for all k, g >= 0."""
    n = len(f)
    C = build_C_for_f(f)
    c = -np.ones(n)
    A_ub = C.toarray()  # Convert for linprog (highs handles dense fine for n<2000)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if res.status != 0 or res.x is None:
        return None
    return np.maximum(res.x, 0)


def solve_lp_cvxpy(f, M, solver='ECOS'):
    """Solve LP using cvxpy."""
    import cvxpy as cp
    n = len(f)
    g_var = cp.Variable(n, nonneg=True)
    # (f * g)(k) = sum_j f[k-j] * g[j] for valid k-j
    # Build as: for each k, the constraint is sum_j f[k-j]*g[j] <= M
    # This is C @ g_var <= M
    C = build_C_for_f(f)
    # Use dense for cvxpy
    constraints = [C.toarray() @ g_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver='SCS')
        except Exception:
            return None
    if g_var.value is None:
        return None
    return np.maximum(g_var.value, 0)


def greedy_lp(f, M):
    """Greedy approximation to the LP."""
    n = len(f)
    g = np.zeros(n)
    conv_vals = np.zeros(2 * n - 1)
    for j in range(n):
        k_start = j
        k_end = j + n - 1
        slack = M - conv_vals[k_start:k_end + 1]
        valid = f > 1e-15
        if not valid.any():
            g[j] = 0
            continue
        max_gj = np.min(slack[valid] / f[valid])
        max_gj = max(0.0, max_gj)
        g[j] = max_gj
        conv_vals[k_start:k_end + 1] += f * max_gj
    return g


def geometric_mean_update(f, g, t):
    """Geometric mean interpolation in log-space."""
    f_safe = np.maximum(f, 1e-30)
    g_safe = np.maximum(g, 1e-30)
    log_new = (1 - t) * np.log(f_safe) + t * np.log(g_safe)
    log_new -= log_new.max()
    a = np.exp(log_new)
    s = a.sum()
    if s < 1e-30:
        return f.copy()
    return a / s


def softmax_from_z(z):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def lbgfs_softmax(a0, time_limit, T, maxiter=5000):
    """L-BFGS in softmax (log-linear) parameterization with smooth max."""
    n = len(a0)
    z0 = np.log(np.maximum(a0, 1e-30))
    z0 -= z0.mean()  # Center for numerical stability
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)
    
    obj_count = [0]
    
    def obj_grad(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15, np.zeros_like(z)
        obj_count[0] += 1
        a = softmax_from_z(z)
        conv = fast_convolve(a)
        shifted = conv / T
        shifted_max = shifted.max()
        w = np.exp(shifted - shifted_max)
        w = w / w.sum()
        log_sum_exp = shifted_max + np.log(w.sum())
        obj_val = 2.0 * n * T * log_sum_exp
        
        # Gradient: 2n * sum_k w[k] * d(conv[k])/da
        # d(conv[k])/da[j] = a[k-j] (for valid k-j in [0,n-1])
        # So grad_a[j] = 2n * sum_k w[k] * a[k-j]
        # This is 2n * (w * a)(j) = 2n * cross-correlation of w and a
        # Actually: grad_a[j] = 2n * sum_k w[k] * a[k-j]
        # Let's compute via FFT: this is the cross-correlation of w and a
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        # cross-correlation: sum_k w[k]*a[k-j] = (w ⊛ a)[j]
        # In FFT: IFFT(conj(W) * A) gives cross-correlation
        grad_a = 2.0 * n * irfft(np.conj(W) * A, fft_size)[:n]
        
        # Project to log-space (softmax gradient)
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        
        return obj_val, grad_z
    
    try:
        res = minimize(
            fun=lambda z: obj_grad(z)[0],
            x0=z0,
            jac=lambda z: obj_grad(z)[1],
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                     'maxfun': maxiter * 10, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        score = compute_score(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception:
        pass
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

        print(f"Initialized: n={len(parent)}, score={best_score:.8f}")
    
    n_parent = len(parent)
    
    # ============ Phase 1: LP Fictitious Play (400s) ============
    rem = deadline - time.monotonic()
    t1 = min(400.0, rem * 0.25)
    print(f"\n=== Phase 1: LP Fictitious Play (budget={t1:.0f}s), best={best_score:.8f} ===")
    if t1 > 30:
        a1, s1 = lp_fictitious_play(best_seq, t1, use_greedy=False, verbose=True)
        best_seq, best_score = check_candidate(a1, best_seq, best_score, "lp1")
    
    # ============ Phase 2: Multi-temperature L-BFGS Annealing (450s) ============
    rem = deadline - time.monotonic()
    t2 = min(450.0, rem * 0.3)
    print(f"\n=== Phase 2: L-BFGS Annealing (budget={t2:.0f}s), best={best_score:.8f} ===")
    if t2 > 30:
        temps = [5000, 2000, 1000, 500, 200, 100, 50, 20]
        # Distribute time across temperatures
        n_temps = len(temps)
        per_temp = t2 / n_temps
        current = best_seq.copy()
        for ti, T_val in enumerate(temps):
            rem_t = deadline - time.monotonic() - 100
            if rem_t < 15:
                break
            t_alloc = min(per_temp, rem_t * 0.5)
            if t_alloc < 10:
                t_alloc = max(5, rem_t * 0.3)
            print(f"  T={T_val}, budget={t_alloc:.0f}s")
            a_t, s_t = lbgfs_softmax(current, t_alloc, T=T_val, maxiter=3000)
            if s_t < best_score:
                best_score = s_t
                best_seq = a_t.copy()
                current = a_t.copy()
                print(f"  Improved: {s_t:.8f}")
            else:
                current = a_t.copy()  # Still use as next init
            # Verify with evaluate_sequence
            best_seq, best_score = check_candidate(current, best_seq, best_score, f"lbgfs_T{T_val}")
    
    # ============ Phase 3: Coordinate Descent (300s) ============
    rem = deadline - time.monotonic()
    t3 = min(300.0, rem * 0.25)
    print(f"\n=== Phase 3: Coordinate Descent (budget={t3:.0f}s), best={best_score:.8f} ===")
    if t3 > 20:
        a3, s3 = coordinate_descent(best_seq, t3, verbose=True)
        best_seq, best_score = check_candidate(a3, best_seq, best_score, "cd")
    
    # ============ Phase 4: Peak Shaving (150s) ============
    rem = deadline - time.monotonic()
    t4 = min(150.0, rem * 0.15)
    print(f"\n=== Phase 4: Peak Shaving (budget={t4:.0f}s), best={best_score:.8f} ===")
    if t4 > 20:
        a4, s4 = peak_shaving(best_seq, t4, verbose=False)
        best_seq, best_score = check_candidate(a4, best_seq, best_score, "peak_shave")
    
    # ============ Phase 5: Second LP pass on refined solution (150s) ============
    rem = deadline - time.monotonic()
    t5 = min(150.0, rem * 0.15)
    print(f"\n=== Phase 5: LP Polish (budget={t5:.0f}s), best={best_score:.8f} ===")
    if t5 > 20:
        a5, s5 = lp_fictitious_play(best_seq, t5, use_greedy=True, verbose=False)
        best_seq, best_score = check_candidate(a5, best_seq, best_score, "lp_polish")
    
    # ============ Phase 6: Length Exploration (200s) ============
    rem = deadline - time.monotonic()
    t6 = min(200.0, rem * 0.2)
    print(f"\n=== Phase 6: Length Exploration (budget={t6:.0f}s), best={best_score:.8f} ===")
    if t6 > 30:
        alt_ns = [n_parent - 4, n_parent - 2, n_parent + 2, n_parent + 4,
                  n_parent - 8, n_parent + 8,
                  n_parent - 16, n_parent + 16]
        alt_ns = [x for x in alt_ns if 64 <= x <= 2000]
        per_len = t6 / max(1, len(alt_ns))
        for alt_n in alt_ns:
            rem_t = deadline - time.monotonic() - 50
            if rem_t < 15:
                break
            t_alloc = min(per_len, rem_t * 0.4)
            if t_alloc < 10:
                continue
            init_alt = interpolate_sequence(best_seq, alt_n)
            # Quick L-BFGS
            a_alt, s_alt = lbgfs_softmax(init_alt, t_alloc * 0.6, T=500, maxiter=2000)
            best_seq, best_score = check_candidate(a_alt, best_seq, best_score, f"n{alt_n}")
            # Quick LP
            rem_t2 = deadline - time.monotonic() - 40
            if rem_t2 > 15:
                a_alt2, s_alt2 = lp_fictitious_play(a_alt, min(rem_t2 * 0.3, 30.0), use_greedy=True, verbose=False)
                best_seq, best_score = check_candidate(a_alt2, best_seq, best_score, f"n{alt_n}_lp")
    
    # ============ Phase 7: Final L-BFGS polish at best length ============
    rem = deadline - time.monotonic()
    t7 = min(80.0, rem * 0.4)
    print(f"\n=== Phase 7: Final Polish (budget={t7:.0f}s), best={best_score:.8f} ===")
    if t7 > 10:
        a7, s7 = lbgfs_softmax(best_seq, t7 * 0.5, T=50, maxiter=2000)
        best_seq, best_score = check_candidate(a7, best_seq, best_score, "final_lbfgs")
        
        rem_t = deadline - time.monotonic() - 20
        if rem_t > 10:
            a7b, s7b = coordinate_descent(best_seq, rem_t * 0.6, verbose=False)
            best_seq, best_score = check_candidate(a7b, best_seq, best_score, "final_cd")
    
    # ============ Final verification ============
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    
    print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
