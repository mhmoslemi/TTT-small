def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return np.array([])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)

    else:

"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return np.array([])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def compute_subgradient_fft(a, conv, peak_fraction=0.15, temperature=0.05):
    """
    Compute approximate subgradient of max_k conv[k] using FFT.
    Uses weighted peak indexing with softmax over peaks above threshold.
    """
    n = len(a)
    conv_len = 2 * n - 1

    # Find peaks: positions with conv[k] above threshold
    threshold = np.percentile(conv, 100.0 * (1.0 - peak_fraction))
    peak_mask = conv >= threshold
    peak_indices = np.where(peak_mask)[0]

    if len(peak_indices) == 0:
        peak_indices = np.array([np.argmax(conv)])

    # Softmax weighting over peaks
    peak_vals = conv[peak_indices]
    temp = temperature * max(peak_vals.max() - peak_vals.min(), 1e-12)
    w = np.exp((peak_vals - peak_vals.max()) / temp)
    w = w / w.sum()

    # Build weight vector
    w_vec = np.zeros(conv_len)
    w_vec[peak_indices] = w

    # Subgradient via FFT: grad[i] = 2 * sum_k w_vec[k] * a[k-i] for valid k-i
    # = 2 * (w_vec * a_rev)[n-1 : n-1+n]
    a_rev = a[::-1].copy()
    # Compute linear convolution of w_vec (length 2n-1) and a_rev (length n)
    total_len = conv_len + n - 1
    fft_sz = 1 << (total_len - 1).bit_length()
    W = rfft(w_vec, fft_sz)
    AR = rfft(a_rev, fft_sz)
    conv_wr = irfft(W * AR, fft_sz)[:total_len]

    grad = 2.0 * conv_wr[n - 1: n - 1 + n]
    return grad


def subgradient_method(a0, time_limit, peak_fraction=0.15, temperature=0.05,
                       momentum_init=0.5, verbose=False):
    """
    Projected subgradient method with Nesterov momentum and adaptive step size.
    Optimizes on the simplex (sum(a)=1, a>=0).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    momentum = momentum_init
    alpha = 1.0 / np.sqrt(n)  # initial step size
    alpha_min = 1e-8
    alpha_max = 0.5
    y_prev = a.copy()  # for momentum
    no_improve = 0
    patience = 500

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        # Nesterov lookahead
        y = a + momentum * (a - y_prev)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        # Compute subgradient at y
        conv_y = fast_convolve(y)
        g = compute_subgradient_fft(y, conv_y, peak_fraction, temperature)

        # Adaptive step size: decrease if no improvement, increase slightly if improving
        score_y = 2.0 * n * conv_y.max()
        if score_y < best_score - 1e-12:
            alpha = min(alpha * 1.02, alpha_max)
        else:
            alpha = max(alpha * 0.97, alpha_min)

        # Update
        a_new = simplex_project(a - alpha * g)

        # Track best
        conv_new = fast_convolve(a_new)
        score_new = 2.0 * n * conv_new.max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adjust momentum
        if no_improve > patience:
            momentum = max(0.3, momentum * 0.95)
            alpha = max(alpha * 0.8, alpha_min)
            no_improve = 0
            if alpha < 1e-7:
                break
        elif no_improve > patience // 2:
            momentum = max(0.3, momentum * 0.99)

        y_prev = a
        a = a_new

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad it={it} ({elapsed:.0f}s): best={best_score:.6f} alpha={alpha:.6f} mom={momentum:.3f}")

    return best_a, best_score


def exact_line_search_coord(a, i, conv, n_trials=20):
    """
    For coordinate i, find the best update using exact evaluation.
    Returns (best_delta, best_new_max, best_new_score).
    Assumes sum(a) = 1.
    """
    n = len(a)
    k_start = i
    k_end = min(i + n - 1, 2 * n - 2)

    # Unaffected max
    if k_start > 0:
        unaff_left = conv[:k_start].max()
    else:
        unaff_left = -1e30
    if k_end < 2 * n - 2:
        unaff_right = conv[k_end + 1:].max()
    else:
        unaff_right = -1e30
    unaff_max = max(unaff_left, unaff_right)

    # Affected values
    affected = conv[k_start:k_end + 1].copy()
    # a_shifted[m] = a[m] where the affected conv[k_start+m] gets contribution 2*delta*a[m]
    a_shifted = a[0:n].copy()

    v = a[i]
    best_delta = 0.0
    best_new_max = conv.max()

    # Try a range of step sizes (both directions)
    # Use log-spaced steps for efficiency
    for trial in range(n_trials):
        if trial == 0:
            t = 0.0
        elif trial <= n_trials // 2:
            # Negative steps (reduce a[i])
            frac = trial / (n_trials // 2)
            t = -0.95 * v * frac
        else:
            # Positive steps (increase a[i])
            frac = (trial - n_trials // 2) / (n_trials // 2)
            t = 0.95 * frac  # can increase significantly

        new_val = v + t
        if new_val < -1e-15:
            continue

        # Compute new affected
        new_affected = affected + 2.0 * t * a_shifted
        if 2 * i >= k_start and 2 * i <= k_end:
            idx_2i = 2 * i - k_start
            new_affected[idx_2i] = conv[2 * i] + 2.0 * t * v + t * t

        new_max = max(unaff_max, float(new_affected.max()))
        if new_max < best_new_max - 1e-16:
            best_new_max = new_max
            best_delta = t

    return best_delta, best_new_max

def fourier_perturb(a0, rng, scale=0.1):
    """
    Perturb in the Fourier domain to preserve the envelope but change fine structure.
    """
    n = len(a0)
    a = a0.copy()

    # Get FFT of the sequence
    A = rfft(a)

    # Perturb mid-to-high frequency components (preserve low-freq envelope)
    n_keep = max(2, n // 20)  # keep first n_keep bins unchanged
    for k in range(n_keep, len(A)):
        if rng.random() < 0.5:
            # Multiply by a random complex factor with magnitude near 1
            mag = 1.0 + scale * rng.standard_normal()
            phase = scale * rng.standard_normal()
            A[k] *= mag + 1j * phase

    # Inverse FFT
    a_new = irfft(A, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a0.copy()
    return a_new / s
