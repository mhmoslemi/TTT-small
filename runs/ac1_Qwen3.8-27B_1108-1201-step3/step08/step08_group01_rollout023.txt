def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a_fixed):
    """Build matrix T such that T @ b = conv(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    if '_lp_solver' in globals() and _lp_solver_returned is not None:
        return _lp_solver_returned
    _lp_solver_returned = cp.ECOS
    for solver_name in [cp.SCS, cp.ECOS, cp.PDLP, cp.CBC, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_returned = solver_name
                return solver_name
        except Exception:
            continue
    return cp.ECOS


_lp_solver_returned = None


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (sum=1, nonneg), find b minimizing max conv(a,b) subject to sum(b)=1, b>=0."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.ECOS, cp.PDLP, cp.CBC]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def coord_descent_bisection(a0, time_limit, label="", n_bisect=12, verbose=False):
    """
    Coordinate descent exploiting 1D convexity.
    For each coordinate, the max-conv as a function of that coordinate (with sum normalized)
    is convex, so we use bisection on the subgradient to find the optimum.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                # Try small positive values
                candidates = [1e-4, 5e-4, 1e-3]
            else:
                # Bisection range: [0, 2*a_i]
                candidates = None

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            # Affected range: conv[k_start:k_end+1]
            # For coordinate i, affected conv indices are [i, i+n-1]
            # conv_k changes by: 2*(x-a_i)*a_{k-i} for k != 2i, (x^2 - a_i^2) for k=2i
            a_shifted = a[0:n]  # a[k-i] for k in [i, i+n-1], i.e., a[0], a[1], ..., a[n-1]

            if a[i] < 1e-15:
                # Try a few small values
                best_x = 0.0
                best_M = M
                for x_try in [1e-5, 5e-5, 1e-4, 5e-4, 1e-3]:
                    delta = x_try
                    new_affected = conv[k_start:k_end + 1] + 2.0 * delta * a_shifted
                    if 2 * i >= k_start and 2 * i <= k_end:
                        idx = 2 * i - k_start
                        new_affected = new_affected.copy()
                        new_affected[idx] = conv[2 * i] + 2.0 * delta * 0 + delta * delta
                    new_M = max(unaff_max, float(new_affected.max()))
                    # After update, sum changes by delta, renormalize
                    new_sum = 1.0 + delta
                    new_score_candidate = 2.0 * n * new_M / (new_sum * new_sum)
                    if new_score_candidate < best_score:
                        best_M = new_M
                        best_x = x_try
                if best_x > 0:
                    a[i] = best_x
                    sum_a = a.sum()
                    a = a / sum_a
                    conv = fast_convolve(a)
                    M = float(conv.max())
                    cur_score = 2.0 * n * M
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()
                    improved = True
                continue

            # Bisection to find optimal x in [0, 2*a_i]
            lo = 0.0
            hi = 2.0 * a[i]
            current_x = a[i]

            # Define g(x) = resulting max conv after setting a_i=x and renormalizing
            def eval_g(x):
                """Evaluate max conv after setting a_i = x and renormalizing."""
                delta = x - a[i]
                if abs(delta) < 1e-15:
                    return M
                # New affected values
                new_aff = conv[k_start:k_end + 1] + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_aff = new_aff.copy()
                    new_aff[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
                new_M = max(unaff_max, float(new_aff.max()))
                # Renormalize: new sum = 1 + delta
                new_sum = 1.0 + delta
                if new_sum < 1e-10:
                    return 1e30
                # Score contribution (we compare 2n*new_M/new_sum^2)
                return new_M / (new_sum * new_sum)

            # Bisection: find x that minimizes eval_g(x) on [lo, hi]
            # Since eval_g is convex, use ternary search
            for _ in range(n_bisect):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                g1 = eval_g(m1)
                g2 = eval_g(m2)
                if g1 < g2:
                    hi = m2
                else:
                    lo = m1

            opt_x = (lo + hi) / 2.0
            opt_val = eval_g(opt_x)
            cur_val = eval_g(a[i])

            if opt_val < cur_val * (1.0 - 1e-12) and opt_x > 1e-15:
                # Apply update
                a[i] = opt_x
                sum_a = a.sum()
                if sum_a < 1e-10:
                    continue
                a = a / sum_a
                conv = fast_convolve(a)
                M = float(conv.max())
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 10:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f} M={M:.8f}")

    return best_a, best_score, sweeps


def coord_descent_fast(a0, time_limit, label="", n_trials=8, verbose=False):
    """Fast coordinate descent with fixed trial steps (for quick polishing)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            a_shifted = a[0:n]
            affected = conv[k_start:k_end + 1]

            best_delta = 0.0
            best_new_M = M

            # Try various fractions
            for factor in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_aff = affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_aff = new_aff.copy()
                    new_aff[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_M = max(unaff_max, float(new_aff.max()))
                if new_M < best_new_M:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-15:
                a[i] = max(a[i] + best_delta, 0.0)
                sum_a = a.sum()
                if sum_a < 1e-10:
                    a[i] = -best_delta
                    sum_a = 1.0
                a = a / sum_a
                conv = fast_convolve(a)
                M = float(conv.max())
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FastCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, sweeps


def fictitious_play(n, time_limit, initial=None, solver=None, verbose=False):
    """Alternating LP best responses with geometric averaging."""
    if solver is None:
        solver = get_lp_solver()
    t0 = time.monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    b = a.copy()
    a_geo = a.copy()
    best_a = a.copy()
    best_score = float('inf')
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        b_new, _ = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
        a_new, _ = solve_lp_best_response(b, solver)
        if a_new is not None:
            # Geometric update for averaging
            if iters == 1:
                a_geo = a_new.copy()
            else:
                a_geo = np.power(np.maximum(a_geo, 1e-15) * np.maximum(a_new, 1e-15), 0.5)
                a_geo = a_geo / a_geo.sum()
            a = a_new

        # Check candidates
        for cand in [a, b, a_geo, 0.5 * (a + b)]:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand_norm = cand / sc
            score = 2.0 * n * fast_convolve(cand_norm).max()
            if score < best_score:
                best_score = score
                best_a = cand_norm.copy()

        if verbose and iters % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP n={n} it={iters} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, iters


def smooth_gradient_descent(a0, time_limit, tau_init=50, tau_end=500, label="", verbose=False):
    """
    Optimize smoothed log-sum-exp objective with projected gradient.
    f(a) = (1/tau) * log(sum_k exp(tau * conv_k(a)))
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    tau = tau_init
    lr = 0.5 / n
    patience = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Animate tau
        progress = min(1.0, it / max(1, int(time_limit / 0.5)))
        tau = tau_init + (tau_end - tau_init) * progress

        conv = fast_convolve(a)
        # Softmax weights
        shifted = conv - conv.max()
        logits = tau * shifted
        w = np.exp(logits - logits.max())
        w = w / w.sum()

        # True score
        M = conv.max()
        score = 2.0 * n * M
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience > 50:
                lr *= 0.7
                patience = 0
                if lr < 1e-8:
                    break

        # Compute gradient: grad = sum_k 2*w[k] * v_k where v_k[j] = a[k-j] (valid)
        grad = np.zeros(n)
        # Vectorized: for each k, the contribution to grad[j] is 2*w[k]*a[k-j]
        # This is a weighted cross-correlation
        # grad[j] = 2 * sum_k w[k] * a[k-j] for valid k-j
        # = 2 * (w cross-correlate a)[j]... let me just do it directly
        for k in range(len(conv)):
            if w[k] < 1e-12:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * w[k] * a[k - hi:k - lo + 1][::-1]

        # Projected gradient step
        a_new = a - lr * grad
        a_new = simplex_project(a_new)

        a = a_new
        lr *= 0.999

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD {label} it={it} ({elapsed:.0f}s): best={best_score:.6f} tau={tau:.0f}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, best_score):
    """Evaluate candidate and return (best_a, best_score) if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return None, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        return a_norm.copy(), score
    return None, best_score


def perturb_sequence(a, rng, magnitude=0.05):
    """Random perturbation to escape local optima."""
    a = a.copy()
    mask = rng.random(len(a)) < 0.3
    a[mask] *= (1.0 + rng.standard_normal(mask.sum()) * magnitude)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    return a / s


def init_fourier(n, rng):
    """Initialize using random Fourier superposition."""
    n_harmonics = int(rng.integers(3, 20))
    amplitudes = rng.random(n_harmonics) * 0.3
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = np.arange(1, n_harmonics + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_cosine(n, rng):
    """Initialize with cosine bell."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * rng.uniform(0.3, 0.6)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    return a / s


def init_u_shape(n, rng):
    """Initialize with U-shape."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    alpha = rng.uniform(0.5, 2.0)
    a = 1.0 + alpha * np.cos(np.pi * t) ** 2
    s = a.sum()
    return a / s

            continue
        t_alloc = min(rem * 0.4, 80.0)
        init_alt = interpolate_seq(best_seq, n_alt)
        a_alt, s_alt, _ = coord_descent_fast(init_alt, t_alloc * 0.7, label=f"p6_n{n_alt}")
        # Polish with bisection
        a_alt2, s_alt2, _ = coord_descent_bisection(a_alt, t_alloc * 0.3, label=f"p6b_n{n_alt}", n_bisect=8)
        for cand in [a_alt, a_alt2]:
            new_best, best_score = check_and_update(cand, best_score)
            if new_best is not None:
                best_seq = new_best
        print(f"  n={n_alt}: best={best_score:.6f}")

    # === Phase 7: Final intensive polishing ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    # Multiple rounds of bisection CD
    for rnd in range(3):
        rem = deadline - time.monotonic() - 30
        if rem < 20:
            break
        t_alloc = min(rem * 0.5, 80.0)
        a_fin, s_fin, _ = coord_descent_bisection(best_seq, t_alloc, label=f"p7r{rnd}", n_bisect=12, verbose=(rnd == 0))
        new_best, best_score = check_and_update(a_fin, best_score)
        if new_best is not None:
            best_seq = new_best
    print(f"  Phase 7 done: {best_score:.6f}")

    # === Phase 8: One more LP round on best ===
    rem = deadline - time.monotonic() - 20
    if rem > 30:
        print(f"\n=== Phase 8: Final LP (best={best_score:.6f}) ===")
        n_fin = len(best_seq)
        a_lp, score_lp, _ = fictitious_play(n_fin, min(rem * 0.6, 60.0), initial=best_seq, solver=solver)
        new_best, best_score = check_and_update(a_lp, best_score)
        if new_best is not None:
            best_seq = new_best
        print(f"  Phase 8 done: {best_score:.6f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
