import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_corr(w, a):
    """Compute g[j] = sum_k w[k] * a[k-j] efficiently via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    a_flip = a[::-1]
    W = rfft(np.asarray(w, dtype=np.float64), fft_size)
    A = rfft(np.asarray(a_flip, dtype=np.float64), fft_size)
    result = irfft(W * A, fft_size)[:size]
    return result[n - 1:n - 1 + n]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def lbgfs_fft_refine(a0, time_limit, T_start=200.0, T_end=0.1, maxiter=10000):
    """L-BFGS with LogSumExp and FFT-based gradient."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    flip_a_cache = {}

    def make_func(T_val):
        def func(z_val):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z_val)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z_val):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_val)
            a = softmax_from_z(z_val)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            # Gradient of conv(a,a) w.r.t. a: grad_a[j] = 2 * sum_k w[k] * a[k-j]
            # = 2 * cross_corr(w, a)[j]
            grad_a = 2.0 * fast_cross_corr(w, a)
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    n_stages = 8
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(30, maxiter // n_stages)

    for idx, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def fista_logsumexp(a0, time_limit, T=50.0, n_steps=50000):
    """FISTA (accelerated proximal gradient) on LogSumExp objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Compute gradient Lipschitz constant estimate
    b0 = fast_convolve(a)
    max_b0 = b0.max()
    # Lipschitz constant of gradient of (2n/T) * logsumexp(T*b) is roughly (4n/T) * max(b)
    L = (4.0 * n / T) * max_b0 * 2  # conservative estimate
    L = max(L, 1e-10)
    
    y = a.copy()
    t_k = 1.0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute gradient at y
        a_y = np.maximum(y, 0)
        s_y = a_y.sum()
        if s_y < 1e-15:
            a_y = np.ones(n) / n
        else:
            a_y = a_y / s_y
        
        b = fast_convolve(a_y)
        Tb = T * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        
        grad_a = 2.0 * (4.0 * n / T) * fast_cross_corr(w, a_y)
        
        # Projected gradient step (projection onto simplex)
        step_size = 1.0 / L
        a_new = a_y - step_size * grad_a
        
        # Project onto probability simplex
        a_new = project_simplex(a_new)
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k**2)) / 2.0
        y = a_new + ((t_k - 1.0) / t_new) * (a_new - a_y)
        a = a_new
        t_k = t_new
        
        # Track best
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Adaptive L
        b_check = fast_convolve(a)
        L = max(L * 0.95, (4.0 * n / T) * b_check.max() * 1.5)
    
    return best_a, best_score


def project_simplex(v):
    """Project vector onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=50000):
    """Nesterov dual averaging LP refinement with periodic restarts."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP: min M s.t. conv(a, b) <= M, b >= 0, sum(b) = 1
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        T_mat = build_conv_matrix(a)
        constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        prob.solve(solver=solver)
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            # Try alternate solver
            for alt in [cp.SCS, cp.ECOS, cp.PDLP]:
                if alt == solver:
                    continue
                try:
                    prob.solve(solver=alt)
                    if prob.status in ('optimal', 'optimal_inaccurate'):
                        break
                except Exception:
                    continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_new = np.asarray(b_var.value, dtype=np.float64)
        b_new = np.maximum(b_new, 0)
        s_b = b_new.sum()
        if s_b < 1e-15:
            continue
        b_new = b_new / s_b
        
        # Nesterov dual averaging update
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = a.copy()
        else:
            y = y / s_y
        
        # Average iterate (converges faster for zero-sum games)
        a = ((it + 1) * a + b_new) / (it + 2)
        a = np.maximum(a, 0)
        s_a = a.sum()
        if s_a < 1e-15:
            a = y.copy()
        else:
            a = a / s_a
        
        # Track best
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
    
    return best_a, best_score


def parametric_init(n, rng):
    """Generate parametric initial sequences."""
    x = np.linspace(0, 1, n)
    choice = rng.integers(0, 6)
    
    if choice == 0:
        # Semicircle-like
        a = np.sqrt(np.maximum(1 - (2*x - 1)**2, 0))
    elif choice == 1:
        # Raised cosine
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif choice == 2:
        # Power of sine
        p = rng.uniform(1.5, 3.0)
        a = np.sin(np.pi * x) ** p
    elif choice == 3:
        # Beta-like
        alpha = rng.uniform(0.5, 2.0)
        beta = alpha
        a = x ** alpha * (1 - x) ** beta
    elif choice == 4:
        # Gaussian
        sigma = rng.uniform(0.3, 0.6)
        a = np.exp(-(x - 0.5)**2 / (2 * sigma**2))
    else:
        # Flat top with rounded edges
        w = rng.uniform(0.1, 0.3)
        a = np.ones(n)
        for i in range(n):
            if x[i] < w:
                a[i] = 0.5 * (1 + np.cos(np.pi * x[i] / w))
            elif x[i] > 1 - w:
                a[i] = 0.5 * (1 + np.cos(np.pi * (x[i] - (1-w)) / w))
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated L-BFGS LogSumExp,
LP-based Nesterov dual averaging, and targeted peak shaving.
Key improvements: O(n log n) gradient via FFT cross-correlation, aggressive
temperature annealing, symmetry exploitation, and optimized time allocation
favoring the LP method (strongest convergence) with fast L-BFGS refinement.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_corr(w, a):
    """Compute g[j] = sum_k w[k] * a[k-j] efficiently via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    a_flip = a[::-1]
    W = rfft(np.asarray(w, dtype=np.float64), fft_size)
    A = rfft(np.asarray(a_flip, dtype=np.float64), fft_size)
    result = irfft(W * A, fft_size)[:size]
    return result[n - 1:n - 1 + n]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def symmetrize(a):
    """Symmetrize and normalize a sequence."""
    a = (a + a[::-1]) / 2.0
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def lbgfs_fft_refine(a0, time_limit, T_start=200.0, T_end=0.1, maxiter=10000):
    """L-BFGS with LogSumExp smoothing and FFT-based gradient."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def make_func(T_val):
        def func(z_val):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z_val)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z_val):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_val)
            a = softmax_from_z(z_val)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = 2.0 * fast_cross_corr(w, a)
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    n_stages = 8
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(30, maxiter // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    result = softmax_from_z(z)
    return result

def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=50000):
    """Nesterov dual averaging LP refinement with stall detection."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_score = best_score
    stall_count = 0
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        T_mat = build_conv_matrix(a)
        constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        prob.solve(solver=solver)
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            for alt in [cp.SCS, cp.ECOS, cp.PDLP]:
                if alt == solver:
                    continue
                try:
                    prob.solve(solver=alt)
                    if prob.status in ('optimal', 'optimal_inaccurate'):
                        break
                except Exception:
                    continue
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_new = np.asarray(b_var.value, dtype=np.float64)
        b_new = np.maximum(b_new, 0)
        s_b = b_new.sum()
        if s_b < 1e-15:
            continue
        b_new = b_new / s_b
        
        # Nesterov dual averaging
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = a.copy()
        else:
            y = y / s_y
        
        # Average iterate
        a = ((it + 1) * a + b_new) / (it + 2)
        a = np.maximum(a, 0)
        s_a = a.sum()
        if s_a < 1e-15:
            a = y.copy()
        else:
            a = a / s_a
        
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        
        if it % 500 == 0:
            if best_score < prev_score - 1e-7:
                stall_count = 0
            else:
                stall_count += 1
            prev_score = best_score
            if stall_count >= 8:
                rng = np.random.default_rng(it + 1)
                noise = rng.random(n) * 0.02
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0
    
    return best_a, best_score


def peak_shave_refine(a0, time_limit, n_steps=20000):
    """Targeted peak shaving focusing on the maximum convolution coefficient."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    consecutive_fails = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if consecutive_fails > 50:
            break
        
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        top_k = np.argsort(contribs)[::-1][:min(6, len(idx))]
        improved = False
        
        for ci in top_k:
            i = idx[ci]
            if a[i] < 1e-14:
                continue
            for factor in [0.98, 0.95, 0.92, 0.88]:
                a_trial = a.copy()
                a_trial[i] *= factor
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            for ci in top_k[:4]:
                i = idx[ci]
                j = peak_k - i
                if i == j or i >= n or j >= n or a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f in [0.97, 0.94, 0.90]:
                    a_trial = a.copy()
                    a_trial[i] *= f
                    a_trial[j] *= f
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            consecutive_fails += 1
        else:
            consecutive_fails = 0
    
    return best_a, best_score


def parametric_init(n, rng):
    """Generate diverse parametric initial sequences."""
    x = np.linspace(0, 1, n)
    choice = rng.integers(0, 7)
    
    if choice == 0:
        a = np.sqrt(np.maximum(1.0 - (2.0 * x - 1.0) ** 2, 0))
    elif choice == 1:
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    elif choice == 2:
        p = rng.uniform(1.5, 3.5)
        a = np.sin(np.pi * x) ** p
    elif choice == 3:
        alpha = rng.uniform(0.5, 2.5)
        a = x ** alpha * (1.0 - x) ** alpha
    elif choice == 4:
        sigma = rng.uniform(0.3, 0.6)
        a = np.exp(-(x - 0.5) ** 2 / (2.0 * sigma ** 2))
    elif choice == 5:
        w = rng.uniform(0.15, 0.35)
        a = np.ones(n)
        for i in range(n):
            if x[i] < w:
                a[i] = 0.5 * (1.0 + np.cos(np.pi * x[i] / w))
            elif x[i] > 1.0 - w:
                a[i] = 0.5 * (1.0 + np.cos(np.pi * (x[i] - (1.0 - w)) / w))
    else:
        a = 1.0 / (1.0 + np.exp(-rng.uniform(3, 8) * (2 * x - 1)))
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

        cur = symmetrize(a_lb2)
        best_state = update_best(best_state, cur, "lowt_p3")
    
    # === Phase 4: Peak shaving ===
    print(f"\n=== Phase 4: Peak Shaving (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 80.0)
    if t_alloc > 10:
        a_ps, ps_score = peak_shave_refine(cur, t_alloc, n_steps=30000)
        cur = symmetrize(a_ps)
        best_state = update_best(best_state, cur, "peak_p4")
    
    # === Phase 5: LP Round 2 ===
    print(f"\n=== Phase 5: LP Round 2 (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.14, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = nesterov_lp_refine(cur, t_alloc, solver=solver, n_max_iters=50000)
        cur = symmetrize(a_lp2)
        best_state = update_best(best_state, cur, "lp_p5")
    
    # === Phase 6: Try alternative length ===
    print(f"\n=== Phase 6: Alt length (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 120:
        alt_n = int(n_best * 1.25)
        if alt_n != n_best and alt_n <= 2048:
            old_x = np.linspace(0, 1, n_best)
            new_x = np.linspace(0, 1, alt_n)
            a_alt = np.interp(new_x, old_x, best_state['seq'])
            a_alt = np.maximum(a_alt, 0)
            a_alt = a_alt / a_alt.sum()
            
            t_total = min(rem * 0.30, 180.0)
            if t_total > 40:
                print(f"  n={alt_n}, budget={t_total:.0f}s")
                t_lb = t_total * 0.45
                a_alt_lb = lbgfs_fft_refine(a_alt, t_lb, T_start=200.0, T_end=1.0, maxiter=6000)
                a_alt = symmetrize(a_alt_lb)
                best_state = update_best(best_state, a_alt, f"lbgfs_n{alt_n}")
                
                rem = deadline - time.monotonic()
                if rem > 30 and best_state['seq'] is not None and len(best_state['seq']) == alt_n:
                    t_lp_alt = min(rem * 0.5, 100.0)
                    if t_lp_alt > 15:
                        a_alt_lp, _ = nesterov_lp_refine(best_state['seq'], t_lp_alt, solver=solver, n_max_iters=25000)
                        a_alt_lp = symmetrize(a_alt_lp)
                        best_state = update_best(best_state, a_alt_lp, f"lp_n{alt_n}")
    
    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_state['score']:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 25:
        t_fin = min(rem * 0.30, 50.0)
        a_fin = low_temp_lbgfs(cur, t_fin, T_start=2.0, T_end=0.01, maxiter=6000)
        cur = symmetrize(a_fin)
        best_state = update_best(best_state, cur, "final_lowt")
    
    rem = deadline - time.monotonic()
    if rem > 15:
        t_ps = min(rem * 0.25, 30.0)
        a_ps2, _ = peak_shave_refine(cur, t_ps, n_steps=15000)
        cur = symmetrize(a_ps2)
        best_state = update_best(best_state, cur, "final_ps")
    
    rem = deadline - time.monotonic()
    if rem > 15:
        t_lp_fin = min(rem * 0.35, 25.0)
        a_lp_fin, _ = nesterov_lp_refine(cur, t_lp_fin, solver=solver, n_max_iters=5000)
        cur = symmetrize(a_lp_fin)
        best_state = update_best(best_state, cur, "final_lp")
    
    # Final verification
    if best_state['seq'] is not None:
        fs = evaluate_safe(best_state['seq'])
        if np.isfinite(fs):
            best_state['score'] = fs
    
    print(f"\nFINAL SCORE: {best_state['score']:.6f} (n={len(best_state['seq'])})")
    return [float(x) for x in best_state['seq']]
