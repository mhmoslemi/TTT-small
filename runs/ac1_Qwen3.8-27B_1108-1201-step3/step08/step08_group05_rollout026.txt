       N = 1 << (len(w) + n - 1 - 1).bit_length()
       W = rfft(w, N)
       F = rfft(f, N)
       return np.real(irfft(np.conj(W) * F, N))[:n]
   
   def project_simplex(v):
       """Euclidean projection onto {x >= 0, sum(x) = 1}."""
       n = len(v)
       u = np.sort(v)[::-1]
       cssv = np.cumsum(u)
       rho_vals = np.arange(1, n + 1)
       cond = u * rho_vals > (cssv - 1.0)
       if not cond.any():
           return np.ones(n) / n
       rho = np.nonzero(cond)[0][-1]
       theta = (cssv[rho] - 1.0) / (rho + 1.0)
       return np.maximum(v - theta, 0.0)
   
   def softmax(x, beta):
       """Softmax with temperature 1/beta."""
       x = x - x.max()
       e = np.exp(beta * x)
       s = e.sum()
       if s < 1e-300:
           return np.ones(len(x)) / len(x)
       return e / s
   
   def smoothed_gradient(f, beta):
       """Compute gradient of (1/beta)*log(sum exp(beta * (f*f)))."""
       c = fast_convolve(f)
       w = softmax(c, beta)
       g = fast_cross_corr(w, f)
       return 2.0 * g, c, w
   
   def lp_solve(f, top_k, max_iter=10, time_limit=30.0):
       """LP with constraint generation."""
       t0 = time.monotonic()
       n = len(f)
       c = fast_convolve(f)
       M = c.max()
       
       if top_k < len(c):
           active = list(np.argpartition(c, -top_k)[-top_k:])
       else:
           active = list(range(len(c)))
       active_set = set(active)
       
       g = None
       for _ in range(max_iter):
           if time.monotonic() - t0 > time_limit:
               break
           
           A = np.zeros((len(active), n))
           for row, k in enumerate(active):
               lo = max(0, k - n + 1)
               hi = min(n - 1, k)
               A[row, lo:hi+1] = f[lo:hi+1]
           
           g_var = cp.Variable(n)
           prob = cp.Problem(cp.Maximize(cp.sum(g_var)), [g_var >= 0, A @ g_var <= M])
           
           try:
               prob.solve(solver=cp.PDLP, max_iters=200000)
           except:
               try:
                   prob.solve(solver=cp.SCS, max_iters=200000)
               except:
                   if g is not None:
                       break
                   return None
           
           if g_var.value is None:
               if g is not None:
                   break
               return None
           
           g = np.maximum(g_var.value, 0.0)
           if np.any(np.isnan(g)):
               return None
            
           # Constraint generation
           fg = fast_cross_convolve(f, g)
           mask = np.zeros(len(c), dtype=bool)
           mask[active] = True
           violations = np.where((fg > M * 1.001) & ~mask)[0]
           
           if len(violations) == 0:
               break
           
           worst = np.argsort(fg[violations])[-min(5, len(violations)):]
           for v in worst:
               if int(v) not in active_set:
                   active_set.add(int(v))
                   active.append(int(v))
       
       return g
   
   def fast_cross_convolve(f, g):

       N = 1 << (size - 1).bit_length()
       F = rfft(f, N)
       G = rfft(g, N)
       return irfft(F * G, N)[:size]
