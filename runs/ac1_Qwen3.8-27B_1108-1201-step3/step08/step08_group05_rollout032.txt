def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""

    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    result = np.maximum(v - theta, 0.0)
    return result * target_sum

    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):

    if n == 0:
        return v
    if n == 1:
        return np.array([max(v[0], 0.0)])

    u = np.sort(v_scaled)[::-1]

    s = w.sum()
    return w / s if s > 1e-10 else np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main optimization: adaptive-temperature LSE descent with FFT gradients,
    multi-peak handling, coordinate descent polish, and length exploration.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1152) / 1152.0
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"No parent, using uniform n=1152, score={p_score:.8f}")

    best_seq = parent.copy()
    best_score = p_score
    n_parent = len(parent)
    print(f"Initial: n={n_parent}, score={best_score:.8f}, target=1.5027")

    # === Phase 1: Main smooth optimization on parent ===
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.55, 900.0)
    if t1 > 30:
        print(f"\n=== Phase 1: Smooth LSE descent (n={n_parent}, budget={t1:.0f}s) ===")
        f1, obj1 = smooth_optimize(parent, time_limit=t1, label="lse_main")
        score1 = float(evaluate_sequence(f1.tolist()))
        print(f"  Phase 1 result: score={score1:.8f}")
        if np.isfinite(score1) and score1 < best_score:
            best_score = score1
            best_seq = f1.copy()
            print(f"  NEW BEST [phase1]: {best_score:.8f}")

    # === Phase 2: Multi-temperature exploration ===
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.2, 250.0)
    if t2 > 30:
        print(f"\n=== Phase 2: Multi-temp (n={len(best_seq)}, budget={t2:.0f}s) ===")
        f2, obj2 = multi_peak_smooth(best_seq, time_limit=t2, label="multi")
        score2 = float(evaluate_sequence(f2.tolist()))
        print(f"  Phase 2 result: score={score2:.8f}")
        if np.isfinite(score2) and score2 < best_score:
            best_score = score2
            best_seq = f2.copy()
            print(f"  NEW BEST [phase2]: {best_score:.8f}")

    # === Phase 3: Coordinate descent polish ===
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.1, 120.0)
    if t3 > 15:
        print(f"\n=== Phase 3: Coord descent (n={len(best_seq)}, budget={t3:.0f}s) ===")
        f3, obj3 = coordinate_descent_polish(best_seq, time_limit=t3, label="cd")
        score3 = float(evaluate_sequence(f3.tolist()))
        print(f"  Phase 3 result: score={score3:.8f}")
        if np.isfinite(score3) and score3 < best_score:
            best_score = score3
            best_seq = f3.copy()
            print(f"  NEW BEST [phase3]: {best_score:.8f}")

    # === Phase 4: Length exploration ===
    rem = deadline - time.monotonic()
    n_base = len(best_seq)
    candidates = [n_base - 128, n_base + 128, n_base - 64, n_base + 64,
                  1024, 1280, 1536]
    candidates = [c for c in candidates if c > 512 and c < 3000 and c != n_base]
    candidates = list(dict.fromkeys(candidates))[:4]

    if rem > 60 and candidates:
        print(f"\n=== Phase 4: Length exploration (best={best_score:.8f}) ===")
        for target_n in candidates:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t4 = min(rem * 0.2, 100.0)
            if t4 < 15:
                break

            # Interpolated initialization
            old_x = np.linspace(0.0, 1.0, len(best_seq))
            new_x = np.linspace(0.0, 1.0, target_n)
            f_interp = np.interp(new_x, old_x, best_seq)
            f_interp = np.maximum(f_interp, 0.0)
            s = f_interp.sum()
            if s > 1e-10:
                f_interp /= s

            print(f"  Trying n={target_n}, budget={t4:.0f}s")
            f4, obj4 = smooth_optimize(f_interp, time_limit=t4,
                                        label=f"n{target_n}")
            score4 = float(evaluate_sequence(f4.tolist()))
            print(f"  n={target_n}: score={score4:.8f}")
            if np.isfinite(score4) and score4 < best_score:
                best_score = score4
                best_seq = f4.copy()
                print(f"  NEW BEST [n{target_n}]: {best_score:.8f}")

    # === Phase 5: Final smooth burst on best ===
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.3, 150.0)
    if t5 > 15:
        print(f"\n=== Phase 5: Final smooth (n={len(best_seq)}, budget={t5:.0f}s) ===")
        f5, obj5 = smooth_optimize(best_seq, time_limit=t5, label="final")
        score5 = float(evaluate_sequence(f5.tolist()))
        print(f"  Phase 5 result: score={score5:.8f}")
        if np.isfinite(score5) and score5 < best_score:
            best_score = score5
            best_seq = f5.copy()
            print(f"  NEW BEST [phase5]: {best_score:.8f}")

    # === Phase 6: Final CD polish ===
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.2, 50.0)
    if t6 > 10:
        f6, obj6 = coordinate_descent_polish(best_seq, time_limit=t6, label="cd_final")
        score6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(score6) and score6 < best_score:
            best_score = score6
            best_seq = f6.copy()
            print(f"  NEW BEST [cd_final]: {best_score:.8f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'=' * 60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"Target: 1.5027")
    print(f"Gap: {best_score - 1.5027:.8f}")
    print(f"{'=' * 60}")

    return [float(x) for x in best_seq]
