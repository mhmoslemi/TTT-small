        mx = shifted.max()
        lse = mx + np.log(np.exp(shifted - mx).sum())
        return (2.0 * n / T) * np.exp(T * lse)

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        zm = z - z.max()
        ez = np.exp(zm)
        a = ez / ez.sum()
        conv = fast_convolve(a)
        shifted = conv / T
        mx = shifted.max()
        w = np.exp(shifted - mx)
        w = w / w.sum()
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
        S = float(np.dot(grad_a, a))
        grad_z = a * (grad_a - S)
        return (2.0 * n / T) * grad_z

    try:
        res = minimize(func, z0, jac=jac, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-16,
                                'gtol': 1e-12, 'maxfun': maxiter * 50,
                                'disp': False})
        zm = res.x - res.x.max()
        ez = np.exp(zm)
        a_out = ez / ez.sum()
        score = compute_score(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception:
        pass
    return best_a, best_score


def interpolate_to_length(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, best_seq, best_score, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    best_seq = None
    best_score = float('inf')

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Initialized: n={n_init}, score={best_score:.8f}")

    n_parent = len(parent)

    # Phase 1: LP-FP with time averaging (main workhorse)
    print(f"\n=== Phase 1: LP-FP Averaged (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = rem * 0.35
    print(f"  Budget: {t1:.0f}s, n={n_parent}")
    a1, s1 = lp_fp_averaged(best_seq, t1, verbose=True)
    best_seq, best_score = check_and_update(a1, best_seq, best_score, "lp-avg")

    # Phase 2: Peak-targeting coordinate descent
    print(f"\n=== Phase 2: Peak CD (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = rem * 0.20
    if rem > 60:
        print(f"  Budget: {t2:.0f}s")
        a2, s2 = peak_targeting_cd(best_seq, t2, n_attempts=300, verbose=True)
        best_seq, best_score = check_and_update(a2, best_seq, best_score, "peak-cd")

    # Phase 3: Multi-length evaluation
    print(f"\n=== Phase 3: Multi-n (best={best_score:.8f}) ===")
    alts = [n_parent - 2, n_parent + 2, n_parent + 4,
            n_parent - 4, n_parent + 6, n_parent + 8]
    for alt_n in alts:
        if alt_n < 64 or alt_n > 8192:
            continue
        rem = deadline - time.monotonic()
        if rem < 20:
            break
        t3 = min(rem * 0.10, 40.0)
        init_alt = interpolate_to_length(best_seq, alt_n)
        a_alt, s_alt = lp_fp_averaged(init_alt, t3, verbose=False)
        best_seq, best_score = check_and_update(a_alt, best_seq, best_score, f"n{alt_n}_lp")
        if rem > 30:
            a_alt2, s_alt2 = peak_targeting_cd(a_alt, min(rem * 0.05, 20.0),
                                               n_attempts=200, verbose=False)
            best_seq, best_score = check_and_update(a_alt2, best_seq, best_score, f"n{alt_n}_cd")

    # Phase 4: L-BFGS polish with multiple temperatures
    print(f"\n=== Phase 4: L-BFGS Polish (best={best_score:.8f}) ===")
    for T_val in [200.0, 500.0, 1000.0]:
        rem = deadline - time.monotonic()
        if rem < 20:
            break
        t4 = min(rem * 0.12, 60.0)
        print(f"  L-BFGS T={T_val:.0f}, budget={t4:.0f}s")
        a4, s4 = lbgfs_polish(best_seq, t4, T=T_val, maxiter=3000)
        best_seq, best_score = check_and_update(a4, best_seq, best_score, f"lbfgs_T{T_val:.0f}")

    # Phase 5: Final LP-FP on best
    print(f"\n=== Phase 5: Final LP (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t5 = min(rem * 0.30, 120.0)
        print(f"  Budget: {t5:.0f}s")
        a5, s5 = lp_fp_averaged(best_seq, t5, verbose=True)
        best_seq, best_score = check_and_update(a5, best_seq, best_score, "final-lp")

    # Phase 6: Final peak CD
    print(f"\n=== Phase 6: Final Peak CD (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t6 = min(rem * 0.40, 60.0)
        print(f"  Budget: {t6:.0f}s")
        a6, s6 = peak_targeting_cd(best_seq, t6, n_attempts=500, verbose=True)
        best_seq, best_score = check_and_update(a6, best_seq, best_score, "final-cd")

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
