def fista_smooth(a0, time_limit, t_init=0.1, t_decay=0.999, verbose=False):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    
    y = a.copy()
    t_k = 1.0
    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t = t_init
    t0 = time.monotonic()
    it = 0
    
    # Lipschitz constant estimate
    L = 1e6
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv_y = fast_convolve(y)
        max_c = conv_y.max()
        
        # Log-sum-exp
        shifted = (conv_y - max_c) / t
        log_sum = max_c / t + np.log(np.sum(np.exp(shifted)))
        f_smooth = t * log_sum
        
        # Softmax weights
        w = np.exp(shifted)
        w /= w.sum()
        
        # Gradient
        grad = 2.0 * np.correlate(w, y, mode='valid')
        
        # Backtracking line search for L
        while True:
            x_new = y - grad / L
            a_new = simplex_project(x_new)
            conv_new = fast_convolve(a_new)
            max_new = conv_new.max()
            shifted_new = (conv_new - max_new) / t
            f_new = t * (max_new / t + np.log(np.sum(np.exp(shifted_new))))
            if f_new <= f_smooth + np.dot(grad, a_new - y) + L / 2 * np.sum((a_new - y)**2):
                break
            L *= 2
        
        a = a_new
        t_next = (1 + np.sqrt(1 + 4 * t_k**2)) / 2
        y = a + (t_k - 1) / t_next * (a - y)
        t_k = t_next
        
        # Update best
        conv_a = fast_convolve(a)
        M = conv_a.max()
        score = 2.0 * n * M
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Decay t
        t *= t_decay
        if t < 0.001:
            t = 0.001
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA it={it} ({elapsed:.0f}s): best={best_score:.6f} t={t:.6f} L={L:.2f}")
    
    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA with log-sum-exp smoothing (proper convex optimization)
2. Active-set QP via cvxpy for exact local solutions
3. Fixed coordinate descent with proper score minimization
4. Parametric initialization for flat autocorrelation
5. Multi-length exploration and simulated annealing escape

import numpy as np
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def compute_gradient_smooth(a, w):
    """Compute gradient of log-sum-exp smoothed max conv.
    grad[i] = 2 * sum_j w[i+j] * a[j] = 2 * cross_correlate(w, a)[i]
    """
    return 2.0 * np.correlate(w, a, mode='valid')


def fista_convex(a0, time_limit, t_init=None, t_floor=1e-4, verbose=False, label=""):
    """FISTA on log-sum-exp smoothed max autocorrelation peak."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    if t_init is None:
        conv0 = fast_convolve(a)
        t_init = max(conv0.max() * 0.05, 1e-4)

    t = t_init
    y = a.copy()
    t_k = 1.0
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    L = 4.0  # Initial Lipschitz estimate
    t0 = time.monotonic()
    it = 0
    stall_count = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv_y = fast_convolve(y)
        max_c = conv_y.max()
        if max_c < 1e-15:
            break

        shifted = (conv_y - max_c) / t
        # Clip for numerical stability
        shifted = np.clip(shifted, -50, 50)
        exp_shifted = np.exp(shifted)
        log_sum = max_c + t * np.log(exp_shifted.sum())
        f_smooth = log_sum  # This is t * log(sum(exp(conv/t)))

        w = exp_shifted / exp_shifted.sum()
        grad = compute_gradient_smooth(y, w)

        # Backtracking line search (limited)
        x_trial = y - grad / L
        a_trial = simplex_project(x_trial)
        for bs in range(20):
            conv_trial = fast_convolve(a_trial)
            max_trial = conv_trial.max()
            shifted_trial = np.clip((conv_trial - max_trial) / t, -50, 50)
            f_trial = max_trial + t * np.log(np.exp(shifted_trial).sum())
            quad_bound = f_smooth + np.dot(grad, a_trial - y) + 0.5 * L * np.sum((a_trial - y) ** 2)
            if f_trial <= quad_bound + 1e-12:
                break
            L *= 1.5
            x_trial = y - grad / L
            a_trial = simplex_project(x_trial)

        a = a_trial
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = a + (t_k - 1.0) / t_next * (a - y)
        y = np.maximum(y, 0.0)
        t_k = t_next

        # Track best (unsmoothed) score
        conv_a = fast_convolve(a)
        M = float(conv_a.max())
        score = 2.0 * n * M
        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Decay smoothing parameter
        if stall_count > 50:
            t *= 0.8
            stall_count = 0
        if it % 200 == 0:
            t *= 0.95
        t = max(t, t_floor)

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA {label} it={it} ({elapsed:.0f}s): best={best_score:.6f} t={t:.6e} L={L:.2f}")

        # Early stop if t is very small and we're stalling
        if t <= t_floor * 1.01 and stall_count > 200:
            break

    return best_a, best_score


def coord_descent_fixed(a0, time_limit, n_trials=8, verbose=False, label=""):
    """Coordinate descent that properly minimizes 2*n*M/sum^2."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    sum_a = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            affected = conv[k_start:k_end + 1]
            a_shifted = a[0:n]  # a[k - k_start + m] pattern; for k in [i, i+n-1], a[k-i]

            best_delta = 0.0
            best_new_score = 2.0 * n * max(unaff_max, float(affected.max())) / (sum_a * sum_a)

            for trial in range(n_trials):
                if trial == 0:
                    delta = -0.5 * a[i]
                elif trial == 1:
                    delta = -0.3 * a[i]
                elif trial == 2:
                    delta = -0.15 * a[i]
                elif trial == 3:
                    delta = -0.05 * a[i]
                elif trial == 4:
                    delta = 0.05 * a[i]
                elif trial == 5:
                    delta = 0.15 * a[i]
                elif trial == 6:
                    delta = 0.3 * a[i]
                else:
                    delta = 0.5 * a[i]

                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx_2i] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_max = max(unaff_max, float(new_affected.max()))
                new_sum = sum_a + delta
                if new_sum < 0.01:
                    continue
                new_score = 2.0 * n * new_max / (new_sum * new_sum)

                if new_score < best_new_score:
                    best_new_score = new_score
                    best_delta = delta

            if best_delta != 0.0 and best_new_score < best_score - 1e-14:
                a[i] = max(a[i] + best_delta, 0.0)
                sum_a += best_delta

                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx_2i] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (sum_a * sum_a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 10:
                break
        else:
            no_improve = 0

        if verbose:
            elapsed = time.monotonic() - t0
            if elapsed % 5 < 1 or it == 1:
                pass

    return best_a, best_score


def build_Q_k_sparse(k, n):
    """Build sparse matrix Q_k such that a^T Q_k a = conv(a,a)[k]."""
    j_start = max(0, k - n + 1)
    j_end = min(k, n - 1)
    rows = []
    cols = []
    vals = []
    for j in range(j_start, j_end + 1):
        i = k - j
        if i < 0 or i >= n:
            continue
        if i == j:
            rows.append(j)
            cols.append(j)
            vals.append(1.0)
        else:
            rows.append(j)
            cols.append(i)
            vals.append(0.5)
            rows.append(i)
            cols.append(j)
            vals.append(0.5)
    if not rows:
        return csr_matrix((n, n))
    return csr_matrix((vals, (rows, cols)), shape=(n, n))


def active_set_qp(a_init, n, solver, max_iters=50, verbose=False, label=""):
    """Active-set QP: solve convex QP with lazy constraint generation."""
    import cvxpy as cp

    a_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    # Start with constraints near the middle (most likely to be binding)
    mid = n - 1
    active_k = sorted(set(list(range(max(0, mid - 10), min(2 * n - 1, mid + 11)))))

    best_a = a_init.copy()
    conv_init = fast_convolve(a_init)
    best_M = float(conv_init.max())

    for it in range(max_iters):
        constraints = [cp.sum(a_var) == 1]
        for k in active_k:
            Q = build_Q_k_sparse(k, n)
            constraints.append(cp.quad_form(a_var, Q) <= M_var)

        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS)
            except Exception:
                break

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break

        a_val = np.asarray(a_var.value, dtype=np.float64)
        a_val = np.maximum(a_val, 0.0)
        s = a_val.sum()
        if s < 1e-10:
            break
        a_val = a_val / s

        conv = fast_convolve(a_val)
        M_val = float(conv.max())
        score = 2.0 * n * M_val

        if score < 2.0 * n * float(fast_convolve(best_a).max()):
            best_M = M_val
            best_a = a_val.copy()

        # Find violated constraints
        M_opt = float(M_var.value) if M_var.value is not None else M_val
        tol = 1e-7
        violated = np.where(conv > M_opt + tol)[0]

        if len(violated) == 0:
            # Check if truly optimal (all constraints satisfied)
            if M_val <= M_opt + 1e-6:
                break

        # Add violated constraints
        if len(violated) > 0:
            # Add the most violated ones
            viol_sorted = violated[np.argsort(conv[violated])[::-1]]
            for k in viol_sorted[:15]:
                if k not in active_k:
                    active_k.append(k)

            # Cap active set size
            if len(active_k) > 100:
                # Keep most violated
                conv_at = fast_convolve(a_val)
                active_k = list(np.argsort(conv_at)[-100:])
                active_k = sorted(active_k)

        if verbose and it % 5 == 0:
            print(f"    QP {label} it={it}: M={M_val:.8f}, active={len(active_k)}, violated={len(violated)}")

    return best_a, best_M


def parametric_init(n, shape, rng):
    """Generate parametric initializations designed for flat autocorrelation."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)

    if shape == 'u_shape':
        # U-shaped: more mass at edges
        alpha = rng.uniform(0.5, 3.0)
        a = 1.0 + alpha * (2.0 * t - 1.0) ** 2
    elif shape == 'double_hump':
        # Two peaks
        c1 = rng.uniform(0.2, 0.35)
        c2 = 1.0 - c1
        w = rng.uniform(0.05, 0.15)
        a = np.exp(-((t - c1) / w) ** 2) + np.exp(-((t - c2) / w) ** 2)
    elif shape == 'cosine_mod':
        # Cosine modulation
        freq = int(rng.integers(2, 8))
        amp = rng.uniform(0.1, 0.4)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + amp * np.cos(2 * np.pi * freq * t + phase)
    elif shape == 'quartic_u':
        # Quartic U-shape
        beta = rng.uniform(1.0, 5.0)
        a = 1.0 + beta * (2.0 * t - 1.0) ** 4
    elif shape == 'sinc_like':
        # Sinc-like (concentrated at edges with oscillation)
        freq = int(rng.integers(3, 15))
        a = 1.0 + 0.3 * np.sinc(freq * (t - 0.5))
        a = np.abs(a)
    elif shape == 'random_smooth':
        # Random smooth via low-freq Fourier
        n_harm = int(rng.integers(2, 10))
        freqs = rng.integers(1, 20, n_harm)
        amps = rng.uniform(0, 0.3, n_harm)
        phases = rng.uniform(0, 2 * np.pi, n_harm)
        a = np.ones(n)
        tt = np.arange(n, dtype=np.float64)
        for f, amp, ph in zip(freqs, amps, phases):
            a += amp * np.cos(2 * np.pi * f * tt / n + ph)
    else:
        a = np.ones(n)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def simulated_annealing_escape(a0, rng, n_steps=100, temp_init=0.02):
    """Random perturbations to escape local optima."""
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    temp = temp_init

    for step in range(n_steps):
        # Random perturbation: pick a few coordinates and scale them
        n_perturb = int(np.random.randint(3, max(4, n // 50)))
        indices = rng.integers(0, n, n_perturb)
        for i in indices:
            a[i] *= (1.0 + rng.standard_normal() * temp)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
            continue
        a = a / s

        conv_new = fast_convolve(a)
        score_new = 2.0 * n * float(conv_new.max())

        if score_new < best_score or rng.random() < np.exp(-(score_new - best_score) / temp):
            a = a.copy()
            if score_new < best_score:
                best_score = score_new
                best_a = a.copy()
        else:
            a = best_a.copy()

        temp *= 0.97

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def get_lp_solver():
    """Find a working LP/QP solver."""
    import cvxpy as cp
    global _lp_solver
    if '_lp_solver' in globals() and _lp_solver is not None:
        return _lp_solver
    _lp_solver = None
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.CLP, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return cp.ECOS


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    # Build constraint matrix: conv(a, b)[k] = sum_j a[j] * b[k-j]
    # For each k: sum_j a[j] * b[k-j] <= M
    # This is linear in b.
    # conv(a,b)[k] = (a_flipped * b)[k] where a_flipped is reversed a
    # Actually: conv(a,b)[k] = sum_{j} a[j] * b[k-j] = sum_{i} a[k-i] * b[i]
    # So the coefficient of b[i] in conv(a,b)[k] is a[k-i] (if valid)

    # Build sparse matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            if 0 <= j < n:
                rows.append(k)
                cols.append(i)
                vals.append(a_fixed[j])

    T = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def fictitious_play(n, time_limit, initial=None, solver=None, verbose=False, label=""):
    """Alternating LP best responses."""
    if solver is None:
        solver = get_lp_solver()
    t0 = time.monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    b = a.copy()
    a_avg = a.copy().astype(np.float64)
    best_a = a.copy()
    best_score = float('inf')
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        b_new, _ = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
        a_new, _ = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new

        a_avg = (iters * a_avg + a) / (iters + 1)

        for cand in [a, b, 0.5 * (a + b), a_avg]:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand_norm = cand / sc
            score = 2.0 * n * float(fast_convolve(cand_norm).max())
            if score < best_score:
                best_score = score
                best_a = cand_norm.copy()

        if verbose and iters % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP {label} n={n} it={iters} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, iters


def propose_candidate():
    """
    Multi-phase optimization:
    1. Diverse parametric initialization + FISTA
    2. Active-set QP polishing
    3. Fixed coordinate descent
    4. LP fictitious play for global search
    5. Simulated annealing escape + re-optimization
    """
    deadline = time.monotonic() + 1770.0
    rng = np.random.default_rng(42)
    solver = get_lp_solver()
    print(f"Using solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    # === Phase 1: Diverse initialization + FISTA ===
    print("\n=== Phase 1: Diverse Init + FISTA ===")
    t1_start = time.monotonic()

    n_targets = [1024]
    if parent is not None:
        np_ = len(parent)
        if np_ not in n_targets:
            n_targets.insert(0, np_)

    shapes = ['u_shape', 'double_hump', 'cosine_mod', 'quartic_u', 'sinc_like', 'random_smooth']
    candidates = []

    for n in n_targets[:4]:
        elapsed_p1 = time.monotonic() - t1_start
        if elapsed_p1 > 300:
            break

        # Parent interpolation
        if parent is not None:
            cand = interpolate_seq(parent, n)
            t_alloc = min(60.0, max(10.0, (300 - (time.monotonic() - t1_start)) / 3.0))
            a_opt, s_opt = fista_convex(cand, t_alloc, t_init=0.001, label=f"p1_n{n}_par")
            candidates.append((s_opt, a_opt, n, "parent"))

        for shape in shapes:
            if time.monotonic() - t1_start > 280:
                break
            cand = parametric_init(n, shape, rng)
            t_alloc = min(30.0, max(5.0, (280 - (time.monotonic() - t1_start)) / 3.0))
            a_opt, s_opt = fista_convex(cand, t_alloc, t_init=0.002, label=f"p1_n{n}_{shape}")
            candidates.append((s_opt, a_opt, n, shape))

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 1 done: {len(candidates)} candidates, best: {candidates[0][0]:.6f}")

    for s_c, a_c, n_c, sh_c in candidates[:8]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_n{n_c}_{sh_c}")

    # === Phase 2: Intensive FISTA on best ===
    print(f"\n=== Phase 2: Intensive FISTA (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        t2_budget = 400.0
        for rnd in range(4):
            rem = deadline - time.monotonic() - 500
            if rem < 30:
                break
            t_alloc = min(rem * 0.3, t2_budget / 4.0)
            print(f"  FISTA round {rnd+1}, n={n_best}, budget={t_alloc:.0f}s")
            a_opt, s_opt = fista_convex(best_seq, t_alloc, t_init=0.0005 * (0.9 ** rnd),
                                        t_floor=1e-5, verbose=True, label=f"p2r{rnd}")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"p2_fista_r{rnd}")

            # Escape via perturbation
            if best_score > 1.503:
                escaped, _ = simulated_annealing_escape(best_seq, rng, n_steps=200, temp_init=0.03)
                a_esc, s_esc = fista_convex(escaped, max(10.0, t_alloc * 0.3), t_init=0.001,
                                            label=f"p2esc{rnd}")
                best_seq, best_score = check_and_update(a_esc, best_seq, best_score, f"p2_esc_r{rnd}")

    # === Phase 3: Active-set QP ===
    print(f"\n=== Phase 3: Active-Set QP (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_qp = min(rem * 0.4, 200.0)
            qp_start = time.monotonic()
            print(f"  QP n={n_best}, budget={t_qp:.0f}s")
            a_qp, M_qp = active_set_qp(best_seq, n_best, solver, max_iters=80,
                                       verbose=True, label="p3")
            best_seq, best_score = check_and_update(a_qp, best_seq, best_score, "qp")

            # Try QP from different starting points
            for shape in ['u_shape', 'double_hump']:
                if time.monotonic() > deadline - 300:
                    break
                cand = parametric_init(n_best, shape, rng)
                a_qp2, _ = active_set_qp(cand, n_best, solver, max_iters=40, label="p3b")
                best_seq, best_score = check_and_update(a_qp2, best_seq, best_score, f"qp_{shape}")

    # === Phase 4: LP Fictitious Play ===
    print(f"\n=== Phase 4: LP Fictitious Play (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        fp_lengths = [n_best]
        for alt_n in [768, 1024, 640, 1280]:
            if alt_n != n_best and 256 <= alt_n <= 2048:
                fp_lengths.append(alt_n)
        fp_lengths = fp_lengths[:3]

        for n_fp in fp_lengths:
            rem = deadline - time.monotonic() - 250
            if rem < 60:
                break
            t_alloc = min(rem * 0.4, 150.0)
            print(f"  FP n={n_fp}, budget={t_alloc:.0f}s")

            initial = interpolate_seq(best_seq, n_fp) if best_seq is not None else None
            a_fp, score_fp, _ = fictitious_play(n_fp, t_alloc, initial=initial,
                                                 solver=solver, verbose=True, label="p4")
            best_seq, best_score = check_and_update(a_fp, best_seq, best_score, f"fp_n{n_fp}")

            # Quick FISTA polish on FP result
            rem = deadline - time.monotonic() - 200
            if rem > 20:
                a_pol, s_pol = fista_convex(a_fp, min(rem * 0.3, 40.0), t_init=0.0005,
                                            label=f"fp_pol_n{n_fp}")
                best_seq, best_score = check_and_update(a_pol, best_seq, best_score, f"fp_pol_n{n_fp}")

    # === Phase 5: Coordinate descent polishing ===
    print(f"\n=== Phase 5: Coord Descent Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_cd = min(rem * 0.5, 150.0)
            print(f"  CD n={len(best_seq)}, budget={t_cd:.0f}s")
            a_cd, s_cd = coord_descent_fixed(best_seq, t_cd, n_trials=12, label="p5")
            best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "cd")

    # === Phase 6: Final FISTA with tight smoothing ===
    print(f"\n=== Phase 6: Final FISTA (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 50
        if rem > 20:
            t_final = min(rem * 0.6, 100.0)
            print(f"  Final FISTA n={len(best_seq)}, budget={t_final:.0f}s")
            a_fin, s_fin = fista_convex(best_seq, t_final, t_init=0.0001, t_floor=1e-6,
                                        verbose=True, label="final")
            best_seq, best_score = check_and_update(a_fin, best_seq, best_score, "final_fista")

    # === Phase 7: Last coordinate descent ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_last, s_last = coord_descent_fixed(best_seq, min(rem * 0.7, 40.0), n_trials=16, label="last")
            best_seq, best_score = check_and_update(a_last, best_seq, best_score, "last_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
