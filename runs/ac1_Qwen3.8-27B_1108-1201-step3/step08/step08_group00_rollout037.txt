            log_sum_exp = max_Tc + np.log(exp_Tc.sum())
            return 2.0 * n * log_sum_exp / T_val
        return func

            a_cur = softmax(z_in)
            a_cur = symmetrize(a_cur)
            a_cur = normalize(a_cur)
            c = fast_convolve(a_cur)
            Tc = T_val * c
            max_Tc = Tc.max()
            exp_Tc = np.exp(Tc - max_Tc)
            w = exp_Tc / exp_Tc.sum()
            # Gradient with respect to a
            g_a = compute_conv_gradient(a_cur, w)
            # Convert to z-gradient
            S = np.dot(a_cur, g_a)
            return a_cur * (g_a - S)
        return jac
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for stage, T_val in enumerate(temps):

        if remaining < 5:

                method='L-BFGS-B',
                options={
                    'maxiter': maxiter_per_stage,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'maxfun': maxiter_per_stage * 50,
                    'disp': False

            z = res.x

        except Exception as e:
            pass

def solve_lp_best_response(a_fixed, solver=None):
    """LP: given fixed a (symmetric, sum=1), find symmetric b (sum=1) minimizing max conv(a,b)."""
    if solver is None:
        solver = get_lp_solver()
    
    n = len(a_fixed)
    half = (n + 1) // 2
    
    b_half = cp.Variable(half, nonneg=True)
    M_var = cp.Variable()
    
    constraints = []
    
    # Sum constraint
    if n % 2 == 0:
        constraints.append(2.0 * cp.sum(b_half) == 1.0)
    else:
        constraints.append(2.0 * cp.sum(b_half[:half-1]) + b_half[half-1] == 1.0)
    
    # Convolution constraints: only k = 0 to n-1 (by symmetry)
    for k in range(n):
        expr = 0.0
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            idx = k - j
            # b_full[idx] = b_half[min(idx, n-1-idx)]
            if idx < half:
                b_elem = b_half[idx]
            else:
                b_elem = b_half[n - 1 - idx]
            expr = expr + a_fixed[j] * b_elem
        constraints.append(expr <= M_var)
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_half_val = np.asarray(b_half.value, dtype=np.float64)
    b_half_val = np.maximum(b_half_val, 0)
    
    b_full_val = np.zeros(n)
    for i in range(half):
        b_full_val[i] = b_half_val[i]
        b_full_val[n-1-i] = b_half_val[i]
    
    b_full_val = normalize(b_full_val)
    return b_full_val, float(M_var.value)


def nesterov_lp(a0, time_limit, solver=None, max_iters=20000):
    """Nesterov-accelerated LP alternating method with symmetry."""
    n = len(a0)
    a = normalize(symmetrize(a0.copy()))
    y = a.copy()
    best_a = a.copy()
    best_score = fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_val = solve_lp_best_response(y, solver)
        if b_new is None:
            stall += 1
            if stall > 50:
                break
            continue
        
        alpha = 2.0 / (it + 2.0)
        y = (1 - alpha) * y + alpha * b_new
        y = normalize(symmetrize(np.maximum(y, 0)))
        
        a_new = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a_new = normalize(symmetrize(np.maximum(a_new, 0)))
        
        score_a = fast_convolve(a_new).max()
        score_y = fast_convolve(y).max()
        
        if score_a < best_score:
            best_score = score_a
            best_a = a_new.copy()
            stall = 0
        a = a_new
        
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        
        if (it + 1) % 100 == 0:
            if best_score > prev_best - 1e-8:
                stall += 1
            else:
                stall = 0
            prev_best = best_score
            
            if stall >= 10:
                # Perturb to escape
                rng = np.random.default_rng(it + 12345)
                y = y + 0.02 * rng.standard_normal(n)
                y = normalize(symmetrize(np.maximum(y, 0)))
                stall = 0
            
            elapsed = time.monotonic() - t0
            print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={2*n*best_score:.6f}")
    
    return best_a


def subgradient_nesterov(a0, time_limit, max_iters=200000, rel_tol=0.01):
    """Fast subgradient Nesterov method."""
    n = len(a0)
    a = normalize(symmetrize(a0.copy()))
    y = a.copy()
    best_a = a.copy()
    best_score = fast_convolve(a).max()
    t0 = time.monotonic()
    eta = 0.5 / np.sqrt(n)
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(y)
        max_c = c.max()
        threshold = max_c * (1 - rel_tol)
        active = np.where(c >= threshold)[0]
        
        # Subgradient: (1/|P|) sum_{k in P} 2 * a_reversed shifted
        g = np.zeros(n)
        for k in active:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                g[i_start:i_end + 1] += 2.0 * a[k - i_end : k - i_start + 1]
        g /= len(active)
        
        # Nesterov update
        t_k = (1 + np.sqrt(1 + 4 * (it + 1) ** 2)) / 2
        t_prev = (1 + np.sqrt(1 + 4 * it ** 2)) / 2 if it > 0 else 1
        
        y_new = y - eta * g
        y_new = project_simplex(symmetrize(y_new))
        
        beta = (t_prev - 1) / t_k if it > 0 else 0
        a = y_new + beta * (y_new - y)
        a = project_simplex(symmetrize(a))
        
        y = y_new
        
        score = fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        eta = eta / (1 + it * 0.0005)
        eta = max(eta, 1e-7)
    
    return best_a


def peak_shave(a0, time_limit, max_steps=30000):
    """Targeted peak shaving with coordinate descent."""
    n = len(a0)
    a = normalize(symmetrize(a0.copy()))
    best_a = a.copy()
    best_score = fast_convolve(a).max()
    t0 = time.monotonic()
    
    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(a)
        max_c = c.max()
        # Find peak (exclude edges for more meaningful peaks)
        c_work = c.copy()
        if len(c_work) > 4:
            c_work[0] = -np.inf
            c_work[-1] = -np.inf
        peak_k = int(np.argmax(c_work))
        
        improved = False
        
        # Find top contributors to the peak
        i_start = max(0, peak_k - n + 1)
        i_end = min(n - 1, peak_k)
        idx = np.arange(i_start, i_end + 1)
        contribs = a[idx] * a[peak_k - idx]
        top = np.argsort(contribs)[::-1][:8]
        
        for ti in top:
            i = idx[ti]
            j = peak_k - i
            if i == j:
                # Single coordinate reduction
                for factor in [0.97, 0.94, 0.90, 0.85]:
                    a_trial = a.copy()
                    a_trial[i] *= factor
                    a_trial = symmetrize(a_trial)
                    a_trial = normalize(a_trial)
                    score = fast_convolve(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
            else:
                # Pair reduction
                for f1 in [0.95, 0.90]:
                    for f2 in [0.95, 0.90]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = symmetrize(a_trial)
                        a_trial = normalize(a_trial)
                        score = fast_convolve(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
            if improved:
                break
        
        if not improved:
            # Try multi-coordinate
            a_trial = a.copy()
            for ti in top[:4]:
                i = idx[ti]
                a_trial[i] *= 0.95
            a_trial = symmetrize(a_trial)
            a_trial = normalize(a_trial)
            score = fast_convolve(a_trial).max()
            if score < best_score:
                a = a_trial
                best_score = score
                best_a = a.copy()
            else:
                break
    
    return best_a


def structured_init(n, rng):
    """Create symmetric structured initial sequence."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 0.5
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 1.0 - 0.3 * np.cos(2 * np.pi * x)
    elif choice == 4:
        a = 1.0 + 0.5 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 0)
    else:
        a = np.ones(n)
    
    a = symmetrize(a)
    a = normalize(a)
    return a


def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length, preserving symmetry."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a_norm = a / a.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0)
    a_interp = symmetrize(a_interp)
    return normalize(a_interp)
