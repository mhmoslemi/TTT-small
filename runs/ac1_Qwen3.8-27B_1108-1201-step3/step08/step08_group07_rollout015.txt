def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    """Build (2n-1) x n dense matrix C such that C @ g = f * g (linear conv)."""
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_best_response(f, M, prev_g=None, perturb=0.0):
    """Solve: max sum(g) s.t. C(f) @ g <= M*(1+perturb), g >= 0."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    C = build_conv_matrix(f)
    M_eff = M * (1.0 + perturb)
    constraints = [C @ g <= M_eff]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    for solver in [cp.CLARABEL, cp.MOSEK, cp.ECOS]:
        try:
            prob.solve(solver=solver, warm_start=(prev_g is not None))
            if g.value is not None:
                return np.maximum(g.value, 0)
        except Exception:
            continue
    return prev_g if prev_g is not None else f.copy()

def subgradient_polish(a0, time_limit, verbose=True):
    """Projected subgradient method for polishing."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    eta = 1.0 / np.sqrt(n)

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        max_val = conv.max()
        # Find active constraints (top 5)
        top_k = np.argsort(conv)[-5:]
        # Compute subgradient: average of gradients at active positions
        grad = np.zeros(n)
        for k in top_k:
            # gradient of b[k] wrt a[j] = 2*a[k-j] (zero-extended)
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    grad[j] += 2.0 * a[idx]
        grad /= len(top_k)

        # Projected subgradient step
        eta_cur = eta / np.sqrt(1.0 + 0.01 * iters)
        a_new = a - eta_cur * grad
        # Project onto simplex {a >= 0, sum(a) = 1}
        a_new = project_simplex(a_new)

        score = compute_score(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        if verbose and iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  Subgrad iter={iters}, best={best_score:.8f}, eta={eta_cur:.6f}, "
                  f"time={elapsed:.0f}s/{time_limit:.0f}s")

    return best_a, best_score


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def lbgfs_smoothed(a0, time_limit, T, maxiter=5000, verbose=False):
    """L-BFGS on smoothed log-sum-exp objective in log-space."""
    n = len(a0)
    z0 = np.log(np.maximum(a0, 1e-30))
    t0 = time.monotonic()
    best_a = a0.copy()

            return 1e15
        return smoothed_objective(z, T, n)

            return np.zeros_like(z)
        return smoothed_gradient(z, T, n)

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,

        score = compute_score(a_out)
        if score < best_score:

            best_score = score
    except Exception:
        pass
    return best_a, best_score


def smoothed_objective(z, T, n):
    a = sp_softmax(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    lse = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
    return (2.0 * n / T) * np.exp(T * lse)


def smoothed_gradient(z, T, n):
    a = sp_softmax(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    w = np.exp(shifted - max_shifted)
    w = w / w.sum()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
    S = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - S)
    return (2.0 * n / T) * grad_z

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def generate_varied_initializations(n, parent, rng):

    # Two-peak
    peak_width = max(1, n // 8)
    for offset in [n // 4, n // 3, n // 2]:
        p = np.exp(-((x - offset) ** 2) / (2 * peak_width ** 2))
        p2 = np.exp(-((x - (n - offset)) ** 2) / (2 * peak_width ** 2))
        c = p + p2
        c = np.maximum(c, 0.001)
        inits.append(c / c.sum())

    # Random with structure
    for alpha in [0.5, 1.0, 2.0]:
        r = rng.random(n)
        c = r ** alpha
        c = np.maximum(c, 0.001)
        inits.append(c / c.sum())

    # Interpolated parent at this length
    if parent is not None:
        inits.append(interpolate_sequence(parent, n))

    return inits


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization with ergodic averaging, multi-length search,
    and diverse initializations to escape local minima.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Initialized: n={n_init}, score={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: LP-FP with ergodic average on parent (main driver) ===
    print(f"\n=== Phase 1: LP-FP Ergodic (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.35, 600.0)
    print(f"  Budget: {t1:.0f}s, n={n_parent}")
    a1, s1 = lp_fictitious_play_ergodic(best_seq, t1, step=0.4, label="LP-Erg")
    best_seq, best_score = check_and_update(a1, "LP-Erg", best_seq, best_score)

    # === Phase 2: LP-FP with geometric average ===
    print(f"\n=== Phase 2: LP-FP Geometric Avg (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.25, 400.0)
    if rem > 100:
        print(f"  Budget: {t2:.0f}s")
        a2, s2 = lp_fictitious_play_avg(best_seq, t2, label="LP-Geo")
        best_seq, best_score = check_and_update(a2, "LP-Geo", best_seq, best_score)

    # === Phase 3: Subgradient polish ===
    print(f"\n=== Phase 3: Subgradient Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.08, 80.0)
    if rem > 60:
        print(f"  Budget: {t3:.0f}s")
        a3, s3 = subgradient_polish(best_seq, t3, verbose=False)
        best_seq, best_score = check_and_update(a3, "Subgrad", best_seq, best_score)

    # === Phase 4: Bussgang spectral flattening ===
    print(f"\n=== Phase 4: Bussgang (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.07, 70.0)
    if rem > 50:
        print(f"  Budget: {t4:.0f}s")
        a4, s4 = bussgang_iteration(best_seq, t4, verbose=False)
        best_seq, best_score = check_and_update(a4, "Bussgang", best_seq, best_score)

    # === Phase 5: L-BFGS polishing with multiple temperatures ===
    print(f"\n=== Phase 5: L-BFGS Polish (best={best_score:.8f}) ===")
    for T_val in [200, 500, 1000, 2000]:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t5 = min(rem * 0.08, 60.0)
        print(f"  L-BFGS T={T_val}, budget={t5:.0f}s")
        a5, s5 = lbgfs_smoothed(best_seq, t5, T=T_val, maxiter=3000)
        best_seq, best_score = check_and_update(a5, f"LBF-T{T_val}", best_seq, best_score)

    # === Phase 6: Improved coordinate descent ===
    print(f"\n=== Phase 6: Coordinate Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.12, 100.0)
    if rem > 40:
        print(f"  Budget: {t6:.0f}s")
        a6, s6 = coordinate_descent_improved(best_seq, t6, n_pairs=400, verbose=False)
        best_seq, best_score = check_and_update(a6, "CD-imp", best_seq, best_score)

    # === Phase 7: Try alternative lengths ===
    print(f"\n=== Phase 7: Alternative Lengths (best={best_score:.8f}) ===")
    alt_lengths = [n_parent - 32, n_parent + 32, n_parent - 64, n_parent + 64]
    # Also try round numbers
    for rl in [512, 1024, 1536]:
        if rl not in alt_lengths and 128 <= rl <= 2048:
            alt_lengths.append(rl)

    for alt_n in alt_lengths:
        if alt_n < 64 or alt_n > 3000:
            continue
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t7 = min(rem * 0.15, 120.0)
        init_alt = interpolate_sequence(best_seq, alt_n)
        print(f"  LP-FP n={alt_n}, budget={t7:.0f}s")
        a_alt, s_alt = lp_fictitious_play_ergodic(init_alt, t7 * 0.6, step=0.35, verbose=False, label=f"LP-n{alt_n}")
        best_seq, best_score = check_and_update(a_alt, f"LP-n{alt_n}", best_seq, best_score)
        # Quick L-BFGS polish
        rem = deadline - time.monotonic()
        if rem > 20:
            a_alt2, s_alt2 = lbgfs_smoothed(a_alt, min(rem * 0.3, 30.0), T=500, maxiter=1000)
            best_seq, best_score = check_and_update(a_alt2, f"LB-n{alt_n}", best_seq, best_score)

    # === Phase 8: Multi-start with diverse initializations on best length ===
    print(f"\n=== Phase 8: Multi-Start (best={best_score:.8f}) ===")
    n_best = len(best_seq)
    inits = generate_varied_initializations(n_best, best_seq, rng)
    for idx, init in enumerate(inits[:4]):  # Try at most 4 different inits
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t8 = min(rem * 0.15, 80.0)
        print(f"  Multi-start init {idx}, budget={t8:.0f}s")
        a_ms, s_ms = lp_fictitious_play_ergodic(init, t8, step=0.4, verbose=False, label=f"MS-{idx}")
        best_seq, best_score = check_and_update(a_ms, f"MS-{idx}", best_seq, best_score)

    # === Phase 9: Final LP polish on the best ===
    print(f"\n=== Phase 9: Final LP Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t9 = min(rem * 0.5, 60.0)
        a9, s9 = lp_fictitious_play_ergodic(best_seq, t9, step=0.2, verbose=False, label="LP-Final")
        best_seq, best_score = check_and_update(a9, "LP-Final", best_seq, best_score)

    # === Phase 10: Final coordinate descent ===
    rem = deadline - time.monotonic()
    if rem > 15:
        t10 = min(rem * 0.7, 30.0)
        a10, s10 = coordinate_descent_improved(best_seq, t10, n_pairs=200, verbose=False)
        best_seq, best_score = check_and_update(a10, "CD-Final", best_seq, best_score)

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(x) for x in best_seq]
