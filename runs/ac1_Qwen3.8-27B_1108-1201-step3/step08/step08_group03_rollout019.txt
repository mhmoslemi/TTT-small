def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])

    if w_sum < 1e-30:
        w = np.ones(len(b)) / len(b)
    else:
        w /= w_sum

    sg = np.zeros(n)

    fft_size = 1 << (size - 1).bit_length()

    return irfft(A * A, fft_size)[:size]


def cross_convolve(f, g):
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def symmetrize(f):
    n = len(f)
    f_sym = (f + f[::-1]) / 2.0
    f_sym = np.maximum(f_sym, 0.0)
    s = f_sym.sum()
    if s > 1e-10:
        f_sym /= s
    return f_sym


def iaa_init(n, n_iters=200, alpha=0.5, rng_seed=42):
    """IAA-based initialization for symmetric sequences."""
    rng = np.random.default_rng(rng_seed)
    f = symmetrize(1.0 + 0.3 * rng.standard_normal(n))
    f = project_simplex(f)
    
    N = 1 << (2 * n - 1).bit_length()
    
    for it in range(n_iters):
        F = rfft(f, N)
        power = np.abs(F) ** 2
        # Blend toward flat spectrum
        avg = power.mean()
        new_power = power * (1 - alpha) + avg * alpha
        new_power = np.maximum(new_power, 1e-20)
        new_mag = np.sqrt(new_power)
        phase = F / (np.abs(F) + 1e-30)
        F_new = new_mag * phase
        f_new = irfft(F_new, N)[:n]
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-10:
            break
        f_new /= s
        f = symmetrize(f_new)
    
    return f


def chirp_init(n, phase=0.0, alpha=1.0):
    """Chirp-based initialization (always non-negative)."""
    j = np.arange(n, dtype=np.float64)
    # Raised cosine chirp: always non-negative
    t = j / (n - 1) if n > 1 else 0.0
    f = 1.0 + alpha * np.cos(np.pi * t * t * 2 + phase)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n)
        s = n
    f /= s
    return symmetrize(f)


def build_sym_lp(f, indices, n):
    """
    Build symmetric LP: max sum(g) s.t. g>=0, g symmetric, (f*g)[k] <= M for k in indices.
    Parameterize g by first half: g[j] = g_half[j] for j < n/2, g[n-1-j] = g_half[j] for j < n/2.
    """
    n_half = n // 2
    # g has 2*n_half variables (for even n)
    # Symmetry: g[j] = g[n-1-j]
    # Constraint (f*g)[k] = sum_j f[j]*g[k-j] <= M
    # With g symmetric: g[k-j] = g[n-1-(k-j)] = g[n-1-k+j]
    
    # Build constraint matrix for the half-variables
    g_var = cp.Variable(n_half)
    constraints = [g_var >= 0]
    
    M = 0  # Will be set by caller
    # Actually, let's just return the variables and let the caller build constraints
    
    return g_var


def solve_sym_lp(f, n, M, active_idx):
    """
    Solve: max sum(g) s.t. g >= 0, g symmetric, (f*g)[k] <= M for k in active_idx.
    g parameterized by first half.
    """
    n_half = n // 2
    g_half = cp.Variable(n_half)
    
    # Build full g from half: g[j] = g_half[min(j, n-1-j)]
    # For constraint (f*g)[k] = sum_j f[j]*g[k-j]:
    # g[k-j] = g_half[min(k-j, n-1-(k-j))]
    
    constraints = [g_half >= 0]
    
    for k in active_idx:
        # (f*g)[k] = sum_{j} f[j] * g[k-j] where 0<=j<n and 0<=k-j<n
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # For each j in [lo, hi]: g[k-j] = g_half[min(k-j, n-1-k+j)]
        row = np.zeros(n_half)
        for j in range(lo, hi + 1):
            g_idx = k - j
            # g[g_idx] = g_half[min(g_idx, n-1-g_idx)]
            h_idx = min(g_idx, n - 1 - g_idx)
            row[h_idx] += f[j]
        constraints.append(cp.sum(row * g_half) <= M)
    
    # Objective: sum(g) = 2 * sum(g_half) (for even n, no middle element)
    prob = cp.Problem(cp.Maximize(2 * cp.sum(g_half)), constraints)
    
    try:
        prob.solve(solver=cp.SCS, max_iters=100000, eps=1e-7)
    except Exception:
        return None
    
    if g_half.value is None:
        try:
            prob.solve(solver=cp.ECOS, max_iters=10000)
        except Exception:
            return None
    
    if g_half.value is None:
        return None
    
    gh = np.maximum(g_half.value, 0.0)
    if np.any(np.isnan(gh)):
        return None
    
    # Reconstruct full g
    g = np.zeros(n)
    for j in range(n_half):
        g[j] = gh[j]
        g[n - 1 - j] = gh[j]
    
    return g
