import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_cross(f, g):
    """Compute cross-convolution f*g via FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_nonneg_sum1(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 1e-15:
        v /= s
    else:
        v = np.ones(len(v)) / len(v)
    return v


def build_constraint_matrix(f, indices, n):
    """Build constraint matrix where row i encodes (f*g)[k_i] = A[i,:] @ g."""
    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp_max_sum(f, n, M, active_idx, solver='SCS', eps=1e-9):
    """Solve: max sum(g) s.t. g>=0, (f*g)[k] <= M for k in active_idx."""
    if len(active_idx) == 0:
        return None
    A = build_constraint_matrix(f, active_idx, n)
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=solver, max_iters=500000, eps=eps)
    except Exception:
        return None
    if g.value is None:
        if n <= 3000:
            try:
                prob.solve(solver=cp.ECOS, max_iters=30000)
            except Exception:
                return None
    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None
    return g_val


def solve_lp_min_peak(f, n, active_idx, solver='SCS', eps=1e-9):
    """Solve: min t s.t. t >= 0, (f*g)[k] <= t, sum(g) >= 1, g >= 0."""
    if len(active_idx) == 0:
        return None, None
    A = build_constraint_matrix(f, active_idx, n)
    g = cp.Variable(n)
    t = cp.Variable()
    constraints = [g >= 0, cp.sum(g) >= 1.0, A @ g <= t, t >= 0]
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=solver, max_iters=500000, eps=eps)
    except Exception:
        return None, None
    if g.value is None:
        if n <= 3000:
            try:
                prob.solve(solver=cp.ECOS, max_iters=30000)
            except Exception:
                return None, None
    if g.value is None:
        return None, None
    g_val = np.maximum(g.value, 0.0)
    t_val = float(t.value) if t.value is not None else None
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None, None
    return g_val, t_val

def spectral_peak_reduction(f_init, n_iters=500, time_limit=None, seed=0):

            break

        F = rfft(f, fft_size)
        R = irfft(F * F, fft_size)[:2*n-1]
        peak = R.max()
        peak_idx = R.argmax()

        # Target: reduce peak by adjusting frequency magnitudes
        # The contribution of frequency k to R[peak_idx] involves |F|^2
        # Simple approach: reduce frequencies that contribute most to the peak
        # R[k] = sum_j |F[j]|^2 * exp(2*pi*j*k/N) ... this is complex
        
        # Alternative: use the "cyclic reduction" approach
        # Reduce the auto-correlation at the peak by modifying the phase
        # But we need to maintain non-negativity...
        
        # Simpler approach: gradient descent on |F|^2 to minimize peak
        # grad_{|F|^2} R[peak] involves the DFT matrix
        
        # Practical approach: iteratively reduce the time-domain sequence at positions
        # that contribute most to the peak
        contrib = np.zeros(n)
        k = peak_idx
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):

        # Reduce the contributing positions

        if max_contrib > 1e-10:

def subgradient_optimize(f_init, time_limit, n_top=50, lr_init=0.003,

    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:

def coordinate_descent(f_init, time_limit, n_sweeps=50):

    t0 = time.monotonic()

            if f[j] < 1e-14:

                        break
                    else:
                        f[j] = 0.0
                        f = project_nonneg_sum1(f)
                continue

            best_val = f[j]
            best_obj_j = best_obj
            
            for mult in [0.7, 0.85, 1.15, 1.3]:
                nv = f[j] * mult
                f[j] = nv
                f = project_nonneg_sum1(f)
                obj = fast_convolve(f).max()
                if obj < best_obj_j - 1e-13:
                    best_obj_j = obj
                    best_val = nv
            
            f[j] = best_val
            f = project_nonneg_sum1(f)
            if best_obj_j < best_obj - 1e-13:
                best_obj = best_obj_j
                best_f = f.copy()
                improved = True

        if not improved and sweep > 5:
            break

    return best_f, best_obj

def convolve_cross(f, g):
    """Compute cross-convolution f*g via FFT."""
    n = max(len(f), len(g))
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_nonneg_sum1(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 1e-15:
        v = v / s
    else:
        v = np.ones(len(v)) / len(v)
    return v


def build_constraint_matrix(f, indices, n):
    """Build constraint matrix where row i encodes (f*g)[k_i] = A[i,:] @ g."""
    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp_max_sum(f, n, M, active_idx, eps=1e-9):
    """Solve: max sum(g) s.t. g>=0, (f*g)[k] <= M for k in active_idx."""
    if len(active_idx) == 0:
        return None
    A = build_constraint_matrix(f, active_idx, n)
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=500000, eps=eps)
    except Exception:
        pass
    if g.value is None:
        try:
            prob.solve(solver=cp.ECOS, max_iters=30000)
        except Exception:
            return None
    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None
    return g_val


def cp_lp_optimize(f_init, time_limit, initial_threshold=0.85, max_constraints=400,
                   min_constraints=30, label="cp", verbose=True):
    """
    Cutting-plane LP optimization with adaptive threshold.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = project_nonneg_sum1(f_init.copy())
    best_f = f.copy()
    b = fast_convolve(f)
    M = b.max()
    best_obj = M
    threshold = initial_threshold
    it = 0
    stuck = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 3:
            break

        # Select constraints: peaks above threshold
        active_mask = b >= threshold * M
        active_idx = np.where(active_mask)[0]

        if len(active_idx) < min_constraints:
            extra = np.argpartition(b, -min_constraints)[-min_constraints:]
            active_idx = np.union1d(active_idx, extra)
        if len(active_idx) > max_constraints:
            active_idx = active_idx[np.argsort(b[active_idx])[-max_constraints:]]
        active_idx = np.sort(active_idx)

        # Solve LP
        g = solve_lp_max_sum(f, n, M, active_idx, eps=1e-9)

        if g is None:
            stuck += 1
            if stuck > 8:
                if threshold > 0.5:
                    threshold -= 0.05
                    stuck = 0
                else:
                    break
            continue

        S_g = g.sum()
        if S_g <= 1.0 + 1e-5:
            stuck += 1
            if stuck > 25:
                if threshold > 0.55:
                    threshold -= 0.03
                    stuck = 0
                else:
                    break
            continue

        stuck = 0

        # Cutting plane: add violated constraints
        fg = convolve_cross(f, g)
        violated = fg > M * (1 + 1e-6)
        if violated.any() and len(active_idx) < max_constraints:
            v_idx = np.where(violated)[0]
            not_active = v_idx[~np.isin(v_idx, active_idx)]
            if len(not_active) > 0:
                n_add = min(25, len(not_active))
                to_add = not_active[np.argsort(fg[not_active])[-n_add:]]
                active_idx = np.unique(np.concatenate([active_idx, to_add]))

        # Normalize direction
        h = g / S_g

        # Adaptive line search
        best_t = 0.05
        best_t_obj = M

        # Scan
        n_scan = 25
        scan_steps = np.linspace(0.005, 0.6, n_scan)
        for t in scan_steps:
            f_trial = np.maximum((1.0 - t) * f + t * h, 0.0)
            f_trial = project_nonneg_sum1(f_trial)
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t

        # Golden section refinement
        if best_t > 0.01:
            lo = max(0.002, best_t - 0.08)
            hi = min(0.95, best_t + 0.12)
            for _ in range(10):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = project_nonneg_sum1(np.maximum((1 - m1) * f + m1 * h, 0.0))
                f2 = project_nonneg_sum1(np.maximum((1 - m2) * f + m2 * h, 0.0))
                o1 = fast_convolve(f1).max()
                o2 = fast_convolve(f2).max()
                if o1 <= o2:
                    hi = m2
                else:
                    lo = m1
            t_mid = (lo + hi) / 2.0
            f_mid = project_nonneg_sum1(np.maximum((1 - t_mid) * f + t_mid * h, 0.0))
            o_mid = fast_convolve(f_mid).max()
            if o_mid < best_t_obj:
                best_t_obj = o_mid
                best_t = t_mid

        # Update
        f = project_nonneg_sum1(np.maximum((1.0 - best_t) * f + best_t * h, 0.0))
        cur_obj = fast_convolve(f).max()
        b = fast_convolve(f)
        M = cur_obj

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            threshold = min(0.97, threshold + 0.004)
        elif it % 15 == 0 and threshold > 0.6:
            threshold -= 0.02

        if verbose and it % 25 == 0:
            print(f"    {label} it={it} obj={best_obj:.12f} score={2.0*n*best_obj:.6f} "
                  f"th={threshold:.3f} |A|={len(active_idx)} t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def subgradient_optimize(f_init, time_limit, n_top=50, lr_init=0.003,
                         momentum=0.9, temp_init=0.002, seed=42):
    """Subgradient method with momentum."""
    t0 = time.monotonic()
    n = len(f_init)
    f = project_nonneg_sum1(f_init.copy())
    v = np.zeros(n)
    lr = lr_init
    temp = temp_init
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)

        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
        else:
            top_idx = np.arange(len(b))

        scaled = b[top_idx] / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        g = np.zeros(n)
        for i, k in enumerate(top_idx):
            if w[i] < 1e-12:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += w[i] * 2.0 * f[k - idx]

        v = momentum * v - lr * g
        f = project_nonneg_sum1(f + v)

        if it % 40 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 4:
                    lr *= 0.9
                    temp *= 1.15
                    no_improve = 0
                if no_improve > 25:
                    f = project_nonneg_sum1(best_f + rng.standard_normal(n) * 0.004)
                    v = np.zeros(n)
                    no_improve = 0

    return best_f, best_obj


def coordinate_descent(f_init, time_limit, n_sweeps=100):
    """Coordinate descent with multiplicative updates."""
    t0 = time.monotonic()
    n = len(f_init)
    f = project_nonneg_sum1(f_init.copy())
    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    step = max(1, n // 300)
    coords = np.arange(0, n, step)
    rng = np.random.default_rng(77)

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        coords_shuffled = coords[rng.permutation(len(coords))]
        improved = False

        for j in coords_shuffled:
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj
            if f[j] < 1e-14:
                continue

            best_val = f[j]
            best_obj_j = best_obj

            for mult in [0.75, 0.88, 1.12, 1.25]:
                f[j] = f[j] * mult
                f = project_nonneg_sum1(f)
                obj = fast_convolve(f).max()
                if obj < best_obj_j - 1e-13:
                    best_obj_j = obj
                    best_val = f[j]

            f[j] = best_val
            f = project_nonneg_sum1(f)
            if best_obj_j < best_obj - 1e-13:
                best_obj = best_obj_j
                best_f = f.copy()
                improved = True

        if not improved and sweep > 10:
            break

    return best_f, best_obj


def micro_perturb_optimize(f_init, time_limit, n_trials=50, seed=0):
    """Try small random perturbations and keep improvements."""
    t0 = time.monotonic()
    n = len(f_init)
    f = project_nonneg_sum1(f_init.copy())
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    rng = np.random.default_rng(seed)
    per_trial = time_limit / n_trials

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break
        if (time.monotonic() - t0) / max(trial, 1) > time_limit:
            break

        # Random small perturbation
        scale = 0.002 + 0.01 * (trial / n_trials)
        mask = rng.random(n) < 0.3
        f_perturbed = f.copy()
        f_perturbed[mask] += rng.standard_normal(mask.sum()) * scale
        f_perturbed = project_nonneg_sum1(np.maximum(f_perturbed, 0.0))

        # Quick subgradient from perturbed point
        t_sub = min(5.0, per_trial * 0.5)
        f_opt, obj = subgradient_optimize(f_perturbed, time_limit=t_sub,
                                           n_top=max(30, n // 12), lr_init=0.004,
                                           seed=trial)
        if obj < best_obj - 1e-11:
            best_obj = obj
            best_f = f_opt.copy()
            f = f_opt.copy()
            if trial % 10 == 0:
                print(f"    micro pert {trial}: obj={best_obj:.12f} score={2.0*n*best_obj:.6f}")

    return best_f, best_obj


def diversify_restart(f_init, n, rng):
    """Create a diversified version of the sequence."""
    # Random local perturbation
    width = max(5, n // 20)
    start = rng.integers(0, n - width)
    f_new = f_init.copy()
    f_new[start:start+width] += rng.standard_normal(width) * 0.03
    f_new = project_nonneg_sum1(np.maximum(f_new, 0.0))

    # Or: blend with a different structured sequence
    r = rng.random()
    if r < 0.3:
        # Blend with Hann window
        hann = np.hanning(n)
        blend = 0.3 * f_new + 0.7 * hann
        f_new = project_nonneg_sum1(np.maximum(blend, 0.0))
    elif r < 0.5:
        # Blend with triangular
        tri = np.concatenate([np.linspace(0, 1, n//2), np.linspace(1, 0, n - n//2)])
        blend = 0.4 * f_new + 0.6 * tri
        f_new = project_nonneg_sum1(np.maximum(blend, 0.0))
    elif r < 0.7:
        # Blend with periodic pattern
        period = rng.integers(3, 12)
        pattern = np.concatenate([np.ones(period // 2), np.zeros(period - period // 2)])
        periodic = np.tile(pattern, n // len(pattern) + 1)[:n]
        blend = 0.5 * f_new + 0.5 * periodic
        f_new = project_nonneg_sum1(np.maximum(blend, 0.0))

    return f_new


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: Multi-phase optimization combining improved cutting-plane LP,
    subgradient, coordinate descent, micro-perturbation, and multi-length exploration.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception as e:
        print(f"Parent load failed: {e}")

    if parent is None:
        parent = np.ones(1024) / 1024
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: uniform n=1024, score={p_score:.6f}")

    best_seq = parent.copy()
    best_score = p_score
    n_parent = len(parent)

    # === Phase 1: Main CP-LP (tight threshold) ===
    print(f"\n=== Phase 1: CP-LP main (n={n_parent}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.28, 450.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s, threshold=0.88, max_c=450")
        f1, obj1 = cp_lp_optimize(parent, time_limit=t1, initial_threshold=0.88,
                                   max_constraints=450, min_constraints=35, label="p1")
        score1 = float(evaluate_sequence(f1.tolist()))
        if np.isfinite(score1) and score1 < best_score:
            best_score = score1
            best_seq = f1.copy()
            print(f"  NEW BEST [p1]: {score1:.6f}")
        else:
            print(f"  p1: {score1:.6f} (no improvement)")

    # === Phase 2: CP-LP (loose threshold, more constraints) ===
    print(f"\n=== Phase 2: CP-LP loose (n={len(best_seq)}, score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.18, 280.0)
    if t2 > 30:
        print(f"  Budget: {t2:.0f}s, threshold=0.72, max_c=550")
        f2, obj2 = cp_lp_optimize(best_seq, time_limit=t2, initial_threshold=0.72,
                                   max_constraints=550, min_constraints=50, label="p2")
        score2 = float(evaluate_sequence(f2.tolist()))
        if np.isfinite(score2) and score2 < best_score:
            best_score = score2
            best_seq = f2.copy()
            print(f"  NEW BEST [p2]: {score2:.6f}")
        else:
            print(f"  p2: {score2:.6f} (no improvement)")

    # === Phase 3: Diversified restarts ===
    print(f"\n=== Phase 3: Diversified restarts (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t3_total = min(rem * 0.18, 280.0)
    if t3_total > 50:
        n_b = len(best_seq)
        n_restarts = 3
        t_per = t3_total / n_restarts
        for r in range(n_restarts):
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            t_r = min(t_per, rem)
            if t_r < 20:
                break
            f_div = diversify_restart(best_seq, n_b, rng)
            # Optimize from diversified point
            f3, _ = cp_lp_optimize(f_div, time_limit=t_r * 0.7, initial_threshold=0.82,
                                    max_constraints=300, min_constraints=30,
                                    label=f"div{r}", verbose=False)
            s3 = float(evaluate_sequence(f3.tolist()))
            if np.isfinite(s3) and s3 < best_score:
                best_score = s3
                best_seq = f3.copy()
                print(f"  NEW BEST [div{r}]: {s3:.6f}")
            else:
                print(f"  div{r}: {s3:.6f}")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.6f}) ===")
    n_base = len(best_seq)
    # Try a diverse set of lengths
    length_candidates = set()
    for factor in [0.85, 0.9, 0.95, 1.05, 1.1, 1.15, 1.25, 1.4, 1.6]:
        tl = int(n_base * factor)
        tl = max(256, min(4096, tl))
        length_candidates.add(tl)
    # Also try some absolute lengths
    for al in [512, 768, 1024, 1280, 1536, 2048, 2560, 3072]:
        length_candidates.add(al)
    # Sort by closeness to current best
    length_candidates = sorted(length_candidates, key=lambda x: abs(x - n_base))
    # Pick top 4 distinct lengths
    selected_lengths = []
    for lc in length_candidates:
        if lc != n_base and lc not in selected_lengths:
            selected_lengths.append(lc)
        if len(selected_lengths) >= 4:
            break

    for target_n in selected_lengths:
        rem = deadline - time.monotonic()
        if rem < 35:
            break
        t4 = min(rem * 0.12, 120.0)
        if t4 < 15:
            break
        old_x = np.linspace(0.0, 1.0, n_base)
        new_x = np.linspace(0.0, 1.0, target_n)
        f_interp = np.interp(new_x, old_x, best_seq)
        f_interp = project_nonneg_sum1(np.maximum(f_interp, 0.0))

        print(f"  CP-LP n={target_n}, budget={t4:.0f}s")
        f4, _ = cp_lp_optimize(f_interp, time_limit=t4, initial_threshold=0.82,
                                max_constraints=300, min_constraints=30,
                                label=f"n{target_n}", verbose=False)
        s4 = float(evaluate_sequence(f4.tolist()))
        if np.isfinite(s4) and s4 < best_score:
            best_score = s4
            best_seq = f4.copy()
            print(f"  NEW BEST [n{target_n}]: {s4:.6f}")
        else:
            print(f"  n={target_n}: {s4:.6f}")

    # === Phase 5: Subgradient polish ===
    print(f"\n=== Phase 5: Subgradient (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.08, 100.0)
    if t5 > 15:
        n_b = len(best_seq)
        print(f"  Budget: {t5:.0f}s, n={n_b}")
        f5, _ = subgradient_optimize(best_seq, time_limit=t5,
                                      n_top=max(40, n_b // 10), lr_init=0.0025,
                                      temp_init=0.0015, seed=99)
        s5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(s5) and s5 < best_score:
            best_score = s5
            best_seq = f5.copy()
            print(f"  NEW BEST [sg]: {s5:.6f}")

    # === Phase 6: Final CP-LP burst ===
    print(f"\n=== Phase 6: Final CP-LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.18, 200.0)
    if t6 > 20:
        n_b = len(best_seq)
        print(f"  Budget: {t6:.0f}s, n={n_b}")
        f6, _ = cp_lp_optimize(best_seq, time_limit=t6, initial_threshold=0.90,
                                max_constraints=500, min_constraints=50, label="final")
        s6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(s6) and s6 < best_score:
            best_score = s6
            best_seq = f6.copy()
            print(f"  NEW BEST [final]: {s6:.6f}")

    # === Phase 7: Micro-perturbation ===
    rem = deadline - time.monotonic()
    if rem > 20:
        print(f"\n=== Phase 7: Micro-perturb (best={best_score:.6f}) ===")
        t7 = min(rem * 0.3, 120.0)
        n_b = len(best_seq)
        f7, _ = micro_perturb_optimize(best_seq, time_limit=t7, n_trials=40, seed=123)
        s7 = float(evaluate_sequence(f7.tolist()))
        if np.isfinite(s7) and s7 < best_score:
            best_score = s7
            best_seq = f7.copy()
            print(f"  NEW BEST [micro]: {s7:.6f}")

    # === Phase 8: Coordinate descent ===
    rem = deadline - time.monotonic()
    if rem > 10:
        print(f"\n=== Phase 8: Coord descent (best={best_score:.6f}) ===")
        t8 = min(rem * 0.6, 80.0)
        f8, _ = coordinate_descent(best_seq, time_limit=t8, n_sweeps=80)
        s8 = float(evaluate_sequence(f8.tolist()))
        if np.isfinite(s8) and s8 < best_score:
            best_score = s8
            best_seq = f8.copy()
            print(f"  NEW BEST [cd]: {s8:.6f}")

    # Final validation
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
