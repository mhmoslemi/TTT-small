"""
Optimization of step function heights to minimize 2*n*max(conv(a,a))/(sum(a))^2.

Uses a hybrid approach:
1. Smooth max (softmax) approximation with adaptive temperature
2. L-BFGS-B for second-order local optimization
3. Projected gradient with momentum for robustness
4. Coordinate descent for final polish
5. Multi-length and multi-restart exploration
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time
import sys

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_smooth_max_and_grad(a, tau):
    """
    Compute smooth max of conv(a,a) and its gradient.
    Returns (value, gradient).
    """
    n = len(a)
    b = fast_convolve(a)
    
    # Softmax weights
    tb = tau * b
    tb_max = tb.max()
    exp_tb = np.exp(tb - tb_max)
    sum_exp = exp_tb.sum()
    w = exp_tb / sum_exp
    
    value = (tb_max + np.log(sum_exp)) / tau
    
    # Gradient via FFT
    # grad[j] = sum_k w[k] * (a[k-j] + a[j-k])
    # First term: cross-correlation of w and a
    # Second term: convolution of w and a
    N = 1 << (3 * n).bit_length()  # FFT size
    A = rfft(a, N)
    W = rfft(w, N)
    
    # Cross-correlation: sum_m w[m] * a[m-j]
    # = irfft(W * conj(A))
    corr = irfft(W * np.conj(A), N)
    
    # Convolution: sum_m w[m] * a[j-m]  
    # = irfft(W * A)
    conv = irfft(W * A, N)
    
    grad = (corr + conv)[:n]
    
    return value, grad

def smooth_max_objective_and_grad(a, tau):
    """Objective and gradient for L-BFGS on the simplex (via penalty)."""
    value, grad = compute_smooth_max_and_grad(a, tau)
    # Add gradient of penalty (sum(a) - 1)^2
    s = a.sum()
    penalty_grad = 2.0 * 100.0 * (s - 1.0) * np.ones_like(a)
    value += 100.0 * (s - 1.0) ** 2
    return value, grad + penalty_grad

def lbfgs_smooth_optimize(a_init, tau_init=10.0, tau_final=500.0, 
                           n_reductions=20, time_limit=300.0, label="lbfgs"):
    """
    L-BFGS optimization with adaptive temperature (smooth max).
    """
    t0 = time.monotonic()
    a = a_init.copy().astype(np.float64)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    tau = tau_init
    # Geometric schedule
    tau_ratios = (tau_final / tau_init) ** (1.0 / n_reductions)
    
    for stage in range(n_reductions):
        if time.monotonic() - t0 > time_limit:
            break
        
        tau = tau_init * (tau_ratios ** stage)
        
        def obj_grad(x):
            val, g = smooth_max_objective_and_grad(x, tau)
            return val, g
        
        try:
            result = minimize(obj_grad, a, jac=True, method='L-BFGS-B',
                             bounds=[(0, None)] * n,
                             options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-10,
                                     'maxls': 50})
            
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = np.maximum(result.x, 0.0)
                s_new = a_new.sum()
                if s_new > 1e-10:
                    a_new = a_new / s_new
                    obj = fast_convolve(a_new).max()
                    if obj < best_obj:
                        best_obj = obj
                        best_a = a_new.copy()
                        a = a_new
        except Exception as e:
            pass
        
        elapsed = time.monotonic() - t0
        if stage % 5 == 0:
            print(f"    {label} stage={stage} tau={tau:.1f} obj={best_obj:.10f} "
                  f"score={2.0*n*best_obj:.6f} t={elapsed:.0f}s")
    
    return best_a, best_obj

def projected_gradient_optimize(a_init, time_limit=200.0, tau_init=5.0, tau_final=200.0,
                                 lr_init=0.001, momentum=0.9, label="pgd"):
    """
    Projected gradient descent with adaptive temperature and momentum.
    """
    t0 = time.monotonic()
    a = a_init.copy().astype(np.float64)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    
    n = len(a)
    v = np.zeros(n)
    lr = lr_init
    tau = tau_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    no_improve = 0
    it = 0
    
    # Adaptive schedule
    total_stages = max(1, int(time_limit / 2))  # ~2s per stage
    tau_ratio = (tau_final / tau_init) ** (1.0 / total_stages)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Update tau
        tau = min(tau * tau_ratio, tau_final)
        
        b = fast_convolve(a)
        M = b.max()
        
        # Softmax weights
        tb = tau * b
        tb_max = tb.max()
        exp_tb = np.exp(tb - tb_max)
        w = exp_tb / exp_tb.sum()
        
        # Gradient
        N_ftt = 1 << (3 * n).bit_length()
        A = rfft(a, N_ftt)
        W = rfft(w, N_ftt)
        corr = irfft(W * np.conj(A), N_ftt)
        conv = irfft(W * A, N_ftt)
        grad = (corr + conv)[:n]
        
        # Momentum update
        v = momentum * v - lr * grad
        a_new = a + v
        
        # Project onto simplex
        a = project_simplex(a_new)
        
        if it % 50 == 0:
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
                lr = min(lr * 1.02, 0.01)
            else:
                no_improve += 1
                if no_improve > 5:
                    lr *= 0.9
                    no_improve = 0
            
            if it % 200 == 0:
                print(f"    {label} it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} "
                      f"tau={tau:.1f} lr={lr:.5f} t={time.monotonic()-t0:.0f}s")
    
    return best_a, best_obj

def coordinate_descent(a_init, time_limit=60.0):
    """Coordinate descent with multiple multipliers."""
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy()
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    
    multipliers = np.array([0.3, 0.5, 0.7, 0.85, 0.9, 0.95, 1.05, 1.1, 1.2, 1.4, 1.8, 2.5])
    
    no_pass_imp = 0
    while time.monotonic() - t0 < time_limit:
        any_imp = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[j] < 1e-14:
                continue
            
            best_j_val = a[j]
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = a[j] * mult
                a_trial = a.copy()
                a_trial[j] = new_val
                s = a_trial.sum()
                if s < 1e-12:
                    continue
                a_trial = a_trial / s
                obj = fast_convolve(a_trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val
            
            if best_j_val != a[j]:
                a[j] = best_j_val
                s = a.sum()
                if s > 1e-12:
                    a = a / s
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    any_imp = True
        
        if not any_imp:
            no_pass_imp += 1
            if no_pass_imp >= 2:
                break
        else:
            no_pass_imp = 0
    
    return best_a, best_obj

def interp_sequence(src, target_n):
    """Interpolate a sequence to a new length."""
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result

"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

Uses:
1. Smooth max (softmax) with adaptive temperature + analytic gradient via FFT
2. L-BFGS-B for second-order local optimization
3. Nesterov-accelerated projected gradient with backtracking
4. Incremental coordinate descent for final polish
5. Multi-length exploration with transfer
6. Multi-restart for robustness
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_max_grad(a, tau):
    """
    Compute smooth max of conv(a,a) and its gradient.
    Returns (value, gradient).
    """
    n = len(a)
    b = fast_convolve(a)

    # Softmax weights (numerically stable)
    tb = tau * b
    tb_max = tb.max()
    exp_tb = np.exp(tb - tb_max)
    sum_exp = exp_tb.sum()
    w = exp_tb / sum_exp

    value = (tb_max + np.log(sum_exp)) / tau

    # Gradient via FFT
    # grad[j] = sum_k w[k] * (a[k-j] + a[j-k])
    N_ftt = 1 << (3 * n - 1).bit_length()
    A = rfft(a, N_ftt)
    W = rfft(w, N_ftt)

    # Cross-correlation: sum_m w[m] * a[m-j]
    S1 = irfft(W * np.conj(A), N_ftt)
    # Convolution: sum_m w[m] * a[j-m]
    S2 = irfft(W * A, N_ftt)

    grad = (S1 + S2)[:n]
    return value, grad


def smooth_max_lj(a, tau, penalty=500.0):
    """Objective + gradient for L-BFGS with sum constraint via penalty."""
    val, grad = compute_smooth_max_grad(a, tau)
    s = a.sum()
    val += penalty * (s - 1.0) ** 2
    grad = grad + 2.0 * penalty * (s - 1.0) * np.ones_like(a)
    return val, grad


def lbfgs_optimize(a_init, tau_init=5.0, tau_final=500.0,
                   n_stages=15, time_limit=300.0, label="lbfgs",
                   max_iter_per_stage=400, restart_every=40):
    """L-BFGS-B optimization with adaptive temperature schedule."""
    t0 = time.monotonic()
    a = a_init.copy().astype(np.float64)
    s = a.sum()
    if s > 1e-15:
        a = a / s

    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    tau_ratio = (tau_final / tau_init) ** (1.0 / max(1, n_stages))

    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        tau = tau_init * (tau_ratio ** stage)

        local_best = a.copy()
        local_best_obj = fast_convolve(a).max()
        stall = 0

        for sub_it in range(max_iter_per_stage):
            if time.monotonic() - t0 > time_limit:
                break

            try:
                res = minimize(smooth_max_lj, a, jac=True, method='L-BFGS-B',
                               bounds=[(0.0, None)] * n,
                               args=(tau,),
                               options={'maxiter': 50, 'ftol': 1e-14,
                                        'gtol': 1e-12, 'maxls': 30,
                                        'disp': False})

                if res.x is not None and np.all(np.isfinite(res.x)):
                    a_new = np.maximum(res.x, 0.0)
                    sn = a_new.sum()
                    if sn > 1e-10:
                        a_new = a_new / sn
                        obj = fast_convolve(a_new).max()
                        if obj < local_best_obj - 1e-13:
                            local_best = a_new.copy()
                            local_best_obj = obj
                            stall = 0
                        else:
                            stall += 1
                        a = a_new
            except Exception:
                stall += 1

            if stall >= restart_every:
                # Perturb and restart
                rng = np.random.default_rng(stage * 100 + sub_it)
                a = project_simplex(local_best + rng.standard_normal(n) * 0.003)
                stall = 0

        if local_best_obj < best_obj - 1e-13:
            best_obj = local_best_obj
            best_a = local_best.copy()

        if stage % 4 == 0:
            print(f"    {label} stage={stage} tau={tau:.1f} best_obj={best_obj:.10f} "
                  f"score={2.0*n*best_obj:.6f} t={time.monotonic()-t0:.0f}s")

    return best_a, best_obj


def nesterov_pgd(a_init, time_limit=200.0, tau_init=3.0, tau_final=300.0,
                 lr_init=0.002, label="pgd"):
    """Nesterov-accelerated projected gradient with adaptive temperature and backtracking."""
    t0 = time.monotonic()
    a = a_init.copy().astype(np.float64)
    s = a.sum()
    if s > 1e-15:
        a = a / s

    n = len(a)
    v = np.zeros(n)
    t_nesterov = 1.0
    lr = lr_init
    tau = tau_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    no_improve = 0
    it = 0

    tau_growth = 1.015

    while time.monotonic() - t0 < time_limit:
        it += 1
        tau = min(tau * tau_growth, tau_final)

        # Compute gradient at look-ahead point for Nesterov
        a_look = a + (t_nesterov / (t_nesterov + 1.0)) * v
        a_look = np.maximum(a_look, 0.0)
        s_look = a_look.sum()
        if s_look > 1e-15:
            a_look = a_look / s_look

        _, grad = compute_smooth_max_grad(a_look, tau)

        # Nesterov momentum
        beta = (1.0 - 1.0 / (t_nesterov + 1.0)) / 2.0
        v = beta * v - lr * grad

        # Update
        a_new = a + ((1.0 + beta) / (t_nesterov + 1.0)) * v - (beta / (t_nesterov + 1.0)) * v
        t_nesterov += 1.0

        # Project onto simplex
        a = project_simplex(a_new)

        # Backtracking: check if we made progress
        if it % 20 == 0:
            obj = fast_convolve(a).max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
                lr = min(lr * 1.01, 0.008)
            else:
                no_improve += 1
                if no_improve > 4:
                    lr *= 0.85
                    # Momentum restart
                    v = np.zeros(n)
                    t_nesterov = 1.0
                    no_improve = 0

            if it % 200 == 0:
                print(f"    {label} it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} "
                      f"tau={tau:.1f} lr={lr:.6f} t={time.monotonic()-t0:.0f}s")

    return best_a, best_obj

                # Second term
                k_start2 = max(0, j - n + 1)
                k_end2 = min(full_len - 1, j)
                if k_start2 <= k_end2:
                    idx2 = k_end2 - k_start2 + 1
                    # a[j-k] for k = k_start2, ..., k_end2
                    # = a[j-k_start2], a[j-k_start2-1], ..., a[j-k_end2]
                    # = a[j-k_start2], ..., a[j-k_end2]
                    # j - k_start2 down to j - k_end2
                    # = a[j-k_start2 : j-k_end2-1 : -1]
                    # Let's just use a fancy index
                    ks = np.arange(k_start2, k_end2 + 1)
                    b_new[ks] += delta * a[j - ks]

                obj_new = b_new.max()
                if obj_new < best_j_obj - 1e-13:
                    best_j_obj = obj_new
                    best_j_val = new_val

            if best_j_val != a[j]:
                delta = best_j_val - a[j]
                a[j] = best_j_val
                # Update b incrementally
                k_start = max(0, int(j))
                k_end = min(full_len - 1, int(j) + n - 1)
                if k_start <= k_end:
                    idx = k_end - k_start + 1
                    b[k_start:k_end+1] += delta * a[0:idx]
                k_start2 = max(0, int(j) - n + 1)
                k_end2 = min(full_len - 1, int(j))
                if k_start2 <= k_end2:
                    ks = np.arange(k_start2, k_end2 + 1)
                    b[ks] += delta * a[int(j) - ks]

                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    any_imp = True

        if not any_imp:
            no_pass_imp += 1
            if no_pass_imp >= 2:
                break
        else:
            no_pass_imp = 0

    # Final normalize
    s = best_a.sum()
    if s > 1e-15:
        best_a = best_a / s

    return best_a, fast_convolve(best_a).max()


def interp_sequence(src, target_n):
    """Interpolate a sequence to a new length."""
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result


def tapered_init(n, alpha=2.0):
    """Create a tapered (raised cosine) initial sequence."""
    x = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x)) ** alpha
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Hybrid optimization combining smooth-max L-BFGS, Nesterov PGD,
    and coordinate descent for minimizing the convolution peak score.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
    except Exception:
        pass

    if parent is not None:
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    else:
        parent = np.ones(512) / 512.0
        p_score = 2.0
        print(f"No parent, using uniform n=512, score=2.0")

    best_seq = parent.copy()
    best_score = p_score

    n_parent = len(parent)

    # ============================================================
    # Phase 1: Multi-length quick screening
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 1: Multi-length screening (best: {best_score:.6f})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    t1 = min(rem * 0.12, 180.0)

    candidate_ns = sorted(set(
        [max(128, min(4000, int(n_parent * f))) for f in
         [0.5, 0.65, 0.8, 0.9, 1.0, 1.1, 1.25, 1.5, 1.8, 2.2]] +
        [256, 384, 512, 768, 1024, 1280, 1536, 2048, 2560, 3072]
    ))

    t_per = max(3.0, t1 / max(1, len(candidate_ns)))
    screened = []

    for n_trial in candidate_ns:
        rem = deadline - time.monotonic()
        if rem < 5:
            break
        alloc = min(t_per, rem * 0.85)
        if alloc < 3:
            continue

        f_init = interp_sequence(parent, n_trial)
        f_res, _ = nesterov_pgd(
            f_init, time_limit=alloc, tau_init=2.0, tau_final=40.0,
            lr_init=0.003, label=f"scr{n_trial}")

        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score):
            screened.append((real_score, n_trial, f_res.copy()))
            if real_score < best_score:
                best_score = real_score
                best_seq = f_res.copy()
                print(f"  NEW BEST [scr n={n_trial}]: {real_score:.6f}")

    # Sort screened by score
    screened.sort(key=lambda x: x[0])
    for sc, nt, _ in screened[:5]:
        print(f"  Screened n={nt}: {sc:.6f}")

    # ============================================================
    # Phase 2: L-BFGS on best candidate
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 2: L-BFGS main (best: {best_score:.6f}, n={len(best_seq)})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    t2 = min(rem * 0.25, 380.0)

    if t2 > 30:
        print(f"  L-BFGS n={len(best_seq)}, budget={t2:.0f}s")
        f_res, _ = lbfgs_optimize(
            best_seq, tau_init=5.0, tau_final=400.0,
            n_stages=18, time_limit=t2, label="lbfgs_main",
            max_iter_per_stage=300, restart_every=30)

        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = f_res.copy()
            print(f"  NEW BEST [lbfgs_main]: {real_score:.6f}")

    # ============================================================
    # Phase 3: Nesterov PGD long run
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 3: PGD long (best: {best_score:.6f})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    t3 = min(rem * 0.18, 250.0)

    if t3 > 20:
        print(f"  PGD n={len(best_seq)}, budget={t3:.0f}s")
        f_res, _ = nesterov_pgd(
            best_seq, time_limit=t3, tau_init=8.0, tau_final=400.0,
            lr_init=0.0015, label="pgd_long")

        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = f_res.copy()
            print(f"  NEW BEST [pgd_long]: {real_score:.6f}")

    # ============================================================
    # Phase 4: L-BFGS on second-best screened candidate
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: L-BFGS on other candidates")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    t4 = min(rem * 0.18, 220.0)

    if t4 > 30 and len(screened) > 1:
        n_best = len(best_seq)
        # Pick up to 2 other candidates with different n
        other_candidates = [(sc, nt, fs) for sc, nt, fs in screened
                           if abs(nt - n_best) > n_best * 0.2][:2]
        t_per4 = t4 / max(1, len(other_candidates))

        for sc, nt, fs in other_candidates:
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            alloc = min(t_per4, rem * 0.85)
            if alloc < 15:
                continue

            # Quick PGD to initialize
            f_pgd, _ = nesterov_pgd(
                fs, time_limit=alloc * 0.25, tau_init=5.0,
                tau_final=80.0, lr_init=0.002, label=f"pgd4_{nt}")

            # L-BFGS
            f_res, _ = lbfgs_optimize(
                f_pgd, tau_init=10.0, tau_final=300.0,
                n_stages=12, time_limit=alloc * 0.75, label=f"lbfgs4_{nt}",
                max_iter_per_stage=250)

            real_score = float(evaluate_sequence(f_res.tolist()))
            if np.isfinite(real_score) and real_score < best_score:
                best_score = real_score
                best_seq = f_res.copy()
                print(f"  NEW BEST [other n={nt}]: {real_score:.6f}")

    # ============================================================
    # Phase 5: Coordinate descent
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: CD polish (best: {best_score:.6f})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    t5 = min(rem * 0.08, 60.0)

    if t5 > 5:
        print(f"  CD n={len(best_seq)}, budget={t5:.0f}s")
        f_res, _ = coordinate_descent(best_seq, time_limit=t5)
        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = f_res.copy()
            print(f"  NEW BEST [cd]: {real_score:.6f}")

    # ============================================================
    # Phase 6: Final L-BFGS with high-temperature precision
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 6: Final L-BFGS (best: {best_score:.6f})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    t6 = min(rem * 0.30, 250.0)

    if t6 > 20:
        print(f"  Final L-BFGS n={len(best_seq)}, budget={t6:.0f}s")
        f_res, _ = lbfgs_optimize(
            best_seq, tau_init=30.0, tau_final=800.0,
            n_stages=20, time_limit=t6, label="lbfgs_final",
            max_iter_per_stage=350, restart_every=25)

        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = f_res.copy()
            print(f"  NEW BEST [lbfgs_final]: {real_score:.6f}")

    # ============================================================
    # Phase 7: Final CD + verification
    # ============================================================
    rem = deadline - time.monotonic()
    if rem > 10:
        t7 = min(rem * 0.4, 40.0)
        f_res, _ = coordinate_descent(best_seq, time_limit=t7)
        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = f_res.copy()
            print(f"  NEW BEST [cd_final]: {real_score:.6f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy() if parent is not None else np.ones(512) / 512.0
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
