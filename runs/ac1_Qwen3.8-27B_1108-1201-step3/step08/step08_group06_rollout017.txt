import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute score assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_C(f):
    """Build LP constraint matrix C where C[k,j] = f[k-j] if 0<=k-j<n else 0."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = f
    return C


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. C*g <= M*1, g >= 0 using HiGHS."""
    n = len(f)
    C = build_C(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=C, b_ub=b_ub, bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True})
        if res.status == 0:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def greedy_lp(f, M):
    """Fast greedy approximation to the LP."""
    n = len(f)
    g = np.zeros(n)
    conv_vals = np.zeros(2 * n - 1)
    for j in range(n):
        k_start = j
        k_end = j + n - 1
        slack = M - conv_vals[k_start:k_end + 1]
        valid = f > 1e-15
        if not valid.any():
            continue
        ratios = np.full(n, np.inf)
        ratios[valid] = slack[valid] / f[valid]
        max_gj = max(0.0, ratios.min())
        g[j] = max_gj
        conv_vals[k_start:k_end + 1] += f * max_gj
    return g


def lp_fictitious_play(f_init, time_limit, verbose=False, use_greedy=False, momentum=0.9):
    """LP-based fictitious play with momentum-averaged updates."""
    f = np.maximum(f_init.astype(np.float64), 0)
    f = f / f.sum()
    n = len(f)
    f_avg = f.copy()
    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()
    iters = 0

    for k in range(10000000):
        if time.monotonic() - t0 > time_limit:
            break

        M = fast_convolve(f).max()

        if use_greedy:
            g = greedy_lp(f, M)
        else:
            g = solve_lp(f, M)
            if g is None:
                g = greedy_lp(f, M)

        if g is None or g.sum() < 1e-15:
            break

        g = g / g.sum()

        # Adaptive step size with decrease
        if k < 100:
            t_step = 0.5
        elif k < 500:
            t_step = 0.3
        elif k < 2000:
            t_step = 0.15
        else:
            t_step = max(0.05, 0.15 * (2000.0 / (k + 1)))

        # Geometric mean update in log-space
        f_safe = np.maximum(f, 1e-30)
        g_safe = np.maximum(g, 1e-30)
        log_new = (1 - t_step) * np.log(f_safe) + t_step * np.log(g_safe)
        log_new -= log_new.max()
        a_new = np.exp(log_new)
        s = a_new.sum()
        if s < 1e-30:
            a_new = f.copy()
        else:
            a_new = a_new / s

        # Momentum average
        f_avg = momentum * f_avg + (1 - momentum) * a_new
        f_avg = f_avg / f_avg.sum()

        f = a_new
        iters += 1

        sc_avg = compute_score(f_avg)
        sc_cur = compute_score(a_new)

        if sc_avg < best_score - 1e-12:
            best_score = sc_avg
            best_f = f_avg.copy()
        if sc_cur < best_score - 1e-12:
            best_score = sc_cur
            best_f = a_new.copy()

        if verbose and iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={iters}, M={M:.6e}, avg={sc_avg:.8f}, best={best_score:.8f}, t={elapsed:.0f}s")

    return best_f, best_score


def softmax_from_z(z):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def lbgfs_logspace(a0, time_limit, T=500, maxiter=3000):
    """L-BFGS in log-space with log-sum-exp smoothing."""
    n = len(a0)
    a = np.maximum(a0, 1e-30)
    a = a / a.sum()
    z0 = np.log(a)
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)

    def func_grad(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15, np.zeros_like(z)
        a_tmp = softmax_from_z(z)
        conv = fast_convolve(a_tmp)
        shifted = conv / T
        max_s = shifted.max()
        w = np.exp(shifted - max_s)
        w = w / w.sum()
        log_sum_exp = max_s + np.log(w.sum())
        obj = (2.0 * n / T) * log_sum_exp * T  # = 2n * log_sum_exp

        # Gradient: 2n * (w cross-correlated with a), projected to log-space
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a_tmp, fft_size)
        grad_a = 2.0 * n * irfft(np.conj(W) * A, fft_size)[:n]
        S = np.dot(grad_a, a_tmp)
        grad_z = a_tmp * (grad_a - S)

        if best_score and obj < best_score * T / (2 * n * T / (2 * n)):
            pass
        return obj, grad_z

    try:
        res = minimize(
            fun=lambda z: func_grad(z)[0],
            x0=z0,
            jac=lambda z: func_grad(z)[1],
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 100, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        score = compute_score(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception:
        pass

    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        best_seq, best_score = check_and_update(a_lb, f"lbgfs_{T_val}", best_seq, best_score)

    # === Phase 5: Random Restarts with short LP ===
    print(f"\n=== Phase 5: Random Restarts (best={best_score:.8f}) ===")
    n_restarts = 5
    for r in range(n_restarts):
        rem = deadline - time.monotonic() - 150
        if rem < 40:
            break
        t_alloc = min(60.0, rem / (n_restarts - r + 0.5))
        if t_alloc < 20:
            break

        # Different perturbation strategies
        if r == 0:
            # Low-frequency perturbation
            freqs = np.fft.rfftfreq(len(best_seq))
            mask = np.abs(freqs) < 0.05
            noise_spec = np.zeros(len(freqs), dtype=complex)
            noise_spec[mask & (freqs != 0)] = (rng.standard_normal(mask.sum()) + 1j * rng.standard_normal(mask.sum())) * 0.01
            noise = np.fft.irfft(noise_spec, len(best_seq))
            candidate = best_seq + noise
        elif r == 1:
            # Gaussian perturbation
            candidate = best_seq + rng.standard_normal(len(best_seq)) * 0.01
        elif r == 2:
            # Permutation of a random subset
            candidate = best_seq.copy()
            perm = rng.permutation(len(candidate))
            idx = rng.choice(len(candidate), size=len(candidate)//2, replace=False)
            candidate[idx] = best_seq[perm[idx]]
        elif r == 3:
            # Local blurring (convolve with small kernel)
            kernel = np.array([0.1, 0.3, 0.3, 0.1])
            candidate = np.convolve(best_seq, kernel, mode='same')
        else:
            # Sparse perturbation
            candidate = best_seq.copy()
            sparse_idx = rng.choice(len(candidate), size=len(candidate)//10, replace=False)
            candidate[sparse_idx] *= (1 + rng.standard_normal(len(sparse_idx)) * 0.1)

        candidate = np.maximum(candidate, 0)
        ss = candidate.sum()
        if ss > 1e-10:
            candidate = candidate / ss
            a_r, sc_r = lp_fictitious_play(candidate, t_alloc * 0.6, verbose=False, use_greedy=True, momentum=0.9)
            best_seq, best_score = check_and_update(a_r, f"restart_{r}", best_seq, best_score)
            # Also try coordinate descent on the perturbed
            rem2 = deadline - time.monotonic() - 100
            if rem2 > 15:
                a_r2, sc_r2 = coordinate_descent(a_r, min(20.0, rem2 * 0.3), verbose=False, seed=100+r)
                best_seq, best_score = check_and_update(a_r2, f"restart_{r}_cd", best_seq, best_score)

    # === Phase 6: Length Exploration ===
    print(f"\n=== Phase 6: Length Exploration (best={best_score:.8f}) ===")
    nearby_ns = sorted(set([n_parent - 2, n_parent - 1, n_parent + 1, n_parent + 2,
                            n_parent - 3, n_parent + 3])).filter if False else sorted(
        set([n_parent - 2, n_parent - 1, n_parent + 1, n_parent + 2, n_parent + 3]))
    for alt_n in nearby_ns:
        if alt_n < 64 or alt_n > 2000:
            continue
        rem = deadline - time.monotonic() - 60
        if rem < 40:
            break
        t_alloc = min(40.0, rem * 0.2)
        if t_alloc < 15:
            break
        init_alt = interpolate_sequence(best_seq, alt_n)
        print(f"  n={alt_n}, budget={t_alloc:.0f}s")
        a_alt, sc_alt = lp_fictitious_play(init_alt, t_alloc * 0.5, verbose=False, use_greedy=True, momentum=0.9)
        best_seq, best_score = check_and_update(a_alt, f"n{alt_n}_lp", best_seq, best_score)
        rem2 = deadline - time.monotonic() - 40
        if rem2 > 15:
            a_alt2, sc_alt2 = coordinate_descent(a_alt, min(15.0, rem2 * 0.4), verbose=False, seed=200)
            best_seq, best_score = check_and_update(a_alt2, f"n{alt_n}_cd", best_seq, best_score)

    # === Phase 7: Final intensive coordinate descent ===
    print(f"\n=== Phase 7: Final CD (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 20
    if rem > 20:
        t_alloc = min(60.0, rem * 0.5)
        a_fin, sc_fin = coordinate_descent(best_seq, t_alloc, verbose=False, seed=999)
        best_seq, best_score = check_and_update(a_fin, "final_cd", best_seq, best_score)

    # === Phase 8: One more LP if time ===
    rem = deadline - time.monotonic() - 10
    if rem > 20 and best_seq is not None:
        t_alloc = min(30.0, rem * 0.8)
        a_last, sc_last = lp_fictitious_play(best_seq, t_alloc, verbose=False, use_greedy=False, momentum=0.95)
        best_seq, best_score = check_and_update(a_last, "final_lp", best_seq, best_score)

    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
