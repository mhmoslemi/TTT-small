import numpy as np
import time
import cvxpy as cp
from scipy.linalg import toeplitz
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    """Build (2n-1) x n dense matrix C such that C @ g = f * g (linear conv)."""
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_best_response(f, M, prev_g=None):
    """Solve: max sum(g) s.t. C(f) @ g <= M, g >= 0."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None))
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            try:
                prob.solve(solver=cp.SCS)
            except Exception:
                return prev_g if prev_g is not None else f.copy()
    if g.value is None:
        return prev_g if prev_g is not None else f.copy()
    return np.maximum(g.value, 0)


def lp_fictitious_play(a0, time_limit, step=0.4, verbose=True, label="LP-FP"):

def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def softmax_from_z(z):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def lse_objective(a, T, n):
    """Compute (2n/T) * log(sum exp(T * conv_k)) - the smoothed objective."""
    conv = fast_convolve(a)
    cmax = conv.max()
    shifted = T * (conv - cmax)
    lse = np.log(np.exp(shifted).sum())
    return 2.0 * n * cmax + (2.0 * n / T) * lse


def lse_gradient(a, T, n):
    """
    Compute gradient of (2n/T) * log(sum exp(T*conv_k)) w.r.t. a.
    Uses cross-correlation: grad_i = 4n * sum_k w_k * a_{k-i}
    where w_k = softmax(T * conv)_k
    """
    conv = fast_convolve(a)
    cmax = conv.max()
    shifted = T * (conv - cmax)
    w = np.exp(shifted - shifted.max())
    w = w / w.sum()

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # Cross-correlation: IFFT(W * conj(A)) gives sum_l w[i+l]*a[l]
    b = irfft(W * np.conj(A), fft_size)[:n]
    grad_a = 4.0 * n * b
    return grad_a


def lbgfs_optimize(a0, time_limit, T=500, maxiter=5000, verbose=False):
    """L-BFGS on corrected log-sum-exp objective in log-space."""
    n = len(a0)
    a_init = np.maximum(a0, 1e-30)
    a_init = a_init / a_init.sum()
    z0 = np.log(a_init)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = softmax_from_z(z)
        return lse_objective(a, T, n)

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        a = softmax_from_z(z)
        grad_a = lse_gradient(a, T, n)
        S = float(np.dot(grad_a, a))
        return a * (grad_a - S)

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 100, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        score = compute_score(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
        if verbose:
            print(f"    L-BFGS T={T}: score={best_score:.8f}, fval={res.fun:.8f}")
    except Exception as e:
        if verbose:
            print(f"    L-BFGS T={T} failed: {e}")

    return best_a, best_score


def fista_optimize(a0, time_limit, T=200, verbose=False):
    """
    FISTA (accelerated proximal gradient) on the smoothed objective
    with projection onto the simplex.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    # Choose step size
    # Lipschitz constant of the gradient is bounded by 4n * max(w) * n (rough)
    # Use a conservative step with backtracking
    eta = 1.0 / (4.0 * n)  # initial step size
    k = 1
    t_k = 1.0

    for it in range(200000):
        if time.monotonic() - t0 > time_limit:
            break

        y = a + ((k - 1) / (k + t_k + 1)) * (a - (a - (eta * lse_gradient(a, T, n))))
        # Simpler FISTA:
        # y = a + ((k-1)/(k+tk+1)) * (a - prev_a)
        # Let's use a simpler momentum scheme

        # Actually let's just do plain projected gradient with diminishing step
        grad = lse_gradient(a, T, n)

        # Backtracking line search
        step = eta
        a_new = a - step * grad
        a_new = project_simplex(a_new)

        f_new = lse_objective(a_new, T, n)
        f_curr = lse_objective(a, T, n)

        # Accept if improved

            a = a_new
            score = compute_score(a)
            if score < best_score - 1e-14:
                best_score = score
                best_a = a.copy()
        else:
            # Reduce step
            eta *= 0.5

        k += 1
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA iter={it}, best={best_score:.8f}, eta={eta:.6f}, time={elapsed:.0f}s")

    return best_a, best_score


def fista_optimized(a0, time_limit, T=200, verbose=False):
    """

    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)

    a = a / s

    t_k = 1.0
    y = a.copy()

        grad_y = lse_gradient(y, T, n)

        t_k = t_new

            best_a = a.copy()

        if verbose and it % 10000 == 0:

def coordinate_descent_exact(a0, time_limit, verbose=True):

            j = rng.integers(0, n)
            if i == j:

            # f(t) = score(a with t moved from i to j), t in [-a[j], a[i]]

                    a_trial_a[j] += a_gs

                    a_trial_b[i] -= b_gs

                a = a_trial
                best_score = sc
                best_a = a.copy()
                improved = True

        if not improved and iters > 10:
            # Perturb to escape local minimum
            noise_scale = max(0.001, 0.01 / np.sqrt(iters))
            noise = rng.standard_normal(n) * noise_scale
            a = best_a.copy() + noise
            a = np.maximum(a, 0)
            a = a / a.sum()

        if verbose and iters % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"  CD iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")

    return best_a, best_score


def check_and_update(a, label, best_seq, best_score, verbose=True):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def warm_start_lp(a0, time_limit, verbose=True):
    """LP with warm start and decreasing step."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    prev_g = None
    alpha = 0.03

    while time.monotonic() - t0 < time_limit:
        iters += 1
        step = 1.0 / (1.0 + alpha * iters)
        conv = fast_convolve(a)
        M = conv.max()
        g = solve_best_response(a, M, prev_g=prev_g)
        prev_g = g
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g_norm = g / g_sum
        a_new = (1 - step) * a + step * g_norm
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        if verbose and iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LP-warm iter={iters}, best={best_score:.8f}, step={step:.5f}, "
                  f"time={elapsed:.0f}s/{time_limit:.0f}s")

    return best_a, best_score


def multi_temperature_lbgfs(a0, time_limit, verbose=True):
    """L-BFGS with temperature annealing."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    temperatures = [5000, 2000, 1000, 500, 200, 100, 50]
    per_T = max(5, time_limit // len(temperatures))

    for T in temperatures:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break
        t_alloc = min(per_T, rem * 0.4)
        if verbose:
            print(f"    L-BFGS T={T}, allocate {t_alloc:.0f}s")
        a_new, sc_new = lbgfs_optimize(best_a, t_alloc, T=T, maxiter=3000)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization with corrected L-BFGS, FISTA, and improved CD.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Initialized: n={n_init}, score={best_score:.8f}")

    n_parent = len(parent)

    # Phase 1: LP Fictitious Play - main driver
    print(f"\n=== Phase 1: LP-FP (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.35, 600.0)
    print(f"  Budget: {t1:.0f}s, n={n_parent}")
    a_lp1, sc_lp1 = lp_fictitious_play(best_seq, t1, step=0.5, label="LP-s0.5")
    best_seq, best_score = check_and_update(a_lp1, "lp_s0.5", best_seq, best_score)

    # Phase 2: LP with decreasing step (refinement)
    print(f"\n=== Phase 2: LP Decreasing Step (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.25, 400.0)
    if rem > 30:
        print(f"  Budget: {t2:.0f}s")
        a_lp2, sc_lp2 = warm_start_lp(best_seq, t2, verbose=True)
        best_seq, best_score = check_and_update(a_lp2, "lp_decr", best_seq, best_score)

    # Phase 3: Corrected multi-temperature L-BFGS
    print(f"\n=== Phase 3: Multi-T L-BFGS (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.20, 250.0)
    if rem > 20:
        print(f"  Budget: {t3:.0f}s")
        a_mt, sc_mt = multi_temperature_lbgfs(best_seq, t3, verbose=True)
        best_seq, best_score = check_and_update(a_mt, "multiT_LBFGS", best_seq, best_score)

    # Phase 4: FISTA polish
    print(f"\n=== Phase 4: FISTA (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.08, 60.0)
    if rem > 15:
        for T_val in [200, 100]:
            rem = deadline - time.monotonic()
            t4 = min(rem * 0.5, 30.0)
            if rem < 10:
                break
            print(f"  FISTA T={T_val}, budget={t4:.0f}s")
            a_f, sc_f = fista_optimized(best_seq, t4, T=T_val, verbose=False)
            best_seq, best_score = check_and_update(a_f, f"fista_T{T_val}", best_seq, best_score)

    # Phase 5: Exact coordinate descent
    print(f"\n=== Phase 5: Exact CD (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.15, 150.0)
    if rem > 15:
        print(f"  Budget: {t5:.0f}s")
        a_cd, sc_cd = coordinate_descent_exact(best_seq, t5, verbose=True)
        best_seq, best_score = check_and_update(a_cd, "exact_CD", best_seq, best_score)

    # Phase 6: Try nearby lengths
    print(f"\n=== Phase 6: Nearby Lengths (best={best_score:.8f}) ===")
    for alt_n in [n_parent + 2, n_parent - 2, n_parent + 4, n_parent + 8, n_parent - 4]:
        if alt_n < 128 or alt_n > 4096:
            continue
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t6 = min(rem * 0.15, 60.0)
        if t6 < 10:
            break
        init_alt = interpolate_sequence(best_seq, alt_n)
        print(f"  L-BFGS n={alt_n}, budget={t6:.0f}s")
        a_alt, sc_alt = lbgfs_optimize(init_alt, t6, T=500, maxiter=2000, verbose=True)
        best_seq, best_score = check_and_update(a_alt, f"n{alt_n}", best_seq, best_score)

    # Phase 7: Final LP polish
    print(f"\n=== Phase 7: Final LP (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t7 = min(rem * 0.30, 100.0)
    if rem > 15:
        print(f"  Budget: {t7:.0f}s")
        a_final, sc_final = lp_fictitious_play(best_seq, t7, step=0.15, label="LP-final")
        best_seq, best_score = check_and_update(a_final, "lp_final", best_seq, best_score)

    # Phase 8: Final L-BFGS with low temperature
    print(f"\n=== Phase 8: Final L-BFGS low-T (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t8 = min(rem * 0.40, 60.0)
    if rem > 10:
        for T_val in [50, 30]:
            rem = deadline - time.monotonic()
            t8 = min(rem * 0.5, 30.0)
            if rem < 5:
                break
            print(f"  Final LBFGS T={T_val}, budget={t8:.0f}s")
            a_fin, sc_fin = lbgfs_optimize(best_seq, t8, T=T_val, maxiter=2000, verbose=True)
            best_seq, best_score = check_and_update(a_fin, f"final_T{T_val}", best_seq, best_score)

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
        print(f"  evaluate_sequence confirms: {fs:.8f}")

    return [float(x) for x in best_seq]
