import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def get_lp_solver():

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver=None, warm_start=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""

        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None

def nesterov_mirror_descent(a0, time_limit, lr_init=0.005, top_k=50, beta_init=50.0):
    """
    Nesterov-accelerated mirror descent with smooth max approximation.
    Uses logsumexp as smooth surrogate for max, with adaptive beta.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()  # Nesterov lookahead
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    beta = beta_init
    no_improve = 0
    gamma = 1.0  # momentum parameter

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov step: y = (1+gamma)/gamma * a - 1/gamma * a_prev
        # Simplified: y = a + momentum * (a - a_prev)
        if it == 1:
            y = a.copy()
        else:
            y = (1.0 + gamma) * a - gamma * y

        # Project y to simplex for valid evaluation
        y_proj = simplex_project(y)

        # Compute smooth gradient using logsumexp
        conv_y = fast_convolve(y_proj)
        # Smooth the max
        max_conv = conv_y.max()
        scaled = beta * (conv_y - max_conv)
        # Clip for numerical stability
        scaled = np.clip(scaled, -50, 0)
        weights = np.exp(scaled)
        Z = weights.sum()
        if Z < 1e-30:
            weights = np.zeros_like(conv_y)
            weights[conv_y.argmax()] = 1.0
            Z = 1.0
        weights = weights / Z

        # Gradient of smooth max w.r.t. a
        grad = np.zeros(n)
        peak_indices = np.where(weights > 1e-10)[0]
        for k in peak_indices:
            w = weights[k]
            if w < 1e-12:
                continue
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += 2.0 * w * y_proj[k - j]

        # Mirror descent update: a_new = a * exp(-lr * grad)
        # Apply to a (not y) for stability
        log_a = np.log(a + 1e-30)
        log_a -= lr * grad
        a_new = np.exp(log_a - log_a.max())
        a = simplex_project(a_new)

        # Update best
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive learning rate and beta
        if no_improve > 100:
            lr *= 0.95
            beta *= 1.5
            no_improve = 0
            if lr < 1e-8:
                break
        elif it % 50 == 0:
            # Gradually increase beta (sharper approximation)
            beta *= 1.02

        # Update momentum (polyak step size)
        gamma = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0

    return best_a, best_score


def fast_coord_descent_v2(a0, time_limit, n_trials=10, use_momentum=False, verbose=False, label=""):
    """
    Fast coordinate descent with vectorized peak tracking.
    Key improvement: track which conv positions are near the max to avoid
    expensive full-array max computations.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    sum_a = 1.0
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve_count = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_max = float(conv[:k_start].max())
            else:
                unaff_max = -1e30
            if k_end < 2 * n - 2:
                unaff_right = float(conv[k_end + 1:].max())
                unaff_max = max(unaff_max, unaff_right)

            affected = conv[k_start:k_end + 1].copy()
            a_shifted = a[0:n]

            best_delta = 0.0
            best_new_max = M

            # Try multiple step sizes
            deltas = []
            for f in [0.01, 0.03, 0.05, 0.1, 0.2, 0.35, 0.5]:
                deltas.append(-f * a[i])
                deltas.append(f * a[i])
            # Add "to zero" and "double"
            deltas.append(-a[i] * 0.99)
            deltas.append(a[i])

            for delta in deltas:
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue
                if new_val > 1.0:
                    continue

                new_affected = affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx_2i] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
                else:
                    new_affected = new_affected.copy()

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-16:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-15:
                a[i] = max(a[i] + best_delta, 0.0)
                sum_a = a.sum()

                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx_2i] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M / (sum_a * sum_a)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve_count += 1
            if no_improve_count > 100:
                break
        else:
            no_improve_count = 0

        if verbose and sweeps % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FCDv2 {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f} M={M:.10f}")

    return best_a, best_score, sweeps

                sum_a += best_delta

                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx_2i] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M / (sum_a * sum_a)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved and sweeps > 50:
            break

    return best_a, best_score, sweeps


def peak_targeted_perturb(a0, rng, strength=0.05):
    """
    Structured perturbation that specifically targets the peak regions
    of the convolution. Reduces values that contribute to peaks and
    increases values that don't.
    """
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a0)
    M = conv.max()

    # Identify top contributing positions to the max peaks
    peak_threshold = M * 0.95
    peak_positions = np.where(conv >= peak_threshold)[0]

    # For each peak at k, the pairs (i, k-i) contribute: a[i]*a[k-i]
    # We want to reduce high-product pairs
    pressure = np.zeros(n)
    for k in peak_positions:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            product = a0[i] * a0[j]
            pressure[i] += product
            if j != i:
                pressure[j] += product * 0.5  # Less pressure on the other side

    # Normalize pressure
    p_max = pressure.max()
    if p_max > 1e-20:
        pressure = pressure / p_max

    # Apply perturbation: reduce high-pressure coords, increase low-pressure
    noise = rng.standard_normal(n)
    a_new = a * (1.0 - strength * pressure + 0.3 * strength * noise)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a0.copy()
    return a_new / s


def random_restructure(a0, rng, frac=0.3):
    """
    Randomly zero out a fraction of coordinates and redistribute mass.
    This helps escape from local optima by fundamentally changing the support.
    """
    n = len(a0)
    a = a0.copy()

    # Choose which coords to keep
    keep = rng.random(n) > frac
    n_keep = keep.sum()
    if n_keep < max(10, n // 10):
        keep = np.ones(n, dtype=bool)
        n_keep = n

    a[~keep] = 0
    # Add some noise to kept elements
    a[keep] = a[keep] * (1.0 + rng.standard_normal(n_keep) * 0.2)
    a = np.maximum(a, 0)

    # Add small random mass to some zeroed positions
    zero_mask = (a < 1e-15)
    if zero_mask.any():
        n_add = max(1, int(zero_mask.sum() * 0.1))
        add_idx = np.where(zero_mask)[0]
        add_idx = rng.choice(add_idx, size=min(n_add, len(add_idx)), replace=False)
        a[add_idx] = rng.random(n_add) * a.max() * 0.5

    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    return a / s


def nesterov_fictitious_play(n, time_limit, initial=None, solver=None, verbose=False):
    """
    Fictitious play with Nesterov-style averaging for faster convergence.
    Uses the "Optimistic" variant: the average tracks a sequent rather than the iterates.
    """
    t0 = time.monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    b = a.copy()
    # Nesterov sequent
    a_sequent = a.copy()
    b_sequent = b.copy()

    best_a = a.copy()
    best_score = float('inf')
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1

        # Solve LP best responses
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
        a_new, M_a = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new

        # Nesterov sequent update (Aginag-Tseng style)
        gamma_it = 2.0 / (iters + 2.0)
        a_sequent = (1.0 - gamma_it) * a_sequent + gamma_it * a
        b_sequent = (1.0 - gamma_it) * b_sequent + gamma_it * b

        # Evaluate candidates
        for cand in [a, b, a_sequent, b_sequent, 0.5 * (a_sequent + b_sequent)]:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand_norm = cand / sc
            score = 2.0 * n * fast_convolve(cand_norm).max()
            if score < best_score:
                best_score = score
                best_a = cand_norm.copy()

        if verbose and iters % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NFP n={n} it={iters} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, iters


def damped_lp_refine(a0, time_limit, solver=None, verbose=False):
    """Damped LP refinement with adaptive step size."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(10000):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            continue
        # Adaptive damping
        alpha = min(1.0, 5.0 / (it + 1))
        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    DLP it={it} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_symmetric(n, rng):
    """Create a symmetric initialization (a_i = a_{n-1-i})."""
    half = n // 2
    # Random profile for half the sequence
    t = np.linspace(0, 1, half)
    # Use a superposition of cosines for the half-profile
    n_comp = int(rng.integers(2, 8))
    profile = np.ones(half)
    for j in range(n_comp):
        freq = rng.uniform(0.5, 4.0)
        amp = rng.uniform(0.1, 0.4)
        phase = rng.uniform(0, np.pi)
        profile += amp * np.cos(np.pi * freq * t + phase)
    profile = np.maximum(profile, 0)

    # Extend to full symmetric sequence
    if n % 2 == 0:
        a = np.concatenate([profile, profile[::-1]])
    else:
        mid = max(profile[-1], profile[0]) * rng.uniform(0.5, 1.5)
        a = np.concatenate([profile, [mid], profile[::-1]])

    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def init_bimodal(n, rng):
    """Initialize with bimodal distribution (mass at two locations)."""
    center1 = n * rng.uniform(0.2, 0.4)
    center2 = n * rng.uniform(0.6, 0.8)
    width1 = n * rng.uniform(0.05, 0.15)
    width2 = n * rng.uniform(0.05, 0.15)
    t = np.arange(n)
    a = np.exp(-0.5 * ((t - center1) / width1) ** 2) + np.exp(-0.5 * ((t - center2) / width2) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def init_flat_with_ridges(n, rng):
    """Flat base with periodic ridges (tries to create flat convolution)."""
    period = int(n / rng.integers(3, 8))
    t = np.arange(n)
    a = np.ones(n) * 0.5
    ripple_amp = rng.uniform(0.1, 0.3)
    a += ripple_amp * np.cos(2 * np.pi * t / period)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate():
    """
    Multi-phase optimization:
    Phase 1: Multi-length diverse initialization + fast CD
    Phase 2: Intensive CD with structured perturbations
    Phase 3: Nesterov mirror descent
    Phase 4: LP fictitious play with Nesterov averaging
    Phase 5: Fine polishing
    Phase 6: Final intensive optimization
    """
    deadline = time.monotonic() + 1790.0
    rng = np.random.default_rng(42)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    # === Phase 1: Multi-length diverse initialization ===
    print("\n=== Phase 1: Diverse Init + Fast CD ===")
    t1_start = time.monotonic()

    n_values = [768, 896, 1024, 1152, 1280]
    if parent is not None:
        np_ = len(parent)
        if np_ not in n_values:
            n_values.insert(0, np_)

    candidates = []

    for n in n_values:
        elapsed_p1 = time.monotonic() - t1_start
        if elapsed_p1 > 150:
            break

        # Try multiple initialization types
        init_cands = []
        if parent is not None:
            init_cands.append(("parent", interpolate_seq(parent, n)))
        init_cands.append(("symmetric", init_symmetric(n, rng)))
        init_cands.append(("bimodal", init_bimodal(n, rng)))
        init_cands.append(("ridges", init_flat_with_ridges(n, rng)))
        init_cands.append(("uniform", np.ones(n) / n))

        for name, cand in init_cands:
            t_left = 150.0 - (time.monotonic() - t1_start)
            if t_left < 3:
                break
            t_alloc = max(2.0, min(5.0, t_left / 3.0))
            a_opt, s_opt, _ = fast_coord_descent_v2(cand, t_alloc, n_trials=8, label=f"p1_{name}_n{n}")
            candidates.append((s_opt, a_opt, n, name))

    candidates.sort(key=lambda x: x[0])
    print(f"  Generated {len(candidates)} candidates, top-3: " +
          ", ".join([f"{c[0]:.6f}({c[2]},{c[3]})" for c in candidates[:3]]))

    for s_c, a_c, n_c, it_c in candidates[:8]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_{it_c}_n{n_c}")

    # === Phase 2: Intensive CD with structured perturbations ===
    print(f"\n=== Phase 2: Intensive CD + Perturb (best={best_score:.6f}) ===")
    t2_start = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        current = best_seq.copy()

        for rnd in range(6):
            rem = deadline - time.monotonic() - 500
            if rem < 20:
                break
            t_alloc = min(rem * 0.25, 120.0)
            print(f"  FCD round {rnd+1}/{6}, n={n_best}, budget={t_alloc:.0f}s, score={best_score:.6f}")
            a_opt, s_opt, sweeps = fast_coord_descent_v2(current, t_alloc, n_trials=12,
                                                          verbose=True, label=f"ph2r{rnd}")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"ph2_fcd_r{rnd}")

            # Structured perturbation to escape local optimum
            if rnd < 4:
                if rnd % 2 == 0:
                    current = peak_targeted_perturb(best_seq, rng, strength=0.03 + 0.02 * rnd)
                else:
                    current = random_restructure(best_seq, rng, frac=0.15 + 0.1 * rnd)
            else:
                current = best_seq.copy()

    # === Phase 3: Nesterov Mirror Descent ===
    print(f"\n=== Phase 3: Nesterov MD (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_md = min(rem * 0.4, 150.0)
            print(f"  NMD n={len(best_seq)}, budget={t_md:.0f}s")
            a_md, s_md = nesterov_mirror_descent(best_seq, t_md, lr_init=0.008, top_k=50, beta_init=30.0)
            best_seq, best_score = check_and_update(a_md, best_seq, best_score, "nesterov_md")

    # === Phase 4: LP Fictitious Play with Nesterov Averaging ===
    print(f"\n=== Phase 4: NFP (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        fp_lengths = [n_best]
        for alt_n in [1024, 768, 1152]:
            if alt_n != n_best and 512 <= alt_n <= 1536:
                fp_lengths.append(alt_n)
        fp_lengths = fp_lengths[:3]

        for n_fp in fp_lengths:
            rem = deadline - time.monotonic() - 300
            if rem < 60:
                break
            t_alloc = min(rem * 0.35, 200.0)
            print(f"  NFP n={n_fp}, budget={t_alloc:.0f}s")

            if best_seq is not None:
                initial = interpolate_seq(best_seq, n_fp)
            else:
                initial = None

            a_fp, score_fp, iters = nesterov_fictitious_play(
                n_fp, t_alloc, initial=initial, solver=solver, verbose=False
            )
            best_seq, best_score = check_and_update(a_fp, best_seq, best_score, f"nfp_n{n_fp}")

            # Damped LP polish
            rem = deadline - time.monotonic() - 250
            if rem > 30:
                t_dl = min(rem * 0.3, 80.0)
                a_dl, _ = damped_lp_refine(a_fp, t_dl, solver=solver)
                best_seq, best_score = check_and_update(a_dl, best_seq, best_score, f"dlp_n{n_fp}")

    # === Phase 5: More Nesterov MD rounds with different params ===
    print(f"\n=== Phase 5: MD variants (best={best_score:.6f}) ===")
    if best_seq is not None:
        for lr_try in [0.003, 0.01, 0.001]:
            rem = deadline - time.monotonic() - 200
            if rem < 30:
                break
            t_md = min(rem * 0.2, 60.0)
            a_md, s_md = nesterov_mirror_descent(best_seq, t_md, lr_init=lr_try, top_k=30, beta_init=80.0)
            best_seq, best_score = check_and_update(a_md, best_seq, best_score, f"md_lr{lr_try}")

    # === Phase 6: Final intensive polishing ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        # Multiple fine CD rounds
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            t_fine = min(rem * 0.4, 120.0)
            print(f"  FineCD n={len(best_seq)}, budget={t_fine:.0f}s")
            a_fine, s_fine, _ = fine_coord_descent(best_seq, t_fine, verbose=False, label="fine1")
            best_seq, best_score = check_and_update(a_fine, best_seq, best_score, "fine_cd_1")

        # Fast CD with many trials
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_cd2 = min(rem * 0.5, 80.0)
            a_cd2, s_cd2, _ = fast_coord_descent_v2(best_seq, t_cd2, n_trials=16, label="fcd_final")
            best_seq, best_score = check_and_update(a_cd2, best_seq, best_score, "fcd_final")

        # Another fine pass
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            a_f2, s_f2, _ = fine_coord_descent(best_seq, min(rem * 0.6, 40.0), label="fine2")
            best_seq, best_score = check_and_update(a_f2, best_seq, best_score, "fine_cd_2")

    # === Phase 7: Try alternative length ===
    print(f"\n=== Phase 7: Alt Length (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 20
        if rem > 30:
            for alt_n in [n_best + 128, n_best + 256, n_best - 128]:
                if alt_n <= 0 or alt_n > 2048:
                    continue
                if rem < 20:
                    break
                t_alt = min(rem * 0.4, 50.0)
                rem = deadline - time.monotonic() - 10
                if rem < 15:
                    break
                init_alt = interpolate_seq(best_seq, alt_n)
                a_alt, s_alt, _ = fast_coord_descent_v2(init_alt, t_alt, n_trials=10, label=f"alt{alt_n}")
                best_seq, best_score = check_and_update(a_alt, best_seq, best_score, f"alt_n{alt_n}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
