def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_A(f, indices, n):
    K = len(indices)
    A = np.zeros((K, n))
    for i in range(K):
        k = indices[i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A

def interp_to_len(seq, target_n):
    """Interpolate a sequence to a new length."""
    src_n = len(seq)
    if src_n == target_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, seq)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result

def run_lp_optimization(f_init, time_limit, threshold_frac=0.005, top_k_max=300,
                        momentum_beta=0.4, label="lp", max_stuck=60):
    """
    LP-based optimization with threshold-based peak selection, momentum,
    and two-stage line search.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s
    else:
        f = np.ones(n) / n

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj

    it = 0
    stuck = 0
    momentum_dir = np.zeros(n)
    no_improve_count = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        b = fast_convolve(f)
        M = b.max()

        # Adaptive peak selection
        threshold = M * (1.0 - threshold_frac)
        idx_thresh = np.where(b >= threshold)[0]
        k_max = min(top_k_max, 2 * n - 1)
        idx_top = np.argpartition(b, -k_max)[-k_max:]
        indices = np.sort(np.union1d(idx_thresh, idx_top))

        max_constraints = min(k_max * 2, 2 * n - 1)
        if len(indices) > max_constraints:
            indices = np.sort(idx_top[:max_constraints])

        # Solve LP
        g = solve_lp_direction(f, indices, M)

        if g is None:
            top_10 = np.sort(np.argpartition(b, -10)[-10:])
            g = solve_lp_direction(f, top_10, M)
            if g is None:
                stuck += 2
                if stuck > max_stuck:
                    break
                continue

        S_g = g.sum()
        if S_g <= 1.0 + 1e-7:
            stuck += 1
            if stuck > max_stuck // 2:
                break
            continue

        h = g / S_g

        # Update momentum
        momentum_dir = momentum_beta * momentum_dir + (1.0 - momentum_beta) * h
        mom_sum = momentum_dir.sum()
        if mom_sum > 1e-10:
            h_mom = momentum_dir / mom_sum
        else:
            h_mom = h.copy()

        # Try both directions
        best_trial = f.copy()
        best_trial_obj = M
        best_trial_t = 0.0

        for direction in [h, h_mom]:
            f_new, obj_new, t_used = fine_line_search(f, direction, M)
            if obj_new < best_trial_obj:
                best_trial_obj = obj_new
                best_trial = f_new
                best_trial_t = t_used

        if best_trial_t <= 0 or best_trial_obj >= best_obj - 1e-15:
            stuck += 1
            no_improve_count += 1

            if stuck > max_stuck:
                break

            if no_improve_count > 12:
                rng = np.random.default_rng(it * 31 + n)
                pert = rng.standard_normal(n) * 0.003
                pert += 0.002 * np.sin(2.0 * np.pi * 3.0 * np.arange(n) / n)
                f = project_simplex(best_f + pert)
                momentum_dir = np.zeros(n)
                no_improve_count = 0
                stuck = 0
        else:
            f = best_trial
            best_obj = best_trial_obj
            best_f = f.copy()
            best_score = 2.0 * n * best_obj
            stuck = 0
            no_improve_count = 0

            if it % 50 == 0:
                threshold_frac = max(0.001, threshold_frac * 0.92)

        if it % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} obj={best_obj:.12f} score={best_score:.6f} "
                  f"K={len(indices)} stuck={stuck} t={elapsed:.0f}s")

    return best_f, best_score


def subgradient_polish(f_init, time_limit, lr_init=0.004, momentum=0.88):
    """Subgradient method with softmax peak weighting."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s

    v = np.zeros(n)
    lr = lr_init
    temp = 0.0008
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        scaled = (b - M) / temp
        w = np.exp(scaled)
        w = w / w.sum()

        g_sub = np.zeros(n)
        active_mask = w > 1e-10
        active_idx = np.where(active_mask)[0]
        for k in active_idx:
            w_k = w[k]
            ki = int(k)
            lo = max(0, ki - n + 1)
            hi = min(n - 1, ki)
            for i in range(lo, hi + 1):
                g_sub[i] += w_k * f[ki - i]

        v = momentum * v - lr * g_sub
        f = project_simplex(f + v)

        if it % 40 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.03, 0.025)
            else:
                no_improve += 1
                if no_improve > 4:
                    lr *= 0.93
                    no_improve = 0

            if it % 200 == 0:
                temp = min(temp * 1.2, 0.5)

            if no_improve > 25:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                v = np.zeros(n)
                no_improve = 0

            if it % 200 == 0:
                print(f"    sg it={it} obj={best_obj:.12f} score={2*n*best_obj:.6f} "
                      f"lr={lr:.5f} t={time.monotonic()-t0:.0f}s")

    return best_f, 2.0 * n * best_obj


def interp_to_len(seq, target_n):
    """Interpolate a sequence to a new length."""
    src_n = len(seq)
    if src_n == target_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, seq)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result
