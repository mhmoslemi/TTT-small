import numpy as np

from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    """Build (2n-1) x n matrix C such that C @ g = linear_conv(f, g)."""
    n = len(f)
    # C[i,j] = f[i-j] if 0 <= i-j < n, else 0
    # This is a lower-left Toeplitz matrix
    from scipy.linalg import toeplitz
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_lp_best_response(f, M, prev_g=None):
    """Solve: max sum(g) s.t. C(f) @ g <= M, g >= 0."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None),
                   verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            return prev_g if prev_g is not None else np.ones(n) / n
    if g.value is None:
        return prev_g if prev_g is not None else np.ones(n) / n
    return np.maximum(g.value, 0)

"""
Multi-phase optimizer: subgradient descent for fast initialization,
LP fictitious play for main optimization across multiple lengths,
and L-BFGS smoothing for final polish.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.linalg import toeplitz


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_lp_best_response(f, M, prev_g=None):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None), verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            return prev_g if prev_g is not None else np.ones(n) / n
    if g.value is None:
        return prev_g if prev_g is not None else np.ones(n) / n
    return np.maximum(g.value, 0)


def subgradient_descent(a0, time_limit, label="SGD"):
    """Fast subgradient descent for the max-conv objective."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        # Process in batches for efficiency
        batch = 100
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit:
                break
            iters += 1
            
            conv = fast_convolve(a)
            k_star = np.argmax(conv)
            M = conv[k_star]
            
            # Subgradient: 2 * a[(k_star - i) mod fft_size] for i in 0..n-1
            # But we need linear conv, so:
            # grad[i] = 2 * a[k_star - i] if 0 <= k_star - i < n else 0
            #           + 2 * a[i - (k_star - (n-1))] if ... 
            # Actually for linear conv: d(conv_k)/d a_i = a[k-i] + a[k-i] (from both sides)
            # Wait: conv_k = sum_j a_j * a_{k-j}
            # d/d a_i = a_{k-i} (from first a) + a_{k-i} (from second a) = 2*a_{k-i}
            # but only if 0 <= k-i < n
            # For the "second" term: a_j * a_{k-j}, d/d a_i of a_{k-j} when k-j=i means j=k-i
            # So it's a_{k-i} (from first factor) + a_{k-i} (from second factor)
            # But wait, the "second factor" contributes when j is such that k-j = i, i.e., j = k-i
            # And the "first factor" contributes when j = i, giving a_{k-i}
            # So total: a_{k-i} + a_{k-i} = 2*a_{k-i} if 0 <= k-i < n
            # But we also need: for the second term, j = k-i must be in [0, n-1], which means 0 <= k-i < n
            # And for the first term, j = i must be in [0, n-1], which means 0 <= i < n (always true)
            # And k-j = k-i must be in [0, n-1], which means 0 <= k-i < n
            # Wait, I think both terms give the same condition. Let me just use:
            # grad[i] = 2 * a[k_star - i] if 0 <= k_star - i < n, else 0
            
            idx = k_star - np.arange(n)
            grad = np.zeros(n)
            valid = (idx >= 0) & (idx < n)
            grad[valid] = 2.0 * a[idx[valid]]
            
            # Step size: eta = c / sqrt(iter)
            eta = 0.5 / np.sqrt(iters + 1)
            
            a_new = a - eta * grad
            a_new = np.maximum(a_new, 0)
            ss = a_new.sum()
            if ss < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / ss
            
            score = compute_score(a_new)
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
            a = a_new
        
        if iters % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, t={elapsed:.0f}s")
    
    return best_a, best_score


def lp_fictitious_play(a0, time_limit, step=0.4, label="LP-FP"):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    prev_g = None
    stall_count = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()
        g = solve_lp_best_response(a, M, prev_g=prev_g)
        prev_g = g
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g_norm = g / g_sum
        a_new = (1 - step) * a + step * g_norm
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1
        a = a_new
        if stall_count > 300:
            step = max(0.05, step * 0.6)
            rng = np.random.default_rng(iters * 7919)
            a = best_a + rng.standard_normal(n) * 0.002
            a = np.maximum(a, 0)
            a = a / a.sum()
            prev_g = None
            stall_count = 0
        if iters % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, step={step:.4f}, t={elapsed:.0f}s/{time_limit:.0f}s")

    return best_a, best_score


def lbgfs_polish(a0, time_limit, T=500, maxiter=5000):
    n = len(a0)
    a = np.maximum(a0, 1e-30)
    z0 = np.log(a / a.sum())
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)
    
    def softmax_z(z):
        z = z - z.max()
        e = np.exp(z)
        return e / e.sum()
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = softmax_z(z)
        conv = fast_convolve(a)
        shifted = conv / T
        m = shifted.max()
        lse = m + np.log(np.exp(shifted - m).sum())
        return 2.0 * n * T * np.exp(T * (lse - m))  # = 2n * T * sum(exp(conv/T - m)) ... hmm
        # Actually: T * lse = T*m + T*log(sum(exp(shifted-m))) = max(conv) + T*log(sum(exp((conv-max)/T)))
        # So 2n * T * lse = 2n * max(conv) + 2n*T*log(sum(...)) which is NOT what we want.
        # We want: 2n * smooth_max(conv) = 2n * T * log(sum(exp(conv_i/T)))
        # = 2n * T * (m/T + log(sum(exp((conv_i - m)/T))))
        # = 2n * (m + T * log(sum(exp((conv_i-m)/T))))
        # = 2n * lse  ... where lse = m + T*log(...) wait no.
        # lse as I defined it = m + log(sum(exp(shifted - m))) where shifted = conv/T
        # = m + log(sum(exp((conv-m)/T)))
        # T * lse = T*m + T*log(sum(exp((conv-m)/T))) = m + T*log(sum(exp((conv-m)/T)))
        # smooth_max = T * log(sum(exp(conv/T))) = T * (log(sum(exp((conv-m)/T))) + m/T) = T*log(sum(...)) + m
        # So smooth_max = T*lse? No: T*lse = T*m + T*log(sum(exp((conv-m)/T)))
        # smooth_max = m + T*log(sum(exp((conv-m)/T)))
        # These are different! T*lse ≠ smooth_max in general.
        # Let me just compute it directly:
        # smooth_max = T * log(sum(exp(conv/T)))
        # For numerical stability: = T * (max(conv)/T + log(sum(exp((conv-max(conv))/T))))
        # = max(conv) + T * log(sum(exp((conv-max(conv))/T)))
        conv = fast_convolve(a)
        m = conv.max()
        exp_terms = np.exp((conv - m) / T)
        smooth_max = m + T * np.log(exp_terms.sum())
        return 2.0 * n * smooth_max
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        a = softmax_z(z)
        conv = fast_convolve(a)
        # softmax weights on conv
        shifted = conv / T
        m_s = shifted.max()
        w = np.exp(shifted - m_s)
        w = w / w.sum()
        # grad of conv w.r.t. a: use FFT
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        # d(conv_k)/d(a) = 2 * circular_conv(w, a) [in the padded space]
        grad_a = 2.0 * irfft(W * A, fft_size)[:n]
        # Project to tangent space of simplex
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        return 2.0 * n * grad_z
    
    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                     'maxfun': maxiter * 100, 'disp': False}
        )
        a_out = softmax_z(res.x)
        score = compute_score(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception as e:
        print(f"  L-BFGS error: {e}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Multi-phase optimizer minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses subgradient descent for fast initialization, LP fictitious play as
main driver across multiple sequence lengths (especially larger n for

Key insight: larger n allows lower peak autocorrelation, beating the

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.linalg import toeplitz


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_lp_best_response(f, M, prev_g=None):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None), verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            return prev_g if prev_g is not None else np.ones(n) / n
    if g.value is None:
        return prev_g if prev_g is not None else np.ones(n) / n
    return np.maximum(g.value, 0)


def subgradient_descent(a0, time_limit, label="SGD"):

            if time.monotonic() - t0 > time_limit:
                break
            iters += 1
            conv = fast_convolve(a)
            k_star = int(np.argmax(conv))

            idx = k_star - np.arange(n)
            grad = np.zeros(n)
            valid = (idx >= 0) & (idx < n)
            grad[valid] = 2.0 * a[idx[valid]]

def lbgfs_polish(a0, time_limit, T=500, maxiter=5000):
    n = len(a0)
    a_clean = np.maximum(a0, 1e-30)
    z0 = np.log(a_clean) - np.log(a_clean.sum())
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)

    def softmax_z(z):
        z = z - z.max()
        e = np.exp(z)
        s = e.sum()
        if s < 1e-300:
            return np.ones(len(z)) / len(z)
        return e / s

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = softmax_z(z)
        conv = fast_convolve(a)
        m = conv.max()
        exp_terms = np.exp((conv - m) / T)
        smooth_max = m + T * np.log(exp_terms.sum())
        return 2.0 * n * smooth_max

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        a = softmax_z(z)
        conv = fast_convolve(a)
        m = conv.max()
        w = np.exp((conv - m) / T)
        w = w / w.sum()
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
        S = float(np.dot(grad_a, a))
        grad_z = a * (grad_a - S)
        return 2.0 * n * grad_z

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                     'maxfun': maxiter * 100, 'disp': False}
        )
        a_out = softmax_z(res.x)
        score = compute_score(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception as e:
        print(f"  L-BFGS error: {e}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n + 1)
    old_vals = np.concatenate([f_norm, [f_norm[0]]])
    new_x = np.linspace(0, 1, new_n + 1)
    new_vals = np.interp(new_x, old_x, old_vals)
    f_interp = new_vals[:-1] if len(new_vals) > new_n else new_vals
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")

        best_seq, best_score = check_and_update(a_final, "lp_final", best_seq, best_score)

    # === Phase 6: Second L-BFGS pass ===
    rem = deadline - time.monotonic()
    if rem > 50:
        t6 = min(rem * 0.5, 80.0)
        print(f"\n=== Phase 6: Final L-BFGS (best={best_score:.8f}) ===")
        a_final2, _ = lbgfs_polish(best_seq, t6, T=500, maxiter=5000)
        best_seq, best_score = check_and_update(a_final2, "lbgfs_end", best_seq, best_score)

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
