    """Compute gradient g_j = 2 * sum_k w_k f_{k-j} efficiently via FFT."""
    n = len(f)

def compute_gradient_fast(f, w):
    """
    Compute g_j = 2 * sum_k w_k * f_{k-j} efficiently.
    w has length 2n-1, f has length n.
    Result: g of length n.
    """
    n = len(f)
    size = 2 * n - 1  # length of w
    f_rev = f[::-1]
    total_size = size + n - 1  # length of conv(w, f_rev)
    fft_size = 1 << (total_size - 1).bit_length()
    W = rfft(w, fft_size)
    FR = rfft(f_rev, fft_size)
    conv = irfft(W * FR, fft_size)
    # h_j = conv[n-1+j] for j = 0, ..., n-1
    h = conv[n-1:n-1+n]
    return 2.0 * h


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_constraint_matrix(f, indices, n):
    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp_direction(f, n, M, active_idx):
    K = len(active_idx)
    if K == 0:
        return None
    A = build_constraint_matrix(f, active_idx, n)
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=200000, eps=1e-7)
    except Exception:
        return None
    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None
    return g_val

            break

            w /= w.sum()

                w = np.ones(len(b)) / len(b)

    w[top_idx] = 1.0 / k
    return w


def build_constraint_matrix(f, indices, n):
    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp(f, n, M, active_idx):
    K = len(active_idx)
    if K == 0:
        return None
    A = build_constraint_matrix(f, active_idx, n)
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=200000, eps=1e-7)
    except Exception:
        return None
    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None
    return g_val


def line_search(f, direction, M_cur):
    """Find best t for f_new = (1-t)*f + t*direction."""
    best_t = 0.0
    best_obj = M_cur
    for t in [0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7]:
        f_trial = np.maximum((1 - t) * f + t * direction, 0.0)
        s = f_trial.sum()
        if s < 1e-10:
            continue
        f_trial /= s
        obj = fast_convolve(f_trial).max()
        if obj < best_obj:
            best_obj = obj
            best_t = t
    return best_t, best_obj


def gradient_phase(f_init, time_limit, label="gd"):
    """Fast gradient descent with log-sum-exp smoothing and adaptive parameters."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    b = fast_convolve(f)
    best_obj = b.max()
    it = 0
    vel = np.zeros(n)
    momentum = 0.85
    lr = 0.005
    beta = best_obj * 0.05
    no_improve = 0
    rng = np.random.default_rng(123)

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2:
            break

        b = fast_convolve(f)
        M_cur = b.max()
        beta = max(M_cur * 0.03, 1e-8)

        if M_cur < best_obj - 1e-14:
            best_obj = M_cur
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        w = softmax_weights(b, beta)
        g = compute_gradient_fast(f, w)

        vel = momentum * vel - lr * g
        f_new = project_simplex(f + vel)

        b_new = fast_convolve(f_new)
        M_new = b_new.max()

        if M_new <= M_cur + 1e-14:
            f = f_new
        else:
            f = project_simplex(f - lr * 0.3 * g)
            lr *= 0.85
            vel *= 0.5

        if no_improve > 30:
            lr = max(lr * 1.5, 0.01)
            momentum = min(momentum + 0.02, 0.95)
            no_improve = 0

        if no_improve > 80:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            vel = np.zeros(n)
            lr = 0.005
            no_improve = 0
            momentum = 0.85

        if it % 200 == 0:
            print(f"    {label} it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} "
                  f"lr={lr:.5f} beta={beta:.2e} t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def lp_phase(f_init, time_limit, initial_k=40, max_k=400, label="lp"):
    """LP-based optimization with cutting planes."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    b = fast_convolve(f)
    M = b.max()
    best_obj = M

    if initial_k < len(b):
        active_idx = np.argpartition(b, -initial_k)[-initial_k:]
    else:
        active_idx = np.arange(len(b))
    active_idx = np.sort(active_idx)

    it = 0
    stuck = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break

        g = solve_lp(f, n, M, active_idx)

        if g is None:
            stuck += 1
            if stuck > 10:
                break
            if len(active_idx) > initial_k:
                active_idx = active_idx[np.argsort(b[active_idx])[-initial_k:]]
            continue

        S_g = g.sum()
        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > 25:
                break
            if len(active_idx) < max_k:
                b_cur = fast_convolve(f)
                new_top = np.argpartition(b_cur, -min(max_k, len(b_cur)))[-min(max_k, len(b_cur)):]
                active_idx = np.unique(np.concatenate([active_idx, new_top]))
            continue

        fg = fast_convolve(f)
        # Check violated constraints
        fg_cross = np.zeros(2 * n - 1)
        fft_size = 1 << (2 * n - 2).bit_length()
        F = rfft(f, fft_size)
        G = rfft(g, fft_size)
        fg_cross = irfft(F * G, fft_size)[:2 * n - 1]
        violated = fg_cross > M * (1 + 1e-7)
        v_count = violated.sum()

        if v_count > 0 and len(active_idx) < max_k:
            v_idx = np.where(violated)[0]
            not_active = v_idx[~np.isin(v_idx, active_idx)]
            if len(not_active) > 0:
                n_add = min(20, len(not_active))
                to_add = not_active[np.argsort(fg_cross[not_active])[-n_add:]]
                active_idx = np.unique(np.concatenate([active_idx, to_add]))
                g = solve_lp(f, n, M, active_idx)
                if g is None:
                    stuck += 1
                    if stuck > 5:
                        break
                    continue
                S_g = g.sum()

        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > 25:
                break
            continue

        stuck = 0
        h = g / S_g
        best_t, best_t_obj = line_search(f, h, M)

        f = np.maximum((1.0 - best_t) * f + best_t * h, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s

        cur_obj = fast_convolve(f).max()
        M = cur_obj

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f.copy()

        if it % 20 == 0:
            b = fast_convolve(f)
            if len(active_idx) > initial_k:
                active_scores = b[active_idx]
                keep = active_idx[np.argsort(active_scores)[-initial_k:]]
                new_top = np.argpartition(b, -initial_k)[-initial_k:]
                active_idx = np.unique(np.concatenate([keep, new_top]))

        if it % 30 == 0:
            print(f"    {label} it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} "
                  f"active={len(active_idx)} stuck={stuck} t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def coordinate_polish(f_init, time_limit):
    """Coordinate descent with multiplicative updates."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    step = max(1, n // 300)

    for sweep in range(30):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for j in range(0, n, step):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-14:
                continue
            best_val = f[j]
            best_obj_j = best_obj
            for mult in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                f[j] = best_val * mult
                s = f.sum()
                if s < 1e-10:
                    f[j] = best_val
                    continue
                f /= s
                obj = fast_convolve(f).max()
                if obj < best_obj_j - 1e-13:
                    best_obj_j = obj
                    best_val = best_val * mult
            f[j] = best_val
            s = f.sum()
            if s > 1e-10:
                f /= s
            if best_obj_j < best_obj - 1e-13:
                best_obj = best_obj_j
                improved = True
        if not improved:
            break
    return f, best_obj


def diversify(f_base, n_vars=10, time_limit=60, label="div"):
    """Create diversified variants and optimize the best one."""
    t0 = time.monotonic()
    n = len(f_base)
    rng = np.random.default_rng(456)
    best_f = f_base.copy()
    best_obj = fast_convolve(f_base).max()

    for i in range(n_vars):
        if time.monotonic() - t0 > time_limit - 10:
            break
        sigma = 0.002 + 0.005 * (i / n_vars)
        f_v = f_base + rng.standard_normal(n) * sigma
        f_v = np.maximum(f_v, 0.0)
        s = f_v.sum()
        if s < 1e-10:
            continue
        f_v /= s
        f_opt, obj_opt = gradient_phase(f_v, time_limit=min(8, time_limit - (time.monotonic() - t0)),
                                        label=f"{label}{i}")
        if obj_opt < best_obj:
            best_obj = obj_opt
            best_f = f_opt.copy()
            print(f"    {label} variant {i}: score={2.0*n*obj_opt:.6f}")

    return best_f, best_obj


def interpolate_sequence(f_orig, target_n):
    """Interpolate a sequence to a new length."""
    n_orig = len(f_orig)
    old_x = np.linspace(0.0, 1.0, n_orig)
    new_x = np.linspace(0.0, 1.0, target_n)
    f_new = np.interp(new_x, old_x, f_orig)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(target_n) / target_n
    return f_new


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: multi-phase optimization combining fast gradient descent,
    LP refinement, multi-length exploration, diversification, and coordinate polish.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception as e:
        print(f"Parent load failed: {e}")

    if parent is None:
        parent = np.ones(1024) / 1024
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: uniform n=1024, score={p_score:.6f}")

    best_seq = parent.copy()
    best_score = p_score
    n_parent = len(parent)

    # === Phase 1: Fast gradient descent on parent ===
    print(f"\n=== Phase 1: Gradient descent (n={n_parent}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.30, 450.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        f1, obj1 = gradient_phase(parent, time_limit=t1, label="gd1")
        score1 = 2.0 * n_parent * obj1
        if np.isfinite(score1) and score1 < best_score:
            best_score = score1
            best_seq = f1.copy()
            print(f"  NEW BEST [gd1]: {score1:.6f}")

    # === Phase 2: LP refinement ===
    print(f"\n=== Phase 2: LP refinement (n={len(best_seq)}, score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.25, 350.0)
    if t2 > 30:
        print(f"  Budget: {t2:.0f}s")
        f2, obj2 = lp_phase(best_seq, time_limit=t2, initial_k=50, max_k=400, label="lp2")
        score2 = float(evaluate_sequence(f2.tolist()))
        if np.isfinite(score2) and score2 < best_score:
            best_score = score2
            best_seq = f2.copy()
            print(f"  NEW BEST [lp2]: {score2:.6f}")

    # === Phase 3: Gradient + LP交替 on best ===
    print(f"\n=== Phase 3: Alternating GD+LP (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.20, 250.0)
    if t3 > 30:
        n_b = len(best_seq)
        t3a = t3 * 0.5
        t3b = t3 * 0.5
        f3a, _ = gradient_phase(best_seq, time_limit=t3a, label="gd3a")
        s3a = float(evaluate_sequence(f3a.tolist()))
        if np.isfinite(s3a) and s3a < best_score:
            best_score = s3a
            best_seq = f3a.copy()
            print(f"  NEW BEST [gd3a]: {s3a:.6f}")
        rem3 = deadline - time.monotonic()
        t3b = min(t3b, rem3 - 5)
        if t3b > 15:
            f3b, _ = lp_phase(best_seq, time_limit=t3b, initial_k=60, max_k=500, label="lp3b")
            s3b = float(evaluate_sequence(f3b.tolist()))
            if np.isfinite(s3b) and s3b < best_score:
                best_score = s3b
                best_seq = f3b.copy()
                print(f"  NEW BEST [lp3b]: {s3b:.6f}")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.6f}) ===")
    n_base = len(best_seq)
    target_lengths = []
    for factor in [0.8, 0.9, 1.05, 1.1, 1.2, 1.4]:
        tl = int(n_base * factor)
        tl = max(512, min(3000, tl))
        if tl != n_base and tl not in target_lengths:
            target_lengths.append(tl)
    target_lengths = sorted(target_lengths, key=lambda x: abs(x - n_base))

    for target_n in target_lengths[:4]:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t4 = min(rem * 0.15, 120.0)
        if t4 < 15:
            break
        f_interp = interpolate_sequence(best_seq, target_n)
        print(f"  Optimizing n={target_n}, budget={t4:.0f}s")
        f4, obj4 = gradient_phase(f_interp, time_limit=t4 * 0.6, label=f"gd_n{target_n}")
        s4 = 2.0 * target_n * obj4
        rem4 = deadline - time.monotonic()
        if rem4 > 20:
            f4b, obj4b = lp_phase(f4, time_limit=min(t4 * 0.4, rem4 - 5),
                                  initial_k=30, max_k=200, label=f"lp_n{target_n}")
            s4b = 2.0 * target_n * obj4b
            if s4b < s4:
                s4 = s4b
                f4 = f4b
        s4_eval = float(evaluate_sequence(f4.tolist()))
        if np.isfinite(s4_eval) and s4_eval < best_score:
            best_score = s4_eval
            best_seq = f4.copy()
            print(f"  NEW BEST [n{target_n}]: {s4_eval:.6f}")

    # === Phase 5: Diversification ===
    print(f"\n=== Phase 5: Diversification (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.20, 200.0)
    if t5 > 30:
        f5, obj5 = diversify(best_seq, n_vars=8, time_limit=t5, label="div")
        s5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(s5) and s5 < best_score:
            best_score = s5
            best_seq = f5.copy()
            print(f"  NEW BEST [div]: {s5:.6f}")

    # === Phase 6: Final LP polish ===
    print(f"\n=== Phase 6: Final LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.25, 200.0)
    if t6 > 20:
        n_b = len(best_seq)
        print(f"  Budget: {t6:.0f}s, n={n_b}")
        f6, obj6 = lp_phase(best_seq, time_limit=t6, initial_k=80, max_k=600, label="final")
        s6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(s6) and s6 < best_score:
            best_score = s6
            best_seq = f6.copy()
            print(f"  NEW BEST [final]: {s6:.6f}")

    # === Phase 7: Coordinate descent ===
    rem = deadline - time.monotonic()
    if rem > 15:
        print(f"\n=== Phase 7: Coord descent (best={best_score:.6f}) ===")
        t7 = min(rem * 0.6, 60.0)
        f7, _ = coordinate_polish(best_seq, time_limit=t7)
        s7 = float(evaluate_sequence(f7.tolist()))
        if np.isfinite(s7) and s7 < best_score:
            best_score = s7
            best_seq = f7.copy()
            print(f"  NEW BEST [cd]: {s7:.6f}")

    # Final evaluation
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'=' * 50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'=' * 50}")

    return [float(x) for x in best_seq]
