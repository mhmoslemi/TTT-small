"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exact 1D coordinate minimization via bisection on target level T
2. Active-set aware coordinate descent (focus on peak-contributing coords)
3. LSE-smoothed gradient optimization (Adam + simplex projection) for global moves
4. Multi-length exploration with diverse initialization
5. Simulated annealing style perturbation for escaping local optima
6. Final ultra-fine coordinate polish

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def exact_coord_delta(a, i, conv, M, n):
    """
    Find the optimal delta for coordinate i that minimizes max(conv) after update.
    Uses bisection on the target level T.
    Returns (best_delta, best_new_max).
    """
    k_start = i
    k_end = min(i + n - 1, 2 * n - 2)

    if k_start > 0:
        unaff_left = conv[:k_start].max()
    else:
        unaff_left = -1e30
    if k_end < 2 * n - 2:
        unaff_right = conv[k_end + 1:].max()
    else:
        unaff_right = -1e30
    unaff_max = max(unaff_left, unaff_right)

    affected_len = k_end - k_start + 1
    base = conv[k_start:k_end + 1]
    coeff = 2.0 * a[:affected_len]  # a[k-i] for k = i, i+1, ..., i+n-1 → a[0], a[1], ...

    quad_idx = None
    if 2 * i >= k_start and 2 * i <= k_end:
        quad_idx = 2 * i - k_start
        # coeff[quad_idx] = 2*a[i] is already correct (linear part)

    # Bisection on T (target max level)
    # Low: best we could hope (unaff_max or min of affected)
    # High: current M
    lo_T = max(unaff_max, base.min() - 1.0)
    hi_T = M + 1e-15

    # Quick check: if unaff_max >= M, we can't improve
    if unaff_max >= M - 1e-18:
        return 0.0, M

    for _ in range(50):
        T = (lo_T + hi_T) / 2.0

        if T < unaff_max:
            lo_T = T
            continue

        # Find feasible interval for delta
        lo_delta = -a[i]  # non-negativity constraint
        hi_delta = np.inf

        # Linear constraints: for each k with coeff[k] > 0:
        #   base[k] + coeff[k]*delta <= T
        #   delta <= (T - base[k]) / coeff[k]
        pos_mask = coeff > 1e-15
        if pos_mask.any():
            upper_bounds = (T - base[pos_mask]) / coeff[pos_mask]
            hi_delta = min(hi_delta, upper_bounds.min())

        # Quadratic constraint (if quad_idx exists):
        #   base[q] + coeff[q]*delta + delta^2 <= T
        #   delta^2 + coeff[q]*delta + (base[q] - T) <= 0
        if quad_idx is not None:
            c_q = coeff[quad_idx]
            b_q = base[quad_idx] - T
            disc = c_q * c_q - 4.0 * b_q
            if disc < 0:
                # No solution → infeasible at this T
                lo_T = T
                continue
            sqrt_disc = np.sqrt(disc)
            q_lo = (-c_q - sqrt_disc) / 2.0
            q_hi = (-c_q + sqrt_disc) / 2.0
            lo_delta = max(lo_delta, q_lo)
            hi_delta = min(hi_delta, q_hi)

        if lo_delta > hi_delta + 1e-15:
            # Infeasible
            lo_T = T
        else:
            # Feasible → T is achievable
            hi_T = T

    # hi_T is the minimum achievable max level
    best_T = hi_T

    # Now find the actual delta that achieves this
    lo_delta = -a[i]
    hi_delta = np.inf

    pos_mask = coeff > 1e-15
    if pos_mask.any():
        upper_bounds = (best_T - base[pos_mask]) / coeff[pos_mask]
        hi_delta = min(hi_delta, upper_bounds.min())

    if quad_idx is not None:
        c_q = coeff[quad_idx]
        b_q = base[quad_idx] - best_T
        disc = c_q * c_q - 4.0 * b_q
        if disc >= 0:
            sqrt_disc = np.sqrt(disc)
            q_lo = (-c_q - sqrt_disc) / 2.0
            q_hi = (-c_q + sqrt_disc) / 2.0
            lo_delta = max(lo_delta, q_lo)
            hi_delta = min(hi_delta, q_hi)

    if lo_delta > hi_delta:
        # Shouldn't happen, but fallback
        return 0.0, M

    # Pick the midpoint of the feasible interval (any point works, but midpoint is safer)
    best_delta = (lo_delta + hi_delta) / 2.0
    best_delta = max(best_delta, -a[i] + 1e-15)
    best_delta = min(best_delta, hi_delta - 1e-15) if hi_delta < np.inf else best_delta

    # Verify
    new_affected = base + coeff * best_delta
    if quad_idx is not None:
        new_affected[quad_idx] += best_delta * best_delta
    new_max = max(unaff_max, new_affected.max())

    if new_max < M - 1e-18:
        return best_delta, new_max
    else:
        return 0.0, M


def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    """Coordinate descent with exact 1D minimization via bisection on T."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve_sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        sweep_improvement = 0.0

        for i in range(n):
            if a[i] < 1e-18:
                continue

            delta, new_M = exact_coord_delta(a, i, conv, M, n)

            if delta != 0.0 and new_M < M - 1e-18:
                # Apply update
                a[i] += delta
                a[i] = max(a[i], 0.0)

                # Update conv incrementally
                k_start = i
                k_end = min(i + n - 1, 2 * n - 2)
                affected_len = k_end - k_start + 1
                changes = 2.0 * delta * a[:affected_len]
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    changes[idx_2i] = 2.0 * delta * (a[i] - delta) + delta * delta
                    # Wait: a[i] has already been updated. Let me redo.
                    # Actually a[i] = old_a_i + delta. The change in conv[2i] is:
                    # (old+delta)^2 - old^2 = 2*old*delta + delta^2
                    # changes[idx_2i] should be 2*(a[i]-delta)*delta + delta^2
                    # which is what we have. Good.
                conv[k_start:k_end + 1] += changes
                M = new_M
                improved = True
                sweep_improvement += (M - new_M)  # This is negative... let me just track count

        if improved:
            cur_score = 2.0 * n * M
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1
            if no_improve_sweeps > 10:
                break

        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ECD {label} sweep={sweeps} ({elapsed:.0f}s): M={M:.10f} score={best_score:.6f}")

    return best_a, best_score, sweeps


def fine_coord_descent(a0, time_limit, verbose=False, label=""):
    """Ultra-fine coordinate descent with very small multiplicative steps."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0

    factors = [0.999, 0.998, 0.997, 0.995, 0.993,
               1.001, 1.002, 1.003, 1.005, 1.007]

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-18:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            affected = conv[k_start:k_end + 1]
            a_shifted = a[:k_end - k_start + 1]
            best_delta = 0.0
            best_new_max = M

            for factor in factors:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx_2i] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
                else:
                    new_affected = new_affected.copy()

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-18:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-18:
                a[i] += best_delta
                a[i] = max(a[i], 0.0)

                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx_2i] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = best_new_max
                improved = True

        if improved:
            cur_score = 2.0 * n * M
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
        else:
            if sweeps > 5:
                break

    return best_a, best_score, sweeps


def lse_gradient_step(a, tau, top_k=None):
    """Compute the LSE-smoothed gradient of max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)

    if top_k is not None and top_k < len(conv):
        # Only use top-k peaks for efficiency
        idx = np.argpartition(conv, -top_k)[-top_k:]
        conv_work = conv[idx].copy()
        shift = conv_work.max()
        exp_v = np.exp(tau * (conv_work - shift))
        q = exp_v / exp_v.sum()
        # Distribute gradient only over top-k
        grad = np.zeros(n)
        for idx_k in range(len(idx)):
            if q[idx_k] < 1e-14:
                continue
            k = int(idx[idx_k])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                j = k - i
                if 0 <= j < n:
                    grad[i] += 2.0 * q[idx_k] * a[j]
    else:
        shift = conv.max()
        exp_v = np.exp(tau * (conv - shift))
        q = exp_v / exp_v.sum()
        grad = np.zeros(n)
        for k in range(len(conv)):
            if q[k] < 1e-14:
                continue
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                j = k - i
                if 0 <= j < n:
                    grad[i] += 2.0 * q[k] * a[j]

    return grad, conv.max()


def lse_optimize(a0, time_limit, verbose=False, label=""):
    """Optimize using LSE-smoothed gradient with Adam + simplex projection."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.75, 0.999, 1e-8
    t = 0
    tau = 20.0
    lr = 0.01
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        t += 1

        grad, cur_max = lse_gradient_step(a, tau, top_k=min(50, 2 * n - 1))

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Project to simplex
        a_new = simplex_project(a_new)

        # Evaluate
        conv_new = fast_convolve(a_new)
        score = 2.0 * n * conv_new.max()

        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Adaptive schedule
        tau = 20.0 + 0.05 * t
        lr = 0.01 / (1.0 + 0.002 * t)

        if no_improve > 200:
            break

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE {label} it={t} ({elapsed:.0f}s): best={best_score:.6f} tau={tau:.0f} lr={lr:.6f}")

    return best_a, best_score


def peak_perturb(a0, rng, strength=0.05):
    """Perturb the sequence to escape local optima, focusing on peak contributors."""
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    M = conv.max()

    # Find positions near the peak
    threshold = M * 0.99
    peak_positions = np.where(conv >= threshold)[0]

    # For each peak position, perturb contributing coordinates
    for k in peak_positions:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        n_contrib = i_end - i_start + 1
        n_perturb = max(1, n_contrib // 5)
        picks = rng.choice(np.arange(i_start, i_end + 1), size=min(n_perturb, n_contrib), replace=False)
        for i in picks:
            a[i] *= (1.0 + rng.standard_normal() * strength)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a0.copy()
    return a / s


def block_perturb(a0, rng, strength=0.03):
    """Perturb blocks of consecutive coordinates."""
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    M = conv.max()
    threshold = M * 0.95
    peak_positions = np.where(conv >= threshold)[0]

    if len(peak_positions) == 0:
        return a0.copy()

    # Pick a random peak and perturb a block around its contributors
    k = int(peak_positions[rng.integers(len(peak_positions))])
    i_start = max(0, k - n + 1)
    i_end = min(n - 1, k)
    block_start = max(0, i_start - n // 10)
    block_end = min(n - 1, i_end + n // 10)
    a[block_start:block_end + 1] *= (1.0 + rng.standard_normal(block_end - block_start + 1) * strength)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a0.copy()
    return a / s


def init_fourier(n, rng):
    """Initialize using random Fourier superposition."""
    n_harmonics = int(rng.integers(3, 20))
    amplitudes = rng.random(n_harmonics) * 0.4
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = np.arange(1, n_harmonics + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_cosine(n, rng):
    """Initialize with a cosine bell shape."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * rng.uniform(0.3, 0.7)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_u_shape(n, rng):
    """Initialize with U-shape (more mass at edges)."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    alpha = rng.uniform(0.5, 2.0)
    a = 1.0 + alpha * np.cos(np.pi * t) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_alternating(n, rng):
    """Initialize with alternating high-low pattern."""
    period = int(rng.integers(3, 12))
    a = np.ones(n)
    for j in range(n):
        a[j] = 1.0 + 0.5 * np.cos(2 * np.pi * j / period + rng.uniform(0, 2 * np.pi))
    a = np.maximum(a, 0.01)
    s = a.sum()
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def active_set_equalize(a, n, conv, M, max_iter=200):
    """
    Try to equalize the active convex values by finding a direction
    in the intersection of their null spaces that reduces the margin.
    """
    # Find active positions (within 0.1% of max)
    threshold = M * 0.999
    active = np.where(conv >= threshold)[0]

    if len(active) < 2:
        return a

    # Build constraint matrix: for each active k, the gradient of conv_k
    # grad_k[i] = 2*a[k-i] for valid i
    m_active = len(active)
    if m_active >= n:
        return a

    # We want direction d such that sum_i a[k_j - i] * d[i] = 0 for all active k_j
    # And we want to reduce the next-highest conv value
    # Simplified: find d in null space of the active gradients, then
    # project -conv onto that null space

    # For efficiency, work with a subset of active constraints
    max_constraints = min(m_active, min(n - 1, 100))
    active_subset = active[:max_constraints]

    # Build A matrix: A[j, i] = a[active_subset[j] - i] (if valid)
    A = np.zeros((max_constraints, n))
    for j, k in enumerate(active_subset):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            A[j, i] = a[k - i]

    # Find null space basis (simplified: use SVD)
    # Actually for large n this is expensive. Let's use a simpler approach:
    # Random direction, project onto null space iteratively (Gauss-Seidel style)

    d = np.random.randn(n)
    for _ in range(max_constraints * 3):
        for j in range(max_constraints):
            # Project d to be orthogonal to A[j]
            norm_sq = np.dot(A[j], A[j])
            if norm_sq > 1e-20:
                d -= np.dot(A[j], d) / norm_sq * A[j]

    # Also project to be orthogonal to 1 (to preserve sum)
    d -= np.mean(d)
    d = np.maximum(d, 0)  # Only increase coordinates (to maintain non-negativity)
    # Actually we need d to be a valid direction: a + t*d >= 0, sum unchanged
    # Let's use a simpler approach: just try small random perturbations

    return a


def propose_candidate():
    """
    Multi-phase optimization:
    Phase 1: Multi-length diverse init + quick exact CD → find best length
    Phase 2: Intensive exact CD on best length
    Phase 3: LSE gradient optimization for global refinement
    Phase 4: Perturbation + re-optimize cycles
    Phase 5: Ultra-fine CD polish
    Phase 6: Try nearby lengths
    Phase 7: Final polish
    """
    deadline = time.monotonic() + 1790.0
    rng = np.random.default_rng(42)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    # === Phase 1: Multi-length exploration with quick exact CD ===
    print("\n=== Phase 1: Multi-length Exploration ===")
    t1_start = time.monotonic()
    n_values = [512, 640, 768, 896, 1024, 1152, 1280]
    if parent is not None:
        np_ = len(parent)
        if np_ not in n_values:
            n_values.insert(0, np_)

    phase1_candidates = []

    for n in n_values:
        elapsed = time.monotonic() - t1_start
        if elapsed > 300:
            break
        if time.monotonic() > deadline - 800:
            break

        # Multiple initializations
        inits = []
        if parent is not None:
            inits.append(interpolate_seq(parent, n))
        inits.append(init_fourier(n, rng))
        inits.append(init_cosine(n, rng))
        inits.append(init_u_shape(n, rng))
        inits.append(init_alternating(n, rng))

        for init_i, cand in enumerate(inits):
            t_alloc = max(3.0, min(15.0, 300.0 / (len(n_values) * len(inits))))
            a_opt, s_opt, _ = exact_coord_descent(cand, t_alloc, label=f"p1_n{n}_i{init_i}")
            phase1_candidates.append((s_opt, a_opt, n, init_i))

    phase1_candidates.sort(key=lambda x: x[0])
    print(f"  {len(phase1_candidates)} candidates, best: {phase1_candidates[0][0]:.6f}")

    for s_c, a_c, n_c, i_c in phase1_candidates[:8]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_n{n_c}_i{i_c}")

    n_best = len(best_seq) if best_seq is not None else 1024
    print(f"  Best after Phase 1: {best_score:.6f} (n={n_best})")

    # === Phase 2: Intensive exact CD on best ===
    print(f"\n=== Phase 2: Intensive Exact CD (best={best_score:.6f}) ===")
    t2_start = time.monotonic()
    if best_seq is not None:
        current = best_seq.copy()
        for rnd in range(4):
            rem = deadline - time.monotonic() - 500
            if rem < 30:
                break
            t_alloc = min(rem * 0.4, 200.0)
            print(f"  ECD round {rnd + 1}, n={n_best}, budget={t_alloc:.0f}s")
            a_opt, s_opt, sweeps = exact_coord_descent(current, t_alloc,
                                                       verbose=True, label=f"p2r{rnd}")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"p2_r{rnd}")
            # Use the result as starting point for next round
            current = best_seq.copy()
            if best_score < 1.503:
                break

    # === Phase 3: LSE gradient optimization ===
    print(f"\n=== Phase 3: LSE Gradient Opt (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_lse = min(rem * 0.5, 200.0)
            print(f"  LSE n={n_best}, budget={t_lse:.0f}s")
            a_lse, s_lse = lse_optimize(best_seq, t_lse, verbose=True, label="p3")
            best_seq, best_score = check_and_update(a_lse, best_seq, best_score, "lse")

            # Follow up with exact CD
            rem = deadline - time.monotonic() - 300
            if rem > 20:
                a_cd, s_cd, _ = exact_coord_descent(a_lse, min(rem * 0.5, 100.0), label="p3_cd")
                best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "lse_cd")

    # === Phase 4: Perturbation + re-optimize cycles ===
    print(f"\n=== Phase 4: Perturbation Cycles (best={best_score:.6f}) ===")
    if best_seq is not None:
        for cyc in range(6):
            rem = deadline - time.monotonic() - 200
            if rem < 60:
                break

            # Perturb
            strength = 0.02 + 0.01 * cyc
            if cyc % 2 == 0:
                perturbed = peak_perturb(best_seq, rng, strength=strength)
            else:
                perturbed = block_perturb(best_seq, rng, strength=strength)

            # Optimize
            t_alloc = min(rem * 0.3, 120.0)
            a_opt, s_opt, _ = exact_coord_descent(perturbed, t_alloc, label=f"p4_c{cyc}")
            best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"perturb{cyc}")

            if best_score < 1.5028:
                print(f"  Breaking early: {best_score:.6f}")
                break

    # === Phase 5: Ultra-fine CD polish ===
    print(f"\n=== Phase 5: Ultra-fine CD (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_fine = min(rem * 0.6, 200.0)
            print(f"  FineCD n={n_best}, budget={t_fine:.0f}s")
            a_fine, s_fine, _ = fine_coord_descent(best_seq, t_fine, verbose=True, label="p5")
            best_seq, best_score = check_and_update(a_fine, best_seq, best_score, "fine")

    # === Phase 6: Try nearby lengths ===
    print(f"\n=== Phase 6: Nearby Lengths (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 40:
            # Try n_best +/- 128
            for delta_n in [-128, 128, -64, 64]:
                new_n = n_best + delta_n
                if new_n < 256 or new_n > 2560:
                    continue
                rem = deadline - time.monotonic() - 40
                if rem < 20:
                    break
                init_alt = interpolate_seq(best_seq, new_n)
                t_alt = min(rem * 0.4, 60.0)
                a_alt, s_alt, _ = exact_coord_descent(init_alt, t_alt, label=f"alt{new_n}")
                best_seq, best_score = check_and_update(a_alt, best_seq, best_score, f"alt_n{new_n}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        # Exact CD
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            t_fin = min(rem * 0.5, 80.0)
            a_fin, s_fin, _ = exact_coord_descent(best_seq, t_fin, label="final_ecd")
            best_seq, best_score = check_and_update(a_fin, best_seq, best_score, "final_ecd")

        # Fine CD
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_fin2, s_fin2, _ = fine_coord_descent(best_seq, min(rem * 0.6, 40.0), label="final_fine")
            best_seq, best_score = check_and_update(a_fin2, best_seq, best_score, "final_fine")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
