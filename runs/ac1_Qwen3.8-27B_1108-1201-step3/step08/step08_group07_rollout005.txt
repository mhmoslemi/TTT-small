"""
Optimized LSE-ADAM + LP fictitious play for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Key fixes: correct gradient cross-correlation direction, proper FFT sizing,
ADAM optimizer with temperature annealing, and multi-restart strategy.

import numpy as np
import time
import cvxpy as cp
from scipy.linalg import toeplitz
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    """Build (2n-1) x n dense matrix C such that C @ g = f * g (linear conv)."""
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_best_response(f, M, prev_g=None):
    """Solve: max sum(g) s.t. C(f) @ g <= M, g >= 0."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None))
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return prev_g if prev_g is not None else f.copy()
    if g.value is None:
        return prev_g if prev_g is not None else f.copy()
    return np.maximum(g.value, 0)


def softmax_from_z(z):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def lse_objective_and_grad(z, T, n):
    """Compute LSE-approximated objective and its gradient in z-space."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    w = np.exp(shifted - max_shifted)
    w = w / w.sum()
    obj = (2.0 * n / T) * (max_shifted + np.log(w.sum() * np.exp(max_shifted - max_shifted)))
    # obj = (2n/T) * log(sum(exp(conv/T))) = (2n/T) * (max_shifted + log(sum(exp(conv/T - max_shifted))))
    obj = (2.0 * n / T) * (max_shifted + np.log((w * np.exp(max_shifted)).sum()))
    # Simpler: obj = (2n/T) * (max_shifted + log(sum(exp(shifted - max_shifted))))
    obj = (2.0 * n / T) * (max_shifted + np.log(np.exp(shifted - max_shifted).sum()))
    
    # Gradient: need S_j = sum_k w_k * a_{k-j} (linear cross-correlation)
    # = IFFT(FFT(w) * conj(FFT(a)))[:n] with proper padding
    size_w = 2 * n - 1
    size_a = n
    # Need fft_size >= size_w + size_a - 1 = 3n - 2
    corr_fft_size = 1 << (size_w + size_a - 2).bit_length()
    W = rfft(w, corr_fft_size)
    A = rfft(a, corr_fft_size)
    # IFFT(FFT(w) * conj(FFT(a))) gives sum_n w_n * a_{n-j} = our S_j
    grad_a = 2.0 * irfft(W * np.conj(A), corr_fft_size)[:n]
    
    S = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - S)
    grad_z *= (2.0 * n / T)
    return obj, grad_z


def adam_lse_optimize(a0, time_limit, T_start=50.0, T_end=5000.0,
                      lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8,
                      verbose=True, label="ADAM-LSE"):
    """ADAM optimization of LSE-approximated objective in log-space with temperature annealing."""
    n = len(a0)
    z0 = np.log(np.maximum(a0, 1e-30))
    z = z0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)
    
    total_iterations_estimate = 5000
    iters_logged = 0
    
    while time.monotonic() - t0 < time_limit:
        # Anneal temperature
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress
        T = max(T, T_start)
        
        t_count += 1
        obj, grad_z = lse_objective_and_grad(z, T, n)
        
        # ADAM update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * grad_z ** 2
        m_hat = m / (1 - beta1 ** t_count)
        v_hat = v / (1 - beta2 ** t_count)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        a = softmax_from_z(z)
        score = compute_score(a)
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
        
        iters_logged += 1
        if verbose and iters_logged % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters_logged}, T={T:.1f}, best={best_score:.8f}, "
                  f"time={elapsed:.0f}s/{time_limit:.0f}s")
    
    return best_a, best_score


def lp_fictitious_play(a0, time_limit, step=0.4, verbose=True, label="LP-FP"):
    """LP-based fictitious play with warm-starting."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    prev_g = None
    stall_count = 0
    stall_threshold = 200

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()

        g = solve_best_response(a, M, prev_g=prev_g)
        prev_g = g
        g_sum = g.sum()

        if g_sum < 1e-10:
            break

        g_norm = g / g_sum
        a_new = (1 - step) * a + step * g_norm
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss

        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1

        a = a_new

        if stall_count > stall_threshold:
            step = max(0.05, step * 0.7)
            rng = np.random.default_rng(iters)
            a = best_a + rng.standard_normal(n) * 0.001
            a = np.maximum(a, 0)
            a = a / a.sum()
            prev_g = None
            stall_count = 0

        if verbose and iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, step={step:.4f}, "
                  f"time={elapsed:.0f}s/{time_limit:.0f}s")

    return best_a, best_score


def coordinate_descent(a0, time_limit, n_pairs=300, verbose=True):
    """Coordinate descent with pairwise transfers and restarts."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(99999)
    iters = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or a[i] < 1e-14:
                continue

            best_local = best_score
            best_t = 0.0
            base = min(a[i], a[j]) if a[j] > 1e-15 else a[i]

            for frac in [0.05, 0.1, 0.2, 0.3, 0.5, 0.8]:
                t = frac * base
                if t >= a[i] - 1e-15:
                    t = a[i] * 0.5
                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_t = t

            if best_t != 0.0:
                a[i] -= best_t
                a[j] += best_t
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = best_local
                best_a = a.copy()
                improved = True
                no_improve = 0

        if not improved:
            no_improve += 1
            if no_improve > 15:
                noise = rng.standard_normal(n) * 0.003
                a = best_a.copy() + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0

        if verbose and iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"  CD iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")

    return best_a, best_score


def check_and_update(a, label, best_seq, best_score, verbose=True):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def subgradient_projected_descent(a0, time_limit, verbose=True, label="SGD"):
    """Subgradient method on the max of conv, projected to simplex."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    lr0 = 0.1
    exponent = 0.6
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        lr = lr0 / (1 + 0.05 * iters) ** exponent
        
        conv = fast_convolve(a)
        M = conv.max()
        # Find argmax indices
        max_idx = np.argmax(conv)
        
        # Subgradient of max(conv(a,a)) at a:
        # grad_j = 2 * a_{max_idx - j} (if 0 <= max_idx - j < n)
        j = np.arange(n)
        k_minus_j = max_idx - j
        valid = (k_minus_j >= 0) & (k_minus_j < n)
        grad = np.zeros(n)
        grad[valid] = 2.0 * a[k_minus_j[valid]]
        
        # Average over all argmax (if multiple)
        max_val = conv.max()
        max_mask = conv >= max_val - 1e-10
        if max_mask.sum() > 1:
            grad = np.zeros(n)
            for mk in np.where(max_mask)[0]:
                kj = mk - j
                v = (kj >= 0) & (kj < n)
                grad[v] += 2.0 * a[kj[v]]
            grad /= max_mask.sum()
        
        # Update
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        
        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        
        if verbose and iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, lr={lr:.6f}, time={elapsed:.0f}s")
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining:
    1. ADAM on corrected LSE surrogate (temperature annealing)
    2. LP fictitious play
    3. Subgradient projected descent
    4. Coordinate descent
    5. Multiple restarts and length variations
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(seed)
    
    best_seq = None
    best_score = float('inf')
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Initialized uniform: n={n_init}, score={best_score:.8f}")
    
    n_parent = len(parent)
    
    # Phase 1: ADAM-LSE with temperature annealing (main driver)
    print(f"\n=== Phase 1: ADAM-LSE Annealing (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.25, 400.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s, n={n_parent}")
        a_adam, sc_adam = adam_lse_optimize(best_seq, t1, T_start=30.0, T_end=3000.0,
                                            lr=0.02, verbose=True, label="ADAM1")
        best_seq, best_score = check_and_update(a_adam, "adam1", best_seq, best_score)
    
    # Phase 2: LP Fictitious Play (medium step)
    print(f"\n=== Phase 2: LP-FP (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.20, 300.0)
    if rem > 60:
        print(f"  Budget: {t2:.0f}s")
        a_lp, sc_lp = lp_fictitious_play(best_seq, t2, step=0.35, label="LP-mid")
        best_seq, best_score = check_and_update(a_lp, "lp_mid", best_seq, best_score)
    
    # Phase 3: ADAM-LSE again from current best (higher T for exploration)
    print(f"\n=== Phase 3: ADAM-LSE High-T (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.15, 200.0)
    if rem > 40:
        print(f"  Budget: {t3:.0f}s")
        a_adam2, sc_adam2 = adam_lse_optimize(best_seq, t3, T_start=100.0, T_end=10000.0,
                                              lr=0.005, verbose=True, label="ADAM2")
        best_seq, best_score = check_and_update(a_adam2, "adam2", best_seq, best_score)
    
    # Phase 4: Subgradient projected descent
    print(f"\n=== Phase 4: Subgradient (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.10, 120.0)
    if rem > 30:
        print(f"  Budget: {t4:.0f}s")
        a_sg, sc_sg = subgradient_projected_descent(best_seq, t4, verbose=True, label="SGD")
        best_seq, best_score = check_and_update(a_sg, "subgrad", best_seq, best_score)
    
    # Phase 5: LP with small step (polish)
    print(f"\n=== Phase 5: LP Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.15, 200.0)
    if rem > 40:
        print(f"  Budget: {t5:.0f}s")
        a_lp2, sc_lp2 = lp_fictitious_play(best_seq, t5, step=0.15, label="LP-polish")
        best_seq, best_score = check_and_update(a_lp2, "lp_polish", best_seq, best_score)
    
    # Phase 6: Coordinate descent
    print(f"\n=== Phase 6: Coordinate Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.10, 100.0)
    if rem > 20:
        print(f"  Budget: {t6:.0f}s")
        a_cd, sc_cd = coordinate_descent(best_seq, t6, n_pairs=400, verbose=True)
        best_seq, best_score = check_and_update(a_cd, "coord_desc", best_seq, best_score)
    
    # Phase 7: Restart from different initialization
    print(f"\n=== Phase 7: Restart (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        # Create perturbed version
        noise = rng.standard_normal(n_parent) * 0.01
        a_restart = best_seq + noise
        a_restart = np.maximum(a_restart, 0)
        a_restart = a_restart / a_restart.sum()
        
        t7 = min(rem * 0.15, 150.0)
        print(f"  Restart budget: {t7:.0f}s")
        a_r, sc_r = adam_lse_optimize(a_restart, t7, T_start=50.0, T_end=5000.0,
                                       lr=0.01, verbose=False, label="ADAM-R")
        best_seq, best_score = check_and_update(a_r, "restart", best_seq, best_score)
        
        rem = deadline - time.monotonic()
        if rem > 30:
            t7b = min(rem * 0.5, 100.0)
            a_lp3, sc_lp3 = lp_fictitious_play(best_seq, t7b, step=0.2, label="LP-R")
            best_seq, best_score = check_and_update(a_lp3, "lp_restart", best_seq, best_score)
    
    # Phase 8: Try nearby lengths
    print(f"\n=== Phase 8: Length Exploration (best={best_score:.8f}) ===")
    for alt_n in [n_parent + 2, n_parent - 2, n_parent + 4, n_parent + 6, n_parent - 4]:
        if alt_n < 128 or alt_n > 4096:
            continue
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t8 = min(rem * 0.20, 80.0)
        init_alt = interpolate_sequence(best_seq, alt_n)
        print(f"  ADAM n={alt_n}, budget={t8:.0f}s")
        a_alt, sc_alt = adam_lse_optimize(init_alt, t8, T_start=50.0, T_end=5000.0,
                                           lr=0.01, verbose=False, label=f"ADAM-n{alt_n}")
        best_seq, best_score = check_and_update(a_alt, f"n{alt_n}", best_seq, best_score)
        
        rem = deadline - time.monotonic()
        if rem > 20:
            t8b = min(rem * 0.3, 40.0)
            a_alt_lp, sc_alt_lp = lp_fictitious_play(best_seq, t8b, step=0.2, label=f"LP-n{alt_n}")
            best_seq, best_score = check_and_update(a_alt_lp, f"lp_n{alt_n}", best_seq, best_score)
    
    # Phase 9: Final ADAM polish with very high T (nearly exact max)
    print(f"\n=== Phase 9: Final ADAM High-T (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t9 = min(rem * 0.3, 80.0)
        a_final, sc_final = adam_lse_optimize(best_seq, t9, T_start=2000.0, T_end=20000.0,
                                              lr=0.002, verbose=False, label="ADAM-FINAL")
        best_seq, best_score = check_and_update(a_final, "adam_final", best_seq, best_score)
    
    # Phase 10: Final LP
    print(f"\n=== Phase 10: Final LP (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t10 = min(rem * 0.5, 60.0)
        a_fin, sc_fin = lp_fictitious_play(best_seq, t10, step=0.1, label="LP-END")
        best_seq, best_score = check_and_update(a_fin, "lp_end", best_seq, best_score)
    
    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
