"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast Smoothed Multiplicative Weight Updates (MWU) with adaptive temperature
2. LP-based direction finding for periodic "jumps"
3. Coordinate descent with golden-section search for final polish
4. Multi-length exploration
The key idea: use O(n log n) MWU iterations (much faster than LP) with
softmax-smoothed gradients to find low-peak autocorrelation sequences.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the optimization score for a normalized sequence."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return 1e10
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def compute_norm_score(a):
    """Score assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def smoothed_gradient(a, T):
    """Compute smoothed gradient of max_k (a*a)[k] using softmax smoothing.
    Returns (grad, conv, w) where grad[j] = 2 * sum_k w[k] * a[k-j] (valid indices)."""
    n = len(a)
    conv = fast_convolve(a)
    max_c = conv.max()
    exp_vals = np.exp((conv - max_c) / T)
    w = exp_vals / exp_vals.sum()

    # grad[j] = 2 * sum_{l=0}^{n-1} a[l] * w[j+l]
    # This is the cross-correlation of a and w at lag j
    # Using FFT: IFFT(conj(FFT(a)) * FFT(w)) gives cross-correlation
    total_len = n + (2 * n - 1) - 1  # n + 2n-2 = 3n-2
    N = 1 << (total_len - 1).bit_length()
    A = rfft(a, N)
    W = rfft(w, N)
    cross = irfft(np.conj(A) * W, N)
    grad = 2.0 * cross[:n]
    return grad, conv, w


def build_C(f):
    """Build constraint matrix C where C[k,j] = f[k-j] if 0<=k-j<n else 0."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = f
    return C


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. C*g <= M*1, g >= 0."""
    n = len(f)
    C = build_C(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=C, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if res.status != 0:
        return None
    return np.maximum(res.x, 0)


def mwu_optimize(a_init, time_limit, T0=None, eta_scale=1.0, use_lp_every=0,
                 n_restarts=3, verbose=True, seed=42):
    """
    Smoothed MWU optimization with adaptive temperature.
    a_init: initial sequence (will be normalized)
    time_limit: seconds to spend
    T0: initial temperature (default: auto)
    eta_scale: learning rate scale
    use_lp_every: if >0, solve LP every this many iterations
    n_restarts: number of restarts
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0)
    a = a / a.sum()

    if T0 is None:
        conv0 = fast_convolve(a)
        T0 = max(conv0.max() / 5.0, 1.0 / n * 0.1)

    best_a = a.copy()
    best_score = compute_norm_score(a)
    a_avg = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    tau = max(n, 100)  # cooling timescale
    total_iters = 0

    for restart in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break

        if restart > 0:
            # Perturb best solution for restart
            noise_scale = 0.01 / (restart + 1)
            a = best_a + rng.standard_normal(n) * noise_scale
            a = np.maximum(a, 0)
            a = a / a.sum()
            a_avg = a.copy()
            # Reset temperature higher for restart
            T0 = T0 * 2.0

        # MWU iteration loop
        for it in range(100000000):
            if time.monotonic() - t0 > time_limit:
                break

            # Adaptive temperature: cool down over iterations
            T = T0 / (1.0 + it / float(tau))
            T = max(T, 1e-6 / n)

            # Compute gradient
            grad, conv, w = smoothed_gradient(a, T)

            # Adaptive learning rate
            max_grad = np.abs(grad).max()
            if max_grad < 1e-15:
                eta = 0.0
            else:
                eta = eta_scale / (max_grad + 1e-15)
                eta = min(eta, 50.0 / n)  # cap to prevent instability

            if eta < 1e-10:
                break

            # MWU update: a_j <- a_j * exp(-eta * grad_j) / Z
            log_update = -eta * grad
            log_update -= log_update.max()  # numerical stability
            update = np.exp(log_update)
            a_new = a * update
            s_new = a_new.sum()
            if s_new < 1e-30:
                a_new = a.copy()
            else:
                a_new = a_new / s_new

            # Update average
            total_iters += 1
            a_avg = (a_avg * (total_iters - 1) + a_new) / total_iters

            # Check scores
            sc_cur = compute_norm_score(a_new)
            if sc_cur < best_score:
                best_score = sc_cur
                best_a = a_new.copy()

            sc_avg = compute_norm_score(a_avg)
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = a_avg.copy()

            a = a_new

            # Periodic LP jump
            if use_lp_every > 0 and it % use_lp_every == 0 and it > 0:
                M = conv.max()
                g = solve_lp(a, M)
                if g is not None and g.sum() > 1e-15:
                    g = g / g.sum()
                    # Blend
                    t_blend = 0.3
                    a = (1 - t_blend) * a + t_blend * g
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    sc_lp = compute_norm_score(a)
                    if sc_lp < best_score:
                        best_score = sc_lp
                        best_a = a.copy()

            if verbose and it % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"    MWU r={restart} it={it} T={T:.4e} score={sc_cur:.8f} "
                      f"avg={sc_avg:.8f} best={best_score:.8f} t={elapsed:.0f}s")

    return best_a, best_score


def coordinate_descent(a_init, time_limit, verbose=True, seed=42):
    """Coordinate descent with golden-section search for 2-swap optimization."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_norm_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    gr = (np.sqrt(5) - 1) / 2
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        conv = fast_convolve(a)
        peak_idx = int(np.argmax(conv))

        # Generate candidate pairs
        n_pairs = min(200, n)
        pairs = set()

        # Peak-focused pairs
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        for _ in range(n_pairs // 2):
            i = int(rng.integers(js, je + 1))
            j = int(rng.integers(js, je + 1))
            if i != j:
                pairs.add((min(i, j), max(i, j)))

        # Random pairs
        for _ in range(n_pairs // 2):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                pairs.add((min(i, j), max(i, j)))

        for (i, j) in list(pairs):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14 and a[j] < 1e-14:
                continue

            t_lo = -a[j] + 1e-15
            t_hi = a[i] - 1e-15
            if t_hi <= t_lo:
                continue

            x1 = t_hi - gr * (t_hi - t_lo)
            x2 = t_lo + gr * (t_hi - t_lo)

            # Compute scores at x1, x2
            a1 = a.copy()
            a1[i] -= x1
            a1[j] += x1
            a1 = np.maximum(a1, 0)
            a1 = a1 / a1.sum()
            f1 = compute_norm_score(a1)

            a2 = a.copy()
            a2[i] -= x2
            a2[j] += x2
            a2 = np.maximum(a2, 0)
            a2 = a2 / a2.sum()
            f2 = compute_norm_score(a2)

            for _ in range(25):
                if time.monotonic() - t0 > time_limit:
                    break
                if f1 < f2:
                    t_hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = t_hi - gr * (t_hi - t_lo)
                    a1 = a.copy()
                    a1[i] -= x1
                    a1[j] += x1
                    a1 = np.maximum(a1, 0)
                    a1 = a1 / a1.sum()
                    f1 = compute_norm_score(a1)
                else:
                    t_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = t_lo + gr * (t_hi - t_lo)
                    a2 = a.copy()
                    a2[i] -= x2
                    a2[j] += x2
                    a2 = np.maximum(a2, 0)
                    a2 = a2 / a2.sum()
                    f2 = compute_norm_score(a2)

            t_opt = (t_lo + t_hi) / 2
            f_opt = min(f1, f2)

            if f_opt < best_score - 1e-13:
                a[i] -= t_opt
                a[j] += t_opt
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = compute_norm_score(a)
                best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 10:
                noise = rng.standard_normal(n) * 0.003
                a = best_a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
        else:
            no_improve = 0

        if verbose and (total_iters if 'total_iters' in dir() else it) % 50 == 0:
            pass

    return best_a, best_score


def sobol_init(n, seed=0):
    """Generate a quasi-random initial sequence using sum of cosines."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 2 * np.pi, n)
    a = np.ones(n)
    for k in range(1, 8):
        freq = rng.uniform(0.5, 5.0) * k
        phase = rng.uniform(0, 2 * np.pi)
        amp = 1.0 / k
        a = a + amp * np.cos(freq * x + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def cosine_taper_init(n, seed=0):
    """Initialize with a cosine-tapered window."""
    x = np.linspace(0, 1, n)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def interpolable_init(n, freq, seed=0):
    """Initialize with a structured pattern."""
    x = np.linspace(0, 1, n)
    a = 1.0 + 0.3 * np.cos(2 * np.pi * freq * x) + 0.1 * np.cos(2 * np.pi * freq * 3 * x)
    a = np.maximum(a, 0)
    s = a.sum()
    return a / s if s > 1e-15 else np.ones(n) / n

        p = np.asarray(height_sequence_1, dtype=np.float64)

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            n_parent = len(parent)
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={n_parent}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        n_parent = 768
        parent = sobol_init(n_parent)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Initialized: n={n_parent}, score={best_score:.8f}")

    print(f"Starting optimization with n={n_parent}, best={best_score:.8f}")

    # === Phase 1: MWU from parent (long run) ===
    rem = total_deadline - time.monotonic()
    t1 = rem * 0.45
    print(f"\n=== Phase 1: MWU from parent (budget={t1:.0f}s) ===")
    a1, sc1 = mwu_optimize(best_seq, t1, T0=None, eta_scale=1.0,
                           use_lp_every=200, n_restarts=2, verbose=True, seed=seed)
    best_seq, best_score = check_and_update(a1, best_seq, best_score, "mwu_parent")

    # === Phase 2: MWU with different temperature ===
    rem = total_deadline - time.monotonic()
    if rem > 60:
        t2 = rem * 0.2
        print(f"\n=== Phase 2: MWU higher T (budget={t2:.0f}s) ===")
        conv_init = fast_convolve(best_seq)
        T_high = conv_init.max() * 2.0
        a2, sc2 = mwu_optimize(best_seq, t2, T0=T_high, eta_scale=0.5,
                               use_lp_every=0, n_restarts=2, verbose=True, seed=seed + 1)
        best_seq, best_score = check_and_update(a2, best_seq, best_score, "mwu_highT")

    # === Phase 3: LP-heavy optimization ===
    rem = total_deadline - time.monotonic()
    if rem > 60:
        t3 = rem * 0.15
        print(f"\n=== Phase 3: LP-heavy MWU (budget={t3:.0f}s) ===")
        a3, sc3 = mwu_optimize(best_seq, t3, T0=None, eta_scale=1.0,
                               use_lp_every=50, n_restarts=1, verbose=True, seed=seed + 2)
        best_seq, best_score = check_and_update(a3, best_seq, best_score, "mwu_lp")

    # === Phase 4: Coordinate descent polish ===
    rem = total_deadline - time.monotonic()
    if rem > 40:
        t4 = rem * 0.25
        print(f"\n=== Phase 4: Coordinate Descent (budget={t4:.0f}s) ===")
        a4, sc4 = coordinate_descent(best_seq, t4, verbose=True, seed=seed + 3)
        best_seq, best_score = check_and_update(a4, best_seq, best_score, "cd")

    # === Phase 5: Multi-init MWU at same length ===
    rem = total_deadline - time.monotonic()
    if rem > 60:
        t5 = rem * 0.15
        print(f"\n=== Phase 5: Multi-init MWU (budget={t5:.0f}s) ===")
        inits = [
            ("sobol", sobol_init(n_parent, seed=100)),
            ("cosine", cosine_taper_init(n_parent)),
            ("freq2", interpolable_init(n_parent, 2)),
            ("freq5", interpolable_init(n_parent, 5)),
        ]
        t_per_init = t5 / len(inits)
        for name, init in inits:
            if time.monotonic() > total_deadline - 30:
                break
            print(f"  Init: {name}")
            a5, sc5 = mwu_optimize(init, t_per_init, T0=None, eta_scale=1.0,
                                   use_lp_every=100, n_restarts=1, verbose=False,
                                   seed=seed + 10 + hash(name) % 100)
            best_seq, best_score = check_and_update(a5, best_seq, best_score, f"init_{name}")

    # === Phase 6: Length exploration ===
    rem = total_deadline - time.monotonic()
    if rem > 60:
        t6 = rem * 0.2
        print(f"\n=== Phase 6: Length Exploration (budget={t6:.0f}s) ===")
        # Try nearby and some non-nearby lengths
        candidate_ns = []
        for dn in [-4, -2, -1, 1, 2, 4, 8, 16, -8, -16]:
            alt_n = n_parent + dn
            if 128 <= alt_n <= 4096:
                candidate_ns.append(alt_n)
        # Also try some structured lengths
        for pn in [512, 1024, 1536, 2048]:
            if pn != n_parent:
                candidate_ns.append(pn)

        # Deduplicate and sort by closeness to n_parent
        seen = set()
        unique_ns = []
        for cn in sorted(candidate_ns, key=lambda x: abs(x - n_parent)):
            if cn not in seen:
                seen.add(cn)
                unique_ns.append(cn)

        t_per_len = t6 / min(len(unique_ns), 6)
        for alt_n in unique_ns[:6]:
            if time.monotonic() > total_deadline - 30:
                break
            print(f"  Trying n={alt_n}")
            # Interpolate best to new length
            old_n = len(best_seq)
            old_x = np.linspace(0, 1, old_n)
            new_x = np.linspace(0, 1, alt_n)
            init_alt = np.interp(new_x, old_x, best_seq)
            init_alt = np.maximum(init_alt, 0)
            init_alt = init_alt / init_alt.sum()

            a6, sc6 = mwu_optimize(init_alt, t_per_len * 0.7, T0=None,
                                   eta_scale=1.0, use_lp_every=0,
                                   n_restarts=1, verbose=False, seed=seed + alt_n)
            # Quick CD polish
            rem2 = total_deadline - time.monotonic() - 20
            if rem2 > 10:
                a6b, sc6b = coordinate_descent(a6, min(rem2 * 0.3, t_per_len * 0.3),
                                               verbose=False, seed=seed + alt_n + 1)
                a6 = a6b if sc6b < sc6 else a6
            best_seq, best_score = check_and_update(a6, best_seq, best_score, f"n{alt_n}")

    # === Phase 7: Final coordinate descent polish ===
    rem = total_deadline - time.monotonic()
    if rem > 20:
        t7 = min(rem * 0.5, 60.0)
        print(f"\n=== Phase 7: Final CD Polish (budget={t7:.0f}s) ===")
        a7, sc7 = coordinate_descent(best_seq, t7, verbose=False, seed=seed + 99)
        best_seq, best_score = check_and_update(a7, best_seq, best_score, "final_cd")

    # === Phase 8: Final MWU with very low temperature ===
    rem = total_deadline - time.monotonic()
    if rem > 15:
        t8 = min(rem * 0.7, 30.0)
        print(f"\n=== Phase 8: Final Low-T MWU (budget={t8:.0f}s) ===")
        conv_final = fast_convolve(best_seq)
        T_low = conv_final.max() * 0.2
        a8, sc8 = mwu_optimize(best_seq, t8, T0=T_low, eta_scale=0.3,
                               use_lp_every=0, n_restarts=1, verbose=False, seed=seed + 200)
        best_seq, best_score = check_and_update(a8, best_seq, best_score, "final_lowT")

    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
