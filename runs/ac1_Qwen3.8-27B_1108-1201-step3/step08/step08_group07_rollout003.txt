def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)

"""
ADAM-based LSE-smoothed optimization with temperature annealing for minimizing
2n * max(autoconv(a)) / (sum(a))^2.
Uses softmax parameterization, adaptive learning rates, and periodic LP polish.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.linalg import toeplitz


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def softmax(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def lse_objective_and_grad_z(z, T, n):
    """
    Compute LSE-smoothed score and its gradient w.r.t. z (log-softmax params).
    score_T = (2n/T) * log(sum_k exp(T * conv[k]))
    """
    a = softmax(z)
    conv = fast_convolve(a)
    
    # Numerically stable LSE
    max_conv = conv.max()
    shifted = T * (conv - max_conv)
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    lse_log = max_conv + (1.0 / T) * np.log(exp_shifted.sum())
    score_val = 2.0 * n * lse_log
    
    # Weights
    w = exp_shifted / exp_shifted.sum()
    
    # Gradient of LSE score w.r.t. a:
    # d score_T / d a[i] = 2n * sum_k w[k] * d conv[k]/d a[i]
    # d conv[k]/d a[i] = 2*a[k-i] if 0 <= k-i < n, else 0
    # So: grad_a[i] = 4n * sum_{k=i}^{i+n-1} w[k] * a[k-i] = 4n * cross_corr(w,a)[i]
    
    fft_size = 1 << (2 * n - 2).bit_length()
    W = rfft(w, fft_size)
    A_ft = rfft(a, fft_size)
    # Linear cross-correlation: (w cross a)[i] = sum_j w[j+i]*a[j]
    # = IFFT(conj(FFT(w)) * FFT(a)) with zero-padding
    cross_corr = irfft(np.conj(W) * A_ft, fft_size)[:n]
    grad_a = 4.0 * n * cross_corr
    
    # Chain rule through softmax:
    # d score / d z[j] = a[j] * (grad_a[j] - dot(grad_a, a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return score_val, grad_z


def adam_lse_optimize(a0, time_limit, T_start=50.0, T_end=2000.0,
                      lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8,
                      verbose=True, label="ADAM-LSE"):
    """ADAM optimizer on LSE-smoothed objective with temperature annealing."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    # Adaptive: reduce learning rate if score doesn't improve
    patience = 0
    max_patience = 300
    lr_scale = 1.0
    
    while time.monotonic() - t0 < time_limit:
        t_count += 1
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        
        # Temperature schedule: exponential annealing
        T = T_start + (T_end - T_start) * elapsed_frac**0.7
        
        score_val, grad_z = lse_objective_and_grad_z(z, T, n)
        
        # ADAM update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * grad_z ** 2
        m_hat = m / (1.0 - beta1 ** t_count)
        v_hat = v / (1.0 - beta2 ** t_count)
        
        z = z - lr * lr_scale * m_hat / (np.sqrt(v_hat) + eps)
        
        # Track best
        a_cur = softmax(z)
        exact_score = compute_score(a_cur)
        if exact_score < best_score - 1e-10:
            best_score = exact_score
            best_a = a_cur.copy()
            patience = 0
        else:
            patience += 1
        
        # Adaptive learning rate
        if patience > max_patience:
            lr_scale *= 0.7
            patience = 0
            # Small random perturbation
            z = z + np.random.default_rng(t_count).standard_normal(n) * 0.01
            lr_scale = max(lr_scale, 0.01)
        
        if verbose and t_count % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} it={t_count}, best={best_score:.8f}, T={T:.1f}, "
                  f"lr={lr*lr_scale:.6f}, time={elapsed:.0f}s/{time_limit:.0f}s")
    
    return best_a, best_score


def build_conv_matrix(f):
    """Build (2n-1) x n matrix C such that C @ g = linear_conv(f, g)."""
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def lp_polish(a0, time_limit, verbose=True, label="LP"):
    """LP-based fictitious play polish."""
    try:
        import cvxpy as cp
    except ImportError:
        return a0.copy(), compute_score(a0)
    
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    prev_g = None
    step = 0.3
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()
        
        g = cp.Variable(n, nonneg=True)
        if prev_g is not None:
            g.value = prev_g
        C = build_conv_matrix(a)
        constraints = [C @ g <= M]
        obj = cp.Maximize(cp.sum(g))
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None))
        except Exception:
            try:
                prob.solve(solver=cp.ECOS)
            except Exception:
                try:
                    prob.solve(solver=cp.SCS)
                except Exception:
                    break
        if g.value is None:
            break
        g_val = np.maximum(g.value, 0)
        g_sum = g_val.sum()
        if g_sum < 1e-10:
            break
        
        g_norm = g_val / g_sum
        a_new = (1 - step) * a + step * g_norm
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        
        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        prev_g = g_val
        
        if verbose and iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")
    
    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, n_pairs=500, verbose=True, label="CD"):
    """Coordinate descent with pairwise mass transfers."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or a[i] < 1e-14:
                continue
            
            best_local = compute_score(a)
            best_t = 0.0
            base = min(a[i], a[j] + 0.1) * a[i]
            
            for frac in [0.1, 0.3, 0.5]:
                t = frac * base
                if t >= a[i] * 0.9:
                    t = a[i] * 0.5
                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_t = t
            
            if best_t != 0.0:
                a[i] -= best_t
                a[j] += best_t
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = best_local
                best_a = a.copy()
                improved = True
        
        if not improved and iters > 50:
            noise = rng.standard_normal(n) * 0.003
            a = best_a.copy() + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
        
        if verbose and iters % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")
    
    return best_a, best_score


defibbon_interpolate(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
