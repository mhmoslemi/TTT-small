import numpy as np
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    """Cross-convolution f*g via FFT."""
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v, target=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target}."""
    n = len(v)
    if n == 0:
        return np.array([])
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    # Scale to unit simplex projection, then scale back
    v_unit = v / max(v.sum(), 1e-30)
    u = np.sort(v_unit)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.full(n, target / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    proj_unit = np.maximum(v_unit - theta, 0.0)
    return proj_unit * target


def compute_gradient_at_lag(a, k):
    """Compute gradient of (a*a)[k] with respect to a. Result is 2*a[k-i] with bounds."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    for i in range(lo, hi + 1):
        j = k - i
        if j == i:
            grad[i] += 2.0 * a[i]
        else:
            grad[i] += 2.0 * a[j]
    return grad


def compute_gradient_vectorized(a, k):
    """Vectorized gradient computation."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    indices = np.arange(lo, hi + 1)
    partners = k - indices
    # For i where i != k-i (i.e., 2*i != k), grad[i] = 2*a[k-i]
    # For i where 2*i == k, grad[i] = 2*a[i] (same formula works)
    grad[indices] = 2.0 * a[partners]
    return grad


def weighted_gradient(a, k_indices, weights):
    """Compute weighted combination of gradients at multiple lags."""
    n = len(a)
    grad = np.zeros(n)
    for idx, w in zip(k_indices, weights):
        if w < 1e-15:
            continue
        lo = max(0, idx - n + 1)
        hi = min(n - 1, idx)
        indices = np.arange(lo, hi + 1)
        partners = idx - indices
        grad[indices] += w * 2.0 * a[partners]
    return grad


def subgradient_nesterov(a_init, time_limit, top_k=10, momentum=0.85, label="sg"):
    """
    Nesterov-accelerated subgradient descent with temperature-annealed lag selection.
    Minimizes max_k (a*a)[k] subject to a >= 0, sum(a) = 1.
    """
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    a_prev = a.copy()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    step = 0.005
    temp = 0.01
    no_improve = 0
    it = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        # Batch of iterations
        batch_end = min(time_limit * 0.02, 5.0)
        t_batch_start = time.monotonic()
        while time.monotonic() - t_batch_start < batch_end:
            it += 1
            b = fast_convolve(a)
            max_b = b.max()

            if max_b < best_obj - 1e-15:
                best_obj = max_b
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            # Select lags: top-k by value, weighted by softmax
            if top_k < len(b):
                top_idx = np.argpartition(b, -top_k)[-top_k:]
            else:
                top_idx = np.arange(len(b))

            # Softmax weights for selected lags
            scaled = b[top_idx] / max(temp, 1e-10)
            scaled -= scaled.max()
            w = np.exp(scaled)
            w_sum = w.sum()
            if w_sum < 1e-30:
                w = np.ones(len(top_idx)) / len(top_idx)
            else:
                w = w / w_sum

            # Compute weighted gradient
            grad = weighted_gradient(a, top_idx, w)

            # Nesterov momentum update
            a_new = a + momentum * (a - a_prev)
            a_new = a_new - step * grad
            a_prev = a.copy()
            a = project_simplex(a_new)

            # Adaptive step size
            new_obj = fast_convolve(a).max()
            if new_obj > max_b * 1.05:
                step *= 0.8
            elif new_obj < max_b * 0.99:
                step = min(step * 1.02, 0.02)

            if it % 200 == 0:
                temp = min(temp * 1.15, 0.5)

        # Periodic check and restart
        if no_improve > 50:
            # Restart with perturbation
            noise = rng.standard_normal(n) * 0.01
            a = project_simplex(best_a + noise)
            a_prev = a.copy()
            step = 0.005
            no_improve = 0
            temp = 0.01
            if it % 500 == 0:
                print(f"    {label} iter={it} obj={best_obj:.12f} "
                      f"score={2.0 * n * best_obj:.8f} restart t={time.monotonic()-t0:.0f}s")

        if it % 500 == 0:
            print(f"    {label} iter={it} obj={best_obj:.12f} "
                  f"score={2.0 * n * best_obj:.8f} step={step:.5f} t={time.monotonic()-t0:.0f}s")

    return best_a, best_obj


def lp_optimize(f_init, time_limit, max_constraints=500, label="lp"):
    """
    LP-based optimization with constraint generation.
    Maximizes sum(g) s.t. g>=0, (f*g)[k] <= M for active constraints.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
        s = 1.0
    f = f / s

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        # Select active constraints: top peaks
        n_constraints = min(max_constraints, len(b))
        # Grow constraints over time
        n_constraints = min(n_constraints, 30 + 10 * (it // 5))

        if n_constraints < len(b):
            top_idx = np.argpartition(b, -n_constraints)[-n_constraints:]
        else:
            top_idx = np.arange(len(b))

        # Build constraint matrix
        A = np.zeros((len(top_idx), n))
        for row_i, k in enumerate(top_idx):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A[row_i, lo:hi + 1] = f[lo:hi + 1]

        # Solve LP
        g_var = cp.Variable(n)
        prob = cp.Problem(cp.Maximize(cp.sum(g_var)),
                          [g_var >= 0, A @ g_var <= M])

        solved = False
        for solver in [cp.ECOS, cp.SCS, cp.PDLP]:
            try:
                prob.solve(solver=solver, max_iters=500000)
                if g_var.value is not None:
                    solved = True
                    break
            except Exception:
                continue

        if not solved or g_var.value is None:
            stuck += 1
            if stuck > 15:
                break
            continue

        g = np.maximum(g_var.value, 0.0)
        if np.any(np.isnan(g)):
            g = np.zeros(n) / n

        S_g = g.sum()
        if S_g <= 1.0 + 1e-4:
            stuck += 1
            if stuck > 20:
                break
            continue

        # Line search: find best t in (0, 1]
        h = g / S_g
        best_t = 0.05
        best_trial = fast_convolve((1 - 0.05) * f + 0.05 * h).max()

        for t in [0.02, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            f_trial = (1 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            st = f_trial.sum()
            if st < 1e-10:
                continue
            f_trial = f_trial / st
            obj = fast_convolve(f_trial).max()
            if obj < best_trial:
                best_trial = obj
                best_t = t

        if best_trial < best_obj - 1e-14:
            f = (1 - best_t) * f + best_t * h
            f = np.maximum(f, 0.0)
            st = f.sum()
            if st > 1e-10:
                f = f / st
            best_obj = best_trial
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 30:
                break

        if it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} obj={best_obj:.12f} "
                  f"score={2.0*n*best_obj:.8f} t={elapsed:.0f}s")

    return best_f, best_obj


def peak_shaving(f_init, time_limit, label="peak_shave"):
    """
    Targeted peak shaving: identify specific high lags and find directions
    that reduce them most effectively.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        # Find top 3 peaks
        top_peaks = np.argpartition(b, -3)[-3:]
        top_peaks = top_peaks[np.argsort(b[top_peaks])[::-1]]

        # For each peak, compute the "obstruction" direction
        # and find a direction that reduces it
        best_direction = None
        best_reduction = 0.0

        for pk in top_peaks:
            lo = max(0, pk - n + 1)
            hi = min(n - 1, pk)
            # The "shadow" of f at lag pk
            shadow = np.zeros(n)
            shadow[lo:hi + 1] = f[pk - lo:pk - lo + (hi - lo + 1)]
            # Reverse
            shadow = shadow[::-1]

            # Direction: reduce where shadow is large
            # Try: f - alpha * shadow, projected
            for alpha in [0.001, 0.005, 0.01, 0.02, 0.05]:
                f_trial = f - alpha * shadow
                f_trial = project_simplex(f_trial)
                obj = fast_convolve(f_trial).max()
                reduction = M - obj
                if reduction > best_reduction:
                    best_reduction = reduction
                    best_direction = f_trial

        if best_direction is None or best_reduction < 1e-12:
            # Try a different strategy: move mass from high-shadow regions
            pk = top_peaks[0]
            lo = max(0, pk - n + 1)
            hi = min(n - 1, pk)
            shadow = np.zeros(n)
            shadow[lo:hi + 1] = f[pk - lo:pk - lo + (hi - lo + 1)][::-1]

            # Find where shadow is largest and move mass from there to where it's smallest
            if shadow.max() > 1e-10:
                f_trial = f.copy()
                # Normalize shadow
                sh_norm = shadow / shadow.sum() if shadow.sum() > 1e-10 else np.ones(n) / n
                # Move a small amount from high-shadow to low-shadow
                # Actually, just try the simple subtraction approach with projection
                for alpha in [0.0005, 0.001, 0.002, 0.005]:
                    f_trial = f - alpha * sh_norm
                    f_trial = project_simplex(f_trial)
                    obj = fast_convolve(f_trial).max()
                    if obj < M - 1e-12:
                        best_direction = f_trial
                        best_reduction = M - obj
                        break

        if best_direction is None or best_reduction < 1e-12:
            # Stuck, try perturbation
            rng = np.random.default_rng(it * 7919)
            f = project_simplex(f + rng.standard_normal(n) * 0.005)
            continue

        f = best_direction
        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()

        if it % 20 == 0:
            print(f"    {label} iter={it} obj={best_obj:.12f} "
                  f"score={2.0*n*best_obj:.8f} t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def coordinate_descent(f_init, time_limit, n_sweeps=100, label="cd"):
    """
    Coordinate descent: for each coordinate, try small perturbations
    and keep the best. Uses a "move mass" strategy.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    rng = np.random.default_rng(999)
    sweep = 0

    while time.monotonic() - t0 < time_limit and sweep < n_sweeps:
        sweep += 1
        # Random order
        order = rng.permutation(n)
        improved = 0

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if f[i] < 1e-8:
                continue

            # Try moving a small fraction of mass from i to a random other coordinate
            for frac in [0.01, 0.05, 0.1]:
                delta = f[i] * frac
                if delta < 1e-10:
                    break

                # Distribute to a partner
                j = rng.integers(0, n)
                if j == i:
                    continue

                f_trial = f.copy()
                f_trial[i] -= delta
                f_trial[j] += delta
                f_trial = np.maximum(f_trial, 0.0)
                st = f_trial.sum()
                if st < 1e-10:
                    continue
                f_trial = f_trial / st

                obj = fast_convolve(f_trial).max()
                if obj < best_obj - 1e-14:
                    f = f_trial
                    best_obj = obj
                    best_f = f.copy()
                    improved += 1
                    break

        if sweep % 20 == 0:
            print(f"    {label} sweep={sweep} obj={best_obj:.12f} "
                  f"score={2.0*n*best_obj:.8f} improved={improved} t={time.monotonic()-t0:.0f}s")

        if improved == 0 and sweep > 5:
            # Try wider perturbations
            for i in rng.choice(n, size=20, replace=False):
                if f[i] < 1e-6:
                    continue
                for frac in [0.2, 0.5]:
                    delta = f[i] * frac
                    j = rng.integers(0, n)
                    if j == i:
                        continue
                    f_trial = f.copy()
                    f_trial[i] -= delta
                    f_trial[j] += delta
                    f_trial = np.maximum(f_trial, 0.0)
                    st = f_trial.sum()
                    if st < 1e-10:
                        continue
                    f_trial = f_trial / st
                    obj = fast_convolve(f_trial).max()
                    if obj < best_obj - 1e-14:
                        f = f_trial
                        best_obj = obj
                        best_f = f.copy()
                        break

    return best_f, best_obj


def interpolated_sequence(f_old, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(f_old)
    if n_old == n_new:
        return f_old.copy()
    old_x = np.linspace(0.0, 1.0, n_old)
    new_x = np.linspace(0.0, 1.0, n_new)
    f_new = np.interp(new_x, old_x, f_old)
    s = f_new.sum()
    if s < 1e-10:
        return np.ones(n_new) / n_new
    return f_new / s

    print(f"\n=== Phase 4: Multi-length exploration (best={best_score:.8f}) ===")
    n_base = len(best_seq)
    candidates = [x for x in [896, 1024, 1280, 1536, 1920, 2048, 2304, 2560]
                  if x != n_base and 512 <= x <= 3000]
    candidates.sort(key=lambda x: abs(x - n_base))

    for target_n in candidates[:4]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t4 = min(rem * 0.12, 180.0)
        if t4 < 25:
            break

        print(f"  Trying n={target_n}, budget={t4:.0f}s")

        # Interpolated initialization
        f_interp = interpolated_sequence(best_seq, target_n)

        # Subgradient first
        t_sg = t4 * 0.5
        t_lp = t4 * 0.4
        t_shave = t4 * 0.1

        f_sg, obj_sg = subgradient_nesterov(f_interp, time_limit=t_sg, top_k=10,
                                             momentum=0.85,
                                             label=f"sg_n{target_n}")
        local_best = f_sg.copy()
        local_score = float(evaluate_sequence(local_best.tolist()))

        if t_lp > 10:
            f_lp, obj_lp = lp_optimize(local_best, time_limit=t_lp,
                                        max_constraints=300, label=f"lp_n{target_n}")
            s_lp = float(evaluate_sequence(f_lp.tolist()))
            if np.isfinite(s_lp) and s_lp < local_score:
                local_best = f_lp.copy()
                local_score = s_lp

        if t_shave > 5 and time.monotonic() < deadline:
            f_sh, obj_sh = peak_shaving(local_best, time_limit=t_shave,
                                         label=f"shave_n{target_n}")
            s_sh = float(evaluate_sequence(f_sh.tolist()))
            if np.isfinite(s_sh) and s_sh < local_score:
                local_best = f_sh.copy()
                local_score = s_sh

        if np.isfinite(local_score) and local_score < best_score:
            best_score = local_score
            best_seq = local_best.copy()
            print(f"  NEW BEST [n{target_n}]: {best_score:.8f}")
        else:
            print(f"  n{target_n}: best={local_score:.8f} (no improvement)")

    # === Phase 5: Final subgradient polish ===
    print(f"\n=== Phase 5: Final Subgradient (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.20, 200.0)
    if t5 > 20:
        n_b = len(best_seq)
        print(f"  Budget: {t5:.0f}s, n={n_b}")
        f5, obj5 = subgradient_nesterov(best_seq, time_limit=t5, top_k=5,
                                         momentum=0.9, label="sg_final")
        score5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(score5) and score5 < best_score:
            best_score = score5
            best_seq = f5.copy()
            print(f"  NEW BEST [sg_final]: {score5:.8f}")
        else:
            print(f"  sg_final: {score5:.8f} (no improvement)")

    # === Phase 6: Final LP ===
    print(f"\n=== Phase 6: Final LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.30, 200.0)
    if t6 > 20:
        print(f"  Budget: {t6:.0f}s")
        f6, obj6 = lp_optimize(best_seq, time_limit=t6, max_constraints=500,
                                label="lp_final")
        score6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(score6) and score6 < best_score:
            best_score = score6
            best_seq = f6.copy()
            print(f"  NEW BEST [lp_final]: {score6:.8f}")
        else:
            print(f"  lp_final: {score6:.8f} (no improvement)")

    # === Phase 7: Final peak shaving ===
    rem = deadline - time.monotonic()
    if rem > 15:
        t7 = min(rem * 0.4, 60.0)
        f7, obj7 = peak_shaving(best_seq, time_limit=t7, label="shave_final")
        score7 = float(evaluate_sequence(f7.tolist()))
        if np.isfinite(score7) and score7 < best_score:
            best_score = score7
            best_seq = f7.copy()
            print(f"  NEW BEST [shave_final]: {score7:.8f}")

    # === Phase 8: Second subgradient pass on best ===
    rem = deadline - time.monotonic()
    if rem > 30:
        t8 = min(rem * 0.5, 100.0)
        f8, obj8 = subgradient_nesterov(best_seq, time_limit=t8, top_k=3,
                                         momentum=0.92, label="sg_polish")
        score8 = float(evaluate_sequence(f8.tolist()))
        if np.isfinite(score8) and score8 < best_score:
            best_score = score8
            best_seq = f8.copy()
            print(f"  NEW BEST [sg_polish]: {score8:.8f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'=' * 60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"Target: 1.5027")
    print(f"Improvement: {1.5027 - best_score:.6f}" if best_score < 1.5027
          else f"Gap to target: {best_score - 1.5027:.6f}")
    print(f"{'=' * 60}")

    return [float(x) for x in best_seq]
