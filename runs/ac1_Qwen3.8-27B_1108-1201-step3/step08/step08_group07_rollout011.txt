"""
Multi-strategy optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses: SBR (Successive Bussgang Resolutions) for flat-spectrum initialization,
LP fictitious play for main optimization, L-BFGS with log-sum-exp smoothing
for polishing, and multi-peak subgradient descent. Tries multiple sequence
lengths and initializations to escape local minima.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.linalg import toeplitz
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    """Build (2n-1) x n matrix C such that C @ g = f * g (linear conv)."""
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_best_response(f, M, prev_g=None):
    """Solve: max sum(g) s.t. C(f) @ g <= M, g >= 0."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None))
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return prev_g if prev_g is not None else f.copy()
    if g.value is None:
        return prev_g if prev_g is not None else f.copy()
    return np.maximum(g.value, 0)


def sbr_iteration(a0, time_limit, tau=0.5, verbose=True, label="SBR"):
    """
    Successive Bussgang Resolutions: iteratively flatten the magnitude
    spectrum while maintaining non-negativity.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        A = rfft(a, fft_size)
        power = np.abs(A) ** 2
        total_power = power.sum()
        target = total_power / fft_size
        ratio = np.where(power > 1e-30, target / power, 1.0)
        ratio = np.clip(ratio, 0.01, 100.0)
        A_new = A * (ratio ** tau)
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        if verbose and iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")

    return best_a, best_score


def sbr_blend(a0, time_limit, tau=0.5, blend=0.5, verbose=True, label="SBR-blend"):
    """SBR with blended magnitude update for stability."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        A = rfft(a, fft_size)
        power = np.abs(A) ** 2
        total_power = power.sum()
        target = total_power / fft_size
        ratio = np.where(power > 1e-30, target / power, 1.0)
        ratio = np.clip(ratio, 0.01, 100.0)
        # Blend the magnitude update
        new_power = (ratio ** tau) * power
        new_power = blend * new_power + (1 - blend) * target
        new_magnitude = np.sqrt(new_power)
        # Keep the phase from current A
        angle = np.angle(A)
        A_new = new_magnitude * np.exp(1j * angle)
        # Make it conjugate-symmetric for real output
        A_new = np.concatenate([A_new[:fft_size//2+1], np.conj(A_new[2:fft_size//2+1][::-1])])
        a_new = ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        if verbose and iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")

    return best_a, best_score


def lp_fictitious_play(a0, time_limit, step=0.4, verbose=True, label="LP-FP"):
    """LP-based fictitious play with warm-starting."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    prev_g = None
    stall_count = 0
    stall_threshold = 300

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()

        g = solve_best_response(a, M, prev_g=prev_g)
        prev_g = g
        g_sum = g.sum()

        if g_sum < 1e-10:
            break

        g_norm = g / g_sum
        a_new = (1 - step) * a + step * g_norm
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss

        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1

        a = a_new

        if stall_count > stall_threshold:
            step = max(0.05, step * 0.7)
            rng = np.random.default_rng(iters)
            a = best_a + rng.standard_normal(n) * 0.001
            a = np.maximum(a, 0)
            a = a / a.sum()
            prev_g = None
            stall_count = 0

        if verbose and iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, step={step:.4f}, "
                  f"time={elapsed:.0f}s/{time_limit:.0f}s")

    return best_a, best_score

def softmax_from_z(z):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-30))


def lse_objective(z, T, n):
    """Log-sum-exp smoothed objective in softmax parameterization."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = T * conv
    max_shifted = shifted.max()
    lse = (max_shifted + np.log(np.exp(shifted - max_shifted).sum())) / T
    return 2.0 * n * lse


def lse_gradient(z, T, n):
    """Gradient of LSE smoothed objective."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = T * conv
    max_shifted = shifted.max()
    w = np.exp(shifted - max_shifted)
    w = w / w.sum()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
    S = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - S)
    return 2.0 * n * grad_z


def lbgfs_optimize(a0, time_limit, T=500, maxiter=5000, verbose=True):
    """L-BFGS on smoothed log-sum-exp objective in log-space."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return lse_objective(z, T, n)

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return lse_gradient(z, T, n)

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14,
                     'maxfun': maxiter * 100, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        score = compute_score(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception:
        pass

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"  L-BFGS T={T}: best={best_score:.8f}, time={elapsed:.0f}s")

    return best_a, best_score


def multi_peak_subgradient(a0, time_limit, eps=0.05, verbose=True, label="MPSG"):
    """
    Multi-peak subgradient descent with projected steps onto the simplex.
    Uses FFT to compute the gradient efficiently.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()
    iters = 0
    lr_base = 0.01

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        max_conv = conv.max()
        threshold = max_conv * (1.0 - eps)
        active = np.where(conv >= threshold)[0]

        # Build weight vector w on the convolution grid
        w = np.zeros(2 * n - 1)
        w[active] = 1.0
        w = w / len(active)

        # Gradient: grad[m] = 2 * sum_k w[k] * a[k-m] (for valid k-m)
        # This is 2 * (w * flip(a)) evaluated appropriately
        # Let b[i] = a[n-1-i] (reversed a)
        b = a[::-1]
        # (w * b)[k] = sum_j w[j] * b[k-j]
        # We want: sum_k w[k] * a[k-m] = sum_k w[k] * b[n-1-(k-m)] = sum_k w[k] * b[m - (n-1-k)]
        # Let l = n-1-k, then k = n-1-l, and sum_l w[n-1-l] * b[m-l] = (flip(w) * b)[m]
        flip_w = w[::-1]
        W = rfft(flip_w, fft_size)
        B = rfft(b, fft_size)
        grad = 2.0 * irfft(W * B, fft_size)[:n]

        # Adaptive learning rate
        lr = lr_base / (1.0 + 0.01 * iters)
        a_trial = a - lr * grad
        a_trial = np.maximum(a_trial, 0)
        ss = a_trial.sum()
        if ss > 1e-15:
            a_trial = a_trial / ss
            score = compute_score(a_trial)
            if score < best_score - 1e-14:
                best_score = score
                best_a = a_trial.copy()
                a = a_trial
            else:
                # If no improvement, reduce lr
                lr_base = max(0.001, lr_base * 0.95)
                a = a + 0.001 * np.random.default_rng(iters).standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
        else:
            break

        if verbose and iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, lr={lr_base:.6f}, time={elapsed:.0f}s")

    return best_a, best_score


def coordinate_descent(a0, time_limit, n_pairs=200, verbose=True):
    """Coordinate descent with pairwise transfers."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or a[i] < 1e-14:
                continue

            best_local = best_score
            best_t = 0.0
            base = min(a[i], a[j]) if a[j] > 1e-15 else a[i]

            for frac in [0.05, 0.1, 0.2, 0.4, 0.6, 0.8]:
                t = frac * base
                if t >= a[i] - 1e-15:
                    t = a[i] * 0.5
                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_t = t

            if best_t != 0.0:
                a[i] -= best_t
                a[j] += best_t
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = best_local
                best_a = a.copy()
                improved = True

        if not improved and iters > 20:
            noise = rng.standard_normal(n) * 0.003
            a = best_a.copy() + noise
            a = np.maximum(a, 0)
            a = a / a.sum()

        if verbose and iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"  CD iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")

    return best_a, best_score


def check_and_update(a, label, best_seq, best_score, verbose=True):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def generate_initializations(n, seed=42):
    """Generate diverse initializations for a given length n."""
    rng = np.random.default_rng(seed)
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine bell
    x = np.arange(n)
    cos_bell = np.cos(np.pi * x / (2.0 * n)) ** 2
    cos_bell = np.maximum(cos_bell, 0)
    inits.append(cos_bell / cos_bell.sum())

    # Gaussian
    center = n / 2.0
    sigma = n / 4.0
    gauss = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    inits.append(gauss / gauss.sum())

    # U-shaped (inverse of cosine bell)
    u_shape = 1.0 / (np.cos(np.pi * x / (2.0 * n)) ** 2 + 0.1)
    inits.append(u_shape / u_shape.sum())

    # Flat-spectrum random
    z = rng.standard_normal(n) + 1j * rng.standard_normal(n)
    Z = fft(z, 2 * n)
    Z_flat = Z / np.abs(Z)
    Z_flat[0] = np.abs(Z_flat[0])  # keep DC real
    # Enforce conjugate symmetry
    Z_sym = np.zeros_like(Z_flat)
    Z_sym[:n+1] = Z_flat[:n+1]
    for k in range(1, n):
        Z_sym[len(Z_flat)-k] = np.conj(Z_flat[k])
    z_flat = ifft(Z_sym)[:n].real
    z_abs = np.abs(z_flat)
    if z_abs.sum() > 1e-10:
        inits.append(z_abs / z_abs.sum())

    # Another flat-spectrum with different seed
    z2 = rng.standard_normal(n) + 1j * rng.standard_normal(n)
    Z2 = fft(z2, 2 * n)
    Z2_flat = Z2 / (np.abs(Z2) + 1e-10)
    Z2_sym = np.zeros_like(Z2_flat)
    Z2_sym[:n+1] = Z2_flat[:n+1]
    for k in range(1, n):
        Z2_sym[len(Z2_flat)-k] = np.conj(Z2_flat[k])
    z2_flat = ifft(Z2_sym)[:n].real
    z2_abs = np.abs(z2_flat)
    if z2_abs.sum() > 1e-10:
        inits.append(z2_abs / z2_abs.sum())

    # Random non-negative
    r = rng.random(n)
    inits.append(r / r.sum())

    # Half-cosine (concentrated at edges)
    edge = np.abs(np.cos(np.pi * x / n))
    inits.append(edge / edge.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase, multi-length optimization:
    1. Generate diverse initializations at multiple lengths
    2. SBR spectral flattening for fast initial progress
    3. LP fictitious play for main optimization
    4. L-BFGS with log-sum-exp smoothing for polish
    5. Multi-peak subgradient descent
    6. Coordinate descent
    7. Final LP polish and nearby length exploration
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Initialized: n={n_init}, score={best_score:.8f}")

    n_parent = len(parent)
    n_candidates = [512, 768, 1024, 1536]
    # Make sure parent length is included
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)
    n_candidates = sorted(set(n_candidates))

    print(f"\nCandidate lengths: {n_candidates}")
    print(f"Initial best: {best_score:.8f} (n={len(best_seq)})")

    # === Phase 1: SBR initialization at multiple lengths ===
    # Allocate time proportionally, favoring the parent length
    phase1_budget = min((deadline - time.monotonic()) * 0.25, 350.0)
    per_length_budget = phase1_budget / len(n_candidates)
    print(f"\n=== Phase 1: SBR Multiple Lengths (budget={phase1_budget:.0f}s) ===")

    sbr_results = {}
    for n_cand in n_candidates:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t_sbr = min(rem * 0.2, per_length_budget)
        if t_sbr < 10:
            break

        # Get best initialization for this length
        if n_cand == n_parent and parent is not None:
            init_a = parent.copy()
        else:
            init_a = interpolate_sequence(best_seq, n_cand)

        # Run SBR with different tau values
        for tau in [0.3, 0.5, 0.7]:
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            t_i = min(rem * 0.15, 40.0)
            a_sbr, sc_sbr = sbr_iteration(init_a, t_i, tau=tau, verbose=False,
                                         label=f"SBR-tau{tau}-n{n_cand}")
            a_sbr, sc_sbr = check_and_update(a_sbr, f"sbr_tau{tau}_n{n_cand}",
                                           best_seq, best_score, verbose=False)
            if sc_sbr < sbr_results.get(n_cand, float('inf')):
                sbr_results[n_cand] = sc_sbr
                sbr_best_a = a_sbr

        # Also try blended SBR
        rem = deadline - time.monotonic()
        if rem > 20:
            t_i = min(rem * 0.1, 30.0)
            a_sbr2, sc_sbr2 = sbr_blend(init_a, t_i, tau=0.5, blend=0.7, verbose=False,
                                       label=f"SBRb-n{n_cand}")
            a_sbr2, sc_sbr2 = check_and_update(a_sbr2, f"sbrb_n{n_cand}",
                                             best_seq, best_score, verbose=False)
            if sc_sbr2 < sbr_results.get(n_cand, float('inf')):
                sbr_results[n_cand] = sc_sbr2
                sbr_best_a = a_sbr2

        print(f"  n={n_cand}: SBR best={sbr_results.get(n_cand, 'N/A')}")

    # === Phase 2: LP Fictitious Play on best candidates ===
    print(f"\n=== Phase 2: LP Fictitious Play (best={best_score:.8f}) ===")
    phase2_budget = min((deadline - time.monotonic()) * 0.35, 600.0)

    # Run LP on the current best
    rem = deadline - time.monotonic()
    t_lp1 = min(rem * 0.4, phase2_budget * 0.4)
    if t_lp1 > 30:
        print(f"  LP step=0.5, n={len(best_seq)}, budget={t_lp1:.0f}s")
        a_lp, sc_lp = lp_fictitious_play(best_seq, t_lp1, step=0.5, verbose=True, label="LP-0.5")
        best_seq, best_score = check_and_update(a_lp, "lp_s0.5", best_seq, best_score)

    # LP with medium step
    rem = deadline - time.monotonic()
    t_lp2 = min(rem * 0.3, phase2_budget * 0.3)
    if t_lp2 > 30:
        print(f"  LP step=0.3, n={len(best_seq)}, budget={t_lp2:.0f}s")
        a_lp, sc_lp = lp_fictitious_play(best_seq, t_lp2, step=0.3, verbose=True, label="LP-0.3")
        best_seq, best_score = check_and_update(a_lp, "lp_s0.3", best_seq, best_score)

    # LP with decreasing step
    rem = deadline - time.monotonic()
    t_lp3 = min(rem * 0.3, phase2_budget * 0.3)
    if t_lp3 > 30:
        print(f"  LP decreasing, n={len(best_seq)}, budget={t_lp3:.0f}s")
        a_lp, sc_lp = lp_decreasing_step(best_seq, t_lp3, alpha=0.03, verbose=True, label="LP-dec")
        best_seq, best_score = check_and_update(a_lp, "lp_dec", best_seq, best_score)

    # === Phase 3: Try LP on other lengths ===
    print(f"\n=== Phase 3: LP on Other Lengths (best={best_score:.8f}) ===")
    for n_cand in n_candidates:
        if n_cand == len(best_seq):
            continue
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_alt = min(rem * 0.15, 80.0)
        if t_alt < 20:
            continue
        init_alt = interpolate_sequence(best_seq, n_cand)
        # Quick SBR first
        t_sbr_quick = min(t_alt * 0.4, 20.0)
        a_s, sc_s = sbr_iteration(init_alt, t_sbr_quick, tau=0.5, verbose=False, label="SBR-quick")
        # Then LP
        t_lp_alt = t_alt - t_sbr_quick - 5
        if t_lp_alt > 10:
            a_a, sc_a = lp_fictitious_play(a_s, t_lp_alt, step=0.4, verbose=False, label=f"LP-n{n_cand}")
            a_a, sc_a = check_and_update(a_a, f"lp_n{n_cand}", best_seq, best_score)

    # === Phase 4: L-BFGS Polish ===
    print(f"\n=== Phase 4: L-BFGS Polish (best={best_score:.8f}) ===")
    for T_val in [200, 500, 1000, 2000]:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t_lb = min(rem * 0.1, 60.0)
        a_l, sc_l = lbgfs_optimize(best_seq, t_lb, T=T_val, maxiter=3000, verbose=True)
        best_seq, best_score = check_and_update(a_l, f"lbgfs_T{T_val}", best_seq, best_score)

    # === Phase 5: Multi-peak Subgradient ===
    print(f"\n=== Phase 5: Multi-Peak Subgradient (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_mpsg = min(rem * 0.1, 60.0)
    if t_mpsg > 15:
        a_m, sc_m = multi_peak_subgradient(best_seq, t_mpsg, eps=0.05, verbose=True, label="MPSG")
        best_seq, best_score = check_and_update(a_m, "mpsg", best_seq, best_score)

    # === Phase 6: Coordinate Descent ===
    print(f"\n=== Phase 6: Coordinate Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_cd = min(rem * 0.1, 60.0)
    if t_cd > 15:
        a_c, sc_c = coordinate_descent(best_seq, t_cd, n_pairs=200, verbose=True)
        best_seq, best_score = check_and_update(a_c, "coord_desc", best_seq, best_score)

    # === Phase 7: Final LP polish ===
    print(f"\n=== Phase 7: Final LP Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_final = min(rem * 0.3, 100.0)
    if t_final > 20:
        a_f, sc_f = lp_fictitious_play(best_seq, t_final, step=0.2, verbose=True, label="LP-final")
        best_seq, best_score = check_and_update(a_f, "lp_final", best_seq, best_score)

    # === Phase 8: Try nearby lengths with L-BFGS ===
    print(f"\n=== Phase 8: Nearby Lengths (best={best_score:.8f}) ===")
    n_best = len(best_seq)
    for alt_n in [n_best + 2, n_best - 2, n_best + 4, n_best - 4, n_best + 6, n_best - 6]:
        if alt_n < 128 or alt_n > 4096:
            continue
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t_alt = min(rem * 0.15, 40.0)
        init_alt = interpolate_sequence(best_seq, alt_n)
        a_alt, sc_alt = lbgfs_optimize(init_alt, t_alt, T=500, maxiter=2000, verbose=True)
        best_seq, best_score = check_and_update(a_alt, f"n{alt_n}", best_seq, best_score)

    # === Phase 9: Final combined polish ===
    print(f"\n=== Phase 9: Final Combined Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        # Quick LP
        t_f1 = min(rem * 0.4, 50.0)
        a_x, sc_x = lp_fictitious_play(best_seq, t_f1, step=0.15, verbose=False, label="LP-end")
        best_seq, best_score = check_and_update(a_x, "lp_end", best_seq, best_score)

        # Quick L-BFGS
        rem = deadline - time.monotonic()
        if rem > 20:
            t_f2 = min(rem * 0.4, 30.0)
            a_y, sc_y = lbgfs_optimize(best_seq, t_f2, T=1000, maxiter=2000, verbose=False)
            best_seq, best_score = check_and_update(a_y, "lbgfs_end", best_seq, best_score)

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
