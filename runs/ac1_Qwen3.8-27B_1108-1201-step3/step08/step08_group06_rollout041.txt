```python
def project_simplex(x):
    """Project x onto the probability simplex."""
    n = len(x)
    if x.sum() <= 1 and (x >= 0).all():
        return x / x.sum() if x.sum() > 0 else np.ones(n)/n
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(x - theta, 0)
```

            state['best_score'] = sc
            nonlocal best_a
            best_a = a_tmp.copy()
        # Scale objective for numerical stability
        return 2.0 * n * sm

    def jac(z):
        if time.monotonic() - state['t0'] > state['limit']:
            return np.zeros_like(z)
        a_tmp = softmax_normalize(z)
        conv = fast_convolve(a_tmp)
        scaled = T * conv
        max_s = scaled.max()
        exp_s = np.exp(scaled - max_s)
        s = exp_s.sum()
        w = exp_s / s
        grad_a = 2.0 * cross_corr_wa(w, a_tmp)
        grad_z = gradient_in_log_space(a_tmp, grad_a)
        return 2.0 * n * grad_z

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-12,
                     'maxfun': maxiter * 100, 'disp': False}
        )
        a_out = softmax_normalize(res.x)
        score = compute_score(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception:
        pass

    return best_a, best_score


def make_initialization(n, method='bump', rng=None):
    """Create diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(0)

    if method == 'uniform':
        a = np.ones(n)
    elif method == 'bump':
        # Triangular bump
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t)
    elif method == 'bimodal':
        # Two bumps
        t = np.linspace(0, 1, n)
        a = np.exp(-20 * (t - 0.25) ** 2) + np.exp(-20 * (t - 0.75) ** 2)
    elif method == 'alternating':
        # Alternating high/low
        a = np.ones(n)
        a[::2] = 1.5
        a[1::2] = 0.5
    elif method == 'fourier':
        # Based on a few Fourier modes
        t = np.linspace(0, 2 * np.pi, n)
        a = 1.0 + 0.3 * np.cos(2 * t) + 0.1 * np.cos(5 * t)
    elif method == 'edge focused':
        # Emphasize edges
        t = np.linspace(0, 1, n)
        a = 1.0 / (0.1 + np.abs(t - 0.5) * 4)
    else:
        a = np.ones(n)

    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score, verbose=True):
    """Evaluate and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimization combining:
    - Annealed smoothed gradient descent (main engine)
    - Subgradient refinement
    - L-BFGS in log-space
    - Targeted coordinate descent
    - Length exploration
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    n_parent = 768
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            n_parent = len(parent)
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={n_parent}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = make_initialization(n_parent, 'bump')
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Initialized from scratch: n={n_parent}, score={best_score:.8f}")

    # === Phase 1: Annealed Smoothed Gradient Descent ===
    rem = deadline - time.monotonic()
    t1 = min(500.0, rem * 0.35)
    print(f"\n=== Phase 1: Annealed Smoothed Gradient (budget={t1:.0f}s, best={best_score:.8f}) ===")
    a1, s1 = annealed_smoothed_gradient(best_seq, t1, seed=seed, verbose=True)
    best_seq, best_score = check_and_update(a1, "smooth_grad", best_seq, best_score)

    # === Phase 2: Subgradient Refinement ===
    rem = deadline - time.monotonic()
    t2 = min(200.0, rem * 0.15)
    if rem > 50:
        print(f"\n=== Phase 2: Subgradient (budget={t2:.0f}s, best={best_score:.8f}) ===")
        a2, s2 = subgradient_descent(best_seq, t2, seed=seed + 1, verbose=True)
        best_seq, best_score = check_and_update(a2, "subgrad", best_seq, best_score)

    # === Phase 3: L-BFGS in Log-Space ===
    rem = deadline - time.monotonic()
    if rem > 50:
        t3 = min(150.0, rem * 0.12)
        print(f"\n=== Phase 3: L-BFGS Log-Space (budget={t3:.0f}s, best={best_score:.8f}) ===")
        for T_val in [3000, 10000]:
            rem_sub = deadline - time.monotonic()
            if rem_sub < 30:
                break
            t_sub = min(t3 / 2.0, rem_sub * 0.6)
            a3, s3 = lbfgs_logspace(best_seq, t_sub, T=T_val, maxiter=2000,
                                     seed=seed + 2, verbose=False)
            best_seq, best_score = check_and_update(a3, f"lbfgs_T{T_val}", best_seq, best_score)

    # === Phase 4: Targeted Coordinate Descent ===
    rem = deadline - time.monotonic()
    t4 = min(350.0, rem * 0.25)
    if rem > 30:
        print(f"\n=== Phase 4: Coordinate Descent (budget={t4:.0f}s, best={best_score:.8f}) ===")
        a4, s4 = coordinate_descent_targeted(best_seq, t4, seed=seed + 3, verbose=True)
        best_seq, best_score = check_and_update(a4, "coord_desc", best_seq, best_score)

    # === Phase 5: Second round of smoothed gradient on refined solution ===
    rem = deadline - time.monotonic()
    t5 = min(200.0, rem * 0.2)
    if rem > 30:
        print(f"\n=== Phase 5: Smoothed Gradient Round 2 (budget={t5:.0f}s, best={best_score:.8f}) ===")
        a5, s5 = annealed_smoothed_gradient(best_seq, t5, seed=seed + 4, verbose=False)
        best_seq, best_score = check_and_update(a5, "smooth_grad2", best_seq, best_score)

    # === Phase 6: Length Exploration ===
    rem = deadline - time.monotonic()
    if rem > 60:
        print(f"\n=== Phase 6: Length Exploration (budget={rem:.0f}s, best={best_score:.8f}) ===")
        nearby_lengths = [n_parent + 2, n_parent - 2, n_parent + 4, n_parent - 4,
                          n_parent + 8, n_parent - 8, n_parent + 1, n_parent - 1]
        for alt_n in nearby_lengths:
            if alt_n < 128 or alt_n > 3000:
                continue
            rem_sub = deadline - time.monotonic()
            if rem_sub < 40:
                break
            t_sub = min(80.0, rem_sub * 0.3)
            if t_sub < 20:
                break
            init_alt = interpolate_sequence(best_seq, alt_n)
            print(f"  Trying n={alt_n} (budget={t_sub:.0f}s)")
            # Quick gradient descent
            a_alt, s_alt = annealed_smoothed_gradient(init_alt, t_sub * 0.5,
                                                       seed=seed + alt_n, verbose=False)
            best_seq, best_score = check_and_update(a_alt, f"n{alt_n}_grad", best_seq, best_score)
            # Quick coordinate descent
            rem_sub2 = deadline - time.monotonic()
            if rem_sub2 > 20:
                a_alt2, s_alt2 = coordinate_descent_targeted(a_alt, min(30.0, rem_sub2 * 0.3),
                                                              seed=seed + alt_n + 1, verbose=False)
                best_seq, best_score = check_and_update(a_alt2, f"n{alt_n}_cd", best_seq, best_score)

    # === Phase 7: Final Polish ===
    rem = deadline - time.monotonic()
    if rem > 30:
        t7 = min(100.0, rem * 0.5)
        print(f"\n=== Phase 7: Final Polish (budget={t7:.0f}s, best={best_score:.8f}) ===")
        a7, s7 = coordinate_descent_targeted(best_seq, t7, seed=seed + 100, verbose=False)
        best_seq, best_score = check_and_update(a7, "final_cd", best_seq, best_score)

    # === Phase 8: Final L-BFGS ===
    rem = deadline - time.monotonic()
    if rem > 20:
        t8 = min(50.0, rem * 0.7)
        a8, s8 = lbfgs_logspace(best_seq, t8, T=5000, maxiter=1000,
                                 seed=seed + 200, verbose=False)
        best_seq, best_score = check_and_update(a8, "final_lbfgs", best_seq, best_score)

    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
