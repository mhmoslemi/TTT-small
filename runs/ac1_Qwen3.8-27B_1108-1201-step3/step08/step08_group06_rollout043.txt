```python
"""
Search for a non-negative step-function height sequence minimizing

    2*n*max_k (a*a)[k] / (sum(a))^2.

The solver combines several complementary local methods:

1. A new "convolution clipping + square-root reconstruction" method.
   Given a = current sequence, compute c = a*a, clip/flatt
en c, and take a causal square root of the clipped target.
   This directly pushes the auto-convolution toward a flatter shape.
2. Active-set linearized-LP descent.  At each step, choose the most
   important peak lags, linearize their quadratic constraints, solve a small
   LP for a feasible descent direction, then do an exact line search.
3. Log-space gradient descent with L2, soft-max, and top-peak weights.
4. Greedy LP fictitious-play polishing.
5. Fast exact two-coordinate pair descent as a final polish.

The code is self-contained and returns the best sequence found within the
budget.
"""

import time
import numpy as np
from operator import itemgetter

try:
    from scipy.fft import rfft, irfft
except Exception:
    from numpy.fft import rfft, irfft

try:
    from scipy.optimize import linprog

    linprog = None

_LS_ALPHAS = (1.0, 0.75, 0.5, 0.35, 0.2, 0.1, 0.05)
_PAIR_LS_ALPHAS = (1.0, 0.75, 0.5, 0.35, 0.25, 0.15, 0.1)


def next_pow2_at_least(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def normalize_nonneg(a):
    arr = np.asarray(a, dtype=np.float64).ravel()
    if arr.size == 0:
        return arr
    arr = arr.copy()
    arr[~np.isfinite(arr)] = 0.0
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 1e-300:
        return arr
    return arr / s


def softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size == 0:
        return z
    z = z - np.max(z)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(z.size, dtype=np.float64) / float(z.size)
    return e / s


def autoconv_normalized(a, P=None):
    """Linear auto-convolution of a, normalized so that sum(a*a)=1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.zeros(2 * n - 1, dtype=np.float64)
    a = a / s
    L = 2 * n - 1
    if P is None or P < L:
        P = next_pow2_at_least(L)
    A = rfft(a, P)
    c = irfft(A * A, P)[:L]
    c = np.maximum(c, 0.0)
    cs = float(c.sum())
    if np.isfinite(cs) and cs > 0.0:
        c = c / cs
    return c


def score_from_conv(c, n):
    if c.size == 0:
        return np.inf
    return 2.0 * float(n) * float(np.max(c))


def internal_score(a):
    a = normalize_nonneg(a)
    if a.size == 0:
        return np.inf
    c = autoconv_normalized(a)
    return score_from_conv(c, a.size)


def consider_candidate(best_seq, best_score, seq, label="", exact=False):
    a = normalize_nonneg(seq)
    if a.size == 0:
        return best_seq, best_score
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        return best_seq, best_score
    a = a / s
    if exact:
        try:
            sc = float(evaluate_sequence(a.tolist()))
        except Exception:
            sc = internal_score(a)
    else:
        c = autoconv_normalized(a)
        sc = score_from_conv(c, a.size)
    if np.isfinite(sc) and sc < best_score:
        best_seq = a.copy()
        best_score = float(sc)
        print(f"  NEW BEST [{label}]: {best_score:.8f} (n={best_seq.size})")
    return best_seq, best_score


def interpolate_sequence(f, new_n):
    f = normalize_nonneg(f)
    old_n = f.size
    if old_n == new_n:
        return f.copy()
    if old_n == 0 or new_n <= 0:
        return np.ones(new_n, dtype=np.float64) / float(max(1, new_n))
    s = float(f.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(new_n, dtype=np.float64) / float(max(1, new_n))
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    g = np.interp(x_new, x_old, f / s)
    return normalize_nonneg(g)


def make_random_smooth(n, rng):
    z = rng.normal(0.0, 1.0, n)
    t = np.arange(n, dtype=np.float64)

    f1 = int(rng.integers(1, max(2, n // 32)))
    p1 = float(rng.uniform(0.0, 2.0 * np.pi))
    z += 0.7 * np.cos(2.0 * np.pi * f1 * t / n + p1)

    f2 = int(rng.integers(1, max(2, n // 8)))
    p2 = float(rng.uniform(0.0, 2.0 * np.pi))
    z += 0.35 * np.cos(2.0 * np.pi * f2 * t / n + p2)

    a = softmax_from_z(z)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def sqrt_flat_prefix(n):
    """
    Causal square root of a flat length-(2n-1) target, matching the first n
    coefficients. This is a useful deterministic starting point.
    """
    if n <= 0:
        return np.array([], dtype=np.float64)
    L = 2 * n - 1
    alpha = 1.0 / float(L)
    p = np.zeros(n, dtype=np.float64)
    p0_sq = alpha
    if p0_sq <= 0.0:
        return np.ones(n, dtype=np.float64) / float(n)
    p0 = np.sqrt(p0_sq)
    p[0] = p0

    for k in range(1, n):
        if k > 1:
            cross = float(np.dot(p[1:k], p[k - 1:0:-1]))
        else:
            cross = 0.0
        num = alpha - cross
        if not np.isfinite(num):
            num = 0.0
        p[k] = num / (2.0 * p0)

    p[~np.isfinite(p)] = 0.0
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    return p / s


def stable_softmax_weights(c, T):
    c = np.asarray(c, dtype=np.float64).ravel()
    if c.size == 0:
        return c
    if not np.isfinite(T) or T <= 0.0:
        T = 1e-8
    x = c / T
    x = x - np.max(x)
    w = np.exp(x)
    s = float(w.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full_like(c, 1.0 / c.size, dtype=np.float64)
    return w / s


def top_weights(c, K=20):
    c = np.asarray(c, dtype=np.float64).ravel()
    n = c.size
    if n == 0:
        return c
    K = max(1, min(K, n))
    idx = np.argsort(c)[::-1][:K]
    w = np.zeros(n, dtype=np.float64)
    w[idx] = 1.0
    s = float(w.sum())
    if s <= 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return w / s


def corr_a_w(a, w, A=None, P=None):
    """
    e[i] = sum_s a[s] * w[s+i] for i=0..n-1.
    This is the derivative kernel for weighted peak objectives.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 3 * n - 2
    if P is None or P < need:
        P = next_pow2_at_least(need)
    if A is None:
        A = rfft(a, P)
    Wr = rfft(w[::-1], P)
    v = irfft(A * Wr, P)
    return v[n - 1:2 * n - 1][::-1]


def greedy_lp(f, M, order):
    f = np.asarray(f, dtype=np.float64).ravel()
    n = f.size
    L = 2 * n - 1
    g = np.zeros(n, dtype=np.float64)
    if n == 0 or M <= 0.0:
        return g

    slack = np.full(L, float(M), dtype=np.float64)
    pos_mask = f > 1e-12
    if not np.any(pos_mask):
        return g

    inv_f = np.zeros(n, dtype=np.float64)
    inv_f[pos_mask] = 1.0 / f[pos_mask]
    inv_pos = inv_f[pos_mask]
    f_pos = f[pos_mask]

    for j in order:
        vals = slack[j:j + n][pos_mask] * inv_pos
        vals = vals[np.isfinite(vals)]
        if vals.size == 0:
            continue
        m = float(np.min(vals))
        if not np.isfinite(m) or m <= 1e-14:
            continue
        g[j] = m
        slack[j:j + n] -= f * m
        # Numerical guard: if serious negative slack appears, stop safely.
        if np.min(slack) < -1e-9:
            slack += 1e-9
    return g


def geometric_mix(f, g, alpha):
    eps = 1e-300
    lf = np.log(np.maximum(f, eps))
    lg = np.log(np.maximum(g, eps))
    h = np.exp((1.0 - alpha) * lf + alpha * lg)
    h = np.maximum(h, 0.0)
    s = float(h.sum())
    if s <= 0.0:
        return normalize_nonneg(f)
    return h / s


def linear_mix(f, g, alpha):
    return normalize_nonneg((1.0 - alpha) * f + alpha * g)


def solve_active_lp(a, c, ks, U):
    if linprog is None:
        return None
    a = np.asarray(a, dtype=np.float64).ravel()
    c = np.asarray(c, dtype=np.float64).ravel()
    ks = np.asarray(ks, dtype=int).ravel()
    n = a.size
    K = ks.size
    if n == 0 or K == 0 or U <= 0.0:
        return None

    A_ub = np.zeros((K, n + 1), dtype=np.float64)
    for r, k in enumerate(ks):
        i0 = max(0, int(k) - n + 1)
        i1 = min(n - 1, int(k))
        if i0 <= i1:
            idx = np.arange(i0, i1 + 1)
            A_ub[r, idx] = 2.0 * a[int(k) - idx]
    A_ub[:, -1] = -1.0
    b_ub = -c[ks].astype(np.float64)

    cvec = np.zeros(n + 1, dtype=np.float64)
    cvec[-1] = 1.0

    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1, dtype=np.float64)

    lo = -a + 1e-14
    hi = np.full(n, float(U), dtype=np.float64)
    # Ensure lo < hi strictly for HS.
    hi = np.maximum(hi, lo + 1e-15)

    bounds = [None] * (n + 1)
    for i in range(n):
        bounds[i] = (float(lo[i]), float(hi[i]))
    bounds[n] = (0.0, None)

    try:
        res = linprog(
            cvec,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options={"disp": False}
        )
    except Exception:
        return None

    if res.status != 0 or res.x is None:
        return None

    delta = np.asarray(res.x[:n], dtype=np.float64)
    if not np.all(np.isfinite(delta)):
        return None

    delta = np.clip(delta, -a + 1e-14, float(U))
    s = float(delta.sum())
    if abs(s) > 1e-12:
        delta = delta - s / float(n)
        delta = np.clip(delta, -a + 1e-14, float(U))
    return delta


def sqrt_reconstruct(target, n):
    """
    Given target c[0],..., find p[0..n-1] such that first n coefficients of
    p*p approximate target. Uses causal square-root recurrence.
    """
    target = np.asarray(target, dtype=np.float64).ravel()
    if n <= 0:
        return np.array([], dtype=np.float64)
    if target.size < n:
        target = np.concatenate([target, np.zeros(n - target.size)])

    p = np.zeros(n, dtype=np.float64)
    p0_sq = float(target[0])
    if not np.isfinite(p0_sq) or p0_sq <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)

    p0 = float(np.sqrt(p0_sq))
    p[0] = p0

    for k in range(1, n):
        if k > 1:
            cross = float(np.dot(p[1:k], p[k - 1:0:-1]))
        else:
            cross = 0.0
        num = float(target[k]) - cross
        if not np.isfinite(num):
            num = 0.0
        p[k] = num / (2.0 * p0)

    p[~np.isfinite(p)] = 0.0
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    return p / s


def sqrt_clip_polish(a0, time_limit=60.0, verbose=True, seed=12345):
    """
    Convolution clipping + causal square-root reconstruction.
    This is a global-ish local method that often makes good initial progress.
    """
    if time_limit <= 1e-3:
        return normalize_nonneg(a0)

    rng = np.random.default_rng(seed)
    a = normalize_nonneg(a0)
    n = a.size
    if n == 0:
        return a

    P = next_pow2_at_least(2 * n - 1)
    c = autoconv_normalized(a, P)
    best = a.copy()
    best_score = score_from_conv(c, n)
    best_l2 = float(np.dot(c, c))

    avg = a.copy()
    delta = 0.25
    no_improve = 0
    it = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        it += 1
        c = autoconv_normalized(a, P)
        M = float(c.max())
        score = score_from_conv(c, n)
        l2 = float(np.dot(c, c))

        if score < best_score:
            best_score = score
            best = a.copy()
            best_l2 = l2

        if M <= 1e-300:
            break

        tau = max(0.0, M * (1.0 - delta))
        r = np.minimum(c, tau)
        r += 1e-15
        rs = float(r.sum())
        if rs <= 0.0:
            r = c.copy()
            rs = float(r.sum())
        r = r / rs

        avg_c = 1.0 / float(r.size)
        if r.size > 0 and r[0] < 0.25 * avg_c:
            r[0] = 0.25 * avg_c
        if r.size > 0 and r[-1] < 0.25 * avg_c:
            r[-1] = 0.25 * avg_c
        r = np.maximum(r, 0.0)
        r = r / float(r.sum())

        p_pre = sqrt_reconstruct(r, n)
        p_suf = sqrt_reconstruct(r[::-1], n)[::-1]
        p_avg = normalize_nonneg(0.5 * p_pre + 0.5 * p_suf)

        roots = (p_pre, p_suf, p_avg)

        # Choose most promising root with a moderate mix.
        best_root = p_pre
        best_root_score = score_from_conv(autoconv_normalized(linear_mix(a, p_pre, 0.5), P), n)
        for p in (p_suf, p_avg):
            mix = linear_mix(a, p, 0.5)
            cm = autoconv_normalized(mix, P)
            sm = score_from_conv(cm, n)
            if sm < best_root_score:
                best_root_score = sm
                best_root = p

        improved = False
        chosen_alpha = 0.0
        for alpha in _LS_ALPHAS:
            if time.monotonic() - t0 > time_limit:
                break
            mix = linear_mix(a, best_root, alpha)
            cm = autoconv_normalized(mix, P)
            sm = score_from_conv(cm, n)
            lm = float(np.dot(cm, cm))
            ok = False
            if sm < best_score - 1e-12:
                ok = True
            elif sm < score - 1e-13 and sm < best_score + 1e-8 and lm < best_l2 - 1e-15:
                ok = True
            if ok:
                a = mix
                score = sm
                best_l2 = lm if lm < best_l2 else best_l2
                improved = True
                chosen_alpha = alpha
                break

        if improved:
            no_improve = 0
            if chosen_alpha < 0.25:
                delta = max(0.005, delta * 0.92)
            else:
                delta = max(0.005, delta * 0.98)
        else:
            no_improve += 1
            delta = min(0.85, delta * 1.5)
            if no_improve >= 18:
                z = np.log(np.maximum(a, 1e-12))
                z -= float(z.mean())
                z += rng.normal(0.0, 0.03, n)
                a = softmax_from_z(z)
                delta = 0.25
                no_improve = 0

        if it % 12 == 0:
            rho = 0.08 if it < 100 else 0.03
            a_avg = (1.0 - rho) * avg + rho * a
            a_avg = normalize_nonneg(a_avg)
            ca = autoconv_normalized(a_avg, P)
            sa = score_from_conv(ca, n)
            if sa < best_score:
                best = a_avg.copy()
                best_score = sa
                best_l2 = float(np.dot(ca, ca))
            avg = a_avg

        if verbose and it % 40 == 0:
            print(
                f"  sqrt_clip it={it}, delta={delta:.4f}, score={score:.8f}, "
                f"best={best_score:.8f}"
            )

    return normalize_nonneg(best)


def generic_log_descent(a0, time_limit=120.0, mode="l2", lr=0.25, momentum=0.9,
                        K_top=20, T_start=0.25, T_end=0.03, verbose=True, seed=777):
    """
    Mirror/log-space descent with several gradient weights:
      mode='l2'   : minimize sum_k (a*a)_k^2  (flattening objective)
      mode='soft' : minimize soft-max of a*a
      mode='top'  : minimize average of top peaks
    """
    if time_limit <= 1e-3:
        return normalize_nonneg(a0)

    rng = np.random.default_rng(seed)
    a = normalize_nonneg(a0)
    n = a.size
    if n == 0:
        return a

    P = next_pow2_at_least(3 * n - 2)
    c = autoconv_normalized(a, P)
    best = a.copy()
    best_score = score_from_conv(c, n)
    best_l2 = float(np.dot(c, c))

    avg = a.copy()
    z = np.log(np.maximum(a, 1e-14))
    z -= float(z.mean())
    m = np.zeros(n, dtype=np.float64)
    step = float(lr)
    no_improve = 0
    it = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        it += 1
        a = softmax_from_z(z)
        A = rfft(a, P)
        c = irfft(A * A, P)[:2 * n - 1]
        c = np.maximum(c, 0.0)
        cs = float(c.sum())
        if cs > 0.0 and np.isfinite(cs):
            c = c / cs

        M = float(c.max())
        score = score_from_conv(c, n)
        l2 = float(np.dot(c, c))

        if score < best_score:
            best_score = score
            best = a.copy()
            best_l2 = l2

        if mode == "l2":
            w = c
        elif mode == "soft":
            frac = T_start + (T_end - T_start) * min(1.0, float(it) / 2000.0)
            T = max(1e-9, M * frac)
            w = stable_softmax_weights(c, T)
        else:
            w = top_weights(c, K_top)

        e = corr_a_w(a, w, A=A, P=P)
        d = e - float(np.dot(a, e))
        d -= float(np.mean(d))
        nrm = float(np.linalg.norm(d))

        if not np.isfinite(nrm) or nrm < 1e-12:
            z += rng.normal(0.0, 0.03, n)
            m = np.zeros(n, dtype=np.float64)
            step = float(lr)
            no_improve += 1
            continue

        d = d / nrm
        m = momentum * m + (1.0 - momentum) * d
        mn = float(np.linalg.norm(m))
        dirv = m / mn if mn > 1e-12 else d

        improved = False
        for factor in (1.0, 0.5, 0.25, 0.1):
            if time.monotonic() - t0 > time_limit:
                break
            zt = z - step * factor * dirv
            at = softmax_from_z(zt)
            At = rfft(at, P)
            ct = irfft(At * At, P)[:2 * n - 1]
            ct = np.maximum(ct, 0.0)
            cts = float(ct.sum())
            if cts > 0.0 and np.isfinite(cts):
                ct = ct / cts

            st = score_from_conv(ct, n)
            lt = float(np.dot(ct, ct))

            ok = False
            if st < best_score - 1e-12:
                ok = True
            elif mode == "l2" and lt < l2 - 1e-15 and st <= best_score + 1e-8:
                ok = True
            elif mode != "l2" and st < score - 1e-13:
                ok = True

            if ok:
                z = zt
                a = at
                score = st
                l2 = lt
                improved = True
                no_improve = 0
                step = min(2.0, step * 1.04)
                break

        if not improved:
            step = max(1e-4, step * 0.8)
            no_improve += 1

        if it % 25 == 0:
            rho = 0.06
            a_avg = (1.0 - rho) * avg + rho * a
            a_avg = normalize_nonneg(a_avg)
            ca = autoconv_normalized(a_avg, P)
            sa = score_from_conv(ca, n)
            if sa < best_score:
                best = a_avg.copy()
                best_score = sa
                best_l2 = float(np.dot(ca, ca))
            avg = a_avg

        if no_improve > 320:
            base = best if rng.random() < 0.55 else avg
            z = np.log(np.maximum(base, 1e-14))
            z -= float(z.mean())
            z += rng.normal(0.0, 0.025, n)
            m = np.zeros(n, dtype=np.float64)
            step = float(lr)
            no_improve = 0

        if verbose and it % 80 == 0:
            print(f"  log_{mode} it={it}, step={step:.4f}, score={score:.8f}, best={best_score:.8f}")

    return normalize_nonneg(best)


def active_lp_polish(a0, time_limit=300.0, K_start=30, K_end=130,
                     U_start=0.02, U_end=0.0004, verbose=True, seed=2024):
    """
    Active-set linearized LP descent for max(autoconv).
    """
    if time_limit <= 1e-3:
        return normalize_nonneg(a0)
    if linprog is None:
        return normalize_nonneg(a0)

    rng = np.random.default_rng(seed)
    a = normalize_nonneg(a0)
    n = a.size
    if n == 0:
        return a

    P = next_pow2_at_least(2 * n - 1)
    c = autoconv_normalized(a, P)
    best = a.copy()
    best_score = score_from_conv(c, n)
    best_l2 = float(np.dot(c, c))

    avg = a.copy()
    U = float(U_start)
    no_improve = 0
    it = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2.0:
            break

        c = autoconv_normalized(a, P)
        M = float(c.max())
        score = score_from_conv(c, n)
        l2 = float(np.dot(c, c))

        if score < best_score:
            best_score = score
            best = a.copy()
            best_l2 = l2

        frac = min(1.0, float(it) / 1200.0)
        K = int(max(K_start, min(K_end, K_start + (K_end - K_start) * frac)))

        threshold = 0.88 * M
        cnt = int(np.sum(c >= threshold))
        Kuse = max(min(K, cnt), min(K_start, c.size))
        ks = np.argsort(c)[::-1][:Kuse]

        U = max(float(U_end), float(U_start) * (0.9955 ** it))
        if no_improve > 35:
            U = max(float(U_end), min(float(U_start), U * 0.7))

        delta = solve_active_lp(a, c, ks, U)
        improved = False
        chosen_alpha = 0.0

        if delta is not None and np.linalg.norm(delta) > 1e-13:
            for alpha in _LS_ALPHAS:
                if time.monotonic() - t0 > time_limit:
                    break
                at = a + alpha * delta
                if np.any(at < -1e-10):
                    continue
                at = np.maximum(at, 0.0)
                s = float(at.sum())
                if s <= 0.0:
                    continue
                at = at / s
                ct = autoconv_normalized(at, P)
                st = score_from_conv(ct, n)
                lt = float(np.dot(ct, ct))

                ok = False
                if st < best_score - 1e-12:
                    ok = True
                elif st < score - 1e-12 and st < best_score + 1e-8 and lt < best_l2 - 1e-15:
                    ok = True

                if ok:
                    a = at
                    score = st
                    l2 = lt
                    improved = True
                    chosen_alpha = alpha
                    break

        if improved:
            no_improve = 0
            if chosen_alpha < 0.25:
                U = max(float(U_end), min(float(U_start) * 2.5, U * 1.15))
            else:
                U = max(float(U_end), U * 1.02)
        else:
            no_improve += 1
            if no_improve > 8:
                U = max(float(U_end), U * 0.82)
            if no_improve > 55:
                z = np.log(np.maximum(a, 1e-14))
                z -= float(z.mean())
                z += rng.normal(0.0, 0.02, n)
                a = softmax_from_z(z)
                U = max(float(U_end), float(U_start) * 0.45)
                no_improve = 0

        if it % 20 == 0:
            rho = 0.05
            a_avg = (1.0 - rho) * avg + rho * a
            a_avg = normalize_nonneg(a_avg)
            ca = autoconv_normalized(a_avg, P)
            sa = score_from_conv(ca, n)
            if sa < best_score:
                best = a_avg.copy()
                best_score = sa
                best_l2 = float(np.dot(ca, ca))
            avg = a_avg

        if verbose and it % 35 == 0:
            print(
                f"  active_lp it={it}, K={Kuse}, U={U:.5f}, score={score:.8f}, "
                f"best={best_score:.8f}"
            )

    return normalize_nonneg(best)


def lp_fictitious_play(a0, time_limit=120.0, verbose=True, seed=999):
    """Greedy LP fictitious play with exact line search."""
    if time_limit <= 1e-3:
        return normalize_nonneg(a0)

    rng = np.random.default_rng(seed)
    f = normalize_nonneg(a0)
    n = f.size
    if n == 0:
        return f

    P = next_pow2_at_least(2 * n - 1)
    best = f.copy()
    cf = autoconv_normalized(f, P)
    best_score = score_from_conv(cf, n)
    avg = f.copy()

    it = 0
    no_improve = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5.0:
            break

        cf = autoconv_normalized(f, P)
        M = float(cf.max())
        score = score_from_conv(cf, n)

        if score < best_score:
            best_score = score
            best = f.copy()

        if M <= 1e-300:
            break

        order = rng.permutation(n)
        g = greedy_lp(f, M, order)
        gs = float(g.sum())
        if gs <= 1e-15:
            break
        g = g / gs

        cur_best = f
        cur_best_score = score
        best_alpha = 0.0

        for alpha in (0.5, 0.25, 0.15, 0.75, 0.35, 0.1):
            mix = linear_mix(f, g, alpha)
            cm = autoconv_normalized(mix, P)
            sm = score_from_conv(cm, n)
            if sm < cur_best_score - 1e-13:
                cur_best = mix
                cur_best_score = sm
                best_alpha = alpha

        if cur_best_score < score - 1e-13:
            f = cur_best
            score = cur_best_score
            no_improve = 0
            if score < best_score:
                best_score = score
                best = f.copy()
        else:
            no_improve += 1
            if no_improve > 25:
                z = np.log(np.maximum(f, 1e-14))
                z -= float(z.mean())
                z += rng.normal(0.0, 0.02, n)
                f = softmax_from_z(z)
                no_improve = 0

        if it % 10 == 0:
            rho = 0.07
            avg = normalize_nonneg((1.0 - rho) * avg + rho * f)
            ca = autoconv_normalized(avg, P)
            sa = score_from_conv(ca, n)
            if sa < best_score:
                best_score = sa
                best = avg.copy()

        if verbose and it % 25 == 0:
            print(f"  greedy_fp it={it}, alpha={best_alpha:.2f}, score={score:.8f}, best={best_score:.8f}")

    return normalize_nonneg(best)


def max_pair_val(base, D, qidx, t):
    val = base + t * D
    val[qidx] += -2.0 * t * t
    return float(np.max(val))


def sample_pairs(n, ks, rng, n_random=300, n_peak=20, n_edge=100):
    pairs = set()
    cap = 2500

    def add_pair(i, j):
        if 0 <= i < n and 0 <= j < n and i != j:
            if i > j:
                i, j = j, i
            pairs.add((i, j))

    for _ in range(n_random):
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        add_pair(i, j)
        if len(pairs) >= cap:
            return list(pairs)

    for k in ks:
        k = int(k)
        i0 = max(0, k - n + 1)
        i1 = min(n - 1, k)
        if i1 - i0 + 1 <= 1:
            continue

        for _ in range(n_peak * 2):
            i = int(rng.integers(i0, i1 + 1))
            j = k - i
            add_pair(i, j)
            if len(pairs) >= cap:
                return list(pairs)

        for _ in range(n_peak):
            i = int(rng.integers(i0, i1 + 1))
            j = int(rng.integers(i0, i1 + 1))
            add_pair(i, j)
            if len(pairs) >= cap:
                return list(pairs)

    edge = max(2, n // 20)
    for _ in range(n_edge):
        i = int(rng.integers(0, edge))
        j = int(rng.integers(n - edge, n))
        add_pair(i, j)

    return list(pairs)


def pair_cd_polish(a0, time_limit=200.0, verbose=True, seed=1010):
    """
    Exact two-coordinate transfer descent.
    Uses O(n) exact updates of the full auto-convolution during line search.
    """
    if time_limit <= 1e-3:
        return normalize_nonneg(a0)

    rng = np.random.default_rng(seed)
    a = normalize_nonneg(a0)
    n = a.size
    if n == 0:
        return a

    P = next_pow2_at_least(2 * n - 1)
    base = autoconv_normalized(a, P)
    best = a.copy()
    best_score = score_from_conv(base, n)

    sweeps = 0
    no_improve_sweeps = 0
    update_count = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 8.0:
            break

        ks = np.argsort(base)[::-1][:20]
        pairs = sample_pairs(n, ks, rng)
        if not pairs:
            break

        improved_this_sweep = False

        for (i, j) in pairs:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-15 or a[j] < 1e-15:
                continue

            t_lo = -float(a[j]) + 1e-15
            t_hi = float(a[i]) - 1e-15
            width = t_hi - t_lo
            if width < 1e-12:
                continue

            D = np.zeros(2 * n - 1, dtype=np.float64)
            D[i:i + n] -= 2.0 * a
            D[j:j + n] += 2.0 * a
            qidx = i + j

            x1 = t_hi - _GOLDEN * width
            x2 = t_lo + _GOLDEN * width
            f1 = max_pair_val(base, D, qidx, x1)
            f2 = max_pair_val(base, D, qidx, x2)

            for _ in range(22):
                if f1 < f2:
                    t_hi = x2
                    x2 = x1
                    f2 = f1
                    width = t_hi - t_lo
                    x1 = t_hi - _GOLDEN * width
                    f1 = max_pair_val(base, D, qidx, x1)
                else:
                    t_lo = x1
                    x1 = x2
                    f1 = f2
                    width = t_hi - t_lo
                    x2 = t_lo + _GOLDEN * width
                    f2 = max_pair_val(base, D, qidx, x2)

            t_opt = 0.5 * (t_lo + t_hi)
            f_opt = min(f1, f2, current_max if False else 0.0) # placeholder, replaced below
            current_max = float(np.max(base))
            f_opt = min(f1, f2, current_max)

            if 2.0 * float(n) * f_opt < best_score - 1e-12:
                # apply exact update
                a[i] -= t_opt
                a[j] += t_opt
                base = base + t_opt * D
                base[qidx] += -2.0 * t_opt * t_opt
                base = np.maximum(base, 0.0)

                # Safety: if numeric drift is noticeable, recompute.
                s = float(a.sum())
                if a[i] < -1e-10 or a[j] < -1e-10 or not np.isfinite(s) or abs(s - 1.0) > 1e-8:
                    a = normalize_nonneg(a)
                    base = autoconv_normalized(a, P)
                else:
                    a = np.maximum(a, 0.0)
                    # Translate by a tiny amount if sum drifted a lot.
                    sa = float(a.sum())
                    if sa > 0.0 and abs(sa - 1.0) > 1e-12:
                        a = a / sa
                        base = autoconv_normalized(a, P)

                sc = score_from_conv(base, n)
                if sc < best_score - 1e-13:
                    best = a.copy()
                    best_score = sc
                    improved_this_sweep = True
                    update_count += 1
                    if update_count % 35 == 0:
                        base = autoconv_normalized(a, P)

        if not improved_this_sweep:
            no_improve_sweeps += 1
            if no_improve_sweeps >= 3:
                z = np.log(np.maximum(a, 1e-14))
                z -= float(z.mean())
                z += rng.normal(0.0, 0.02, n)
                a = softmax_from_z(z)
                base = autoconv_normalized(a, P)
                no_improve_sweeps = 0
        else:
            no_improve_sweeps = 0

        if verbose and sweeps % 10 == 0:
            print(f"  pair_cd sweep={sweeps}, score={score_from_conv(autoconv_normalized(a, P), n):.8f}, best={best_score:.8f}")

    return normalize_nonneg(best)


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main entry point. Runs within budget_s seconds and returns the best list
    of non-negative floats found.
    """
    t_start = time.monotonic()
    # Leave a safety margin so the outer harness can collect the return value.
    deadline = t_start + max(1.0, float(budget_s) - 45.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")

    # Load parent if available.
    parent = None
    try:
        p = height_sequence_1
        p = np.asarray(p, dtype=np.float64).ravel()
        p = p[np.isfinite(p)]
        p = np.clip(p, 0.0, 1000.0)
        if p.size > 0 and p.sum() > 1e-10:
            parent = p / p.sum()
            best_seq, best_score = consider_candidate(
                best_seq, best_score, parent, "parent", exact=False
            )
            # Make the bookkeeping score an honest exact score for the parent.
            try:
                ep = float(evaluate_sequence(parent.tolist()))
                if np.isfinite(ep):
                    best_score = min(best_score, ep)
            except Exception:
                pass
            print(f"Parent: n={parent.size}, internal/exact~{best_score:.8f}")
    except Exception as e:
        print("Could not load parent:", e)

    # Candidate lengths to explore.
    if parent is not None:
        n_parent = int(parent.size)
        lengths = [
            n_parent,
            max(64, n_parent - 2),
            n_parent + 1,
            n_parent + 2,
            n_parent + 4,
            min(n_parent + 128, 1536),
            1024,
            1280,
        ]
    else:
        parent = normalize_nonneg(np.ones(768) / 768)
        n_parent = 768
        lengths = [768, 1024]

    cleaned_lengths = []
    for x in lengths:
        x = int(x)
        if 32 <= x <= 2200 and x not in cleaned_lengths:
            cleaned_lengths.append(x)
    if not cleaned_lengths:
        cleaned_lengths = [768]

    candidates = []
    for n in cleaned_lengths:
        seqs = []
        if parent is not None:
            seqs.append(interpolate_sequence(parent, n))
        seqs.append(sqrt_flat_prefix(n))
        seqs.append(make_random_smooth(n, rng))

        for s in seqs:
            s = normalize_nonneg(s)
            if s.size != n:
                continue
            c = autoconv_normalized(s)
            sc = score_from_conv(c, n)
            if np.isfinite(sc):
                candidates.append((sc, s, n))

    if not candidates:
        best_seq = normalize_nonneg(np.ones(768) / 768)
        best_score = internal_score(best_seq)
        print("No valid candidates; using uniform fallback.")
        return [float(x) for x in best_seq]

    # Keep the best start per length, then sort by score.
    best_by_len = {}
    for sc, s, n in candidates:
        if n not in best_by_len or sc < best_by_len[n][0]:
            best_by_len[n] = (sc, s)

    top_items = [(sc, s, n) for n, (sc, s) in best_by_len.items()]
    top_items.sort(key=itemgetter(0))
    print("Top initial candidates by internal score:")
    for sc, s, n in top_items[:6]:
        print(f"  n={n}, score={sc:.8f}")

    # Quick multi-start polishing to avoid a bad basin.
    for idx in range(min(3, len(top_items))):
        rem = deadline - time.monotonic()
        if rem < 140:
            break
        sc, s, n = top_items[idx]
        print(f"\n[Quick polish {idx}] n={n}, init score={sc:.8f}, rem={rem:.0f}s")

        t1 = min(80.0, rem * 0.30)
        if t1 > 20.0:
            seq = sqrt_clip_polish(s, t1, verbose=False, seed=1000 + idx)
            best_seq, best_score = consider_candidate(best_seq, best_score, seq, f"quick_sqrt_{n}")

        rem = deadline - time.monotonic()
        if rem > 80:
            t2 = min(60.0, rem * 0.25)
            if t2 > 15.0:
                seq = active_lp_polish(seq if "seq" in locals() else s, t2,
                                       K_start=25, K_end=80, U_start=0.015,
                                       verbose=False, seed=2000 + idx)
                best_seq, best_score = consider_candidate(best_seq, best_score, seq, f"quick_lp_{n}")

        rem = deadline - time.monotonic()
        if rem > 60:
            t3 = min(45.0, rem * 0.25)
            if t3 > 10.0:
                seq = generic_log_descent(best_seq if best_seq is not None else s, t3,
                                          mode="l2", verbose=False, seed=3000 + idx)
                best_seq, best_score = consider_candidate(best_seq, best_score, seq, f"quick_l2_{n}")

    if best_seq is None:
        best_seq, _ = top_items[0][1], top_items[0][1]

    print(f"\n=== Main phase 1: active LP (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 120:
        t1 = min(520.0, rem * 0.33)
        seq = active_lp_polish(best_seq, t1, K_start=35, K_end=140,
                               U_start=0.02, U_end=0.0004, verbose=True, seed=11)
        best_seq, best_score = consider_candidate(best_seq, best_score, seq, "main_active_lp", exact=True)

    print(f"\n=== Main phase 2: sqrt-clip reconstruction (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 90:
        t2 = min(330.0, rem * 0.33)
        seq = sqrt_clip_polish(best_seq, t2, verbose=True, seed=22)
        best_seq, best_score = consider_candidate(best_seq, best_score, seq, "main_sqrt", exact=True)

    print(f"\n=== Main phase 3: L2 log descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t3 = min(240.0, rem * 0.33)
        seq = generic_log_descent(best_seq, t3, mode="l2", lr=0.22,
                                  momentum=0.9, verbose=True, seed=33)
        best_seq, best_score = consider_candidate(best_seq, best_score, seq, "main_l2", exact=True)

    print(f"\n=== Main phase 4: fine active LP (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t4 = min(240.0, rem * 0.33)
        seq = active_lp_polish(best_seq, t4, K_start=70, K_end=180,
                               U_start=0.006, U_end=0.0002, verbose=True, seed=44)
        best_seq, best_score = consider_candidate(best_seq, best_score, seq, "fine_active_lp", exact=True)

    print(f"\n=== Main phase 5: greedy fictitious play (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 70:
        t5 = min(140.0, rem * 0.35)
        seq = lp_fictitious_play(best_seq, t5, verbose=True, seed=55)
        best_seq, best_score = consider_candidate(best_seq, best_score, seq, "greedy_fp", exact=True)

    print(f"\n=== Main phase 6: soft/top peak log descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t6a = min(90.0, rem * 0.5)
        if t6a > 15.0:
            seq = generic_log_descent(best_seq, t6a, mode="soft",
                                      T_start=0.20, T_end=0.03, verbose=False, seed=66)
            best_seq, best_score = consider_candidate(best_seq, best_score, seq, "soft", exact=True)

        rem = deadline - time.monotonic()
        if rem > 40:
            t6b = min(70.0, rem * 0.6)
            if t6b > 10.0:
                seq = generic_log_descent(best_seq, t6b, mode="top", K_top=18,
                                          verbose=False, seed=77)
                best_seq, best_score = consider_candidate(best_seq, best_score, seq, "top", exact=True)

    print(f"\n=== Main phase 7: pair coordinate polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t7 = min(350.0, rem * 0.75)
        seq = pair_cd_polish(best_seq, t7, verbose=True, seed=88)
        best_seq, best_score = consider_candidate(best_seq, best_score, seq, "pair_cd", exact=True)

    # Short exploration of nearby lengths, if time remains.
    print(f"\n=== Extra length exploration (best={best_score:.8f}) ===")
    n_best = int(best_seq.size)
    nearby = [n_best - 2, n_best - 1, n_best + 1, n_best + 2, n_best + 4, n_best - 4, n_best + 6]
    nearby = [x for x in nearby if 32 <= x <= 2200]
    seen = {n_best}
    for x in nearby:
        if x not in seen:
            seen.add(x)

    for n in nearby:
        rem = deadline - time.monotonic()
        if rem < 110:
            break
        print(f"  Trying n={n}, rem={rem:.0f}s")
        seq0 = interpolate_sequence(best_seq, n)
        t1 = min(70.0, rem * 0.40)
        if t1 <= 10.0:
            break
        seq1 = sqrt_clip_polish(seq0, t1, verbose=False, seed=9000 + n)
        best_seq, best_score = consider_candidate(best_seq, best_score, seq1, f"len{n}_sqrt", exact=False)

        rem = deadline - time.monotonic()
        if rem > 50:
            t2 = min(45.0, rem * 0.45)
            if t2 > 10.0:
                seq2 = active_lp_polish(best_seq, t2, K_start=40, K_end=100,
                                        U_start=0.008, U_end=0.0003, verbose=False, seed=9100 + n)
                best_seq, best_score = consider_candidate(best_seq, best_score, seq2, f"len{n}_lp", exact=False)

    # Final small polish.
    rem = deadline - time.monotonic()
    if rem > 45:
        tfin = min(60.0, rem * 0.65)
        seq = generic_log_descent(best_seq, tfin, mode="l2", lr=0.15,
                                  momentum=0.85, verbose=False, seed=1234)
        best_seq, best_score = consider_candidate(best_seq, best_score, seq, "final_l2", exact=False)

    rem = deadline - time.monotonic()
    if rem > 40:
        tfin2 = min(45.0, rem * 0.65)
        seq = pair_cd_polish(best_seq, tfin2, verbose=False, seed=4321)
        best_seq, best_score = consider_candidate(best_seq, best_score, seq, "final_pair", exact=False)

    # Final exact verification.
    best_seq = normalize_nonneg(best_seq)
    if best_seq.size == 0:
        best_seq = normalize_nonneg(np.ones(768) / 768)

    try:
        final_exact = float(evaluate_sequence(best_seq.tolist()))
    except Exception:
        final_exact = internal_score(best_seq)

    print(f"\nFINAL INTERNAL SCORE: {best_score:.8f}")
    print(f"FINAL EXACT SCORE:    {final_exact:.8f} (n={best_seq.size})")

    # If exact is finite and slightly better than tracked score, trust it.
    if np.isfinite(final_exact) and final_exact < best_score:
        best_score = final_exact

    return [float(x) for x in best_seq]
```
