def subgradient_nesterov(a0, time_limit, n_max_iters=2000000, top_k_frac=0.05):

    t0 = time.monotonic()

    # Nesterov momentum
    a_prev = a.copy()
    a_curr = a.copy()
    beta = 1.0  # momentum coefficient (1 - 1/sqrt(t+1))

    # Adaptive step size
    alpha = 0.02
    no_improve = 0
    top_k = max(3, int(n * top_k_frac))

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if time.monotonic() - t0 > time_limit - 2:
            break

        # Nesterov extrapolation
        beta = (it) / (it + 1) if it > 0 else 1.0
        y = a_curr + beta * (a_curr - a_prev)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a_curr.copy()
            sy = 1.0
        y = y / sy

        # Compute convolution and find top peaks
        b = fast_convolve(y)
        max_b = b.max()

        # Select top-K peaks for subgradient
        if len(b) > top_k:
            threshold_idx = np.argpartition(b, -top_k)[-top_k]
            threshold = b[threshold_idx]
            active = np.where(b >= threshold)[0]
        else:
            active = np.arange(len(b))

        # Compute subgradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
             je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += y[k - idx]
        g /= len(active)

        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm

        # Update with projection onto simplex
        step = alpha * g_norm  # scale step by gradient magnitude
        a_new = y - step * g
        # Project onto simplex {a >= 0, sum(a) = 1}
        a_new = simplex_project(a_new)

        score_new = 2.0 * n * fast_convolve(a_new).max()

        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            no_improve = 0
            # Slightly increase step if improving

        else:
            no_improve += 1
            if no_improve >= 500:
                alpha *= 0.7
                no_improve = 0
            elif no_improve >= 200:
                alpha *= 0.9
                no_improve = 0

        # Decay alpha slowly
        alpha = max(alpha / (1.0 + it * 0.00001), 0.003)

        a_prev = a_curr
        a_curr = a_new

        if (it + 1) % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad iter={it+1} ({elapsed:.0f}s): score={best_score:.6f} alpha={alpha:.6f}")

    return best_a, best_score


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    proj = np.maximum(v - theta, 0)
    return proj


def lbgfs_refine(a0, time_limit, T_start=50.0, T_end=0.1, maxiter=15000):
    """L-BFGS with LogSumExp smoothing - optimized for fine-tuning."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def make_func(T_val):
        def func(z_inner):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a_inner = softmax_from_z(z_inner)
            b_inner = fast_convolve(a_inner)
            Tb = T_val * b_inner
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z_inner):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_inner)
            a_inner = softmax_from_z(z_inner)
            b_inner = fast_convolve(a_inner)
            Tb = T_val * b_inner
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b_inner)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_inner[k - idx]
            S = np.dot(grad_a, a_inner)
            return a_inner * (grad_a - S)
        return jac

    n_stages = 6
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(100, maxiter // n_stages)

    for stage_idx, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
        elapsed = time.monotonic() - t0
        sc = 2.0 * n * fast_convolve(softmax_from_z(z)).max()
        print(f"    L-BFGS stage {stage_idx+1} T={stage_T:.1f} ({elapsed:.0f}s): score={sc:.6f}")

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        try_update(a_sg, "subgrad_nesterov")

    # === Phase 3: L-BFGS fine-tuning (low temperature) ===
    print(f"\n=== Phase 3: L-BFGS polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s, T: 30->0.05")
        a_lb = lbgfs_refine(best_seq, t_alloc, T_start=30.0, T_end=0.05, maxiter=20000)
        try_update(a_lb, "lbgfs_main")

    # === Phase 4: Second Nesterov LP round on refined solution ===
    print(f"\n=== Phase 4: Nesterov LP r2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=100000)
        try_update(a_lp2, "nesterov_lp_2")

    # === Phase 5: Quick subgradient + L-BFGS cycle ===
    print(f"\n=== Phase 5: Quick cycle (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_sg = min(rem * 0.2, 80.0)
        t_lb = min(rem * 0.2, 80.0)
        if t_sg > 10:
            a_sg2, _ = subgradient_nesterov(best_seq, t_sg, n_max_iters=2000000, top_k_frac=0.12)
            try_update(a_sg2, "subgrad_2")
        rem = deadline - time.monotonic()
        if rem > 20 and t_lb > 10:
            a_lb2 = lbgfs_refine(best_seq, min(rem * 0.3, 60.0), T_start=10.0, T_end=0.02, maxiter=8000)
            try_update(a_lb2, "lbgfs_2")

    # === Phase 6: Try alternative length ===
    print(f"\n=== Phase 6: Alternative length (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 120:
        alt_n = int(n_best * 1.25)  # 1200 for n=960
        alt_n = min(alt_n, 2048)
        init_alt = interpolate_sequence(best_seq, alt_n)
        t_alt = min(rem * 0.35, 250.0)
        print(f"  Trying n={alt_n}, budget={t_alt:.0f}s")

        # Quick LP refinement
        t_lp_alt = t_alt * 0.55
        a_alt, _ = nesterov_lp_refine(init_alt, t_lp_alt, solver=solver, n_max_iters=50000)
        try_update(a_alt, f"lp_n{alt_n}")

        rem = deadline - time.monotonic()
        if rem > 30 and best_seq is not None and len(best_seq) == alt_n:
            t_lb_alt = min(rem * 0.3, 80.0)
            if t_lb_alt > 15:
                a_alt2 = lbgfs_refine(best_seq, t_lb_alt, T_start=20.0, T_end=0.1, maxiter=5000)
                try_update(a_alt2, f"lbgfs_n{alt_n}")

    # === Phase 7: Final polish on best found ===
    print(f"\n=== Phase 7: Final polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.5, 60.0)
        if t_fin > 10:
            a_fin = lbgfs_refine(best_seq, t_fin, T_start=5.0, T_end=0.01, maxiter=5000)
            try_update(a_fin, "final_lbgfs")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
