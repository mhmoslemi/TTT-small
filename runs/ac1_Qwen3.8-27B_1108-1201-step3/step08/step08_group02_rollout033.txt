    return irfft(A * A, fft_size)[:size]

def compute_score(f):
    s = f.sum()
    if s < 1e-10:
        return np.inf
    b = fast_convolve(f)
    return 2.0 * len(f) * b.max() / (s * s)

def project_simplex(v, target=1.0):
    n = len(v)
    if n == 1:
        return np.array([max(0.0, float(v[0]))])
    v = v.astype(np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - target)
    if not cond.any():
        return np.full(n, target / n)
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - target) / (idx + 1)
    return np.maximum(v - theta, 0.0)

def lp_direction(f, peaks, M):
    """Solve max sum(g) s.t. g>=0, (f*g)(k)<=M for k in peak positions."""
    n = len(f)
    K = len(peaks)
    if K == 0:
        return None
    
    # Build sparse constraint matrix
    rows_list = []
    cols_list = []
    vals_list = []
    
    for idx, k in enumerate(peaks):
        k = int(k)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows_list.append(idx)
            cols_list.append(j)
            vals_list.append(f[k - j])
    
    A = csr_matrix((vals_list, (rows_list, cols_list)), shape=(K, n))
    
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Maximize(cp.sum(g))
    prob = cp.Problem(prob, constraints)
    
    for solver in [cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None:
                gv = np.maximum(g.value, 0.0)
                if not np.any(np.isnan(gv)):
                    return gv
        except Exception:
            continue
    return None

def line_search_golden(f, h, max_iter=50):
    """Golden section search for best t in [0,1] minimizing max(conv(f_t, f_t))"""
    s = f.sum()
    phi = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    a, b = 0.0, 1.0
    
    def obj(t):
        ft = (1 - t) * f + t * h
        ft = np.maximum(ft, 0)
        ss = ft.sum()
        if ss < 1e-15:
            return np.inf
        ft = ft / ss
        return fast_convolve(ft).max()
    
    c = b - phi * (b - a)
    d = a + phi * (b - a)
    fc = obj(c)
    fd = obj(d)
    
    for _ in range(max_iter):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - phi * (b - a)
            fc = obj(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + phi * (b - a)
            fd = obj(d)
    
    t = (a + b) / 2
    return t, obj(t)

def lp_optimize(f_init, time_limit, initial_frac=0.95, final_frac=0.80,
                max_peaks=400, patience=30, label="lp", seed=42):
    """Main LP-based optimization loop."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f = f / f.sum()
    best_f = f.copy()
    best_M = fast_convolve(f).max()
    score = compute_score(f)
    print(f"  [{label}] start: score={score:.6f}, M={best_M:.10f}")
    
    it = 0
    stuck = 0
    peak_frac = initial_frac
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()
        
        # Adaptive peak selection
        threshold = peak_frac * M
        peaks = np.where(b >= threshold)[0]
        if len(peaks) > max_peaks:
            peaks = np.argpartition(b, -max_peaks)[-max_peaks:]
        
        g = lp_direction(f, peaks, M)
        if g is None:
            stuck += 1
            if stuck > patience:
                break
            continue
        
        Sg = g.sum()
        if Sg <= 1.0 + 1e-6:
            # No improvement possible with this peak set
            stuck += 1
            if stuck > patience:
                # Try wider peak set
                new_frac = max(final_frac, peak_frac - 0.05)
                if new_frac < peak_frac - 1e-6:
                    peak_frac = new_frac
                    stuck = 0
                else:
                    # Perturb
                    f = f + rng.standard_normal(n) * 0.001
                    f = np.maximum(f, 0)
                    f /= f.sum()
                    stuck = 0
            continue
        
        # Normalize direction
        h = g / Sg
        
        # Line search
        t_best, obj_best = line_search_golden(f, h, max_iter=40)
        
        if t_best < 1e-6 or obj_best >= M - 1e-14:
            stuck += 1
            if stuck > patience:
                break
            continue
        
        # Update
        f_new = (1 - t_best) * f + t_best * h
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-10:
            break
        f = f_new / s
        
        cur_M = fast_convolve(f).max()
        if cur_M < best_M - 1e-14:
            best_M = cur_M
            best_f = f.copy()
            stuck = 0
            score = 2.0 * n * best_M  # since sum=1
            if it % 20 == 0:
                print(f"  [{label}] it={it} score={score:.6f} M={best_M:.10f} "
                      f"t={t_best:.3f} Sg={Sg:.4f} peaks={len(peaks)} "
                      f"time={time.monotonic()-t0:.0f}s")
        else:
            stuck += 1
        
        # Gradually widen peak set
        if it % 25 == 0:
            peak_frac = max(final_frac, peak_frac - 0.003)
        
        # Perturbation when very stuck
        if stuck > patience // 2:
            f = f * (1 + rng.standard_normal(n) * 0.002)
            f = np.maximum(f, 0)
            f /= f.sum()
            stuck = max(0, stuck - patience // 4)
    
    return best_f, best_M

def coordinate_descent(f_init, time_limit, label="cd"):
    """Cyclic coordinate descent with multiplicative steps."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_M = fast_convolve(f).max()
    best_f = f.copy()
    multipliers = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0]
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-18:
                continue
            best_val = f[j]
            best_obj = best_M
            for m in multipliers:
                new_val = f[j] * m
                f[j] = new_val
                s = f.sum()
                if s < 1e-10:
                    f[j] = best_val
                    continue
                f_norm = f / s
                obj = fast_convolve(f_norm).max()
                if obj < best_obj - 1e-15:
                    best_obj = obj
                    best_val = new_val
            f[j] = best_val
            if best_val != f[j]:
                f[j] = best_val
                s = f.sum()
                f /= s
                if best_obj < best_M - 1e-15:
                    best_M = best_obj
                    best_f = f.copy()
                    improved = True
        if not improved:
            break
    
    return best_f, best_M

LP-based optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines adaptive LP direction finding, golden-section line search,
mirror descent polishing, coordinate descent, and multi-length search.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score_normalized(f):
    """Compute score assuming sum(f)=1: 2*n*max(conv(f,f))."""
    M = fast_convolve(f).max()
    return 2.0 * len(f) * M


def project_simplex(v, target=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, float(v[0]))])
    v = np.asarray(v, dtype=np.float64).copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - target)
    if not np.any(cond):
        return np.full(n, target / n)
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - target) / (idx + 1)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f, peaks):
    """Build the constraint matrix A where A[i,j] gives coefficient of g[j] in (f*g)(peaks[i])."""
    n = len(f)
    K = len(peaks)
    A = np.zeros((K, n), dtype=np.float64)
    for i in range(K):
        k = int(peaks[i])
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        length = j_hi - j_lo + 1
        # A[i, j_lo:j_hi+1] = f[k - j_hi : k - j_lo + 1][::-1]
        A[i, j_lo:j_hi + 1] = f[k - j_hi:k - j_lo + 1][::-1]
    return A


def lp_solve(f, peaks, M):
    """
    Solve: max sum(g) s.t. g >= 0, (f*g)(k) <= M for k in peaks.
    Returns g or None.
    """
    n = len(f)
    K = len(peaks)
    if K == 0 or M <= 0:
        return None
    
    A = build_lp_matrix(f, peaks)
    
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    
    best_g = None
    best_obj = -np.inf
    
    for solver in [cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None:
                gv = np.asarray(g.value, dtype=np.float64)
                if not np.any(np.isnan(gv)) and not np.any(np.isinf(gv)):
                    gv = np.maximum(gv, 0.0)
                    obj = gv.sum()
                    if obj > best_obj:
                        best_obj = obj
                        best_g = gv
        except Exception:
            continue
    
    return best_g


def golden_line_search(f, h, max_iter=50):
    """
    Golden section search for t in [0, 1] minimizing max(conv(f_t, f_t))
    where f_t = (1-t)*f + t*h, normalized to sum 1.
    """
    phi = (np.sqrt(5) - 1) / 2
    
    def obj_at(t):
        ft = (1.0 - t) * f + t * h
        ft = np.maximum(ft, 0.0)
        s = ft.sum()
        if s < 1e-15:
            return np.inf
        ft = ft / s
        return fast_convolve(ft).max()
    
    a, b = 0.0, 1.0
    c = b - phi * (b - a)
    d = a + phi * (b - a)
    fc = obj_at(c)
    fd = obj_at(d)
    
    best_t = c if fc < fd else d
    best_val = min(fc, fd)
    
    for _ in range(max_iter):
        if b - a < 1e-8:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - phi * (b - a)
            fc = obj_at(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + phi * (b - a)
            fd = obj_at(d)
        cur_val = min(fc, fd)
        if cur_val < best_val:
            best_val = cur_val
            best_t = c if fc < fd else d
    
    return best_t, best_val

def project_simplex(v, target=1.0):

def build_lp_matrix(f, peaks):
    """Build dense constraint matrix for the LP.
    A[i, j] = coefficient of g[j] in (f*g)(peaks[i]).
    (f*g)(k) = sum_j f[k-j] * g[j] for valid j.

        j_hi = min(n - 1, k)
        A[i, j_lo:j_hi + 1] = f[k - j_hi:k - j_lo + 1][::-1]
    return A

def mirror_descent_polish(f_init, time_limit, label="md", seed=55):
    """Mirror descent polishing with softmax peak weights."""
    t0 = time.monotonic()
    n = len(f_init)
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = f / f.sum()
    best_f = f.copy()
    best_M = fast_convolve(f).max()
    lr = 0.02
    temp = 0.003
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()
        scaled = b / (temp * M)
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        # grad[i] = sum_k w[k] * f[k-i] via FFT
        w_padded = np.zeros(2 * n - 1)
        w_padded[:len(w)] = w
        fw = fast_convolve(w_padded)
        # fw[k] = sum_j w_padded[j] * w_padded[k-j], we want cross-correlation
        # Actually let's compute grad directly via the correlation formula
        # grad[i] = sum_k w[k] * f[k-i] for valid k-i
        # This is (w * f)(i) = sum_k w[k] * f[i-k] ... hmm
        # Let me use: (f*w)(k) = sum_i f[i]*w[k-i]
        # grad[i] = sum_k w[k]*f[k-i]
        # Let m = k-i, then grad[i] = sum_m w[m+i]*f[m] = (w_shifted * f)(i) where w_shifted[m] = w[m+i]
        # This is the cross-correlation. Use FFT:
        # cross_corr[i] = sum_k f[k] * w[k+i] = (f * flip(w))(i + offset)
        # Simpler: just do it with a loop for active w
        active = np.where(w > 1e-10)[0]
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            # grad[i] += w[k] * f[k-i] for i where k-i in [0,n-1]
            # i ranges from max(0, k-n+1) to min(n-1, k)
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            grad[i_lo:i_hi + 1] += w[k] * f[k - i_hi:k - i_lo + 1][::-1]
        log_f = np.log(f + 1e-30)
        log_f_new = log_f - lr * grad
        log_f_new -= log_f_new.max()
        f_new = np.exp(log_f_new)
        s = f_new.sum()
        if s < 1e-30:
            break
        f = f_new / s
        if it % 30 == 0:
            cur_M = fast_convolve(f).max()
            if cur_M < best_M - 1e-14:
                best_M = cur_M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 4:
                    lr *= 0.93
                    no_improve = 0
            if no_improve > 20:
                f = f * (1.0 + rng.standard_normal(n) * 0.005)
                f = np.maximum(f, 0.0)
                f /= f.sum()
                lr = 0.02
                no_improve = 0
            if it % 100 == 0:
                print(f"  [{label}] it={it} score={2.0*n*best_M:.6f} lr={lr:.4f} t={time.monotonic()-t0:.0f}s")
    return best_f, best_M


def coordinate_descent(f_init, time_limit, label="cd"):
    """Cyclic coordinate descent with multiplicative steps."""
    t0 = time.monotonic()
    n = len(f_init)
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = f / f.sum()
    best_M = fast_convolve(f).max()
    best_f = f.copy()
    multipliers = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0]
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-18:
                continue
            best_val = f[j]
            best_obj = best_M
            for m in multipliers:
                new_val = f[j] * m
                f[j] = new_val
                s = f.sum()
                if s < 1e-10:
                    f[j] = best_val
                    continue
                f_norm = f / s
                obj = fast_convolve(f_norm).max()
                if obj < best_obj - 1e-15:
                    best_obj = obj
                    best_val = new_val
            f[j] = best_val
            s = f.sum()
            if s > 1e-10:
                f = f / s
            if best_val != f[j] * (f.sum() / f.sum()):
                pass
            if best_obj < best_M - 1e-15:
                best_M = best_obj
                best_f = f.copy()
                improved = True
        if not improved:
            break
    return best_f, best_M


def interpolate_length(f, new_n):
    """Interpolate sequence to new length."""
    n = len(f)
    if n == new_n:
        return f.copy()
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

    tri = np.arange(1, n_p + 1, dtype=np.float64)
    tri /= tri.sum()
    alternatives.append(("triangle", tri))
    x = np.linspace(-3, 3, n_p)
    gauss = np.exp(-x ** 2)
    gauss /= gauss.sum()
    alternatives.append(("gaussian", gauss))
    # Asymmetric: mass on one side
    asym = np.exp(-np.linspace(0, 4, n_p))
    asym /= asym.sum()
    alternatives.append(("asymmetric", asym))

    for name, alt in alternatives:
        sc = compute_score_normalized(alt)
        print(f"  Alt '{name}': score={sc:.6f}")
        if sc < best_score:
            best_score = sc
            best_seq = alt.copy()

    # === Phase 1: Main LP (50% time) ===
    rem = deadline - time.monotonic()
    t1 = rem * 0.48
    print(f"\n=== Phase 1: Main LP (budget={t1:.0f}s) ===")
    f1, M1 = lp_optimize(best_seq, time_limit=t1, initial_frac=0.95,
                         final_frac=0.80, max_peaks=350, patience=35,
                         label="lp1", seed=42)
    sc1 = 2.0 * len(f1) * M1
    print(f"  Phase 1: score={sc1:.6f}")
    if sc1 < best_score:
        best_score = sc1
        best_seq = f1.copy()

    # === Phase 2: Deeper LP (20% time) ===
    rem = deadline - time.monotonic()
    t2 = rem * 0.22
    print(f"\n=== Phase 2: LP deep (budget={t2:.0f}s) ===")
    f2, M2 = lp_optimize(best_seq, time_limit=t2, initial_frac=0.92,
                         final_frac=0.72, max_peaks=500, patience=40,
                         label="lp2", seed=123)
    sc2 = 2.0 * len(f2) * M2
    print(f"  Phase 2: score={sc2:.6f}")
    if sc2 < best_score:
        best_score = sc2
        best_seq = f2.copy()

    # === Phase 3: Multi-length (15% time) ===
    rem = deadline - time.monotonic()
    n_base = len(best_seq)
    target_ns = []
    for factor in [0.88, 1.08, 1.15, 1.25]:
        tn = int(n_base * factor)
        tn = max(256, min(2500, tn))
        if tn != n_base and tn not in target_ns:
            target_ns.append(tn)
    target_ns.sort(key=lambda x: abs(x - n_base))
    for idx, tn in enumerate(target_ns[:2]):
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t3 = rem * 0.15
        if t3 < 20:
            break
        f_interp = interpolate_length(best_seq, tn)
        print(f"\n=== Phase 3: LP n={tn} (budget={t3:.0f}s) ===")
        f3, M3 = lp_optimize(f_interp, time_limit=t3, initial_frac=0.93,
                             final_frac=0.78, max_peaks=300, patience=25,
                             label=f"lp_n{tn}", seed=777 + idx)
        sc3 = 2.0 * tn * M3
        print(f"  Phase 3 (n={tn}): score={sc3:.6f}")
        if sc3 < best_score:
            best_score = sc3
            best_seq = f3.copy()

    # === Phase 4: Mirror descent (8% time) ===
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.10, 100.0)
    if t4 > 15:
        print(f"\n=== Phase 4: Mirror descent (budget={t4:.0f}s) ===")
        f4, M4 = mirror_descent_polish(best_seq, time_limit=t4, label="md", seed=55)
        sc4 = 2.0 * len(f4) * M4
        print(f"  Phase 4: score={sc4:.6f}")
        if sc4 < best_score:
            best_score = sc4
            best_seq = f4.copy()

    # === Phase 5: Coordinate descent (7% time) ===
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.10, 100.0)
    if t5 > 10:
        print(f"\n=== Phase 5: Coordinate descent (budget={t5:.0f}s) ===")
        f5, M5 = coordinate_descent(best_seq, time_limit=t5, label="cd")
        sc5 = 2.0 * len(f5) * M5
        print(f"  Phase 5: score={sc5:.6f}")
        if sc5 < best_score:
            best_score = sc5
            best_seq = f5.copy()

    # === Phase 6: Final LP burst (remaining time) ===
    rem = deadline - time.monotonic()
    t6 = rem * 0.50
    if t6 > 15:
        print(f"\n=== Phase 6: Final LP (budget={t6:.0f}s) ===")
        f6, M6 = lp_optimize(best_seq, time_limit=t6, initial_frac=0.90,
                             final_frac=0.72, max_peaks=500, patience=30,
                             label="lp_fin", seed=999)
        sc6 = 2.0 * len(f6) * M6
        print(f"  Phase 6: score={sc6:.6f}")
        if sc6 < best_score:
            best_score = sc6
            best_seq = f6.copy()

    # Final coordinate descent with remaining time
    rem = deadline - time.monotonic()
    if rem > 10:
        f7, M7 = coordinate_descent(best_seq, time_limit=rem * 0.8, label="cd_fin")
        sc7 = 2.0 * len(f7) * M7
        if sc7 < best_score:
            best_score = sc7
            best_seq = f7.copy()

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if not np.isfinite(final_score):
        best_seq = parent.copy()
        final_score = float(evaluate_sequence(parent.tolist()))
    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {final_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")
    return [float(x) for x in best_seq]
