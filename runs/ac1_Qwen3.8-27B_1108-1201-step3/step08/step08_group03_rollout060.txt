When is sum(g*) = 1 exactly? When the current f is already "optimally flat" in the sense that no feasible direction can increase the sum while keeping (f*g)[k] <= M. This happens at the optimum of the original problem.

In practice, with numerical precision, the LP might report sum(g*) = 1 + ε for some tiny ε, giving a noise direction. I should check that the improvement is significant before accepting it.

"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses full-constraint LP direction finding with PDLP solver, spectral
ratio-matching for diversification, and multi-start exploration.
Key improvement: uses ALL convolution constraints in the LP (no cutting
planes needed for moderate n), giving a tighter and more reliable search.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size).astype(np.complex128)
    return irfft(A * A, fft_size)[:size]


def cross_convolve(f, g):
    """Linear cross-convolution f*g via FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size).astype(np.complex128)
    G = rfft(g, fft_size).astype(np.complex128)
    return irfft(F * G, fft_size)[:size]


def build_full_constraint_matrix(f, n):
    """Build the full (2n-1) x n constraint matrix for (f*g)[k] <= M."""
    K = 2 * n - 1
    A = np.zeros((K, n))
    for k in range(K):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi+1] = f[lo:hi+1]
    return A


def solve_full_lp(f, n, M, timeout_hint=None):
    """Solve: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    K = 2 * n - 1
    A = build_full_constraint_matrix(f, n)
    
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    
    g_val = None
    try:
        prob.solve(solver=cp.PDP, max_iters=500000, eps_abs=1e-9, eps_rel=1e-9)
        if g.value is not None:
            g_val = g.value
    except Exception:
        pass
    
    if g_val is None or np.any(np.isnan(g_val)):
        try:
            prob.solve(solver=cp.SCS, max_iters=500000, eps=1e-9)
            if g.value is not None:
                g_val = g.value
        except Exception:
            pass
    
    if g_val is None or np.any(np.isnan(g_val)):
        if n <= 3000:
            try:
                prob.solve(solver=cp.ECOS, max_iters=50000)
                if g.value is not None:
                    g_val = g.value
            except Exception:
                pass
    
    if g_val is None:
        return None
    
    g_val = np.maximum(g_val, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None
    if g_val.sum() < 1e-10:
        return None
    return g_val


def ratio_match(f_init, n_iters=100, flatten_power=0.7, min_ratio=0.1):
    """
    Spectral ratio matching: iteratively flatten the power spectrum
    while preserving non-negativity in the time domain.
    """
    n = len(f_init)
    f = f_init.copy()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    for it in range(n_iters):
        F = rfft(f, fft_size).astype(np.complex128)
        R = np.abs(F) ** 2
        R = np.maximum(R, 1e-30)
        
        # Flatten: R_new = R^p (reduces dynamic range)
        R_mean = R.mean()
        R_flat = R_mean * (R / R_mean) ** flatten_power
        
        # New magnitude
        new_mag = np.sqrt(R_flat)
        
        # Preserve phase
        phase = F / (np.abs(F) + 1e-30)
        F_new = new_mag * phase
        
        # Inverse transform
        f_new = irfft(F_new, fft_size)[:n]
        
        # Project to non-negative
        f_new = np.maximum(f_new, 0.0)
        
        # Normalize
        s = f_new.sum()
        if s < 1e-10:
            f = f_init.copy()
            break
        f = f_new / s
    
    return f

                best_t = t

                best_t = t_mid

def subgradient_polish(f_init, time_limit, lr_init=0.001, n_top=100):
    """Subgradient method with momentum."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    v = np.zeros(n)
    lr = lr_init
    best_f = f.copy()
    best_M = fast_convolve(f).max()
    momentum = 0.9
    it = 0
    rng = np.random.default_rng(123)
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        
        # Top-k peaks with softmax weights
        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
        else:
            top_idx = np.arange(len(b))
        
        w = np.zeros(len(b))
        bw = b[top_idx]
        bw_shifted = bw - bw.max()
        w_vals = np.exp(bw_shifted / (0.01 * bw.max() + 1e-10))
        w_vals /= w_vals.sum()
        w[top_idx] = w_vals
        
        # Subgradient
        g = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-12:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += w[k] * 2.0 * f[k - idx]
        
        v = momentum * v - lr * g
        f_new = f + v
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-10:
            f_new = f.copy()
        else:
            f_new = f_new / s
        f = f_new
        
        if it % 50 == 0:
            obj = fast_convolve(f).max()
            if obj < best_M - 1e-12:
                best_M = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve % 10 == 0:
                    lr *= 0.85
                if no_improve > 50:
                    f = best_f + rng.standard_normal(n) * 0.002
                    f = np.maximum(f, 0.0)
                    s = f.sum()
                    f = f / s
                    v = np.zeros(n)
                    no_improve = 0
    
    return best_f, best_M
