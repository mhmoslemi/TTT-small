def project_simplex(v, target_sum=1.0):
    """Project onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    v = v.copy()
    v *= target_sum  # scale to sum = target_sum
    # Standard simplex projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)

import numpy as np
from scipy.fft import rfft, irfft
import time

def auto_convolve(a):
    """Fast linear auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    v = v * s / max(v.sum(), 1e-30)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u * rho > (cssv - s)
    if not cond.any():
        return np.full(n, s/n)
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - s) / (idx + 1)
    return np.maximum(v - theta, 0.0)

def compute_smooth_gradient(a, T, b=None):
    """
    Compute gradient of 2n * LSE(T * auto_conv(a)).
    Returns (gradient, smoothed_value).
    """
    n = len(a)
    if b is None:
        b = auto_convolve(a)
    
    # Softmax weights
    b_max = b.max()
    exponents = T * (b - b_max)
    # Clip to avoid underflow
    exponents = np.clip(exponents, -50, 50)
    w = np.exp(exponents)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(b))
        w[np.argmax(b)] = 1.0
    else:
        w = w / w_sum
    
    # Smoothed max
    smoothed_max = b_max + np.log(w_sum) / T
    
    # Gradient: g_j = 2n * 2 * sum_m w[j+m] * a[m]
    # = 4n * (cross_corr(w, a))[j]
    fft_size = 1 << (len(w) + n - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    cross = irfft(W * A.conj(), fft_size)
    grad = 4.0 * n * cross[:n]
    
    return grad, smoothed_max

"""
Nesterov accelerated gradient with log-sum-exp smoothing for minimizing
2n * max(auto-conv(a)) / (sum(a))^2 on the simplex.

Key innovations over the LP approach:
1. FFT-based gradient computation (O(n log n) per step vs O(n^2) LP)
2. Log-sum-exp smoothing enables momentum and acceleration
3. Geometric temperature schedule (continuation method)
4. Multi-length exploration with diverse initialization
5. Active-set refinement at high temperature
6. Much more iterations possible in same time budget
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def auto_convolve(a):
    """Fast linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, s=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    v_sum = v.sum()
    if v_sum < 1e-30:
        return np.full(n, s / n)
    v = v * s / v_sum
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u * rho > (cssv - s)
    if not cond.any():
        return np.full(n, s / n)
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - s) / (idx + 1)
    return np.maximum(v - theta, 0.0)


def smooth_gradient(a, T, b=None, top_k=None):
    """
    Compute gradient of 2n * LSE(T * b) where b = auto_conv(a).
    Returns (gradient_vector, smoothed_max_value).
    """
    n = len(a)
    if b is None:
        b = auto_convolve(a)

    # Restrict to top_k peaks for efficiency at high T
    m = len(b)
    if top_k is not None and top_k < m:
        idx = np.argpartition(b, -top_k)[-top_k:]
        b_use = b[idx].copy()
        w_full = None
    else:
        b_use = b
        w_full = None

    b_max = b_use.max()
    exponents = T * (b_use - b_max)
    exponents = np.clip(exponents, -700, 700)
    w = np.exp(exponents)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(b_use))
        w[np.argmax(b_use)] = 1.0
    else:
        w = w / w_sum

    smoothed_max = b_max + np.log(max(w_sum, 1e-300)) / T

    if top_k is not None and top_k < m:
        # Build full weight vector
        w_full = np.zeros(m)
        w_full[idx] = w
    else:
        w_full = w

    # Gradient: g_j = 4n * sum_m w_full[j+m] * a[m]
    # This is 4n * cross_correlation(w_full, a)[j]
    fft_size = 1 << (m + n - 1).bit_length()
    W = rfft(w_full, fft_size)
    A = rfft(a, fft_size)
    cross = irfft(W * A.conj(), fft_size)
    grad = 4.0 * n * cross[:n]

    return grad, smoothed_max


def nesterov_optimize(a_init, time_limit, T_start=5.0, T_end=20000.0,
                      momentum_init=0.9, momentum_end=0.5,
                      lr_init=0.5, top_k_switch=200, top_k_max=None,
                      label="nest", verbose=True):
    """
    Nesterov accelerated gradient descent on the simplex with adaptive temperature.
    """
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(n) / n

    best_a = a.copy()
    best_obj = auto_convolve(a).max()
    best_score = 2.0 * n * best_obj

    a_prev = a.copy()
    y = a.copy()

    it = 0
    no_improve = 0
    check_interval = max(50, n // 10)

    # Temperature: geometric schedule
    n_phases = 30
    T_ratio = (T_end / T_start) ** (1.0 / n_phases) if T_end > T_start else 1.0
    T = T_start
    phase = 0
    steps_per_phase = max(3, int(50000.0 / n))

    # Momentum annealing
    mom_ratio = (momentum_end / momentum_init) ** (1.0 / n_phases)
    momentum = momentum_init

    # Learning rate decreases with T
    lr = lr_init

    # Top-k for gradient (activated at high T)
    use_top_k = False
    current_top_k = top_k_max if top_k_max else (2 * n - 1)

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Temperature and momentum schedule
        if it % steps_per_phase == 0 and phase < n_phases - 1:
            T *= T_ratio
            momentum *= mom_ratio
            phase += 1
            # Activate top-k at high temperature

                current_top_k = max(50, min(200, 2 * n - 1))

        y = project_simplex(y)

        # Compute gradient

        if use_top_k:

            grad, _ = smooth_gradient(y, T, b_y)

        # Gradient step
        a_new = project_simplex(y - lr * grad)

        # Update Nesterov state
        a_prev = a
        a = a_new

        # Periodic evaluation
        if it % check_interval == 0:
            obj = auto_convolve(a).max()
            if obj < best_obj - 1e-13:

                best_score = 2.0 * n * obj

                no_improve = 0

                no_improve += 1
                if no_improve > 20:
                    # Restart from best with small perturbation

                    a = best_a + rng.standard_normal(n) * (0.002 / np.sqrt(n))
                    a = project_simplex(a)
                    a_prev = a.copy()
                    no_improve = 0
                    momentum = min(momentum + 0.1, 0.95)

        if verbose and it % (check_interval * 20) == 0:

    return best_a, best_score

def coordinate_descent(a_init, time_limit, n_sweeps=3):
    """
    Coordinate descent: for each coordinate, find the optimal value
    that minimizes the max auto-convolution.
    """
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s

    best_a = a.copy()
    best_obj = auto_convolve(a).max()

    multipliers = [0.3, 0.5, 0.7, 0.85, 0.95, 1.0, 1.05, 1.15, 1.3, 1.5, 1.8, 2.5]

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        any_improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[j] < 1e-18:
                continue

            b_current = auto_convolve(a)
            M_current = b_current.max()

            best_val = a[j]
            best_M = M_current
            for mult in multipliers:
                trial = a.copy()
                trial[j] = a[j] * mult
                ts = trial.sum()
                if ts < 1e-15:
                    continue
                trial = trial / ts
                b_trial = auto_convolve(trial)
                M_trial = b_trial.max()
                if M_trial < best_M - 1e-14:
                    best_M = M_trial
                    best_val = a[j] * mult

            if best_val != a[j]:
                a[j] = best_val
                ts = a.sum()
                if ts > 1e-15:
                    a = a / ts
                if best_M < best_obj - 1e-14:
                    best_obj = best_M
                    best_a = a.copy()
                    any_improved = True

        if not any_improved:
            break

    return best_a, 2.0 * n * best_obj


def interp_sequence(src, target_n):
    """Interpolate a sequence to a new length, maintaining non-negativity and unit sum."""
    src_n = len(src)
    if src_n == target_n:
        result = src.copy()
    elif src_n == 1:
        result = np.full(target_n, src[0])
    else:
        old_x = np.linspace(0.0, 1.0, src_n)
        new_x = np.linspace(0.0, 1.0, target_n)
        result = np.interp(new_x, old_x, src)
        result = np.maximum(result, 0.0)

    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result


def diverse_initialization(n, seed=0):
    """Create diverse initial sequences for exploration."""
    rng = np.random.default_rng(seed)
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Triangular
    x = np.linspace(0, 1, n)
    tri = 1 - np.abs(2 * x - 1)
    tri = tri / tri.sum()
    inits.append(tri)

    # U-shaped
    u_shape = x**2 + (1-x)**2
    u_shape = u_shape / u_shape.sum()
    inits.append(u_shape)

    # Exponential
    exp_a = np.exp(2 * x)
    exp_a = exp_a / exp_a.sum()
    inits.append(exp_a)

    # Random (Pareto-like)
    rand = rng.pareto(1.5, n) + 0.1
    rand = rand / rand.sum()
    inits.append(rand)

    # Gaussian
    gauss = np.exp(-8 * (x - 0.5)**2)
    gauss = gauss / gauss.sum()
    inits.append(gauss)

    return inits

            real_score = float(evaluate_sequence(f_res.tolist()))
            update_best(f_res, real_score, f"p1 n={n_trial}")

    # ============================================================
    # Phase 2: Diverse initialization on best n
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 2: Diverse init (best: {best_score:.8f}, n={len(best_seq)})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    t2 = min(rem * 0.2, 300.0)

    if t2 > 30:
        n_best = len(best_seq)
        inits = diverse_initialization(n_best, seed=seed)

        per_init = t2 / len(inits)
        for i, f_init in enumerate(inits):
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            alloc = min(per_init, rem * 0.9)
            if alloc < 8:
                break

            print(f"  Diverse init {i+1}/{len(inits)}, budget={alloc:.0f}s")
            f_res, score = nesterov_optimize(
                f_init, time_limit=alloc,
                T_start=10.0, T_end=10000.0,
                momentum_init=0.9, momentum_end=0.4,
                lr_init=0.5, label=f"d{i}")

            real_score = float(evaluate_sequence(f_res.tolist()))
            update_best(f_res, real_score, f"d{i}")

    # ============================================================
    # Phase 3: Deep Nesterov on best with high T range
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 3: Deep Nesterov (best: {best_score:.8f}, n={len(best_seq)})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    t3 = min(rem * 0.25, 400.0)

    if t3 > 30:
        n_best = len(best_seq)
        print(f"  Deep Nesterov n={n_best}, budget={t3:.0f}s")
        f_res, score = nesterov_optimize(
            best_seq, time_limit=t3,
            T_start=50.0, T_end=50000.0,
            momentum_init=0.85, momentum_end=0.3,
            lr_init=0.3, top_k_switch=100,
            top_k_max=min(200, 2*n_best-1),
            label="deep")

        real_score = float(evaluate_sequence(f_res.tolist()))
        update_best(f_res, real_score, "deep")

    # ============================================================
    # Phase 4: Try nearby n values with refinement
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Nearby n refinement (best: {best_score:.8f})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    t4 = min(rem * 0.15, 200.0)

    if t4 > 20:
        n_best = len(best_seq)
        nearby = []
        for delta in [-64, -32, 32, 64, 128, 256]:
            n_val = n_best + delta
            if 200 <= n_val <= 4000 and n_val != n_best:
                nearby.append(n_val)

        if nearby:
            per_n = t4 / len(nearby)
            for n_trial in nearby:
                rem = deadline - time.monotonic()
                if rem < 10:
                    break
                alloc = min(per_n, rem * 0.9)
                if alloc < 8:
                    break

                f_init = interp_sequence(best_seq, n_trial)
                print(f"  Refine n={n_trial}, budget={alloc:.0f}s")
                f_res, score = nesterov_optimize(
                    f_init, time_limit=alloc,
                    T_start=20.0, T_end=20000.0,
                    momentum_init=0.85, momentum_end=0.4,
                    lr_init=0.4, label=f"r{n_trial}")

                real_score = float(evaluate_sequence(f_res.tolist()))
                update_best(f_res, real_score, f"r n={n_trial}")

    # ============================================================
    # Phase 5: Subgradient polish on best
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Subgradient polish (best: {best_score:.8f})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    t5 = min(rem * 0.15, 180.0)

    if t5 > 15:
        n_best = len(best_seq)
        n_top = max(30, n_best // 15)
        print(f"  SG n={n_best}, top_k={n_top}, budget={t5:.0f}s")
        f_res, score = subgradient_polish(
            best_seq, time_limit=t5,
            n_top=n_top, lr_init=0.005,
            momentum_init=0.7, label="sg")

        real_score = float(evaluate_sequence(f_res.tolist()))
        update_best(f_res, real_score, "sg")

    # ============================================================
    # Phase 6: Coordinate descent final polish
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 6: CD polish (best: {best_score:.8f})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    t6 = min(rem * 0.1, 90.0)

    if t6 > 10:
        print(f"  CD n={len(best_seq)}, budget={t6:.0f}s")
        f_res, score = coordinate_descent(best_seq, time_limit=t6, n_sweeps=2)
        real_score = float(evaluate_sequence(f_res.tolist()))
        update_best(f_res, real_score, "cd")

    # ============================================================
    # Final verification
    # ============================================================
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
