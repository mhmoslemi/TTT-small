import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_cross(f, g):
    """Compute cross-convolution f*g via FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_A(f, indices, n):
    """Build constraint matrix for cross-correlation."""
    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp_direction(f, n, M, active_idx, use_pdlp=True):
    """Solve LP to find descent direction."""
    K = len(active_idx)
    if K == 0:
        return None
    A = build_A(f, active_idx, n)

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    g_val = None
    if use_pdlp and n >= 200:
        try:
            prob.solve(solver=cp.PDLP, verbose=False, max_iter=500000)
            g_val = g.value
        except Exception:
            g_val = None
    if g_val is None:
        try:
            prob.solve(solver=cp.SCS, max_iters=200000, eps=1e-9, verbose=False)
            g_val = g.value
        except Exception:
            g_val = None
    if g_val is None and n <= 3000:
        try:
            prob.solve(solver=cp.ECOS, max_iters=20000, verbose=False)
            g_val = g.value
        except Exception:
            g_val = None

    if g_val is None:
        return None
    g_val = np.maximum(g_val, 0.0)
    if np.any(np.isnan(g_val)) or g_val.sum() < 1e-12:
        return None
    return g_val


def line_search(f, direction, M, max_t=1.0):
    """Golden-section line search to minimize peak."""
    def obj_at_t(t):
        f_t = (1.0 - t) * f + t * direction
        f_t = np.maximum(f_t, 0.0)
        s = f_t.sum()
        if s < 1e-12:
            return 1e10
        f_t = f_t / s
        return fast_convolve(f_t).max()

    # Initial scan
    best_t = 0.0
    best_obj = M
    scan = [0.02, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
    for t in scan:
        o = obj_at_t(t)
        if o < best_obj:
            best_obj = o
            best_t = t

    # Golden section refinement
    if best_t > 0.005:
        lo = max(0.0, best_t - 0.08)
        hi = min(1.0, best_t + 0.12)
        invphi = (np.sqrt(5.0) - 1.0) / 2.0
        for _ in range(10):
            m1 = lo + invphi * (hi - lo)
            m2 = hi - invphi * (hi - lo)
            f1 = obj_at_t(m1)
            f2 = obj_at_t(m2)
            if f1 < f2:
                hi = m2
                if f1 < best_obj:
                    best_obj = f1
                    best_t = m1
            else:
                lo = m1
                if f2 < best_obj:
                    best_obj = f2
                    best_t = m2
        t_mid = (lo + hi) / 2.0
        o_mid = obj_at_t(t_mid)
        if o_mid < best_obj:
            best_obj = o_mid
            best_t = t_mid

    return best_t, best_obj


def lp_cutting_plane(f_init, time_limit, initial_k=50, max_k=500, label="lp"):
    """Main LP optimization with cutting planes."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-12:
        f = np.ones(n) / n
    else:
        f /= s

    b = fast_convolve(f)
    M = b.max()
    best_f = f.copy()
    best_obj = M

    active_idx = np.argpartition(b, -min(initial_k, len(b)))[-min(initial_k, len(b)):]
    active_idx = np.sort(active_idx)

    it = 0
    stuck = 0
    use_pdlp = True

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break

        g = solve_lp_direction(f, n, M, active_idx, use_pdlp=use_pdlp)

        if g is None:
            stuck += 1
            if stuck > 10:
                # Try without PDLP
                use_pdlp = not use_pdlp
                g = solve_lp_direction(f, n, M, active_idx, use_pdlp=use_pdlp)
                if g is None:
                    if stuck > 20:
                        break
                    continue
            else:
                continue

        S_g = g.sum()
        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > 25:
                break
            # Expand active set
            if len(active_idx) < max_k:
                b_cur = fast_convolve(f)
                new_top = np.argpartition(b_cur, -min(max_k, len(b_cur)))[-min(max_k, len(b_cur)):]
                active_idx = np.unique(np.concatenate([active_idx, new_top]))
            continue

        # Check for violated constraints (cutting plane)
        fg = convolve_cross(f, g)
        violated = fg > M * (1 + 1e-8)
        v_idx = np.where(violated)[0]
        not_in_active = v_idx[~np.isin(v_idx, active_idx)]

        if len(not_in_active) > 0 and len(active_idx) < max_k:
            n_add = min(50, len(not_in_active))
            to_add = not_in_active[np.argsort(fg[not_in_active])[-n_add:]]
            active_idx = np.unique(np.concatenate([active_idx, to_add]))
            active_idx = active_idx[np.argsort(fg[active_idx])[-max_k:]]

            g = solve_lp_direction(f, n, M, active_idx, use_pdlp=use_pdlp)
            if g is None:
                stuck += 1
                if stuck > 10:
                    break
                continue
            S_g = g.sum()

        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > 30:
                break
            continue

        stuck = 0
        h = g / S_g

        # Line search
        best_t, best_t_obj = line_search(f, h, M)

        if best_t < 1e-6:
            stuck += 5
            if stuck > 15:
                break
            continue

        # Update
        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s < 1e-12:
            f = np.ones(n) / n
        else:
            f /= s

        cur_obj = fast_convolve(f).max()
        M = cur_obj

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()

        # Periodically refresh active set
        if it % 20 == 0:
            b = fast_convolve(f)
            keep_size = min(len(active_idx), max_k)
            active_idx = np.sort(b[np.argsort(b)[-keep_size:]])

        if it % 50 == 0:

            print(f"  {label} it={it} obj={best_obj:.12f} score={score:.8f} "
                  f"K={len(active_idx)} t={elapsed:.0f}s")

    return best_f, best_obj


def spectral_diversify(f_init, n_iters=200, aggression=0.3):
    """Flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(f_init)
    f = f_init.copy()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    for it in range(n_iters):
        F = rfft(f, fft_size)
        R = irfft(F * F, fft_size)[:size]

        # Flatten the correlation: reduce peaks, boost valleys
        peak = R.max()
        floor = aggression * peak
        R_mod = np.clip(R, floor, peak)

        # Preserve energy
        energy = R.sum()
        if R_mod.sum() > 1e-12:
            R_mod *= energy / R_mod.sum()

        # Update amplitude spectrum
        R_fft = rfft(R_mod, fft_size)
        new_amp = np.sqrt(np.maximum(R_fft.real, 0.0))
        abs_F = np.abs(F)
        phase = F / (abs_F + 1e-30)
        F_new = new_amp * phase
        f = irfft(F_new, fft_size)[:n]
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-12:
            f /= s
        else:
            break

    return f


def subgradient_polish(f_init, time_limit, lr_init=0.001, temp_init=0.002):
    """Subgradient with momentum and adaptive learning rate."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    v = np.zeros(n)
    lr = lr_init
    temp = temp_init
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    it = 0
    momentum = 0.9
    rng = np.random.default_rng(123)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)

        # Soft weights on peaks
        scaled = b / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        # Subgradient
        g = np.zeros(n)
        nonzero = w > 1e-10
        for k in np.where(nonzero)[0]:
            wk = w[k]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            g[lo:hi + 1] += wk * 2.0 * f[k - hi:lo - 1:-1] if lo < hi else wk * f[k]

        # Actually let me redo the subgradient computation properly
        g = np.zeros(n)
        for k in np.where(nonzero)[0]:
            wk = w[k]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # b[k] = sum_{i=lo}^{hi} f[i] * f[k-i]
            # db[k]/d f[j] = f[k-j] if lo <= j <= hi (and 0 <= k-j < n)
            for j in range(lo, hi + 1):
                kj = k - j
                if 0 <= kj < n:
                    g[j] += wk * f[kj]

        v = momentum * v - lr * g
        f = project_simplex(f + v)

        if it % 100 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 5:
                    lr *= 0.9
                    temp *= 1.15
                    no_improve = 0
                if no_improve > 50:
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                    v = np.zeros(n)
                    no_improve = 0

    return best_f, best_obj


def coordinate_descent(f_init, time_limit):
    """Coordinate descent with multiplicative updates."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    step = max(1, n // 300)
    factors = [0.8, 0.9, 1.1, 1.2]

    for sweep in range(30):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for j in range(0, n, step):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-14:
                continue
            orig = f[j]
            best_v = orig
            best_o = best_obj
            for mult in factors:
                f[j] = orig * mult
                s = f.sum()
                if s < 1e-12:
                    f[j] = orig
                    continue
                f /= s
                o = fast_convolve(f).max()
                if o < best_o - 1e-14:
                    best_o = o
                    best_v = orig * mult
            f[j] = best_v
            s = f.sum()
            if s > 1e-12:
                f /= s
            if best_o < best_obj - 1e-14:
                best_obj = best_o
                improved = True
        if not improved:
            break
    return f, best_obj


def make_bump(n, width=0.3):
    """Create a Gaussian bump initial sequence."""
    x = np.linspace(0, 1, n)
    f = np.exp(-((x - 0.5) / width) ** 2)
    f = np.maximum(f, 0)
    s = f.sum()
    if s > 1e-12:
        f /= s
    else:
        f = np.ones(n) / n
    return f


def make_raised_cosine(n):
    """Create a raised cosine initial sequence."""
    x = np.linspace(0, 1, n)
    f = 0.5 * (1 + np.cos(2 * np.pi * x))
    f = np.maximum(f, 1e-6)
    s = f.sum()
    if s > 1e-12:
        f /= s
    else:
        f = np.ones(n) / n
    return f

            best_seq = f8.copy()
            print(f"  NEW BEST [cd]: {s8:.8f}")

    # Final validation
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'=' * 50}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'=' * 50}")

    return [float(x) for x in best_seq]
