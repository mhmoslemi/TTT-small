def build_full_constraint_matrix(f, n):
    """Build (2n-1) x n constraint matrix for full LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[j])
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

        f_trial = f_trial / f_trial.sum()
        cur_obj = fast_convolve(f_trial).max()
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_score = 2.0 * n * best_obj
            best_f = f_trial.copy()
            f = f_trial.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 40:
                break
        
        if it % 15 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} obj={best_obj:.12f} score={best_score:.6f} "
                  f"K={len(indices)} t={elapsed:.0f}s")
    
    return best_f, best_score


def run_smoothed_grad_opt(f_init, time_limit, label="sg", T_ratio=0.03, mom_init=0.6, lr_init=0.001):
    """Smoothed log-sum-exp gradient with momentum."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s
    
    v = np.zeros(n)
    lr = lr_init
    T = T_ratio * fast_convolve(f).max()
    mom = mom_init
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    no_improve = 0
    it = 0
    rng = np.random.default_rng(777)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        grad = smooth_grad_fft(f, T)
        
        v = mom * v - lr * grad
        f_new = project_simplex(f + v)
        f = f_new
        
        if it % 10 == 0:
            obj = fast_convolve(f).max()
            T = T_ratio * obj + 1e-12
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_score = 2.0 * n * obj
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.04, 0.03)
                mom = min(mom + 0.005, 0.85)
            else:
                no_improve += 1
                if no_improve > 5:
                    lr *= 0.88
                    mom = max(mom - 0.02, 0.4)
                    no_improve = 0
            
            if it % 50 == 0:
                print(f"    {label} it={it} obj={best_obj:.12f} score={best_score:.6f} "
                      f"lr={lr:.5f} mom={mom:.2f} t={time.monotonic()-t0:.0f}s")
            
            if no_improve > 30:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.006)
                v = np.zeros(n)
                no_improve = 0
    
    return best_f, best_score


def run_partial_lp_opt(f_init, time_limit, label="plp", top_k=150):
    """Partial LP optimization with fewer constraints (faster per iteration)."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    it = 0
    stuck = 0
    full_len = 2 * n - 1
    solvers = ['PDLP', 'ECOS', 'SCS']
    rng = np.random.default_rng(314)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()
        
        k_sel = min(top_k, full_len)
        indices = np.sort(np.argpartition(b, -k_sel)[-k_sel:])
        
        g, _ = solve_lp_direction(f, indices, M, solvers=solvers)
        if g is None:
            stuck += 2
            if stuck > 20:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.008)
                stuck = 0
            continue
        
        S_g = g.sum()
        if S_g <= 1.0 + 1e-7:
            stuck += 1
            if stuck > 30:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.01)
                stuck = 0
            elif stuck > 15:
                break
            continue
        
        h = g / S_g
        t_found, obj_found = line_search(f, h, M)
        
        if t_found == 0.0:
            stuck += 1
            if stuck > 30:
                break
            continue
        
        f_trial = (1.0 - t_found) * f + t_found * h
        f_trial = np.maximum(f_trial, 0.0)
        f_trial = f_trial / f_trial.sum()
        cur_obj = fast_convolve(f_trial).max()
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_score = 2.0 * n * best_obj
            best_f = f_trial.copy()
            f = f_trial.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 30:
                break
        
        if it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} obj={best_obj:.12f} score={best_score:.6f} t={elapsed:.0f}s")
    
    return best_f, best_score


def coordinate_descent_polish(f_init, time_limit):
    """Coordinate descent with adaptive multipliers."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    best_f = f.copy()
    
    multipliers = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0]
    no_pass_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        any_improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-14:
                continue
            
            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = f[j] * mult
                if new_val < 0:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                st = f_trial.sum()
                if st < 1e-12:
                    continue
                f_trial = f_trial / st
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val
            
            if abs(best_j_val - f[j]) > 1e-15:
                f[j] = best_j_val
                st = f.sum()
                if st > 1e-12:
                    f = f / st
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_score = 2.0 * n * best_obj
                    best_f = f.copy()
                    any_improved = True
        
        if not any_improved:
            no_pass_improve += 1
            if no_pass_improve >= 2:
                break
        else:
            no_pass_improve = 0
    
    return best_f, best_score


def interp_sequence(src, target_n):
    """Interpolate a sequence to a new length."""
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result


def make_rised_cosine(n):
    """Create a raised-cosine window as initialization."""
    x = np.linspace(0, np.pi, n)
    w = 0.5 * (1 + np.cos(x))
    w = w / w.sum()
    return w


def make_one_sided(n, frac=0.6):
    """Create a one-sided concentrated initialization."""
    m = int(n * frac)
    f = np.zeros(n)
    f[:m] = 1.0 / m
    return f


def make_bump(n, center_frac=0.4, width_frac=0.3):
    """Create a Gaussian bump initialization."""
    x = np.linspace(0, 1, n)
    center = center_frac
    width = width_frac
    f = np.exp(-0.5 * ((x - center) / width) ** 2)
    f = f / f.sum()
    return f


def proposal_score(f):
    """Compute the score of a sequence."""
    try:
        sc = float(evaluate_sequence(f.tolist()))
        return sc
    except Exception:
        return np.inf

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
    except Exception:
        pass
    
    if parent is not None:
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    else:
        parent = np.ones(512) / 512.0
        p_score = 2.0
        print(f"No parent, using uniform n=512, score=2.0")

    best_score = p_score
    
    # ============================================================
    # Phase 1: Multi-n quick exploration (~300s)

    t1 = min(rem * 0.18, 320.0)

        n_parent = len(parent)

        for factor in [0.5, 0.65, 0.8, 0.9, 1.0, 1.1, 1.25, 1.4]:
            n_c = int(n_parent * factor)
            n_c = max(256, min(2000, n_c))
            if n_c not in n_candidates:
                n_candidates.append(n_c)
        n_candidates.sort()
        
        per_n = t1 / len(n_candidates)

                interp_sequence(parent, n_trial),
                make_rised_cosine(n_trial),
                make_bump(n_trial, 0.35, 0.35),
