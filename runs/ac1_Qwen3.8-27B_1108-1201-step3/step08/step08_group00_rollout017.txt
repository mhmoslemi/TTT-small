                s = score_normalized(a_trial)
                if s < best_score - 1e-12:
                    best_score = s
                    best_a = a_trial.copy()
                    a = a_trial
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try simultaneous reduction of top 2 contributors
            if len(top_ci) >= 2:
                a_trial = a.copy()
                a_trial[idx[top_ci[0]]] *= 0.95
                a_trial[idx[top_ci[1]]] *= 0.95
                a_trial = a_trial / a_trial.sum()
                s = score_normalized(a_trial)
                if s < best_score - 1e-12:
                    best_score = s
                    best_a = a_trial.copy()
                    a = a_trial
                else:
                    break

    return best_a, best_score

def structured_init(n, rng):
    """Generate a structured initial sequence."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.exp(-3 * (x - 0.5) ** 2)
    elif choice == 2:
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        a = np.exp(-5 * x * (1 - x))
    elif choice == 4:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = a ** 1.3
    else:
        a = np.ones(n) * (1.0 + 0.3 * np.sin(4 * np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a)
    a_interp = np.maximum(a_interp, 0)
    s = a_interp.sum()
    if s > 1e-15:
        a_interp = a_interp / s
    return a_interp

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    Phase 1: Quick L-BFGS on parent to improve
    Phase 2: Extended multi-temperature L-BFGS
    Phase 3: Exact subgradient + Nesterov
    Phase 4: Log-domain coordinate descent
    Phase 5: Peak rebalance + pairwise coordinate descent
    Phase 6: Multi-start from perturbations
    Phase 7: Alternative length exploration
    Phase 8: Final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)}) [t={time.monotonic()-deadline:.0f}s remaining]")

    # === Phase 1: Quick L-BFGS warm-up ===
    print(f"\n=== Phase 1: Quick L-BFGS (n={n_best}, score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.08, 80.0)
    if t1 > 10:
        a1 = lbgfs_anneal(best_seq, t1, T_start=100.0, T_end=5.0, maxiter=3000)
        try_update(a1, "lbgfs_warmup")

    # === Phase 2: Extended L-BFGS with aggressive annealing ===
    print(f"\n=== Phase 2: Extended L-BFGS (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.30, 450.0)
    if t2 > 30:
        print(f"  Budget: {t2:.0f}s, T: 500 -> 0.05")
        a2 = lbgfs_anneal(best_seq, t2, T_start=500.0, T_end=0.05, maxiter=20000)
        try_update(a2, "lbgfs_main")

    # === Phase 3: Second L-BFGS pass from current best ===
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.15, 200.0)
    if t3 > 20:
        print(f"\n=== Phase 3: L-BFGS r2 (score={best_score:.6f}) ===")
        a3 = lbgfs_anneal(best_seq, t3, T_start=50.0, T_end=0.02, maxiter=10000)
        try_update(a3, "lbgfs_r2")

    # === Phase 4: Exact subgradient with Nesterov ===
    print(f"\n=== Phase 4: Subgradient (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.12, 150.0)
    if t4 > 15:
        a4, _ = exact_subgradient_refine(best_seq, t4, n_steps=100000)
        try_update(a4, "subgrad")

    # === Phase 5: Log-domain coordinate descent ===
    print(f"\n=== Phase 5: Coord Descent Log (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.12, 150.0)
    if t5 > 15:
        a5, _ = coordinate_descent_log(best_seq, t5, n_sweeps=30)
        try_update(a5, "coord_log")

    # === Phase 6: Peak rebalance + pairwise ===
    print(f"\n=== Phase 6: Peak Rebalance (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t6a = min(rem * 0.06, 60.0)
    if t6a > 10:
        a6, _ = peak_rebalance(best_seq, t6a)
        try_update(a6, "peak_rebal")

    rem = deadline - time.monotonic()
    t6b = min(rem * 0.05, 50.0)
    if t6b > 10:
        a6b, _ = coordinate_descent_pairs(best_seq, t6b, n_trials=10000)
        try_update(a6b, "pair_coord")

    # === Phase 7: Multi-start from perturbations ===
    print(f"\n=== Phase 7: Multi-Start (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 120:
        n_pert = 4
        t_per_pert = min(rem / n_pert * 0.8, 80.0)
        pert_scales = [0.02, 0.05, 0.10, 0.15]

        for pi in range(n_pert):
            rem = deadline - time.monotonic()
            if rem < 30:
                break

            scale = pert_scales[pi]
            n_cur = len(best_seq)
            # Structured perturbation (low-frequency)
            noise = np.zeros(n_cur)
            for freq in [1, 2, 3, 4]:
                phase = rng.uniform(0, 2 * np.pi)
                noise += rng.standard_normal() * np.sin(2 * np.pi * freq * np.linspace(0, 1, n_cur) + phase)
            noise += rng.standard_normal(n_cur) * 0.5
            perturbed = best_seq + scale * noise / np.sqrt(n_cur)
            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()

            t_lp = t_per_pert * 0.7
            a_p = lbgfs_anneal(perturbed, min(t_lp, rem - 10), T_start=20.0, T_end=0.1, maxiter=5000)
            try_update(a_p, f"pert_{pi}_T{scale}")

            rem = deadline - time.monotonic()
            if rem > 20 and len(best_seq) == n_cur:
                t_cd = min(rem * 0.3, 30.0)
                if t_cd > 10:
                    a_p2, _ = coordinate_descent_log(best_seq, t_cd, n_sweeps=10)
                    try_update(a_p2, f"pert_{pi}_cd")

    # === Phase 8: Alternative length exploration ===
    print(f"\n=== Phase 8: Alt Lengths (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    alt_candidates = []
    for mult in [1.25, 1.5, 0.8]:
        alt_n = int(n_best * mult)
        if 256 <= alt_n <= 3000 and alt_n != n_best:
            alt_candidates.append(alt_n)
    alt_candidates = sorted(set(alt_candidates))[:2]

    for alt_n in alt_candidates:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        print(f"  Trying n={alt_n}")
        t_alt = min(rem * 0.4, 180.0)

        init_alt = interpolate_to_length(best_seq, alt_n)
        a_alt = lbgfs_anneal(init_alt, t_alt * 0.8, T_start=50.0, T_end=0.5, maxiter=5000)
        try_update(a_alt, f"alt_n{alt_n}")

        rem = deadline - time.monotonic()
        if rem > 20:
            a_alt2, _ = coordinate_descent_log(best_seq, min(rem * 0.3, 40.0), n_sweeps=8)
            try_update(a_alt2, f"alt_n{alt_n}_cd")

    # === Phase 9: Final intensive polish on best ===
    print(f"\n=== Phase 9: Final Polish (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.5, 80.0)
        a_fin = lbgfs_anneal(best_seq, t_fin, T_start=10.0, T_end=0.01, maxiter=5000)
        try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 15:
        a_ps, _ = peak_rebalance(best_seq, min(rem * 0.4, 40.0))
        try_update(a_ps, "final_peak")

    rem = deadline - time.monotonic()
    if rem > 10:
        a_cd, _ = coordinate_descent_log(best_seq, min(rem * 0.5, 30.0), n_sweeps=15)
        try_update(a_cd, "final_cd")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
