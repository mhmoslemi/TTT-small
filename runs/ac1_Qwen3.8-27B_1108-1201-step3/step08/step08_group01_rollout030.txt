import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def spectral_equalize(a0, time_limit, n_iterations=200, temp_init=0.1):
    """
    Equalize the spectrum of a to minimize the peak of the autocorrelation.
    Uses iterated soft-thresholding in the frequency domain.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    temp = temp_init
    
    for it in range(n_iterations):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute spectrum
        A = rfft(a, n)
        mag = np.abs(A)
        
        # Find the "target" magnitude: we want |A[k]| to be as equal as possible
        # Subject to A[0] = 1 (DC = sum(a) = 1)

        if it % 10 == 0:
            conv = fast_convolve(a)
            M = conv.max()
            score = 2.0 * n * M
            if score < best_score:
                best_score = score
                best_a = a.copy()
        
        # Adjust spectrum: push towards uniform magnitude
        # But we need to be careful: changing the phase matters too
        # For real signals, A[n-k] = conj(A[k])
        
        # Alternative: adjust in time domain based on spectral flatness
        # Compute the "pressure" at each time point from the spectral peaks
        
        # Find frequency bins with highest power
        power = mag ** 2
        if len(power) > 1:
            # We want to reduce the spread of power
            # Use exponential tilting: boost low-power bins, reduce high-power bins
            mean_power = power[1:].mean()  # Exclude DC
            std_power = power[1:].std()
            if std_power > 1e-10:
                # Soft thresholding on the power spectrum
                target = mean_power
                # New magnitude: compress the dynamic range
                alpha = 0.1  # Compression factor
                new_mag = mag.copy()
                new_mag[1:] = target + (mag[1:] - target) * (1 - alpha)
                # Preserve phase
                phase = np.angle(A)
                A_new = new_mag * np.exp(1j * phase)
                # Ensure DC is preserved

                a_new = np.maximum(a_new, 0)
                ss = a_new.sum()

                    a_new = a_new / ss
                    a = a_new

    score = 2.0 * n * conv.max()
    if score < best_score:
        best_score = score
        best_a = a.copy()
    
    return best_a, best_score

def nesterov_subgradient(a0, time_limit, verbose=False):
    """
    Nesterov-accelerated projected subgradient method for minimizing max_k (a*a)[k]
    subject to a >= 0, sum(a) = 1.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    y = a.copy()  # Nesterov momentum variable
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    t = 1
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        # Compute subgradient at y
        conv_y = fast_convolve(y)
        M_y = conv_y.max()
        
        # Find active peaks (within a tolerance of the max)
        tol = 0.01 * M_y
        active = np.where(conv_y >= M_y - tol)[0]
        
        # Compute subgradient: weighted sum of 2*C_k*a for active k
        # (C_k * a)_j = a[k-j] (with appropriate bounds)
        # For subgradient, we use y instead of a (evaluated at y)
        grad = np.zeros(n)
        for k in active:
            # (C_k * y)_j = y[k-j] for valid j
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            # grad[j] += 2 * y[k-j]
            # j ranges from j_start to j_end, k-j ranges from k-j_end to k-j_start
            idx = np.arange(j_start, j_end + 1)
            grad[idx] += 2.0 * y[k - idx]
        
        # Normalize gradient (it's a subgradient, scale matters)
        # Polyak step size: eta = (f(y) - f*) / ||g||^2
        # We don't know f*, use a lower bound
        # Lower bound: by Parseval, sum_k (a*a)[k] = sum_j a[j]^2 >= 1/n (by Cauchy-Schwarz)
        # So max_k (a*a)[k] >= (1/(2n-1)) * sum_j a[j]^2 >= 1/((2n-1)*n)
        f_lower = 1.0 / (n * (2 * n - 1))  # Very rough lower bound
        f_current = M_y
        eta = (f_current - f_lower) / (np.dot(grad, grad) + 1e-10)
        eta = min(eta, 1.0 / (2.0 * np.sqrt(t)))  # Cap with Nesterov rate
        
        # Nesterov update
        theta = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        # a_new = project(y - eta * grad)
        a_candidate = simplex_project(y - eta * grad)
        
        # Update y for next iteration
        y_new = a_candidate + (t / (t + 1.0)) * (a_candidate - a)
        
        a = a_candidate
        y = y_new
        
        # Check improvement
        if t % max(1, int(0.01 * n)) == 0:
            conv_a = fast_convolve(a)
            score = 2.0 * n * conv_a.max()
            if score < best_score - 1e-14:
                best_score = score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            if verbose and t % (max(1, int(0.1 * n))) == 0:
                elapsed = time.monotonic() - t0
                print(f"    Nesterov t={t} ({elapsed:.0f}s): best={best_score:.6f}")
            
            if no_improve > 100:
                # Reset momentum
                y = a.copy()
                no_improve = 0
    
    return best_a, best_score

def active_set_coord_descent(a0, time_limit, verbose=False):
    """
    Coordinate descent that focuses on coordinates contributing to the active peaks.
    Uses a priority queue to process the most impactful coordinates first.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and no_improve < 20:
        no_improve += 1
        sweep_improved = False
        
        # Identify active peaks
        tol = 0.05 * M
        active_peaks = np.where(conv >= M - tol)[0]
        
        # For each active peak, identify contributing coordinates
        # Process coordinates in order of their contribution to peaks
        coord_priority = np.zeros(n)
        for k in active_peaks:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                coord_priority[j] += conv[k] * a[j] * a[k - j]
        
        # Process in priority order
        order = np.argsort(-coord_priority)
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                return best_a, best_score
            if a[i] < 1e-15:
                continue
            
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            
            # Unaffected max
            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, conv[:k_start].max())
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, conv[k_end + 1:].max())
            
            affected = conv[k_start:k_end + 1]
            a_shifted = a  # a[k-i] for k = i, ..., i+n-1
            
            # Try multiple step sizes (both increase and decrease)
            best_delta = 0.0
            best_new_max = M + 1e-10
            
            # Binary search for optimal delta
            # The function new_max(delta) is piecewise linear (with one quadratic piece)
            # Try a few strategic points
            for factor in [0.95, 0.9, 0.85, 0.8, 0.97, 0.99, 1.01, 1.03, 1.05, 1.1, 1.15]:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue
                
                new_affected = affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    if new_affected is affected:
                        new_affected = new_affected.copy()
                    new_affected[idx_2i] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
                
                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_delta = delta
            
            if best_delta != 0.0 and best_new_max < M - 1e-12:
                a[i] = max(a[i] + best_delta, 0.0)
                
                # Update conv
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    if changes is (2.0 * best_delta * a_shifted):
                        changes = changes.copy()
                    changes[idx_2i] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                sweep_improved = True
        
        if sweep_improved:
            no_improve = 0
    
    return best_a, best_score

def interpolate_seq(f, new_n):

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def coordinate_descent(a0, time_limit, verbose=False, label="", n_trials=8):
    """
    Fast coordinate descent with incremental convolution updates.
    For each coordinate, tries multiple step sizes and picks the best.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            # Unaffected max
            unaff_max = -1e30
            if k_start > 0:
                unaff_max = conv[:k_start].max()
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, conv[k_end + 1:].max())

            affected = conv[k_start:k_end + 1]
            # a_shifted[m] = a[m] corresponds to a[k-i] where k = k_start + m = i + m
            a_shifted = a

            best_delta = 0.0
            best_new_max = M

            # Try multiple step sizes
            for trial in range(n_trials):
                if trial == 0:
                    delta = -0.5 * a[i]
                elif trial == 1:
                    delta = -0.25 * a[i]
                elif trial == 2:
                    delta = -0.1 * a[i]
                elif trial == 3:
                    delta = -0.05 * a[i]
                elif trial == 4:
                    delta = 0.05 * a[i]
                elif trial == 5:
                    delta = 0.1 * a[i]
                elif trial == 6:
                    delta = 0.25 * a[i]
                else:
                    delta = 0.5 * a[i]

                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx_2i] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-14:
                a[i] += best_delta
                a[i] = max(a[i], 0.0)

                # Update conv incrementally
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx_2i] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def pairwise_coord_descent(a0, time_limit, n_pairs_per_sweep=200, verbose=False):
    """
    Pairwise mass-transfer coordinate descent.
    Transfers mass from position i to position j (and vice versa)
    to reduce the peak of the autocorrelation.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit and no_improve < 10:
        sweeps += 1
        improved = False

        # Identify active peaks
        tol = 0.1 * M
        active_peaks = np.where(conv >= M - tol)[0]

        # Get coordinates that contribute to active peaks
        contributing = set()
        for k in active_peaks[:20]:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            contributing.update(range(j_start, j_end + 1))
        contributing = list(contributing)

        if len(contributing) < 2:
            break

        # Try random pairs from contributing set
        n_try = min(n_pairs_per_sweep, len(contributing) * 2)
        for _ in range(n_try):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, len(contributing))
            j = rng.integers(0, len(contributing))
            i = contributing[i]
            j = contributing[j]
            if i == j or a[i] < 1e-15 or a[j] < 1e-15:
                continue

            # Find the range of affected conv values
            # Changing a[i] by d and a[j] by -d affects:
            # conv[m] for m in [i, i+n-1] (from a[i]) and [j, j+n-1] (from a[j])
            k_min = min(i, j)
            k_max = max(i + n - 1, j + n - 1, 2 * n - 2)
            k_min = max(k_min, 0)
            k_max = min(k_max, 2 * n - 2)

            # Unaffected max
            unaff_max = -1e30
            if k_min > 0:
                unaff_max = conv[:k_min].max()
            if k_max < 2 * n - 2:
                unaff_max = max(unaff_max, conv[k_max + 1:].max())

            affected = conv[k_min:k_max + 1]

            # For delta: a[i] += delta, a[j] -= delta
            # Change in conv[m]:
            # From a[i]: +delta * a[m-i] (for m in [i, i+n-1])
            # From a[j]: -delta * a[m-j] (for m in [j, j+n-1])
            # Quadratic: +delta^2 if m = 2i, -delta^2 if m = 2j
            # Cross: -2*delta^2 if m = i+j

            # Build the sensitivity vector
            sens = np.zeros(k_max - k_min + 1)
            for m in range(k_min, k_max + 1):
                val = 0.0
                if i <= m < i + n:
                    val += a[m - i]
                if j <= m < j + n:
                    val -= a[m - j]
                sens[m - k_min] = val

            # Try a few delta values
            max_delta = 0.3 * min(a[i], a[j])
            best_delta = 0.0
            best_new_max = M

            for frac in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, -0.1, -0.2, -0.3, -0.5]:
                delta = frac * max_delta
                if a[i] + delta < -1e-15 or a[j] - delta < -1e-15:
                    continue

                new_affected = affected + delta * sens
                # Quadratic corrections
                if 2 * i >= k_min and 2 * i <= k_max:
                    new_affected = new_affected.copy()
                    new_affected[2 * i - k_min] += delta * delta
                if 2 * j >= k_min and 2 * j <= k_max:
                    if new_affected is affected:
                        new_affected = new_affected.copy()
                    new_affected[2 * j - k_min] -= delta * delta
                if i + j >= k_min and i + j <= k_max:
                    if new_affected is affected:
                        new_affected = new_affected.copy()
                    new_affected[i + j - k_min] -= 2.0 * delta * delta

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-14:
                a[i] += best_delta
                a[j] -= best_delta
                a[i] = max(a[i], 0.0)
                a[j] = max(a[j], 0.0)

                # Update conv
                conv[k_min:k_max + 1] += best_delta * sens
                if 2 * i >= k_min and 2 * i <= k_max:
                    conv[2 * i] += best_delta * best_delta
                if 2 * j >= k_min and 2 * j <= k_max:
                    conv[2 * j] -= best_delta * best_delta
                if i + j >= k_min and i + j <= k_max:
                    conv[i + j] -= 2.0 * best_delta * best_delta

                M = max(unaff_max, float(conv[k_min:k_max + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
        else:
            no_improve = 0

        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PairCD sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def nesterov_subgradient(a0, time_limit, verbose=False):
    """
    Nesterov-accelerated projected subgradient method for minimizing
    max_k (a*a)[k] subject to a >= 0, sum(a) = 1.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    best_a = a.copy()
    conv0 = fast_convolve(a)
    best_score = 2.0 * n * conv0.max()
    t0 = time.monotonic()
    t = 1
    no_improve = 0
    f_lower = 1.0 / (2 * n - 1)  # lower bound on max_k (a*a)[k]

    while time.monotonic() - t0 < time_limit:
        t += 1

        # Compute subgradient at y
        conv_y = fast_convolve(y)
        M_y = float(conv_y.max())

        # Active peaks (within 2% of max)
        tol = 0.02 * M_y
        active = np.where(conv_y >= M_y - tol)[0]

        # Compute subgradient
        grad = np.zeros(n)
        for k in active:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            grad[idx] += 2.0 * y[k - idx]

        # Step size (Polyak-like with decay)
        grad_norm_sq = float(np.dot(grad, grad))
        if grad_norm_sq < 1e-20:
            break
        eta = (M_y - f_lower) / grad_norm_sq
        eta = min(eta, 1.0 / (2.0 * np.sqrt(t)))
        eta = max(eta, 1e-10)

        # Nesterov update
        a_new = simplex_project(y - eta * grad)
        theta = t / (t + 1.0)
        y = a_new + theta * (a_new - a)
        a = a_new

        # Check improvement periodically
        check_interval = max(1, n // 50)
        if t % check_interval == 0:
            conv_a = fast_convolve(a)
            score = 2.0 * n * conv_a.max()
            if score < best_score - 1e-14:
                best_score = score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            if verbose and t % (check_interval * 20) == 0:
                elapsed = time.monotonic() - t0
                print(f"    Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}")

            if no_improve > 200:
                # Reset momentum
                y = a.copy()
                no_improve = 0

    return best_a, best_score


def spectral_equalize(a0, time_limit, verbose=False):
    """
    Frequency-domain spectral equalization.
    Iteratively compresses the dynamic range of the power spectrum
    to flatten the autocorrelation.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Compute spectrum
        fft_size = 1 << (2 * n - 1).bit_length()
        A = rfft(a, fft_size)
        mag = np.abs(A)

        # Compress dynamic range (excluding DC)
        if it % 20 == 0 or it <= 5:
            conv = fast_convolve(a)
            score = 2.0 * n * conv.max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

        if mag.size <= 1:
            break

        # Target: make |A[1:]| as uniform as possible
        non_dc = mag[1:]
        target = np.median(non_dc)
        if target < 1e-15:
            break

        # Compress: new_mag = target + (mag - target) * compression
        compression = 0.85  # Start strong, weaken over time
        comp = 0.85 - 0.005 * min(it, 100)
        comp = max(comp, 0.5)

        new_mag = mag.copy()
        new_mag[1:] = target + (mag[1:] - target) * comp
        # Preserve phase
        phase = np.angle(A)
        A_new = new_mag * np.exp(1j * phase)
        A_new[0] = A[0]  # Preserve DC

        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        ss = a_new.sum()
        if ss < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / ss

        # Use blending to avoid instability
        blend = 0.3
        a = (1 - blend) * a + blend * a_new

    conv = fast_convolve(a)
    score = 2.0 * n * conv.max()
    if score < best_score:
        best_score = score
        best_a = a.copy()

    return best_a, best_score


def perturbed_restart(a0, rng, magnitude=0.02):
    """Create a perturbed version of a0 to escape local minima."""
    n = len(a0)
    a = a0.copy()
    # Add random perturbation
    noise = rng.standard_normal(n) * magnitude
    a = a + noise
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    return a / s


def propose_candidate():
    """
    Multi-phase optimization combining coordinate descent, pairwise updates,
    Nesterov subgradient, spectral equalization, and random restarts.
    """
    deadline = time.monotonic() + 1785.0
    rng = np.random.default_rng(2024)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = 2.0
        print(f"Default: n=1024, score=2.0")

    # === Phase 1: Spectral equalization ===
    print(f"\n=== Phase 1: Spectral Equalization (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 800
    t1 = min(rem, 120.0)
    if t1 > 20:
        a_se, s_se = spectral_equalize(best_seq, t1, verbose=True)
        if s_se < best_score:
            best_score = s_se
            best_seq = a_se.copy()
            print(f"  After spectral eq: {best_score:.8f}")

    # === Phase 2: Coordinate descent (main workhorse) ===
    print(f"\n=== Phase 2: Coordinate Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 600
    t2 = min(rem, 350.0)
    if t2 > 20:
        a_cd, s_cd = coordinate_descent(best_seq, t2, verbose=True, label="main", n_trials=10)
        if s_cd < best_score:
            best_score = s_cd
            best_seq = a_cd.copy()
            print(f"  After CD: {best_score:.8f}")

    # === Phase 3: Pairwise mass transfer ===
    print(f"\n=== Phase 3: Pairwise CD (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 500
    t3 = min(rem, 250.0)
    if t3 > 20:
        a_pw, s_pw = pairwise_coord_descent(best_seq, t3, n_pairs_per_sweep=300, verbose=True)
        if s_pw < best_score:
            best_score = s_pw
            best_seq = a_pw.copy()
            print(f"  After PairCD: {best_score:.8f}")

    # === Phase 4: Nesterov subgradient for global exploration ===
    print(f"\n=== Phase 4: Nesterov (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 400
    t4 = min(rem, 250.0)
    if t4 > 20:
        a_ns, s_ns = nesterov_subgradient(best_seq, t4, verbose=True)
        if s_ns < best_score:
            best_score = s_ns
            best_seq = a_ns.copy()
            print(f"  After Nesterov: {best_score:.8f}")

            # Follow up with coordinate descent
            rem = deadline - time.monotonic() - 300
            t4b = min(rem, 100.0)
            if t4b > 15:
                a_cd2, s_cd2 = coordinate_descent(best_seq, t4b, label="post_ns", n_trials=8)
                if s_cd2 < best_score:
                    best_score = s_cd2
                    best_seq = a_cd2.copy()
                    print(f"  After post-NS CD: {best_score:.8f}")

    # === Phase 5: Random restarts + re-optimization ===
    print(f"\n=== Phase 5: Random Restarts (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 200
    if rem > 60:
        n_restarts = 5
        t_per_restart = rem / n_restarts
        for r in range(n_restarts):
            if time.monotonic() > deadline - 100:
                break
            mag = 0.01 + 0.01 * r
            a_pert = perturbed_restart(best_seq, rng, magnitude=mag)
            # Quick CD on perturbed
            t_alloc = min(t_per_restart * 0.6, 40.0)
            a_opt, s_opt = coordinate_descent(a_pert, t_alloc, label=f"restart{r}", n_trials=6)
            if s_opt < best_score:
                best_score = s_opt
                best_seq = a_opt.copy()
                print(f"  Restart {r}: {best_score:.8f}")

    # === Phase 6: Fine polishing ===
    print(f"\n=== Phase 6: Fine Polishing (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 80
    if rem > 20:
        t6 = min(rem, 150.0)
        a_fine, s_fine = fine_coordinate_descent(best_seq, t6, verbose=True, label="fine")
        if s_fine < best_score:
            best_score = s_fine
            best_seq = a_fine.copy()
            print(f"  After fine CD: {best_score:.8f}")

    # === Phase 7: Final pairwise + CD ===
    rem = deadline - time.monotonic() - 40
    if rem > 30:
        t7a = min(rem * 0.5, 60.0)
        a_pw2, s_pw2 = pairwise_coord_descent(best_seq, t7a, n_pairs_per_sweep=400)
        if s_pw2 < best_score:
            best_score = s_pw2
            best_seq = a_pw2.copy()

        rem = deadline - time.monotonic() - 20
        if rem > 15:
            t7b = min(rem, 50.0)
            a_fin, s_fin = coordinate_descent(best_seq, t7b, label="final", n_trials=12)
            if s_fin < best_score:
                best_score = s_fin
                best_seq = a_fin.copy()

    # === Phase 8: Try different length ===
    rem = deadline - time.monotonic() - 30
    if rem > 40:
        n_cur = len(best_seq)
        for alt_n in [n_cur + 128, n_cur - 128, n_cur + 64]:
            if alt_n < 256 or alt_n > 2560:
                continue
            if time.monotonic() > deadline - 20:
                break
            a_alt = interpolate_seq(best_seq, alt_n)
            t_alt = min(rem * 0.4, 60.0)
            a_opt, s_opt = coordinate_descent(a_alt, t_alt, label=f"alt{alt_n}", n_trials=8)
            if s_opt < best_score:
                best_score = s_opt
                best_seq = a_opt.copy()
                print(f"  Alt n={alt_n}: {best_score:.8f}")

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
