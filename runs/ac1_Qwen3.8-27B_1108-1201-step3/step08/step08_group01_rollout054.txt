    # This is like a "valid" cross-correlation of w and a
    # g = np.correlate(w, a, mode='valid') would give g[i] = sum_{m=0}^{len(a)-1} w[i+m]*a[m]
    # But np.correlate is slow for large arrays. Let's use FFT.
    
    # g[i] = sum_m a[m] * w[i+m]
    # This is (a * w_reversed)[i + n - 1] in convolution notation? Let me think...
    # Actually, it's the cross-correlation: (a ⊛ w)[i] = sum_m a[m] * w[i+m]
    # In terms of convolution: (a ⊛ w)[i] = (reverse(a) * w)[i + n - 1]
    # where * denotes linear convolution
    
    # Let's just use FFT-based approach
    fft_size = 1 << (2 * n - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    # cross-correlation via FFT:
    # (a ⊛ w)[i] = sum_j a[j] * conj(w[j-i]) for complex
    # For real: (a ⊛ w)[i] = sum_j a[j] * w[j-i]
    # Hmm, this is getting confusing. Let me just do it directly for n=1024.
    
    # For n=1024, direct computation is O(n^2) = about 10^6, which is fine.
    # g[i] = (2/T) * sum_{m=0}^{n-1} w[i+m] * a[m] for 0 ≤ i+m < 2n-1
    
    # Actually, let me use the cross-correlation definition properly.
    # numpy.correlate(w, a, 'valid')[i] = sum_{m=0}^{len(a)-1} w[i+m] * a[m]
    # That's exactly what we want!
    
    g_direct = np.correlate(w, a, mode='valid')  # length = (2n-1) - (n-1) = n
    g = (2.0 / T) * g_direct
    
    # grad_x[i] = a[i] * (g[i] - a[i] * sum_j g[j])
    sum_g_a = np.dot(a, g)
    grad = a * (g - a * sum_g_a)
    
    return obj, grad

def l_bfgs_optimize(a_init, n, T_schedule, max_iter=2000, verbose=False, label=""):
    """
    Optimize using L-BFGS-B on the softmax-parameterized softened objective.
    """
    # Initialize in log-space
    a_init = np.maximum(a_init, 1e-30)
    x0 = np.log(a_init / a_init.sum())  # log of normalized
    
    T_list = T_schedule
    best_x = x0.copy()
    best_obj = float('inf')
    
    t0 = time.monotonic()
    
    for T in T_list:
        result = minimize(
            softened_obj_and_grad,
            x0,
            method='L-BFGS-B',
            jac=True,
            options={'maxiter': max_iter, 'maxfun': max_iter * 3, 'ftol': 1e-16, 'gtol': 1e-10, 'iprint': 0}
        )
        if result.success or result.fun < best_obj:
            if result.fun < best_obj:
                best_obj = result.fun
                best_x = result.x.copy()
        x0 = result.x
        
        elapsed = time.monotonic() - t0
        if verbose:
            a_cur = softmax(result.x)
            true_obj = 2.0 * n * fast_convolve(a_cur).max()
            print(f"    L-BFGS {label} T={T:.6f} ({elapsed:.0f}s): softened={result.fun:.8f}, true={true_obj:.6f}")
    
    return softmax(best_x), softmax(x0)

def fast_coord_descent(a0, time_limit, n_trials=8, verbose=False, label=""):
    """
    Fast coordinate descent using sensitivity analysis.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    sum_a = 1.0
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve_sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        sweep_start_score = best_score

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            affected = conv[k_start:k_end + 1].copy()
            a_shifted = a[0:n].copy()

            best_delta = 0.0
            best_new_max = M

            for trial in range(n_trials):
                if trial == 0:
                    delta = -0.4 * a[i]
                elif trial == 1:
                    delta = -0.2 * a[i]
                elif trial == 2:
                    delta = -0.1 * a[i]
                elif trial == 3:
                    delta = -0.05 * a[i]
                elif trial == 4:
                    delta = 0.05 * a[i]
                elif trial == 5:
                    delta = 0.1 * a[i]
                elif trial == 6:
                    delta = 0.2 * a[i]
                else:
                    delta = 0.4 * a[i]

                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx_2i] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-16:
                a[i] += best_delta
                a[i] = max(a[i], 0.0)
                sum_a += best_delta

                ks = np.arange(k_start, k_end + 1)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx_2i] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M / (sum_a * sum_a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > 30:
                break
        else:
            no_improve_sweeps = 0

        if verbose and sweeps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f} M={M:.8f}")

    return best_a, best_score, sweeps

def fine_coord_descent(a0, time_limit, verbose=False, label=""):
    """Fine coordinate descent with smaller step sizes for polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()

    if s < 1e-15:

    sum_a = 1.0
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            affected = conv[k_start:k_end + 1]
            a_shifted = a[0:n]

            best_delta = 0.0
            best_new_max = M

            for factor in [0.97, 0.94, 0.91, 0.985, 1.015, 1.03, 1.06, 1.09]:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx_2i] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
                else:
                    new_affected = new_affected.copy()

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-17:
                a[i] += best_delta
                a[i] = max(a[i], 0.0)
                sum_a += best_delta

                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx_2i] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M / (sum_a * sum_a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved and sweeps > 10:
            break

    return best_a, best_score, sweeps

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def init_fourier(n, rng):
    """Initialize using random Fourier superposition."""
    n_harmonics = int(rng.integers(3, 18))
    amplitudes = rng.random(n_harmonics) * 0.35
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = np.arange(1, n_harmonics + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def init_u_shape(n, rng):
    """Initialize with U-shape (more mass at edges)."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    alpha = rng.uniform(0.5, 2.0)
    a = 1.0 + alpha * np.cos(np.pi * t) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def init_arcosine(n, rng):
    """Initialize with arccosine-based shape."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    a = np.abs(np.arccos(2.0 * t - 1.0))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS-B optimization on softmax-parameterized log-sum-exp softened objective
2. Adaptive geometric temperature cooling
3. Multiple diverse initializations and random restarts with perturbation
4. Multi-length exploration

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax(x):
    """Stable softmax."""
    x_max = x.max()
    e = np.exp(x - x_max)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def softened_obj_and_grad(x, T):
    """
    Compute the softened objective 2*n*T*log(sum exp(conv(a,a)/T)) and its gradient
    with respect to x where a = softmax(x).
    Returns (objective, gradient).
    """
    a = softmax(x)
    n = len(a)
    conv = fast_convolve(a)

    # Compute weights w[k] = exp(conv[k]/T) / sum_j exp(conv[j]/T)
    conv_scaled = conv / T
    conv_scaled_max = conv_scaled.max()
    exp_scaled = np.exp(conv_scaled - conv_scaled_max)
    w = exp_scaled / exp_scaled.sum()

    # Objective: 2*n*T*log(sum exp(conv/T))
    log_Z = conv_scaled_max + np.log(exp_scaled.sum())
    obj = 2.0 * n * T * log_Z

    # Compute g[i] = (2/T) * sum_{m=0}^{n-1} w[i+m] * a[m]
    # This is np.correlate(w, a, mode='valid') scaled by 2/T
    g = (2.0 / T) * np.correlate(w, a, mode='valid')

    # grad_x[i] = a[i] * (g[i] - a[i] * sum_j a[j]*g[j])
    dot_ago = np.dot(a, g)
    grad = a * (g - a * dot_ago)

    return obj, grad


def fast_coord_descent(a0, time_limit, n_trials=8, verbose=False, label=""):

    Fast coordinate descent using sensitivity analysis per coordinate update.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()

    if s < 1e-15:

    else:
        a = a / s

    conv = fast_convolve(a)

    sum_a = 1.0

    no_improve_sweeps = 0

        sweeps += 1
        improved = False

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            a_shifted = a[0:n].copy()

                a[i] = max(a[i], 0.0)

                if 2 * i >= k_start and 2 * i <= k_end:

                    best_a = a.copy()

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_fourier(n, rng):
    """Initialize using random Fourier superposition."""
    n_harmonics = int(rng.integers(3, 18))
    amplitudes = rng.random(n_harmonics) * 0.35
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = np.arange(1, n_harmonics + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_u_shape(n, rng):
    """Initialize with U-shape."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    alpha = rng.uniform(0.5, 2.0)
    a = 1.0 + alpha * np.cos(np.pi * t) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_bimodal(n, rng):
    """Initialize with a bimodal distribution (mass concentrated at two peaks)."""
    center1 = n * rng.uniform(0.2, 0.4)
    center2 = n * rng.uniform(0.6, 0.8)
    width = n * rng.uniform(0.05, 0.15)
    t = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((t - center1) / width) ** 2) + np.exp(-0.5 * ((t - center2) / width) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def check_and_update(a, best_seq, best_score, label=""):

    s = a.sum()

        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def run_lbfgs(a_init, T, maxiter=400, gtol=1e-12, ftol=1e-18):
    """Run L-BFGS-B on the softened objective from a given starting point."""
    a_safe = np.maximum(a_init, 1e-30)
    a_norm = a_safe / a_safe.sum()
    x0 = np.log(a_norm)

    try:
        result = minimize(
            softened_obj_and_grad,
            x0,
            args=(T,),
            method='L-BFGS-B',
            jac=True,
            options={'maxiter': maxiter, 'maxfun': maxiter * 4,
                     'ftol': ftol, 'gtol': gtol, 'iprint': 0}
        )
        a_out = softmax(result.x)
        a_out = np.maximum(a_out, 0)
        s = a_out.sum()
        if s < 1e-15:
            return a_init.copy()
        return a_out / s
    except Exception as e:
        print(f"    L-BFGS error: {e}")
        return a_init.copy()

                a_new = run_lbfgs(a_curr, T, maxiter=300, gtol=1e-12, ftol=1e-18)
                score_new = 2.0 * n * fast_convolve(a_new).max()
                score_cur = 2.0 * n * fast_convolve(a_curr).max()
                if score_new <= score_cur + 1e-12:
                    a_curr = a_new

            best_seq, best_score = check_and_update(a_curr, best_seq, best_score, f"p1_n{n}_i{init_idx}")

            # CD polish
            rem = deadline - time.monotonic() - 300
            if rem > 10:
                a_pol, s_pol, _ = fast_coord_descent(a_curr, min(15.0, rem * 0.1), n_trials=10, label=f"p1pol_n{n}")
                best_seq, best_score = check_and_update(a_pol, best_seq, best_score, f"p1pol_n{n}_i{init_idx}")

    # === Phase 2: Intensive optimization on best ===
    print(f"\n=== Phase 2: Intensive Opt (best={best_score:.6f}) ===")
    t_phase2 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)

        # Fine L-BFGS cooling on the best
        a_curr = best_seq.copy()
        conv_cur = fast_convolve(a_curr)
        M_cur = conv_cur.max()
        T_schedule2 = np.geomspace(M_cur * 0.25, M_cur * 0.01, 8)

        for T in T_schedule2:
            if time.monotonic() - t_phase2 > 250:
                break
            a_new = run_lbfgs(a_curr, T, maxiter=500, gtol=1e-14, ftol=1e-20)
            score_new = 2.0 * n_best * fast_convolve(a_new).max()
            if score_new < best_score:
                best_score = score_new
                best_seq = a_new.copy()
                print(f"  NEW BEST [ph2_T{T:.8f}]: {score_new:.6f}")
            a_curr = a_new

        # CD polish after L-BFGS
        rem = deadline - time.monotonic() - 350
        if rem > 30:
            a_cd2, s_cd2, _ = fast_coord_descent(best_seq, min(100.0, rem * 0.4), n_trials=12, verbose=True, label="ph2_cd")
            best_seq, best_score = check_and_update(a_cd2, best_seq, best_score, "ph2_cd")

        # Random restarts
        for r in range(5):
            if time.monotonic() - t_phase2 > 400:
                break
            sigma = 0.005 + 0.003 * r
            a_pert = best_seq + rng.standard_normal(n_best) * sigma / np.sqrt(n_best)
            a_pert = np.maximum(a_pert, 0)
            sp = a_pert.sum()
            if sp < 1e-10:
                continue
            a_pert = a_pert / sp

            conv_p = fast_convolve(a_pert)
            T_r = conv_p.max() * 0.2
            a_new = run_lbfgs(a_pert, T_r, maxiter=300, gtol=1e-12, ftol=1e-16)
            # Quick CD
            rem_r = deadline - time.monotonic() - 200
            if rem_r > 5:
                a_new, _, _ = fast_coord_descent(a_new, min(20.0, rem_r * 0.15), n_trials=10, label=f"restart{r}")
            best_seq, best_score = check_and_update(a_new, best_seq, best_score, f"restart{r}")

    # === Phase 3: Try alternative lengths ===
    print(f"\n=== Phase 3: Alt Lengths (best={best_score:.6f}) ===")
    t_phase3 = time.monotonic()

    if best_seq is not None:
        n_best = len(best_seq)
        alt_ns = [n_best + 128, n_best - 128, n_best + 256, n_best - 256, 2048]
        alt_ns = [x for x in alt_ns if 256 <= x <= 3000 and x != n_best]

        for alt_n in alt_ns[:4]:
            if time.monotonic() - t_phase3 > 250:
                break

            a_init = interpolate_seq(best_seq, alt_n)
            # Quick CD
            a_cd, s_cd, _ = fast_coord_descent(a_init, 15.0, n_trials=8, label=f"alt{alt_n}_cd")
            # L-BFGS
            conv_c = fast_convolve(a_cd)
            T_alt = conv_c.max() * 0.2
            a_lb = run_lbfgs(a_cd, T_alt, maxiter=300, gtol=1e-12, ftol=1e-16)
            # Polish
            rem_alt = deadline - time.monotonic() - 150
            if rem_alt > 5:
                a_lp, _, _ = fast_coord_descent(a_lb, min(25.0, rem_alt * 0.2), n_trials=10, label=f"alt{alt_n}_pol")
                a_lb = a_lp
            best_seq, best_score = check_and_update(a_lb, best_seq, best_score, f"alt_n{alt_n}")

    # === Phase 4: Final polishing ===
    print(f"\n=== Phase 4: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        # Fine CD
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t_fine = min(rem * 0.4, 80.0)
            a_fine, s_fine, _ = fine_coord_descent(best_seq, t_fine, verbose=True, label="finet")
            best_seq, best_score = check_and_update(a_fine, best_seq, best_score, "final_fine_cd")

        # L-BFGS with very low T
        rem = deadline - time.monotonic() - 50
        if rem > 15:
            conv_f = fast_convolve(best_seq)
            T_final = max(conv_f.max() * 0.02, 1e-7)
            a_f = run_lbfgs(best_seq, T_final, maxiter=400, gtol=1e-15, ftol=1e-20)
            best_seq, best_score = check_and_update(a_f, best_seq, best_score, "final_lbfgs")

        # Final CD
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            a_ff, s_ff, _ = fast_coord_descent(best_seq, min(rem * 0.7, 40.0), n_trials=14, label="last")
            best_seq, best_score = check_and_update(a_ff, best_seq, best_score, "last_cd")

        # Ultra-fine CD
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_uf, s_uf, _ = fine_coord_descent(best_seq, min(rem * 0.8, 15.0), label="ultra")
            best_seq, best_score = check_and_update(a_uf, best_seq, best_score, "ultra_fine")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
