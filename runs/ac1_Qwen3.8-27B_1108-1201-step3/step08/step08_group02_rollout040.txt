        k = int(peaks[row_i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)

    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):

            if stuck > 30:

                g2 = solve_lp(f, n, fewer_peaks, M)

                    f /= f.sum()
                    stuck = 0
                    continue

def coordinate_descent(f_init, time_limit, label="cd"):
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_b = fast_convolve(f).max()
    multipliers = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0]
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue
            best_val = f[j]
            best_obj = best_b
            for mult in multipliers:
                f_trial = f.copy()
                f_trial[j] = f[j] * mult
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_val = f[j] * mult
            if best_val != f[j]:
                f[j] = best_val
                f /= f.sum()
                if best_obj < best_b - 1e-14:
                    best_b = best_obj
                    improved = True
        if not improved:
            break
    
    return f, best_b

def interpolate_to_length(f, new_n):
    n = len(f)
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.ones(n) * target_sum / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp(f, n, peaks, M):
    K = len(peaks)
    if K == 0:
        return None, None
    A = np.zeros((K, n))
    for row_i in range(K):
        k = int(peaks[row_i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]

    g = cp.Variable(n)
    c = A @ g - M
    constraints = [g >= 0, c <= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    solved = False
    for solver in [cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, max_iters=2000000)
            solved = True
            break
        except Exception:
            continue

    if not solved or g.value is None:
        return None, None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None, None
    dual_vals = None
    try:
        dual_vals = np.maximum(np.asarray(c.dual_value, dtype=np.float64), 0.0)
    except Exception:
        pass
    return g_val, dual_vals


def smoothed_grad_descent(f_init, time_limit, label="sgd"):
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    v = np.zeros(n)
    best_f = f.copy()
    best_b = fast_convolve(f).max()
    lr = 0.015
    tau = 30.0
    beta = 0.9
    no_improve = 0
    it = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        scaled = tau * b
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(b))
            w[np.argmax(b)] = 1.0
        else:
            w /= w_sum

        grad = np.zeros(n)
        active_idx = np.where(w > 1e-7)[0]
        for k in active_idx:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * w[k] * f[k - hi:k - lo + 1][::-1]

        v = beta * v - lr * grad
        f = project_simplex(f + v)

        if it % 100 == 0:
            obj = fast_convolve(f).max()
            if obj < best_b - 1e-12:
                best_b = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 4:
                    lr *= 0.92
                    no_improve = 0

            tau = max(5.0, tau * 0.995)

            if no_improve > 15:
                f = best_f + rng.standard_normal(n) * 0.003 * best_f
                f = np.maximum(f, 0.0)
                f /= f.sum()
                v = np.zeros(n)
                lr = 0.015
                no_improve = 0

            if it % 500 == 0:
                score = 2.0 * n * best_b
                print(f"  {label} it={it} obj={best_b:.10f} score={score:.6f} "
                      f"lr={lr:.4f} tau={tau:.1f} t={time.monotonic() - t0:.0f}s")

    return best_f, best_b


def lp_optimize_dynamic(f_init, time_limit, label="lp", max_peaks=500,
                         min_peaks=20, initial_margin=0.02, final_margin=0.001):
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_b = fast_convolve(f).max()
    stuck = 0
    it = 0
    rng = np.random.default_rng(42)
    total_it_expected = max(1, int(time_limit / 0.15))

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        progress = min(1.0, it / (total_it_expected * 0.7))
        margin = initial_margin + (final_margin - initial_margin) * progress
        margin = max(final_margin, margin)

        threshold = M * (1.0 - margin)
        peaks = np.where(b >= threshold)[0]

        if len(peaks) > max_peaks:
            peaks = np.argpartition(b, -max_peaks)[-max_peaks:]
        if len(peaks) < min_peaks:
            peaks = np.argpartition(b, -min_peaks)[-min_peaks:]

        g, dual_vals = solve_lp(f, n, peaks, M)

        if g is None:
            stuck += 1
            if stuck > 40:
                break
            continue

        S_g = g.sum()
        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > 10:
                fewer_peaks = np.argpartition(b, -max(5, len(peaks) // 4))[-max(5, len(peaks) // 4):]
                g2, _ = solve_lp(f, n, fewer_peaks, M)
                if g2 is not None and g2.sum() > 1.0 + 1e-6:
                    g = g2
                    S_g = g.sum()
                else:
                    if stuck > 40:
                        break
                    if stuck % 10 == 0:
                        f = f + rng.standard_normal(n) * 0.001 * f
                        f = np.maximum(f, 0.0)
                        f /= f.sum()
                    continue
            else:
                continue

        h = g / S_g
        best_t = 0.0
        best_obj = M
        t_candidates = [0.03, 0.05, 0.08, 0.12, 0.18, 0.25, 0.35, 0.5, 0.7, 0.9]
        for t in t_candidates:
            f_trial = (1.0 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_t = t

        if best_t > 0:
            for dt in [best_t * 0.7, best_t * 1.3, best_t * 1.6, best_t * 0.4]:
                if dt <= 0 or dt > 0.95:
                    continue
                f_trial = (1.0 - dt) * f + dt * h
                f_trial = np.maximum(f_trial, 0.0)
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_t = dt

        if best_t <= 0:
            stuck += 1
            if stuck > 40:
                break
            continue

        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        f /= f.sum()

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_b - 1e-12:
            best_b = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1

        if it % 20 == 0:
            score = 2.0 * n * best_b
            print(f"  {label} it={it} obj={best_b:.10f} score={score:.6f} "
                  f"peaks={len(peaks)} S_g={S_g:.4f} margin={margin:.4f} "
                  f"t={time.monotonic() - t0:.0f}s")

        if stuck > 25 and stuck % 8 == 0:
            f = best_f + rng.standard_normal(n) * 0.002 * best_f
            f = np.maximum(f, 0.0)
            f /= f.sum()

    return best_f, best_b


def coordinate_descent(f_init, time_limit, label="cd"):
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_b = fast_convolve(f).max()
    multipliers = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0]

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue
            best_val = f[j]
            best_obj = best_b
            for mult in multipliers:
                f_trial = f.copy()
                f_trial[j] = f[j] * mult
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_val = f[j] * mult
            if best_val != f[j]:
                f[j] = best_val
                f /= f.sum()
                if best_obj < best_b - 1e-14:
                    best_b = best_obj
                    improved = True
        if not improved:
            break

    return f, best_b


def interpolate_to_length(f, new_n):
    n = len(f)
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new
