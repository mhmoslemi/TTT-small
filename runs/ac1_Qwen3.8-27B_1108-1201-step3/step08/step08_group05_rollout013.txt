def flat_top_window(n):
    w = np.ones(n)
    a = [0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368]
    for i, ai in enumerate(a):
        w -= ai * np.cos(2 * np.pi * i * np.arange(n) / max(1, n - 1) - i * np.pi)
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s

"""
Auto-convolution peak minimization using LP with adaptive constraint generation,
heavy-ball subgradient polish, symmetric perturbation search, and multi-length
exploration. Targets score < 1.5027 (lower is better).
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(f, b, n_top=30, temperature=1.0):
    """Compute softmax-weighted subgradient of max auto-conv peak."""
    n = len(f)
    if n_top < len(b):
        top_idx = np.argpartition(b, -n_top)[-n_top:]
    else:
        top_idx = np.arange(len(b))

    b_top = b[top_idx]
    b_max = b_top.max()
    if b_max < 1e-15:
        return np.zeros(n), top_idx

    inv_temp = 1.0 / max(temperature * b_max, 1e-15)
    exp_vals = np.exp((b_top - b_max) * inv_temp)
    w_top = exp_vals / exp_vals.sum()

    g = np.zeros(n)
    for i, k in enumerate(top_idx):
        wi = w_top[i]
        if wi < 1e-13:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        g[idx] += wi * f[k - idx]

    return g, top_idx


def solve_lp(f, n, active_lags, M):
    """Solve: max sum(g) s.t. g >= 0, (f*g)[k] <= M for k in active_lags."""
    try:
        A = np.zeros((len(active_lags), n))
        for i, k in enumerate(active_lags):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A[i, lo:hi + 1] = f[lo:hi + 1]

        g_var = cp.Variable(n)
        constraints = [g_var >= 0, A @ g_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)

        prob.solve(solver=cp.PDLP, max_iters=300000)
        if g_var.value is None:
            return None, None
        g = np.maximum(g_var.value, 0.0)
        if np.any(np.isnan(g)) or np.any(np.isinf(g)):
            return None, None
        dual_vals = None
        for con in prob.constraints:
            try:
                if con.dual_value is not None:
                    dual_vals = np.asarray(con.dual_value).flatten()
                    break
            except Exception:
                pass
        return g, dual_vals
    except Exception:
        return None, None

            grad, _ = compute_gradient(f, b, n_top=n_top, temperature=temperature)

            lr = lr_init / (1.0 + it * 0.002)
            v = beta * v - lr * grad
            f_new = project_simplex(f + v)
            f = f_new

        # Check improvement
        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
            temperature = min(temperature * 1.1, 2.0)
        else:
            no_improve += 1
            temperature = max(temperature * 0.95, 0.1)
            if no_improve > 3:
                # Perturb to escape
                f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                v = np.zeros(n)
                no_improve = 0

        if it % 500 == 0:
            print(f"    hb iter={it} obj={best_obj:.12f} "
                  f"score={2.0 * n * best_obj:.6f} t={time.monotonic() - t0:.0f}s")

    return best_f, best_obj


def perturbation_search(f_init, time_limit):
    """Adaptive-scale random perturbation search."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    f = f / f.sum()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    rng = np.random.default_rng(555)
    scale = 0.008
    min_scale = 0.0005

    while time.monotonic() - t0 < time_limit:
        # Try a batch of perturbations
        n_batch = min(500, max(50, int((time_limit - (time.monotonic() - t0)) * 5)))
        for _ in range(n_batch):
            if time.monotonic() - t0 > time_limit:
                break
            pert = rng.standard_normal(n) * scale
            f_trial = f + pert
            f_trial = project_simplex(f_trial)
            obj = fast_convolve(f_trial).max()
            if obj < best_obj - 1e-14:
                f = f_trial
                best_f = f.copy()
                best_obj = obj
                scale = min(scale * 1.03, 0.03)
            else:
                scale = max(scale * 0.995, min_scale)

    return best_f, best_obj


def coordinate_descent_polish(f_init, time_limit):
    """Coordinate descent: optimize each coefficient one at a time (cyclic with random restarts)."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    f = f / f.sum()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    rng = np.random.default_rng(31337)

    while time.monotonic() - t0 < time_limit:
        # Cyclic pass or random pass
        if rng.random() < 0.5:
            order = np.arange(n)
        else:
            order = rng.permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj

            # Try modifying f[j]
            # The contribution of f[j] to the auto-convolution at lag k is f[j] * f[k-j]
            # We want to find the best f[j] that minimizes the peak
            # Simplified: try a few values
            old_val = f[j]
            if old_val < 1e-12:
                # Try small positive values
                candidates = [1e-6, 1e-5, old_val * 2]
            else:
                candidates = [0.0, old_val * 0.5, old_val * 0.8, old_val * 1.2, old_val * 1.5]

            best_val = old_val
            best_trial = best_obj

            f_old = f.copy()
            for cv in candidates:
                f[j] = cv
                # Renormalize
                s = f.sum()
                if s < 1e-10:
                    f = f_old.copy()
                    continue
                f_norm = f / s
                obj = fast_convolve(f_norm).max()
                if obj < best_trial - 1e-14:
                    best_trial = obj
                    best_val = cv
            f = f_old.copy()
            f[j] = best_val
            s = f.sum()
            if s > 1e-10:
                f = f / s

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()

    return best_f, best_obj


def flat_top_window(n):
    w = np.ones(n)
    a = [0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368]
    for i, ai in enumerate(a):
        w -= ai * np.cos(2 * np.pi * i * np.arange(n) / max(1, n - 1) - i * np.pi)
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def hann_window(n):
    w = 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n) / max(1, n)))
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def interpolate_sequence(src, target_n):
    """Interpolate a sequence to a new length."""
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s < 1e-10:
        return np.ones(target_n) / target_n
    return result / s


def try_symmetrize(f):
    """Try symmetrizing and check if it improves."""
    f_sym = (f + f[::-1]) / 2.0
    f_sym = np.maximum(f_sym, 0.0)
    s = f_sym.sum()
    if s < 1e-10:
        return f, fast_convolve(f).max()
    f_sym /= s
    obj_orig = fast_convolve(f).max()
    obj_sym = fast_convolve(f_sym).max()
    if obj_sym < obj_orig:
        return f_sym, obj_sym
    return f, obj_orig
