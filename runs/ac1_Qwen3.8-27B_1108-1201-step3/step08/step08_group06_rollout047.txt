Alternatively, we could use L-BFGS which only needs matrix-vector products. The Hessian-vector product for `c_k` is:
`Q_k * v` where `(Q_k v)_i = v_{k-i}` (with 0 padding). This is O(n) per k, or O(n * |A|) for the full Hessian-vector product. With FFT, we can do it in O(n log n).

So L-BFGS is feasible! Let me implement it.

Actually, the smoothed objective `f_T(a) = 2n * T * log(sum exp(c_k/T))` has:
- Gradient: `2n * sum_k w_k * ∇c_k` where `∇c_k = 2 * (a shifted)` (O(n) per k, or O(n log n) via FFT)
- Hessian-vector: `2n * [sum_k w_k * (Q_k v) - 2/T * sum_k w_k * (∇c_k^T v) * ∇c_k]` (the second term is the "covariance" correction from the softmax)

This is all computable in O(n log n).

Let me implement L-BFGS on the smoothed objective with the simplex constraint handled by:
- Optimization in the "interior" (using log-barrier or just ignoring the constraint and projecting)
- Or, optimization in log-space (softmax parameterization)

I think the log-space approach is cleaner. Let `a_i = exp(z_i) / sum_j exp(z_j)`. Then we optimize over `z` (unconstrained) and the non-negativity and sum-to-1 are automatically satisfied.

The gradient in z-space:
`∂f/∂z_i = a_i * (∂f/∂a_i - sum_j a_j * ∂f/∂a_j)`

(Let me verify: `f = f(a(z))`, `∂f/∂z_i = sum_k ∂f/∂a_k * ∂a_k/∂z_i = sum_k ∂f/∂a_k * a_k * (δ_{ki} - a_i) = a_i * ∂f/∂a_i - a_i * sum_k a_k ∂f/∂a_k`)

Yes, that's correct.

So the algorithm is:
1. Parameterize `a = softmax(z)`
2. Compute gradient `g_z = a * (g_a - a^T g_a)` where `g_a = ∇_a f`
3. Use L-BFGS to optimize `z`

For the smoothed objective:
`f_T(a) = 2n * T * log(sum_k exp(c_k/T))`
`g_a_i = 2n * sum_k w_k * 2 * a_{k-i} * [0<=k-i<n]` (where `w_k = softmax(c/T)`)

This is computable in O(n log n) using FFT.

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast mirror descent (multiplicative weight) with uniform subgradient of max-convolution
2. FISTA (accelerated proximal gradient) on log-sum-exp smoothed objective
3. L-BFGS in log-space (softmax parameterization) on smoothed objective

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score_norm(a):
    """Score assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def subgradient_max(a, frac=0.05):
    """Compute uniform subgradient of max(conv(a,a)) w.r.t. a.
    Returns gradient vector g of same length as a.
    g_i = 2/|A| * sum_{k in A, 0<=k-i<n} a[k-i]
    """
    conv = fast_convolve(a)
    n = len(a)
    max_c = conv.max()
    if max_c < 1e-30:
        return np.zeros(n)
    # Active set: top frac fraction of peaks
    threshold = max_c * (1.0 - frac)
    active_idx = np.where(conv >= threshold)[0]
    active_float = np.zeros(2 * n - 1)
    active_float[active_idx] = 1.0
    
    # g_i/2 = (1/|A|) * sum_{j=0}^{n-1} a[j] * active[i+j]
    # This is the cross-correlation of a with active at various shifts
    # corr[i] = sum_j a[j] * active[i+j] for i in [0, n-1]
    
    # Using FFT: corr[i] = (a * flip(active))[2n-2-i]
    b = active_float[::-1]
    total_size = len(a) + len(b) - 1
    fft_size = 1 << (total_size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    conv_ab = irfft(A * B, fft_size)
    
    # corr[i] = conv_ab[2n-2-i] for i in [0, n-1]
    # 2n-2-i for i=0 gives 2n-2, for i=n-1 gives n-1
    corr = conv_ab[n-1 : 2*n-1][::-1]
    
    g = 2.0 * corr / len(active_idx)
    return g


def mirror_descent(a_init, time_limit, eta0=1.0, cooling=0.9995, frac=0.05, verbose=True):
    """Mirror descent with exponential updates for max-convolution minimization."""
    a = np.maximum(a_init.astype(np.float64).copy(), 1e-30)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_score = compute_score_norm(a)
    t0 = time.monotonic()
    eta = eta0
    
    for it in range(50000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        g = subgradient_max(a, frac)
        g_mean = g.mean()
        g_centered = g - g_mean  # Centering doesn't change the update (normalization handles it)
        
        # Multiplicative weight update
        update = -eta * g_centered
        # Clip for numerical stability
        update = np.clip(update, -50, 50)
        new_a = a * np.exp(update)
        s = new_a.sum()
        if s < 1e-30 or not np.isfinite(s):
            eta *= 0.5
            continue
        new_a = new_a / s
        
        score = compute_score_norm(new_a)
        if score < best_score:
            best_score = score
            best_a = new_a.copy()
        
        a = new_a
        eta = max(eta * cooling, 0.001)
        
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  MD it={it}, score={best_score:.8f}, eta={eta:.4f}, elapsed={elapsed:.0f}s")
    
    return best_a, best_score

def project_simplex(v):
    """Project vector onto the probability simplex (sum=1, all >= 0)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(1 - u / np.arange(1, n + 1) > 0)
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def lbgfs_logspace(a_init, time_limit, T=100.0, verbose=True):

        z_shift = z - z.max()

        w = np.exp(shifted - max_s)
        w = w / w.sum()

        S = np.dot(grad_a, a_tmp)

        return obj, grad_z
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        obj, _ = func_and_grad(z)
        return obj

        _, g = func_and_grad(z)
        return g

        if time.monotonic() - t0 > time_limit:
            break
        grad, _ = gradient_smoothed(y, T)
        step = 1.0 / L
        a_trial = y - step * grad
        a_trial = project_simplex(a_trial)
        # Simple backtracking: check if step was good
        if it % 200 == 0 and it > 0:
            # Estimate gradient Lipschitz from a and a_trial
            grad_a, _ = gradient_smoothed(a, T)
            if np.max(np.abs(grad - grad_a)) > 0:
                L = max(L * 0.95, 1.0)
            else:
                L = max(L * 1.1, 2.0 * n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = a_trial + (t_k - 1.0) / t_new * (a_trial - a)
        y = project_simplex(y)
        t_k = t_new
        a = a_trial
        score = compute_score_norm(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  FISTA it={it}, score={best_score:.8f}, L={L:.1f}, T={T:.1f}, elapsed={elapsed:.0f}s")
    return best_a, best_score


def lbgfs_logspace(a_init, time_limit, T=50.0, verbose=True):
    """L-BFGS in log-space on smoothed objective."""
    n = len(a_init)
    a0 = np.maximum(a_init, 1e-30)
    a0 = a0 / a0.sum()
    z0 = np.log(a0)
    z0 -= z0.max()
    best_a = a0.copy()
    best_score = compute_score_norm(a0)
    t0 = time.monotonic()

    def func_jac(z):
        exp_z = np.exp(z - z.max())
        a_tmp = exp_z / exp_z.sum()
        conv = fast_convolve(a_tmp)
        shifted = conv / T
        max_s = shifted.max()
        log_sum_exp = max_s + np.log(np.exp(shifted - max_s).sum())
        obj = 2.0 * n * T * log_sum_exp
        w = np.exp(shifted - max_s)
        w = w / w.sum()
        corr = crosscorr_with_active(a_tmp, w)
        grad_a = 4.0 * n * corr
        S = np.dot(grad_a, a_tmp)
        grad_z = a_tmp * (grad_a - S)
        return obj, grad_z

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        obj, _ = func_jac(z)
        return obj

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        _, g = func_jac(z)
        return g

    try:
        res = minimize(fun=func, x0=z0, jac=jac, method='L-BFGS-B',
                       options={'maxiter': 50000, 'ftol': 1e-15, 'gtol': 1e-10,
                                'maxfun': 300000, 'disp': False})
        z_out = res.x
        a_out = np.exp(z_out - z_out.max())
        a_out = a_out / a_out.sum()
        score = compute_score_norm(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception as e:
        if verbose:
            print(f"  L-BFGS error: {e}")
    return best_a, best_score


def coordinate_descent(a_init, time_limit, verbose=True):
    """Pair coordinate descent with golden section search."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64).copy(), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score_norm(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(456)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        conv = fast_convolve(a)
        top_peak = int(np.argmax(conv))
        js = max(0, top_peak - n + 1)
        je = min(n - 1, top_peak)
        n_pairs = min(n, 3000)
        pairs = []
        for _ in range(n_pairs // 2):
            i = int(rng.integers(js, je + 1))
            j = int(rng.integers(js, je + 1))
            if i != j:
                pairs.append((min(i, j), max(i, j)))
        for _ in range(n_pairs // 3):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                pairs.append((min(i, j), max(i, j)))
        edge = max(3, n // 6)
        for _ in range(n_pairs // 6):
            i = int(rng.integers(0, edge))
            j = int(rng.integers(n - edge, n))
            if i != j:
                pairs.append((min(i, j), max(i, j)))

        for (i, j) in pairs:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14 and a[j] < 1e-14:
                continue
            t_lo = -a[j] + 1e-15
            t_hi = a[i] - 1e-15
            if t_hi <= t_lo:
                continue
            gr = (np.sqrt(5) - 1) / 2
            x1 = t_hi - gr * (t_hi - t_lo)
            x2 = t_lo + gr * (t_hi - t_lo)

            def eval_t(t):
                ai = a[i] - t
                aj = a[j] + t
                if ai < -1e-15 or aj < -1e-15:
                    return 1e10
                ai = max(0.0, ai)
                aj = max(0.0, aj)
                a_trial = a.copy()
                a_trial[i] = ai
                a_trial[j] = aj
                ss = a_trial.sum()
                if ss < 1e-15:
                    return 1e10
                a_trial /= ss
                return compute_score_norm(a_trial)

            f1 = eval_t(x1)
            f2 = eval_t(x2)
            for _ in range(25):
                if time.monotonic() - t0 > time_limit:
                    break
                if f1 < f2:
                    t_hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = t_hi - gr * (t_hi - t_lo)
                    f1 = eval_t(x1)
                else:
                    t_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = t_lo + gr * (t_hi - t_lo)
                    f2 = eval_t(x2)
            t_opt = (t_lo + t_hi) / 2
            f_opt = min(f1, f2)
            if f_opt < best_score - 1e-13:
                a[i] = max(0.0, a[i] - t_opt)
                a[j] = max(0.0, a[j] + t_opt)
                a = a / a.sum()
                best_score = compute_score_norm(a)
                best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 15:
                noise = rng.standard_normal(n) * 0.002
                a = best_a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
        else:
            no_improve = 0

        if verbose and no_improve % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"  CD no_improve={no_improve}, score={best_score:.8f}, elapsed={elapsed:.0f}s")
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def check_and_update(a, label, best_seq, best_score, verbose=True):
    """Evaluate candidate and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy fast optimization:
    - Mirror descent (multiplicative weight with subgradient)
    - L-BFGS in log-space on smoothed objective
    - FISTA on smoothed objective
    - Coordinate descent polish
    - Length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 45.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        t_arr = np.linspace(0, 2 * np.pi * 2, n_init)
        parent = parent + 0.03 * np.cos(t_arr)
        parent = np.maximum(parent, 0)
        parent = parent / parent.sum()
        best_seq = parent.copy()
        best_score = compute_score_norm(parent)
        print(f"Initialized: n={n_init}, score={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: Mirror Descent ===
    print(f"\n=== Phase 1: Mirror Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(300.0, rem * 0.2)
    print(f"  Budget: {t_alloc:.0f}s")
    a_md, sc_md = mirror_descent(best_seq, t_alloc, eta0=0.3, cooling=0.9999,
                                 frac_start=0.2, frac_end=0.02, verbose=True)
    best_seq, best_score = check_and_update(a_md, "md", best_seq, best_score)

    # === Phase 2: L-BFGS in log-space ===
    print(f"\n=== Phase 2: L-BFGS Log-Space (best={best_score:.8f}) ===")
    for T_val in [30, 100, 300]:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t_alloc = min(rem * 0.15, 120.0)
        print(f"  L-BFGS T={T_val}, budget={t_alloc:.0f}s")
        a_lb, sc_lb = lbgfs_logspace(best_seq, t_alloc, T=T_val, verbose=True)
        best_seq, best_score = check_and_update(a_lb, f"lbgfs_T{T_val}", best_seq, best_score)

    # === Phase 3: FISTA ===
    print(f"\n=== Phase 3: FISTA Smoothed (best={best_score:.8f}) ===")
    for T_val in [20, 50]:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t_alloc = min(rem * 0.12, 100.0)
        print(f"  FISTA T={T_val}, budget={t_alloc:.0f}s")
        a_f, sc_f = fista_smoothed(best_seq, t_alloc, T=T_val, verbose=True)
        best_seq, best_score = check_and_update(a_f, f"fista_T{T_val}", best_seq, best_score)

    # === Phase 4: More Mirror Descent (refined) ===
    print(f"\n=== Phase 4: Mirror Descent Refined (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 50:
        t_alloc = min(rem * 0.15, 150.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a_md2, sc_md2 = mirror_descent(best_seq, t_alloc, eta0=0.1, cooling=0.99995,
                                       frac_start=0.05, frac_end=0.01, verbose=True)
        best_seq, best_score = check_and_update(a_md2, "md2", best_seq, best_score)

    # === Phase 5: Coordinate Descent Polish ===
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_alloc = min(rem * 0.2, 250.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a_cd, sc_cd = coordinate_descent(best_seq, t_alloc, verbose=True)
        best_seq, best_score = check_and_update(a_cd, "cd", best_seq, best_score)

    # === Phase 6: L-BFGS Polish ===
    print(f"\n=== Phase 6: L-BFGS Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_alloc = min(rem * 0.15, 80.0)
        a_lp, sc_lp = lbgfs_logspace(best_seq, t_alloc, T=20.0, verbose=False)
        best_seq, best_score = check_and_update(a_lp, "lbgfs_polish", best_seq, best_score)

    # === Phase 7: Length Exploration ===
    print(f"\n=== Phase 7: Length Exploration (best={best_score:.8f}) ===")
    nearby_ns = [n_parent + 1, n_parent - 1, n_parent + 2, n_parent - 2,
                 n_parent + 4, n_parent - 4, n_parent + 8, n_parent + 16]
    for alt_n in nearby_ns:
        if alt_n < 64 or alt_n > 5000:
            continue
        rem = deadline - time.monotonic()
        if rem < 20:
            break
        t_alloc = min(rem * 0.1, 50.0)
        if t_alloc < 10:
            break
        init_alt = interpolate_sequence(best_seq, alt_n)
        print(f"  n={alt_n}, budget={t_alloc:.0f}s")
        a_alt, sc_alt = lbgfs_logspace(init_alt, t_alloc * 0.5, T=50.0, verbose=False)
        best_seq, best_score = check_and_update(a_alt, f"n{alt_n}_lbgfs", best_seq, best_score)
        rem2 = deadline - time.monotonic()
        if rem2 > 10:
            a_alt2, sc_alt2 = mirror_descent(a_alt, min(rem2 * 0.3, 30.0), eta0=0.1,
                                             frac_start=0.1, frac_end=0.02, verbose=False)
            best_seq, best_score = check_and_update(a_alt2, f"n{alt_n}_md", best_seq, best_score)

    # === Phase 8: Final CD Polish ===
    print(f"\n=== Phase 8: Final CD (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t_alloc = min(rem * 0.4, 60.0)
        a_fin, sc_fin = coordinate_descent(best_seq, t_alloc, verbose=False)
        best_seq, best_score = check_and_update(a_fin, "final_cd", best_seq, best_score)

    # Final verification
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
