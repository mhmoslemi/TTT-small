One more consideration: the simplex projection. For the gradient method, I need to project onto {f >= 0, sum(f) = 1}. The standard algorithm (using the Hungarian algorithm or the "water-filling" approach) runs in O(n log n) time.

Here's the projection:
```python
def project_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0) * target_sum])
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho = np.arange(1, n+1)
    cond = v_sorted * rho > (cssv - target_sum)
    rho_star = np.max(rho[cond])
    theta = (cssv[rho_star-1] - target_sum) / rho_star
    return np.maximum(v - theta, 0.0)
```

"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines entropy-regularized gradient descent (smooth warm-up),
cutting-plane LP (precise non-smooth optimization), adaptive
diversification, and multi-length exploration.
Key improvements: smooth-to-sharp annealing, better constraint
management, and structured restarts.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_convolve(f, g):
    """Cross-convolution f*g via FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Project onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0) * target_sum])
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho = np.arange(1, n + 1)
    cond = v_sorted * rho > (cssv - target_sum)
    if not cond.any():
        return np.ones(n) * (target_sum / n)
    rho_star = np.max(rho[cond])
    theta = (cssv[rho_star - 1] - target_sum) / rho_star
    return np.maximum(v - theta, 0.0)


def build_A(f, indices, n):
    """Build constraint matrix where row i gives (f*g)[indices[i]] as linear in g."""
    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[k - hi:k - lo + 1][::-1]
    return A


def pick_solver():
    """Find the best available LP solver."""
    for s in ['GUROBI', 'MOSEK', 'PDLP', 'GLPK', 'ECOS', 'CBC', 'SCS']:
        try:
            if s in cp.installed_solvers():
                return s
        except Exception:
            pass
    return 'SCS'


SOLVER = pick_solver()


def solve_lp(f, n, M, active_idx):
    """Solve: max sum(g) s.t. g >= 0, A@g <= M. Returns g or None."""
    K = len(active_idx)
    if K == 0:
        return None
    A = build_A(f, active_idx, n)
    g_var = cp.Variable(n)
    constraints = [g_var >= 0, A @ g_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
    try:
        if SOLVER in ('GUROBI', 'MOSEK', 'GLPK', 'CBC'):
            prob.solve(solver=SOLVER)
        elif SOLVER == 'PDLP':
            prob.solve(solver='PDLP', max_iter=100000)
        elif SOLVER == 'ECOS':
            prob.solve(solver='ECOS', max_iters=50000)
        else:
            prob.solve(solver='SCS', max_iters=500000, eps=1e-8)
    except Exception:
        try:
            prob.solve(solver='SCS', max_iters=500000, eps=1e-7)
        except Exception:
            return None
    if g_var.value is None:
        return None
    g_val = np.array(g_var.value).flatten()
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None
    return np.maximum(g_val, 0.0)


def line_search(f, h, b_current, n):
    """Find best step size t along direction h from f."""
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    H = rfft(h, fft_size)

    def obj_at(t):
        ft = (1.0 - t) * f + t * h
        Aft = (1.0 - t) * F + t * H
        return irfft(Aft * Aft, fft_size)[:size].max()

    best_t = 0.05
    best_obj = obj_at(0.05)

    for t in [0.02, 0.04, 0.06, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7]:
        o = obj_at(t)
        if o < best_obj:
            best_obj = o
            best_t = t

    lo = max(0.0, best_t - 0.08)
    hi = min(1.0, best_t + 0.12)
    for _ in range(20):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        o1 = obj_at(m1)
        o2 = obj_at(m2)
        if o1 < o2:
            hi = m2
        else:
            lo = m1
        if hi - lo < 1e-6:
            break
    t_mid = (lo + hi) / 2.0
    o_mid = obj_at(t_mid)
    if o_mid < best_obj:
        best_obj = o_mid
        best_t = t_mid

    return best_t, best_obj


def gradient_warmup(f_init, time_limit, beta_start=50.0, beta_end=5000.0,
                    lr_init=0.05, momentum=0.9):
    """Entropy-regularized gradient descent for rapid initial progress."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    v = np.zeros(n)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    size = 2 * n - 1

    total_time = max(time_limit, 1.0)
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        frac = min(1.0, it / max(1, int(time_limit / 0.05)))
        beta = beta_start * (beta_end / beta_start) ** frac

        b = fast_convolve(f)
        scaled = beta * b
        scaled -= scaled.max()
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.ones(size) / size
        else:
            p /= p_sum

        # Gradient: g[i] = 2 * sum_j p[i+j] * f[j]
        # Use direct computation (fast enough for n~1000-2000)
        g = np.zeros(n)
        for i in range(n):
            g[i] = 2.0 * np.dot(p[i:i + n], f)

        lr = lr_init / (1.0 + 0.01 * it)
        v = momentum * v - lr * g
        f_new = project_simplex(f + v)

        if time.monotonic() - t0 > time_limit:
            break

        f = f_new
        if it % 50 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_f = f.copy()

    return best_f, best_obj


def lp_optimize(f_init, time_limit, initial_k=50, max_k=350, label="lp"):
    """Cutting-plane LP optimization."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    size = 2 * n - 1

    b = fast_convolve(f)
    M = b.max()
    best_f = f.copy()
    best_obj = M

    if initial_k < size:
        active_idx = np.argpartition(b, -initial_k)[-initial_k:]
    else:
        active_idx = np.arange(size)
    active_idx = np.sort(active_idx)

    it = 0
    stuck = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2.0:
            break

        g = solve_lp(f, n, M, active_idx)

        if g is None:
            stuck += 1
            if stuck > 10:
                break
            if len(active_idx) > initial_k:
                scores = b[active_idx]
                active_idx = active_idx[np.argsort(scores)[-initial_k:]]
            continue

        S_g = g.sum()
        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > 25:
                # Try expanding active set
                if len(active_idx) < max_k:
                    new_top = np.argpartition(b, -min(max_k, size))[-min(max_k, size):]
                    active_idx = np.unique(np.concatenate([active_idx, new_top]))
                    active_idx = active_idx[np.argsort(b[active_idx])[-max_k:]]
                    stuck = 10
                else:
                    break
            continue

        stuck = 0
        h = g / S_g

        # Check violated constraints for cutting plane
        fg = cross_convolve(f, h)
        violated = fg > M * (1 + 1e-7)
        v_idx = np.where(violated)[0]
        if len(v_idx) > 0 and len(active_idx) < max_k:
            not_active = v_idx[~np.isin(v_idx, active_idx)]
            if len(not_active) > 0:
                n_add = min(20, len(not_active))
                to_add = not_active[np.argsort(fg[not_active])[-n_add:]]
                active_idx = np.unique(np.concatenate([active_idx, to_add]))
                active_idx = active_idx[np.argsort(b[active_idx])[-max_k:]]
                # Re-solve
                g = solve_lp(f, n, M, active_idx)
                if g is not None:
                    S_g = g.sum()
                    if S_g > 1.0 + 1e-6:
                        h = g / S_g
                    else:
                        h = g / max(S_g, 1e-10)

        # Line search
        best_t, best_t_obj = line_search(f, h, b, n)

        if best_t < 1e-4:
            stuck += 1
            if stuck > 15:
                break
            continue

        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        else:
            f = f_init.copy()
            break

        cur_obj = fast_convolve(f).max()
        M = cur_obj

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f.copy()

        # Periodic active set refresh
        if it % 20 == 0:
            b = fast_convolve(f)
            if len(active_idx) > initial_k:
                keep_scores = b[active_idx]
                n_keep = min(len(active_idx), initial_k)
                keep = active_idx[np.argsort(keep_scores)[-n_keep:]]
                new_top = np.argpartition(b, -initial_k)[-initial_k:]
                active_idx = np.unique(np.concatenate([keep, new_top]))

        if it % 50 == 0:
            print(f"    {label} it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} "
                  f"active={len(active_idx)} t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def subgradient_polish(f_init, time_limit, lr_init=0.003, n_top=40):
    """Subgradient method with adaptive parameters."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    size = 2 * n - 1
    v = np.zeros(n)
    lr = lr_init
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(123)
    momentum = 0.9

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)

        if n_top < size:
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            top_vals = b[top_idx]
            w = np.exp((top_vals - top_vals.max()) / 0.01)
            w_sum = w.sum()
            if w_sum < 1e-30:
                w = np.ones(len(top_idx)) / len(top_idx)
            else:
                w /= w_sum
        else:
            top_idx = np.arange(size)
            w = np.ones(size) / size

        g = np.zeros(n)
        for idx_k in range(len(top_idx)):
            k = top_idx[idx_k]
            wi = w[idx_k]
            if wi < 1e-12:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            j_range = np.arange(lo, hi + 1)
            g[j_range] += wi * 2.0 * f[k - j_range]

        v = momentum * v - lr * g
        f = project_simplex(f + v)

        if it % 40 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 5:
                    lr *= 0.85
                    no_improve = 0
                if no_improve > 30:
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                    v = np.zeros(n)
                    no_improve = 0

    return best_f, best_obj


def coordinate_polish(f_init, time_limit):
    """Coordinate descent with multiplicative updates."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    step = max(1, n // 300)

    for sweep in range(30):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for j in range(0, n, step):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-14:
                continue
            best_val = f[j]
            best_obj_j = best_obj
            for mult in [0.7, 0.85, 1.15, 1.3]:
                nv = f[j] * mult
                f[j] = nv
                s = f.sum()
                if s < 1e-10:
                    f[j] = best_val
                    continue
                f /= s
                obj = fast_convolve(f).max()
                if obj < best_obj_j - 1e-13:
                    best_obj_j = obj
                    best_val = nv
            f[j] = best_val
            s = f.sum()
            if s > 1e-10:
                f /= s
            if best_obj_j < best_obj - 1e-13:
                best_obj = best_obj_j
                improved = True
        if not improved:
            break
    return f, best_obj

    # === Phase 4: Diversification ===
    print(f"\n=== Phase 4: Diversification (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    noise_levels = [0.015, 0.03, 0.05]
    for ml in noise_levels:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t4 = min(rem * 0.08, 80.0)
        if t4 < 20:
            break
        f_noisy = best_f + rng.standard_normal(len(best_f)) * ml
        f_noisy = np.maximum(f_noisy, 0.0)
        s = f_noisy.sum()
        if s > 1e-10:
            f_noisy /= s
        f4, _ = lp_optimize(f_noisy, time_limit=t4, initial_k=40, max_k=250,
                            label=f"div_{ml}")
        score4 = float(evaluate_sequence(f4.tolist()))
        if np.isfinite(score4) and score4 < best_score:
            best_score = score4
            best_f = f4.copy()
            print(f"  NEW BEST [div_{ml}]: {score4:.8f}")
        else:
            print(f"  div_{ml}: {score4:.8f} (no improvement)")

    # === Phase 5: Length exploration ===
    print(f"\n=== Phase 5: Length exploration (best={best_score:.8f}) ===")
    n_base = len(best_f)
    target_lengths = []
    for delta in [-150, -100, -50, 50, 100, 150, 200]:
        tl = n_base + delta
        if 512 <= tl <= 2500:
            target_lengths.append(tl)
    target_lengths = sorted(set(target_lengths), key=lambda x: abs(x - n_base))

    for target_n in target_lengths[:4]:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t5 = min(rem * 0.08, 100.0)
        if t5 < 20:
            break
        old_x = np.linspace(0.0, 1.0, n_base)
        new_x = np.linspace(0.0, 1.0, target_n)
        f_interp = np.interp(new_x, old_x, best_f)
        f_interp = np.maximum(f_interp, 0.0)
        s = f_interp.sum()
        if s > 1e-10:
            f_interp /= s
        else:
            f_interp = np.ones(target_n) / target_n

        print(f"  Length {target_n}, budget={t5:.0f}s")
        f5, _ = lp_optimize(f_interp, time_limit=t5, initial_k=40, max_k=250,
                            label=f"n{target_n}")
        score5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(score5) and score5 < best_score:
            best_score = score5
            best_f = f5.copy()
            print(f"  NEW BEST [n{target_n}]: {score5:.8f}")
        else:
            print(f"  n{target_n}: {score5:.8f} (no improvement)")

    # === Phase 6: Subgradient polish ===
    rem = deadline - time.monotonic()
    if rem > 30:
        print(f"\n=== Phase 6: Subgradient polish (best={best_score:.8f}) ===")
        t6 = min(rem * 0.15, 120.0)
        n_b = len(best_f)
        f6, _ = subgradient_polish(best_f, time_limit=t6,
                                   lr_init=0.002, n_top=max(30, n_b // 15))
        score6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(score6) and score6 < best_score:
            best_score = score6
            best_f = f6.copy()
            print(f"  NEW BEST [sg]: {score6:.8f}")

    # === Phase 7: Final LP burst ===
    rem = deadline - time.monotonic()
    if rem > 30:
        print(f"\n=== Phase 7: Final LP (best={best_score:.8f}) ===")
        t7 = min(rem * 0.4, 150.0)
        n_b = len(best_f)
        print(f"  Budget: {t7:.0f}s, n={n_b}, k=100->500")
        f7, _ = lp_optimize(best_f, time_limit=t7, initial_k=100, max_k=500,
                            label="final_lp")
        score7 = float(evaluate_sequence(f7.tolist()))
        if np.isfinite(score7) and score7 < best_score:
            best_score = score7
            best_f = f7.copy()
            print(f"  NEW BEST [final_lp]: {score7:.8f}")

    # === Phase 8: Coordinate descent ===
    rem = deadline - time.monotonic()
    if rem > 15:
        print(f"\n=== Phase 8: Coord descent (best={best_score:.8f}) ===")
        t8 = min(rem * 0.5, 60.0)
        f8, _ = coordinate_polish(best_f, time_limit=t8)
        score8 = float(evaluate_sequence(f8.tolist()))
        if np.isfinite(score8) and score8 < best_score:
            best_score = score8
            best_f = f8.copy()
            print(f"  NEW BEST [cd]: {score8:.8f}")

    # Final validation
    fs = float(evaluate_sequence(best_f.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_f = parent.copy()
        best_score = p_score

    print(f"\n{'='*55}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_f)})")
    print(f"{'='*55}")

    return [float(x) for x in best_f]
