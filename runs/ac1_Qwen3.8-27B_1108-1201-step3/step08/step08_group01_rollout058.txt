def fast_conv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1

    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.GLPK, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.GLPK, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def softmax_adam(a0, time_limit, T_init=2.0, T_max=500.0, lr_init=0.05,
                 T_growth=1.02, verbose=False, label=""):
    """
    Soft-max gradient descent with Adam optimizer.
    Minimizes f_T(a) = (1/T) * log(sum_k exp(T * conv(a,a)_k))
    subject to a >= 0, sum(a) = 1.
    Temperature T increases over time (starts smooth, ends sharp).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_conv(a)
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    step = 0

    t0 = time.monotonic()
    T = T_init
    lr = lr_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Compute convolution
        conv = fast_conv(a)
        M = float(conv.max())
        score = 2.0 * n * M

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive temperature: increase T when not improving
        if no_improve > 50:
            T = min(T * 1.5, T_max)
            no_improve = 0

        # Compute soft-max weights
        T_eff = max(T, 1.0)
        conv_shifted = conv - conv.max()
        log_w = T_eff * conv_shifted
        log_w -= log_w.max()
        w = np.exp(log_w)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros_like(conv)
            w[int(np.argmax(conv))] = 1.0
        else:
            w = w / w_sum

        # Compute gradient: grad[i] = 2 * sum_m w[i+m] * a[m]
        # This is 2 * correlate(w, a, 'valid')
        grad = 2.0 * correlate(w, a, mode='valid')
        grad = np.maximum(grad, 0)  # gradient should be non-negative for non-neg a

        # Adam update
        step += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)
        update = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - update
        # Project onto simplex
        a_new = simplex_project(a_new)

        # Check if update improved
        conv_new = fast_conv(a_new)
        score_new = 2.0 * n * float(conv_new.max())

        if score_new < score + 1e-10:
            a = a_new
        else:
            # Rejected, reduce lr
            lr *= 0.95

        # Decay learning rate slowly
        lr = max(0.001, lr_init * (0.9999 ** it))

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SoftMaxAdam {label} it={it} ({elapsed:.0f}s): best={best_score:.6f} cur={score:.6f} T={T:.1f} lr={lr:.5f}")

    return best_a, best_score


def peak_weighted_grad(a0, time_limit, n_peaks=5, lr_init=0.02, verbose=False, label=""):
    """
    Gradient descent using hard peak weighting (no soft-max smoothing).
    Focuses optimization on the top n_peaks of the convolution.
    Uses a custom line search for stability.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_conv(a)
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_conv(a)
        M = float(conv.max())
        score = 2.0 * n * M

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                lr *= 0.9
                no_improve = 0
                if lr < 1e-6:
                    break

        # Find top peaks
        k_peaks = min(n_peaks, len(conv))
        peak_idx = np.argpartition(conv, -k_peaks)[-k_peaks:]
        peak_vals = conv[peak_idx]

        # Use softmax weighting over peaks
        temp = 0.05 * max(peak_vals.max(), 1e-10)
        pw = np.exp((peak_vals - peak_vals.max()) / temp)
        pw = pw / pw.sum()

        # Compute weighted gradient
        grad = np.zeros(n)
        for p in range(len(peak_idx)):
            if pw[p] < 1e-10:
                continue
            k = int(peak_idx[p])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_end < i_start:
                continue
            idx = np.arange(i_start, i_end + 1)
            # grad[i] += 2 * pw[p] * a[k-i]
            contributions = 2.0 * pw[p] * a[k - idx]
            grad[i_start:i_end + 1] += contributions

        # Gradient step with projection
        a_trial = a - lr * grad
        a_trial = simplex_project(a_trial)

        # Line search: accept if score doesn't increase too much
        conv_trial = fast_conv(a_trial)
        score_trial = 2.0 * n * float(conv_trial.max())

        if score_trial <= score + 1e-10:
            a = a_trial
        else:
            lr *= 0.9
            if lr < 1e-7:
                break

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakGrad {label} it={it} ({elapsed:.0f}s): best={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score


def coord_descent_peaks(a0, time_limit, n_trials=10, n_peaks_focus=3, verbose=False, label=""):
    """
    Coordinate descent that focuses on coordinates contributing to the top peaks.
    Uses incremental convolution updates for efficiency.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_conv(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        # Get current top peaks
        k_p = min(n_peaks_focus, len(conv))
        peak_positions = np.argpartition(conv, -k_p)[-k_p:]

        # Find all coordinates that contribute to any top peak
        focus_set = set()
        for k in peak_positions:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                focus_set.add(i)
        focus_list = list(focus_set)

        # Process focus coordinates first, then others
        all_coords = focus_list + [i for i in range(n) if i not in focus_set]

        for i in all_coords:
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            if k_end < k_start:
                continue

            # Unaffected max
            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            affected = conv[k_start:k_end + 1]
            # a_shifted[m] = a[m] (for the change in conv due to a[i] -> a[i]+delta)
            # conv[k] changes by 2*delta*a[k-i] for k in [i, i+n-1]
            a_shifted = a[0:n]

            best_delta = 0.0
            best_new_max = M

            # Try multiple delta values
            for trial in range(n_trials):
                if trial == 0:
                    delta = -0.5 * a[i]
                elif trial == 1:
                    delta = -0.3 * a[i]
                elif trial == 2:
                    delta = -0.15 * a[i]
                elif trial == 3:
                    delta = -0.05 * a[i]
                elif trial == 4:
                    delta = -0.02 * a[i]
                elif trial == 5:
                    delta = 0.02 * a[i]
                elif trial == 6:
                    delta = 0.05 * a[i]
                elif trial == 7:
                    delta = 0.15 * a[i]
                elif trial == 8:
                    delta = 0.3 * a[i]
                else:
                    delta = 0.5 * a[i]

                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                # New affected values
                new_affected = affected + 2.0 * delta * a_shifted
                # Handle k = 2i (diagonal term)
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx_2i] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-16:
                # Apply
                a[i] += best_delta
                a[i] = max(a[i], 0.0)

                # Update conv
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx_2i] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved and sweeps > 10:
            break

        if verbose and sweeps % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CDPeaks {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def fine_coord_descent(a0, time_limit, verbose=False, label=""):
    """Very fine coordinate descent with small multiplicative step factors."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_conv(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0

    factors = [0.99, 0.97, 0.95, 0.93, 1.03, 1.05, 1.07, 1.09]

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-18:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            affected = conv[k_start:k_end + 1]
            a_shifted = a[0:n]

            best_delta = 0.0
            best_new_max = M

            for factor in factors:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx_2i] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-17:
                a[i] += best_delta
                a[i] = max(a[i], 0.0)

                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx_2i] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved and sweeps > 5:
            break

    return best_a, best_score


def lp_alternating(a0, time_limit, solver=None, alpha_start=0.5, verbose=False, label=""):
    """
    Alternating LP best response with adaptive damping.
    alpha decays from alpha_start to small values.
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_conv(a).max()
    t0 = time.monotonic()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        alpha = alpha_start * (0.995 ** iters)
        alpha = max(alpha, 0.01)

        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            continue
        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        score = 2.0 * n * fast_conv(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        if verbose and iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPAlt {label} it={iters} ({elapsed:.0f}s): best={best_score:.6f} alpha={alpha:.4f}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length (preserving sum=1)."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_fourier(n, rng):
    """Initialize with random Fourier superposition."""
    n_harm = int(rng.integers(3, 20))
    amps = rng.random(n_harm) * 0.4
    phases = rng.uniform(0, 2 * np.pi, n_harm)
    freqs = np.arange(1, n_harm + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harm):
        a += amps[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_gauss(n, rng):
    """Gaussian bell init."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * rng.uniform(0.25, 0.5)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_u_shape(n, rng):
    """U-shape init (mass at edges)."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    alpha = rng.uniform(1.0, 3.0)
    a = 1.0 + alpha * (2 * t - 1) ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_noise(n, base, rng, sigma=0.1):
    """Add noise to base."""
    a = base.copy() + rng.standard_normal(n) * sigma
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def perturb_escape(a0, rng, magnitude=0.05):
    """Perturb to escape local optimum."""
    n = len(a0)
    a = a0.copy()
    conv = fast_conv(a)
    top_k = min(5, len(conv))
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    for k in peaks:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1)
        n_pick = max(1, len(idx) // 5)
        picks = rng.choice(idx, size=min(n_pick, len(idx)), replace=False)
        for i in picks:
            a[i] *= max(0, 1.0 + rng.standard_normal() * magnitude)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a0.copy()
    return a / s


def evaluate_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate using the actual evaluate_sequence function."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def multi_phase_optimize(a0, time_budget, solver, rng, label=""):
    """
    Run a multi-phase optimization pipeline on a single starting sequence.
    Returns the best sequence found.
    """
    n = len(a0)
    a = a0.copy()
    best_local = a.copy()
    best_local_score = 2.0 * n * fast_conv(a).max()
    t0 = time.monotonic()

    # Phase A: SoftMax-Adam
    tA = min(time_budget * 0.35, 120.0)
    if tA > 5:
        a_a, s_a = softmax_adam(a, tA, T_init=1.0, T_max=300.0, lr_init=0.03,
                                verbose=False, label=f"{label}/sm")
        if s_a < best_local_score:
            best_local_score = s_a
            best_local = a_a.copy()
        a = a_a

    # Phase B: Peak-weighted gradient
    tB = min(time_budget * 0.2, 60.0)
    if tB > 5:
        a_b, s_b = peak_weighted_grad(a, tB, n_peaks=3, lr_init=0.01,
                                      verbose=False, label=f"{label}/pg")
        if s_b < best_local_score:
            best_local_score = s_b
            best_local = a_b.copy()
        a = a_b

    # Phase C: Peak-focused coordinate descent
    tC = min(time_budget * 0.25, 80.0)
    if tC > 5:
        a_c, s_c = coord_descent_peaks(a, tC, n_trials=10, n_peaks_focus=3,
                                       verbose=False, label=f"{label}/cd")
        if s_c < best_local_score:
            best_local_score = s_c
            best_local = a_c.copy()
        a = a_c

    # Phase D: LP alternating
    tD = min(time_budget * 0.15, 50.0)
    if tD > 5:
        a_d, s_d = lp_alternating(a, tD, solver=solver, alpha_start=0.4,
                                  verbose=False, label=f"{label}/lp")
        if s_d < best_local_score:
            best_local_score = s_d
            best_local = a_d.copy()
        a = a_d

    # Phase E: Fine polish
    tE = max(0, time_budget - (time.monotonic() - t0))
    if tE > 5:
        a_e, s_e = fine_coord_descent(a, tE, verbose=False, label=f"{label}/fine")
        if s_e < best_local_score:
            best_local_score = s_e
            best_local = a_e.copy()

    return best_local, best_local_score


def propose_candidate():
    """
    Multi-phase optimization combining soft-max gradient descent,
    peak-targeted coordinate descent, LP alternation, and diverse
    multi-length search to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(42)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    # === Phase 1: Quick exploration with diverse inits at multiple lengths ===
    print("\n=== Phase 1: Diverse Init + Quick Opt ===")
    t1_start = time.monotonic()
    t1_budget = 300.0

    n_values = [1024, 1152, 1280, 960]
    if parent is not None:
        np_parent = len(parent)
        if np_parent not in n_values:
            n_values.insert(0, np_parent)

    initial_candidates = []

    for n in n_values:
        elapsed = time.monotonic() - t1_start
        if elapsed > t1_budget - 30:
            break
        t_per_len = (t1_budget - elapsed) / (len(n_values) - n_values.index(n))
        t_per_init = max(5.0, t_per_len / 5.0)

        inits = []
        if parent is not None:
            inits.append(('parent', interpolate_seq(parent, n)))
        inits.append(('fourier', init_fourier(n, rng)))
        inits.append(('gauss', init_gauss(n, rng)))
        inits.append(('u_shape', init_u_shape(n, rng)))
        if parent is not None:
            inits.append(('noise', init_noise(n, interpolate_seq(parent, n), rng, 0.15)))

        for name, init in inits:
            elapsed = time.monotonic() - t1_start
            if elapsed > t1_budget - 10:
                break
            # Quick coordinate descent
            a_opt, s_opt = multi_phase_optimize(init, min(t_per_init, 4.0), solver, rng, label=f"p1/{name}/{n}")
            initial_candidates.append((s_opt, a_opt, n, name))

    initial_candidates.sort(key=lambda x: x[0])
    print(f"  {len(initial_candidates)} candidates, best: {initial_candidates[0][0]:.6f}")

    for s_c, a_c, n_c, name_c in initial_candidates[:5]:
        best_seq, best_score = evaluate_and_update(a_c, best_seq, best_score, f"p1/{name_c}/{n_c}")

    # === Phase 2: Intensive optimization on best candidates ===
    print(f"\n=== Phase 2: Intensive Optimization (best={best_score:.6f}) ===")
    t2_start = time.monotonic()
    t2_budget = 500.0
    n_best = len(best_seq) if best_seq is not None else 1024

    # Run full pipeline on the best sequence
    t_alloc = min(t2_budget * 0.4, deadline - time.monotonic() - 800 + t2_start)
    if t_alloc > 20:
        print(f"  Full pipeline on n={n_best}, budget={t_alloc:.0f}s")
        a_res, s_res = multi_phase_optimize(best_seq, t_alloc, solver, rng, label="p2/main")
        best_seq, best_score = evaluate_and_update(a_res, best_seq, best_score, "p2/main")

    # Try with perturbation
    t_alloc = min(t2_budget * 0.3, deadline - time.monotonic() - 600 + t2_start)
    if t_alloc > 20 and best_score > 1.503:
        print(f"  Perturbed pipeline, budget={t_alloc:.0f}s")
        perturbed = perturb_escape(best_seq, rng, magnitude=0.04)
        a_res, s_res = multi_phase_optimize(perturbed, t_alloc, solver, rng, label="p2/pert")
        best_seq, best_score = evaluate_and_update(a_res, best_seq, best_score, "p2/pert")

    # Try another length
    t_alloc = min(t2_budget * 0.3, deadline - time.monotonic() - 500 + t2_start)
    if t_alloc > 20:
        alt_n = 1280 if n_best != 1280 else 1152
        print(f"  Alt length n={alt_n}, budget={t_alloc:.0f}s")
        init_alt = interpolate_seq(best_seq, alt_n)
        a_res, s_res = multi_phase_optimize(init_alt, t_alloc, solver, rng, label=f"p2/alt{alt_n}")
        best_seq, best_score = evaluate_and_update(a_res, best_seq, best_score, f"p2/alt{alt_n}")

    # === Phase 3: Focused refinement ===
    print(f"\n=== Phase 3: Focused Refinement (best={best_score:.6f}) ===")
    t3_start = time.monotonic()
    t3_budget = 400.0
    remaining = deadline - time.monotonic() - 200
    t3_budget = min(t3_budget, max(0, remaining))

    if t3_budget > 30:
        # Extended soft-max with higher temperature
        t_sm = t3_budget * 0.35
        print(f"  Extended SoftMax, budget={t_sm:.0f}s")
        a_sm, s_sm = softmax_adam(best_seq, t_sm, T_init=5.0, T_max=1000.0,
                                  lr_init=0.01, T_growth=1.03, verbose=True, label="p3/sm")
        best_seq, best_score = evaluate_and_update(a_sm, best_seq, best_score, "p3/sm")

        # Peak gradient focus
        remaining = deadline - time.monotonic() - 150
        if remaining > 30:
            t_pg = min(remaining * 0.4, t3_budget * 0.25)
            print(f"  Peak Gradient, budget={t_pg:.0f}s")
            a_pg, s_pg = peak_weighted_grad(best_seq, t_pg, n_peaks=3, lr_init=0.005,
                                            verbose=True, label="p3/pg")
            best_seq, best_score = evaluate_and_update(a_pg, best_seq, best_score, "p3/pg")

        # LP alternating with more iterations
        remaining = deadline - time.monotonic() - 100
        if remaining > 30:
            t_lp = min(remaining * 0.4, t3_budget * 0.25)
            print(f"  LP Alternating, budget={t_lp:.0f}s")
            a_lp, s_lp = lp_alternating(best_seq, t_lp, solver=solver, alpha_start=0.2,
                                        verbose=True, label="p3/lp")
            best_seq, best_score = evaluate_and_update(a_lp, best_seq, best_score, "p3/lp")

        # Coordinate descent
        remaining = deadline - time.monotonic() - 50
        if remaining > 20:
            t_cd = min(remaining * 0.5, t3_budget * 0.15)
            print(f"  CoordDescent, budget={t_cd:.0f}s")
            a_cd, s_cd = coord_descent_peaks(best_seq, t_cd, n_trials=12, n_peaks_focus=2,
                                             verbose=True, label="p3/cd")
            best_seq, best_score = evaluate_and_update(a_cd, best_seq, best_score, "p3/cd")

    # === Phase 4: Final intensive polish ===
    print(f"\n=== Phase 4: Final Polish (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic() - 30
    if remaining > 20:
        t_fine = min(remaining * 0.5, 120.0)
        print(f"  Fine CD, budget={t_fine:.0f}s")
        a_fine, s_fine = fine_coord_descent(best_seq, t_fine, verbose=True, label="p4/fine")
        best_seq, best_score = evaluate_and_update(a_fine, best_seq, best_score, "p4/fine")

        remaining = deadline - time.monotonic() - 15
        if remaining > 10:
            a_pol, s_pol = coord_descent_peaks(best_seq, min(remaining * 0.7, 60.0),
                                               n_trials=14, n_peaks_focus=2,
                                               label="p4/polish")
            best_seq, best_score = evaluate_and_update(a_pol, best_seq, best_score, "p4/polish")

    # === Phase 5: Last resort - another perturb + optimize cycle ===
    remaining = deadline - time.monotonic() - 15
    if remaining > 30 and best_score > 1.503:
        print(f"\n=== Phase 5: Last Perturb (best={best_score:.6f}) ===")
        pert = perturb_escape(best_seq, rng, magnitude=0.02)
        t_last = min(remaining * 0.6, 60.0)
        a_last, s_last = multi_phase_optimize(pert, t_last, solver, rng, label="p5")
        best_seq, best_score = evaluate_and_update(a_last, best_seq, best_score, "p5")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
