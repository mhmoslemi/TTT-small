def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)

    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def proj_simplex(v, s=1.0):
    """Project onto {x >= 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = 1.0 - np.arange(1, n + 1) * u / s
    rho = np.sum(cond > 0)
    if rho < 1:
        return np.full(n, s / n)
    theta = (cssv[rho - 1] - s) / rho
    return np.maximum(v - theta, 0.0)

def solve_lp_best_response(a_fixed, solver=None):

    # This is a Toeplitz system. For efficiency, build it as a dense matrix.
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):

        T[start:end, j] = a_fixed

    s = b_val.sum()

    b_val = b_val / s

def lse_objective_and_grad(a, T):
    """Compute LogSumExp objective and gradient.
    f_T(a) = (1/T) * log(sum_k exp(T * C_k(a)))
    Returns (f_T(a), grad)."""
    n = len(a)
    c = fast_convolve(a)
    Tc = T * c
    Tc_max = Tc.max()
    exp_Tc = np.exp(Tc - Tc_max)
    w = exp_Tc / exp_Tc.sum()
    obj = (Tc_max + np.log(exp_Tc.sum())) / T
    
    # Gradient: (1/T) * sum_k w[k] * grad C_k
    # grad C_k[i] = 2 * a[k-i] if 0 <= k-i < n else 0
    grad = np.zeros(n)
    for k in range(len(c)):

            a_vals = a[k - i_vals]
            grad[i_vals] += 2.0 * w[k] * a_vals
    
    grad = grad / T
    return obj, grad
