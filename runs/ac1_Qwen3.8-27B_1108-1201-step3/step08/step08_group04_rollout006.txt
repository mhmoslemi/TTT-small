import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_gradient(f, w, n):
    """Compute grad[j] = 2 * sum_i w[i+j] * f[i] for j=0..n-1."""
    grad = np.zeros(n)
    for j in range(n):
        i_max = min(n - 1, 2 * n - 2 - j)
        if i_max >= 0:
            grad[j] = 2.0 * np.dot(f[:i_max+1], w[j:j+i_max+1])
    return grad


def fista_optimize(f_init, time_limit, tau_init=0.5, tau_min=0.001, tau_decay=0.9995):
    """FISTA with smoothed max objective and adaptive temperature."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s
    
    f_prev = f.copy()
    t_k = 1.0
    tau = tau_init
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    full_len = 2 * n - 1
    rng = np.random.default_rng(12345)
    no_improve = 0
    step_base = 1.0 / n  # initial step size heuristic
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()
        
        # Compute softmax weights
        scaled = (b - M) / tau
        # Clip to avoid overflow
        scaled = np.clip(scaled, -50, 0)
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum > 1e-30:
            w = w / w_sum
        else:
            w = np.zeros(full_len)
            w[np.argmax(b)] = 1.0
        
        # Compute gradient
        grad = compute_smoothed_gradient(f, w, n)
        
        # FISTA momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        momentum = (t_k - 1.0) / t_new
        f_y = f + momentum * (f - f_prev)
        
        # Step with backtracking
        step = step_base
        f_trial = f_y - step * grad
        f_trial = np.maximum(f_trial, 0.0)
        s_trial = f_trial.sum()
        if s_trial < 1e-15:
            f_trial = f.copy()
        else:
            f_trial = f_trial / s_trial
        
        # Check improvement
        obj_trial = fast_convolve(f_trial).max()
        while obj_trial > M + 1e-10 and step > 1e-8:
            step *= 0.5
            f_trial = f_y - step * grad
            f_trial = np.maximum(f_trial, 0.0)
            s_trial = f_trial.sum()
            if s_trial < 1e-15:
                break
            f_trial = f_trial / s_trial
            obj_trial = fast_convolve(f_trial).max()
        
        if obj_trial <= M + 1e-10:
            f_prev = f
            f = f_trial
            t_k = t_new
            step_base = min(step * 1.5, 1.0)  # increase step for next time
        
        if obj_trial < best_obj - 1e-14:
            best_obj = obj_trial
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Anneal temperature
        tau = max(tau * tau_decay, tau_min)
        
        # Periodic restart
        if no_improve > 50 and it % 10 == 0:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.01)
            f_prev = f.copy()
            t_k = 1.0
            no_improve = 0
            step_base = 1.0 / n
        
        if it % 100 == 0:
            elapsed = time.monotonic() - t0

            print(f"      FISTA it={it} score={score:.6f} tau={tau:.5f} t={elapsed:.0f}s")
    
    return best_f, 2.0 * n * best_obj


def exact_line_search(f, h, b_ff, b_fh, b_hh):
    """Ternary search for optimal t in [0,1]."""
    lo, hi = 0.0, 1.0
    for _ in range(50):
        t1 = lo + (hi - lo) / 3.0
        t2 = hi - (hi - lo) / 3.0
        g1 = ((1-t1)**2 * b_ff + 2*t1*(1-t1) * b_fh + t1**2 * b_hh).max()
        g2 = ((1-t2)**2 * b_ff + 2*t2*(1-t2) * b_fh + t2**2 * b_hh).max()
        if g1 < g2:

        else:
            lo = t1
    t = (lo + hi) / 2.0

def build_lp_matrix(f, indices, n):
    """Build constraint matrix for LP."""
    K = len(indices)
    A = np.zeros((K, n))
    for i in range(K):
        k = indices[i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi+1] = f[lo:hi+1]
    return A


def solve_lp(f, indices, M):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    A = build_lp_matrix(f, indices, n)
    
    g = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])
    
    try:
        if n <= 1000:
            prob.solve(solver=cp.ECOS, max_iters=30000)
        else:
            prob.solve(solver=cp.SCS, max_iters=100000, eps_abs=1e-9, eps_rel=1e-9)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=50000, eps_abs=1e-8, eps_rel=1e-8)
        except Exception:
            return None
    
    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None
    return g_val

    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def coordinate_descent(f_init, time_limit):
    """Coordinate descent polish."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    multipliers = [0.3, 0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0]
    
    no_pass = 0
    while time.monotonic() - t0 < time_limit:
        any_imp = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-14:
                continue
            best_val = f[j]
            best_obj_j = best_obj
            for mult in multipliers:
                nv = f[j] * mult
                f[j] = nv
                s = f.sum()
                if s < 1e-12:
                    f[j] = best_val
                    continue
                f_n = f / s
                obj = fast_convolve(f_n).max()
                if obj < best_obj_j - 1e-13:
                    best_obj_j = obj
                    best_val = nv
            f[j] = best_val
            if best_val != f_init[j] or True:  # check if improved
                s = f.sum()
                f = f / s
                if best_obj_j < best_obj - 1e-13:
                    best_obj = best_obj_j
                    best_f = f.copy()
                    any_imp = True
        if not any_imp:
            no_pass += 1
            if no_pass >= 2:
                break
        else:
            no_pass = 0
    
    return best_f, 2.0 * n * best_obj


def interp_sequence(src, target_n):
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    return result
