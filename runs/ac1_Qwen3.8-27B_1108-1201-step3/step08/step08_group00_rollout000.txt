import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver_cached = None


def get_lp_solver():
    """Find a working LP solver."""
    global _lp_solver_cached
    if _lp_solver_cached is not None:
        return _lp_solver_cached
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):

        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)

    s = b_val.sum()

def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def lbgfs_refine(a0, time_limit, T_start=100.0, T_end=0.1, maxiter=10000):
    """L-BFGS with LogSumExp smoothing and geometric temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def make_func(T_val):
        def func(z):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    n_stages = 8
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(40, maxiter // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)

        contrib_count = np.zeros(n)
        for k in near_peaks[:8]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k - idx]
            top_idx = np.argsort(contribs)[::-1][:8]
            for ci in top_idx:
                contrib_count[idx[ci]] += 1

        top_coords = np.argsort(contrib_count)[::-1][:15]

        improved = False
        for i in top_coords:
            if a[i] < 1e-14:
                continue
            for factor in [0.99, 0.97, 0.95, 0.92, 0.88, 0.83, 0.78]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                if others.sum() < 1e-15:
                    continue
                a_trial[mask] = others + diff * others / others.sum()
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_contrib = np.argsort(contribs)[::-1][:6]
            for ci in top_contrib:
                i = idx[ci]
                j = peak_k - i
                if i == j or i >= n or j >= n:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.97, 0.94, 0.90]:
                    for f2 in [0.97, 0.94, 0.90]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-15:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            a_trial = a.copy()
            for i in top_coords[:4]:
                if a[i] > 1e-14:
                    a_trial[i] *= 0.98
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-15:
                a = a_trial
                best_score = score
                best_a = a.copy()
            else:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create structured initial sequence."""
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = a ** 1.5
    else:
        x = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + 20.0 * (x - 0.5) ** 2)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def windowed_sinc_init(n):
    """Initialize with a windowed sinc-like sequence (flat spectrum ideal)."""
    x = np.arange(n)
    center = (n - 1) / 2.0
    t = (x - center) / center
    # Hamming window
    w = 0.54 + 0.46 * np.cos(2 * np.pi * t / (n - 1))
    a = w ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization with extended LP, tight L-BFGS polishing,
    multi-length exploration, and multi-restart perturbation.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        best_seq = windowed_sinc_init(n_init)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Aggressive Nesterov LP on parent length ===
    print(f"\n=== Phase 1: Nesterov LP n={n_best} (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 50:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=200000)
        try_update(a_lp, "nesterov_lp_1")

    # === Phase 2: L-BFGS tight polishing ===
    print(f"\n=== Phase 2: L-BFGS Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.18, 280.0)
    if t_alloc > 30:
        print(f"  L-BFGS T: 100->0.05, budget={t_alloc:.0f}s")
        a_lb = lbgfs_refine(best_seq, t_alloc, T_start=100.0, T_end=0.05, maxiter=15000)
        try_update(a_lb, "lbgfs_polish")

    # === Phase 3: Second LP round from polished point ===
    print(f"\n=== Phase 3: Nesterov LP r2 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.18, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=100000)
        try_update(a_lp2, "nesterov_lp_2")

    # === Phase 4: L-BFGS final tight polish ===
    print(f"\n=== Phase 4: L-BFGS Final (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 180.0)
    if t_alloc > 20:
        print(f"  L-BFGS T: 20->0.01, budget={t_alloc:.0f}s")
        a_lb2 = lbgfs_refine(best_seq, t_alloc, T_start=20.0, T_end=0.01, maxiter=10000)
        try_update(a_lb2, "lbgfs_final")

    # === Phase 5: Peak shaving ===
    print(f"\n=== Phase 5: Peak Shaving (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.06, 80.0)
    if t_alloc > 10:
        a_ps, _ = peak_shave_local(best_seq, t_alloc, n_steps=30000)
        try_update(a_ps, "peak_shave")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Length Exploration (best={best_score:.8f}) ===")
    # Try lengths near parent and some multiples
    candidate_lengths = set()
    for delta in [-32, -16, -8, 8, 16, 32]:
        cand = n_best + delta
        if 128 <= cand <= 4096:
            candidate_lengths.add(cand)
    for mult in [1.5, 2.0]:
        cand = int(n_best * mult)
        if 128 <= cand <= 4096:
            candidate_lengths.add(cand)
    # Also try some "round" lengths
    for ln in [512, 1024, 1536, 2048]:
        if 128 <= ln <= 4096 and ln != n_best:
            candidate_lengths.add(ln)
    candidate_lengths = sorted([x for x in candidate_lengths if x != n_best])

    explored = 0
    for alt_n in candidate_lengths:
        rem = deadline - time.monotonic()
        if rem < 100 or explored >= 3:
            break
        t_total = min(rem * 0.20, 200.0)
        if t_total < 40:
            break
        print(f"  Trying n={alt_n}, budget={t_total:.0f}s")
        explored += 1

        init_alt = interpolate_sequence(best_seq, alt_n)

        # LP on new length
        t_lp = t_total * 0.60
        a_alt, _ = nesterov_lp_refine(init_alt, t_lp, solver=solver, n_max_iters=30000)
        try_update(a_alt, f"lp_n{alt_n}")

        # L-BFGS on new length
        rem = deadline - time.monotonic()
        if rem > 30:
            t_lb = min(rem * 0.15, 80.0)
            if t_lb > 15:
                a_alt2 = lbgfs_refine(best_seq, t_lb, T_start=50.0, T_end=0.5, maxiter=5000)
                try_update(a_alt2, f"lbgfs_n{alt_n}")

    print(f"\n=== Phase 7: Multi-Restart (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()

        t_per_restart = min(rem * 0.08, 100.0)

            n_cur = len(best_seq)
            noise_scale = 0.03 + 0.02 * rs
            perturbed = best_seq.copy()

            x = np.linspace(0, 2 * np.pi * (rs + 2), n_cur)
            mod = 1.0 + noise_scale * np.sin(x + rs * 1.3)
            perturbed = perturbed * mod
            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()

            # Quick LP + L-BFGS
            t_lp_rs = t_per_restart * 0.5
            a_rs, _ = nesterov_lp_refine(perturbed, t_lp_rs, solver=solver, n_max_iters=15000)
            try_update(a_rs, f"restart_{rs}")

            rem = deadline - time.monotonic()
            if rem > 20:
                t_lb_rs = min(rem * 0.2, 40.0)

                    try_update(a_rs2, f"restart_lb_{rs}")

    # === Phase 8: Final aggressive polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:

        try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 25:
        t_ps = min(rem * 0.3, 40.0)
        a_ps2, _ = peak_shave_local(best_seq, t_ps, n_steps=15000)
        try_update(a_ps2, "final_ps")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_lp_fin = min(rem * 0.5, 30.0)

        try_update(a_lp_fin, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
