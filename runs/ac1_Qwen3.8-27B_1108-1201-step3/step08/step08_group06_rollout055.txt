"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multiplicative Peak Reduction (MPR) - novel approach using mirror descent
   with soft-max smoothed gradient and adaptive step size
2. Smoothed L-BFGS for local refinement
3. Coordinate descent with exact line search
4. Length exploration around parent
Key insight: the gradient of the convolution peak is a shifted version of a itself,
enabling efficient FFT-based gradient computation.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score_norm(a):
    """Score for normalized sequence (sum=1): 2*n*max(conv)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def mpr_algorithm(a_init, time_limit, T_scale=1.0, eta_base=1.0, verbose=True):
    """
    Multiplicative Peak Reduction via mirror descent.
    Uses soft-max smoothing of the max and multiplicative (log-space) updates.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 1e-15)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_score_norm(a)
    t0 = time.monotonic()
    
    # Precompute FFT of reversed a (updated each iteration)
    fft_size = 1 << (2 * n - 2).bit_length()
    
    iter_count = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        iter_count += 1
        
        # Temperature: anneal from high to low
        progress = min(1.0, elapsed / time_limit)
        T = T_scale / n * (100.0 * (1 - progress) + 1.0)
        
        # Step size: start moderate, decrease
        eta = eta_base / (1.0 + 0.1 * progress)
        
        # Compute convolution
        b = fast_convolve(a)
        
        # Soft-max weights
        b_scaled = b / T
        b_max = b_scaled.max()
        if b_max > 500:  # overflow protection
            w = np.zeros_like(b)
            w[b == b.max()] = 1.0 / np.sum(b == b.max())
        else:
            exp_b = np.exp(b_scaled - b_max)
            w = exp_b / exp_b.sum()
        
        # Compute gradient: grad_m = 2 * sum_k w_k * a_{k-m}
        # = 2 * (w * a_rev)[n-1+m]
        a_rev = a[::-1]
        W = rfft(w, fft_size)
        Ar = rfft(a_rev, fft_size)
        conv_w_ar = irfft(W * Ar, fft_size)
        grad = 2.0 * conv_w_ar[n-1:2*n-1]
        
        # Ensure gradient is non-negative (it should be since a, w >= 0)
        grad = np.maximum(grad, 0)
        
        # Line search: try a few step sizes
        best_trial_score = best_score
        best_eta = 0
        
        # Try geometrically spaced step sizes
        for eta_try in [eta * 0.1, eta * 0.3, eta, eta * 3, eta * 10, eta * 30]:
            if eta_try <= 0:
                continue
            log_a = np.log(a) - eta_try * grad
            log_a -= log_a.max()
            a_trial = np.exp(log_a)
            s = a_trial.sum()
            if s < 1e-300:
                continue
            a_trial = a_trial / s
            sc = compute_score_norm(a_trial)
            if sc < best_trial_score:
                best_trial_score = sc
                best_eta = eta_try
        
        if best_eta > 0:
            log_a = np.log(a) - best_eta * grad
            log_a -= log_a.max()
            a = np.exp(log_a)
            a = a / a.sum()
            if best_trial_score < best_score:
                best_score = best_trial_score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 200:
            # Perturb to escape
            rng = np.random.default_rng(iter_count)
            noise = rng.standard_normal(n) * 0.01
            a = best_a + noise
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            no_improve = 0
        
        if verbose and iter_count % 200 == 0:
            print(f"  MPR iter={iter_count}, T={T:.4e}, score={best_score:.8f}, elapsed={elapsed:.0f}s")
    
    return best_a, best_score


def smoothed_lbgfs(a_init, time_limit, T=50.0, maxiter=3000, verbose=True):
    """L-BFGS on the smoothed (log-sum-exp) objective in log-space."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 1e-15)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_score_norm(a)
    t0 = time.monotonic()
    
    # Work in log-space: z = log(a), a = softmax(z)
    z = np.log(a)
    
    def objective(z):
        a_tmp = np.exp(z - z.max())
        a_tmp = a_tmp / a_tmp.sum()
        b = fast_convolve(a_tmp)
        # Log-sum-exp approximation of max
        b_scaled = b / T
        b_max = b_scaled.max()
        lse = T * (b_max + np.log(np.sum(np.exp(b_scaled - b_max))))
        return 2.0 * n * lse
    
    def gradient(z):
        a_tmp = np.exp(z - z.max())
        a_tmp = a_tmp / a_tmp.sum()
        b = fast_convolve(a_tmp)
        b_scaled = b / T
        b_max = b_scaled.max()
        w = np.exp(b_scaled - b_max)
        w = w / w.sum()
        
        # grad of (2n * T * lse) w.r.t. a: 2n * (w convolved with a_rev)[n-1:2n-1]
        # = 2n * sum_k w_k * a_{k-m} * 2 (from the gradient of b_k w.r.t. a_m)
        # Wait, let me recompute.
        # f(a) = 2n * T * log(sum exp(b_k/T))
        # df/da_m = 2n * T * sum_k (exp(b_k/T) / sum_j exp(b_j/T)) * (db_k/da_m) / T
        # = 2n * sum_k w_k * db_k/da_m
        # = 2n * sum_k w_k * 2*a_{k-m}
        # = 4n * (w * a_rev)[n-1+m]
        
        a_rev = a_tmp[::-1]
        fft_size = 1 << (2 * n - 2).bit_length()
        W = rfft(w, fft_size)
        Ar = rfft(a_rev, fft_size)
        conv_w_ar = irfft(W * Ar, fft_size)
        grad_a = 4.0 * n * conv_w_ar[n-1:2*n-1]
        
        # Project to log-space: grad_z = a * (grad_a - sum(a * grad_a))
        S = np.dot(grad_a, a_tmp)
        grad_z = a_tmp * (grad_a - S)
        
        return grad_z
    
    try:
        res = minimize(
            fun=objective, x0=z, jac=gradient,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-16,
                     'maxfun': maxiter * 100, 'disp': False}
        )
        a_out = np.exp(res.x - res.x.max())
        a_out = a_out / a_out.sum()
        score = compute_score_norm(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception as e:
        if verbose:
            print(f"  L-BFGS error: {e}")
    
    # Time check
    if verbose and time.monotonic() - t0 < time_limit:
        print(f"  L-BFGS done, score={best_score:.8f}, elapsed={time.monotonic()-t0:.0f}s")
    
    return best_a, best_score


def coordinate_descent(a_init, time_limit, verbose=True):
    """Coordinate descent with golden-section search for pair transfers."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score_norm(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        if no_improve > 500:
            # Perturb
            noise = rng.standard_normal(n) * 0.005
            a = best_a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            no_improve = 0
        
        # Compute convolution to guide pair selection
        b = fast_convolve(a)
        peak_idx = int(np.argmax(b))
        
        # Generate candidate pairs
        pairs = set()
        
        # Pairs involving positions that contribute to the peak
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        n_peak = je - js + 1
        for _ in range(min(200, n_peak * 2)):
            i = js + rng.integers(0, n_peak)
            j = rng.integers(0, n)
            if i != j:
                pairs.add((min(i,j), max(i,j)))
        
        # Random pairs
        for _ in range(200):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                pairs.add((min(i,j), max(i,j)))
        
        # Edge pairs
        edge = max(2, n // 20)
        for _ in range(100):
            i = rng.integers(0, edge)
            j = rng.integers(n - edge, n)
            if i != j:
                pairs.add((min(i,j), max(i,j)))
        
        improved = False
        for (i, j) in list(pairs):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-15 and a[j] < 1e-15:
                continue
            
            # Golden section search
            t_lo = -a[j] + 1e-15
            t_hi = a[i] - 1e-15
            if t_hi <= t_lo:
                continue
            
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            x1 = t_hi - gr * (t_hi - t_lo)
            x2 = t_lo + gr * (t_hi - t_lo)
            
            def eval_t(t):
                a[i] -= t
                a[j] += t
                if a[i] < -1e-15 or a[j] < -1e-15:
                    return 1e10
                a[i] = max(0.0, a[i])
                a[j] = max(0.0, a[j])
                ss = a.sum()
                if ss < 1e-15:
                    return 1e10
                a_norm = a / ss
                sc = compute_score_norm(a_norm)
                a[i] += t  # restore
                a[j] -= t
                return sc
            
            f1 = eval_t(x1)
            f2 = eval_t(x2)
            
            for _ in range(15):
                if time.monotonic() - t0 > time_limit:
                    break
                if f1 < f2:
                    t_hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = t_hi - gr * (t_hi - t_lo)
                    f1 = eval_t(x1)
                else:
                    t_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = t_lo + gr * (t_hi - t_lo)
                    f2 = eval_t(x2)
            
            t_opt = (t_lo + t_hi) / 2
            f_opt = min(f1, f2)
            
            if f_opt < best_score - 1e-12:
                a[i] = max(0.0, a[i] - t_opt)
                a[j] = max(0.0, a[j] + t_opt)
                a = a / a.sum()
                best_score = compute_score_norm(a)
                best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
        else:
            no_improve = 0
        
        if verbose:
            pass  # Skip verbose for CD (too slow to print every iteration)
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score, verbose=True):
    """Evaluate candidate via the actual evaluate_sequence and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score
