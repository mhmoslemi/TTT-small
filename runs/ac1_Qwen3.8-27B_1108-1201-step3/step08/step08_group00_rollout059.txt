import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Evaluate sequence safely using the provided evaluate_sequence function."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver_cached = None

def get_lp_solver():

    if _lp_solver_cached is not None:
        return _lp_solver_cached
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])

                _lp_solver_cached = solver_name
                return solver_name
        except Exception:

def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def lbgfs_refine(a0, time_limit, T_start=100.0, T_end=0.1, maxiter=20000):
    """L-BFGS with LogSumExp smoothing and geometric temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def make_func(T_val):
        def func(z_vec):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z_vec)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z_vec):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_vec)
            a = softmax_from_z(z_vec)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    # Adaptive number of stages based on time
    n_stages = 10
    temps = np.geomspace(T_start, T_end, n_stages)

def peak_target_descent(a0, time_limit, n_steps=50000):

    Direct peak-targeting: identify top convolution peaks and reduce them

    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    step_size = 0.02

        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        
        if 2.0 * n * max_b < best_score - 1e-15:
            best_score = 2.0 * n * max_b
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Find top peaks (excluding boundary effects)
        b_interior = b.copy()
        if n > 2:
            b_interior[:min(n//10, 10)] = -np.inf
            b_interior[-min(n//10, 10):] = -np.inf
        peak_positions = np.where(b_interior > max_b * 0.95)[0]
        
        if len(peak_positions) == 0:
            peak_positions = np.array([int(np.argmax(b))])
        
        improved = False
        for k in peak_positions[:3]:
            # Find contributing indices
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k - idx]
            
            # Reduce the top contributor
            top_i = idx[np.argmax(contribs)]
            
            # Try reducing top_i
            old_val = a[top_i]
            new_val = old_val * (1.0 - step_size)
            if new_val < 1e-15:
                continue
            
            a_trial = a.copy()
            a_trial[top_i] = new_val
            diff = old_val - new_val
            # Redistribute to maintain sum

            mask[top_i] = False

            if others.sum() < 1e-15:
                continue
            a_trial[mask] = others + diff * others / others.sum()
            a_trial = a_trial / a_trial.sum()
            
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-12:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            # Try pair reduction
            k = int(np.argmax(b))

def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=20000):
    """Nesterov dual averaging LP refinement (simplified)."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create structured initial sequence."""
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x)
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x / 2) ** 4
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-x**2 * 3)
        a = a / a[-1]
        a = 0.5 * (a + a[::-1])
    else:
        x = np.linspace(0, 1, n)
        a = 1.0 - 0.3 * np.sin(2 * np.pi * x)
        a = a ** 1.5

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def lse_func(z_vec, n_val, T_val):
    a = softmax_from_z(z_vec)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    return (2.0 * n_val / T_val) * log_sum_exp

    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
