def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)

def softmax(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def compute_g_fft(weights, p, n):
    """Compute g[i] = 2 * sum_k weights[k] * p[k-i] using FFT cross-correlation."""
    N = 1 << (3 * n).bit_length()  # N >= 3n-1
    W = np.fft.rfft(weights, N)
    P = np.fft.rfft(p, N)
    cross = np.fft.irfft(W * np.conj(P), N)[:n]
    return 2.0 * cross

    Compute objective and gradient for 2n * smooth_max(conv(p,p)) where p = softmax(w).

def build_conv_matrix(a):
    """Build matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth gradient-based optimization (log-sum-exp + L-BFGS-B) with FFT-based gradients
2. Fast coordinate descent with O(n) sensitivity analysis per coordinate
3. LP-based fictitious play for global improvements
4. Basin hopping for escaping local minima
5. Multi-length exploration with diverse initialization

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def compute_g_fft(weights, p, n):
    """Compute g[i] = 2 * sum_k weights[k] * p[k-i] using FFT cross-correlation."""
    N = 1 << (3 * n).bit_length()
    W = np.fft.rfft(weights, N)
    P = np.fft.rfft(p, N)
    cross = np.fft.irfft(W * np.conj(P), N)[:n]
    return 2.0 * cross


def obj_and_grad(w, n, T):
    """Compute objective and gradient for 2n * smooth_max(conv(p,p)) where p = softmax(w)."""
    p = softmax_fn(w)
    conv = fast_convolve(p)
    c = conv.max()
    exp_T = np.exp(T * (conv - c))
    smax = c + np.log(exp_T.sum()) / T
    weights = exp_T / exp_T.sum()
    obj = 2.0 * n * smax
    g = compute_g_fft(weights, p, n)
    g_dot_p = np.dot(g, p)

        result = minimize(fun_and_grad, w0, jac=True, method='L-BFGS-B', options=options)
        p_opt = softmax_fn(result.x)
        p_opt = np.maximum(p_opt, 0)
        s = p_opt.sum()
        if s > 1e-15:
            p_opt = p_opt / s
        score = 2.0 * n * fast_convolve(p_opt).max()
        if score < best_score:
            best_score = score
            best_p = p_opt.copy()
        if verbose:
            print(f"    L-BFGS (T={T}): {result.message}, iters={result.nit}, score={score:.8f}")
    except Exception as e:
        if verbose:
            print(f"    L-BFGS (T={T}) error: {e}")

    return best_p, best_score


def fast_coord_descent(a0, time_limit, n_trials=12, verbose=False, label=""):
    """Fast coordinate descent with O(n) sensitivity per step."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    sum_a = 1.0
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, conv[:k_start].max())
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, conv[k_end + 1:].max())

            affected = conv[k_start:k_end + 1]
            a_shifted = a

            best_delta = 0.0
            best_new_max = M

            for t in range(n_trials):
                if t == 0:
                    delta = -0.5 * a[i]
                elif t == 1:
                    delta = -0.25 * a[i]
                elif t == 2:
                    delta = -0.125 * a[i]
                elif t == 3:
                    delta = -0.05 * a[i]
                elif t == 4:
                    delta = -0.02 * a[i]
                elif t == 5:
                    delta = -0.005 * a[i]
                elif t == 6:
                    delta = 0.005 * a[i]
                elif t == 7:
                    delta = 0.02 * a[i]
                elif t == 8:
                    delta = 0.05 * a[i]
                elif t == 9:
                    delta = 0.125 * a[i]
                elif t == 10:
                    delta = 0.25 * a[i]
                else:
                    delta = 0.5 * a[i]

                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-18:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-17:
                a[i] += best_delta
                a[i] = max(a[i], 0.0)
                sum_a += best_delta

                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M / (sum_a * sum_a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def build_conv_matrix(a):
    """Build matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    global _lp_solver
    if '_lp_solver' in globals() and _lp_solver is not None:
        return _lp_solver
    _lp_solver = None
    for solver_name in [cp.ECOS, cp.SCS, cp.GLPK, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return _lp_solver
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.GLPK, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    b_val = b_val / s
    return b_val


def fictitious_play(n, time_limit, initial=None, solver=None, verbose=False):
    """Alternating LP best responses (fictitious play)."""
    if solver is None:
        solver = get_lp_solver()
    t0 = time.monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    b = a.copy()
    best_a = a.copy()
    best_score = float('inf')
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        b_new = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
        a_new = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new

        for cand in [a, b, 0.5 * (a + b)]:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand_norm = cand / sc
            score = 2.0 * n * fast_convolve(cand_norm).max()
            if score < best_score:
                best_score = score
                best_a = cand_norm.copy()

        if verbose and iters % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP n={n} it={iters} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score
