The problem can be written as:

- `sum(a) = 1`
- `a_i >= 0`

- Maximize `sum(g)`
- Subject to: `g >= 0`
- `(f*g)_k <= M` for active `k`

This is NOT the same as the original problem. The original problem minimizes `max_k (a*a)_k` subject to `sum(a) = 1, a >= 0`. The LP is a different (dual-like) problem that gives a direction to move.

1. Fix current `f`
2. Find `g` that maximizes mass while keeping `f*g` bounded by current peak `M`
3. Blend `f` towards `g` to reduce the peak

This is a "peeling" or "peaking" approach. The direction `g` is chosen so that when we move towards it, the peaks of `f*f` decrease.

Let me think about the smooth approximation:
`max_k b_k ≈ (1/tau) * log(sum_k exp(tau * b_k))`

The gradient of this w.r.t. `a_i` is:
`d/d a_i [log(sum_k exp(tau * b_k))] = sum_k [tau * b_k' * exp(tau * b_k) / sum_j exp(tau * b_j)]`

where `b_k' = sum_j a_{k-j}` (the derivative of `(a*a)_k` w.r.t. `a_i`).

This gives us a smooth objective whose minimum approximates the original.

"""
Multi-strategy optimization for minimum peak auto-convolution sequences.
Combines LP-based peeling, smooth approximation gradient descent,
coordinate descent, and multi-length exploration.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_cross_convolve(f, g):
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_logsumexp_grad(f, tau):
    """Compute gradient of (1/tau)*log(sum exp(tau*b_k)) w.r.t. f."""
    b = fast_convolve(f)
    scaled = tau * b
    scaled -= scaled.max()  # numerical stability
    w = np.exp(scaled)
    w_sum = w.sum()
    w = w / w_sum  # softmax weights
    
    n = len(f)
    grad = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad[idx] += w[k] * f[k - idx]
    return grad * tau / tau  # (1/tau) * tau * weighted_sum = weighted_sum
    
    # Actually: d/da_i of (1/tau)*log(sum_k exp(tau*b_k))
    # = (1/tau) * sum_k tau * (db_k/da_i) * exp(tau*b_k) / sum_j exp(tau*b_j)
    # = sum_k (db_k/da_i) * w_k
    # where db_k/da_i = f[k-i] if 0<=k-i<n else 0
    # So grad = sum_k w_k * f[k-i] for valid k,i
    # Which is exactly what we computed above (without the tau factor)
    
def smooth_logsumexp_grad_vec(f, tau, n_top=100):
    """Vectorized version: gradient of smooth approximation, restricted to top peaks."""
    b = fast_convolve(f)
    n = len(f)
    
    # Restrict to top peaks for efficiency
    if n_top < len(b):
        top_idx = np.argpartition(b, -n_top)[-n_top:]
    else:
        top_idx = np.arange(len(b))
    
    scaled = tau * b[top_idx]
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    w = w / w_sum
    
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        if w[i] < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad[idx] += w[i] * f[k - idx]
    
    return grad

def lp_peel_direction(f, top_k=50, max_constraints=300):
    """Solve LP to find peeling direction."""
    b = fast_convolve(f)
    M = b.max()
    n = len(f)
    
    if M < 1e-15:
        return None
    
    top_idx = np.argpartition(b, -top_k)[-top_k:]
    top_idx = np.sort(top_idx)
    
    K = len(top_idx)
    A = np.zeros((K, n))
    for row_i in range(K):
        k = int(top_idx[row_i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]
    
    g_var = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), [g_var >= 0, A @ g_var <= M])
    
    for solver in [cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, max_iters=200000)
            if g_var.value is not None and not np.any(np.isnan(g_var.value)):
                g = np.maximum(g_var.value, 0.0)
                return g
        except Exception:
            continue
    
    return None

def coordinate_descent_pass(f, n_iters=3, time_limit=None, t_start=None):
    """Do coordinate descent: for each position, find optimal value."""
    n = len(f)
    f = f.copy()

            # = delta_i * 2 * sum_j a[j] * a[k-i-j] for valid j
            # But this is complicated. Let's just try a few values.
            base_sum = f.sum()
            for scale in [0.5, 0.8, 1.2, 1.5, 0.0]:
                f_trial = f.copy()

def smooth_gradient_descent(f, time_limit, tau_init=50.0, lr_init=0.01, n_top=50):
    """Gradient descent on smooth approximation of max-convolution."""
    t0 = time.monotonic()
    n = len(f)
    f = f.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    tau = tau_init
    lr = lr_init
    v = np.zeros(n)  # momentum
    
    while time.monotonic() - t0 < time_limit:
        grad = smooth_logsumexp_grad_vec(f, tau, n_top)
        
        # Nesterov-like step
        v = 0.9 * v - lr * grad
        f_new = project_simplex(f + v)
        
        obj = fast_convolve(f_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            f = f_new
            lr = min(lr * 1.01, 0.05)
        else:
            f = f_new  # still accept to maintain momentum
            lr *= 0.95
        
        # Increase temperature slowly
        if (int((time.monotonic() - t0) * 10) % 100) == 0:
            tau = min(tau * 1.05, 500.0)
    
    return best_f, best_obj

def peak_shaving(f, time_limit, n_peaks=10):
    """Directly reduce peaks by reducing contributions at peak lags."""
    t0 = time.monotonic()
    n = len(f)
    f = f.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(f)
        top_idx = np.argpartition(b, -n_peaks)[-n_peaks:]
        
        # For each peak, identify which positions contribute most
        # and try to reduce them
        contributions = np.zeros(n)
        for k in top_idx:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                contributions[j] += f[k - j]
        
        # Direction: reduce high-contribution positions, add to low-contribution
        # Project onto simplex after perturbation
        perturbation = contributions - contributions.mean()
        
        best_alpha = None
        for alpha in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1]:
            f_trial = project_simplex(f - alpha * perturbation)
            obj = fast_convolve(f_trial).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_alpha = alpha
                best_f = f_trial.copy()
        
        if best_alpha is None:
            break
        
        f = best_f.copy()
    
    return best_f, best_obj

                obj = fast_convolve(f_trial).max()
                score = 2 * target_n * obj

                    results['best_score'] = score
                    results['best_seq'] = f_trial.copy()
                    results['best_n'] = target_n
        
        budget = budget - t_alloc

"""
Multi-strategy optimization for minimum peak auto-convolution sequences.
Combines LP-based peeling with adaptive constraint generation, smooth log-sum-exp
gradient descent with Nesterov momentum, multi-length exploration via interpolation,
and random restarts to escape local optima. Targets score < 1.5027.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    """Cross-convolution f*g via FFT."""
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_peel_direction(f, top_k=50, max_constraints=300, solver_timeout=15):
    """
    Solve LP: maximize sum(g) s.t. g >= 0, (f*g)_k <= M for active lags k.
    Returns the optimal g (or None if LP fails).
    """
    b = fast_convolve(f)
    M = b.max()
    n = len(f)

    if M < 1e-15:
        return None

    # Select active constraints (top peaks)
    if top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(b))
    top_idx = np.sort(top_idx)

    K = len(top_idx)
    if K > max_constraints:
        # Keep only the most violated (top by b value)
        order = np.argsort(b[top_idx])[::-1][:max_constraints]
        top_idx = top_idx[order]
        K = len(top_idx)

    # Build constraint matrix A where A[k, j] = f[k - j] if valid
    A = np.zeros((K, n))
    for row_i in range(K):
        k = int(top_idx[row_i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]

    g_var = cp.Variable(n)
    constraints = [g_var >= 0, A @ g_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)

    for solver in [cp.PDLP, cp.SCS]:
        try:

            if g_var.value is not None:

                g = np.maximum(g, 0.0)
                if not np.any(np.isnan(g)) and not np.any(np.isinf(g)):
                    return g
        except Exception:
            continue
    return None


def smooth_grad(f, tau, n_top=80):
    """
    Gradient of (1/tau)*log(sum_k exp(tau*(f*f)_k)) w.r.t. f,
    restricted to top n_top peaks for efficiency.
    """
    b = fast_convolve(f)
    n = len(f)

    if n_top < len(b):
        top_idx = np.argpartition(b, -n_top)[-n_top:]
    else:
        top_idx = np.arange(len(b))

    scaled = tau * b[top_idx]
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(len(top_idx))
        w[b[top_idx].argmax()] = 1.0
    else:
        w /= ws

    grad = np.zeros(n)
    for i in range(len(top_idx)):
        if w[i] < 1e-14:

        k = int(top_idx[i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad[idx] += w[i] * f[k - idx]

    return grad

            continue

        S_g = g.sum()
        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > 25:
                break
            continue

        h = g / S_g

        # Adaptive line search: try several step sizes
        best_t = 0.05
        best_obj_t = fast_convolve((1 - 0.05) * f + 0.05 * h).max()
        candidates = [0.01, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]
        for t in candidates:
            f_trial = (1.0 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s > 1e-10:
                f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_obj_t:
                best_obj_t = obj
                best_t = t

        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 35:
                break

        if it % 15 == 0:
            score = 2.0 * n * best_obj
            print(f"    {label} it={it} score={score:.8f} top_k={top_k} t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def random_restart_optimize(f_best, time_limit, n_restarts=5, label="restart"):
    """
    Try random perturbations of the best sequence to escape local optima.
    """
    t0 = time.monotonic()
    n = len(f_best)
    best_f = f_best.copy()
    best_obj = fast_convolve(f_best).max()
    rng = np.random.default_rng(12345)

    t_per_restart = time_limit / n_restarts

    for r in range(n_restarts):
        rem = time_limit - (time.monotonic() - t0)
        if rem < 10:
            break

        # Perturbation magnitude decreases for later restarts
        mag = 0.003 / (1 + 0.3 * r)
        f_pert = f_best + rng.standard_normal(n) * mag
        f_pert = project_simplex(f_pert)

        # Run a short LP peeling from perturbed point
        t_alloc = min(t_per_restart - 5, rem - 5)
        if t_alloc < 5:
            break

        f_opt, obj_opt = lp_peeling_loop(f_pert, t_alloc, top_k_start=40,
                                          top_k_max=120, label=f"{label}_r{r}")

        if obj_opt < best_obj - 1e-14:
            best_obj = obj_opt
            best_f = f_opt.copy()
            score = 2.0 * n * best_obj
            print(f"    {label} restart {r}: NEW BEST score={score:.8f}")

    return best_f, best_obj


def interpolate_sequence(f, target_n):
    """Interpolate a sequence to a new length, preserving non-negativity and normalization."""
    old_x = np.linspace(0.0, 1.0, len(f))
    new_x = np.linspace(0.0, 1.0, target_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    return f_new
