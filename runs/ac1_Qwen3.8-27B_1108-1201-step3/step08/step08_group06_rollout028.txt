import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the optimization score for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def smooth_max_grad(conv, T):
    """Compute the smoothed max value and the probability weights."""
    shifted = conv / T
    shifted -= shifted.max()
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    smooth_val = T * (shifted.max() + np.log(exp_shifted.sum()))  # = T * log(sum(exp(conv/T)))
    return smooth_val, w


def compute_subgradient(a, w):
    """Compute subgradient: 2 * sum_k w[k] * a[k-m] for m = 0..n-1.
    Uses FFT cross-correlation."""
    n = len(a)
    size_w = len(w)  # 2n-1
    # We want R[m] = sum_k w[k] * a[k-m] for m = 0, ..., n-1
    # This is the cross-correlation of w and a.

    # Standard: corr(x, y)[k] = sum_n x[n+k] * y[n]
    # FFT: corr = ifft(FFT(x) * conj(FFT(y)))

    W = rfft(w, L)

def multiplicative_weights_step(a, subgrad, eta):
    """One step of multiplicative weights (mirror descent).
    a_new = a * exp(-eta * subgrad) / sum(a * exp(-eta * subgrad))"""
    exponent = -eta * subgrad
    # For numerical stability
    exponent -= exponent.max()
    weights = np.exp(exponent)
    a_new = a * weights
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    a_new = a_new / s
    return a_new

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    total_iters = 50000
    eta = eta_init
    
    for it in range(total_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Annealing schedule
        progress = min(1.0, it / total_iters)
        T = T_init * (T_final / T_init) ** progress
        
        # Compute convolution
        conv = fast_convolve(a)
        
        # Compute smoothed gradient
        _, w = smooth_max_grad(conv, T)
        subgrad = compute_subgradient(a, w)
        
        # Adaptive step size: scale with 1/sqrt(iteration) and problem parameters
        eta = eta_init / (1.0 + 0.01 * it)
        
        # Multiplicative weights update
        a_new = multiplicative_weights_step(a, subgrad, eta)
        
        # Check improvement
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  MD iter={it}, T={T:.4f}, eta={eta:.4f}, score={score:.8f}, "
                  f"best={best_score:.8f}, elapsed={elapsed:.0f}s")
    
    return best_a, best_score


def build_C_sparse(f):
    """Build the constraint for the LP.
    The constraint is: for each lag k (0 to 2n-2), sum_j f[j] * g[k-j] <= M
    which is: (f * g)(k) <= M for all k.
    In matrix form: C @ g <= M * 1, where C[k, j] = f[k-j] if valid.
    """
    n = len(f)
    return f  # We'll use FFT to compute f*g instead of building C explicitly


def solve_lp_for_g(f, M, method='highs'):
    """Solve LP: max sum(g) s.t. (f*g)(k) <= M for all k, g >= 0.
    Returns the optimal g (normalized to sum=1) and the optimal value."""
    n = len(f)
    # The constraint matrix C has shape (2n-1, n)
    # C[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # Building this explicitly is O(n^2) which might be slow for large n
    # Alternative: use the fact that C is a Toeplitz-like matrix and use a custom solver
    
    # For n up to ~2000, building C explicitly is feasible (4M entries)
    if n <= 1500:
        C = np.zeros((2 * n - 1, n))
        for j in range(n):
            C[j:j+n, j] = f
        c_vec = -np.ones(n)
        b_vec = np.full(2 * n - 1, M)
        res = linprog(c_vec, A_ub=C, b_ub=b_vec, bounds=[(0, None)] * n, method=method)
        if res.status != 0:
            return None, None
        g = np.maximum(res.x, 0)
        total = g.sum()
        if total < 1e-15:
            return None, None
        return g / total, total
    else:
        # For larger n, use a greedy approximation
        return greedy_lp(f, M), None


def greedy_lp(f, M):
    """Greedy approximation to the LP."""
    n = len(f)
    g = np.zeros(n)
    conv_fg = np.zeros(2 * n - 1)
    
    for j in range(n):
        # Max g[j] such that conv_fg[k] + f[k-j]*g[j] <= M for all k in [j, j+n-1]
        k_start = j
        k_end = j + n - 1
        slack = M - conv_fg[k_start:k_end + 1]
        f_slice = f[:]  # f[0], f[1], ..., f[n-1]
        valid = f_slice > 1e-15
        if not valid.any():
            continue
        max_gj = np.min(slack[valid] / f_slice[valid])
        max_gj = max(0.0, max_gj)
        g[j] = max_gj
        conv_fg[k_start:k_end + 1] += f_slice * max_gj
    
    total = g.sum()
    if total < 1e-15:
        return np.ones(n) / n
    return g / total


def lp_fictitious_play(f_init, time_limit, verbose=True, use_greedy=False):
    """LP-based fictitious play with geometric mean updates."""
    f = f_init.copy()
    f = np.maximum(f, 0)
    f = f / f.sum()
    n = len(f)
    
    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()
    
    for k in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M = fast_convolve(f).max()
        
        if use_greedy or n > 1500:
            g = greedy_lp(f, M)
        else:
            g, _ = solve_lp_for_g(f, M)
            if g is None:
                g = greedy_lp(f, M)
        
        if g is None or g.sum() < 1e-15:
            break
        g = g / g.sum()
        
        # Adaptive step size
        t_step = 2.0 / (k + 2)
        t_step = max(0.01, min(1.0, t_step))
        
        # Geometric mean update
        f_safe = np.maximum(f, 1e-30)
        g_safe = np.maximum(g, 1e-30)
        log_new = (1 - t_step) * np.log(f_safe) + t_step * np.log(g_safe)
        log_new -= log_new.max()
        a_new = np.exp(log_new)
        s = a_new.sum()
        if s < 1e-30:
            break
        a_new = a_new / s
        
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_f = a_new.copy()
        
        f = a_new
        
        if verbose and k % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LP iter={k}, M={M:.6e}, score={score:.8f}, "
                  f"best={best_score:.8f}, elapsed={elapsed:.0f}s")
    
    return best_f, best_score


def coordinate_descent(a0, time_limit, verbose=True):
    """Coordinate descent with golden-section search."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    iters = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        
        conv = fast_convolve(a)
        n_pairs = min(n, 300)
        pairs = set()
        
        # Peak-focused pairs
        top_peak = int(np.argmax(conv))
        js = max(0, top_peak - n + 1)
        je = min(n - 1, top_peak)
        for _ in range(n_pairs // 2):
            i = rng.integers(js, je + 1)
            j = rng.integers(js, je + 1)
            if i != j:
                pairs.add((min(i, j), max(i, j)))
        
        # Random pairs
        for _ in range(n_pairs // 2):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                pairs.add((min(i, j), max(i, j)))
        
        for (i, j) in list(pairs):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14 and a[j] < 1e-14:
                continue
            
            t_lo = -a[j] + 1e-15
            t_hi = a[i] - 1e-15
            if t_hi <= t_lo:
                continue
            
            gr = (np.sqrt(5) - 1) / 2
            x1 = t_hi - gr * (t_hi - t_lo)
            x2 = t_lo + gr * (t_hi - t_lo)
            
            def eval_transfer(t):
                a[i] -= t
                a[j] += t
                if a[i] < 0 or a[j] < 0:
                    a[i] += t
                    a[j] -= t
                    return 1e10
                sc = compute_score(a)
                a[i] += t
                a[j] -= t
                return sc
            
            f1 = eval_transfer(x1)
            f2 = eval_transfer(x2)
            
            for _ in range(15):
                if time.monotonic() - t0 > time_limit:
                    break
                if f1 < f2:
                    t_hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = t_hi - gr * (t_hi - t_lo)
                    f1 = eval_transfer(x1)
                else:
                    t_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = t_lo + gr * (t_hi - t_lo)
                    f2 = eval_transfer(x2)
            
            t_opt = (t_lo + t_hi) / 2
            f_opt = min(f1, f2)
            
            if f_opt < best_score - 1e-14:
                a[i] -= t_opt
                a[j] += t_opt
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = compute_score(a)
                best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 3:
                noise = rng.standard_normal(n) * 0.001
                a = best_a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
        else:
            no_improve = 0
        
        if verbose and iters % 10 == 0:
            print(f"  CD iter={iters}, score={best_score:.8f}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate and return updated best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score
