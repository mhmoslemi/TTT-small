Minimize 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Soft-max projected gradient descent (smooth approximation to max)
2. Cutting-plane LP for local refinement
3. Multi-length and multi-start exploration
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time

# --- Helper functions ---

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def cross_convolve(u, v):
    """Linear cross-convolution u*v via FFT. Returns full result."""
    len_u, len_v = len(u), len(v)
    size = len_u + len_v - 1
    fft_size = 1 << (size - 1).bit_length()
    U = rfft(u, fft_size)
    V = rfft(v, fft_size)
    return irfft(U * V, fft_size)[:size]

def project_simplex(v):
    """Projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_sm_gradient(f, w):
    """
    Compute gradient of sum_k w[k] * (f*f)[k] w.r.t. f.
    Result: g[m] = 2 * sum_k w[k] * f[k-m] (valid range only).
    """
    n = len(f)
    f_rev = f[::-1]
    conv = cross_convolve(w, f_rev)
    # g[m] = conv[m + n - 1] for m = 0, ..., n-1
    return 2.0 * conv[n - 1: 2 * n - 1]

def softmax_obj(f, beta):
    """Compute soft-max objective and gradient."""
    b = fast_convolve(f)
    max_b = b.max()
    scaled = beta * (b - max_b)
    e = np.exp(scaled)
    w = e / e.sum()
    obj = (max_b + np.log(e.sum()) / beta)  # = (1/beta) * log(sum exp(beta*b))
    grad = compute_sm_gradient(f, w)
    return obj, grad, b, w

# --- Main optimization routines ---

def softmax_gradient_descent(f_init, time_limit, beta_start=2.0, beta_end=80.0, 
                               momentum=0.85, lr_start=0.5, label="sgd"):
    """
    Soft-max projected gradient descent with momentum and adaptive beta.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = project_simplex(f_init.copy())
    v = np.zeros(n)
    best_f = f.copy()
    
    # Initial evaluation
    b = fast_convolve(f)
    best_peak = b.max()
    
    it = 0
    total_iters = 0
    
    # Phase-based beta schedule
    n_phases = 5
    phase_duration = time_limit / n_phases
    
    for phase in range(n_phases):
        beta = beta_start * (beta_end / beta_start) ** (phase / max(1, n_phases - 1))
        lr = lr_start / (1 + 0.5 * phase)
        phase_start = time.monotonic()
        
        while time.monotonic() - phase_start < phase_duration:
            if time.monotonic() - t0 > time_limit:
                break
            it += 1
            total_iters += 1
            
            # Compute gradient
            obj, grad, b, w = softmax_obj(f, beta)
            
            # Check best
            cur_peak = b.max()
            if cur_peak < best_peak:
                best_peak = cur_peak
                best_f = f.copy()
            
            # Momentum update
            v = momentum * v - lr * grad
            f_new = project_simplex(f + v)
            
            # Simple backtracking if objective worsens significantly
            if it % 100 == 0:
                b_check = fast_convolve(f_new)
                if b_check.max() > best_peak * 1.5:
                    # Too much overshooting, reduce step
                    v *= 0.5
                    f_new = project_simplex(f + v)
            
            f = f_new
            
            if it % 1000 == 0:
                print(f"    {label} phase={phase} beta={beta:.1f} it={it} "
                      f"peak={best_peak:.8f} t={time.monotonic()-t0:.0f}s")
    
    return best_f, best_peak

def lp_refine(f_init, time_limit, initial_k=30, max_k=300, label="lp"):
    """Cutting-plane LP for local refinement."""
    t0 = time.monotonic()
    n = len(f_init)
    f = project_simplex(f_init.copy())
    best_f = f.copy()
    b = fast_convolve(f)
    M = b.max()
    best_M = M
    
    # Initial active set
    if initial_k < len(b):
        active_idx = np.argpartition(b, -initial_k)[-initial_k:]
    else:
        active_idx = np.arange(len(b))
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2:
            break
        
        # Build constraint matrix
        K = len(active_idx)
        A = np.zeros((K, n))
        for i, k in enumerate(active_idx):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A[i, lo:hi+1] = f[lo:hi+1]
        
        # Solve LP: max sum(g) s.t. g >= 0, A@g <= M
        g = cp.Variable(n)
        constraints = [g >= 0, A @ g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=200000, eps=1e-8)
        except Exception:
            try:
                prob.solve(solver=cp.PDPD, max_iters=200000)
            except Exception:
                break
        
        if g.value is None:
            break
        
        g_val = np.maximum(g.value, 0)
        if np.any(np.isnan(g_val)):
            break
        
        S_g = g_val.sum()
        if S_g <= 1.0 + 1e-6:
            # Check for violated constraints
            fg = cross_convolve(f, g_val)
            violated = fg > M * (1 + 1e-6)
            if violated.any() and len(active_idx) < max_k:
                v_idx = np.where(violated)[0]
                new_idx = v_idx[~np.isin(v_idx, active_idx)]
                if len(new_idx) > 0:
                    n_add = min(20, len(new_idx))
                    to_add = new_idx[np.argsort(fg[new_idx])[-n_add:]]
                    active_idx = np.sort(np.concatenate([active_idx, to_add]))
                    active_idx = active_idx[-max_k:]
            else:
                break
        
        h = g_val / S_g
        
        # Line search
        best_t = 0.05
        best_t_obj = M
        for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            ft = np.maximum((1-t)*f + t*h, 0)
            s = ft.sum()
            if s > 1e-10:
                ft /= s
                obj = fast_convolve(ft).max()
                if obj < best_t_obj:
                    best_t_obj = obj
                    best_t = t
        
        f = np.maximum((1-best_t)*f + best_t*h, 0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        
        cur_M = fast_convolve(f).max()
        if cur_M < best_M:
            best_M = cur_M
            best_f = f.copy()
            M = cur_M
        
        if it % 10 == 0:
            print(f"    {label} it={it} M={best_M:.8f} t={time.monotonic()-t0:.0f}s")
    
    return best_f, best_M

def make_cosine_window(n):
    """Create a cosine window of length n, normalized to sum=1."""
    x = np.linspace(0, np.pi, n)
    w = 0.5 * (1 + np.cos(x))
    w /= w.sum()
    return w

def make_kaiser_window(n, beta=5.0):
    """Create a Kaiser window of length n, normalized to sum=1."""
    from scipy.signal import kaiser
    w = kaiser(n, beta)
    w = np.maximum(w, 0)
    w /= w.sum()
    return w

def make_gaussian_window(n, sigma_frac=0.3):
    """Create a Gaussian window of length n, normalized to sum=1."""
    x = np.arange(n)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    w = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    w /= w.sum()
    return w
