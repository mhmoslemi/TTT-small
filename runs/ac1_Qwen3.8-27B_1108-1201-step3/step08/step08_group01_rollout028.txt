import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a_fixed):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

                new_score = 2.0 * n * new_max / (new_sum * new_sum)

                if new_max < best_new_max:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-16:
                a[i] = max(a[i] + best_delta, 0.0)
                sum_a += best_delta

                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx_2i] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M / (sum_a * sum_a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > 30:
                break
        else:
            no_improve_sweeps = 0

        if verbose and sweeps % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, sweeps


def fine_coord_descent(a0, time_limit, verbose=False, label=""):
    """Fine coordinate descent with very small steps for final polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    sum_a = 1.0
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            a_shifted = a

            best_delta = 0.0
            best_new_max = M

            for factor in [0.98, 0.96, 0.94, 0.99, 1.01, 1.03, 1.05, 1.07]:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = conv[k_start:k_end + 1] + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx_2i] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
                else:
                    new_affected = new_affected.copy()

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-17:
                a[i] = max(a[i] + best_delta, 0.0)
                sum_a += best_delta

                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx_2i] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M / (sum_a * sum_a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved and sweeps > 10:
            break

        if verbose and sweeps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FineCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, sweeps


def block_coord_descent(a0, time_limit, block_size=3, verbose=False):
    """Block coordinate descent: update blocks of coordinates jointly."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    sum_a = 1.0
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        # Process blocks
        n_blocks = n // block_size
        for b_idx in range(n_blocks):
            idx = np.arange(b_idx * block_size, (b_idx + 1) * block_size)
            if a[idx].max() < 1e-15:
                continue

            # Determine affected range
            k_start = idx[0]
            k_end = min(idx[-1] + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            # Try small perturbations of the block
            for scale in [0.95, 0.98, 1.02, 1.05]:
                delta_vec = (scale - 1.0) * a[idx]
                new_vals = a[idx] + delta_vec
                if new_vals.min() < -1e-15:
                    continue

                # Compute change in conv
                new_conv_range = conv[k_start:k_end + 1].copy()
                for j, i in enumerate(idx):
                    ks = i
                    ke = min(i + n - 1, 2 * n - 2)
                    # Clip to our range
                    cs = max(ks, k_start)
                    ce = min(ke, k_end)
                    if cs > ce:
                        continue
                    a_seg = a[cs - i:ce - i + 1]
                    new_conv_range[cs - k_start:ce - k_start + 1] += 2.0 * delta_vec[j] * a_seg
                    # Handle the quadratic term at k=2i
                    if 2 * i >= k_start and 2 * i <= k_end:
                        idx2 = 2 * i - k_start
                        new_conv_range[idx2] += delta_vec[j] * delta_vec[j]
                        # Correct the linear term (we added 2*delta*a[i] but should add 2*delta*(a[i]+delta/2) + ... )
                        # Actually the correct update is: new_val = old + 2*delta*a[i] + delta^2
                        # We added 2*delta*a[i], so we need to add delta^2 (already done above)
                        # But we also need to subtract delta*a[i] from the linear term... 
                        # Actually: (a[i]+d)^2 - a[i]^2 = 2*a[i]*d + d^2
                        # And for k != 2i: (a[i]+d)*a[k-i] - a[i]*a[k-i] = d*a[k-i]
                        # So for k=2i: change = 2*d*a[i] + d^2 (correct as computed)
                        # For k != 2i: change = d*a[k-i] (we used 2*d*a[k-i] which is wrong!)
                        # Wait no - for auto-conv, conv[k] = sum_i a[i]*a[k-i]
                        # If we change a[i] by d, then:
                        #   conv[k] changes by d*a[k-i] + a[i]*d (if k-i != i, i.e., k != 2i)
                        #   = 2*d*a[k-i] (if k != 2i)
                        #   = 2*d*a[i] + d^2 (if k == 2i, since both terms get the change)
                        # Actually no: conv[k] = sum_j a[j]*a[k-j]
                        # If we change a[i] by d:
                        #   Terms involving a[i]: a[i]*a[k-i] and a[k-i]*a[i] (if k-i != i)
                        #   So change = d*a[k-i] + a[k-i]*d = 2*d*a[k-i] (if k-i != i and k-i valid)
                        #   If k-i == i (i.e., k = 2i): change = (a[i]+d)^2 - a[i]^2 = 2*d*a[i] + d^2
                        #   But wait, if k-i == i, then the term is a[i]*a[i] and it appears only ONCE in the sum
                        #   So change = (a[i]+d)^2 - a[i]^2 = 2*d*a[i] + d^2
                        # This is what we have. Good.
                        pass

                new_max = max(unaff_max, float(new_conv_range.max()))
                new_sum = sum_a + delta_vec.sum()
                if new_sum < 0.01:
                    continue

                if new_max < M - 1e-16:
                    # Apply update
                    a[idx] = new_vals
                    conv[k_start:k_end + 1] = new_conv_range
                    M = new_max
                    sum_a = new_sum
                    improved = True

                    cur_score = 2.0 * n * M / (sum_a * sum_a)
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()
                    break

        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    BlockCD sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def fictitious_play(n, time_limit, initial=None, solver=None, verbose=False):
    """
    Fictitious play for the zero-sum game:
    min_a max_b a^T (sum_k y_k T_k) b
    At Nash equilibrium, a = b gives the optimal solution.
    """
    if solver is None:
        solver = get_lp_solver()
    t0 = time.monotonic()
    if initial is None:
        a = init_u_shape(n, np.random.default_rng(42))
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = init_u_shape(n, np.random.default_rng(42))
        else:
            a = a / s

    b = a.copy()
    a_avg = a.copy().astype(np.float64)
    best_a = a.copy()
    best_score = float('inf')
    iters = 0
    last_improve_iter = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1

        # Get best response to current a
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new

        # Get best response to current b
        a_new, M_a = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new

        # Osher averaging
        a_avg = (iters * a_avg + a) / (iters + 1)

        # Evaluate candidates
        for cand in [a, b, 0.5 * (a + b), a_avg]:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand_norm = cand / sc
            score = 2.0 * n * fast_convolve(cand_norm).max()
            if score < best_score:
                best_score = score
                best_a = cand_norm.copy()
                last_improve_iter = iters

        if verbose and iters % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP n={n} it={iters} ({elapsed:.0f}s): best={best_score:.6f}")

        # Early stopping if no improvement in many iterations
        if iters - last_improve_iter > 100 and iters > 50:
            break

    return best_a, best_score, iters


def damped_lp_refine(a0, time_limit, solver=None):
    """Damped alternating LP refinement with adaptive damping."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0

    for it in range(10000):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            continue
        alpha = min(1.0, 5.0 / (it + 1))
        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                alpha = max(0.1, alpha * 0.5)
                no_improve = 0

    return best_a, best_score


def mirror_descent(a0, time_limit, lr_init=0.01, top_k=30, verbose=False):
    """Mirror descent (multiplicative weight update)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        score = 2.0 * n * M

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 100:
            lr *= 0.8
            no_improve = 0
            if lr < 1e-7:
                break

        # Compute subgradient using top-k peaks with softmax weighting
        k_act = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]
        temp = 0.05 * max(pv.max(), 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        w = w / ws

        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += 2.0 * w[idx] * a[k - j]

        # Mirror descent update
        scaled = a * np.exp(-lr * grad)
        a = simplex_project(scaled)
        lr *= 0.9995

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD it={it} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_u_shape(n, rng, sharpness=None):
    """U-shaped initialization (edge-concentrated)."""
    if sharpness is None:
        sharpness = rng.uniform(1.0, 4.0)
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    # U-shape: high at edges, low in middle
    a = 1.0 + sharpness * np.cos(np.pi * t) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_fourier(n, rng):
    """Initialize using random Fourier superposition."""
    n_harmonics = int(rng.integers(3, 20))
    amplitudes = rng.random(n_harmonics) * 0.4
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = np.arange(1, n_harmonics + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_cosine_window(n, rng):
    """Initialize with a raised cosine (Hanning) window shape."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    a = 0.5 * (1 - np.cos(2 * np.pi * t))
    # Add some randomness
    a += rng.uniform(0, 0.2) * (1 - a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_sharp_u(n, rng):
    """Very sharp U-shape: most mass at the very edges."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    alpha = rng.uniform(2.0, 8.0)
    a = 1.0 + alpha * (2 * t - 1) ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def perturb_and_optimize(a0, rng, n_perturb, time_limit, label=""):
    """Perturb a sequence randomly, then optimize."""
    n = len(a0)
    a = a0.copy()
    # Random perturbation
    mask = rng.random(n) < 0.3
    a[mask] *= (1.0 + rng.standard_normal(mask.sum()) * 0.1)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = a0.copy()
    else:
        a = a / s
    a_opt, s_opt, _ = fast_coord_descent(a, time_limit, n_trials=8, label=label)
    return a_opt, s_opt


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate():

    Combines LP fictitious play, coordinate descent, mirror descent,
    and diverse multi-length search.

            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    # === Phase 1: LP Fictitious Play (main optimization) ===
    print("\n=== Phase 1: LP Fictitious Play ===")
    t1_start = time.monotonic()

                best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "post_md_cd")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-Length (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        alt_lengths = []
        for alt_n in [n_best * 2, n_best + 256, n_best - 256, 1536, 2048, 1280]:
            if 256 <= alt_n <= 3000 and alt_n != n_best and alt_n not in alt_lengths:
                alt_lengths.append(alt_n)

        for alt_n in alt_lengths[:4]:
            rem = deadline - time.monotonic() - 200
            if rem < 20:
                break
            t_alloc = min(rem * 0.3, 120.0)
            init_alt = interpolate_seq(best_seq, alt_n)
            a_alt, s_alt, _ = fast_coord_descent(init_alt, t_alloc, n_trials=10,
                                                 label=f"alt{alt_n}")
            best_seq, best_score = check_and_update(a_alt, best_seq, best_score, f"alt_n{alt_n}")

    # === Phase 5: Diverse initializations ===
    print(f"\n=== Phase 5: Diverse Inits (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            inits = [
                init_u_shape(n_best, rng, sharpness=3.0),
                init_sharp_u(n_best, rng),
                init_cosine_window(n_best, rng),
                init_fourier(n_best, rng),
            ]
            # Also try noisy version of best
            noisy = best_seq + rng.standard_normal(n_best) * 0.05
            noisy = np.maximum(noisy, 0)
            noisy = noisy / noisy.sum()
            inits.append(noisy)

            t_per = max(5.0, (rem - 10) / len(inits))
            for idx, init in enumerate(inits):
                rem = deadline - time.monotonic() - 100
                if rem < 5:
                    break
                a_d, s_d, _ = fast_coord_descent(init, t_per, n_trials=8,
                                                 label=f"div{idx}")
                best_seq, best_score = check_and_update(a_d, best_seq, best_score, f"div{idx}")

    # === Phase 6: Block coordinate descent ===
    print(f"\n=== Phase 6: Block CD (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_blk = min(rem * 0.5, 80.0)
            a_blk, s_blk = block_coord_descent(best_seq, t_blk, block_size=3, verbose=True)
            best_seq, best_score = check_and_update(a_blk, best_seq, best_score, "block_cd")
            # Follow up with regular CD
            rem = deadline - time.monotonic() - 80
            if rem > 10:
                a_f, s_f, _ = fast_coord_descent(a_blk, min(rem * 0.6, 50.0), n_trials=10, label="post_blk")
                best_seq, best_score = check_and_update(a_f, best_seq, best_score, "post_blk_cd")

    # === Phase 7: Final polishing ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        # Fine coordinate descent
        rem = deadline - time.monotonic() - 50
        if rem > 20:
            t_fine = min(rem * 0.5, 120.0)
            print(f"  FineCD n={len(best_seq)}, budget={t_fine:.0f}s")
            a_fine, s_fine, _ = fine_coord_descent(best_seq, t_fine, verbose=True, label="fine")
            best_seq, best_score = check_and_update(a_fine, best_seq, best_score, "fine_cd")

        # Another fast CD round
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            a_cd2, s_cd2, _ = fast_coord_descent(best_seq, min(rem * 0.7, 80.0),
                                                 n_trials=16, label="last_cd")
            best_seq, best_score = check_and_update(a_cd2, best_seq, best_score, "last_cd")

    # Fallback
    if best_seq is None:
        best_seq = init_u_shape(1024, rng, sharpness=2.0)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
