"""
Optimization using Adam with logsumexp smoothing for the auto-convolution peak minimization.
Multi-restart, multi-n exploration, with subgradient polish.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_smooth_gradient(a, w, fft_size):
    """
    Compute gradient of logsumexp(T, c) with respect to a.
    w is the softmax weight vector of length 2n-1.
    Returns gradient of length n.
    """
    n = len(a)
    # S1[m] = cross-correlation of w and a at lag m (m = 0..n-1)
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    S1 = irfft(W * np.conj(A), fft_size)[:n]
    # S2[m] = convolution of a and w at lag m (m = 0..n-1)
    S2 = irfft(A * W, fft_size)[:n]
    return S1 + S2

def adam_optimize(f_init, time_limit, T_init=0.01, T_final=0.001, lr=0.01, 
                  beta1=0.9, beta2=0.999, eps=1e-8, anneal_every=200):
    """
    Adam optimizer with logsumexp smoothing.
    f_init is normalized (sum=1, non-negative).
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    step = 0
    T = T_init
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        # Anneal temperature
        if step % anneal_every == 0:
            T = max(T_final, T * 0.95)
        
        # Compute convolution
        c = fast_convolve(f)
        
        # Compute softmax weights
        c_scaled = c / T
        c_scaled -= c_scaled.max()  # Numerical stability
        w = np.exp(c_scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(c))
            w[c.argmax()] = 1.0
        else:
            w /= w_sum
        
        # Compute gradient
        grad = compute_smooth_gradient(f, w, fft_size)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)
        
        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)
        
        # Check objective
        if step % 50 == 0 or step == 1:
            obj = fast_convolve(f_new).max()
            if obj < best_obj:
                best_obj = obj
                best_f = f_new.copy()
            f = f_new
    
    return best_f, best_obj


def subgradient_polish(f_init, time_limit, n_top=20, lr=0.005):
    """Subgradient descent on the top-k peaks."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    v = np.zeros(n)  # Momentum
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(f)
        # Get top-k peaks
        if n_top < len(c):
            top_idx = np.argpartition(c, -n_top)[-n_top:]
        else:
            top_idx = np.arange(len(c))
        
        # Compute subgradient (average gradient at top peaks)
        grad = np.zeros(n)
        for k in top_idx:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                grad[j] += f[k - j]
        grad /= len(top_idx)
        
        # Momentum step
        v = 0.9 * v - lr * grad
        f_new = project_simplex(f + v)
        
        obj = fast_convolve(f_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        f = f_new
        
        if no_improve > 100:
            lr *= 0.9
            no_improve = 0
            if lr < 1e-6:
                break
        
        if step % 500 == 0:
            print(f"    sg: obj={best_obj:.10f}")
    
    return best_f, best_obj

        c_scaled -= c_scaled.max()
        w = np.exp(c_scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(c))
            w[c.argmax()] = 1.0
        else:
            w /= w_sum

        # Compute gradient via FFT
        grad = compute_gradient_fft(f, w, fft_size)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)
        f = f_new

        # Periodically check objective
        if step % check_every == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 5:
                    # Reduce learning rate and reset momentum
                    lr *= 0.85
                    m *= 0.5
                    v *= 0.5
                    no_improve = 0
                    if lr < 1e-7:
                        break

    return best_f, best_obj


def subgradient_polish(f_init, time_limit, n_top=20, lr_init=0.003):
    """
    Subgradient descent focusing on top-k peaks with momentum.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    momentum = np.zeros(n)
    lr = lr_init
    no_improve = 0
    step = 0
    fft_size = 1 << (3 * n - 2 - 1).bit_length()

    while time.monotonic() - t0 < time_limit:
        step += 1
        c = fast_convolve(f)

        # Get top-k peaks
        if n_top < len(c):
            top_idx = np.argpartition(c, -n_top)[-n_top:]
        else:
            top_idx = np.arange(len(c))

        # Build weight vector from top peaks (uniform weights)
        w_top = np.zeros(len(c))
        w_top[top_idx] = 1.0 / len(top_idx)

        # Compute gradient via FFT
        grad = compute_gradient_fft(f, w_top, fft_size)

        # Momentum step
        momentum = 0.9 * momentum - lr * grad
        f_new = project_simplex(f + momentum)
        f = f_new

        if step % 50 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 10:
                    lr *= 0.9
                    no_improve = 0
                    if lr < 1e-8:
                        break

    return best_f, best_obj


def make_gaussian_init(n, sigma_frac=0.25):
    """Gaussian-centered initialization."""
    x = np.arange(n)
    center = (n - 1) / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_raised_cosine_init(n):
    """Raised cosine (Hann-like) initialization."""
    x = np.linspace(0, 1, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_asymmetric_init(n, shift=0.1):
    """Asymmetric initialization: shifted and truncated Gaussian."""
    x = np.arange(n)
    center = n * (0.5 - shift)
    sigma = n * 0.2
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_bimodal_init(n, sep=0.3):
    """Bimodal initialization."""
    x = np.arange(n)
    c1 = n * (0.5 - sep)
    c2 = n * (0.5 + sep)
    sigma = n * 0.1
    a = np.exp(-0.5 * ((x - c1) / sigma) ** 2) + np.exp(-0.5 * ((x - c2) / sigma) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_tukey_init(n, r=0.5):
    """Tukey (tapered cosine) window initialization."""
    x = np.arange(n)
    a = np.ones(n)
    edges = int(r * n)
    if edges > 0:
        t = np.arange(edges) / edges
        taper = 0.5 * (1 + np.cos(np.pi * (2 * t / r - 1)))
        a[:edges] = taper
        a[-edges:] = taper[::-1]
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def interpolate_sequence(src, target_n):
    """Interpolate a sequence to a new length."""
    old_n = len(src)
    if old_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    a = np.interp(new_x, old_x, src)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    Phase 1: Quick Adam exploration across multiple (n, init) combinations.
    Phase 2: Refined Adam on top candidates with lower temperature.
    Phase 3: Subgradient polish on the best.
    Phase 4: Final LP-free refinement with very small steps.
    """
    deadline = time.monotonic() + max(10.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={parent_score:.8f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1024) / 1024.0
        parent_score = 2.0
        print("No parent, using uniform n=1024")

    best_seq = parent.copy()
    best_score = parent_score

    # === Phase 1: Quick exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 1: Quick Adam exploration")
    print(f"{'='*60}")

    # Candidate configurations: (n, init_function, label)
    n_parent = len(parent)
    configs = []

    # Parent at its own length
    configs.append((n_parent, lambda: parent.copy(), "parent"))

    # Parent interpolated to other lengths
    for target_n in [1024, 1280, 1536, 2048, 2560]:
        if target_n != n_parent and 512 <= target_n <= 3000:
            configs.append((target_n, lambda tn=target_n: interpolate_sequence(parent, tn),
                           f"parent_interp_{target_n}"))

    # Different initializations at selected lengths
    for target_n in [n_parent, 1536, 2048]:
        if 512 <= target_n <= 3000:
            configs.append((target_n, lambda tn=target_n: make_gaussian_init(tn, 0.25),
                           f"gauss_{target_n}"))
            configs.append((target_n, lambda tn=target_n: make_raised_cosine_init(tn),
                           f"raised_cos_{target_n}"))
            configs.append((target_n, lambda tn=target_n: make_asymmetric_init(tn, 0.1),
                           f"asym_{target_n}"))
            configs.append((target_n, lambda tn=target_n: make_bimodal_init(tn, 0.15),
                           f"bimodal_{target_n}"))
            configs.append((target_n, lambda tn=target_n: make_tukey_init(tn, 0.4),
                           f"tukey_{target_n}"))

    # Random initializations
    for target_n in [n_parent, 2048]:
        if 512 <= target_n <= 3000:
            configs.append((target_n, lambda tn=target_n: project_simplex(
                rng.standard_normal(tn)), f"random_{target_n}"))

    print(f"Total configs: {len(configs)}")

    # Allocate time per config
    phase1_time = min(700.0, (deadline - time.monotonic()) * 0.4)
    time_per_config = phase1_time / len(configs)
    candidates = []

    for idx, (n, init_fn, label) in enumerate(configs):
        rem = deadline - time.monotonic()
        if rem < time_per_config * 2:
            break
        t_alloc = min(time_per_config, rem * 0.9)
        if t_alloc < 5:
            break

        f_init = init_fn()
        if f_init is None or len(f_init) != n or f_init.sum() < 1e-10:
            continue

        f_best, obj_best = adam_optimize(
            f_init, time_limit=t_alloc,
            T_init=0.01, T_min=0.001, lr=0.008,
            anneal_every=50, check_every=50
        )
        score = float(evaluate_sequence(f_best.tolist()))
        candidates.append((score, f_best, n, label))
        marker = " *" if score < best_score else ""
        print(f"  [{idx+1}/{len(configs)}] {label}: n={n}, score={score:.8f}{marker}")

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = f_best.copy()

    # Sort candidates by score
    candidates.sort(key=lambda x: x[0])
    print(f"\nPhase 1 best: {candidates[0][3]} score={candidates[0][0]:.8f}")

    # === Phase 2: Refined optimization on top candidates ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Refined Adam on top candidates")
    print(f"{'='*60}")

    phase2_time = min(600.0, (deadline - time.monotonic()) * 0.35)
    top_k = min(4, len(candidates))
    time_per_candidate = phase2_time / top_k

    for idx in range(top_k):
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        score_c, f_c, n_c, label_c = candidates[idx]
        t_alloc = min(time_per_candidate, rem * 0.9)
        if t_alloc < 15:
            break

        # Try multiple Adam configurations
        best_for_cand = f_c.copy()
        best_score_cand = score_c

        for lr_try in [0.003, 0.005, 0.002]:
            rem_inner = deadline - time.monotonic()
            if rem_inner < 10:
                break
            t_inner = min(t_alloc / 3.0, rem_inner * 0.9)
            if t_inner < 5:
                break

            f_r, obj_r = adam_optimize(
                f_c, time_limit=t_inner,
                T_init=0.003, T_min=0.0003, lr=lr_try,
                anneal_every=80, check_every=80
            )
            s_r = float(evaluate_sequence(f_r.tolist()))
            if np.isfinite(s_r) and s_r < best_score_cand:
                best_score_cand = s_r
                best_for_cand = f_r.copy()

        if best_score_cand < best_score:
            best_score = best_score_cand
            best_seq = best_for_cand.copy()
            print(f"  Refined [{label_c}]: score={best_score:.8f} *NEW BEST*")
        else:
            print(f"  Refined [{label_c}]: score={best_score_cand:.8f}")

    # === Phase 3: Subgradient polish ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Subgradient polish (n={len(best_seq)}, score={best_score:.8f})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    t3 = min(rem * 0.25, 300.0)
    if t3 > 15:
        n_b = len(best_seq)
        print(f"  Budget: {t3:.0f}s, n={n_b}")
        f_sg, obj_sg = subgradient_polish(best_seq, time_limit=t3, n_top=30, lr_init=0.002)
        score_sg = float(evaluate_sequence(f_sg.tolist()))
        if np.isfinite(score_sg) and score_sg < best_score:
            best_score = score_sg
            best_seq = f_sg.copy()
            print(f"  NEW BEST [sg]: {best_score:.8f}")
        else:
            print(f"  sg: {score_sg:.8f} (no improvement)")

    # === Phase 4: Final fine-tuning with very low temperature ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Final fine-tuning (score={best_score:.8f})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    t4 = min(rem * 0.3, 200.0)
    if t4 > 10:
        print(f"  Budget: {t4:.0f}s")
        f_fin, obj_fin = adam_optimize(
            best_seq, time_limit=t4,
            T_init=0.001, T_min=0.0001, lr=0.001,
            anneal_every=200, check_every=100
        )
        score_fin = float(evaluate_sequence(f_fin.tolist()))
        if np.isfinite(score_fin) and score_fin < best_score:
            best_score = score_fin
            best_seq = f_fin.copy()
            print(f"  NEW BEST [final]: {best_score:.8f}")
        else:
            print(f"  final: {score_fin:.8f} (no improvement)")

    # === Phase 5: One more subgradient pass ===
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.5, 100.0)
    if t5 > 10:
        f_sg2, obj_sg2 = subgradient_polish(best_seq, time_limit=t5, n_top=50, lr_init=0.001)
        score_sg2 = float(evaluate_sequence(f_sg2.tolist()))
        if np.isfinite(score_sg2) and score_sg2 < best_score:
            best_score = score_sg2
            best_seq = f_sg2.copy()
            print(f"  NEW BEST [sg2]: {best_score:.8f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"Target: 1.5027")
    print(f"Difference: {best_score - 1.5027:.6f}")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
