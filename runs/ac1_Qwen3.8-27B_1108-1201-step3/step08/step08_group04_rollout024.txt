def fast_convolve(a):

    A = rfft(a, fft_size)

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def softmax_gradient(f, temperature=0.01, n_top=100):
    """Compute smoothed gradient of max(conv(f,f))."""

    full_len = 2 * n - 1

"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) FFT-based smoothed gradient descent with mirror descent,
(2) away-step Frank-Wolfe, (3) coordinate descent polish,
(4) multi-start with diverse initializations, (5) multi-length search.

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_gradient(f, temperature, n_top=None):
    """Compute softmax-smoothed gradient of max(conv(f,f))."""
    n = len(f)
    full_len = 2 * n - 1
    fft_size = 1 << (full_len - 1).bit_length()
    
    b = fast_convolve(f)
    M = b.max()
    
    # Softmax weights
    scaled = (b - M) / temperature
    scaled = np.clip(scaled, -100, 0)
    w = np.exp(scaled)
    
    if n_top is not None and n_top < full_len:
        # Zero out weights except top n_top
        top_idx = np.argpartition(b, -n_top)[-n_top:]
        w_masked = np.zeros(full_len)
        w_masked[top_idx] = w[top_idx]
        w = w_masked
    
    w_sum = w.sum()
    if w_sum < 1e-30:
        # Fallback
        g = np.zeros(n)
        k_star = np.argmax(b)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        g[lo:hi+1] = f[k_star - np.arange(lo, hi+1)]
        return g, M, b
    
    w = w / w_sum
    
    # g[j] = sum_k w[k] * f[k-j] = cross-correlation(w, f)[j]
    w_padded = np.zeros(fft_size)
    w_padded[:full_len] = w
    f_padded = np.zeros(fft_size)
    f_padded[:n] = f
    
    W = rfft(w_padded, fft_size)
    F = rfft(f_padded, fft_size)
    g = irfft(W * np.conj(F), fft_size)[:n]
    
    return g, M, b


def mirror_descent_step(f, g, lr):
    """Mirror descent update on the simplex."""
    # a_i <- a_i * exp(-lr * g_i)
    # This maintains non-negativity
    f_new = f * np.exp(-lr * g)
    s = f_new.sum()
    if s < 1e-30:
        return f.copy()
    return f_new / s


def adam_step(f, g, v, m, lr, beta1=0.9, beta2=0.999, eps=1e-8, t=1):
    """Adam optimizer step with simplex projection."""
    v = beta1 * v + (1 - beta1) * g
    m = beta2 * m + (1 - beta2) * (g * g)
    v_hat = v / (1 - beta1**t)
    m_hat = m / (1 - beta2**t)
    f_new = f - lr * v_hat / (np.sqrt(m_hat) + eps)
    return project_simplex(f_new), v, m


def randomized_smoothed_optimize(f_init, time_limit, seed=0, 
                                  temp_init=None, temp_min=0.001,
                                  lr_init=0.05, use_mirror=True,
                                  use_adam=False, n_top=None,
                                  label="opt"):

    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s
    
    if temp_init is None:
        temp_init = 0.1 * n  # Initial temperature (in units of b[k])
        # Actually, b[k] has magnitude ~1/n, so temperature should be ~1/n
        temp_init = 0.05 / n
    
    temp = temp_init
    lr = lr_init
    v = np.zeros(n)
    m = np.zeros(n)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)
    t_count = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        t_count += 1
        
        # Adapt temperature: decrease slowly
        temp = max(temp_min, temp_init * 0.995**it)
        
        # Adapt learning rate

            lr *= 0.9
        elif no_improve == 0:
            lr = min(lr * 1.02, lr_init * 2)
        
        g, M, b = compute_smoothed_gradient(f, temp, n_top)
        
        # Choose update method
        if use_mirror and (not use_adam or it % 3 != 0):

            no_improve += 1

            f = f / f.sum()
            v = np.zeros(n)
            m = np.zeros(n)

def frank_wolfe_optimize(f_init, time_limit, seed=0, label="fw"):
    """Away-step Frank-Wolfe for the smoothed objective."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(seed)
    temp = 0.05 / n
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        temp = max(0.001 / n, temp * 0.999)
        
        g, M, b = compute_smoothed_gradient(f, temp)
        
        # Frank-Wolfe step
        # Move towards argmin(g)
        j_min = np.argmin(g)
        # Away step: move away from argmax(g * f)
        load = g * f
        j_away = np.argmax(load)
        
        # Step size
        t_fw = 2.0 / (it + 3)
        
        f_new = f.copy()
        f_new[j_min] += t_fw
        if f_new[j_away] > t_fw * 0.5:
            f_new[j_away] -= t_fw * 0.5
        
        f_new = np.maximum(f_new, 0)
        s_new = f_new.sum()
        if s_new > 1e-15:
            f_new = f_new / s_new
            f = f_new
        
        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 30:
            # Perturb
            f = best_f + rng.standard_normal(n) * 0.003
            f = np.maximum(f, 0)
            f = f / f.sum()
            no_improve = 0
            
        if it % 100 == 0:
            print(f"  {label} it={it} score={2*n*best_obj:.6f} t={time.monotonic()-t0:.0f}s")
    
    return best_f, 2.0 * n * best_obj

    best_f = f.copy()

    multipliers = [0.3, 0.5, 0.7, 0.8, 0.9, 0.95, 1.05, 1.1, 1.2, 1.4, 1.7, 2.0, 3.0]
    
    while time.monotonic() - t0 < time_limit:
        any_improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                # Try turning on

                    f_trial = f.copy()
                    f_trial[j] = mult / n
                    f_trial = f_trial / f_trial.sum()
                    obj = fast_convolve(f_trial).max()
                    if obj < best_obj - 1e-14:
                        f = f_trial
                        best_obj = obj
                        best_score = 2.0 * n * best_obj
                        best_f = f.copy()
                        any_improved = True
                        break
                continue
            
            best_j_obj = best_obj
            best_j_val = f[j]
            for mult in multipliers:
                f_trial = f.copy()
                f_trial[j] = f[j] * mult
                s = f_trial.sum()
                if s < 1e-12:
                    continue
                f_trial = f_trial / s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-14:
                    best_j_obj = obj
                    best_j_val = f[j] * mult
            
            if best_j_val != f[j]:
                f[j] = best_j_val
                f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:

            if no_pass_improve >= 2:

def interp_sequence(src, target_n):
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result


def generate_initialization(n, method, rng):
    """Generate diverse initializations."""
    if method == 'uniform':
        f = np.ones(n) / n
    elif method == 'sin2':
        f = np.sin(np.pi * np.arange(n) / n)**2
        f = np.maximum(f, 0) / f.sum()
    elif method == 'triangle':
        mid = (n - 1) / 2.0
        f = 1 - np.abs(np.arange(n) - mid) / (mid + 1)
        f = np.maximum(f, 0) / f.sum()
    elif method == 'bimodal':
        # Two peaks
        center1 = n * 0.25
        center2 = n * 0.75
        width = n * 0.1
        f = np.exp(-(np.arange(n) - center1)**2 / (2*width**2)) + \
            np.exp(-(np.arange(n) - center2)**2 / (2*width**2))
        f = f / f.sum()
    elif method == 'random':
        f = rng.random(n)
        f = f / f.sum()
    elif method == 'random_sparse':
        f = rng.random(n)
        f[f < 0.7] = 0
        f = f / f.sum()
    elif method == 'sine_mod':
        # Modulated sine
        f = (1 + 0.5*np.sin(2*np.pi*3*np.arange(n)/n)) * np.sin(np.pi*np.arange(n)/n)**2
        f = np.maximum(f, 0) / f.sum()
    else:
        f = np.ones(n) / n
    f = np.maximum(f, 0)
    s = f.sum()
    if s > 1e-15:
        f = f / s
    return f

def griffiths_stern(f_init, time_limit, target_ratio=1.0):
    """
    Griffiths-Stern-like peak reduction.
    target_ratio: target max_b / (sum_a^2 / (2n-1))
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    f = f / f.sum()
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    target = target_ratio * 1.0 / (2*n - 1)  # target_b = ratio * average_b
    epsilon = 1e-6
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()
        
        if M <= target + epsilon:
            break
        
        # Find all peaks above target
        peaks = np.where(b > target)[0]
        
        # For each peak, reduce the contributing terms
        for k in peaks:
            # b[k] = sum_i f[i]*f[k-i]
            # Find the largest term
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            i_range = np.arange(lo, hi + 1)
            terms = f[i_range] * f[k - i_range]
            max_term_idx = np.argmax(terms)
            i_star = i_range[max_term_idx]
            j_star = k - i_star
            
            # Reduce both f[i_star] and f[j_star]
            # Amount to reduce: we want to reduce b[k] by (b[k] - target)
            # If we reduce f[i] by delta, b[k] decreases by approximately delta * f[j]
            excess = b[k] - target
            delta = min(excess / (f[j_star] + 1e-15), f[i_star] * 0.1)
            delta = min(delta, f[j_star] * 0.1)
            
            if i_star == j_star:
                # Diagonal term: f[i]^2, reduction needs 2*delta*f[i]
                delta = min(excess / (2*f[i_star] + 1e-15), f
