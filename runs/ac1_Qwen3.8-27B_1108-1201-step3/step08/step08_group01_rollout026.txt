import numpy as np
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:

    try:

                prob.solve(solver=alt)
                break

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s

def exact_1d_minimize(f, lo, hi, tol=1e-14, max_iter=200):
    """Golden section search for 1D minimization."""
    gr = (np.sqrt(5) - 1) / 2  # golden ratio ~0.618
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < tol:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) / 2

def score_from_conv(a, conv):
    """Compute the score given a and its convolution."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    M = conv.max()
    return 2.0 * n * M / (s * s)

def priority_coord_descent(a0, time_limit, verbose=False, label=""):

        s = 1.0
    else:
        a = a / s
        s = 1.0

    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0

        sweeps += 1
        improved = False

        # Determine priority: coordinates that contribute to peaks

        cur_score = 2.0 * n * M / (a.sum() ** 2)

            best_score = cur_score

        for i in order:

                # Recompute affected part
                # Actually, let's just recompute the full conv for correctness

                M = conv.max()
                improved = True

def lp_refine(a0, time_limit, solver=None, verbose=False, label=""):

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def peak_perturb(a0, rng, magnitude=0.05):
    """Perturb coordinates that contribute to peaks to escape local optima."""
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    top_k = min(5, len(conv))
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    for k in peaks:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1)
        n_pick = max(1, len(idx) // 5)
        picks = rng.choice(idx, size=min(n_pick, len(idx)), replace=False)
        for i in picks:
            a[i] *= (1.0 + rng.standard_normal() * magnitude)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a0.copy()
    return a / s

def mirror_descent_refine(a0, time_limit, lr_init=0.01, verbose=False):
    """Mirror descent with peak-weighted subgradient."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        score = 2.0 * n * M

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            lr *= 0.95
            no_improve = 0
            if lr < 1e-7:
                break

        # Softmax-weighted subgradient over peaks
        k_top = min(50, len(conv))
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        pv = conv[peaks]
        temp = 0.05 * (pv.max() - pv.min() + 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        w = w / w.sum()

        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for j in range(i_start, i_end + 1):
                grad[j] += 2.0 * w[idx] * a[k - j]

        # Mirror descent step
        scaled = a * np.exp(-lr * grad)
        a = simplex_project(scaled)
        lr *= 0.9995

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD it={it} ({elapsed:.0f}s): best={best_score:.8f} lr={lr:.6f}")

    return best_a, best_score

def check_and_update(a, best_seq, best_score, label=""):

5. Careful time management to maximize progress within 1800s
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    global _lp_solver
    if '_lp_solver' in globals():
        try:
            if _lp_solver is not None:
                return _lp_solver
        except Exception:
            pass
    _lp_solver = None
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.CLARABEL
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k].
    Also returns dual variables for the convolution constraints."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    conv_constraints = T @ b_var <= M_var
    constraints = [
        conv_constraints,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, None
    b_val = b_val / s
    M_val = float(M_var.value)

    # Try to get dual variables
    dual = None
    try:
        dual_val = np.asarray(conv_constraints.dual_value, dtype=np.float64)
        if dual_val is not None and np.isfinite(dual_val).all():
            dual = np.maximum(dual_val, 0)
    except Exception:
        pass

    return b_val, M_val, dual


def smart_coord_descent(a0, time_limit, verbose=False, label=""):
    """
    Coordinate descent with smart step selection:
    - Tries equalization-aware steps (making top peaks equal)
    - Uses incremental convolution updates
    - Priority ordering based on peak contribution
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        sweep_best = best_score

        # Determine update order: prioritize coords contributing to top peaks
        top_k_peaks = min(8, 2 * n - 1)
        peak_indices = np.argpartition(conv, -top_k_peaks)[-top_k_peaks:]
        peak_vals = conv[peak_indices]
        peak_order = peak_indices[np.argsort(peak_vals)[::-1]]

        priority = np.zeros(n)
        for k in peak_order:
            weight = conv[k] / max(M, 1e-15)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                j = k - i
                if j < n:
                    priority[i] += weight * a[j] * a[i]
        order = np.argsort(priority)[::-1]

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-16:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            a_shifted = a  # a[k-i] for k in [i,...,i+n-1] is a[0],a[1],...,a[n-1]
            affected_start = k_start
            affected = conv[affected_start:k_end + 1]

            cur_score = 2.0 * n * M / (s * s)

            best_delta = 0.0
            best_new_score = cur_score

            # Generate candidate deltas
            # 1. Equalization: find t that makes affected peak = unaff_max
            # 2. Standard geometric steps
            candidates = []

            # Try to equalize the max affected value with unaff_max
            aff_max_idx = int(np.argmax(affected))
            aff_max_val = affected[aff_max_idx]
            if aff_max_val > unaff_max and a_shifted[aff_max_idx] > 1e-15:
                # Want: affected[aff_max_idx] + 2*delta*a_shifted[aff_max_idx] = unaff_max
                # (ignoring quadratic term for now)
                t_eq = (unaff_max - aff_max_val) / (2.0 * a_shifted[aff_max_idx])
                if -a[i] * 0.99 < t_eq < a[i] * 3:
                    candidates.append(t_eq)

            # Also try to equalize top 2 affected values
            if len(affected) > 1:
                sorted_idx = np.argsort(affected)[::-1]
                if len(sorted_idx) >= 2:
                    i1, i2 = sorted_idx[0], sorted_idx[1]
                    v1, v2 = affected[i1], affected[i2]
                    s1, s2 = a_shifted[i1], a_shifted[i2]
                    if abs(s1 - s2) > 1e-15:
                        t_eq2 = (v2 - v1) / (2.0 * (s1 - s2))
                        if -a[i] * 0.99 < t_eq2 < a[i] * 3:
                            candidates.append(t_eq2)

            # Geometric steps
            for factor in [0.3, 0.5, 0.7, 0.85, 0.93, 0.97, 1.03, 1.07, 1.15, 1.3, 1.6, 2.5]:
                candidates.append(a[i] * (factor - 1.0))

            # Also try zeroing
            candidates.append(-a[i])

            # Evaluate candidates
            for delta in candidates:
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue
                new_sum = s + delta
                if new_sum < 1e-10:
                    continue

                new_affected = affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx_2i] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_max = max(unaff_max, float(new_affected.max()))
                new_score = 2.0 * n * new_max / (new_sum * new_sum)

                if new_score < best_new_score - 1e-15:
                    best_new_score = new_score
                    best_delta = delta

            if best_delta != 0.0:
                a[i] = max(a[i] + best_delta, 0.0)
                s += best_delta

                # Incremental convolution update
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx_2i] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                improved = True

                cur_score = 2.0 * n * M / (s * s)
                if cur_score < sweep_best:
                    sweep_best = cur_score

        if best_score > sweep_best:
            best_score = sweep_best
            best_a = a.copy()

        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f} M={M:.8f}")

    return best_a, best_score


def lp_subgradient_descent(a0, time_limit, solver=None, verbose=False, label=""):
    """
    Uses LP to get dual variables (subgradient of M) and performs
    projected gradient descent on the simplex.
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    lr = 0.1 / n
    no_improve = 0
    avg_grad = np.zeros(n)

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Get LP solution and dual
        b_lp, M_lp, dual = solve_lp_best_response(a, solver)
        if b_lp is None:
            break

        if dual is not None and dual.sum() > 1e-12:
            # Use dual as subgradient
            # grad_j = sum_k dual_k * (partial conv_k / partial a_j) = sum_k dual_k * (2*a_{k-j} for valid k)
            # But dual is for the LP with fixed a, so it gives info about conv(a, b) not conv(a, a)
            # Instead, use the subgradient of max_k conv(a,a)[k] directly
            pass

        # Direct subgradient of M = max_k (a*a)[k]
        # At the current point, the subgradient is in the convex hull of
        # {grad_k : k in argmax(conv)}
        # Use softmax weighting over top peaks for a smooth approximation
        top_k = min(30, 2 * n - 1)
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        pmax = pv.max()
        pmin = pv.min()
        temp = 0.1 * max(pmax - pmin, 1e-10)
        w = np.exp((pv - pmax) / temp)
        w = w / w.sum()

        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for j in range(i_start, i_end + 1):
                grad[j] += 2.0 * w[idx] * a[k - j]

        # Averaged gradient (Polyak averaging)
        avg_grad = (it * avg_grad + grad) / (it + 1)

        # Gradient step with projection to simplex
        a_new = a - lr * avg_grad
        a_new = simplex_project(a_new)

        # Also try the LP solution mixed in
        mix = min(1.0, 5.0 / (it + 1))
        a_mix = (1.0 - mix) * a_new + mix * b_lp
        a_mix = np.maximum(a_mix, 0)
        a_mix = a_mix / a_mix.sum()

        # Pick the better one
        score_new = 2.0 * n * fast_convolve(a_new).max()
        score_mix = 2.0 * n * fast_convolve(a_mix).max()

        if score_mix < score_new:
            a = a_mix
            cur_score = score_mix
        else:
            a = a_new
            cur_score = score_new

        conv = fast_convolve(a)

        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 100:
            lr *= 0.8
            no_improve = 0
            if lr < 1e-8:
                break

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-GD {label} it={it} ({elapsed:.0f}s): best={best_score:.8f} lr={lr:.2e}")

    return best_a, best_score


def lp_alternating(a0, time_limit, solver=None, verbose=False, label=""):
    """
    Alternating LP with strong duality exploitation.
    At each iteration, solve LP for best response, then mix.
    Uses the LP value as a lower bound certificate.
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0

    # Exponential moving average of iterates
    a_ema = a.copy()
    ema_decay = 0.9

    while time.monotonic() - t0 < time_limit:
        it += 1

        b_new, M_lp, dual = solve_lp_best_response(a, solver)
        if b_new is None:
            break

        # The LP value M_lp is a lower bound on the optimal M
        # (since the true optimum requires both sides to be optimal simultaneously)
        lower_bound = 2.0 * n * M_lp

        # Try multiple mixing strategies
        candidates = []
        for alpha in [min(1.0, 3.0 / (it + 1)), 0.5, 0.3]:
            cand = (1.0 - alpha) * a + alpha * b_new
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc > 1e-15:
                cand = cand / sc
                candidates.append(cand)

        # Also try the EMA
        a_ema = ema_decay * a_ema + (1.0 - ema_decay) * b_new
        a_ema = a_ema / a_ema.sum()
        candidates.append(a_ema)

        best_cand_score = best_score
        best_cand = a
        for cand in candidates:
            sc = 2.0 * n * fast_convolve(cand).max()
            if sc < best_cand_score:
                best_cand_score = sc
                best_cand = cand

        if best_cand_score < best_score - 1e-14:
            best_score = best_cand_score
            best_a = best_cand.copy()

        a = best_cand

        if verbose and it % 25 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Alt {label} it={it} ({elapsed:.0f}s): best={best_score:.8f} LB={lower_bound:.8f}")

        # Convergence check
        if it > 10 and best_score - lower_bound < 1e-6:
            break

    return best_a, best_score


def paired_coord_descent(a0, time_limit, verbose=False, label=""):
    """
    Updates pairs of coordinates (i, j) that jointly contribute to a peak.
    For peak at position k = i + j, reducing both a[i] and a[j] can be more effective.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        # Find top peaks
        top_k = min(5, 2 * n - 1)
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peaks = peaks[np.argsort(conv[peaks])[::-1]]

        for k in peaks:
            if time.monotonic() - t0 > time_limit:
                break

            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)

            # Find the dominant pair contributing to this peak
            contribs = []
            for i in range(i_start, i_end + 1):
                j = k - i
                if j < n:
                    contribs.append((a[i] * a[j], i, j))
            contribs.sort(reverse=True)

            # Try reducing the top contributing pair
            for _, i, j in contribs[:3]:
                if i == j:
                    # Single coordinate
                    delta = -0.1 * a[i]
                    new_val = a[i] + delta
                    if new_val < 0:
                        continue
                    a_try = a.copy()
                    a_try[i] = new_val
                    conv_try = conv.copy()
                    ks = i
                    ke = min(i + n - 1, 2 * n - 2)
                    a_sh = a  # sensitivity
                    conv_try[ks:ke + 1] += 2.0 * delta * a_sh
                    if 2 * i >= ks and 2 * i <= ke:
                        conv_try[2 * i] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
                    new_M = conv_try.max()
                    new_s = s + delta
                    new_score = 2.0 * n * new_M / (new_s * new_s)
                    if new_score < best_score - 1e-14:
                        a = a_try
                        s = new_s
                        conv = conv_try
                        M = new_M
                        best_score = new_score
                        best_a = a.copy()
                        improved = True
                else:
                    # Paired update: reduce both a[i] and a[j]
                    for frac in [0.05, 0.1, 0.2, 0.3]:
                        di = -frac * a[i]
                        dj = -frac * a[j]
                        ni = a[i] + di
                        nj = a[j] + dj
                        if ni < 0 or nj < 0:
                            continue
                        a_try = a.copy()
                        a_try[i] = ni
                        a_try[j] = nj
                        conv_try = fast_convolve(a_try)
                        new_s = s + di + dj
                        if new_s < 1e-10:
                            continue
                        new_score = 2.0 * n * conv_try.max() / (new_s * new_s)
                        if new_score < best_score - 1e-14:
                            a = a_try
                            s = new_s
                            conv = conv_try
                            M = conv_try.max()
                            best_score = new_score
                            best_a = a.copy()
                            improved = True
                            break

        if not improved and sweeps > 20:
            break

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def peak_perturb(a0, rng, magnitude=0.03):
    """Perturb coordinates contributing to peaks."""
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    top_k = min(5, len(conv))
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    for k in peaks:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1)
        n_pick = max(1, len(idx) // 4)
        picks = rng.choice(idx, size=min(n_pick, len(idx)), replace=False)
        for i in picks:
            a[i] *= (1.0 + rng.standard_normal() * magnitude)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a0.copy()
    return a / s


def mirror_descent_refine(a0, time_limit, lr_init=0.02, verbose=False):
    """Mirror descent (exponentiated gradient) for the max-convolution problem."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        score = 2.0 * n * M

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            lr *= 0.9
            no_improve = 0
            if lr < 1e-8:
                break

        # Softmax-weighted subgradient
        k_top = min(40, len(conv))
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        pv = conv[peaks]
        temp = 0.05 * (pv.max() - pv.min() + 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        w = w / w.sum()

        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for j in range(i_start, i_end + 1):
                grad[j] += 2.0 * w[idx] * a[k - j]

        scaled = a * np.exp(-lr * grad)
        a = simplex_project(scaled)
        lr *= 0.999

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD it={it} ({elapsed:.0f}s): best={best_score:.8f} lr={lr:.6f}")

    return best_a, best_score


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate():
    """
    Multi-phase optimization combining:
    1. Smart coordinate descent with equalization-aware steps
    2. LP alternating method with lower bound tracking
    3. LP subgradient descent using peaked subgradients
    4. Paired coordinate updates for peak elimination
    5. Mirror descent for smooth global exploration
    6. Multi-length exploration and final polishing
    """
    deadline = time.monotonic() + 1785.0
    rng = np.random.default_rng(12345)
    solver = None
    try:
        solver = get_lp_solver()
        print(f"Using LP solver: {solver}")
    except Exception as e:
        print(f"LP solver unavailable: {e}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default init: n=1024, score={best_score:.8f}")

    # === Phase 1: Smart coordinate descent (most likely to improve quickly) ===
    print(f"\n=== Phase 1: Smart CD (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.2, 250.0)
    print(f"  Time budget: {t1:.0f}s, n={len(best_seq)}")
    a1, s1 = smart_coord_descent(best_seq, t1, verbose=True, label="p1")
    best_seq, best_score = check_and_update(a1, best_seq, best_score, "p1_scd")

    # === Phase 2: LP alternating (global improvement) ===
    print(f"\n=== Phase 2: LP Alternating (best={best_score:.8f}) ===")
    if solver is not None:
        rem = deadline - time.monotonic()
        t2 = min(rem * 0.2, 200.0)
        print(f"  Time budget: {t2:.0f}s, n={len(best_seq)}")
        a2, s2 = lp_alternating(best_seq, t2, solver=solver, verbose=True, label="p2")
        best_seq, best_score = check_and_update(a2, best_seq, best_score, "p2_lp")

    # === Phase 3: Smart CD again (exploit LP improvements) ===
    print(f"\n=== Phase 3: Smart CD round 2 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.15, 150.0)
    if t3 > 20:
        print(f"  Time budget: {t3:.0f}s")
        a3, s3 = smart_coord_descent(best_seq, t3, verbose=True, label="p3")
        best_seq, best_score = check_and_update(a3, best_seq, best_score, "p3_scd")

    # === Phase 4: Paired coordinate descent (peak elimination) ===
    print(f"\n=== Phase 4: Paired CD (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.1, 100.0)
    if t4 > 15:
        print(f"  Time budget: {t4:.0f}s")
        a4, s4 = paired_coord_descent(best_seq, t4, verbose=True, label="p4")
        best_seq, best_score = check_and_update(a4, best_seq, best_score, "p4_pcd")

    # === Phase 5: Mirror descent (smooth global optimization) ===
    print(f"\n=== Phase 5: Mirror Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.1, 100.0)
    if t5 > 20:
        print(f"  Time budget: {t5:.0f}s")
        a5, s5 = mirror_descent_refine(best_seq, t5, lr_init=0.015, verbose=True)
        best_seq, best_score = check_and_update(a5, best_seq, best_score, "p5_md")

    # === Phase 6: LP subgradient (if still time) ===
    print(f"\n=== Phase 6: LP Subgradient (best={best_score:.8f}) ===")
    if solver is not None:
        rem = deadline - time.monotonic()
        t6 = min(rem * 0.15, 120.0)
        if t6 > 30:
            print(f"  Time budget: {t6:.0f}s")
            a6, s6 = lp_subgradient_descent(best_seq, t6, solver=solver, verbose=True, label="p6")
            best_seq, best_score = check_and_update(a6, best_seq, best_score, "p6_lpgd")

    # === Phase 7: Perturbation + re-optimization ===
    print(f"\n=== Phase 7: Perturb + Optimize (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        for mag in [0.02, 0.01]:
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            pert = peak_perturb(best_seq, rng, magnitude=mag)
            t_p = min(rem * 0.25, 80.0)
            a_p, _ = smart_coord_descent(pert, t_p, label=f"pert{mag}")
            best_seq, best_score = check_and_update(a_p, best_seq, best_score, f"pert_{mag}")

    # === Phase 8: Try different length ===
    print(f"\n=== Phase 8: Alt Length (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 50:
        n_cur = len(best_seq)
        for alt_n in [n_cur + 128, n_cur - 128, 1152, 896]:
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            if alt_n < 256 or alt_n > 2560:
                continue
            if alt_n == n_cur:
                continue
            init_alt = interpolate_seq(best_seq, alt_n)
            t_a = min(rem * 0.3, 80.0)
            a_a, _ = smart_coord_descent(init_alt, t_a, label=f"alt{alt_n}")
            best_seq, best_score = check_and_update(a_a, best_seq, best_score, f"alt_{alt_n}")

    # === Phase 9: Final intensive polish ===
    print(f"\n=== Phase 9: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_f = min(rem * 0.5, 120.0)
        a_f, _ = smart_coord_descent(best_seq, t_f, verbose=True, label="final")
        best_seq, best_score = check_and_update(a_f, best_seq, best_score, "final")

    # Final LP polish if time remains
    if solver is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            t_fl = min(rem * 0.5, 60.0)
            a_fl, _ = lp_alternating(best_seq, t_fl, solver=solver, label="final_lp")
            best_seq, best_score = check_and_update(a_fl, best_seq, best_score, "final_lp")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
