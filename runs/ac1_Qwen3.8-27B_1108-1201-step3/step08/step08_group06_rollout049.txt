Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast Exponentiated Gradient (EG) with adaptive temperature and momentum
2. LP-based fictitious play with geometric mean updates for polishing
3. Coordinate descent with golden-section search for final refinement
4. Multi-start with diverse initializations
5. Length exploration
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Score for normalized sequence (sum=1). Lower is better."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def cross_correlate(a, p):
    """
    Compute (a ⋆ p)[m] = sum_l a[l] * p[m+l] for m = 0..n-1.
    a has length n, p has length 2n-1.
    """
    n = len(a)
    fft_size = 1 << (n + 2*n - 1 - 1 - 1).bit_length()  # n + (2n-1) - 1 = 3n-2
    A = rfft(a, fft_size)
    P = rfft(p, fft_size)
    cross = irfft(A * np.conj(P), fft_size)
    return cross[:n]


def eg_step(a, T, eta, momentum_beta=0.0, log_a=None, v=None):
    """
    One step of exponentiated gradient with optional momentum in log-space.
    Returns new a, new log_a, new v.
    """
    n = len(a)
    conv = fast_convolve(a)
    
    # Softmax weights
    shifted = conv / T
    shifted -= shifted.max()
    p = np.exp(shifted)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.ones(len(conv)) / len(conv)
    else:
        p /= p_sum
    
    # Gradient (without the 2/T factor, absorbed into eta)
    g = cross_correlate(a, p)
    
    if momentum_beta > 0 and log_a is not None and v is not None:
        # Nesterov momentum in log-space
        new_v = momentum_beta * v - eta * g
        new_log_a = log_a + new_v
        new_log_a -= new_log_a.max()
        new_a = np.exp(new_log_a)
        s = new_a.sum()
        if s < 1e-300:
            return a, log_a, v
        new_a /= s
        return new_a, new_log_a, new_v
    else:
        # Simple EG without momentum
        new_log_a = np.log(np.maximum(a, 1e-300)) - eta * g
        new_log_a -= new_log_a.max()
        new_a = np.exp(new_log_a)
        s = new_a.sum()
        if s < 1e-300:
            return a, log_a, v
        new_a /= s
        return new_a, new_log_a, v

    a = np.maximum(a, 1e-15)
    a /= a.sum()
    
    log_a = np.log(a)
    v = np.zeros(n)
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    total_decay = max_iters
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive temperature
        progress = min(1.0, it / (total_decay * 0.8))
        T = T_init * (T_min / T_init) ** progress
        
        # Step size: decrease over time
        eta = eta_init / (1.0 + it * 0.001)
        
        a, log_a, v = eg_step(a, T, eta, momentum_beta, log_a, v)
        
        # Check score (every 10 iters to save time)
        if it % 10 == 0:
            sc = compute_score(a)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()
        
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            sc = compute_score(a)
            print(f"    EG it={it}, T={T:.4f}, score={sc:.8f}, best={best_score:.8f}, t={elapsed:.0f}s")
    
    # Final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score


def build_C(f):
    """Build constraint matrix C where C[k,j] = f[k-j] if 0<=k-j<n else 0."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = f
    return C


def solve_lp_scipy(f, M):
    """Solve LP: max sum(g) s.t. C*g <= M*1, g >= 0 using scipy linprog."""
    n = len(f)
    C = build_C(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=C, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if res.status != 0:
        return None
    return np.maximum(res.x, 0)


def greedy_lp(f, M):
    """Greedy approximation to the LP."""
    n = len(f)
    g = np.zeros(n)
    conv_vals = np.zeros(2 * n - 1)
    
    for j in range(n):
        k_start = j
        k_end = j + n - 1
        slack = M - conv_vals[k_start:k_end + 1]
        f_slice = f[:n]
        valid = f_slice > 1e-15
        if not valid.any():
            continue
        max_gj = np.inf
        for idx in range(n):
            if f_slice[idx] > 1e-15:
                cand = slack[idx] / f_slice[idx]
                if cand < max_gj:
                    max_gj = cand
        max_gj = max(0.0, max_gj)
        g[j] = max_gj
        conv_vals[k_start:k_end + 1] += f_slice * max_gj
    
    return g


def lp_fictitious_play(f_init, time_limit, max_iters=5000, use_greedy=False, verbose=True):
    """LP-based fictitious play with geometric mean updates."""
    f = f_init.copy()
    f = np.maximum(f, 1e-15)
    f = f / f.sum()
    n = len(f)
    
    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()
    
    # Also track running average for better results
    f_avg = f.copy()
    
    for k in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        M = fast_convolve(f).max()
        
        if use_greedy:
            g = greedy_lp(f, M)
        else:
            g = solve_lp_scipy(f, M)
            if g is None:
                g = greedy_lp(f, M)
        
        if g is None or g.sum() < 1e-15:
            break
        
        g = g / g.sum()
        
        # Geometric mean update with adaptive step
        t_step = min(0.5, 2.0 / (k + 2))
        f_safe = np.maximum(f, 1e-30)
        g_safe = np.maximum(g, 1e-30)
        log_new = (1 - t_step) * np.log(f_safe) + t_step * np.log(g_safe)
        log_new -= log_new.max()
        a_new = np.exp(log_new)
        a_new /= a_new.sum()
        
        # Update running average
        f_avg = (k * f_avg + a_new) / (k + 1)
        
        sc_cur = compute_score(a_new)
        sc_avg = compute_score(f_avg)
        
        if sc_cur < best_score - 1e-13:
            best_score = sc_cur
            best_f = a_new.copy()
        if sc_avg < best_score - 1e-13:
            best_score = sc_avg
            best_f = f_avg.copy()
        
        f = a_new
        
        if verbose and k % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP it={k}, M={M:.6e}, best={best_score:.8f}, t={elapsed:.0f}s")
    
    return best_f, best_score


def coordinate_descent(a0, time_limit, max_rounds=100, verbose=True):
    """Coordinate descent with golden-section search for pair transfers."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        
        improved = False
        conv = fast_convolve(a)
        
        # Focus on pairs involving the top peak
        top_peak = int(np.argmax(conv))
        js = max(0, top_peak - n + 1)
        je = min(n - 1, top_peak)
        
        # Try random pairs
        n_pairs = min(200, n)
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            
            if _ < n_pairs // 3:
                # Peak-focused
                i = int(rng.integers(js, je + 1))
                j = int(rng.integers(js, je + 1))
            elif _ < 2 * n_pairs // 3:
                # Random
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
            else:
                # Edge
                if rng.integers(0, 2) == 0:
                    i = int(rng.integers(0, max(2, n // 10)))
                    j = int(rng.integers(0, n))
                else:
                    i = int(rng.integers(0, n))
                    j = int(rng.integers(n - n // 10, n))
            
            if i == j or a[i] < 1e-14 or a[j] < 1e-14:
                continue
            
            # Golden section search for best transfer
            t_lo = -a[j] + 1e-15
            t_hi = a[i] - 1e-15
            if t_hi <= t_lo:
                continue
            
            gr = (np.sqrt(5) - 1) / 2
            x1 = t_hi - gr * (t_hi - t_lo)
            x2 = t_lo + gr * (t_hi - t_lo)
            
            def eval_transfer(t):
                a[i] -= t
                a[j] += t
                if a[i] < -1e-15 or a[j] < -1e-15:
                    a[i] += t
                    a[j] -= t
                    return 1e10
                s = a.sum()
                a_norm = a / s
                sc = compute_score(a_norm)
                a[i] += t
                a[j] -= t
                return sc
            
            f1 = eval_transfer(x1)
            f2 = eval_transfer(x2)
            
            for _ in range(15):
                if time.monotonic() - t0 > time_limit:
                    break
                if f1 < f2:
                    t_hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = t_hi - gr * (t_hi - t_lo)
                    f1 = eval_transfer(x1)
                else:
                    t_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = t_lo + gr * (t_hi - t_lo)
                    f2 = eval_transfer(x2)
            
            t_opt = (t_lo + t_hi) / 2
            f_opt = min(f1, f2)
            
            if f_opt < best_score - 1e-12:
                a[i] -= t_opt
                a[j] += t_opt
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = compute_score(a)
                best_a = a.copy()
                improved = True
        
        if verbose and rnd % 10 == 0:
            print(f"    CD round={rnd}, best={best_score:.8f}")
    
    return best_a, best_score


def make_initializations(n, rng):
    """Generate diverse initial sequences of length n."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Power law decay
    j = np.arange(n)
    a = (j + 1) ** (-0.5)
    inits.append(a / a.sum())
    
    # Power law rise
    a = (n - j) ** (-0.5)
    inits.append(a / a.sum())
    
    # Asymmetric power law
    a = (j + 1) ** (-0.3) * (n - j) ** (-0.3)
    inits.append(a / a.sum())
    
    # Gaussian centered
    mu = n // 2
    sigma = n / 4
    a = np.exp(-0.5 * ((j - mu) / sigma) ** 2)
    if a.sum() > 0:
        inits.append(a / a.sum())
    
    # Gaussian off-center
    mu = n // 3
    sigma = n / 5
    a = np.exp(-0.5 * ((j - mu) / sigma) ** 2)
    if a.sum() > 0:
        inits.append(a / a.sum())
    
    # Linear decreasing
    a = np.linspace(1, 0.1, n)
    inits.append(a / a.sum())
    
    # Linear increasing
    a = np.linspace(0.1, 1, n)
    inits.append(a / a.sum())
    
    # Random Dirichlet
    for alpha in [0.5, 1.0, 2.0]:
        a = rng.dirichlet(np.full(n, alpha))
        inits.append(a)
    
    # "U-shaped" (high at edges, low in middle)
    a = 1.0 / (np.sin(np.pi * (j + 0.5) / n) + 0.1)
    inits.append(a / a.sum())
    
    return inits


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, best_seq, best_score):
    """Check if a is better and update."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        return a_norm.copy(), score
    return best_seq, best_score
