import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    if '_lp_solver' in globals():
        try:
            if _lp_solver is not None:
                return _lp_solver
        except Exception:
            pass
    _lp_solver = None
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.GLPK, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.GLPK, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def compute_subgradient(a, top_k=10, temperature=0.005):
    """
    Compute a (soft) subgradient of max_k conv(a,a)[k].
    Uses softmax weighting over the top-k peaks.
    Returns: subgradient vector of length n, and the max value M.
    """
    n = len(a)
    conv = fast_convolve(a)
    M = float(conv.max())
    
    # Softmax weighting over all (or top-k) peaks
    if top_k < len(conv):
        # Only consider top_k
        top_indices = np.argpartition(conv, -top_k)[-top_k:]
        w = np.zeros(len(conv))
        top_vals = conv[top_indices]
        w_vals = np.exp((top_vals - top_vals.max()) / max(temperature, 1e-10))
        w[top_indices] = w_vals
        w = w / w.sum()
    else:
        w = np.exp((conv - M) / max(temperature, 1e-10))
        w = w / w.sum()
    
    # Compute subgradient: grad[i] = 2 * sum_k w[k] * a[k-i]
    # This is 2 * cross_correlation(w, a)[i]
    # where cross_correlation(w, a)[i] = sum_k w[k] * a[k-i]
    
    # Using FFT for efficiency
    # cross_correlation(w, a)[i] = ifft(fft(w) * conj(fft(a)))[i]
    # But w has length 2n-1 and a has length n.
    # We need to be careful about padding to avoid circular effects.
    
    # The result has length (2n-1) + n - 1 = 3n - 2 (for cross-correlation)
    # But we only need indices 0 to n-1.
    # For the computation sum_k w[k] * a[k-i], the valid range of k-i is [0, n-1],
    # so k ranges from i to i+n-1. Since k is in [0, 2n-2], i is in [0, n-1].
    
    # Let's use a direct FFT approach:
    # Pad both to a size that prevents wraparound
    fft_size = 1 << (3 * n - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    
    # Cross-correlation: c[m] = sum_k w[k] * a[k+m]
    # = ifft(W * conj(A))
    # But we want d[i] = sum_k w[k] * a[k-i] = c[-i]
    # c is periodic with period fft_size, so c[-i] = c[fft_size - i] for i > 0
    
    c = np.fft.irfft(W * np.conj(A), fft_size)
    # c[m] = sum_k w[k] * a[(k+m) mod fft_size]
    # For our purposes (fft_size > 3n), this equals the linear cross-correlation.
    
    # d[i] = c[-i mod fft_size] = c[fft_size - i] for i = 1, ..., n-1, and c[0] for i=0
    subgrad = np.zeros(n)
    subgrad[0] = 2.0 * c[0]
    if n > 1:
        subgrad[1:] = 2.0 * c[fft_size - 1: fft_size - n: -1]
    
    return subgrad, M, w, conv

def subgradient_momentum(a0, time_limit, top_k=15, temp_init=0.01, verbose=False, label=""):
    """
    Subgradient descent with Nesterov momentum and backtracking line search.
    Optimizes max_k conv(a,a)[k] on the simplex.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    
    # Momentum state
    y = a.copy()  # lookahead point
    v = np.zeros(n)  # velocity
    beta = 0.9  # momentum coefficient
    beta_decay = 0.99999
    lr = 0.05  # initial learning rate
    lr_decay = 0.9999
    temperature = temp_init
    temp_decay = 0.9999
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    f_best = best_score
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute subgradient at y (lookahead)
        subgrad, M_y, w, conv_y = compute_subgradient(y, top_k=top_k, temperature=temperature)
        
        # Backtracking line search
        best_trial = None
        best_trial_score = float('inf')
        
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.125, lr * 0.0625]:
            # Momentum step
            new_v = beta * v - trial_lr * subgrad
            new_y = y + new_v
            new_a = simplex_project(new_y)
            
            # Evaluate
            conv_new = fast_convolve(new_a)
            score_new = 2.0 * n * float(conv_new.max())
            
            if score_new < best_trial_score:
                best_trial_score = score_new
                best_trial = (new_a, new_y, new_v, trial_lr)
            
            if score_new < f_best break if False else score_new < f_best:
                break
        
        if best_trial is None or best_trial_score >= f_best * 1.001:
            # No improvement; try smaller lr
            lr *= 0.9
            no_improve += 1
            if no_improve > 500:
                break
            continue
        
        # Accept the step
        new_a, new_y, new_v, accepted_lr = best_trial
        a = new_a
        y = new_y
        v = new_v
        lr = accepted_lr * 1.05  # slightly increase lr if we used a reduced step
        lr = min(lr, 0.5)
        
        # Update best
        cur_score = 2.0 * n * float(fast_convolve(a).max())
        if cur_score < f_best:
            f_best = cur_score
            best_a = a.copy()
            no_improve = 0
        
        # Decay parameters
        beta = max(0.5, beta * beta_decay)
        lr *= lr_decay
        temperature = max(0.001, temperature * temp_decay)
        top_k = min(30, int(10 + it * 0.01))  # gradually increase active set
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad {label} it={it} ({elapsed:.0f}s): best={f_best:.6f} lr={lr:.6f} beta={beta:.3f}")
    
    return best_a, f_best

def block_coord_descent(a0, time_limit, block_size=4, n_trials=6, verbose=False, label=""):
    """
    Block coordinate descent: update `block_size` consecutive coordinates at a time.
    For each block, try multiple perturbation patterns.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve_sweeps = 0
    
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        
        # Process blocks
        for start in range(0, n, block_size):
            end = min(start + block_size, n)
            bsz = end - start
            
            if a[start:end].sum() < 1e-15:
                continue
            
            # Determine affected range in conv
            # Changing a[start:end] affects conv[k] for k in [start, end-1+n-1]
            k_start = start
            k_end = min(end - 1 + n - 1, 2 * n - 2)
            
            if k_start > 0:
                unaff_max = conv[:k_start].max()
            else:
                unaff_max = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
                unaff_max = max(unaff_max, unaff_right)
            
            current_block = a[start:end].copy()
            best_block = current_block.copy()
            best_block_score = best_score
            
            # Try perturbations
            for trial in range(n_trials):
                if trial == 0:
                    # Reduce max element
                    max_idx = np.argmax(current_block)
                    pert = np.zeros(bsz)
                    pert[max_idx] = -0.15 * current_block[max_idx]
                elif trial == 1:
                    # Reduce min element (increase others relatively)
                    min_idx = np.argmin(current_block)
                    pert = np.zeros(bsz)
                    pert[min_idx] = 0.1 * current_block[start + min_idx] if a[start+min_idx] > 0 else 0.01
                elif trial == 2:
                    # Smooth: move toward block mean
                    mean = current_block.mean()
                    pert = 0.1 * (mean - current_block)
                elif trial == 3:
                    # Random perturbation
                    pert = np.random.uniform(-0.05, 0.05, bsz)
                elif trial == 4:
                    # Reduce all slightly (redistribute)
                    pert = -0.05 * current_block
                else:
                    # Increase min, decrease max
                    max_idx = np.argmax(current_block)
                    min_idx = np.argmin(current_block)
                    pert = np.zeros(bsz)
                    pert[max_idx] = -0.1 * current_block[max_idx]
                    pert[min_idx] = 0.1 * current_block[max_idx]
                
                new_block = current_block + pert
                new_block = np.maximum(new_block, 0.0)
                
                # Normalize block to maintain sum
                block_sum = new_block.sum()
                if block_sum < 1e-15:
                    continue
                # Scale to maintain overall sum = 1
                old_block_sum = current_block.sum()
                if old_block_sum > 1e-15:
                    new_block = new_block * (old_block_sum / block_sum)
                
                # Compute new conv values in affected range
                # conv_new[k] = conv[k] + 2*sum_{j in block} delta[j]*a[k-j] + sum_{i,j in block} delta[i]*delta[j] (if k-i=j)
                delta = new_block - current_block
                new_conv_affected = conv[k_start:k_end + 1].copy()
                
                for k in range(k_start, k_end + 1):
                    # Change = 2 * sum_j delta[j] * a[k-j] + sum_{i,j} delta[i]*delta[j] * [k-i==j]
                    change = 0.0
                    for j_idx in range(bsz):
                        j = start + j_idx
                        if j_idx >= 0 and k - j >= 0 and k - j < n:
                            change += 2.0 * delta[j_idx] * a[k - j]
                    # Quadratic term
                    for i_idx in range(bsz):
                        i = start + i_idx
                        j = k - i
                        if start <= j < end:
                            change += delta[i_idx] * delta[j - start]
                    new_conv_affected[k - k_start] += change
                
                new_max = max(unaff_max, float(new_conv_affected.max()))
                if new_max < best_block_score / (2.0 * n) - 1e-15:
                    best_block_score = 2.0 * n * new_max
                    best_block = new_block.copy()
            
            # Apply best block update
            if not np.allclose(best_block, current_block):
                a[start:end] = best_block
                # Update conv
                delta = best_block - current_block
                for k in range(k_start, k_end + 1):
                    change = 0.0
                    for j_idx in range(bsz):
                        j = start + j_idx
                        if j_idx >= 0 and k - j >= 0 and k - j < n:
                            change += 2.0 * delta[j_idx] * a[k - j]
                    for i_idx in range(bsz):
                        i = start + i_idx
                        j = k - i
                        if start <= j < end:
                            change += delta[i_idx] * delta[j - start]
                    conv[k] += change
                M = float(conv.max())
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > 10:
                break
        else:
            no_improve_sweeps = 0
        
        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    BCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score

def fine_coord_descent(a0, time_limit, n_trials=12, verbose=False, label=""):
    """Fine coordinate descent with many trial steps."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n

    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        
        for i in range(n):
            if a[i] < 1e-15:
                continue
            
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            
            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:

            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)
            
            affected = conv[k_start:k_end + 1]
            a_shifted = a[0:n]
            
            best_delta = 0.0
            best_new_max = M
            
            # Try various step sizes
            factors = [0.995, 0.99, 0.98, 0.97, 0.95, 1.05, 1.03, 1.015, 1.01, 1.005, 0.96, 0.985]
            for factor in factors:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue
                
                new_affected = affected + 2.0 * delta * a_shifted
                new_affected = new_affected.copy()
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    new_affected[idx_2i] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
                
                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_delta = delta
            
            if best_delta != 0.0 and best_new_max < M - 1e-16:
                a[i] += best_delta
                a[i] = max(a[i], 0.0)
                
                changes = 2.0 * best_delta * a_shifted
                changes = changes.copy()
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    changes[idx_2i] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 5:
                break
        else:
            no_improve = 0
    
    return best_a, best_score

def fictitious_play(n, time_limit, initial=None, solver=None, verbose=False):

    t0 = time.monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:

        else:
            a = a / s

    iters = 0

        a_new, M_a = solve_lp_best_response(b, solver)
        if a_new is not None:

            if sc < 1e-10:

            cand_norm = cand / sc

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def init_fourier(n, rng):
    """Initialize using random Fourier superposition."""
    n_harmonics = int(rng.integers(3, 20))
    amplitudes = rng.random(n_harmonics) * 0.35
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = np.arange(1, n_harmonics + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def init_cosine(n, rng):
    """Initialize with a cosine bell shape."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * rng.uniform(0.3, 0.7)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def init_u_shape(n, rng):
    """Initialize with U-shape."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    alpha = rng.uniform(0.5, 2.0)
    a = 1.0 + alpha * np.cos(np.pi * t) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def init_noise(n, base, rng, sigma=0.1):
    """Add Gaussian noise to a base sequence."""
    a = base.copy() + rng.standard_normal(n) * sigma
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a
