import numpy as np
import time
import cvxpy as cp
from scipy.linalg import toeplitz
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    """Build (2n-1) x n dense matrix C such that C @ g = f * g (linear conv)."""
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_best_response(f, M, prev_g=None):
    """Solve: max sum(g) s.t. C(f) @ g <= M, g >= 0."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = np.maximum(prev_g, 0)
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None))
    except Exception:
        try:
            prob.solve(solver=cp.SCS)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS)
            except Exception:
                return prev_g if prev_g is not None else f.copy()
    if g.value is None:
        return prev_g if prev_g is not None else f.copy()
    return np.maximum(g.value, 0)

    if s < 1e-15:

    best_score = compute_score(a)

        step = 2.0 / (iters + 2.0)

        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss

        score = compute_score(a_new)
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        if verbose and iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"  AggPeak iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")

    return best_a, best_score


def softmax_from_z(z):
    """Numerically stable softmax."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Log of positive values (for softmax parameterization)."""
    return np.log(np.maximum(a, 1e-30))


def lse_obj_and_grad(z, T, n):
    """Compute LSE-smoothed objective and its gradient w.r.t. z."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    Tconv = T * conv
    m = Tconv.max()
    exp_shifted = np.exp(Tconv - m)
    lse = (m + np.log(exp_shifted.sum())) / T
    w = exp_shifted / exp_shifted.sum()

    # Cross-correlation: cross[j] = sum_m w[m+j] * a[m]
    size = 2 * n - 1
    N = 1 << (size + n - 2).bit_length()
    W = np.fft.fft(w, N)
    A = np.fft.fft(a, N)
    cross = np.fft.ifft(W * np.conj(A), N).real[:n]

    # grad_a = 4n * cross (gradient of smoothed objective w.r.t. a)
    grad_a = 4.0 * n * cross
    # grad_z = a * (grad_a - dot(a, grad_a))
    S_grad = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - S_grad)

    obj = 2.0 * lse
    return obj, grad_z


def lbgfs_optimize_z(a0, time_limit, T, maxiter=5000, verbose=True):
    """L-BFGS on log-sum-exp smoothed objective in z-space (softmax parameterization)."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        obj, _ = lse_obj_and_grad(z, T, n)
        return obj

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        _, g = lse_obj_and_grad(z, T, n)
        return g

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 80, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        score = compute_score(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception:
        pass

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"  L-BFGS T={T}: best={best_score:.8f}, time={elapsed:.1f}s")

    return best_a, best_score


def coordinate_descent(a0, time_limit, n_pairs=500, verbose=True):
    """Coordinate descent with random pairwise transfers."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(99999)
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or a[i] < 1e-14:
                continue

            best_local = best_score
            best_t = 0.0
            base = a[i]

            for frac in [0.02, 0.05, 0.1, 0.2, 0.35, 0.5]:
                t = frac * base
                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_t = t

            if best_t != 0.0:
                a[i] -= best_t
                a[j] += best_t
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = best_local
                best_a = a.copy()
                improved = True

        if not improved and iters > 30:
            noise = rng.standard_normal(n) * 0.003
            a = best_a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()

        if verbose and iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  CD iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")

    return best_a, best_score


def check_and_update(a, label, best_seq, best_score, verbose=True):
    """Check if a new candidate improves the best score."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def get_parent_sequence():
    """Retrieve the parent sequence from the global variable."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            return parent, p_score
    except Exception:
        pass
    # Fallback: use a good initialization
    n_init = 768
    parent = np.ones(n_init) / n_init
    # Apply a slight ramp to break symmetry (helps get below 2)
    ramp = np.linspace(0.5, 1.5, n_init)
    parent = ramp / ramp.sum()
    p_score = compute_score(parent)
    print(f"Initialized (ramp): n={n_init}, score={p_score:.8f}")
    return parent, p_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    Phase 1: LP fictitious play with Osgood step (main search, ~45% time)
    Phase 2: Aggressive peak reduction LP (~10% time)
    Phase 3: Multi-temperature L-BFGS polishing (~35% time)
    Phase 4: Coordinate descent + final LP polish (~10% time)
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq, best_score = get_parent_sequence()
    n_parent = len(best_seq)
    print(f"\nStarting optimization with n={n_parent}, score={best_score:.8f}")
    print(f"Time budget: {budget_s:.0f}s, deadline in {deadline - time.monotonic():.0f}s")

    # Phase 1: LP Fictitious Play with Osgood step
    print(f"\n=== Phase 1: LP-FP Osgood (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.45, 700.0)
    if rem > 60:
        print(f"  Budget: {t1:.0f}s")
        a_lp, sc_lp = lp_fictitious_play(best_seq, t1, verbose=True)
        best_seq, best_score = check_and_update(a_lp, "lp_ osgood", best_seq, best_score)

    # Phase 2: Aggressive peak reduction
    print(f"\n=== Phase 2: Aggressive Peak Reduction (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.10, 150.0)
    if rem > 40:
        print(f"  Budget: {t2:.0f}s")
        a_agg, sc_agg = aggressive_peak_reduction(best_seq, t2, verbose=True)
        best_seq, best_score = check_and_update(a_agg, "agg_peak", best_seq, best_score)

    # Phase 3: Multi-temperature L-BFGS
    print(f"\n=== Phase 3: Multi-T L-BFGS (best={best_score:.8f}) ===")
    temperatures = [50, 150, 400, 1000, 3000, 8000, 20000]
    for T_val in temperatures:
        rem = deadline - time.monotonic()
        if rem < 20:
            break
        t3 = min(rem * 0.10, 60.0)
        print(f"  L-BFGS T={T_val}, budget={t3:.0f}s")
        a_lb, sc_lb = lbgfs_optimize_z(best_seq, t3, T=T_val, maxiter=3000, verbose=True)
        best_seq, best_score = check_and_update(a_lb, f"lbgfs_T{T_val}", best_seq, best_score)

    # Phase 4: Coordinate descent
    print(f"\n=== Phase 4: Coordinate Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.15, 100.0)
    if rem > 30:
        print(f"  Budget: {t4:.0f}s")
        a_cd, sc_cd = coordinate_descent(best_seq, t4, n_pairs=400, verbose=True)
        best_seq, best_score = check_and_update(a_cd, "coord_desc", best_seq, best_score)

    # Phase 5: Final L-BFGS with highest temperature
    print(f"\n=== Phase 5: Final High-T L-BFGS (best={best_score:.8f}) ===")
    for T_val in [50000, 100000]:
        rem = deadline - time.monotonic()
        if rem < 15:
            break
        t5 = min(rem * 0.15, 50.0)
        a_lb2, sc_lb2 = lbgfs_optimize_z(best_seq, t5, T=T_val, maxiter=2000, verbose=False)
        best_seq, best_score = check_and_update(a_lb2, f"lbgfs_final_T{T_val}", best_seq, best_score)

    # Phase 6: Final LP polish
    print(f"\n=== Phase 6: Final LP Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t6 = min(rem * 0.4, 100.0)
        a_final, sc_final = lp_fictitious_play(best_seq, t6, verbose=False)
        best_seq, best_score = check_and_update(a_final, "lp_final", best_seq, best_score)

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
