def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_direction(f, n, peak_indices, M, solver_list=None):
    """
    Solve: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for k in peak_indices.
    """
    if solver_list is None:
        solver_list = [cp.PDLP, cp.SCS, cp.ECOS]
    
    K = len(peak_indices)
    if K == 0:
        return None
    
    # Build constraint matrix: A[row, col] = f[col] where the convolution (f*g)[k] = sum_j f[j]*g[k-j]
    # (f*g)[k] = sum_{j} f[j] * g[k-j], where 0<=j<n and 0<=k-j<n
    # So for a given k, the terms are f[j] * g[k-j] for j in range(max(0,k-n+1), min(n,k+1))
    # Let i = k-j, so j = k-i, and i ranges from max(0, k-n+1) to min(n-1, k)
    # (f*g)[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} f[k-i] * g[i]
    
    A = np.zeros((K, n))
    for row_i in range(K):
        k = int(peak_indices[row_i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # A[row, i] = f[k-i] for i in [lo, hi]
        for i in range(lo, hi + 1):
            A[row_i, i] = f[k - i]
    
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    
    for solver in solver_list:

            prob.solve(solver=solver)
            if g.value is not None:
                g_val = np.maximum(g.value, 0.0)
                if np.all(np.isfinite(g_val)):
                    return g_val
        except Exception:
            continue
    
    return None

        scaled = beta * c
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum > 1e-300:
            w = w / w_sum
        else:
            w = np.zeros_like(c)
            w[M == c.max()] = 1.0 / np.sum(c == c.max())
        
        # Gradient: grad_i = sum_k w_k * y[k-i] (with appropriate bounds)
        grad = np.zeros(n)
        for k in range(len(c)):
            if w[k] < 1e-15:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += w[k] * y[k - i]
        
        # FISTA update
        lr_t = lr / (1.0 + it * 0.1)  # decreasing step size
        z = y - lr_t * grad
        f_new = project_simplex(z)
        
        t_new = 0.5 * (1.0 + math.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = f_new + ((t_mom - 1.0) / t_new) * (f_new - f)
        t_mom = t_new
        f = f_new
        
        # Check improvement
        if it % 10 == 0:
            obj = fast_convolve(f).max()
            if obj < best_b - 1e-14:
                best_b = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            if no_improve > 10:
                beta = min(beta * 1.5, 500.0)
                lr = lr_init
                no_improve = 0
            
            if no_improve > 30:
                # Restart with perturbation
                f = best_f + np.random.standard_normal(n) * 0.01 * best_f
                f = np.maximum(f, 0)
                s = f.sum()
                if s > 1e-10:
                    f = f / s
                y = f.copy()
                t_mom = 1.0
                no_improve = 0
        
        # Adaptively increase beta
        if it % 50 == 0:
            beta = min(beta * 1.3, 200.0)
            lr = lr_init / math.sqrt(it / 100 + 1)
    
    return best_f, best_b

def lp_optimize(f_init, time_limit, label="lp", 
                start_peaks=30, end_peaks=500, 
                patience=30, solver_names=None):

    LP-based optimization with adaptive peak count.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-10:
        f = f / s
    best_f = f.copy()
    best_b = fast_convolve(f).max()
    it = 0
    stuck = 0
    rng = np.random.default_rng(12345)
    
    solver_names = solver_names or ['PDLP', 'SCS', 'ECOS', 'GLPK']
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()
        
        # Adaptive peak selection
        frac_progress = min(1.0, it / (2.0 * patience))
        n_peaks = int(start_peaks + (end_peaks - start_peaks) * frac_progress)
        n_peaks = min(n_peaks, len(b) - 1)
        
        # Select top peaks
        if n_peaks < len(b):
            peaks = np.argpartition(b, -n_peaks)[-n_peaks:]
        else:
            peaks = np.arange(len(b))
        
        # Also ensure we always include the global max
        max_idx = np.argmax(b)
        if max_idx not in peaks:
            peaks = np.append(peaks, max_idx)
        
        g = solve_lp_direction(f, n, peaks, M, solver_list=solver_names)
        
        if g is None:
            stuck += 1
            if stuck > patience:
                break
            continue
        
        S_g = g.sum()
        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > patience:
                # Try random perturbation
                if stuck % 10 == 0:
                    f = f + rng.standard_normal(n) * 0.005 * f
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                continue
        else:
            stuck = 0
        
        # Normalize direction
        h = g / S_g
        
        # Line search
        best_t, _ = line_search_convex(f, h, M)
        
        if best_t <= 0:
            stuck += 1
            continue
        
        # Update
        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f = f / s
        else:
            break
        
        # Check improvement
        cur_obj = fast_convolve(f).max()
        if cur_obj < best_b - 1e-14:
            best_b = cur_obj
            best_f = f.copy()
        
        if it % 50 == 0:
            score = 2.0 * n * best_b
            print(f"  {label} it={it} obj={best_b:.12f} score={score:.6f} peaks={n_peaks} t={time.monotonic()-t0:.0f}s")
    
    return best_f, best_b

def coordinate_descent_optimize(f_init, time_limit, label="cd"):
    """Coordinate descent with multiple multipliers per coordinate."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_b = fast_convolve(f).max()
    multipliers = np.array([0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0])
    n_full_passes = 0
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        n_full_passes += 1
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-18:
                # Try setting it to a small positive value
                f_trial = f.copy()
                f_trial[j] = 0.001 / n
                s = f_trial.sum()
                f_trial = f_trial / s
                obj = fast_convolve(f_trial).max()
                if obj < best_b - 1e-14:
                    f[j] = 0.001 / n
                    f = f / f.sum()
                    best_b = obj
                    improved = True
                continue
            
            best_j_val = f[j]
            best_j_obj = best_b
            for mult in multipliers:
                f_trial = f.copy()
                f_trial[j] = f[j] * mult
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial = f_trial / s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-15:
                    best_j_obj = obj
                    best_j_val = f[j] * mult
            
            if best_j_val != f[j]:
                f[j] = best_j_val
                f = f / f.sum()
                best_b = best_j_obj
                improved = True
        
        if not improved:
            break
    
    return f, best_b

def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    n = len(f)
    if n == new_n:
        return f.copy()
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new = f_new / s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses smooth gradient (FISTA), LP-based direction finding with adaptive peak
selection, coordinate descent, and multi-length exploration.
Key improvements over baseline: faster LP solving with multiple solvers,
adaptive peak count scheduling, momentum restarts, and better line search.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        val = max(float(v[0]), 0.0)
        return np.array([val])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f, peaks):
    """Build matrix A where A[row, i] = f[k - i] for convolution constraint at k."""
    n = len(f)
    K = len(peaks)
    A = np.zeros((K, n))
    for row in range(K):
        k = int(peaks[row])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        length = hi - lo + 1
        if length > 0:
            A[row, lo:hi + 1] = f[hi:hi - length + 1 if length > 1 else hi + 1][::-1]
    return A


def solve_lp_direction(f, n, peak_indices, M):
    """
    Solve: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for k in peak_indices.
    Returns g vector or None.
    """
    K = len(peak_indices)
    if K == 0:
        return None

    A = np.zeros((K, n))
    for row in range(K):
        k = int(peak_indices[row])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            A[row, i] = f[k - i]

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    solvers_to_try = [cp.PDLP, cp.SCS, cp.ECOS, cp.GLPK]
    for solver in solvers_to_try:
        try:
            prob.solve(solver=solver)
            if g.value is not None:
                g_val = np.maximum(g.value, 0.0)
                if np.all(np.isfinite(g_val)) and g_val.sum() > 0:
                    return g_val
        except Exception:
            continue
    return None

    best_t = 0.0
    best_obj = M_current

    for t in t_grid:

        f_trial = np.maximum(f_trial, 0.0)
        s = f_trial.sum()
        if s < 1e-10:

        f_trial = f_trial / s
        obj = fast_convolve(f_trial).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_t = t

                if obj < best_obj - 1e-15:

                    best_t = t

        z = y - lr_t * grad
        f_new = project_simplex(z)
        t_new = 0.5 * (1.0 + math.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = f_new + ((t_mom - 1.0) / t_new) * (f_new - f)
        if not np.all(np.isfinite(y)):
            y = f_new.copy()
        t_mom = t_new
        f = f_new

        if it % 20 == 0:
            obj = fast_convolve(f).max()
            if obj < best_b - 1e-14:
                best_b = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 8:
                beta = min(beta * 1.4, 300.0)
                no_improve = 0
            if no_improve > 20:
                f = best_f + rng.standard_normal(n) * 0.01 * best_f
                f = np.maximum(f, 1e-15)
                f = f / f.sum()
                y = f.copy()
                t_mom = 1.0
                no_improve = 0
            if it % 100 == 0:
                score = 2.0 * n * best_b
                print(f"  {label} it={it} obj={best_b:.10f} score={score:.6f} beta={beta:.0f} t={time.monotonic()-t0:.0f}s")

    return best_f, best_b


def lp_optimize(f_init, time_limit, label="lp",
                start_peaks=20, end_peaks=500, patience=35):
    """
    LP-based optimization with adaptive peak count and restart logic.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-10:
        f = f / s
    best_f = f.copy()
    best_b = fast_convolve(f).max()
    it = 0
    stuck = 0
    rng = np.random.default_rng(99999)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        progress = min(1.0, it / (3.0 * patience))
        n_peaks = int(start_peaks + (end_peaks - start_peaks) * progress)
        n_peaks = max(5, min(n_peaks, len(b) - 1))

        if n_peaks < len(b):
            peaks = np.argpartition(b, -n_peaks)[-n_peaks:]
        else:
            peaks = np.arange(len(b))
        max_idx = int(np.argmax(b))
        if max_idx not in peaks:
            peaks = np.append(peaks, max_idx)

        g = solve_lp_direction(f, n, peaks, M)

        if g is None:
            stuck += 1
            if stuck > patience:
                break
            continue

        S_g = g.sum()
        if S_g <= 1.0 + 1e-5:
            stuck += 1
            if stuck > 10 and stuck % 5 == 0:
                f = f + rng.standard_normal(n) * 0.003 * f
                f = np.maximum(f, 0)
                s2 = f.sum()
                if s2 > 1e-10:
                    f = f / s2
            if stuck > patience:
                break
            continue
        else:
            stuck = 0

        h = g / S_g
        best_t, _ = line_search(f, h, M)

        if best_t <= 0:
            stuck += 1
            if stuck > patience:
                break
            continue

        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f = f / s
        else:
            break

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_b - 1e-15:
            best_b = cur_obj
            best_f = f.copy()

        if it % 30 == 0:
            score = 2.0 * n * best_b
            print(f"  {label} it={it} obj={best_b:.12f} score={score:.6f} peaks={n_peaks} Sg={S_g:.4f} t={time.monotonic()-t0:.0f}s")

    return best_f, best_b


def coordinate_descent(f_init, time_limit, label="cd"):
    """Coordinate descent with adaptive multipliers."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_b = fast_convolve(f).max()
    multipliers = np.array([0.4, 0.6, 0.8, 0.9, 0.95, 1.02, 1.05, 1.1, 1.2, 1.4, 1.7, 2.2])

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            base = f[j]
            if base < 1e-18:
                f_trial = f.copy()
                f_trial[j] = 0.5 / n
                f_trial = f_trial / f_trial.sum()
                obj = fast_convolve(f_trial).max()
                if obj < best_b - 1e-15:
                    f[j] = 0.5 / n
                    f = f / f.sum()
                    best_b = obj
                    improved = True
                continue

            best_val = base
            best_obj = best_b
            for mult in multipliers:
                f_trial = f.copy()
                f_trial[j] = base * mult
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial = f_trial / s
                obj = fast_convolve(f_trial).max()
                if obj < best_obj - 1e-16:
                    best_obj = obj
                    best_val = base * mult

            if best_val != base:
                f[j] = best_val
                f = f / f.sum()
                best_b = best_obj
                improved = True

        if not improved:
            break

    return f, best_b


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    n = len(f)
    if n == new_n:
        return f.copy()
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new = f_new / s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: FISTA smooth -> LP main -> LP deep -> CD ->
    multi-length -> final LP -> final CD.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        n0 = 1024
        parent = np.ones(n0) / n0
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Using uniform n={n0}, score={p_score:.8f}")

    best_seq = parent.copy()
    best_score = p_score

    # Phase 1: Smooth FISTA
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.12, 150.0)
    if t1 > 30:
        n_p = len(parent)
        print(f"\n=== Phase 1: FISTA smooth (n={n_p}, budget={t1:.0f}s) ===")
        f1, _ = smooth_fista(parent, time_limit=t1, label="fista")
        vs1 = float(evaluate_sequence(f1.tolist()))
        if np.isfinite(vs1) and vs1 < best_score:
            best_score = vs1
            best_seq = f1.copy()
            print(f"  NEW BEST [fista]: {vs1:.8f}")
        else:
            print(f"  fista: {vs1:.8f} (no improvement over {best_score:.8f})")

    # Phase 2: Main LP
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.42, 700.0)
    if t2 > 60:
        n_b = len(best_seq)
        print(f"\n=== Phase 2: LP main (n={n_b}, budget={t2:.0f}s) ===")
        f2, _ = lp_optimize(best_seq, time_limit=t2, label="lp1",
                             start_peaks=15, end_peaks=400, patience=40)
        vs2 = float(evaluate_sequence(f2.tolist()))
        if np.isfinite(vs2) and vs2 < best_score:
            best_score = vs2
            best_seq = f2.copy()
            print(f"  NEW BEST [lp1]: {vs2:.8f}")

    # Phase 3: Deep LP (more peaks, tighter)
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.28, 400.0)
    if t3 > 40:
        n_b = len(best_seq)
        print(f"\n=== Phase 3: LP deep (n={n_b}, budget={t3:.0f}s) ===")
        f3, _ = lp_optimize(best_seq, time_limit=t3, label="lp2",
                             start_peaks=80, end_peaks=600, patience=50)

        if np.isfinite(vs3) and vs3 < best_score:
            best_score = vs3
            best_seq = f3.copy()
            print(f"  NEW BEST [lp2]: {vs3:.8f}")

    # Phase 4: Coordinate descent
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.12, 120.0)
    if t4 > 10:
        print(f"\n=== Phase 4: CD (budget={t4:.0f}s) ===")
        f4, _ = coordinate_descent(best_seq, time_limit=t4, label="cd")

            best_seq = f4.copy()
            print(f"  NEW BEST [cd]: {vs4:.8f}")

    # Phase 5: Multi-length exploration
    rem = deadline - time.monotonic()
    n_base = len(best_seq)
    target_lengths = []
    for factor in [0.85, 1.1, 1.25, 1.4, 0.75]:
        tl = int(n_base * factor)
        tl = max(256, min(3000, tl))
        if tl != n_base and tl not in target_lengths:
            target_lengths.append(tl)
    target_lengths.sort(key=lambda x: abs(x - n_base))

    for target_n in target_lengths[:3]:

        if rem < 45:
            break
        t5 = min(rem * 0.14, 120.0)
        if t5 < 20:
            break
        f_interp = interpolate_to_length(best_seq, target_n)

        f5, _ = lp_optimize(f_interp, time_limit=t5, label=f"lp_n{target_n}",
                             start_peaks=15, end_peaks=300, patience=25)
        vs5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(vs5) and vs5 < best_score:
            best_score = vs5
            best_seq = f5.copy()
            print(f"  NEW BEST [n{target_n}]: {vs5:.8f}")

    # Phase 6: Final LP burst

    t6 = min(rem * 0.30, 150.0)

            print(f"  NEW BEST [lp_final]: {vs6:.8f}")

            print(f"  NEW BEST [cd_final]: {vs7:.8f}")
