"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using reduced-constraint LP alternating
optimization with Nesterov acceleration, Adam-based LogSumExp optimization,
multi-length exploration, and careful coordinate descent polishing.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def build_conv_matrix_sparse(a):
    """Build convolution matrix as a list of (row, col, val) tuples."""
    n = len(a)
    size = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for i in range(n):
            k = i + j
            if a[i] > 0:
                rows.append(k)
                cols.append(j)
                vals.append(a[i])
    return np.array(rows), np.array(cols), np.array(vals)


def convolve_with(a, b):
    """Compute convolve(a, b) where both have same length."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def solve_lp_reduced(a, prev_b, prev_t, solver, n_keep=300, eps_fraction=0.02):
    """Solve LP best response with reduced constraints.
    
    Only keep constraints that are near-binding.
    """
    n = len(a)
    if prev_b is not None and prev_t is not None:
        # Compute all T @ prev_b values
        T_prev = convolve_with(a, prev_b)
        # Find near-binding constraints
        threshold = prev_t - eps_fraction * prev_t
        active = np.where(T_prev >= threshold)[0]
        if len(active) < 50:
            active = np.argsort(T_prev)[::-1][:n_keep]
        if len(active) > n_keep:
            active = np.argsort(T_prev)[::-1][:n_keep]
    else:
        active = np.arange(2 * n - 1)
    
    # Build submatrix
    T = build_conv_matrix(a)
    T_sub = T[active, :]
    
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T_sub @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        # Fallback: use full LP
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver)
        except Exception:
            pass
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def adam_logsumexp(a0, time_limit, T_start=50.0, T_end=0.5, lr_init=0.01, n_steps=200000):
    """Adam optimization of LogSumExp-smoothed objective in log-space."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    t0 = time.monotonic()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    n_stages = 8
    temps = np.geomspace(T_start, T_end, n_stages)
    steps_per_stage = max(50, n_steps // n_stages)
    
    for si, T in enumerate(temps):
        stage_steps = steps_per_stage
        for step in range(stage_steps):
            global_step = si * steps_per_stage + step
            if time.monotonic() - t0 > time_limit:
                break
            
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            
            # Gradient with respect to a
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += w[k] * a[k - idx]
            
            # Gradient with respect to z (softmax Jacobian)
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)
            
            # Scale gradient
            grad_z = (2.0 * n / T) * grad_z
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * (grad_z ** 2)
            m_hat = m / (1 - beta1 ** (global_step + 1))
            v_hat = v / (1 - beta2 ** (global_step + 1))
            
            # Learning rate with warmup and decay
            warmup_steps = 100
            if global_step < warmup_steps:
                lr = lr_init * (global_step + 1) / warmup_steps
            else:
                progress = (global_step - warmup_steps) / max(1, n_steps - warmup_steps)
                lr = lr_init * 0.5 * (1 + np.cos(np.pi * progress))
            
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Check improvement
            a_new = softmax_from_z(z)
            score = 2.0 * n * fast_convolve(a_new).max()
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
            
            if (global_step + 1) % 10000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    Adam step={global_step+1} ({elapsed:.0f}s): score={best_score:.8f} T={T:.2f}")
        
        if time.monotonic() - t0 > time_limit:
            break
    
    return best_a, best_score


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def gradient_logsumexp(a, T):
    """Compute gradient of LogSumExp objective w.r.t. a for fast use."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue

        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    return grad_a, w


def coordinate_descent_polish(a0, time_limit, n_steps=50000):
    """Careful coordinate descent for final polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)

        peak_val = b[peak_k]

        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

                not_contrib[idx] = False

                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

                break

                    i, j = top_coords[i_idx], top_coords[j_idx]

def lbgfs_refine_v2(a0, time_limit, T_start=20.0, T_end=0.1, maxiter=10000):
    """Improved L-BFGS with LogSumExp."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()

    def make_func(T_val):
        def func(z_val):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z_val)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z_val):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_val)
            a = softmax_from_z(z_val)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    n_stages = 10
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(30, maxiter // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_ab(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):

    for j in range(n):

        M_var = cp.Variable()
        constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver)
        except Exception:
            pass
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_reduced(a0, time_limit, solver, n_max_iters=500000, n_keep=300, eps_frac=0.02):
    """Nesterov LP alternating with reduced constraints and adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_b = None
    prev_t = best_score / (2.0 * n)
    stall_count = 0
    prev_best = best_score

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_reduced(y, prev_b, prev_t, solver, n_keep=n_keep, eps_frac=eps_frac)
        if b_new is None:
            b_new, M_b = solve_lp_reduced(y, None, None, solver, n_keep=n_keep, eps_frac=eps_frac)
            if b_new is None:
                continue

        prev_b = b_new.copy()
        prev_t = M_b

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        numerator = (it + 1) * a + (alpha * (it + 2)) * b_new
        denominator = (it + 1) + alpha * (it + 2)
        a = numerator / denominator
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if it % 200 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            if stall_count >= 8:
                rng_local = np.random.default_rng(it * 7919)
                noise = rng_local.random(n) * 0.03
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0
                prev_b = None
                prev_t = None

        if (it + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-reduced iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def lbgfs_func(z):
    """L-BFGS objective using global state."""
    T_val = _lbgfs_state['T']
    n = _lbgfs_state['n']
    t0 = _lbgfs_state['t0']
    tl = _lbgfs_state['time_limit']
    if time.monotonic() - t0 > tl:
        return 1e15
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    return (2.0 * n / T_val) * log_sum_exp


def lbgfs_jac(z):
    """L-BFGS gradient using global state."""
    T_val = _lbgfs_state['T']
    n = _lbgfs_state['n']
    t0 = _lbgfs_state['t0']
    tl = _lbgfs_state['time_limit']
    if time.monotonic() - t0 > tl:
        return np.zeros_like(z)
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    return a * (grad_a - S)


def lbgfs_refine(a0, time_limit, T_start=20.0, T_end=0.1, maxiter=10000):
    """L-BFGS with LogSumExp smoothing and geometric temperature annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()

    n_stages = 12
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(30, maxiter // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        _lbgfs_state['T'] = float(stage_T)
        _lbgfs_state['n'] = n
        _lbgfs_state['t0'] = t0
        _lbgfs_state['time_limit'] = time_limit
        try:
            res = minimize(
                fun=lbgfs_func,
                x0=z,
                jac=lbgfs_jac,
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def adam_logsumexp(a0, time_limit, T_start=50.0, T_end=0.5, lr_init=0.005, n_steps=200000):
    """Adam optimization of LogSumExp-smoothed objective in log-space."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    t0 = time.monotonic()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    n_stages = 10
    temps = np.geomspace(T_start, T_end, n_stages)
    steps_per_stage = max(50, n_steps // n_stages)

    for si, T in enumerate(temps):
        for step in range(steps_per_stage):
            global_step = si * steps_per_stage + step
            if time.monotonic() - t0 > time_limit:
                break

            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()

            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += w[k] * a[k - idx]

            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)
            grad_z = (2.0 * n / T) * grad_z

            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * (grad_z ** 2)
            m_hat = m / (1 - beta1 ** (global_step + 1))
            v_hat = v / (1 - beta2 ** (global_step + 1))

            warmup_steps = 200
            if global_step < warmup_steps:
                lr = lr_init * (global_step + 1) / warmup_steps
            else:
                progress = (global_step - warmup_steps) / max(1, n_steps - warmup_steps)
                lr = lr_init * 0.5 * (1 + np.cos(np.pi * progress))

            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            a_new = softmax_from_z(z)
            score = 2.0 * n * fast_convolve(a_new).max()
            if score < best_score:
                best_score = score
                best_a = a_new.copy()

            if (global_step + 1) % 20000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    Adam step={global_step+1} ({elapsed:.0f}s): score={best_score:.8f} T={T:.2f}")

        if time.monotonic() - t0 > time_limit:
            break

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, n_steps=50000):
    """Careful coordinate descent targeting peak contributors."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 200:
            break

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        top_n = min(10, len(idx))
        top_coords = idx[np.argsort(contribs)[::-1][:top_n]]

        improved = False
        for i in top_coords:
            if a[i] < 1e-15:
                continue
            for frac in [0.995, 0.99, 0.98, 0.97, 0.95, 0.93, 0.90]:
                new_val = a[i] * frac
                diff = a[i] - new_val
                if diff < 1e-18:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                not_peak = np.ones(n, dtype=bool)
                not_peak[idx] = False
                candidates = np.where(mask & not_peak)[0]
                if len(candidates) == 0:
                    candidates = np.where(mask)[0]
                if candidates.sum() == 0:
                    continue
                a_trial[candidates] += diff * a_trial[candidates] / a_trial[candidates].sum()
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Pair modification
            for ii in range(min(4, len(top_coords))):
                for jj in range(ii + 1, min(4, len(top_coords))):
                    i, j = top_coords[ii], top_coords[jj]
                    if a[i] < 1e-15 or a[j] < 1e-15:
                        continue
                    for f1 in [0.98, 0.96, 0.94]:
                        for f2 in [0.98, 0.96, 0.94]:
                            a_trial = a.copy()
                            a_trial[i] *= f1
                            a_trial[j] *= f2
                            a_trial = a_trial / a_trial.sum()
                            score = 2.0 * n * fast_convolve(a_trial).max()
                            if score < best_score - 1e-15:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                improved = True
                                break
                        if improved:
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            no_improve += 1
        else:
            no_improve = 0

        if (step + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc step={step+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create structured initial sequence."""
    choice = int(rng.integers(0, 7))
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif choice == 4:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = a ** 1.5
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * (x - 0.5) ** 2) + 0.1 * np.exp(-20 * (x - 0.3) ** 2) + 0.1 * np.exp(-20 * (x - 0.7) ** 2)
    else:
        x = np.linspace(0, 1, n)
        a = (1 + np.cos(4 * np.pi * x)) * np.exp(-2 * x * (1 - x))

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: reduced-constraint LP alternating, Adam LogSumExp,
    L-BFGS refinement, coordinate descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1024
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.8f}")

    n_best = len(best_seq)

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return False
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a_norm)})")
            return True
        return False

    # === Phase 1: Reduced-constraint LP alternating (primary) ===
    print(f"\n=== Phase 1: LP-reduced n={n_best} (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.38, 580.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s, n_keep=300")
        a_lp, _ = nesterov_lp_reduced(best_seq, t_alloc, solver,
                                       n_max_iters=500000, n_keep=300, eps_frac=0.02)
        do_try_update(a_lp, "lp_reduced_1")

    # === Phase 2: Second LP round with different parameters ===
    print(f"\n=== Phase 2: LP-reduced r2 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 180.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s, n_keep=400, eps=0.01")
        # Slight perturbation to escape potential local min
        perturbed = best_seq + 0.01 * rng.standard_normal(len(best_seq))
        perturbed = np.maximum(perturbed, 0)
        perturbed = perturbed / perturbed.sum()
        a_lp2, _ = nesterov_lp_reduced(perturbed, t_alloc, solver,
                                        n_max_iters=200000, n_keep=400, eps_frac=0.01)
        do_try_update(a_lp2, "lp_reduced_2")

    # === Phase 3: Adam LogSumExp ===
    print(f"\n=== Phase 3: Adam LogSumExp (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.18, 270.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s, T: 40->0.3, lr=0.004")
        a_adam = adam_logsumexp(best_seq, t_alloc, T_start=40.0, T_end=0.3,
                                lr_init=0.004, n_steps=300000)
        do_try_update(a_adam, "adam")

    # === Phase 4: L-BFGS refinement ===
    print(f"\n=== Phase 4: L-BFGS (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 180.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s, T: 15->0.05")
        a_lb = lbgfs_refine(best_seq, t_alloc, T_start=15.0, T_end=0.05, maxiter=10000)
        do_try_update(a_lb, "lbgfs")

    # === Phase 5: Coordinate descent polishing ===
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 10:
        a_cd = coordinate_descent_polish(best_seq, t_alloc, n_steps=100000)
        do_try_update(a_cd, "coord_desc")

    # === Phase 6: Try nearby lengths ===
    print(f"\n=== Phase 6: Alternative Lengths (best={best_score:.8f}) ===")
    alt_lengths = []
    for delta in [-96, -48, 48, 96, 144, 192, 256]:
        alt_n = n_best + delta
        if 512 <= alt_n <= 2560:
            alt_lengths.append(alt_n)
    alt_lengths = sorted(set(alt_lengths))

    n_tried = 0
    for alt_n in alt_lengths:
        if n_tried >= 3:
            break
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_total = min(rem * 0.25, 200.0)
        if t_total < 30:
            break
        n_tried += 1
        print(f"  n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_seq, alt_n)

        t_lp = t_total * 0.65
        a_alt, _ = nesterov_lp_reduced(init_alt, t_lp, solver,
                                        n_max_iters=30000, n_keep=250, eps_frac=0.03)
        do_try_update(a_alt, f"lp_n{alt_n}")

        rem = deadline - time.monotonic()
        if rem > 20 and best_seq is not None and len(best_seq) == alt_n:
            t_lb = min(rem * 0.3, 60.0)
            if t_lb > 10:
                a_alt2 = lbgfs_refine(best_seq, t_lb, T_start=10.0, T_end=0.5, maxiter=3000)
                do_try_update(a_alt2, f"lbgfs_n{alt_n}")

    # === Phase 7: Final polish on best found ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 25:
        t_fin = min(rem * 0.35, 60.0)
        if t_fin > 10:
            a_fin = lbgfs_refine(best_seq, t_fin, T_start=5.0, T_end=0.02, maxiter=5000)
            do_try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_cd = min(rem * 0.35, 40.0)
        if t_cd > 5:
            a_cd2 = coordinate_descent_polish(best_seq, t_cd, n_steps=30000)
            do_try_update(a_cd2, "final_cd")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_lp_fin = min(rem * 0.5, 30.0)
        if t_lp_fin > 5:
            a_lp_fin, _ = nesterov_lp_reduced(best_seq, t_lp_fin, solver,
                                               n_max_iters=10000, n_keep=200, eps_frac=0.01)
            do_try_update(a_lp_fin, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
            print(f"  Verified: {best_score:.8f}")

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
