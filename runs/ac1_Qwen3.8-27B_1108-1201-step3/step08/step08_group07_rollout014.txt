                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_t = t
            if best_t != 0.0:
                a[i] -= best_t
                a[j] += best_t
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = best_local
                best_a = a.copy()
                improved = True
                stall = 0
        if not improved:
            stall += 1
            if stall > 10:
                noise = rng.standard_normal(n) * 0.001
                a = best_a.copy() + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
        if verbose and iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={iters}, best={best_score:.8f}, t={elapsed:.0f}s")
    return best_a, best_score


def check_and_update(a, label, best_seq, best_score, verbose=True):
    """Evaluate and update best sequence."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
