def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the optimization score for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s


def cross_pressure(a, peak_positions, weights):
    """Compute the pressure vector: p_j = sum_k w_k * a_{k-j}."""
    n = len(a)
    p = np.zeros(n)
    for k, w in zip(peak_positions, weights):
        k = int(k)
        # p_j += w * a[k-j] for valid j
        # a[k-j] is valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            # a[k-j] for j in [j_start, j_end]
            # k-j ranges from k-j_start to k-j_end
            idx = k - np.arange(j_start, j_end + 1)
            p[j_start:j_end + 1] += w * a[idx]
    return p


def peak_reduction_mirror(a, time_limit, verbose=True, rng=None):

    Fast multiplicative peak reduction using mirror descent.
    a_j <- a_j * (1 - eta * p_j / max_p) then normalize.
    p_j is the pressure from top peaks.

    eta_min = 0.01
    no_improve = 0
    max_no_improve = 30
    n_top_peaks = min(20, n)

    while time.monotonic() - t0 < time_limit:
        iteration += 1
        
        conv = fast_convolve(a)
        current_score = 2.0 * n * conv.max()

            best_score = current_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        num_peaks = min(n_top_peaks, len(conv))
        peak_indices = np.argsort(conv)[::-1][:num_peaks]

        peak_values = conv[peak_indices]
        base = 0.5 * conv.max()
        weights = np.maximum(peak_values - base, 0)
        w_sum = weights.sum()
        if w_sum < 1e-30:
            weights = np.ones(num_peaks) / num_peaks
        else:
            weights = weights / w_sum

                a = normalize(a)
                eta = 0.5
                no_improve = 0
        
        if verbose and iteration % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MR iter={iteration}, score={best_score:.8f}, eta={eta:.4f}, elapsed={elapsed:.0f}s")
    
    return best_a, best_score


def subgradient_peak_cancellation(a, time_limit, verbose=True, rng=None):
    """
    Subgradient-based peak cancellation with projection.
    At each step, compute subgradient at the peak, take a projected step.
    Uses Polyak-like adaptive step size.
    """
    if rng is None:
        rng = np.random.default_rng(456)
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    # Lower bound: for uniform, score = 2, so we know score >= 2/n * ... 
    # Actually the absolute lower bound is 2 (achieved by... no, uniform gives 2)
    # A better lower bound: score >= 2*n*(1/n) = 2 by the constraint on (a*a)_{n-1}
    # Wait no. Let's use score >= 2 as a crude lower bound.
    # Actually for our problem the minimum is around 1.5, so let's use 1.0 as lower bound.
    lower_bound = 1.0
    
    step = 1.0 / np.sqrt(10)
    momentum = np.zeros(n)
    beta = 0.9  # momentum coefficient
    
    iteration = 0
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        if iteration % 10000 == 0:
            step *= 0.995  # very slow decay
        
        conv = fast_convolve(a)
        current_score = 2.0 * n * conv.max()
        
        if current_score < best_score - 1e-14:
            best_score = current_score
            best_a = a.copy()
        
        # Polyak step size
        gap = current_score - lower_bound
        if gap > 1e-10:
            polyak_step = gap / (4.0 * n)  # Lipschitz constant ~ 4n
            step = min(step, polyak_step * 10)
        
        # Find top peaks (use top 3 for smoother gradient)
        num_peaks = min(3, len(conv))
        peak_indices = np.argsort(conv)[::-1][:num_peaks]
        peak_values = conv[peak_indices]
        max_peak = peak_values[0]
        weights = np.exp(peak_values / (0.1 * max_peak)) 
        weights = weights / weights.sum()
        
        # Compute subgradient: g_j = 2 * sum_k w_k * a_{k-j}
        g = cross_pressure(a, peak_indices, weights) * 2.0
        
        # Center the gradient (project onto sum=0 subspace)
        g -= g.mean()
        
        # Momentum update
        momentum = beta * momentum + (1 - beta) * g
        direction = -momentum
        
        # Take step
        a_new = a + step * direction
        a_new = normalize(a_new)

        conv_new = fast_convolve(a_new)

            a_new = normalize(a_new)
            conv_new = fast_convolve(a_new)
            score_new = 2.0 * n * conv_new.max()

                a = a_new

                    a = best_a + rng.standard_normal(n) * 0.003
                    a = normalize(a)
                    momentum *= 0.1
        
        if verbose and iteration % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG iter={iteration}, score={best_score:.8f}, step={step:.6f}, elapsed={elapsed:.0f}s")

def solve_lp_for_direction(f, M, method='highs'):
    """
    Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0.
    Returns the optimal g (normalized).
    """
    n = len(f)
    # Build A_ub matrix: A_ub[k, j] = f[k-j] if 0<=k-j<n else 0
    # Size: (2n-1) x n
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = j + n
        A_ub[k_start:k_end, j] = f
    
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method=method)
    if res.status != 0 or res.x is None:
        return None
    g = np.maximum(res.x, 0)
    if g.sum() < 1e-15:
        return None
    return g / g.sum()

def smooth_logsumexp(a, T):
    """Compute T * log(sum(exp(conv/T))) - a smooth approximation of max(conv)."""
    conv = fast_convolve(a)
    m = conv.max()
    if m < 1e-30:
        return 0.0
    shifted = (conv - m) / T
    return m + T * np.log(np.exp(shifted).sum())


def lbgfs_smooth(a0, time_limit, T=500, verbose=False):
    """L-BFGS optimization with smooth max approximation."""
    n = len(a0)
    a0 = normalize(a0.copy())
    best_a = a0.copy()
    best_score = compute_score(a0)
    t0 = time.monotonic()
    
    def func(a_vec):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a_t = normalize(a_vec)
        conv = fast_convolve(a_t)
        m = conv.max()
        shifted = (conv - m) / T
        lse = m + T * np.log(np.exp(shifted).sum())
        return 2.0 * n * lse
    
    def jac(a_vec):
        if time.monotonic() - t0 > time_limit:
            return np.zeros(len(a_vec))
        a_t = normalize(a_vec)
        conv = fast_convolve(a_t)
        m = conv.max()
        shifted = (conv - m) / T
        w = np.exp(shifted)
        w = w / w.sum()
        # grad of smooth max w.r.t. a: 2 * (w * a) cross-correlation, projected
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a_t, fft_size)
        grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
        # Project to sum=0 (tangent to simplex)
        grad_a -= grad_a.mean()
        return 2.0 * n * grad_a
    
    try:
        res = minimize(
            fun=func, x0=a0, jac=jac,
            method='L-BFGS-B',
            bounds=[(0, None)] * n,
            options={'maxiter': 3000, 'ftol': 1e-20, 'gtol': 1e-12,
                     'maxfun': 50000, 'disp': False}
        )
        a_out = normalize(res.x)
        score = compute_score(a_out)
        if score < best_score - 1e-14:
            best_a = a_out
            best_score = score
    except Exception:
        pass
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)

    elif style == 'raised_cosine':

    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    if parent is None:
        n_init = 768
        parent = make_initialization(n_init, 'hanning')
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Initialized: n={n_init}, score={best_score:.8f}")
    
    n_parent = len(parent)
    
    # === Phase 1: Mirror descent peak reduction (main workhorse) ===
    print(f"\n=== Phase 1: Mirror Descent Peak Reduction (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(300.0, rem * 0.2)
    print(f"  Budget: {t_alloc:.0f}s")
    a_mr, sc_mr = peak_reduction_mirror(best_seq, t_alloc, verbose=True, rng=rng)
    best_seq, best_score = check_and_update(a_mr, "mirror", best_seq, best_score)
    
    # === Phase 2: Subgradient peak cancellation ===
    print(f"\n=== Phase 2: Subgradient Peak Cancellation (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 600
    if rem > 50:
        t_alloc = min(200.0, rem * 0.2)
        print(f"  Budget: {t_alloc:.0f}s")
        a_sg, sc_sg = subgradient_peak_cancellation(best_seq, t_alloc, verbose=True, rng=rng)
        best_seq, best_score = check_and_update(a_sg, "subgrad", best_seq, best_score)
    
    # === Phase 3: LP Fictitious Play ===
    print(f"\n=== Phase 3: LP Fictitious Play (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 500
    if rem > 60:
        t_alloc = min(200.0, rem * 0.25)
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, sc_lp = lp_fictitious_play(best_seq, t_alloc, verbose=True)
        best_seq, best_score = check_and_update(a_lp, "lp_fp", best_seq, best_score)
    
    # === Phase 4: Coordinate Descent ===
    print(f"\n=== Phase 4: Coordinate Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 400
    if rem > 50:
        t_alloc = min(250.0, rem * 0.3)
        print(f"  Budget: {t_alloc:.0f}s")
        a_cd, sc_cd = coordinate_descent(best_seq, t_alloc, verbose=True, rng=rng)
        best_seq, best_score = check_and_update(a_cd, "coord_descent", best_seq, best_score)
    
    # === Phase 5: L-BFGS Polish ===
    print(f"\n=== Phase 5: L-BFGS Polish (best={best_score:.8f}) ===")
    for T_val in [200, 1000, 5000]:
        rem = deadline - time.monotonic() - 200
        if rem < 30:
            break
        t_alloc = min(80.0, rem * 0.2)
        print(f"  L-BFGS T={T_val}, budget={t_alloc:.0f}s")
        a_lb, sc_lb = lbgfs_smooth(best_seq, t_alloc, T=T_val)
        best_seq, best_score = check_and_update(a_lb, f"lbgfs_T{T_val}", best_seq, best_score)
    
    # === Phase 6: Mirror descent again (second pass) ===
    print(f"\n=== Phase 6: Mirror Descent Pass 2 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 200
    if rem > 50:
        t_alloc = min(150.0, rem * 0.3)
        print(f"  Budget: {t_alloc:.0f}s")
        a_mr2, sc_mr2 = peak_reduction_mirror(best_seq, t_alloc, verbose=True, rng=rng)
        best_seq, best_score = check_and_update(a_mr2, "mirror2", best_seq, best_score)
    
    # === Phase 7: Multi-n exploration ===
    print(f"\n=== Phase 7: Multi-n Exploration (best={best_score:.8f}) ===")
    candidate_ns = [n_parent + i for i in [-10, -5, -3, -2, -1, 1, 2, 3, 5, 10, 15, 20, 32, 64]]
    candidate_ns = [x for x in candidate_ns if 200 <= x <= 3000]
    candidate_ns = sorted(set(candidate_ns))
    
    for alt_n in candidate_ns:
        rem = deadline - time.monotonic() - 100
        if rem < 60:
            break
        t_alloc = min(60.0, rem * 0.2)
        if t_alloc < 20:
            break
        
        init_alt = interpolate_sequence(best_seq, alt_n)
        
        # Quick mirror descent
        a_alt, sc_alt = peak_reduction_mirror(init_alt, t_alloc * 0.5, verbose=False, rng=rng)
        best_seq, best_score = check_and_update(a_alt, f"n{alt_n}_mr", best_seq, best_score)
        
        # Quick coordinate descent
        rem2 = deadline - time.monotonic() - 80
        if rem2 > 20:
            a_alt2, sc_alt2 = coordinate_descent(a_alt, min(rem2 * 0.3, 30.0), verbose=False, rng=rng)
            best_seq, best_score = check_and_update(a_alt2, f"n{alt_n}_cd", best_seq, best_score)
    
    # === Phase 8: Final coordinate descent polish ===
    print(f"\n=== Phase 8: Final CD Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 30
    if rem > 30:
        t_alloc = min(80.0, rem * 0.5)
        a_final, sc_final = coordinate_descent(best_seq, t_alloc, verbose=True, rng=rng)
        best_seq, best_score = check_and_update(a_final, "final_cd", best_seq, best_score)
    
    # === Phase 9: Final mirror descent ===
    rem = deadline - time.monotonic() - 10
    if rem > 15:
        t_alloc = min(40.0, rem * 0.7)
        a_fin, sc_fin = peak_reduction_mirror(best_seq, t_alloc, verbose=False, rng=rng)
        best_seq, best_score = check_and_update(a_fin, "final_mr", best_seq, best_score)
    
    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
