"""
Min-max LP with cutting planes for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses direct peak-minimizing LP directions, aggressive line search,
perturbation restarts, and multi-length exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_convolve(f, g):
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_minmax_lp_cutting(f, active_idx, n):
    """
    Solve: min t
    s.t. (f*g)[k] <= t for k in active_idx
         sum(g) = 1
         g >= 0
    Returns (g*, t*) or (None, None) on failure.
    """
    g = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [g >= 0, cp.sum(g) == 1]
    
    for k in active_idx:
        lo = max(0, int(k) - n + 1)
        hi = min(n - 1, int(k))
        idx = np.arange(lo, hi + 1)
        constraints.append(cp.sum(cp.multiply(f[lo:hi+1], g[idx])) <= t)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    try:
        prob.solve(solver=cp.SCS, max_iters=200000, eps=1e-10)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iters=50000)
        except Exception:
            return None, None
    
    if g.value is None:
        return None, None
    
    g_val = np.maximum(g.value, 0.0)
    s = g_val.sum()
    if s < 1e-10:
        return None, None
    g_val /= s
    
    t_val = t.value if t.value is not None else cross_convolve(f, g_val).max()
    return g_val, t_val


def line_search(f, g, f0_conv, n):
    """
    Find t* = argmin_t max_k ((1-t)f + tg) * ((1-t)f + tg)[k]
    using grid scan + golden section refinement.
    Returns (t*, obj*).
    """
    # Grid scan
    grid = np.linspace(0.0, 1.0, 40)
    best_t = 0.0
    best_obj = np.inf
    
    for t in grid:
        f_trial = (1.0 - t) * f + t * g
        f_trial = np.maximum(f_trial, 0.0)
        s = f_trial.sum()
        if s < 1e-10:
            continue
        f_trial /= s
        obj = fast_convolve(f_trial).max()
        if obj < best_obj:
            best_obj = obj
            best_t = t
    
    if best_t <= 0:
        return 0.0, f0_conv
    
    # Golden section refinement
    idx = int(round(best_t * (len(grid) - 1)))
    lo = grid[max(0, idx - 2)]
    hi = grid[min(len(grid) - 1, idx + 2)]
    
    gr = (np.sqrt(5) - 1) / 2
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    
    for _ in range(30):
        fc = _eval_obj(f, g, c)
        fd = _eval_obj(f, g, d)
        if fc < fd:
            hi = d
            d = c
            c = hi - gr * (hi - lo)
        else:
            lo = c
            c = d
            d = lo + gr * (hi - lo)
    
    t_opt = (lo + hi) / 2
    obj_opt = _eval_obj(f, g, t_opt)
    
    if obj_opt < best_obj:
        best_obj = obj_opt
        best_t = t_opt
    
    return best_t, best_obj


def _eval_obj(f, g, t):
    f_trial = (1.0 - t) * f + t * g
    f_trial = np.maximum(f_trial, 0.0)
    s = f_trial.sum()
    if s < 1e-10:
        return np.inf
    f_trial /= s
    return fast_convolve(f_trial).max()


def minmax_optimize(f_init, time_limit, initial_k=50, max_k=None, label="mm"):
    """
    Min-max LP with cutting planes optimization.
    """
    t0 = time.monotonic()
    n = len(f_init)
    if max_k is None:
        max_k = n  # use all unique positions
    
    f = f_init.copy()
    f /= f.sum()
    b = fast_convolve(f)
    M = b.max()
    best_f = f.copy()
    best_obj = M
    
    # Use symmetry: only positions 0..n-1 matter
    all_positions = np.arange(n)
    
    # Initialize active set with top peaks
    b_first_half = b[:n]
    if initial_k < n:
        active_idx = np.argpartition(b_first_half, -initial_k)[-initial_k:]
    else:
        active_idx = all_positions.copy()
    
    it = 0
    no_improve = 0
    stuck = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 15:
            break
        
        # Solve min-max LP
        g, t_lp = solve_minmax_lp_cutting(f, active_idx, n)
        
        if g is None:
            stuck += 1
            if stuck > 5:
                break
            continue
        stuck = 0
        
        # Check for violated constraints (cutting plane)
        fg = cross_convolve(f, g)
        fg_first_half = fg[:n]
        violated = fg_first_half > t_lp + 1e-8
        v_idx = np.where(violated)[0]
        not_active = v_idx[~np.isin(v_idx, active_idx)]
        
        if len(not_active) > 0 and len(active_idx) < max_k:
            # Add violated constraints (most violated first)
            n_add = min(20, len(not_active))
            to_add = not_active[np.argsort(fg_first_half[not_active])[-n_add:]]
            active_idx = np.unique(np.concatenate([active_idx, to_add]))
            # Re-solve
            g, t_lp = solve_minmax_lp_cutting(f, active_idx, n)
            if g is None:
                stuck += 1
                if stuck > 3:
                    break
                continue
        
        # Line search
        b = fast_convolve(f)
        M_cur = b.max()
        t_opt, obj_opt = line_search(f, g, M_cur, n)
        
        if t_opt <= 0:
            no_improve += 1
            if no_improve > 30:
                break
            # Try expanding active set
            if len(active_idx) < max_k:
                new_top = np.argpartition(fg_first_half, -min(max_k, n))[-min(max_k, n):]
                active_idx = np.unique(np.concatenate([active_idx, new_top]))
            continue
        
        # Update
        f_new = (1.0 - t_opt) * f + t_opt * g
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-10:
            break
        f_new /= s
        
        f = f_new
        b = fast_convolve(f)
        M_cur = b.max()
        
        if M_cur < best_obj - 1e-12:
            best_obj = M_cur
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Refresh active set periodically
        if it % 20 == 0:
            top_n = min(max_k, n)
            active_idx = np.argpartition(b[:n], -top_n)[-top_n:]
        
        if it % 10 == 0:
            print(f"    {label} it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} "
                  f"active={len(active_idx)} no_improve={no_improve} t={time.monotonic()-t0:.0f}s")
        
        if no_improve > 50:
            break
    
    return best_f, best_obj


def perturb_and_optimize(f_init, time_limit, n_perturb=5, label="perturb"):
    """Try random perturbations + re-optimization."""
    t0 = time.monotonic()
    n = len(f_init)
    best_f = f_init.copy()
    best_obj = fast_convolve(f_init).max()
    rng = np.random.default_rng(123)
    
    per_perturb_time = time_limit / n_perturb
    
    for p in range(n_perturb):
        if time.monotonic() - t0 > time_limit - 10:
            break
        
        # Random perturbation
        noise = rng.standard_normal(n) * 0.05
        f_pert = f_init + noise
        f_pert = np.maximum(f_pert, 0)
        s = f_pert.sum()
        if s < 1e-10:
            continue
        f_pert /= s
        
        # Optimize
        t_rem = per_perturb_time - (time.monotonic() - t0) - p * 0  # rough
        t_rem = min(per_perturb_time, time_limit - (time.monotonic() - t0) - 10)
        if t_rem < 20:
            break
        
        f_opt, obj = minmax_optimize(f_pert, time_limit=t_rem, initial_k=30, max_k=n, label=f"{label}{p}")
        
        if obj < best_obj:
            best_obj = obj
            best_f = f_opt.copy()
            print(f"  {label} perturbation {p} improved: obj={best_obj:.10f} score={2*n*best_obj:.6f}")
    
    return best_f, best_obj


def coordinate_polish(f_init, time_limit):
    """Coordinate descent with multiplicative updates."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    step = max(1, n // 300)
    
    for sweep in range(50):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for j in range(0, n, step):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-15:
                continue
            best_val = f[j]
            best_obj_j = best_obj
            for mult in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                nv = f[j] * mult
                f[j] = nv
                s = f.sum()
                if s < 1e-10:
                    f[j] = best_val
                    continue
                f_temp = f / s
                obj = fast_convolve(f_temp).max()
                if obj < best_obj_j - 1e-14:
                    best_obj_j = obj
                    best_val = nv
            f[j] = best_val
            s = f.sum()
            if s > 1e-10:
                f = f / s
            if best_obj_j < best_obj - 1e-14:
                best_obj = best_obj_j
                improved = True
        if not improved:
            break
    return f, best_obj

"""
Min-max LP with cutting planes for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses direct peak-minimizing LP direction finding, aggressive line search with
golden section refinement, perturbation restarts to escape local minima,
and multi-length exploration. Key improvement over previous: the LP directly
minimizes the peak of the cross-convolution rather than using a relaxed
formulation, giving more effective descent directions.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_convolve(f, g):
    """Linear cross-convolution f*g via FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_row(f, k, n):
    """Build the coefficient vector for constraint (f*g)[k] <= t.
    Returns (coeffs, lo, hi) where coeffs @ g[lo:hi+1] = (f*g)[k]."""
    lo = max(0, int(k) - n + 1)
    hi = min(n - 1, int(k))
    # coeff of g[i] for i in [lo,hi] is f[k-i]
    # So coeffs = [f[k-lo], f[k-lo-1], ..., f[k-hi]]
    coeffs = f[k - lo : k - hi - 1 : -1] if k - lo != k - hi else np.array([f[k - lo]])
    if len(coeffs) != hi - lo + 1:
        coeffs = np.array([f[k - i] for i in range(lo, hi + 1)])
    return coeffs, lo, hi


def solve_minmax_lp(f, active_idx, n):
    """
    Solve: min t
           s.t. (f*g)[k] <= t  for k in active_idx
                sum(g) = 1
                g >= 0
    Returns (g*, t*) or (None, None).
    """
    g = cp.Variable(n)
    t = cp.Variable()
    constraints = [g >= 0, cp.sum(g) == 1]

    for k in active_idx:
        coeffs, lo, hi = build_conv_row(f, int(k), n)
        constraints.append(coeffs @ g[lo:hi + 1] <= t)

    prob = cp.Problem(cp.Minimize(t), constraints)

    g_val = None
    t_val = None
    try:
        prob.solve(solver=cp.SCS, max_iters=300000, eps=1e-9)
        if g.value is not None:
            g_val = np.maximum(g.value, 0.0)
            t_val = float(t.value) if t.value is not None else float('inf')
    except Exception:
        pass

    if g_val is None or not np.all(np.isfinite(g_val)):
        try:
            prob.solve(solver=cp.ECOS, max_iters=100000)
            if g.value is not None:
                g_val = np.maximum(g.value, 0.0)
                t_val = float(t.value) if t.value is not None else float('inf')
        except Exception:
            pass

    if g_val is None:
        return None, None

    s = g_val.sum()
    if s < 1e-10:
        return None, None
    g_val /= s

    if t_val is None or not np.isfinite(t_val):
        t_val = cross_convolve(f, g_val).max()

    return g_val, t_val


def eval_step(f, g, t):
    """Evaluate objective at (1-t)*f + t*g (normalized, projected to non-negative)."""
    f_trial = (1.0 - t) * f + t * g
    f_trial = np.maximum(f_trial, 0.0)
    s = f_trial.sum()
    if s < 1e-10:
        return np.inf
    f_trial /= s
    return fast_convolve(f_trial).max()


def line_search(f, g):
    """
    Find t* = argmin_t max_k conv((1-t)f+tg, (1-t)f+tg)[k]
    using grid scan + golden section refinement.
    Returns (t*, obj*).
    """
    # Grid scan
    grid = np.linspace(0.0, 1.0, 30)
    best_t = 0.0
    best_obj = np.inf

    for t in grid:
        obj = eval_step(f, g, t)
        if obj < best_obj:
            best_obj = obj
            best_t = t

    if best_t <= 0.01:
        return best_t, best_obj

    # Narrow the interval
    idx = int(round(best_t * (len(grid) - 1)))
    lo = grid[max(0, idx - 2)]
    hi = grid[min(len(grid) - 1, idx + 2)]

    # Golden section refinement
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = eval_step(f, g, c)
    fd = eval_step(f, g, d)

    for _ in range(25):
        if fc < fd:
            hi = d
            d = c
            fc = fd
            c = hi - gr * (hi - lo)
            fd = eval_step(f, g, c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = eval_step(f, g, d)

    t_opt = (lo + hi) / 2.0
    obj_opt = eval_step(f, g, t_opt)

    if obj_opt < best_obj:
        best_obj = obj_opt
        best_t = t_opt

    return best_t, best_obj


def minmax_optimize(f_init, time_limit, initial_k=80, max_k=None, label="mm"):
    """
    Min-max LP with cutting planes.
    Iteratively solves the LP, checks for violated constraints, adds them,
    and performs line search.
    """
    t0 = time.monotonic()
    n = len(f_init)
    if max_k is None:
        max_k = n

    f = f_init.copy().astype(np.float64)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f /= s

    b = fast_convolve(f)
    best_f = f.copy()
    best_obj = b.max()

    # Initialize active set from current peaks (use first half by symmetry)
    b_half = b[:n]
    k_init = min(initial_k, n)
    if k_init < n:
        active_idx = np.argpartition(b_half, -k_init)[-k_init:]
    else:
        active_idx = np.arange(n)
    active_idx = np.sort(active_idx)

    it = 0
    no_improve = 0
    stuck = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 20:
            break

        # Solve min-max LP
        g, t_lp = solve_minmax_lp(f, active_idx, n)

        if g is None:
            stuck += 1
            if stuck > 6:
                break
            continue
        stuck = 0

        # Cutting plane: find violated constraints
        fg = cross_convolve(f, g)
        fg_half = fg[:n]
        violated = fg_half > t_lp + 1e-7
        v_idx = np.where(violated)[0]
        not_active = v_idx[~np.isin(v_idx, active_idx)]

        if len(not_active) > 0 and len(active_idx) < max_k:
            n_add = min(25, len(not_active))
            to_add = not_active[np.argsort(fg_half[not_active])[-n_add:]]
            active_idx = np.unique(np.concatenate([active_idx, to_add]))
            active_idx = np.sort(active_idx)
            # Re-solve with expanded constraints
            g, t_lp = solve_minmax_lp(f, active_idx, n)
            if g is None:
                stuck += 1
                if stuck > 4:
                    break
                continue

        # Line search
        t_opt, obj_opt = line_search(f, g)

        if t_opt < 0.001:
            no_improve += 1
            if no_improve > 40:
                break
            # Try expanding active set
            if len(active_idx) < max_k:
                top_k = min(max_k, n)
                b_new = fast_convolve(f)
                active_idx = np.argpartition(b_new[:n], -top_k)[-top_k:]
            continue

        # Update
        f_new = (1.0 - t_opt) * f + t_opt * g
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-10:
            break
        f_new /= s_new
        f = f_new

        b = fast_convolve(f)
        cur_obj = b.max()

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Periodically refresh active set to focus on current peaks
        if it % 15 == 0:
            top_k = min(max_k, n)
            active_idx = np.argpartition(b[:n], -top_k)[-top_k:]
            active_idx = np.sort(active_idx)

        if it % 8 == 0:
            print(f"    {label} it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} "
                  f"active={len(active_idx)} no_impr={no_improve} t={time.monotonic()-t0:.0f}s")

        if no_improve > 60:
            break

    return best_f, best_obj


def subgradient_step(f, n_top=50):
    """One subgradient step to reduce the peak of auto-convolution."""
    n = len(f)
    b = fast_convolve(f)
    # Find top peaks
    k_top = min(n_top, n)
    b_half = b[:n]
    top_idx = np.argpartition(b_half, -k_top)[-k_top:]

    # Subgradient of max_k conv(f,f)[k] is 2 * sum of shifted f for active k
    g = np.zeros(n)
    for k in top_idx:
        lo = max(0, int(k) - n + 1)
        hi = min(n - 1, int(k))
        for i in range(lo, hi + 1):
            g[i] += 2.0 * f[int(k) - i]

    # Normalize subgradient
    g_norm = np.linalg.norm(g)
    if g_norm < 1e-15:
        return f.copy()

    # Project negative gradient onto tangent of simplex
    g_tangent = g - np.mean(g)
    f_new = f - 0.01 * g_tangent
    f_new = project_simplex(f_new)
    return f_new


def perturb_and_optimize(f_init, time_limit, n_perturb=4, label="pert"):
    """Random perturbations followed by min-max LP optimization."""
    t0 = time.monotonic()
    n = len(f_init)
    best_f = f_init.copy()
    best_obj = fast_convolve(f_init).max()
    rng = np.random.default_rng(777)

    per_time = (time_limit - 10) / n_perturb

    for p in range(n_perturb):
        if time.monotonic() - t0 > time_limit - 15:
            break

        # Different perturbation strategies
        if p == 0:
            noise = rng.standard_normal(n) * 0.03
        elif p == 1:
            noise = rng.standard_normal(n) * 0.06
        elif p == 2:
            # Sparse perturbation
            noise = np.zeros(n)
            idx = rng.choice(n, size=max(1, n // 10), replace=False)
            noise[idx] = rng.standard_normal(len(idx)) * 0.1
        else:
            # Smooth perturbation
            noise = np.convolve(rng.standard_normal(n), np.ones(5) / 5, mode='same') * 0.04

        f_pert = f_init + noise
        f_pert = np.maximum(f_pert, 0)
        s = f_pert.sum()
        if s < 1e-10:
            continue
        f_pert /= s

        t_rem = min(per_time, time_limit - (time.monotonic() - t0) - 10)
        if t_rem < 25:
            break

        f_opt, obj = minmax_optimize(f_pert, time_limit=t_rem, initial_k=40, max_k=n,
                                     label=f"{label}{p}")
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f_opt.copy()
            print(f"  {label} pert {p}: obj={best_obj:.10f} score={2*n*best_obj:.6f}")

    return best_f, best_obj


def coordinate_polish(f_init, time_limit):
    """Coordinate descent with multiplicative updates for final polish."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    step = max(1, n // 400)

    for sweep in range(100):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for j in range(0, n, step):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-14:
                continue
            best_val = f[j]
            best_obj_j = best_obj
            for mult in [0.8, 0.9, 0.95, 1.05, 1.1, 1.2]:
                f[j] = best_val * mult
                s = f.sum()
                if s < 1e-10:
                    continue
                f_temp = f / s
                obj = fast_convolve(f_temp).max()
                if obj < best_obj_j - 1e-14:
                    best_obj_j = obj
            f[j] = best_val
            s = f.sum()
            if s > 1e-10:
                f /= s
            if best_obj_j < best_obj - 1e-14:
                best_obj = best_obj_j
                improved = True
        if not improved:
            break

    return f, best_obj
