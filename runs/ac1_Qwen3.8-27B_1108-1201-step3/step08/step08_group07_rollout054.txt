Improved LP fictitious play with iterate averaging, active-set LP reduction,
FFT-based gradient polishing, and multi-length exploration for minimizing
2n * max(conv(a,a)) / (sum(a))^2.

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.linalg import toeplitz

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def build_conv_matrix(f):
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)

def solve_best_response(f, M, prev_g=None, active_set=None):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    if active_set is not None:
        # Use only active constraints for speed
        C = build_conv_matrix(f)
        constraints = [C[active_set] @ g <= M * np.ones(len(active_set))]
    else:
        C = build_conv_matrix(f)
        constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None))
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return prev_g if prev_g is not None else f.copy()
    if g.value is None:
        return prev_g if prev_g is not None else f.copy()
    return np.maximum(g.value, 0)

    """LP fictitious play with iterate averaging."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    prev_g = None
    step = step_init
    stall_count = 0
    stall_threshold = 100
    
    # Iterate averaging
    avg_a = a.copy()
    total_weight = 0.0
    
    while time.monotonic() - t0 < time_limit:

        conv = fast_convolve(a)
        M = conv.max()

            k_active = max(n, (2*n - 1) // 2)
            active_set = np.argsort(conv)[-k_active:]
        else:

        prev_g = g
        g_sum = g.sum()
        
        if g_sum < 1e-10:

        g_norm = g / g_sum
        a_new = (1 - step) * a + step * g_norm
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:

        total_weight += 1.0
        avg_a = (avg_a * (total_weight - 1.0) + a_new) / total_weight
        
        score = compute_score(a_new)

            stall_count = 0

        a = a_new

"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Phase 1: LP fictitious play with iterate averaging and adaptive steps.
Phase 2: L-BFGS-B on log-sum-exp smoothed objective with FFT gradients.
Phase 3: Multi-length exploration.
Phase 4: Final polishing with small-step LP and coordinate descent.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from scipy.linalg import toeplitz


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    """Build (2n-1) x n Toeplitz matrix for convolution with f."""
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_best_response(f, M, prev_g=None):
    """Solve: max sum(g) s.t. C(f) @ g <= M, g >= 0."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None))
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return prev_g if prev_g is not None else f.copy()
    if g.value is None:
        return prev_g if prev_g is not None else f.copy()
    return np.maximum(g.value, 0)


def lp_fictitious_play(a0, time_limit, step_init=0.5, verbose=True, label="LP"):
    """LP fictitious play with iterate averaging and adaptive step."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    prev_g = None
    step = step_init
    stall_count = 0
    
    # Running average
    avg_a = a.copy()
    avg_sum = a.copy()
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()
        
        g = solve_best_response(a, M, prev_g=prev_g)
        prev_g = g
        g_sum = g.sum()
        
        if g_sum < 1e-10:
            break
        
        g_norm = g / g_sum
        a_new = (1 - step) * a + step * g_norm
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        
        # Update running average
        avg_sum += a_new
        avg_a = avg_sum / iters
        
        score = compute_score(a_new)
        avg_score = compute_score(avg_a)
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            stall_count = 0
        elif avg_score < best_score - 1e-12:
            best_score = avg_score
            best_a = avg_a.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        a = a_new
        
        # Adaptive step
        if stall_count > 50:
            step = max(0.05, step * 0.7)
            stall_count = 0
        
        # Geometric decay
        if iters % 100 == 0:
            step = max(0.03, step * 0.93)
        
        if verbose and iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, "
                  f"step={step:.4f}, time={elapsed:.0f}s/{time_limit:.0f}s")
    
    return best_a, best_score


def softmax_from_z(z):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-30))


def smoothed_obj_and_grad(z, T, n):

    conv = fast_convolve(a)
    
    # Log-sum-exp for numerical stability

    shifted = T * (conv - max_conv)
    exp_shifted = np.exp(shifted)
    log_sum_exp = max(T * max_conv, 0) + np.log(exp_shifted.sum())

    obj = (2.0 * n / T) * np.exp(log_sum_exp)
    # Wait, that's not right. The smoothed objective is:
    # f_T(a) = (2n/T) * log(sum_k exp(T * (a*a)_k))
    # = (2n/T) * log_sum_exp
    # Not (2n/T) * exp(log_sum_exp)
    # Let me fix this.
    
    # Actually, the smoothed objective that converges to 2n*max(conv) as T->inf is:
    # f_T(a) = (2n/T) * log(sum_k exp(T * (a*a)_k))
    # As T->inf, this approaches 2n * max_k (a*a)_k. Correct.
    
    obj = (2.0 * n / T) * (T * max_conv + np.log(exp_shifted.sum()))
    # = 2n * max_conv + (2n/T) * log(exp_shifted.sum())
    
    # Weights for gradient
    w = exp_shifted / exp_shifted.sum()
    
    # Gradient of f_T with respect to a:
    # grad_a[i] = 2n * sum_k w_k * d(a*a)_k / da_i
    # = 2n * sum_k w_k * a_{k-i} (where valid)
    # = 2n * cross_correlation(w, a)[i]
    # where cross_correlation(w, a)[i] = sum_j w[i+j] * a[j]
    
    size = 1 << (4 * n - 1).bit_length()
    W = rfft(w, size)
    A = rfft(a, size)
    # corr[i] = sum_j a[j] * w[j+i] = ifft(conj(fft(a)) * fft(w))[i]
    corr = irfft(np.conj(A) * W, size)
    grad_a = 2.0 * n * corr[:n]
    
    # Gradient in z-space: grad_z = a * (grad_a - dot(grad_a, a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

def lbgfs_smoothed(a0, time_limit, T_start=200, T_end=5000, n_temps=5, maxiter_per_T=2000, verbose=True):
    """L-BFGS-B on smoothed objective with decreasing T (increasing sharpness)."""
    n = len(a0)
    z = z_from_a(a0.copy())
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)
    
    temps = np.geomspace(T_start, T_end, n_temps)
    
    for T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        
        remaining = time_limit - (time.monotonic() - t0)
        maxiter = min(maxiter_per_T, int(remaining * 10))  # rough estimate
        
        def func(z_inner):
            obj, _ = smoothed_obj_and_grad(z_inner, T, n)
            return obj
        
        def jac(z_inner):
            _, grad = smoothed_obj_and_grad(z_inner, T, n)
            return grad
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxfun': maxiter * 30, 'disp': False,
                         'maxls': 50}
            )
            a_new = softmax_from_z(res.x)
            score = compute_score(a_new)
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
                z = res.x.copy()
            else:
                # Keep the z from the optimization even if score didn't improve
                # (the smoothed objective might have improved)
                z = res.x.copy()
        except Exception as e:
            print(f"  L-BFGS error at T={T}: {e}")
        
        # Also update z from the current best
        z = z_from_a(best_a)
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"  L-BFGS T={T:.0f}, best={best_score:.8f}, time={elapsed:.0f}s")
    
    return best_a, best_score

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    """Build (2n-1) x n Toeplitz matrix C such that C @ g = f * g."""
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_best_response(f, M, prev_g=None):
    """Solve: max sum(g) s.t. C(f) @ g <= M, g >= 0."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None),
                   max_iter=500)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return prev_g if prev_g is not None else f.copy()
    if g.value is None:
        return prev_g if prev_g is not None else f.copy()
    return np.maximum(g.value, 0)


def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if not np.all(np.isfinite(v)):
        return np.ones(n) / n
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def power_mean_objective(a, p):

def power_mean_gradient(a, p):

    v = conv ** (p - 1)
    size = 1 << (4 * n - 1).bit_length()
    V = rfft(v, size)
    A = rfft(a, size)
    corr = irfft(np.conj(A) * V, size)
    return p * corr[:n]

def lbgfs_smoothed(a0, time_limit, T_values=None, maxiter_per_T=3000, verbose=True):
    """L-BFGS-B on log-sum-exp smoothed objective in softmax parameterization."""
    n = len(a0)
    if T_values is None:
        T_values = [100, 300, 1000, 3000, 10000]

    z = np.log(np.maximum(a0, 1e-30))

    best_a = a0.copy()
    best_score = compute_score(a0)

    for T in T_values:
        if time.monotonic() - t0 > time_limit:
            break

        remaining = time_limit - (time.monotonic() - t0)
        maxiter = min(maxiter_per_T, max(100, int(remaining * 5)))

        def func(z_inner):
            a_local = softmax_from_z_top(z_inner)
            conv = fast_convolve(a_local)
            max_c = conv.max()
            shifted = T * (conv - max_c)
            lse = T * max_c + np.log(np.exp(shifted).sum())
            return (2.0 * n / T) * lse

        def jac(z_inner):
            a_local = softmax_from_z_top(z_inner)
            conv = fast_convolve(a_local)
            max_c = conv.max()
            shifted = T * (conv - max_c)
            exp_s = np.exp(shifted)
            w = exp_s / exp_s.sum()
            size = 1 << (4 * n - 1).bit_length()

            A = rfft(a_local, size)
            corr = irfft(np.conj(A) * W, size)
            grad_a = 2.0 * n * corr[:n]
            S = np.dot(grad_a, a_local)
            return a_local * (grad_a - S)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxfun': maxiter * 30, 'disp': False, 'maxls': 50}

            a_new = softmax_from_z_top(res.x)
            score = compute_score(a_new)

                best_score = score
                best_a = a_new.copy()
            z = res.x.copy()
        except Exception as e:
            print(f"  L-BFGS error at T={T}: {e}")

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"  L-BFGS T={T}, best_score={best_score:.8f}, time={elapsed:.0f}s")

    return best_a, best_score


def softmax_from_z_top(z):
    """Top-level softmax (no closure)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def coordinate_descent(a0, time_limit, n_pairs=200, verbose=True):

            best_t = 0.0

                a_trial = a_trial / ss

                    best_t = t

                best_score = best_local

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        a_lp, sc_lp = lp_fictitious_play_avg(best_seq, t2, step_init=0.4,
                                             label="LP-avg")
        best_seq, best_score = check_and_update(a_lp, "lp_avg", best_seq, best_score)

    # === Phase 3: L-BFGS smoothing ===
    print(f"\n=== Phase 3: L-BFGS Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.15, 200.0)
    if rem > 40:
        print(f"  Budget: {t3:.0f}s")
        a_lb, sc_lb = lbgfs_smoothed(best_seq, t3,
                                     T_values=[200, 500, 1500, 5000],
                                     maxiter_per_T=2000)
        best_seq, best_score = check_and_update(a_lb, "lbgfs", best_seq, best_score)

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-Length (best={best_score:.8f}) ===")
    alt_lengths = []
    for delta in [2, 4, 8, -2, -4]:
        alt_n = n_parent + delta
        if alt_n >= 128 and alt_n <= 4096 and alt_n not in alt_lengths:
            alt_lengths.append(alt_n)

    for alt_n in alt_lengths[:3]:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t4 = min(rem * 0.12, 100.0)
        init_alt = interpolate_sequence(best_seq, alt_n)
        # Quick PG + LP
        a_alt_pg, _ = convex_pg_power_mean(init_alt, t4 * 0.5, p_start=4, p_end=40,
                                           n_phases=4, verbose=False, label="PG-alt")
        a_alt, sc_alt = lp_fictitious_play_avg(a_alt_pg, t4 * 0.5, step_init=0.3,
                                               verbose=False, label="LP-alt")
        best_seq, best_score = check_and_update(a_alt, f"n{alt_n}", best_seq, best_score)

    # === Phase 5: Final LP polish on best ===
    print(f"\n=== Phase 5: Final LP Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.15, 100.0)
    if rem > 20:
        print(f"  Budget: {t5:.0f}s")
        a_fp, sc_fp = lp_fictitious_play_avg(best_seq, t5, step_init=0.2,
                                             label="LP-final")
        best_seq, best_score = check_and_update(a_fp, "lp_final", best_seq, best_score)

    # === Phase 6: Coordinate descent ===
    print(f"\n=== Phase 6: Coordinate Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.10, 60.0)
    if rem > 15:
        print(f"  Budget: {t6:.0f}s")
        a_cd, sc_cd = coordinate_descent(best_seq, t6, n_pairs=150, verbose=False)
        best_seq, best_score = check_and_update(a_cd, "coord_desc", best_seq, best_score)

    # Final verification
    if best_seq is not None:
        try:
            fs = float(evaluate_sequence(best_seq.tolist()))
            if np.isfinite(fs) and fs < best_score:
                best_score = fs
        except Exception:
            pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
