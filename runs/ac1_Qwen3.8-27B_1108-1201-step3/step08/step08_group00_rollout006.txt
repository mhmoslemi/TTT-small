"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-stage L-BFGS with
LogSumExp annealing, LP-based initialization, multi-length exploration,
and coordinate descent polishing.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def lse_gradient(a, T):
    """Compute gradient of (2n/T)*logsumexp(T*conv(a,a)) w.r.t. z (log-domain of a)."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    
    # grad_a: gradient of LSE w.r.t. a
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    
    # Convert to z-gradient via softmax Jacobian
    S = np.dot(grad_a, a)
    return a * (grad_a - S)


def lse_value(a, T):
    """Compute (2n/T)*logsumexp(T*conv(a,a))."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    return (2.0 * n / T) * log_sum_exp

    n = len(a0)
    z = z_from_a(a0).copy()

                return 1e15
            a = softmax_from_z(z_)
            return lse_value(a, T_val)
        return func

                return np.zeros_like(z_)
            a = softmax_from_z(z_)
            return lse_gradient(a, T_val)
        return jac
    
    temps = np.geomspace(T_start, T_end, n_stages)

def build_conv_matrix(a):

    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def structured_init(n, rng):
    """Create structured initial sequence."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 4
    elif choice == 3:
        x = np.linspace(0, 1, n)
        # Asymmetric bump
        a = np.exp(-6 * (x - 0.3) ** 2)
    elif choice == 4:
        x = np.linspace(0, 1, n)
        # Two bumps
        a = np.exp(-8 * (x - 0.3) ** 2) + np.exp(-8 * (x - 0.7) ** 2)
    elif choice == 5:
        # Power law
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 2
    elif choice == 6:
        # Asymmetric power
        x = np.linspace(0, 1, n)
        a = x ** 1.5
    else:
        # Random smooth
        a = rng.random(n)
        # Smooth it
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-stage L-BFGS with
LogSumExp annealing, LP-based initialization, coordinate descent peak shaving,
and multi-length exploration. Key innovation: very aggressive low-temperature

import cvxpy as cp

from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def lse_value_from_a(a, T):
    """Compute (2n/T)*logsumexp(T*conv(a,a))."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    return (2.0 * n / T) * log_sum_exp


def lse_grad_from_a(a, T):
    """Compute gradient w.r.t. z (log-domain of a) of (2n/T)*logsumexp(T*conv(a,a))."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    return a * (grad_a - S)


def lse_obj(z, T, t0, time_limit):

    if time.monotonic() - t0 > time_limit:
        return 1e15
    a = softmax_from_z(z)
    return lse_value_from_a(a, T)

def lbgfs_refine(a0, time_limit, T_start=500.0, T_end=0.05, n_stages=12):
    """L-BFGS with LogSumExp smoothing and geometric temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)

    for stage_i, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        # More iterations at lower temperatures for fine-tuning
        if stage_T > 50:
            maxiter = 200
        elif stage_T > 10:
            maxiter = 400
        elif stage_T > 1:
            maxiter = 600
        else:
            maxiter = 1000

        maxfun = maxiter * 500

        try:
            res = minimize(
                fun=lse_obj,
                x0=z,
                jac=lse_grad,
                args=(stage_T, t0, time_limit),
                method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-15,
                         'maxfun': maxfun, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        a = softmax_from_z(z)
        score = 2.0 * n * fast_convolve(a).max()
        elapsed = time.monotonic() - t0
        print(f"    L-BFGS stage {stage_i+1}/{n_stages} (T={stage_T:.3f}, {elapsed:.0f}s): score={score:.8f}")

    return softmax_from_z(z)


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T
