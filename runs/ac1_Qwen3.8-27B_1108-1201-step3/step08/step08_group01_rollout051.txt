30. **One more idea: "spectral" approach**

If I take the DFT of a, say A[m] = sum_k a[k] * exp(-2πimk/n), then for circular convolution:
(a*a)_circ[k] = (1/n) sum_m |A[m]|^2 * exp(2πimk/n)

The peak of this is bounded by (1/n) * max_k |sum_m |A[m]|^2 * exp(2πimk/n)| <= (1/n) * sum_m |A[m]|^2 = (1/n) * n * sum_k a[k]^2 = sum_k a[k]^2

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

                    delta = 0.05 * a[i]
                elif trial == 5:
                    delta = 0.1 * a[i]
                elif trial == 6:
                    delta = 0.2 * a[i]
                elif trial == 7:
                    delta = -0.7 * a[i]
                elif trial == 8:
                    delta = 0.3 * a[i]
                else:
                    delta = 0.4 * a[i]

                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx_2i] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-14:
                a[i] = max(a[i] + best_delta, 0.0)

                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx_2i] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0

    return best_a, best_score, sweeps


def two_d_pivot(a0, time_limit, max_peaks=15, top_cand=6, label=""):
    """2-coordinate pivot optimization. For each peak, try best (i,j) pairs."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    s = 1.0

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    iterations = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        iterations += 1
        M = float(conv.max())
        current_score = 2.0 * n * M

        # Find top peaks
        n_p = min(max_peaks, len(conv))
        peak_indices = np.argpartition(conv, -n_p)[-n_p:]
        peak_indices = peak_indices[np.argsort(conv[peak_indices])[::-1]]

        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0
        best_new_conv = None
        best_new_M = M

        for k_peak in peak_indices:
            # Get partner values for this peak
            partners = get_partners(a, n, k_peak)
            
            # Best i: maximize partner (large contribution to peak)
            # Best j: minimize partner (small contribution to peak)
            # Exclude a[i] < 1e-15 for i
            valid_i = [idx for idx in np.argsort(partners)[::-1][:top_cand * 2] if a[idx] > 1e-8]
            valid_j = [idx for idx in np.argsort(partners)[:top_cand * 2] if a[idx] > 1e-8]
            
            # Take top candidates
            top_i_list = valid_i[:top_cand]
            top_j_list = valid_j[:top_cand]

            for i in top_i_list:
                for j in top_j_list:
                    if i == j:
                        continue
                    # Compute change vector
                    c = compute_change_vector(a, n, i, j)
                    
                    # Find best delta using ternary search
                    # delta in [-a[j], a[i]]
                    d_lo = -a[j]
                    d_hi = a[i]
                    if d_lo >= d_hi:
                        continue
                    
                    # Quick evaluation at a few points
                    candidates_d = [0.0, d_hi * 0.25, d_hi * 0.5, d_hi * 0.75, d_hi]
                    # Also add the point where peak equals second highest
                    sorted_conv = np.sort(conv)[::-1]
                    if len(sorted_conv) > 1 and c[k_peak] < 0:
                        # Peak will decrease; find where it crosses second max
                        second_max = sorted_conv[1]
                        d_cross = (M - second_max) / (-c[k_peak]) if c[k_peak] != 0 else 0
                        if d_lo < d_cross < d_hi:
                            candidates_d.append(d_cross)
                    
                    best_local_d = 0.0
                    best_local_max = M
                    for d in candidates_d:
                        new_conv_vals = conv + d * c
                        new_max = float(new_conv_vals.max())
                        if new_max < best_local_max:
                            best_local_max = new_max
                            best_local_d = d
                    
                    # Refine around best_local_d with ternary search
                    if best_local_d != 0.0:
                        lo = max(d_lo, best_local_d * 0.5 - 0.01 / max(n, 1))
                        hi = min(d_hi, best_local_d * 1.5 + 0.01 / max(n, 1))
                        lo = max(lo, d_lo)
                        hi = min(hi, d_hi)
                        if hi - lo > 1e-15:
                            for _ in range(30):
                                m1 = lo + (hi - lo) / 3.0
                                m2 = hi - (hi - lo) / 3.0
                                f1 = float((conv + m1 * c).max())
                                f2 = float((conv + m2 * c).max())
                                if f1 < f2:
                                    hi = m2
                                else:
                                    lo = m1
                            d_mid = (lo + hi) / 2.0
                            f_mid = float((conv + d_mid * c).max())
                            if f_mid < best_local_max:
                                best_local_max = f_mid
                                best_local_d = d_mid

                    if best_local_max < best_new_M:
                        best_new_M = best_local_max
                        best_i = i
                        best_j = j
                        best_delta = best_local_d
                        best_new_conv = conv + best_delta * c

        # Apply best update
        if best_i >= 0 and best_new_M < M - 1e-12:
            a[best_i] = max(a[best_i] - best_delta, 0.0)
            a[best_j] = a[best_j] + best_delta
            a = np.maximum(a, 0.0)
            s_new = a.sum()
            if s_new < 1e-15:
                break
            a = a / s_new
            conv = best_new_conv / (s_new * s_new) if abs(s_new - 1.0) > 1e-10 else best_new_conv
            # Recompute conv exactly to avoid drift
            conv = fast_convolve(a)
            M = float(conv.max())

            cur_score = 2.0 * n * M
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                break

        if iterations % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    2D-pivot {label} it={iterations} ({elapsed:.0f}s): score={2*n*M:.6f} best={best_score:.6f}")

    return best_a, best_score, iterations


def active_set_lp(a0, time_limit, label="", solver=None):
    """LP refinement using only active (near-peak) constraints."""
    import cvxpy as cp
    
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0

    # Determine solver
    if solver is None:
        solver = _get_lp_solver()

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Identify active set: positions where conv >= M - 0.05*M (top 5% threshold)
        threshold = M - 0.05 * M
        active_mask = conv >= threshold
        n_active = int(active_mask.sum())
        
        if n_active < 3:
            # Use top 5 positions
            n_active = 5
            peak_idx = np.argpartition(conv, -5)[-5:]
            active_mask = np.zeros(len(conv), dtype=bool)
            active_mask[peak_idx] = True
        elif n_active > 50:
            # Limit to top 50
            peak_idx = np.argpartition(conv, -50)[-50:]
            active_mask = np.zeros(len(conv), dtype=bool)
            active_mask[peak_idx] = True

        # Build reduced LP
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        T_active = build_conv_matrix(a)[active_mask]
        
        constraints = [
            T_active @ b_var <= M_var,
            cp.sum(b_var) == 1
        ]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        try:
            prob.solve(solver=solver)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS)
            except Exception:
                break
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        
        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        s_b = b_val.sum()
        if s_b < 1e-15:
            break
        b_val = b_val / s_b

        # Damped update
        alpha = min(1.0, 5.0 / (it + 1))
        a_new = (1 - alpha) * a + alpha * b_val
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        conv_new = fast_convolve(a_new)
        M_new = float(conv_new.max())
        score_new = 2.0 * n * M_new

        if score_new < best_score - 1e-10:
            best_score = score_new
            best_a = a_new.copy()
        a = a_new
        conv = conv_new
        M = M_new

        if it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ActiveLP {label} it={it} ({elapsed:.0f}s): score={score_new:.6f} best={best_score:.6f} n_act={n_active}")

    return best_a, best_score, it


def _get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    global _cached_solver
    if '_cached_solver' in globals() and _cached_solver is not None:
        return _cached_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _cached_solver = solver_name
                return solver_name
        except Exception:
            continue
    _cached_solver = cp.ECOS
    return cp.ECOS


def peak_aware_perturb(a0, rng, magnitude=0.05, n_peaks=5):
    """Perturb coordinates that contribute to peaks."""
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    top_k = min(n_peaks, len(conv))
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    
    for k in peaks:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1)
        n_pick = max(1, len(idx) // 5)
        picks = rng.choice(idx, size=min(n_pick, len(idx)), replace=False)
        for i in picks:
            a[i] *= (1.0 + rng.standard_normal() * magnitude)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a0.copy()
    return a / s


def random_perturb(a0, rng, magnitude=0.03):
    """Random multiplicative perturbation."""
    n = len(a0)
    a = a0 * (1.0 + rng.standard_normal(n) * magnitude)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a0.copy()
    return a / s


def sparse_perturb(a0, rng, n_changes=20, magnitude=0.3):
    """Perturb only a few random coordinates."""
    n = len(a0)
    a = a0.copy()
    idx = rng.choice(n, size=min(n_changes, n), replace=False)
    a[idx] *= (1.0 + rng.standard_normal(len(idx)) * magnitude)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a0.copy()
    return a / s


def block_shift_perturb(a0, rng, block_size=5, magnitude=0.5):
    """Shift a block of mass to another location."""
    n = len(a0)
    a = a0.copy()
    # Pick source and destination blocks
    src = rng.integers(0, n - block_size)
    dst = rng.integers(0, n - block_size)
    if src == dst:
        return a
    # Move some mass
    frac = rng.uniform(0.1, 0.4) * magnitude
    for offset in range(block_size):
        a[dst + offset] += frac * a[src + offset]
        a[src + offset] -= frac * a[src + offset]
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a0.copy()
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_fourier(n, rng):
    """Initialize using random Fourier superposition."""
    n_harmonics = int(rng.integers(3, 18))
    amplitudes = rng.random(n_harmonics) * 0.35
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = np.arange(1, n_harmonics + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_cosine(n, rng):
    """Initialize with a cosine bell shape."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * rng.uniform(0.3, 0.7)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_u_shape(n, rng):
    """Initialize with U-shape."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    alpha = rng.uniform(0.5, 2.0)
    a = 1.0 + alpha * np.cos(np.pi * t) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def local_optimize(a0, time_limit, label=""):
    """Combined local optimizer: 1D CD then 2D pivots."""
    n = len(a0)
    t0 = time.monotonic()
    
    # Phase A: 1D CD
    t_cd = min(30.0, time_limit * 0.4)
    a1, s1, _ = one_d_cd(a0, t_cd, n_trials=10, label=label+"_cd")
    
    # Phase B: 2D pivots
    t_rem = time_limit - (time.monotonic() - t0) - 5.0
    if t_rem > 10:
        a2, s2, _ = two_d_pivot(a1, t_rem, max_peaks=12, top_cand=5, label=label+"_2d")
        if s2 < s1:
            return a2, s2
    return a1, s1

                if t5_rem < 15:
                    break
                t_alloc = min(25.0, t5_rem * 0.4)
                a_d, s_d = local_optimize(init_fn(), t_alloc, label=f"p5_{lbl}")
                best_seq, best_score = check_and_update(a_d, best_seq, best_score, f"p5_{lbl}")
                t5_rem = deadline - time.monotonic() - 100

        print(f"  After Phase 5: {best_score:.6f}")

    # === Phase 6: Second round of ILS with smaller perturbations ===
    print(f"\n=== Phase 6: Fine ILS (best={best_score:.6f}) ===")
    t6 = min(200.0, deadline - time.monotonic() - 50)
    if t6 > 30 and best_seq is not None:
        current_best = best_seq.copy()
        t6_start = time.monotonic()
        ills2 = 0
        while time.monotonic() - t6_start < t6:
            ills2 += 1
            # Smaller perturbations for fine-tuning
            perturbed = peak_aware_perturb(current_best, rng, magnitude=0.01 + 0.005 * min(ills2 // 10, 3))
            t_local = min(25.0, deadline - time.monotonic() - 30)
            if t_local < 5:
                break
            a_opt, s_opt = local_optimize(perturbed, t_local, label=f"fine_ils{ills2}")
            if s_opt < best_score:
                best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"fine_ils_{ills2}")
                current_best = a_opt.copy()
            if ills2 % 10 == 0:
                print(f"    Fine ILS it={ills2}: best={best_score:.6f}")
        print(f"  After Phase 6: {best_score:.6f}")

    # === Phase 7: Final intensive polishing ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        # 2D pivot with more peaks
        t7a = min(100.0, deadline - time.monotonic() - 20)
        if t7a > 15:
            a7a, s7a, _ = two_d_pivot(best_seq, t7a, max_peaks=25, top_cand=10, label="final_2d")
            best_seq, best_score = check_and_update(a7a, best_seq, best_score, "final_2d")
        
        # 1D CD with many trials
        t7b = min(60.0, deadline - time.monotonic() - 10)
        if t7b > 10:
            a7b, s7b, _ = one_d_cd(best_seq, t7b, n_trials=14, label="final_cd")
            best_seq, best_score = check_and_update(a7b, best_seq, best_score, "final_cd")

    # Final verification
    if best_seq is None:
        best_seq = np.ones(512) / 512
    
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
