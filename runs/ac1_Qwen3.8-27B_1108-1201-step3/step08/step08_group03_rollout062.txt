def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_pair(f, g):
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def logsumexp_objective(f, T):
    """Compute F_T(f) = T * log(sum exp(b_k/T)) and its gradient."""
    b = fast_convolve(f)
    scaled = b / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    obj = T * (scaled.max() + np.log(p_sum))
    # gradient
    n = len(f)
    g = np.zeros(n)
    p_norm = p / p_sum
    for k in range(len(b)):
        if p_norm[k] < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        g[idx] += p_norm[k] * 2.0 * f[k - idx]
    return obj, g

    rng = np.random.default_rng(rng_seed)

    no_improve = 0

        it += 1
        b = fast_convolve(f)
        max_b = b.max()
        
        # Subgradient: use top n_subgrad peaks weighted
        top_k = min(n_subgrad, len(b))
        top_idx = np.argpartition(b, -top_k)[-top_k:]
        w = b[top_idx]
        w = np.maximum(w, 0)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(top_k) / top_k
        else:
            w /= ws
        
        g = np.zeros(n)
        for i, k in enumerate(top_idx):
            if w[i] < 1e-14:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += w[i] * 2.0 * f[k - idx]
        
        # Linear minimization over simplex: pick argmin of g
        s_idx = np.argmin(g)
        s = np.zeros(n)
        s[s_idx] = 1.0
        
        # Step size: full or half
        # Try a few step sizes
        best_t = 0.0
        best_t_obj = best_obj
        for t in [2.0 / (it + 2), 0.1, 0.2, 0.3, 0.5]:
            f_trial = (1 - t) * f + t * s
            f_trial = np.maximum(f_trial, 0.0)
            st = f_trial.sum()
            if st < 1e-10:
                continue
            f_trial /= st
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t
        
        if best_t > 0:
            f = (1 - best_t) * f + best_t * s
            f = np.maximum(f, 0.0)
            s_f = f.sum()
            if s_f > 1e-10:
                f /= s_f
        
        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-13:

            best_f = f.copy()
            no_improve = 0

    return best_f, best_obj

        # Option 1: Add noise to magnitude
        magnitude_scale = 1.0 + 0.3 * rng.standard_normal(len(abs_F))

def build_constraint_matrix(f, indices, n):
    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A

def solve_lp_direction(f, n, M, active_idx, eps=1e-8):
    K = len(active_idx)
    if K == 0:
        return None
    A = build_constraint_matrix(f, active_idx, n)
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=300000, eps=eps)
    except Exception:
        g.value = None
    if g.value is None:
        if n <= 3000:
            try:
                prob.solve(solver=cp.ECOS, max_iters=30000)
            except Exception:
                return None
        else:
            return None
    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None
    return g_val

def lp_cutting_plane(f_init, time_limit, initial_k=40, max_k=300, label="lp"):
    """Cutting-plane LP optimization."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    b = fast_convolve(f)
    M = b.max()
    best_obj = M
    
    if initial_k < len(b):
        active_idx = np.argpartition(b, -initial_k)[-initial_k:]
    else:
        active_idx = np.arange(len(b))
    active_idx = np.sort(active_idx)
    
    it = 0
    stuck = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break
        
        g = solve_lp_direction(f, n, M, active_idx)
        
        if g is None:
            stuck += 1
            if stuck > 10:
                break
            continue
        
        S_g = g.sum()
        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > 15:
                # Try expanding
                if len(active_idx) < max_k:
                    b_cur = fast_convolve(f)
                    new_top = np.argpartition(b_cur, -min(max_k, len(b_cur)))[-min(max_k, len(b_cur)):]
                    active_idx = np.unique(np.concatenate([active_idx, new_top]))
                continue
        
        # Check violated constraints
        fg = convolve_pair(f, g)
        violated = fg > M * (1 + 1e-7)
        if violated.any() and len(active_idx) < max_k:
            v_idx = np.where(violated)[0]
            not_active = v_idx[~np.isin(v_idx, active_idx)]
            if len(not_active) > 0:
                n_add = min(20, len(not_active))
                to_add = not_active[np.argsort(fg[not_active])[-n_add:]]
                active_idx = np.unique(np.concatenate([active_idx, to_add]))
                g = solve_lp_direction(f, n, M, active_idx)
                if g is None:
                    stuck += 1
                    if stuck > 5:
                        break
                    continue
                S_g = g.sum()
        
        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > 20:
                break
            continue
        
        stuck = 0
        h = g / S_g
        
        # Line search
        best_t = 0.05
        best_t_obj = M
        for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            f_trial = np.maximum((1 - t) * f + t * h, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t
        
        f = np.maximum((1 - best_t) * f + best_t * h, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        
        cur_obj = fast_convolve(f).max()
        M = cur_obj
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f.copy()
        
        # Refresh active set
        if it % 10 == 0:
            b = fast_convolve(f)
            if len(active_idx) > initial_k:
                active_idx = np.argpartition(b, -min(max_k, len(b)))[-min(max_k, len(b)):]
                active_idx = np.sort(active_idx)
        
        if it % 30 == 0:
            print(f"    {label} it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} t={time.monotonic()-t0:.0f}s")
    
    return best_f, best_obj

def coordinate_descent_polish(f_init, time_limit):
    """Coordinate descent with multiplicative updates on blocks."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    block_size = max(4, n // 100)
    
    for sweep in range(50):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for start in range(0, n, block_size):
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj
            end = min(start + block_size, n)
            if f[start:end].max() < 1e-14:
                continue
            
            best_block = f[start:end].copy()
            best_block_obj = best_obj
            
            for mult in [0.7, 0.85, 1.15, 1.3]:
                f_trial = f.copy()
                f_trial[start:end] *= mult
                f_trial = np.maximum(f_trial, 0.0)
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_block_obj - 1e-14:
                    best_block_obj = obj
                    best_block = f_trial[start:end].copy()
            
            f[start:end] = best_block
            s = f.sum()
            if s > 1e-10:
                f /= s
            if best_block_obj < best_obj - 1e-14:
                best_obj = best_block_obj
                improved = True
        
        if not improved:
            break
    
    best_f = f.copy()
    return best_f, best_obj

"""
Multi-method optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Primary method: log-sum-exp smoothed mirror descent with T-annealing, combined
with Frank-Wolfe, spectral diversification, cutting-plane LP, and coordinate descent.
Key difference from previous: uses smooth approximation and multiplicative weight
updates as the main driver, with different direction-finding strategies.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_pair(f, g):
    """Compute cross-convolution f*g via FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_gradient(f, b, T, n_top=300):
    """Compute gradient of T*log(sum exp(b_k/T)) w.r.t. f."""
    n = len(f)
    size = len(b)
    scaled = b / T
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-30:
        return np.zeros(n), np.ones(size) / size
    p /= p_sum

    # Focus on top n_top components for efficiency
    if n_top < size:
        top_idx = np.argpartition(p, -n_top)[-n_top:]
    else:
        top_idx = np.arange(size)

    g = np.zeros(n)
    for k in top_idx:
        pk = p[k]
        if pk < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        j_vals = np.arange(lo, hi + 1)
        g[j_vals] += 2.0 * pk * f[k - j_vals]

    return g, p


def mirror_descent_smoothed(f_init, time_limit, T_init=None, T_min=0.05,
                            decay=0.997, lr_init=0.05, restart_interval=80,
                            restart_strength=0.03, seed=42, n_top=300):
    """Mirror descent (multiplicative weights) with log-sum-exp smoothing and T-annealing."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    rng = np.random.default_rng(seed)

    if T_init is None:
        b = fast_convolve(f)
        T_init = b.max() / 4.0

    T = T_init
    lr = lr_init
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        g, p = compute_lse_gradient(f, b, T, n_top=n_top)

        # Multiplicative weight update
        # Scale gradient to have reasonable magnitude
        g_scale = g.max()
        if g_scale > 1e-14:
            g_norm = g / g_scale
        else:
            g_norm = g

        f_new = f * np.exp(-lr * g_norm)
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-10:
            f_new = f.copy()
        else:
            f_new /= s

        f = f_new

        # Anneal T
        T = max(T_min, T * decay)

        # Evaluate
        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 25:
                lr *= 0.92
                no_improve = 0
                if lr < 0.001:
                    lr = lr_init * 0.5

        # Periodic random perturbations to escape local minima
        if it % restart_interval == 0:
            noise = rng.standard_normal(n) * restart_strength
            f = f + noise
            f = np.maximum(f, 0.0)
            s = f.sum()
            if s > 1e-10:
                f /= s
            # Also try a Dirichlet mix
            if rng.random() < 0.3:
                alpha = rng.dirichlet(np.ones(n) * 0.1)
                f = 0.7 * f + 0.3 * alpha
                f = np.maximum(f, 0.0)
                f /= f.sum()

        if it % 200 == 0:
            print(f"    md it={it} T={T:.4f} lr={lr:.4f} obj={best_obj:.10f} "
                  f"score={2.0*n*best_obj:.6f} t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def frank_wolfe(f_init, time_limit, n_subgrad=5, rng_seed=123):
    """Frank-Wolfe (conditional gradient) for minimizing max(conv(f,f))."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    rng = np.random.default_rng(rng_seed)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        max_b = b.max()

        # Subgradient from top peaks
        top_k = min(n_subgrad, len(b))
        top_idx = np.argpartition(b, -top_k)[-top_k:]
        w = b[top_idx].copy()
        w = np.maximum(w, 0)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(top_k) / top_k
        else:
            w /= w ** 1.5  # Sharper weighting toward peaks
            w /= w.sum()

        g = np.zeros(n)
        for i, k in enumerate(top_idx):
            if w[i] < 1e-14:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += w[i] * 2.0 * f[k - idx]

        # Linear minimization over simplex
        s_idx = np.argmin(g)
        s = np.zeros(n)
        s[s_idx] = 1.0

        # Step size selection
        best_t = 2.0 / (it + 2.0)
        best_t_obj = best_obj
        for t in [2.0 / (it + 2), 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            f_trial = (1 - t) * f + t * s
            f_trial = np.maximum(f_trial, 0.0)
            st = f_trial.sum()
            if st < 1e-10:
                continue
            f_trial /= st
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t

        if best_t > 0:
            f = (1 - best_t) * f + best_t * s
            f = np.maximum(f, 0.0)
            s_f = f.sum()
            if s_f > 1e-10:
                f /= s_f

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 60:
                # Random restart
                mix = rng.random()
                f = (1 - mix) * f + mix * rng.dirichlet(np.ones(n) * 0.5)
                f = np.maximum(f, 0.0)
                f /= f.sum()
                no_improve = 0

        if it % 300 == 0:
            print(f"    fw it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} "
                  f"t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def spectral_diversify(f, n_variants=4, seed=77):
    """Create diverse variants by modifying spectral content."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    rng = np.random.default_rng(seed)
    variants = []

    for v in range(n_variants):
        F = rfft(f, fft_size)
        abs_F = np.abs(F)
        phase = F / (abs_F + 1e-30)

        if v == 0:
            # Magnitude noise
            scale = 1.0 + 0.25 * rng.standard_normal(len(abs_F))
            scale = np.maximum(scale, 0.2)
            new_abs = abs_F * scale
        elif v == 1:
            # Phase perturbation
            random_phase = np.cos(0.15 * rng.standard_normal(len(phase))) + \
                           1j * np.sin(0.15 * rng.standard_normal(len(phase)))
            phase = phase * random_phase
            new_abs = abs_F
        elif v == 2:
            # Low-pass filtering of deviations from uniform
            uniform_spec = np.zeros_like(abs_F)
            uniform_spec[0] = n
            dev = abs_F - uniform_spec
            # Low-pass: keep only first 30% of frequencies
            cutoff = len(abs_F) // 3
            dev[cutoff:] *= 0.3
            new_abs = uniform_spec + dev
            new_abs = np.maximum(new_abs, 0.01)
        else:
            # Random band enhancement
            band_start = rng.integers(1, len(abs_F) - 2)
            band_end = min(band_start + len(abs_F) // 5, len(abs_F))
            new_abs = abs_F.copy()
            new_abs[band_start:band_end] *= (1.0 + 0.3 * rng.random())
            new_abs[len(abs_F) - 1 - band_end:len(abs_F) - band_start] *= (1.0 + 0.3 * rng.random())

        F_new = new_abs * phase
        f_new = irfft(F_new, fft_size)[:n]
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s > 1e-10:
            f_new /= s
        else:
            f_new = f.copy()
        variants.append(f_new)

    return variants


def build_constraint_matrix(f, indices, n):
    """Build constraint matrix for LP."""
    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp_direction(f, n, M, active_idx, eps=1e-8):
    """Solve LP: max sum(g) s.t. g>=0, A@g <= M."""
    K = len(active_idx)
    if K == 0:
        return None
    A = build_constraint_matrix(f, active_idx, n)
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=300000, eps=eps)
    except Exception:
        g.value = None
    if g.value is None:
        if n <= 3000:
            try:
                prob.solve(solver=cp.ECOS, max_iters=30000)
            except Exception:
                return None
        else:
            return None
    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None
    return g_val


def lp_cutting_plane(f_init, time_limit, initial_k=40, max_k=300, label="lp"):
    """Cutting-plane LP direction finding with line search."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    b = fast_convolve(f)
    M = b.max()
    best_obj = M

    if initial_k < len(b):
        active_idx = np.argpartition(b, -initial_k)[-initial_k:]
    else:
        active_idx = np.arange(len(b))
    active_idx = np.sort(active_idx)

    it = 0
    stuck = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break

        g = solve_lp_direction(f, n, M, active_idx)

        if g is None:
            stuck += 1
            if stuck > 10:
                break
            continue

        S_g = g.sum()
        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > 20:
                if len(active_idx) < max_k:
                    b_cur = fast_convolve(f)
                    new_top = np.argpartition(b_cur, -min(max_k, len(b_cur)))[-min(max_k, len(b_cur)):]
                    active_idx = np.unique(np.concatenate([active_idx, new_top]))
                continue

        # Cutting plane
        fg = convolve_pair(f, g)
        violated = fg > M * (1 + 1e-7)
        if violated.any() and len(active_idx) < max_k:
            v_idx = np.where(violated)[0]
            not_active = v_idx[~np.isin(v_idx, active_idx)]
            if len(not_active) > 0:
                n_add = min(25, len(not_active))
                to_add = not_active[np.argsort(fg[not_active])[-n_add:]]
                active_idx = np.unique(np.concatenate([active_idx, to_add]))
                g = solve_lp_direction(f, n, M, active_idx)
                if g is None:
                    stuck += 1
                    if stuck > 5:
                        break
                    continue
                S_g = g.sum()
                if S_g <= 1.0 + 1e-6:
                    stuck += 1
                    if stuck > 15:
                        break
                    continue

        stuck = 0
        h = g / S_g

        # Line search
        best_t = 0.05
        best_t_obj = M
        for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            f_trial = np.maximum((1 - t) * f + t * h, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t

        f = np.maximum((1 - best_t) * f + best_t * h, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        else:
            f = best_f.copy()
            break

        cur_obj = fast_convolve(f).max()
        M = cur_obj
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f.copy()

        # Refresh active set periodically
        if it % 12 == 0:
            b = fast_convolve(f)
            active_idx = np.argpartition(b, -min(max_k, len(b)))[-min(max_k, len(b)):]
            active_idx = np.sort(active_idx)

        if it % 40 == 0:

    return best_f, best_obj


def block_coordinate_descent(f_init, time_limit):
    """Block coordinate descent with multiplicative updates."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    block_size = max(4, n // 120)

    for sweep in range(100):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for start in range(0, n, block_size):
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj
            end = min(start + block_size, n)
            if f[start:end].max() < 1e-14:
                continue

            best_block_obj = best_obj
            best_block = f[start:end].copy()

            for mult in [0.6, 0.8, 1.2, 1.4]:
                f_trial = f.copy()
                f_trial[start:end] *= mult
                f_trial = np.maximum(f_trial, 0.0)
                s = f_trial.sum()
                if s < 1e-10:

                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_block_obj - 1e-14:
                    best_block_obj = obj
                    best_block = f_trial[start:end].copy()

            f[start:end] = best_block
            s = f.sum()
            if s > 1e-10:
                f /= s
            if best_block_obj < best_obj - 1e-14:
                best_obj = best_block_obj
                best_f = f.copy()
                improved = True

        if not improved:
            break

    return best_f, best_obj


def zero_coordinate_search(f_init, time_limit, n_trials=200):
    """Try zeroing individual coordinates or small groups."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    rng = np.random.default_rng(456)

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break
        # Pick 1-3 random coordinates to zero out
        n_zero = rng.integers(1, 4)
        indices = rng.choice(n, size=n_zero, replace=False)
        f_trial = f.copy()
        f_trial[indices] = 0.0
        s = f_trial.sum()
        if s < 1e-10:
            continue
        f_trial /= s
        obj = fast_convolve(f_trial).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f_trial.copy()
            f = f_trial.copy()

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: multi-method optimization combining mirror descent with
    log-sum-exp smoothing, Frank-Wolfe, spectral diversification,
    cutting-plane LP, block coordinate descent, and zero-coordinate search.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception as e:
        print(f"Parent load failed: {e}")

    if parent is None:
        parent = np.ones(1024) / 1024.0
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: uniform n=1024, score={p_score:.6f}")

    best_seq = parent.copy()
    best_score = p_score

    # === Phase 1: Mirror descent on parent ===
    print(f"\n=== Phase 1: Mirror descent (n={len(parent)}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.22, 350.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        f1, obj1 = mirror_descent_smoothed(parent, time_limit=t1,
                                            T_init=fast_convolve(parent).max() / 4.0,
                                            T_min=0.05, decay=0.996,
                                            lr_init=0.06, restart_interval=70,
                                            restart_strength=0.04, seed=42)
        score1 = float(evaluate_sequence(f1.tolist()))
        if np.isfinite(score1) and score1 < best_score:
            best_score = score1
            best_seq = f1.copy()
            print(f"  NEW BEST [md1]: {score1:.6f}")

    # === Phase 2: Second mirror descent with different params ===
    print(f"\n=== Phase 2: Mirror descent v2 (n={len(best_seq)}, best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.15, 200.0)
    if t2 > 20:
        # Start from a slight perturbation
        rng = np.random.default_rng(999)
        f_start = best_seq + rng.standard_normal(len(best_seq)) * 0.01
        f_start = np.maximum(f_start, 0.0)
        f_start /= f_start.sum()
        print(f"  Budget: {t2:.0f}s")
        f2, obj2 = mirror_descent_smoothed(f_start, time_limit=t2,
                                            T_init=fast_convolve(f_start).max() / 3.0,
                                            T_min=0.08, decay=0.995,
                                            lr_init=0.04, restart_interval=50,
                                            restart_strength=0.05, seed=777)
        score2 = float(evaluate_sequence(f2.tolist()))
        if np.isfinite(score2) and score2 < best_score:
            best_score = score2
            best_seq = f2.copy()
            print(f"  NEW BEST [md2]: {score2:.6f}")

    # === Phase 3: Frank-Wolfe ===
    print(f"\n=== Phase 3: Frank-Wolfe (n={len(best_seq)}, best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.12, 180.0)
    if t3 > 20:
        print(f"  Budget: {t3:.0f}s")
        f3, obj3 = frank_wolfe(best_seq, time_limit=t3, n_subgrad=8, rng_seed=55)
        score3 = float(evaluate_sequence(f3.tolist()))
        if np.isfinite(score3) and score3 < best_score:
            best_score = score3
            best_seq = f3.copy()
            print(f"  NEW BEST [fw]: {score3:.6f}")

    # === Phase 4: Spectral diversification + optimization ===
    print(f"\n=== Phase 4: Spectral diversity (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t4_total = min(rem * 0.20, 280.0)
    if t4_total > 80:
        n_b = len(best_seq)
        variants = spectral_diversify(best_seq, n_variants=4, seed=222)
        t4_each = t4_total / len(variants)
        for vi, v in enumerate(variants):
            rem_v = deadline - time.monotonic()
            if rem_v < 20:
                break
            t4_i = min(t4_each, rem_v * 0.9)
            if t4_i < 15:
                break
            print(f"  Variant {vi}: budget={t4_i:.0f}s")

            # Mirror descent
            f4a, _ = mirror_descent_smoothed(v, time_limit=t4_i * 0.45,
                                              T_init=fast_convolve(v).max() / 4.0,
                                              T_min=0.05, decay=0.997,
                                              lr_init=0.05, restart_interval=50,
                                              restart_strength=0.04, seed=300 + vi)
            s4a = float(evaluate_sequence(f4a.tolist()))
            if np.isfinite(s4a) and s4a < best_score:
                best_score = s4a
                best_seq = f4a.copy()
                print(f"    NEW BEST [sd{vi}]: {s4a:.6f}")

            # LP refinement
            rem_v2 = deadline - time.monotonic()
            if rem_v2 > 15:
                f4b, _ = lp_cutting_plane(f4a, time_limit=min(t4_i * 0.45, rem_v2 * 0.85),
                                           initial_k=30, max_k=250, label=f"sd{vi}")
                s4b = float(evaluate_sequence(f4b.tolist()))
                if np.isfinite(s4b) and s4b < best_score:
                    best_score = s4b
                    best_seq = f4b.copy()
                    print(f"    NEW BEST [sd{vi}_lp]: {s4b:.6f}")

    # === Phase 5: LP on best ===
    print(f"\n=== Phase 5: LP main (n={len(best_seq)}, best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.18, 250.0)
    if t5 > 30:
        print(f"  Budget: {t5:.0f}s")
        f5, obj5 = lp_cutting_plane(best_seq, time_limit=t5,
                                     initial_k=60, max_k=400, label="lp_main")
        score5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(score5) and score5 < best_score:
            best_score = score5
            best_seq = f5.copy()
            print(f"  NEW BEST [lp]: {score5:.6f}")

    # === Phase 6: Multi-length ===
    print(f"\n=== Phase 6: Multi-length (best={best_score:.6f}) ===")
    n_base = len(best_seq)
    target_lengths = []
    for factor in [0.88, 0.94, 1.06, 1.12, 1.25]:
        tl = int(n_base * factor)
        tl = max(512, min(2500, tl))
        if tl != n_base and tl not in target_lengths:
            target_lengths.append(tl)
    target_lengths.sort(key=lambda x: abs(x - n_base))

    for target_n in target_lengths[:3]:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t6 = min(rem * 0.12, 120.0)
        if t6 < 20:
            break

        old_x = np.linspace(0.0, 1.0, n_base)
        new_x = np.linspace(0.0, 1.0, target_n)
        f_interp = np.interp(new_x, old_x, best_seq)
        f_interp = np.maximum(f_interp, 0.0)
        s = f_interp.sum()
        if s > 1e-10:
            f_interp /= s

        print(f"  n={target_n}, budget={t6:.0f}s")
        f6a, _ = mirror_descent_smoothed(f_interp, time_limit=t6 * 0.4,
                                          T_init=fast_convolve(f_interp).max() / 4.0,
                                          T_min=0.05, decay=0.997,
                                          lr_init=0.05, restart_interval=50,
                                          restart_strength=0.03, seed=500 + target_n)
        s6a = float(evaluate_sequence(f6a.tolist()))
        if np.isfinite(s6a) and s6a < best_score:
            best_score = s6a
            best_seq = f6a.copy()
            print(f"    NEW BEST [ml_{target_n}]: {s6a:.6f}")

        rem6 = deadline - time.monotonic()
        if rem6 > 15:
            f6b, _ = lp_cutting_plane(f6a, time_limit=min(t6 * 0.45, rem6 * 0.8),
                                       initial_k=30, max_k=250, label=f"ml_{target_n}")
            s6b = float(evaluate_sequence(f6b.tolist()))
            if np.isfinite(s6b) and s6b < best_score:
                best_score = s6b
                best_seq = f6b.copy()
                print(f"    NEW BEST [ml_{target_n}_lp]: {s6b:.6f}")

    # === Phase 7: Final LP burst ===
    print(f"\n=== Phase 7: Final LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t7 = min(rem * 0.15, 200.0)
    if t7 > 20:
        n_b = len(best_seq)
        print(f"  Budget: {t7:.0f}s, n={n_b}")
        f7, _ = lp_cutting_plane(best_seq, time_limit=t7,
                                  initial_k=80, max_k=500, label="final_lp")
        score7 = float(evaluate_sequence(f7.tolist()))
        if np.isfinite(score7) and score7 < best_score:
            best_score = score7
            best_seq = f7.copy()
            print(f"  NEW BEST [final_lp]: {score7:.6f}")

    # === Phase 8: Block coordinate descent ===
    rem = deadline - time.monotonic()
    if rem > 15:
        print(f"\n=== Phase 8: Block CD (best={best_score:.6f}) ===")
        t8 = min(rem * 0.35, 80.0)
        f8, obj8 = block_coordinate_descent(best_seq, time_limit=t8)
        s8 = float(evaluate_sequence(f8.tolist()))
        if np.isfinite(s8) and s8 < best_score:
            best_score = s8
            best_seq = f8.copy()
            print(f"  NEW BEST [bcd]: {s8:.6f}")

    # === Phase 9: Zero coordinate search ===
    rem = deadline - time.monotonic()
    if rem > 5:
        t9 = min(rem * 0.3, 30.0)
        if t9 > 3:
            f9, obj9 = zero_coordinate_search(best_seq, time_limit=t9, n_trials=300)
            s9 = float(evaluate_sequence(f9.tolist()))
            if np.isfinite(s9) and s9 < best_score:
                best_score = s9
                best_seq = f9.copy()
                print(f"  NEW BEST [zc]: {s9:.6f}")

    # === Phase 10: Final mirror descent polish ===
    rem = deadline - time.monotonic()
    if rem > 20:
        print(f"\n=== Phase 10: Final polish (best={best_score:.6f}) ===")
        t10 = min(rem * 0.7, 100.0)
        f10, _ = mirror_descent_smoothed(best_seq, time_limit=t10,
                                          T_init=fast_convolve(best_seq).max() / 8.0,
                                          T_min=0.02, decay=0.999,
                                          lr_init=0.02, restart_interval=100,
                                          restart_strength=0.01, seed=888)
        s10 = float(evaluate_sequence(f10.tolist()))
        if np.isfinite(s10) and s10 < best_score:
            best_score = s10
            best_seq = f10.copy()
            print(f"  NEW BEST [final_md]: {s10:.6f}")

    # Final validation
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'=' * 50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'=' * 50}")

    return [float(x) for x in best_seq]
