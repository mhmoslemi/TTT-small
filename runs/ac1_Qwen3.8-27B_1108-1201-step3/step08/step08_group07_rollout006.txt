"""
Multi-method optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Combines Bussgang spectral flattening, projected subgradient with Nesterov
acceleration, LP best-response, and coordinate descent.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.linalg import toeplitz
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def subgradient_step(a, eta):
    """One step of projected subgradient descent."""
    n = len(a)
    conv = fast_convolve(a)
    k_star = np.argmax(conv)
    # Subgradient: g_j = 2 * a_{k_star - j} (with bounds)
    g = np.zeros(n)
    for j in range(n):
        idx = k_star - j
        if 0 <= idx < n:
            g[j] = 2.0 * a[idx]
    v = a - eta * g
    return project_simplex(v)

    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)

    x = a.copy()  # Nesterov's y

    iters = 0

    t_nesterov = 1.0

        iters += 1

        x = a.copy()

        power = np.abs(A) ** 2
        total_power = power.sum()
        target = total_power / fft_size
        ratio = np.where(power > 1e-30, target / power, 1.0)
        ratio = np.clip(ratio, 0.01, 100.0)
        A_new = A * (ratio ** tau)
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        score = compute_score(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        
        if verbose and iters % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  Bussgang iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")
    
    return best_a, best_score


def build_conv_matrix(f):
    """Build (2n-1) x n dense matrix C such that C @ g = f * g (linear conv)."""
    n = len(f)
    # C[k, j] = f[k-j] if 0 <= k-j < n, else 0
    k_vals = np.arange(2 * n - 1)
    j_vals = np.arange(n)
    # C[k, j] = f[k-j] if 0 <= k-j < n
    diff = k_vals[:, None] - j_vals[None, :]
    C = np.zeros((2 * n - 1, n))
    mask = (diff >= 0) & (diff < n)
    C[mask] = f[diff[mask]]
    return C


def solve_best_response(f, M, prev_g=None):
    """Solve: max sum(g) s.t. C(f) @ g <= M, g >= 0."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None), verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, warm_start=(prev_g is not None), verbose=False)
        except Exception:
            return prev_g if prev_g is not None else f.copy()
    if g.value is None:
        return prev_g if prev_g is not None else f.copy()
    return np.maximum(g.value, 0)


def lp_fictitious_play(a0, time_limit, step=0.4, verbose=True):
    """LP-based fictitious play with warm-starting."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    prev_g = None
    stall_count = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()
        
        g = solve_best_response(a, M, prev_g=prev_g)
        prev_g = g
        g_sum = g.sum()
        
        if g_sum < 1e-10:
            break
        
        g_norm = g / g_sum
        a_new = (1 - step) * a + step * g_norm
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        
        score = compute_score(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        a = a_new
        
        if stall_count > 200:
            step = max(0.05, step * 0.7)
            rng = np.random.default_rng(iters)
            a = best_a + rng.standard_normal(n) * 0.0003
            a = np.maximum(a, 0)
            a = a / a.sum()
            prev_g = None
            stall_count = 0
        
        if verbose and iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LP-FP iter={iters}, best={best_score:.8f}, step={step:.4f}, "
                  f"time={elapsed:.0f}s/{time_limit:.0f}s")
    
    return best_a, best_score


def coordinate_descent(a0, time_limit, verbose=True):
    """Systematic coordinate descent with line search."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        
        # Random pairs with systematic sweep
        n_pairs = min(500, n * 2)
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            
            # Try transferring mass from i to j and vice versa
            for direction in [1, -1]:
                if direction == 1 and a[i] < 1e-15:
                    continue
                if direction == -1 and a[j] < 1e-15:
                    continue
                
                best_local = compute_score(a)
                best_t = 0.0
                base = min(a[i], a[j]) if a[j] > 1e-15 else a[i]
                
                for frac in [0.02, 0.05, 0.1, 0.2, 0.4]:
                    t = frac * base * direction
                    a_trial = a.copy()
                    a_trial[i] -= t
                    a_trial[j] += t
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    sc = compute_score(a_trial)
                    if sc < best_local - 1e-14:
                        best_local = sc
                        best_t = t
                
                if best_t != 0.0:
                    a[i] -= best_t
                    a[j] += best_t
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    best_score = best_local
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved and iters > 10:
            noise = rng.standard_normal(n) * 0.001
            a = best_a.copy() + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
        
        if verbose and iters % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"  CD iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")
    
    return best_a, best_score


def multiplicative_weight_update(a0, time_limit, verbose=True):
    """
    Multiplicative weight updates (mirror descent with entropy).
    a_i <- a_i * exp(-eta * g_i) / Z
    where g is the subgradient at the current point.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    
    # Adaptive eta
    eta = 0.1
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        k_star = np.argmax(conv)
        
        # Subgradient
        g = np.zeros(n)
        low = max(0, k_star - (n - 1))
        high = min(n - 1, k_star)
        js = np.arange(low, high + 1)
        g[js] = 2.0 * a[k_star - js]
        
        # Multiplicative update
        a_new = a * np.exp(-eta * g)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        
        score = compute_score(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            # Reduce eta and try again from best
            eta *= 0.8
            if eta < 1e-6:
                # Add perturbation
                rng = np.random.default_rng(iters)
                a = best_a + rng.standard_normal(n) * 0.001
                a = np.maximum(a, 0)
                a = a / a.sum()
                eta = 0.05
            else:
                a = a_new  # still update to explore
        
        if verbose and iters % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  MWU iter={iters}, best={best_score:.8f}, eta={eta:.6f}, time={elapsed:.0f}s")
    
    return best_a, best_score


def maximize_subgradient(a0, time_limit, verbose=True):
    """
    Instead of minimizing, we MAXIMIZE the subgradient direction.
    This is the LP best-response but done greedily:
    At each step, find the direction that most reduces the peak.
    
    Actually, let's do something different: use the "heaviest" lag to guide
    a proportional update.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        k_star = np.argmax(conv)
        
        # The subgradient tells us which coordinates to decrease
        # g_j = 2 * a_{k_star - j}
        # Large g_j means a_{k_star - j} is large, so we should decrease a_j
        # But we want to REDUCE the convolution at lag k_star
        # This means we want to decrease the product a_i * a_{k_star - i} for all i
        # The "bottleneck" is the pair (i, k_star - i) with the largest product
        
        # Strategy: reduce the largest a_i * a_{k_star-i} pair
        # Find i that maximizes a_i * a_{k_star - i}
        low = max(0, k_star - (n - 1))
        high = min(n - 1, k_star)
        js = np.arange(low, high + 1)
        products = a[js] * a[k_star - js]
        i_max = js[np.argmax(products)]
        
        # Reduce a[i_max] and redistribute
        # Find j where a_{k_star - j} is small (to give mass to)
        # j should be such that k_star - j is in range and a[k_star-j] is small
        candidates = js.copy()
        # Prioritize coordinates where the "partner" a[k_star - j] is small
        partner_vals = a[k_star - js]
        # We want to move mass from i_max to a coordinate j where partner_vals[j] is small
        # Sort by partner value (ascending) - we want to give mass to coordinates
        # where the partner is small
        sorted_idx = np.argsort(partner_vals)
        
        # Try moving a small amount from i_max to the best candidate
        move_amount = a[i_max] * 0.05
        target_j = sorted_idx[0]  # coordinate with smallest partner
        
        a_new = a.copy()
        a_new[i_max] -= move_amount
        a_new[target_j] += move_amount
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        
        score = compute_score(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller move
            move_amount = a[i_max] * 0.01
            a_new = a.copy()
            a_new[i_max] -= move_amount
            a_new[target_j] += move_amount
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            score = compute_score(a_new)
            if score < best_score - 1e-14:
                best_score = score
                best_a = a_new.copy()
                a = a_new
            else:
                # Reset
                a = best_a.copy()
        
        if verbose and iters % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  MaxSubGrad iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")
    
    return best_a, best_score
