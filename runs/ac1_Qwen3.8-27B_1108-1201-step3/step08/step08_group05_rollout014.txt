"""
Hybrid optimizer for auto-convolution peak minimization.
Uses FISTA with smoothed objective, LP refinement with adaptive constraints,
cosine-modulated explorations, and peak-specific refinement.
Targets score < 1.5027 (lower is better).
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    """Cross-convolution f*g via FFT."""
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_gradient(a, mu):
    """
    Compute gradient of (1/mu)*sum_k w[k]*b[k] where w = softmax(b/mu), b = conv(a,a).
    This is the gradient of the log-sum-exp smoothed objective (up to factor 1/mu).
    Returns (grad, peak_value, w)
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    A = rfft(a, fft_size)
    b = irfft(A * A, fft_size)[:size]

    b_max = b.max()
    if b_max < 1e-15:
        return np.zeros(n), 0.0, np.ones(size) / size

    w = np.exp((b - b_max) / mu)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(size)
        w[b.argmax()] = 1.0
    else:
        w /= w_sum

    # grad[i] = a[i] * S(i) + C(i)
    # S(i) = sum_{k=i}^{i+n-1} w[k] (w zero-padded beyond size-1)
    # C(i) = sum_{j=0}^{n-1} w[j+i] * a[j] (w zero-padded)

    # S(i) via cumulative sum
    w_ext = np.zeros(size + n)
    w_ext[:size] = w
    cumw = np.concatenate([[0.0], np.cumsum(w_ext)])
    S = cumw[n:n + n] - cumw[:n]

    # C(i) via FFT cross-correlation
    # C(i) = sum_j w[j+i] * a[j]
    # Use: ifft(FFT(w) * conj(FFT(a)))[i] for zero-padded circular correlation
    N = 1 << (size + n - 1).bit_length()
    W_full = np.fft.fft(w, N)
    A_full = np.fft.fft(a, N)
    corr = np.fft.ifft(W_full * np.conj(A_full)).real[:n]

    grad = a * S + corr
    return grad, b.max(), w


def fista_optimize(f_init, time_limit, mu_init=0.05, mu_min=0.002, mu_decay=0.95):
    """FISTA for smoothed peak minimization with adaptive step size."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    y = f.copy()
    t_k = 1.0
    mu = mu_init
    best_f = f.copy()
    best_peak = fast_convolve(f).max()
    step = 1.0
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        batch_end = min(time.monotonic() + 3.0, t0 + time_limit)
        while time.monotonic() < batch_end:
            it += 1
            grad, peak, w = compute_smooth_gradient(y, mu)

            # Adaptive step size
            z = y - step * grad
            x_new = project_simplex(z)

            # FISTA momentum
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            y = x_new + ((t_k - 1.0) / t_new) * (x_new - f)
            f = x_new
            t_k = t_new

            if it % 20 == 0:
                p = fast_convolve(f).max()
                if p < best_peak - 1e-14:
                    best_peak = p
                    best_f = f.copy()
                    no_improve = 0
                else:
                    no_improve += 1
                    if no_improve >= 5:
                        step *= 0.9
                        no_improve = 0

            if it % 100 == 0:
                step = min(step * 1.02, 5.0)

        # Decrease mu
        mu = max(mu * mu_decay, mu_min)

        if it % 200 == 0:
            p = fast_convolve(f).max()
            score = 2.0 * n * p
            print(f"  FISTA: iter={it} mu={mu:.5f} score={score:.6f} t={time.monotonic()-t0:.0f}s")

    return best_f, best_peak


def lp_refine(f_init, time_limit, top_k_init=50, top_k_max=300, max_cg_iters=20):
    """LP-based refinement with constraint generation."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_peak = fast_convolve(f).max()
    it = 0
    stuck = 0
    top_k = top_k_init

    while time.monotonic() - t0 < time_limit:
        it += 1

        b = fast_convolve(f)
        M = b.max()

        # Select active constraints
        if top_k < len(b):
            top_idx = np.argpartition(b, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(b))
        top_idx = np.sort(top_idx)
        active_list = top_idx.tolist()
        active_set = set(active_list)

        g = None
        for cg_iter in range(max_cg_iters):
            if time.monotonic() - t0 > time_limit:
                break

            K = len(active_list)
            if K > 500:
                b_active = np.array(b[active_list])
                keep = np.argsort(b_active)[-500:]
                active_list = [active_list[i] for i in keep]
                active_set = set(active_list)
                K = len(active_list)

            # Build constraint matrix
            A_mat = np.zeros((K, n))
            for row_i in range(K):
                k = active_list[row_i]
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                A_mat[row_i, lo:hi + 1] = f[lo:hi + 1]

            g_var = cp.Variable(n)
            prob = cp.Problem(
                cp.Maximize(cp.sum(g_var)),
                [g_var >= 0, A_mat @ g_var <= M]
            )

            solved = False
            for solver in [cp.PDLP, cp.SCS, cp.ECOS]:
                try:
                    prob.solve(solver=solver, max_iters=200000)
                    solved = True
                    break
                except Exception:
                    continue

            if not solved or g_var.value is None:
                break

            g = np.maximum(g_var.value, 0.0)
            if np.any(np.isnan(g)) or np.any(np.isinf(g)):
                break

            # Check for violated constraints
            if time.monotonic() - t0 > time_limit * 0.8:
                break

            fg = fast_cross_convolve(f, g)
            mask = np.zeros(len(b), dtype=bool)
            for ai in active_list:
                mask[ai] = True
            violations = np.where((fg > M + 1e-8) & (~mask))[0]

            if len(violations) == 0:
                break

            worst = np.argsort(fg[violations])[-min(10, len(violations)):]
            for v in worst:
                vi = int(v)
                if vi not in active_set:
                    active_set.add(vi)
                    active_list.append(vi)

        if g is None or g.sum() <= 1.0 + 1e-5:
            stuck += 1
            if stuck > 15:
                break
            top_k = min(top_k * 2, top_k_max)
            continue

        S_g = g.sum()
        h = g / S_g

        # Line search: try multiple step sizes
        best_t = 0.05
        best_trial = M
        for t in [0.01, 0.02, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3]:
            f_trial = (1.0 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial = f_trial / s
            obj = fast_convolve(f_trial).max()
            if obj < best_trial:
                best_trial = obj
                best_t = t

        if best_t == 0.0:
            break

        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s

        cur_peak = fast_convolve(f).max()
        if cur_peak < best_peak - 1e-14:
            best_peak = cur_peak
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 30:
                break

        if it % 20 == 0:
            score = 2.0 * n * best_peak
            print(f"  LP: iter={it} score={score:.6f} top_k={top_k} t={time.monotonic()-t0:.0f}s")

    return best_f, best_peak


def peak_shave(f_init, time_limit, n_peaks=5):
    """Directly optimize to reduce the top peaks using coordinate-style updates."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_peak = fast_convolve(f).max()
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(f)

        # Get top peaks
        if n_peaks < len(b):
            top_idx = np.argpartition(b, -n_peaks)[-n_peaks:]
        else:
            top_idx = np.arange(len(b))
        top_idx = np.sort(top_idx)

        # Compute gradient from top peaks (uniform weighting)
        grad = np.zeros(n)
        for k in top_idx:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                grad[j] += f[k - j] + f[j]  # db[k]/da[j]

        # Try multiple step sizes with projection
        best_alpha = None
        best_trial = best_peak
        for alpha in [0.001, 0.003, 0.01, 0.03, 0.05, 0.1, 0.15]:
            f_trial = f - alpha * grad
            f_trial = project_simplex(f_trial)
            obj = fast_convolve(f_trial).max()
            if obj < best_trial - 1e-14:
                best_trial = obj
                best_alpha = alpha

        if best_alpha is None:
            # Try adding noise
            f_trial = project_simplex(f + rng.standard_normal(n) * 0.002)
            obj = fast_convolve(f_trial).max()
            if obj < best_peak:
                f = f_trial
                best_peak = obj
                best_f = f.copy()
            else:
                break

        f = f - best_alpha * grad
        f = project_simplex(f)
        cur_peak = fast_convolve(f).max()

        if cur_peak < best_peak - 1e-14:
            best_peak = cur_peak
            best_f = f.copy()

    return best_f, best_peak


def cosine_modulated_init(n, eps, p, phase):
    """Create a cosine-modulated uniform initialization."""
    i = np.arange(n)
    w = 1.0 + eps * np.cos(2.0 * np.pi * p * i / n + phase)
    w = np.maximum(w, 0.0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s

def smooth_window_init(n, shape='hann'):
    """Create a smooth window initialization."""
    if shape == 'hann':
        w = 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n) / max(1, n)))
    elif shape == 'flat_top':
        a = [0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368]
        w = np.ones(n)
        for i, ai in enumerate(a):
            w -= ai * np.cos(2 * np.pi * i * np.arange(n) / max(1, n - 1) - i * np.pi)
        w = np.maximum(w, 0)
    elif shape == 'raised_cos':
        x = np.linspace(0, 1, n)
        w = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.5)))
        w = np.maximum(w, 0)
    elif shape == 'skewed':
        x = np.arange(n) / max(1, n - 1)
        w = (1 - x) ** 1.5 + 0.1
    elif shape == 'bimodal':
        x = np.arange(n) / max(1, n - 1)
        w = np.exp(-20 * (x - 0.25) ** 2) + np.exp(-20 * (x - 0.75) ** 2) + 0.05
    else:
        w = np.ones(n)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return (w / s).astype(np.float64)


def interpolate_sequence(src, target_n):
    """Interpolate a sequence to a new length."""
    src_n = len(src)
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s < 1e-10:
        return np.ones(target_n) / target_n
    return result / s
