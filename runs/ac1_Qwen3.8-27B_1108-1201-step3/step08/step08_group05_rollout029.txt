def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Multi-strategy optimization for auto-convolution peak minimization.
Combines LP with constraint generation, log-sum-exp smoothed mirror descent,
peak shaving, and multi-length exploration to minimize 2*n*max(conv(a,a))/sum(a)^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    """Cross-convolution f*g via FFT."""
    nf = len(f)
    ng = len(g)
    size = nf + ng - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_objective(b, T):
    """Log-sum-exp: T * log(sum(exp(b/T)))."""
    scaled = b / T
    scaled -= scaled.max()
    return T * (scaled.max() + np.log(np.sum(np.exp(scaled))))


def lse_gradient(b, T, a):
    """
    Gradient of T*log(sum(exp(b/T))) w.r.t. a.
    b = conv(a,a), length 2n-1. a has length n.
    grad[i] = 2 * sum_k w[k] * a[k-i] where w[k] = exp(b[k]/T)/sum(exp(b/T)).
    Computed as 2 * correlate(w, a, mode='valid').
    """
    n = len(a)
    scaled = b / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(b))
        w[b.argmax()] = 1.0
    else:
        w /= w_sum
    # grad[i] = 2 * sum_j w[i+j] * a[j] for j=0..n-1
    # = 2 * np.correlate(w, a, mode='valid')[i]
    # np.correlate(w, a, 'valid')[i] = sum_j w[i+j]*a[j] for j=0..len(a)-1
    # This requires len(w) >= len(a), which is (2n-1) >= n. Good.
    grad = 2.0 * np.correlate(w, a, mode='valid')
    return grad


def mirror_descent_step(a, grad, lr):
    """
    Mirror descent step on the simplex using entropy divergence.
    Works in log-space: p = log(a), update p -= lr*grad, then a = exp(p)/sum(exp(p)).
    """
    n = len(a)
    # Work in log space for stability
    a_clipped = np.maximum(a, 1e-30)
    p = np.log(a_clipped)
    p -= p.max()  # numerical stability
    p -= lr * grad
    p -= p.max()  # again for stability
    a_new = np.exp(p)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    a_new /= s
    return a_new


def lp_solve(f, n, top_k_init, max_constraints, time_limit, rng):
    """
    Solve LP with constraint generation.
    Maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for active k.
    Returns (g, M) or (None, M).
    """
    t0 = time.monotonic()
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return None, M

    # Initial active set: top constraints
    if top_k_init < len(b):
        top_idx = np.argpartition(b, -top_k_init)[-top_k_init:]
    else:
        top_idx = np.arange(len(b))
    active_list = sorted(top_idx.tolist())
    active_set = set(active_list)

    g = None
    prev_obj = -np.inf

    for cg_iter in range(25):
        if time.monotonic() - t0 > time_limit:
            break

        K = len(active_list)
        if K > max_constraints:
            b_active = b[active_list]
            keep = np.argsort(b_active)[-max_constraints:]
            active_list = [active_list[i] for i in sorted(keep)]
            active_set = set(active_list)
            K = len(active_list)

        if K == 0:
            break

        # Build constraint matrix: A[row, j] = f[j] for constraint k=active_list[row]
        # (f*g)[k] = sum_j f[j] * g[k-j] <= M
        # So A[row, j] = f[j] if 0 <= k-row - j < n, i.e., the constraint is sum_j f[j]*g[j'] where j' = k-j
        # Actually: (f*g)[k] = sum_{j} f[j] * g[k-j]
        # For g to have length n, we need 0 <= k-j < n, i.e., k-n+1 <= j <= k
        # So (f*g)[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} f[j] * g[k-j]
        # Let i = k-j, then i ranges from k-min(n-1,k) to k-max(0,k-n+1)
        # = max(0, k-(n-1)) to min(n-1, k) ... j ranges from max(0,k-n+1) to min(n-1,k)
        # and g index is k-j which ranges from k-min(n-1,k) to k-max(0,k-n+1)
        # = max(0, k-n+1) to min(n-1, k) ... same range!
        # So (f*g)[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} f[k-i] * g[i]
        # The constraint row for g is: sum_i f[k-i] * g[i] <= M
        # So A[row, i] = f[k-i] for i in [max(0,k-n+1), min(n-1,k)]

        A = np.zeros((K, n))
        for row_i in range(K):
            k = active_list[row_i]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # g index i goes from lo to hi, and f[k-i] is the coefficient
            for i in range(lo, hi + 1):
                A[row_i, i] = f[k - i]

        g_var = cp.Variable(n)
        prob = cp.Problem(
            cp.Maximize(cp.sum(g_var)),
            [g_var >= 0, A @ g_var <= M]
        )

        solved = False
        for solver in [cp.PDLP, cp.SCS, cp.ECOS]:
            try:
                prob.solve(solver=solver, max_iters=500000)
                solved = True
                break
            except Exception:
                continue

        if not solved or g_var.value is None:
            if g is not None:
                break
            return None, M

        g_new = np.maximum(g_var.value, 0.0)
        if np.any(np.isnan(g_new)) or np.any(np.isinf(g_new)):
            if cg_iter > 0:
                break
            return None, M

        obj_new = g_new.sum()
        if g is not None and abs(obj_new - prev_obj) < 1e-10:
            break
        g = g_new
        prev_obj = obj_new

        # Check for violated constraints
        if time.monotonic() - t0 > time_limit * 0.7:
            break

        fg = fast_cross_convolve(f, g)
        mask = np.zeros(len(b), dtype=bool)
        for ai in active_list:
            mask[ai] = True
        violations = np.where((fg > M + 1e-9) & (~mask))[0]

        if len(violations) == 0:
            break

        # Add most violated constraints
        n_add = min(8, len(violations))
        worst = np.argsort(fg[violations])[-n_add:]
        for v in worst:
            vi = int(v)
            if vi not in active_set:
                active_set.add(vi)
                active_list.append(vi)

    return g, M


def run_lp_optimization(f_init, time_limit, top_k_init=50, top_k_max=300, label="lp"):
    """LP-based optimization with constraint generation and line search."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-10:
        f = f / s
    else:
        return f_init, fast_convolve(f_init).max()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0
    top_k = top_k_init
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 30 == 0 and top_k < top_k_max:
            top_k = min(top_k + 15, top_k_max)

        remaining = time_limit - (time.monotonic() - t0)
        t_lp = max(3.0, min(remaining * 0.15, 40.0))

        g, M = lp_solve(f, n, top_k, 400, t_lp, rng)

        if g is None:
            # Fallback: gradient-based direction
            b = fast_convolve(f)
            top_peaks = np.argpartition(b, -min(8, len(b)))[-min(8, len(b)):]
            grad = np.zeros(n)
            for k in top_peaks:
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                for j in range(lo, hi + 1):
                    grad[j] += f[k - j]
            g = project_simplex(f - 0.01 * grad)
            M = b.max()

        S_g = g.sum()
        if S_g <= 1.0 + 1e-4:
            stuck += 1
            if stuck > 30:
                # Try perturbation
                f_pert = project_simplex(f + rng.standard_normal(n) * 0.002)
                obj_pert = fast_convolve(f_pert).max()
                if obj_pert < best_obj:
                    f = f_pert
                    best_obj = obj_pert
                    best_f = f.copy()
                    stuck = 0
                else:
                    break
            else:
                top_k = min(top_k * 2, top_k_max)
                continue

        h = g / S_g

        # Line search: minimize max(conv((1-t)*f + t*h, (1-t)*f + t*h))
        best_t = 0.05
        best_trial = best_obj

        # Coarse search
        for t in [0.01, 0.02, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            f_trial = (1.0 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            st = f_trial.sum()
            if st < 1e-10:
                continue
            f_trial = f_trial / st
            obj = fast_convolve(f_trial).max()
            if obj < best_trial:
                best_trial = obj
                best_t = t

        if best_t == 0.0 and best_trial >= best_obj:
            stuck += 1
            if stuck > 15:
                break
            continue

        # Fine search around best_t
        if best_t > 0:
            step = max(0.005, best_t * 0.3)
            for t in np.linspace(max(0.001, best_t - step), best_t + step, 15):
                f_trial = (1.0 - t) * f + t * h
                f_trial = np.maximum(f_trial, 0.0)
                st = f_trial.sum()
                if st < 1e-10:
                    continue
                f_trial = f_trial / st
                obj = fast_convolve(f_trial).max()
                if obj < best_trial:
                    best_trial = obj
                    best_t = t

        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s2 = f.sum()
        if s2 > 1e-10:
            f = f / s2

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1

        if it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} obj={best_obj:.12f} "
                  f"score={2.0*n*best_obj:.6f} top_k={top_k} t={elapsed:.0f}s")

    return best_f, best_obj


def run_lse_optimization(f_init, time_limit, label="lse"):
    """
    Log-sum-exp smoothed mirror descent on the simplex.
    Adaptive temperature: start high, decrease over time.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-10:
        f = f / s
    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    # Initial temperature
    b = fast_convolve(f)
    T_init = 0.3 * b.max()
    T_min = 0.01 * b.max()

    it = 0
    stuck = 0
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        # Adaptive temperature
        progress = (time.monotonic() - t0) / time_limit
        T = T_init * (T_min / T_init) ** (progress ** 0.5)
        T = max(T, T_min)

        # Learning rate (decreases over time)
        lr = 0.02 * (1.0 - 0.5 * progress)

        # Batch of steps
        batch_size = max(10, int(50 * (1.0 - progress)))
        for _ in range(batch_size):
            if time.monotonic() - t0 > time_limit:
                break
            it += 1

            b = fast_convolve(f)
            grad = lse_gradient(b, T, f)

            # Mirror descent step
            f_new = mirror_descent_step(f, grad, lr)

            # Occasional projection to simplex (mirror descent already stays positive)
            f = f_new

        # Check improvement
        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 5:
                # Try perturbation
                f_pert = mirror_descent_step(f, grad, lr * rng.uniform(0.5, 3.0))
                obj_pert = fast_convolve(f_pert).max()
                if obj_pert < best_obj:
                    f = f_pert
                    best_obj = obj_pert
                    best_f = f.copy()
                stuck = 0

        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} T={T:.8f} obj={best_obj:.12f} "
                  f"score={2.0*n*best_obj:.6f} t={elapsed:.0f}s")

    return best_f, best_obj


def peak_shaving(f_init, time_limit, label="peak"):
    """
    Direct peak shaving: iteratively reduce the peak by adjusting coordinates.
    For the peak at lag k, we want to reduce b[k] = sum_j a[j]*a[k-j].
    Strategy: find the pair (i, k-i) contributing most to b[k] and reduce it.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-10:
        f = f / s
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        peak_k = b.argmax()
        peak_val = b[peak_k]

        if peak_val < best_obj * 0.999:
            best_obj = peak_val
            best_f = f.copy()
            stuck = 0
        elif stuck > 50:
            break

        # Find top contributors to b[peak_k]
        lo = max(0, peak_k - n + 1)
        hi = min(n - 1, peak_k)
        products = np.array([f[j] * f[peak_k - j] for j in range(lo, hi + 1)])
        total = products.sum()
        if total < 1e-30:
            break

        # Target: reduce the largest product
        # Try moving mass from the larger element of the top pair
        top_local = np.argsort(products)[-min(5, len(products)):]
        improved = False

        for idx in top_local:
            j = lo + idx
            i = peak_k - j
            if i < 0 or i >= n or j < 0 or j >= n:
                continue
            if i == j:
                # Diagonal: a[i]^2 contributes. Reduce a[i] slightly.
                for frac in [0.001, 0.003, 0.005, 0.01]:
                    f_trial = f.copy()
                    reduction = frac * f[i]
                    f_trial[i] -= reduction
                    # Redistribute to non-peak-contributing positions
                    # Find positions that don't contribute to peak
                    contrib_mask = np.zeros(n, dtype=bool)
                    for jj in range(lo, hi + 1):
                        contrib_mask[jj] = True
                        if peak_k - jj >= 0 and peak_k - jj < n:
                            contrib_mask[peak_k - jj] = True
                    redist_mask = ~contrib_mask
                    if redist_mask.sum() > 0:
                        f_trial[redist_mask] += reduction / redist_mask.sum()
                        f_trial = np.maximum(f_trial, 0.0)
                        st = f_trial.sum()
                        if st > 1e-10:
                            f_trial /= st
                            obj = fast_convolve(f_trial).max()
                            if obj < best_obj:
                                best_obj = obj
                                best_f = f_trial.copy()
                                f = f_trial.copy()
                                improved = True
                                break

            else:
                # Off-diagonal: reduce the larger of a[i], a[j]
                if f[i] > f[j]:
                    big_idx = i
                    small_idx = j
                else:
                    big_idx = j
                    small_idx = i

                for frac in [0.001, 0.003, 0.005, 0.01, 0.02]:
                    f_trial = f.copy()
                    reduction = frac * f[big_idx]
                    f_trial[big_idx] -= reduction
                    # Redistribute to positions less involved in the peak
                    # Positions far from the peak lag
                    dist = np.abs(np.arange(n) - peak_k) + np.abs(np.arange(n) + n - peak_k)
                    # Actually, position p contributes to b[peak_k] if a[p]*a[peak_k-p] is large
                    # So prefer to add mass where a[peak_k - p] is small
                    contributions = np.zeros(n)
                    for p in range(n):
                        q = peak_k - p
                        if 0 <= q < n:
                            contributions[p] = f[q]
                    # Add mass where contribution is small
                    add_weights = np.maximum(1.0 - contributions / max(contributions.max(), 1e-10), 0.01)
                    add_weights /= add_weights.sum()
                    f_trial += reduction * add_weights
                    f_trial = np.maximum(f_trial, 0.0)
                    st = f_trial.sum()
                    if st > 1e-10:
                        f_trial /= st
                        obj = fast_convolve(f_trial).max()
                        if obj < best_obj - 1e-15:
                            best_obj = obj
                            best_f = f_trial.copy()
                            f = f_trial.copy()
                            improved = True
                            break

        if not improved:
            stuck += 1
        else:
            stuck = 0

        if it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} obj={best_obj:.12f} "
                  f"score={2.0*n*best_obj:.6f} t={elapsed:.0f}s")

    return best_f, best_obj


def coordinate_descent(f_init, time_limit, label="coord"):
    """
    Coordinate-wise optimization: for each coordinate i, find the optimal value
    given all others fixed, minimizing the peak of the auto-convolution.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-10:
        f = f / s
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    rng = np.random.default_rng(555)

    # Precompute which lags each coordinate affects
    # Coordinate i affects b[k] for k = i (through a[i]*a[0]) to k = 2i+n-2... 
    # Actually b[k] = sum_j a[j]*a[k-j], so a[i] appears in b[k] when j=i (b[k] += a[i]*a[k-i])
    # or k-j=i (b[k] += a[k-i]*a[i]). So a[i] appears in b[k] for all k where 0<=k-i<n and 0<=i<n.
    # That's k in [i, i+n-1] intersect [0, 2n-2], so k in [i, min(i+n-1, 2n-2)].

    while time.monotonic() - t0 < time_limit:
        it += 1
        improved_any = False

        # Random coordinate order
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            # Current value
            ai = f[i]
            if ai < 1e-15:
                continue

            # The peak depends on ai quadratically: b[k] = (stuff without a[i]) + a[i]*a[k-i] + a[k-i]*a[i] - a[i]^2 (if k=2i)
            # More precisely: b[k] = b0[k] + a[i]*(a[k-i] + a[k-i]) - a[i]^2 * delta_{k,2i}
            # where b0[k] is b[k] with a[i] set to 0
            # = b0[k] + 2*a[i]*a[k-i] - a[i]^2 * delta_{k,2i}
            # For k != 2i: b[k] is linear in a[i]
            # For k = 2i: b[k] is quadratic in a[i]

            # Find the optimal ai that minimizes max_k b[k]
            # b[k] = b0[k] + 2*ai*a[k-i] - ai^2 * delta(k,2i)
            # Let c[k] = b0[k] + 2*ai*a[k-i] for k != 2i
            # and c[2i] = b0[2i] + 2*ai*a[i] - ai^2 = b0[2i] + 2*a[i]^2 - a[i]^2 = b0[2i] + a[i]^2
            # Wait, that's circular. Let me redo.
            # 
            # Let f' = f with f'[i] = x (the variable). All other coordinates fixed.
            # b'[k] = sum_{j != i} f[j]*f'[k-j] + f'[i]*f'[k-i] + f'[k-i]*f'[i] - delta(k-i,i)*f'[i]^2
            # Hmm, this is getting complicated. Let me just do a simple line search.

            # Simple approach: try a few multipliers for a[i]
            current_peak = fast_convolve(f).max()

            best_x = ai
            best_val = current_peak
            for mult in [0.5, 0.8, 1.0, 1.2, 1.5, 2.0]:
                x = ai * mult
                f_trial = f.copy()
                # Maintain sum = 1: adjust another coordinate
                delta = x - ai
                if delta != 0:
                    # Move mass to a random other coordinate (or spread it)
                    # Find coordinate with smallest involvement in peak
                    f_trial[i] = x
                    # Redistribute
                    f_trial = np.maximum(f_trial, 0.0)
                    st = f_trial.sum()
                    if st > 1e-10:
                        f_trial /= st
                        obj = fast_convolve(f_trial).max()
                        if obj < best_val:
                            best_val = obj
                            best_x = x

            if best_x != ai and best_val < best_obj - 1e-15:
                f[i] = best_x
                f = np.maximum(f, 0.0)
                st = f.sum()
                if st > 1e-10:
                    f /= st
                best_obj = best_val
                best_f = f.copy()
                improved_any = True

        if not improved_any and it > 3:
            break

        if it % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} obj={best_obj:.12f} "
                  f"score={2.0*n*best_obj:.6f} t={elapsed:.0f}s")

    return best_f, best_obj


def flat_top_window(n):
    """Generate a normalized flat top window."""
    w = np.ones(n)
    a = [0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368]
    for i, ai in enumerate(a):
        w -= ai * np.cos(2 * np.pi * i * np.arange(n) / max(1, n - 1) - i * np.pi)
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def hann_window(n):
    """Generate a normalized Hann window."""
    w = 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n) / max(1, n)))
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def tukey_window(n, r=0.3):
    """Generate a normalized Tukey (tapered cosine) window."""
    n = int(n)
    w = np.ones(n)
    m = int(np.floor(r * (n - 1) / 2))
    if m > 0:
        for i in range(m):
            w[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (2 * m) - 1)))
            w[n - 1 - i] = w[i]
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main optimization: multi-strategy approach combining LP, LSE smoothed
    mirror descent, peak shaving, coordinate descent, and multi-length exploration.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1024) / 1024.0
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"No parent, using uniform n=1024, score={p_score:.8f}")

    best_seq = parent.copy()
    best_score = p_score
    n_parent = len(parent)
    print(f"Initial: n={n_parent}, score={best_score:.8f}")

    print(f"\n=== Phase 1: LP main (n={n_parent}) ===")

    if t1 > 30:

                                        top_k_max=250, label="lp_main")

            best_score = score1
            best_seq = f1.copy()

            print(f"  lp_main: {score1:.8f} (best remains {best_score:.8f})")

            print(f"  lse: {score2:.8f} (best remains {best_score:.8f})")

        else:
            print(f"  peak: {score4:.8f} (best remains {best_score:.8f})")

    # === Phase 5: LSE again (from improved point) ===
    print(f"\n=== Phase 5: LSE polish (n={len(best_seq)}, score={best_score:.8f}) ===")
