    """Linear auto-convolution via FFT."""
    n = len(a)

    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_A_matrix(f, active_indices, n):
    """Build K x n constraint matrix: A[k,i] = f[k-i] if 0<=k-i<n else 0."""

                ik = i + k
                if 0 <= ik < n:
                    val += y[ik]
                ik2 = i - k
                if 0 <= ik2 < n:
                    val += y[ik2]
                grad[i] += val

        # Normalize subgradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm

        # Update
        lr = lr_base / (it ** 0.5)
        f = project_simplex(y - lr * grad)
        y = f + (1.0 - theta_t) * (f - y)
        # Fix: standard Nesterov update
        y = (1.0 - theta_t) * f + theta_t * y * 0 + (1.0 + theta_t) * f - theta_t * y
        # Simpler: y = f + (theta_t - 1) * (prev_y - f) ... let me just use standard
        # y_new = f + (it/(it+2)) * (f - y_old)
        # Actually let's just use: y = f (no momentum) for simplicity, or proper Nesterov

        # Proper Nesterov:
        # y_t = f_t + (t-1)/(t+1) * (f_t - f_{t-1})
        # Let me just track prev_f
        if it > 1:
            y = f + ((it - 1.0) / (it + 1.0)) * (f - y)

        if it % 100 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_f = f.copy()
            if it % 500 == 0:
                print(f"    {label} it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} "
                      f"t={time.monotonic()-t0:.0f}s")

        if it > 20000:
            break

    return best_f, 2.0 * n * best_obj


def subgradient_fast(f_init, time_limit, label="sg"):
    """Fast subgradient with vectorized gradient computation."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s
    else:
        f = np.ones(n) / n

    y = f.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    n_top = min(50, 2 * n - 1)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(y)
        M = b.max()

        # Top peaks
        top_idx = np.argpartition(b, -n_top)[-n_top:]

        # Compute subgradient vectorized
        grad = np.zeros(n)
        for k in top_idx:
            # (grad_k)(i) = y[i+k] (if valid) + y[i-k] (if valid)
            lo1 = 0
            hi1 = min(n, n - k) if k < n else 0
            # y[i+k]: i in [0, n-k-1]
            if k < n:
                grad[:n-k] += y[k:]
            # y[i-k]: i in [k, n-1]
            if k > 0:
                grad[k:] += y[:n-k]

        # Nesterov: y = f + (it-1)/(it+1) * (f - y_prev)
        if it > 1:
            y = f + ((it - 1.0) / (it + 1.0)) * (f - y)

        # Update f
        lr = 0.02 / np.sqrt(it)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm
        f = project_simplex(y - lr * grad)

        if it % 200 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_f = f.copy()
            if it % 1000 == 0:
                print(f"    {label} it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} "
                      f"t={time.monotonic()-t0:.0f}s")

        if it > 30000:
            break

    return best_f, 2.0 * n * best_obj


def coordinate_descent(f_init, time_limit, label="cd"):
    """Coordinate descent with geometric search."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    multipliers = [0.5, 0.6, 0.7, 0.8, 0.9, 1.0, 1.1, 1.2, 1.4, 1.6, 2.0]
    no_pass = 0

    while time.monotonic() - t0 < time_limit:
        any_imp = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_f, 2.0 * n * best_obj
            if f[j] < 1e-14:
                continue

            best_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                if mult == 1.0:
                    continue
                trial = f.copy()
                trial[j] = f[j] * mult
                st = trial.sum()
                if st < 1e-12:
                    continue
                trial = trial / st
                obj = fast_convolve(trial).max()
                if obj < best_j_obj - 1e-14:
                    best_j_obj = obj
                    best_val = f[j] * mult

            if best_val != f[j]:
                f[j] = best_val
                st = f.sum()
                if st > 1e-12:
                    f = f / st
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    any_imp = True

        if not any_imp:
            no_pass += 1
            if no_pass >= 2:
                break
        else:
            no_pass = 0

    return best_f, 2.0 * n * best_obj


def interp_sequence(src, target_n):
    """Interpolate a sequence to a new length."""
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    Phase 1: Explore multiple n values with fast LP (55% time)
    Phase 2: Deep LP on best n (25% time)
    Phase 3: Subgradient + CD polish (20% time)
    """
    np.random.seed(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
    except Exception:
        pass

    if parent is not None:
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    else:
        parent = np.ones(512) / 512.0
        p_score = 2.0
        print(f"No parent, using uniform n=512")

    best_seq = parent.copy()
    best_score = p_score

    # ============================================================
    # Phase 1: Explore n values with fast LP
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 1: Multi-n LP exploration")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    t1 = rem * 0.55

    n_parent = len(parent)
    # Generate candidate n values
    n_candidates = []
    for factor in [0.6, 0.7, 0.8, 0.9, 1.0, 1.1, 1.25, 1.4, 1.6]:
        n_c = int(n_parent * factor)
        n_c = max(128, min(2500, n_c))
        if n_c not in n_candidates:
            n_candidates.append(n_c)
    n_candidates.sort()

    # Allocate more time to n values closer to parent
    weights = []
    for n_c in n_candidates:
        dist = abs(n_c - n_parent) / n_parent
        weights.append(1.0 / (1.0 + 5.0 * dist))
    w_sum = sum(weights)
    weights = [w / w_sum for w in weights]

    for i, n_trial in enumerate(n_candidates):
        rem = deadline - time.monotonic()
        if rem < 20:
            break
        alloc = t1 * weights[i] * 0.85
        alloc = max(alloc, 15.0)
        alloc = min(alloc, rem * 0.9)

        f_init = interp_sequence(best_seq, n_trial)
        print(f"  LP n={n_trial}, budget={alloc:.0f}s")
        f_res, lp_score = run_fast_lp(
            f_init, time_limit=alloc,
            top_k_init=10, top_k_max=40,
            label=f"n{n_trial}")

        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = f_res.copy()
            print(f"  NEW BEST [n={n_trial}]: {real_score:.6f}")

    # ============================================================
    # Phase 2: Deep LP on best n
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 2: Deep LP (best={best_score:.6f}, n={len(best_seq)})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    t2 = rem * 0.25

    if t2 > 30:
        print(f"  Deep LP n={len(best_seq)}, budget={t2:.0f}s")
        f_res, deep_score = run_fast_lp(
            best_seq, time_limit=t2,
            top_k_init=20, top_k_max=60,
            label="deep")

        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = f_res.copy()
            print(f"  NEW BEST [deep]: {real_score:.6f}")

    # ============================================================
    # Phase 3: Polish with subgradient + coordinate descent
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 3: Polish (best={best_score:.6f})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    t3_sg = rem * 0.12
    t3_cd = rem * 0.08

    if t3_sg > 10:
        print(f"  Subgradient n={len(best_seq)}, budget={t3_sg:.0f}s")
        f_res, sg_score = subgradient_fast(
            best_seq, time_limit=t3_sg, label="sg")
        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = f_res.copy()
            print(f"  NEW BEST [sg]: {real_score:.6f}")

    if t3_cd > 5:
        rem = deadline - time.monotonic()
        if rem > 5:
            print(f"  CD n={len(best_seq)}, budget={min(t3_cd, rem-5):.0f}s")
            f_res, cd_score = coordinate_descent(
                best_seq, time_limit=min(t3_cd, rem - 5), label="cd")
            real_score = float(evaluate_sequence(f_res.tolist()))
            if np.isfinite(real_score) and real_score < best_score:
                best_score = real_score
                best_seq = f_res.copy()
                print(f"  NEW BEST [cd]: {real_score:.6f}")

    # ============================================================
    # Phase 4: Final LP burst with remaining time
    # ============================================================
    rem = deadline - time.monotonic()
    t4 = rem * 0.7
    if t4 > 20:
        print(f"\n{'='*60}")
        print(f"Phase 4: Final LP (best={best_score:.6f}, budget={t4:.0f}s)")
        print(f"{'='*60}")
        f_res, final_score = run_fast_lp(
            best_seq, time_limit=t4,
            top_k_init=20, top_k_max=50,
            label="final")

        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = f_res.copy()
            print(f"  NEW BEST [final]: {real_score:.6f}")

    # ============================================================
    # Phase 5: Final CD polish
    # ============================================================
    rem = deadline - time.monotonic()
    if rem > 10:
        f_res, _ = coordinate_descent(best_seq, time_limit=rem - 5, label="final_cd")
        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = f_res.copy()
            print(f"  NEW BEST [final_cd]: {real_score:.6f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy() if parent is not None else np.ones(512) / 512.0
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
