def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def softmax_from_z(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:

    return e / s

    """O(n log n) gradient of (2n/T)*logsumexp(conv(softmax(z),softmax(z))/T) w.r.t. z."""
    a = softmax_from_z(z)
    n = len(a)

    Tb = b / T
