            if g_var.value is not None:
                g_val = np.array(g_var.value, dtype=np.float64)
                g_val = np.maximum(g_val, 0.0)
                if not np.any(np.isnan(g_val)) and not np.any(np.isinf(g_val)):
                    break
                g_val = None
        except Exception:
            continue

    if g_val is None:
        return None, M
    return g_val, g_val.sum()


def line_search_peak(f, direction, n_candidates=12):
    """Find best interpolation parameter t minimizing the peak."""
    best_t = 0.1
    best_obj = float('inf')
    
    for t in np.linspace(0.01, 0.5, n_candidates):
        f_trial = (1.0 - t) * f + t * direction
        f_trial = np.maximum(f_trial, 0.0)
        s = f_trial.sum()
        if s < 1e-10:
            continue
        f_trial = f_trial / s
        obj = fast_convolve(f_trial).max()
        if obj < best_obj:
            best_obj = obj
            best_t = t
    
    # Refine around best
    t_lo = max(0.0, best_t - 0.1)
    t_hi = min(1.0, best_t + 0.1)
    for t in np.linspace(t_lo, t_hi, 8):
        f_trial = (1.0 - t) * f + t * direction
        f_trial = np.maximum(f_trial, 0.0)
        s = f_trial.sum()
        if s < 1e-10:
            continue
        f_trial = f_trial / s
        obj = fast_convolve(f_trial).max()
        if obj < best_obj:
            best_obj = obj
            best_t = t
    
    return best_t


def lp_optimize(f_init, time_limit, top_k_init=30, top_k_max=200, label="lp"):
    """
    Main LP-based optimization loop with constraint generation.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    
    b = fast_convolve(f)
    best_obj = b.max()
    it = 0
    stuck = 0
    top_k = top_k_init

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        # Compute current convolution and find active lags
        b = fast_convolve(f)
        current_peak = b.max()

        # Get active lags (top peaks)
        if top_k < len(b):
            top_idx = np.argpartition(b, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(b))
        
        # Sort by value descending
        top_idx = top_idx[np.argsort(b[top_idx])[::-1]]
        active_lags = top_idx.tolist()

        M = current_peak

        # Solve LP
        g, sum_g = lp_solve(f, n, active_lags, M)

        if g is None or sum_g <= 1.0 + 1e-6:
            # Try with relaxed M
            M_relaxed = current_peak * 1.05
            g, sum_g = lp_solve(f, n, active_lags, M_relaxed)
            if g is None or sum_g <= 1.0 + 1e-6:
                # Try adding more constraints
                if top_k < top_k_max:
                    top_k = min(top_k * 2, top_k_max)
                stuck += 1
                if stuck > 30:
                    break
                continue

        direction = g / sum_g

        # Line search
        best_t = line_search_peak(f, direction)

        # Update
        f_new = (1.0 - best_t) * f + best_t * direction
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-10:
            stuck += 1
            continue
        f_new = f_new / s

        # Check improvement
        b_new = fast_convolve(f_new)
        obj_new = b_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            stuck = 0
        else:
            # Try smaller step
            best_t2 = best_t * 0.3
            f_try = (1.0 - best_t2) * f + best_t2 * direction
            f_try = np.maximum(f_try, 0.0)
            s2 = f_try.sum()
            if s2 > 1e-10:
                f_try = f_try / s2
                obj_try = fast_convolve(f_try).max()
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_f = f_try.copy()
                    f = f_try
                    stuck = 0
                else:
                    stuck += 1
            else:
                stuck += 1

        if it % 30 == 0:
            score = 2.0 * n * best_obj
            print(f"    {label} it={it} peak={best_obj:.10f} score={score:.8f} "
                  f"top_k={top_k} t={time.monotonic()-t0:.0f}s")

        # Gradually increase top_k
        if it % 50 == 0 and top_k < top_k_max:
            top_k = min(top_k + 10, top_k_max)

    return best_f, best_obj


def subgradient_optimize(f_init, time_limit, n_top=30, lr_init=0.0005, momentum=0.9):
    """
    Subgradient optimization with momentum and adaptive step size.
    Minimizes the smooth max of the auto-convolution.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy().astype(np.float64)
    v = np.zeros(n)
    lr = lr_init
    best_f = f.copy()
    
    b = fast_convolve(f)
    best_obj = b.max()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(123)

    while time.monotonic() - t0 < time_limit:
        # Batch of steps
        batch = min(50, max(10, int((time_limit - (time.monotonic() - t0)) * 3)))
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit:
                break
            it += 1

            b = fast_convolve(f)

            # Weight the top peaks with softmax
            if n_top < len(b):
                threshold_idx = np.argpartition(b, -n_top)[-n_top:]
                b_top = b[threshold_idx].copy()
            else:
                threshold_idx = np.arange(len(b))
                b_top = b.copy()

            # Softmax weights (temperature dependent on it)
            temp = 0.005 + 0.01 * np.exp(-it / 200.0)
            shifted = b_top - b_top.max()
            w = np.exp(shifted / temp)
            w_sum = w.sum()
            if w_sum < 1e-30:
                w = np.zeros(len(b_top))
                w[b_top.argmax()] = 1.0
            else:
                w = w / w_sum

            # Compute gradient: sum of weighted flipped sequences
            grad = np.zeros(n)
            for idx_i, k in enumerate(threshold_idx):
                wi = w[idx_i]
                if wi < 1e-12:
                    continue
                # grad[m] += wi * f[k-m] (if valid)
                m_lo = max(0, k - n + 1)
                m_hi = min(n - 1, k)
                for m in range(m_lo, m_hi + 1):
                    grad[m] += wi * f[k - m]

            # Momentum update
            v = momentum * v - lr * grad
            f_new = project_simplex(f + v)
            f = f_new

        # Periodic check
        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
            lr = min(lr * 1.02, 0.005)
        else:
            no_improve += 1
            if no_improve % 5 == 0:
                lr *= 0.9
            if no_improve > 20:
                # Restart with perturbation
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                v = np.zeros(n)
                no_improve = 0
                lr = lr_init

        if it % 1000 == 0:
            score = 2.0 * n * best_obj
            print(f"    sg it={it} peak={best_obj:.10f} score={score:.8f} "
                  f"lr={lr:.6f} t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def focused_peak_reduction(f, time_limit, n_peaks=5):
    """
    Directly reduce the top peaks by finding directions that decrease them.
    Uses a simple gradient-like approach on the peak values.
    """
    t0 = time.monotonic()
    n = len(f)
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    b = fast_convolve(f)
    best_obj = b.max()
    stuck = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(f)
        
        if n_peaks < len(b):
            top_idx = np.argpartition(b, -n_peaks)[-n_peaks:]
        else:
            top_idx = np.arange(len(b))

        # For each top peak, compute the "gradient" direction to reduce it
        # grad_k[m] = f[k-m] (the derivative of (f*f)[k] w.r.t. f[m])
        # We want to decrease (f*f)[k], so move in direction -grad_k
        # But we need to keep sum = 1 and f >= 0
        
        combined_grad = np.zeros(n)
        for k in top_idx:
            m_lo = max(0, k - n + 1)
            m_hi = min(n - 1, k)
            for m in range(m_lo, m_hi + 1):
                combined_grad[m] += f[k - m]

        if combined_grad.max() < 1e-15:
            break

        # Find the best alpha by trying several values
        best_alpha = None
        best_trial = best_obj
        for alpha in [0.001, 0.003, 0.005, 0.01, 0.02, 0.03, 0.05]:
            f_trial = f - alpha * combined_grad
            f_trial = project_simplex(f_trial)
            obj = fast_convolve(f_trial).max()
            if obj < best_trial - 1e-15:
                best_trial = obj
                best_alpha = alpha

        if best_alpha is None:
            stuck += 1
            if stuck > 50:
                break
            continue

        f = f - best_alpha * combined_grad
        f = project_simplex(f)
        obj = fast_convolve(f).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1

    return best_f, best_obj


def hann_window(n):
    """Normalized Hann window."""
    w = 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n) / max(1, n - 1)))
    w = np.maximum(w, 0.0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def flat_top_window(n):
    """Normalized flat-top window."""
    w = np.ones(n)
    a_coeffs = [0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368]
    for i, ai in enumerate(a_coeffs):
        w -= ai * np.cos(2.0 * np.pi * i * np.arange(n) / max(1, n - 1) - i * np.pi)
    w = np.maximum(w, 0.0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return f_new / s
