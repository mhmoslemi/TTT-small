So the uniform sequence gives score 2. To get below 2, we need to make b[n-1] < 1/n while keeping all other b[k] also < 1/n.

This means we need `sum_j a[j] * a[n-1-j] < 1/n` while `sum a[j] = 1`.

By Cauchy-Schwarz, `sum_j a[j] * a[n-1-j] >= (sum_j sqrt(a[j] * a[n-1-j]))^2 / n`... hmm, that's not directly useful.

Actually, `sum_j a[j] * a[n-1-j]` can be made small if `a[j]` and `a[n-1-j]` are "anti-correlated" (one is large when the other is small). This is possible if the sequence is non-symmetric.

For example, if `a[j]` is large for small j and small for large j (a "decaying" sequence), then `a[n-1-j]` is small for small j and large for large j. The product `a[j] * a[n-1-j]` would be small for all j.

But then other lags might be large. For example, for lag k close to 0, `b[k] ≈ a[k] * a[0]` which could be large.

So the optimization is a trade-off: reduce the center lag at the cost of increasing edge lags, and find the balance that minimizes the maximum.

This is a classic "equiripple" problem, similar to minimax polynomial approximation (Chebyshev polynomials). The optimal solution likely has the property that several lags are all at the same maximum value (equiripple).

def fast_convolve(a):

    n = len(a)

    A = rfft(a, fft_size)

def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_constraint_matrix(f, n, peak_indices):
    """Build the A matrix for the LP: row k has f[lo:hi+1] at appropriate columns."""
    K = len(peak_indices)
    A = np.zeros((K, n))
    for row_i in range(K):
        k = int(peak_indices[row_i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp_direction(f, n, peak_indices, M, eps=1e-7):
    """Solve: maximize sum(g) s.t. g >= 0, A@g <= M. Returns g or None."""
    K = len(peak_indices)
    if K == 0:
        return None
    A = build_constraint_matrix(f, n, peak_indices)
    
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    
    g_val = None
    for solver in [cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, verbose=False)

def mirror_descent_step(f, b, lr, n_top, temp, rng=None):
    """One mirror descent step. Returns updated f."""
    n = len(f)
    n_b = len(b)
    M = b.max()
    
    # Softmax weights on peaks
    scaled = b / (temp * M)
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    # Focus on top peaks
    if n_top < n_b:
        top_idx = np.argpartition(b, -n_top)[-n_top:]
        w_sel = w[top_idx].copy()
        ws = w_sel.sum()
        if ws > 1e-30:
            w_sel /= ws
        w = np.zeros(n_b)
        w[top_idx] = w_sel
    
    # Compute subgradient: grad[i] = sum_k w[k] * a[k-i] (valid range)
    grad = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for k in active:
        lo = max(0, int(k) - n + 1)
        hi = min(n - 1, int(k))
        idx = np.arange(lo, hi + 1)
        grad[idx] += w[k] * f[k - idx]
    
    # Multiplicative update
    log_f = np.log(np.maximum(f, 1e-30))
    log_f_new = log_f - lr * grad
    log_f_new -= log_f_new.max()
    f_new = np.exp(log_f_new)
    s = f_new.sum()
    if s < 1e-30:
        return f
    f_new /= s
    return f_new

def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    n = len(f)
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

            print(f"  NEW BEST [lp_main]: {vs2:.6f}")
    
    # === Phase 3: Mirror descent polish ===
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.10, 120.0)
    if t3 > 20:
        n_b = len(best_seq)
        print(f"\n=== Phase 3: MD polish (n={n_b}, budget={t3:.0f}s) ===")
        f3, obj3 = md_polish(best_seq, time_limit=t3, label="md1")
        vs3 = float(evaluate_sequence(f3.tolist()))
        if np.isfinite(vs3) and vs3 < best_score:
            best_score = vs3
            best_seq = f3.copy()
            print(f"  NEW BEST [md1]: {vs3:.6f}")
    
    # === Phase 4: Multi-length exploration ===
    rem = deadline - time.monotonic()
    n_base = len(best_seq)
    target_lengths = []
    for factor in [0.88, 0.95, 1.05, 1.12, 1.25]:
        tl = int(n_base * factor)
        tl = max(512, min(3000, tl))
        if tl != n_base and tl not in target_lengths:
            target_lengths.append(tl)
    target_lengths.sort(key=lambda x: abs(x - n_base))
    
    for target_n in target_lengths[:4]:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t4 = min(rem * 0.10, 100.0)
        if t4 < 15:
            break
        f_interp = interpolate_to_length(best_seq, target_n)
        print(f"\n=== Phase 4: Hybrid n={target_n} (budget={t4:.0f}s) ===")
        f4, obj4 = run_hybrid_optimize(f_interp, time_limit=t4, label=f"hyb_n{target_n}",
                                        lp_every=8, peak_frac_start=0.93,
                                        peak_frac_end=0.82, max_peaks=300,
                                        md_lr=0.02, md_temp=0.003, n_top_md=60)
        vs4 = float(evaluate_sequence(f4.tolist()))
        if np.isfinite(vs4) and vs4 < best_score:
            best_score = vs4
            best_seq = f4.copy()
            print(f"  NEW BEST [n{target_n}]: {vs4:.6f}")
    
    # === Phase 5: Final LP burst ===
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.20, 150.0)
    if t5 > 15:
        n_b = len(best_seq)
        print(f"\n=== Phase 5: Final LP (n={n_b}, budget={t5:.0f}s) ===")
        f5, obj5 = run_pure_lp(best_seq, time_limit=t5, label="lp_final",
                                peak_frac=0.85, max_peaks=500)
        vs5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(vs5) and vs5 < best_score:
            best_score = vs5
            best_seq = f5.copy()
            print(f"  NEW BEST [lp_final]: {vs5:.6f}")
    
    # === Phase 6: Coordinate descent ===
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.15, 80.0)
    if t6 > 10:
        print(f"\n=== Phase 6: CD (budget={t6:.0f}s) ===")
        f6, obj6 = coordinate_descent(best_seq, time_limit=t6, label="cd")
        vs6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(vs6) and vs6 < best_score:
            best_score = vs6
            best_seq = f6.copy()
            print(f"  NEW BEST [cd]: {vs6:.6f}")
    
    # === Phase 7: Final short MD + LP ===
    rem = deadline - time.monotonic()
    if rem > 20:
        t7 = min(rem * 0.5, 60.0)
        if t7 > 10:
            n_b = len(best_seq)
            f7, obj7 = run_hybrid_optimize(best_seq, time_limit=t7, label="hyb_final",
                                            lp_every=5, peak_frac_start=0.90,
                                            peak_frac_end=0.80, max_peaks=400,
                                            md_lr=0.01, md_temp=0.002, n_top_md=50)
            vs7 = float(evaluate_sequence(f7.tolist()))
            if np.isfinite(vs7) and vs7 < best_score:
                best_score = vs7
                best_seq = f7.copy()
                print(f"  NEW BEST [hyb_final]: {vs7:.6f}")
        
        rem = deadline - time.monotonic()
        if rem > 10:
            f8, obj8 = coordinate_descent(best_seq, time_limit=min(rem * 0.5, 30.0), label="cd_final")
            vs8 = float(evaluate_sequence(f8.tolist()))
            if np.isfinite(vs8) and vs8 < best_score:
                best_score = vs8
                best_seq = f8.copy()
                print(f"  NEW BEST [cd_final]: {vs8:.6f}")
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
    
    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")
    
    return [float(x) for x in best_seq]
