def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build the convolution matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver, max_constraints=500, warm_start=None):
    """
    Solve the LP: maximize sum(b) subject to conv(a,b) <= M, b >= 0.
    Uses constraint reduction for efficiency.
    Returns the optimal b or None.
    """
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    
    # Start with a smart subset of constraints
    # Use all constraints initially, then reduce if too many
    if size <= max_constraints:
        constraints = [T @ b_var <= M]
    else:
        # Start with all, solve, then reduce to active
        constraints = [T @ b_var <= M]
    
    if warm_start is not None:
        b_var.value = warm_start
    
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    if b_var.value is None:
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    
    # Verify and potentially refine with active constraints
    if size > max_constraints:
        circ = T @ b_val
        tol = 1e-6 * M
        active = circ >= M - tol
        n_active = int(active.sum())
        
        if n_active < size and n_active >= 20:
            # Re-solve with just active constraints
            prob2 = cp.Problem(cp.Maximize(cp.sum(b_var)), [T[active, :] @ b_var <= M[active]])
            try:
                prob2.solve(solver=solver, verbose=False)
            except Exception:
                pass
            if prob2.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
                b_new = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
                # Verify the new solution satisfies all constraints
                circ_new = T @ b_new
                if circ_new.max() <= M * (1 + 1e-4):
                    b_val = b_new
    
    return b_val


def nesterov_lp(a0, time_limit, solver, verbose=True, gamma_init=2.0, max_constraints=500):
    """
    Fessler-Parks accelerated LP iteration.
    Minimizes max convolution with sum-normalized, non-negative a.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    
    y = a.copy()
    y_prev = a.copy()
    t = time.monotonic()
    n_iters = 0
    stall_count = 0
    gamma = gamma_init
    M_history = []
    
    while time.monotonic() - t < time_limit:
        n_iters += 1
        t_iter = n_iters
        
        M_curr = fast_autocorr(y).max()
        M_history.append(M_curr)
        if M_curr < 1e-14:
            break
        
        # Solve LP
        b = lp_best_response(y, M_curr, solver, max_constraints=max_constraints)
        if b is None:
            stall_count += 1
            if stall_count > 50:
                # Restart
                rng = np.random.default_rng(n_iters * 7 + 3)
                y = np.maximum(y + rng.normal(0, 0.02 * y.mean(), n), 0)
                y = y * norm / y.sum()
                y_prev = y.copy()
                stall_count = 0
                M_history = []
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        stall_count = 0
        
        # Normalize b to match
        g = b * norm / b_sum
        
        # Fessler-Parks update
        alpha = 2.0 / (t_iter + 2.0 - gamma)
        alpha = min(alpha, 0.5)
        
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        # Momentum
        if t_iter == 1:
            a_new = g.copy()
        else:
            momentum = (t_iter + 1.0 - gamma) / (t_iter + 2.0 - gamma)
            momentum = min(momentum, 1.0)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y_prev = y.copy()
        y = y_new
        a = a_new
        
        # Track best
        M_y = fast_autocorr(y).max()
        if M_y < best_M:
            best_M = M_y
            best_a = y.copy()
        
        M_a = fast_autocorr(a).max()
        if M_a < best_M:
            best_M = M_a
            best_a = a.copy()
        
        # Adaptive gamma
        if len(M_history) > 200:
            recent = M_history[-100:]
            diffs = np.diff(recent)
            osc = np.sum(np.abs(diffs)) / (100 * max(recent[0], 1e-15))
            if osc > 0.03:
                gamma = max(1.5, gamma - 0.1)
            elif osc < 0.01:
                gamma = min(2.0, gamma + 0.05)
        
        # Nesterov restart test
        if len(M_history) > 10:
            if M_history[-1] > M_history[-10] * 1.01:
                # Non-monotone, restart
                y = y_prev.copy()  # Go back
                y_prev = y.copy()
                M_history = M_history[-10:]
                gamma = max(1.5, gamma - 0.1)
        
        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t
            print(f"    LP-Neu iter={n_iters} ({elapsed:.0f}s): best_M={best_M:.8f}, M={M_curr:.6f}, gamma={gamma:.2f}")
    
    return best_a, best_M


def nesterov_subgradient(a0, time_limit, verbose=True, n_top=3):
    """
    Nesterov-accelerated subgradient method for minimizing max autocorrelation.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    
    y = a.copy()
    y_prev = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    L = best_M  # Lipschitz estimate
    
    while time.monotonic() - t0 < time_limit:
        n_iters += 1
        t_k = n_iters
        
        # Compute autocorrelation at y
        b = fast_autocorr(y)
        M_curr = b.max()
        
        # Find active peaks
        top_idx = np.argsort(b)[::-1][:n_top]
        top_vals = b[top_idx]
        
        # Compute subgradient (uniform over top peaks)
        # grad_j = sum_{p in top} 2 * a_{p-j} for valid j
        grad = np.zeros(n)
        for p in top_idx:
            js = max(0, p - n + 1)
            je = min(n - 1, p)
            if js <= je:
                grad[js:je+1] += 2.0 * y[p - np.arange(js, je+1)]
        
        # Normalize gradient
        grad /= n_top  # Average over peaks
        
        # Project gradient onto sum constraint (remove mean)
        grad -= np.mean(grad)
        
        # Step size
        step = 1.0 / (L * t_k)
        # Actually use Polyak-like: step = (M_curr - best_M) / ||grad||^2
        g_norm_sq = np.dot(grad, grad)
        if g_norm_sq > 1e-30:
            polyak_step = (M_curr - best_M * 0.999) / g_norm_sq
            step = max(step, min(1.0 / (n * 0.1), max(polyak_step, 1e-8 / n)))
        
        # Nesterov step
        y_new = np.maximum(y - step * grad, 0.0)
        s_new = y_new.sum()
        if s_new < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_new
        
        # Momentum
        if t_k == 1:
            a_new = y_new.copy()
        else:
            beta = (t_k - 1) / (t_k + 1)
            a_new = y_new + beta * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        y_prev = y.copy()
        y = y_new
        a = a_new
        
        # Track best
        M_y = fast_autocorr(y).max()
        if M_y < best_M:
            best_M = M_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        
        M_a = fast_autocorr(a).max()
        if M_a < best_M:
            best_M = M_a
            best_a = a.copy()
            stall = 0
        
        # Adapt L
        if stall > 50:
            L *= 1.1
            stall = 0
        
        if verbose and n_iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad-Neu iter={n_iters} ({elapsed:.0f}s): best_M={best_M:.8f}, M={M_curr:.6f}")
    
    return best_a, best_M

                    M_trial = fast_autocorr(a_trial).max()

def lse_smooth_descent(a0, time_limit, verbose=True, T_start=None, T_end=None):

def interpolated_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

def fast_autocorr(a):
    """Compute full autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

    """Safely evaluate a sequence."""

def build_conv_matrix(a):
    """Build the convolution matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

                pass
            if prob2.status in ('optimal', 'optimal_inaccurate') and b_var2.value is not None:
                b_new = np.maximum(np.asarray(b_var2.value, dtype=np.float64), 0.0)
                circ_new = T @ b_new
                if circ_new.max() <= M * (1 + 1e-3):
                    b_val = b_new

    return b_val


def nesterov_lp(a0, time_limit, solver, verbose=True, gamma_init=2.0, max_constraints=500):
    """
    Fessler-Parks accelerated LP iteration for minimizing max autocorrelation.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_M = fast_autocorr(a).max()

    y = a.copy()
    y_prev = a.copy()
    t_start = time.monotonic()
    n_iters = 0
    stall_count = 0
    gamma = gamma_init
    M_history = []
    last_improve_iter = 0

    while time.monotonic() - t_start < time_limit:
        n_iters += 1
        t_k = n_iters

        M_curr = fast_autocorr(y).max()
        M_history.append(M_curr)
        if M_curr < 1e-14:
            break

        # Keep M_history bounded
        if len(M_history) > 500:
            M_history = M_history[-250:]

        # Solve LP
        b = lp_best_response(y, M_curr, solver, max_constraints=max_constraints)
        if b is None:
            stall_count += 1
            if stall_count > 30:
                rng = np.random.default_rng(n_iters * 7 + 3)
                perturb = 0.02 * y.mean()
                y = np.maximum(y + rng.normal(0, perturb, n), 0)
                s_y = y.sum()
                if s_y < 1e-15:
                    y = a.copy()
                else:
                    y = y * norm / s_y
                y_prev = y.copy()
                stall_count = 0
                M_history = []
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            stall_count += 1
            continue
        stall_count = 0

        # Normalize b
        g = b * norm / b_sum

        # Fessler-Parks update
        alpha = 2.0 / (t_k + 2.0 - gamma)
        alpha = min(alpha, 0.5)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        # Momentum
        if t_k == 1:
            a_new = g.copy()
        else:
            momentum = (t_k + 1.0 - gamma) / (t_k + 2.0 - gamma)
            momentum = min(momentum, 1.0)
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y_prev = y.copy()
        y = y_new
        a = a_new

        # Track best
        M_y = fast_autocorr(y).max()
        if M_y < best_M:
            best_M = M_y
            best_a = y.copy()
            last_improve_iter = n_iters

        M_a = fast_autocorr(a).max()
        if M_a < best_M:
            best_M = M_a
            best_a = a.copy()
            last_improve_iter = n_iters

        # Adaptive gamma: reduce if oscillating, increase if smooth
        if len(M_history) > 100:
            recent = M_history[-50:]
            diffs = np.diff(recent)
            osc = np.sum(np.abs(diffs)) / (50 * max(recent[0], 1e-15))
            if osc > 0.05:
                gamma = max(1.2, gamma - 0.1)
            elif osc < 0.005 and gamma < 2.0:
                gamma = min(2.0, gamma + 0.02)

        # Nesterov restart: if non-monotone over last 20 iters
        if len(M_history) > 40:
            if M_history[-1] > max(M_history[-20:]) * 1.02:
                y = y_prev.copy()
                y_prev = y.copy()
                gamma = max(1.2, gamma - 0.1)
                M_history = M_history[-20:]

        # Long stall: bigger restart
        if n_iters - last_improve_iter > 500:
            rng = np.random.default_rng(n_iters * 13 + 7)
            perturb = 0.05 * y.mean()
            y = np.maximum(y + rng.normal(0, perturb, n), 0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y * norm / s_y
            y_prev = y.copy()
            a = np.maximum(y + rng.normal(0, perturb * 0.5, n), 0)
            s_a = a.sum()
            if s_a > 1e-15:
                a = a * norm / s_a
            last_improve_iter = n_iters
            gamma = max(1.2, gamma - 0.2)
            M_history = []

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    LP-Neu iter={n_iters} ({elapsed:.0f}s): best_M={best_M:.8f}, M={M_curr:.6f}, g={gamma:.2f}")

    return best_a, best_M


def lse_nesterov(a0, time_limit, verbose=True):
    """
    Nesterov-accelerated gradient descent on LogSumExp smoothing of max autocorrelation.
    Uses efficient top-peak only gradient computation.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_M = fast_autocorr(a).max()

    y = a.copy()
    y_prev = a.copy()
    t_start = time.monotonic()
    steps = 0

    T_start = max(0.05, best_M * 0.03)
    T_end = max(0.001, best_M * 0.002)

    while time.monotonic() - t_start < time_limit:
        steps += 1
        progress = min(1.0, (time.monotonic() - t_start) / time_limit)
        T = T_start * (T_end / T_start) ** progress

        b = fast_autocorr(y)
        M_max = b.max()
        if M_max < 1e-15:
            break

        # Compute gradient using top peaks (efficient when T is small)
        # Number of peaks to use scales with T
        n_peaks = min(int(4.0 / max(T, 0.001)) + 3, 2 * n - 1)
        n_peaks = min(n_peaks, 50)

        sorted_idx = np.argsort(b)[::-1][:n_peaks]
        exponents = (b[sorted_idx] - M_max) / T
        w_top = np.exp(exponents)
        w_top_sum = w_top.sum()
        if w_top_sum < 1e-30:
            break
        w_top = w_top / w_top_sum

        # Check if we're capturing enough weight
        all_exponents = (b - M_max) / T
        all_w_sum = np.exp(all_exponents).sum()
        captured_frac = w_top_sum / all_w_sum if all_w_sum > 0 else 1.0

        # Compute gradient from top peaks only
        grad = np.zeros(n)
        for i, p in enumerate(sorted_idx):
            wp = w_top[i]
            if wp < 1e-8:
                continue
            js = max(0, p - n + 1)
            je = min(n - 1, p)
            if js <= je:
                grad[js:je + 1] += 2.0 * wp * y[p - np.arange(js, je + 1)]

        # Scale by 1/T for the LSE gradient
        grad /= T

        # Project: remove mean for sum constraint
        grad -= np.mean(grad)

        # Adaptive step size
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-30:
            break
        # Use 1/L type step
        step = 0.5 / (g_norm * np.sqrt(n))
        step = min(step, 1.0 / n)

        # Nesterov step
        y_new = np.maximum(y - step * grad, 0.0)
        s_new = y_new.sum()
        if s_new < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_new

        # Momentum
        beta = (steps + np.sqrt(1 + 4 * steps * steps) - 1) / 2
        beta = beta / (steps + 1 + np.sqrt(1 + 4 * steps * steps))
        if steps == 1:
            a_new = y_new.copy()
        else:
            a_new = y_new + beta * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y_prev = y.copy()
        y = y_new
        a = a_new

        # Track best
        M_y = fast_autocorr(y).max()
        if M_y < best_M:
            best_M = M_y
            best_a = y.copy()

        M_a = fast_autocorr(a).max()
        if M_a < best_M:
            best_M = M_a
            best_a = a.copy()

        if verbose and steps % 200 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    LSE-Neu step={steps} ({elapsed:.0f}s): T={T:.4f}, best_M={best_M:.8f}")

    return best_a, best_M


def subgradient_nesterov(a0, time_limit, verbose=True, n_top=3):
    """
    Nesterov-accelerated subgradient method using top peaks.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_M = fast_autocorr(a).max()

    y = a.copy()
    y_prev = a.copy()
    t_start = time.monotonic()
    n_iters = 0
    stall = 0
    L = best_M / n

    while time.monotonic() - t_start < time_limit:
        n_iters += 1
        t_k = n_iters

        b = fast_autocorr(y)
        M_curr = b.max()

        # Top peaks
        top_idx = np.argsort(b)[::-1][:n_top]

        # Subgradient: average of gradients of active peaks
        grad = np.zeros(n)
        for p in top_idx:
            js = max(0, p - n + 1)
            je = min(n - 1, p)
            if js <= je:
                grad[js:je + 1] += 2.0 * y[p - np.arange(js, je + 1)]
        grad /= n_top

        # Project: remove mean
        grad -= np.mean(grad)

        # Step size: Polyak-like
        g_norm_sq = np.dot(grad, grad)
        if g_norm_sq < 1e-30:
            break
        gap = M_curr - best_M
        if gap > 0:
            polyak_step = gap / g_norm_sq
            step = min(1.0 / (L * t_k), max(polyak_step * 0.5, 1e-8 / n))
        else:
            step = 1.0 / (L * t_k)
        step = max(step, 1e-10 / n)

        # Nesterov step
        y_new = np.maximum(y - step * grad, 0.0)
        s_new = y_new.sum()
        if s_new < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_new

        if t_k == 1:
            a_new = y_new.copy()
        else:
            beta = (t_k - 1) / (t_k + 1)
            a_new = y_new + beta * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y_prev = y.copy()
        y = y_new
        a = a_new

        M_y = fast_autocorr(y).max()
        if M_y < best_M:
            best_M = M_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        M_a = fast_autocorr(a).max()
        if M_a < best_M:
            best_M = M_a
            best_a = a.copy()
            stall = 0

        # Adapt L
        if stall > 100:
            L *= 1.2
            stall = 0

        if verbose and n_iters % 300 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    SubGrad iter={n_iters} ({elapsed:.0f}s): best_M={best_M:.8f}")

    return best_a, best_M


def coordinate_polish(a0, time_limit, verbose=True, seed=42):
    """
    Coordinate-wise and pair-wise local optimization with simulated annealing.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    iters = 0
    n_improve = 0
    n_stall = 0

    while time.monotonic() - t_start < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t_start) / time_limit)
        iters += 1

        b = fast_autocorr(a)
        M_curr = b.max()
        p_star = int(np.argmax(b))

        # Choose move type
        move_type = rng.random()

        if move_type < 0.4:
            # Single coordinate adjustment
            # Pick coordinate contributing to max peak
            js = max(0, p_star - n + 1)
            je = min(n - 1, p_star)
            if js <= je and rng.random() < 0.7:
                idx = rng.integers(js, je + 1)
            else:
                idx = rng.integers(n)

            old_val = a[idx]
            if old_val < 1e-15:
                # Try setting it
                candidate_vals = [norm / n * 0.1, norm / n * 0.5, norm / n]
            else:
                candidate_vals = [old_val * 0.3, old_val * 0.6, old_val * 0.8,
                                  old_val * 1.2, old_val * 1.5, 0.0]

            best_trial_M = M_curr
            best_v = old_val
            for v in candidate_vals:
                if v < 0 or v > 1000:
                    continue
                a_trial = a.copy()
                a_trial[idx] = v
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_trial_M:
                    best_trial_M = M_trial
                    best_v = v

            if best_v != old_val and best_trial_M < M_curr:
                a[idx] = best_v
                a = a * norm / a.sum()
                M_new = fast_autocorr(a).max()
                if M_new < best_M:
                    best_M = M_new
                    best_a = a.copy()
                    n_improve += 1
                    n_stall = 0
                else:
                    n_stall += 1
            else:
                n_stall += 1

        elif move_type < 0.8:
            # Pair move: transfer mass from one coordinate to another
            i1 = rng.integers(n)
            # Prefer source with high value, destination far from i1
            i2 = rng.integers(n)
            if i1 == i2:
                n_stall += 1
                continue

            # Choose direction: from higher to lower value (or random)
            if a[i1] < a[i2] and rng.random() < 0.5:
                i1, i2 = i2, i1

            src_val = a[i1]
            if src_val < 1e-15:
                n_stall += 1
                continue

            for frac in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
                amt = src_val * frac
                a_trial = a.copy()
                a_trial[i1] -= amt
                a_trial[i2] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_M:
                    a = a_trial
                    best_M = M_trial
                    best_a = a.copy()
                    n_improve += 1
                    n_stall = 0
                    break
                elif M_trial < M_curr:
                    # Accept with annealing
                    T = best_M * 0.002 * (1.0 - elapsed_frac)
                    if rng.random() < np.exp(-(M_trial - best_M) / max(T, 1e-10)):
                        a = a_trial
                        n_stall += 1
                        break
            else:
                n_stall += 1

        else:
            # Multi-coordinate perturbation
            n_pert = rng.integers(3, max(4, n // 20))
            idx_pert = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for idx in idx_pert:
                amt = a[idx] * rng.uniform(0.1, 0.4)
                j = rng.integers(n)
                a_trial[idx] -= amt
                a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial * norm / s_t
                M_trial = fast_autocorr(a_trial).max()
                if M_trial < best_M:
                    a = a_trial
                    best_M = M_trial
                    best_a = a.copy()
                    n_improve += 1
                    n_stall = 0
                elif M_trial < best_M * 1.01:
                    T = best_M * 0.003 * (1.0 - elapsed_frac)
                    if rng.random() < np.exp(-(M_trial - best_M) / max(T, 1e-10)):
                        a = a_trial
                        n_stall += 1
                else:
                    n_stall += 1
            else:
                n_stall += 1

        # Big reset if very stuck
        if n_stall > 500:
            n_pert = max(10, n // 10)
            idx_pert = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for idx in idx_pert:
                amt = a[idx] * rng.uniform(0.2, 0.7)
                j = rng.integers(n)
                a_trial[idx] -= amt
                a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial * norm / s_t
                a = a_trial
            n_stall = 0

        if verbose and n_improve > 0 and n_improve % 500 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    CoordPolish improve={n_improve} ({elapsed:.0f}s): best_M={best_M:.8f}")

    return best_a, best_M


def interpolated_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving normalized shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function combining multiple optimization strategies:
    1. LP Fessler-Parks (primary workhorse with PDLP)
    2. LSE Nesterov for smoothing
    3. Subgradient Nesterov
    4. Multi-length exploration
    5. Coordinate polishing
    6. Final LP polish with many constraints
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, Solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.6f}")
    except Exception:
        n_base = 1600
        print("No parent found, using default init")

    def update_best(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize if needed
    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    # === Phase 1: LP Nesterov at base length (primary) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP-Neu n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.22, 380.0)
    if t1 > 30:
        n_cur = len(best_seq)
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t1, solver, verbose=True,
                               gamma_init=2.0, max_constraints=500)
        update_best(a_out, f"lp1_n{n_cur}")

    # === Phase 2: LSE Nesterov (smoothing transition) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: LSE-Neu n={n_cur} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.08, 100.0)
    if t2 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lse_nesterov(a_norm, t2, verbose=True)
        update_best(a_out, f"lse_n{n_cur}")

    # === Phase 3: LP Nesterov round 2 (finer constraints) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 3
