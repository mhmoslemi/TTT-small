    # Gradient: dLSE/da_j = sum_k w_k * db_k/da_j

    fft_size = 1 << (3 * n - 2).bit_length()

    A_fft = rfft(a_pad, fft_size)
    W_fft = rfft(w_pad, fft_size)
    cross = irfft(A_fft * np.conj(W_fft), fft_size)[:n]
    return 4.0 * cross  # factor 2 from each side of the product
