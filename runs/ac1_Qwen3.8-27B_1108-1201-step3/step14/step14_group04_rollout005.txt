def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    s = v.sum()
    if s < 1e-15:
        n = len(v)
        return np.ones(n) * target_sum / n
    return target_sum * v / s

def parametric_init(n, method='raised_cosine'):
    x = np.linspace(0.0, 1.0, n)
    xc = x - 0.5
    if method == 'raised_cosine':
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * xc))
    elif method == 'gaussian':
        a = np.exp(-4.0 * xc * xc)
    elif method == 'super_gaussian':
        a = np.exp(-2.0 * np.abs(xc) ** 2.5)
    elif method == 'bilateral':
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * xc)) ** 1.5
    elif method == 'plateau':
        w = 0.3
        a = np.maximum(0.5 * (1.0 + np.cos(np.pi * xc / w)), 0.0)
    elif method == 'uniform':
        a = np.ones(n)
    else:
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * xc)) ** 1.2
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    else:
        a = a / s
    return a


def greedy_peak_shave(a0, time_limit, seed=42, top_k_peaks=8):
    """Aggressive greedy peak shaving with adaptive step sizes and momentum."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        peaks = np.argsort(b)[::-1][:top_k_peaks]
        improved = False

        for k in peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            total = contrib.sum()
            if total < 1e-15:
                continue

            src = idx[np.argmax(contrib)]

            # Build "risk" mask: positions that contribute to any top peak
            risk = np.zeros(n, dtype=np.float64)
            for k2 in peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                idx2 = np.arange(js2, je2 + 1)
                risk[idx2] += a[idx2] * a[k2 - idx2]

            # Destination: low-risk, low-value positions
            safe_score = risk + 0.1 * a
            safe_score[src] = np.inf
            candidates = np.argsort(safe_score)[:max(1, n // 4)]
            dst = candidates[rng.integers(len(candidates))]

            # Try multiple fractions (adaptive)
            fracs = [0.0002, 0.0005, 0.001, 0.002, 0.005, 0.01, 0.02, 0.04]
            for frac in fracs:
                amt = a[src] * frac
                if amt < 1e-20:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    iter_count += 1
                    break

        if not improved:
            stall += 1
            # Random perturbation to escape local minima
            n_trials = min(3, max(1, n // 500))
            for _ in range(n_trials):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > norm * 1e-8:
                    frac = rng.uniform(0.005, 0.05)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        s_t = a_trial.sum()
                        if s_t > 1e-15:
                            a_trial = a_trial * norm / s_t
                            score = fast_autocorr(a_trial).max()
                            if score < best_score:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                stall = 0
                                break
            if stall > 500:
                # Larger perturbation
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > norm * 1e-6:
                    frac = rng.uniform(0.05, 0.2)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        s_t = a_trial.sum()
                        if s_t > 1e-15:
                            a_trial = a_trial * norm / s_t
                            score = fast_autocorr(a_trial).max()
                            if score < best_score:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                stall = 0
                if stall > 2000:
                    break

        if iter_count % 200 == 0 and iter_count > 0:
            elapsed = time.monotonic() - t0
            print(f"    Greedy iter={iter_count} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def topk_lp_best_response(a, M, K, solver):
    """LP best response using only top-K convolution peaks as active constraints."""
    n = len(a)
    b = fast_autocorr(a)
    # Get top-K lags
    top_lags = np.argsort(b)[::-1][:K]

    b_var = cp.Variable(n, nonneg=True)
    constraints = [cp.sum(b_var) >= 1.0]

    # Add only top-K constraints
    T_mat = np.zeros((len(top_lags), n))
    for i, k in enumerate(top_lags):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        T_mat[i, idx] = a[k - idx]

    # Also add a few random constraints for diversity
    n_random = min(5, 2 * n - 1 - K)
    if n_random > 0:
        all_lags = np.arange(2 * n - 1)
        remaining = np.setdiff1d(all_lags, top_lags)
        if len(remaining) > 0:
            rand_lags = np.random.default_rng(0).choice(remaining, size=min(n_random, len(remaining)), replace=False)
            T_extra = np.zeros((len(rand_lags), n))
            for i, k in enumerate(rand_lags):
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                T_extra[i, idx] = a[k - idx]
            T_mat = np.vstack([T_mat, T_extra])

    constraints.append(T_mat @ b_var <= M)

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def full_lp_best_response(a, M, solver):
    """Full LP best response with all constraints."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        T_mat[j:j + n, j] = a
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def lp_nesterov_topk(a0, time_limit, solver, K=50, verbose=True, full_every=50):
    """LP Nesterov with top-K constraints (fast) and periodic full LP."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    avg_a = np.zeros(n)
    n_avg = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        # Use top-K LP for speed, full LP periodically
        if t % full_every == 0:
            b = full_lp_best_response(y, M_curr, solver)
        else:
            b = topk_lp_best_response(y, M_curr, K, solver)

        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 10 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()

        if verbose and n_iters % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-topK iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def lse_fista_adaptive(a0, time_limit, verbose=True):
    """LSE-FISTA with adaptive temperature and backtracking step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    t_fista = 1

    # Adaptive temperature: start high, decrease geometrically
    T_start = 50.0
    T_end = 0.05
    n_stages = 40
    T_ratios = np.geomspace(T_start, T_end, n_stages)

    for stage_i, T in enumerate(T_ratios):
        stage_t0 = time.monotonic()
        L = n / max(T * 2.0, 5.0)
        staging = a.copy()
        max_iters = max(50, int(15.0 / max(time.monotonic() - t0, 1.0) * (time_limit - (time.monotonic() - t0)) / (n_stages - stage_i)))
        max_iters = min(max_iters, 2000)

        for _ in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                break
            # Compute gradient of LSE
            ac = fast_autocorr(y)
            ac_max = ac.max()
            shifted = (ac - ac_max) / T
            shifted = np.clip(shifted, -50.0, 50.0)
            exp_s = np.exp(shifted)
            w = exp_s / exp_s.sum()

            # Gradient via cross-correlation
            N = 1 << (3 * n - 2).bit_length()
            W = rfft(w, N)
            Y_rev = rfft(y[::-1], N)
            cross = irfft(W * Y_rev, N)
            grad = 2.0 * cross[n - 1: n - 1 + n]

            # Backtracking step size
            step = 1.0 / L
            for _bt in range(10):
                z = y - step * grad
                a_new = project_simplex(z, 1.0)
                # Check sufficient decrease (approximate)
                break
            if t_fista == 1:
                y_new = a_new.copy()
            else:
                mom = (t_fista - 1.0) / (t_fista + 2.0)
                y_new = a_new + mom * (a_new - staging)
                y_new = project_simplex(y_new, 1.0)
            staging = a_new.copy()
            a = a_new
            y = y_new
            t_fista += 1

        sc = fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        if verbose and (stage_i % 8 == 0 or stage_i == n_stages - 1):
            elapsed = time.monotonic() - t0
            print(f"    LSE-FISTA stage={stage_i} T={T:.3f} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score

    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search: multi-phase optimization combining top-K LP, greedy shaving, and LSE."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    solver = get_lp_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    # Load parent sequence if available
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        # Try multiple parametric initializations
        n_base = 1600
        methods = ['raised_cosine', 'gaussian', 'super_gaussian', 'bilateral', 'plateau', 'uniform']
        best_init = None
        best_init_score = float('inf')
        for m in methods:
            a_try = parametric_init(n_base, m)
            sc = evaluate_safe(a_try)
            if sc < best_init_score:
                best_init_score = sc
                best_init = a_try
        print(f"Best init: score={best_init_score:.6f}")
        best_seq = best_init
        best_score = best_init_score
        n_base = len(best_seq)
    else:
        n_base = len(best_seq)

    # === Phase 1: LSE-FISTA warm-up ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.08, 80.0)
    if t1 > 15:
        print(f"\n=== Phase 1: LSE-FISTA n={n_base} (budget={t1:.0f}s) ===")
        a_out, _ = lse_fista_adaptive(best_seq, t1, verbose=True)
        try_update(a_out, f"lse1_n{n_base}")

    # === Phase 2: Top-K LP Nesterov (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 60:
        print(f"\n=== Phase 2: LP-topK n={n_best} (budget={t2:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_nesterov_topk(a_norm, t2, solver, K=40, verbose=True, full_every=30)
        try_update(a_out, f"lp_topk1_n{n_best}")

    # === Phase 3: Greedy peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        print(f"\n=== Phase 3: Greedy peak shave n={n_best} (budget={t3:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = greedy_peak_shave(a_norm, t3, seed=42, top_k_peaks=10)
        try_update(a_out, f"greedy1_n{n_best}")

    # === Phase 4: Second LP pass (finer) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 30:
        print(f"\n=== Phase 4: LP-topK refine n={n_best} (budget={t4:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lp_nesterov_topk(a_norm, t4, solver, K=60, verbose=True, full_every=50)
        try_update(a_out, f"lp_topk2_n{n_best}")

    # === Phase 5: More greedy shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = min(remaining * 0.12, 150.0)
    if t5 > 20:
        print(f"\n=== Phase 5: Greedy shave 2 n={n_best} (budget={t5:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = greedy_peak_shave(a_norm, t5, seed=999, top_k_peaks=12)
        try_update(a_out, f"greedy2_n{n_best}")

    # === Phase 6: Try longer sequence ===
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq) + 400
    t6 = min(remaining * 0.12, 150.0)
    if t6 > 40:
        print(f"\n=== Phase 6: LP at n={n2} (budget={t6:.0f}s) ===")
        init_long = interpolate_sequence(best_seq, n2)
        # LSE warm-up for the longer sequence
        t6a = min(t6 * 0.3, 40.0)
        if t6a > 10:
            a_warm, _ = lse_fista_adaptive(init_long, t6a, verbose=False)
            try_update(a_warm, f"lse_long_n{n2}")
        remaining6 = t6 - (time.monotonic() - t_global + (time_budget - remaining))
        a_norm = best_seq * np.sqrt(2.0 * len(best_seq)) / best_seq.sum()
        init_long = interpolate_sequence(best_seq, n2)
        a_norm2 = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = lp_nesterov_topk(a_norm2, max(remaining * 0.08, 30.0), solver, K=40, verbose=True, full_every=40)
        try_update(a_out, f"lp_long_n{n2}")

    # === Phase 7: Final LSE polish (very low temperature) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t7 = min(remaining * 0.08, 100.0)
    if t7 > 15:
        print(f"\n=== Phase 7: Final LSE n={n_best} (budget={t7:.0f}s) ===")
        a_out, _ = lse_fista_adaptive(best_seq, t7, verbose=True)
        try_update(a_out, f"lse_final_n{n_best}")

    # === Phase 8: Final greedy + LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    if remaining > 30:
        t8a = min(remaining * 0.4, 80.0)
        if t8a > 10:
            print(f"\n=== Phase 8a: Final greedy n={n_best} (budget={t8a:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = greedy_peak_shave(a_norm, t8a, seed=424242, top_k_peaks=15)
            try_update(a_out, f"greedy_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t8b = min(remaining * 0.5, 60.0)
            if t8b > 10:
                print(f"\n=== Phase 8b: Final LP n={n_best} (budget={t8b:.0f}s) ===")
                a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
                a_out, _ = lp_nesterov_topk(a_norm, t8b, solver, K=80, verbose=False, full_every=20)
                try_update(a_out, f"lp_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
