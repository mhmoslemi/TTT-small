import cvxpy as cp

from scipy.fft import rfft, irfft

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_grad_softmax(a, T):
    """Compute gradient of logsumexp(conv(a,a)/T) w.r.t. a."""
    n = len(a)
    size = 2 * n - 1
    c = fast_autocorr(a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    p = exp_c / exp_c.sum()
    
    # grad[i] = 2 * sum_{m=0}^{n-1} p[i+m] * a[m]
    # = 2 * (p * reverse(a))[i+n-1]
    b = a[::-1]
    fft_size = 1 << (3 * n - 3).bit_length()
    P = rfft(p, fft_size)
    B = rfft(b, fft_size)
    conv_pb = irfft(P * B, fft_size)
    grad = 2.0 * conv_pb[n-1 : 2*n-1]
    return grad

def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T_mat = np.zeros((size, n))
    for j in range(n):
        T_mat[j:j + n, j] = a

def lp_best_response(a, M, solver):
    n = len(a)
    size = 2 * n - 1
    T_mat = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_mat @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def lag_guided_cd(a0, time_budget, rng, verbose=True):
    """
    Coordinate descent guided by which lags are peaked.
    For each peaked lag, find the best pair swap to reduce it.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0
    
    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    
    no_improve = 0
    n_passes = 0
    
    while time.monotonic() - t0 < time_budget:
        n_passes += 1
        pass_improved = False
        conv = fast_autocorr(a)
        max_val = conv.max()
        
        # Find top peaked lags
        top_k = 5
        peak_indices = np.argsort(conv)[-top_k:]
        peak_indices = peak_indices[np.argsort(conv[peak_indices])[::-1]]  # descending
        
        perm = rng.permutation(n)
        
        for idx in range(n):
            if time.monotonic() - t0 > time_budget:
                break
            
            i = perm[idx]
            if a[i] < norm * 1e-12:
                continue
            
            # For each peaked lag, find best j to swap with
            best_total_decrease = 0
            best_j = -1
            best_d = 0
            
            for k in peak_indices:
                # c_k = sum_m a[m] * a[k-m]
                # Moving delta from i to j changes c_k by:
                # delta * (partner_at_i - partner_at_j) * 2
                # where partner_at_i = a[k-i] if 0<=k-i<n else 0
                # and partner_at_j = a[k-j] if 0<=k-j<n else 0
                
                pi = a[k - i] if (0 <= k - i < n) else 0.0
                # We want j such that a[k-j] < pi, to decrease c_k
                # Candidate j's: those where a[k-j] is small
                # For efficiency, just try nearby j's
                for offset in range(1, 50):
                    for j in [(i + offset) % n, (i - offset) % n]:
                        if j == i:
                            continue
                        pj = a[k - j] if (0 <= k - j < n) else 0.0
                        # Potential decrease in c_k: 2 * delta * (pi - pj)
                        if pi - pj > 0:
                            # Max delta is a[i] (can't go below 0)
                            d_max = a[i] * 0.5  # conservative
                            # Actual decrease in c_k
                            dec_k = 2.0 * d_max * (pi - pj)
                            if dec_k > best_total_decrease:
                                best_total_decrease = dec_k
                                best_j = j
                                best_d = d_max
            
            if best_j >= 0 and best_d > 0:
                # Try the swap
                d = best_d
                a[i] -= d
                a[best_j] += d
                s = a.sum()
                if s < 1e-15:
                    a[i] += d
                    a[best_j] -= d
                    continue
                a = a * norm / s
                new_conv = fast_autocorr(a)
                new_max = new_conv.max()
                
                if new_max < best_score - 1e-13:
                    best_score = new_max
                    best_a = a.copy()
                    pass_improved = True
                    no_improve = 0
                    conv = new_conv
                else:
                    # Undo
                    a[best_j] -= d

Let me think about the computational complexity:

- For each j (1600 values), for each target lag k (5 values), compute `a[k-j]` (O(1))
- For each pair (i,j): gain = max_k (a[k-i] - a[k-j]) (O(5))
- Total: O(n * |K| + n^2 * |K|) = O(n^2 * |K|) for the full pass

For n=1600, |K|=5: about 12.8 million operations per pass. At ~1ns per operation, that's ~13ms per pass. Very fast!

But the issue is that after each swap, the autocorrelation changes, so we need to recompute. If we do one swap at a time and recompute the full autocorrelation, it's O(n log n) per swap, and we might do many swaps per pass.

A better approach: do a "greedy" pass where we compute the gain for all pairs, pick the best one, do the swap, update the autocorrelation incrementally, and repeat.

Incremental update of autocorrelation after swapping δ from i to j:
`c_new = c_old + 2δ * (e_i - e_j) * a` (in the convolution sense)

Where `e_i` is the indicator at position i. More precisely:
`c_new[k] = c_old[k] + 2δ * (a[k-i] - a[k-j])` (with bounds)
And also adjust the diagonal: `c_new[2i] -= 2δ*a[i] + δ^2` and `c_new[2j] += 2δ*a[j] + δ^2` (approximately, for small δ).

This is O(n) per swap (to update all c_k values), which is much better than O(n log n).

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_grad_softmax(a, T):
    n = len(a)
    size = 2 * n - 1
    c = fast_autocorr(a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    p = exp_c / exp_c.sum()
    b = a[::-1]
    fft_size = 1 << (3 * n - 3).bit_length()
    P = rfft(p, fft_size)
    B = rfft(b, fft_size)
    conv_pb = irfft(P * B, fft_size)
    grad = 2.0 * conv_pb[n - 1:2 * n - 1]
    return grad, c

                if 0 <= idx < n:

                    )
                if 2 * j_orig < size:

                    # Simpler: just compute the change directly

                        if conv.max() < best_score - 1e-13:
                            best_score = conv.max()
                            best_a = a.copy()
                            pass_improved = True
                            no_improve = 0

    n1 = n_base
    t1 = min(remaining * 0.30, 300.0)
    if t1 > 20:
        print(f"\n=== Phase 1: Softmax MW n={n1} (budget={t1:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n1) / best_seq.sum()
        a_out, _ = softmax_mw_descent(a_norm, t1, verbose=True)
        try_update(a_out, f"mw_n{n1}")

    # Phase 2: LP Nesterov (35% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq)
    t2 = min(remaining * 0.35, 400.0)
    if t2 > 30:
        print(f"\n=== Phase 2: LP Nesterov n={n2} (budget={t2:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n2) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t2, solver, verbose=True)
        try_update(a_out, f"lp_n{n2}")

    # Phase 3: Efficient pair CD (20% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n3 = len(best_seq)
    t3 = min(remaining * 0.20, 200.0)
    if t3 > 15:
        print(f"\n=== Phase 3: Pair CD n={n3} (budget={t3:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n3) / best_seq.sum()
        if n3 <= 2000:
            a_out, _ = efficient_pair_cd(a_norm, t3, rng, n_top_lags=5, verbose=True)
        else:
            a_out, _ = windowed_cd(a_norm, t3, rng, window=50, verbose=True)
        try_update(a_out, f"pcd_n{n3}")

    # Phase 4: Second LP Nesterov round (15% time)
    remaining = time_budget - (time.monotonic() - t_global)
    n4 = len(best_seq)
    t4 = min(remaining * 0.15, 150.0)
    if t4 > 20:
        print(f"\n=== Phase 4: LP Nesterov R2 n={n4} (budget={t4:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n4) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, verbose=True)
        try_update(a_out, f"lp2_n{n4}")

    # Phase 5: Final CD polish (remaining time)
    remaining = time_budget - (time.monotonic() - t_global)
    n5 = len(best_seq)
    if remaining > 15:
        print(f"\n=== Phase 5: Final CD n={n5} (budget={remaining:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n5) / best_seq.sum()
        if n5 <= 2000:
            a_out, _ = efficient_pair_cd(a_norm, min(remaining, 200.0), rng, n_top_lags=3, verbose=True)
        else:
            a_out, _ = windowed_cd(a_norm, min(remaining, 200.0), rng, window=30, verbose=True)
        try_update(a_out, f"pcd2_n{n5}")

    # Final evaluation
    if best_seq is not None:
        fs = _evaluate(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
