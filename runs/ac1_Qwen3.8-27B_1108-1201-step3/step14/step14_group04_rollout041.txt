def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum):
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(len(v)) * target_sum / len(v)
    return target_sum * v / s

def build_conv_matrix(a):
    """Build the matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2*n-1, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def lp_solve(a, M, solver, get_dual=False):
    """Solve: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T_mat = build_conv_matrix(a)
    constraint = T_mat @ b_var <= M
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [constraint])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return (None, None) if get_dual else None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return (None, None) if get_dual else None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return (None, None) if get_dual else None
    if get_dual:
        dual = constraint.dual_value
        if dual is not None:
            dual = np.maximum(np.asarray(dual, dtype=np.float64), 0.0)
        else:
            dual = np.zeros(2*n-1)
        return b_val, dual
    return b_val

def subgradient_max_conv(a):
    """Compute subgradient of max_k conv(a,a)[k] at a."""
    n = len(a)
    conv_a = fast_autocorr(a)
    M = conv_a.max()
    # Find active set (positions where conv is close to max)
    threshold = M * 0.999
    active = np.where(conv_a >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(conv_a)])
    # Compute subgradient: average of gradients at active positions
    grad = np.zeros(n)
    for k in active:
        for i in range(max(0, k-n+1), min(n, k+1)):
            grad[i] += 2.0 * a[k-i]
    grad /= len(active)
    return grad, M, active

def lse_gradient(a, T_temp):
    """Compute gradient of smoothed max: T*log(sum(exp(conv/T)))."""
    n = len(a)
    conv_a = fast_autocorr(a)
    M = conv_a.max()
    if M < 1e-15:
        return np.zeros(n)
    shifted = (conv_a - M) / T_temp
    shifted = np.clip(shifted, -50.0, 50.0)
    w = np.exp(shifted)
    w /= w.sum()
    # grad = sum_k w[k] * grad(conv[k])
    grad = np.zeros(n)
    for k in range(2*n-1):
        if w[k] > 1e-12:
            for i in range(max(0, k-n+1), min(n, k+1)):
                grad[i] += 2.0 * w[k] * a[k-i]
    return grad

    t0 = time.monotonic()

    for T in T_schedule:

            if t_fista == 1:
                y_new = a_new.copy()
            else:

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss
