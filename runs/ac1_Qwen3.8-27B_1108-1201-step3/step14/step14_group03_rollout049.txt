    no_improve = 0

        if n_passes % 100 == 0 and no_improve > 200:
            break
        conv = fast_autocorr(a)
        max_val = conv.max()

        improved_this_pass = False

            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-12:
                continue

            # Try moving mass from i to nearby positions
            offsets = [1, -1, 2, -2, 3, -3, 5, -5, 10, -10, 20, -20]
            for off in offsets:
                j = (i + off) % n
                if j == i or a[j] >= a[i]:
                    continue
                # Try a few fractions
                for frac in [0.3, 0.5, 0.7]:
                    d = a[i] * frac
                    a[i] -= d
                    a[j] += d
                    new_conv = fast_autocorr(a)
                    new_max = new_conv.max()
                    if new_max < best_score - 1e-13:
                        best_score = new_max
                        best_a = a.copy()
                        conv = new_conv
                        improved_this_pass = True
                        no_improve = 0
                        break
                    else:
                        a[i] += d
                        a[j] -= d

        if not improved_this_pass:
            no_improve += 1
            if no_improve >= 50:
                # Perturb and try again
                noise = rng.normal(0, norm * 0.005, n)
                a_pert = a + noise
                a_pert = np.maximum(a_pert, 0)
                s_pert = a_pert.sum()
                if s_pert > 1e-15:
                    a = a_pert * norm / s_pert
                no_improve = 0

        if verbose and n_passes % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    TCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

                break
            i = perm[idx]
            if a[i] < norm * 1e-10:
                continue

            for offset in range(1, window + 1):
                for j in [(i + offset) % n, (i - offset) % n]:
                    if j == i:
                        continue
                    if a[i] < norm * 1e-10:
                        break

                    # Compute delta in convolution
                    delta_vec = np.zeros(size)
                    lo1 = i
                    hi1 = min(size, i + n)
                    if hi1 > lo1:
                        delta_vec[lo1:hi1] += a[0:hi1 - lo1]
                    lo2 = max(0, n - 1 - i)
                    hi2 = min(size, n - 1 + i + 1)
                    if hi2 > lo2:
                        idx2 = np.arange(lo2, hi2)
                        src2 = n - 1 - idx2
                        valid = (src2 >= 0) & (src2 < n)
                        if valid.any():
                            delta_vec[lo2:hi2][valid] += a[src2[valid]]

                    lo3 = j
                    hi3 = min(size, j + n)
                    if hi3 > lo3:
                        delta_vec[lo3:hi3] -= a[0:hi3 - lo3]
                    lo4 = max(0, n - 1 - j)
                    hi4 = min(size, n - 1 + j + 1)
                    if hi4 > lo4:
                        idx4 = np.arange(lo4, hi4)
                        src4 = n - 1 - idx4
                        valid4 = (src4 >= 0) & (src4 < n)
                        if valid4.any():
                            delta_vec[lo4:hi4][valid4] -= a[src4[valid4]]

                    delta_vec *= 2.0

                    best_d = 0.0
                    best_val = conv.max()
                    n_samples = 10
                    deltas = np.linspace(0.0, a[i] * 0.95, n_samples)
                    for d in deltas:
                        if d <= 0:
                            continue
                        new_conv = conv + d * delta_vec
                        if 2 * i < size:
                            new_conv[2 * i] = conv[2 * i] - 2 * a[i] * d + d * d
                        if 2 * j < size:
                            new_conv[2 * j] = conv[2 * j] + 2 * a[j] * d + d * d
                        new_max = new_conv.max()
                        if new_max < best_val - 1e-13:
                            best_val = new_max
                            best_d = d

                    if best_d > 0:
                        a[i] -= best_d
                        a[j] += best_d
                        s = a.sum()
                        if s < 1e-15:
                            a[i] += best_d
                            a[j] -= best_d
                            continue
                        a = a * norm / s
                        conv = fast_autocorr(a)
                        cur_max = conv.max()
                        if cur_max < best_score - 1e-13:
                            best_score = cur_max
                            best_a = a.copy()
                            pass_improved = True
                            no_improve = 0

        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break

        if verbose and n_passes % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def random_restart_search(a0, time_limit, solver, n_restarts=3, verbose=True):
    """Try random perturbations + quick LP optimization."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    best_a = np.asarray(a0, dtype=np.float64).copy()
    best_score = fast_autocorr(best_a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(4242)
    time_per_restart = time_limit / max(1, n_restarts)

    for r in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break
        # Perturb
        scale = norm * 0.05 * (r + 1)
        noise = rng.laplace(0, scale, n)
        a_pert = best_a + noise
        a_pert = np.maximum(a_pert, 0)
        s = a_pert.sum()
        if s < 1e-15:
            continue
        a_pert = a_pert * norm / s

        # Quick LP optimization
        t_remaining = time_limit - (time.monotonic() - t0)
        t_sub = min(time_per_restart * 0.7, t_remaining * 0.8)
        if t_sub < 5:
            break
        a_out, sc = nesterov_lp_optimize(a_pert, t_sub, solver, verbose=False)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_out.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    RR restart {r}: improved to {best_score:.8f} ({elapsed:.0f}s)")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-9:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        n_base = 1600
        a_init = np.ones(n_base) / n_base
        best_seq = a_init.copy()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LP Nesterov (35% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n1 = n_base
    t1 = remaining * 0.35
    print(f"\n=== Phase 1: LP Nesterov n={n1} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n1) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n1}")

    # Phase 2: Mirror Descent (20% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t2 = remaining * 0.20
    print(f"\n=== Phase 2: Mirror Descent n={n_best} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = mirror_descent_subgrad(a_norm, t2, top_k=25, verbose=True)
        do_try_update(a_out, f"md_n{n_best}")

    # Phase 3: Targeted CD (15% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t3 = remaining * 0.15
    print(f"\n=== Phase 3: Targeted CD n={n_best} (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = targeted_cd(a_norm, t3, top_lags=8, verbose=True)
        do_try_update(a_out, f"tcd_n{n_best}")

    # Phase 4: LP Nesterov round 2 (20% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = remaining * 0.20
    print(f"\n=== Phase 4: LP Nesterov R2 n={n_best} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, verbose=True)
        do_try_update(a_out, f"lp2_n{n_best}")

    # Phase 5: Full CD polish (5% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t5 = remaining * 0.05
    print(f"\n=== Phase 5: Full CD n={n_best} (budget={t5:.0f}s) ===")
    if t5 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = full_cd_window(a_norm, t5, window=15, verbose=True)
        do_try_update(a_out, f"cd_n{n_best}")

    # Phase 6: Random restarts (5% of time)
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_best = len(best_seq)
        t6 = remaining * 0.5
        print(f"\n=== Phase 6: Random Restarts n={n_best} (budget={t6:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = random_restart_search(a_norm, t6, solver, n_restarts=2, verbose=True)
        do_try_update(a_out, f"rr_n{n_best}")

    # Phase 7: Final LP + CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        n_best = len(best_seq)
        t7a = min(remaining * 0.5, 60.0)
        t7b = remaining - t7a - 5
        if t7a > 10:
            print(f"\n=== Phase 7a: Final LP n={n_best} (budget={t7a:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = nesterov_lp_optimize(a_norm, t7a, solver, verbose=False)
            do_try_update(a_out, f"lp3_n{n_best}")
        remaining2 = time_budget - (time.monotonic() - t_global)
        if remaining2 > 10:
            print(f"\n=== Phase 7b: Final CD n={n_best} (budget={remaining2:.0f}s) ===")
            n_best = len(best_seq)
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = full_cd_window(a_norm, remaining2 * 0.8, window=25, verbose=False)
            do_try_update(a_out, f"cd3_n{n_best}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
