def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def build_active_rows(a, active_indices):
    """Build only the rows of the convolution matrix for active constraints.
    Returns a (m, n) matrix where m = len(active_indices)."""
    n = len(a)
    m = len(active_indices)
    T = np.zeros((m, n))
    for idx, k in enumerate(active_indices):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        T[idx, js:je+1] = a[k - np.arange(js, je+1)]
    return T


def lp_best_response_cutting_plane(a, M, solver, max_iters=8, max_active=150):
    """Solve the best response LP using a cutting-plane approach.
    
    Start with a subset of constraints, solve, check for violations,
    add violated constraints, repeat.
    """
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    
    # Compute current convolution to identify good initial constraints
    conv_ab = fast_autocorr(a)  # This is conv(a,a), we need conv(a,b) check later
    # Actually for the initial constraint set, use the lags where conv(a,a) is large
    sorted_lags = np.argsort(conv_ab)[::-1]
    
    # Start with top max_active lags
    active_set = sorted_lags[:max_active].copy()
    
    b_val = None
    for cp_iter in range(max_iters):
        # Build constraint matrix
        active_indices = active_set
        T = build_active_rows(a, active_indices)
        M_sub = M  # All constraints use the same M
        
        constraints = [T @ b_var <= M_sub]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            if b_val is None:
                return None
            break
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            if b_val is None:
                return None
            break
        if b_var.value is None:
            break
        
        b_new = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        s = b_new.sum()
        if s < 1e-15:
            if b_val is not None:
                b_val = b_new
            continue
        
        # Check if solution violates any constraints
        # conv(a, b_new)[k] for all k
        fft_size = 1 << (size - 1).bit_length()
        A = rfft(a, fft_size)
        B = rfft(b_new, fft_size)
        conv_ab = irfft(A * B.conj(), fft_size)[:size]
        
        # Actually for real signals, conv(a,b) via FFT is:
        # We need np.convolve(a, b) which is the same as irfft(rfft(a) * rfft(b))
        # Wait, np.convolve(a, b) is NOT the same as the circular convolution via FFT
        # unless we zero-pad. Let me use the correct formula.
        # np.convolve(a, b)[k] = sum_j a[j] * b[k-j]
        # This is the same as the cross-correlation of a and reverse(b)
        # Or equivalently, it's the convolution (not correlation)
        
        # Actually: np.convolve(a, b) = irfft(rfft(a, N) * rfft(b, N))[:len(a)+len(b)-1]
        # where N >= len(a)+len(b)-1
        # This is correct for linear convolution.
        
        # But wait - I computed A * B which gives the linear convolution of a and b.
        # That's exactly what we want: conv(a, b).
        
        # Check violations
        violations = conv_ab > M * (1 + 1e-7)
        n_viol = int(violations.sum())
        
        if n_viol == 0:
            b_val = b_new
            break
        
        # Add violated constraints (up to max_active total)
        viol_indices = np.where(violations)[0]
        # Sort by violation magnitude
        viol_vals = conv_ab[viol_indices]
        viol_sorted = viol_indices[np.argsort(viol_vals)[::-1]]
        
        # Merge with existing active set
        current_active = set(active_set.tolist())
        new_active = list(active_set)
        for vi in viol_sorted:
            if len(new_active) >= max_active:
                # Remove the least violated constraint
                current_conv = conv_ab[np.array(new_active)]
                min_idx = np.argmin(current_conv)
                removed = new_active[min_idx]
                current_active.discard(removed)
                new_active[min_idx] = vi
                current_active.add(vi)
            elif vi not in current_active:
                new_active.append(vi)
                current_active.add(vi)
        
        active_set = np.array(new_active)
        b_val = b_new
    
    if b_val is None or b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_lp(a0, time_limit, solver, verbose=True, max_active=150, max_cp_iters=6):
    """Fessler-Parks accelerated LP with cutting-plane best response."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    
    y = a.copy()
    prev_y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    gamma = 1.7
    M_history = []
    
    while time.monotonic() - t0 < time_limit:
        t = n_iters + 1
        M_curr = fast_autocorr(y).max()
        M_history.append(M_curr)
        
        if len(M_history) > 200:
            M_history = M_history[-100:]
        
        # Solve LP
        b = lp_best_response_cutting_plane(y, M_curr, solver, 
                                             max_iters=max_cp_iters, max_active=max_active)
        if b is None:
            stall += 1
            if stall > 50:
                # Restart
                rng = np.random.default_rng(n_iters * 7 + 3)
                perturb = 0.02 * y.mean()
                y = np.maximum(y + rng.normal(0, perturb, n), 0)
                y = y * norm / y.sum()
                a = np.maximum(a + rng.normal(0, perturb, n), 0)
                a = a * norm / a.sum()
                prev_y = y.copy()
                stall = 0
                gamma = 1.7
                M_history = []
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            stall += 1
            continue
        g = b * norm / b_sum
        n_iters += 1
        
        # Fessler-Parks step
        alpha = 2.0 / (t + 2.0 - gamma)
        alpha = min(alpha, 0.5)
        
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        # Momentum
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t + 1.0 - gamma) / (t + 2.0 - gamma)
            momentum = min(momentum, 1.0)
            a_new = y_new + momentum * (y_new - prev_y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        prev_y = y.copy()
        y = y_new
        a = a_new
        
        # Track best
        for candidate in [y, a]:
            sc = fast_autocorr(candidate).max()
            if sc < best_M:
                best_M = sc
                best_a = candidate.copy()
                stall = 0
        
        # Adaptive gamma
        if len(M_history) > 50:
            recent = M_history[-30:]
            if len(recent) > 2:
                diffs = np.abs(np.diff(recent))
                osc = diffs.mean() / max(recent[0], 1e-15)
                if osc > 0.03:
                    gamma = max(1.2, gamma - 0.05)
                elif osc < 0.005:
                    gamma = min(1.95, gamma + 0.02)
        
        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NL iter={n_iters} ({elapsed:.0f}s): M={M_curr:.6f}, best={best_M:.8f}, gamma={gamma:.2f}")
        
        # Restart
        if stall > 400:
            rng = np.random.default_rng(n_iters * 13 + 7)
            # Structured perturbation: change effective "width"
            spread = rng.uniform(0.8, 1.2)
            center = n / 2.0
            x = np.arange(n)
            weight = np.exp(-(x - center)**2 / (2 * (n * spread / 4)**2))
            perturbed = y * weight
            perturbed = np.maximum(perturbed + rng.normal(0, 0.01 * y.mean(), n), 0)
            perturbed = perturbed * norm / perturbed.sum()
            y = perturbed
            a = perturbed.copy()
            prev_y = y.copy()
            stall = 0
            gamma = 1.7
            M_history = []
            if verbose:
                print(f"    [Structured restart at iter {n_iters}]")
    
    return best_a, best_M


def polyak_subgradient(a0, time_limit, verbose=True, max_peaks=8):
    """Polyak subgradient method with known lower bound."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    # Lower bound on max(conv) when sum = norm
    # sum(conv) = (sum a)^2 = norm^2 = 2n
    # max(conv) >= 2n / (2n-1)
    M_lower = 2.0 * n / (2.0 * n - 1.0)
    
    t0 = time.monotonic()
    steps = 0
    step_size = 1.0 / np.sqrt(n)
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        
        b = fast_autocorr(a)
        max_b = b.max()
        
        if max_b <= M_lower * (1 + 1e-10):
            break
        
        # Select top peaks
        K = min(max_peaks, 2 * n - 1)
        # Use softmax weighting with adaptive temperature
        T = max_b * (0.01 + 0.1 * elapsed_frac)
        sorted_idx = np.argsort(b)[::-1][:K]
        peak_vals = b[sorted_idx]
        w = np.exp((peak_vals - max_b) / T)
        w = w / w.sum()
        
        # Compute subgradient: sum_k w[k] * 2 * shifted(a)
        grad = np.zeros(n)
        for m, k in enumerate(sorted_idx):
            if w[m] < 1e-8:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            # grad[i] += 2 * w[m] * a[k - i] for i in [js, je]
            # a[k - i] for i in [js..je] is a[k-je .. k-js]
            grad[js:je+1] += 2.0 * w[m] * a[k - np.arange(js, je+1)]
        
        # Project onto tangent space (sum = 0 in the subspace)
        grad -= grad.mean()
        
        g_norm_sq = np.dot(grad, grad)
        if g_norm_sq < 1e-30:
            break
        
        # Polyak step size
        polyak = (max_b - M_lower) / g_norm_sq
        # Decay step size with time
        decay = 1.0 / (1.0 + 0.01 * steps)
        step = min(step_size, polyak) * decay
        step = max(step, 1e-10)
        
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        score_new = fast_autocorr(a_new).max()
        
        if score_new < best_M:
            best_M = score_new
            best_a = a_new.copy()
            a = a_new
            step_size = min(step_size * 1.1, 5.0 / np.sqrt(n))
        elif score_new < max_b * 1.005:
            # Accept with probability based on improvement potential
            a = a_new
            step_size *= 0.9
        else:
            step_size *= 0.5
        
        steps += 1
        
        if verbose and steps % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PG step={steps} ({elapsed:.0f}s): M={max_b:.6f}, best={best_M:.8f}, step={step:.2e}")
        
        # Periodic restart
        if steps % 3000 == 0 and step_size < 1e-8:
            rng = np.random.default_rng(steps + seed_offset)
            a = np.maximum(a + rng.normal(0, 0.005 * a.mean(), n), 0)
            a = a * norm / a.sum()
            step_size = 1.0 / np.sqrt(n)
    
    return best_a, best_M

def peak_shave(a0, time_limit, seed=42, verbose=True):
    """Targeted peak shaving with quadratic line search."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_imp = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        p_star = int(np.argmax(b))
        
        # Find contributing positions
        js = max(0, p_star - n + 1)
        je = min(n - 1, p_star)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[p_star - idx]
        
        if contrib.max() < 1e-20:
            stall += 1
            if stall > 200:
                break
            continue
        
        # Source: position with max contribution to peak
        src = idx[np.argmax(contrib)]
        
        # Destinations: positions NOT contributing to the peak
        # Position j contributes to peak p if max(0, p-n+1) <= j <= min(n-1, p)
        non_contrib = np.ones(n, dtype=bool)
        non_contrib[js:je+1] = False
        non_contrib[src] = True  # exclude source from dest
        
        # Also exclude positions that contribute to other high peaks
        top5_peaks = np.argsort(b)[::-1][:5]
        for p in top5_peaks[1:]:
            p_js = max(0, p - n + 1)
            p_je = min(n - 1, p)
            non_contrib[p_js:p_je+1] = False  # Wait, this is wrong logic
            # Actually we want positions that DON'T contribute to ANY top peak
            # Let me redo this
        
        # Better: find positions that minimize contribution to ALL top peaks
        top5_peaks = np.argsort(b)[::-1][:5]
        peak_positions = np.zeros(n)
        for p in top5_peaks:
            p_js = max(0, p - n + 1)
            p_je = min(n - 1, p)
            for j in range(p_js, p_je + 1):
                peak_positions[j] += a[j] * a[p - j]
        
        # Destination: position with lowest total peak contribution and non-zero value
        candidates = np.where((np.arange(n) != src))[0]
        # Prefer positions with low peak contribution
        cand_scores = peak_positions[candidates]
        top_cands = candidates[np.argsort(cand_scores)[:20]]
        
        if len(top_cands) == 0:
            stall += 1
            continue
        
        dst = top_cands[rng.integers(len(top_cands))]
        
        # Line search: try multiple transfer fractions
        src_val = a[src]
        best_trial_M = max_b
        best_frac = 0
        
        fracs = [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]
        for frac in fracs:
            amt = src_val * frac
            if amt < 1e-20:
                break
            a_t = a.copy()
            a_t[src] -= amt
            a_t[dst] += amt
            a_t = np.maximum(a_t, 0)
            s_t = a_t.sum()
            if s_t < 1e-15:
                continue
            a_t = a_t * norm / s_t
            sc = fast_autocorr(a_t).max()
            if sc < best_trial_M:
                best_trial_M = sc
                best_frac = frac
        
        if best_frac > 0:
            a_t = a.copy()
            amt = a_t[src] * best_frac
            a_t[src] -= amt
            a_t[dst] += amt
            a_t = np.maximum(a_t, 0)
            a_t = a_t * norm / a_t.sum()
            sc = fast_autocorr(a_t).max()
            
            if sc < best_M:
                a = a_t
                best_M = sc
                best_a = a.copy()
                n_imp += 1
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        
        # Random exploration when stalled
        if stall > 30:
            n_pert = rng.integers(max(3, n // 100), max(5, n // 50))
            pert_mask = rng.choice(n, size=n_pert, replace=False)
            a_t = a.copy()
            for i in pert_mask:
                j = rng.integers(n)
                if i != j:
                    amt = a_t[i] * rng.uniform(0.05, 0.3)
                    a_t[i] -= amt
                    a_t[j] += amt
            a_t = np.maximum(a_t, 0)
            a_t = a_t * norm / a_t.sum()
            sc = fast_autocorr(a_t).max()
            if sc < best_M * 1.01:
                a = a_t
                if sc < best_M:
                    best_M = sc
                    best_a = a.copy()
                stall = 0
        
        if stall > 500:
            break
        
        if verbose and n_imp % 500 == 0 and n_imp > 0:
            elapsed = time.monotonic() - t0
            print(f"    PS imp={n_imp} ({elapsed:.0f}s): best={best_M:.8f}")
    
    return best_a, best_M


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    # Power law

    # Bimodal
    starts.append(np.exp(-((x - 0.3)**2) / 0.02) + np.exp(-((x - 0.7)**2) / 0.02))
    
    # Random smooth

    # Triangle

        ss = s.sum()
        if ss > 0:
            s = s / ss
