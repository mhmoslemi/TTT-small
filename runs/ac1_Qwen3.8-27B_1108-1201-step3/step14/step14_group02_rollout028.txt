        return np.ones_like(a) / len(a)
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def compute_gradient_at_a(a, k):
    """Compute gradient of conv_k(a) at point a.
    grad[j] = 2 * a[k-j] if 0 <= k-j < n, else 0."""
    n = len(a)
    grad = np.zeros(n)
    j_start = max(0, k - n + 1)
    j_end = min(n, k + 1)
    for j in range(j_start, j_end):
        idx = k - j
        if 0 <= idx < n:
            grad[j] = 2.0 * a[idx]
    return grad

def compute_soft_subgrad(a, ac, top_k=30, beta=3.0):
    """Soft subgradient using top-k active constraints with softmax weights."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    
    # Softmax weights with temperature
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n, k_int + 1)
        for j in range(j_start, j_end):
            idx = k_int - j
            if 0 <= idx < n:
                grad[j] += 2.0 * wi * a[idx]
    return grad

def compute_soft_subgrad_fast(a, ac, top_k=30, beta=3.0):
    """Vectorized soft subgradient computation."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum

    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n, k_int + 1)
        idxs = np.arange(j_start, j_end)
        partners = k_int - idxs
        grad[idxs] += 2.0 * wi * a[partners]
    return grad

def mirror_descent_optimize(a0, time_limit, top_k=30, beta=3.0, verbose=True):
    """Mirror descent with soft subgradient and adaptive step size."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Nesterov momentum
    y = a.copy()
    
    gamma = max(1.0, np.sqrt(n) * best_m)
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        step = gamma / np.sqrt(t)
        
        ac = fast_autocorr(y)
        g = compute_soft_subgrad_fast(y, ac, top_k=top_k, beta=beta)

        # Mirror step (exponential family / KL mirror)
        log_y = np.log(y + 1e-30)
        log_y_new = log_y - step * g
        max_val = log_y_new.max()
        a_new = np.exp(log_y_new - max_val)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new
        
        # Nesterov update for y
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * a_new
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y = y_new
        else:
            y = y_new / sy

        a = a_new
        M_new = fast_autocorr(a).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a.copy()

        # Also check Nesterov point
        My = fast_autocorr(y).max()
        if My < best_m:
            best_m = My
            best_a = y.copy()

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    MD t={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m

def lp_frank_wolfe(a0, time_limit, solver, verbose=True):
    """LP-based Frank-Wolfe with Nesterov acceleration."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    x = a.copy()
    iter_count = 0

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break

        # Build the LP: max sum(b) s.t. T @ b <= M_curr, b >= 0
        b_var = cp.Variable(n, nonneg=True)
        
        # Build T matrix efficiently
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j+n, j] = y
        
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g = b_val / b_sum

        # Standard FW step size
        alpha = 2.0 / (t + 2.0)
        
        # Nesterov momentum
        if t == 1:
            y_new = g.copy()
            x_new = g.copy()
        else:
            beta_mom = (t - 1.0) / (t + 2.0)
            y_new = (1.0 - alpha) * y + alpha * g
            y_new = np.maximum(y_new, 0.0)
            sy = y_new.sum()
            if sy < 1e-15:
                y_new = g.copy()
            else:
                y_new = y_new / sy
            
            x_new = y_new + beta_mom * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx

        y = y_new
        x = x_new
        iter_count += 1

        # Track best
        my = fast_autocorr(y).max()
        if my < best_m:
            best_m = my
            best_a = y.copy()

        mx = fast_autocorr(x).max()
        if mx < best_m:
            best_m = mx
            best_a = x.copy()

        if verbose and t % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_m
            print(f"    LP iter={t} ({elapsed:.1f}s): J={J:.6f}")

    return best_a, best_m

            if contrib.max() < 1e-20:
                continue

            src_local = np.argmax(contrib)
            src = idx[src_local]
            src_partner = partners[src_local]

            # Try moving mass from src (or src_partner) to reduce this peak
            # While also not increasing other peaks too much
            for move_from in [src, src_partner]:
                if a[move_from] < 1e-12:
                    continue
                
                # Find best destination
                # Score: for each candidate dst, compute the change in top peaks
                best_dst = -1
                best_delta = 0.0

                samples = rng.integers(0, n, n_samples)

                        continue

                    delta_max = 0.0
                    for k2 in top_peaks[:5]:
                        k2i = int(k2)
                        # Change in conv[k2] from moving epsilon from move_from to dst

                        # d(conv[k2])/d(a[dst]) = 2*a[k2-dst] (if valid)

                        break
                    else:

            i = rng.integers(n)
            j = rng.integers(n)

                M_new = fast_autocorr(a).max()

**Key improvement: Use exact gradient (not soft subgradient) with random tie-breaking**

The function f(a) = max_k conv_k(a) is piecewise quadratic. At a generic point, exactly one k achieves the max. The gradient is then 2*P_k(a) where P_k(a)_j = a_{k-j}.

The issue is that at the optimum, multiple k's might achieve the max (it's a "kink" point). In that case, we need a subgradient, which is any convex combination of the active gradients.

A good strategy: at each iteration, randomly select from the top-3 (or top-5) argmax indices and use that single gradient. This gives a valid subgradient and might escape degenerate situations.

def project_simplex(v, target_sum=1.0):
    """Project v onto the simplex {x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    # Scale to simplify
    u = v / target_sum
    # Project onto {x >= 0, sum(x) = 1}
    dv = np.sort(u)[::-1]
    cssv = np.cumsum(dv)
    rho = np.nonzero(dv * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(u - theta, 0)
    return w * target_sum

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def compute_gradient_at_a(a, k):
    """Compute gradient of conv_k(a) at point a. Vectorized."""
    n = len(a)
    grad = np.zeros(n)
    j_start = max(0, k - n + 1)
    j_end = min(n, k + 1)
    idxs = np.arange(j_start, j_end)
    partners = k - idxs
    grad[idxs] = 2.0 * a[partners]
    return grad

def compute_soft_subgrad_fast(a, ac, top_k=30, beta=3.0):
    """Vectorized soft subgradient."""
    n = len(a)
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(ac)[::-1][:k_max]
    top_vals = np.maximum(ac[top_idx], 1e-30)
    w = top_vals ** beta
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(k_max) / k_max
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        k_int = int(k)
        j_start = max(0, k_int - n + 1)
        j_end = min(n, k_int + 1)
        idxs = np.arange(j_start, j_end)
        partners = k_int - idxs
        grad[idxs] += 2.0 * wi * a[partners]
    return grad
