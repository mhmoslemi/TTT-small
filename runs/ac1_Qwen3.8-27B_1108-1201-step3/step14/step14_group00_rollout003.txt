def coordinate_descent(a0, time_limit, rng, verbose=True):

            if time.monotonic() - t0 > time_limit:

            if ai < 1e-12:

def interpolate_to(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    fi = np.interp(new_x, old_x, f)
    fi = np.maximum(fi, 0)
    si = fi.sum()
    if si < 1e-15:
        return np.ones(new_n) / new_n
