    y = a.copy()
    prev_a = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    
    # Start with all constraints
    active_set = None
    duals = None
    
    stall_count = 0
    stall_threshold = 80
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        
        # Solve LP
        b, active_set, duals = solve_lp_best_response(y, M_curr, solver, active_set)
        
        if b is None:
            # Retry with full constraints
            if t % 10 == 0:
                active_set = None
                b, active_set, duals = solve_lp_best_response(y, M_curr, solver, None)
            if b is None:
                continue
        
        # Check for violations of inactive constraints
        if active_set is not None and len(active_set) < size:
            inactive = np.setdiff1d(np.arange(size), active_set)
            if len(inactive) > 0:
                T_inactive = build_conv_matrix(y, inactive)
                violations = T_inactive @ b
                viol_mask = violations > M_curr * (1.0 + 1e-8)
                if np.any(viol_mask):
                    viol_idx = inactive[viol_mask]
                    active_set = np.union1d(active_set, viol_idx)
                    if len(active_set) > 0.8 * size:
                        active_set = None
                    b, active_set, duals = solve_lp_best_response(y, M_curr, solver, active_set)
                    if b is None:
                        continue
        
        # Update active set based on duals
        if duals is not None and len(duals) == len(active_set):
            # Keep constraints with significant dual values
            threshold = M_curr * 1e-4
            keep_mask = duals > threshold
            if np.sum(keep_mask) < max(10, int(0.05 * size)):
                keep_mask = duals > M_curr * 1e-6
            if np.sum(keep_mask) < 5:
                keep_mask = np.ones(len(active_set), dtype=bool)
            new_active = active_set[keep_mask]
            # Also always keep the most violated constraints
            conv_y = fast_autocorr(y)
            top_k = np.argsort(conv_y)[-max(20, int(0.02 * size)):][::-1]
            active_set = np.union1d(new_active, top_k)
            if len(active_set) > 0.7 * size:
                active_set = None
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        
        g = b * norm / b_sum
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.9)
            a_new = y_new + momentum * (y_new - prev_a)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        prev_a = y
        y = y_new
        a = a_new
        
        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        
        # Ergodic average
        avg_a += a
        n_avg += 1
        if n_avg % 5 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    stall_count = 0
        
        # Restart if stalled
        if stall_count > stall_threshold:
            y = best_a.copy()
            a = best_a.copy()
            prev_a = best_a.copy()
            avg_a = np.zeros(n)
            n_avg = 0
            active_set = None
            stall_count = 0
        
        if t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP t={t} ({elapsed:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score

def pair_cd(a0, time_limit, seed=0):
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    size = 2 * n - 1
    
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0
    
    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        pass_improved = False
        perm = rng.permutation(n)
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue
            
            # Try multiple random partners
            for attempt in range(6):
                j = int(rng.integers(n))
                if j == i:
                    continue
                
                # Compute change vector for moving delta from i to j
                # delta_a[i] -= d, delta_a[j] += d
                # Change in conv[k] = -d * (a shifted by i) + d * (a shifted by j)
                #                   + second order: -d^2 at 2i, +d^2 at 2j, +2*d*a[j]*... etc
                
                # First order: g[k] = -a_{k-i} + a_{k-j} (where valid)
                # Actually: (a+delta_a)*(a+delta_a) - a*a 
                # = a*delta_a + delta_a*a + delta_a*delta_a
                # delta_a has -d at i and +d at j
                # (a*delta_a)[k] = -d * a[k-i] + d * a[k-j]
                # (delta_a*a)[k] = -d * a[k-i] + d * a[k-j]  (same by symmetry of autocorrelation? No!)
                # Wait: (a*delta_a)[k] = sum_i a_i * delta_{k-i} = a[k-i]*(-d) + a[k-j]*(d)
                # (delta_a*a)[k] = sum_i delta_i * a[k-i] = (-d)*a[k-i] + d*a[k-j]
                # So (a*delta + delta*a)[k] = -2d*a[k-i] + 2d*a[k-j]
                # delta*delta: only non-zero at k=2i (value d^2), k=2j (value d^2), k=i+j (value -2d^2... wait)
                # Actually delta_a = -d*e_i + d*e_j
                # (delta_a * delta_a)[k] = delta_a[i]*delta_a[k-i] + ... 
                # = (-d)*(-d) if k-i=i, i.e., k=2i: d^2
                # = d*d if k-j=j, i.e., k=2j: d^2
                # = (-d)*d if k-i=j, i.e., k=i+j: -d^2
                # = d*(-d) if k-j=i, i.e., k=i+j: -d^2
                # So (delta*delta)[i+j] = -2d^2 (if i != j)
                # And (delta*delta)[2i] = d^2, (delta*delta)[2j] = d^2
                
                # Total change: conv_new[k] = conv[k] - 2d*(a[k-i] - a[k-j]) + (delta*delta)[k]
                # where a[k-i] means a[k-i] if 0<=k-i<n else 0
                
                # We want to minimize max_k conv_new[k]
                # For small d, the first-order term dominates
                # g[k] = -2*(a[k-i] - a[k-j])
                
                # Compute g efficiently
                g = np.zeros(size)
                lo_i = max(0, i)
                hi_i = min(size, i + n)
                if hi_i > lo_i:
                    src = np.arange(lo_i - i, hi_i - i)
                    g[lo_i:hi_i] -= 2.0 * a[src]
                
                lo_j = max(0, j)
                hi_j = min(size, j + n)
                if hi_j > lo_j:
                    src = np.arange(lo_j - j, hi_j - j)
                    g[lo_j:hi_j] += 2.0 * a[src]
                
                # Find optimal d using bisection on the maximin
                # We want the largest d such that max_k(conv[k] + d*g[k] + second_order[k]) < conv.max()
                # For simplicity, use a few candidate values
                candidates = [ai * f for f in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]]
                best_d = 0.0
                best_val = conv.max()
                
                for d in candidates:
                    if d <= 0 or d > ai:
                        continue
                    # First order
                    new_conv = conv + d * g
                    # Second order corrections
                    if 2 * i < size:
                        new_conv[2 * i] += d * d
                    if 2 * j < size:
                        new_conv[2 * j] += d * d
                    if i + j < size:
                        new_conv[i + j] -= 2.0 * d * d
                    
                    nm = new_conv.max()
                    if nm < best_val - 1e-13:
                        best_val = nm
                        best_d = d
                
                if best_d > 0:
                    a[i] -= best_d
                    a[j] += best_d
                    s = a.sum()
                    if s < 1e-15:
                        a[i] += best_d
                        a[j] -= best_d
                        continue
                    a = a * norm / s
                    conv = fast_autocorr(a)
                    cur_max = conv.max()
                    if cur_max < best_score - 1e-13:
                        best_score = cur_max
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0
                    else:
                        a[j] -= best_d
                        a[i] += best_d
                        a = a * norm / a.sum()
                        conv = fast_autocorr(a)
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0
    
    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

def perturb_seq(a, rng, scale=0.05):
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.1)
    a_pert = a * noise
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s
