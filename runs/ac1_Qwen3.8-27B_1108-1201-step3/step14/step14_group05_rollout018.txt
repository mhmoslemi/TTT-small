def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build the Toeplitz matrix T where T @ b = conv(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        start = j
        T[start:start+n, j] = a
    return T

def build_conv_matrix_sparse_indices(a):
    """Return COO format data for the convolution matrix."""
    n = len(a)
    size = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(n):
            rows.append(j + k)
            cols.append(j)
            vals.append(a[k])
    return np.array(rows), np.array(cols), np.array(vals)

def lp_solve_constr_gen(a, M, solver, max_iters=30, active_tol=1e-8):
    """
    LP best response with constraint generation.
    Maximizes sum(b) subject to conv(a,b) <= M, b >= 0.
    Uses cutting plane to iteratively add violated constraints.
    """
    n = len(a)
    size = 2 * n - 1
    
    # Build full matrix once (needed for violation checking)
    T = build_conv_matrix(a)
    
    b_var = cp.Variable(n, nonneg=True)
    
    # Start with a subset of constraints: peak and boundaries
    M_arr = np.full(size, M)
    
    # Initial active set: indices near the peak of conv(a,a)
    b_a = T @ np.ones(n)  # This is not quite right, let's just use all initially for small n
    # Actually, let's start with all constraints if n is small enough
    if n <= 500:
        active_idx = np.arange(size)
    else:
        # Start with top constraints
        conv_aa = fast_autocorr(a)
        # Pick lags where conv(a,a) is closest to M
        sorted_lags = np.argsort(-conv_aa)
        active_idx = sorted_lags[:min(200, size)]
    
    active_idx_list = list(active_idx)
    
    b_val = None
    for cg_iter in range(max_iters):
        constraints = [T[active_idx_list, :] @ b_var <= M_arr[active_idx_list]]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        if b_var.value is None:
            break
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        
        # Check for violated constraints
        if cg_iter < max_iters - 1:
            conv_ab = T @ b_val
            violated = conv_ab > M * (1 + active_tol)
            n_violated = int(violated.sum())
            
            if n_violated == 0:
                break
            
            # Add top violated constraints (up to 100 per iteration)
            violated_idx = np.where(violated)[0]
            if n_violated > 100:
                violated_order = np.argsort(-conv_ab[violated_idx])
                violated_idx = violated_idx[violated_order[:100]]
            
            # Merge with existing active set
            existing_set = set(active_idx_list)
            new_idx = [i for i in violated_idx if i not in existing_set]
            active_idx_list.extend(new_idx)
            
            # Cap total constraints
            if len(active_idx_list) > 500:
                # Keep the most constraint-binding ones
                conv_ab_full = T @ b_val
                all_idx = np.array(active_idx_list)
                ranked = all_idx[np.argsort(-conv_ab_full[all_idx])]
                active_idx_list = list(ranked[:500])
    
    if b_val is None:
        return None
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

def nesterov_lp(a0, time_limit, solver, verbose=True, max_active=200, restart_every=200):
    """
    Accelerated Nesterov LP iterations.
    a0: initial sequence (will be normalized internally)
    Returns (best_a, best_M) where best_M = max autocorrelation of best_a.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    
    y = a.copy()
    prev_y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    
    # Build the conv matrix once (for the fixed length)
    # We'll rebuild it each time y changes, but that's O(n^2)
    
    while time.monotonic() - t0 < time_limit:
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        
        # Solve LP
        b = lp_solve_constr_gen(y, M_curr, solver, max_iters=20)
        if b is None:
            # Fallback: use y itself
            b = y.copy()
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            b = y.copy()
            b_sum = b.sum()
        
        g = b * norm / b_sum
        n_iters += 1
        
        # Nesterov step
        t = float(n_iters)
        alpha = 2.0 / (t + 1.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
            s_y = y.sum()
        y_new = y_new * norm / s_y
        
        if n_iters == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 1.0)
            a_new = y_new + momentum * (y_new - prev_y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        prev_y = y.copy()
        y = y_new
        a = a_new
        
        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_M:
            best_M = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        
        score_a = fast_autocorr(a).max()
        if score_a < best_M:
            best_M = score_a
            best_a = a.copy()
            stall = 0
        
        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best_M={best_M:.8f}, stall={stall}")
        
        # Restart if stalled
        if stall > restart_every:
            rng = np.random.default_rng(n_iters * 7 + 3)
            pert = 0.05 * a.std()
            y = np.maximum(y + rng.normal(0, pert, n), 0)
            y = y * norm / y.sum()
            a = np.maximum(a + rng.normal(0, pert, n), 0)
            a = a * norm / a.sum()
            prev_y = y.copy()
            stall = 0
            if verbose:
                print(f"    [Restart at iter {n_iters}]")
    
    return best_a, best_M

def lse_adam(a0, time_limit, verbose=True):
    """LSE-smoothed optimization with Adam."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    
    T_start = max(0.5, best_score * 0.1)
    T_end = max(0.01, best_score * 0.005)
    
    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress
        
        b = fast_autocorr(a)
        max_b = b.max()
        
        # Compute LSE gradient
        w = np.exp((b - max_b) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum
        
        # Gradient: grad_j = 2 * sum_k w_k * a_{k-j} (cross-correlation of w with a)
        # conv(w, a_rev) gives us this
        a_rev = a[::-1]
        fft_size = 1 << (3 * n - 2).bit_length()
        W = rfft(w, fft_size)
        AR = rfft(a_rev, fft_size)
        conv_wa = irfft(W * AR, fft_size)
        grad = 2.0 * conv_wa[n - 1:2 * n - 1]
        
        # Project onto tangent space (sum = norm)
        grad -= np.mean(grad)
        
        # Adam step
        steps += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** steps)
        v_hat = v / (1 - beta2 ** steps)
        
        lr = 0.5 / (T * np.sqrt(n) + eps)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new
        
        a = a_new
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if verbose and steps % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-Adam step={steps} ({elapsed:.0f}s): T={T:.4f}, best={best_score:.8f}")
        
        # Occasional restart
        if steps % 2000 == 0:
            rng = np.random.default_rng(steps)
            a = np.maximum(a + rng.normal(0, 0.02 * a.std(), n), 0)
            a = a * norm / a.sum()
            m = np.zeros(n)
            v = np.zeros(n)
            steps = 0
    
    return best_a, best_score

def coordinate_peak_shave(a0, time_limit, verbose=True, seed=12345):
    """
    Direct coordinate-level peak shaving.
    Identifies the peak lag and redistributes mass from contributing coordinates.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)

    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    n_improve = 0
    n_stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        p_star = int(np.argmax(b))
        
        # Find contributing indices for peak at p_star
        js = max(0, p_star - n + 1)
        je = min(n - 1, p_star)
        idx = np.arange(js, je + 1)
        if len(idx) == 0:
            n_stall += 1
            if n_stall > max_stall:
                break
            continue
        
        contrib = a[idx] * a[p_star - idx]
        
        # Try moving mass from high-contributing source to low-contributing destination
        # Source: index with highest contribution to the peak
        src_local = int(np.argmax(contrib))
        src = int(idx[src_local])
        src_val = a[src]
        
        if src_val < 1e-15:
            n_stall += 1
            if n_stall > max_stall:
                break
            continue
        
        # Destination: prefer indices that don't contribute to the peak and have low value
        # Index j contributes to peak p if js <= j <= je
        non_contrib_mask = np.ones(n, dtype=bool)
        non_contrib_mask[js:je+1] = False
        non_contrib_mask[src] = True  # always allow src as potential (won't be chosen)
        
        dst_candidates = np.where(non_contrib_mask)[0]
        if len(dst_candidates) == 0:
            dst_candidates = np.array([i for i in range(n) if i != src])
        
        if len(dst_candidates) == 0:

        n_try = min(5, len(dst_candidates))
        # Prefer low-value destinations

        best_trial = None

                amt = src_val * frac

                s_t = a_trial.sum()
                if s_t < 1e-15:

                a_trial = a_trial * norm / s_t
                sc = fast_autocorr(a_trial).max()

                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial * norm / s_t
                    sc = fast_autocorr(a_trial).max()
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()

                    n_stall += 1

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function: combines LP Nesterov, LSE-Adam, and peak shaving
    with multi-length exploration and adaptive time allocation.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1720.0)
    
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            print(f"Parent loaded: n={n_base}, score={best_score:.6f}")
    except (NameError, Exception) as e:
        n_base = 1600
        print(f"No parent found ({e}), using default n={n_base}")
    
    # Initialize if needed
    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    
    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # === Phase 1: LP Nesterov at base length (primary) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t1 = min(remaining * 0.35, 550.0)
    print(f"\n=== Phase 1: LP Nesterov n={n_cur} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t1, solver, verbose=True, restart_every=150)
        do_try_update(a_out, f"lp1_n{n_cur}")
    
    # === Phase 2: LSE-Adam refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.15, 250.0)
    print(f"\n=== Phase 2: LSE-Adam n={n_cur} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lse_adam(a_norm, t2, verbose=True)
        do_try_update(a_out, f"lse_n{n_cur}")
    
    # === Phase 3: LP Nesterov round 2 (refined start) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.25, 400.0)
    print(f"\n=== Phase 3: LP Nesterov r2 n={n_cur} (budget={t3:.0f}s) ===")
    if t3 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t3, solver, verbose=True, restart_every=100)
        do_try_update(a_out, f"lp2_n{n_cur}")
    
    # === Phase 4: Try nearby lengths ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_base = len(best_seq)
    lengths_to_try = [n_base + 80, n_base - 50, n_base + 150]
    # Remove invalid lengths and deduplicate
    lengths_to_try = [l for l in lengths_to_try if l > 100 and l != n_base]
    lengths_to_try = list(dict.fromkeys(lengths_to_try))  # deduplicate preserving order
    
    for i, n_try in enumerate(lengths_to_try):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        t_alloc = min(remaining * 0.2, 150.0)
        if t_alloc < 30:
            break
        print(f"\n=== Phase 4.{i}: LP Nesterov n={n_try} (budget={t_alloc:.0f}s) ===")
        init_long = interpolate_sequence(best_seq, n_try)
        a_norm = init_long * np.sqrt(2.0 * n_try) / init_long.sum()
        a_out, _ = nesterov_lp(a_norm, t_alloc, solver, verbose=True, restart_every=80)
        do_try_update(a_out, f"lp_n{n_try}")
    
    # === Phase 5: Peak shaving on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t5 = min(remaining * 0.2, 200.0)
    print(f"\n=== Phase 5: PeakShave n={n_cur} (budget={t5:.0f}s) ===")
    if t5 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = coordinate_peak_shave(a_norm, t5, verbose=True, seed=77777)
        do_try_update(a_out, f"peak_n{n_cur}")
    
    # === Phase 6: Final LP polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t6 = min(remaining * 0.5, 200.0)
    print(f"\n=== Phase 6: Final LP n={n_cur} (budget={t6:.0f}s) ===")
    if t6 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp(a_norm, t6, solver, verbose=True, restart_every=50)
        do_try_update(a_out, f"lp_final_n{n_cur}")
    
    # === Phase 7: Final peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t7 = max(remaining * 0.8, 20.0)
    t7 = min(t7, 150.0)
    print(f"\n=== Phase 7: Final PeakShave n={n_cur} (budget={t7:.0f}s) ===")
    if t7 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = coordinate_peak_shave(a_norm, t7, verbose=True, seed=314159)
        do_try_update(a_out, f"peak_final_n{n_cur}")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
