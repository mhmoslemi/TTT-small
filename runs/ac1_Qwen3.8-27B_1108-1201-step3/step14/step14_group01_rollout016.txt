            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            
            # Gradient: grad_i = 2 * sum_k w[k] * (a[k-i] + a[i-k])
            W = rfft(w, fft_size)
            A = rfft(y, fft_size)
            
            # First term: sum_k w[k] * y[k-i] -> cross-correlation
            g1 = irfft(np.conj(A) * W, fft_size)[:n]
            # Second term: conv(y, w)[:n]
            g2 = irfft(A * W, fft_size)[:n]
            
            grad = 2.0 * (g1 + g2)
            
            # FISTA step
            t_k = (1.0 + np.sqrt(1.0 + 4.0 * k * k)) / 2.0
            beta = (t_k - 1.0) / t_k
            step = 1.0 / L
            
            # Proximal step (project onto simplex with non-negativity)
            z = y - step * grad
            z = np.maximum(z, 0.0)
            sz = z.sum()
            if sz > 1e-15:
                z = z * norm / sz
            else:
                z = y.copy()
            
            # FISTA extrapolation
            y_new = z + beta * (z - y)
            y_new = np.maximum(y_new, 0.0)
            sy = y_new.sum()
            if sy > 1e-15:
                y_new = y_new * norm / sy
            y = y_new
            k += 1
            
            # Check best
            sc = fast_conv_self(z).max()
            if sc < best_score:
                best_score = sc
                best_a = z.copy()
            
            # Adaptive Lipschitz (backtracking)
            if (it + 1) % 10 == 0:
                conv_z = fast_conv_self(z)
                # Estimate local Lipschitz
                if np.linalg.norm(grad) > 1e-15:
                    L = max(L * 0.9, 4.0 * norm / (T_val * np.sqrt(n)))
        
        if verbose:
            print(f"    FISTA stage {si} T={T_val:.2f}: best={best_score:.8f}")
    
    return best_a, best_score


def lp_nesterov_optimize(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration and adaptive screening."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    y = a.copy()
    no_improve = 0
    stagnant_score = best_score
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break
        
        # Adaptive constraint screening
        # Use a tighter threshold as we progress
        fraction = min(0.9, 0.5 + 0.4 * (t / min(t, 100)))
        threshold = M * fraction
        active = np.where(conv_y >= threshold)[0]
        
        if len(active) < 10:
            active = np.where(conv_y >= M * 0.3)[0]
        if len(active) > size:
            active = None
        
        b, dual_y = lp_best_response(y, M, solver, active)
        
        if b is None:
            b, dual_y = lp_best_response(y, M, solver, None)
        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        
        # Verify no dropped constraints are badly violated
        if active is not None and len(active) < size:
            check = fast_conv_cross(y, g)
            violated = np.any(check > M * (1 + 1e-3))
            if violated:
                b2, _ = lp_best_response(y, M, solver, None)
                if b2 is not None and b2.sum() > 1e-15:
                    g = b2 * norm / b2.sum()
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / sy
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            sa = a_new.sum()
            if sa < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / sa
        
        y = y_new
        a = a_new
        
        # Track best among y, a, and running average
        sc_y = fast_conv_self(y).max()
        sc_a = fast_conv_self(a).max()
        
        # Running average every 50 steps
        if t % 50 == 0:
            avg = (y + a) / 2.0
            avg = np.maximum(avg, 0.0)
            sa_avg = avg.sum()
            if sa_avg > 1e-15:
                avg = avg * norm / sa_avg
                sc_avg = fast_conv_self(avg).max()
                if sc_avg < best_score:
                    best_score = sc_avg
                    best_a = avg.copy()
        
        if min(sc_y, sc_a) < best_score:
            best_score = min(sc_y, sc_a)
            best_a = (y.copy() if sc_y < sc_a else a.copy())
            no_improve = 0
            stagnant_score = best_score
        else:
            no_improve += 1
        
        # Restart if stuck
        if no_improve == 200:
            y = best_a * (1.0 + 0.05 * np.random.randn(n))
            y = np.maximum(y, 0.0)
            sy = y.sum()
            if sy > 1e-15:
                y = y * norm / sy
            no_improve = 0
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M:.6f}")
    
    return best_a, best_score


def lp_multiplicative_weights(a0, time_limit, solver, verbose=True):
    """LP best-response with multiplicative weight (exponential weights) updates."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    eps = 1e-10
    no_improve = 0
    
    for t in range(1, 300000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_a = fast_conv_self(a)
        M = conv_a.max()
        if M < 1e-12:
            break
        
        # Constraint screening
        threshold = M * 0.7
        active = np.where(conv_a >= threshold)[0]
        if len(active) < 10:
            active = np.where(conv_a >= M * 0.4)[0]
        if len(active) > size:
            active = None
        
        b, _ = lp_best_response(a, M, solver, active)
        if b is None:
            b, _ = lp_best_response(a, M, solver, None)
        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = (b + eps) / (b_sum + eps * n)  # Smoothed direction
        
        # Multiplicative weight update
        eta = 1.0 / np.sqrt(t + 1.0)
        ratio = g / (a + eps)
        a_new = a * np.power(ratio, eta)
        a_new = np.maximum(a_new, 0.0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / sa
        a = a_new
        
        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Restart
        if no_improve == 300:
            a = best_a * (1.0 + 0.1 * np.random.randn(n))
            a = np.maximum(a, 0.0)
            sa = a.sum()
            if sa > 1e-15:
                a = a * norm / sa
            no_improve = 0
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-MW t={t} ({elapsed:.0f}s): best={best_score:.8f}, eta={eta:.4f}")
    
    return best_a, best_score


def dual_guided_gradient_descent(a0, time_limit, solver, verbose=True):
    """Use LP dual variables to guide a smoothed gradient step."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_a = fast_conv_self(a)
        M = conv_a.max()
        
        # Solve LP and get dual
        active = np.where(conv_a >= M * 0.6)[0]
        if len(active) < 10:
            active = None
        b, dual_y = lp_best_response(a, M, solver, active)
        
        if b is None:
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        
        # Also compute standard gradient from dual
        if dual_y is not None and dual_y.sum() > 1e-15:
            # dual_y gives weights for each lag
            w = dual_y / dual_y.sum()
            # Gradient of max-conv (smoother version)
            W = rfft(w, 1 << (size - 1).bit_length())
            A = rfft(a, 1 << (size - 1).bit_length())
            grad = 2.0 * (irfft(np.conj(A) * W, 1 << (size - 1).bit_length())[:n] +
                          irfft(A * W, 1 << (size - 1).bit_length())[:n])
            
            # Combine LP direction and dual gradient
            step = 0.3 / (t + 1)
            a_new = a + step * (g - a) - step * 0.5 * grad
            a_new = np.maximum(a_new, 0.0)
            sa = a_new.sum()
            if sa > 1e-15:
                a_new = a_new * norm / sa
        else:
            # Fallback: Nesterov-like
            alpha = 2.0 / (t + 2.0)
            a_new = (1.0 - alpha) * a + alpha * g
            a_new = np.maximum(a_new, 0.0)
            sa = a_new.sum()
            if sa > 1e-15:
                a_new = a_new * norm / sa
        
        a = a_new
        sc = fast_conv_self(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Dual-GD t={t} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent with vectorized delta search."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s
    
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False
        
        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue
            
            # Find good j candidates
            peak_lags = np.where(conv >= cur_max * 0.9)[0]
            
            j_candidates = list(rng.integers(n, size=6))
            for kl in peak_lags[:4]:
                j1 = int((kl - i) % n)
                j2 = int((kl + i) % n)
                if j1 != i:
                    j_candidates.append(j1)
                if j2 != i:
                    j_candidates.append(j2)
            
            best_delta = 0.0
            best_val = cur_max
            best_j = -1
            
            for j in set(j_candidates):
                if j == i:
                    continue
                aj = a[j]
                
                # Compute gain vector
                g = np.zeros(size)
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    g[lo:hi] += a[0:hi - lo]
                lo2 = max(0, j - n + 1)
                hi2 = min(size, j + 1)
                if hi2 > lo2:
                    g[lo2:hi2] += a[j - np.arange(lo2, hi2)]
                lo3 = max(0, i)
                hi3 = min(size, i + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[0:hi3 - lo3]
                lo4 = max(0, i - n + 1)
                hi4 = min(size, i + 1)
                if hi4 > lo4:
                    g[lo4:hi4] -= a[i - np.arange(lo4, hi4)]
                
                # Find optimal delta
                n_samples = 20
                deltas = np.linspace(0.01, 0.95, n_samples) * ai
                best_d = 0.0
                best_v = cur_max
                
                for d in deltas:
                    new_conv = conv + d * g
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * ai * d + d * d
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * aj * d + d * d
                    # Cross term at k = i + j
                    ij = i + j
                    if ij < size:
                        new_conv[ij] = conv[ij] + d * (a[i] + a[j] - a[i])  # approximate
                    v = new_conv.max()
                    if v < best_v:
                        best_v = v
                        best_d = d
                
                if best_d > 0 and best_v < best_val - 1e-14:
                    best_val = best_v
                    best_delta = best_d
                    best_j = j
            
            if best_j >= 0 and best_delta > 1e-15:
                a[i] -= best_delta
                a[best_j] += best_delta
                sa = a.sum()
                if sa > 1e-15:
                    a = a * norm / sa
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-14:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                        if total_improvements % 500 == 0 and verbose:
                            print(f"    PCD imp={total_improvements}: best={best_score:.8f}")
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()
        
        if not pass_improved:
            no_improve += 1
            if no_improve >= 8:
                break
            if no_improve == 4:
                n_pert = max(10, n // 30)
                idx_pert = rng.choice(n, size=n_pert, replace=False)
                a[idx_pert] *= (1.0 + 0.15 * rng.standard_normal(n_pert))
                a = np.maximum(a, 0.0)
                sa = a.sum()
                if sa > 1e-15:
                    a = a * norm / sa
        else:
            no_improve = 0
    
    return best_a, best_score


def structured_init(n, rng, style=None):
    """Generate diverse structured initializations."""
    if style is None:
        style = int(rng.integers(0, 10))
    x = np.linspace(0.0, 1.0, n)
    
    if style == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif style == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif style == 2:
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.15 * np.cos(10 * np.pi * x)
    elif style == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif style == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif style == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif style == 6:
        a = np.exp(-3.0 * x) + 0.5
    elif style == 7:
        a = 0.5 + 0.3 * np.sin(8 * np.pi * x) * np.cos(4 * np.pi * x)
    elif style == 8:
        # Multi-peak Gaussian
        n_peaks = int(rng.integers(3, 7))
        centers = sorted(rng.uniform(0.1, 0.9, size=n_peaks))
        a = 0.2 * np.ones(n)
        for c in centers:
            width = 0.05 + 0.05 * rng.random()
            a += np.exp(-(x - c) ** 2 / (2 * width ** 2))
    else:
        # Symmetric random
        half_n = n // 2
        a = np.ones(n) * 0.3
        a[:half_n] = np.abs(rng.standard_normal(half_n)) * 0.5 + 0.3
        a[half_n:] = a[:half_n][::-1]
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function. Returns best sequence found within time budget."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")
    
    best_seq = None
    best_score = float('inf')
    
    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng, style=0)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)
    
    # ========== Phase 1: LSE-FISTA (fast initial progress) ==========
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.15, 180.0)
    print(f"\n=== Phase 1: LSE-FISTA n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 20:
        a_work = best_seq.copy()
        a_out, _ = lse_fista(a_work, t1, verbose=False)
        try_update(a_out, f"fista_n{n_base}")
    
    # ========== Phase 2: LP Nesterov (main workhorse) ==========
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.25, 350.0)
    print(f"\n=== Phase 2: LP-Nesterov n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = lp_nesterov_optimize(best_seq, t2, solver, verbose=True)
        try_update(a_out, f"lp_nest_{len(best_seq)}")
    
    # ========== Phase 3: Pair CD polish ==========
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 150.0)
    print(f"\n=== Phase 3: Pair CD n={len(best_seq)} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = pair_cd_optimize(best_seq, t3, rng, verbose=True)
        try_update(a_out, f"pcd_{len(best_seq)}")
    
    # ========== Phase 4: LP Multiplicative Weights ==========
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.20, 250.0)
    print(f"\n=== Phase 4: LP-MW n={len(best_seq)} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_out, _ = lp_multiplicative_weights(best_seq, t4, solver, verbose=True)
        try_update(a_out, f"lp_mw_{len(best_seq)}")
    
    # ========== Phase 5: Second LP Nesterov from improved point ==========
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 5: LP-Nesterov round 2 n={len(best_seq)} (budget={t5:.0f}s) ===")
    if t5 > 30:
        a_out, _ = lp_nesterov_optimize(best_seq, t5, solver, verbose=False)
        try_update(a_out, f"lp2_{len(best_seq)}")
    
    # ========== Phase 6: Try longer length ==========
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_base + 400
    t6 = min(remaining * 0.20, 200.0)
    print(f"\n=== Phase 6: Longer n={n_long} (budget={t6:.0f}s) ===")
    if t6 > 40 and remaining > 80:
        init_long = interpolate_seq(best_seq, n_long)
        a_out, _ = lse_fista(init_long, min(t6 * 0.3, 50.0), verbose=False)
        try_update(a_out, f"fista_long_{n_long}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40:
            a_out, _ = lp_nesterov_optimize(best_seq if len(best_seq) == n_base else a_out,
                                             min(remaining * 0.5, 80.0), solver, verbose=False)
            try_update(a_out, f"lp_long_{n_long}")
    
    # ========== Phase 7: Final polish - alternate FISTA and LP ==========
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final polish n={len(best_seq)} (budget={remaining:.0f}s) ===")
    if remaining > 30:
        # Quick FISTA
        t_half = remaining * 0.3
        a_out, _ = lse_fista(best_seq, min(t_half, 60.0), verbose=False)
        try_update(a_out, "fista_final")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            # Quick LP
            a_out, _ = lp_nesterov_optimize(best_seq, min(remaining * 0.3, 60.0), solver, verbose=False)
            try_update(a_out, "lp_final")
        
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            # Final CD
            a_out, _ = pair_cd_optimize(best_seq, min(remaining * 0.5, 40.0), rng, verbose=False)
            try_update(a_out, "pcd_final")
    
    # ========== Phase 8: Try a completely different initialization ==========
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 100:
        print(f"\n=== Phase 8: Alternative init (budget={remaining:.0f}s) ===")
        for style in [2, 8, 7]:
            if remaining < 30:
                break
            n_alt = n_base
            a_alt = structured_init(n_alt, rng, style=style)
            t_sub = min(remaining * 0.3, 60.0)
            if t_sub > 15:
                a_out, _ = lse_fista(a_alt, t_sub, verbose=False)
                try_update(a_out, f"alt_style{style}_fista")
                remaining = time_budget - (time.monotonic() - t_global)
            if remaining > 30:
                t_sub2 = min(remaining * 0.3, 50.0)
                a_out, _ = lp_nesterov_optimize(best_seq, t_sub2, solver, verbose=False)
                try_update(a_out, f"alt_style{style}_lp")
                remaining = time_budget - (time.monotonic() - t_global)
    
    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
    
    return [float(x) for x in best_seq]
