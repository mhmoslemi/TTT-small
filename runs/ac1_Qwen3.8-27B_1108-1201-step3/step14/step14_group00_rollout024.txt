            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    try:
        a_list = [float(x) for x in np.asarray(a, dtype=np.float64)]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def build_conv_matrix_rows(a, k_indices):
    n = len(a)
    size = 2 * n - 1
    rows = np.zeros((len(k_indices), n))
    for row_idx, k in enumerate(k_indices):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if hi > lo:
            j_range = np.arange(lo, hi)
            idx_range = k - j_range
            valid = (idx_range >= 0) & (idx_range < n)
            if np.any(valid):
                rows[row_idx, j_range[valid]] = a[idx_range[valid]]
    return rows


def build_conv_matrix_full(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver, active_set=None):
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)

    if active_set is not None and len(active_set) > 0 and len(active_set) < size:
        T = build_conv_matrix_rows(a, active_set)
        constraints = [T @ b_var <= M]
    else:
        T = build_conv_matrix_full(a)
        constraints = [T @ b_var <= M]
        active_set = np.arange(size)

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, active_set

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, active_set

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None, active_set

    return b_val, active_set


def check_and_expand_active_set(a, b, M, active_set):
    n = len(a)
    size = 2 * n - 1
    complement = np.setdiff1d(np.arange(size), active_set)
    if len(complement) == 0:
        return None, active_set
    T_comp = build_conv_matrix_rows(a, complement)
    violations = T_comp @ b
    viol_idx = np.where(violations > M * (1 + 1e-6))[0]
    if len(viol_idx) > 0:
        new_active = np.union1d(active_set, complement[viol_idx])
        if len(new_active) > 0.9 * size:
            return None, None
        return b, new_active
    return None, active_set


def smooth_gradient(a, p):
    n = len(a)
    size = 2 * n - 1
    L = 1 << (2 * size).bit_length()

    A = np.zeros(L)
    A[:n] = a
    P = np.zeros(L)
    P[:size] = p

    FA = np.fft.fft(A)
    FP = np.fft.fft(P)

    term1 = np.fft.ifft(FP * np.conj(FA)).real[:n]
    term2 = np.fft.ifft(FA * np.conj(FP)).real[:n]
    return term1 + term2


def smooth_gradient_step(a, alpha, verbose=False, t_label=""):
    n = len(a)
    norm_val = np.sqrt(2.0 * n)
    conv = fast_autocorr(a)
    max_c = conv.max()
    if max_c < 1e-12:
        return a, max_c
    shifted = np.minimum(alpha * (conv - max_c), 0.0)
    p = np.exp(shifted)
    ps = p.sum()
    if ps < 1e-300:
        p = np.zeros(2 * n - 1)
        p[int(np.argmax(conv))] = 1.0
    else:
        p = p / ps

    g = smooth_gradient(a, p)
    g_norm = np.linalg.norm(g)
    if g_norm < 1e-15:
        return a, max_c

    g = g / g_norm
    lr = 0.3 / np.sqrt(1.0 + alpha * 0.1)
    a_new = a - lr * g
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a, max_c
    a_new = a_new * norm_val / s
    score = fast_autocorr(a_new).max()
    if verbose:
        print(f"      SM {t_label} alpha={alpha:.0f}: {score:.10f}")
    return a_new, score


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, seed=0,
                         active_tol_start=0.03, active_tol_end=0.005,
                         smooth_kick_interval=200):
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    size = 2 * n - 1

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    n_iters = 0
    t0 = time.monotonic()
    active_set = None
    stall_count = 0
    stall_threshold = 80

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        # Adaptive active set
        progress = min(1.0, t / 2000.0)
        tol = active_tol_start + (active_tol_end - active_tol_start) * progress
        if active_set is None or t % 80 == 0:
            conv_y = fast_autocorr(y)
            threshold = M_curr * (1.0 - tol)
            active_set = np.where(conv_y > threshold)[0]
            if len(active_set) < 3:
                active_set = np.where(conv_y > M_curr * 0.7)[0]

        b, active_set_new = lp_best_response(y, M_curr, solver, active_set)

        if b is not None and active_set_new is not None and len(active_set_new) < size:
            b_check, active_set_exp = check_and_expand_active_set(y, b, M_curr, active_set_new)
            if b_check is not None:
                b, active_set_new = lp_best_response(y, M_curr, solver, active_set_exp)

        if b is None:
            if t % 30 == 0:
                active_set = None
                b, active_set_new = lp_best_response(y, M_curr, solver, None)
            if b is None:
                continue

        active_set = active_set_new
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Nesterov update
        alpha_step = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha_step) * y + alpha_step * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.97)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        prev_y = y
        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        # Smooth gradient kick to escape local optima
        if smooth_kick_interval > 0 and t % smooth_kick_interval == 0:
            kick_alpha = 10.0 + 20.0 * (t // smooth_kick_interval)
            a_kicked, score_kicked = smooth_gradient_step(a, kick_alpha)
            if score_kicked < best_score:
                best_score = score_kicked
                best_a = a_kicked.copy()
                y = a_kicked.copy()
                a = a_kicked.copy()
                stall_count = 0
                active_set = None

        # Ergodic average check
        if t % 50 == 0 and t > 100:
            avg_candidate = (y + a) * 0.5
            s_avg = avg_candidate.sum()
            if s_avg > 1e-15:
                avg_candidate = avg_candidate * norm / s_avg
                sc_avg = fast_autocorr(avg_candidate).max()
                if sc_avg < best_score:
                    best_score = sc_avg
                    best_a = avg_candidate.copy()
                    stall_count = 0

        # Restart if stalled
        if stall_count > stall_threshold:
            y = best_a.copy()
            a = best_a.copy()
            active_set = None
            stall_count = 0
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    LP restart at iter {t} ({elapsed:.0f}s): best={best_score:.10f}")

        if verbose and n_iters % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={n_iters} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def pair_coordinate_descent(a0, time_limit, rng, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    n_passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            for attempt in range(3):
                j = int(rng.integers(n))
                if j == i:
                    continue

                # Compute gain vector for moving mass from i to j
                g = np.zeros(size)
                # Removing from i
                for k in range(max(0, i), min(size, i + n)):
                    m = k - i
                    g[k] -= a[m]
                for k in range(max(0, n - 1 - i), min(size, n + i)):
                    m = n - 1 - k + i
                    if 0 <= m < n:
                        g[k] -= a[m]
                # Adding to j
                for k in range(max(0, j), min(size, j + n)):
                    m = k - j
                    g[k] += a[m]
                for k in range(max(0, n - 1 - j), min(size, n + j)):
                    m = n - 1 - k + j
                    if 0 <= m < n:
                        g[k] += a[m]

                # Try candidate deltas
                candidates = [ai * 0.05, ai * 0.1, ai * 0.25, ai * 0.5]
                best_d = 0.0
                best_val = conv.max()

                for d in candidates:
                    if d <= 0 or d > ai:
                        continue
                    new_max = (conv + d * g).max()
                    di = 2 * i
                    if di < size:
                        new_max = max(new_max, conv[di] - 2 * a[i] * d + d * d)
                    dj = 2 * j
                    if dj < size:
                        new_max = max(new_max, conv[dj] + 2 * a[j] * d + d * d)
                    if new_max < best_val - 1e-14:
                        best_val = new_max
                        best_d = d

                if best_d > 0:
                    a[i] -= best_d
                    a[j] += best_d
                    s = a.sum()
                    if s < 1e-15:
                        a[i] += best_d
                        a[j] -= best_d
                        continue
                    a = a * norm / s
                    conv = fast_autocorr(a)
                    cur_max = conv.max()
                    if cur_max < best_score - 1e-14:
                        best_score = cur_max
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0
                    else:
                        a[i] += best_d
                        a[j] -= best_d
                        a = a * norm / a.sum()
                        conv = fast_autocorr(a)
                    break  # Only one successful move per (i, attempts)

        if not pass_improve if False else not pass_improved:
            no_improve += 1
            if no_improve >= 5:
                break

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def perturb_sequence(a, rng, scale=0.03):
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.05)
    a_pert = a * noise
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def structured_init(n, rng):
    choice = rng.integers(0, 6)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.sin(np.pi * x) ** 1.2
    elif choice == 1:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.8
    elif choice == 2:
        a = np.exp(-8.0 * (x - 0.35) ** 2) + 0.8 * np.exp(-8.0 * (x - 0.65) ** 2)
    elif choice == 3:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * x) * np.exp(-2.0 * x * (1 - x))
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2 * (1 + 0.15 * np.cos(6 * np.pi * x))
    else:
        a = np.sin(np.pi * x) ** 0.6 * (1 - 0.2 * np.cos(4 * np.pi * x))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
