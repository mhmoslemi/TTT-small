    # Actually, the standard result is: cross-corr(a,w)[k] = sum_n a[n+k]*conj(w[n]) = IFFT(conj(FFT(a))*FFT(w))
    # But I want sum_j a[j]*w[j+i], which is a different convention.
    # sum_j a[j]*w[j+i] = cross-corr(w,a)[i] (with the convention cross-corr(x,y)[k] = sum_n x[n+k]*y[n])
    # = IFFT(conj(FFT(w)) * FFT(a))
    # So: cross = IFFT(conj(FFT(w)) * FFT(a))
    
    cross = irfft(np.conj(W_fft) * A_fft, fft_size)[:n]
    
    # d(obj)/d(a_i) where obj = 2*n*lse/S^2 (for S fixed)
    # = 2*n/S^2 * d(lse)/d(a_i)
    # = 2*n/S^2 * 2 * cross[i]
    # But we also have the constraint sum(a) = S, so the effective gradient for the
    # constrained problem is: grad[i] - mean(grad)
    # (since we project onto the affine constraint sum(a)=S)
    
    grad = 2.0 * n / (S * S) * 2.0 * cross
    # For the constrained problem (sum(a) = S), the projected gradient is:
    # grad_proj = grad - (sum(a*grad)/S) * (1/S) ... 
    # Actually for projection onto {a: sum(a)=S, a>=0}, the linear part is:
    # grad_proj = grad - mean over the support of (grad - grad_bar) where grad_bar = sum(a*grad)/S
    # For simplicity in the inner iteration, I'll just use:
    grad_proj = grad - a * (grad @ a) / S
    # This is the projection of grad onto the tangent space of {sum(a)=S}
    # (treating it as a manifold constraint)
    
    return grad_proj

def lse_value(a, T, S):
    """Compute LSE-smoothed objective value."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    exp_args = T * (b - max_b)
    exp_vals = np.exp(exp_args)
    lse = max_b + np.log(exp_vals.sum()) / T
    return 2.0 * n * lse / (S * S)

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    # Binary search for the threshold
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho = np.where(v_sorted * np.arange(1, n+1) > (cssv - S))[0][-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    result = np.maximum(v - theta, 0.0)
    # Ensure exact sum
    result += (S - result.sum()) / n
    result = np.maximum(result, 0.0)
    return result

def bb_step(prev_a, curr_a, prev_grad, curr_grad):
    """Barzilai-Borwein step size."""
    s = curr_a - prev_a
    y = curr_grad - prev_grad
    ss = np.dot(s, s)
    ys = np.dot(y, s)
    if ss < 1e-15:
        return 0.01
    if abs(ys) < 1e-15:
        return 0.01
    # Use the BB+ variant (always positive)
    alpha_s = ss / ys if ys > 0 else ss / ys  # BB-s
    alpha_y = ys / np.dot(y, y) if np.dot(y, y) > 1e-15 else 0.01  # BB-y
    # Use BB-s (more common)
    if alpha_s > 0 and alpha_s < 10:
        return alpha_s
    return 0.01

def nesterov_lse_optimize(a0, time_limit, T_start=50.0, T_end=1.0, verbose=True):
    """Nesterov-accelerated LSE gradient descent with BB step size."""
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s0
    
    best_a = a.copy()
    best_score = eval_obj(a)
    
    y = a.copy()  # Nesterov momentum variable
    x_prev = a.copy()
    grad_prev = None
    t0 = time.monotonic()
    iters = 0
    
    for it in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive temperature
        progress = min(it / 5000.0, 1.0)
        T = T_start * (T_end / T_start) ** progress
        
        # Nesterov lookahead
        if it == 1:
            y_new = a.copy()
        else:
            beta = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
            momentum = (beta - 1.0) / beta
            y_new = y + momentum * (y - x_prev)
        
        # Compute gradient at y_new
        S_y = y_new.sum()
        if S_y < 1e-10:
            y_new = a.copy()
            S_y = S
        grad_y = lse_grad(y_new, T, S_y)
        
        # BB step size
        if grad_prev is not None and it > 5:
            alpha = bb_step(x_prev, y, grad_prev, grad_y)
            alpha = min(alpha, 1.0 / T)  # Cap step size
        else:
            alpha = 2.0 / (T * (it + 1.0))
        
        # Gradient step
        x_new = y_new - alpha * grad_y
        x_new = project_simplex(x_new, S)
        
        # Check improvement
        score = eval_obj(x_new)
        if score < best_score:
            best_score = score
            best_a = x_new.copy()
        
        x_prev = x_a = x_new
        y = x_new
        S_y = y.sum()
        grad_y = lse_grad(y, T, S_y)
        grad_prev = grad_y
        iters += 1
        
        if verbose and iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE iter={iters} ({elapsed:.0f}s): T={T:.2f}, best={best_score:.8f}")
    
    return best_a, best_score

def peak_reducing_search(a0, time_limit, seed=42):
    """Targeted search that moves mass away from convolution peaks."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s0
    
    best_a = a.copy()
    best_score = eval_obj(a)
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_conv(a)
        # Find top peaks
        top_k = min(8, len(b))
        peak_indices = np.argsort(b)[::-1][:top_k]
        
        improved = False
        
        for pk in peak_indices[:4]:
            # Find which pairs (i, j) with i+j=pk contribute to this peak
            i_min = max(0, pk - n + 1)
            i_max = min(n - 1, pk)
            if i_min > i_max:
                continue
            
            pairs = np.arange(i_min, i_max + 1)
            contrib = a[pairs] * a[pk - pairs]
            top_pairs_idx = pairs[np.argsort(contrib)[::-1][:5]]
            
            # For each top-contributing pair, try moving mass
            for src_i in top_pairs_idx:
                if a[src_i] < 1e-12:
                    continue
                
                # Find destination: positions NOT contributing to any top peak
                bad_positions = set()
                for pk2 in peak_indices[:4]:
                    i2_min = max(0, pk2 - n + 1)
                    i2_max = min(n - 1, pk2)
                    bad_positions.update(range(i2_min, i2_max + 1))
                
                good_positions = [p for p in range(n) if p not in bad_positions]
                if not good_positions:
                    good_positions = [p for p in range(n) if p != src_i]
                
                # Sort by current value (prefer moving to low-value positions)
                good_positions.sort(key=lambda p: a[p])
                
                for dst in good_positions[:8]:
                    for frac in [0.01, 0.03, 0.08, 0.15, 0.3]:
                        amt = a[src_i] * frac
                        if amt < 1e-14:
                            continue
                        trial = a.copy()
                        trial[src_i] -= amt
                        trial[dst] += amt
                        if trial[src_i] < -1e-12:
                            continue
                        sc = eval_obj(trial)
                        if sc < best_score - 1e-12:
                            a = trial
                            best_a = trial.copy()
                            best_score = sc
                            stall = 0
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall > 5:
                # Random perturbation
                n_perturb = max(2, n // 200)
                perturb_idx = rng.choice(n, size=n_perturb, replace=False)
                trial = a.copy()
                for idx in perturb_idx:
                    trial[idx] *= rng.uniform(0.7, 1.3)
                trial = np.maximum(trial, 0.0)
                sc = eval_obj(trial)
                if sc < best_score:
                    a = trial
                    best_a = trial.copy()
                    best_score = sc
                    stall = 0
                elif stall > 100:
                    break
            elif stall > 50:
                # Add Gaussian noise
                noise = rng.normal(0, 0.01 * a.mean(), n)
                trial = np.maximum(a + noise, 0.0)
                sc = eval_obj(trial)
                if sc < best_score:
                    a = trial
                    best_a = trial.copy()
                    best_score = sc
                    stall = 0
    
    return best_a, best_score

def spectral_polish(a0, time_limit, seed=42):
    """Spectral domain post-processing."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s0
    
    best_a = a.copy()
    best_score = eval_obj(a)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        # Compute spectrum
        fft_size = 1 << (2 * n - 1).bit_length()
        A = rfft(a, fft_size)
        power = np.abs(A)**2
        
        # The convolution is IFFT(power)
        # Try to reduce peaks by modifying the spectrum
        b = fast_conv(a)
        max_b = b.max()
        peak_frac = max_b / b.sum()  # Should be ~1/(2n-1) if flat
        
        # Strategy: reduce high-power frequency components that align with peaks
        # For each peak at position p, the contribution from frequency f is:
        # power[f] * cos(2*pi*f*p/fft_size) (real part)
        # We want to reduce power at frequencies where this is positive for peak positions
        
        # Simple approach: reduce the variance of the power spectrum
        mean_power = power.mean()
        # Shrink high-power bins
        shrink = 0.05
        new_power = power.copy()
        high_mask = power > mean_power * 1.5
        new_power[high_mask] = mean_power * 1.5 + (power[high_mask] - mean_power * 1.5) * (1 - shrink)
        
        # Reconstruct: a_new = IFFT(sqrt(new_power) * phase(A))
        # But this might give negative values
        phase = A / np.abs(A)
        phase[np.abs(A) < 1e-12] = 1.0
        new_A = np.sqrt(new_power) * phase
        a_new = irfft(new_A, fft_size)[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-10:
            a_new = a_new * S / s_new
            sc = eval_obj(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
            else:
                # Try smaller shrinkage
                break
        
        # Alternate: try increasing low-power bins
        new_power2 = power.copy()
        low_mask = power < mean_power * 0.5
        new_power2[low_mask] = mean_power * 0.5 - (mean_power * 0.5 - power[low_mask]) * shrink
        new_A2 = np.sqrt(new_power2) * phase
        a_new2 = irfft(new_A2, fft_size)[:n]
        a_new2 = np.maximum(a_new2, 0.0)
        s_new2 = a_new2.sum()
        if s_new2 > 1e-10:
            a_new2 = a_new2 * S / s_new2
            sc2 = eval_obj(a_new2)
            if sc2 < best_score:
                best_score = sc2
                best_a = a_new2.copy()
                a = a_new2
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LSE-smoothed projected gradient descent with Nesterov acceleration + BB step size
2. Peak-targeted mass redistribution (gradient-guided local search)
3. Spectral domain post-processing
4. Multi-length exploration with interpolation transfer
5. Simulated annealing style exploration for escaping local minima
Key derivative: under sum(a)=S constraint, grad = 2*(c - mean(c)) where c[i] = sum_j a[j]*w[i+j]
and w = softmax(T * (conv(a,a) - max)). Computed via FFT cross-correlation.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_conv(a):
    """Compute conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_obj(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2 for non-negative a."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    S = a.sum()
    if S < 1e-15:
        return float('inf')
    return 2.0 * n * max_b / (S * S)


def compute_lse_grad(a, T):
    """
    Compute gradient of LSE-smoothed objective under constraint sum(a)=S.
    Returns the direction to subtract (i.e., positive gradient of objective).
    grad[i] = 2*(c[i] - mean(c)) where c[i] = sum_j a[j]*w[i+j]
    """
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    exp_args = T * (b - max_b)
    exp_vals = np.exp(exp_args)
    w = exp_vals / exp_vals.sum()
    
    # c[i] = sum_{j=0}^{n-1} a[j] * w[i+j]
    # This is cross-correlation: corr(w, a)[i] = sum_j w[j+i] * a[j]
    # = IFFT(conj(FFT(w)) * FFT(a)) with zero padding
    fft_size = 1 << (3 * n - 2).bit_length()
    a_pad = np.zeros(fft_size)
    a_pad[:n] = a
    w_pad = np.zeros(fft_size)
    w_pad[:2 * n - 1] = w
    rW = rfft(w_pad)
    rA = rfft(a_pad)
    c = irfft(np.conj(rW) * rA, fft_size)[:n]
    
    # Gradient direction (to subtract): 2*(c - mean(c))
    grad = 2.0 * (c - c.mean())
    return grad, w, b


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho_candidates = np.where(v_sorted * np.arange(1, n + 1) > (cssv - S))[0]
    if len(rho_candidates) == 0:
        return np.full(n, S / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    result = np.maximum(v - theta, 0.0)
    # Fix numerical drift
    diff = S - result.sum()
    if abs(diff) > 1e-12:
        result += diff / n
        result = np.maximum(result, 0.0)
    return result


def bb_step_size(s, y):
    """Barzilai-Borwein step size (BB-s variant with safeguards)."""
    ss = np.dot(s, s)
    ys = np.dot(y, s)
    if ss < 1e-20:
        return 0.1
    if ys < 1e-15:
        return 0.1
    alpha = ss / ys
    if alpha < 1e-6:
        alpha = 0.1
    elif alpha > 10.0:
        alpha = 10.0
    return alpha


def nesterov_lse(a0, time_limit, T_start=40.0, T_end=0.3, verbose=True):
    """
    Nesterov-accelerated gradient descent on LSE-smoothed objective.
    Uses BB step size with backtracking safeguard.
    """
    n = len(a0)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s0
    
    best_a = a.copy()
    best_score = eval_obj(a)
    
    y = a.copy()
    x_prev = a.copy()
    grad_prev = None
    t0 = time.monotonic()
    iters = 0
    last_improve = t0
    
    for it in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive temperature: exponential decay
        progress = min(it / 3000.0, 1.0)
        T = T_start * (T_end / T_start) ** progress
        
        # Nesterov momentum
        if it == 1:
            y_new = a.copy()
        else:
            t_val = float(it)
            beta = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
            momentum = (beta - 1.0) / beta
            y_new = y + momentum * (y - x_prev)
            y_new = np.maximum(y_new, 0.0)
            y_new = y_new * S / max(y_new.sum(), 1e-15)
        
        # Compute gradient at y_new
        grad_y, w_y, b_y = compute_lse_grad(y_new, T)

        if grad_prev is not None and it > 3:
            s_vec = y - x_prev
            y_vec = grad_y - grad_prev
            alpha = bb_step_size(s_vec, y_vec)
            alpha = min(alpha, 5.0 / T)
        else:
            alpha = 2.0 / (T * (it + 1.0))
        
        # Gradient step + projection
        x_new = y_new - alpha * grad_y
        x_new = project_simplex(x_new, S)
        
        # Evaluate
        score = eval_obj(x_new)
        if score < best_score:
            best_score = score
            best_a = x_new.copy()
            last_improve = time.monotonic()
        
        # Update
        x_prev = y.copy()
        y = x_new
        grad_y, _, _ = compute_lse_grad(y, T)
        grad_prev = grad_y
        iters += 1
        
        # Stall detection
        if it % 300 == 0 and time.monotonic() - last_improve > 60:

                print(f"      LSE stall at iter={iters}, T={T:.2f}, best={best_score:.8f}")
            # Restart from best
            y = best_a.copy()
            x_prev = y.copy()
            grad_prev = None
            last_improve = time.monotonic()
        
        if verbose and iters % 100 == 0:

            print(f"      LSE iter={iters} ({elapsed:.0f}s): T={T:.2f}, best={best_score:.8f}")
    
    return best_a, best_score

        improved = False

                        sc = eval_obj(trial)
                        if sc < best_score - 1e-13:

                            best_a = trial.copy()
                            best_score = sc
                            stall = 0
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall % 3 == 0 and stall <= 15:
                # Small random perturbations

                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i == j or a[i] < 1e-12:

                    trial[i] -= amt
                    trial[j] += amt
                    sc = eval_obj(trial)

                break
    
    return best_a, best_score


def spectral_polish(a0, time_limit, seed=42):
    """
    Work in frequency domain: modify the power spectrum to reduce time-domain
    convolution peaks, then project back to non-negative time domain.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s0
    
    best_a = a.copy()
    best_score = eval_obj(a)
    t0 = time.monotonic()
    
    fft_size = 1 << (2 * n - 1).bit_length()
    
    shrinks = [0.02, 0.05, 0.1, 0.2]
    for shrink in shrinks:
        if time.monotonic() - t0 > time_limit:
            break
        for _ in range(20):
            if time.monotonic() - t0 > time_limit:
                break
            
            A = rfft(a, fft_size)
            mag = np.abs(A)
            phase = np.ones_like(A)
            nonzero = mag > 1e-12
            phase[nonzero] = A[nonzero] / mag[nonzero]
            power = mag ** 2
            
            # Reduce spectral peaks (high power bins)
            mean_p = power.mean()
            threshold = mean_p * (1.0 + 2.0 * shrink)
            high_mask = power > threshold
            new_power = power.copy()
            new_power[high_mask] = threshold + (power[high_mask] - threshold) * (1.0 - shrink)
            
            # Also slightly boost low bins
            low_threshold = mean_p * (1.0 - shrink)
            low_mask = power < low_threshold
            new_power[low_mask] = low_threshold - (low_threshold - power[low_mask]) * shrink * 0.5
            
            # Reconstruct
            new_mag = np.sqrt(np.maximum(new_power, 0))
            new_A = new_mag * phase
            a_new = irfft(new_A, fft_size)[:n]
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new * S / s_new
                sc = eval_obj(a_new)
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    a = a_new
    
    return best_a, best_score


def gradient_local_search(a0, time_limit, seed=42):
    """
    Hybrid: use LSE gradient to identify good directions, then do
    line search along that direction with multiple step sizes.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    S = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s0
    
    best_a = a.copy()
    best_score = eval_obj(a)
    t0 = time.monotonic()
    stall = 0
    
    T = 10.0  # Fixed moderate temperature for gradient computation
    
    while time.monotonic() - t0 < time_limit:
        grad, w, b = compute_lse_grad(a, T)
        
        # Normalize gradient direction
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-12:
            break
        direction = -grad / g_norm  # Descent direction
        
        # Try multiple step sizes along this direction
        best_step = 0
        best_trial_score = best_score
        max_step = a.max() * 0.5
        for alpha in [0.001, 0.003, 0.01, 0.03, 0.1, 0.3]:
            step = alpha * max_step
            trial = a + step * direction
            trial = project_simplex(trial, S)
            sc = eval_obj(trial)
            if sc < best_trial_score:
                best_trial_score = sc
                best_step = step
        
        if best_step > 0 and best_trial_score < best_score - 1e-13:
            a = project_simplex(a + best_step * direction, S)
            best_a = a.copy()
            best_score = best_trial_score
            stall = 0
        else:
            stall += 1
            if stall > 5:
                # Random restart
                for _ in range(30):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i == j or a[i] < 1e-12:
                        continue
                    frac = rng.uniform(0.01, 0.1)
                    amt = a[i] * frac
                    trial = a.copy()
                    trial[i] -= amt
                    trial[j] += amt
                    sc = eval_obj(trial)
                    if sc < best_score:
                        a = trial
                        best_a = trial.copy()
                        best_score = sc
                        stall = 0
                        break
                if stall > 50:
                    break
    
    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function: returns best sequence found within time budget."""
    t_start = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    rng = np.random.default_rng(seed)
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_obj(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception as e:
        print(f"No parent loaded: {e}")
        best_seq = np.ones(1600) / 1600
        best_score = 2.0
        print(f"Default init: n=1600, score={best_score:.6f}")
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        sc = eval_obj(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    n_base = len(best_seq)
    
    # === Phase 1: LSE Nesterov (primary optimizer, 30% time) ===
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 1: LSE Nesterov (n={n_base}, budget={remaining*0.30:.0f}s) ===")
    t1 = remaining * 0.30
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lse(a_norm, t1, T_start=50.0, T_end=0.5, verbose=True)
        try_update(a_out, "lse_nest")
    
    # === Phase 2: Peak local search (20% time) ===
    remaining = time_budget - (time.monotonic() - t_start)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Peak search (n={n_best}, budget={remaining*0.20:.0f}s) ===")
    t2 = remaining * 0.20
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_local_search(a_norm, t2, seed=101)
        try_update(a_out, "peak1")
    
    # === Phase 3: LSE pass 2 with noise (15% time) ===
    remaining = time_budget - (time.monotonic() - t_start)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LSE pass 2 (n={n_best}, budget={remaining*0.15:.0f}s) ===")
    t3 = remaining * 0.15
    if t3 > 20:
        a_noisy = best_seq + rng.normal(0, 0.003 * best_seq.mean(), n_best)
        a_noisy = np.maximum(a_noisy, 0.0)
        a_norm = a_noisy * np.sqrt(2.0 * n_best) / a_noisy.sum()
        a_out, _ = nesterov_lse(a_norm, t3, T_start=20.0, T_end=0.3, verbose=False)
        try_update(a_out, "lse2")
    
    # === Phase 4: Gradient local search (15% time) ===
    remaining = time_budget - (time.monotonic() - t_start)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Grad local (n={n_best}, budget={remaining*0.15:.0f}s) ===")
    t4 = remaining * 0.15
    if t4 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = gradient_local_search(a_norm, t4, seed=202)
        try_update(a_out, "grad_local")
    
    # === Phase 5: Spectral polish (10% time) ===
    remaining = time_budget - (time.monotonic() - t_start)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Spectral (n={n_best}, budget={remaining*0.10:.0f}s) ===")
    t5 = remaining * 0.10
    if t5 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = spectral_polish(a_norm, t5, seed=303)
        try_update(a_out, "spectral")
    
    # === Phase 6: Length exploration (10% time) ===
    remaining = time_budget - (time.monotonic() - t_start)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Length exploration ===")
    for n_try in [n_best + 100, n_best - 100, n_best + 200]:
        if time.monotonic() - t_start > time_budget - 90:
            break
        t_rem = time_budget - (time.monotonic() - t_start) - 60
        t_alloc = min(t_rem * 0.25, 100.0)
        if t_alloc < 15:
            break
        print(f"  Trying n={n_try} ({t_alloc:.0f}s)")
        init = interpolate_seq(best_seq, n_try)
        a_norm = init * np.sqrt(2.0 * n_try) / init.sum()
        a_out, _ = nesterov_lse(a_norm, t_alloc * 0.6, T_start=30.0, T_end=0.5, verbose=False)
        try_update(a_out, f"len{n_try}")
        if time.monotonic() - t_start < time_budget - 60:
            a_out2, _ = peak_local_search(a_out, t_alloc * 0.4, seed=404)
            try_update(a_out2, f"len{n_try}_pk")
    
    # === Phase 7: Final intensive peak search (remaining time) ===
    remaining = time_budget - (time.monotonic() - t_start)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final peak polish (n={n_best}, {remaining:.0f}s) ===")
    if remaining > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_local_search(a_norm, remaining * 0.7, seed=505)
        try_update(a_out, "final_peak")
        # One more LSE pass
        remaining2 = time_budget - (time.monotonic() - t_start)
        if remaining2 > 15:
            a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out2, _ = nesterov_lse(a_norm, remaining2, T_start=10.0, T_end=0.2, verbose=False)
            try_update(a_out2, "final_lse")
    
    # Final validation
    if best_seq is not None:
        final_score = eval_obj(best_seq)
        print(f"\nFINAL: score={final_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_start:.0f}s")
        # Ensure all values are in valid range
        result = np.clip(best_seq, 0.0, 1000.0)
        assert result.sum() > 0.01, "Sum too small"
        return [float(x) for x in result]
    else:
        print("ERROR: No valid sequence!")
        return [1.0]
