import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize as scipy_minimize

_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    return _global_solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_fast(a):
    """Fast evaluation using FFT autocorrelation."""
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    max_b = b.max()
    n = len(a)
    return float(2.0 * n * max_b / (s * s))


def evaluate_safe(a):
    try:
        a_arr = np.asarray(a, dtype=np.float64).ravel()
        a_list = [float(x) for x in a_arr]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def lp_best_response(a, M, solver):
    """Solve max sum(b) s.t. (a*b)_k <= M for all k, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    size = 2 * n - 1
    # Build constraint: T @ b <= M where T[k,i] = a[k-i] if valid
    # Use a more efficient construction
    rows = []
    cols = []
    vals = []
    for i in range(n):
        for k in range(max(0, i), min(i + n, size)):
            j = k - i
            if 0 <= j < n:
                rows.append(k)
                cols.append(i)
                vals.append(a[j])
    from scipy.sparse import csr_matrix
    T = csr_matrix((vals, (rows, cols)), shape=(size, n))
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def lp_best_response_sparse(a, M, solver):
    """More efficient LP using direct sparse construction."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    size = 2 * n - 1

    # Build the convolution constraint matrix efficiently
    # (a * b)[k] = sum_i a[|k-i|] * b[i] for valid indices
    # For each k, the valid i range is [max(0, k-n+1), min(k, n-1)]
    import scipy.sparse as sp

    row_list = []
    col_list = []
    data_list = []
    for i in range(n):
        k_start = max(0, i - (n - 1))  # Wait, let me think more carefully
        # (a*b)[k] = sum_i a[i'] * b[k - i'] where i' is the index into a
        # Actually np.convolve(a, b)[k] = sum_j a[j] * b[k-j]
        # So for fixed i (index into b), the contribution to position k is a[k-i]*b[i]
        # valid when 0 <= k-i < n, i.e., i <= k <= i+n-1
        k_start = i
        k_end = min(i + n, size)
        for k in range(k_start, k_end):
            j = k - i  # index into a
            row_list.append(k)
            col_list.append(i)
            data_list.append(a[j])

    T = sp.csr_matrix((data_list, (row_list, col_list)), shape=(size, n))
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None
    b_val = np.maximum(b_val, 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration and line search."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    last_improve = time.monotonic()
    consecutive_no_improve = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        # Also break if we've been going for a while without improvement
        if t > 100 and time.monotonic() - last_improve > time_limit * 0.3:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response_sparse(y, M_curr, solver)
        if b is None:
            consecutive_no_improve += 1
            if consecutive_no_improve > 20:
                break
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        consecutive_no_improve = 0
        g = b * norm / b_sum
        n_iters += 1

        # Adaptive line search
        alpha_base = 2.0 / (t + 2.0)
        alphas = [alpha_base, alpha_base * 0.5, alpha_base * 1.5, alpha_base * 2.0, 1.0]
        best_alpha = alpha_base
        best_cand_obj = fast_autocorr(np.maximum((1 - alpha_base) * y + alpha_base * g, 0.0)).max()
        for al in alphas[1:]:
            al = min(al, 1.0)
            cand = np.maximum((1 - al) * y + al * g, 0.0)
            sc = fast_autocorr(cand).max()
            if sc < best_cand_obj:
                best_cand_obj = sc
                best_alpha = al
        alpha = best_alpha

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        # Nesterov momentum
        if t == 1:
            a_new = y_new.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            last_improve = time.monotonic()

        # Averaging
        avg_a += a
        n_avg += 1
        if n_avg % 5 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    last_improve = time.monotonic()

        # Restart if stalling
        if t % 300 == 0:
            if t > 600 and time.monotonic() - last_improve > 60.0:
                y = best_a * norm / best_a.sum()
                a = y.copy()
                avg_a = np.zeros(n)
                n_avg = 0
                if verbose:
                    print(f"    Restart at iter={n_iters}, best={best_score:.8f}")

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    # Final averaging check
    if n_avg > 0:
        avg_final = avg_a / n_avg
        s_f = avg_final.sum()
        if s_f > 1e-15:
            avg_final = avg_final * norm / s_f
            score_f = fast_autocorr(avg_final).max()
            if score_f < best_score:
                best_score = score_f
                best_a = avg_final.copy()

    return best_a, best_score


def lse_objective(a_flat, n, T_temp):
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10
    exp_vals = np.exp(T_temp * (b - max_b))
    lse = max_b + np.log(exp_vals.sum()) / T_temp
    return 2.0 * n * lse / (s * s)


def lse_gradient(a_flat, n, T_temp):
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return np.zeros(n)
    exp_vals = np.exp(T_temp * (b - max_b))
    w = exp_vals / exp_vals.sum()
    lse = max_b + np.log(exp_vals.sum()) / T_temp

    # Gradient of lse w.r.t. a: (1/T) * sum_k w_k * d(b_k)/da
    # b_k = (a*a)[k], db_k/da_i = a[k-i] + a[i-k] (with valid indices)
    # Using FFT: grad_i = (4/T) * sum_k w_k * (a*a_conv_structure)
    # More precisely: d(b_k)/d(a_i) = a[k-i] if 0<=k-i<n, plus a[i-(k-n+1)] if valid
    # This is the "cross-correlation" of w with a at each position
    fft_size = 1 << (3 * n - 2).bit_length()
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a
    w_padded = np.zeros(fft_size)
    w_padded[:2 * n - 1] = w
    A_fft = rfft(a_padded, fft_size)
    W_fft = rfft(w_padded, fft_size)
    # The gradient of b_k w.r.t. a_i involves (a * w)[i] and (w * a)[i]
    # Actually: d/d(a_i) sum_k w_k * (a*a)[k] = 2 * sum_k w_k * (a * flip(a))_i... 
    # Let me use: d/d(a_i) (a*a)[k] = a[k-i] + a[i-k] (for valid ranges)
    # So grad_i = 2 * sum_k w_k * (a[k-i] if valid) = 2 * (w * a_flipped @ i) approximately
    # Using the FFT approach: convolve w with a
    cross = irfft(W_fft * A_fft, fft_size)[:n]
    # Actually this gives (w*a)[i] = sum_j w[j]*a[i-j], which is what we want for one side
    # The full gradient is 4 * cross (factor of 2 from symmetry, factor of 2 from the two terms)
    conv_grad = 4.0 * cross
    obj_val = 2.0 * n * lse / (s * s)
    grad = conv_grad / (s * s) - 2.0 * obj_val * a / s
    return grad.ravel()


def lse_scipy_optimize(a0, time_limit, T_start=30.0, T_end=1.0, n_stages=5):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        stage_time = remaining * 0.8 / max(1, n_stages - si)
        stage_time = min(stage_time, 120.0)

        x0 = a.copy()

        def obj_func(x):
            return lse_objective(x, n, T)

        def grad_func(x):
            return lse_gradient(x, n, T)

        try:
            res = scipy_minimize(obj_func, x0, method='L-BFGS-B',
                                  jac=grad_func,
                                  bounds=[(0, None)] * n,
                                  options={'maxiter': 200, 'maxfun': 500})
            if res.x is not None:
                a_candidate = np.maximum(res.x, 0.0)
                s_c = a_candidate.sum()
                if s_c > 1e-10:
                    a_candidate = a_candidate * norm / s_c
                    score_c = fast_autocorr(a_candidate).max()
                    if score_c < best_score:
                        best_score = score_c
                        best_a = a_candidate.copy()
                        a = a_candidate
        except Exception:
            pass

    return best_a, best_score


def gradient_coordinate_descent(a0, time_limit, seed=42):
    """
    Gradient-guided coordinate descent.
    At each step, identify the peak lag(s) and redistribute mass
    from high-contribution positions to low-impact positions.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    def try_update(trial):
        nonlocal a, best_a, best_score, stall
        trial = np.maximum(trial, 0.0)
        s_t = trial.sum()
        if s_t < 1e-10:
            return False
        trial = trial * norm / s_t
        score = fast_autocorr(trial).max()
        if score < best_score - 1e-15:
            a = trial
            best_a = trial.copy()
            best_score = score
            stall = 0
            return True
        return False

    iteration = 0
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        b = fast_autocorr(a)
        # Find top peaks
        n_peaks = min(8, 2 * n - 1)
        top_peaks = np.argsort(b)[::-1][:n_peaks]

        improved = False

        for k_peak in top_peaks[:4]:
            # Positions that contribute to this peak
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k_peak - idx]
            total_contrib = contrib.sum()
            if total_contrib < 1e-15:
                continue
            # Top contributors
            top_idx = idx[np.argsort(contrib)[::-1][:4]]

            # Build mask of positions involved in any top peak
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:6]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                if je2 >= js2:
                    peak_mask[js2:je2 + 1] = True

            for src in top_idx:
                if a[src] < 1e-12:
                    continue
                # Find good destinations: not in peak mask, low current value
                candidates = np.where(~peak_mask)[0]
                if len(candidates) == 0:
                    candidates = np.where([i for i in range(n) if i != src])[0]
                if len(candidates) == 0:
                    continue
                # Sort by a value (move to lower values for more uniformity)
                candidates_sorted = candidates[np.argsort(a[candidates])]
                n_try = min(8, len(candidates_sorted))

                for dst in candidates_sorted[:n_try]:
                    for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if try_update(a_trial):
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random restarts within the coordinate descent
            if stall % 3 == 0:
                for _ in range(50):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i == j or a[i] < 1e-12:
                        continue
                    frac = rng.uniform(0.005, 0.1)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if try_update(a_trial):
                        break

            # Periodic perturbation
            if stall % 20 == 1:
                n_perturb = max(5, n // 80)
                indices = rng.choice(n, size=n_perturb, replace=False)
                a_trial = a.copy()
                for idx in indices:
                    a_trial[idx] *= rng.uniform(0.7, 1.3)
                if try_update(a_trial):
                    pass

            # Multi-step random walk
            if stall % 50 == 1:
                a_trial = a.copy()
                n_steps = n // 20
                for _ in range(n_steps):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i == j or a[i] < 1e-12:
                        continue
                    amt = a[i] * rng.uniform(0.01, 0.05)
                    a_trial[i] -= amt
                    a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                if try_update(a_trial):
                    pass

            if stall > 500:
                break

    return best_a, best_score


def projected_gradient_step(a, lr, n):
    """
    One step of projected gradient descent on max_k (a*a)(k)
    with the constraint sum(a) = sqrt(2n), a >= 0.
    Uses subgradient of the max.
    """
    b = fast_autocorr(a)
    max_b = b.max()
    peak_indices = np.where(np.abs(b - max_b) < 1e-10)[0]
    
    # Subgradient: average of gradients at active peaks
    # d(b_k)/d(a_i) = sum over valid j of a[k-i+j]... 
    # More precisely for convolution: d/d(a_i) (a*a)[k] = a[k-i] + a[i-k] (valid terms)
    # Using FFT for efficiency
    n_seq = len(a)
    fft_size = 1 << (3 * n_seq - 2).bit_length()
    a_padded = np.zeros(fft_size)
    a_padded[:n_seq] = a
    A_fft = rfft(a_padded, fft_size)
    
    # Gradient at a peak k: (a * delta_k)[i] where we convolve a with the indicator
    # Actually: d(b_k)/d(a_i) where b = a*a
    # b_k = sum_j a_j * a_{k-j}
    # d(b_k)/d(a_i) = a_{k-i} + a_{i-k} (when indices valid)
    # This is (a * a_reflected)_i evaluated at the right offset... 
    # Simpler: use the fact that grad = 2 * (P_k * a) where P_k is the shift operator
    # For a single peak k: grad_i = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)
    
    grad = np.zeros(n_seq)
    for k in peak_indices:
        for i in range(n_seq):
            val = 0.0
            j1 = k - i
            if 0 <= j1 < n_seq:
                val += a[j1]
            j2 = i - k
            if 0 <= j2 < n_seq:
                val += a[j2]
            grad[i] += val
    grad /= len(peak_indices)
    
    # Projected gradient step
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new * np.sqrt(2.0 * n_seq) / s
    return a_new


def interp_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

        if best_seq is not None:
            init_seq = interp_sequence(best_seq, n_scan)
        else:
            init_seq = np.ones(n_scan) / n_scan
        a_norm = init_seq * np.sqrt(2.0 * n_scan) / init_seq.sum()
        scan_time = min((time_budget - phase0_budget) - (time.monotonic() - t_global), 60.0)
        if scan_time < 10:
            break
        a_out, sc_out = nesterov_lp_optimize(a_norm, scan_time, solver, verbose=False)
        scan_results[n_scan] = (a_out, sc_out)
        do_try_update(a_out, f"scan_n{n_scan}")
        print(f"  n={n_scan}: score={sc_out:.8f} (t={time.monotonic()-t_global:.0f}s)")

    # Determine best length
    if best_seq is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = evaluate_safe(best_seq)
    else:
        n_best = len(best_seq)
        # Use the length that gave the best scan result if available
        if scan_results:
            best_scan_n = min(scan_results, key=lambda k: scan_results[k][1])
            if scan_results[best_scan_n][1] < best_score * 1.001:
                n_base = best_scan_n
                best_seq = scan_results[best_scan_n][0]
                best_score = scan_results[best_scan_n][1]
            else:
                n_base = n_best
        else:
            n_base = n_best

    # Phase 1: Main LP optimization at best length
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP main n={n_base} (budget={remaining:.0f}s) ===")
    t1 = remaining * 0.40
    a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
    a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
    do_try_update(a_out, f"lp_main_n{n_base}")

    # Phase 2: LSE smooth optimization
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LSE scipy n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.12, 150.0)
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_scipy_optimize(a_norm, t2, T_start=40.0, T_end=2.0, n_stages=5)
        do_try_update(a_out, f"lse_n{n_best}")

    # Phase 3: Try longer length
    n_long = n_base + 200
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LP longer n={n_long} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.18, 250.0)
    if t3 > 30:
        init_long = interp_sequence(best_seq, n_long)
        a_norm = init_long * np.sqrt(2.0 * n_long) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=True)
        do_try_update(a_out, f"lp_long_n{n_long}")

    # Phase 4: Noisy restart LP
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: LP noisy restart n={n_best} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.12, 150.0)
    if t4 > 20:
        a_noisy = best_seq + rng.normal(0, 0.03 * best_seq.mean(), n_best)
        a_noisy = np.maximum(a_noisy, 0.0)
        a_noisy = a_noisy * np.sqrt(2.0 * n_best) / a_noisy.sum()
        a_out, _ = nesterov_lp_optimize(a_noisy, t4, solver, verbose=False)
        do_try_update(a_out, f"lp_noisy_n{n_best}")

    # Phase 5: Gradient coordinate descent (extended)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Grad coord descent n={n_best} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.25, 350.0)
    if t5 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = gradient_coordinate_descent(a_norm, t5, seed=99999)
        do_try_update(a_out, "grad_cd_1")

    # Phase 6: LSE low temperature polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: LSE polish n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.12, 120.0)
    if t6 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_scipy_optimize(a_norm, t6, T_start=15.0, T_end=0.5, n_stages=4)
        do_try_update(a_out, "lse_polish")

    # Phase 7: Final coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final coord descent n={n_best} (budget={remaining:.0f}s) ===")
    t7 = max(remaining - 30, 30)
    if t7 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = gradient_coordinate_descent(a_norm, t7, seed=123456)
        do_try_update(a_out, "final_cd")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
