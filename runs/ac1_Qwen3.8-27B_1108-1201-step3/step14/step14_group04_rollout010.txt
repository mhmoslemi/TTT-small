        je = min(n-1, k)
        for j in range(js, je+1):

Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:

2. LP-based Nesterov with warm-starting, ergodic averaging, and adaptive M

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.sparse import csc_matrix

_lp_pattern_cache = {}
_global_solver = None


def get_lp_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.CLARABEL
    print("  LP solver (fallback): CLARABEL")
    return _global_solver


def fast_autocorr(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    s = v.sum()
    if s < 1e-15:
        n = len(v)
        return np.ones(n) * target_sum / n
    return target_sum * v / s


def precompute_lp_pattern(n):
    if n in _lp_pattern_cache:
        return _lp_pattern_cache[n]
    rows = []
    cols = []
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            rows.append(k)
            cols.append(j)
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    _lp_pattern_cache[n] = (rows, cols)
    return rows, cols


def build_lp_matrix(a, n):
    rows, cols = precompute_lp_pattern(n)
    idx = rows - cols
    vals = a[idx]
    return csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, solver, warm_b=None):
    n = len(a)
    A = build_lp_matrix(a, n)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [A @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def subgrad_fista(a0, time_limit, verbose=True):
    """Fast FISTA using top-K subgradients of max autocorrelation."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    t1 = 1
    t0 = time.monotonic()
    K_active = 5

    while time.monotonic() - t0 < time_limit:
        ac_y = fast_autocorr(y)
        M_curr = ac_y.max()
        if M_curr < 1e-15:
            break
        # Get top-K active lags
        top_lags = np.argsort(ac_y)[::-1][:K_active]
        weights = ac_y[top_lags]
        weights = weights / weights.sum()
        # Subgradient: weighted sum of 2*T_k*y for active k
        grad = np.zeros(n)
        for w, k in zip(weights, top_lags):
            # (T_k * y)_i = y_{i+k} + y_{i-k} (with boundary handling)
            if k == 0:
                grad += 2.0 * w * 2.0 * y
            else:
                # (T_k y)_i = y_{i+k} if i+k < n else 0, plus y_{i-k} if i-k >= 0 else 0
                shifted1 = np.zeros(n)
                shifted2 = np.zeros(n)
                if k < n:
                    shifted1[:n-k] = y[k:]
                    shifted2[k:] = y[:n-k]
                grad += 2.0 * w * (shifted1 + shifted2)
        # Lipschitz estimate
        L = 4.0 * n / 10.0 + 1.0
        step = 1.0 / L
        z = y - step * grad
        a_new = project_simplex(z, norm_val)
        if t1 == 1:
            y_new = a_new.copy()
        else:
            mom = (t1 - 1.0) / (t1 + 2.0)
            y_new = a_new + mom * (a_new - y)
            y_new = project_simplex(y_new, norm_val)
        y = y_new
        a = a_new
        t1 += 1
        if t1 % 200 == 0:
            sc = fast_autocorr(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                if verbose:
                    elapsed = time.monotonic() - t0
                    print(f"    SubGrad-FISTA iter={t1} ({elapsed:.0f}s): best={best_score:.8f}")
    sc = fast_autocorr(a).max()
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def lp_nesterov_improved(a0, time_limit, solver, verbose=True, restart_interval=500):
    """LP-based Nesterov with ergodic averaging and periodic restarts."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    # Exponential moving average
    ema_a = a.copy()
    ema_beta = 0.99
    t0 = time.monotonic()
    n_iters = 0
    t = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm_val / b_sum
        n_iters += 1

        # Nesterov step with periodic restart
        if t % restart_interval == 1:
            t_eff = t % restart_interval
            alpha = 1.0 / (t_eff + 1.0)
        else:
            alpha = 2.0 / (t + 2.0)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_val / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm_val / s_a

        y = y_new
        a = a_new

        # Update EMA
        ema_a = ema_beta * ema_a + (1.0 - ema_beta) * a
        s_ema = ema_a.sum()
        if s_ema > 1e-15:
            ema_a_scaled = ema_a * norm_val / s_ema

        # Check scores
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        # Check EMA periodically
        if n_iters % 30 == 0:
            score_ema = fast_autocorr(ema_a_scaled).max()
            if score_ema < best_score:
                best_score = score_ema
                best_a = ema_a_scaled.copy()

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    # Final check
    score_final = fast_autocorr(a).max()
    if score_final < best_score:
        best_score = score_final
        best_a = a.copy()
    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, verbose=True):
    """Coordinate descent with incremental autocorrelation updates."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    b = fast_autocorr(a)
    t0 = time.monotonic()
    sweep = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        # Random order for diversity
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            current_val = a[i]
            if current_val < 1e-12:
                continue
            # Try multiplicative changes
            for factor in [0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = current_val * factor
                if new_val < 0:
                    continue
                delta = new_val - current_val
                # Update autocorrelation incrementally
                # b_k = b_k + delta * (a[i+k] + a[i-k]) + delta^2 for k=0
                new_b = b.copy()
                for k in range(2 * n - 1):
                    contrib = 0.0
                    if i + k < n:
                        contrib += a[i + k]
                    if i - k >= 0:
                        contrib += a[i - k]
                    new_b[k] = b[k] + delta * contrib
                    if k == 0:
                        new_b[k] += delta * delta
                new_peak = new_b.max()
                # Compare with current peak
                old_peak = b.max()
                if new_peak < best_score * (s / norm_val) ** 2 - 1e-12:
                    a[i] = new_val
                    # Renormalize
                    s_new = a.sum()
                    if s_new < 1e-15:
                        a[i] = current_val
                        continue
                    a = a * norm_val / s_new
                    b = fast_autocorr(a)
                    sc = b.max()
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
                    break
        # Renormalize after sweep
        s = a.sum()
        if s > 1e-15:
            a = a * norm_val / s
        b = fast_autocorr(a)
        sc = b.max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDescent sweep={sweep} ({elapsed:.0f}s): best={best_score:.8f}")
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    # === Phase 1: Subgradient FISTA warm-up ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.10, 120.0)
    if t1 > 20:
        print(f"\n=== Phase 1: SubGrad-FISTA n={n_base} (budget={t1:.0f}s) ===")
        a_out, _ = subgrad_fista(best_seq, t1, verbose=True)
        try_update(a_out, "subgrad_fista")

    # === Phase 2: LP-Nesterov (main) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = min(remaining * 0.45, 700.0)
    if t2 > 60:
        print(f"\n=== Phase 2: LP-Nesterov n={n_cur} (budget={t2:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lp_nesterov_improved(a_norm, t2, solver, verbose=True)
        try_update(a_out, "lp_nesterov_main")

    # === Phase 3: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 30:
        print(f"\n=== Phase 3: CoordDescent n={n_cur} (budget={t3:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = coordinate_descent_polish(a_norm, t3, verbose=True)
        try_update(a_out, "coord_descent")

    # === Phase 4: Local perturbation search ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 30:
        print(f"\n=== Phase 4: Local Perturbation n={n_cur} (budget={t4:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = local_perturbation_search(a_norm, t4, solver, verbose=True)
        try_update(a_out, "local_perturb")

    # === Phase 5: LP polish after all local methods ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 30:
        print(f"\n=== Phase 5: LP Polish n={n_cur} (budget={t5:.0f}s) ===")
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = lp_nesterov_improved(a_norm, t5, solver, verbose=True, restart_interval=300)
        try_update(a_out, "lp_polish")

    # === Phase 6: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_cur = len(best_seq)
        lengths_to_try = []
        for dn in [-200, -100, 100, 200, 300]:
            nl = n_cur + dn
            if nl >= 400 and nl <= 3000:
                lengths_to_try.append(nl)
        print(f"\n=== Phase 6: Multi-length exploration (budget={remaining:.0f}s) ===")
        for nl in lengths_to_try:
            if time.monotonic() - t_global > time_budget - 30:
                break
            t_interp = min(remaining * 0.3, 120.0)
            if t_interp < 20:
                break
            print(f"  Trying n={nl}...")
            init_long = interpolate_sequence(best_seq, nl)
            a_norm = init_long * np.sqrt(2.0 * nl) / init_long.sum()
            a_out, _ = lp_nesterov_improved(a_norm, t_interp, solver, verbose=False, restart_interval=300)
            try_update(a_out, f"lp_n{nl}")
            # Also try coordinate descent on the best
            if time.monotonic() - t_global < time_budget - 30:
                t_cd = min(30.0, (time_budget - (time.monotonic() - t_global)) * 0.3)
                if t_cd > 10:
                    a_cd, _ = coordinate_descent_polish(a_norm if a_out is None else a_out, t_cd, verbose=False)
                    try_update(a_cd, f"cd_n{nl}")

    # === Phase 7: Final LP if time remains ===
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30:
        n_cur = len(best_seq)
        t7 = min(remaining * 0.5, 100.0)
        if t7 > 20:
            print(f"\n=== Phase 7: Final LP n={n_cur} (budget={t7:.0f}s) ===")
            a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
            a_out, _ = lp_nesterov_improved(a_norm, t7, solver, verbose=False, restart_interval=500)
            try_update(a_out, "lp_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
