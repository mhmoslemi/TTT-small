1. Evaluate `f(delta)` at several candidate `delta` values.
2. Use bisection to find the minimizer.

def coordinate_descent(a0, time_limit, rng, verbose=False):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(a0.astype(np.float64).copy(), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    passes = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        perm = rng.permutation(n)

        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            ai = a[i]
            if ai < 1e-12 * norm:
                continue

            # Gradient direction: g[k] = 2*a[k-i] for valid k
            lo = max(0, i)
            hi = min(size, i + n)
            g = np.zeros(size)
            g[lo:hi] = 2.0 * a[lo - i:hi - i]

            # Candidate deltas
            k_star = np.argmax(conv)
            g_at_kstar = g[k_star]
            if g_at_kstar > 0:
                d_sub = -ai * 0.5 / g_at_kstar
            else:
                d_sub = -ai * 0.5

            candidates = [d_sub, d_sub * 0.5, d_sub * 0.25, -d_sub * 0.25, -ai * 0.5]
            best_d = 0.0
            best_val = conv.max()

            for d in candidates:
                if d == 0:
                    continue
                if ai + d < -1e-12:
                    continue
                new_conv = conv + d * g
                new_conv = np.maximum(new_conv, 0.0)
                # Add quadratic at k=2i
                if 2 * i < size:
                    new_conv[2 * i] += d * d
                nm = new_conv.max()
                if nm < best_val - 1e-14:
                    best_val = nm
                    best_d = d

            if best_d != 0.0:
                a[i] = max(0.0, ai + best_d)
                s = a.sum()
                if s < 1e-15:
                    a[i] = ai
                    continue
                a = a * norm / s
                conv = fast_autocorr(a)
                cur = conv.max()
                if cur < best_score - 1e-14:
                    best_score = cur
                    best_a = a.copy()
                    improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0

        if verbose and passes % 20 == 0:
            print(f"    CD pass={passes} ({time.monotonic()-t0:.0f}s): best={best_score:.10f}")

    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def perturb_seq(a, rng, scale=0.05):

    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.1)
    a_pert = a * noise
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s

    n_cur = len(best_seq)
    print(f"\n=== Phase 2: CD n={n_cur} ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = coordinate_descent(a_norm, t2, rng, verbose=True)
        do_update(a_out, f"cd1_n{n_cur}")

    # Phase 3: Perturb + LP restarts
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Perturb+LP restarts ===")
    t3_total = min(remaining * 0.20, 300.0)
    for r in range(3):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 80:
            break
        t3 = min(t3_total / 3.0, remaining - 60)
        if t3 < 25:
            break
        n_c = len(best_seq)
        pert = perturb_seq(best_seq, rng, scale=0.03 + 0.02 * r)
        a_norm = pert * np.sqrt(2.0 * n_c) / pert.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=False, seed=100 + r)
        do_update(a_out, f"pert{r}_n{n_c}")

    # Phase 4: Longer length
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq) + 400
    print(f"\n=== Phase 4: LP n={n2} ===")
    t4 = min(remaining * 0.15, 240.0)
    if t4 > 30:
        init_long = interpolate_seq(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, verbose=True, seed=7)
        do_update(a_out, f"lp_n{n2}")

    # Phase 5: CD on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 5: CD n={n_cur} ===")
    t5 = min(remaining * 0.12, 180.0)
    if t5 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = coordinate_descent(a_norm, t5, rng, verbose=True)
        do_update(a_out, f"cd2_n{n_cur}")

    # Phase 6: FW corrected
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: FW n={n_cur} ===")
    t6 = min(remaining * 0.10, 120.0)
    if t6 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = frank_wolfe_corrected(a_norm, t6, verbose=False)
        do_update(a_out, f"fw_n{n_cur}")

    # Phase 7: Final LP
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: Final LP n={n_cur} ===")
    t7 = min(remaining * 0.15, 150.0)
    if t7 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t7, solver, verbose=True, seed=99)
        do_update(a_out, f"lp_final_n{n_cur}")

    # Phase 8: Final CD
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 8: Final CD n={n_cur} ===")
    t8 = min(remaining * 0.20, 150.0)
    if t8 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = coordinate_descent(a_norm, t8, rng, verbose=True)
        do_update(a_out, f"cd_final_n{n_cur}")

    # Phase 9: Try shorter length
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        n_short = max(800, len(best_seq) - 400)
        print(f"\n=== Phase 9: LP n={n_short} ===")
        t9 = min(remaining * 0.3, 80.0)
        if t9 > 15:
            init_short = interpolate_seq(best_seq, n_short)
            a_norm = init_short * np.sqrt(2.0 * n_short) / init_short.sum()
            a_out, _ = nesterov_lp_optimize(a_norm, t9, solver, verbose=False, seed=55)
            do_update(a_out, f"lp_n{n_short}")

    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, t={time.monotonic()-t_global:.0f}s")
    else:
        print("ERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
