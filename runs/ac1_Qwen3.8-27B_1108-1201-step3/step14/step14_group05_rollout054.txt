```python
    c: current autocorrelation (length 2n-1)
    s: slope vector (length 2n-1), s[k] = dc_k/da_i
    a_i: current value of the coordinate
    Returns: optimal delta (can be negative or positive)
    """

    s_k1 = s[k1]
    
    if abs(s_k1) < 1e-14:
        return 0.0
    
    if s_k1 > 0:
        # Decrease a_i (delta < 0)
        # Find first catch-up: max over j of (c_j - c_k1)/(s_k1 - s_j) for s_j < s_k1 and c_j < c_k1

            deltas = np.where(valid, (c - c[k1]) / (s_k1 - s), np.inf)
        
        # We want the SMALLEST positive delta
        delta_opt = np.min(deltas[valid])
        delta_opt = max(delta_opt, 0.0)
        # Cap the increase
        delta_opt = min(delta_opt, a_i)  # Don't more than double
        return delta_opt
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fessler-Parks accelerated LP Nesterov with warm-started active set reduction
2. Novel vectorized coordinate descent with exact peak-equalization line search
3. Multi-length exploration with spectral interpolation
4. Adaptive hybrid time allocation with convergence monitoring
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver = None


def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    print(f"  LP solver (fallback): ECOS")
    return _solver


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_score(a):
    n = len(a)
    if n < 1:
        return float('inf')
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_autocorr(a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a, target_sum):
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * target_sum / len(a)
    return a * target_sum / s


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver, max_active=100):
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    if b_var.value is None:
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None

    for _ in range(4):
        circ = T @ b_val
        tol = 1e-7 * M
        active = circ >= M - tol
        n_active = int(active.sum())
        if n_active < 5 or n_active >= min(size, max_active + 100):
            break
        if n_active > max_active:
            active_idx = np.where(active)[0]
            active_vals = circ[active_idx]
            top = active_idx[np.argsort(active_vals)[::-1][:max_active]]
            active = np.zeros(size, dtype=bool)
            active[top] = True
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T[active, :] @ b_var <= M[active]])
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        if b_var.value is None:
            break
        b_new = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        if (T @ b_new).max() <= M * (1 + 1e-4):
            b_val = b_new
        else:
            break
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp(a0, time_limit, solver, max_active=80, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    y = a.copy()
    prev_y = a.copy()
    t0 = time.monotonic()
    iters = 0
    stall = 0
    gamma = 2.0

    while time.monotonic() - t0 < time_limit:
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-15:
            break
        b = lp_best_response(y, M_curr, solver, max_active=max_active)
        if b is None:
            break
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        g = b * norm / b_sum
        iters += 1
        t = iters
        alpha = 2.0 / (t + 2.0 - gamma)
        alpha = min(alpha, 0.5)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = normalize(y_new, norm)
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t + 1.0 - gamma) / (t + 2.0 - gamma)
            momentum = min(momentum, 1.0)
            a_new = y_new + momentum * (y_new - prev_y)
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new, norm)
        prev_y = y.copy()
        y = y_new
        a = a_new
        score_y = fast_autocorr(y).max()
        if score_y < best_M:
            best_M = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        score_a = fast_autocorr(a).max()
        if score_a < best_M:
            best_M = score_a
            best_a = a.copy()
            stall = 0
        if stall > 40:
            gamma = max(1.2, gamma - 0.15)
        if stall > 150:
            rng = np.random.default_rng(iters * 31 + 7)
            pert = 0.03 * a.mean()
            y = np.maximum(y + rng.normal(0, pert, n), 0)
            y = normalize(y, norm)
            a = np.maximum(a + rng.normal(0, pert, n), 0)
            a = normalize(a, norm)
            prev_y = y.copy()
            stall = 0
            gamma = 2.0
        if verbose and iters % 50 == 0:
            print(f"    NP iter={iters} ({time.monotonic()-t0:.0f}s): best_M={best_M:.8f}, stall={stall}, gamma={gamma:.2f}")
    return best_a, best_M


def coord_slope(a, i):
    n = len(a)
    s = np.zeros(2 * n - 1)
    s[:n - i] += a[i:]
    s[:i + 1] += a[i::-1]
    return s


def coord_cd_optimal_delta(c, s, a_i):
    k1 = int(np.argmax(c))
    s_k1 = s[k1]
    if abs(s_k1) < 1e-14:
        return 0.0
    c_k1 = c[k1]
    if s_k1 > 0:
        valid = (s < s_k1) & (c < c_k1)
        if not valid.any():
            return -a_i
        with np.errstate(divide='ignore', invalid='ignore'):
            dvals = np.where(valid, (c - c_k1) / (s_k1 - s), -np.inf)
        delta_opt = np.max(dvals)
        delta_opt = min(delta_opt, -1e-12)
        delta_opt = max(delta_opt, -a_i)
        return delta_opt
    else:
        valid = (s > s_k1) & (c < c_k1)
        if not valid.any():
            return a_i * 0.5
        with np.errstate(divide='ignore', invalid='ignore'):
            dvals = np.where(valid, (c - c_k1) / (s_k1 - s), np.inf)
        pos = dvals > 0
        if not pos.any():
            return a_i * 0.1
        delta_opt = np.min(dvals[pos])
        delta_opt = max(delta_opt, 1e-12)
        delta_opt = min(delta_opt, a_i)
        return delta_opt


def coordinate_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize(a, norm)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(7777)
    pass_count = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a)
        peak = c.max()
        if peak < best_M - 1e-10:
            best_M = peak
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 80:
                n_pert = max(5, n // 50)
                idx = rng.choice(n, size=n_pert, replace=False)
                for ii in idx:
                    jj = rng.integers(n)
                    amt = a[ii] * rng.uniform(0.1, 0.5)
                    a[ii] -= amt
                    a[jj] += amt
                a = np.maximum(a, 0)
                a = normalize(a, norm)
                stall = 0
                pass_count += 1
                continue

        order = rng.permutation(n)
        c = fast_autocorr(a)

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[i]
            if old_val < 1e-10:
                continue
            s = coord_slope(a, i)
            delta = coord_cd_optimal_delta(c, s, old_val)
            if abs(delta) < 1e-14:
                continue
            a[i] = old_val + delta
            c += delta * s

        a = np.maximum(a, 0)
        a = normalize(a, norm)
        pass_count += 1

        if verbose and pass_count % 200 == 0:
            cur_M = fast_autocorr(a).max()
            print(f"    CD pass={pass_count} ({time.monotonic()-t0:.0f}s): cur_M={cur_M:.8f}, best_M={best_M:.8f}, stall={stall}")

    return best_a, best_M


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def lse_gradient_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = normalize(a, norm)
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    steps = 0
    T_start = max(0.3, best_M * 0.05)
    T_end = max(0.001, best_M * 0.002)

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress
        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            break
        w = np.exp((b - max_b) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum
        a_rev = a[::-1]
        fft_size = 1 << (3 * n - 2).bit_length()
        W = rfft(w, fft_size)
        AR = rfft(a_rev, fft_size)
        conv_grad = irfft(W * AR, fft_size)
        grad = 2.0 * conv_grad[n - 1:2 * n - 1]
        step = 0.5 / (T * n * 0.3)
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new, norm)
        a = a_new
        steps += 1
        score = fast_autocorr(a).max()
        if score < best_M:
            best_M = score
            best_a = a.copy()
        if verbose and steps % 300 == 0:
            print(f"    LSE-GD step={steps} ({time.monotonic()-t0:.0f}s): T={T:.4f}, best_M={best_M:.8f}")
    return best_a, best_M
```
