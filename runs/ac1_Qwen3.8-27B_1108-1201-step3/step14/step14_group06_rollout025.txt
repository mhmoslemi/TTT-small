def lse_objective_and_grad(a_flat, n, T_temp):
    a = np.maximum(a_flat.reshape(n), 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-15:
        return 1e10, np.zeros(n)
    shifted = T_temp * (b - max_b)
    shifted = np.minimum(shifted, 500.0)
    exp_vals = np.exp(shifted)
    log_sum = max_b + np.log(exp_vals.sum()) / T_temp
    w = exp_vals / exp_vals.sum()

    fft_size = 1 << (2 * n - 1).bit_length()
    w_full = np.zeros(fft_size)
    w_full[:2 * n - 1] = w
    a_full = np.zeros(fft_size)
    a_full[:n] = a

    W_fft = rfft(w_full, fft_size)
    A_fft = rfft(a_full, fft_size)
    cross = irfft(W_fft * np.conj(A_fft), fft_size)[:n]

    obj_val = 2.0 * n * log_sum / (s * s)
    grad = 2.0 * n * cross / (s * s) - 2.0 * obj_val * a / s
    return obj_val, grad

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, use_tight_M=True):

    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)

    n_iters = 0
    last_improve = time.monotonic()
    stall_count = 0
    perturbation_count = 0

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        b_y = fast_autocorr(y)
        M_curr = b_y.max()
        if M_curr < 1e-12:
            break

        if use_tight_M:
            progress = min(1.0, n_iters / 200.0)
            eps = 0.01 + 0.04 * progress
            M_used = M_curr * (1.0 - eps)
        else:
            M_used = M_curr

        T = build_constraint_matrix(y, n)
        b = lp_best_response(y, M_used, solver, T)
        if b is None:
            b = lp_best_response(y, M_curr, solver, T)
            if b is None:
                break

        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        g = b * norm_val / b_sum
        n_iters += 1

        M_new = fast_autocorr(g).max()
        alpha = 2.0 / (t + 2.0)
        if M_curr > 0 and M_new > 0:
            ratio = min(M_new / M_curr, 1.5)
            alpha = min(alpha / max(ratio, 0.1), 1.0)
            alpha = max(alpha, 0.01)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_val / s_y

            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:

            else:

        a = a_new

def gradient_coord_descent(a0, time_limit, seed=42):

        top_contrib = sorted_idx[:20]

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    choice = rng.integers(0, 7)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.3) ** 2) + 0.5 * np.exp(-4.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.5) ** 2)
    elif choice == 5:
        # Asymmetric bump
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-5.0 * (x - 0.4) ** 2) + 0.3 * np.exp(-5.0 * (x - 0.8) ** 2)
    else:
        # Two well-separated bumps (to reduce center convolution peak)
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-30.0 * (x - 0.25) ** 2) + np.exp(-30.0 * (x - 0.75) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def attempt_update(best_state, candidate_a, label, t_global):
    best_seq, best_score = best_state
    if candidate_a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(candidate_a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function combining LP, gradient, and coordinate descent methods."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LP Nesterov with adaptive tight M (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 1: LP-Nest adaptive n={n_best} (budget={remaining:.0f}s) ===")
    t1 = remaining * 0.45
    a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
    a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True, use_tight_M=True)
    best_seq, best_score = attempt_update((best_seq, best_score), a_out, f"lp_n{n_best}", t_global)

    # Phase 2: LSE gradient optimization
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: LSE gradient n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 180.0)
    if t2 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_scipy_optimize(a_norm, t2, T_start=60.0, T_end=2.0, n_stages=4)
        best_seq, best_score = attempt_update((best_seq, best_score), a_out, f"lse_n{n_best}", t_global)

    # Phase 3: Gradient coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Coord descent n={n_best} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 250.0)
    if t3 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = gradient_coord_descent(a_norm, t3, seed=42)
        best_seq, best_score = attempt_update((best_seq, best_score), a_out, f"coord1_n{n_best}", t_global)

    # Phase 4: Multi-peak coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Multi-peak coord n={n_best} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 180.0)
    if t4 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = multi_peak_coord_descent(a_norm, t4, seed=999)
        best_seq, best_score = attempt_update((best_seq, best_score), a_out, f"mpc1_n{n_best}", t_global)

    # Phase 5: LP Nesterov round 2
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LP-Nest round 2 n={n_best} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t5, solver, verbose=False, use_tight_M=True)
        best_seq, best_score = attempt_update((best_seq, best_score), a_out, f"lp_r2_n{n_best}", t_global)

    # Phase 6: Try a perturbation restart
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Perturbation + LP n={n_best} (budget={remaining:.0f}s) ===")
    if remaining > 60:
        t6a = remaining * 0.3
        t6b = remaining * 0.3
        t6c = remaining * 0.4

        # Perturb and run short LP
        rng2 = np.random.default_rng(12345)
        for pert_idx in range(3):
            if time.monotonic() - t_global > time_budget - 60:
                break
            noise = rng2.normal(0, 0.03 * (pert_idx + 1), n_best)
            a_pert = best_seq + noise
            a_pert = np.maximum(a_pert, 0.0)
            s_p = a_pert.sum()
            if s_p < 1e-10:
                continue
            a_pert = a_pert * np.sqrt(2.0 * n_best) / s_p

            remaining6 = time_budget - (time.monotonic() - t_global) - 60
            if remaining6 < 30:
                break
            t_pert = min(remaining6 * 0.25, 80.0)
            a_out, _ = nesterov_lp_optimize(a_pert, t_pert, solver, verbose=False, use_tight_M=True)
            best_seq, best_score = attempt_update((best_seq, best_score), a_out, f"pert{pert_idx+1}", t_global)

        # Short coord descent on perturbed result
        remaining6 = time_budget - (time.monotonic() - t_global) - 60
        if remaining6 > 20:
            a_norm = best_seq * np.sqrt(2.0 * len(best_seq)) / best_seq.sum()
            a_out, _ = gradient_coord_descent(a_norm, min(remaining6 * 0.5, 60.0), seed=777)
            best_seq, best_score = attempt_update((best_seq, best_score), a_out, "coord_pert", t_global)

    # Phase 7: Final LSE polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final LSE n={n_best} (budget={remaining:.0f}s) ===")
    if remaining > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_scipy_optimize(a_norm, min(remaining * 0.5, 80.0), T_start=30.0, T_end=1.0, n_stages=3)
        best_seq, best_score = attempt_update((best_seq, best_score), a_out, "lse_final", t_global)

    # Phase 8: Final coord descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final coord n={n_best} (budget={remaining:.0f}s) ===")
    if remaining > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = gradient_coord_descent(a_norm, min(remaining * 0.8, 60.0), seed=31415)
        best_seq, best_score = attempt_update((best_seq, best_score), a_out, "coord_final", t_global)

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
