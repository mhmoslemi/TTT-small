"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated LogSumExp (LSE) gradient descent with adaptive temperature and backtracking
2. LP best-response refinement via cvxpy
3. Coordinate descent polish with greedy moves
4. Multi-length exploration with interpolation transfer
Key improvement: proper Nesterov momentum on LSE (previous used vanilla GD), 
adaptive step size via backtracking, and smarter time allocation between phases.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft


_global_solver = None


def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print(f"  LP solver (fallback): ECOS")
    return _global_solver


def fast_autocorr(a):
    """Compute full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_gradient(a, T):
    """
    Compute gradient of LSE-smoothed max autocorrelation.
    f(a) = T * log(sum_k exp(b[k]/T)) where b = conv(a,a)
    grad[j] = 2 * sum_m p[j+m] * a[m], where p[k] = softmax(b[k]/T)
    """
    n = len(a)
    b = fast_autocorr(a)
    
    if T < 1e-10:
        p_idx = int(np.argmax(b))
        grad = np.zeros(n)
        js = max(0, p_idx - n + 1)
        je = min(n - 1, p_idx)
        if js <= je:
            grad[js:je + 1] = 2.0 * a[p_idx - np.arange(js, je + 1)]
        return grad
    
    # Compute softmax weights
    b_shifted = b - b.max()
    p = np.exp(b_shifted / T)
    p_sum = p.sum()
    if p_sum < 1e-30:
        p = np.zeros(len(b))
        p[int(np.argmax(b))] = 1.0
    else:
        p = p / p_sum
    
    # grad[j] = 2 * sum_{m=0}^{n-1} p[j+m] * a[m]
    # = 2 * conv(flip(a), p)[n-1 : 2n-1]
    a_flip = a[::-1]
    fft_size = 1 << (3 * n - 2).bit_length()
    A_f = fft(a_flip.astype(np.complex128), fft_size)
    P = fft(p.astype(np.complex128), fft_size)
    cross = ifft(A_f * P, fft_size).real
    grad = 2.0 * cross[n - 1:2 * n - 1]
    
    return grad


def project_nonneg_sum(a, norm_val):
    """Project onto {a >= 0, sum(a) = norm_val}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), norm_val / len(a))
    return a * norm_val / s


def nesterov_lse_optimize(a_init, time_limit, verbose=True):
    """
    Nesterov-accelerated LSE gradient descent with adaptive temperature.
    Uses backtracking line search for step size.
    """
    n = len(a_init)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    a_prev = a.copy()
    y = a.copy()
    t = 0
    t0 = time.monotonic()
    
    # Temperature schedule: exponential cooling
    b_init = fast_autocorr(a)
    max_b_init = b_init.max()
    T_start = max_b_init * 0.12
    T_end = max_b_init * 0.002
    
    # Step size
    eta = 1.0 / (n * 0.5)
    eta_min = 1e-12
    eta_max = 1.0 / (n * 0.01)
    
    stall = 0
    stall_threshold = 3000
    n_restarts = 0
    max_restarts = 30
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress
        
        # Compute gradient at y
        grad = lse_gradient(y, T)
        # Project onto tangent space of sum constraint
        grad = grad - np.mean(grad)
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Backtracking line search
        f_y = None  # lazy compute
        a_trial = project_nonneg_sum(y - eta * grad, norm_val)
        
        # Simple acceptance: check if exact score improved
        score_trial = fast_autocorr(a_trial).max()
        
        backtracks = 0
        while score_trial > best_score * (1 + 1e-4) + 1e-10 and backtracks < 15:
            eta = max(eta * 0.5, eta_min)
            a_trial = project_nonneg_sum(y - eta * grad, norm_val)
            score_trial = fast_autocorr(a_trial).max()
            backtracks += 1
        
        if backtracks >= 15:
            # Too many backtracks, perturb and reset
            eta = max(eta_max * 0.1, eta_min)
            a_trial = a.copy()
            score_trial = best_score
        
        a_new = a_trial
        score_new = fast_autocorr(a_new).max()
        
        # Nesterov momentum update
        a_x = a.copy()
        a = a_new
        if t >= 2:
            momentum = (t - 1.0) / (t + 1.0)
            y = a + momentum * (a - a_x)
            y = project_nonneg_sum(y, norm_val)
        else:
            y = a.copy()
        
        a_prev = a_x
        
        # Track best
        if score_new < best_score:
            best_score = score_new
            best_a = a.copy()
            stall = 0
            # Slowly increase eta if making progress
            eta = min(eta * 1.02, eta_max)
        else:
            stall += 1
        
        # Restart if stalled
        if stall > stall_threshold and n_restarts < max_restarts:
            n_restarts += 1
            rng = np.random.default_rng(t * 7 + n_restarts)
            perturb_scale = 0.02 * a.mean()
            a = np.maximum(a + rng.normal(0, perturb_scale, n), 0)
            a = project_nonneg_sum(a, norm_val)
            y = a.copy()
            a_prev = a.copy()
            eta = 1.0 / (n * 0.5)
            stall = 0
            # Also boost temperature temporarily
            stall_threshold = int(stall_threshold * 1.3)
        
        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NAG-LSE t={t} ({elapsed:.0f}s): best={best_score:.8f}, T={T:.5f}, eta={eta:.2e}, stall={stall}")
    
    return best_a, best_score


def build_conv_matrix(a):
    """Build the convolution matrix T where (T@b)[k] = conv(a,b)[k]."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver, max_active=200):
    """
    Solve LP: max sum(b) s.t. T@b <= M, b >= 0.
    With constraint reduction to active set.
    """
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    
    # Use active constraints for efficiency
    circ = T @ a
    active = circ >= M * 0.5
    n_active = int(active.sum())
    
    if n_active > max_active:
        active_idx = np.where(active)[0]
        active_vals = circ[active_idx]
        top_idx = active_idx[np.argsort(active_vals)[::-1][:max_active]]
        active = np.zeros(size, dtype=bool)
        active[top_idx] = True
    
    constraints = [T[active, :] @ b_var <= M[active]]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    if b_var.value is None:
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    
    # Verify feasibility
    circ_check = T @ b_val
    if circ_check.max() > M * (1 + 1e-4):
        return None
    
    return b_val


def lp_nesterov_refine(a_init, time_limit, solver, verbose=True, max_active=120):
    """
    Fessler-Parks accelerated LP Nesterov refinement.
    """
    n = len(a_init)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    
    y = a.copy()
    prev_y = a.copy()
    t = 0
    t0 = time.monotonic()
    stall = 0
    gamma = 1.7
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        
        b = lp_best_response(y, M_curr, solver, max_active=max_active)
        if b is None:
            stall += 100
            if stall > 500:
                break
            continue
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        
        g = b * norm_val / b_sum
        
        # Fessler-Parks step
        alpha = 2.0 / (t + 2.0 - gamma)
        alpha = min(alpha, 0.5)
        
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_val / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t + 1.0 - gamma) / (t + 2.0 - gamma)
            momentum = min(momentum, 1.0)
            a_new = y_new + momentum * (y_new - prev_y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm_val / s_a
        
        prev_y = y.copy()
        y = y_new
        a = a_new
        
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        
        # Adaptive gamma
        if stall > 200:
            gamma = max(1.0, gamma - 0.1)
        
        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-NAG t={t} ({elapsed:.0f}s): best={best_score:.8f}, stall={stall}")
        
        if stall > 500:
            break
    
    return best_a, best_score

        for j in order:

            b = fast_autocorr(a)
            max_b = b.max()
            p_star = int(np.argmax(b))

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving relative shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

    print(f"\n=== Phase 3: Nesterov LSE r2 n={n_cur} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 280.0)
    if t3 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, sc = nesterov_lse_optimize(a_norm, t3, verbose=True)
        do_try_update(a_out, f"nlse2_n{n_cur}")
    
    # === Phase 4: Coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 4: CoordDesc n={n_cur} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, sc = coordinate_polish(a_norm, t4, seed=777, verbose=True)
        do_try_update(a_out, f"coord_n{n_cur}")
    
    # === Phase 5: Try longer length ===
    n2 = n_base + 100
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Nesterov LSE n={n2} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.25, 300.0)
    if t5 > 30:
        init_long = interpolate_sequence(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, sc = nesterov_lse_optimize(a_norm, t5, verbose=True)
        do_try_update(a_out, f"nlse_n{n2}")
    
    # === Phase 6: LP refinement on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 6: LP r2 n={n_cur} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.25, 250.0)
    if t6 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, sc = lp_nesterov_refine(a_norm, t6, solver, verbose=True, max_active=80)
        do_try_update(a_out, f"lp2_n{n_cur}")
    
    # === Phase 7: Final coordinate polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    print(f"\n=== Phase 7: CoordDesc final n={n_cur} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.25, 200.0)
    if t7 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, sc = coordinate_polish(a_norm, t7, seed=31415, verbose=True)
        do_try_update(a_out, f"coord2_n{n_cur}")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]
    
    return [float(x) for x in best_seq]
