import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_autocorr(a):
    """Compute linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_grad_lse(a, w):
    """Compute gradient of LSE objective using FFT.
    
    Objective: T * log(sum(exp(c[k]/T))) where c = conv(a,a)
    Gradient w.r.t. a[j] = 2 * sum_k w[k] * dc[k]/da[j]
    
    Using the FFT trick: conv(w, a[::-1])[n-1 + j] = sum_k w[k] * a[k-j]
    """
    n = len(a)
    w = np.asarray(w, dtype=np.float64)
    a_rev = a[::-1].copy()
    fft_size = 1 << (len(w) + n - 1 - 1).bit_length()
    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)
    conv_result = irfft(W * AR, fft_size)
    # We want conv(w, a_rev)[j + n - 1] for j = 0, ..., n-1
    grad = 2.0 * conv_result[n-1:2*n-1]
    return grad

def fast_autocorr(a):
    """Compute linear autocorrelation conv(a,a) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def next_pow2(m):
    return 1 << (m - 1).bit_length()


def project_simplex(v, s):
    """Project v onto {x >= 0, sum(x) = s} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - s))[0]
    if len(rho_candidates) == 0:
        return np.full(n, s / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """Compute gradient of LSE objective: grad[i] = 2 * sum_m a[m]*w[m+i]."""
    n = len(a)
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    cross = irfft(A * np.conj(W), fft_size)
    return 2.0 * cross[:n]


def lse_gd_optimize(a0, time_limit, verbose=True, seed=42):
    """FFT-based LSE gradient descent with Nesterov momentum and adaptive temperature."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    c_init = fast_autocorr(a)
    best_obj = 2.0 * n * c_init.max()
    M0 = c_init.max()

    fft_size = next_pow2(2 * n - 1)
    v = np.zeros(n)
    beta = 0.85

    T_start = M0 * 0.4
    T_end = M0 * 0.003
    t0 = time.monotonic()
    step = 0
    stagnation = 0
    last_best = best_obj

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** progress
        T = max(T, M0 * 0.001)

        # Compute autocorrelation and weights
        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < 1e-18:
            break

        w = np.exp((c - c_max) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c))] = 1.0
        else:
            w /= w_sum

        # Compute gradient via FFT
        grad = compute_lse_grad(a, w, fft_size)

        # Project gradient to tangent space of {sum = 1}
        grad -= np.mean(grad)

        # Adaptive step size with backtracking
        grad_norm_sq = np.dot(grad, grad)
        if grad_norm_sq < 1e-30:
            break
        lr = 0.5 / grad_norm_sq
        lr = min(lr, 1.0 / (n * T * 0.5))

        # Nesterov momentum step
        v = beta * v - lr * grad
        a_new = a + v
        a_new = project_simplex(a_new, 1.0)

        # Evaluate
        c_new = fast_autocorr(a_new)
        obj_new = 2.0 * n * c_new.max()

        if obj_new < best_obj:
            if obj_new < last_best * (1.0 - 1e-6):
                last_best = obj_new
                stagnation = 0
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Backtrack: reduce step
            v *= 0.3
            stagnation += 1

        step += 1

        # Restart if stagnating
        if stagnation > 2000:
            # Increase temperature and perturb
            perturbation = 0.02 * a.mean()
            a = np.maximum(a + rng.normal(0, perturbation, n), 0)
            a = project_simplex(a, 1.0)
            v = np.zeros(n)
            beta = 0.85
            stagnation = 0
            last_best = 2.0 * n * fast_autocorr(a).max()

        if verbose and step % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE step={step} ({elapsed:.0f}s): T={T:.6f}, obj={obj_new:.8f}, best={best_obj:.8f}")

    return best_a, best_obj


def build_conv_matrix_sparse(a):
    """Build sparse matrix T where (T @ b)[k] = conv(a,b)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_solve(a, M, solver, max_constr=300):
    """Solve Fessler-Parks LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    T = build_conv_matrix_sparse(a)
    b_var = cp.Variable(n, nonneg=True)

    # Start with all constraints
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    if b_var.value is None:
        return None

    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None

    # Reduce to active constraints
    for _ in range(4):
        circ = T @ b_val
        tol = 1e-6 * M
        active = circ >= M - tol
        n_active = int(active.sum())

        if n_active < 10 or n_active >= max_constr:
            break

        if n_active > max_constr:
            active_idx = np.where(active)[0]
            active_vals = circ[active_idx]
            top_idx = active_idx[np.argsort(active_vals)[::-1][:max_constr]]
            active = np.zeros(len(active), dtype=bool)
            active[top_idx] = True

        prob2 = cp.Problem(cp.Maximize(cp.sum(b_var)), [T[active, :] @ b_var <= M[active]])
        try:
            prob2.solve(solver=solver, verbose=False)
        except Exception:
            break
        if prob2.status not in ('optimal', 'optimal_inaccurate'):
            break
        if b_var.value is None:
            break
        b_new = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        circ_new = T @ b_new
        if circ_new.max() <= M * (1 + 1e-4) and b_new.sum() >= b_sum * 0.99:
            b_val = b_new
            b_sum = b_new.sum()
        else:
            break

    return b_val


def fessler_parks_optimize(a0, time_limit, solver, verbose=True, max_active=200):
    """Fessler-Parks accelerated LP Nesterov iteration."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    prev_y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    gamma = 1.7
    M_hist = []

    while time.monotonic() - t0 < time_limit:
        M_curr = fast_autocorr(y).max()
        M_hist.append(M_curr)
        if M_curr < 1e-12:
            break

        b = lp_solve(y, M_curr, solver, max_constr=max_active)
        if b is None:
            stall += 10
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        # Fessler-Parks step
        t = n_iters + 1
        alpha = 2.0 / (t + 2.0 - gamma)
        alpha = min(alpha, 0.5)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t + 1.0 - gamma) / (t + 2.0 - gamma)
            momentum = min(momentum, 1.0)
            a_new = y_new + momentum * (y_new - prev_y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        prev_y = y.copy()
        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0

        # Adaptive gamma
        if len(M_hist) > 80:
            recent = M_hist[-40:]
            if recent[0] > 1e-15:
                osc = np.sum(np.abs(np.diff(recent))) / (40 * recent[0])
                if osc > 0.04:
                    gamma = max(1.0, gamma - 0.1)
                elif osc < 0.01:
                    gamma = min(2.0, gamma + 0.05)

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}, gamma={gamma:.2f}, stall={stall}")

        # Restart
        if stall > 400:
            rng = np.random.default_rng(n_iters * 7 + 3)
            perturb_scale = 0.02 * a.mean()
            y = np.maximum(y + rng.normal(0, perturb_scale, n), 0)
            y = y * norm / y.sum()
            a = np.maximum(a + rng.normal(0, perturb_scale, n), 0)
            a = a * norm / a.sum()
            prev_y = y.copy()
            stall = 0
            gamma = 1.7
            M_hist = []

    return best_a, best_score


def peak_shave_refine(a0, time_limit, seed=123, verbose=True):
    """Targeted peak shaving with line search."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    n_improve = 0
    n_stall = 0
    total_steps = 0

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        c = fast_autocorr(a)
        max_c = c.max()
        p_star = int(np.argmax(c))

        # Find contributing pairs
        js = max(0, p_star - n + 1)
        je = min(n - 1, p_star)
        idx = np.arange(js, je + 1)
        if len(idx) == 0:
            break
        contrib = a[idx] * a[p_star - idx]

        if contrib.max() < 1e-20:
            n_stall += 1
            if n_stall > 200:
                break
            continue

        src = idx[np.argmax(contrib)]
        src_val = a[src]

        # Find non-contributing destinations
        non_contrib = np.ones(n, dtype=bool)
        non_contrib[js:je + 1] = False
        non_contrib[src] = True
        non_contrib_idx = np.where(non_contrib)[0]

        if len(non_contrib_idx) == 0:
            non_contrib_idx = np.array([i for i in range(n) if i != src])

        # Select destination from low-value non-contributing positions
        n_select = min(20, len(non_contrib_idx))
        sorted_dst = non_contrib_idx[np.argsort(a[non_contrib_idx])][:n_select]
        dst = sorted_dst[rng.integers(len(sorted_dst))]

        # Line search
        best_trial_score = max_c
        best_frac = 0
        for frac in [0.03, 0.07, 0.12, 0.2, 0.35]:
            amt = src_val * frac
            if amt < 1e-18:
                break
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                break
            a_trial = a_trial / s_t
            sc = 2.0 * n * fast_autocorr(a_trial).max()
            if sc < best_trial_score:
                best_trial_score = sc
                best_frac = frac

        total_steps += 1

        if best_frac > 0:
            a_trial = a.copy()
            a_trial[src] -= src_val * best_frac
            a_trial[dst] += src_val * best_frac
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial / s_t
                sc = 2.0 * n * fast_autocorr(a_trial).max()
                if sc < best_score:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    n_improve += 1
                    n_stall = 0
                else:
                    # Annealing acceptance
                    T_anneal = best_score * 0.0005 * (1.0 - elapsed_frac)
                    if T_anneal > 1e-12 and rng.random() < np.exp(-(sc - best_score) / T_anneal):
                        a = a_trial
                    n_stall += 1
        else:
            n_stall += 1

        # Periodic random exploration
        if n_stall > 80:
            n_pert = max(3, n // 80)
            idx_pert = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for idx in idx_pert:
                if a_trial[idx] > 1e-12:
                    amt = a_trial[idx] * rng.uniform(0.1, 0.4)
                    j = rng.integers(n)
                    a_trial[idx] -= amt
                    a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t > 1e-15:
                a_trial = a_trial / s_t
                sc = 2.0 * n * fast_autocorr(a_trial).max()
                if sc < best_score * 1.01:
                    a = a_trial
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
                        n_improve += 1
            n_stall = 0

        if verbose and n_improve > 0 and n_improve % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave improve={n_improve} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def combined_optimize(a0, time_limit, solver, verbose=True, seed=42):
    """Combined LSE GD + LP refinement in one loop."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    M0 = fast_autocorr(a).max()

    fft_size = next_pow2(2 * n - 1)
    v = np.zeros(n)
    beta = 0.85
    T_start = M0 * 0.35
    T_end = M0 * 0.002
    step = 0
    stagnation = 0
    lp_interval = max(3, int(60.0 / max(0.001, (time_limit / max(1, int(time_limit / 60))))))

    rng = np.random.default_rng(seed)
    last_lp_time = 0.0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)

        # Periodic LP refinement every ~60 seconds
        if elapsed - last_lp_time > 60.0 and time_limit - elapsed > 120.0:
            last_lp_time = elapsed
            # LP refinement
            a_norm = a * np.sqrt(2.0 * n) / a.sum()
            M_curr = fast_autocorr(a_norm).max()
            b = lp_solve(a_norm, M_curr, solver, max_constr=150)
            if b is not None and b.sum() > 1e-15:
                g = b / b.sum()
                # Blend
                blend = 0.3
                a_new = (1.0 - blend) * a + blend * g
                a_new = np.maximum(a_new, 0)
                s_new = a_new.sum()
                if s_new > 1e-15:
                    a_new = a_new / s_new
                    sc_new = 2.0 * n * fast_autocorr(a_new).max()
                    if sc_new < best_score:
                        best_score = sc_new
                        best_a = a_new.copy()
                        a = a_new
                        if verbose:
                            print(f"    [LP] ({elapsed:.0f}s): score={sc_new:.8f}")
                        stagnation = 0
                    else:
                        # Try different blend
                        for bl in [0.1, 0.5, 0.7]:
                            a_try = (1.0 - bl) * a + bl * g
                            a_try = np.maximum(a_try, 0)
                            s_try = a_try.sum()
                            if s_try > 1e-15:
                                a_try = a_try / s_try
                                sc_try = 2.0 * n * fast_autocorr(a_try).max()
                                if sc_try < best_score:
                                    best_score = sc_try
                                    best_a = a_try.copy()
                                    a = a_try
                                    break

                v = np.zeros(n)

        # LSE gradient step
        T = T_start * (T_end / T_start) ** progress
        T = max(T, M0 * 0.001)

        c = fast_autocorr(a)
        c_max = c.max()
        if c_max < 1e-18:
            break

        w = np.exp((c - c_max) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c))] = 1.0
        else:
            w /= w_sum

        grad = compute_lse_grad(a, w, fft_size)
        grad -= np.mean(grad)

        grad_norm_sq = np.dot(grad, grad)
        if grad_norm_sq < 1e-30:
            break
        lr = 0.5 / grad_norm_sq
        lr = min(lr, 2.0 / (n * T))

        v = beta * v - lr * grad
        a_new = a + v
        a_new = project_simplex(a_new, 1.0)

        c_new = fast_autocorr(a_new)
        obj_new = 2.0 * n * c_new.max()

        if obj_new < best_score:
            best_score = obj_new
            best_a = a_new.copy()
            a = a_new
            stagnation = 0
        else:
            v *= 0.3
            stagnation += 1

        step += 1

        # Restart if stagnating
        if stagnation > 3000:
            perturbation = 0.015 * a.mean()
            a = np.maximum(a + rng.normal(0, perturbation, n), 0)
            a = project_simplex(a, 1.0)
            v = np.zeros(n)
            beta = 0.85
            stagnation = 0

        if verbose and step % 5000 == 0:
            print(f"    Combined step={step} ({elapsed:.0f}s): T={T:.6f}, best={best_score:.8f}, stall={stagnation}")

    return best_a, best_score


def propose_candidate():
    """Main search function: multi-phase optimization with LSE GD, LP, and peak shaving."""
    global _best_seq_global, _best_score_global

    t_global = time.monotonic()
    TIME_BUDGET = 1740.0  # leave 60s margin

    solver = get_solver()
    print(f"Solver: {solver}, Time budget: {TIME_BUDGET:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    n_base = 1600
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            n_base = len(best_seq)
            print(f"Parent loaded: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        # Default initialization: smooth bump
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 1.2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Combined LSE+LP at base length (55% of time) ===
    n_cur = len(best_seq)
    remaining = TIME_BUDGET - (time.monotonic() - t_global)
    t1 = remaining * 0.50
    print(f"\n=== Phase 1: Combined LSE+LP n={n_cur} (budget={t1:.0f}s) ===")
    a_out, _ = combined_optimize(best_seq, t1, solver, verbose=True, seed=42)
    try_update(a_out, "combined_p1")

    # === Phase 2: Dedicated LP refinement (20% of time) ===
    remaining = TIME_BUDGET - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t2 = remaining * 0.22
    print(f"\n=== Phase 2: LP Fessler-Parks n={n_cur} (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_cur) / best_seq.sum()
        a_out, _ = fessler_parks_optimize(a_norm, t2, solver, verbose=True, max_active=150)
        try_update(a_out, f"lp_p2_n{n_cur}")

    # === Phase 3: LSE GD at longer length (15% of time) ===
    n_long = n_cur + 200
    remaining = TIME_BUDGET - (time.monotonic() - t_global)
    t3 = remaining * 0.20
    print(f"\n=== Phase 3: LSE GD n={n_long} (budget={t3:.0f}s) ===")
    if t3 > 30:
        init_long = interpolate_sequence(best_seq, n_long)
        a_out, _ = lse_gd_optimize(init_long, t3, verbose=True, seed=777)
        try_update(a_out, f"lse_long_n{n_long}")

    # === Phase 4: LP at longer length (10% of time) ===
    remaining = TIME_BUDGET - (time.monotonic() - t_global)
    n_best = len(best_seq)
    t4 = remaining * 0.15
    print(f"\n=== Phase 4: LP n={n_best} (budget={t4:.0f}s) ===")
    if t4 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = fessler_parks_optimize(a_norm, t4, solver, verbose=True, max_active=100)
        try_update(a_out, f"lp_p4_n{n_best}")

    # === Phase 5: Peak shaving polish (remaining time) ===
    remaining = TIME_BUDGET - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t5 = remaining * 0.9
    print(f"\n=== Phase 5: Peak Shaving n={n_cur} (budget={t5:.0f}s) ===")
    if t5 > 10:
        a_out, _ = peak_shave_refine(best_seq, t5, seed=31415, verbose=True)
        try_update(a_out, f"peak_n{n_cur}")

    # === Final: try a few more lengths ===
    remaining = TIME_BUDGET - (time.monotonic() - t_global)
    if remaining > 60:
        for n_try in [len(best_seq) + 300, len(best_seq) + 600]:
            remaining = TIME_BUDGET - (time.monotonic() - t_global)
            if remaining < 40:
                break
            t_try = min(remaining * 0.4, 80.0)
            print(f"\n=== Extra: n={n_try} (budget={t_try:.0f}s) ===")
            init_try = interpolate_sequence(best_seq, n_try)
            a_out, _ = combined_optimize(init_try, t_try, solver, verbose=False, seed=n_try)
            try_update(a_out, f"extra_n{n_try}")
            # Quick peak shave
            remaining = TIME_BUDGET - (time.monotonic() - t_global)
            if remaining > 30 and len(best_seq) == n_try:
                a_out, _ = peak_shave_refine(best_seq, min(remaining * 0.5, 40.0), seed=n_try + 1, verbose=False)
                try_update(a_out, f"peak_extra_n{n_try}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
