a_new[j] = a[j] * exp(-eta * g[j]) / Z

Where Z = sum_j a[j] * exp(-eta * g[j]).

a_new[j] = a[j] / Z (for j outside the range, where g[j]=0)

This makes sense: we're decreasing a[j] proportionally to a[k*-j], which reduces the contribution to conv[k*].

```python
import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def subgradient_at_peak(a, k):
    n = len(a)
    lo = max(0, k - n + 1)
    hi = min(n, k + 1)
    g = np.zeros(n)
    # g[j] = 2*a[k-j] for j in [lo, hi)
    # a[k-j] for j in [lo, hi): indices are k-lo, k-lo-1, ..., k-hi+1
    # So it's a[k-hi+1 : k-lo+1][::-1]
    idx_start = k - hi + 1
    idx_end = k - lo + 1
    g[lo:hi] = 2.0 * a[idx_start:idx_end][::-1]
    return g

def multiplicative_weights_step(a, g, eta):
    n = len(a)
    log_weights = -eta * g
    # For numerical stability, subtract the max
    log_weights -= log_weights.max()
    weights = np.exp(log_weights)
    a_new = a * weights
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    return a_new / s

def optimize_mw(a0, time_limit, n_lp=1000, verbose=True):
    """Optimize using multiplicative weights with periodic LP."""
    n = len(a0)
    a = a0.copy()
    s = a.sum()
    a = a / s  # Normalize to sum 1
    
    best_a = a.copy()
    best_conv = fast_autocorr(a)
    best_score = best_conv.max()
    t0 = time.monotonic()
    
    eta_0 = 0.5
    tau = 500
    lp_every = 200
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_autocorr(a)
        peak = conv.max()
        if peak < best_score:
            best_score = peak
            best_a = a.copy()
        
        k_star = int(np.argmax(conv))
        g = subgradient_at_peak(a, k_star)
        
        eta = eta_0 / (1.0 + t / tau)
        a = multiplicative_weights_step(a, g, eta)
        
        # Periodic LP correction
        if t % lp_every == 0:
            a = lp_correction(a, time_limit - (time.monotonic() - t0))
        
        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MW t={t} ({elapsed:.0f}s): best={best_score:.10f}")
    
    return best_a, best_score

def lp_correction(a, time_remaining):
    """Solve LP: max sum(b) s.t. conv(a,b) <= max(conv(a,a)), b >= 0."""
    n = len(a)
    conv_a = fast_autocorr(a)
    M = conv_a.max()
    size = 2 * n - 1
    
    b_var = cp.Variable(n, nonneg=True)
    
    # Build constraint matrix (full)
    # T[k, j] = a[k-j] if 0 <= k-j < n, else 0
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    solver = None
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solver = s
                break
        except:
            continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return a
    
    b_val = b_val / b_sum  # Normalize to sum 1
    
    # Blend: a = (1-t)*a + t*b_val
    t = 0.1
    a_new = (1 - t) * a + t * b_val
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    return a_new / s
```
