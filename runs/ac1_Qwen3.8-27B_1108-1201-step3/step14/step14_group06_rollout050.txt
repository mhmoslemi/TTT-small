But this bound is not achievable because the c[k] values are not independent (they're determined by a single sequence a).

A better bound: c[k] = c[2n-2-k] (symmetry), so we have n independent values (k=0 to n-1, with c[n-1] being the "center" and appearing only once).

Actually, c[k] = sum_{valid i} a[i]*a[k-i]. The constraint is that all these are determined by the n values of a.

A information-theoretic argument: we have n degrees of freedom (the values a[0], ..., a[n-1]) and we want to make 2n-1 values small. This is an underdetermined problem in some sense, but the non-negativity constraint limits our freedom.

I think the practical minimum for n ~ 1600 is around 1.5, which is what we're seeing.

def build_constraint_matrix(a, n):
    """Build the (2n-1) x n matrix T where T[k,i] = a[k-i] if valid."""
    size = 2 * n - 1
    T = np.zeros((size, n))

        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        T[idx, i] = a[idx - i]
    return T

def build_constraint_matrix_sparse(a, n):
    """Build T more efficiently using the banded structure."""

def spectral_refine(a0, time_limit, n_iters=500):
    """Frequency-domain refinement using GERCHBERG-SAXTON style iteration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, fast_autocorr(a0).max()
    a = a / s  # normalize to sum = 1
    
    best_a = a.copy()
    best_peak = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    N = 1 << (2 * n - 1).bit_length()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = fft(a, N)
        C = ifft(A * A, N)[:2n-1].real  # linear convolution
        
        # Find peak and apply spectral reduction
        peak_val = C.max()
        if peak_val < 1e-15:
            break
        
        # Compute which frequencies contribute to the peak
        peak_idx = int(np.argmax(C))
        
        # Method: reduce the magnitude of A where it contributes to the peak
        # The contribution of frequency k to C[j] is |A[k]|^2 * exp(2πijk/N)
        # We want to reduce contributions to C[peak_idx]
        
        # Simpler approach: apply a "peak reduction" in frequency domain
        # Reduce amplitudes proportionally to their contribution to the peak
        phases = np.angle(A)
        mags = np.abs(A)
        
        # Compute contribution of each frequency to the peak
        freq_contrib = mags**2  # power at each frequency
        # Weight by how much they contribute to the peak vs. average
        C_mean = C.mean()
        if C_mean > 0:
            ratio = peak_val / C_mean
            # Reduce frequencies that contribute to high peaks
            reduction = 1.0 - 0.1 * np.minimum(ratio / 10.0, 1.0)
            mags = mags * reduction
            A_new = mags * np.exp(1j * phases)
        
        a_new = np.real(ifft(A_new, N))[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new
        
        peak_new = fast_autocorr(a_new).max()
        if peak_new < best_peak:
            best_peak = peak_new
            best_a = a_new.copy()
            a = a_new.copy()
    
    return best_a, best_peak

            if max_b < 1e-15:

            w = exp_vals / exp_vals.sum()

def combine_sequences(seq_list):
    """Combine multiple sequences by averaging (exploiting convexity)."""
    n = len(seq_list[0])
    norm_val = np.sqrt(2.0 * n)
    # Normalize all to same sum
    normalized = []
    for s in seq_list:
        s = np.maximum(np.asarray(s, dtype=np.float64), 0.0)
        ss = s.sum()
        if ss > 1e-15:
            s = s * norm_val / ss
        else:
            s = np.ones(n) * norm_val / n
        normalized.append(s)
    
    # Try different combinations
    best_combo = None
    best_score = float('inf')
    
    # Average of all
    avg = np.mean(normalized, axis=0)
    avg = np.maximum(avg, 0)
    s_avg = avg.sum()
    if s_avg > 1e-15:
        avg = avg * norm_val / s_avg
        score = fast_autocorr(avg).max()
        if score < best_score:
            best_score = score
            best_combo = avg.copy()
    
    # Pairwise averages
    for i in range(len(normalized)):
        for j in range(i+1, len(normalized)):
            pair_avg = 0.5 * (normalized[i] + normalized[j])
            pair_avg = np.maximum(pair_avg, 0)
            s_p = pair_avg.sum()
            if s_p > 1e-15:
                pair_avg = pair_avg * norm_val / s_p
                score = fast_autocorr(pair_avg).max()
                if score < best_score:
                    best_score = score
                    best_combo = pair_avg.copy()
    
    # Weighted combinations
    for w in [0.3, 0.5, 0.7]:
        for i in range(len(normalized)):
            for j in range(i+1, len(normalized)):
                combo = w * normalized[i] + (1-w) * normalized[j]
                combo = np.maximum(combo, 0)
                s_c = combo.sum()
                if s_c > 1e-15:
                    combo = combo * norm_val / s_c
                    score = fast_autocorr(combo).max()
                    if score < best_score:
                        best_score = score
                        best_combo = combo.copy()
    
    return best_combo, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def structured_init(n, rng):
    """Generate structured initializations."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
        a = np.maximum(a, 0)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.5) ** 2)
    elif choice == 5:
        # Multi-peak
        centers = rng.uniform(0.1, 0.9, 3)
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-10.0 * (np.linspace(0, 1, n) - c) ** 2)
    elif choice == 6:
        # Alternating pattern
        a = np.ones(n)
        a[::2] = 0.5
    else:
        # Random smooth
        a = rng.random(n)
        # Smooth with moving average
        kernel_size = max(1, n // 50)
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same')
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
