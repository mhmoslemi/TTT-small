            g2 = irfft(A * W, fft_size)[:n]
            grad = 2.0 * (g1 + g2)

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-18:
                break
            step = norm / (T_val * np.sqrt(n)) * 0.5

            for trial in range(8):
                a_new = a - step * grad
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    step *= 0.5
                    continue
                a_new = a_new * norm / s_new
                sc = compute_max_conv(a_new)
                if sc <= max_c + 0.05 * T_val:
                    break
                step *= 0.5

            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = compute_max_conv(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE T={T_val:.2f}: best={best_score:.8f}")

    return best_a, best_score

def lp_solve_active(a, M, solver, max_active=200):
    """Solve LP best-response using active set method."""
    n = len(a)
    size = 2 * n - 1
    conv_y = fast_conv_self(a)

    # Start with top constraints
    order = np.argsort(-conv_y)
    active = order[min(max_active, len(order))].copy()
    # Always include lag 0
    if 0 not in active:
        active = np.insert(active, 0, 0)
        active = active[:max_active + 1]

    T = None
    b = None
    for iteration in range(30):
        T = np.zeros((len(active), n))
        for idx, m in enumerate(active):
            for j in range(n):
                T[idx, j] = a[m - j] if 0 <= m - j < n else 0.0

        b_var = cp.Variable(n, nonneg=True)
        constraints = [T @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            return None
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None
        b = np.asarray(b_var.value, dtype=np.float64)
        b = np.maximum(b, 0.0)

        if b.sum() < 1e-15:
            return None

        # Check constraints
        check_conv = fast_conv_cross(a, b)
        violated = np.where(check_conv > M * (1 + 1e-6))[0]
        if len(violated) == 0:
            return b
        # Add most violated constraints
        most_violated = violated[np.argmax(check_conv[violated])]
        active = np.append(active, most_violated)
        active = np.unique(active)
        if len(active) > 500:
            # Keep only top 500 by violation
            viol_amounts = np.maximum(check_conv - M, 0)
            active = np.argsort(-viol_amounts)[::-1][:500]

    return b

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = compute_max_conv(a)
    t0 = time.monotonic()
    y = a.copy()

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        b = lp_solve_active(y, M, solver, max_active=150)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Adaptive Nesterov: fast early, slow late
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        sc_y = compute_max_conv(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        sc_a = compute_max_conv(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        # Check for stagnation
        if t % 200 == 0:
            diff = np.max(np.abs(g - y / y.sum() * g.sum()))
            if diff < 1e-8:
                if verbose:
                    print(f"    LP Nesterov converged at t={t}")
                break

        if verbose and t % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def pair_cd_optimize(a0, time_limit, rng, verbose=True):
    """Pair coordinate descent with analytical gain and bisection."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = compute_max_conv(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0
    total_improvements = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-9:
                continue

            # Find peak lags for j selection
            peak_threshold = cur_max * 0.9
            peak_lags = np.where(conv >= peak_threshold)[0]

            # Candidate j's
            j_candidates = []
            for _ in range(6):
                j_candidates.append(int(rng.integers(n)))
            for kl in peak_lags[:8]:
                j1 = (kl - i) % n
                j2 = (kl + i + n) % n
                j_candidates.append(j1)
                j_candidates.append(j2)
            j_candidates.append((i + n // 2) % n)

            seen = set()
            j_candidates = [j for j in j_candidates if j != i and j not in seen and not seen.add(j)]

            best_delta = 0.0
            best_val = cur_max
            best_j = -1

            for j in j_candidates[:12]:
                aj = a[j]
                if aj < 0:
                    continue

                # Compute slope s_k = 2*(a[k-j] - a[k-i])
                s_k = np.zeros(size)
                # a[k-j] valid for k in [j, j+n-1]
                lo = max(0, j)
                hi = min(size, j + n)
                if hi > lo:
                    s_k[lo:hi] += a[0:hi - lo]
                # a[k-i] valid for k in [i, i+n-1]
                lo2 = max(0, i)
                hi2 = min(size, i + n)
                if hi2 > lo2:
                    s_k[lo2:hi2] -= a[0:hi2 - lo2]
                s_k *= 2.0

                # Second order corrections at k=2i, 2j, i+j
                c_2i = 1.0 if 2 * i < size else 0.0
                c_2j = 1.0 if 2 * j < size else 0.0
                c_ij = -2.0 if i + j < size else 0.0

                # Bisection on d in [0, ai]
                d_lo = 0.0
                d_hi = ai
                for _b in range(18):
                    d_mid = 0.5 * (d_lo + d_hi)
                    new_vals = conv + d_mid * s_k
                    if 2 * i < size:
                        new_vals[2 * i] += c_2i * d_mid * d_mid
                    if 2 * j < size:
                        new_vals[2 * j] += c_2j * d_mid * d_mid
                    if i + j < size:
                        new_vals[i + j] += c_ij * d_mid * d_mid
                    v = new_vals.max()
                    if v < best_val:
                        best_val = v
                        best_delta = d_mid
                        best_j = j
                    # Determine bisection direction using subgradient
                    # Find argmax
                    k_star = int(np.argmax(new_vals))
                    slope_at_star = s_k[k_star]
                    quad_at_star = 0.0
                    if k_star == 2 * i:
                        quad_at_star = c_2i * d_mid
                    elif k_star == 2 * j:
                        quad_at_star = c_2j * d_mid
                    elif k_star == i + j:
                        quad_at_star = c_ij * d_mid
                    total_slope = slope_at_star + 2.0 * quad_at_star
                    if total_slope > 0:
                        d_hi = d_mid
                    else:
                        d_lo = d_mid

            if best_j >= 0 and best_delta > 1e-12:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new < 1e-15:
                    a[i] += best_delta
                    a[best_j] -= best_delta
                else:
                    a = a * norm / s_new
                    new_max = compute_max_conv(a)
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()
                        conv = fast_conv_self(a)
                        cur_max = conv.max()

        if not pass_improved:
            no_improve += 1
            if no_improve >= 8:
                break
            if no_improve == 5:
                n_perturb = max(3, n // 80)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a[idx_perturb] *= (1.0 + 0.05 * rng.standard_normal(n_perturb))
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    best_score = compute_max_conv(a)
                    best_a = a.copy()
        else:
            no_improve = 0

        if verbose and total_improvements % 200 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD improvements={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def structured_init(n, rng):
    choice = rng.integers(0, 10)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.15 * np.cos(2 * np.pi * 5 * x)
    elif choice == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = np.exp(-3.0 * x) + 0.5
    elif choice == 7:
        a = 0.5 + 0.3 * np.sin(4 * np.pi * x) * np.cos(2 * np.pi * x)
    elif choice == 8:
        a = np.sin(np.pi * x) ** 0.5 * (1 + 0.2 * np.cos(3 * np.pi * x))
    else:
        a = np.exp(-1.5 * x) * (1 + 0.3 * np.sin(5 * np.pi * x))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = eval_fn(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 0: Quick LSE to get a good starting point
    remaining = time_budget - (time.monotonic() - t_global)
    t0_phase = min(remaining * 0.08, 80.0)
    print(f"\n=== Phase 0: Quick LSE n={n_base} (budget={t0_phase:.0f}s) ===")
    if t0_phase > 10:
        a_out, _ = lse_gradient_descent(best_seq, t0_phase, verbose=False)
        try_update(a_out, f"lse_init_n{n_base}")

    # Phase 1: LP Nesterov (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    n1 = len(best_seq)
    t1 = min(remaining * 0.28, 400.0)
    print(f"\n=== Phase 1: LP Nesterov n={n1} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t1, solver, verbose=True)
        try_update(a_out, f"lp1_n{n1}")

    # Phase 2: Pair CD polish
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq)
    t2 = min(remaining * 0.22, 300.0)
    print(f"\n=== Phase 2: Pair CD n={n2} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_out, _ = pair_cd_optimize(best_seq, t2, rng, verbose=True)
        try_update(a_out, f"pcd1_n{n2}")

    # Phase 3: Another LP round
    remaining = time_budget - (time.monotonic() - t_global)
    n3 = len(best_seq)
    t3 = min(remaining * 0.20, 250.0)
    print(f"\n=== Phase 3: LP Nesterov round 2 n={n3} (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, _ = nesterov_lp_optimize(best_seq, t3, solver, verbose=True)
        try_update(a_out, f"lp2_n{n3}")

    # Phase 4: Pair CD round 2
    remaining = time_budget - (time.monotonic() - t_global)
    n4 = len(best_seq)
    t4 = min(remaining * 0.18, 200.0)
    print(f"\n=== Phase 4: Pair CD round 2 n={n4} (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = pair_cd_optimize(best_seq, t4, rng, verbose=True)
        try_update(a_out, f"pcd2_n{n4}")

    # Phase 5: Try longer length
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_base + 200
    t5 = min(remaining * 0.20, 150.0)
    print(f"\n=== Phase 5: Longer length n={n_long} (budget={t5:.0f}s) ===")
    if t5 > 30 and remaining > 100:
        init_long = interpolate_seq(best_seq, n_long)
        a_out, _ = nesterov_lp_optimize(init_long, min(t5, 80.0), solver, verbose=False)
        try_update(a_out, f"lp_long_{n_long}")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 50:
            a_out2, _ = pair_cd_optimize(best_seq, min(remaining * 0.4, 60.0), rng, verbose=False)
            try_update(a_out2, f"pcd_long_{n_long}")

    # Phase 6: Try a different length
    remaining = time_budget - (time.monotonic() - t_global)
    n_alt = max(800, n_base - 200)
    t6 = min(remaining * 0.15, 100.0)
    print(f"\n=== Phase 6: Alt length n={n_alt} (budget={t6:.0f}s) ===")
    if t6 > 30 and remaining > 80:
        init_alt = interpolate_seq(best_seq, n_alt)
        a_out, _ = nesterov_lp_optimize(init_alt, min(t6, 60.0), solver, verbose=False)
        try_update(a_out, f"lp_alt_{n_alt}")

    # Phase 7: Final LSE polish
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Final LSE polish n={len(best_seq)} (budget={remaining:.0f}s) ===")
    if remaining > 20:
        a_out, _ = lse_gradient_descent(best_seq, min(remaining * 0.6, 60.0), verbose=False)
        try_update(a_out, f"lse_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_out, _ = pair_cd_optimize(best_seq, min(remaining * 0.7, 40.0), rng, verbose=False)
            try_update(a_out, f"pcd_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
