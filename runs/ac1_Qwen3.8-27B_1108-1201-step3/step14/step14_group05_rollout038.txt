import cvxpy as cp
import time
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Compute full convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    """Safely evaluate the sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def build_conv_matrix(a):
    """Build the convolution matrix T such that (T @ b) = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def lp_best_response(a, M, solver):
    """Solve the LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    prob.solve(solver=solver, verbose=False)
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    if b_var.value is None:
        return None
    return np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)

def fessler_parks(a0, time_limit, solver, verbose=True):
    """Fessler-Parks accelerated method with full LP."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    
    y = a.copy()
    prev_y = a.copy()
    t0 = time.monotonic()
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        t = iters + 1
        M_curr = fast_autocorr(y).max()
        
        # Solve full LP
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        iters += 1
        
        # Fessler-Parks step
        gamma = 1.8
        alpha = min(0.5, 2.0 / (t + 2.0 - gamma))
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        y_new = y_new * norm / y_new.sum()
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = min(1.0, (t + 1.0 - gamma) / (t + 2.0 - gamma))
            a_new = y_new + momentum * (y_new - prev_y)
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new * norm / a_new.sum()
        
        prev_y = y.copy()
        y = y_new
        a = a_new
        
        # Update best
        for candidate in [y, a]:
            M_cand = fast_autocorr(candidate).max()
            if M_cand < best_M:
                best_M = M_cand
                best_a = candidate.copy()
        
        if verbose and iters % 50 == 0:
            print(f"  FP iter={iters} ({time.monotonic()-t0:.0f}s): M={M_curr:.8f}, best={best_M:.8f}")
    
    return best_a, best_M

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_conv(a, b):
    """Compute convolve(a, b) using FFT (for the LP constraint evaluation)."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def eval_score(a):
    """Evaluate the objective for sequence a (assumes sum(a) is fixed)."""
    b = fast_conv(a)
    return float(b.max())


def bounded_eval(a):
    """Evaluate the actual score via the evaluator function."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def solve_lp_reduced(a, M, constraint_indices, solver):
    """
    Solve the LP: max sum(b) s.t. (T b)_k <= M_k for k in constraint_indices, b >= 0.
    Uses pre-built constraint matrix for speed.
    """
    n = len(a)
    idx = constraint_indices
    n_cons = len(idx)
    
    # Build the constraint matrix for selected rows
    T_sel = np.zeros((n_cons, n))
    for row_i, k in enumerate(idx):
        # Row k of T: T[k,j] = a[k-j] if 0 <= k-j < n
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            T_sel[row_i, js:je+1] = a[k - np.arange(js, je+1)]
    
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T_sel @ b_var <= M[idx]]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    prob.solve(solver=solver, verbose=False)
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    if b_var.value is None:
        return None
    return np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)


def fessler_parks_optimize(a0, time_limit, solver, verbose=True, max_cons=200):
    """
    Fessler-Parks accelerated LP method with adaptive constraint management.
    """
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    y = a.copy()
    y_prev = a.copy()
    t_start = time.monotonic()
    iters = 0
    stall = 0
    gamma = 1.8
    
    # Initial constraint set: all
    full_idx = np.arange(2 * n - 1)
    
    # History of M for oscillation detection
    M_hist = []
    
    while time.monotonic() - t_start < time_limit:
        it = iters + 1
        M_curr = fast_conv(y).max()
        M_hist.append(M_curr)
        
        if M_curr < 1e-12:
            break
        
        # Select constraints
        if it <= 3:
            # First few iterations: use all constraints
            cs_idx = full_idx
        else:
            circ = cross_conv(y, y)  # y * y
            # Active set: where circ is close to M_curr
            tol = 0.01 * M_curr + 1e-10
            active = np.where(circ >= M_curr - tol)[0]
            
            # If too many active, select top max_cons
            if len(active) > max_cons:
                # Sort by value (descending) and take top
                sorted_active = active[np.argsort(circ[active])[::-1][:max_cons]]
                cs_idx = sorted_active
            elif len(active) < 20:
                # If too few, add some random constraints
                rng = np.random.default_rng(it * 7)
                extra = rng.choice(2*n-1, size=min(max_cons - len(active), (2*n-1) - len(active)), replace=False)
                cs_idx = np.concatenate([active, extra])
            else:
                cs_idx = active
        
        # Solve LP
        b = solve_lp_reduced(y, M_curr, cs_idx, solver)
        if b is None:
            iters += 1
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            iters += 1
            continue
        g = b * norm_val / b_sum
        iters += 1
        
        # Fessler-Parks step
        alpha = min(0.5, 2.0 / (it + 2.0 - gamma))
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_val / sy
        
        # Momentum
        if it == 1:
            a_new = g.copy()
        else:
            momentum = min(1.0, (it + 1.0 - gamma) / (it + 2.0 - gamma))
            a_new = y_new + momentum * (y_new - y_prev)
            a_new = np.maximum(a_new, 0.0)
            sa = a_new.sum()
            if sa < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm_val / sa
        
        y_prev = y.copy()
        y = y_new
        a = a_new
        
        # Track best
        for cand in [y, a]:
            Mc = fast_conv(cand).max()
            if Mc < best_M:
                best_M = Mc
                best_a = cand.copy()
                stall = 0
            else:
                stall += 1
        
        # Adaptive gamma
        if len(M_hist) > 50:
            recent = np.array(M_hist[-30:])
            osc = np.mean(np.abs(np.diff(recent))) / max(recent[0], 1e-15)
            if osc > 0.03:
                gamma = max(1.0, gamma - 0.05)
            elif osc < 0.005:
                gamma = min(2.0, gamma + 0.02)
        
        if verbose and iters % 100 == 0:
            print(f"    FP iter={iters} ({time.monotonic()-t_start:.0f}s): M={M_curr:.8f}, best={best_M:.8f}, gamma={gamma:.2f}, stall={stall}, n_cons={len(cs_idx)}")
        
        # Restart if stuck
        if stall > 200:
            rng = np.random.default_rng(iters * 13 + 42)
            scale = 0.02 * a.mean()
            y = np.maximum(y + rng.normal(0, scale, n), 0)
            y = y * norm_val / y.sum()
            a = np.maximum(a + rng.normal(0, scale, n), 0)
            a = a * norm_val / a.sum()
            y_prev = y.copy()
            stall = 0
            gamma = 1.8
            M_hist = []
            if verbose:
                print(f"    [Restart at iter {iters}]")
    
    return best_a, best_M


def lse_adam_optimize(a0, time_limit, verbose=True):

    f_T(a) = T * log(sum_k exp((Ta)_k / T)) approximates max_k (Ta)_k
    Gradient: (1/T) * sum_k exp((Ta)_k / T) * (Ta)_k^T / sum_k exp((Ta)_k / T)
    """
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()

    best_M = fast_conv(a).max()
    t_start = time.monotonic()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.01
    
    # Temperature schedule
    T0 = best_M * 0.1
    T_min = max(0.001, best_M * 0.005)
    
    steps = 0
