_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    return _global_solver


def fast_autocorr(a):
    """Compute full autoconvolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(np.asarray(a, dtype=np.float64), fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_score(a):
    """Compute the objective function value."""
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return float(2 * n * max_b / (sum_a ** 2))


def build_T(a):
    """Build the convolution matrix T where (T@b)[k] = conv(a,b)[k]."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T


def solve_lp_best_response(a, M, solver, max_constraints=300):
    """Solve the LP best response: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    size = 2 * n - 1
    T = build_T(a)
    
    # Compute current conv(a, a) to identify important constraints
    # We'll use the constraints where conv(a,b) is most likely to be binding
    # Initially use all, then reduce
    b_var = cp.Variable(n, nonneg=True)
    
    # Phase 1: Solve with all constraints
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    if b_var.value is None:
        return None
    
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    
    # Phase 2: Reduce to active constraints and re-solve
    conv_ab = T @ b_val
    for _ in range(3):
        tol = 1e-6 * M + 1e-10
        active = conv_ab >= M - tol
        n_active = int(active.sum())
        
        if n_active < 5 or n_active > max_constraints:
            # If too many, keep top max_constraints
            if n_active > max_constraints:
                active_idx = np.where(active)[0]
                top_idx = active_idx[np.argsort(conv_ab[active_idx])[::-1][:max_constraints]]
                active = np.zeros(size, dtype=bool)
                active[top_idx] = True
            else:
                break
        
        constraints = [T[active, :] @ b_var <= M[active]]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            break
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        if b_var.value is None:
            break
        b_new = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        conv_new = T @ b_new
        if conv_new.max() <= M * (1 + 1e-4):
            b_val = b_new
            conv_ab = conv_new
        else:
            break
    
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp(a0, time_limit, solver, verbose=True, max_constraints=300):
    """Fessler-Parks accelerated LP Nesterov method."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    
    y = a.copy()
    prev_y = a.copy()
    t_start = time.monotonic()
    n_iters = 0
    stall = 0
    gamma = 2.0
    
    while time.monotonic() - t_start < time_limit:
        t = n_iters + 1
        M_curr = fast_autocorr(y).max()
        
        if M_curr < 1e-12:
            break
        
        b = solve_lp_best_response(y, M_curr, solver, max_constraints=max_constraints)
        if b is None:
            stall += 1
            if stall > 50:
                break
            continue
        stall = 0
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1
        
        # Fessler-Parks step
        alpha = 2.0 / (t + 2.0 - gamma)
        alpha = min(alpha, 0.5)
        
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y
        
        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t + 1.0 - gamma) / (t + 2.0 - gamma)
            momentum = min(momentum, 1.0)
            if momentum < 0:
                momentum = 0
            a_new = y_new + momentum * (y_new - prev_y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a
        
        prev_y = y.copy()
        y = y_new
        a = a_new
        
        # Track best
        score_y = fast_autocorr(y).max()
        if score_y < best_M:
            best_M = score_y
            best_a = y.copy()
        
        score_a = fast_autocorr(a).max()
        if score_a < best_M:
            best_M = score_a
            best_a = a.copy()
        
        # Adaptive gamma
        if M_curr > best_M * 1.01 and n_iters > 100:
            gamma = max(1.0, gamma - 0.2)
        
        # Restart if stalled
        if n_iters > 50 and score_y > best_M * 1.05:
            stall += 1
            if stall > 100:
                rng = np.random.default_rng(n_iters * 31 + 7)
                pert = 0.05 * a.mean()
                y = np.maximum(y + rng.normal(0, pert, n), 0)
                y = y * norm / y.sum()
                a = np.maximum(a + rng.normal(0, pert, n), 0)
                a = a * norm / a.sum()
                prev_y = y.copy()
                stall = 0
                gamma = 2.0
        
        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best_M={best_M:.8f}, M_curr={M_curr:.6f}, gamma={gamma:.2f}")
    
    return best_a, best_M


def coordinate_descent_polish(a0, time_limit, seed=42, verbose=True):
    """Coordinate descent polish: move mass between positions to reduce the peak."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_M = fast_autocorr(a).max()
    t_start = time.monotonic()
    n_improve = 0
    n_stall = 0
    
    while time.monotonic() - t_start < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        p_star = int(np.argmax(b))
        
        # Find the most contributing pair to the peak
        js = max(0, p_star - n + 1)
        je = min(n - 1, p_star)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[p_star - idx]
        
        if contrib.max() < 1e-18:
            n_stall += 1
            if n_stall > 200:
                break
            # Random move
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                amt = a[i] * rng.uniform(0.02, 0.15)
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial * norm / a_trial.sum()
                sc = fast_autocorr(a_trial).max()
                if sc < best_M:
                    a = a_trial
                    best_M = sc
                    best_a = a.copy()
                    n_improve += 1
                    n_stall = 0
                else:
                    n_stall += 1
            continue
        
        src = idx[np.argmax(contrib)]
        
        # Try moving mass to various destinations
        best_trial_M = max_b
        best_move = None
        
        # Try non-contributing positions first
        non_contrib = np.ones(n, dtype=bool)
        non_contrib[js:je+1] = False
        non_contrib[src] = False  # can't move to self

        if len(nc_idx) > 20:

            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial * norm / a_trial.sum()
            sc = fast_autocorr(a_trial).max()
            if sc < best_trial_M:
                best_trial_M = sc
                best_move = (src, dst, amt)
        
        if best_move is not None and best_trial_M < best_M:
            src, dst, amt = best_move
            a[src] -= amt
            a[dst] += amt
            a = np.maximum(a, 0)
            a = a * norm / a.sum()
            best_M = best_trial_M
            best_a = a.copy()
            n_improve += 1
            n_stall = 0
        else:
            n_stall += 1
            # Annealing: accept with small probability

                dst = rng.integers(n)
                if dst != src:
                    amt = a[src] * rng.uniform(0.05, 0.2)
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial * norm / a_trial.sum()
                    sc = fast_autocorr(a_trial).max()
                    if sc < best_M * 1.01:
                        a = a_trial
                        if sc < best_M:
                            best_M = sc
                            best_a = a.copy()
                    n_stall = 0 if sc < best_M * 1.005 else n_stall

            s_t = a_trial.sum()
            if s_t > 1e-10:

def interpolate_seq(f, new_n):
    """Interpolate sequence to new length, preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss
