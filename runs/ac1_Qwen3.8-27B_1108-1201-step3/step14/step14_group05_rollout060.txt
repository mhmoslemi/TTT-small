2. Set beta = beta_start (e.g., 10).
3. Iterate:
   a. b = conv(a, a) via FFT (O(n log n))
   b. w = softmax(beta * b) = exp(beta*(b - max(b))) / sum(exp(beta*(b - max(b))))
   c. grad = cross_corr(a, w) via FFT (O(n log n))
   d. Step: a_new = a - lr * grad
   e. Project: a_new = max(a_new, 0), normalize sum = 1
   f. Evaluate: if improved, accept.
4. Periodically increase beta (anneal).
5. When beta is very large, w is concentrated on the max element(s), and the gradient is essentially the subgradient of the max function. This is equivalent to the LP approach.

The advantage of this approach over LP:
- Each iteration is O(n log n) instead of O(n^2) (for the LP).
- We can do many more iterations in the same time.
- The smoothness of the log-sum-exp approximation allows for better convergence with gradient-based methods.

The disadvantage:
- We need very large beta to approximate the max exactly, which can cause numerical issues.
- The log-sum-exp gradient is a "weighted average" of subgradients, which might be less aggressive than the LP's exact subgradient.

To combine the best of both:
- Use the smooth gradient approach for most of the time (fast iterations).
- Occasionally (every N iterations or when stuck), do an LP solve for an exact subgradient step.

import numpy as np

# Global state
_best_seq = None

def get_fft_size(n):
    """Get next power of 2 >= n."""
    return 1 << (n - 1).bit_length()

def fast_convolve(a):
    """Compute full linear convolution of a with itself via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = get_fft_size(size)
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_cross_corr(a, w):
    """
    Compute grad_j = sum_{m=0}^{n-1} a[m] * w[m+j] for j = 0, ..., n-1.
    This is the cross-correlation of a (length n) with w (length 2n-1).
    """
    n = len(a)
    w_len = 2 * n - 1
    fft_size = get_fft_size(n + w_len - 1)  # >= 3n-2
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    result = irfft(A * np.conj(W), fft_size)
    return result[:n]

def logsumexp(x, beta):
    """Compute (1/beta) * log(sum(exp(beta * x))) safely."""
    x = np.asarray(x)
    bx = beta * x
    m = bx.max()
    if m > 700:  # prevent overflow
        return x.max() + (1.0 / beta) * np.log(x.size)  # approximation
    return m / beta + np.log(np.exp(bx - m).sum()) / beta

def softmax_safe(x, beta):
    """Compute softmax(beta * x) safely."""
    bx = beta * x
    m = bx.max()
    e = np.exp(bx - m)
    return e / e.sum()

def normalize(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        a = np.ones(len(a))
        s = a.sum()
    return a / s

def score_normalized(a):
    """Compute score for a with sum(a) = 1. Score = 2n * max_b."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

    """Compute smooth approximation of score for sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * logsumexp(b, beta)

def adam_smooth_optimize(a0, time_limit, beta_start=50.0, beta_end=2000.0, 
                         lr=1e-4, verbose=True, seed=42):
    """
    Adam optimizer on the smooth log-sum-exp objective.
    Projects onto {a>=0, sum(a)=1} after each step.
    Anneals beta from beta_start to beta_end.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_score = score_normalized(a)
    true_best_score = best_score  # Track with true max
    
    t0 = time.monotonic()
    steps = 0
    
    # Adam state
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    
    # Learning rate schedule
    lr_start = lr
    lr_end = lr * 0.01
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = elapsed / time_limit
        
        # Anneal beta
        beta = beta_start * (beta_end / beta_start) ** progress
        # Anneal learning rate
        lr_cur = lr_start * (lr_end / lr_start) ** progress
        
        # Compute gradient
        score, grad = smooth_score_gradient(a, beta)
        
        # Project gradient onto tangent space of {sum(a)=1}
        # The constraint is sum(a) = 1, so the normal is (1,1,...,1)
        # Project: grad - mean(grad) * ones
        grad = grad - np.mean(grad)
        
        # Also, for the a >= 0 constraint, zero out gradient at positions where a is 0
        # (active set method)
        a_below = a < 1e-12
        grad[a_below] = 0.0
        # Re-project
        grad_active = grad[~a_below]
        if len(grad_active) > 0:
            grad[~a_below] = grad_active - np.mean(grad_active)
        
        # Update
        steps += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** steps)
        v_hat = v / (1 - beta2 ** steps)
        
        a_new = a - lr_cur * m_hat / (np.sqrt(v_hat) + eps)
        a_new = normalize(a_new)
        
        # Evaluate with true max
        new_true_score = score_normalized(a_new)
        
        # Also track smooth score for monitoring
        if new_true_score < true_best_score:
            true_best_score = new_true_score
            best_a = a_new.copy()
        
        a = a_new
        steps += 1
        
        if verbose and steps % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam step={steps} ({elapsed:.0f}s): beta={beta:.0f}, true_score={true_best_score:.8f}, smooth={score:.8f}")
        
        # Stochastic restart if stuck
        if steps % 50000 == 0:
            # Check if we've made progress recently
            pass  # For simplicity, no restart in this version
    
    return best_a, true_best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast smooth gradient (log-sum-exp + FFT cross-correlation) with Adam - O(n log n) per step
2. LP Nesterov (Fessler-Parks) for exact subgradient refinement
3. Adaptive multi-peak subgradient with annealed temperature
4. Peak shaving for final polish
5. Multi-length exploration with interpolation
Key innovation: FFT-based smooth gradient enables 100x+ more iterations than LP,
combined with periodic LP solves for exact subgradient steps.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp

# ===================== Helper Functions =====================

def _fft_size(n):
    return 1 << (n - 1).bit_length()


def _fast_convolver(a):
    """Full linear autocorrelation via FFT."""
    n = len(a)
    sz = 2 * n - 1
    fs = _fft_size(sz)
    A = rfft(a, fs)
    return irfft(A * A, fs)[:sz]


def _fast_cross_corr(a, w):
    """
    Compute cross_corr[j] = sum_{m=0}^{n-1} a[m] * w[m+j] for j=0..n-1.
    a has length n, w has length 2n-1.
    """
    n = len(a)
    fs = _fft_size(3 * n - 2)
    A = rfft(a, fs)
    W = rfft(w, fs)
    return irfft(A * np.conj(W), fs)[:n]


def _normalize(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s


def _score_norm(a):
    """Score = 2n * max_b when sum(a)=1."""
    return 2.0 * len(a) * _fast_convolver(a).max()


def _safe_evaluate(a):
    try:
        lst = [float(x) for x in a]
        return float(evaluate_sequence(lst))
    except Exception:
        return float('inf')


def _softmax_scaled(b, beta):
    """Safe softmax of beta * b."""
    bx = beta * b
    m = bx.max()
    e = np.exp(bx - m)
    s = e.sum()
    if s < 1e-300:
        return np.zeros_like(b)
    return e / s


# ===================== Smooth Gradient Optimizer =====================

def _adam_smooth(a0, time_limit, beta_start=10.0, beta_end=10000.0,
                 lr=None, verbose=True, seed=42, check_interval=5000):
    """
    Adam optimizer on smooth log-sum-exp objective.
    Projects onto {a>=0, sum(a)=1}. Anneals beta.
    """
    n = len(a0)
    if lr is None:
        lr = 1.0 / n
    rng = np.random.default_rng(seed)
    a = _normalize(np.asarray(a0, dtype=np.float64).copy())
    best_a = a.copy()
    best_score = _score_norm(a)

    t0 = time.monotonic()
    steps = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    n_improve = 0
    n_stall = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)

        # Anneal beta (exponential schedule)
        beta = beta_start * (beta_end / beta_start) ** (progress ** 0.7)
        # Learning rate with gentle decay
        lr_cur = lr * (1.0 - 0.3 * progress)

        # Compute gradient
        b = _fast_convolver(a)
        w = _softmax_scaled(b, beta)
        cross = _fast_cross_corr(a, w)
        # Gradient of score = 4n * cross_corr, projected onto sum=0
        grad = 4.0 * n * cross
        grad = grad - np.mean(grad)
        # Zero out at inactive constraints (a_j = 0)
        mask = a > 1e-14
        if mask.sum() < 2:
            break
        grad[mask] = grad[mask] - np.mean(grad[mask])
        grad[~mask] = 0.0

        # Adam update
        steps += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** steps)
        v_hat = v / (1 - beta2 ** steps)

        a_new = a - lr_cur * m_hat / (np.sqrt(v_hat) + eps)
        a_new = _normalize(a_new)

        # Evaluate (periodically check true score)
        if steps % check_interval == 0 or steps <= 10:
            sc = _score_norm(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                n_improve += 1
                n_stall = 0
            else:
                n_stall += 1
        a = a_new

        # Restart if stuck for long
        if steps % (check_interval * 3) == 0 and n_stall > 3:
            a = _normalize(a + 0.02 * rng.standard_normal(n) * a.mean())
            m[:] = 0
            v[:] = 0
            steps = 0
            n_stall = 0

        if verbose and steps % (check_interval * 2) == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            print(f"    Adam s={steps} ({time.monotonic()-t0:.0f}s) beta={beta:.0f} best={best_score:.8f}")

    # Final check
    sc = _score_norm(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ===================== LP Nesterov Optimizer =====================

def _get_solver():
    for s in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return s
        except Exception:
            continue
    return cp.SCS


def _lp_best_response(a, M, solver, max_constraints=300):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0. With constraint reduction."""
    n = len(a)
    sz = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)

    # Build constraint matrix T: T @ b_var = conv(a, b_var)
    # T[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # But we solve with reduced constraints for speed
    # First solve full problem
    T = np.zeros((sz, n))
    for j in range(n):
        T[j:j+n, j] = a

    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    if b_var.value is None:
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None

    # Try to reduce to active constraints
    circ = T @ b_val
    tol = max(1e-8 * M, 1e-12)
    active = circ >= M - tol
    n_active = int(active.sum())

    if 10 < n_active < max_constraints:
        prob2 = cp.Problem(cp.Maximize(cp.sum(b_var)), [T[active] @ b_var <= M[active]])
        try:
            prob2.solve(solver=solver, verbose=False)
            if prob2.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
                b_new = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
                if (T @ b_new).max() <= M * (1 + 1e-6) and b_new.sum() >= b_val.sum() * 0.99:
                    b_val = b_new
        except Exception:
            pass
    elif n_active >= max_constraints:
        # Keep top max_constraints active
        active_idx = np.where(active)[0]
        top_k = active_idx[np.argsort(circ[active_idx])[::-1][:max_constraints]]
        active2 = np.zeros(sz, dtype=bool)
        active2[top_k] = True
        prob2 = cp.Problem(cp.Maximize(cp.sum(b_var)), [T[active2] @ b_var <= M[active2]])
        try:
            prob2.solve(solver=solver, verbose=False)
            if prob2.status in ('optimal', 'optimal_inaccurate') and b_var.value is not None:
                b_new = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
                if (T @ b_new).max() <= M * (1 + 1e-5) and b_new.sum() > b_val.sum() * 0.95:
                    b_val = b_new
        except Exception:
            pass

    return b_val


def _nesterov_lp(a0, time_limit, solver, verbose=True, max_active=200):
    """Fessler-Parks accelerated LP Nesterov."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_M = _fast_convolver(a).max()

    y = a.copy()
    prev_y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    gamma = 1.7

    while time.monotonic() - t0 < time_limit:
        M_curr = _fast_convolver(y).max()
        if M_curr < 1e-12:
            break

        b = _lp_best_response(y, M_curr, solver, max_constraints=max_active)
        if b is None:
            n_iters += 1
            if n_iters % 100 == 0:
                print(f"    LP iter={n_iters} ({time.monotonic()-t0:.0f}s): LP failed, skipping")
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            n_iters += 1
            continue

        g = b * norm / b_sum
        n_iters += 1

        # Fessler-Parks
        t = n_iters + 1
        alpha = 2.0 / (t + 2.0 - gamma)
        alpha = min(alpha, 0.5)

        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / sy

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t + 1.0 - gamma) / (t + 2.0 - gamma)
            momentum = min(momentum, 1.0)
            a_new = y_new + momentum * (y_new - prev_y)
            a_new = np.maximum(a_new, 0.0)
            sa = a_new.sum()
            if sa < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / sa

        prev_y = y.copy()
        y = y_new
        a = a_new

        # Track best
        My = _fast_convolver(y).max()
        if My < best_M:
            best_M = My
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        Ma = _fast_convolver(a).max()
        if Ma < best_M:
            best_M = Ma
            best_a = a.copy()
            stall = 0

        # Adaptive gamma
        if stall > 200:
            gamma = max(1.1, gamma - 0.05)
        elif stall == 0:
            gamma = min(1.9, gamma + 0.02)

        # Restart
        if stall > 500:
            rng = np.random.default_rng(n_iters + 7)
            noise = 0.02 * a.mean() * rng.standard_normal(n)
            y = np.maximum(y + noise, 0)
            y = y * norm / y.sum()
            a = np.maximum(a + noise, 0)
            a = a * norm / a.sum()
            prev_y = y.copy()
            stall = 0
            gamma = 1.7

        if verbose and n_iters % 300 == 0:
            print(f"    LP iter={n_iters} ({time.monotonic()-t0:.0f}s): best_M={best_M:.6f} gamma={gamma:.2f} stall={stall}")

    return best_a, best_M


# ===================== Peak Shaving =====================

def _peak_shave(a0, time_limit, seed=42, verbose=True):
    """Targeted peak shaving with line search."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = _normalize(np.asarray(a0, dtype=np.float64).copy())
    best_a = a.copy()
    best_score = _score_norm(a)
    t0 = time.monotonic()
    n_stall = 0

    while time.monotonic() - t0 < time_limit:
        b = _fast_convolver(a)
        p = int(np.argmax(b))
        max_b = b[p]

        # Find contributing indices
        js = max(0, p - n + 1)
        je = min(n - 1, p)
        idx = np.arange(js, je + 1)
        if len(idx) == 0:
            break
        contrib = a[idx] * a[p - idx]
        src = idx[np.argmax(contrib)]

        # Find good destinations (low value, not contributing to peak)
        non_contrib = np.ones(n, dtype=bool)
        non_contrib[js:je+1] = False
        non_contrib[src] = True
        nc_idx = np.where(non_contrib)[0]
        if len(nc_idx) == 0:
            nc_idx = np.array([i for i in range(n) if i != src])

        # Try best destination
        best_sc = best_score
        best_move = None
        n_try = min(8, len(nc_idx))
        dst_candidates = nc_idx[np.argsort(a[nc_idx])[:n_try]]

        for dst in dst_candidates:
            for frac in [0.05, 0.1, 0.2, 0.3]:
                amt = a[src] * frac
                if amt < 1e-18:
                    continue
                a_t = a.copy()
                a_t[src] -= amt
                a_t[dst] += amt
                a_t = _normalize(a_t)
                sc = _score_norm(a_t)
                if sc < best_sc:
                    best_sc = sc
                    best_move = (src, dst, frac)

        if best_move is not None and best_sc < best_score:
            src, dst, frac = best_move
            amt = a[src] * frac
            a[src] -= amt
            a[dst] += amt
            a = _normalize(a)
            best_score = best_sc
            best_a = a.copy()
            n_stall = 0
        else:
            n_stall += 1
            # Random exploration
            if n_stall % 20 == 0:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.05, 0.2)
                    a_t = a.copy()
                    a_t[i] -= amt
                    a_t[j] += amt
                    a_t = _normalize(a_t)
                    sc = _score_norm(a_t)
                    if sc < best_score * 1.01:
                        a = a_t
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
                        n_stall = 0

        if n_stall > 200:
            # Bigger perturbation
            n_pert = max(5, n // 100)
            pp = rng.choice(n, size=n_pert, replace=False)
            a_t = a.copy()
            for idx_p in pp:
                j = rng.integers(n)
                amt = a_t[idx_p] * rng.uniform(0.1, 0.4)
                a_t[idx_p] -= amt
                a_t[j] += amt
            a_t = _normalize(a_t)
            sc = _score_norm(a_t)
            if sc < best_score * 1.02:
                a = a_t
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            n_stall = 0

        if verbose and (n_stall == 0 and time.monotonic() - t0) % 30 < 0.1:
            print(f"    PeakShave ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

    return best_a, best_score


# ===================== Multi-peak Subgradient (fast, no LP) =====================

def _multi_peak_subgrad(a0, time_limit, verbose=True, seed=42):
    """
    Multi-peak subgradient with annealed temperature.
    Each iteration: compute b, weight peaks, compute gradient via FFT, take step.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = _normalize(np.asarray(a0, dtype=np.float64).copy())
    best_a = a.copy()
    best_score = _score_norm(a)
    t0 = time.monotonic()
    steps = 0
    step_size = 0.5 / n

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        # Temperature annealing (beta for peak weighting)
        beta = 20.0 + 500.0 * (progress ** 1.5)

        b = _fast_convolver(a)
        w = _softmax_scaled(b, beta)
        cross = _fast_cross_corr(a, w)
        grad = 4.0 * n * cross
        grad = grad - np.mean(grad)
        mask = a > 1e-14
        if mask.sum() >= 2:
            grad[mask] = grad[mask] - np.mean(grad[mask])
        grad[~mask] = 0.0

        # Polyak-type step
        gn2 = np.dot(grad, grad)
        if gn2 < 1e-30:
            break
        max_b = b.max()
        target = best_score / (2.0 * n)  # target max_b in normalized form
        polyak = (max_b - target * 0.99) / gn2 if max_b > target else step_size
        step = min(step_size, polyak) if polyak > 0 else step_size
        step = max(step, 1e-8 / n)

        a_new = a - step * grad
        a_new = _normalize(a_new)
        sc = _score_norm(a_new)

        if sc < best_score:
            a = a_new
            best_score = sc
            best_a = a_new.copy()
            step_size = min(step_size * 1.02, 2.0 / n)
        else:
            # Annealing accept
            T = best_score * 0.002 * (1.0 - progress)
            if rng.random() < np.exp(-(sc - best_score) / max(T, 1e-10)):
                a = a_new
            step_size *= 0.85

        steps += 1
        if verbose and steps % 5000 == 0:
            print(f"    SubGrad s={steps} ({time.monotonic()-t0:.0f}s): best={best_score:.8f} beta={beta:.0f} step={step:.2e}")

    return best_a, best_score


# ===================== Interpolation =====================

def _interpolate(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    fi = np.interp(new_x, old_x, f_norm)
    return _normalize(fi)

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search combining smooth gradient (fast), LP Nesterov (exact), and peak shaving.
    """

        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = _safe_evaluate(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        x = np.linspace(0.0, 1.0, n_base)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = _normalize(best_seq)
        best_score = _safe_evaluate(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_base = len(best_seq)

    def _try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = _safe_evaluate(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_global:.0f}s)")

    # === Try multiple initializations ===
    print("\n=== Trying multiple initializations ===")
    inits = []
    # Sine^2
    x = np.linspace(0, 1, n_base)
    inits.append(_normalize(np.sin(np.pi * x) ** 2))
    # Raised cosine
    inits.append(_normalize(0.5 * (1 - np.cos(2 * np.pi * x))))
    # Sine^4 (more peaked)
    inits.append(_normalize(np.sin(np.pi * x) ** 4))
    # Asymmetric: more mass on left
    inits.append(_normalize(np.sin(np.pi * x ** 0.7)))
    # Asymmetric: more mass on right
    inits.append(_normalize(np.sin(np.pi * (1 - x) ** 0.7)))
    # Gaussian
    inits.append(_normalize(np.exp(-((x - 0.5) / 0.15) ** 2)))
    # Parent (if available)
    if best_seq is not None:
        inits.append(best_seq.copy())

    # Pick best initialization
    for i, init in enumerate(inits):
        sc = _score_norm(init)
        if sc < best_score / (2.0 * n_base) * 2.0 * n_base:  # compare properly
            pass
    # Just use parent as main start, but test all with brief smooth runs
    main_init = best_seq.copy()
    print(f"  Best init score: {best_score:.8f} (n={n_base})")

    # === Phase 1: Smooth gradient (main workhorse, fast) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.40, 600.0)
    print(f"\n=== Phase 1: Adam Smooth n={n_base} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, _ = _adam_smooth(main_init, t1, beta_start=5.0, beta_end=8000.0,
                                lr=1.0/n_base, verbose=True, seed=seed)

    # === Phase 2: Multi-peak subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 2: MultiPeak SubGrad n={len(best_seq)} (budget={t2:.0f}s) ===")
    if t2 > 20:
        a_out, _ = _multi_peak_subgrad(best_seq, t2, verbose=True, seed=seed+1)
        _try_update(a_out, f"subgrad_n{len(best_seq)}")

    # === Phase 3: LP Nesterov ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t3 = min(remaining * 0.20, 350.0)
    print(f"\n=== Phase 3: LP Nesterov n={n_cur} (budget={t3:.0f}s) ===")
    if t3 > 30:

    remaining = time_budget - (time.monotonic() - t_global)

    print(f"\n=== Phase 4: Adam Smooth r2 n={n_cur} (budget={t4:.0f}s) ===")

                                lr=0.5/n_cur, verbose=True, seed=seed+2)

    # === Phase 5: Try different n ===
    remaining = time_budget - (time.monotonic() - t_global)

    t5 = min(remaining * 0.12, 150.0)
    print(f"\n=== Phase 5: Explore n={n_try} (budget={t5:.0f}s) ===")
    if t5 > 20:
        init_long = _interpolate(best_seq, n_try)
        a_out, _ = _adam_smooth(init_long, t5 * 0.6, beta_start=10.0, beta_end=5000.0,
                                lr=1.0/n_try, verbose=False, seed=seed+3)

                                     solver, verbose=False, max_active=150)
            _try_update(a_out, f"lp_n{n_try}")

    # === Phase 6: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t6 = min(remaining * 0.10, 120.0)

    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t7 = min(remaining * 0.15, 150.0)
    print(f"\n=== Phase 7: Final LP n={n_cur} (budget={t7:.0f}s) ===")
    if t7 > 20:
        a_out, _ = _nesterov_lp(best_seq, t7, solver, verbose=True, max_active=150)
        _try_update(a_out, f"lp_final_n{n_cur}")

    # === Phase 8: Final peak shave ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t8 = min(remaining * 0.15, 100.0)
    print(f"\n=== Phase 8: Final PeakShave n={n_cur} (budget={t8:.0f}s) ===")
    if t8 > 10:
        a_out, _ = _peak_shave(best_seq, t8, seed=seed+99, verbose=True)
        _try_update(a_out, f"peak_final_n{n_cur}")

    # === Phase 9: One more smooth pass if time ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(best_seq)
    t9 = min(remaining * 0.20, 120.0)
    print(f"\n=== Phase 9: Final Adam n={n_cur} (budget={t9:.0f}s) ===")
    if t9 > 15:
        a_out, _ = _adam_smooth(best_seq, t9, beta_start=100.0, beta_end=50000.0,
                                lr=0.3/n_cur, verbose=True, seed=seed+200)
        _try_update(a_out, f"adam_final_n{n_cur}")

    # Final verification
    if best_seq is not None:
        fs = _safe_evaluate(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
