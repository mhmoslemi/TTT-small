def sLP_step(a, S, solver):
    """One SLP iteration: linearize the QCQP and solve the LP."""
    n = len(a)
    size = 2*n - 1
    c = fast_autocorr(a)  # a * a
    G = build_G(a)
    
    a_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [cp.sum(a_var) == S, cp.sum(G @ a_var - size * t_var) <= cp.sum(c)]
    # Actually, the constraints are: G[k,:] @ a <= t + c[k] for each k
    # In matrix form: G @ a - 1*t <= c
    
    # This is a bit tricky in cvxpy. Let me use element-wise constraints.
    # G @ a_var has shape (size,), t_var is scalar
    # Constraint: G @ a_var <= t_var + c (element-wise)
    constraints = [cp.sum(a_var) == S]
    constraints.append(G @ a_var <= t_var + c)
    constraints.append(t_var >= 0)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    prob.solve(solver=solver, verbose=False)
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    a_new = np.maximum(np.asarray(a_var.value, dtype=np.float64), 0.0)
    t_new = max(0.0, float(t_var.value))
