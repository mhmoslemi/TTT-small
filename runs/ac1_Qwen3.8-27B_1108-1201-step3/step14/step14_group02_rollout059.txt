def solve_lp_subproblem(a_y, M_curr, solver):

def newton_refinement(a0, max_iters=50, verbose=False):

    - conv(a,a)[k] = M for k in A
    - sum(a) = 1
    - a >= 0

    We solve this iteratively.
    """
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    
    for it in range(max_iters):
        ac = fast_autocorr(a)
        M = ac.max()
        
        # Identify active set (top K lags at or near max)
        threshold = M * (1 - 1e-6)
        active = np.where(ac >= threshold)[0]
        if len(active) < 2:
            active = np.argsort(ac)[::-1][:5]
        
        # Try to equalize the top active lags
        # For each active lag k, the gradient of conv[k] w.r.t. a is:
        # d(conv[k])/da[j] = a[k-j] (if valid) + a[j-k] (if valid)
        
        # Simplified: use the "push down the peaks" approach
        # For the lag k* that achieves the max, reduce the contribution
        k_max = int(np.argmax(ac))

        js = max(0, k_max - n + 1)

            break
        
        # Try to reduce by adjusting a[idx] and a[k_max - idx]
        # Move mass from the larger of the pair to reduce the product
        i = idx[np.argmax(contrib)]
        j = k_max - i
        
        if i == j:
            # Can't reduce a[i]*a[i] by moving within the pair
            break
        
        if a[i] > a[j]:
            src, dst = i, j
        else:
            src, dst = j, i
        
        # Reduce a[src] slightly, increase a[dst]
        # But this might increase other peaks...
        # Just do a small step
        delta = a[src] * 0.01
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        M_new = fast_autocorr(a_new).max()
        if M_new < M:
            a = a_new
            if M_new < best_m:
                best_m = M_new
                best_a = a.copy()
        else:
            break
    
    if verbose:
        J = 2.0 * n * best_m
        print(f"    Newton ref: J={J:.8f}")

1. Nesterov-accelerated LP-based Frank-Wolfe (primary method, long run)
2. FW with adaptive line search for fine-tuning

def get_solver():

    if _global_solver is not None:
        return _global_solver
    for s in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                print(f"  LP solver: {s}")
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    print("  LP solver (fallback): ECOS")
    return _global_solver

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):

    except Exception:

def norm_sum1(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()

    if s < 1e-15:
        return np.ones(new_n) / new_n

def solve_lp(a_y, M_curr, solver):
    n = len(a_y)
    b_var = cp.Variable(n, nonneg=True)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_y
    constraints = [T @ b_var <= M_curr]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    b_sum = b_val.sum()
    if b_sum < 1e-15:
        return None
    return b_val / b_sum

        print(f"    Coord ({elapsed:.1f}s): J={J:.8f}")
    return best_a, best_m

def kkt_equalize(a0, max_rounds=200, verbose=False):
    """Equalize the top autocorrelation peaks using Newton-like steps."""
    n = len(a0)
    a = norm_sum1(a0)
    best_a = a.copy()
    best_m = fast_autocorr(a).max()
    
    for rnd in range(max_rounds):
        ac = fast_autocorr(a)
        M = ac.max()
        threshold = M * (1.0 - 1e-5)
        active = np.where(ac >= threshold)[0]
        
        if len(active) < 2:
            break
        
        target = np.mean(ac[active])
        
        for k in active:
            k_int = int(k)
            js = max(0, k_int - n + 1)
            je = min(n - 1, k_int)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_int - idx]
            
            if contrib.max() < 1e-20:
                continue
            
            excess = ac[k_int] - target
            if excess <= 1e-18:
                continue
            
            src = idx[np.argmax(contrib)]
            partner = k_int - src
            
            if partner == src:
                continue
            if partner < 0 or partner >= n:
                continue
            
            if a[src] <= a[partner]:
                continue
            
            delta = excess / (2.0 * a[partner] + 1e-30)
            delta = min(delta, a[src] * 0.1)
            if delta < 1e-18:
                continue
            
            a[src] -= delta
            a[partner] += delta
        
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        
        M_new = fast_autocorr(a).max()
        if M_new < best_m:
            best_m = M_new
            best_a = a.copy()
        elif M_new > best_m * (1 + 1e-6):
            a = best_a.copy()
            break
    
    if verbose:
        J = 2.0 * n * best_m
        print(f"    KKT-eq: J={J:.8f}")
    return best_a, best_m
