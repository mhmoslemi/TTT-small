def lp_best_response(a, M, solver, active_set=None):
    n = len(a)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    if active_set is not None and len(active_set) > 0:
        T = build_conv_matrix_rows(a, active_set)
        constraints = [T @ b_var <= M]
    else:
        T = build_conv_matrix_rows(a, np.arange(size))
        constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, active_set
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, active_set
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None, active_set
    return b_val, active_set


def nesterov_lp_optimize(a0, time_limit, solver, verbose=False, seed=0):
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    t0 = time.monotonic()
    active_set = None
    stall_count = 0
    stall_threshold = 35

    for t in range(1, 2000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        conv_y = fast_autocorr(y)
        if active_set is None or t % 50 == 0:
            tol = 0.025 + 0.01 * np.exp(-t / 400.0)
            active_set = np.where(conv_y > M_curr * (1.0 - tol))[0]
            if len(active_set) < 3:
                active_set = np.where(conv_y > M_curr * 0.75)[0]

        b, active_set_new = lp_best_response(y, M_curr, solver, active_set)
        if b is None:
            if t % 10 == 0:
                active_set = None
                b, active_set_new = lp_best_response(y, M_curr, solver, None)
            if b is None:
                continue

        active_set = active_set_new
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = min((t - 1.0) / (t + 2.0), 0.93)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        if stall_count > stall_threshold:
            y = best_a.copy()
            a = best_a.copy()
            active_set = None
            stall_count = 0

        if verbose and t % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP t={t} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def compute_smoothed_max_gradient(a, weights):
    """Compute gradient of sum_k weights_k * (a*a)_k w.r.t. a.
    Uses the formula: grad_j = sum_m a_m * w_{j+m} * (1 + [m != j])
    """
    n = len(a)
    size = 2 * n - 1
    # Build the gradient using cross-correlation structure
    # grad_j = sum_{m=0}^{n-1} a_m * w_{j+m} * (1 + [m != j])
    #        = sum_{m=0}^{n-1} a_m * w_{j+m} + a_j * w_{2j} (if 2j < size)
    
    # Part 1: cross-correlation of a with w at shifts 0..n-1
    # (a ⋆ w)_j = sum_m a_m * w_{j+m}
    # Use FFT for efficiency
    fft_size = 1 << (2 * n - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(weights, fft_size)
    # cross-correlation: ifft(conj(A) * W) gives sum_m a_{m-j} * w_m
    # We want sum_m a_m * w_{m+j} which is correlate(w, a) at lag j
    # numpy correlate: np.correlate(w, a, 'full')[n-1+j] = sum_m w[m] * a[m+n-1+j-(n-1)] ... 
    # Let's just use the direct approach via matrix for correctness, 
    # or use the FFT formula properly.
    # 
    # sum_m a_m * w_{m+j} = ifft(conj(FFT(a)) * FFT(w)) evaluated at index j
    # Actually: ifft(conj(A) * W)[j] = sum_m a_m * w_{(m-j) mod N}
    # That's not quite right either. Let me use:
    # sum_m a_m * w_{m+j} = (a * w_rev) where w_rev is reversed... 
    # 
    # Simplest correct approach: use the Toeplitz-like structure
    # Offset matrix: Row j of the matrix M has w[j], w[j+1], ..., w[j+n-1] (padded with 0)
    # Then part1 = M @ a
    
    # For n up to ~3000, the O(n^2) direct computation is fine with numpy
    # Build offset sums efficiently
    part1 = np.zeros(n)
    for j in range(n):
        lo = 0
        hi = min(n, size - j)
        if hi > lo:
            part1[j] = np.dot(a[:hi], weights[j:j+hi])
    
    # Part 2: diagonal correction a_j * w_{2j}
    diag_idx = 2 * np.arange(n)
    diag_mask = diag_idx < size
    part2 = np.zeros(n)
    part2[diag_mask] = a[diag_mask] * weights[diag_idx[diag_mask]]
    
    return part1 + part2


def smoothed_max_grad_descent(a0, time_limit, verbose=False, seed=0):
    """Gradient descent on log-sum-exp smoothed max of autocorrelation."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Adam state
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.05

    T_start = 0.05
    T_end = 0.005
    total_iters = 30000

    for t in range(1, total_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        # Anneal temperature
        frac = min(1.0, t / (total_iters * 0.8))
        T = T_start * (T_end / T_start) ** frac
        T = max(T, 0.003)

        lr_t = base_lr / (1.0 + 0.05 * t / 1000.0)

        conv = fast_autocorr(a)
        conv_max = conv.max()

        # Log-sum-exp weights
        conv_shifted = (conv - conv_max) / T
        exp_vals = np.exp(conv_shifted)
        weights = exp_vals / exp_vals.sum()

        # Compute gradient
        grad = compute_smoothed_max_gradient(a, weights)

        # Add small regularization to prevent degeneracy
        grad += 0.001 * a

        # Adam update
        m_adam = beta1 * m_adam + (1 - beta1) * grad
        v_adam = beta2 * v_adam + (1 - beta2) * grad**2
        m_hat = m_adam / (1 - beta1**t)
        v_hat = v_adam / (1 - beta2**t)

        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s

        a = a_new

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Occasional random perturbation to escape local minima
        if t % 5000 == 0 and best_score > 1.4:
            noise = 0.01 * rng.standard_normal(n)
            a_pert = a + noise
            a_pert = np.maximum(a_pert, 0.0)
            sp = a_pert.sum()
            if sp > 1e-15:
                a_pert = a_pert * norm / sp
                sp_score = fast_autocorr(a_pert).max()
                if sp_score < best_score:
                    best_score = sp_score
                    best_a = a_pert.copy()
                    a = a_pert.copy()

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GDL t={t} ({elapsed:.0f}s): T={T:.5f}, best={best_score:.10f}")

    return best_a, best_score


def pair_coordinate_descent(a0, time_limit, rng, verbose=False):
    """Enhanced pair CD with multi-pair moves and better line search."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm * 1e-8:
                continue

            # Try multiple random partners
            best_move = None
            best_move_val = conv.max()

            for attempt in range(6):
                j = int(rng.integers(n))
                if j == i:
                    continue

                # Compute gain vector g (first-order + second-order for diagonal)
                g = np.zeros(size)
                # Remove from i
                lo1 = max(0, i)
                hi1 = min(size, i + n)
                if hi1 > lo1:
                    src = np.arange(lo1, hi1) - i
                    g[lo1:hi1] -= src * 0 + a[src]  # a[k-i] for k in [i, i+n)
                    # Actually: contribution of a_i to conv_k is a_i * a_{k-i}
                    # Removing a_i: subtract a_i * a_{k-i} from conv_k
                    # But since we're moving mass, the effect is:
                    # a_i decreases by d, a_j increases by d
                    # Effect on conv_k: -d * a_{k-i} + d * a_{k-j} (roughly)
                    # Plus second order: -d^2 at k=2i, +d^2 at k=2j, +2*d*a_j at k=i+j, etc.
                    # For first order: g_k = a_{k-j} - a_{k-i} (with bounds)
                
                # Let me redo this properly
                g = np.zeros(size)
                # Effect of -d at position i on conv_k: -d * a_{k-i} (if valid)
                # Effect of +d at position j on conv_k: +d * a_{k-j} (if valid)
                # Plus symmetric terms: -d * a_{i-k}... wait
                # conv_k = sum_p a_p * a_{k-p}
                # d(conv_k)/d(a_i) = a_{k-i} + a_{i-k} (if both valid, and 2a_{k-i} if k=2i... no)
                # d(conv_k)/d(a_i) = a_{k-i} if 0<=k-i<n, plus a_{i-k} if 0<=i-k<n and i-k != k-i
                # = a_{k-i} * [0<=k-i<n] * (1 + [k != 2i and 0<=i-k<n])
                # Wait, i-k = -(k-i), so 0<=i-k<n means 0<=k-i<... no.
                # i-k >= 0 means k <= i, and i-k < n means k > i-n.
                # k-i >= 0 means k >= i, and k-i < n means k < i+n.
                # So the two terms are:
                # Term 1: a_{k-i} for k in [i, i+n-1] (i.e., 0<=k-i<n)
                # Term 2: a_{i-k} for k in [max(0,i-n+1), i-1] (i.e., 0<=i-k<n and k!=i... well k<i)
                # Wait, if k=i, then k-i=0 and i-k=0, same index. The term a_0 appears only once in conv_i.
                # Actually conv_i = sum_p a_p * a_{i-p}. The term p=i gives a_i * a_0. The term p=0 gives a_0 * a_i. These are the same!
                # So conv_i includes a_0*a_i only once (from p=0 or p=i, but they're the same product).
                # No wait: conv_i = sum_{p: 0<=p<n, 0<=i-p<n} a_p * a_{i-p}
                # For p=0: a_0 * a_i
                # For p=i: a_i * a_0
                # These are different terms in the sum! They're both a_0*a_i.
                # So d(conv_i)/d(a_i) gets a_0 from p=0 term AND a_0 from p=i term = 2*a_0.
                # But wait, that's only if both p=0 and p=i are valid range indices.
                # For p=0: need 0<=i-0<n, i.e., 0<=i<n. Yes.
                # For p=i: need 0<=i-i<n=0<n. Yes.
                # So d(conv_i)/d(a_i) = 2*a_0 (from p=0) + ... hmm this is getting confusing.
                
                # Let me just use the formula: d(conv_k)/d(a_j) = sum of a_{k-p} for all p where the term a_p*a_{k-p} contains a_j
                # The term a_p * a_{k-p} contains a_j if p=j or k-p=j
                # If p=j: gives a_{k-j} (valid if 0<=k-j<n)
                # If k-p=j (p=k-j): gives a_j (valid if 0<=k-j<n and k-j != j)
                # If k-j=j (k=2j): only one term, gives a_j
                # So: d(conv_k)/d(a_j) = a_{k-j}*[0<=k-j<n] + a_{k-j}*[0<=k-j<n and k!=2j]
                # = a_{k-j}*[0<=k-j<n]*(1+[k!=2j])
                
                # For the move: delta = -d at i, +d at j
                # Effect on conv_k: d * (d(conv_k)/d(a_j) - d(conv_k)/d(a_i)) [first order]
                # Plus second order: -d^2 * [2i==k] + d^2 * [2j==k] + 2*d*(a_j - a_i) * [i+j==k]... 
                # Actually the second order from the cross term: when we change a_i and a_j, 
                # the cross term in conv_{i+j} gets 2*d*(a_j - a_i)... this is getting complex.
                # Let me just evaluate at candidate d values directly.
                
                candidates = [0.1, 0.25, 0.5, 0.75, 0.9, 1.0]
                for frac in candidates:
                    d = ai * frac
                    a_trial = a.copy()
                    a_trial[i] -= d
                    a_trial[j] += d
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial * norm / st
                    conv_trial = fast_autocorr(a_trial)
                    val = conv_trial.max()
                    if val < best_move_val - 1e-12:
                        best_move_val = val
                        best_move = (i, j, d)

            if best_move is not None:
                i_m, j_m, d_m = best_move
                a[i_m] -= d_m
                a[j_m] += d_m
                s = a.sum()
                if s < 1e-15:
                    a[i_m] += d_m
                    a[j_m] -= d_m
                    continue
                a = a * norm / s
                conv = fast_autocorr(a)
                if best_move_val < best_score - 1e-13:
                    best_score = best_move_val
                    best_a = a.copy()
                    pass_improved = True
                    no_improve = 0
                else:
                    # Revert
                    a[j_m] -= d_m
                    a[i_m] += d_m
                    a = a * norm / a.sum()
                    conv = fast_autocorr(a)

        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    PCD ({elapsed:.0f}s): best={best_score:.10f})")

    return best_a, best_score


def peak_balance(a0, time_limit, rng, verbose=False):
    """Try to reduce the top peaks by moving mass from near-peak positions."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    best_score = fast_autocorr(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1

    while time.monotonic() - t0 < time_limit:
        conv = fast_autocorr(a)
        # Find top peaks
        top_k = np.argsort(conv)[-5:][::-1]
        improved = False

        for k_peak in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            # The peak at k_peak involves pairs (i, k_peak-i)
            # Try reducing mass at positions that contribute most to this peak
            lo = max(0, k_peak - n + 1)
            hi = min(n, k_peak + 1)
            contributors = list(range(lo, hi))
            
            # For each contributor i, try moving mass to a random other position
            for i in contributors[:5]:  # Limit to top 5
                if a[i] < norm * 1e-7:
                    continue
                # Try moving to a position that doesn't contribute to this peak
                for attempt in range(3):
                    j = int(rng.integers(n))
                    if j == i or (lo <= j < hi):
                        continue
                    # Check if j contributes to k_peak
                    if 0 <= k_peak - j < n:
                        continue
                    
                    d = a[i] * 0.3
                    a_trial = a.copy()
                    a_trial[i] -= d
                    a_trial[j] += d
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial * norm / st
                    conv_trial = fast_autocorr(a_trial)
                    if conv_trial.max() < best_score - 1e-12:
                        a = a_trial.copy()
                        best_score = conv_trial.max()
                        best_a = a.copy()
                        improved = True
                        break

        if not improved:
            # Try small random perturbations
            idx = int(rng.integers(n))
            jdx = int(rng.integers(n))
            if idx != jdx and a[idx] > norm * 1e-6:
                d = a[idx] * 0.1
                a_trial = a.copy()
                a_trial[idx] -= d
                a_trial[jdx] += d
                st = a_trial.sum()
                if st > 1e-15:
                    a_trial = a_trial * norm / st
                    if fast_autocorr(a_trial).max() < best_score - 1e-12:
                        a = a_trial.copy()
                        best_score = fast_autocorr(a).max()
                        best_a = a.copy()
                        improved = True

        if not improved:
            break

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    PB ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 12)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2.0) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.15 * np.cos(2 * np.pi * 5 * x) + 0.05 * np.cos(2 * np.pi * 9 * x)
    elif choice == 3:
        a = np.exp(-8.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-8.0 * (x - 0.7) ** 2)
    elif choice == 4:
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    elif choice == 5:
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    elif choice == 6:
        a = 1.0 + 0.3 * np.cos(4 * np.pi * x) + 0.1 * np.cos(8 * np.pi * x)
    elif choice == 7:
        a = np.sin(np.pi * x) ** 0.5 * (1 + 0.2 * np.cos(6 * np.pi * x))
    elif choice == 8:
        # Optimal transport-like: high in middle, slightly asymmetric
        a = np.exp(-3.0 * (x - 0.5) ** 2 / 0.1) + 0.3
    elif choice == 9:
        # Water-drop shape
        a = 0.4 + 0.6 * np.sin(np.pi * x) ** 2
    elif choice == 10:
        # Bimodal with different peaks
        a = np.exp(-10.0 * (x - 0.35) ** 2) + 0.8 * np.exp(-10.0 * (x - 0.65) ** 2) + 0.2
    else:
        # Very flat with slight modulation
        a = 1.0 + 0.1 * np.cos(2 * np.pi * 3 * x) + 0.05 * np.cos(2 * np.pi * 7 * x)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def perturb_sequence(a, rng, scale=0.05):
    n = len(a)
    noise = 1.0 + scale * rng.standard_normal(n)
    noise = np.maximum(noise, 0.1)
    a_pert = a * noise
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s

        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")
    else:
        n_base = len(best_seq)

    remaining = time_budget - (time.monotonic() - t_global)

    # === Phase 1: LP Nesterov at base length ===
    print(f"\n=== Phase 1: LP Nesterov n={n_base} ===")
    t1 = min(remaining * 0.22, 350.0)
    if t1 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True, seed=0)
        do_try_update(a_out, f"lp1_n{n_base}")

    # === Phase 2: Gradient descent (NEW) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Smoothed-max GradDescent n={n_best} ===")
    t2 = min(remaining * 0.18, 300.0)
    if t2 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = smoothed_max_grad_descent(a_norm, t2, verbose=True, seed=1)
        do_try_update(a_out, f"gd1_n{n_best}")

    # === Phase 3: Pair CD ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Pair CD n={n_best} ===")
    t3 = min(remaining * 0.12, 180.0)
    if t3 > 15:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pair_coordinate_descent(a_norm, t3, rng, verbose=True)
        do_try_update(a_out, f"pcd1_n{n_best}")

    # === Phase 4: Peak balance (NEW) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Peak Balance n={n_best} ===")
    t4 = min(remaining * 0.08, 100.0)
    if t4 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_balance(a_norm, t4, rng, verbose=True)
        do_try_update(a_out, f"pb1_n{n_best}")

    # === Phase 5: Perturbation restart + LP ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Perturbation restarts ===")
    t5_total = min(remaining * 0.20, 250.0)
    n_restarts = 3
    t5_each = t5_total / n_restarts
    for r in range(n_restarts):
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 60:
            break
        t5 = min(t5_each, remaining - 60)
        if t5 < 30:
            break
        n_cur = len(best_seq)
        scale = 0.02 + 0.02 * r
        pert = perturb_sequence(best_seq, rng, scale=scale)
        a_norm = pert * np.sqrt(2.0 * n_cur) / pert.sum()
        if r % 2 == 0:
            a_out, _ = nesterov_lp_optimize(a_norm, t5, solver, verbose=False, seed=200 + r)
        else:
            a_out, _ = smoothed_max_grad_descent(a_norm, t5, verbose=False, seed=300 + r)
        do_try_update(a_out, f"pert{r}_n{n_cur}")

    # === Phase 6: Try longer length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n2 = len(best_seq) + 200
    print(f"\n=== Phase 6: LP Nesterov n={n2} ===")
    t6 = min(remaining * 0.12, 180.0)
    if t6 > 30:
        init_long = interpolate_sequence(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t6, solver, verbose=True, seed=7)
        do_try_update(a_out, f"lp_long_n{n2}")

    # === Phase 7: Gradient on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: GradDescent n={n_best} ===")
    t7 = min(remaining * 0.10, 150.0)
    if t7 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = smoothed_max_grad_descent(a_norm, t7, verbose=True, seed=99)
        do_try_update(a_out, f"gd2_n{n_best}")

    # === Phase 8: Final LP Nesterov ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: Final LP Nesterov n={n_best} ===")
    t8 = min(remaining * 0.15, 200.0)
    if t8 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t8, solver, verbose=True, seed=777)
        do_try_update(a_out, f"lp_final_n{n_best}")

    # === Phase 9: Final Pair CD + Peak Balance ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 9: Final PCD + PB n={n_best} ===")
    t9 = min(remaining * 0.50, 200.0)
    if t9 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pair_coordinate_descent(a_norm, t9 * 0.6, rng, verbose=True)
        do_try_update(a_out, f"pcd_final_n{n_best}")
        a_norm2 = best_seq * np.sqrt(2.0 * len(best_seq)) / best_seq.sum()
        n_best2 = len(best_seq)
        remaining2 = time_budget - (time.monotonic() - t_global)
        if remaining2 > 20:
            a_out2, _ = peak_balance(a_norm2, min(remaining2 * 0.5, 80.0), rng, verbose=True)
            do_try_update(a_out2, f"pb_final_n{n_best2}")

    # === Final check ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
