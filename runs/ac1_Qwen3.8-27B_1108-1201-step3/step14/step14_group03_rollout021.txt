    Log-sum-exp gradient descent for minimizing max autocorrelation.
    Uses mirror descent (multiplicative) updates for non-negativity.
    """

    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a * norm / s
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    # Need larger FFT for cross-correlation: >= 3n-2
    fft_size_grad = 1 << (3 * n - 2 - 1).bit_length()
    
    best_a = a.copy()

        T = T_start * (T_end / T_start) ** frac

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. Log-sum-exp smooth surrogate gradient descent with projected gradient steps
2. Peak-targeted coordinate descent for local refinement
3. LP-based best response with Nesterov acceleration for large jumps
4. Adaptive length exploration
Key innovation: smooth gradient-based optimization via log-sum-exp surrogate,
combined with LP-based large jumps, to escape local minima better than pure LP or pure CD.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    try:
        import cvxpy as cp
        for s in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                x = cp.Variable(4, nonneg=True)
                prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
                prob.solve(solver=s)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    _global_solver = s
                    return s
            except Exception:
                continue
        _global_solver = cp.ECOS
        return cp.ECOS
    except Exception:
        return None


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(x, target_sum):
    """Project x onto {a >= 0, sum(a) = target_sum}."""
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho) < 1:
        return np.full(n, target_sum / n)
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(x - theta, 0.0)


def compute_lse_grad(a, p, fft_size_grad):
    """Gradient of sum_m p[m]*conv(a,a)[m] w.r.t. a."""
    P = rfft(p, fft_size_grad)
    A = rfft(a, fft_size_grad)
    cross = irfft(np.conj(P) * A, fft_size_grad)
    return 2.0 * cross[:len(a)]


def lse_projected_grad(a0, time_limit, verbose=True):
    """Log-sum-exp projected gradient descent."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm_val / a.sum()

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    fft_size_grad = 1 << (3 * n - 3).bit_length()

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    T_schedule = [8.0, 4.0, 2.5, 1.5, 1.0, 0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08]

    for phase, T in enumerate(T_schedule):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        phases_left = len(T_schedule) - phase
        phase_budget = max(remaining / phases_left * 1.3, 5.0)
        n_steps = max(int(phase_budget * 40), 30)

        eta = T / (norm_val + 1e-10)

        for step in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break

            c = fast_autocorr(a)
            c_max = c.max()
            exp_c = np.exp((c - c_max) / T)
            p_sum = exp_c.sum()
            if p_sum < 1e-300:
                p = np.zeros(size)
                p[np.argmax(c)] = 1.0
            else:
                p = exp_c / p_sum

            grad = compute_lse_grad(a, p, fft_size_grad)

            # Backtracking line search
            step_size = eta
            accepted = False
            for bt in range(8):
                a_trial = a - step_size * grad
                a_trial = project_simplex(a_trial, norm_val)
                score = fast_autocorr(a_trial).max()
                if score <= best_score + 0.01 * T:
                    a = a_trial
                    accepted = True
                    if score < best_score:
                        best_score = score
                        best_a = a.copy()
                    break
                step_size *= 0.5

            if not accepted:
                a = a - step_size * grad
                a = project_simplex(a, norm_val)
                score = fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()

            if verbose and step % 200 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LSE-PGD T={T:.2f} step={step} ({elapsed:.0f}s): best={best_score:.8f}")

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"  LSE-PGD phase T={T:.2f} done ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def peak_cd(a0, time_limit, rng, verbose=True):

    n = len(a0)
    norm_val = np.sqrt(2.0 * n)

    s = a.sum()

    size = 2 * n - 1
    window = min(50, n // 4)

    n_passes = 0
    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            ai = a[i]
            if ai < norm_val * 1e-10:
                continue

            for offset in range(1, window + 1):
                for j in [(i + offset) % n, (i - offset) % n]:
                    if j == i:
                        continue
                    if a[i] < norm_val * 1e-10:
                        break

                    # Compute delta in convolution if we move mass from i to j
                    delta_vec = np.zeros(size)
                    # Remove contribution of a[i]
                    for m in range(n):
                        if 0 <= m <= size - 1:
                            delta_vec[m] -= a[i] * a[m]  # simplified
                    # This is too slow; use FFT-based approach
                    # Actually let me use the formula: delta[m] = 2*(a[j]*a[m-j] - a[i]*a[m-i])... 
                    # Simplified: just evaluate a few candidate moves
                    d = ai * 0.5  # try moving half
                    a[i] -= d
                    a[j] += d
                    new_max = fast_autocorr(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        conv = fast_autocorr(a)
                    else:
                        a[j] -= d

        if verbose and n_passes % 5 == 0:

def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver):
    import cvxpy as cp
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val


def nesterov_lp(a0, time_limit, solver, verbose=True):
    """LP-based best response with Nesterov acceleration."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    y = a.copy()
    t0 = time.monotonic()
    n_iters = 0
    stuck = 0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, solver)
        if b is None:
            stuck += 1
            if stuck > 30:
                break
            continue
        stuck = 0

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm_val / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_val / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm_val / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        if verbose and n_iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pair_cd_window(a0, time_limit, rng, window=30, verbose=True):
    """Pair coordinate descent with window constraint."""
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm_val / a.sum()

    conv = fast_autocorr(a)
    best_score = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    size = 2 * n - 1

    n_passes = 0
    no_improve = 0
    window = min(window, n // 3)

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm_val * 1e-10:
                continue

            for offset in range(1, window + 1):
                for j in [(i + offset) % n, (i - offset) % n]:
                    if j == i:
                        continue
                    if a[i] < norm_val * 1e-10:
                        break

                    # Try moving a fraction of a[i] to a[j]
                    for frac in [0.3, 0.5, 0.7]:
                        d = a[i] * frac
                        a[i] -= d
                        a[j] += d
                        s = a.sum()
                        if s < 1e-15:
                            a[i] += d
                            a[j] -= d
                            break
                        a = a * norm_val / s
                        new_max = fast_autocorr(a).max()
                        if new_max < best_score - 1e-12:
                            best_score = new_max
                            best_a = a.copy()
                            pass_improved = True
                            conv = fast_autocorr(a)
                        else:
                            a[j] -= d
                            a[i] += d
                            a = a * norm_val / a.sum()
                            conv = fast_autocorr(a)

        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and n_passes % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score
