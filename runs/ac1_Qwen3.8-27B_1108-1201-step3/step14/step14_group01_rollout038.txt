                s_new = a_new.sum()
                if s_new > 1e-15:
                    a_new = a_new * norm / s_new
                    a = a_new
                v *= 0.5

            sc = fast_conv_self(a).max()

def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration and adaptive restart."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()
    
    # Nesterov with restart
    momentum = 0.0
    prev_best = best_score
    patience = 0
    MAX_PATIENCE = 200

    max_iters = 300000
    count = 0

    for t in range(1, max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        count += 1

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Solve LP best response
        b = lp_solve_with_generation(y, M, solver, max_rounds=30)
        if b is None:
            # Fallback: use y itself
            b = y.copy()
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Nesterov update with adaptive step
        alpha = 2.0 / (t + 2.0)
        
        if t == 1:
            y_new = g.copy()
        else:
            # Adaptive momentum: restart if objective worsens
            y_new = (1.0 - alpha) * y + alpha * g
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y

        y = y_new

        # Check improvement
        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            patience = 0
        else:
            patience += 1

        # Restart mechanism: if no improvement, reduce momentum
        if patience > MAX_PATIENCE:
            # Reset: jump towards best_a
            y = 0.5 * y + 0.5 * best_a
            y = np.maximum(y, 0.0)
            y = y * norm / y.sum()
            patience = 0

        if verbose and count % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}, patience={patience}")

    return best_a, best_score

def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 10)
    x = np.linspace(0.0, 1.0, n)
    if choice == 0:
        # Skewed: most mass on the left
        a = np.exp(-4.0 * x) + 0.3
    elif choice == 1:
        # Two peaks
        a = np.exp(-15.0 * (x - 0.2) ** 2) + 0.8 * np.exp(-15.0 * (x - 0.8) ** 2)
    elif choice == 2:
        # Ramp down
        a = (1.0 - x) ** 2 + 0.1
    elif choice == 3:
        # Cosine window
        a = np.cos(np.pi * x / 2.0) ** 3
    elif choice == 4:
        # Asymmetric sine
        a = np.sin(np.pi * x * 0.7) ** 2 + 0.1
    elif choice == 5:
        # Hamming-like
        a = 0.54 + 0.46 * np.cos(2 * np.pi * x)
    elif choice == 6:
        # Triangle
        a = 1.0 - np.abs(2.0 * x - 0.3)
    elif choice == 7:
        # Flat with slight variation
        a = 1.0 + 0.2 * np.sin(4 * np.pi * x)
    elif choice == 8:
        # Geometric decay
        a = 0.95 ** (x * n)
    else:
        # Random smooth
        k = 5
        coeffs = rng.standard_normal(k)
        a = sum(c * np.cos(2*np.pi*j*x + rng.standard_normal()) for j, c in enumerate(coeffs, 1))
        a = np.abs(a) + 0.1
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def fast_lp_solve(a, M, solver, active_constraints=None):
    """Fast LP solve with pre-built constraint matrix."""
    n = len(a)
    size = 2 * n - 1
    
    if active_constraints is not None and len(active_constraints) > 0:
        rows = active_constraints
    else:
        rows = np.arange(size)
    
    # Build constraint matrix efficiently
    n_constraints = len(rows)
    T = np.zeros((n_constraints, n))
    for ii, k in enumerate(rows):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if hi > lo:
            T[ii, lo:hi] = a[lo:hi]
    
    b_var = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    val = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(val, 0.0)

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_solver = None

def get_solver():
    global _solver
    if _solver is not None:
        return _solver
    for s in [cp.CLARABEL, cp.MOSEK, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver = s
                print(f"  Using solver: {s}")
                return s
        except Exception:
            continue
    _solver = cp.ECOS
    print(f"  Using solver (fallback): ECOS")

def fast_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_constraint_matrix(a, rows):
    """Build rows of the convolution matrix T where T[k,j] = a[j] if 0<=k-j<n."""
    n = len(a)
    T = np.zeros((len(rows), n))
    for ii, k in enumerate(rows):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if hi > lo:
            T[ii, lo:hi] = a[lo:hi]
    return T


def lp_solve_gen(a, M, solver, max_rounds=40):
    """LP best response with constraint generation."""
    n = len(a)
    size = 2 * n - 1

    # Initial constraint set: peak neighborhood + evenly spaced
    conv_a = fast_conv_self(a)
    peak_idx = int(np.argmax(conv_a))

    n_init = min(40, size)
    spaced = np.linspace(0, size - 1, n_init, dtype=int)
    neighbors = np.arange(max(0, peak_idx - 10), min(size, peak_idx + 11))
    active = np.unique(np.concatenate([spaced, neighbors]))
    active = active.astype(int)

    for _ in range(max_rounds):
        T = build_constraint_matrix(a, active)
        b_var = cp.Variable(n, nonneg=True)
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M])
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            return None
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None
        b = np.asarray(b_var.value, dtype=np.float64)
        b = np.maximum(b, 0.0)
        if b.sum() < 1e-15:
            return None

        # Check for violated constraints
        conv_ab = fast_conv_cross(a, b)
        tol = M * 1e-5 + 1e-12
        violated = np.where(conv_ab > M + tol)[0]
        if len(violated) == 0:
            return b

        # Add only the worst violators
        if len(violated) > 30:
            worst = violated[np.argsort(conv_ab[violated])[-30:]]
            violated = worst
        active = np.unique(np.concatenate([active, violated]))

    # Final solve with all accumulated constraints
    T = build_constraint_matrix(a, active)
    b_var = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [T @ b_var <= M])
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(b, 0.0)


def lp_solve_full(a, M, solver, active=None):
    """LP solve with given active constraints or all constraints."""
    n = len(a)
    size = 2 * n - 1
    if active is None:
        active = np.arange(size)
    T = build_constraint_matrix(a, active)
    b_var = cp.Variable(n, nonneg=True)

    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b = np.asarray(b_var.value, dtype=np.float64)
    return np.maximum(b, 0.0)


def lse_gradient_descent(a0, time_limit, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    n_stages = 10
    temps = np.geomspace(40.0, 0.1, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        n_steps = max(20, min(400, int(remaining * 10)))

        v = np.zeros(n)
        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_conv_self(a)
            max_c = conv.max()
            if max_c < 1e-15:
                break
            shifted = (conv - max_c) / T_val
            shifted = np.maximum(shifted, -200.0)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()

            A = rfft(a, fft_size)
            W = rfft(w, fft_size)
            ca = irfft(np.conj(A) * W, fft_size)
            g1 = ca[:n]
            aw = irfft(A * W, fft_size)
            g2 = aw[:n]
            grad = 2.0 * (g1 + g2)

            beta = 0.9
            v = beta * v + (1.0 - beta) * grad
            grad_norm = np.linalg.norm(v)
            if grad_norm < 1e-15:
                step = 0.0
            else:
                step = norm / (T_val * np.sqrt(n)) / (1.0 + 0.03 * it)

            a_new = a - step * v
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s_new
            a = a_new

            sc = fast_conv_self(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if verbose:
            print(f"    LSE stage {si} T={T_val:.2f}: best={best_score:.8f}")

    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """LP best-response with Nesterov acceleration, constraint generation, adaptive restart, and averaging."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    y = a.copy()

    # Averaging
    avg = np.zeros(n)
    n_avg = 0

    patience = 0
    MAX_PATIENCE = 300
    best_at_restart = a.copy()

    count = 0
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        count += 1

        conv_y = fast_conv_self(y)
        M = conv_y.max()
        if M < 1e-12:
            break

        # Solve LP best response
        b = lp_solve_gen(y, M, solver, max_rounds=30)
        if b is None:
            continue
        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum

        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        if t == 1:
            y_new = g.copy()
        else:
            y_new = (1.0 - alpha) * y + alpha * g
            y_new = np.maximum(y_new, 0.0)
            s_y = y_new.sum()
            if s_y < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new * norm / s_y

        y = y_new

        # Track average
        avg += y
        n_avg += 1

        # Check y
        sc_y = fast_conv_self(y).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            best_at_restart = y.copy()
            patience = 0

        # Check average periodically
        if n_avg % 100 == 0 and n_avg >= 200:
            avg_candidate = avg / n_avg
            s_avg = avg_candidate.sum()
            if s_avg > 1e-15:
                avg_candidate = avg_candidate * norm / s_avg
                sc_avg = fast_conv_self(avg_candidate).max()
                if sc_avg < best_score:
                    best_score = sc_avg
                    best_a = avg_candidate.copy()
                    patience = 0

        if verbose and count % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP Nesterov t={t} ({elapsed:.0f}s): best={best_score:.8f}, pat={patience}")

        # Patience-based restart
        if sc_y >= best_score:
            patience += 1
        if patience > MAX_PATIENCE:
            # Restart: blend current with best
            y = 0.3 * y + 0.7 * best_at_restart
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y * norm / s_y
            patience = 0

    return best_a, best_score

def coordinate_descent(a0, time_limit, rng, verbose=True):
    """Efficient coordinate descent: transfer mass between pairs using gain vectors."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, compute_score(a)
    a = a * norm / s
    best_a = a.copy()
    best_score = fast_conv_self(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    total_improvements = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve >= 4:
            break
        conv = fast_conv_self(a)
        cur_max = conv.max()
        perm = rng.permutation(n)
        pass_improved = False

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(perm[idx])
            ai = a[i]
            if ai < norm * 1e-10:
                continue

            # Compute gain vector for removing mass at i
            # gain_i[k] = a[k-i] + a[i-k] (contribution of a[i] to conv[k])
            gain_i = np.zeros(size)
            lo1 = i
            hi1 = min(size, i + n)
            gain_i[lo1:hi1] += a[0:hi1 - lo1]
            lo2 = max(0, i - n + 1)
            hi2 = i + 1
            gain_i[lo2:hi2] += a[i - np.arange(lo2, hi2)]

            # Try several candidate j's
            n_candidates = 12
            j_list = list(rng.integers(0, n, size=n_candidates))
            # Add structured candidates: positions where a is small
            small_positions = np.argsort(a)[:5]
            j_list.extend(int(x) for x in small_positions)
            # Add position n//2 away from peak
            peak_k = int(np.argmax(conv))
            j_list.append(int((peak_k - i + n) % n))
            j_list = [j for j in set(j_list) if j != i]

            best_delta = 0.0
            best_v = cur_max
            best_j = -1

            for j in j_list[:15]:
                if j == i:
                    continue
                aj = a[j]

                # Gain vector for adding mass at j
                gain_j = np.zeros(size)
                lo3 = j
                hi3 = min(size, j + n)
                gain_j[lo3:hi3] += a[0:hi3 - lo3]
                lo4 = max(0, j - n + 1)
                hi4 = j + 1
                gain_j[lo4:hi4] += a[j - np.arange(lo4, hi4)]

                # Net gain: adding at j, removing from i
                gain = gain_j - gain_i

                # Try delta values
                for frac in [0.15, 0.3, 0.5, 0.7, 0.85]:
                    d = ai * frac
                    new_conv = conv + d * gain
                    v = new_conv.max()
                    if v < best_v - 1e-12:
                        best_v = v
                        best_delta = d
                        best_j = j

            if best_j >= 0 and best_delta > 0:
                a[i] -= best_delta
                a[best_j] += best_delta
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * norm / s_new
                    new_max = fast_conv_self(a).max()
                    if new_max < best_score - 1e-12:
                        best_score = new_max
                        best_a = a.copy()
                        pass_improved = True
                        total_improvements += 1
                    else:
                        a[best_j] -= best_delta
                        a[i] += best_delta
                        a = a * norm / a.sum()

        if not pass_improved:
            no_improve += 1
            # Perturb
            n_pert = max(3, n // 200)
            idx_p = rng.choice(n, size=n_pert, replace=False)
            a[idx_p] *= (1.0 + 0.1 * rng.standard_normal(n_pert))
            a = np.maximum(a, 0.0)
            a = a * norm / a.sum()
        else:
            no_improve = 0

        if verbose and total_improvements % 200 == 0 and total_improvements > 0:
            elapsed = time.monotonic() - t0
            print(f"    CD improvements={total_improvements} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search: minimize 2n*max(conv(a,a))/sum(a)^2 over non-negative sequences."""
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)
    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = eval_fn(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if best_seq is None:
        n_base = 1600
        for trial in range(4):
            a_init = structured_init(n_base, rng)
            a_out, _ = lse_gradient_descent(a_init, 40.0, verbose=False)
            try_update(a_out, f"init{trial}")
    else:
        n_base = len(best_seq)

    print(f"\nMain optimization: n={n_base}, score={best_score:.6f}")

    # Phase 1: LSE GD (8% time)
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = min(remaining * 0.08, 100.0)
    print(f"\n=== Phase 1: LSE GD (budget={t1:.0f}s) ===")
    if t1 > 15:
        a_out, _ = lse_gradient_descent(best_seq, t1, verbose=True)
        try_update(a_out, "lse1")

    # Phase 2: LP Nesterov - main (40% time)
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = min(remaining * 0.45, 650.0)
    print(f"\n=== Phase 2: LP Nesterov (budget={t2:.0f}s) ===")
    if t2 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t2, solver, verbose=True)
        try_update(a_out, "lp_main")

    # Phase 3: CD polish (12% time)
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 3: CD Polish (budget={t3:.0f}s) ===")
    if t3 > 15:
        a_out, _ = coordinate_descent(best_seq, t3, rng, verbose=True)
        try_update(a_out, "cd1")

    # Phase 4: LP round 2 (20% time)
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = min(remaining * 0.25, 300.0)
    print(f"\n=== Phase 4: LP Round 2 (budget={t4:.0f}s) ===")
    if t4 > 20:
        a_out, _ = nesterov_lp_optimize(best_seq, t4, solver, verbose=True)
        try_update(a_out, "lp2")

    # Phase 5: CD round 2 (10% time)
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.15, 150.0)
    print(f"\n=== Phase 5: CD Round 2 (budget={t5:.0f}s) ===")
    if t5 > 10:
        a_out, _ = coordinate_descent(best_seq, t5, rng, verbose=False)
        try_update(a_out, "cd2")

    # Phase 6: Try longer length
    remaining = time_budget - (time.monotonic() - t_global)
    n_long = n_base + 400
    print(f"\n=== Phase 6: Longer n={n_long} (remaining={remaining:.0f}s) ===")
    if remaining > 100:
        init_long = interpolate_seq(best_seq, n_long)
        t6 = min(remaining * 0.45, 180.0)
        a_out, _ = nesterov_lp_optimize(init_long, t6, solver, verbose=False)
        try_update(a_out, f"lp_long_{n_long}")

        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 40 and best_seq is not None:
            t6b = min(remaining * 0.4, 80.0)
            a_out2, _ = coordinate_descent(best_seq, t6b, rng, verbose=False)
            try_update(a_out2, "cd_long")

    # Final polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 30 and best_seq is not None:
        print(f"\n=== Final polish (remaining={remaining:.0f}s) ===")
        t_fp = min(remaining * 0.6, 60.0)
        a_out, _ = lse_gradient_descent(best_seq, t_fp, verbose=False)
        try_update(a_out, "lse_final")
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            a_out, _ = coordinate_descent(best_seq, min(remaining * 0.7, 30.0), rng, verbose=False)
            try_update(a_out, "cd_final")

    # Final evaluation
    if best_seq is not None:
        final_score = eval_fn(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
