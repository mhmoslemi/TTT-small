def build_cross_matrix(a, n):
    """Build the (2n-1) x n matrix C where (C @ b)_k = sum_i a_{k-i} b_i."""
    size = 2 * n - 1
    C = np.zeros((size, n))
    for i in range(n):
        start = i
        end = min(i + n, size)
        idx = np.arange(start, end)
        C[idx, i] = a[idx - i]
    return C


def lp_best_response(a, M, solver, C=None):
    """Solve: max sum(b) s.t. C @ b <= M, b >= 0."""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    if C is None:
        C = build_cross_matrix(a, n)
    constraints = [C @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None, None
    b_val = np.maximum(b_val, 0.0)
    # Try to get dual variables
    dual_vals = None
    try:
        dual_vals = np.asarray(prob.constraints[0].dual_value, dtype=np.float64)
        if dual_vals is not None:
            dual_vals = np.maximum(dual_vals, 0.0)
    except Exception:
        pass
    return b_val, dual_vals

        # alpha_t = 2/(t+2)
        # y_{t+1} = (1-alpha_t) y_t + alpha_t * g
        # x_{t+1} = y_{t+1} + ((t)/(t+2)) * (y_{t+1} - y_t)
        
        alpha = 2.0 / (t + 2.0)
        
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm_val / s_y

        if t == 1:
            x_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            x_new = y_new + momentum * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new * norm_val / s_x

        y = y_new
        x = x_new

        # Check improvement
        score_y = fast_autocorr(y).max()
        if score_y < best_score - 1e-14:
            best_score = score_y
            best_a = y.copy()
            last_improve = time.monotonic()

        score_x = fast_autocorr(x).max()
        if score_x < best_score - 1e-14:
            best_score = score_x
            best_a = x.copy()
            last_improve = time.monotonic()

        # Running average (helps with convergence)
        avg_x += x
        avg_count += 1
        if avg_count % 100 == 0:
            avg_candidate = avg_x / avg_count
            s_avg = avg_candidate.sum()
            if s_avg > 1e-15:
                avg_candidate = avg_candidate * norm_val / s_avg
                score_avg = fast_autocorr(avg_candidate).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_candidate.copy()
                    last_improve = time.monotonic()

        # Stall detection
        if time.monotonic() - last_improve > 90.0:
            stall_count += 1
            if stall_count > 8:
                if verbose:
                    print(f"    Stalling at iter={n_iters}, best={best_score:.8f}")
                break
        else:
            stall_count = 0

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def multi_peak_coord_descent(a0, time_limit, seed=42):
    """
    Coordinate descent that targets multiple peaks simultaneously.
    Tries to reduce the top-3 peaks by moving mass from peak-contributing positions.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm_val / n
    else:
        a = a * norm_val / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        
        # Find top 3 peaks
        top_peaks = np.argsort(b)[::-1][:3]
        
        # For each peak, find the top contributing pairs
        candidates = []
        for k_peak in top_peaks:
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_peak - idx]
            # Sort by contribution
            sorted_pos = np.argsort(contrib)[::-1]
            top_pos = idx[sorted_pos[:10]]
            for p in top_pos:
                candidates.append((p, k_peak))
        
        # Also find low-value positions (good targets for receiving mass)
        low_positions = np.argsort(a)[:20]
        
        improved = False
        # Try targeted moves
        for src, k_peak in candidates[:40]:
            if a[src] < 1e-12:
                continue
            for dst in low_positions[:12]:
                if dst == src:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.08, 0.12, 0.2]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t < 1e-10:
                        continue
                    a_trial = a_trial * norm_val / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial.copy()
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Random moves
            for _ in range(200):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j or a[i] < 1e-12:
                    continue
                frac = rng.uniform(0.005, 0.1)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial * norm_val / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial.copy()
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            # Try block moves
            improved = False
            for _ in range(50):
                # Move a block of consecutive elements
                block_size = rng.integers(2, 6)
                start = rng.integers(0, n - block_size)
                i1 = start
                i2 = start + block_size - 1
                j1 = rng.integers(0, n - block_size)
                if i1 == j1:
                    continue
                j2 = j1 + block_size - 1
                amt = a[i1:i2+1].sum() * rng.uniform(0.1, 0.3)
                # Scale the block
                block_sum = a[i1:i2+1].sum()
                if block_sum < 1e-12:
                    continue
                scale = amt / block_sum
                a_trial = a.copy()
                a_trial[i1:i2+1] -= a[i1:i2+1] * scale
                # Distribute to target
                a_trial[j1:j2+1] += amt / block_size
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial = a_trial * norm_val / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial.copy()
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            stall += 1
            if stall > 200:
                break
        else:
            stall = 0

    return best_a, best_score


def lse_gradient_optimize(a0, time_limit, seed=42):
    """
    Optimize using log-sum-exp smoothed objective with L-BFGS-B.
    Uses log-parameterization to ensure positivity.
    """
    n = len(a0)
    norm_val = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-10)
    s0 = a.sum()
    a = a * norm_val / s0

    # Log parameterization: x = log(a), so a = exp(x)
    # But we need to handle the normalization constraint.
    # Alternative: use x directly with softplus.
    # Let's just use the standard parameterization with bounds.
    x0 = np.asarray(a, dtype=np.float64).copy()
    
    best_a = x0.copy()
    best_score = fast_autocorr(x0).max()
    t0 = time.monotonic()

    def lse_obj_grad(x, T):
        a = np.maximum(x, 1e-15)
        s = a.sum()
        if s < 1e-10:
            return 1e10, np.zeros(len(x))
        b = fast_autocorr(a)
        max_b = b.max()
        if max_b < 1e-15:
            return 1e10, np.zeros(len(x))
        
        # Log-sum-exp for stability
        shifted = (b - max_b) * T
        exp_vals = np.exp(shifted)
        log_sum_exp = max_b + np.log(exp_vals.sum()) / T
        w = exp_vals / exp_vals.sum()
        
        # Gradient of log_sum_exp w.r.t. a:
        # d/d a_i [log sum_k exp(T * b_k)] / T
        # = sum_k w_k * T * (a * e_i)_k / T = sum_k w_k * (a_i if k >= i and k-i < n)
        # = (w * a)_i... wait, let me think more carefully.
        # b_k = sum_j a_j a_{k-j}
        # db_k / da_i = a_{k-i} + a_i... no.
        # b_k = sum_j a_j * a_{k-j}
        # db_k / da_i = a_{k-i} (if 0 <= k-i < n) + a_i (if 0 <= k-i... wait)
        # Actually: b_k = sum_j a_j * a_{k-j}
        # If we vary a_i: db_k/da_i appears when j=i (giving a_{k-i}) or when k-j=i i.e. j=k-i (giving a_{k-i} again if k-i != i, or a_i^2 derivative = 2*a_i if k-i=i)
        # So db_k/da_i = a_{k-i} * [0 <= k-i < n] + a_{k-i} * [0 <= k-i < n and k-i != i] + ... 
        
        # Actually, b_k = sum_j a_j a_{k-j}. The derivative w.r.t. a_i:
        # = a_{k-i} (from term j=i) + a_i (from term j=k-i, if k-i != i) 
        # Wait no. Let me be more careful.
        # b_k = sum_{j: 0<=j<n, 0<=k-j<n} a_j * a_{k-j}
        # db_k/da_i = a_{k-i} [if 0<=k-i<n] + a_{k-i} [if k-i==i, i.e., k=2i] 
        # Hmm, actually:
        # b_k = sum_j a_j * a_{k-j}
        # The terms involving a_i are: j=i (gives a_i * a_{k-i}) and j=k-i (gives a_{k-i} * a_i)
        # If k-i != i, these are two distinct terms, each contributing a_{k-i} to db_k/da_i.
        # If k-i == i (i.e., k=2i), it's a single term a_i^2, contributing 2*a_i.
        # So db_k/da_i = a_{k-i} * 2 [if 0<=k-i<n and k-i != i] + 2*a_i [if k=2i and 0<=i<n]
        # = 2 * a_{k-i} [if 0<=k-i<n] (this works because if k-i=i, a_{k-i}=a_i, and we get 2*a_i)
        
        # So the gradient of the LSE w.r.t. a_i is:
        # sum_k w_k * 2 * a_{k-i} [0<=k-i<n] / ... 
        # This is 2 * (w * a)_i where (w * a) is the correlation of w and a.
        
        # Actually, let me just compute it via FFT for efficiency.
        # grad_a = 2 * corr(w, a) where (corr(w,a))_i = sum_k w_k * a_{k-i} [0<=k-i<n]
        
        fft_size = 1 << (2 * n - 1).bit_length()
        w_full = np.zeros(fft_size)
        w_full[:2*n-1] = w
        a_full = np.zeros(fft_size)
        a_full[:n] = a
        
        # We want: grad_a_i = 2 * sum_k w_k * a_{k-i} for valid k-i
        # This is the cross-correlation of w and a.
        # cross_corr_i = sum_k w_{k+i} * a_k = (w flipped) * a at lag i
        # Or: sum_k w_k * a_{k-i} = (a * w_reversed)_i... 
        
        # Let me use the definition: (w * a)_k = sum_j w_j a_{k-j}
        # We want sum_k w_k a_{k-i} for fixed i.
        # Let m = k-i, so k = m+i: sum_m w_{m+i} a_m
        # This is the cross-correlation: (w ∘ a)(i) = sum_m w(m+i) a(m)
        # In FFT terms: if W = FFT(w), A = FFT(a), then cross-correlation = IFFT(conj(W) * A)
        # But we need to be careful about the shift.
        
        # Actually, let's just compute it directly for clarity (n is at most ~2000):
        # grad_a_i = 2 * sum_{k: 0<=k<2n-1, 0<=k-i<n} w_k * a_{k-i}
        
        # For efficiency, note that this is the convolution of w with flipped a, evaluated at specific points.
        # Let a_flip = a[::-1]. Then (w * a_flip)_m = sum_j w_j * a_flip[m-j] = sum_j w_j * a[n-1-(m-j)]
        # This is getting complicated. Let me just do it directly.
        
        # More efficient: note that sum_k w_k a_{k-i} is the i-th element of the "anti-convolution"
        # which can be computed as: take the FFT of w (padded to 2n-1) and FFT of a (padded to 2n-1)
        # then the cross-correlation is IFFT(conj(FFT(w)) * FFT(a))
        
        W_fft = rfft(w_full, fft_size)
        A_fft = rfft(a_full, fft_size)
        
        # Cross-correlation: (w ∘ a)(i) = sum_m w(m+i) a(m)
        # In frequency domain: IFFT(conj(W) * A) gives sum_m w(m) a(m+i) which is (a ∘ w)(i)
        # We want sum_m w(m+i) a(m) = (w ∘ a)(i) = IFFT(conj(A) * W)[i]... 
        # Hmm, let me just verify with a small example numerically and use the correct formula.
        
        # For now, let's compute it as: (W_fft * conj(A_fft)) -> IFFT gives sum w(m) a(m+i) shifted
        # Actually the standard cross-correlation in numpy is:
        # np.correlate(w, a, mode='full')[n-1+i] = sum_m w[m] * a[m+i] (for valid i)
        # But we want sum_m w[m+i] * a[m] which is np.correlate(a, w, mode='full')[n-1-i]... 
        
        # This is getting too complicated for the FFT approach. Let me just compute it directly.
        # For n ~ 2000, a direct O(n * (2n)) computation is 4 * 10^6 which is fine.
        
        grad_a = np.zeros(n)
        # For each i, grad_a_i = 2 * sum_k w[k] * a[k-i] where 0 <= k-i < n
        # i.e., k ranges from i to i+n-1 (clamped to [0, 2n-2])
        for i in range(n):
            k_start = i
            k_end = min(i + n, 2*n - 1)
            grad_a[i] = 2.0 * np.dot(w[k_start:k_end], a[0:k_end-k_start])
        
        # Gradient of objective = 2*n*log_sum_exp / s^2
        # d(obj)/da_i = 2*n * grad_a_i / s^2 - 2*obj * a_i / s
        obj_val = 2.0 * n * log_sum_exp / (s * s)
        grad = 2.0 * n * grad_a / (s * s) - 2.0 * obj_val * a / s
        
        return obj_val, grad

    # Temperature schedule
    temps = [100.0, 50.0, 20.0, 10.0, 5.0, 3.0]
    n_stages = len(temps)
    stage_time = time_limit / n_stages

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 15:
            break
        this_time = min(remaining / (n_stages - si), stage_time)

        def obj_func(x):
            val, _ = lse_obj_grad(x, T)
            return val

        def grad_func(x):
            _, g = lse_obj_grad(x, T)
            return g

        try:
            res = scipy_minimize(obj_func, x0.copy(), method='L-BFGS-B',
                                  jac=grad_func, bounds=[(1e-15, None)] * n,
                                  options={'maxiter': 200, 'maxfun': 500,
                                           'gtol': 1e-9, 'ftol': 1e-12,
                                           'maxls': 30})
            if res.x is not None:
                a_candidate = np.maximum(res.x, 1e-15)
                s_c = a_candidate.sum()
                if s_c > 1e-10:
                    a_candidate = a_candidate * norm_val / s_c
                    score_c = fast_autocorr(a_candidate).max()
                    if score_c < best_score:
                        best_score = score_c
                        best_a = a_candidate.copy()
                        x0 = a_candidate.copy()
                        print(f"    LSE T={T:.0f}: score={score_c:.8f}")
        except Exception as e:
            print(f"    LSE T={T:.0f} error: {e}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def structured_init(n, rng):
    """Generate a structured initial sequence."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-3.0 * (x - 0.5) ** 2)
    elif choice == 5:
        center = n // 2
        idx = np.arange(n) - center
        sigma = n / 5.0
        a = np.exp(-0.5 * (idx / sigma) ** 2)
    elif choice == 6:
        # Quadratic residue-inspired pattern
        a = np.ones(n)
        for i in range(n):
            a[i] = 1.0 + 0.3 * np.sin(2.0 * np.pi * i * np.sqrt(2))
    else:
        # Low-discrepancy inspired
        golden_ratio = (1 + np.sqrt(5)) / 2
        idx = np.arange(n)
        a = 1.0 + 0.2 * np.sin(2.0 * np.pi * idx / golden_ratio)
    
    a = np.maximum(a, 0.01)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def attempt_update(best_seq, best_score, candidate_a, label, t_global):
    """Try to update best state."""
    if candidate_a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(candidate_a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a.copy()
        elapsed = time.monotonic() - t_global
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Uses LP best-response with Nesterov acceleration as the
    primary method, supplemented by coordinate descent and LSE gradient refinement.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Try to use parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # === Phase 1: LP Nesterov (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 1: LP-Nest n={n_best} (budget={remaining:.0f}s) ===")
    t1 = remaining * 0.35
    a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
    a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True, max_its=100000)
    best_seq, best_score = attempt_update(best_seq, best_score, a_out, f"lp1_n{n_best}", t_global)

    # === Phase 2: Multi-peak coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Multi-peak coord descent n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 150.0)
    if t2 > 20:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = multi_peak_coord_descent(a_norm, t2, seed=42)
        best_seq, best_score = attempt_update(best_seq, best_score, a_out, "coord1", t_global)

    # === Phase 3: LP Nesterov round 2 ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: LP-Nest round 2 n={n_best} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.25, 300.0)
    if t3 > 30:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
