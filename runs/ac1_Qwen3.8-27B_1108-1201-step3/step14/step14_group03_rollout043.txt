def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def lse_smoothed_objective_and_grad(a, T):
    """
    Compute the log-sum-exp smoothed value and gradient of max(conv(a,a)).
    Returns (lse_value, gradient_array_of_length_n).
    """
    n = len(a)
    size = 2 * n - 1
    
    conv = fast_autocorr(a)
    max_conv = conv.max()
    shifted = (conv - max_conv) / T
    exp_shifted = np.exp(shifted)
    lse_value = max_conv + T * np.log(exp_shifted.sum())
    w = exp_shifted / exp_shifted.sum()
    
    # Need FFT size for cross-convolution of a (len n) and w (len size)
    # Result length: n + size - 1 = 3n - 2
    conv_fft_size = 1 << (3 * n - 2 - 1).bit_length()
    
    A_full = rfft(a, conv_fft_size)
    W_full = rfft(w, conv_fft_size)
    
    # Part A: sum_j a[j] * w[j+i] -> IFFT(conj(FFT(a)) * FFT(w))[:n]
    part_a = irfft(np.conj(A_full) * W_full, conv_fft_size)[:n]
    # Part B: sum_j a[j] * w[i-j] -> IFFT(FFT(a) * FFT(w))[:n]
    part_b = irfft(A_full * W_full, conv_fft_size)[:n]
    
    grad = part_a + part_b
    return lse_value, grad

def lse_gradient_descent(a0, time_limit, verbose=True):
    """
    Smoothed gradient descent with multiplicative (mirror) updates.
    Anneals T from high to low.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0
    
    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Annealing schedule: T from T_start to T_end
    T_start = 2.0
    T_end = 0.05
    n_total_iters = 5000
    
    for it in range(n_total_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Anneal T
        frac = it / max(1, n_total_iters - 1)
        T = T_start * (T_end / T_start) ** frac
        
        lse_val, grad = lse_smoothed_objective_and_grad(a, T)
        
        # Multiplicative update: a_i <- a_i * exp(-eta * grad_i)
        # Choose eta adaptively
        eta = 1.0 / T  # natural scale
        
        # Clip to avoid overflow
        log_update = -eta * grad
        log_update = np.clip(log_update, -50, 50)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-15)  # keep strictly positive for multiplicative updates
        
        # Rescale
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s
        
        a = a_new
        
        # Check score
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-GD iter={it} T={T:.4f} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def projected_lse_gd(a0, time_limit, verbose=True):
    """

    """
    n = len(a0)

    t0 = time.monotonic()
    
    T_start = 3.0
    T_end = 0.02
    n_total = 8000
    
    for it in range(n_total):
        if time.monotonic() - t0 > time_limit:
            break
        
        frac = it / max(1, n_total - 1)
        T = T_start * (T_end / T_start) ** frac
        
        lse_val, grad = lse_smoothed_objective_and_grad(a, T)
        
        # Backtracking line search
        alpha = 1.0 / (it + 10)
        for _ in range(20):
            a_trial = a - alpha * grad
            # Project onto {x >= 0, sum(x) = norm}
            a_proj = _project_simplex(a_trial, norm)
            score = fast_autocorr(a_proj).max()
            # Simple improvement check
            if score < lse_val + 0.1 * T:  # relaxed condition
                break
            alpha *= 0.5
        
        a = _project_simplex(a - alpha * grad, norm)
        
        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PGD iter={it} T={T:.4f} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def _project_simplex(v, target_sum):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64)
    # Water-filling: find tau such that sum(max(v - tau, 0)) = target_sum
    # Use bisection
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.maximum(v - mid, 0.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2
    result = np.maximum(v - tau, 0.0)
    # Fix rounding
    s = result.sum()
    if s > 0:
        result = result * target_sum / s
    return result

def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def lp_best_response(a, M, solver):
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

            continue

        n_iters += 1

            s_a = a_new.sum()
            if s_a < 1e-15:

            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 10 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:

                    best_a = avg_check.copy()

        if verbose and n_iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_value_and_grad(a, T):
    """Compute LSE-smoothed max of conv(a,a) and its gradient w.r.t. a."""
    n = len(a)
    size = 2 * n - 1
    conv = fast_autocorr(a)
    max_c = conv.max()
    shifted = (conv - max_c) / T
    exp_s = np.exp(shifted)
    lse = max_c + T * np.log(exp_s.sum())
    w = exp_s / exp_s.sum()

    # grad[i] = 2 * sum_m a[m] * w[m+i]
    # = 2 * (rev_a * w)[n-1+i] where * is linear convolution
    rev_a = a[::-1]
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    Ra = rfft(rev_a, fft_size)
    W = rfft(w, fft_size)
    conv_result = irfft(Ra * W, fft_size)
    grad = 2.0 * conv_result[n - 1: 2 * n - 1]
    return lse, grad


def lse_mirror_descent(a0, time_limit, verbose=True):
    """
    LSE-smoothed mirror descent (multiplicative updates) with T annealing.
    This is the primary new method.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    T_start = 3.0
    T_end = 0.03
    n_total = 20000

    for it in range(n_total):
        if time.monotonic() - t0 > time_limit:
            break

        frac = it / max(1, n_total - 1)
        # Exponential annealing
        T = T_start * (T_end / T_start) ** frac

        lse_val, grad = lse_value_and_grad(a, T)

        # Mirror descent (multiplicative) update:
        # a_i <- a_i * exp(-eta * grad_i)
        # Optimal eta for LSE smoothing is eta = 1/T (dual of the entropy-regularized problem)
        eta = 1.0 / T
        log_update = -eta * grad
        # Clip for numerical stability
        log_update = np.clip(log_update, -80.0, 80.0)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, norm * 1e-15)

        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s

        a = a_new

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-MD iter={it} T={T:.4f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def _project_simplex(v, target_sum):
    """Project v onto {x >= 0, sum(x) = target_sum} via water-filling."""
    v = np.asarray(v, dtype=np.float64)
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.maximum(v - mid, 0.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    result = np.maximum(v - tau, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result * target_sum / s
    return result


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver):
    n = len(a)
    size = 2 * n - 1
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)

    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val

            for k in range(max(0, i), min(size, i + n)):
                cross[k] += a_i0[k - i]
            for k in range(max(0, i - n + 1), min(size, i + 1)):
                cross[k] += a_i0[i - k] if 0 <= i - k < n else 0.0

            # Self term at k = 2i
            self_idx = 2 * i
            has_self = self_idx < size

            # Find best v by sampling
            n_samples = 20
            v_candidates = np.linspace(0.0, old_val * 1.5, n_samples + 1)
            best_v = old_val
            best_max = conv.max()

            for v in v_candidates:
                conv_new = conv_base + v * cross
                if has_self:
                    conv_new[self_idx] += v * v
                new_max = conv_new.max()
                if new_max < best_max:
                    best_max = new_max
                    best_v = v

            if abs(best_v - old_val) > old_val * 1e-6:
                a[i] = best_v
                s = a.sum()
                if s > 1e-15:
                    a = a * norm / s
                    conv = fast_autocorr(a)
                    cur_max = conv.max()
                    if cur_max < best_score:
                        best_score = cur_max
                        best_a = a.copy()
                        pass_improved = True

        if verbose and n_passes % 2 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

        if not pass_improved:
            break

    return best_a, best_score


def pair_cd_fast(a0, time_limit, rng, window=15, verbose=True):
    """
    Fast pair coordinate descent: move mass between nearby pairs.
    """
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        return a, float('inf')
    a = a * norm / s0

    conv = fast_autocorr(a)
    best_a = a.copy()
    best_score = conv.max()
    t0 = time.monotonic()
    size = 2 * n - 1

    no_improve = 0
    while time.monotonic() - t0 < time_limit and no_improve < 3:
        pass_improved = False
        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            if a[i] < norm * 1e-10:
                continue

            for offset in range(1, window + 1):
                j = (i + offset) % n
                if j == i or a[i] < norm * 1e-10:
                    continue
                if a[i] < norm * 1e-8:
                    break

                # Try moving a fraction of a[i] to j
                # Use 5 sample fractions
                best_frac = 0.0
                best_new_max = conv.max()
                fracs = [0.1, 0.2, 0.3, 0.5, 0.7]
                for frac in fracs:
                    delta = frac * a[i]
                    # Approximate new conv:
                    # Change from removing delta from i and adding to j
                    # conv changes by: -delta*(a[k-i]+a[i-k]) + delta*(a[k-j]+a[j-k]) approximately
                    # (ignoring second-order terms for speed)
                    change = np.zeros(size)
                    for k in range(max(0, i), min(size, i + n)):
                        change[k] -= delta * a[k - i]
                    for k in range(max(0, i - n + 1), min(size, i + 1)):
                        if 0 <= i - k < n:
                            change[k] -= delta * a[i - k]
                    for k in range(max(0, j), min(size, j + n)):
                        change[k] += delta * a[k - j]
                    for k in range(max(0, j - n + 1), min(size, j + 1)):
                        if 0 <= j - k < n:
                            change[k] += delta * a[j - k]

                    new_conv = conv + change
                    # Fix self-terms
                    if 2 * i < size:
                        new_conv[2 * i] = conv[2 * i] - 2 * a[i] * delta + delta * delta
                    if 2 * j < size:
                        new_conv[2 * j] = conv[2 * j] + 2 * a[j] * delta + delta * delta

                    new_max = new_conv.max()
                    if new_max < best_new_max:
                        best_new_max = new_max
                        best_frac = frac

                if best_frac > 0:
                    delta = best_frac * a[i]
                    a[i] -= delta
                    a[j] += delta
                    s = a.sum()
                    if s > 1e-15:
                        a = a * norm / s
                        conv = fast_autocorr(a)
                        cur_max = conv.max()
                        if cur_max < best_score - 1e-12:
                            best_score = cur_max
                            best_a = a.copy()
                            pass_improved = True
                        else:
                            # Revert
                            a[j] -= delta
                            a[i] += delta
                            a = a * norm / a.sum()
                            conv = fast_autocorr(a)

        if not pass_improved:
            no_improve += 1
        else:
            no_improve = 0

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_to_n(f, new_n):
    """Interpolate sequence f to length new_n, preserving relative shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss
