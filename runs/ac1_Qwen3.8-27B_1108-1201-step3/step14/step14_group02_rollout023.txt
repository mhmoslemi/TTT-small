    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)

def compute_soft_gradient(a, ac, top_k=50, beta=50.0):
    """Compute the gradient of the soft max of autocorrelation."""
    n = len(a)
    ac_pos = np.maximum(ac, 0.0)
    max_ac = ac_pos.max()
    if max_ac < 1e-20:
        return np.zeros(n)
    
    exponents = beta * (ac_pos - max_ac)
    np.maximum(exponents, -500, out=exponents)  # prevent underflow
    weights = np.exp(exponents)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        return np.zeros(n)
    weights /= w_sum
    
    # Find top-K indices by weight
    k_max = min(top_k, len(ac))
    top_idx = np.argsort(weights)[::-1][:k_max]
    
    grad = np.zeros(n)
    for k_i in top_idx:
        k = int(k_i)
        w = weights[k_i]
        if w < 1e-12:
            continue
        if k == 0:
            grad += 2.0 * w * a
        else:
            # d(ac[k])/da[j] = a[j-k] (for j>=k) + a[j+k] (for j<n-k)
            grad[k:] += w * a[:n-k]
            grad[:n-k] += w * a[k:]
    
    return grad

def lp_frank_wolfe_reduced(a0, time_limit, solver, M_constraints=200, verbose=True):
    """LP Frank-Wolfe with only top-M constraints."""
    n = len(a0)
    x = norm_sum1(a0)
    y = x.copy()
    best = x.copy()
    best_M = fast_autocorr(x).max()
    t0 = time.monotonic()
    
    iter_count = 0
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        t = iter_count
        
        ac = fast_autocorr(y)
        M_curr = ac.max()
        if M_curr < 1e-14:
            break
        
        # Use only top-M constraints
        top_M_idx = np.argsort(ac)[::-1][:min(M_constraints, len(ac))]
        
        b_var = cp.Variable(n, nonneg=True)
        
        # Build T only for selected rows
        T = np.zeros((len(top_M_idx), n))
        for i, k in enumerate(top_M_idx):
            k_int = int(k)
            j_start = max(0, k_int - n + 1)
            j_end = min(n, k_int + 1)
            js = np.arange(j_start, j_end)
            T[i, js] = y[k_int - js]
        
        constraints = [T @ b_var <= M_curr]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=solver, verbose=False)
        except Exception:
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue
        
        b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        b_sum = b_val.sum()
        if b_sum < 1e-15:
            continue
        g = b_val / b_sum
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / sy
        
        if t == 1:
            x_new = g.copy()
        else:
            beta_mom = (t - 1.0) / (t + 2.0)
            x_new = y_new + beta_mom * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            sx = x_new.sum()
            if sx < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / sx
        
        y = y_new
        x = x_new
        
        my = fast_autocorr(y).max()
        if my < best_M:
            best_M = my
            best = y.copy()
        
        mx = fast_autocorr(x).max()
        if mx < best_M:
            best_M = mx
            best = x.copy()
        
        if verbose and iter_count % 5 == 0:
            elapsed = time.monotonic() - t0
            J = 2.0 * n * best_M
            print(f"    LP t={iter_count} ({elapsed:.1f}s): J={J:.6f}")
    
    return best, best_M

def coordinate_polish(a0, time_limit, verbose=True):
    """Efficient coordinate descent polish."""
    n = len(a0)
    a = norm_sum1(a0)
    best = a.copy()
    best_M = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    stall = 0
    max_stall = 500
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        ac = fast_autocorr(a)
        
        # Find the lag with maximum autocorrelation
        k_max = int(np.argmax(ac))
        
        # Find the pair (i, i+k_max) with maximum product
        js = max(0, k_max - n + 1)
        je = min(n, k_max + 1)
        idx = np.arange(js, je)
        products = a[idx] * a[k_max - idx]
        
        if products.max() < 1e-20:
            stall += 1
            break
        
        src = idx[np.argmax(products)]
        
        # Find best dst: try to move mass to where it causes least increase
        # Score each position by how much it would increase the max autocorr
        # Approximation: sum of a[|k - dst|] for top lags
        
        top_lags = np.argsort(ac)[::-1][:30]
        
        # Compute sensitivity for all positions
        sensitivity = np.zeros(n)
        for k in top_lags:
            k_i = int(k)
            w = ac[k_i]
            if k_i == 0:
                sensitivity += w * 2.0 * a
            else:
                sensitivity[k_i:] += w * a[:n-k_i]
                sensitivity[:n-k_i] += w * a[k_i:]
        
        # Remove sensitivity at src
        sensitivity[src] = -np.inf
        
        # Pick the position with lowest sensitivity
        dst = int(np.argmin(sensitivity))
        
        # Try different step sizes
        improved = False
        for frac in [0.01, 0.03, 0.05, 0.1, 0.2, 0.5]:
            amt = a[src] * frac
            if amt < 1e-18:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_autocorr(a_trial).max()
            if M_trial < best_M - 1e-15:
                a = a_trial
                best_M = M_trial
                best = a.copy()
                improved = True
                stall = 0
                break
        
        if not improved:
            stall += 1
            # Random restart
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-8:
                frac = rng.uniform(0.01, 0.1)
                amt = a[i] * frac
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                s_t = a.sum()
                if s_t > 1e-15:
                    a = a / s_t
                    M_new = fast_autocorr(a).max()
                    if M_new < best_M:
                        best_M = M_new
                        best = a.copy()
                        stall = 0
    
    if verbose:
        J = 2.0 * n * best_M
        print(f"    Coord ({time.monotonic()-t0:.1f}s): J={J:.6f}")
    
    return best, best_M
